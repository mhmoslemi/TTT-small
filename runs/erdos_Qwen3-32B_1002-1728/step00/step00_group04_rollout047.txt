import time

def run(seed=42, budget_s=1100, **kwargs):

    global initial_h_values

    np.random.seed(seed)

    start_time = time.time()

    # get initial h

    h = initial_h_values.copy().astype(np.float64)

    n_points = len(h)

    dx = 2.0 / n_points

    best_h = h.copy()

    best_C5 = float('inf')

    # iterations counter

    while time.time() - start_time < budget_s:

        # compute overlaps

        corr = np.correlate(h, 1 - h, mode='full') * dx

        C_values = corr[n_points : 2*n_points -1]

        k_max = np.argmax(C_values)

        current_C5 = C_values[k_max]

        if current_C5 < best_C5:

            best_C5 = current_C5

            best_h = h.copy()

        # Perform coordinate-wise optimization

        for i in range(n_points):

            # compute gradient for current k_max

            g_i = 0.0

            if i + k_max < n_points:

                g_i += (1.0 - h[i + k_max])

            if i - k_max >= 0:

                g_i -= h[i - k_max]

            # adjust h[i]

            if g_i > 0:

                h[i] = 0.0

                h[i] = 1.0

        current_sum = np.sum(h)
