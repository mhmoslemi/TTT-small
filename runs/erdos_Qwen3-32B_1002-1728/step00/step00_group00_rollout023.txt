            particle = np.clip(particle, 0, 1)
        particles.append(particle)
        velocities.append(np.zeros_like(particle))
        pbest_vals.append(compute_c5(particle))
        pbest.append(particle.copy())
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    pbest = np.array(pbest)
    pbest_vals = np.array(pbest_vals)
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    dx = 2.0 / n_points
    def compute_c5(h):
        return np.max(np.correlate(h, 1-h, mode='full')) * dx
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            # Update position
            particles[i] += velocities[i]
            # Feasibility repair
            particles[i] = np.clip(particles[i], 0, 1)
            s = particles[i].sum()
            if s > 0:
                scale = target_sum / s
                particles[i] *= scale
                particles[i] = np.clip(particles[i], 0, 1)
            # Update bests
            current_val = compute_c5(particles[i])
            if current_val < pbest_vals[i]:
                pbest_vals[i] = current_val
                pbest[i] = particles[i].copy()
                if current_val < gbest_val:
                    gbest_val = current_val
                    gbest = particles[i].copy()
    
    return (gbest, gbest_val, n_points)
