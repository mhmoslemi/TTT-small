import time

def compute_C5(h, dx):
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    start_time = time.time()

    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: half 1s, half 0s
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0

    # Since the initial_h_values should already satisfy this.

    dx = 2.0 / n_points
    current_C5 = compute_C5(h, dx)
    best_C5 = current_C5
    best_h = h.copy()

    T = 1.0  # initial temperature
    alpha = 0.99  # cooling rate
    max_no_improvement = 100
    no_improvement_counter = 0

    while time.time() - start_time < budget_s - 10:  # leave 10s for safety
        # Perturbation
        i, j = np.random.choice(n_points, 2, replace=False)
        h_i = h[i]
        h_j = h[j]
        max_delta = min(1 - h_i, h_j)
        if max_delta <= 0:
            continue  # no move possible
        delta = np.random.uniform(0, max_delta)

        new_h[i] = h_i + delta
        new_h[j] = h_j - delta

        # Compute new_C5
        new_C5 = compute_C5(new_h, dx)
        delta_C5 = new_C5 - current_C5

        # Acceptance
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = new_h
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()

                no_improvement_counter += 1
        else:

        T *= alpha

        # Adjust cooling rate if no improvement
        if no_improvement_counter > max_no_improvement:
