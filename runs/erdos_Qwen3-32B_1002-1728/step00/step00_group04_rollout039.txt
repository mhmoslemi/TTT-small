        initial_h_data = None

    m = 30
    target_sum = n_points / 2.0
    np.random.seed(seed)

    def compute_c5(h):
        dx = 2.0 / n_points
        ones_h = 1.0 - h
        cor = scipy.signal.correlate(h, ones_h, mode='full')
        return np.max(cor) * dx

    def project_feasible(h_clipped):
        h = cp.Variable(n_points)
        obj = cp.Minimize(0.5 * cp.sum_squares(h - h_clipped))
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return h.value
        except:
            return h_clipped

    def generate_random_h():
        h = np.random.beta(2, 2, size=n_points)
        return project_feasible(h)

    particles = []
    if initial_h_data is not None:
        particles.append(initial_h_data)
        for _ in range(m - 1):
            particles.append(generate_random_h())
    else:
        for _ in range(m):
            particles.append(generate_random_h())

    particles = np.array(particles, dtype=float)
    velocities = 0.1 * np.random.randn(m, n_points)
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([compute_c5(p) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    omega = 0.7
    omega_min = 0.4
    c1, c2 = 1.5, 1.5
    omega_decay = 0.001
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(m):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - particles[i]) +
                             c2 * r2 * (global_best_position - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = project_feasible(particles[i])
            new_score = compute_c5(particles[i])
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_best_positions[i] = particles[i].copy()
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best_position = particles[i].copy()
        omega = max(omega_min, omega - omega_decay)

    return (global_best_position, global_best_score, n_points)
