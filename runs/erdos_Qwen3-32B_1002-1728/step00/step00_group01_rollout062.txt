    c = cp.Variable()

        z >= -1,
        z <= 1,
        cp.sum(z) == 0
    ]
    
    for m in range(n_points):
        indices = [(i, i + m) for i in range(n_points - m)]
        terms = [z[i] * z[j] for i, j in indices]
        expr = cp.sum(terms)
        constraints.append(expr >= c)
    
    objective = cp.Maximize(c)
    problem = cp.Problem(objective, constraints)
    
    try:

    elapsed = time.time() - start_time
    if problem.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE} or elapsed > budget_s - 10:
        return (initial_h, evaluate_erdos_solution(initial_h, n_points), n_points)
