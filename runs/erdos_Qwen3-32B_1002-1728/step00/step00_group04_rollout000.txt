            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_steps = 0
            alpha = 0.01
        else:
            no_improvement_steps += 1
            if no_improvement_steps >= decay_steps:
                alpha *= 0.9
                if alpha < min_alpha:
                    break
    
    return best_h, best_c5, n_points
