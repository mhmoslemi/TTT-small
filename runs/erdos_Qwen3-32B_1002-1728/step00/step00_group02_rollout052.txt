def compute_C5(h, dx):
    full_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(full_corr)
    C5 = max_corr * dx
    return C5

def project_feasible(h, n_points):
    h_clip = np.clip(h, 0, 1)
    sum_h = h_clip.sum()
    if sum_h == 0:
        return initial_h_values.copy()
    scale = (n_points / 2) / sum_h
    return h_clip * scale

    new_h = h.copy()
    choice = np.random.rand()
    # Single-point adjust
    if choice < 0.5:
        idx = np.random.randint(n_points)
        new_h[idx] = np.random.rand()

        start = np.random.randint(n_points)
        max_len = min(10, n_points - start)
        length = np.random.randint(2, max_len + 1)
        end = start + length
        segment = new_h[start:end]
        shifted = np.roll(segment, 1)
        new_h[start:end] = shifted
    else:  # Swap
        i, j = np.random.choice(n_points, 2, replace=False)
        new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    if n_points == 0:
        raise ValueError("initial_h_values is empty")
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_C = compute_C5(h, dx)
    best_h = h.copy()
    current_h = h.copy()
    current_C = best_C

    T = 1.0
    alpha = 0.95
    no_improvement_count = 0

    while time.time() < start_time + budget_s:
        new_h = perturb(current_h, n_points)
        new_h = project_feasible(new_h, n_points)
        new_C = compute_C5(new_h, dx)
        delta = new_C - current_C

            current_h = new_h.copy()
            current_C = new_C

import numpy as np
import time

def compute_C5(h, dx):
    full_corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(full_corr)
    return max_corr * dx

def project_feasible(h, n_points):
    h_clip = np.clip(h, 0, 1)
    sum_h = h_clip.sum()
    if sum_h == 0:
        return initial_h_values.copy()
    scale = (n_points / 2) / sum_h
    return h_clip * scale

def perturb(h, n_points):
    new_h = h.copy()
    choice = np.random.rand()
    if choice < 0.5:
        idx = np.random.randint(n_points)
        new_h[idx] = np.random.rand()
    elif choice < 0.8:
        start = np.random.randint(n_points)
        max_len = min(10, n_points - start)
        length = np.random.randint(2, max_len + 1)
        end = start + length
        segment = new_h[start:end]
        shifted = np.roll(segment, 1)
        new_h[start:end] = shifted
    else:
        i, j = np.random.choice(n_points, 2, replace=False)
        new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_C = compute_C5(h, dx)
    best_h = h.copy()
    current_h = h.copy()
    current_C = best_C

    T = 1.0
    alpha = 0.95
    no_improvement_count = 0

    while time.time() < start_time + budget_s:
        new_h = perturb(current_h, n_points)
        new_h = project_feasible(new_h, n_points)
        new_C = compute_C5(new_h, dx)
        delta = new_C - current_C
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h.copy()
            current_C = new_C
            if new_C < best_C:
                best_C = new_C
                best_h = current_h.copy()
                no_improvement_count = 0

            no_improvement_count += 1

        if no_improvement_count >= 20:
            T *= alpha
            no_improvement_count = 0

        if T < 1e-6:
            break

    return (best_h, best_C, n_points)
