import numpy as np
from scipy.optimize import minimize

def run():
    import time

    start_time = time.time()
    budget_s = 1100  # Time budget

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective_func(h_vec):
        h_current = h_vec.reshape(n_points)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        return np.max(corr) * dx

    constraint_sum = {'type': 'eq', 'fun': lambda v: np.sum(v.reshape(n_points)) - n_points / 2}
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(
        objective_func,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint_sum,
        options={'maxiter': 200, 'disp': False}
    )
    best_solution = result.x if result.success else h

    # Project to feasible set in case of numerical errors
    h_best = best_solution.reshape(n_points)
    total = np.sum(h_best)
    adjust = (n_points / 2 - total) / n_points
    h_best = np.clip(h_best + adjust, 0, 1)

    c5_bound = evaluate_erdos_solution(h_best)
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h_best = initial_h_values

    return (h_best.tolist(), c5_bound, n_points)
