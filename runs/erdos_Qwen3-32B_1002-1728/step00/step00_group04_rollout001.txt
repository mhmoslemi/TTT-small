def project_h(h_initial, n_points):
    h_proj = cp.Variable(shape=(n_points,))
    constraints = [h_proj >= 0, h_proj <= 1, cp.sum(h_proj) == n_points / 2]
    objective = cp.Minimize(0.5 * cp.norm2(h_proj - h_initial)**2)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS)
    if problem.status != cp.OPTIMAL:
        raise ValueError("Projection failed")
    return np.array(h_proj.value).flatten()
