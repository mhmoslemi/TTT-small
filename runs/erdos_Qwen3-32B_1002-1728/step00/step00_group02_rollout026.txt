import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    current_sum = np.sum(h)

        h *= target_sum / current_sum

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    patience = 0

    max_iterations = 1000000  # High upper bound, controlled by time
    learning_rate = 0.01

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5_current = np.max(corr)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience = 0

            patience += 1
        if patience >= max_patience:
            learning_rate *= 0.9  # Reduce learning rate if no progress
            patience = 0

        # Find the index of the maximum overlap
        k_star_idx = np.argmax(corr)
