            if not np.isclose(curr_sum, n_points / 2, atol=1e-6):
                diff = (n_points / 2) - curr_sum
                curr_h[0] += diff
                curr_h = np.clip(curr_h, 0.0, 1.0)
            curr_c5 = objective(curr_h, n_points)
            if best_c5 is None or curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)
    return best_h, c5_bound, n_points
