import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {
        'type': 'eq',
        'fun': lambda h_flat: np.sum(h_flat) - n_points / 2
    }

    # First optimization run
    result = scipy.optimize.minimize(
        objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[constraint],
        options={'maxiter': 300, 'disp': False}
    )

    best_h = result.x.reshape(n_points)
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0.0, 1.0)
    best_c5 = objective(best_h)

    if time.time() - start_time < budget_s - 30:
        # Second run with perturbed initial guess
        h_pert = best_h + 0.1 * np.random.randn(n_points)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        sum_h_pert = np.sum(h_pert)
        if not np.isclose(sum_h_pert, n_points / 2):
            scale = (n_points / 2) / sum_h_pert
            h_pert = np.clip(h_pert * scale, 0.0, 1.0)
        try:
            result2 = scipy.optimize.minimize(
                objective,
                x0=h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=[constraint],
                options={'maxiter': 200, 'disp': False}
            )
            if result2.success:
                h2 = np.clip(result2.x.reshape(n_points), 0.0, 1.0)
                sum_h2 = np.sum(h2)

                    scale = (n_points / 2) / sum_h2

            pass

    return (best_h, best_c5, n_points)
