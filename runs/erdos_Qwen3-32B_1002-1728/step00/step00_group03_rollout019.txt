        h_clipped = np.clip(h_clipped, 0, 1)
        h[:] = h_clipped

        # Occasionally perturb
        if iter % perturb_interval == 0 and iter > 0:
            noise = np.random.normal(0, 0.01, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            # readjust sum again
            current_sum = np.sum(h)
            delta = (s - current_sum) / n_points
            h += delta
            h = np.clip(h, 0, 1)

    # Final evaluation
    overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(overlaps)
    return best_h.tolist(), final_c5, n_points
