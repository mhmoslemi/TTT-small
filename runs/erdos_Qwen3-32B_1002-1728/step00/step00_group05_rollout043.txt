import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    h = h * (target_sum / h_sum) if h_sum != 0.0 else np.zeros(n_points)
    h[:int(n_points/2)] = 1.0 if h_sum == 0.0 else h

    def objective_func(h_vec):
        h_f = h_vec.reshape(n_points)
        corr = np.correlate(h_f, 1 - h_f, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    bounds_func = [(0.0, 1.0) for _ in range(n_points)]
    constraints_func = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}

    best_h = h.copy()
    best_c5 = objective_func(best_h)

    while time.time() - start_time < budget_s - 1.0:
        res = minimize(
            objective_func,
            best_h,
            method='SLSQP',
            bounds=bounds_func,
            constraints=[constraints_func],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        optimized_h = np.clip(res.x, 0.0, 1.0)
        optimized_h_sum = optimized_h.sum()
        if optimized_h_sum != 0.0:
            optimized_h = optimized_h * (target_sum / optimized_h_sum)
        current_c5 = objective_func(optimized_h)
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
