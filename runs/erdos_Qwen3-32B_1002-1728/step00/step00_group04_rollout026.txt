import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = initial_h_values.copy()
    n_points = len(h)
    required_sum = n_points / 2.0

    h = h / h.sum() * required_sum  # Normalize initial h

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    temp = 1.0
    cooling_rate = 0.995
    patience = 100
    no_improve = 0

    while time.time() - start_time < budget_s and temp > 1e-4:
        mask = np.random.rand(n_points) < 0.1
        delta = np.random.normal(0, 0.2, size=n_points)
        new_h = h.copy()
        new_h[mask] += delta[mask]
        new_h = np.clip(new_h, 0, 1)
        s = new_h.sum()
        if s > 1e-8:
            new_h *= (required_sum / s)

        try:
            new_c5 = evaluate_erdos_solution(new_h)
        except:
            new_c5 = float('inf')

        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0:
            h[:] = new_h
            best_h[:] = new_h
            best_c5 = new_c5
            no_improve = 0
        else:
            if np.random.rand() < np.exp(-delta_c5 / temp):
                h[:] = new_h
                no_improve = 0
            else:
                no_improve += 1

        temp *= cooling_rate
        if no_improve > patience:
            temp = max(temp * 0.5, 1e-4)
            no_improve = 0

    return (best_h.tolist(), best_c5, n_points)
