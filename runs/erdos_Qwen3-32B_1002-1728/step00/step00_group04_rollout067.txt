            if i+1 >= len(selected):
                continue
            if random.random() < 0.8:
                child = two_point_crossover(selected[i], selected[i+1])
                new_population.append(child)

        # Mutation
        for i in range(population_size):
            if random.random() < mutation_rate:
                new_population.append(mutate(random.choice(selected)))

        # Local search
        for i in range(int(population_size * 0.2)):
            im = int(random.random() * len(selected))
            h_ls = gradient_descent(selected[im].copy())
            new_population.append(h_ls)

        population = np.array(new_population[:population_size])

        # Update best
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0].copy()

        # Progress every 100 iterations
        if update_counter % 100 == 0:
            print(f"Generation {update_counter} Best C5: {best_c5:.8f} Time: {time.time()-start_time:.1f}s")

    best_h = project(best_h)
    return best_h.tolist(), best_c5, n_points
