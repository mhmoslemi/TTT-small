import numpy as np
import scipy.interpolate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_coarse = 70
    n_fine = 280
    target_coarse = n_coarse / 2.0
    target_fine = n_fine / 2.0
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine

    def compute_c5(h, dx):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    def project(h, target):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s != 0:
            h = h * (target / s)
        h = np.clip(h, 0, 1)
        return h

    def optimize_step(h, dx, target, step_size):
        n = len(h)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap_idx = np.argmax(overlaps)
        k = max_overlap_idx - n + 1

        gradient = np.zeros(n)
        if k == 0:
            gradient = (1 - 2 * h) * dx
        else:
            abs_k = abs(k)
            valid_i = np.arange(n - abs_k)
            if k > 0:
                gradient[valid_i] += (1 - h[valid_i + k]) * dx
                gradient[valid_i + k] += -h[valid_i] * dx
            else:
                gradient[valid_i + abs_k] += (1 - h[valid_i]) * dx
                gradient[valid_i] += -h[valid_i + abs_k] * dx
        h_grad = h - step_size * gradient
        h_proj = project(h_grad, target)
        return h_proj, compute_c5(h_proj, dx)

    start_time = time.time()
    best_h, best_c5, n_pts = h.copy(), compute_c5(h, dx_coarse), n_coarse

    # Coarse grid optimization
    for _ in range(1000):
        if time.time() - start_time > budget_s * 0.5:

        if c5 < best_c5:
