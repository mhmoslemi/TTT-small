def compute_overlap_gradient(h, dx, k_max):
    # Compute gradient of the overlap for shift k_max
    n = len(h)
    grad = np.zeros_like(h)
    # For shift k_max, compute how each h[i] affects the overlap
    for i in range(n):
        # Find j such that x_i + k_max * dx = x_j
        # Since x_i = i * dx, and x_j = j*dx

        j = i + k_max

            grad[i] += 1

            grad[i] += (1 - h[j])
