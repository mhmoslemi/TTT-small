def run(seed=42, budget_s=1100, **kwargs):

    random.seed(seed)
    import time

    # initial_h_values is a global variable provided by the harness
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def project_feasible(candidate):
        # clip and scale
        clipped = np.clip(candidate, 0, 1)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:
            return initial_h_values.copy()  # fallback
        scale = (n_points / 2) / sum_clipped
        return clipped * scale

    def compute_c5(h):
        # returns the maximum overlap multiplied by dx
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    T = 1.0
    alpha = 0.95
    max_iter_per_temp = 100
    no_improvement_count = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Perturb
        perturbation_type = np.random.rand()
        candidate = current_h.copy()

        if perturbation_type < 0.5:
            # Single-point adjustment
            i = np.random.randint(n_points)
            new_val = np.random.uniform(0, 1)
            candidate[i] = new_val
        elif perturbation_type < 0.8:
            # Segment slide
            start = np.random.randint(n_points)
            length = np.random.randint(2, 11)
            if length > n_points:
                length = n_points
            direction = np.random.choice([-1, 1])
            #indices = [(start + k) % n_points for k in range(length)]
            # TODO: need vectorized indices
            indices = (start + np.arange(length)) % n_points
            shifted_indices = (indices + direction) % n_points
            original_vals = candidate[indices]
            candidate[shifted_indices] = original_vals
        else:
            # Pair swap
            i, j = np.random.choice(n_points, 2, replace=False)
            candidate[i], candidate[j] = candidate[j], candidate[i]

        # Project to feasible region
        proj_h = project_feasible(candidate)
