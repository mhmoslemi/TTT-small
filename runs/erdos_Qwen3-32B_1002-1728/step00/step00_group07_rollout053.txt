import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    initial_temperature = 0.2
    cooling_rate = 0.995
    T = initial_temperature
    t0 = time.time()
    max_time = budget_s
    
    while time.time() - t0 < max_time:
        i, j = np.random.choice(n_points, size=2, replace=False)
        h_i = h[i]
        h_j = h[j]
        
        delta_plus = min(h_i, 1 - h_j)
        delta_minus = min(h_j, 1 - h_i)
        
        if delta_plus + delta_minus <= 0:
            continue
        
        delta = np.random.uniform(-delta_minus, delta_plus)
        
        new_h = h.copy()
        new_h[i] = h_i - delta
        new_h[j] = h_j + delta
        
        new_C5 = compute_C5(new_h)
        delta_C5 = new_C5 - current_C5
        
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = new_h
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        T *= cooling_rate
    
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Invalid h values"
    assert np.isclose(np.sum(best_h), n_points / 2), f"Sum constraint violated: {np.sum(best_h)} vs {n_points/2}"
    
    return best_h, best_C5, n_points
