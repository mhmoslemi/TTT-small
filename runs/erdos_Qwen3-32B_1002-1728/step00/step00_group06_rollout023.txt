import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 95
        h = np.full(n_points, 0.5)
    
    required_sum = n_points / 2
    h_sum = np.sum(h)
    if not np.allclose(h_sum, required_sum):
        h *= required_sum / h_sum
    
    def objective(x):
        x = np.clip(x, 0, 1)
        _, c5 = evaluate_erdos_solution(x.tolist())
        return c5
    
    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum}
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_time = time()
    
    start = time()
    while time() - start < budget_s - 10:
        try:
            res = opt.minimize(
                objective, best_h, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 200, 'ftol': 1e-6}
            )
            if res.success and res.fun < best_c5:
                best_h = np.clip(res.x, 0, 1).copy()
                best_c5 = res.fun
                best_time = time()

            pass

            pert = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(best_h + pert, 0, 1)
            new_sum = np.sum(new_h)
            scale = required_sum / new_sum
            new_h = np.clip(new_h * scale, 0, 1)
            best_h = new_h.copy()
            best_time = time()
