def run(seed=42, budget_s=1100, **kwargs):
    tf.random.set_seed(seed)
    np.random.seed(seed)
    
    # Load and normalize initial h
    h_init = initial_h_values.copy()
    n_current = len(h_init)
    target_sum = n_current / 2.0
    h_current = h_init * (target_sum / np.sum(h_init))
    
    # Convert to TensorFlow variable
    h = tf.Variable(h_current, dtype=tf.float32)
    
    optimizer = tf.keras.optimizers.Adam(learning_rate=0.01)
    best_c5 = np.inf
    best_h = h.numpy().copy()
    best_n = n_current
    
    dx = 2.0 / n_current
    start_time = time.time()
    max_iterations = 1000
    
    # Refinement steps: 70 -> 140 -> 280 -> 560
    for refinement in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        for iter in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            with tf.GradientTape() as tape:
                h_clipped = tf.clip_by_value(h, 0.0, 1.0)
                current_sum = tf.reduce_sum(h_clipped)
                target = tf.cast(tf.shape(h_clipped)[0] * 0.5, tf.float32)
                h_norm = h_clipped * (target / current_sum)

                c5 = tf.reduce_max(corr) * dx
            
            gradients = tape.gradient(c5, h)

            current_sum = tf.reduce_sum(h_clipped)

                best_c5 = c5_val
                best_h = h_norm.numpy().copy()
