import numpy as np
import time

def compute_c5(h, n_points):
    max_Ck = 0.0
    dx = 2.0 / n_points
    for k in range(1, n_points):
        valid = n_points - k
        if valid <= 0:
            continue
        sum_c = np.sum(h[:valid] * (1 - h[k:k+valid]))
        c = sum_c * dx
        if c > max_Ck:
            max_Ck = c
    return max_Ck

def compute_gradient(h, k_max, n_points):
    grad = np.zeros(n_points)
    k = k_max
    valid = n_points - k
    for i in range(valid):
        grad[i] += (1 - h[i + k])
        grad[i + k] += -h[i]
    return grad

def project(h, n_points):
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    # Scale to sum h = n_points / 2
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        # This shouldn't happen if all h are 0, but since the initial sum is n/2, we don't expect it
        return h_clipped * (n_points / 2)
    scale = (n_points / 2.0) / current_sum
    return h_clipped * scale
