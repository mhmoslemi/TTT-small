import numpy as np
import scipy.optimize as opt
import time

def run():
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    best_h = initial_h.copy()

    def objective(h_flat):

    bounds = [(0, 1) for _ in range(n_points)]

    budget_s = 1100

    result = opt.minimize(
        objective,
        x0=initial_h,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-6, 'disp': False}
    )

    h_opt = result.x
    cor = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(cor * dx)

    # Final check to ensure constraint is satisfied
    if not np.isclose(np.sum(h_opt), n_points / 2):
        # Project to sum constraint
        h_opt = np.clip(h_opt, 0, 1)
        current_sum = np.sum(h_opt)
        if current_sum > 0:
            h_opt *= (n_points / 2) / current_sum
        else:
            h_opt[:] = 0.5
    cor = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(cor * dx)

    return h_opt, c5, n_points
