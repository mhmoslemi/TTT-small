    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    grids = [140, 280, 560]  # refinements to try
    for target_n in grids:
        if time() - start_time > budget_s - 10:
            break
        if target_n <= n_points:
            continue
        
        # Interpolate to higher resolution
        def upsample(x_old, y_old, x_new):
            f = interp1d(x_old, y_old, kind='linear', fill_value='extrapolate')
            y_new = f(x_new)
            y_new = np.clip(y_new, 0, 1)
            current_sum = y_new.sum()
            required_sum = target_n / 2.0
            y_new = y_new * (required_sum / current_sum)
            return np.clip(y_new, 0, 1)

        x_new = np.linspace(0, 2, target_n)
        h_new = upsample(x_old, best_h, x_new)
        
        # Optimization
        bounds = [(0, 1) for _ in range(target_n)]
        def constraint(h):
            return h.sum() - (target_n / 2.0)

            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
