def project_to_feasible(y, target_sum):
    y_clipped = np.clip(y, 0, 1)
    current_sum = np.sum(y_clipped)
    if np.isclose(current_sum, target_sum):
        return y_clipped.copy()
    delta = target_sum - current_sum
    if delta > 0:
        indices = np.argsort(y_clipped)
        for i in range(len(indices)):
            idx = indices[i]
            space = 1 - y_clipped[idx]
            if space <= 0:
                continue
            add = np.min((delta, space))
            y_clipped[idx] += add
            delta -= add
            if delta <= 1e-8:
                break
    elif delta < 0:
        indices = np.argsort(-y_clipped)
        for i in range(len(indices)):

            space = y_clipped[idx]
            if space <= 0:
                continue
            sub = np.min((-delta, space))
            y_clipped[idx] -= sub
            delta += sub
            if delta >= -1e-8:
                break
    return y_clipped.copy()

def evaluate_individual(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def crossover(parent1, parent2, n_points):

    child = np.where(mask, parent1, parent2)
    return project_to_feasible(child, n_points / 2.0)
