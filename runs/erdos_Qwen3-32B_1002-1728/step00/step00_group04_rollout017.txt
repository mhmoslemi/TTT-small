def optimize_with_gradient_descent(h, max_steps=50, learning_rate=0.05, lr_decay=0.95):

    h_current = h.copy()
    for step in range(max_steps):
        # Compute k_star
        anti_h = 1.0 - h_current
        correl = np.correlate(h_current, anti_h, mode="full")
        relevant_correl = correl[n: 2 * n - 1]
        max_idx = np.argmax(relevant_correl)
        k_star = max_idx + 1  # shift k=1 to n-1
        # Compute gradient for k_star
        grad = np.zeros(n)

            if m >= k_star:
