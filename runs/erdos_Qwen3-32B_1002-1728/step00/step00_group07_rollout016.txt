import numpy as np
import time

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum == 0:
        h.fill(0.0)
        h[0] = target_sum
    else:
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    k_values = np.correlate(h, 1 - h, mode='full')
    return np.max(k_values) * dx

def tournament_selection(population, fitnesses, tournament_size=3):
    indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitnesses[i] for i in indices]
    best_idx = indices[np.argmin(tournament_fitness)]
    return population[best_idx]

def uniform_crossover(parent1, parent2):
    mask = np.random.rand(len(parent1)) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(individual, sigma=0.05):
    noise = np.random.normal(0, sigma, len(individual))
    return individual + noise

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 80
    target_sum = n_points / 2.0
    population_size = 50
    generations_no_improve_limit = 100

    # Initialize population with initial_h_values if available
    population = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = initial_h_values.copy()
        population.append(repair_h(h0, target_sum))
    else:
        h0 = np.random.rand(n_points)
        population.append(repair_h(h0, target_sum))

    while len(population) < population_size:
        h = np.random.rand(n_points)
        population.append(repair_h(h, target_sum))
    population = np.array(population)

    # Evaluate initial fitness
    try:
        fitness = np.array([compute_c5(h, n_points) for h in population])
    except:
        return population[0].tolist(), 1.0, n_points

    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx].copy()
    no_improve_count = 0

    while time.time() - start_time < budget_s:
        offspring = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            child = uniform_crossover(parent1, parent2)
            child = mutate(child)
            child = repair_h(child, target_sum)
            offspring.append(child)

        new_offspring = np.array(offspring)
        try:
            new_fitness = np.array([compute_c5(h, n_points) for h in new_offspring])
        except:
            new_fitness = np.full(new_offspring.shape[0], 1.0)

        combined_pop = np.concatenate([population, new_offspring])
        combined_fitness = np.concatenate([fitness, new_fitness])
        indices = np.argsort(combined_fitness)[:population_size]
        population = combined_pop[indices]
        fitness = combined_fitness[indices]

        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= generations_no_improve_limit:
            break

    return best_h.tolist(), best_c5, n_points
