import numpy as np
import time
import random
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.isclose(np.sum(h), n_points / 2), "Invalid initial h"
    
    def compute_c5(h_array):
        one_minus_h = 1.0 - h_array
        corr = np.correlate(h_array, one_minus_h, mode='full')
        return np.max(corr * dx)
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    temp = 1000.0
    alpha = 0.999
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        h_prev = h.copy()
        indices = np.random.choice(n_points, size=2, replace=False)
        i, j = indices
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        delta = np.random.uniform(0, max_delta)
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        new_c5 = compute_c5(h)
        delta_c5 = current_c5 - new_c5
        
        if delta_c5 > 0:
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = h.copy()
            current_c5 = new_c5
        else:
            metropolis = math.exp(delta_c5 / temp)
            if random.random() < metropolis:
                current_c5 = new_c5
            else:
                h[:] = h_prev[:]
        
        temp *= alpha
        
        if time.time() - start_time > budget_s * 0.999:
            break
    
    return (best_h.tolist(), best_c5, n_points)
