import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_vec):

        dh = 1 - h
        corr = np.correlate(h, dh, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vec) - n_points / 2

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    x0 = h_initial.copy()

    # Try initial optimization
    result = minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'disp': False}
    )
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    else:
        best_c5 = objective(x0)
        best_h = x0

    # Perturb and restart if time allows
    restarts = 0
    while time.time() - start_time < budget_s * 0.95 and restarts < 3:

        total = np.sum(h_pert)
