import numpy as np
import time

def project_h(h, n_points):
    h_clipped = np.clip(h, 0., 1.)
    current_sum = h_clipped.sum()
    target_sum = n_points / 2
    scaling = target_sum / current_sum if current_sum != 0 else 1.0
    h_clipped *= scaling
    return np.clip(h_clipped, 0., 1.)

def find_max_c5_and_k(h, n_points):
    dx = 2.0 / n_points
    ccorr = np.correlate(h, 1 - h, mode='full')
    start = n_points - 1
    relevant = ccorr[start: start + n_points]
    max_c5 = (relevant * dx).max()
    max_k = np.argmax(relevant)
    return max_c5, max_k

def compute_gradient(h, k, n_points):
    grad = np.zeros(n_points)
    if k < n_points:
        grad[:n_points - k] += 1 - h[k:]
        grad[k:] -= h[:n_points - k]
    return grad
