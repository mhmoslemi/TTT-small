def run(seed=42, budget_s=1100, **kwargs):

    h0 = np.array(initial_h_values)
    n = len(h0)

    def compute_overlap(h):
        dx = 2.0 / n
        max_C = 0.0
        for k in range(1, n):
            indices = np.arange(n - k)
            total = np.sum(h[indices] * (1.0 - h[indices + k]))
            current_C = total * dx
            if current_C > max_C:
                max_C = current_C
        return max_C

    h0_transformed = h0[:-1]

    def objective_transformed(h_vars):
        h_vars = np.clip(h_vars, 0, 1)
        current_sum = np.sum(h_vars)
        h_last = target_sum - current_sum
        h_last = np.clip(h_last, 0, 1)
        h = np.append(h_vars, h_last)
        return compute_overlap(h)

    start_time = time.time()
    bounds = [(0.0, 1.0) for _ in range(n - 1)]

    # Limit iterations to fit within budget
    max_iter = int(10000)  # Heuristic for 1100s with n=42

    result = minimize(
        objective_transformed,
        x0=h0_transformed,
        bounds=bounds,
        method='Nelder-Mead',
