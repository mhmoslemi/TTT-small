            best_h = current_h
            best_c5 = current_c5

    # Final enforcement of constraints
    best_h = best_h * (desired_sum / np.sum(best_h))
    best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
