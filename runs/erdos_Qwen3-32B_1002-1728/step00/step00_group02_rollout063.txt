import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    S0 = n_points / 2.0

    best_h = h.copy()
    best_c5 = None
    start_time = time.time()

    learning_rate = 0.02
    decay_rate = 0.995
    iterations = 0

    while time.time() - start_time < budget_s - 0.1:
        iterations += 1

        # Compute overlap
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = corr.max()
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find shift with maximum
        k_max_idx = np.argmax(corr)
        actual_k = k_max_idx - (n_points - 1)

        # Determine valid indices
        lower_i = max(0, -actual_k)
        upper_i = min(n_points - 1, (n_points - 1) - actual_k)
        valid_is = np.arange(lower_i, upper_i + 1)

        # Coordinate updates
        for i in valid_is:
            j = i + actual_k
            if 0 <= j < n_points:
                dh = learning_rate * (1 - h[j])
                h[i] = np.clip(h[i] - dh, 0, 1)

        # Project and normalize
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum > 1e-8:
            h = h * (S0 / h_sum)
            h = np.clip(h, 0, 1)

        # Learning rate decay
        learning_rate *= decay_rate

    return (best_h.tolist(), best_c5, n_points)
