        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 150, 'disp': False}
    )

    h_opt = np.clip(result.x, 0.0, 1.0)
    current_sum = h_opt.sum()
    target_sum = n_points / 2.0
    if not np.isclose(current_sum, target_sum, atol=1e-5):
        delta = target_sum - current_sum
        h_opt += delta / n_points
        h_opt = np.clip(h_opt, 0.0, 1.0)

    final_c5 = objective_func(h_opt)
    elapsed = time.time() - start_time

    return h_opt.tolist(), final_c5, n_points
