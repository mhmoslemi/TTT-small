def repair_candidate(h, target_sum):

    import cvxpy as cp

    n_points = len(h)

    h_var = cp.Variable(n_points)

    objective = cp.Minimize(cp.sum_squares(h_var - h))

    constraints = [cp.sum(h_var) == target_sum, h_var >= 0, h_var <= 1]

    problem = cp.Problem(objective, constraints)

    problem.solve(solver=cp.ECOS, verbose=False)

    if problem.status != cp.OPTIMAL:

        return np.clip(h, 0, 1)

    return h_var.value.copy()
