import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    max_iterations = int(budget_s * 100)
    gamma = 0.05
    patience = 0
    max_patience = 200

    for iter in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break

        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(overlaps)
        k_star = max_idx - (n_points - 1)

        # Compute gradient for worst k
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k_star
            if 0 <= i_plus_k < n_points:
                term1 = 1 - h[i_plus_k]
            else:
                term1 = 1.0
            i_minus_k = i - k_star
            if 0 <= i_minus_k < n_points:
                term2 = h[i_minus_k]
            else:
                term2 = 0.0
            grad[i] = (term1 - term2) * dx

        # Gradient step and projection
        h_temp = h - gamma * grad
        h_temp = np.clip(h_temp, 0, 1)
        current_sum = np.sum(h_temp)
        if current_sum > 1e-8:
            h_temp = h_temp * (n_points / 2) / current_sum
        h = h_temp

        # Track best solution
        overlaps_current = np.correlate(h, 1 - h, mode='full') * dx
        c5_current = np.max(overlaps_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                gamma = max(gamma / 2, 1e-3)
                patience = 0

    overlaps_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(overlaps_final), n_points
