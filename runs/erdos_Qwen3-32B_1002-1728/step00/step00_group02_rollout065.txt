import numpy as np
import time
from scipy.interpolate import interp1d

def project_feasible(h, n_points):
    h_clipped = np.clip(h.copy(), 0, 1)
    total = np.sum(h_clipped)
    if total == 0:
        return h.copy()
    scale = (n_points / 2.0) / total
    return h_clipped * scale
