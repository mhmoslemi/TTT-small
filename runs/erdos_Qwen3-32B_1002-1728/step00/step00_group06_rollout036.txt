import time

def run(seed=42, budget_s=1100):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = None
    start_time = time.time()
    no_improvement = 0

    while time.time() - start_time < budget_s - 0.1:
        # Compute correlations
        other = 1.0 - h
        correlations = np.correlate(h, other, mode="full")
        max_corr = np.max(correlations)
        q_max = np.argmax(correlations)
        k = q_max - (n_points - 1)

        # Compute gradient
        g = np.zeros(n_points)
        for m in range(n_points):
            if 0 <= m + k < n_points:
                g[m] += 1.0 - h[m + k]
            if 0 <= m - k < n_points:
                g[m] -= h[m - k]

        # Update h
        h -= learning_rate * g * dx
        h = np.clip(h, 0.0, 1.0)

        # Normalize to constraint
        current_sum = h.sum()
        if current_sum == 0:
            h = best_h.copy()
            continue
        desired_sum = n_points / 2.0
        scale = desired_sum / current_sum
        h *= scale
        h = np.clip(h, 0.0, 1.0)

        # Evaluate current C5
        correlations_new = np.correlate(h, 1.0 - h, mode="full")
        c5_candidate = np.max(correlations_new * dx)

        # Update best solution
        if best_c5 is None or c5_candidate < best_c5:
            best_h = h.copy()
            best_c5 = c5_candidate
            no_improvement = 0
        else:
            no_improvement += 1

        # Adjust learning rate
        if no_improvement > 50:
            learning_rate *= 0.5
            no_improvement = 0
            if learning_rate < 1e-6:
                break

    return best_h, best_c5, n_points
