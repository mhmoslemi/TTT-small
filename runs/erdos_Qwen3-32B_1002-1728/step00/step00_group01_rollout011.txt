import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    z_initial = 2.0 * h_initial - 1.0

    z = cp.Variable(n_points)
    c = cp.Variable()

    constraints = []
    constraints.append(cp.sum(z) == 0)

    for i in range(n_points):
        constraints.append(z[i] >= -1.0)
        constraints.append(z[i] <= 1.0)

    # m=0 case: sum(z^2) >= c * n_points
    sum_z_sq = cp.sum([z[i] ** 2 for i in range(n_points)])
    constraints.append(sum_z_sq >= c * n_points)

    # Add shift constraints for m=1 to n-1
    for m in range(1, n_points):
        terms = [z[i] * z[i + m] for i in range(n_points - m)]
        corr = cp.sum(terms)
        constraints.append(corr >= c * (n_points - m))

    objective = cp.Maximize(c)
    problem = cp.Problem(objective, constraints)
    z.value = z_initial  # Provide initial guess

    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
            print("Optimization failed, using initial solution.")
            return h_initial.copy(), 0.38085, n_points
    except Exception:
        print("Error during optimization.")
        return h_initial.copy(), 0.38085, n_points

    h_values = (z.value + 1.0) / 2.0
    h_values = np.clip(h_values, 0.0, 1.0)

    current_sum = h_values.sum()
    adjustment = (n_points / 2.0 - current_sum) / n_points
    h_values += adjustment
    h_values = np.clip(h_values, 0.0, 1.0)

    dx = 2.0 / n_points
    one_minus_h = 1.0 - h_values
    cross_corr = np.correlate(h_values, one_minus_h, mode='full')
    c5_bound = np.max(cross_corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Budget exceeded.")

    return h_values, c5_bound, n_points
