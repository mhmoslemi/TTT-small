from scipy.optimize import minimize, differential_evolution

    start_time = time.time()

        h_initial = initial_h_values.copy()

    except NameError:

        h_initial = np.full(n_points, 0.5)

    def objective(h):

        corr = np.correlate(h, ones_minus_h, mode="full")
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    target_sum = n_points / 2.0
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)

    best_h = None
    best_c5 = float('inf')

    def try_optimize(optimizer_func, h0, method, **kwargs):
        nonlocal best_h, best_c5
        result = optimizer_func(h0, method=method, bounds=bounds, constraints=constraints, **kwargs)
        if result.success:
            h_candidate = np.clip(result.x, 0.0, 1.0)
            sum_adj = target_sum - np.sum(h_candidate)
            h_candidate[0] += sum_adj * (h_candidate[0] > 0.0)
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            c5_val = objective(h_candidate)
            if c5_val < best_c5:
                best_h = h_candidate
                best_c5 = c5_val
        return result

    try_optimize(minimize, h_initial, 'SLSQP', options={'maxiter': 300})

    while time.time() - start_time < budget_s * 0.9:

    elapsed = time.time() - start_time
