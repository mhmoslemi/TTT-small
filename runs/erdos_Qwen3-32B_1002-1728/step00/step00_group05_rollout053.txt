import scipy.optimize as opt
from time import time

def run(seed=42, budget_s=1100, **kwargs):

    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n

        return np.max(corr) * dx

    def optimize_h(n, prev_h):
        target_sum = n / 2.0
        # Interpolate and scale to match integral constraint
        x = np.linspace(0, 2, len(prev_h))

        h0 = np.interp(x_new, x, prev_h)
        current_sum = h0.sum()
        h0 *= target_sum / current_sum
        h0 = np.clip(h0, 0, 1)

        cons = ({'type': 'eq', 'fun': lambda h: h.sum() - target_sum})
