    constraints.append(cp.sum(z_var) == 0)
    constraints.extend([z_var[i] <= 1 for i in range(n_points)])
    constraints.extend([z_var[i] >= -1 for i in range(n_points)])

    for m in range(n_points):
        terms = [z_var[i] * z_var[i + m] for i in range(n_points - m)]
        constraints.append(cp.sum(terms) >= c_var * (n_points - m))

    objective = cp.Maximize(c_var)
    problem = cp.Problem(objective, constraints)

    try:
        remaining_time = max(0, budget_s - (time.time() - start_time))
        problem.solve(solver=cp.ECOS, verbose=False, ecos_args={"max_iter": 5000})
    except:
        h_values = initial_h_values
        c5_bound = evaluate_erdos_solution(h_values, n_points)
        return h_values, c5_bound, n_points

    h_values = (np.clip(z_var.value, -1, 1) + 1) / 2
    h_sum = np.sum(h_values)
    h_values = h_values * (n_points / 2) / h_sum
    h_values = np.clip(h_values, 0, 1)

    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_values = initial_h_values
    c5_bound = evaluate_erdos_solution(h_values, n_points)

    return h_values.tolist(), c5_bound, n_points
