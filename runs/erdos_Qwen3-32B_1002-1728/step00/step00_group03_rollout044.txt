    return np.max(overlap) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h_sum = initial_h.sum()
        if not np.isclose(initial_h_sum, n_points / 2):
            initial_h = initial_h / initial_h_sum * (n_points / 2)
    except:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / initial_h.sum() * (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)
    
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        total = h.sum()
        if total < 1e-6:
            h = np.random.rand(n_points)
        h = h / total * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        population.append(h)
    
    def evaluate_population(pop):
        return [compute_c5(h, dx) for h in pop]
    
    best_c5 = min(evaluate_population(population))
    best_h = population[np.argmin(evaluate_population(population))].copy()
    
    start_time = time.time()
    stagnation = 0
    
    while time.time() - start_time < budget_s and stagnation < 200:
        tournament = np.random.choice(population, size=int(0.3*population_size), replace=False)
        scores = np.array([compute_c5(h, dx) for h in tournament])
        best_indices = np.argsort(scores)[:int(0.5*len(tournament))]
        selected = [tournament[i] for i in best_indices]
        
        new_candidates = []
        for _ in range(population_size // 2):
            p1 = selected[np.random.randint(len(selected))]
            p2 = selected[np.random.randint(len(selected))]
            child = crossover(p1, p2)
            if np.random.rand() < 0.1:
                child = mutate(child)
            new_candidates.append(child)
        
        population = population + new_candidates
        scores = [compute_c5(h, dx) for h in population]
        rankings = np.argsort(scores)
        population = [population[i] for i in rankings[:population_size]]
        
        current_c5 = scores[rankings[0]]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0].copy()
            stagnation = 0
        else:
            stagnation += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * (n_points / 2)
    return (best_h, best_c5, n_points)
