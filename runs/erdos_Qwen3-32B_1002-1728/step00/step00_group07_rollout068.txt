def repair_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-8:
        return h_clipped
    delta = target_sum - current_sum
    non_clipped = np.where((h_clipped > 0) & (h_clipped < 1))[0]

        return h
