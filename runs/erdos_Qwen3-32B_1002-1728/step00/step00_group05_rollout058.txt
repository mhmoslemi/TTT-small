import numpy as np
import time

def project_to_constraints(h, n_points):
    h = np.clip(h, 0, 1).copy()
    total = np.sum(h)
    target = n_points / 2.0
    if abs(total - target) < 1e-8:
        return h
    delta = target - total
    if delta > 0:
        indices = np.where(h < 1)[0]
        if len(indices):
            amount = delta / len(indices)
            h[indices] += amount
            h[indices] = np.clip(h[indices], 0, 1)
    else:
        indices = np.where(h > 0)[0]
        if len(indices):
            amount = (-delta) / len(indices)
            h[indices] -= amount
            h[indices] = np.clip(h[indices], 0, 1)
    return h

def evaluate_candidate(h, n_points):
    dx = 2.0 / n_points
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    return np.max(corr) * dx
