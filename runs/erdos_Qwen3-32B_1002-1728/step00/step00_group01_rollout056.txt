import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()
    n_points = 47  # Use 47 to leverage initial_h_values if available

    try:
        initial_h = initial_h_values
        z_initial = 2 * initial_h - 1
        z_initial = z_initial - np.mean(z_initial) * n_points  # Ensure sum(z) = 0
        z_initial = np.clip(z_initial, -1, 1)
    except NameError:
        np.random.seed(seed)
        z_initial = np.random.uniform(-1, 1, n_points)
        z_initial = z_initial - np.mean(z_initial) * n_points
        z_initial = np.clip(z_initial, -1, 1)

    z = cp.Variable(n_points)
    c = cp.Variable()
    constraints = [
        z >= -1,
        z <= 1,
        cp.sum(z) == 0
    ]

    for m in range(1, n_points):
        expr = sum(z[i] * z[i + m] for i in range(n_points - m))
        constraints.append(expr >= c)

    objective = cp.Maximize(c)
    problem = cp.Problem(objective, constraints)

    best_h = None
    best_c5 = float('inf')

    try:
        problem.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            z_val = z.value
            h = (z_val + 1) / 2
            h = h * (n_points / 2) / h.sum()
            h = np.clip(h, 0, 1)
            c5 = evaluate(h, n_points)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    except Exception:
        pass

    if best_h is None:
        if 'initial_h' in locals():
            best_h = initial_h
        else:
            h = (z_initial + 1) / 2
            h = h * (n_points / 2) / h.sum()
        best_c5 = evaluate(best_h, n_points)

    return (best_h.tolist(), best_c5, n_points)

def evaluate(h, n_points):
    h = np.array(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
