def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h_vals = h_flat.reshape(-1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr * dx)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    if not np.isclose(np.sum(initial_h), n_points / 2):
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    result = minimize(
        objective, initial_h, method='SLSQP', bounds=bounds,
        constraints=constraints, options={'maxiter': 20, 'ftol': 1e-6, 'disp': False}
    )

    optimized_h = result.x
    best_c5 = evaluate_erdos_solution(optimized_h)

    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        next_result = minimize(
            objective, optimized_h, method='SLSQP', bounds=bounds,
            constraints=constraints, options={'maxiter': 10, 'ftol': 1e-6, 'disp': False}
        )
        new_c5 = evaluate_erdos_solution(next_result.x)
        if new_c5 < best_c5:
            best_c5 = new_c5
            optimized_h = next_result.x
        if iter_count > 3:
            break  # Prevent infinite refinement

    return (optimized_h, best_c5, n_points)
