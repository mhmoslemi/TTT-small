import numpy as np
import time

def compute_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):

    else:
        n_points = 70

    if not np.allclose(np.sum(h), n_points / 2):
        diff = np.sum(h) - n_points / 2
        h += -diff / n_points
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    current_h = h.copy()
    current_C5 = best_C5
    
    T = 1.0
    alpha = 0.99
    
    while time.time() - start_time < budget_s - 10:
        for _ in range(100):

            max_negative = -min(1.0 - original_i, original_j)

            epsilon = np.clip(epsilon, max_negative, max_positive)
