import numpy as np

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def repair(ind, target_sum, n_points):
    ind = np.clip(ind, 0, 1)
    current_sum = np.sum(ind)
    delta = target_sum - current_sum
    if delta != 0:
        for i in range(n_points):
            if delta > 0 and ind[i] < 1:
                add = min(delta, 1 - ind[i])
                ind[i] += add
                delta -= add
            elif delta < 0 and ind[i] > 0:
                subtract = min(-delta, ind[i])
                ind[i] -= subtract
                delta += subtract
            if delta == 0:
                break
    # Final check in case of failure
    if np.abs(np.sum(ind) - target_sum) > 1e-8:
        # This should not happen, but just in case, add a small adjustment
        factor = target_sum / np.sum(ind)
        ind = ind * factor
        ind = np.clip(ind, 0, 1)
    return ind

def create_initial_individual(initial_h, target_sum, n_points):
