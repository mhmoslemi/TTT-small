def repair_individual(h, desired_sum):
    h_clamped = np.clip(h, 0.0, 1.0)
    current_sum = h_clamped.sum()
    if np.isclose(current_sum, 0.0):
        h_clamped[:] = 0.5
        return h_clamped
    scale = desired_sum / current_sum
    h_scaled = h_clamped * scale
    h_final = np.clip(h_scaled, 0.0, 1.0)
    current_sum_final = h_final.sum()
    delta = desired_sum - current_sum_final
    if abs(delta) < 1e-8:
        return h_final
    idx = np.random.randint(0, h_final.size)
    old_val = h_final[idx]
    new_val = min(1.0, max(0.0, old_val + delta))
    h_final[idx] = new_val
    return h_final

def mutate(h, mutation_rate, n_points):
    h = h.copy()
    if np.random.rand() < mutation_rate:
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h[idx1], h[idx2] = h[idx2], h[idx1]
    return h

def crossover(parent1, parent2, crossover_rate, n_points, desired_sum):
