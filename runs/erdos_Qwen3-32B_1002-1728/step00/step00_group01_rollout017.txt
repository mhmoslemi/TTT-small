    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    current_sum = h.sum()
    adjustment = (n_points / 2) - current_sum
    h += adjustment / n_points
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    T = 1000.0
    alpha = 0.999
    while time.time() - start_time < budget_s - 0.1:
        try:
            i, j = np.random.choice(n_points, size=2, replace=False)
        except:
            continue
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        delta = np.random.uniform(0, max_delta)
        h_prev_i = h[i]
        h_prev_j = h[j]
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        delta_energy = new_c5 - current_c5
        if delta_energy < 0:
            current_c5 = new_c5
        else:
            if np.random.rand() < np.exp(-delta_energy / T):
                current_c5 = new_c5
            else:
                h[i] = h_prev_i
                h[j] = h_prev_j
        T *= alpha
    return best_h.tolist(), best_c5, n_points
