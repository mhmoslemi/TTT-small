import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = initial_h_values.size
    h = initial_h_values.copy()
    dx = 2.0 / n_points

    def compute_C5(h_eval):
        return np.max(np.correlate(h_eval, 1 - h_eval, mode='full') * dx)

    def project_feasible(h_proj):
        h_clipped = np.clip(h_proj, 0, 1)
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0:
            return initial_h_values.copy()
        scale = (n_points / 2) / sum_clipped
        return np.clip(h_clipped * scale, 0, 1)

    def perturb(h_perturb):
        new_h = h_perturb.copy()
        strategy = np.random.choice(['single', 'segment', 'swap'], p=[0.5, 0.3, 0.2])
        if strategy == 'single':
            i = np.random.randint(n_points)
            new_h[i] += np.random.uniform(-0.1, 0.1)
        elif strategy == 'segment':
            len_seg = np.random.randint(2, min(11, n_points))
            start = np.random.randint(0, n_points - len_seg + 1)
            end = start + len_seg
            segment = new_h[start:end]
            shift = np.random.choice([-1, 1])
            new_h[start:end] = np.roll(segment, shift)
        else:  # swap
            i, j = np.random.choice(n_points, 2, replace=False)
            new_h[i], new_h[j] = h_perturb[j], h_perturb[i]
        return new_h

    best_h = h.copy()
    best_C5 = compute_C5(h)

    T = 1.0
    alpha = 0.95
    max_iter_per_T = 100
    no_improvement = 0

    while time.time() - start_time < budget_s:
        improved = False
        for _ in range(max_iter_per_T):
            if time.time() - start_time > budget_s:
                break
            new_h = perturb(h)
            new_h = project_feasible(new_h)
            new_C5 = compute_C5(new_h)
            delta = new_C5 - best_C5
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                h = new_h
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h.copy()
                    improved = True
                    no_improvement = 0
