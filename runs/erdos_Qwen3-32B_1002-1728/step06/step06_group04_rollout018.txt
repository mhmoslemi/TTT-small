        if current_sum > 1e-10:
            return vec_clipped * target_sum / current_sum
        return np.full(n_points, target_sum / n_points)

    h = clip_and_rescale(h_initial)

        constraint_violation = h.sum() - target_sum

        return overlap + λ * constraint_violation + ρ / 2 * constraint_violation ** 2

def calculate_c5(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
