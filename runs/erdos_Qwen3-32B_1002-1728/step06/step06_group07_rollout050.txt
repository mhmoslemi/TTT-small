            max_corr = np.max(corr_vals)
            fitness.append(max_corr * dx)
        fitness = np.array(fitness)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]

        top_percent = int(0.2 * pop_size)
        new_population = list(sorted_population[:top_percent])

        remaining = pop_size - top_percent
        while len(new_population) < pop_size:
            top_half_size = pop_size // 2
            parent1 = sorted_population[np.random.randint(top_half_size)].copy()
            parent2 = sorted_population[np.random.randint(top_half_size)].copy()

            p1 = parent1.astype(bool)
            p2 = parent2.astype(bool)
            common = np.logical_and(p1, p2)
            diff = np.logical_xor(p1, p2)
            diff_indices = np.where(diff)[0]
            if len(diff_indices) == 0:
                child = parent1.copy()
            else:
                np.random.shuffle(diff_indices)
                x = len(diff_indices) // 2
                selected = diff_indices[:x]
                child = np.zeros(n_points, dtype=bool)
                child[common] = True
                child[selected] = True
                child = child.astype(float)
            if np.random.rand() < 0.1:
                ones = np.where(child == 1.0)[0]
                zeros = np.where(child == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = np.random.choice(ones)
                    j = np.random.choice(zeros)
                    child[i], child[j] = child[j], child[i]
            new_population.append(child)
        population = new_population

    best_binary = sorted_population[0]
    h0 = best_binary.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        lambda x: evaluate_real(x, T, n_points, dx),
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        final_h = result.x
    else:
        final_h = best_binary

    final_h = np.clip(final_h, 0, 1)
    current_sum = np.sum(final_h)
    if abs(current_sum - T) > 1e-6:
        final_h = (final_h * T / current_sum).clip(0, 1)

    corr_vals = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr_vals) * dx

    return (final_h.tolist(), final_c5, n_points)
