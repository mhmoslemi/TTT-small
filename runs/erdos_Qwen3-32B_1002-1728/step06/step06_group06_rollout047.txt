import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):

        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    np.random.seed(seed)
    n_candidates = 30
    candidates = []
    for _ in range(n_candidates):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + noise
        h_perturbed = clip_and_rescale(h_perturbed)
        candidates.append(h_perturbed)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
