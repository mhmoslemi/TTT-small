        de_h = np.clip(de_h, 0, 1)
        best_h = de_h.copy()
        best_c5 = float(objective(de_h))

    def refine_candidate(x0):
        x0 = np.clip(x0, 0, 1)
        x0_total = np.sum(x0)
        if x0_total > 0:
            x0 = (x0 / x0_total) * target_sum
        x0 = np.clip(x0, 0, 1)
        result = minimize(
            objective, x0=x0, method='SLSQP', bounds=bounds_per_var, constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        return np.clip(result.x if result.success else x0, 0, 1)

    while time.time() - start_time < budget_s - 15:
        perturbation = np.random.randn(n_points) * 0.08
        new_h = np.clip(best_h + perturbation, 0, 1)
        refined_h = refine_candidate(new_h)
        refined_c5 = objective(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h.copy()
            best_c5 = refined_c5

    # Final validation
    h_full = np.clip(best_h, 0, 1)
    total = np.sum(h_full)
    if abs(total - target_sum) > 1e-6:
        h_full = (h_full / total) * target_sum
    h_full = np.clip(h_full, 0, 1)
    final_c5 = float(objective(h_full))

    return (h_full.tolist(), final_c5, n_points)
