    # Construct design matrix for Fourier terms
    A = np.zeros((n_points, 2*N))
    for k in range(N):
        # Cosine terms
        col = 2*k
        A[:, col] = np.cos((k+1) * np.pi * x_vals)
        # Sine terms
        col = 2*k + 1
        A[:, col] = np.sin((k+1) * np.pi * x_vals)

    # Compute initial coefficients using least squares
    y_initial = h_initial - 0.5
    coeffs_initial = np.linalg.lstsq(A, y_initial, rcond=None)[0]

    a_initial = coeffs_initial[::2]
    b_initial = coeffs_initial[1::2]

    # Define variables for optimization: a_1 to a_N, b_1 to b_N

    b_vars = cp.Variable(N)

    constraints = [h_samples >= 0, h_samples <= 1]

    objective = cp.Minimize(cp.sum_squares(a_vars) + cp.sum_squares(b_vars))

    # Solve with MOSEK if available
    problem = cp.Problem(objective, constraints)

        problem.solve(solver=cp.SCS, verbose=False)
