            method='L-BFGS-B',
            bounds=[(0, 1)] * half_len,
            constraints={'type': 'eq', 'fun': sum_constraint},
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res_h.success and np.isfinite(res_h.fun):
            h_half_opt = res_h.x
            h_full_opt = np.zeros(n_points)
            h_full_opt[:half_len] = h_half_opt
            h_full_opt[-half_len:] = h_half_opt[::-1]
            h_sum = np.sum(h_full_opt)
            if abs(h_sum - target_sum) < 1e-6:
                new_c5 = np.max(np.correlate(h_full_opt, 1 - h_full_opt, mode='full')) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_full_opt

    # Fallback: refine best solution
    h_full_best = best_h if (best_c5 < np.inf) else h_initial
    corr_vals = np.correlate(h_full_best, 1 - h_full_best, mode='full')
    final_c5 = np.max(corr_vals) * dx

    return (h_full_best.tolist(), final_c5, n_points)
