import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    n_particles = 40
    max_iter_pso = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5

    np.random.seed(seed)
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5s = []

    for _ in range(n_particles):
        pos = h_initial + np.random.normal(0, 0.1, n_points)
        pos = clip_and_rescale(pos)
        vel = np.random.normal(0, 0.05, n_points)
        particles.append(pos)
        velocities.append(vel)
        pb = pos.copy()
        pbest_c5 = np.max(np.correlate(pb, 1 - pb, mode='full')) * dx
        personal_best.append(pb)
        personal_best_c5s.append(pbest_c5)

    best_global = h_initial.copy()
    best_global_c5 = personal_best_c5s[0]
    for i in range(personal_best_c5s[0]):
        if personal_best_c5s[i] < best_global_c5:
            best_global = personal_best[i].copy()
            best_global_c5 = personal_best_c5s[i]

    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best[i] - particles[i]) +
                             social * r2 * (best_global - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = clip_and_rescale(new_pos)
            particles[i] = new_pos
            corr_vals = np.correlate(new_pos, 1 - new_pos, mode='full')
            c5_new = np.max(corr_vals) * dx
            if c5_new < personal_best_c5s[i]:
                personal_best[i] = new_pos.copy()
                personal_best_c5s[i] = c5_new
                if c5_new < best_global_c5:
                    best_global = new_pos.copy()
                    best_global_c5 = c5_new

    top_candidates = sorted(zip(personal_best, personal_best_c5s), key=lambda x: x[1])[:10]

    for candidate, _ in top_candidates:
        if time.time() - start_time > budget_s - 10:
            break

        def obj(x):
            x = np.reshape(x, (-1,))
            x_proj = clip_and_rescale(x)
            corr_vals = np.correlate(x_proj, 1 - x_proj, mode='full')
            return np.max(corr_vals) * dx

        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            fun=obj, x0=candidate, method='SLSQP', bounds=bounds,
            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
        )
        if res.success:
            x_opt = clip_and_rescale(res.x)
            corr_vals = np.correlate(x_opt, 1 - x_opt, mode='full')
            c5_opt = np.max(corr_vals) * dx
            if c5_opt < best_global_c5:
                best_global = x_opt.copy()
                best_global_c5 = c5_opt

    h_final = best_global
    return (h_final.tolist(), best_global_c5, n_points)
