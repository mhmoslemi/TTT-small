import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h) = n_points / 2.0 for full h

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s < 1e-8:
            h_half[:] = target_sum_half / half_n
        else:
            h_half = h_half / s * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, fit, k=3):
        idx = np.random.choice(len(pop), k, replace=False)
        winner_idx = idx[np.argmin([fit[i] for i in idx])]
        return pop[winner_idx].copy()

    def create_population(initial_half, size=20):
        np.random.seed(seed)
        population = []
        for i in range(size):
            p = initial_half.copy() if i == 0 else np.random.normal(loc=initial_half, scale=0.1, size=half_n)
            population.append(repair_half(p))
        return population

    initial_half = h_initial[:half_n].copy()
    population = create_population(initial_half, size=20)
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic Algorithm Phase
    pop_size = 20
    max_gen = 300
    mutation_rate = 0.1
    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.8:
            break
        new_pop = [best_h_half.copy()]
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            child = p1.copy()
            # Crossover
            if np.random.rand() < 0.3:
                cp = np.random.randint(1, half_n)
                child[:cp] = p2[:cp]
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate = np.random.choice(half_n, round(half_n * 0.2), replace=False)
                child[mutate] += np.random.normal(0, 0.1, mutate.size)
                child[mutate] = np.clip(child[mutate], 0, 1)
            child = repair_half(child)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

    # Gradient Refinement Phase
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0.0, 1.0)] * half_n
    result = minimize(compute_c5, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    if result.success:
        refined = repair_half(result.x)
        c5 = compute_c5(refined)
        if c5 < best_c5:

    h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
