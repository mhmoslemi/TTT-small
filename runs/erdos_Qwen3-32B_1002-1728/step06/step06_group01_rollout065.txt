        method_idx = (large_iter * 2) % 4
        method = 'L-BFGS-B' if method_idx == 0 else ('Nelder-Mead' if method_idx == 1 else 'SLSQP')

        res = minimize(
            objective, x0=perturbed_half, method=method,
            bounds=bounds_per_var, constraints=sum_cons,
            tol=1e-6 if method != 'Nelder-Mead' else 1e-5,
            options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        )
        if res.success:
            new_h_half = np.clip(res.x, 0, 1)
            new_h = create_full_h(new_h_half)
            new_h = np.clip(new_h, 0, 1)
            if abs(np.sum(new_h) - n_points / 2.0) < 1e-6:
                new_c5 = objective(new_h_half)
                if new_c5 < best_c5 - recent_improvement_tol:
                    best_h = new_h
                    best_c5 = new_c5
                    patience_counter = 0
                else:
                    patience_counter += 1
            else:
                patience_counter += 1
        else:
            patience_counter += 1

    h_full = np.clip(best_h, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
