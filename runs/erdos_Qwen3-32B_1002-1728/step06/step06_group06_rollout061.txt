import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return np.zeros_like(h)
    scale = target_sum / current_sum
    return np.clip(h_clipped * scale, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = clip_and_rescale(h_initial, target_sum)

    np.random.seed(seed)
    candidates = [h_initial.copy()]
    perturbation_scale = 0.1
    for _ in range(24):
        perturb = h_initial + np.random.normal(0, perturbation_scale, n_points)
        perturb = clip_and_rescale(perturb, target_sum)
        candidates.append(perturb)

    def objective(h_params):
        h = np.clip(h_params, 0, 1)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    best_c5 = np.inf
    best_h = None

    for candidate in candidates:
        if time.time() - start_time > budget_s - 60:
            break
        x0 = candidate.copy()
        result = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=[(0,1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 200, 'gtol': 1e-10, 'ftol': 1e-12, 'disp': False}
        )
        if result.success:
            h_opt = clip_and_rescale(result.x, target_sum)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()

    h_final = best_h if best_h is not None else h_initial
    c5_final = best_c5 if best_c5 < np.inf else np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    return (h_final.tolist(), c5_final, n_points)
