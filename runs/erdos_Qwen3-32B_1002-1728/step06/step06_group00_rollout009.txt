                    population[i] = (population[i] / population[i].sum()) * target_sum_half

    # Local optimize best solution
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    h_full = (h_full / h_full.sum()) * target_sum_full

    def area(x):
        x = np.clip(x, 0, 1).copy()
        x = x[:half_n]
        x_full = np.zeros(n_points)
        x_full[:half_n] = x
        x_full[-half_n:] = x[::-1]
        return x_full

    def objective(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return np.inf
        x = (x / np.sum(x)) * target_sum_half
        x_full = area(x)
        corr = np.correlate(x_full, 1 - x_full, mode='full') * dx
        return np.max(corr)

    bounds_local = [(0, 1)] * half_n
    result = minimize(objective, x0=best_h_half, bounds=bounds_local,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum_half}],
                     method='L-BFGS-B', options={'maxiter': 50, 'ftol': 1e-8})
    
    if result.success:
        best_h_half = result.x
        h_final = area(best_h_half)
    else:
        h_final = area(best_h_half)

    # Final repair
    h_final = np.clip(h_final, 0, 1)
    h_final_sum = h_final.sum()
    if h_final_sum != 0:
        h_final = (h_final / h_final_sum) * target_sum_full
        h_final = np.clip(h_final, 0, 1)
        
    h_final_corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5_bound = np.max(h_final_corr) * dx
    return (h_final.tolist(), c5_bound, n_points)
