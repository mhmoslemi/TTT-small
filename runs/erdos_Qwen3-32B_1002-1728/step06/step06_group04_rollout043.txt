from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Prepare initial h with correct constraints
    h_full_clipped = np.clip(h_full_initial, 0.0, 1.0)
    current_sum = h_full_clipped.sum()
    if current_sum > 1e-10:
        h_full_clipped *= target_sum / current_sum
    else:
        h_full_clipped[:] = 0.5
    best_h = h_full_clipped.copy()
    best_c5 = calculate_c5(h_full_clipped, dx)

    np.random.seed(seed)

    # Differential Evolution step
    bounds_de = [(0.0, 1.0) for _ in range(n_points)]

    def objective_de(x):
        h = np.array(x)
        total = h.sum()
        if total < 1e-10:
            h[:] = target_sum / n_points
        else:
            h = h * (target_sum / total)
        return calculate_c5(h, dx)

    # Check remaining time
    if time.time() - start_time > budget_s - 30:

    # Run DE
    result_de = differential_evolution(
        func=objective_de,
        bounds=bounds_de,
        strategy='best1bin',
        maxiter=20,
        popsize=15,  # Total of 1050 individuals for n=70
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        polish=False,
        disp=False
    )

    # Get h from DE result
    x_de = result_de.x
    h_de = np.array(x_de)
    total_de = h_de.sum()

    de_c5 = calculate_c5(h_de, dx)

        best_c5 = de_c5

    # Starting point
    x0 = h_de.copy()
