            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best_pos[i] - positions[i]) + c2 * r2 * (gbest_pos - positions[i])
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            positions[i] += velocities[i]
            positions[i] = process_h_half(positions[i])
            score = pso_obj(positions[i])
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_pos[i] = positions[i].copy()
                if score < best_score:
                    best_score = score
                    gbest_pos = positions[i].copy()
            if time.time() - start_time >= budget_s * 0.6:
                break

    def slsqp_obj(x):
        x = process_h_half(x)
        h_full = create_full(x)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0, 1)] * half_n
    res = minimize(slsqp_obj, x0=gbest_pos, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
    final_h_half = res.x if res.success else gbest_pos
    final_h_half = process_h_half(final_h_half)
    final_score = pso_obj(final_h_half)

    h_full = create_full(final_h_half)
    h_sum = np.sum(h_full)
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
    h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), final_score, n_points)
