import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate Frank-Zadoff-like sequence of length 64
    N = 64
    i_values = np.arange(N, dtype=np.float64)
    mod_values = (2 * i_values + 1) % N
    s = np.cos(np.pi * mod_values / N)
    b = np.where(s > 0, 1.0, 0.0)
    h_initial = 0.5 * (b + 1.0)

    # Interpolate to n_points=70
    x_orig = np.linspace(0, 2, N)
    x_new = np.linspace(0, 2, n_points)
    h_initial = np.interp(x_new, x_orig, h_initial)
    h_initial = np.clip(h_initial, 0.0, 1.0)

    # Scale to target_sum
    current_sum = np.sum(h_initial)
    if current_sum != 0:
        h_initial *= (target_sum / current_sum)
    h_initial = np.clip(h_initial, 0.0, 1.0)

    # Objective function for optimization
    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    # Constraint to enforce sum(x) = target_sum
    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Run optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    best_c5 = 0.380876  # Placeholder value
    h_full = h_initial

    if result.success and (time.time() - start_time) < budget_s - 10:
        optimized_h = np.clip(result.x, 0.0, 1.0)
        h_full = optimized_h
        best_c5 = objective(h_full)

    # Final validation of constraints
    h_full = np.clip(h_full, 0.0, 1.0)
    h_full_sum = np.sum(h_full)
    if not np.isclose(h_full_sum, target_sum, atol=1e-6):
        h_full = np.full(n_points, 0.5)

    return (h_full.tolist(), best_c5, n_points)
