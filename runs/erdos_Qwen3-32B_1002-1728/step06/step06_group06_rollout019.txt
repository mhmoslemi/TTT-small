import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    # PSO parameters
    n_particles = 30
    max_iter_pso = 30
    inertia = 0.72984  # Standard value for PSO
    cognitive = 1.49445
    social = 1.49445

    best_global = h_initial.copy()
    best_global_c5 = np.max(np.correlate(best_global, 1 - best_global, mode='full')) * dx

    particles = []
    velocities = []
    personal_best = []
    personal_best_c5s = []

    np.random.seed(seed)
    for _ in range(n_particles):
        pos = h_initial + np.random.normal(0, 0.1, n_points)
        pos = clip_and_rescale(pos)
        vel = np.random.normal(0, 0.05, n_points)
        particles.append(pos)
        velocities.append(vel)
        pb = pos.copy()
        pbest_c5 = np.max(np.correlate(pb, 1 - pb, mode='full')) * dx
        personal_best.append(pb)
        personal_best_c5s.append(pbest_c5)
        if pbest_c5 < best_global_c5:
            best_global = pb.copy()
            best_global_c5 = pbest_c5

    no_improve = 0
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s - 120:
            break
        prev_best_c5 = best_global_c5
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best[i] - particles[i]) +
                            social * r2 * (best_global - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = clip_and_rescale(new_pos)
            particles[i] = new_pos
            corr_vals = np.correlate(new_pos, 1 - new_pos, mode='full')
            c5_new = np.max(corr_vals) * dx
            if c5_new < personal_best_c5s[i]:
                personal_best[i] = new_pos.copy()
                personal_best_c5s[i] = c5_new
                if c5_new < best_global_c5:
                    best_global = new_pos.copy()
                    best_global_c5 = c5_new
        if abs(prev_best_c5 - best_global_c5) < 1e-8:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve >= 7:
            break

    # Select top 8 candidates for local search
    all_candidates = list(zip(personal_best, personal_best_c5s))
    all_candidates.sort(key=lambda x: x[1])
    top_candidates = all_candidates[:8]

        def objective(x):
            x = np.reshape(x, (-1,))

            return np.max(corr_vals) * dx
