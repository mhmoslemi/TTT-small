    current_h = best_h_half.copy()
    
    for step in range(steps):
        if time.time() - start_time > budget_s - 10:
            break
            
        h_full = create_full_h(current_h)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_idx = np.argmax(corr)
        
        grad = np.zeros(half_n)
        for i in range(half_n):
            # Forward perturbation
            h_up = current_h.copy()
            h_up[i] = min(1.0, h_up[i] + eps)
            h_up = repair_h(h_up)
            up_corr = np.correlate(create_full_h(h_up), 1 - create_full_h(h_up), mode='full')
            up_val = up_corr[max_idx] * dx
            
            # Backward perturbation
            h_down = current_h.copy()
            h_down[i] = max(0.0, h_down[i] - eps)
            h_down = repair_h(h_down)
            down_corr = np.correlate(create_full_h(h_down), 1 - create_full_h(h_down), mode='full')
            down_val = down_corr[max_idx] * dx
            
            grad[i] = (up_val - down_val) / (2*eps)
        
        # Normalize gradient
        grad /= (np.linalg.norm(grad) + 1e-6)
        
        # Gradient step
        current_h = current_h - lr * grad
        
        # Repair constraints
        current_h = repair_h(current_h)
        
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h_half = current_h
            best_c5 = current_c5

    # Final result
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    final_sum = h_full.sum()
    if abs(final_sum - target_sum_full) > 1e-6:
        h_full = (h_full / final_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
