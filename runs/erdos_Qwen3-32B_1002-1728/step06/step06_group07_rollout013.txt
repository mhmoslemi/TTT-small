def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0  # Target sum of h values

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    # Generate Fourier-based initial guess
    h_initial = clip_and_scale(h_full_initial.copy())
    # Compute FFT and truncate
    h_fft = np.fft.fft(h_initial) / n_points  # Normalize
    n_terms = min(10, n_points // 2)
    h_fft_truncated = np.zeros_like(h_fft)
    h_fft_truncated[:n_terms] = h_fft[:n_terms]
    h_fft_truncated[-(n_terms - 1):] = h_fft[-(n_terms - 1):]  # Maintain symmetry
    h_fourier = np.fft.ifft(h_fft_truncated * n_points).real
    h_fourier = clip_and_scale(h_fourier)
    
    # Choose the initial guess with better C5
    c5_full_initial = objective(h_initial)
    c5_fourier = objective(h_fourier)
    if c5_fourier < c5_full_initial:
        best_h = h_fourier.copy()
        best_c5 = c5_fourier
    else:
        best_h = h_initial.copy()
        best_c5 = c5_full_initial

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Differential Evolution phase
    de_result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=8, popsize=12, tol=1e-4, mutation=(0.5, 1),
        recombination=0.85, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        de_c5 = objective(de_h)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h

    # Local refinement with SLSQP
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break

    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), best_c5, n_points)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Differential Evolution phase with full bounds and iterative refinement
    de_result = differential_evolution(
        objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=10,  # Increased iterations for better exploration
        popsize=12,
        tol=1e-5,
        mutation=(0.6, 1),
        recombination=0.9,
        seed=seed,
        workers=1
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        de_c5 = objective(de_h)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h

    # Local optimization
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            options={
                'maxiter': 150,
                'ftol': 1e-8,
                'disp': False
            }
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Final validation
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Constraint violated"
    return (final_h.tolist(), best_c5, n_points)
