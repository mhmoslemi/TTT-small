import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    # Convert initial to binary
    initial_binary = (h_initial > 0.5).astype(float)
    assert abs(np.sum(initial_binary) - T) < 1e-8

    # Initialize population
    pop_size = 50
    population = [initial_binary.copy()]
    np.random.seed(seed)
    for _ in range(pop_size - 1):
        indices = np.random.permutation(n_points)
        sample = np.zeros(n_points)
        sample[indices[:int(T)]] = 1.0
        population.append(sample)

    generations = 100
    for gen in range(generations):
        if time.time() - start_time > budget_s - 60:
            break
        # Evaluate fitness
        fitness = []
        for indiv in population:
            corr_vals = np.correlate(indiv, 1 - indiv, mode='full')
            c5 = np.max(corr_vals) * dx
            fitness.append(c5)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        best_individual = sorted_population[0].copy()
        best_c5 = fitness[0]
        # Select survivors (20%)
        survivors = sorted_population[:int(0.2 * pop_size)]
        new_pop = survivors.copy()
        # Generate new individuals
        while len(new_pop) < pop_size:
            if time.time() - start_time > budget_s - 60:
                break
            parent = survivors[np.random.randint(len(survivors))].copy()
            child = parent.copy()
            # Mutate by swapping two bits
            found = False
            for _ in range(10):
                a, b = np.random.choice(n_points, size=2, replace=False)
                if child[a] != child[b]:
                    child[a], child[b] = child[b], child[a]
                    found = True
                    break
            if found:
                new_pop.append(child)
        population = new_pop

    # Post-GA refinement
    def objective_opt(x):
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return float(np.max(corr_vals) * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        objective_opt, x0=best_individual, method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - T}],
        options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
    )
    if not result.success:
        # Fallback to initial if optimization fails
        final_h = best_individual.copy()
    else:
        final_h = np.clip(result.x, 0, 1)

    # Final validation
    final_sum = final_h.sum()
    assert abs(final_sum - T) < 1e-6, f"Sum violation: {final_sum} vs {T}"
    corr_vals = np.correlate(final_h, 1 - final_h, mode='full')
    c5_final = float(np.max(corr_vals) * dx)
    return (final_h.tolist(), c5_final, n_points)
