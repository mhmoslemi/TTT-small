def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale_factor = T / current_sum
        scaled = np.clip(h_clipped * scale_factor, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    current_h = h_initial.copy()
    best_c5 = objective(current_h)
    best_h = current_h.copy()

    alpha = 0.95  # Dampening factor
    admm_iterations = 50  # Tune this number based on budget

    for iter_idx in range(admm_iterations):
        if time.time() - start_time > budget_s - 300:

        h_temp = np.fft.ifft(current_h_fft).real

        # Evaluate and track best solution

        if current_c5 < best_c5:

            best_c5 = current_c5
            best_h = current_h.copy()

    # Now apply local optimization to refine the solution
    h_opt = best_h.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for _ in range(2):  # 2 rounds of SLSQP to prevent premature termination
        if time.time() - start_time > budget_s - 60:
            break

        result = minimize(
            objective,
            x0=h_opt,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            h_new = clip_and_scale(result.x)
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_opt = h_new.copy()
            else:
                h_opt = h_new.copy()  # Still use this as the next starting point

    # Final check
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"

    return (final_h.tolist(), best_c5, n_points)
