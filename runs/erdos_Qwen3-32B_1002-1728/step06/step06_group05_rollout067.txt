import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def enforce_feasibility(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        return np.clip(x, 0, 1)

    h_current = enforce_feasibility(h_initial)

    def evaluate(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    best_h = h_current.copy()
    best_c5 = evaluate(best_h)

    np.random.seed(seed)
    T0 = 1.0
    alpha = 0.99
    max_iter = 1000
    T = T0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
        # Mutation
        mutation = np.random.normal(0, T, n_points)
        h_new = h_current + mutation
        h_new = enforce_feasibility(h_new)
        c5_new = evaluate(h_new)
        # Metropolis criterion
        delta = c5_new - best_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        T *= alpha

    # Local refinement
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(
        fun=lambda x: evaluate(enforce_feasibility(x)),

    return (final_h.tolist(), final_c5, n_points)
