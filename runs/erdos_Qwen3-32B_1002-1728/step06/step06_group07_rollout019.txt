import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Fourier parameterization with N_terms
    N_terms = 20
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Precompute Fourier basis vectors
    cos_vectors = []
    sin_vectors = []
    for k in range(1, N_terms + 1):
        cos_k = np.cos(k * np.pi * x)
        sin_k = np.sin(k * np.pi * x)
        cos_vectors.append(cos_k)
        sin_vectors.append(sin_k)
    
    # Initialize Fourier coefficients from h_initial
    a_initial = np.zeros(N_terms)
    b_initial = np.zeros(N_terms)
    for k in range(N_terms):
        k_idx = k + 1
        cos_k = np.cos(k_idx * np.pi * x)
        sin_k = np.sin(k_idx * np.pi * x)
        a_initial[k] = np.sum(h_initial * cos_k) * dx
        b_initial[k] = np.sum(h_initial * sin_k) * dx
    
    coeffs_initial = np.concatenate([a_initial, b_initial])
    
    # Build linear constraints: 0 <= h(x_i) <= 1
    A_constraints = []
    for i in range(n_points):
        row = []
        for k in range(N_terms):
            row.append(cos_vectors[k][i])
        for k in range(N_terms):
            row.append(sin_vectors[k][i])
        A_constraints.append(row)
    A_constraints = np.array(A_constraints)
    linear_constraints = LinearConstraint(A_constraints, -0.5, 0.5)
    
    # Objective function computes max overlap integral
    def objective(coeffs):
        a = coeffs[:N_terms]
        b = coeffs[N_terms:]
        h = 0.5 * np.ones(n_points)
        for k_idx in range(N_terms):
            h += a[k_idx] * cos_vectors[k_idx] + b[k_idx] * sin_vectors[k_idx]
        # Clip to [0,1] to ensure validity during evaluation
        h_clipped = np.clip(h, 0, 1)
        # Normalize to ensure integral = 1
        sum_h = h_clipped.sum()
        if sum_h == 0:
            return float('inf')
        h_scaled = h_clipped * (n_points / 2) / sum_h
        # Compute correlation
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
        return float(np.max(corr))
    
    # Run optimization with SLSQP from Fourier initial guess
    result = minimize(
        objective,
        x0=coeffs_initial,
        constraints=linear_constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8}
    )

    a_best = best_coeffs[:N_terms]

    if sum_h != 0:
