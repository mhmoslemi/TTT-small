import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = (h / current_sum) * target_sum
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def downsample(h_full):
        h_coarse = (h_full[::2] + h_full[1::2]) / 2
        target = len(h_coarse) / 2.0
        current_sum = np.sum(h_coarse)
        if current_sum != 0:
            h_coarse = h_coarse * (target / current_sum)
        h_coarse = np.clip(h_coarse, 0, 1)
        return h_coarse

    def upsample(h_coarse):
        full = np.zeros(2 * len(h_coarse))
        for i in range(len(h_coarse)):
            full[2 * i] = h_coarse[i]
            full[2 * i + 1] = h_coarse[i]
        return full

    def asymmetric_perturb(h_half, noise_scale):
        perturbed = h_half + np.random.normal(0, noise_scale, len(h_half))
        full = np.zeros(n_points)
        full[:half_n] = perturbed
        return repair_h(full)

    # Phase 1: Initialize and perturb
    h_fixed = repair_h(h_initial)
    perturbed = []
    for i in range(10):
        if time.time() - start_time > 0.2 * budget_s:
            break
        noise_scale = 0.1 if i < 5 else 0.2
        p = asymmetric_perturb(h_fixed[:half_n], noise_scale)
        perturbed.append((p, compute_c5(p)))

    # Phase 2: Coarse optimization
    best_coarse = []
    for p, _ in sorted(perturbed, key=lambda x: x[1])[:3]:
        if time.time() - start_time > 0.5 * budget_s:
            break
        h_coarse_init = downsample(p)
        bounds = [(0, 1)] * len(h_coarse_init)
        def obj(x):
            x = np.clip(x, 0, 1)
            return compute_c5(upsample(x))
        result = differential_evolution(obj, bounds, strategy='currenttobest1bin', 
                                       tol=1e-5, recombination=0.9, 
                                       mutation=(0.5, 1.5), seed=seed)
        if result.success:
            h_coarse_opt = np.clip(result.x, 0, 1)
            h_full_opt = upsample(h_coarse_opt)
            best_coarse.append((h_full_opt, compute_c5(h_full_opt)))

    # Phase 3: Fine-scale optimization
    candidates = best_coarse + sorted(perturbed, key=lambda x: x[1])[:3]
    best = min(candidates, key=lambda x: x[1])
    while time.time() - start_time < budget_s - 10:
        w = np.random.uniform(0.25, 0.75)
        idxs = np.random.choice(len(candidates), 2, replace=False)
        h_mix = w * candidates[idxs[0]][0] + (1 - w) * candidates[idxs[1]][0]
        h_mix = repair_h(h_mix)
        bounds = Bounds(0, 1)
        result = minimize(lambda x: compute_c5(x), h_mix, method='SLSQP', 
                         bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if result.success:
            h_new = repair_h(result.x)
            c_new = compute_c5(h_new)
            if c_new < best[1]:
                best = (h_new, c_new)
                candidates.append((h_new, c_new))

    # Final selection
    h_best, c_best = best
    h_best = repair_h(h_best)
    return (h_best.tolist(), c_best, n_points)
