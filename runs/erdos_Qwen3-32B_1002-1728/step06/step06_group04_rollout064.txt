import numpy as np
from scipy.optimize import minimize
import time

def h_from_theta(theta, n_points, N_terms):
    x = np.linspace(0, 2, n_points)
    a0 = theta[0]
    an = theta[1:1 + N_terms]
    bn = theta[1 + N_terms:1 + 2 * N_terms]
    fourier_sum = a0 * np.ones_like(x)
    for n in range(1, N_terms + 1):
        fourier_sum += an[n - 1] * np.cos(n * np.pi * x)
        fourier_sum += bn[n - 1] * np.sin(n * np.pi * x)
    h_values = 1.0 / (1.0 + np.exp(-fourier_sum))
    target_sum = n_points / 2.0
    h_values *= target_sum / np.sum(h_values)
    return h_values
