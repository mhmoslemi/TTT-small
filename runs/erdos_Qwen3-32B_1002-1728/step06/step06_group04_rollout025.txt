from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)

    bounds = [(0.0, 1.0)] * n_points

    def objective(x):
        x_clipped = np.clip(x, 0.0, 1.0)

        if abs(total - target_sum) > 1e-10:
            x_clipped = x_clipped * (target_sum / total)
        return calculate_c5(x_clipped, dx)

            'polish': False,
            'disp': False,
