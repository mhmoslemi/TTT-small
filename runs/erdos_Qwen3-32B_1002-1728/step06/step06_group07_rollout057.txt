import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def clip_and_scale(h_candidate):
    h_clipped = np.clip(h_candidate, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return np.zeros_like(h_candidate)
    scale = (len(h_candidate) / 2.0) / current_sum
    scaled = np.clip(h_clipped * scale, 0, 1)
    return scaled

def objective(h_candidate):
    h = clip_and_scale(h_candidate)
    corr_vals = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    return float(np.max(corr_vals) * dx)

    n_points = len(h_initial_np)

    dx = 2.0 / n_points

    # Fourier terms
    N = 10  # Number of Fourier terms

    # Compute initial Fourier coefficients from h_initial
    a0 = 0.5  # Fixed to ensure integral is correct
    a = cp.Parameter(N)
    b = cp.Parameter(N)

    # Create variables for optimization
    a_var = cp.Variable(N)
    b_var = cp.Variable(N)

    # Objective is to minimize the sum of squares of a_var and b_var
    obj = cp.Minimize(cp.sum_squares(a_var) + cp.sum_squares(b_var))

    # Constraints for h(x) in [0,1]
    constraints = []
    for i in range(n_points):

            h_i += b_var[n] * cp.sin((n+1) * np.pi * x[i] / 2)
        constraints += [h_i >= 0, h_i <= 1]

    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.OSQP, verbose=False)

    # Get optimized a and b
    optimized_a = a_var.value
    optimized_b = b_var.value

    # Compute h from the Fourier series
    h_fourier = np.zeros(n_points)
    for i in range(n_points):
        h = a0
        for n in range(N):
            h += optimized_a[n] * np.cos((n+1) * np.pi * x[i] / 2)
            h += optimized_b[n] * np.sin((n+1) * np.pi * x[i] / 2)
        h_fourier[i] = h

    h_fourier = clip_and_scale(h_fourier)

    # Local optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    h_current = h_fourier.copy()
    c5_current = objective(h_current)
    best_c5 = c5_current
    best_h = h_current.copy()

    # Perform SLSQP optimization
    max_steps = 3
    for step in range(max_steps):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(

    problem = cp.Problem(cp.Minimize(cp.sum_squares(a_var) + cp.sum_squares(b_var)), constraints)
    problem.solve(solver=cp.OSQP, verbose=False)

    optimized_a = a_var.value
    optimized_b = b_var.value

    h_fourier = np.zeros(n_points)
    for i in range(n_points):
        h = a0
        for n in range(N):
            h += optimized_a[n] * np.cos((n + 1) * np.pi * x[i] / 2)
            h += optimized_b[n] * np.sin((n + 1) * np.pi * x[i] / 2)
        h_fourier[i] = h

    h_fourier = clip_and_scale(h_fourier)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h_fourier.copy()
    best_c5 = objective(best_h)

    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        res = minimize(
            objective, x0=best_h, method='SLSQP',
            bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            new_h = clip_and_scale(res.x)
            c5 = objective(new_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = new_h

    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - n_points / 2) < 1e-6
    return (final_h.tolist(), best_c5, n_points)
