            if time.time() - start_time > budget_s - 10:
                break
            h_current = clip_and_rescale(h_current)
            corr_vals = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(corr_vals) * dx
            if abs(c5 - prev_c5) < 1e-8:
                break
            prev_c5 = c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()

    best_h = clip_and_rescale(best_h)
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_vals) * dx

    return (best_h.tolist(), final_c5, n_points)
