import time

def objective(x):

    target_sum = n_points / 2.0
    x = np.clip(x, 0, 1)
    sum_x = np.sum(x)
    if sum_x != 0:
        x = (x / sum_x) * target_sum
    x = np.clip(x, 0, 1)
    dx = 2.0 / n_points
    corr_vals = np.correlate(x, 1 - x, mode='full')
