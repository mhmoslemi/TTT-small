            mutation=(0.5, 1.5), recombination=0.9, seed=seed, workers=1
        )
        if de_result.success and de_result.fun < best_c5:
            final_h_half = np.clip(de_result.x, 0, 1.0)
            final_h_half = (final_h_half / final_h_half.sum()) * target_sum_half
            final_h_half = np.clip(final_h_half, 0, 1.0)
            best_c5 = objective(final_h_half)
    except Exception:
        pass

    # Phase 2: Local optimization with gradient-based approach
    remaining_time = budget_s - (time.time() - start_time)
    perturb_steps = int(remaining_time / 2.5) if remaining_time > 5 else 0
    for _ in range(perturb_steps):
        perturbed = final_h_half.copy()
        perturbed += np.random.normal(0, 0.1, half_n)
        perturbed = np.clip(perturbed, 0, 1.0)
        if np.sum(perturbed) != 0:
            perturbed = (perturbed / np.sum(perturbed)) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1.0)
        
        try:
            result = minimize(
                objective, x0=perturbed, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 200, 'ftol': 1e-10}
            )
            if result.success:
                new_h_half = np.clip(result.x, 0, 1.0)
                new_h_half /= np.sum(new_h_half) * target_sum_half
                new_h_half = np.clip(new_h_half, 0, 1.0)
                new_c5 = objective(new_h_half)
                if new_c5 < best_c5:
                    final_h_half = new_h_half
                    best_c5 = new_c5
        except:
            pass

    # Final validation and repair
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1.0)
    h_full_sum = np.sum(h_full)
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1.0)
    return (h_full.tolist(), best_c5, n_points)
