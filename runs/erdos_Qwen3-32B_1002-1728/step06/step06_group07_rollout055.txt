def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0:
            return np.zeros_like(h_candidate)
        scale = T / sum_clipped
        return np.clip(h_clipped * scale, 0, 1)

    def objective_time(h_candidate):
        h = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr_vals) * dx)

    # Fourier coefficient functions
    def compute_fourier_coefficients(h, n_coefficients):
        x = np.linspace(0, 2, n_points, endpoint=False)
        coeffs = []
        for k in range(1, n_coefficients + 1):
            c = np.sum(h * np.cos(k * np.pi * x)) * dx
            d = np.sum(h * np.sin(k * np.pi * x)) * dx
            coeffs.extend([c, d])
        return np.array(coeffs)

    def h_from_fourier_params(params, n_coefficients):
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = 0.5 * np.ones(n_points)  # a0=1 corresponds to base 0.5
        for idx in range(0, len(params), 2):
            k = idx // 2 + 1
            c = params[idx]
            d = params[idx+1]
            h += c * np.cos(k * np.pi * x) + d * np.sin(k * np.pi * x)
        return h

    def objective_fourier(params, n_coefficients):
        h = h_from_fourier_params(params, n_coefficients)
        h_processed = clip_and_scale(h)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return np.max(corr_vals) * dx

    best_c5 = objective_time(h_initial)
    best_h = h_initial.copy()

    # Fourier-based optimization
    if time.time() - start_time < budget_s - 600:
        N_FOURIER = 16
        try:
            fourier_params = compute_fourier_coefficients(h_initial, N_FOURIER)
            bounds_fourier = [(-2.0, 2.0) for _ in fourier_params]
            
            result_fft = differential_evolution(

                popsize=6,
                seed=seed,
                workers=1,
                disp=False
