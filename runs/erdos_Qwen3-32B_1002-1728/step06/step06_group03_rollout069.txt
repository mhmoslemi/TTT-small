import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize and clip initial solution
    h_clipped = np.clip(h_initial, 0.0, 1.0)
    h_sum = h_clipped.sum()
    if h_sum != 0:
        h_clipped = (h_clipped / h_sum) * target_sum
    h_initial = np.clip(h_clipped, 0.0, 1.0)
    current_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    best_h = h_initial.copy()
    best_c5 = current_c5

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Optimization parameters
    bounds = [(0.0, 1.0)] * n_points
    constraint = {'type': 'eq', 'fun': lambda h: h.sum() - target_sum}
    np.random.seed(seed)

    # Perturbation and optimization loop
    n_perturb = 50
    noise_scale = 0.05
    noise_offset = 0.01
    max_iter = 200
    tol = 1e-8

    for _ in range(n_perturb):
        if time.time() - start_time > 0.7 * budget_s:
            break

        # Generate perturbed candidate
        noise = (noise_scale * h_initial + noise_offset) * np.random.randn(n_points)
        h_candidate = h_initial + noise
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_sum = h_candidate.sum()
        if h_sum != 0:
            h_candidate = (h_candidate / h_sum) * target_sum
        h_candidate = np.clip(h_candidate, 0.0, 1.0)

        # Local optimization
        result = minimize(
            compute_c5, 
            x0=h_candidate, 
            method='SLSQP', 
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': max_iter, 'ftol': tol}
        )
        if result.success:
            h_opt = np.clip(result.x, 0.0, 1.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Final refinement
    if time.time() - start_time < budget_s * 0.9:
        result = minimize(
            compute_c5, 
            x0=best_h, 
            method='SLSQP', 
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 300, 'ftol': 1e-10}
        )
        if result.success:
            final_h = np.clip(result.x, 0.0, 1.0)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5

    # Final validation

        best_h = (best_h / h_sum) * target_sum
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
