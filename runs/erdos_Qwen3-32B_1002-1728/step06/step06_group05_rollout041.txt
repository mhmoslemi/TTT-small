            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            optimized_h = np.clip(result.x, 0, 1)
            sum_opt = np.sum(optimized_h)
            if sum_opt != 0:
                optimized_h = (optimized_h / sum_opt) * target_sum
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_obj = fun(optimized_h)
            if optimized_obj < best_c5:
                best_c5 = optimized_obj
                global_best = optimized_h

    h_final = np.clip(global_best, 0, 1)
    sum_h = np.sum(h_final)
    if sum_h != 0:
        h_final = (h_final / sum_h) * target_sum
    h_final = np.clip(h_final, 0, 1)
    final_c5 = objective(h_final, target_sum, dx, n_points)
    return (h_final.tolist(), final_c5, n_points)

def objective(h, target_sum, dx, n_points):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = (h / sum_h) * target_sum
    h = np.clip(h, 0, 1)
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
