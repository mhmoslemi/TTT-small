import numpy as np
from scipy.optimize import minimize
import time

def enforce_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total != 0:
        h = (h / total) * target_sum
    return np.clip(h, 0, 1)

def objective(x, target_sum, dx):
    x = enforce_constraints(x, target_sum)
    corr_vals = np.correlate(x, 1 - x, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = enforce_constraints(h_initial, target_sum)

    pop_size = 30
    max_iter_pso = 30
    particles = np.zeros((pop_size, n_points))
    velocities = np.zeros((pop_size, n_points))
    particles[0] = h_initial.copy()

    np.random.seed(seed)
    for i in range(1, pop_size):
        if np.random.rand() < 0.5:
            perturbed = h_initial + np.random.normal(0, 0.1, n_points)
        else:
            perturbed = np.random.uniform(0, 1, n_points)
        particles[i] = enforce_constraints(perturbed, target_sum)
        velocities[i] = np.random.uniform(-0.05, 0.05, n_points)

    personal_best = particles.copy()
    personal_best_obj = np.array([objective(p, target_sum, dx) for p in particles])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = particles[global_best_idx].copy()
    best_c5 = personal_best_obj[global_best_idx]

    for iter_num in range(max_iter_pso):
        if time.time() - start_time > budget_s - 60:
            break
        w = 0.9 - 0.5 * (iter_num / max_iter_pso)
        for i in range(pop_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             0.5 * r1 * (personal_best[i] - particles[i]) +
                             0.5 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = enforce_constraints(particles[i], target_sum)
        obj_vals = np.array([objective(p, target_sum, dx) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        current_best_idx = np.argmin(personal_best_obj)
        if personal_best_obj[current_best_idx] < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = personal_best_obj[current_best_idx]

    # Local grid search on top candidates
    top_indices = np.argsort(personal_best_obj)[:5]
    for idx in top_indices:
        if time.time() - start_time > budget_s - 60:
            break
        current_h = personal_best[idx].copy()
        for _ in range(5):
            new_h = current_h.copy()
            indices = np.random.choice(n_points, size=5, replace=False)
            perturb = np.random.uniform(-0.05, 0.05, size=5)
            new_h[indices] += perturb
            new_h = enforce_constraints(new_h, target_sum)
            c5 = objective(new_h, target_sum, dx)
            if c5 < personal_best_obj[idx]:
                personal_best[idx] = new_h.copy()
                personal_best_obj[idx] = c5

    # Final refinement with SLSQP
    candidates = personal_best[np.argsort(personal_best_obj)][:3]
    for candidate in candidates:
        if time.time() - start_time > budget_s - 30:
            break
        def fun(x):
            return objective(x, target_sum, dx)
        def constraint_sum(x):
            return np.sum(x) - target_sum
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        result = minimize(
            fun=fun,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            optimized_h = enforce_constraints(result.x, target_sum)
            c5 = fun(optimized_h)
            if c5 < best_c5:
                best_c5 = c5
                global_best = optimized_h

    h_full = enforce_constraints(global_best, target_sum)
    return (h_full.tolist(), best_c5, n_points)
