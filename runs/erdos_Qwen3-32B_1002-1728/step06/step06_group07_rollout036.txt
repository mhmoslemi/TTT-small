import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2  # Constraint sum(h) = 1 corresponds to T = n_points / 2

    # Fourier parameterization parameters
    K = 20  # Number of Fourier modes to use

    # Compute initial Fourier coefficients from h_initial
    fft_h = np.fft.fft(h_initial)
    real_f = fft_h[1:K+1].real.copy()
    imag_f = fft_h[1:K+1].imag.copy()
    initial_coeffs = np.concatenate([real_f, imag_f])

    def evaluate_h_from_coeffs(coeffs):
        """Generate h(x) from Fourier coefficients."""
        h = np.zeros(n_points)
        for i in range(n_points):
            x = dx * i
            val = 0.5  # DC term (fixed at 0.5 because ∫h=1)
            for k in range(K):
                a_k = coeffs[2*k]
                b_k = coeffs[2*k + 1]
                freq = (k + 1)  # Frequency index
                val += a_k * np.cos(freq * np.pi * x) + b_k * np.sin(freq * np.pi * x)
            h[i] = val
        return h

    def optional_scale(h):
        """Clip and scale h to satisfy constraints."""
        h_clipped = np.clip(h, 0, 1)
        sum_vals = h_clipped.sum()
        if sum_vals == 0:
            return np.zeros_like(h)
        scale = T / sum_vals
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        return h_scaled

    def objective_f(coeffs):
        """Objective function for Fourier optimization."""
        h_f = evaluate_h_from_coeffs(coeffs)
        h_scaled = optional_scale(h_f)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return float(np.max(corr_vals) * dx)

    # Bounds for Fourier coefficients
    bounds_f = [(-1.0, 1.0)] * 2 * K

    best_c5 = np.inf
    best_h = None

    # First phase: Fourier domain optimization with DE
    if time.time() - start_time < budget_s - 60:
        de_result = differential_evolution(
            objective_f, bounds_f,
            strategy='best1bin',
            maxiter=8,
            popsize=6,
            tol=1e-4,
            mutation=(0.5, 0.9),
            recombination=0.85,
            seed=seed,
            workers=1,
            disp=False
        )
        if de_result.success:
            coeffs = de_result.x
            h_fft = evaluate_h_from_coeffs(coeffs)
            h_fft = optional_scale(h_fft)
            c5_fft = objective_f(coeffs)  # Already scaled
            if c5_fft < best_c5:
                best_c5 = c5_fft
                best_h = h_fft

    # Fallback to initial_coeffs
    if best_h is None:
        h_fft = evaluate_h_from_coeffs(initial_coeffs)
        h_fft = optional_scale(h_fft)
        c5_fft = objective_f(initial_coeffs)
        best_c5 = c5_fft
        best_h = h_fft

    # Second phase: Local refinement in time domain
    if time.time() - start_time < budget_s - 30:
        result_time = minimize(
            objective_f, x0=initial_coeffs,  # Note: minimize Fourier coeffs, but in time domain
            method='L-BFGS-B',
            bounds=bounds_f,
            options={'maxiter': 75, 'ftol': 1e-8, 'disp': False}
        )
        if result_time.success:
            coeffs = result_time.x
            h_time = evaluate_h_from_coeffs(coeffs)
            h_time = optional_scale(h_time)
            c5_time = np.max(np.correlate(h_time, 1 - h_time, mode='full')) * dx
            if c5_time < best_c5:
                best_c5 = c5_time
                best_h = h_time

    # Third phase: Final local polish in time domain
    if best_h is not None and time.time() - start_time < budget_s - 20:
        bounds_t = [(0.0, 1.0)] * n_points
        result_local = minimize(
            lambda x: np.max(np.correlate(optional_scale(x), 1 - optional_scale(x), mode='full')) * dx,
            x0=best_h, method='L-BFGS-B', bounds=bounds_t,
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        if result_local.success:
            final_h = optional_scale(result_local.x)
            c5_final = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = final_h

    # Ensure constraint check
    assert abs(best_h.sum() - T) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
