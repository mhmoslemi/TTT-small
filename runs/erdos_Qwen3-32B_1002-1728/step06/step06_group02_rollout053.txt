import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is feasible
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-6:
        h_initial = h_initial * (target_sum / current_sum)
        h_initial = np.clip(h_initial, 0, 1)

    # Define objective and gradient
    def objective_and_gradient(h_vec):
        h = h_vec.copy()
        n = len(h)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr_vals) * (2.0 / n)
        max_idx = np.argmax(corr_vals)
        k = max_idx - (n - 1)
        dx = 2.0 / n

        indices_plus = np.arange(n) + k
        valid_plus = (indices_plus >= 0) & (indices_plus < n)
        term1 = np.zeros(n)
        term1[valid_plus] = 1.0 - h[indices_plus[valid_plus]]
        term1[~valid_plus] = 0.0

        indices_minus = np.arange(n) - k
        valid_minus = (indices_minus >= 0) & (indices_minus < n)
        term2 = np.zeros(n)
        term2[valid_minus] = -h[indices_minus[valid_minus]]
        term2[~valid_minus] = 0.0

        grad = dx * (term1 + term2)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        'jac': lambda h: np.ones_like(h),
    })

    options = {'maxiter': 300, 'ftol': 1e-8, 'disp': False, 'iprint': 0}

    # Optimization
    result = minimize(
        objective_and_gradient,
        x0=h_initial,
        method='SLSQP',
        jac=True,
        bounds=bounds,
