import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Clip and rescale the initial solution
    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    # Generate perturbed initial points
    n_starts = 20
    np.random.seed(seed)
    starts = []
    for _ in range(n_starts):
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + perturbation
        perturbed = clip_and_rescale(perturbed)
        starts.append(perturbed)

    # Objective function
    def objective(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # Constraints and bounds for minimize
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_h = h_initial.copy()
    best_c5 = objective(h_initial)

    for idx, start in enumerate(starts):
        if time.time() - start_time > budget_s:
            break
        # Run SLSQP optimization
        result = minimize(
            fun=objective,
            x0=start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'gtol': 1e-10, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            h_candidate = clip_and_rescale(result.x)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate

    # Return best solution found
    return (best_h.tolist(), best_c5, n_points)

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):

        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

        starts.append(perturbed)

    best_h = h_initial.copy()
    best_c5 = objective(h_initial)

    for idx, start in enumerate(starts):
        if time.time() - start_time >= budget_s - 30:
            break
        result = minimize(
            fun=objective,
            x0=start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
