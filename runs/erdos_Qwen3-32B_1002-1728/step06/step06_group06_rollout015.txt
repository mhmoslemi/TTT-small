import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    np.random.seed(seed)

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    def compute_c5(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # Generate diverse perturbed candidates
    n_candidates = 20
    perturb_std = 0.1
    candidates = [clip_and_rescale(h_initial + perturb_std * np.random.randn(n_points)) for _ in range(n_candidates)]

    best_h = clip_and_rescale(h_initial)
    best_c5 = compute_c5(best_h)

    # Refine each candidate with alternating projections
    for candidate in candidates:
        current_h = candidate.copy()
        current_c5 = compute_c5(current_h)
        for _ in range(100):
            prev_c5 = current_c5
            current_h = clip_and_rescale(current_h)
            current_c5 = compute_c5(current_h)
            if abs(current_c5 - prev_c5) < 1e-8:
                break
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed

    # Final refinement with SLSQP if time permits
    if remaining_time > 30:
        def objective(x):
            return compute_c5(clip_and_rescale(x))

        cons = ({'type': 'eq', 'fun': lambda x: clip_and_rescale(x).sum() - target_sum},)
        bounds = [(0, 1)] * n_points
        result = minimize(
            fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'gtol': 1e-08}
        )
        if result.success:
            opt_h = clip_and_rescale(result.x)
            opt_c5 = compute_c5(opt_h)
            if opt_c5 < best_c5:
                best_h = opt_h
                best_c5 = opt_c5

    return best_h.tolist(), best_c5, n_points
