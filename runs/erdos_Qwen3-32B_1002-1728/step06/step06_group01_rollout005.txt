import numpy as np
import time
import scipy.fft

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_dct = scipy.fft.dct(h_initial, norm='ortho')
    params = initial_dct.copy()
    n_params = len(params)

    # Adam optimizer parameters
    alpha = 0.05  # Learning rate
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_params)
    v = np.zeros(n_params)
    t = 0

    def compute_h(p):
        h = scipy.fft.idct(p, norm='ortho')
        h = 1 / (1 + np.exp(-h))  # Sigmoid to [0,1]
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s != 0:
            h = h * (target_sum / s)
        h = np.clip(h, 0, 1)
        return h

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = np.inf
    best_h = None

    while time.time() - start_time < budget_s - 30:
        h = compute_h(params)
        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Finite difference gradient
        grad = np.zeros_like(params)
        delta = 1e-4
        for i in range(n_params):
            p_up = params.copy()
            p_up[i] += delta
            h_up = compute_h(p_up)
            val_up = obj(h_up)

            p_down = params.copy()
            p_down[i] -= delta
            h_down = compute_h(p_down)
            val_down = obj(h_down)

            grad[i] = (val_up - val_down) / (2 * delta)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        params -= alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        # Learning rate decay
        if t % 50 == 0:

    # Final computation and validation
    h_final = compute_h(params)
    final_c5 = obj(h_final)
