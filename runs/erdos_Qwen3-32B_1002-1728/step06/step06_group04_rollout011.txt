import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Fourier parameterization setup
    x = np.linspace(0, 2.0, n_points, endpoint=False)
    N_harmonics = 20
    features = []
    for n in range(1, N_harmonics + 1):
        features.append(np.cos(n * np.pi * x))
        features.append(np.sin(n * np.pi * x))
    features = np.column_stack(features)

    # Initialize coefficients from initial_h_values via least squares
    h_initial_clipped = np.clip(h_initial, 1e-10, 1 - 1e-10)
    arg_initial = np.log(h_initial_clipped / (1.0 - h_initial_clipped))
    coeffs_initial = np.linalg.lstsq(features, arg_initial, rcond=None)[0].flatten()

    # Fourier space optimization
    def objective_fourier(coeffs):
        arg = features @ coeffs
        h = 1.0 / (1.0 + np.exp(-arg))
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    result_fourier = minimize(
        objective_fourier,
        coeffs_initial,
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_coeff = result_fourier.x if result_fourier.success else coeffs_initial

    # Convert to sample space and enforce sum constraint
    arg_best = features @ best_coeff
    h_best = 1.0 / (1.0 + np.exp(-arg_best))
    h_best = np.clip(h_best, 0.0, 1.0)
    scaling = (n_points / 2.0) / h_best.sum()
    h_best_scaled = h_best * scaling

    # Real space refinement
    def objective_real(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    bounds_real = [(0.0, 1.0)] * n_points
    cons_real = [{'type': 'eq', 'fun': lambda h: h.sum() - n_points / 2.0}]

    result_real = minimize(
        objective_real,
        h_best_scaled,
        method='SLSQP',
        bounds=bounds_real,
        constraints=cons_real,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    if result_real.success:
        h_final = result_real.x
        c5_bound = objective_real(h_final)
    else:
        h_final = h_best_scaled
        c5_bound = objective_real(h_final)

    # Final checks and return
    h_final = np.clip(h_final, 0.0, 1.0)
    assert abs(h_final.sum() - n_points / 2.0) < 1e-6
    return (h_final.tolist(), c5_bound, n_points)
