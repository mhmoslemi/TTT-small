import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    # Ensure initial_h is feasible
    h_feasible = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_feasible)
    if current_sum != 0:
        h_feasible = (h_feasible / current_sum) * target_sum
    h_feasible = np.clip(h_feasible, 0, 1)
    best_h = h_feasible.copy()
    best_c5 = compute_c5(best_h)

    noise_scale = 0.05 * h_initial + 0.01
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)

    np.random.seed(seed)
    for _ in range(50):
        if time.time() - start_time > budget_s - 30:
            break

        perturbation = np.random.randn(n_points) * noise_scale
        new_h = np.clip(best_h + perturbation, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = (new_h / sum_new) * target_sum
        new_h = np.clip(new_h, 0, 1)

        def obj(x):
            return compute_c5(x)

        result = minimize(
            obj, x0=new_h, method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h.copy()

    total_sum = np.sum(best_h)
    if total_sum != 0:

    best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
