                    new_h_full[:half_n] = new_h
                    new_h_full[-half_n:] = new_h[::-1]
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h_full.copy()
            except Exception:
                continue

    # Final assert
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
