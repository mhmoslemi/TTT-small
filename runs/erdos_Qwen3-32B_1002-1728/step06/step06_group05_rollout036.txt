            x = (x / s) * target_sum_coarse
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * (2.0 / n_coarse)

    personal_best = particles.copy()
    personal_best_obj = np.array([objective(p) for p in particles])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = particles[global_best_idx].copy()
    best_c5 = personal_best_obj[global_best_idx]

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
        w = w_start - (w_start - w_end) * (iter / max_iter)
        for i in range(pop_size):
            r1 = np.random.rand(n_coarse)
            r2 = np.random.rand(n_coarse)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = (particles[i] / sum_p) * target_sum_coarse
            particles[i] = np.clip(particles[i], 0, 1)
        obj_vals = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        current_best_idx = np.argmin(personal_best_obj)
        current_c5 = personal_best_obj[current_best_idx]
        if current_c5 < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = current_c5

    # Upscale to fine grid
    h_fine = np.repeat(global_best, 2)
    sum_fine = np.sum(h_fine)
    if sum_fine != 0:
        h_fine = (h_fine / sum_fine) * target_sum
    h_fine = np.clip(h_fine, 0, 1)

    def fun(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s != 0:
            x = (x / s) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    result = minimize(
        fun=fun,
        x0=h_fine,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        optimized_h = np.clip(result.x, 0, 1)
        s = np.sum(optimized_h)
        if s != 0:
            optimized_h = (optimized_h / s) * target_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        c5_fine = fun(optimized_h)
        if c5_fine < fun(h_fine):
            h_fine = optimized_h

    h_final = np.clip(h_fine, 0, 1)
    s = np.sum(h_final)
    if s != 0:
        h_final = (h_final / s) * target_sum
    h_final = np.clip(h_final, 0, 1)

    return (h_final.tolist(), fun(h_final), n_points)
