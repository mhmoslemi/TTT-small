            total = np.sum(de_h_half)
            if total != 0:
                de_h_half = (de_h_half / total) * target_sum_half
            de_h_half = np.clip(de_h_half, 0, 1)
            de_c5 = objective(de_h_half)
            if de_c5 < best_c5:
                best_h_half = de_h_half
                best_c5 = de_c5
    except Exception:
        pass

    # Aggressive local optimization loop with adaptive strategies
    np.random.seed(seed)
    while time.time() - start_time < budget_s - 60:
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.5 * (1 - frac**2)  # More aggressive noise initially

        for _ in range(4):  # Increase number of perturbations
            seed = seed + 1
            rnd = np.random.RandomState(seed)
            idxs = np.random.choice(half_n, size=min(half_n, 30), replace=False)
            perturbed = best_h_half.copy()
            perturbed[idxs] += rnd.normal(0, noise_scale, len(idxs))
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total != 0:
                perturbed = (perturbed / total) * target_sum_half
            perturbed = np.clip(perturbed, 0, 1)

            # Use multiple local optimization methods with higher tolerance
            for method in ['L-BFGS-B', 'SLSQP', 'COBYLA']:
                result = minimize(
                    fun=objective, x0=perturbed, method=method, bounds=bounds_per_var, constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
                )
                if result.success:
                    new_h_half = np.clip(result.x, 0, 1)
                    if abs(np.sum(new_h_half) - target_sum_half) < 1e-6:
                        new_c5 = objective(new_h_half)
                        if new_c5 < best_c5:
                            best_h_half = new_h_half
                            best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
