            callback=lambda x: time.time() - start_time < budget_s
        )
        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_h_half = new_h
                best_c5 = new_c5
    
    def objective_perturbed(x):
        h_pert = x.copy()
        h_pert = np.clip(h_pert / np.sum(h_pert) * target_sum_half, 0, 1)
        return objective(h_pert)
    
    bounds_de = [(0.0, 1.0)] * half_n
    de_result = differential_evolution(
        objective_perturbed,
        bounds_de,
        strategy='best1bin',
        popsize=10,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed+1,
        disp=False,
        tol=1e-4,
        maxiter=10
    )
    if de_result.success and time.time() - start_time < budget_s - 30:
        de_h_half = np.clip(de_result.x, 0, 1)
        de_c5 = objective(de_h_half)
        if de_c5 < best_c5:
            best_h_half = de_h_half
            best_c5 = de_c5
    
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Sum constraint violated"
    return (h_full.tolist(), best_c5, n_points)
