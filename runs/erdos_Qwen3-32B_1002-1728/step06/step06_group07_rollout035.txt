        recombination=0.9, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        de_c5 = objective(de_h)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h

    # Local refinement with L-BFGS-B
    for _ in range(6):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective, x0=best_h, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': 300, 'ftol': 1e-8}
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Final validation
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), best_c5, n_points)
