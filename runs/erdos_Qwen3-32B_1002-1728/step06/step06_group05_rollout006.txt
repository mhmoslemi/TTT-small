import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = (h / h_sum) * target_sum
        return np.clip(h, 0, 1)

    h_initial = clip_and_scale(h_initial)

    # PSO parameters
    pop_size = 30
    max_pso_iter = 30
    w_max, w_min = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    np.random.seed(seed)

    # Initialization: 50% from initial with perturb, 50% random
    particles = np.zeros((pop_size, n_points))
    particles[0] = h_initial.copy()
    for i in range(1, pop_size):
        if i < pop_size // 2:
            perturbation = np.random.normal(0, 0.1, n_points)
            perturbed = np.clip(h_initial + perturbation, 0, 1)
        else:
            perturbed = np.random.rand(n_points)
        particles[i] = clip_and_scale(perturbed)
    velocities = np.random.uniform(-0.05, 0.05, (pop_size, n_points))

    def objective(x):
        x = clip_and_scale(x)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    personal_best = particles.copy()
    personal_best_obj = np.array([objective(p) for p in particles])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = particles[global_best_idx].copy().copy()
    best_c5 = personal_best_obj[global_best_idx]

    # PSO main loop
    for iter in range(max_pso_iter):
        w = w_max - (w_max - w_min) * (iter / max_pso_iter)
        if time.time() - start_time > budget_s - 50:
            break
        for i in range(pop_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = clip_and_scale(particles[i])
        obj_vals = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        current_best_idx = np.argmin(personal_best_obj)
        if personal_best_obj[current_best_idx] < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = personal_best_obj[current_best_idx]

    # Local grid search on top 5
    top_indices = np.argsort(personal_best_obj)[:5]
    for idx in top_indices:
        base = personal_best[idx].copy()
        for _ in range(5):  # Generate 5 random neighbors per top entry
            perturbed = base.copy()
            delta_indices = np.random.choice(n_points, size=np.random.randint(1, 5))
            for di in delta_indices:
                perturbed[di] += np.random.uniform(-0.03, 0.03)
            perturbed = clip_and_scale(perturbed)
            c5 = objective(perturbed)
            if c5 < personal_best_obj[idx]:
                personal_best[idx] = perturbed.copy()
                personal_best_obj[idx] = c5
                if c5 < best_c5:
                    global_best[:] = perturbed
                    best_c5 = c5

    # Final SLSQP on top 3 after grid search
    top3_indices = np.argsort(personal_best_obj)[:3]
    for idx in top3_indices:
        if time.time() - start_time > budget_s - 30:
            break
        h0 = personal_best[idx].copy()
        def constraint(x):
            return np.sum(x) - target_sum
        def wrapped_objective(x):
            return objective(x)
        res = minimize(
            fun=wrapped_objective,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if res.success:
            sol = clip_and_scale(res.x)
            c5 = objective(sol)
            if c5 < best_c5:
                global_best[:] = sol
                best_c5 = c5

    final_h = clip_and_scale(global_best)
    return (final_h.tolist(), best_c5, n_points)
