    candidates = [clip_and_rescale(h_initial + np.random.normal(0, 0.1, n_points)) for _ in range(num_starts)]

    def objective(h_vec):
        h = h_vec.copy()
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    def grad_objective(h_vec):
        h = h_vec
        n = len(h)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_vals)
        k_val = max_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + k_val < n:
                grad[i] += (1 - h[i + k_val])
            if 0 <= i - k_val < n:
                grad[i] -= h[i - k_val]
        return grad * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for candidate in candidates:
        if time.time() - start_time > budget_s - 60:
            break
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            jac=grad_objective,
            options={'maxiter': 150, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            opt_h = np.clip(clip_and_rescale(result.x), 0, 1)
            c5_candidate = objective(opt_h)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = opt_h

    top_indices = np.argsort([objective(c) for c in candidates])[:5]
    for idx in top_indices:
        if time.time() - start_time > budget_s - 15:
            break
        c = candidates[idx]
        result = minimize(
            fun=objective,
            x0=c,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            jac=grad_objective,
            options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
        )
        if result.success:
            opt_h = np.clip(clip_and_rescale(result.x), 0, 1)
            c5_candidate = objective(opt_h)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = opt_h

    best_h = clip_and_rescale(best_h)
    return (best_h.tolist(), best_c5, n_points)
