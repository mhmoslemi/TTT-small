                       options={'maxiter': int(remaining_time // 1.5)})
        if res.success and res.fun < best_c5:
            refined_h = np.clip(res.x, 0, 1)
            refined_sum = refined_h.sum()
            refined_h = refined_h * (target_sum / refined_sum) if refined_sum != 0 else (target_sum / n_points)
            refined_h = np.clip(refined_h, 0, 1)
            best_c5 = evaluate_c5(refined_h, dx)
            best_h = refined_h

    return (best_h.tolist(), best_c5, n_points)
