        h = h_from_θ(θ)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    cons = ({'type': 'eq', 'fun': lambda θ: h_from_θ(θ, return_sum=True)[1] - target_sum})

    # Optimize with Fourier parameterization
    iter_count, max_iter = 0, 50
    while time.time() - start_time < budget_s - 60 and iter_count < max_iter:
        result = minimize(
            objective,
            θ_init,
            method='SLSQP',
            constraints=cons,
            bounds=[(None, None)] * len(θ_init),
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        if not result.success:
            break
        new_h = h_from_θ(result.x)
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            θ_init = result.x
        iter_count += 1

    # Final refinement in real space
    def real_objective(h_flat):
        h = h_flat.copy().clip(0.0, 1.0)
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-6:
            h *= target_sum / current_sum
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    bounds_real = [(0, 1)] * n_points
    result_refine = minimize(
        real_objective,
        best_h,
        method='SLSQP',
        bounds=bounds_real,
        constraints=({'type': 'eq', 'fun': lambda h: h.sum() - target_sum}),
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result_refine.success:
        h_refined = result_refine.x.clip(0.0, 1.0)
        h_refined *= target_sum / h_refined.sum()
        c5_refine = np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx
        if c5_refine < best_c5:
            best_h = h_refined
            best_c5 = c5_refine

    best_h = best_h.clip(0.0, 1.0)
    assert abs(best_h.sum() - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
