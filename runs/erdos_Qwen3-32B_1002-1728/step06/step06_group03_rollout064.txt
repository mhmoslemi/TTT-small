```python
import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def create_full_h(h_vars):
        return np.concatenate((h_vars, h_vars[::-1]))

    def objective(h_vars):
        h_full = create_full_h(np.clip(h_vars, 0, 1))
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    def normalize(h_half):
        total = np.sum(h_half)
        if total == 0:
            return np.zeros_like(h_half)
        h_half = (h_half / total) * (target_sum / 2)
        return np.clip(h_half, 0, 1)

    np.random.seed(seed)
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = normalize(h_half_initial)

    n_particles = 30
    positions = []
    velocities = []
    for _ in range(n_particles):
        perturbation = np.random.randn(half_n) * 0.05
        pos = np.clip(h_half_initial + perturbation, 0, 1)
        pos = normalize(pos)
        positions.append(pos)
        velocities.append(np.random.rand(half_n) * 0.02)
    positions = np.array(positions)
    velocities = np.array(velocities)

    personal_best_positions = positions.copy()
    personal_best_scores = np.array([objective(p) for p in positions])
    global_idx = np.argmin(personal_best_scores)
    global_best = personal_best_positions[global_idx]
    best_c5 = personal_best_scores[global_idx]

    w = 0.7
    c1 = 1.2
    c2 = 1.2

    while time.time() - start_time < 0.6 * budget_s:
        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        velocities = w * velocities + c1 * r1 * (personal_best_positions - positions) + c2 * r2 * (global_best - positions)
        velocities = np.clip(velocities, -0.1, 0.1)
        positions += velocities
        positions = np.clip(positions, 0, 1)
        for i in range(n_particles):
            positions[i] = normalize(positions[i])
        current_scores = np.array([objective(p) for p in positions])
        improved = current_scores < personal_best_scores
        personal_best_positions[improved] = positions[improved]
        personal_best_scores[improved] = current_scores[improved]
        current_global_idx = np.argmin(personal_best_scores)
        if personal_best_scores[current_global_idx] < best_c5:
            global_best = personal_best_positions[current_global_idx]
            best_c5 = personal_best_scores[current_global_idx]

    candidates = personal_best_positions[np.argsort(personal_best_scores)][:3]
    for candidate in candidates:
        bounds = [(0.0, 1.0)] * half_n
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - (target_sum / 2.)})
        result = minimize(objective, x0=candidate, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
        if result.success:
            refined = normalize(np.clip(result.x, 0, 1))
            c5 = objective(refined)
            if c5 < best_c5:
                global_best = refined.copy()
                best_c5 = c5

    h_full = create_full_h(np.clip(global_best, 0, 1))
    h_total = np.sum(h_full)
    if abs(h_total - target_sum) > 1e-6:
        h_full = (h_full / h_total) * target_sum
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
```
