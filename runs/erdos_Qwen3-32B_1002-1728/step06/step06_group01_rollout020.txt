            gbest = pbest_positions[current_index].copy()
            gbest_value = current_min

        # Periodic local optimization
        if iteration % 10 == 0:
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
            bounds = [(0.0, 1.0) for _ in range(half_n)]
            result = minimize(objective, x0=gbest.copy(), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                refined = np.clip(result.x, 0, 1)
                refined = project_to_constraints(refined)
                refined_value = objective(refined)
                if refined_value < gbest_value:
                    gbest = refined.copy()
                    gbest_value = refined_value

    best_h_half = gbest.copy()
    best_c5 = gbest_value

    # Final local optimization
    if time.time() - start_time < budget_s - 10:
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
        bounds = [(0.0, 1.0) for _ in range(half_n)]
        result = minimize(objective, x0=best_h_half.copy(), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            refined = np.clip(result.x, 0, 1)
            refined = project_to_constraints(refined)
            refined_value = objective(refined)
            if refined_value < best_c5:
                best_h_half = refined.copy()
                best_c5 = refined_value

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - target_sum_full) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
