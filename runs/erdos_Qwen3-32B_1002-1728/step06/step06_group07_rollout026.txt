import numpy as np
import time
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    T = n_points / 2.0

    np.random.seed(seed)

    def make_binary(h):
        binary = (h > 0.5).astype(int)
        current_sum = binary.sum()
        delta = current_sum - T
        if delta != 0:
            idx = np.argsort(h)
            if delta > 0:
                binary[idx[-delta:]] = 0
            else:
                binary[idx[:int(-delta)]] = 1
        return binary

    binary_initial = make_binary(h_initial)
    population = [binary_initial]
    while len(population) < 50:
        perm = np.random.permutation(n_points)
        h = np.zeros(n_points, dtype=int)
        h[perm[:int(T)]] = 1
        population.append(h)

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h = binary_initial

    while time.time() - start_time < budget_s - 300:
        scores = []
        for h in population:
            score = compute_c5(h)
            if score < best_c5:
                best_c5 = score
                best_h = h.copy()
            scores.append(score)

        new_pop = []
        for _ in range(25):
            a, b = np.random.choice(len(population), 2, replace=False)
            p1 = population[a] if scores[a] < scores[b] else population[b]
            a, b = np.random.choice(len(population), 2, replace=False)
            p2 = population[a] if scores[a] < scores[b] else population[b]

            cross = np.random.randint(n_points)
            child = np.concatenate([p1[:cross], p2[cross:]]).copy()

            if np.random.rand() < 0.1:
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i, j = np.random.choice(ones), np.random.choice(zeros)

            new_pop.append(child)

    dx = 2.0 / n_points
    refined_h = best_h.astype(float) * 0.7 + 0.15

    refined_h = np.clip(refined_h, 0, 1)

        corr = np.correlate(x, 1 - x, mode='full')
