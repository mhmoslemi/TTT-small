import numpy as np
import time
from scipy.optimize import minimize
import functools

def create_particle(h_base, noise_level=0.05, target_sum=None):
    n_points = len(h_base)
    h = h_base + noise_level * (np.random.rand(n_points) - 0.5)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != 0 and target_sum is not None:
        h = (h / h_sum) * target_sum
    h = np.clip(h, 0, 1)
    return h

def objective(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):

        particles.append(h_p)
        velocities.append(np.zeros(n_points))

    gbest_idx = np.argmin(pbest_scores)
