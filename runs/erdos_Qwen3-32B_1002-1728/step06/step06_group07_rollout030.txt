    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    # Fourier-filtered initialization
    h_fft = np.fft.fft(h_full_initial)
    N_terms = 20
    lowpass_mask = np.zeros_like(h_fft, dtype=bool)
    lowpass_mask[:N_terms] = True
    if n_points % 2 == 0:
        lowpass_mask[-(N_terms-1):] = True
    else:
        lowpass_mask[-N_terms:] = True
    h_fft[~lowpass_mask] = 0
    h_smooth = np.fft.ifft(h_fft).real
    h_initial = clip_and_scale(h_smooth)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    # Differential Evolution with enhanced sampling
    de_result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=15, popsize=15, tol=1e-4, mutation=(0.5, 1),
        recombination=0.85, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        de_c5 = objective(de_h)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h

    # Iterative SLSQP refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 60:
            break
        result = minimize(
            objective, x0=best_h, method='SLSQP',
            bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6
    return (final_h.tolist(), best_c5, n_points)
