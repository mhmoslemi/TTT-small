        grad = compute_gradient(h, dx, tau, target_sum)
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum != 0:
            h_new = h_new * (target_sum / current_sum)
        else:
            h_new[:] = target_sum / n_points
        h_new = np.clip(h_new, 0, 1)
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5 and np.all((0 <= h_new) & (h_new <= 1)):
            best_h = h_new.copy()
            best_c5 = new_c5
        h = h_new.copy()
        tau *= tau_decay

        if iteration % 10 == 0 and n_refine_cobyla > 0:
            def objective_cobyla(x):
                return evaluate_c5(x, dx)
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            res = minimize(objective_cobyla, x0=h, method='COBYLA', bounds=bounds, options={'maxiter': 200})
            if res.success:
                refined_h = res.x
                refined_h = np.clip(refined_h, 0, 1)
                sum_r = np.sum(refined_h)
                if sum_r != 0:
                    refined_h *= target_sum / sum_r
                else:
                    refined_h[:] = target_sum / n_points
                refined_h = np.clip(refined_h, 0, 1)
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h.copy()
                    best_c5 = refined_c5
                n_refine_cobyla -= 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
