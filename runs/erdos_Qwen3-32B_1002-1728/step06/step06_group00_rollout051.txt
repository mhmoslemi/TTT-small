                options={'maxiter': int(150 * (1 - frac_remaining ** 2)), 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h_sum = new_h.sum()
                if abs(new_h_sum - target_sum_full) < 1e-6:
                    new_c5 = objective_full(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
        except:
            pass

    # Final cleanup and constraint enforcement
    h_full = best_h.copy()
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
