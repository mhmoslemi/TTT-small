import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    x_vals = np.linspace(0, 2.0, n_points, endpoint=False)

    # Frequency domain parameterization setup
    N_terms = 10  # Number of Fourier terms to maintain tractability
    a0 = 0.5  # Fixed by the integral constraint

    # Compute initial Fourier coefficients from h_full_initial
    a_coeffs_initial = np.zeros(N_terms)
    for k in range(N_terms):
        idx = k + 1
        coeff = 0.0
        for i in range(n_points):
            coeff += h_full_initial[i] * np.cos(idx * np.pi * x_vals[i] / 2) * dx
        a_coeffs_initial[k] = coeff

    # Objective function for frequency domain optimization
    def objective_freq(a_coeffs_flat):
        a_coeffs = a_coeffs_flat
        h_freq = np.full(n_points, a0)
        for k in range(N_terms):
            term = a_coeffs[k] * np.cos((k + 1) * np.pi * x_vals / 2.0)
            h_freq += term
        h_clipped = np.clip(h_freq, 0, 1)
        corrs = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return float(np.max(corrs) * dx)

    # Bounds for frequency domain optimization
    bounds_freq = [(-0.3, 0.3)] * N_terms

    # Frequency domain differential evolution
    freq_budget = 300  # 5 minutes allocated for this phase
    freq_end = start_time + freq_budget
    freq_result = differential_evolution(
        objective_freq, bounds_freq, strategy='best1bin',
        maxiter=30, popsize=10, tol=1e-3, mutation=(0.5, 1.4),  # Various mutation to avoid local optima
        recombination=0.8, seed=seed, workers=1,
        polish=False, updating='deferred'
    )

    best_a_coeffs = freq_result.x if freq_result.success else a_coeffs_initial

    # Generate time-domain approximation from frequency solution
    h_freq = np.full(n_points, a0)
    for k in range(N_terms):
        h_freq += best_a_coeffs[k] * np.cos((k + 1) * np.pi * x_vals / 2.0)
    h_freq_clipped = np.clip(h_freq, 0, 1)

    # Time domain optimization (only runs if time is available)
    def objective(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        corrs = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return float(np.max(corrs) * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = objective(h_freq_clipped)
    best_h = h_freq_clipped.copy()

    # Hybrid time domain refinement: DE followed by SLSQP
    remaining_time = budget_s - max(freq_budget, 0)
    if remaining_time > 20:
        # Differential evolution with fewer iterations and popsize to save time
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=8, popsize=12, tol=1e-4, mutation=(0.5, 1.4),
            recombination=0.85, seed=seed, workers=1
        )
        if de_result.success:
            new_h = np.clip(de_result.x, 0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        # Post DE refinement with SLSQP
        slsqp_time = remaining_time - (time.time() - start_time)
        if slsqp_time > 40:
            slsqp_result = minimize(
                objective, x0=best_h, method='SLSQP',
                bounds=bounds, options={'maxiter': 150, 'ftol': 1e-8}
            )
            if slsqp_result.success:
                post_h = np.clip(slsqp_result.x, 0, 1)
                post_c5 = objective(post_h)
                if post_c5 < best_c5:
                    best_h = post_h
                    best_c5 = post_c5

    # Final validation of constraints
    assert abs(best_h.sum() - n_points / 2) < 1e-6, "Sum constraint violated"
    return (best_h.tolist(), best_c5, n_points)
