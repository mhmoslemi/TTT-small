    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        scale_factor = target_sum_full / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    def de_objective(x):

        if total != 0:
            x_scaled = (x / total) * target_sum_half
        else:
            x_scaled = x.copy()
        x_clipped = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_clipped)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    de_budget_seconds = budget_s * 0.4
    popsize = 30  # increased population size
    maxiter = 20  # increased generations
    try:
        de_result = differential_evolution(
            de_objective, bounds, strategy='best1bin',

        )
        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
