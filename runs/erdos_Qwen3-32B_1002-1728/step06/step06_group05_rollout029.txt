    h_initial = np.clip(h_scaled, 0, 1)
    
    # Post-processing to ensure sum constraint
    sum_h = np.sum(h_initial)
    if sum_h != 0:
        h_initial = (h_initial / sum_h) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    if result.success:
        optimized_h = np.clip(result.x, 0, 1)
        sum_opt = np.sum(optimized_h)
        if sum_opt != 0:
            optimized_h = (optimized_h / sum_opt) * target_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        current_c5 = objective(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

    # Final validation
    final_sum = np.sum(best_h)
    if not np.isclose(final_sum, target_sum):
        best_h = np.clip(best_h, 0, 1)
        best_h = (best_h / np.sum(best_h)) * target_sum

    return (best_h.tolist(), best_c5, n_points)
