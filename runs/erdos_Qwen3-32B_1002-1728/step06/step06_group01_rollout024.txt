    c_initial = dct(h_initial, type=2, norm='ortho')

    # Objective function using correct IDCT type 3
    def objective_func(c_vec):
        try:
            h_dct = idct(c_vec, type=3, norm='ortho')
        except:
            return np.inf
        # Apply sigmoid and normalize
        h_scaled = 1 / (1 + np.exp(-h_dct))
        current_sum = np.sum(h_scaled)
        if current_sum == 0:
            return np.inf
        scale_factor = (n_points / 2.0) / current_sum
        h_scaled *= scale_factor
        h_clipped = np.clip(h_scaled, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        h_clipped *= (n_points / 2.0) / total
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(corr_vals)

    bounds = [(-20, 20) for _ in range(n_points)]
    best_c = c_initial.copy()
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(1 - best_h, best_h, mode='full')) * dx

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 20:
        # Perturbed initial guess for new attempt
        perturbation = np.random.normal(0, 0.5, size=n_points)
        current_c = best_c + perturbation * np.sqrt(1 - (time.time() - start_time)/budget_s)

        # Adaptive maxiter based on remaining time
        remaining = budget_s - (time.time() - start_time)
        max_iter = max(10, int(min(100, remaining / 5)))
        result = minimize(objective_func, current_c, bounds=bounds, method='L-BFGS-B', 
                         options={'maxiter': max_iter, 'ftol': 1e-6})

        # Update best solution if valid
        if result.status == 0:
            optimized_c = result.x
            h_dct = idct(optimized_c, type=3, norm='ortho')
            h_scaled = 1/(1 + np.exp(-h_dct))
            current_sum = np.sum(h_scaled)
            if current_sum != 0:
                scale_factor = (n_points / 2.0) / current_sum
                h_scaled *= scale_factor
            h_clipped = np.clip(h_scaled, 0, 1)
            total = np.sum(h_clipped)
            if total != 0:
                h_clipped *= (n_points / 2.0) / total
            h_final = np.clip(h_clipped, 0, 1)
            corr_vals = np.correlate(h_final, 1 - h_final, mode='full') * dx
            current_c5 = np.max(corr_vals)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_c = optimized_c
                best_h = h_final.copy()

        # Skip low-quality results
        if time.time() - start_time > budget_s - 20:
            break

    # Final constraint check
    total = np.sum(best_h)
    if abs(total - n_points / 2.0) > 1e-6:
        best_h *= (n_points / 2.0) / total
        best_h = np.clip(best_h, 0, 1)

    corr_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_vals)
    return (best_h.tolist(), final_c5, n_points)
