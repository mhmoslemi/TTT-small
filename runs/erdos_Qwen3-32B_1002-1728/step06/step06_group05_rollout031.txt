import numpy as np
from scipy.interpolate import interp1d
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = 35.0

    # Generate Frank-Zadoff like initial guess
    n_frank = 64
    x_frank = np.linspace(0, 2, n_frank)
    b_frank = np.zeros(n_frank, dtype=float)
    b_frank[:16] = 1.0
    b_frank[48:] = 1.0  # 48 to 63 is 16 elements
    
    x_new = np.linspace(0, 2, n_points)
    f_interpolate = interp1d(x_frank, b_frank, kind='linear')
    h_initial = f_interpolate(x_new)

    assert np.isclose(np.sum(h_initial), target_sum), "Initial sum mismatch"
    
    def objective(x):
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full') * dx
        return np.max(corr_vals)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
    
    # Final validation and aggregation
    h_final = h_initial.copy()
    c5_final = objective(h_final)

        h_candidate = np.clip(result.x, 0, 1)

        h_candidate = np.clip(h_candidate, 0, 1)
        c5_candidate = objective(h_candidate)
        if c5_candidate < c5_final:
            h_final = h_candidate
            c5_final = c5_candidate
    
    return (h_final.tolist(), c5_final, n_points)
