import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = np.array(initial_h_values).copy().astype(float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is feasible
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, target_sum, rtol=1e-6):
        h *= target_sum / h_sum
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Hyperparameters
    max_iter = 200
    lr = 0.05
    tau = 0.005  # Smoothing temperature for soft maximum
    n_refine = 50  # Refine with COBYLA every n_refine steps
    decay_factor = 0.995  # Learning rate decay

    for it in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break

        # Compute correlation values
        corr_vals = np.correlate(h, 1 - h, mode='full')
        c = corr_vals * dx
        valid = c != 0  # Avoid exp(0) issues
        expc = np.zeros_like(c)
        expc[valid] = np.exp(c[valid] / tau)
        expc[~valid] = 0.0
        sum_exp = np.sum(expc)
        weights = expc / sum_exp

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            g_i = 0.0
            for m in range(len(corr_vals)):
                k_shift = m - (n_points - 1)
                # Term 1: contribution from h[i] * (1 - h[j])
                j = i + k_shift
                if 0 <= j < n_points:
                    term1 = 1.0 - h[j]
                else:
                    term1 = 1.0
                # Term 2: contribution from h[x] * (1 - h[i])
                x = i - k_shift
                if 0 <= x < n_points:
                    term2 = -h[x]
                else:
                    term2 = 0.0
                contrib = (term1 + term2) * dx
                g_i += weights[m] * contrib
            grad[i] = g_i

        # Gradient descent step
        h -= lr * grad

        # Clip and project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h *= target_sum / h_sum
        else:
            h[:] = target_sum / n_points
        h = np.clip(h, 0, 1)

        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            # Adaptive learning rate: reduce if no progress
            lr *= decay_factor

        # Periodic refinement with COBYLA
        if it % n_refine == 0 and it > 0:
            x0 = h.copy()
            def objective(x):
                return evaluate_c5(x, dx)
            bounds = [(0, 1)] * n_points
            res = minimize(objective, x0=x0, method='COBYLA', bounds=bounds,
                           options={'maxiter': 100, 'tol': 1e-8})
            if res.success:
                refined_h = res.x.copy()
                refined_h = np.clip(refined_h, 0, 1)
                h_sum_r = np.sum(refined_h)
                if h_sum_r != 0:
                    refined_h *= target_sum / h_sum_r
                else:
                    refined_h[:] = target_sum / n_points
                refined_h = np.clip(refined_h, 0, 1)

        best_h *= target_sum / h_sum
        best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
