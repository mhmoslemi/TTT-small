import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    np.random.seed(seed)
    candidates = []
    for _ in range(30):
        pos = h_initial + np.random.normal(0, 0.1, n_points)
        pos = clip_and_rescale(pos)
        candidates.append(pos)

    def evaluate_c5(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    best_c5 = evaluate_c5(h_initial)
    best_h = h_initial.copy()

    for idx, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 60:
            break
        def obj(x):
            x = x.reshape(-1)
            x = clip_and_rescale(x)
            return evaluate_c5(x)
        def constraint(x):
            return np.sum(x) - target_sum
        bounds = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': constraint})
        result = minimize(
            obj, x0=candidate, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 200, 'gtol': 1e-8, 'disp': False}
        )
        if result.success:
            h_opt = clip_and_rescale(result.x)
            c5_val = evaluate_c5(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

    top_indices = np.argsort([evaluate_c5(c) for c in candidates])[:5]
    for idx in top_indices:
        if time.time() - start_time > budget_s - 10:
            break
        candidate = candidates[idx]
        def obj_refine(x):
            x = x.reshape(-1)
            x = clip_and_rescale(x)
            return evaluate_c5(x)
        result_refine = minimize(
            obj_refine, x0=candidate, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 150, 'gtol': 1e-9, 'disp': False}
        )
        if result_refine.success:
            h_ref = clip_and_rescale(result_refine.x)
            c5_ref = evaluate_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    return (best_h.tolist(), best_c5, n_points)
