        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def generate_h_from_c(c, K):
        x = np.arange(n_points) * dx
        k_values = np.arange(1, K+1)
        cos_terms = np.cos(k_values[:, None] * np.pi * x[None, :])
        h = 0.5 + np.dot(c, cos_terms)
        return h

    def objective_fourier(c):
        K = len(c)
        h = generate_h_from_c(c, K)
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.inf
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        corr_vals = np.correlate(scaled, 1 - scaled, mode='full')
        return float(np.max(corr_vals) * dx)

    def compute_initial_coefficients(h, K):
        n_points_sample = len(h)

        x = np.arange(n_points_sample) * dx_sample
        c = []

            term = (h - 0.5) * np.cos(k * np.pi * x)
            c_k = (2.0 / n_points_sample) * np.sum(term)
            c.append(c_k)
        return np.array(c)

    K = 20  # Number of cosine terms
    c_initial = compute_initial_coefficients(h_initial, K)

    # Frequency domain optimization
    best_c5 = np.inf

    if time.time() - start_time < budget_s - 300:
