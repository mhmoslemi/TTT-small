    # Final refinement near best solution
    refine_params = best_params if results else initial_params
    bounds = [(max(p[0], x-0.1), min(p[1], x+0.1)) for p, x in zip(param_bounds, refine_params)]
    if len(results) > 0:
        res_refine = minimize(objective, refine_params, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 50, 'ftol': 1e-10})
        if res_refine.success:
            final_h = generate_h(res_refine.x)
            final_c5 = evaluate_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
