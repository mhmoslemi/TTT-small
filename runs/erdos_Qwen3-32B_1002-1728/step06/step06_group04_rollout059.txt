        fitness_vals = [calculate_c5(clip_and_rescale(sol)) for sol in solutions]
        es.tell(solutions, fitness_vals)
    best_candidate = es.result.xbest
    best_candidate_repaired = clip_and_rescale(best_candidate)
    best_c5_cma = calculate_c5(best_candidate_repaired)
    if best_c5_cma < best_c5:
        best_h = best_candidate_repaired
        best_c5 = best_c5_cma

    # SLSQP local refinement
    def objective_slsqp(x):
        return calculate_c5(clip_and_rescale(x))
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(clip_and_rescale(x)) - target_sum}]
    result = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        refined_h = clip_and_rescale(result.x)
        refined_c5 = calculate_c5(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    return (best_h.tolist(), best_c5, n_points)
