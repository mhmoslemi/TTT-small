            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'gtol': 1e-8}
        )
        if result.success:
            optimized = result.x.copy()
            c5 = compute_c5(optimized)
            if c5 < best_c5:
                best_c5 = c5
                best_h = optimized.copy()

    intermediate_results = [(c, compute_c5(c)) for c in perturbed_candidates]
    intermediate_results.sort(key=lambda x: x[1])
    top_intermediates = intermediate_results[:5]

    for candidate, _ in top_intermediates:
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            fun=objective,
            x0=candidate.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'gtol': 1e-9}
        )
        if result.success:
            optimized = result.x.copy()
            c5 = compute_c5(optimized)
            if c5 < best_c5:
                best_c5 = c5
                best_h = optimized.copy()

    return (best_h.tolist(), best_c5, n_points)
