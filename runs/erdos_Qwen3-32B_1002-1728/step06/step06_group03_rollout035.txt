    sum_h = np.sum(h_clipped)
    if sum_h != 0:
        h_scaled = (h_clipped / sum_h) * target_sum
    else:
        h_scaled = h_clipped.copy()
    h_final = np.clip(h_scaled, 0, 1)
    c5_bound = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx

    return (h_final.tolist(), c5_bound, n_points)
