        h_clipped = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum != 0:
            h_scaled = h_clipped * target_sum / current_sum
        else:
            h_scaled = h_clipped * 0
        h_scaled = np.clip(h_scaled, 0, 1)
        overlaps = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
        max_overlap = np.linalg.norm(overlaps, p_norm) ** (1 / p_norm)
        return max_overlap + lambda_l1 * np.linalg.norm(theta_flat, 1)

    best_theta = theta.copy()
    best_c5 = float('inf')

    # Main optimization loop
    max_iter = 200
    for phase in range(2):
        # Perturb initial guess
        np.random.seed(seed + phase)
        perturbation = np.random.normal(0, 0.1, len(theta))
        perturbed_theta = theta + perturbation
        perturbed_theta = np.clip(perturbed_theta, -np.max(np.abs(theta)) * 2, np.max(np.abs(theta)) * 2)
        
        # Run optimization
        result = minimize(
            objective, perturbed_theta,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-9, 'gtol': 1e-8}
        )
        if result.success:
            theta_opt = result.x
            theta_opt = np.clip(theta_opt, -np.max(np.abs(theta)) * 2, np.max(np.abs(theta)) * 2)
            h_full = inverse_haar(theta_opt)
            h_full = np.clip(h_full, 0, 1)
            current_sum = np.sum(h_full)
            if current_sum != 0:
                h_full = h_full * target_sum / current_sum
            h_full = np.clip(h_full, 0, 1)
            overlaps = np.correlate(h_full, 1 - h_full, mode='full') * dx
            c5 = np.max(overlaps)
            if c5 < best_c5:
                best_theta = theta_opt
                best_c5 = c5

        # Time check
        if time.time() - start_time > budget_s - 30:
            break
        # Decay lambda_l1 in second phase
        if phase == 1:
            lambda_l1 *= 0.5

    # Final processing
    h_final = inverse_haar(best_theta)
    h_final = np.clip(h_final, 0, 1)
    current_sum = np.sum(h_final)
    if current_sum != 0:
        h_final = h_final * target_sum / current_sum
    h_final = np.clip(h_final, 0, 1)

    # Final evaluation
    overlaps = np.correlate(h_final, 1 - h_final, mode='full') * dx
    final_c5 = np.max(overlaps)
    
    return (h_final.tolist(), final_c5, N)
