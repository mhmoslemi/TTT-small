import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def enforce_feasibility(h):
        h_np = np.array(h, copy=True)
        h_np = np.clip(h_np, 0, 1)
        sum_h = np.sum(h_np)
        if sum_h != 0:
            h_np = (h_np / sum_h) * target_sum
        h_np = np.clip(h_np, 0, 1)
        return h_np

    h_current = enforce_feasibility(h_initial).copy()

    def evaluate(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    current_c5 = evaluate(h_current)
    best_h = h_current.copy()
    best_c5 = current_c5

    def mutate(h, T):
        mutation = np.random.normal(0, T, len(h))
        h_new = h + mutation
        h_new = enforce_feasibility(h_new)
        return h_new

    T0 = 1.0
    alpha = 0.99
    max_iter_sa = 1000
    T = T0
    np.random.seed(seed)

    for _ in range(max_iter_sa):
        if time.time() - start_time > budget_s - 30:
            break
        h_new = mutate(h_current, T)
        c5_new = evaluate(h_new)
        delta = c5_new - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            current_c5 = c5_new
            if delta < 0:
                best_h = h_current.copy()
                best_c5 = current_c5
        T *= alpha

    def fun(x):
        return evaluate(x)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    result = minimize(
        fun=fun,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    if result.success:
        optimized_h = result.x
        optimized_c5 = evaluate(optimized_h)
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            best_h = optimized_h

    h_full = enforce_feasibility(best_h)
    return (h_full.tolist(), best_c5, n_points)
