    def create_full_h(h_vars):

        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

        h_full = create_full_h(h_half)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    # Projection function for particles
    def project_particle(candidate):
        clipped = np.clip(candidate, 0, 1)
        current_sum = np.sum(clipped)
        if current_sum == 0:
            return np.zeros_like(candidate)
        scaled = (target_sum_half / current_sum) * clipped
        scaled_clipped = np.clip(scaled, 0, 1)
        new_sum = np.sum(scaled_clipped)
        if abs(new_sum - target_sum_half) > 1e-6:
            # Rescale
            scaling_factor = target_sum_half / new_sum
            rescaled = scaled_clipped * scaling_factor
            rescaled = np.clip(rescaled, 0, 1)
            return rescaled
        else:

    h_full_sym = h_full_initial.copy()

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_half):
        h_full = create_full_h(h_half)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    def project_particle(candidate):
        clipped = np.clip(candidate, 0, 1)
        current_sum = np.sum(clipped)
        if current_sum == 0:
            return np.zeros_like(candidate)
        scaled = (target_sum_half / current_sum) * clipped
        scaled = np.clip(scaled, 0, 1)
        new_sum = np.sum(scaled)
        if abs(new_sum - target_sum_half) > 1e-6:
            scale_factor = target_sum_half / new_sum
            scaled = scaled * scale_factor
            scaled = np.clip(scaled, 0, 1)
        return scaled

    np.random.seed(seed)
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum_full = np.sum(h_full_sym)
    if abs(current_sum_full - (n_points / 2.0)) > 1e-6:
        h_full_sym = (h_full_sym / current_sum_full) * (n_points / 2.0)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_init = h_full_sym[:half_n]
    h_half_init = project_particle(h_half_init)

    n_particles = 20
    particles = []
    for _ in range(n_particles):
        perturbation = np.random.randn(half_n) * 0.15
        noisy = h_half_init + perturbation
        particles.append(project_particle(noisy))
    particles = np.array(particles)
    velocities = np.random.randn(n_particles, half_n) * 0.05

    pbest_pos = particles.copy()
    pbest_vals = np.array([objective(p) for p in pbest_pos])
    gbest_idx = np.argmin(pbest_vals)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

    w, c1, c2 = 0.7, 1.5, 1.5
    max_iter = int(100)
    for it in range(max_iter):
        if time.time() - start_time >= budget_s - 20:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(half_n), np.random.rand(half_n)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest_pos[i] - particles[i]) +
                             c2 * r2 * (gbest_pos - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = project_particle(new_pos)
            particles[i] = new_pos
        current_vals = np.array([objective(p) for p in particles])
        for i in range(n_particles):
            if current_vals[i] < pbest_vals[i]:
                pbest_vals[i] = current_vals[i]
                pbest_pos[i] = particles[i].copy()
        new_gbest_idx = np.argmin(pbest_vals)
        if pbest_vals[new_gbest_idx] < gbest_val:
            gbest_val = pbest_vals[new_gbest_idx]
            gbest_pos = pbest_pos[new_gbest_idx].copy()

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    result = minimize(
        fun=objective,
        x0=gbest_pos,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result.success:
        final_half = project_particle(result.x)
        final_c5 = objective(final_half)
    else:
        final_half = gbest_pos
        final_c5 = objective(final_half)

    h_full = create_full_h(final_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), final_c5, n_points)
