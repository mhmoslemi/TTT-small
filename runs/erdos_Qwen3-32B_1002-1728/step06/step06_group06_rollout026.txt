import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    n_particles = 20
    np.random.seed(seed)
    start_points = []
    for _ in range(n_particles):
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + perturbation
        feasible = clip_and_rescale(perturbed)
        start_points.append(feasible.copy())

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    for start_point in start_points:
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            fun=objective,
            x0=start_point,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8, 'disp': False}
        )
        if result.success:
            optimized = result.x
            optimized = clip_and_rescale(optimized)
            c5 = objective(optimized)
            if c5 < best_c5:
                best_h[:] = optimized
                best_c5 = c5

    return (best_h.tolist(), best_c5, n_points)
