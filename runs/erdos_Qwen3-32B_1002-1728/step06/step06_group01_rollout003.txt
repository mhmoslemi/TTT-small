    h_full = np.clip(h_full, 0, 1)
    scale_full = (n_points / 2.0) / np.sum(h_full)
    h_full *= scale_full
    assert np.isclose(np.sum(h_full), n_points / 2.0, atol=1e-6)
    return (h_full.tolist(), best_c5, n_points)
