    def compute_h(coeffs):
        a0 = coeffs[0]
        a = coeffs[1:N+1]
        b = coeffs[N+1:]
        series = a0 + sum(a_n * np.cos((n+1)*np.pi*x_array) + b_n * np.sin((n+1)*np.pi*x_array) for n, (a_n, b_n) in enumerate(zip(a, b)))
        return 1 / (1 + np.exp(-series))

    # Objective function in Fourier space
    def objective(coeffs):
        h = compute_h(coeffs)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: sum(h) = n_points/2
    def constraint_sum(coeffs):
        return np.sum(compute_h(coeffs)) - n_points / 2.0

    # Initial coefficients fitting to h_initial
    eps = 1e-8
    log_odds = np.log((h_initial + eps) / (1 - h_initial + eps + (h_initial == 1)))
    a0_init = np.mean(log_odds)
    coeffs_init = np.concatenate([[a0_init], np.zeros(N), np.zeros(N)])

    # Fit Fourier coeffs to rough estimate
    def residuals(coeffs):
        return (compute_h(coeffs) - h_initial).ravel()

    fit_result = least_squares(residuals, coeffs_init, max_nfev=30)
    coeffs = fit_result.x.copy()

    # Fourier space optimization with SLSQP
    bounds_coeff = Bounds([-10] * (1 + 2*N), [10] * (1 + 2*N))
    cons = ({'type': 'eq', 'fun': constraint_sum})
    result = minimize(
        objective,
        coeffs,
        method='SLSQP',
        bounds=bounds_coeff,
        constraints=cons,
        options={'maxiter': 40, 'ftol': 1e-6}
    )
    if time.time() - start_time > budget_s - 120:
        coeffs_best = result.x
    else:
        # Gradient-based local refinement in real space
        h_fg = compute_h(result.x)
        def real_obj(h_vec):
            return np.max(np.correlate(h_vec, 1-h_vec, mode='full')) * dx
        cons_real = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2})
        real_result = minimize(
            real_obj, h_fg,
            method='SLSQP',
            bounds=Bounds([0]*n_points, [1]*n_points),
            constraints=cons_real,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        h_best = real_result.x if real_result.success else h_fg
        # Final check
        if time.time() - start_time < budget_s:
            # Further polish using Fourier parameters
            coeffs_best = result.x
        else:
            coeffs_best = result.x

    h_final = compute_h(coeffs_best)
    c5 = objective(coeffs_best)
    return (h_final.tolist(), c5, n_points)
