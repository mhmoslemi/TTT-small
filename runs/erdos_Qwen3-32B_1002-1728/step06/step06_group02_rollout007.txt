    n_points = len(h_initial)

    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-6:
        h_initial = h_initial * (target_sum / current_sum)
        h_initial = np.clip(h_initial, 0, 1)

    # Initialize ADMM variables
    h = h_initial.copy()
    z = h_initial.copy()
    u = np.zeros(n_points)
    rho = 1.0
    max_rho = 10.0
    rho_decay_rate = 0.95
    lr = 0.1  # Initial learning rate
    best_c5 = np.inf
    best_h = h.copy()

    # Function to compute gradients for max overlap k
    def compute_overlap(h_eval):
        n = len(h_eval)
        overlap = np.zeros(n)
        for k in range(n):
            sum_term = 0.0
            for i in range(n):
                idx_plus = (i +k) %n
                sum_term += h_eval[i] * (1.0 - h_eval[idx_plus])
            overlap[k] = sum_term * dx
        return overlap

    # Function to compute gradient for given k
    def compute_gradient(h_eval, k):
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx_plus = (j +k) %n_points
            idx_minus = (j -k) %n_points
            grad[j] = (1.0 - h_eval[idx_plus] - h_eval[idx_minus]) * dx
        return grad

    # ADMM iterations
    while time.time() - start_time < budget_s - 50:  # leave some time for final steps
        # Compute overlaps for current h
        overlaps = compute_overlap(h)
        current_c5 = np.max(overlaps)
        k_max = np.argmax(overlaps)
