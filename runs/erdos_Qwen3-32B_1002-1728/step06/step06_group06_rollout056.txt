        current_sum = np.sum(h_clipped)
        if current_sum < 1e-8:
            h_clipped[:] = 0.5
        else:
            h_clipped *= target_sum / current_sum
        return np.clip(h_clipped, 0, 1)

    cognitive = 1.5
    social = 1.5
