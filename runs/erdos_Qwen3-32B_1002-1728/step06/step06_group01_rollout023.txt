            alpha_pert = project_simplex(alpha_pert)
            h_pert = compute_h(alpha_pert)
            h_pert_clipped = np.clip(h_pert, 0, 1)
            corr_pert = np.correlate(h_pert_clipped, 1 - h_pert_clipped, mode='full')[max_idx]
            grad[i] = (corr_pert - max_val) / eps_fd

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = project_simplex(alpha - step)
        alpha = project_simplex((alpha - step) / 2.0 + alpha / 2.0)

        # Schedule learning rate and pattern renewal
        if t % 200 == 0:
            lr *= 0.9
            if t % 400 == 0:
                new_patterns = patterns.copy()
                indices = np.random.choice(n_points, size=int(half_sum), replace=False)
                new_patterns[1] = np.zeros(n_points)
                new_patterns[1][indices] = 1.0
                patterns = new_patterns

    # Final check
    h_best = compute_h(best_alpha)
    h_best_clipped = np.clip(h_best, 0, 1)
    h_best_clipped = h_best_clipped / np.sum(h_best_clipped) * half_sum
    h_best_clipped = np.clip(h_best_clipped, 0, 1)
    corr_vals = np.correlate(h_best_clipped, 1 - h_best_clipped, mode='full')
    best_c5 = np.max(corr_vals) * (2.0 / n_points)
    assert np.allclose(np.sum(h_best_clipped), half_sum, atol=1e-6)
    return (h_best_clipped.tolist(), best_c5, n_points)
