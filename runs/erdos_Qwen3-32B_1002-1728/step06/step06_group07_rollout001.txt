        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    bounds_time = [(0.0, 1.0) for _ in range(n_points)]

    # Use up remaining budget for local optimization
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        result = minimize(
            time_domain_objective, x0=best_h, method='SLSQP',
            bounds=bounds_time,
            options={'maxiter': int(remaining_time // 0.5), 'ftol': 1e-8},
            tol=1e-4
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = time_domain_objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Ensure constraint satisfaction
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-5, "Sum invalid"
    return (final_h.tolist(), best_c5, n_points)
