from scipy.optimize import minimize
import time

def enforce_feasibility(h):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total != 0:

    h = np.clip(h, 0, 1)
    return h

    n = len(h)
    dx = 2.0 / n
    return np.max(corr) * dx

    h_new = h + mutation
    h_new = enforce_feasibility(h_new)
    return h_new

    best_c5 = evaluate(best_h)
    current_h = h_current.copy()
    current_c5 = best_c5
    T = T0

    np.random.seed(seed)
