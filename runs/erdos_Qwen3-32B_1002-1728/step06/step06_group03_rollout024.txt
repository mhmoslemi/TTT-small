    l10 = base_decay * (1 + 0.4 * asymmetry)
    l20 = base_decay * (1 - 0.4 * asymmetry)
    
    initial_params = [a0, b0, c0, w10, w20, l10, l20]
    bounds = [
        (0.05, 0.95),  # a
        (0.05, 0.95),  # b
        (0.3, 1.7),    # c
        (0.1, 0.7),    # w1
        (0.1, 0.7),    # w2
        (0.5, 3.0),    # l1
        (0.5, 3.0),    # l2
    ]
    
    def progress_callback(result):
        elapsed = time.time() - start_time
        if elapsed > 850:
            return "terminate"
        return "continue"
    
    opt_result = minimize(
        fun=lambda p: objective_function(p, n_points, dx),
        x0=initial_params,
        bounds=bounds,
        method='L-BFGS-B',
        jac=lambda p: np.float64(0),  # placeholder
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-4},
        tol=None
    )
    
    best_params = opt_result.x if opt_result.success else initial_params
    h_best = generate_normalized_h(best_params, n_points)
    
    # Final safeguard: double-check constraint
    if abs(np.sum(h_best) - n_points/2) > 1e-6:
        h_best = h_best * (n_points/2) / np.sum(h_best)
        h_best = np.clip(h_best, 0, 1)
    
    corr_vals = np.correlate(h_best, 1 - h_best, mode='full')
    best_c5 = np.max(corr_vals) * dx
    
    return (h_best.tolist(), best_c5, n_points)
