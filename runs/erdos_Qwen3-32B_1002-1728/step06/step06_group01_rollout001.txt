import numpy as np
from scipy.optimize import minimize, PSO
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # because sum(h_full) is n/2, half_n entries contribute half that

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def clip_and_scale_h(h_vars):
        h_full = create_full_h(h_vars)
        h_full = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_full)
        if np.abs(current_sum - n_points / 2.0) > 1e-6 and current_sum != 0:
            scale = (n_points / 2.0) / current_sum
            h_full *= scale
        return h_full

    def evaluate_h(h_vars):
        h_full = clip_and_scale_h(h_vars)
        h_flat = h_full.copy()
        corr_vals = np.correlate(h_flat, 1 - h_flat, mode='full') * dx
        return np.max(corr_vals)

    # Particle Swarm Optimization
    np.random.seed(seed)
    n_particles = 20
    personal_best = h_initial[:half_n].copy()
    personal_best_eval = evaluate_h(personal_best)
    global_best = personal_best.copy()
    global_best_eval = personal_best_eval

    # Initialize particles
    particles = np.random.rand(n_particles, half_n) * 0.1 + h_initial[:half_n]
    velocities = np.random.rand(n_particles, half_n) * 0.05 - 0.025
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([evaluate_h(p) for p in particles])

    for idx in range(n_particles):
        if personal_best_scores[idx] < personal_best_eval:
            personal_best = particles[idx].copy()
            personal_best_eval = personal_best_scores[idx]
            if personal_best_eval < global_best_eval:
                global_best = personal_best.copy()
                global_best_eval = personal_best_eval

    # PSO parameters
    inertia = 0.9
    cognitive = 2.0
    social = 2.0

    while time.time() - start_time < budget_s * 0.8:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + \
                            cognitive * r1 * (personal_best_positions[i] - particles[i]) + \
                            social * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            total = np.sum(particles[i])
            if total != 0:
                particles[i] = (particles[i] / total) * target_sum
            particles[i] = np.clip(particles[i], 0, 1)

        current_scores = np.array([evaluate_h(p) for p in particles])
        improved = current_scores < personal_best_scores
        personal_best_scores[improved] = current_scores[improved]
        personal_best_positions[improved] = particles[improved]

        new_global_index = np.argmin(current_scores)
        if current_scores[new_global_index] < global_best_eval:
            global_best = particles[new_global_index].copy()
            global_best_eval = current_scores[new_global_index]

        if time.time() - start_time > budget_s:
            break

    # Local refinement of best solution
    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective_local(x):
        return evaluate_h(x)

    result = minimize(
        fun=objective_local,
        x0=global_best,
        method='SLSQP',
        bounds=bounds_per_var,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9}
    )
    if result.success:

        final_c5 = evaluate_h(refined)
        if final_c5 < global_best_eval:

            global_best_eval = final_c5

    h_full = clip_and_scale_h(global_best)
    assert np.abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Constraint violation"
    return (h_full.tolist(), global_best_eval, n_points)
