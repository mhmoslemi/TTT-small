        optimized_c5 = fun(optimized_h)
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            best_h = optimized_h

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        best_h = (best_h / sum_best) * target_sum
    best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
