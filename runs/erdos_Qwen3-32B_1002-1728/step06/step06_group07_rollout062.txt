                new_ind = parent.copy()
                new_ind[i_zero], new_ind[i_one] = 1, 0
            else:
                new_ind = parent.copy()
            new_pop.append(new_ind)
        pop = new_pop

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective_continuous(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    refined_h = clip_and_scale(best_binary.astype(float))
    best_h = refined_h.copy()

    result = minimize(
        objective_continuous,
        x0=refined_h,
        method='SLSQP',
        bounds=[(0,1)]*n_points,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result.success:
        h_candidate = clip_and_scale(result.x)
        c5_candidate = objective_continuous(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    assert abs(best_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (best_h.tolist(), best_c5, n_points)
