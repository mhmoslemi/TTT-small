        # Compute smoothed objective and gradient

        if sum_exp == 0:

        else:
            h_proposed[:] = target_sum / n_points
        h_proposed = np.clip(h_proposed, 0, 1)
        h = h_proposed

        # Update temperature and learning rate
        tau *= tau_decay
        learning_rate *= lr_decay
