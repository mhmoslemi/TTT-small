import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_initial = project(h_initial)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    np.random.seed(seed)
    n_candidates = 20
    candidates = []
    for _ in range(n_candidates):
        perturbation = np.random.normal(0, 0.1, n_points)
        candidate = h_initial + perturbation
        candidate = project(candidate)
        candidates.append(candidate)

    max_refine_iter = 100
    convergence_eps = 1e-8

    for candidate in candidates:
        current_h = candidate.copy()
        current_c5 = compute_c5(current_h)
        for _ in range(max_refine_iter):
            if time.time() - start_time > budget_s - 5:
                break
            current_h = project(current_h)
            new_c5 = compute_c5(current_h)
            if abs(new_c5 - current_c5) < convergence_eps:
                break
            current_c5 = new_c5
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
