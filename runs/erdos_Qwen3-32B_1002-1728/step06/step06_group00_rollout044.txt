                best_h = np.copy(ind)

        # Selection via tournament
        new_population = []
        for _ in range(len(population)):
            contestants = np.random.choice(population, size=3)
            selected = min(contestants, key=lambda x: np.max(np.correlate(create_full_h(repair(x)), 1 - create_full_h(repair(x)), mode='full') * dx))
            new_population.append(np.copy(selected))

        # Crossover and mutation
        for i in range(len(population)):
            parent1 = new_population[i % len(new_population)]
            parent2 = new_population[(i + 1) % len(new_population)]

            # Differential evolution style mutation
            trial = parent1 * 0.5 + parent2 * 0.5 + np.random.normal(0, 0.05, half_n)
            trial = np.clip(trial, 0, 1)
            trial = repair(trial)

            h_trial = create_full_h(trial)
            h_parent = create_full_h(parent1)

            # Use rank-based recombination
            trial_c5 = np.max(np.correlate(h_trial, 1 - h_trial, mode='full') * dx)
            parent_c5 = np.max(np.correlate(h_parent, 1 - h_parent, mode='full') * dx)
            if trial_c5 < parent_c5:
                new_population[i] = trial

        population = new_population

    # Local optimization
    h_half_local = best_h.copy()
    def objective(x):
        x = np.clip(x, 0, 1)
        x = repair(x)
        h = create_full_h(x)
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    result = minimize(
        objective, 
        h_half_local,
        bounds=[(0, 1)] * half_n,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * 2 / n_points - target_sum / 2},
        method='L-BFGS-B',
        options={'maxiter': 200, 'ftol': 1e-6}
    )

    if result.success:
        final_h = result.x
    else:
        final_h = best_h

    final_h = repair(final_h)
    h_full = create_full_h(final_h)
    h_full = np.clip(h_full, 0, 1)
    h_full *= (target_sum / h_full.sum())  # Final renormalization

    final_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full') * dx)
    return (h_full.tolist(), final_c5, n_points)
