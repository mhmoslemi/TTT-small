import numpy as np
import time
import cvxpy as cp

def calculate_c5(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n
    target_sum = n / 2.0

    h = cp.Variable(shape=h_initial.shape, nonneg=True)
    p_vars = {}
    shift_to_p = {}

    for k in range(-n + 1, n):
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                var_name = f'p_{i}_{k}'
                p = cp.Variable(name=var_name, nonneg=True)
                p_vars[(i, k)] = p
                if k not in shift_to_p:
                    shift_to_p[k] = []
                shift_to_p[k].append(p)

    C = cp.Variable(pos=True)
    constraints = [
        cp.sum(h) == target_sum,
        0 <= h,
        h <= 1
    ]

    for (i, k), p in p_vars.items():
        j = i + k
        constraints += [
            p >= h[i] - h[j],
            p <= h[i],
            p <= 1 - h[j]
        ]

    for k, p_list in shift_to_p.items():
        constraints.append(sum(p_list) * dx <= C)

    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)

    time_remaining = max(0.0, budget_s - (time.time() - start_time))
    if time_remaining < 10:

        best_h *= target_sum / np.sum(best_h)

    problem.solve(solver=cp.GLPK, verbose=False, time_limit=time_remaining)

    status_ok = problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
