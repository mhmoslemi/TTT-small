import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)

    dx = 2.0 / n_points

    def generate_h(params):
        a, b, c, w1, w2, lam1, lam2 = params
        h_full = np.zeros(n_points)
        for i in range(n_points):
            x = dx * i
            if x < c:
                left_start = c - w1 / 2
                left_end = c + w1 / 2
                if left_start <= x <= left_end:
                    distance = abs(x - c)
                    tri_val = a * (1 - distance / (w1 / 2))
                    if tri_val > 1:
                        tri_val = 1.0
                    else:
                        tri_val = 0.0
                else:
                    tri_val = 0.0
                h_val = tri_val * np.exp(-lam1 * x)
            else:
                right_start = c - w2 / 2
                right_end = c + w2 / 2
                if right_start <= x <= right_end:
                    distance = abs(x - c)
                    tri_val = b * (1 - distance / (w2 / 2))
                    if tri_val > 1:
                        tri_val = 1.0
                    else:
                        tri_val = 0.0
                else:
                    tri_val = 0.0
                h_val = tri_val * np.exp(-lam2 * (x - c))
            h_full[i] = h_val
        h_sum = np.sum(h_full)
        if h_sum > 1e-6:
            h_full *= (n_points / 2) / h_sum
        h_full = np.clip(h_full, 0.0, 1.0)
        return h_full

    def objective(params):
        h_full = generate_h(params)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initial guess (consistent with strategy)
    initial_params = [0.8, 0.7, 1.0, 0.5, 0.5, 1.5, 1.5]

        (1.0, 2.5),  # lam1
        (1.0, 2.5)   # lam2
    ]

    # Optimization setup
    options = {'maxiter': 500, 'ftol': 1e-6, 'disp': False}
