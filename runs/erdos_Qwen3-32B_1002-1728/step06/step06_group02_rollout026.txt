        for k in range(n_points):
            shifted = np.roll(h, -k)
            overlap_vals[k] = 1.0 - dx * np.sum(h * shifted)
        max_overlap = np.max(overlap_vals)
        k_max = np.argmax(overlap_vals)

        # Compute gradient for k_max
        grad_Ck = np.zeros(n_points)
        for j in range(n_points):
            idx1 = (j + k_max) % n_points
            idx2 = (j - k_max) % n_points
            grad_Ck[j] = dx * (1.0 - h[idx1] - h[idx2])
        # ADMM gradient term
        grad_admm = rho * (h - z + lambda_ / rho)
        grad_total = grad_Ck + grad_admm

        # Gradient descent step
        h_new = h - step_size * grad_total
        h_new = np.clip(h_new, 0 - 1e-8, 1 + 1e-8)  # Allow for numerical stability
        h = h_new

        # Project z onto feasible region
        z_new = h + lambda_ / rho
        z_new = np.clip(z_new, 0.0, 1.0)
        current_sum_z = np.sum(z_new)
        if np.isclose(current_sum_z, 0.0):
            z_new[:] = target_sum / n_points
        else:
            z_new *= target_sum / current_sum_z
        z_new = np.clip(z_new, 0.0, 1.0)
        z = z_new

        # Update lambda
        lambda_ += rho * (z - h)

        # Cool down rho
        if iter_count % 100 == 0 and iter_count > 0:
            rho *= 0.95

        # Check current C5
        current_c5 = np.max(overlap_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Local refinement with COBYLA
    # Now, perform a final optimization with COBYLA on the best_h
    # Define objective function
    def objective(x):
        x = np.array(x)
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if abs(sum_x - target_sum) > 1e-6:
            x = x * (target_sum / sum_x)
            x = np.clip(x, 0, 1)
        overlap_vals = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(x, -k)
            overlap_vals[k] = 1.0 - dx * np.sum(x * shifted)
        return np.max(overlap_vals)

    bounds = [(0, 1) for _ in range(n_points)]
    from scipy.optimize import minimize
    result = minimize(objective, x0=best_h, method='COBYLA', bounds=bounds, options={'maxiter': 100})
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        sum_ref = np.sum(refined_h)
        if abs(sum_ref - target_sum) > 1e-6:
            refined_h = refined_h * (target_sum / sum_ref)
            refined_h = np.clip(refined_h, 0, 1)
        overlap_vals = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(refined_h, -k)
            overlap_vals[k] = 1.0 - dx * np.sum(refined_h * shifted)
        refined_c5 = np.max(overlap_vals)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    # Final check
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
