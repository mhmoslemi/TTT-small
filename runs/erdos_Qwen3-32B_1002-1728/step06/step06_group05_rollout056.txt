import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = (h / sum_h) * target_sum
        return np.clip(h, 0, 1)

    h_initial = enforce_constraints(h_initial)

    # PSO Parameters
    pop_size = 30
    max_iter = 30
    particles = np.zeros((pop_size, n_points))
    velocities = np.zeros((pop_size, n_points))

    # Initialize particles
    np.random.seed(seed)
    perturbed = h_initial + np.random.normal(0, 0.2, (pop_size // 2, n_points))
    random_init = np.random.uniform(0, 1, (pop_size // 2, n_points))
    random_init = np.array([enforce_constraints(r) for r in random_init])
    particles[:pop_size // 2] = perturbed
    particles[pop_size // 2:] = random_init
    velocities = np.random.uniform(-0.05, 0.05, (pop_size, n_points))

    def objective(x):
        x = enforce_constraints(x)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    personal_best = particles.copy()
    personal_best_obj = np.array([objective(p) for p in particles])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = particles[global_best_idx].copy()
    best_c5 = personal_best_obj[global_best_idx]

    # PSO Optimizer
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 100:
            break
        w = 0.9 - (iter / max_iter) * (0.9 - 0.4)
        r1 = np.random.rand(pop_size, n_points)
        r2 = np.random.rand(pop_size, n_points)
        velocities = (w * velocities +
                      1.2 * r1 * (personal_best - particles) +
                      1.2 * r2 * (global_best[np.newaxis, :] - particles))
        particles += velocities
        particles = np.array([enforce_constraints(p) for p in particles])
        obj_vals = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        current_best_idx = np.argmin(personal_best_obj)
        if personal_best_obj[current_best_idx] < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = personal_best_obj[current_best_idx]

    # Local Grid Search
    top_indices = np.argsort(personal_best_obj)[:5]
    for idx in top_indices:
        top_candidate = personal_best[idx].copy()
        for _ in range(5):
            neighbor = top_candidate + np.random.uniform(-0.02, 0.02, n_points)
            neighbor = enforce_constraints(neighbor)
            neighbor_obj = objective(neighbor)
            if neighbor_obj < personal_best_obj[idx]:
                personal_best[idx] = neighbor.copy()
                personal_best_obj[idx] = neighbor_obj
                if neighbor_obj < best_c5:
                    global_best = neighbor.copy()
                    best_c5 = neighbor_obj

    # Final SLSQP Optimization
    final_candidates = personal_best[np.argsort(personal_best_obj)][:3]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    for i in range(len(final_candidates)):
        if time.time() - start_time > budget_s - 50:
            break
        result = minimize(
            fun=objective,
            x0=final_candidates[i],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            optimized_h = result.x
            optimized_h = enforce_constraints(optimized_h)
            current_c5 = objective(optimized_h)
            if current_c5 < best_c5:
                global_best = optimized_h
                best_c5 = current_c5

    h_final = enforce_constraints(global_best)
    return (h_final.tolist(), best_c5, n_points)
