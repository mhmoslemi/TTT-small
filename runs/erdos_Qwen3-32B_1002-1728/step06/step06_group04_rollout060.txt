    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Prepare initial h with correct constraints
    h_full_clipped = np.clip(h_full_initial, 0.0, 1.0)
    current_sum = h_full_clipped.sum()
    if current_sum > 1e-10:
        h_full_clipped *= target_sum / current_sum
    else:
        h_full_clipped[:] = 0.5
    best_h = h_full_clipped.copy()
    best_c5 = calculate_c5(best_h, dx)

    # Define the objective function for differential evolution
    def objective_de(x):
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        if sum_x < 1e-10:
            return float('inf')
        rescaled_x = x * target_sum / sum_x
        corr_vals = np.correlate(rescaled_x, 1 - rescaled_x, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result_de = None
    try:
        result_de = differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            maxiter=20,  # Parameter tuned for time efficiency
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            workers=1,
            polish=False,
            updating='deferred',
        )
    except Exception:
        pass  # Handle failure gracefully

    if time.time() - start_time > budget_s - 30:
        return (best_h.tolist(), best_c5, n_points)

    # If DE succeeded, use the solution for local optimization
    if result_de and result_de.success:
        x_de = result_de.x
        x_de_clipped = np.clip(x_de, 0.0, 1.0)
        sum_de = x_de_clipped.sum()
        if sum_de > 1e-10:
            x_de_clipped *= target_sum / sum_de
        else:
            x_de_clipped[:] = 0.5

        # Run SLSQP local refinement
        def objective_slsqp(x):
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        result_slsqp = minimize(
            objective_slsqp,
            x0=x_de_clipped,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_h[:] = result_slsqp.x
                best_c5 = current_c5

    # Final safety-net local optimization if needed
    def objective_local(x):
        return calculate_c5(x, dx)

    if time.time() - start_time < budget_s - 30:
        bounds_local = [(0.0, 1.0) for _ in range(n_points)]
        cons_local = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        result_local = minimize(
            objective_local,
            x0=best_h.copy(),
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons_local,
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        if result_local.success:
            current_c5 = result_local.fun
            if current_c5 < best_c5:
                best_h[:] = result_local.x
                best_c5 = current_c5

    # Final constraint checks
    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
