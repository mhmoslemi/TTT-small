import numpy as np
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_gradient(h, k_shift, n_points):
    if k_shift == 0:
        return 1.0 - h - h
    abs_k = abs(k_shift)
    sign_k = 1 if k_shift > 0 else -1
    if sign_k > 0:  # Positive shift
        right = np.zeros(n_points)
        right[:n_points - abs_k] = h[abs_k:]
        left = np.zeros(n_points)
        left[abs_k:] = h[:-abs_k]
    else:  # Negative shift
        abs_k = -k_shift
        right = np.zeros(n_points)
        right[abs_k:] = h[:-abs_k]
        left = np.zeros(n_points)
        left[:n_points - abs_k] = h[abs_k:]
    grad = 1.0 - right - left
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h to be feasible
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * target_sum / h_sum
    else:
        h[:] = target_sum / n_points
    h = np.clip(h, 0, 1)

    z = h.copy()
    lambda_ = np.zeros(n_points)
    rho = 1.0
    max_admm_iter = 500
    step_size = 0.05
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    for iter in range(max_admm_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute maximum overlap correlation
        corr_vals = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr_vals)
        k_shift = max_index - (n_points - 1)

        # Compute gradient for current k_shift
        grad = compute_gradient(h, k_shift, n_points)

        # Update h using gradient descent in ADMM framework
        aug_grad = grad + 2 * rho * (h - z) - lambda_
        h_new = h - step_size * aug_grad

        # Project z onto [0,1] and adjust to meet sum constraint
        z = np.clip(h_new, 0, 1)
        z_sum = np.sum(z)
        if z_sum < 1e-10:
            z[:] = target_sum / n_points
        else:
            z *= target_sum / z_sum
        z = np.clip(z, 0, 1)

        # Update lambda and h
        lambda_ += rho * (z - h_new)
        h = h_new.copy()

        # Cool rho every 100 iterations
        if iter % 100 == 0 and iter > 0:
            rho *= 0.95

        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final checks and return
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    return (best_h.tolist(), best_c5, n_points)
