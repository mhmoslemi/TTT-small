from scipy.optimize import minimize, differential_evolution

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0
    lambda_tv = 0.001
    p_norm = 20

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def normalize_h(h_vars):
        h_full = create_full_h(h_vars)
        h_clipped = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_clipped)
        target_sum = n_points / 2.0
        if current_sum > 1e-9:
            h_full_result = h_clipped * (target_sum / current_sum)
        else:
            h_full_result = h_clipped.copy()
        return np.clip(h_full_result, 0, 1)

    def objective_with_tv(h_half):
        h_full = normalize_h(h_half)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full') * dx

        tv = np.sum(np.abs(np.diff(h_full)))
        return approx_max + lambda_tv * tv

    def objective_de(x_vars):
        x = np.array(x_vars)
        total = np.sum(x)
        if total != 0:
            x_scaled = (x / total) * target_sum_half
        else:
            x_scaled = x.copy()
        x_clipped = np.clip(x_scaled, 0, 1)
        return objective_with_tv(x_clipped)

    bounds_de = [(0.0, 1.0)] * half_n
    np.random.seed(seed)
    de_result = differential_evolution(
        objective_de, bounds_de, strategy='best1bin',

    )
    best_h_half = de_result.x * (target_sum_half / np.clip(np.sum(de_result.x), 1e-9, None))
    best_h_half = np.clip(best_h_half, 0, 1)
    best_c5 = objective_with_tv(best_h_half)

    # Local refinement with perturbation-based optimization
    np.random.seed(seed + 1)
    while time.time() - start_time < budget_s - 30:
        idxs = np.random.choice(half_n, size=min(half_n, 8), replace=False)
        perturbed = best_h_half.copy()
        perturbed[idxs] += np.random.normal(0, 0.15, len(idxs))
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        
        def objective_refine(x, sum_target=target_sum_half):
            x = np.clip(x, 0, 1)
            total = np.sum(x)
            if total > 0:

            else:
                x_scaled = x.copy()
            return objective_with_tv(x_scaled)
        
        result = minimize(

            if new_c5 < best_c5:

    h_full = normalize_h(best_h_half)
