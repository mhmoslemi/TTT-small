import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is feasible
    h = np.clip(h_initial, 0, 1).copy()
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)

    z = h.copy()
    lagrange = np.zeros_like(h)

    best_c5 = np.inf
    best_h = h.copy()

    # ADMM parameters
    rho = 1.0
    step_size = 0.1
    iterations = 0

    while time.time() - start_time < budget_s - 60:
        # Compute current overlap and find max k
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr_vals)
        k = max_index

        # Compute gradient for current k
        grad_overlap = np.zeros_like(h)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            term1 = 1.0 - h[i_plus_k]
            term2 = h[i_minus_k]
            grad_overlap[i] = (term1 - term2) * dx

        # ADMM gradient = overlap gradient + rho*(h - z + lagrange)
        grad = grad_overlap + rho * (h - z + lagrange)

        # Update h
        new_h = h - step_size * grad
        new_h = np.clip(new_h, 0, 1)

        # Update z: clip and normalize
        new_z = np.clip(new_h, 0, 1).copy()
        new_sum = new_z.sum()
        new_z = (new_z / new_sum) * target_sum if new_sum > 1e-8 else new_z * target_sum / (n_points)
        new_z = np.clip(new_z, 0, 1)

        # Update Lagrange multiplier
        new_lagrange = lagrange + (new_z - new_h)

        # Assign new values
        h, z, lagrange = new_h, new_z, new_lagrange

        iterations += 1
        if iterations % 100 == 0:

        # Check if current h improves C5
        current_corr = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_corr < best_c5:
            best_c5 = current_corr
            best_h = h.copy()

        # Check time
        if iterations > 1000:
            break

    # Final local refinement with gradient descent
    h_refine = best_h.copy()
    learning_rate = 0.1

            break
        corr_vals = np.correlate(h_refine, 1 - h_refine, mode='full') * dx

            i_plus_k = (i + k) % n_points
