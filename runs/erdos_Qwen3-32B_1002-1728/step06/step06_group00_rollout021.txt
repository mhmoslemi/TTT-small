            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = h_perturbed.sum()
            if current_sum != 0:
                h_perturbed = (h_perturbed / current_sum) * target_sum_full
            h_perturbed = np.clip(h_perturbed, 0, 1)
            perturbations.append(h_perturbed)
        return perturbations

    perturbations = generate_perturbations(10)
    best_c5 = objective(h_full_initial)
    best_h = h_full_initial.copy()
    coarse_optimized = []

    for h_pert in perturbations:
        if time.time() - start_time > budget_s * 0.2:
            break
        h_coarse = downsample(h_pert)
        current_sum_coarse = h_coarse.sum()
        if current_sum_coarse != 0:
            h_coarse = h_coarse / current_sum_coarse * target_sum_coarse
        h_coarse = np.clip(h_coarse, 0, 1)
        cons_coarse = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_coarse})
        bounds_coarse = [(0.0, 1.0)] * len(h_coarse)
        try:
            result_coarse = minimize(
                lambda x: objective(upsample(x)),
                x0=h_coarse,
                method='SLSQP',
                bounds=bounds_coarse,
                constraints=cons_coarse,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result_coarse.success:
                optimized_coarse = result_coarse.x
                optimized_coarse = np.clip(optimized_coarse, 0, 1)
                coarse_optimized.append(optimized_coarse)
        except:
            pass

    for idx, h_coarse in enumerate(coarse_optimized[:2]):
        h_fine = upsample(h_coarse)
        current_sum_fine = h_fine.sum()
        if current_sum_fine != 0:
            h_fine = (h_fine / current_sum_fine) * target_sum_full
        h_fine = np.clip(h_fine, 0, 1)
        c5 = objective(h_fine)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fine.copy()

    if len(coarse_optimized) >= 2:
        h1 = coarse_optimized[0]
        h2 = coarse_optimized[1]
        for alpha in [0.4, 0.6]:
            h_mixed = alpha * h1 + (1 - alpha) * h2
            h_mixed = np.clip(h_mixed, 0, 1)
            current_sum_coarse = h_mixed.sum()
            if current_sum_coarse != 0:
                h_mixed = h_mixed / current_sum_coarse * target_sum_coarse
            h_mixed = np.clip(h_mixed, 0, 1)
            try:
                result_fine_coarse = minimize(
                    lambda x: objective(upsample(x)),
                    x0=h_mixed,
                    method='SLSQP',
                    bounds=bounds_coarse,
                    constraints=cons_coarse,
                    options={'maxiter': 100, 'ftol': 1e-8}
                )
                if result_fine_coarse.success:
                    optimized_h = result_fine_coarse.x
                    h_fine = upsample(optimized_h)
                    h_fine = np.clip(h_fine, 0, 1)
                    current_sum_fine = h_fine.sum()
                    if current_sum_fine != 0:
                        h_fine = (h_fine / current_sum_fine) * target_sum_full
                    h_fine = np.clip(h_fine, 0, 1)
                    c5 = objective(h_fine)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_fine.copy()
            except:
                pass

    h_fine = best_h.copy()
    bounds_fine = [(0.0, 1.0)] * n_points
    cons_fine = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_full})
    np.random.seed(seed)

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        perturbed = h_fine.copy()
        mask = np.random.choice([False, True], size=n_points, p=[0.3, 0.7])
        noise = np.random.normal(0, 0.05, size=n_points)
        perturbed[mask] += noise[mask]
        perturbed = np.clip(perturbed, 0, 1)
        current_sum_fine = perturbed.sum()
        if current_sum_fine != 0:
            perturbed = perturbed / current_sum_fine * target_sum_full
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result_fine = minimize(
                objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds_fine,
                constraints=cons_fine,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            if result_fine.success:
                h_opt = result_fine.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt_sum = h_opt.sum()
                if h_opt_sum != 0:
                    h_opt = h_opt / h_opt_sum * target_sum_full
                h_opt = np.clip(h_opt, 0, 1)
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass

    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final_sum = h_final.sum()
    if abs(h_final_sum - target_sum_full) > 1e-6:
        h_final = h_final / h_final_sum * target_sum_full
        h_final = np.clip(h_final, 0, 1)

    return (h_final.tolist(), best_c5, n_points)
