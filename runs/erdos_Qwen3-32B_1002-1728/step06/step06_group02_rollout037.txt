import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):

    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_gradient(h, corr_vals, dx, tau):
    n = len(h)
    I_vals = corr_vals * dx
    exp_vals = np.exp(tau * I_vals)
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        return np.zeros(n)
    weights = exp_vals / sum_exp

    grad = np.zeros(n)
    for k in range(len(I_vals)):
        s = k - (n - 1)  # Compute lag from k
        x_start = max(0, -s)
        x_end = min(n - 1, n - 1 - s)
        if x_start > x_end:
            continue
        x_valid = np.arange(x_start, x_end + 1)
        x_plus_s = x_valid + s
        w = weights[k]
        grad[x_valid] += w * (1.0 - h[x_plus_s])
        grad[x_plus_s] += w * (-h[x_valid])
    return grad

    total = np.sum(clipped)
    if total <= 1e-10:
        return np.full_like(h, target_sum / h.size)
    return clipped * (target_sum / total)
