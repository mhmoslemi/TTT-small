    max_velocity = 0.1  # Adaptive range to balance exploration/exploitation

    # PSO Arrays
    particles = np.random.rand(n_particles, half_n) * 0.1  # Initial perturbations
    particles[0] = h_half_initial  # Seed with initial solution
    velocities = np.zeros((n_particles, half_n))
    pbest_positions = particles.copy()
    pbest_scores = np.array([objective(p) for p in particles])
    gbest_index = pbest_scores.argmin()
    gbest_position = pbest_positions[gbest_index]
    gbest_score = pbest_scores[gbest_index]

    # Project all particles into feasible space
    for i in range(n_particles):
        p = particles[i]
        total = np.sum(p)
        if total != 0:
            particles[i] = (target_sum_half / total) * p
        particles[i] = np.clip(particles[i], 0, 1)

    # PSO Main Loop
    while time.time() - start_time < budget_s - 30:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, half_n)
            cognitive = c1 * r1 * (pbest_positions[i] - particles[i])
            social = c2 * r2 * (gbest_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            total = np.sum(particles[i])
            if total != 0:
                particles[i] = (target_sum_half / total) * particles[i]
            particles[i] = np.clip(particles[i], 0, 1)

            # Update personal best
            current_score = objective(particles[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i]
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i]
        if time.time() - start_time > budget_s - 30:
            break

    # Final Local Optimization
    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    result = minimize(
        fun=objective,
        x0=gbest_position,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 300}
    )

    if result.success:
        refined = np.clip(result.x, 0, 1)
        total = np.sum(refined)
        if total != 0:
            refined = (target_sum_half / total) * refined
        refined = np.clip(refined, 0, 1)
        final_score = objective(refined)
        if final_score < gbest_score:
            gbest_score = final_score
            gbest_position = refined

    # Final Validation and Output
    h_full = create_full_h(gbest_position)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), gbest_score, n_points)
