    gbest_idx = int(np.argmin(pbest_scores))
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    w = 0.7
    c1 = 1.2
    c2 = 1.2
    max_iter = 200

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(len(particles)):
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel_update = w * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_position - particles[i])
            velocities[i] = vel_update
            new_pos = particles[i] + vel_update
            new_pos_clipped = np.clip(new_pos, 0, 1)
            sum_clipped = np.sum(new_pos_clipped)
            if sum_clipped != 0:
                new_pos_scaled = new_pos_clipped * (target_sum_half / sum_clipped)
            else:
                new_pos_scaled = target_sum_half * initial_h_half / np.sum(initial_h_half)
            particles[i] = np.clip(new_pos_scaled, 0, 1)
            score = evaluate_particle(particles[i])
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest_position = pbest_positions[i].copy()

    candidates = [(score, pos) for score, pos in zip(pbest_scores, pbest_positions)]
    candidates.sort()
    top_candidates = [pos for score, pos in candidates[:3]]
    best_refined = gbest_position.copy()
    best_c5 = gbest_score

    for pos in top_candidates:
        result = minimize(
            evaluate_particle, 
            x0=pos, 
            method='SLSQP', 
            bounds=[(0,1) for _ in range(half_n)], 
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            refined_pos = np.clip(result.x, 0, 1)
            score_refined = evaluate_particle(refined_pos)
            if score_refined < best_c5:
                best_refined = refined_pos.copy()
                best_c5 = score_refined

    h_full = create_full_h(np.clip(best_refined, 0, 1))
    total_sum = np.sum(h_full)
    if abs(total_sum - n_points / 2) > 1e-6:
        h_full = (h_full / total_sum) * (n_points / 2)
    h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
