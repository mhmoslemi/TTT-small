import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h != 0:
        h_initial = h_initial / sum_h * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    def enforce_constraints(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum != 0:
            h_scaled = h_clipped / h_sum * target_sum
        else:
            h_scaled = h_clipped.copy()
        return np.clip(h_scaled, 0, 1)

    def evaluate(h):
        h_constrained = enforce_constraints(h, target_sum)
        corr_vals = np.correlate(h_constrained, 1 - h_constrained, mode='full')
        return np.max(corr_vals) * dx

    pop_size = 30
    max_pso_iter = 30
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0

    np.random.seed(seed)
    particles = np.zeros((pop_size, n_points))
    velocities = np.zeros((pop_size, n_points))
    particles[0] = h_initial.copy()

    # Initialize first half with perturbed initial_h
    for i in range(pop_size // 2):
        if i == 0:
            continue
        perturbation = np.random.normal(0, 0.1, n_points)
        particles[i] = h_initial + perturbation
        particles[i] = enforce_constraints(particles[i], target_sum)
    # Initialize second half with random samples
    for i in range(pop_size // 2, pop_size):
        rand_vals = np.random.uniform(0, 1, n_points)
        particles[i] = rand_vals
        particles[i] = enforce_constraints(particles[i], target_sum)

    velocities = np.random.uniform(-0.05, 0.05, (pop_size, n_points))
    personal_best = particles.copy()
    personal_best_obj = np.array([evaluate(pb) for pb in personal_best])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = personal_best[global_best_idx].copy()
    best_c5 = personal_best_obj[global_best_idx]

    for iter in range(max_pso_iter):
        if time.time() - start_time > budget_s - 60:
            break
        w = w_start - (w_start - w_end) * (iter / max_pso_iter)
        # Update velocities and positions
        r1 = np.random.rand(pop_size, n_points)
        r2 = np.random.rand(pop_size, n_points)
        velocities = (w * velocities +
                      c1 * r1 * (personal_best - particles) +
                      c2 * r2 * (global_best - particles))
        particles += velocities
        particles = np.array([enforce_constraints(p, target_sum) for p in particles])
        # Update personal best
        obj_vals = np.array([evaluate(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        # Update global best
        current_best_idx = np.argmin(personal_best_obj)
        current_c5 = personal_best_obj[current_best_idx]
        if current_c5 < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = current_c5

    # Local grid search on top candidates
    top_candidates = 5
    indices = np.argsort(personal_best_obj)[:top_candidates]
    for idx in indices:
        current_candidate = personal_best[idx].copy()
        for _ in range(5):  # 5 local perturbations per candidate
            if time.time() - start_time > budget_s - 60:
                break
            # Perturb random 5 points
            perturb_indices = np.random.choice(n_points, size=5, replace=False)
            noise = np.random.uniform(-0.05, 0.05, 5)
            new_candidate = current_candidate.copy()
            new_candidate[perturb_indices] += noise
            new_candidate = enforce_constraints(new_candidate, target_sum)
            current_obj = evaluate(current_candidate)
            new_obj = evaluate(new_candidate)
            if new_obj < current_obj:
                current_candidate = new_candidate.copy()
                best_c5 = min(best_c5, new_obj)
        if evaluate(current_candidate) < evaluate(global_best):
            global_best = current_candidate.copy()

    # Final SLSQP optimization
    def fun(x):
        return evaluate(x)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Take top 3 candidates from local search
    final_candidates = [personal_best[i] for i in np.argsort(personal_best_obj)[:3]]
    for x0 in final_candidates:
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            fun=fun,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            optimized = enforce_constraints(result.x, target_sum)
            current_c5 = evaluate(optimized)
            if current_c5 < best_c5:
                best_c5 = current_c5
                global_best = optimized

    h_final = enforce_constraints(global_best, target_sum)
    c5 = evaluate(h_final)
    return (h_final.tolist(), c5, n_points)
