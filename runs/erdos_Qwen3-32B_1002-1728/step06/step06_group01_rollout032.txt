import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    best_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(best_h_half)

    # Generate binary-like initial guess
    num_ones = 17
    positions = [2*i for i in range(num_ones)]  # 0, 2, ..., 32
    h_half_binary = np.zeros(half_n)
    h_half_binary[positions] = 1.0
    zero_indices = [i for i in range(half_n) if i not in positions]
    if zero_indices:
        h_half_binary[zero_indices[0]] = 0.5
    h_half_binary = np.clip(h_half_binary, 0, 1)
    total = np.sum(h_half_binary)
    if total != 0:
        h_half_binary = (h_half_binary / total) * target_sum_half
    h_half_binary = np.clip(h_half_binary, 0, 1)

    # Perturbation and local optimization
    np.random.seed(seed)
    current_h_half = h_half_binary.copy()
    best_h_half = current_h_half.copy()
    best_c5 = objective(best_h_half)

    while time.time() - start_time < budget_s - 60:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.2 * (1 - frac_remaining ** 1.5)
        for _ in range(2):
            seed = (seed + 1) % (2**32)
            rnd = np.random.RandomState(seed)
            idxs = np.random.choice(half_n, size=min(half_n, 10), replace=False)
            perturbed = best_h_half.copy()
            perturbed[idxs] += rnd.normal(0, noise_scale, len(idxs))
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total != 0:
                perturbed = (perturbed / total) * target_sum_half
            perturbed = np.clip(perturbed, 0, 1)

            result = minimize(
                fun=objective, x0=perturbed, method='L-BFGS-B', bounds=bounds_per_var, 
                constraints=cons, options={'maxiter': 400, 'ftol': 1e-10}
            )
            if result.success:
                new_h_half = np.clip(result.x, 0, 1)
                if abs(np.sum(new_h_half) - target_sum_half) < 1e-6:
                    new_c5 = objective(new_h_half)
                    if new_c5 < best_c5:
                        best_h_half = new_h_half
                        best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - (n_points / 2.0)) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
