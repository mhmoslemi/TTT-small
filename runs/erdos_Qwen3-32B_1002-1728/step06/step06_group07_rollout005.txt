from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    N = 20  # Number of Fourier cosine terms
    basis = np.zeros((n_points, N))
    for k in range(N):
        basis[:, k] = np.cos(k * np.pi * x / 1)  # Scale for interval [0,2]
    
    a_coeffs = cp.Variable(N)
    h_fourier = 0.5 + basis @ a_coeffs
    constraints = [h_fourier >= 0, h_fourier <= 1]
    objective = cp.Minimize(cp.sum_squares(a_coeffs))
    
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.MOSEK, verbose=False)
