import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)
    np.random.seed(seed)

    perturbations = []
    for _ in range(30):
        perturbed = h_initial + np.random.normal(0, 0.1, n_points)
        perturbations.append(clip_and_rescale(perturbed))

    best_c5 = np.inf
    best_h = h_initial.copy()

    for candidate in perturbations:
        if time.time() - start_time > budget_s - 30:
            break
        def objective(x):
            x = x.reshape(-1)
            x = clip_and_rescale(x)
            return np.max(np.correlate(x, 1 - x, mode='full')) * dx
        bounds = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum},)
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'gtol': 1e-8, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            x_opt = clip_and_rescale(result.x)
            c5 = np.max(np.correlate(x_opt, 1 - x_opt, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_opt

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = clip_and_rescale(perturbed)
        result = minimize(
            fun=objective,

            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        )
        if result.success:
            x_opt = clip_and_rescale(result.x)
            c5 = np.max(np.correlate(x_opt, 1 - x_opt, mode='full')) * dx

                best_c5 = c5
                best_h = x_opt

    return (best_h.tolist(), best_c5, n_points)
