import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = (n_points / 2.0) / 2.0
    target_sum_full = n_points / 2.0

    # Repair initial solution
    h_initial = np.clip(h_initial, 0, 1)
    scale = target_sum_full / max(1e-8, h_initial.sum())
    h_initial = np.clip(h_initial * scale, 0, 1)
    remaining = target_sum_full - h_initial.sum()
    if abs(remaining) > 1e-6:
        idx = np.unravel_index(np.argsort(h_initial, axis=None)[::-1], h_initial.shape)
        gain = np.minimum(remaining, 1 - h_initial[idx])
        h_initial[idx] += gain

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h.clip(0, 1)

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
    best_x = h_initial[:half_n].copy()
    best_val = objective(best_x)

    # Global optimization with adaptive DE
    np.random.seed(seed)
    de_start = time.time()
    de_popsize = 25
    de_maxiter = int((budget_s * 0.4) / 30)
    de_result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=de_maxiter, popsize=de_popsize, 
        mutation=(0.8, 1.2), recombination=0.9, 
        tol=1e-6, seed=seed, disp=False
    )
    if de_result.success and de_result.fun < best_val:
        best_x = np.clip(de_result.x, 0, 1)
        best_val = de_result.fun

    # Local optimization with COBYLA
    def perturb_and_optimize():
        while time.time() - start_time < budget_s - 5:
            frac = min(1.0, max(0.0, 1.0 - (time.time() - start_time)/budget_s))
            sample = best_x.copy()
            idx = np.random.choice(half_n, size=min(half_n, int(half_n * 0.4 * frac**0.5)), replace=False)
            noise = np.random.rand(len(idx)) * 0.3
            sample[idx] = np.clip(sample[idx] + (noise - 0.15), 0, 1)
            sample[idx] = np.clip(sample[idx] / sample.sum() * target_sum_half, 0, 1)
            try:
                result = minimize(
                    objective, sample, bounds=bounds, method='COBYLA',
                    constraints=cons, options={'maxfun': 250, 'ftol': 1e-8, 'eps': 1e-4}
                )
                if result.success:
                    new_x = np.clip(result.x, 0, 1)
                    if abs(new_x.sum() - target_sum_half) < 1e-6:
                        val = objective(new_x)
                        if val < best_val:
                            nonlocal best_x, best_val
                            best_x = new_x
                            best_val = val
            except:
                pass

    perturb_and_optimize()

    # Final refinement
    if time.time() - start_time < budget_s - 10:
        ref_x = best_x.copy()
        ref_x = np.clip(ref_x / max(1e-8, ref_x.sum()) * target_sum_half, 0, 1)
        refine_result = minimize(
            objective, ref_x, bounds=bounds, method='Nelder-Mead',
            constraints=cons, options={'maxiter': 100, 'xatol': 1e-6}
        )
        if refine_result.success:
            refine_x = np.clip(refine_result.x, 0, 1)
            final_val = objective(refine_x)
            if final_val < best_val:
                best_x = refine_x
                best_val = final_val

    h_full = create_full_h(best_x)
    h_full = np.clip(h_full, 0, 1)
    h_full = np.where(h_full < 1e-8, 0, h_full)
    h_full = np.where(h_full > 1 - 1e-8, 1, h_full)
    assert abs(h_full.sum() - target_sum_full) < 1e-3, (h_full.sum(), target_sum_full)
    return (h_full.tolist(), best_val, n_points)
