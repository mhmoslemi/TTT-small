        seed += 1
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.15 * (1 - frac ** 1.5)
        perturbed = best_h_half.copy()
        idxs = np.random.choice(half_n, size=min(half_n, 10), replace=False)
        perturbed[idxs] += np.random.normal(0, noise_scale, len(idxs))
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (target_sum_half / total) * perturbed
        perturbed = np.clip(perturbed, 0, 1)
        method_idx = int((time.time() - start_time) * 1000) % 4
        method = 'COBYLA' if method_idx == 0 else ('L-BFGS-B' if method_idx == 1 else 'SLSQP')
        result = minimize(
            fun=lambda x: np.max(np.correlate(create_full_h(x), 1 - create_full_h(x), 'full')) * dx,
            x0=perturbed,
            method=method,
            bounds=[(0, 1)] * half_n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
            options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            x = np.clip(result.x, 0, 1)
            total = np.sum(x)
            if total != 0:
                x = (target_sum_half / total) * x
            x = np.clip(x, 0, 1)
            c5 = np.max(np.correlate(create_full_h(x), 1 - create_full_h(x), 'full')) * dx
            if c5 < best_c5:
                best_h_half = x
                best_c5 = c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    total_sum = np.sum(h_full)
    assert abs(total_sum - n_points / 2.0) < 1e-6, f"Feasibility violation: {total_sum} vs {n_points/2.0}"
    return (h_full.tolist(), best_c5, n_points)
