from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def project_to_constraints(h, target_sum):

    if current_sum > 1e-6:
        h *= target_sum / current_sum
    else:

    h = np.clip(h, 0, 1)
