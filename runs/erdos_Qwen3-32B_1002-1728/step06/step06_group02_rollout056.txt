        sum_p = np.sum(proposed)
        if sum_p == 0:
            proposed = np.full(n_points, target_sum / n_points)
        else:
            proposed = proposed * (target_sum / sum_p)
        proposed = np.clip(proposed, 0, 1)
        new_c5 = evaluate_c5(proposed, dx)
        delta_c5 = new_c5 - current_c5

        # Metropolis criterion
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h[:] = proposed
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h[:] = current_h
                best_c5 = current_c5
        
        T *= cooling_rate

    # Final gradient-based optimization
    if time.time() - start_time < budget_s - 30:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def objective(x):
            x = np.clip(x, 0, 1)
            s = np.sum(x)
            if s == 0:
                x[:] = target_sum / n_points
            else:
                x = x * (target_sum / s)
            x = np.clip(x, 0, 1)
            return evaluate_c5(x, dx)
        res = minimize(objective, x0=best_h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200})
        if res.success:
            refined_h = res.x.copy()
            refined_h = np.clip(refined_h, 0, 1)
            s = np.sum(refined_h)
            if s != 0:
                refined_h = refined_h * (target_sum / s)
            else:
                refined_h[:] = target_sum / n_points
            refined_h = np.clip(refined_h, 0, 1)
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
