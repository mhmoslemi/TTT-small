        x = np.linspace(0, 2, n_points, endpoint=False)
        L = 2.0
        h_continuous = theta_vec[0]  # a0
        for n in range(1, N_terms + 1):
            a_idx = 1 + 2 * (n - 1)
            b_idx = a_idx + 1
            a_n = theta_vec[a_idx]
            b_n = theta_vec[b_idx]
            term = a_n * np.cos(n * np.pi * x / L) + b_n * np.sin(n * np.pi * x / L)
            h_continuous += term
        h_samples = 1.0 / (1.0 + np.exp(-h_continuous + 1e-8))  # Sigmoid with clip
        return np.clip(h_samples, 1e-8, 1 - 1e-8)

    def objective(theta_vec):
        h_eval = fourier_to_h(theta_vec)
        h_eval = h_eval * target_sum / h_eval.sum()
        corr = np.correlate(h_eval, 1 - h_eval, mode='full')
        return np.max(corr) * dx

    def constraint_func(theta_vec):
        h_eval = fourier_to_h(theta_vec)
        return h_eval.sum() - target_sum

    opt_result = minimize(
        objective,
        theta,
        method='SLSQP',
        jac=False,
        constraints={'type': 'eq', 'fun': constraint_func},
        bounds=[(-10, 10) for _ in range(theta_size)],
        options={
            'maxiter': 150,
            'ftol': 1e-8,
            'disp': False
        },
    )

    if opt_result.success:
        theta_opt = opt_result.x
        h_opt = fourier_to_h(theta_opt)
        h_opt = h_opt * target_sum / h_opt.sum()
        h_opt = np.clip(h_opt, 0, 1)
        new_c5 = calculate_c5(h_opt, dx)
        if new_c5 < best_c5:
            best_h, best_c5 = h_opt, new_c5

    # Final verification and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / best_h.sum())
    return (best_h.tolist(), best_c5, n_points)

def calculate_c5(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
