import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        if not np.isclose(np.sum(h), target_sum):
            h = (h / np.sum(h)) * target_sum
        return h

    h_initial = enforce_constraints(h_initial)

    # PSO parameters
    pop_size = 30
    max_iter_pso = 30
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    velocities_rate = 0.05  # Fraction of dx for velocity

    np.random.seed(seed)
    particles = np.zeros((pop_size, n_points))
    velocities = np.zeros((pop_size, n_points))  # As fraction of dx

    # Initialize with Latin Hypercube Sampling and perturbations
    # 50% from Latin Hypercube Sampling
    num_latin = pop_size // 2
    for i in range(num_latin):
        latin_vals = np.zeros(n_points)
        positions = np.random.permutation(n_points)
        for j in range(int(n_points / 4)):
            latin_vals[positions[j]] = 1.0
        particles[i] = enforce_constraints(latin_vals)

    # 50% from initial_h_values with perturbation
    for i in range(num_latin, pop_size):
        perturbation = np.random.normal(0, 0.3, n_points)  # 30% perturbation
        particles[i] = enforce_constraints(h_initial + perturbation)

    # Initialize velocities
    velocities = np.random.uniform(-velocities_rate, velocities_rate, (pop_size, n_points))

    # PSO objective
    def objective(x):
        x = enforce_constraints(x)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    # Initialize particles
    personal_best = particles.copy()
    personal_best_obj = np.array([objective(p) for p in particles])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = particles[global_best_idx].copy().copy()
    best_c5 = personal_best_obj[global_best_idx]

    # PSO loop
    for iter in range(max_iter_pso):
        if time.time() - start_time > budget_s - 400:
            break
        w = w_max - (w_max - w_min) * (iter / max_iter_pso)  # Linear decay

        for i in range(pop_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = enforce_constraints(particles[i])

        # Update personal best and global best
        current_obj = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if current_obj[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = current_obj[i]
        current_best_idx = np.argmin(personal_best_obj)
        if personal_best_obj[current_best_idx] < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = personal_best_obj[current_best_idx]

    # Local Grid Search around top 5 PSO candidates
    top_indices = np.argsort(personal_best_obj)[:5]
    for i in top_indices:
        if time.time() - start_time > budget_s - 200:
            break
        current = personal_best[i].copy()
        grid_size = 5
        for j in range(grid_size):
            noise = np.random.uniform(-0.1, 0.1, n_points)  # 10% neighborhood
            perturbed = current + noise
            trial = enforce_constraints(perturbed)
            trial_obj = objective(trial)
            if trial_obj < personal_best_obj[i]:
                personal_best[i] = trial.copy()
                personal_best_obj[i] = trial_obj
                if trial_obj < best_c5:
                    global_best = trial.copy()
                    best_c5 = trial_obj

    # Final optimization with SLSQP on top 3 refined candidates
    top_refined = np.argsort(personal_best_obj)[:3]
    for idx in top_refined:
        if time.time() - start_time > budget_s - 30:
            break
        def fun(x):
            x = enforce_constraints(x)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        def constraint_sum(x):
            return np.sum(x) - target_sum

        x0 = personal_best[idx].copy()
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            fun=fun,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            optimized_h = enforce_constraints(result.x)
            optimized_c5 = fun(optimized_h)
            if optimized_c5 < best_c5:
                best_c5 = optimized_c5
                global_best = optimized_h

    h_full = enforce_constraints(global_best)
    return (h_full.tolist(), best_c5, n_points)
