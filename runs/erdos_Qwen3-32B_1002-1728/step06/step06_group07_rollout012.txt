        scaled_h = np.clip(h_processed * scale, 0, 1)
        corr_vals = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return float(np.max(corr_vals * dx))

    result_slsqp = minimize(objective_ordinary, x0=final_h, method='SLSQP',
                          bounds=[(0.0, 1.0) for _ in range(n_points)],
                          options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    else:
        best_h = final_h
        best_c5 = best_c5_fft

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    sum_f = final_h.sum()
    if not np.isclose(sum_f, T, atol=1e-6):
        # If the sum is not correct, clip and scale
        final_h = np.clip(final_h, 0, 1)

        if sum_f != 0:

import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0
    x_grid = np.linspace(0, 2, n_points, endpoint=False)

    K = 5  # Number of Fourier terms for approximating h(x)

    # Compute initial Fourier coefficients from h_initial
    a_coeff = []
    b_coeff = []
    for k in range(1, K+1):
        sum_cos = np.sum([h_initial[i] * np.cos(np.pi * k * x_grid[i]) for i in range(n_points)])
        sum_sin = np.sum([h_initial[i] * np.sin(np.pi * k * x_grid[i]) for i in range(n_points)])
        a_coeff.append(sum_cos * dx)
        b_coeff.append(sum_sin * dx)

    initial_coeff_flat = np.array(a_coeff + b_coeff)

    def objective_fourier(coeff_concat):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            return float('inf')
        a_new = coeff_concat[:K]
        b_new = coeff_concat[K:]
        h_new = np.zeros(n_points)
        for i in range(n_points):
            x_i = x_grid[i]
            h_val = 0.5  # A0 fixed at 0.5
            for k in range(K):
                h_val += a_new[k] * np.cos(np.pi * (k+1) * x_i)
                h_val += b_new[k] * np.sin(np.pi * (k+1) * x_i)
            h_new[i] = h_val
        # Clip and scale
        clipped_h = np.clip(h_new, 0, 1)
        sum_clipped = np.sum(clipped_h)
        if sum_clipped == 0:
            return float('inf')
        scale = T / sum_clipped
        scaled_h = np.clip(clipped_h * scale, 0, 1)
        # Compute C₅
        corr_vals = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return float(np.max(corr_vals) * dx)

    bounds_fft = [(-0.5, 0.5)] * (2*K)
    best_c5 = float('inf')
    best_coeff = initial_coeff_flat.copy()

    try:
        # Fourier domain differential evolution
        de_result = differential_evolution(
            objective_fourier, bounds_fft, strategy='best1bin',
            maxiter=30, popsize=15, tol=1e-4, mutation=(0.5, 1),
            recombination=0.85, seed=seed, workers=1, disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            best_c5 = de_result.fun
            best_coeff = de_result.x
    except:
        pass  # Preserve original best_coeff

    # Construct final h from best Fourier coefficients
    h_fft = np.zeros(n_points)
    for i in range(n_points):
        x_i = x_grid[i]
        val = 0.5
        for k in range(K):
            val += best_coeff[k] * np.cos(np.pi * (k+1) * x_i)
            val += best_coeff[K + k] * np.sin(np.pi * (k+1) * x_i)
        h_fft[i] = val

    # Clip and scale final h
    h_fft = np.clip(h_fft, 0, 1)
    sum_h = np.sum(h_fft)
    if sum_h > 1e-6:
        h_fft = np.clip(h_fft * (T / sum_h), 0, 1)

    # Local refinement in time domain
    def time_domain_objective(h_candidate):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            return float('inf')
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = T / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return float(np.max(corr_vals) * dx)

    final_bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Apply L-BFGS-B for efficiency
    lbfgs_result = minimize(time_domain_objective, x0=h_fft, 
                           method='L-BFGS-B', bounds=final_bounds,
                           options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
    if lbfgs_result.success:
        final_h = lbfgs_result.x
        final_corr = lbfgs_result.fun
    else:
        final_h = h_fft
        final_corr = objective_fourier(best_coeff)

    # Final validation
    final_h = np.clip(final_h, 0, 1)
    sum_h = np.sum(final_h)
    if not np.isclose(sum_h, T, atol=1e-6):
        final_h = (final_h * (T / sum_h)).clip(0, 1)

    return (final_h.tolist(), final_corr, n_points)
