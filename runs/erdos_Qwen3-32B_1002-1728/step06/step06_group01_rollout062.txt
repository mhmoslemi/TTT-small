def adam_update(grad, theta, m, v, t, alpha=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    t += 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad**2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    theta -= alpha * m_hat / (np.sqrt(v_hat) + eps)
    return theta, m, v, t

import numpy as np
import time

def haar_transform(h):
    n = len(h)
    if n % 2 != 0:
        raise ValueError("Haar transform requires even-length input")
    m = n // 2
    ha = np.zeros(m)
    hd = np.zeros(m)
    for i in range(m):
        a = (h[2*i] + h[2*i + 1]) / np.sqrt(2)
        d = (h[2*i] - h[2*i + 1]) / np.sqrt(2)
        ha[i] = a
        hd[i] = d
    return np.concatenate([ha, hd])

def inverse_haar_transform(theta):
    m = len(theta) // 2
    n = 2 * m
    h = np.zeros(n)
    for i in range(m):
        a = theta[i]
        d = theta[m + i]
        h[2*i] = (a + d) / np.sqrt(2)
        h[2*i + 1] = (a - d) / np.sqrt(2)
    return h

def objective(theta, p, dx, n_points, lambda_l1):
    h = inverse_haar_transform(theta)
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)

        return float('inf')

    norm_p = (np.sum(corrs**p)) ** (1/p)

        theta_plus[i] += eps
        val_plus = objective_func(theta_plus, p, dx, n_points, lambda_l1)
        grad[i] = (val_plus - baseline) / eps
    return grad
