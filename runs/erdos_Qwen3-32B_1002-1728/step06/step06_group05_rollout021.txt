        return np.max(corr) * dx_coarse
    
    personal_best = particles.copy()
    personal_best_val = np.array([objective_coarse(p) for p in particles])
    best_idx = np.argmin(personal_best_val)
    global_best = particles[best_idx].copy()
    best_val = personal_best_val[best_idx]
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
        w = w_max - (w_max - w_min) * (iter / max_iter)
        for i in range(pop_size):
            r1, r2 = np.random.rand(n_coarse), np.random.rand(n_coarse)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best[i] - particles[i]) + 
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = particles[i].sum()
            if sum_p != 0:
                particles[i] = (particles[i] / sum_p) * target_sum_coarse
            particles[i] = np.clip(particles[i], 0, 1)
        currents = np.array([objective_coarse(p) for p in particles])
        for i in range(pop_size):
            if currents[i] < personal_best_val[i]:
                personal_best[i] = particles[i].copy()
                personal_best_val[i] = currents[i]
        current_best_idx = np.argmin(personal_best_val)
        if personal_best_val[current_best_idx] < best_val:
            global_best = personal_best[current_best_idx].copy()
            best_val = personal_best_val[current_best_idx]
    
    # Upscale to fine grid
    h_fine = np.repeat(global_best, 2).copy()
    
    # SLSQP optimization on fine grid
    def objective_fine(x):
        x = np.clip(x, 0, 1)
        sum_x = x.sum()
        if sum_x != 0:
            x = (x / sum_x) * target_sum
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return x.sum() - target_sum
    
    x0 = h_fine.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    best_c5 = objective_fine(x0)
    
    result = minimize(
        fun=objective_fine,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        h_opt = np.clip(result.x, 0, 1)
        sum_opt = h_opt.sum()
        if sum_opt != 0:
            h_opt = (h_opt / sum_opt) * target_sum
        h_opt = np.clip(h_opt, 0, 1)
        c5_opt = objective_fine(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            x0 = h_opt
    
    h_full = np.clip(x0, 0, 1)
    sum_h = h_full.sum()
    if sum_h != 0:
        h_full = (h_full / sum_h) * target_sum
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
