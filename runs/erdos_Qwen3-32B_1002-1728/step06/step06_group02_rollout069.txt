            best_h = h_candidate.copy()
            h = h_candidate.copy()
        else:
            step_size *= 0.5

        # COBYLA refinement
        if iter % 200 == 0:
            def objective(x):
                return evaluate_c5(x, dx)
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            res = minimize(
                objective,
                x0=h.copy(),
                method='COBYLA',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 200, 'tol': 1e-8}
            )
            if res.success:
                refined = res.x
                refined = np.clip(refined, 0, 1)
                current_sum_r = np.sum(refined)
                if current_sum_r != 0:
                    refined *= target_sum / current_sum_r
                else:
                    refined[:] = target_sum / n_points
                refined = np.clip(refined, 0, 1)
                refined_c5 = evaluate_c5(refined, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
