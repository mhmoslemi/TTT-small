from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # 35.0

    # Generate Frank-Zadoff-like initial sequence
    N_frank = 64  # 8^2
    s_frank = np.zeros(N_frank)
    for i in range(N_frank):
        mod_val = (2 * i + 1) % N_frank
        angle = np.pi / N_frank * mod_val
        s_frank[i] = np.cos(angle)

    # Pad to n_points by linear interpolation between end and start
    h_frank = np.zeros(n_points)
    h_frank[:N_frank] = s_frank
    for i in range(N_frank, n_points):
        t = (i - N_frank + 1) / (n_points - N_frank + 1)
        h_frank[i] = s_frank[-1] * (1 - t) + s_frank[0] * t

    # Normalize to [0, 1] and ensure exact integral constraint
    h_initial = (h_frank + 1.0) / 2.0
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial = (h_initial / np.sum(h_initial)) * target_sum
    h_initial = np.clip(h_initial, 0.0, 1.0)

    # Local optimization with constraint
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0.0, 1.0)
        x = (x / np.sum(x)) * target_sum
        x = np.clip(x, 0.0, 1.0)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
    )

    if result.success:
        optimized_h = np.clip(result.x, 0.0, 1.0)
        sum_opt = np.sum(optimized_h)
        if abs(sum_opt - target_sum) > 1e-6:
            optimized_h = (optimized_h / sum_opt) * target_sum
