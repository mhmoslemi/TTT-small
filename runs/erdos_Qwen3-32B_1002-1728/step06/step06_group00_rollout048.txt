            maxiter=40,
            polish=False
        )
        if result_de.success:
            candidate = repair_h(result_de.x)
            c5_candidate = np.max(np.correlate(candidate, 1 - candidate, mode='full')) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()
    except:
        pass
    
    remaining_time = budget_s - (time.time() - start_time)
    
    # Phase 2: Local optimization with multiple restarts
    for _ in range(int(remaining_time / 10) if remaining_time > 0 else 0):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Perturb starting point
        idx = np.random.choice(n_points, min(15, n_points), replace=False)
        perturbed = best_h.copy()
        perturbed[idx] += np.random.normal(0, 0.15, len(idx))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        
        local_result = minimize(
            objective, perturbed,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'ftol': 1e-10, 'gtol': 1e-7, 'maxiter': 300}
        )
        if local_result.success:
            h_candidate = repair_h(local_result.x)
            c5_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    
    # Final constraint verification
    best_h = repair_h(best_h)
    return (best_h.tolist(), best_c5, n_points)
