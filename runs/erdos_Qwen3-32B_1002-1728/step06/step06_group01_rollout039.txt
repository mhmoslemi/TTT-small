import numpy as np
from scipy.optimize import minimize
from scipy.fft import dct, idct
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values).astype(float)
    n = len(h_initial)
    m = n // 2
    dx = 2.0 / n
    target_sum = n / 2.0

    def create_full_h(h_half):
        h = np.zeros(n)
        h[:m] = h_half
        h[-m:] = h_half[::-1]
        return h

    h_half_initial = h_initial[:m]
    coeffs_initial = dct(h_half_initial, norm='ortho')

    def objective(coeffs):
        h_half = idct(coeffs, norm='ortho', n=m)
        h_full = create_full_h(h_half)
        h_clipped = np.clip(h_full, 0, 1)
        total = np.sum(h_clipped)
        if total != 0:
            h_scaled = h_clipped * (target_sum / total)
        else:
            h_scaled = h_clipped
        h_final = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        return np.max(corr) * dx

    bounds = [(-100.0, 100.0) for _ in range(m)]
    result = minimize(objective, coeffs_initial, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-6})
    coeffs_final = result.x
    h_half_final = idct(coeffs_final, norm='ortho', n=m)
    h_full = create_full_h(h_half_final)
    h_clipped = np.clip(h_full, 0, 1)
    total = np.sum(h_clipped)
    h_scaled = h_clipped * (target_sum / total) if total != 0 else h_clipped
    h_final = np.clip(h_scaled, 0, 1)
    corr_vals = np.correlate(h_final, 1 - h_final, mode='full')
    c5 = np.max(corr_vals) * dx

    # Perturbation and refinement within remaining time
    elapsed = time.time()
    while time.time() - elapsed < budget_s - 30:
        perturb = np.random.normal(0, 0.1, size=len(coeffs_final))
        coeffs_new = coeffs_final + perturb
        coeffs_new = np.clip(coeffs_new, -100, 100)
        res = minimize(objective, coeffs_new, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            c_new = res.x
            h_new = create_full_h(idct(c_new, norm='ortho', n=m))
            h_new_clipped = np.clip(h_new, 0, 1)
            total_new = np.sum(h_new_clipped)
            if total_new != 0:
                h_new_scaled = h_new_clipped * (target_sum / total_new)

                h_new_scaled = h_new_clipped
            h_new_final = np.clip(h_new_scaled, 0, 1)
            corr_new = np.correlate(h_new_final, 1 - h_new_final, mode='full') * dx
            if np.max(corr_new) < c5:
                coeffs_final = c_new
                h_final = h_new_final
                c5 = np.max(corr_new)
        elapsed = time.time()

    return (h_final.tolist(), c5, n)
