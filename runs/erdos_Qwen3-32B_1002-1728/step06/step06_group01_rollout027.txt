                corrs = np.correlate(h_final, 1 - h_final, mode='full')
                l_p = np.linalg.norm(corrs, ord=p)
                reg = lambda_reg * np.linalg.norm(theta, ord=1)
                return l_p * dx + reg

            bounds_theta = [(-2.0, 2.0) for _ in range(n_points)]
            result = minimize(
                fun=objective,
                x0=theta_initial,
                method='L-BFGS-B',
                bounds=bounds_theta,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if result.success:
                theta_opt = result.x
                h_opt = fft.idct(theta_opt, type=3, norm='ortho')
                h_clipped = np.clip(h_opt, 0, 1)
                sum_current = np.sum(h_clipped)
                if sum_current < 1e-8:
                    h_scaled = h_clipped.copy()
                else:
                    h_scaled = h_clipped * (sum_target / sum_current)
                h_final = np.clip(h_scaled, 0, 1)
                corrs = np.correlate(h_final, 1 - h_final, mode='full')
                max_corr = np.max(corrs) * dx
                if max_corr < best_c5:
                    best_c5 = max_corr
                    best_h = h_final

    return (best_h.tolist(), best_c5, n_points)
