import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair(vec):
        vec_clipped = np.clip(vec, 0, 1)
        total = vec_clipped.sum()
        if total > 1e-10:
            return vec_clipped * target_sum / total
        return np.full(n_points, target_sum / n_points)

    def calculate_c5(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    h_repaired = repair(h_initial)
    best_c5 = calculate_c5(h_repaired)
    best_h = h_repaired.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    remaining_time = budget_s * 0.9  # Reserve time for local search
    de_maxiter = 100
    de_pop_size = 20

    if time.time() - start_time < remaining_time:
        def objective(x):
            return calculate_c5(repair(x))
        result_de = differential_evolution(
            objective, bounds,
            strategy='best1bin',
            popsize=de_pop_size,
            maxiter=de_maxiter,
            mutation=(0.5, 1.0),
            recombination=0.9,
            polish=False,
            workers=1,
            seed=seed,
            update_interval=10
        )
        candidate = repair(result_de.x)
        current_c5 = calculate_c5(candidate)
        if current_c5 < best_c5:
            best_h[:] = candidate
            best_c5 = current_c5

    def slsqp_objective(x):
        return calculate_c5(x)

    if time.time() - start_time < budget_s:
        result_sl = minimize(
            slsqp_objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: x.sum() - target_sum},
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        if result_sl.success:
            candidate_sl = np.clip(result_sl.x, 0, 1)
            if abs(candidate_sl.sum() - target_sum) > 1e-10:
                candidate_sl = repair(candidate_sl)
            current_c5 = calculate_c5(candidate_sl)
            if current_c5 < best_c5:
                best_h[:] = candidate_sl
                best_c5 = current_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
