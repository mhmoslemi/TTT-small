import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_de(x_flat):
        h = np.clip(x_flat, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return 1.0e6
        h_scaled = (h / current_sum) * target_sum
        h_clipped = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0_flat = h_full_initial.copy()

    # Global search with DE, tracking time
    remaining_time = budget_s * 0.7
    result_de = differential_evolution(
        objective_de, bounds,
        strategy='best1bin',
        maxiter=20,
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        workers=1,
        disp=False
    )
    de_duration = time.time() - start_time
    remaining_time = budget_s - de_duration

    h_best = np.clip(result_de.x, 0, 1)
    current_sum = np.sum(h_best)
    if current_sum != 0:
        h_best = (h_best / current_sum) * target_sum
    h_best = np.clip(h_best, 0, 1)
    best_c5 = objective_de(h_best)

    if remaining_time < 10:
        return (h_best.tolist(), best_c5, n_points)

    # Local refinement with SLSQP
    def objective_local(x_flat):
        h = np.clip(x_flat, 0, 1)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    cons_local = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_local = [(0.0, 1.0)] * n_points

    result_local = minimize(
        objective_local,
        x0=h_best,
        method='SLSQP',
        bounds=bounds_local,
        constraints=cons_local,
        options={'maxiter': 200, 'ftol': 1e-10}
    )
    if result_local.success:
        h_local = np.clip(result_local.x, 0, 1)
        corr_vals = np.correlate(h_local, 1 - h_local, mode='full')
        c5_local = np.max(corr_vals) * dx
        if c5_local < best_c5:
            best_c5 = c5_local
            h_best = h_local

    # Final validation
    h_full = np.clip(h_best, 0, 1)
    h_sum = np.sum(h_full)
    if abs(h_sum - target_sum) > 1e-6:
        h_full = (h_full / h_sum) * target_sum
    h_valid = np.clip(h_full, 0, 1)

    return (h_valid.tolist(), best_c5, n_points)
