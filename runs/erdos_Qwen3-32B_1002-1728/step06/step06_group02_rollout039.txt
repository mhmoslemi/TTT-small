            noise = np.random.uniform(-max_perturbation, max_perturbation)
            proposed_h[idx] += noise
            proposed_h[idx] = np.clip(proposed_h[idx], 0, 1)
        # Normalize and clip
        current_sum_p = np.sum(proposed_h)
        if current_sum_p != 0:
            proposed_h *= target_sum / current_sum_p
        else:
            proposed_h[:] = target_sum / n_points
        proposed_h = np.clip(proposed_h, 0, 1)
        # Evaluate new solution
        new_c5 = evaluate_c5(proposed_h, dx)
        # Accept based on SA rules
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            h = proposed_h.copy()
            current_c5 = new_c5
        # Update global best
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        # Cool down
        temperature *= cooling_rate

    # Final refinement with constrained optimization
    if time.time() - start_time < budget_s - 10:
        def objective(x):
            return evaluate_c5(x, dx)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(
            objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if res.success:
            refined_h = np.clip(res.x, 0, 1)
            current_sum_r = np.sum(refined_h)
            if current_sum_r != 0:
                refined_h *= target_sum / current_sum_r
            else:
                refined_h[:] = target_sum / n_points
            refined_h = np.clip(refined_h, 0, 1)
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
