import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h_clipped)
    if abs(sum_h) > 1e-10:
        return h_clipped * target_sum / sum_h
    else:
        return np.full_like(h_clipped, target_sum / len(h_clipped))

def objective_augmented(h, dx, target_sum, λ, ρ):
    c5 = calculate_c5(h, dx)
    sum_h = np.sum(h)
    constraint_violation = sum_h - target_sum
    lagrangian = c5 + λ * constraint_violation + 0.5 * ρ * (constraint_violation ** 2)
    return lagrangian

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = clip_and_rescale(h_full_initial, target_sum)
    best_h = h.copy()
    best_c5 = calculate_c5(best_h, dx)
    np.random.seed(seed)

    λ = 0.0
    ρ = 1.0
    max_outer_iterations = 50
    ρ_increase = 1.2
    ρ_decrease = 0.9
    threshold = 1e-4

    for iter_outer in range(max_outer_iterations):
        if time.time() - start_time > budget_s - 30:
            break

        result = minimize(
            objective_augmented,
            x0=h,

            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False},

        current_sum = np.sum(h_opt)
        constraint_violation = current_sum - target_sum
