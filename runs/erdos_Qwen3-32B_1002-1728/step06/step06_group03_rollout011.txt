    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    local_result = minimize(
        lambda x: evaluate_objective_global(x, dx, target_sum),
        x0=global_best_position,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    if local_result.success:
        refined = local_result.x
        refined = np.clip(refined, 0, 1)
        x_sum = np.sum(refined)
        if x_sum != 0:
            refined = (refined / x_sum) * target_sum
        refined = np.clip(refined, 0, 1)
        final_score = evaluate_objective_global(refined, dx, target_sum)
        if final_score < global_best_score:
            global_best_position = refined
            global_best_score = final_score

    # Final validation and adjustment
    h_full = global_best_position
    h_full = np.clip(h_full, 0, 1)
    total = np.sum(h_full)
    if abs(total - target_sum) > 1e-6:
        h_full = (h_full / total) * target_sum
    h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), global_best_score, n_points)
