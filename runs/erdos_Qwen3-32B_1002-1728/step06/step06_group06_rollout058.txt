import numpy as np
from scipy.optimize import minimize
import time

def project(h, target_sum, n_points, max_iter=5):
    h_proj = np.copy(h)
    for _ in range(max_iter):
        h_clipped = np.clip(h_proj, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            h_proj = np.zeros_like(h_proj)
        else:
            scale = target_sum / current_sum
            h_proj = h_clipped * scale
    return h_proj

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        h_valid = project(h, target_sum, n_points)
        corr_vals = np.correlate(h_valid, 1 - h_valid, mode='full')
        return np.max(corr_vals) * dx

    best_global = project(h_initial, target_sum, n_points)
    best_global_c5 = evaluate(best_global)

    n_particles = 40
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter_pso = 50

    np.random.seed(seed)
    particles = [project(h_initial + np.random.normal(0, 0.1, n_points), target_sum, n_points) for _ in range(n_particles)]
    velocities = [np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_c5 = [evaluate(p) for p in particles]
    best_global = personal_best[np.argmin(personal_best_c5)]
    best_global_c5 = min(personal_best_c5)

    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s - 60:
            break
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best[i] - particles[i]) +
                            social * r2 * (best_global - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = project(new_pos, target_sum, n_points, max_iter=5)
            particles[i] = new_pos
            c5_new = evaluate(new_pos)
            if c5_new < personal_best_c5[i]:
                personal_best[i] = new_pos.copy()
                personal_best_c5[i] = c5_new
                if c5_new < best_global_c5:
                    best_global = new_pos.copy()
                    best_global_c5 = c5_new

    top_candidates = [personal_best[i] for i in np.argsort(personal_best_c5)[:10]]

    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 30:
