    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is feasible
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-6:
        h_initial = h_initial * (target_sum / current_sum)
        h_initial = np.clip(h_initial, 0, 1)

    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Gradient descent parameters
    max_iterations = 100
    step_size = 0.1
    epsilon = 1e-5
    iterations = 0

    while iterations < max_iterations and time.time() - start_time < budget_s - 30:
        # Find current max C_k and its index
        h_clipped = np.clip(best_h, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        max_index = np.argmax(corr_vals)
        current_max_c5 = corr_vals[max_index] * dx

        # Compute gradient at this max_index
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward step
            h_forward = best_h.copy()
            h_forward[i] += epsilon
            h_forward = np.clip(h_forward, 0, 1)
            sum_forward = np.sum(h_forward)
            if sum_forward != 0:
                h_forward *= target_sum / sum_forward
            h_forward = np.clip(h_forward, 0, 1)
            corr_forward = np.correlate(h_forward, 1 - h_forward, mode='full')
            val_forward = corr_forward[max_index] * dx

            # Backward step
            h_backward = best_h.copy()
            h_backward[i] -= epsilon
            h_backward = np.clip(h_backward, 0, 1)
            sum_backward = np.sum(h_backward)
            if sum_backward != 0:
                h_backward *= target_sum / sum_backward
            h_backward = np.clip(h_backward, 0, 1)
            corr_backward = np.correlate(h_backward, 1 - h_backward, mode='full')
            val_backward = corr_backward[max_index] * dx

            grad[i] = (val_forward - val_backward) / (2 * epsilon)
        
        # Normalize and take step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad /= grad_norm
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h *= target_sum / sum_new
        else:
            new_h[:] = target_sum / n_points
        new_h = np.clip(new_h, 0, 1)

        # Check improvement
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        # Else: reduce step size
        elif step_size > 1e-4:
            step_size *= 0.5
        else:
            break  # stuck in local min with tiny steps

        iterations += 1

        # Periodic refinement with COBYLA
        if iterations % 10 == 0:
            def objective(x):
                return evaluate_c5(x, dx)
            bounds = [(0.0, 1.0)] * n_points
            res = minimize(
                objective,
                x0=best_h.copy(),
                method='COBYLA',
                bounds=bounds,
                options={'maxiter': 100, 'tol': 1e-6}
            )
            if res.success:
                refined_h = res.x.copy()
                refined_h = np.clip(refined_h, 0, 1)
                sum_r = np.sum(refined_h)
                if sum_r != 0:
                    refined_h *= target_sum / sum_r
                else:
                    refined_h[:] = target_sum / n_points
                refined_h = np.clip(refined_h, 0, 1)
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h

    # Final feasibility check
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
