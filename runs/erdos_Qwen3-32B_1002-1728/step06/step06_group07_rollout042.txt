    population = np.array(population)

    best_c5 = float('inf')
    best_ind = h_binary_initial

    # GA loop
    max_generations = 200
    gen = 0
    while time.time() - start_time < budget_s - 300 and gen < max_generations:
        gen += 1
        scores = [objective_binary(ind) for ind in population]
        # Update best
        current_best_idx = np.argmin(scores)
        if scores[current_best_idx] < best_c5:
            best_c5 = scores[current_best_idx]
            best_ind = population[current_best_idx].copy()

        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            p1 = select_parent(population, scores)
            p2 = select_parent(population, scores)
            c1, c2 = crossover(p1, p2)
            c1 = repair(c1, T_int)
            c2 = repair(c2, T_int)
            c1 = mutate(c1)
            c2 = mutate(c2)
            new_population.extend([c1, c2])
        population = np.array(new_population)

    # Convert best binary solution to continuous
    h_refined = best_ind.astype(np.float64)

    # Refinement with SLSQP
    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h_candidate)
        scale = T / total
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def final_objective(h):
        h_processed = clip_and_scale(h)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    result = minimize(final_objective, h_refined, method='SLSQP',
                      bounds=[(0, 1) for _ in range(n_points)],
                      options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})
    if result.success:
        h_new = clip_and_scale(result.x)
        c5_new = final_objective(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_ind = h_new

    # Final validation
    best_h = clip_and_scale(best_ind)
    assert abs(np.sum(best_h) - T) < 1e-6, "Sum constraint violated"
    return (best_h.tolist(), best_c5, n_points)
