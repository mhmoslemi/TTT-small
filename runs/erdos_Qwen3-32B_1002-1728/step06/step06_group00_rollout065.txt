            scale_factor = 0.0
        else:
            scale_factor = target_sum_full / total
        h_scaled = np.clip(h * scale_factor, 0, 1)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr_vals) * dx

    # Phase 1: Adaptive global search
    bounds_de = [(0.0, 1.0)] * n_points
    np.random.seed(seed)
    de_budget = budget_s * 0.35
    popsize = 10
    try:
        de_result = differential_evolution(
            de_objective, bounds_de, strategy='best1bin',
            popsize=popsize, tol=1e-5, mutation=(0.5, 1.5),
            recombination=0.9, seed=seed, workers=1, disp=False
        )
        if de_result.success:
            best_h = np.clip(de_result.x, 0, 1)
            best_c5 = de_result.fun
    except:
        pass

    # Phase 2: Local optimization with gradient-based refinement
    def local_objective(x):
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    bounds_local = list(zip(np.zeros(n_points), np.ones(n_points)))
    cons_local = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_full})
    iter_count = 0
    while time.time() - start_time < budget_s - 10:
        # Perturb current solution
        perturb_idx = np.random.choice(n_points, min(8, n_points), replace=False)
        x0 = np.clip(best_h + np.random.normal(0, 0.02, n_points), 0, 1)
        x0_sum = x0.sum()
        if x0_sum != 0:
            x0 = (x0 / x0_sum) * target_sum_full
        x0 = np.clip(x0, 0, 1)
        
        # Local phase with constraint projection
        try:
            result = minimize(
                local_objective, x0=x0, method='SLSQP',
                bounds=bounds_local, constraints=cons_local,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if result.success and result.fun < best_c5:
                best_h[:] = np.clip(result.x, 0, 1)
                best_c5 = result.fun
        except:
            pass
        iter_count += 1

    # Final constraint enforcement
    best_sum = best_h.sum()
    if abs(best_sum - target_sum_full) > 1e-6:
        best_h = np.clip(best_h / best_sum * target_sum_full, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
