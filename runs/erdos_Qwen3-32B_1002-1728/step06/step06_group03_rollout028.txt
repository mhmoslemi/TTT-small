    target_half_sum = target_sum / 2.0
    np.random.seed(seed)

    n_particles = 30
    positions = []
    velocities = []

    # Initialize particles around the initial guess
    half_initial = h_full_initial[:half_n]
    if half_n == 0:
        half_initial = h_full_initial

    # Initialize first particle as the initial half
    positions.append(half_initial.copy())
    velocities.append(np.zeros(half_n))

    # Initialize remaining particles with random perturbations
    for i in range(1, n_particles):
        noise = np.random.uniform(-0.1, 0.1, size=half_n)
        new_pos = np.clip(half_initial + noise, 0, 1)
        sum_new = np.sum(new_pos)
        if sum_new > 1e-10:
            new_pos = (new_pos / sum_new) * target_half_sum
        new_pos = np.clip(new_pos, 0, 1)
        positions.append(new_pos)
        velocities.append(np.zeros(half_n))

    positions = np.array(positions)
    velocities = np.array(velocities)

    # Objective function for PSO
    def pso_objective(pos):
        h_full = create_full_h(pos, n_points)
        h_clipped = np.clip(h_full, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped > 1e-10:
            h_scaled = h_clipped * (target_sum / sum_clipped)
        else:
            h_scaled = h_clipped
        h_final = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_final, 1 - h_final, mode='full')
        return np.max(corr_vals) * dx

    # Initialize personal bests and global best
    personal_best = positions.copy()
    personal_best_scores = np.array([pso_objective(pos) for pos in positions])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_pos = positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # PSO parameters
    w_inertia = 0.7
    c1 = 1.2
    c2 = 1.2
    velocity_min = -0.1
    velocity_max = 0.1

    # PSO main loop
    while time.time() - start_time < 0.6 * budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = (w_inertia * velocities[i] +
                          c1 * r1 * (personal_best[i] - positions[i]) +
                          c2 * r2 * (global_best_pos - positions[i]))
            vel_update = np.clip(vel_update, velocity_min, velocity_max)
            velocities[i] = vel_update
            positions[i] += vel_update

            # Clip and normalize position
            positions[i] = np.clip(positions[i], 0, 1)
            pos_sum = np.sum(positions[i])
            if pos_sum > 1e-10:
                positions[i] = (positions[i] / pos_sum) * target_half_sum
            positions[i] = np.clip(positions[i], 0, 1)

            # Evaluate fitness
            score = pso_objective(positions[i])
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = positions[i].copy()

            if score < global_best_score:
                global_best_score = score
                global_best_pos = positions[i].copy()

        if time.time() - start_time >= 0.9 * budget_s:
            break

    # Final fitness evaluation of all particles
    personal_best_scores = np.array([pso_objective(pos) for pos in positions])
    if np.min(personal_best_scores) < global_best_score:
        best_idx = np.argmin(personal_best_scores)
        global_best_score = personal_best_scores[best_idx]
        global_best_pos = positions[best_idx].copy()

    # Local refinement with SLSQP
    def refine_objective(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum > 1e-10:
            x = (x / x_sum) * target_half_sum
        x = np.clip(x, 0, 1)
        return pso_objective(x)

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons_refine = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_half_sum}]
    x0 = global_best_pos.copy()
    x0_sum = np.sum(x0)
    if x0_sum > 1e-10:
        x0 = (x0 / x0_sum) * target_half_sum
    x0 = np.clip(x0, 0, 1)

    remaining_time = budget_s - (time.time() - start_time)

    try:
        result = minimize(
            refine_objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_refine,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            refined_pos = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_pos)
            if refined_sum > 1e-10:
                refined_pos = (refined_pos / refined_sum) * target_half_sum
            refined_pos = np.clip(refined_pos, 0, 1)
            refined_score = refine_objective(refined_pos)
            if refined_score < global_best_score:
                global_best_pos = refined_pos
                global_best_score = refined_score
    except:
        pass

    # Final full h construction
    h_full = create_full_h(global_best_pos, n_points)
    h_clipped = np.clip(h_full, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum > 1e-10:
        h_final = (h_clipped / h_sum) * target_sum
    else:
        h_final = h_clipped
    h_final = np.clip(h_final, 0, 1)

    return (h_final.tolist(), global_best_score, n_points)
