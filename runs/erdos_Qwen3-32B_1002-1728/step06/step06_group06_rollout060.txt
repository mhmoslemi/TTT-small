import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h.copy()
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    np.random.seed(seed)

            method='SLSQP',
            bounds=bounds,
