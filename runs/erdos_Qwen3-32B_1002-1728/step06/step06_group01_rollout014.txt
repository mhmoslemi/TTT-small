import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n // 2

    # Binary initialization
    top_idx = np.argsort(h_init)[::-1][:target_sum]
    bin_h = np.zeros(n)
    bin_h[top_idx] = 1.0

    def eval_overlap(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    best_h = bin_h.copy()
    best_c5 = eval_overlap(best_h)

    # Binary local optimization
    for _ in range(1000):
        if time.time() - start_time > budget_s - 120:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        mid = len(corr) // 2
        max_k = np.argmax(corr) - mid
        candidates = []
        for i in range(n):
            if best_h[i] == 1:
                j = (i + max_k) % n
                if 0 <= j < n and best_h[j] == 0:
                    candidates.append((i, j))
        if candidates:
            i, j = candidates[np.random.choice(len(candidates))]
            best_h[i], best_h[j] = best_h[j], best_h[i]
        if eval_overlap(best_h) < best_c5:
            best_c5 = eval_overlap(best_h)
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]  # revert on no improvement

    # Continuous optimization
    def objective(x, is_binary=False):
        x = np.clip(x, 0, 1)
        if is_binary or np.isclose(np.sum(x), target_sum):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        else:
            return float('inf')

    # Differential evolution with sum constraint
    bounds = [(0, 1) for _ in range(n)]
    def de_penalized(x):
        x_clip = np.clip(x, 0, 1)
        if abs(np.sum(x_clip) - target_sum) > 1e-6:
            return 1e6
        return objective(x_clip)

    if time.time() - start_time < budget_s - 90:
        result = differential_evolution(
            de_penalized, bounds, seed=seed, strategy='best1bin',
            maxiter=10, popsize=10, mutation=(0.7, 1.0), recombination=0.9,
            workers=1, disp=False
        )
        if result.success:
            cont_opt_h = np.clip(result.x, 0, 1)
            best_c5 = min(best_c5, objective(cont_opt_h))

    # Local refinement
    def obj_min(x):
        return objective(x)

    if time.time() - start_time < budget_s - 60:
        for _ in range(3):
            pert = np.random.RandomState(seed).uniform(-0.1, 0.1, n)
            x0 = np.clip(best_h + pert, 0, 1)
            x0 = (x0 / np.sum(x0)) * target_sum
            x0 = np.clip(x0, 0, 1)
            res = minimize(obj_min, x0, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h = np.clip(res.x, 0, 1)
                best_c5 = min(best_c5, objective(h, is_binary=False))

    final_h = best_h if best_c5 < 0.38085 else h_init
    return (final_h.tolist(), best_c5, n)
