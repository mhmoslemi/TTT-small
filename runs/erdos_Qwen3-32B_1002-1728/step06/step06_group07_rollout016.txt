            parent2_idx = candidates[np.argmin([fitness[i] for i in candidates])]
            parent1, parent2 = population[parent1_idx], population[parent2_idx]

            child1 = crossover(parent1, parent2, T)
            child1 = mutate(child1, mutation_rate=0.05)
            child2 = crossover(parent2, parent1, T)
            child2 = mutate(child2, mutation_rate=0.05)
            new_population.extend([child1, child2])
        new_population = elite + new_population[:pop_size - len(elite)]
        population = new_population

    # Final binary to continuous
    h_best = clip_and_scale(best_binary.astype(float))
    h_best_scaled = h_best
    assert abs(h_best_scaled.sum() - T) < 1e-6, "Constraint violated"

    # Local refine
    best_h = h_best_scaled
    best_c5 = objective(best_h)
    for _ in range(3 if time.time() - start_time < budget_s - 100 else 1):
        res = minimize(objective, x0=best_h, method='SLSQP', bounds=[(0, 1)] * n_points,
                       options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})
        if res.success:
            new_h = clip_and_scale(res.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    final_h = clip_and_scale(best_h)
    return (final_h.tolist(), best_c5, n_points)
