import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    np.random.seed(seed)
    n_perturb_initial = 30
    n_refine = 3  # per top candidate

    # Generate initial perturbed candidates
    perturbed_candidates = []
    for _ in range(n_perturb_initial):
        pert = h_initial + np.random.normal(0, 0.1, n_points)
        pert = clip_and_rescale(pert)
        perturbed_candidates.append(pert)

    best_global = h_initial.copy()
    best_c5 = np.max(np.correlate(best_global, 1 - best_global, mode='full')) * dx

    def optimize(x0):
        def obj(x):
            x = np.reshape(x, -1)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx
        cons = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum
        })
        bounds = [(0, 1)] * n_points
        result = minimize(
            obj, x0=x0, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 200, 'gtol': 1e-8, 'disp': False}
        )
        if not result.success:
            return x0, np.max(np.correlate(x0, 1 - x0, mode='full')) * dx
        x_opt = clip_and_rescale(result.x)
        c5 = np.max(np.correlate(x_opt, 1 - x_opt, mode='full')) * dx
        return x_opt, c5

    # Optimize initial perturbed candidates
    for cand in perturbed_candidates:
        if time.time() - start_time > budget_s - 60:
            break
        x_opt, c5 = optimize(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_global = x_opt.copy()

    # Sort and select top candidates for refinement
    all_candidates = [(optimize(cand)[0], np.max(np.correlate(optimize(cand)[0], 1 - optimize(cand)[0], mode='full')) * dx) for cand in perturbed_candidates]
    all_candidates.sort(key=lambda x: x[1])
    top_candidates = all_candidates[:5]

    # Generate refined perturbations from top candidates
    for x_top, _ in top_candidates:
        for _ in range(n_refine):
            if time.time() - start_time > budget_s - 10:
                break
            # Generate a new perturbation with higher noise
            pert = x_top + np.random.normal(0, 0.2, n_points)
            pert = clip_and_rescale(pert)
            x_opt, c5 = optimize(pert)
            if c5 < best_c5:
                best_c5 = c5
                best_global = x_opt.copy()

    return (best_global.tolist(), best_c5, n_points)
