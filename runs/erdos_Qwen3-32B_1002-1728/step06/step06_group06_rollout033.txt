import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scale = target_sum / sum_clipped
        h_rescaled = h_clipped * scale
        h_rescaled = np.clip(h_rescaled, 0, 1)
        if abs(np.sum(h_rescaled) - target_sum) > 1e-8:
            h_rescaled = h_clipped * target_sum / sum_clipped
            h_rescaled = np.clip(h_rescaled, 0, 1)
        return h_rescaled

    np.random.seed(seed)
    n_candidates = 20
    candidates = []
    for _ in range(n_candidates):
        pert = h_initial + np.random.normal(0, 0.1, n_points)
        pert = clip_and_rescale(pert)
        candidates.append(pert)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for h in candidates:
        if time.time() - start_time > budget_s - 30:
            break
        current_h = h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 30:
                break
            corr_vals = np.correlate(current_h, 1 - current_h, mode='full')
            idx = np.argmax(corr_vals)
            k_max = idx
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k_max) % n_points
                grad[i] += (1 - current_h[j]) - current_h[i]
            grad *= dx
            current_h -= 0.1 * grad  # Learning rate
            current_h = clip_and_rescale(current_h)
        test_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if test_c5 < best_c5:
            best_c5 = test_c5
            best_h = current_h.copy()

    return (best_h.tolist(), best_c5, n_points)
