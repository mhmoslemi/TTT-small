import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        return h_scaled
    
    h_initial = clip_and_rescale(h_initial)
    
    best_h = h_initial.copy()
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr_vals) * dx
    
    np.random.seed(seed)
    
    # Generate diverse initial candidates
    num_candidates = 30
    perturbation_scale = 0.15
    candidates = []
    for _ in range(num_candidates):
        perturbed = h_initial + np.random.normal(0, perturbation_scale, n_points)
        perturbed = clip_and_rescale(perturbed)
        candidates.append(perturbed)
    
    # Alternating projection refinement
    max_projection_steps = 200
    convergence_threshold = 1e-9
    
    for cand in candidates:
        current_h = cand.copy()
        prev_c5 = np.inf
        
        for step in range(max_projection_steps):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Clip to [0,1]
            current_h_clipped = np.clip(current_h, 0, 1)
            current_sum = np.sum(current_h_clipped)
            
            # Scale to satisfy target sum
            if current_sum == 0:
                current_scaled = np.zeros_like(current_h_clipped)
            else:
                scale_factor = target_sum / current_sum
                current_scaled = np.clip(current_h_clipped * scale_factor, 0, 1)
            
            # Compute C5 for current candidate
            corr_vals = np.correlate(current_scaled, 1 - current_scaled, mode='full')
            current_c5 = np.max(corr_vals) * dx
            
            # Update best solution
            if current_c5 < best_c5:
                best_h[:] = current_scaled.copy()
                best_c5 = current_c5
            
            # Early stopping
            if abs(prev_c5 - current_c5) < convergence_threshold:
                break
            prev_c5 = current_c5
            
            current_h = current_scaled.copy()
    
    # Final check for remaining time
    if time.time() - start_time < budget_s - 60:
        # Refine top candidate further
        top_candidate = best_h.copy()
        for _ in range(10):

                if current_sum == 0:
