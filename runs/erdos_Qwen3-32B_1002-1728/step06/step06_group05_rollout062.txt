import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    
    # Normalize initial_h_values to meet the constraint
    h_initial = h_initial / h_initial.sum() * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    
    # DE parameters
    pop_size = 30
    F = 0.8
    CR = 0.9
    max_iter = 200
    
    np.random.seed(seed)
    
    # Initialize population
    population = np.zeros((pop_size, n_points))
    population[0] = h_initial.copy()
    
    for i in range(1, pop_size):
        perturbation = np.random.normal(0, 0.1, n_points)
        h = h_initial.copy() + perturbation
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != 0:
            h = (h / sum_h) * target_sum
        h = np.clip(h, 0, 1)
        population[i] = h
    
    # Evaluate initial population
    def evaluate(h):
        return evaluate_erdos_solution(h)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    # Main DE loop
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 30:
            break
        
        for i in range(pop_size):
            # Mutation
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            
            # Crossover
            trial = np.zeros(n_points)
            r = np.random.rand(n_points) < CR
            idx = np.random.randint(0, n_points)
            r[idx] = True
            trial = np.where(r, mutant, population[i])
            
            # Apply constraints
            trial = np.clip(trial, 0, 1)
            sum_trial = trial.sum()
            if sum_trial != 0:
                trial = (trial / sum_trial) * target_sum
            trial = np.clip(trial, 0, 1)
            
            # Selection
            trial_c5 = evaluate(trial)
            if trial_c5 < evaluate(population[i]):
                population[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_h[:] = trial.copy()
                    best_c5 = trial_c5
    
    # Post-processing refinement
    def fun(x):
        x = np.clip(x, 0, 1)
        total = x.sum()
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        return evaluate_erdos_solution(x)
    
    def constraint_sum(x):
        return x.sum() - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(
        fun=fun,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    if result.success:
        optimized = np.clip(result.x, 0, 1)
        sum_opt = optimized.sum()
        if sum_opt != 0:
            optimized = (optimized / sum_opt) * target_sum
        optimized = np.clip(optimized, 0, 1)
        optimized_c5 = evaluate_erdos_solution(optimized)
        if optimized_c5 < best_c5:
            best_h[:] = optimized
            best_c5 = optimized_c5
    
    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    sum_best = best_h.sum()
    if sum_best != 0:
        best_h = (best_h / sum_best) * target_sum
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), best_c5, n_points)
