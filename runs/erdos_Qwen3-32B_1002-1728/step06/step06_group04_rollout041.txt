                    scaled_candidate = np.full(n_points, 0.5)
                else:
                    scaled_candidate = candidate_clipped * (target_sum / candidate_sum)
                c5_de = calculate_c5(scaled_candidate, dx)
                if c5_de < best_c5:
                    best_h[:] = scaled_candidate
                    best_c5 = c5_de
        except Exception as e:
            pass 

    # Local refinement with SLSQP
    bounds_sl = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    if time.time() - start_time < budget_s - 10:
        result_slsqp = minimize(
            lambda x: calculate_c5(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds_sl,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result_slsqp.success:
            candidate_slsqp = result_slsqp.x
            c5_slsqp = calculate_c5(candidate_slsqp, dx)
            if c5_slsqp < best_c5:
                best_h[:] = candidate_slsqp
                best_c5 = c5_slsqp

    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
