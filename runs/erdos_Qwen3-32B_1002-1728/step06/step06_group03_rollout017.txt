    while time.time() - start_time < budget_s - 60:
        propose_h, propose_c5 = refine_step(best_h_proj)
        if propose_c5 < best_c5:
            best_h_proj, best_c5 = propose_h, propose_c5

    final_h_proj = np.clip(project_to_feasible(best_h_proj), 0, 1)
    final_c5 = objective(final_h_proj)
    final_sum = np.sum(final_h_proj)
    if abs(final_sum - target_sum) > 1e-6:
        final_h_proj = (final_h_proj / final_sum) * target_sum
        final_h_proj = np.clip(final_h_proj, 0, 1)
        final_c5 = objective(final_h_proj)

    return (final_h_proj.tolist(), final_c5, n_points)
