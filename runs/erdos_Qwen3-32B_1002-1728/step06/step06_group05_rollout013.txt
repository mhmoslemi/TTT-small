import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 70  # Fixed resolution matches initial construction
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate Frank-Zadoff based initial guess
    m = 8
    N = m**2  # Nearest square to n_points = 64
    x_old = np.linspace(0, 1, N)
    x_new = np.linspace(0, 1, n_points)
    s = np.zeros(N)
    for i in range(N):
        mod_val = (2 * i + 1) % N
        phase = np.pi * mod_val / N
        s[i] = np.cos(phase)
    s_scaled = (s + 1.0) / 2.0  # Scale to [0,1]
    h_initial = np.interp(x_new, x_old, s_scaled)
    # Normalize to meet sum constraint
    current_sum = np.sum(h_initial)
    scaling_factor = target_sum / current_sum if current_sum != 0 else 1.0
    h_initial = np.clip(h_initial * scaling_factor, 0, 1)

    # Define objective function
    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Run SLSQP optimization
    result = minimize(
        fun=objective,
        x0=h_initial.copy(),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-4, 'disp': False}
