import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(x):
        h = np.clip(x, 0, 1)
        s = h.sum()
        if s != 0:
            h_scaled = (h / s) * target_sum
        else:
            h_scaled = np.zeros_like(h)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0)] * n_points

    # Global optimization phase
    np.random.seed(seed)
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=10, popsize=15, mutation=(0.5, 1.5),
        recombination=0.8, tol=1e-4, seed=seed,
        workers=1, disp=False
    )
    best_candidate = np.clip(result_de.x, 0, 1)
    best_value = objective(best_candidate)

    # Local refinement
    def local_func(x):
        h = np.clip(x, 0, 1)
        s = h.sum()
        if s != 0:
            h_scaled = (h / s) * target_sum
        else:
            h_scaled = np.zeros_like(h)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr_vals) * dx

    local_result = minimize(
        local_func, best_candidate, method='L-BFGS-B',
        bounds=bounds, options={'maxiter': 150, 'ftol': 1e-10}
    )
    if local_result.success:
        best_candidate = np.clip(local_result.x, 0, 1)
        best_value = objective(best_candidate)

    # Final constraint check
    h_final = np.clip(best_candidate, 0, 1)
    s_final = h_final.sum()
    if s_final != 0:
        h_final = (h_final / s_final) * target_sum
    h_final = np.clip(h_final, 0, 1)

    return (h_final.tolist(), best_value, n_points)
