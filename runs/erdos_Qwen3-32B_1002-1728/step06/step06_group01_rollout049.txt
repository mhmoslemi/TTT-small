                val = 1.0
            else:
                val = (2.0 - x[i]) / (2.0 - b) if (2.0 - b) > 1e-8 else 0
            h[i] = val
        scale = target_sum / np.sum(h)
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    # Test triangle centroid
    for m in [0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 1.8, 2.0]:
        h_test = make_triang_h(m)
        c5_test = compute_c5(h_test)
        if c5_test < best_c5:
            best_h = h_test.copy()
            best_c5 = c5_test
        if time.time() - start_time > budget_s * 0.8:
            break

    # Local refinement of best candidate
    def opt_func(x):
        x = x[0]
        h = make_triang_h(x)
        return compute_c5(h)

    def opt_asymm_func(x):
        a, b, c = x
        h = make_asymm_h(a, b, c)
        return compute_c5(h)

    if time.time() - start_time < budget_s:
        result = minimize(opt_func, x0=[1.0], bounds=[(0.1, 1.9)], method="Nelder-Mead")
        if result.success:
            h_test = make_triang_h(result.x[0])
            best_c5 = min(best_c5, compute_c5(h_test))
            best_h = h_test

    if time.time() - start_time < budget_s:
        result = minimize(opt_asymm_func, 
                         x0=[0.5, 1.5, 1.0],
                         bounds=[(0.1, 1.9), (0.1, 1.9), (0.1, 2.0)],
                         method="Nelder-Mead")
        if result.success:
            h_test = make_asymm_h(*result.x)
            best_c5 = min(best_c5, compute_c5(h_test))
            best_h = h_test

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
