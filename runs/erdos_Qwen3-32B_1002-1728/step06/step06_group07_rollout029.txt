import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0  # Since sum(h) * dx = 1 => sum(h) = T

    x_grid = np.linspace(0, 2, n_points, endpoint=False)

    # Initialize Fourier coefficients from initial_h_values
    N_terms = 10
    a = np.zeros(N_terms)
    b = np.zeros(N_terms)
    for k in range(1, N_terms + 1):
        cos_k = np.cos(k * np.pi * x_grid / 2)
        sin_k = np.sin(k * np.pi * x_grid / 2)
        a_k = 0.5 * np.sum(h_initial * cos_k) * dx
        b_k = 0.5 * np.sum(h_initial * sin_k) * dx
        a[k-1] = a_k
        b[k-1] = b_k
    initial_coeffs = np.concatenate([a, b])

    def fourier_to_h(coeffs):
        a_fft = coeffs[:N_terms]
        b_fft = coeffs[N_terms:]
        h = 0.5 * np.ones_like(x_grid)
        for k in range(1, N_terms + 1):
            h += a_fft[k-1] * np.cos(k * np.pi * x_grid / 2) + b_fft[k-1] * np.sin(k * np.pi * x_grid / 2)
        h_proj = np.clip(h, 0, 1)
        current_sum = h_proj.sum()
        if current_sum <= 1e-10:
            return h_proj
        scale_factor = T / (current_sum)
        h_scaled = h_proj * scale_factor
        return np.clip(h_scaled, 0, 1)

    def objective_fft(coeffs):
        h_fft = fourier_to_h(coeffs)
        corr_vals = np.correlate(h_fft, 1 - h_fft, mode='full')
        return float(np.max(corr_vals) * dx)

    # Fourier differential evolution phase
    bounds_fft = [(-1.0, 1.0)] * (2 * N_terms)
    try:
        fft_result = differential_evolution(
            objective_fft, bounds_fft, strategy='best1bin',
            maxiter=10, popsize=12, tol=1e-4, mutation=(0.5, 1),
            recombination=0.9, seed=seed, workers=1
        )
        best_fft_coeffs = fft_result.x
        h_fft_best = fourier_to_h(best_fft_coeffs)
        best_c5_fft = objective_fft(best_fft_coeffs)
    except:
        h_fft_best = fourier_to_h(initial_coeffs)
        best_c5_fft = objective_fft(initial_coeffs)
        print("FFT DE failed")

    best_h = h_fft_best.copy()
    best_c5 = best_c5_fft

    # Time-domain refinement
    def objective_time(h_candidate):
        h_cl = np.clip(h_candidate, 0, 1)
        s = h_cl.sum()
        if s == 0:
            return float('inf')
        h_scaled = h_cl * (T / s)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
        return corr

    if time.time() - start_time < budget_s - 100:
        # Local optimization from Fourier-based h
        try:
            time_result = minimize(
                objective_time, x0=best_h, method='SLSQP',
                bounds=[(0, 1)]*n_points, options={'maxiter': 500, 'ftol': 1e-8}
            )
            if time_result.success:
                h_cl = np.clip(time_result.x, 0, 1)
                s = h_cl.sum()
                if s > 1e-10:
                    h_scaled = h_cl * (T / s)
                    h_scaled = np.clip(h_scaled, 0, 1)
                    c5 = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_scaled
        except:
            pass

    final_h = np.clip(best_h, 0, 1)
    assert abs(final_h.sum() - T) < 1e-6, "Constraint violation"
    return (final_h.tolist(), best_c5, n_points)
