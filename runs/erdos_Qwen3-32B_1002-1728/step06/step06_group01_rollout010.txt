    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half.copy()
        h_full[-half_n:] = h_half[::-1].copy()
        return h_full

    h_half_initial = h_full_initial[:half_n].copy()
    current_sum = np.sum(h_half_initial)
    if abs(current_sum - target_sum_half) > 1e-10:
        scale_factor = target_sum_half / current_sum
        h_half_initial = h_half_initial * scale_factor
    h_half_initial = np.clip(h_half_initial, 0.0, 1.0)

    p = 100
    lambda_tv = 0.01
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}

    best_h_half = h_half_initial.copy()
    best_c5 = np.inf

    while time.time() - start_time < budget_s:
        # Perturbation
        h_half_pert = best_h_half.copy()
        np.random.seed(int((time.time() - start_time) * 1e6 % 4294967296))
        perturbation = (np.random.rand(half_n) - 0.5) * 0.1
        h_half_pert = np.clip(h_half_pert + perturbation, 0.0, 1.0)
        total = np.sum(h_half_pert)
        if total != 0:
            h_half_pert = (h_half_pert / total) * target_sum_half
        h_half_pert_clipped = np.clip(h_half_pert, 0, 1)

        def objective_with_regularization(h_half_vars):
            h_half_clipped = np.clip(h_half_vars, 0, 1)
            total = np.sum(h_half_clipped)
            if abs(total - target_sum_half) > 1e-10:
                scale = target_sum_half / total
                h_half_clipped = h_half_clipped * scale
                h_half_clipped = np.clip(h_half_clipped, 0, 1)
            h_full = create_full_h(h_half_clipped)
            corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
            x = corr_vals * dx
            lp_norm = np.linalg.norm(x, ord=p)
            tv = np.sum(np.abs(np.diff(h_full)))
            return lp_norm + lambda_tv * tv

        result = minimize(
            fun=objective_with_regularization,
            x0=h_half_pert_clipped,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 30, 'ftol': 1e-8}
        )

        if result.success:
            new_h_half = result.x
            h_full_candidate = create_full_h(new_h_half)
            h_full_candidate = np.clip(h_full_candidate, 0, 1)
            corr_vals_candidate = np.correlate(h_full_candidate, 1 - h_full_candidate, mode='full')
            current_c5 = np.max(corr_vals_candidate) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_half = new_h_half.copy()

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        # Reduce regularization strength as time progresses
        progress = 1.0 - remaining / (budget_s * 0.9)
        lambda_tv = 0.01 * (1 - progress**2)

    h_full_final = create_full_h(best_h_half)
    h_full_final = np.clip(h_full_final, 0, 1)
    corr_vals_final = np.correlate(h_full_final, 1 - h_full_final, mode='full')
    final_c5 = np.max(corr_vals_final) * dx
    assert abs(np.sum(h_full_final) - n_points / 2.0) < 1e-6
    return (h_full_final.tolist(), final_c5, n_points)
