    w_start = 0.9
    w_end = 0.4
    c1 = 2.0

    np.random.seed(seed)
    particles = np.zeros((pop_size, n_points))
    velocities = np.zeros((pop_size, n_points))
    particles[0] = h_initial.copy()

    # Initialize particles with 20% perturbations and 80% random samples
    num_perturb = int(0.2 * pop_size)
    for i in range(1, num_perturb + 1):

        particles[i] = np.clip(particles[i], 0, 1)
    for i in range(num_perturb + 1, pop_size):

        if sum_rand != 0:
            h_random = (h_random / sum_rand) * target_sum
        h_random = np.clip(h_random, 0, 1)
        particles[i] = h_random

    velocities = np.random.uniform(-0.05, 0.05, (pop_size, n_points))

    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum

        corr_vals = np.correlate(x, 1 - x, mode='full')

    personal_best = particles.copy()

    global_best_idx = np.argmin(personal_best_obj)

    best_c5 = personal_best_obj[global_best_idx]

    for it in range(max_iter):

        current_w = w_start - (w_start - w_end) * (it / max_iter)

            velocities[i] = (current_w * velocities[i] +
