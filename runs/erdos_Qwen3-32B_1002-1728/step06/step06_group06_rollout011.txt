                best_c5 = current_c5

    if optimized_solutions:
        optimized_solutions.sort(key=lambda x: x[1])
        top_k = min(5, len(optimized_solutions))
        for idx in range(top_k):
            if time.time() - start_time > budget_s - 10:
                break
            optimized, _ = optimized_solutions[idx]
            noise = np.random.normal(0, 0.2, n_points)
            perturbed = clip_and_rescale(optimized + noise)

            def refined_obj(x):
                x = clip_and_rescale(x)
                corr_vals = np.correlate(x, 1 - x, mode='full')
                return np.max(corr_vals) * dx

            result = minimize(
                fun=refined_obj, x0=perturbed, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 200, 'gtol': 1e-8, 'disp': False}
            )
            if result.success:
                refined = clip_and_rescale(result.x)
                refined_c5 = compute_c5(refined)
                if refined_c5 < best_c5:
                    best_h = refined.copy()
                    best_c5 = refined_c5

    return (best_h.tolist(), best_c5, n_points)
