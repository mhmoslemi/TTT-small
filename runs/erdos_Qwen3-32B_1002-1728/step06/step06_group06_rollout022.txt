import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        max_iter = 10
        for _ in range(max_iter):
            h_clipped = np.clip(h, 0, 1)
            current_sum = h_clipped.sum()
            if current_sum == 0:
                return np.zeros_like(h)
            scale = target_sum / current_sum
            h = h_clipped * scale
        return np.clip(h, 0, 1)

    h_initial = project(h_initial)

    np.random.seed(seed)
    n_particles = 40
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter_pso = 50
    best_global = h_initial.copy()
    best_global_c5 = np.max(np.correlate(best_global, 1 - best_global, mode='full')) * dx

    particles = []
    velocities = []
    personal_best = []
    personal_best_c5s = []

    for _ in range(n_particles):
        pos = h_initial + np.random.normal(0, 0.1, n_points)
        pos = project(pos)
        vel = np.random.normal(0, 0.05, n_points)
        particles.append(pos)
        velocities.append(vel)
        pb = pos.copy()
        pbest_c5 = np.max(np.correlate(pb, 1 - pb, mode='full')) * dx
        personal_best.append(pb)
        personal_best_c5s.append(pbest_c5)
        if pbest_c5 < best_global_c5:
            best_global = pb.copy()
            best_global_c5 = pbest_c5

    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s - 90:
            break
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best[i] - particles[i]) +
                            social * r2 * (best_global - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = project(new_pos)
            particles[i] = new_pos
            corr_vals = np.correlate(new_pos, 1 - new_pos, mode='full')
            c5_new = np.max(corr_vals) * dx
            if c5_new < personal_best_c5s[i]:
                personal_best[i] = new_pos.copy()
                personal_best_c5s[i] = c5_new
                if c5_new < best_global_c5:
                    best_global = new_pos.copy()
                    best_global_c5 = c5_new

    all_candidates = list(zip(personal_best, personal_best_c5s))
    all_candidates.sort(key=lambda x: x[1])
    top_candidates = all_candidates[:10]

    for candidate, _ in top_candidates:
        if time.time() - start_time > budget_s - 30:
            break
        def obj(x):
            x = x.reshape(-1)
            x = project(x)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        result = minimize(
            fun=obj, x0=candidate, method='Nelder-Mead',
            bounds=[(0, 1)] * n_points, options={'maxiter': 100, 'xatol': 1e-8, 'disp': False}
        )
        if result.success:
            x_opt = project(result.x)
            corr_vals = np.correlate(x_opt, 1 - x_opt, mode='full')
            c5_opt = np.max(corr_vals) * dx
            if c5_opt < best_global_c5:
                best_global = x_opt.copy()
                best_global_c5 = c5_opt

    h_final = best_global
    return (h_final.tolist(), best_global_c5, n_points)
