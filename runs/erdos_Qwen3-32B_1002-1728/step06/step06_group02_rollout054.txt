import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_gradient(h, k_shift):
    n_points = len(h)
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Add contribution from h[i] * (1 - h[i + k_shift])
        if 0 <= i + k_shift < n_points:
            grad[i] += (1 - h[i + k_shift])
        # Subtract contribution from h[i - k_shift] * (1 - h[i])
        if 0 <= i - k_shift < n_points:
            grad[i] -= h[i - k_shift]
    # Scale by dx because the gradient of sum is multiplied by dx
    grad *= dx
    return grad

def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum < 1e-8:
        return np.full_like(h_clipped, target_sum / len(h_clipped))
    scale = target_sum / current_sum
    h_clipped *= scale
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

    h = project_h(h_initial, target_sum)
    best_c5 = evaluate_c5(h, dx)

    # Gradient descent parameters
    max_gradient_steps = 1000  # Will adjust based on time
    learning_rate = 0.01
    # COBYLA local steps
    cobyla_freq = 20  # every 20 gradient steps

    current_h = h.copy()
    best_h = current_h.copy()

    # Main loop
    for step in range(max_gradient_steps):
        # Time check
        if time.time() - start_time > budget_s - 30:

        corr_vals = np.correlate(current_h, 1 - current_h, mode='full')
        index_in_result = np.argmax(corr_vals)

        grad = compute_gradient(current_h, k_shift)

        new_h = current_h - learning_rate * grad

        # Evaluate and update best
        current_c5 = evaluate_c5(current_h, dx)
        if current_c5 < best_c5:

                method='COBYLA',

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_gradient(h, k_shift):
    n_points = len(h)
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        if 0 <= i + k_shift < n_points:
            grad[i] += (1 - h[i + k_shift])
        if 0 <= i - k_shift < n_points:
            grad[i] -= h[i - k_shift]
    grad *= dx
    return grad

def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum < 1e-8:
        return np.full_like(h_clipped, target_sum / h.size)
    scale = target_sum / current_sum
    h_clipped *= scale
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped
