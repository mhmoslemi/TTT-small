def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0  # sum(h) = T

    def clip_and_scale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = T / current_sum
        scaled = h_clipped * scale
        scaled = np.clip(scaled, 0, 1)
        return scaled

    def fourier_to_h(coeffs, K):
        xs = np.linspace(0, 2, n_points, endpoint=False)
        h = np.zeros(n_points)
        dc = coeffs[0]
        for k in range(1, K+1):
            if k >= len(coeffs):
                break
            ak = coeffs[k]
            h += ak * np.cos((k * np.pi * xs) / 2.0)
        h += dc
        return h

    # Compute initial Fourier coefficients (K=20)
    K = 20
    xs = np.linspace(0, 2, n_points, endpoint=False)
    A = np.zeros((n_points, K + 1))
    A[:, 0] = 1.0
    for k in range(1, K + 1):
        A[:, k] = np.cos((k * np.pi * xs) / 2.0)
    coeffs_init, _, _, _ = np.linalg.lstsq(A, h_initial, rcond=None)

    def objective_fourier(coeffs):
        h_unscaled = fourier_to_h(coeffs, K)
        h = clip_and_scale(h_unscaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr_vals) * dx)

    bounds = [(0.0, 1.0)] + [(-0.5, 0.5)] * K  # DC + K cosine coeffs

    best_c5 = objective_fourier(coeffs_init)
    best_coeffs = coeffs_init.copy()

    # Global optimization in Fourier space
    if time.time() - start_time < budget_s - 300:
        de_result = differential_evolution(
