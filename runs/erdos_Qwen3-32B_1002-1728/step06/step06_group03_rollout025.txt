            options={'maxiter': 25, 'ftol': 1e-8}
        )
        if result.success:
            new_c5 = objective(result.x)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_params = result.x
                best_h_vals = compute_h_vals(result.x)
        # Perturb more if no improvement
        if tries % 3 == 0:
            best_params = np.clip(
                best_params * (1 + np.random.randn(7) * 0.2),
                [b[0] for b in bounds],
                [b[1] for b in bounds]
            )

    # Final normalization check
    total = np.sum(best_h_vals) * dx
    if abs(total - target_sum) > 1e-6:
        best_h_vals *= target_sum / total
    best_h_vals = np.clip(best_h_vals, 0, 1)

    return (best_h_vals.tolist(), best_c5, n_points)
