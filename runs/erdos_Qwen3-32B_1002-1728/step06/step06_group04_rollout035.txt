from scipy.optimize import minimize
import time

def calculate_c5(h, dx):

    return np.max(corr_vals) * dx

def objective_aug(h, lam, rho, dx, target_sum):
    max_overlap_val = calculate_c5(h, dx)
    sum_h = np.sum(h)
    constraint = sum_h - target_sum
    return max_overlap_val + lam * constraint + 0.5 * rho * (constraint ** 2)
