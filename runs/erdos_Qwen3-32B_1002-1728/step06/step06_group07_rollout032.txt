    h_initial_values = np.array(initial_h_values)
    n_points = len(h_initial_values)
    dx = 2.0 / n_points
    T = n_points / 2.0

    sorted_indices = np.argsort(h_initial_values)[::-1]
    binary_initial = np.zeros(n_points, dtype=int)
    binary_initial[sorted_indices[:int(T)]] = 1

    population_size = 50
    generations = 500
    mutation_rate = 0.1
    tournament_size = 5

    def eval_objective(individual):
        h = individual.astype(np.float64)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals * dx)

    def fix_ones_count(individual):
        current_sum = np.sum(individual)
        target = T
        if current_sum < target:
            zero_indices = np.where(individual == 0)[0]
            np.random.shuffle(zero_indices)
            flip_indices = zero_indices[:int(target - current_sum)]
            individual[flip_indices] = 1
        elif current_sum > target:
            one_indices = np.where(individual == 1)[0]
            np.random.shuffle(one_indices)
            flip_indices = one_indices[:int(current_sum - target)]
            individual[flip_indices] = 0
        return individual

    def crossover(parent1, parent2):
        a, b = sorted(np.random.choice(n_points, 2, replace=False))
        child1 = np.copy(parent1)
        child1[a:b] = parent2[a:b]
        child2 = np.copy(parent2)
        child2[a:b] = parent1[a:b]
        return fix_ones_count(child1), fix_ones_count(child2)

    def mutate(individual):
        _individual = np.copy(individual)
        ones_mask = _individual == 1
        zeros_mask = _individual == 0
        one_indices = np.where(ones_mask)[0]
        zero_indices = np.where(zeros_mask)[0]
        if len(one_indices) > 0 and len(zero_indices) > 0:
            one_idx = np.random.choice(one_indices)
            zero_idx = np.random.choice(zero_indices)
            _individual[one_idx] = 0
            _individual[zero_idx] = 1
        return _individual

    population = [binary_initial.copy() for _ in range(population_size)]
    for i in range(population_size):
        if i > 0:
            for _ in range(2):
                population[i] = mutate(population[i])

    best_c5 = float('inf')
    best_individual = binary_initial.copy()

    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
        fitnesses = [eval_objective(ind) for ind in population]
        for i in range(population_size):
            if fitnesses[i] < best_c5:
                best_c5 = fitnesses[i]
                best_individual = population[i].copy()

        def tournament_selection(population, fitnesses):
            selected = []
            for _ in range(population_size // 2):
                candidates = np.random.choice(population_size, tournament_size, replace=False)
                best_idx = candidates[0]
                for idx in candidates[1:]:
                    if fitnesses[idx] < fitnesses[best_idx]:
                        best_idx = idx
                selected.append(population[best_idx])
            return selected

        parents = tournament_selection(population, fitnesses)
        new_population = []
        for _ in range(population_size // 2 - 1):
            if not parents:
                break
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            ch1, ch2 = crossover(p1, p2)
            new_population.append(ch1)
            new_population.append(ch2)
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                new_population[i] = mutate(new_population[i])
        if len(new_population) < population_size:
            new_population.extend([best_individual.copy()] * (population_size - len(new_population)))
        population = new_population

    h_continuous = best_individual.astype(np.float64)
    h_continuous = np.clip(h_continuous, 0, 1)

    def objective_continuous(x):
        h = x.copy()
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum <= 1e-10:
            return float('inf')
        scale = T / current_sum
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr_vals * dx)

    bounds_continuous = [(0.0, 1.0)] * n_points
    x0 = h_continuous

    refine_result = minimize(
        objective_continuous, x0=x0, method='SLSQP',
        bounds=bounds_continuous, options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
    )

    h_refined = refine_result.x
    h_refined = np.clip(h_refined, 0, 1)
    final_c5 = objective_continuous(h_refined)

    assert abs(np.sum(h_refined) - T) < 1e-6, "Constraint violated"
    return (h_refined.tolist(), final_c5, n_points)
