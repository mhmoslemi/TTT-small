    if current_sum > 1e-10:
        h *= target_sum / current_sum
    else:
        h[:] = 0.5
    best_h = h.copy()
    best_c5 = calculate_c5_and_k(best_h, dx)[0]
    assert abs(np.sum(best_h) - target_sum) < 1e-6

    np.random.seed(seed)

    def calculate_c5_and_k(h, dx):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        start = len(h) - 1
        relevant = corr_vals[start:start + len(h)]
        max_c5 = np.max(relevant) * dx
        max_k = np.argmax(relevant)
        return max_c5, max_k

    def compute_gradient(h, k, dx, n_p):
        g = np.zeros_like(h)
        for i in range(n_p - k):
            g[i] += (1.0 - h[i + k]) * dx
            g[i + k] -= h[i] * dx
        return g

    def line_search(h, g, current_c5, dx, target_sum, n_points, current_max_k):
        alpha = 1.0
        beta = 0.5
        alpha_max = 2.0
        max_iter = 20
        for _ in range(max_iter):
            if alpha < 1e-6:
                break
            new_h = h - alpha * g
            new_h_clipped = np.clip(new_h, 0.0, 1.0)
            s = new_h_clipped.sum()
            if s < 1e-10:
                new_h = np.full_like(h, target_sum / n_points)

                new_h = new_h_clipped * target_sum / s
            new_c5, _ = calculate_c5_and_k(new_h, dx)
            if new_c5 < current_c5:
                return new_h, new_c5
            alpha *= beta
