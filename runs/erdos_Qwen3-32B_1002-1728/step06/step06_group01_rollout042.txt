        kernel = (1 - h_scaled).flip(-1)[None, None, :]  # Shape: (1, 1, n_points)
        corr = F.conv1d(h_pad, kernel, padding=n_points-1)
        max_corr = torch.max(corr)
        loss = max_corr * dx
        
        loss.backward()
        optimizer.step()
        
        # Clip parameters to maintain numerical stability
        with torch.no_grad():
            params.data = torch.clamp(params, -2.0, 2.0)
        
        # Convert to numpy for loss tracking and constraint validation
        h_np = h_scaled.detach().numpy()
        corr_vals = np.correlate(h_np, 1 - h_np, mode='full') * dx
        current_c5 = np.max(corr_vals)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_np.copy()
            print(f"Improved bound: {best_c5:.6f} ({time.time()-start_time:.1f}s elapsed)")
        
        # Early exit if no progress
        if optimizer.param_groups[0]['lr'] < 1e-5 and best_c5 < 0.3809:
            break
    
    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    assert abs(best_h.sum() - target_sum) < 1e-6, f"Violation: {best_h.sum()} vs {target_sum}"
    return (best_h.tolist(), best_c5, n_points)
