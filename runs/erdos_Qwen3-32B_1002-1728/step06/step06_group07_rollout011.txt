import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    # Define objective function for time-domain optimization
    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    # Fourier-based initialization
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    N = 10  # Number of Fourier terms
    a_initial = []
    b_initial = []
    for k in range(1, N+1):
        cos_term = np.cos(k * np.pi * x_vals / 2.0)
        sin_term = np.sin(k * np.pi * x_vals / 2.0)
        a_k = 0.5 * np.sum(h_initial * cos_term) * dx * 2
        b_k = 0.5 * np.sum(h_initial * sin_term) * dx * 2
        a_initial.append(a_k)
        b_initial.append(b_k)
    a_initial = np.array(a_initial)
    b_initial = np.array(b_initial)

    # Objective function for Fourier coefficients
    def obj_fourier(coeffs):
        a = coeffs[:N]
        b = coeffs[N:]
        h_fourier = 0.5 * np.ones(n_points)
        for i in range(N):
            k = i + 1
            h_fourier += a[i] * np.cos(k * np.pi * x_vals / 2.0) + b[i] * np.sin(k * np.pi * x_vals / 2.0)
        h_clipped = np.clip(h_fourier, 0, 1)
        h_scaled = clip_and_scale(h_clipped)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return float(np.max(corr_vals) * dx)

    # Optimize Fourier coefficients
    bounds_fourier = [(-1, 1) for _ in range(N)] + [(-1, 1) for _ in range(N)]
    initial_guess_fourier = np.concatenate([a_initial, b_initial])
    optimized_fourier = minimize(obj_fourier, x0=initial_guess_fourier, method='L-BFGS-B', bounds=bounds_fourier, options={'maxiter': 50})
    if optimized_fourier.success:
        a_opt = optimized_fourier.x[:N]
        b_opt = optimized_fourier.x[N:]
        h_fourier = 0.5 * np.ones(n_points)
        for i in range(N):
            k = i + 1
            h_fourier += a_opt[i] * np.cos(k * np.pi * x_vals / 2.0) + b_opt[i] * np.sin(k * np.pi * x_vals / 2.0)
        h_fourier = clip_and_scale(h_fourier)
    else:
        h_fourier = h_initial

    # Differential Evolution phase
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h_fourier.copy()

    de_result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=8, popsize=12, tol=1e-4, mutation=(0.5, 1),
        recombination=0.85, seed=seed, workers=1
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        if objective(de_h) < objective(best_h):
            best_h = de_h

    # Local refinement with SLSQP
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective, x0=best_h, method='SLSQP',
            bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            if objective(new_h) < objective(best_h):
                best_h = new_h

    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), objective(final_h), n_points)
