    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        scale_factor = target_sum_full / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective_local(x):
        sum_x = x.sum()
        if sum_x <= 1e-8:
            return float('inf')
        x_scaled = (x / sum_x) * target_sum_half
        x_scaled_clipped = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_scaled_clipped)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    def de_objective(x):
        sum_x = x.sum()
        if sum_x <= 1e-8:
            return float('inf')
        x_scaled = (x / sum_x) * target_sum_half
        x_scaled_clipped = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_scaled_clipped)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective_local(final_h_half)

    # Phase 1: Global search with differential evolution
    de_budget = budget_s * 0.45
    maxiter = max(1, int((de_budget * 30) // n_points**2))
    popsize = max(10, min(30, int(10 * n_points**0.5)))

    np.random.seed(seed)
    try:
        de_result = differential_evolution(
            de_objective, bounds,
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            mutation=(0.5, 1.5),
            recombination=0.9,
            tol=1e-6,
            seed=seed,
            workers=1
        )
        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
            best_c5 = de_result.fun
            final_h_half = best_h_half
    except:
        pass

    # Phase 2: Local optimization with adaptive L-BFGS-B
    num_locals = 0
    while time.time() - start_time < budget_s - 5:
        num_locals += 1
        perturbed = final_h_half.copy()
        indices = np.random.choice(half_n, size=min(half_n, 10), replace=False)
        perturbed[indices] += np.random.normal(0, 0.1, len(indices))
        perturbed = np.clip(perturbed, 0, 1)

        # Enforce sum constraint with scaling
        sum_p = perturbed.sum()
        if sum_p <= 1e-8:
            continue
        perturbed = (perturbed / sum_p) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        try:
            result = minimize(
                objective_local, x0=perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-12}
            )
            if result.success:
                new_h_half = result.x
                new_c5 = result.fun
                if new_c5 < best_c5:
                    final_h_half = new_h_half
                    best_c5 = new_c5
        except:
            pass

    # Final repair
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        h_full = (h_full / current_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
