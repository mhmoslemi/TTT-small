import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    x_i = np.arange(n_points) * dx

    # Frequency domain optimization parameters
    K = 20
    C = np.zeros((n_points, K))
    for k in range(1, K + 1):
        for i in range(n_points):
            C[i, k - 1] = np.cos(np.pi * k * x_i[i])

    a = cp.Variable(K)
    constraints = [C @ a >= -0.5, C @ a <= 0.5]
    objective = cp.Minimize(cp.sum_squares(a))
    problem = cp.Problem(objective, constraints)

    try:
        problem.solve(solver=cp.MOSEK, verbose=False)
        mosek_time = time.time() - start_time
        if problem.status == 'optimal':
            a_values = a.value
            h_f = 0.5 * np.ones(n_points) + C @ a_values
            h_f = np.clip(h_f, 0, 1)
        else:
            h_f = h_initial.copy()
    except:
        h_f = h_initial.copy()

    def objective_C5(h_candidate):
        corr_vals = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return float(np.max(corr_vals) * dx)

    best_c5 = objective_C5(h_f)
    best_h = h_f.copy()

    # Local refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        res = minimize(
            objective_C5, 
            x0=best_h, 
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if res.success:
            candidate = np.clip(res.x, 0, 1)
            val = objective_C5(candidate)
            if val < best_c5:
                best_c5 = val
                best_h = candidate

    # Final validation
    assert abs(best_h.sum() - n_points / 2.0) <= 1e-6, "Integral constraint violated"
    assert np.all((0 <= best_h) & (best_h <= 1)), "Bounds constraint violated"
    return (best_h.tolist(), best_c5, n_points)
