        obj_vals = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        current_best_idx = np.argmin(personal_best_obj)
        current_best_c5 = personal_best_obj[current_best_idx]
        if current_best_c5 < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = current_best_c5

    def fun(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    top_indices = np.argsort(personal_best_obj)[:5]
    for idx in top_indices:
        result = minimize(
            fun=fun,
            x0=personal_best[idx],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            optimized_h = np.clip(result.x, 0, 1)
            optimized_sum = np.sum(optimized_h)
            if optimized_sum != 0:
                optimized_h = (optimized_h / optimized_sum) * target_sum
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_c5 = fun(optimized_h)
            if optimized_c5 < best_c5:
                best_c5 = optimized_c5
                global_best = optimized_h

    result = minimize(
        fun=fun,
        x0=global_best,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        optimized_h = np.clip(result.x, 0, 1)
        optimized_sum = np.sum(optimized_h)
        if optimized_sum != 0:
            optimized_h = (optimized_h / optimized_sum) * target_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_c5 = fun(optimized_h)
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            global_best = optimized_h

    h_final = np.clip(global_best, 0, 1)
    h_sum = np.sum(h_final)
    if h_sum != 0:
        h_final = (h_final / h_sum) * target_sum
    h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), best_c5, n_points)
