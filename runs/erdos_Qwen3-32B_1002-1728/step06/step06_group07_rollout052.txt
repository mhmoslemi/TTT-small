            term = a[k] * np.cos(kth * np.pi * x_samples) + b[k] * np.sin(kth * np.pi * x_samples)
            h += term
        return h

    # Extract initial Fourier coefficients from h_initial
    a = []
    b = []
    for k in range(N):
        kth = k + 1
        ak = np.sum(h_initial * np.cos(kth * np.pi * x_samples)) * dx
        bk = np.sum(h_initial * np.sin(kth * np.pi * x_samples)) * dx
        a.append(ak)
        b.append(bk)
    initial_coeffs = np.array(a + b)

    def objective(coeffs):
        h = compute_h_fft(coeffs)
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr_vals)

    # Generate inequality constraints for all samples
    def make_lower(i):
        def lower_constraint(coeffs):
            h = compute_h_fft(coeffs)
            return h[i]
        return lower_constraint

    def make_upper(i):
        def upper_constraint(coeffs):
            h = compute_h_fft(coeffs)
            return 1.0 - h[i]
        return upper_constraint

    constraints = []
    for i in range(n_points):
        constraints.append({'type': 'ineq', 'fun': make_lower(i)})
        constraints.append({'type': 'ineq', 'fun': make_upper(i)})

    # Fourier domain optimization
    result_fft = minimize(
        objective,
        x0=initial_coeffs,
        method='SLSQP',
        constraints=constraints,
        options={'maxiter': 30, 'ftol': 1e-6, 'disp': False}
    )

    if result_fft.success:
        coeffs_fft_opt = result_fft.x
        h_fft_opt = compute_h_fft(coeffs_fft_opt)
        h_fft_opt = np.clip(h_fft_opt, 0, 1)
        sum_h = h_fft_opt.sum()
        if sum_h != 0:
            h_fft_opt *= (T / sum_h)
    else:
        h_fft_opt = compute_h_fft(initial_coeffs)

    # Time-domain local refinement with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_time(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(corr_vals)

    # Start with Fourier-refined h
    result_time1 = minimize(
        objective_time,
        x0=h_fft_opt,
        method='SLSQP',
        bounds=bounds,
        options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
    )

    best_h = h_fft_opt
    best_c5 = objective(best_h)
    if result_time1.success:
        h_candidate = result_time1.x
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = h_candidate.sum()
        if sum_h != 0:
            h_candidate *= (T / sum_h)
        c5_candidate = objective_time(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Quick final validation of constraints
    assert abs(best_h.sum() - T) < 1e-6, "Sum constraint violated"
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5_verified = np.max(corr_vals)

    return (best_h.tolist(), best_c5_verified, n_points)
