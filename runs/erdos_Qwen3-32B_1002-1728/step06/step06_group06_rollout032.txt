import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        rescaled = h_clipped * scale
        return np.clip(rescaled, 0, 1)

    def compute_c5(h):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    n_candidates = 20
    candidates = []
    for _ in range(n_candidates):
        pos = h_initial + np.random.normal(0, 0.1, size=n_points)
        pos = clip_and_rescale(pos)
        candidates.append(pos)

    best_c5 = float('inf')
    best_h = h_initial.copy()
    for h in candidates:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    initial_c5 = compute_c5(h_initial)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = h_initial.copy()

    max_refine_iter = 100
    for h in candidates:
        for _ in range(max_refine_iter):
            if time.time() - start_time > budget_s - 10:
                break
            h_new = clip_and_rescale(h)
            current_c5 = compute_c5(h_new)
            if abs(current_c5 - best_c5) < 1e-8:
                break
            h[:] = h_new[:]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        if time.time() - start_time > budget_s - 10:
            break

    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        best_h = clip_and_rescale(best_h)
        current_c5 = compute_c5(best_h)
        if abs(current_c5 - best_c5) < 1e-8:
            break
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
