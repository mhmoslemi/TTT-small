                          options={'maxiter': 300, 'ftol': 1e-7})
        if result.success:
            new_c5 = objective(result.x)
            if new_c5 < best_c5:
                best_gv = result.x
                best_h = create_full_h(best_gv)
                best_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
