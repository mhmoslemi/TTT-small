    h_full_binary = np.clip(h_full_binary, 0, 1)
    h_half = h_full_binary[:half_n]
    
    best_h_half = h_half.copy()
    best_c5 = np.max(np.correlate(h_full_binary, 1 - h_full_binary, mode='full')) * (2.0 / n_points)
    
    # Differential Evolution with enhanced configuration
    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full
    
    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * (2.0 / n_points)
    
    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_full / 2})
    
    # Parallel DE with higher mutation
    def objective_de(x_vars):
        x = np.array(x_vars)
        scaled = (x / np.sum(x)) * (target_sum_full / 2)
        scaled = np.clip(scaled, 0, 1)
        h_full = create_full_h(scaled)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * (2.0 / n_points)
    
    bounds_de = bounds_per_var
    np.random.seed(seed)
    de_result = differential_evolution(
        objective_de, bounds_de, strategy='best1bin',
        maxiter=50, popsize=30, tol=1e-4, mutation=(0.8, 1.2),
        recombination=0.9, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        de_h_half = np.clip(de_result.x, 0, 1)
        de_h_half_scaled = (de_h_half / np.sum(de_h_half)) * (target_sum_full / 2)
        de_h_half_scaled = np.clip(de_h_half_scaled, 0, 1)
        de_c5 = objective(de_h_half_scaled)
        if de_c5 < best_c5:
            best_h_half = de_h_half_scaled
            best_c5 = de_c5
    
    # Hybrid Perturbation Optimization
    while time.time() - start_time < budget_s - 30:
        seed += 1
        np.random.seed(seed)
        idxs = np.random.choice(half_n, size=int(half_n ** 0.5), replace=False)
        perturbed = best_h_half.copy()
        
        # Apply ternary perturbation pattern (boost, cut, hybrid)
        for i, idx in enumerate(idxs):
            if i % 3 == 0:
                # Boost
                perturbed[idx] = min(1.0, perturbed[idx] + 0.2)
            elif i % 3 == 1:
                # Cut
                perturbed[idx] = max(0.0, perturbed[idx] - 0.15)
            else:
                # Hybrid
                perturbed[idx] = np.clip(perturbed[idx] * (1 + 0.15 * np.random.randn()), 0, 1)
        
        # Enforce sum constraint
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * (target_sum_full / 2)
        perturbed = np.clip(perturbed, 0, 1)
        
        # Local refinement using multiple solvers
        for solver in ['L-BFGS-B', 'SLSQP', 'COBYLA']:
            if time.time() - start_time >= budget_s - 30:
                break
            result = minimize(
                fun=objective,
                x0=perturbed,
                method=solver,
                bounds=bounds_per_var,
                constraints=cons,
                options={'maxiter': 200 + int(100 * np.random.rand()), 'ftol': 1e-8}
            )
            if result.success:
                new_h_half = np.clip(result.x, 0, 1)
                new_h_half = (new_h_half / np.sum(new_h_half)) * (target_sum_full / 2)
                new_h_half = np.clip(new_h_half, 0, 1)
                new_c5 = objective(new_h_half)
                if new_c5 < best_c5:
                    best_h_half = new_h_half
                    best_c5 = new_c5
    
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - target_sum_full) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
