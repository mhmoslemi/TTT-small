import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    np.random.seed(seed)

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    # Generate diverse perturbed candidates
    n_candidates = 30
    candidates = [clip_and_rescale(h_initial + np.random.normal(0, 0.1, n_points)) for _ in range(n_candidates)]

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Local optimization for each candidate
    for candidate in candidates:
        if time.time() - start_time > budget_s - 20:
            break

        def objective(x):
            x = x.reshape(-1)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        # Enforce hard constraints
        constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        bounds = [(0, 1)] * n_points

        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )

        if result.success:
            h_opt = result.x.copy()
            # Final feasibility correction
            h_opt = clip_and_rescale(h_opt)
            corr_vals = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(corr_vals) * dx
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5

    # Post-processing refinement on top feasible candidates
    refinement_candidates = sorted([(np.max(np.correlate(c, 1 - c, mode='full')) * dx, c) for c in candidates + [best_h]], key=lambda x: x[0])[:5]
    for _, candidate in refinement_candidates:
        if time.time() - start_time > budget_s - 10:
            break

        def refine_objective(x):
            x = x.reshape(-1)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        result = minimize(
            fun=refine_objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )

        if result.success:
            h_opt = result.x.copy()
            h_opt = clip_and_rescale(h_opt)
            corr_vals = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(corr_vals) * dx

                best_h = h_opt.copy()

    return (best_h.tolist(), best_c5, n_points)
