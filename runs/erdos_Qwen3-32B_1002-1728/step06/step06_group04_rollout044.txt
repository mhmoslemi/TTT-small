import numpy as np
from scipy.optimize import differential_evolution, minimize, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points

    de_timeout = budget_s - 60  # Reserve 60s for refinement and final checks

        total = x.sum()
