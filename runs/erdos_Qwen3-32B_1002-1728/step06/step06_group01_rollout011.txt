    w_initial = 0.5  # Initial width (refined during optimization)

        if w <= 0:
            return np.inf

        area = np.sum(t) * dx
        if area < 1.0:
            return np.inf

        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx
