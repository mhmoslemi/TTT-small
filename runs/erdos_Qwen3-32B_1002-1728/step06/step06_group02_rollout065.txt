import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_smooth_objective(h, dx, temperature):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    overlaps = corr_vals * dx
    exp_vals = np.exp(overlaps / temperature)
    return np.log(np.sum(exp_vals))

def project_h(h, target_sum):
    h_proj = np.clip(h, 0.0, 1.0)
    current_sum = h_proj.sum()
    if current_sum == 0:
        h_proj[:] = target_sum / len(h_proj)
    else:
        h_proj *= (target_sum / current_sum)
    return h_proj

def compute_gradient(h, dx, temperature, epsilon=1e-5):
    n = len(h)
    base_obj = compute_smooth_objective(h, dx, temperature)
    grad = np.zeros(n)
    for i in range(n):
        h_pert = h.copy()
        h_pert[i] += epsilon
        obj_pert = compute_smooth_objective(h_pert, dx, temperature)
        grad[i] = (obj_pert - base_obj) / epsilon
    return grad

    np.random.seed(seed)
    h_initial = np.array(initial_h_values).copy()
