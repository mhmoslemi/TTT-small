        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        
        def objective(x):
            x = np.clip(x, 0.0, 1.0)
            x_sum = np.sum(x)
            if x_sum != 0:
                x = x / x_sum * target_sum
            x = np.clip(x, 0.0, 1.0)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            x_opt = np.clip(result.x, 0.0, 1.0)
            x_sum = np.sum(x_opt)
            if x_sum != 0:
                x_opt = x_opt / x_sum * target_sum
            x_opt = np.clip(x_opt, 0.0, 1.0)
            c5 = compute_c5(x_opt)
            if c5 < best_c5:
                best_h[:] = x_opt.copy()
                best_c5 = c5

    n_perturbations = 40
    while time.time() - start_time < budget_s - 10 and n_perturbations > 0:
        # Perturb h_initial
        noise_scale = 0.05 * h_initial + 0.01
        noise = np.random.normal(0, 1, n_points) * noise_scale
        perturbed = h_initial + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = perturbed / perturbed_sum * target_sum
        perturbed = np.clip(perturbed, 0.0, 1.0)
        optimize_candidate(perturbed)
        n_perturbations -= 1

    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h.tolist(), best_c5, n_points)
