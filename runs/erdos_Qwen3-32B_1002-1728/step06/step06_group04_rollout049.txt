    current_sum = np.sum(h_preprocessed)

    else:

    # Precompute (i, k) pairs
    ik_pairs = []
    for k in range(0, n_points):
        max_i = n_points - k

    # Group p variables by k
    k_to_p_indices = {}
    for idx, (i, k) in enumerate(ik_pairs):
        if k not in k_to_p_indices:
            k_to_p_indices[k] = []
        k_to_p_indices[k].append(idx)

    # CVXPY variables
    h = cp.Variable(n_points, nonneg=True)
    C5 = cp.Variable()

    # p variables
    p = cp.Variable(len(ik_pairs))

    constraints = []

    # Constraints on h
    constraints += [cp.sum(h) == target_sum]
    constraints += [0 <= h, h <= 1]

    # McCormick constraints on p
    for idx, (i, k) in enumerate(ik_pairs):

        term2 = p[idx] <= h[i]

    # Objective: minimize C5
    objective = cp.Minimize(C5)

    prob = cp.Problem(objective, constraints)

    # Solve the problem
    prob.solve(solver=cp.ECOS, verbose=True, max_iters=1000, warm_start=False)

import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_preprocessed = np.clip(h_initial, 0.0, 1.0)
    current_sum = np.sum(h_preprocessed)
    if current_sum > 1e-10:
        h_preprocessed *= target_sum / current_sum
    else:
        h_preprocessed.fill(0.5)

    # Precompute (i, k) pairs for positive shifts
    ik_pairs = []
    for k in range(0, n_points):
        max_i = n_points - k
        for i in range(0, max_i):
            ik_pairs.append((i, k))

    k_to_p_indices = {}
    for idx, (i, k) in enumerate(ik_pairs):
        if k not in k_to_p_indices:
            k_to_p_indices[k] = []
        k_to_p_indices[k].append(idx)

    h = cp.Variable(n_points, nonneg=True)
    C5 = cp.Variable()

    p = cp.Variable(len(ik_pairs))

    constraints = [
        cp.sum(h) == target_sum,
        0 <= h,
        h <= 1
    ]

    for idx, (i, k) in enumerate(ik_pairs):
        j = i + k
        constraints += [
            p[idx] >= h[i] + (1 - h[j]) - 1,
            p[idx] <= h[i],
            p[idx] <= 1 - h[j],
            p[idx] >= 0
        ]

    for k in k_to_p_indices:
        idxs = k_to_p_indices[k]
        p_sum = cp.sum(p[idxs])
        constraints.append(p_sum * dx <= C5)

    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_itrs=500)
        except Exception:
            pass

    solution_h = h.value
    if solution_h is None or np.isnan(solution_h).any():
        solution_h = h_preprocessed.copy()

    solution_h = np.clip(solution_h, 0, 1)
    current_sum = np.sum(solution_h)
    if abs(current_sum - target_sum) > 1e-6:
        solution_h *= target_sum / current_sum

    def calculate_c5(h, dx):
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    c5_bound = calculate_c5(solution_h, dx)

    return (solution_h.tolist(), c5_bound, n_points)
