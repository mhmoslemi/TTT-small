        h_full_candidate = np.clip(h_full_candidate, 0, 1)
        corr_vals = np.correlate(h_full_candidate, 1 - h_full_candidate, mode='full') * dx
        smooth_max = 0.1 * np.log(np.sum(np.exp(corr_vals / 0.1)))
        return smooth_max

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    np.random.seed(seed)

    while time.time() - start_time < budget_s * 0.9:
        init_guess = np.random.uniform(0, 1, half_n)
        result = minimize(objective_smooth, init_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            x_opt = result.x
            x_opt = np.clip(x_opt, 0, 1)
            total = np.sum(x_opt)
            if total != 0:
                x_opt = (x_opt / total) * target_sum_half
            x_opt = np.clip(x_opt, 0, 1)
            h_full_candidate = create_full_h(x_opt)
            h_full_candidate = np.clip(h_full_candidate, 0, 1)
            c5_candidate = np.max(np.correlate(h_full_candidate, 1 - h_full_candidate, mode='full') * dx)
            if c5_candidate < best_c5:
                best_h_half = x_opt.copy()
                best_c5 = c5_candidate

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, f"Sum constraint violated by {abs(np.sum(h_full) - n_points / 2.0)}"
    return (h_full.tolist(), best_c5, n_points)
