    target_sum_half = n_points / 4.0  # Since full h sums to n_points/2 and we optimize half the variables

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def compute_c5(h_full):
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    lambda_val = 1e-3
    p = 20  # Approximate max with Lp norm

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        corr_integrals = corr_vals * dx
        l_p = np.linalg.norm(corr_integrals, ord=p)
        l1_term = lambda_val * np.sum(np.abs(h_vars))
        return l_p + l1_term

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    h_half = h_full_initial[:half_n].copy()
    total = np.sum(h_half)
    if total != 0:
        h_half = (h_half / total) * target_sum_half
    h_half = np.clip(h_half, 0, 1)
    best_h_half = h_half.copy()
    best_c5 = compute_c5(create_full_h(best_h_half))

    def objective_de(x_vars):
        x = np.array(x_vars)
        total = np.sum(x)
        if total != 0:
            x_scaled = (x / total) * target_sum_half
        else:
            x_scaled = x.copy()
        x_clipped = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_clipped)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        corr_integrals = corr_vals * dx
        l_p = np.linalg.norm(corr_integrals, ord=p)
        l1_term = lambda_val * np.sum(np.abs(x_vars))
        return l_p + l1_term

    bounds_de = [(0.0, 1.0)] * half_n
    np.random.seed(seed)
    try:
        de_result = differential_evolution(
            objective_de, bounds_de, strategy='best1bin', maxiter=15, popsize=15, tol=1e-4, mutation=(0.7, 1),
            recombination=0.9, seed=seed, workers=1, disp=False
        )
        if de_result.success:
            de_h_half = np.clip(de_result.x, 0, 1)
            total = np.sum(de_h_half)
            if total != 0:
                de_h_half = (de_h_half / total) * target_sum_half
            de_h_half = np.clip(de_h_half, 0, 1)
            new_c5 = compute_c5(create_full_h(de_h_half))
            if new_c5 < best_c5:
                best_h_half = de_h_half
                best_c5 = new_c5
    except Exception:
        pass

    # Perturbation and local optimization loop
    np.random.seed(seed)
    while time.time() - start_time < budget_s - 30:
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.25 * (1 - frac**1.5)
        seed = seed + 1
        rnd = np.random.RandomState(seed)
        idxs = np.random.choice(half_n, size=min(half_n, 10), replace=False)
        perturbed = best_h_half.copy()
        perturbed[idxs] += rnd.normal(0, noise_scale, len(idxs))
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        method_idx = np.random.randint(0, 3)
        if method_idx == 0:
            method = 'SLSQP'
        elif method_idx == 1:
            method = 'L-BFGS-B'
        else:
            method = 'COBYLA'
        result = minimize(
            fun=objective, x0=perturbed, method=method, bounds=bounds_per_var, constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h_half = np.clip(np.array(result.x), 0, 1)
            total = np.sum(new_h_half)
            if abs(total - target_sum_half) < 1e-6:
                new_h_full = create_full_h(new_h_half)
                new_c5 = compute_c5(new_h_full)
                if new_c5 < best_c5:
                    best_h_half = new_h_half
                    best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
