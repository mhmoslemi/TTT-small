import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points, endpoint=False)
    N = 20  # Number of harmonics

    # Build Fourier basis matrix A
    A = np.zeros((n_points, 1 + 2 * N))
    A[:, 0] = 1.0
    for n in range(1, N + 1):
        A[:, n] = np.cos(n * np.pi * x)
        A[:, N + n] = np.sin(n * np.pi * x)

    # Least squares fit initial theta
    theta_linear, _, _, _ = np.linalg.lstsq(A, h_full_initial, rcond=None)

    # Define objective and constraint
    def objective(theta):
        a0 = theta[0]
        a_n = theta[1:1 + N]
        b_n = theta[1 + N:1 + 2 * N]
        z = a0 * np.ones_like(x)
        for idx in range(N):
            harmonic = idx + 1
            z += a_n[idx] * np.cos(harmonic * np.pi * x)
            z += b_n[idx] * np.sin(harmonic * np.pi * x)
        h_samples = 1.0 / (1.0 + np.exp(-z))
        corr_vals = np.correlate(h_samples, 1 - h_samples, mode='full')
        return np.max(corr_vals) * dx

    def constraint_func(theta):
        a0 = theta[0]
        a_n = theta[1:1 + N]
        b_n = theta[1 + N:1 + 2 * N]
        z = a0 * np.ones_like(x)
        for idx in range(N):
            harmonic = idx + 1
            z += a_n[idx] * np.cos(harmonic * np.pi * x)
            z += b_n[idx] * np.sin(harmonic * np.pi * x)
        h_samples = 1.0 / (1.0 + np.exp(-z))
        return np.sum(h_samples) - target_sum

    cons = ({'type': 'eq', 'fun': constraint_func})
    result = minimize(
        objective,
        x0=theta_linear,
        method='SLSQP',
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )

    theta_opt = result.x
    a0 = theta_opt[0]

    assert abs(h_best.sum() - target_sum) < 1e-6
    best_c5 = objective(theta_opt)
