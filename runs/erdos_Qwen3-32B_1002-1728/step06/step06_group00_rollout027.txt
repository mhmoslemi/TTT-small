def repair_individual(individual, target_sum):
    # Clip values to [0, 1]
    individual_clipped = np.clip(individual, 0, 1)
    # Compute current sum
    current_sum = np.sum(individual_clipped)
    # Scale to match target sum
    if current_sum != 0:
        individual_repaired = (individual_clipped / current_sum) * target_sum
    else:
        # All zeros, distribute equally
        individual_repaired = np.full_like(individual_clipped, target_sum / len(individual_clipped))
    # Clip again in case of scaling overshoot
    individual_repaired = np.clip(individual_repaired, 0, 1)
    return individual_repaired

def create_full_h(individual, half_n, n_points):
    h = np.zeros(n_points)
    h[:half_n] = individual
    h[-half_n:] = individual[::-1]
    return h

def repair_individual(individual, target_sum):
    individual_clipped = np.clip(individual, 0, 1)
    current_sum = np.sum(individual_clipped)
    if current_sum != 0:
        individual_repaired = (individual_clipped / current_sum) * target_sum
    else:
        individual_repaired = np.full_like(individual_clipped, target_sum / len(individual_clipped))
    individual_repaired = np.clip(individual_repaired, 0, 1)
    return individual_repaired

def evaluate(individual, dx, n_points, half_n):
    h_full = create_full_h(individual, half_n, n_points)
    h_full = np.clip(h_full, 0, 1)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return -np.max(corr) * dx  # Fitness is negative of C5

def tournament_selection(population, fitnesses, tournament_size=3):
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2):

    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])

    ind[mask] += noise

            fitnesses.append(fitness)

        elite_count = max(1, int(0.05 * len(population)))
        elite_indices = np.argsort(fitnesses)[::-1][:elite_count]

    h_full = create_full_h(best_half, half_n, n_points)
    h_full = np.clip(h_full, 0, 1)
    corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5_bound = np.max(corr_vals) * dx
