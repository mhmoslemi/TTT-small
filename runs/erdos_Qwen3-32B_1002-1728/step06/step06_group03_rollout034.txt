import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_clipped = np.clip(h_vars, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return float('inf')
        scaled = (h_clipped / total) * target_sum
        h_scaled = np.clip(scaled, 0, 1)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr_vals) * dx

    bounds_full = Bounds([0.0] * n_points, [1.0] * n_points)

    # Global search with differential evolution
    de_result = differential_evolution(
        objective, bounds_full, strategy='best1bin',
        maxiter=50, popsize=25, tol=1e-5, mutation=(0.7, 1.2), recombination=0.9,
        seed=seed, workers=1, disp=False
    )

    best_h = de_result.x if de_result.success else h_full_initial.copy()

    # Local refinement with SLSQP
    result = minimize(
        objective, best_h, method='SLSQP', bounds=bounds_full,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - target_sum}],
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    if result.success:
        best_h = result.x

    # Final validation and normalization
    final_h = np.clip(best_h, 0, 1)
    total = np.sum(final_h)
    if total != 0:
        final_h = (final_h / total) * target_sum
        final_h = np.clip(final_h, 0, 1)
    else:
        final_h[0] = target_sum

    return (final_h.tolist(), objective(final_h), n_points)
