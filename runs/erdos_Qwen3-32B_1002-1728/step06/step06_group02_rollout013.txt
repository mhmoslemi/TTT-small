                first = 0.0 
            # Compute second contribution 
            j2 = i - k_shift 
            if 0 <= j2 < n:
                second = - h[j2] * dx 
            else:
                second = 0.0 
            # total contribution from this shift
            contribution = weight * (first + second)
            total += contribution 
        grad[i] = total 
    return grad 

def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        h_clipped[:] = target_sum / len(h)
    else:
        h_clipped = h_clipped * (target_sum / current_sum)
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

    tau = 0.5  # initial temperature
    step_size = 0.1
    max_iterations = 500
    n_points = len(h_initial)
    
    h = h_initial.copy()
    
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Compute gradient
        grad = compute_gradient_smoothed_c5(h, dx, tau)
        
        # Line search
        alpha = step_size
        h_candidate = h - alpha * grad
        h_candidate = project_h(h_candidate, target_sum)
        
        # Check if candidate is better
        c5_candidate = evaluate_c5(h_candidate, dx)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            h = h_candidate.copy()
        else:
            # Backtracking line search
            for _ in range(10):
                alpha *= 0.5
                if alpha < 1e-8:
                    break
                h_candidate = h - alpha * grad
                h_candidate = project_h(h_candidate, target_sum)
                c5_candidate = evaluate_c5(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    h = h_candidate.copy()
                    break
        # Decrease tau
        tau *= 0.99
        
        if iter % 50 ==0:
            print(f"Iter {iter} best_c5 {best_c5} time {time.time()-start_time}")
            
    # Local refinement with COBYLA
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        c5 = evaluate_c5(h, dx)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Add constraint for sum
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    if res.success and res.fun < best_c5:
        best_h = np.clip(res.x, 0, 1)
        best_c5 = res.fun

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_smoothed_c5(h, dx, tau):
    corr_vals_unscaled = np.correlate(h, 1 - h, mode='full')
    corr_vals = corr_vals_unscaled * dx
    exp_vals = np.exp(corr_vals / tau)
    sum_exp = np.sum(exp_vals)
    return np.log(sum_exp) * dx 

def compute_gradient_smoothed_c5(h, dx, tau):

    corr_vals_unscaled = np.correlate(h, 1 - h, mode='full')

    sum_exp = np.sum(exp_vals)
