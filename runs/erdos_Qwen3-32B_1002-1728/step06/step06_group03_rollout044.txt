import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # Half of total target sum (n/2 / 2)
    dx = 2.0 / n_points

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    # PSO parameters
    n_particles = 30
    w = 0.7  # inertia
    c1 = 1.2  # cognitive
    c2 = 1.2  # social
    max_vel = 0.1

    np.random.seed(seed)
    positions = np.array([h_full_initial[:half_n] + np.random.uniform(-0.1, 0.1, half_n) for _ in range(n_particles)])
    velocities = np.random.uniform(-max_vel, max_vel, (n_particles, half_n))

    pbest_positions = positions.copy()
    pbest_scores = np.array([objective(np.clip(pos, 0, 1)) for pos in positions])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    # PSO main loop
    while time.time() - start_time < budget_s - 30:
        for i in range(n_particles):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest_positions[i] - positions[i]) + c2 * r2 * (gbest_position - positions[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            # Position update and constraint handling
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)

            # Sum constraint normalization
            clipped = np.clip(positions[i], 0, 1)
            s = np.sum(clipped)
            if s != 0:
                scaled = clipped * (target_sum_half / s)
            else:
                scaled = clipped
            positions[i] = np.clip(scaled, 0, 1)

            # Update objective score
            score = objective(positions[i])
            if score < pbest_scores[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_scores[i] = score

        # Update global best
        candidates = pbest_scores if time.time() - start_time >= budget_s - 30 else pbest_scores
        new_idx = np.argmin(candidates)
        if pbest_scores[new_idx] < gbest_score:
            gbest_position = pbest_positions[new_idx].copy()
            gbest_score = pbest_scores[new_idx]

    # Local refinement on top candidates
    indices = np.argsort(pbest_scores)[:max(1, int(0.1 * n_particles))]
    for idx in indices:
        if time.time() - start_time >= budget_s:
            break
        x0 = pbest_positions[idx].copy()
        bounds = [(0.0, 1.0)] * half_n
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
