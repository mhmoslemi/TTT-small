        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    def objective_de(x_de):
        x_de = np.array(x_de).flatten()
        x_feasible = clip_and_rescale(x_de)
        corr_vals = np.correlate(x_feasible, 1 - x_feasible, mode='full')
        return np.max(corr_vals) * dx

    bounds_de = [(0, 1) for _ in range(n_points)]

    result_de = None
    try:
        # Run Differential Evolution with time tracking
        maxiter_de = 30
        popsize_de = 5
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 60:  # Avoid using more than 1 minute for DE
            result_de = differential_evolution(
                objective_de, bounds_de,
                strategy='best1bin',
                maxiter=maxiter_de,
                popsize=popsize_de,
                tolerance=1e-7,
                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed,
                polish=False,
                init='sobol',
                disp=False
            )
    except:
        pass

    h_best = h_initial.copy()
    c5_best = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx

    if result_de is not None and result_de.success:
        h_de_params = result_de.x
        h_feasible = clip_and_rescale(h_de_params)
        c5_de = result_de.fun
        if c5_de < c5_best:
            h_best = h_feasible
            c5_best = c5_de
    else:
        h_de_params = h_initial.copy()

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        # Refine with SLSQP
        bounds_sl = [(0, 1) for _ in range(n_points)]
        def objective_sl(x):
            x_feasible = clip_and_rescale(x)
            corr_vals = np.correlate(x_feasible, 1 - x_feasible, mode='full')
            return np.max(corr_vals) * dx

        result_sl = minimize(
            fun=objective_sl,
            x0=h_de_params,
            method='SLSQP',
            bounds=bounds_sl,
            options={'maxiter': 200, 'gtol': 1e-8, 'disp': False}
        )

        if result_sl.success:
            x_opt = clip_and_rescale(result_sl.x)
            corr_vals = np.correlate(x_opt, 1 - x_opt, mode='full')
            c5_opt = np.max(corr_vals) * dx
            if c5_opt < c5_best:
                h_best = x_opt
                c5_best = c5_opt

    return (h_best.tolist(), c5_best, n_points)
