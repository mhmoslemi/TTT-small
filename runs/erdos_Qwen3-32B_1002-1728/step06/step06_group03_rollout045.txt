            h_full[-half_n:] = h_half[::-1]
            h_full = np.clip(h_full, 0, 1)
            best_h_full = h_full.copy()

    # Refine top candidates
    top_indices = np.argsort(pbest_fitness)[:max(1, n_particles // 10)]
    for idx in top_indices:
        if time.time() - start_time >= budget_s - 10:
            break
        h_half_refine = pbest_positions[idx].copy()
        def refine_obj(optim):
            param = np.clip(optim, 0, 1)
            total = np.sum(param)
            param = param * (target_sum_half / total) if total != 0 else np.zeros_like(param)
            param = np.clip(param, 0, 1)
            h_full = np.zeros(n_points)
            h_full[:half_n] = param
            h_full[-half_n:] = param[::-1]
            corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr_vals) * dx
        bounds = [(0.0, 1.0) for _ in range(half_n)]
        res = minimize(refine_obj, x0=h_half_refine, method='SLSQP', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        if res.success:
            refined = np.clip(res.x, 0, 1)
            total = np.sum(refined)
            refined = refined * (target_sum_half / total) if total != 0 else np.zeros_like(refined)
            refined = np.clip(refined, 0, 1)
            c5 = objective_half(refined)
            if c5 < best_c5:
                best_c5 = c5
                h_half = refined
                h_full = np.zeros(n_points)
                h_full[:half_n] = h_half
                h_full[-half_n:] = h_half[::-1]
                h_full = np.clip(h_full, 0, 1)
                best_h_full = h_full.copy()

    # Final validation
    if best_h_full is None:
        h_full = h_full_initial
    else:
        h_full = best_h_full
    h_sum = np.sum(h_full)
    if abs(h_sum - n_points / 2) > 1e-6:
        h_full = (h_full / h_sum) * (n_points / 2)
    h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
