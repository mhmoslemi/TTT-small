        sum_opt = np.sum(h_fine_opt)
        if abs(sum_opt) > 1e-10:
            h_fine_opt = (h_fine_opt / sum_opt) * target_sum_fine
        h_fine_opt = np.clip(h_fine_opt, 0, 1)
        refined_c5_fine = fun_fine(h_fine_opt)
        if refined_c5_fine < best_c5_fine:
            best_c5_fine = refined_c5_fine
            global_best_fine = h_fine_opt

    # Final check: use the best between coarse-upsampled and fine best
    # Possibly, the initial fine has a better C5?

    # Get current best solution (from fine grid)
    h_full = global_best_fine.copy()
    h_full_sum = np.sum(h_full)
    if abs(h_full_sum) > 1e-10:
        h_full = (h_full / h_full_sum) * target_sum_fine
    h_full = np.clip(h_full, 0, 1)
    final_c5 = objective_fine(h_full)

    # Also check if the refined coarse solution upsampled is better
    h_coarse_upsampled = upsample(h_coarse_opt)
    h_coarse_upsampled_sum = np.sum(h_coarse_upsampled)
    if abs(h_coarse_upsampled_sum) > 1e-10:
        h_coarse_upsampled = (h_coarse_upsampled / h_coarse_upsampled_sum) * target_sum_fine
    h_coarse_upsampled = np.clip(h_coarse_upsampled, 0, 1)
    c5_coarse_up = objective_fine(h_coarse_upsampled)
    if c5_coarse_up < final_c5:
        h_full = h_coarse_upsampled
        final_c5 = c5_coarse_up

    return (h_full.tolist(), final_c5, N_fine)
