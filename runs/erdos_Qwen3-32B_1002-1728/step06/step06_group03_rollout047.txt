from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(initial_h_values)

    half_n = n_points // 2

    dx = 2.0 / n_points

    target_sum = n_points / 2.0

    target_sum_half = target_sum / 2.0

    np.random.seed(seed)

    def normalize_half(half_h, target_sum_half):

        half_h_clipped = np.clip(half_h, 0, 1)

        current_sum = np.sum(half_h_clipped)

        if current_sum == 0:

            return np.zeros_like(half_h_clipped)

        scaled = (half_h_clipped / current_sum) * target_sum_half

        return np.clip(scaled, 0, 1)

    def compute_c5(h_half):

        h_full = np.zeros(n_points)

        h_full[:half_n] = h_half

        h_full[-half_n:] = h_half[::-1]

        h_full_clipped = np.clip(h_full, 0, 1)

        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')

        return np.max(corr_vals) * dx

    # Initialize particles

    n_particles = 30

    positions = np.zeros((n_particles, half_n))

    positions[0] = initial_h_values[:half_n].copy()

    # Generate remaining particles

    for i in range(1, n_particles):

        perturbation = np.random.normal(0, 0.05, half_n)

        positions[i] = positions[0] + perturbation

        positions[i] = normalize_half(positions[i], target_sum_half)

    # Initialize velocities

    velocities = np.zeros((n_particles, half_n))

    # Evaluate initial fitness

    pbest_positions = positions.copy()

    pbest_fitness = np.array([compute_c5(p) for p in positions])

    global_best_idx = np.argmin(pbest_fitness)

    global_best_fitness = pbest_fitness[global_best_idx]

    c2 = 1.2  # social

        new_global_idx = np.argmin(pbest_fitness)

        if pbest_fitness[new_global_idx] < global_best_fitness:

            global_best_position = pbest_positions[new_global_idx].copy()

            global_best_fitness = pbest_fitness[new_global_idx]

        # Exit condition based on time

import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    target_sum_half = target_sum / 2.0

    np.random.seed(seed)

    def normalize_half(half_h, target_sum_half):
        half_h_clipped = np.clip(half_h, 0, 1)
        current_sum = np.sum(half_h_clipped)
        if current_sum == 0:
            return np.zeros_like(half_h_clipped)
        scaled = (half_h_clipped / current_sum) * target_sum_half
        return np.clip(scaled, 0, 1)

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * dx

    n_particles = 30
    positions = np.zeros((n_particles, half_n))
    positions[0] = initial_h_values[:half_n].copy()
    for i in range(1, n_particles):
        perturbation = np.random.normal(0, 0.05, half_n)
        positions[i] = positions[0] + perturbation
        positions[i] = normalize_half(positions[i], target_sum_half)
    velocities = np.zeros((n_particles, half_n))

    pbest_positions = positions.copy()
    pbest_fitness = np.array([compute_c5(p) for p in positions])
    global_best_idx = np.argmin(pbest_fitness)
    global_best_position = pbest_positions[global_best_idx].copy()
    global_best_fitness = pbest_fitness[global_best_idx]

    w = 0.7
    c1 = 1.2
    c2 = 1.2

    while time.time() - start_time < budget_s * 0.6:
        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        vel_cognitive = c1 * r1 * (pbest_positions - positions)
        vel_social = c2 * r2 * (global_best_position - positions)
        velocities = w * velocities + vel_cognitive + vel_social
        velocities = np.clip(velocities, -0.1, 0.1)
        new_positions = positions + velocities

        positions = new_positions

            current_fitness = compute_c5(positions[i])
