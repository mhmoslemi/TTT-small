import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:

        scale = target_sum / current_sum

    h_initial = clip_and_rescale(h_initial)

    np.random.seed(seed)

    n_initial = 20
    initial_candidates = []
    for _ in range(n_initial):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + noise
        feasible = clip_and_rescale(perturbed)
        initial_candidates.append(feasible)

    best_c5 = np.inf

    start_time = time.time()
