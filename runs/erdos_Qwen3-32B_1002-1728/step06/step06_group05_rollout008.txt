    F = 0.8
    CR = 0.9
    max_iter = 200

        candidate = pop[i - 1] + perturbation
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate != 0:
            candidate = (candidate / sum_candidate) * target_sum
        candidate = np.clip(candidate, 0, 1)
        pop[i] = candidate

    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    best_h = h_initial
    best_c5 = objective(h_initial)

    for gen in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break

        for i in range(pop_size):
            temp_pop = [j for j in range(pop_size) if j != i]
            a, b, c = np.random.choice(temp_pop, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = np.copy(pop[i])
            cross_mask = np.random.rand(n_points) < CR
            if not np.any(cross_mask):
                cross_mask[np.random.randint(n_points)] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = np.clip(trial, 0, 1)

            if sum_trial != 0:
                trial = (trial / sum_trial) * target_sum

                pop[i] = trial
                if objective(trial) < best_c5:
                    best_c5 = objective(trial)
                    best_h = trial

    candidates = pop[np.argsort([objective(ind) for ind in pop])[:5]]
    refined_candidates = []

            x = np.clip(x, 0, 1)

            return np.max(np.correlate(x, 1 - x, mode='full')) * dx
