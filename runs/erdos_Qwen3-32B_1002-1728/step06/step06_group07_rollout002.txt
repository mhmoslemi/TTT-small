        h = 0.5 + a @ cos_basis + b @ sin_basis
        return objective(h)

    best_h = h_full_initial[:]
    best_c5 = objective(best_h)

    # Fourier-based exploration
    if time.time() - start_time < budget_s - 30:
        res_f = differential_evolution(
            fourier_obj, [(-10, 10)]*(2*n_harmonics),
            strategy='best1bin', popsize=10, tol=1e-3,
            maxiter=10, seed=seed, workers=1, disp=False
        )
        if res_f.success:
            a_opt, b_opt = res_f.x[:n_harmonics], res_f.x[n_harmonics:]
            h_fft = 0.5 + a_opt @ cos_basis + b_opt @ sin_basis
            curr_c5 = objective(h_fft)
            if curr_c5 < best_c5:
                best_h = h_fft
                best_c5 = curr_c5

    # Time-domain refinement
    bounds = [(0, 1)] * n_points
    for _ in range(2):
        if time.time() - start_time > budget_s - 30:
            break
        res_de = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=10,
            tol=1e-4, mutation=(0.5, 0.8), maxiter=5, seed=seed,
            workers=1, disp=False
        )
        if res_de.x is not None:
            h_de = clip_and_scale(res_de.x)
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de

    # Final polish
    if time.time() - start_time < budget_s - 10:
        res_slsqp = minimize(
            objective, x0=best_h, method='SLSQP', bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res_slsqp.success:
            h_slsqp = clip_and_scale(res_slsqp.x)
            best_c5 = min(best_c5, objective(h_slsqp))
            best_h = h_slsqp

    final_h = clip_and_scale(best_h)
    return (final_h.tolist(), best_c5, n_points)
