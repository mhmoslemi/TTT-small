            method='SLSQP',

            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            optimized_h = enforce_constraints(result.x, target_sum)
            optimized_c5 = objective(optimized_h, target_sum, dx)
            if optimized_c5 < best_c5:
                best_c5 = optimized_c5
                global_best = optimized_h.copy()

    h_final = enforce_constraints(global_best, target_sum)
    return (h_final.tolist(), best_c5, n_points)
