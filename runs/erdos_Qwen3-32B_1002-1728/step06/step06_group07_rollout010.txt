import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = clip_and_scale(h_initial)
    best_c5 = objective(best_h)

    # Enhanced differential evolution with more population and iterations
    de_result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=15, popsize=20, tol=1e-4, mutation=(0.5, 1),
        recombination=0.85, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        de_c5 = objective(de_h)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h

    # More extensive local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective, x0=best_h, method='SLSQP',
            bounds=bounds, options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Final validation
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), best_c5, n_points)
