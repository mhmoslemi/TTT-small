import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    np.random.seed(seed)
    n_candidates = 20
    candidates = []
    for _ in range(n_candidates):
        if time.time() - start_time > budget_s - 10:
            break
        pos = h_initial + np.random.normal(0, 0.1, n_points)
        pos = clip_and_rescale(pos)
        candidates.append(pos)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for idx, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 5:
            break
        h_proj = candidate.copy()
        prev_c5 = np.inf
        for iter_proj in range(100):
            h_proj = clip_and_rescale(h_proj)
            corr_vals = np.correlate(h_proj, 1 - h_proj, mode='full')
            current_c5 = np.max(corr_vals) * dx
            if np.abs(prev_c5 - current_c5) < 1e-8:
                break
            prev_c5 = current_c5
        if current_c5 < best_c5:
            best_h = h_proj.copy()
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
