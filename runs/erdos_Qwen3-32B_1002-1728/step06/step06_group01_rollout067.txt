        perturbed = perturbed * (target_sum_half / np.sum(perturbed))
        h_candidates.append(perturbed)

    for candidate in h_candidates:
        result = minimize(
            fun=objective,
            x0=candidate,
            method='L-BFGS-B',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0, 1)
            h_full = np.concatenate([new_h, new_h[::-1]])
            current_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                h_half = new_h
                best_h_full = h_full

    h_full = np.clip(best_h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, (np.sum(h_full), n_points)
    return (h_full.tolist(), best_c5, n_points)
