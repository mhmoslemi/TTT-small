import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def enforce_feasibility(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = np.sum(x_clipped)
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum
            x_scaled = np.clip(x_scaled, 0.0, 1.0)
        else:
            x_scaled = x_clipped
        return x_scaled

    h_current = enforce_feasibility(h_initial)

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    T0 = 1.0
    alpha = 0.99
    T = T0
    max_iter = 5000
    best_h = h_current.copy()
    best_c5 = evaluate(best_h)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
        mutation = np.random.normal(0, T, n_points)
        h_new = enforce_feasibility(h_current + mutation)
        c5_new = evaluate(h_new)
        delta = best_c5 - c5_new
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        T *= alpha

    def fun(x):
        x = enforce_feasibility(x)
        return evaluate(x)

    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(
        fun=fun,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        h_opt = enforce_feasibility(result.x)
        c5_opt = evaluate(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    return (enforce_feasibility(best_h).tolist(), best_c5, n_points)
