def adjust_sum(indiv, T):
    indiv = indiv.copy()
    sum_current = indiv.sum()
    if sum_current == T:
        return indiv
    diff = sum_current - T
    if diff > 0:
        idxs = np.where(indiv == 1)[0]
        if len(idxs) == 0:
            return indiv
        np.random.shuffle(idxs)
        to_flip = idxs[:diff]
        indiv[to_flip] = 0
    else:
        idxs = np.where(indiv == 0)[0]

def adjust_sum(indiv, T):
    indiv = indiv.copy()
    sum_current = indiv.sum()
    if np.isclose(sum_current, T, atol=1e-6):
        return indiv
    diff = sum_current - T
    if diff > 0:
        idxs = np.where(indiv == 1)[0]
        if len(idxs) == 0:
            return indiv
        np.random.shuffle(idxs)
        to_flip = idxs[:int(np.round(diff))]
        indiv[to_flip] = 0
    else:
        idxs = np.where(indiv == 0)[0]
