            a[k] = dx * np.sum(h * np.cos(k_plus_1 * np.pi * x / 2))
            b[k] = dx * np.sum(h * np.sin(k_plus_1 * np.pi * x / 2))
        return a, b

    a_init, b_init = get_initial_coeffs(h_initial, n_terms)
    params_init = np.concatenate([a_init, b_init])

    def reconstruct_and_scale(params):
        a = params[:n_terms]
        b = params[n_terms:]
        x = np.linspace(0, 2, n_points)
        series_sum = 0.5 * np.ones(n_points)
        for k in range(n_terms):
            k_plus_1 = k + 1
            cos_term = a[k] * np.cos(k_plus_1 * np.pi * x / 2)
            sin_term = b[k] * np.sin(k_plus_1 * np.pi * x / 2)
            series_sum += cos_term + sin_term
        h = series_sum
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(params):
        h_scaled = reconstruct_and_scale(params)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return float(np.max(corr_vals) * dx)

    bounds = [(-5.0, 5.0)] * (2 * n_terms)
    best_params = params_init.copy()
    best_c5 = objective(best_params)

    de_result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=10, popsize=20, tol=1e-4, mutation=(0.5, 1),
        recombination=0.9, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        de_params = de_result.x
        de_c5 = objective(de_params)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_params = de_params

    if time.time() - start_time < budget_s - 60:
        result = minimize(
            objective, x0=best_params, method='SLSQP',
            bounds=bounds, options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
        )
        if result.success:
            slsqp_params = result.x
            slsqp_c5 = objective(slsqp_params)
            if slsqp_c5 < best_c5:
                best_c5 = slsqp_c5
                best_params = slsqp_params

    final_h = reconstruct_and_scale(best_params)
    assert abs(final_h.sum() - n_points / 2) < 1e-6, "Integral constraint violated"
    return (final_h.tolist(), best_c5, n_points)
