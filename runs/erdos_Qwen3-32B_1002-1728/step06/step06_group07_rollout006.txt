import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective_binary(candidate):
        h_processed = candidate.astype(float)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    def mutate(candidate):
        h = candidate.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return h
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0, 1
        return h

    np.random.seed(seed)
    h_binary_initial = (h_full_initial > 0.5).astype(int)
    population_size = 50
    population = [h_binary_initial.copy()]
    for _ in range(population_size - 1):
        indices = np.random.permutation(n_points)
        h_binary = np.zeros(n_points, dtype=int)
        h_binary[indices[:int(T)]] = 1
        population.append(h_binary)

    best_c5 = float('inf')
    best_h = h_binary_initial.copy()

    while time.time() - start_time < budget_s - 100:
        current_objectives = [objective_binary(c) for c in population]
        current_min = min(current_objectives)
        if current_min < best_c5:
            best_c5 = current_min
            idx = np.argmin(current_objectives)
            best_h = population[idx].copy()

        new_population = []
        for _ in range(population_size):
            while True:
                i_a = np.random.randint(population_size)
                i_b = np.random.randint(population_size)
                if i_a != i_b:
                    break
            a, b = population[i_a], population[i_b]
            if current_objectives[i_a] <= current_objectives[i_b]:
                parent = a
            else:
                parent = b
            child = mutate(parent)
            new_population.append(child)
        population = new_population

    h_initial_continuous = clip_and_scale(best_h.astype(float))
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_continuous(x):
        h = clip_and_scale(x)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    final_h = h_initial_continuous.copy()
    best_c5_refined = best_c5
    start_refinement = time.time()
    while time.time() - start_time < budget_s:
        if time.time() - start_refinement > 60:
            break
        result = minimize(
            objective_continuous, final_h, method='SLSQP', bounds=bounds,
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            h_candidate = clip_and_scale(result.x)
            c5_candidate = objective_continuous(h_candidate)
            if c5_candidate < best_c5_refined:
                best_c5_refined = c5_candidate
                final_h = h_candidate.copy()

    final_h = clip_and_scale(final_h)
    return (final_h.tolist(), best_c5_refined, n_points)
