    )
    if result.success:
        current_c5 = calculate_c5(result.x, dx)
        if current_c5 < best_c5:
            best_h[:] = result.x

    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)

def calculate_c5(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
