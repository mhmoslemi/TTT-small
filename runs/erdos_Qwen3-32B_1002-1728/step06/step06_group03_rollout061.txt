            lambda p: np.max(np.correlate(compute_h(*p), 1 - compute_h(*p), 'full')) * dx,
            x0=trial,
            bounds=bounds,
            method='SLSQP',
            options={'maxiter': 200, 'ftol': 1e-10}
        )

        if result.success:
            a_new, b_new, c_new, w1_new, w2_new, l1_new, l2_new = result.x
            h_new = compute_h(a_new, b_new, c_new, w1_new, w2_new, l1_new, l2_new)
            c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_params = result.x

        # Update for next perturbation
        a, b, c, w1, w2, l1, l2 = best_params if best_params is not None else [a, b, c, w1, w2, l1, l2]

    if best_h is None:
        best_h = compute_h(*get_initial_params(h_full_initial))
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, 'full')) * dx

    return (best_h.tolist(), best_c5, n_points)
