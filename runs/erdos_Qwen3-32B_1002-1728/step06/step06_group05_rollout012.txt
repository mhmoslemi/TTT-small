import numpy as np
import scipy.interpolate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 70
    target_sum = n_points / 2.0  # 35.0

    # Generate Frank-Zadoff sequence
    N_fz = 64
    i = np.arange(N_fz)
    s = np.cos(np.pi / N_fz * ((2 * i + 1) % N_fz))  # Real Frank-Zadoff-like sequence
    binary_seq = (s >= 0).astype(float)  # Threshold to binary
    original_x = np.linspace(0, 2, N_fz)
    new_x = np.linspace(0, 2, n_points)
    f_inter = scipy.interpolate.interp1d(original_x, binary_seq, kind='linear', fill_value='extrapolate')
    interpolated_seq = f_inter(new_x)
    current_sum = np.sum(interpolated_seq)
    scaling_factor = target_sum / current_sum
    h_initial = np.clip(interpolated_seq * scaling_factor, 0, 1)

    # Optimization settings
    dx = 2.0 / n_points

    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)

        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )

    best_h = h_initial
    best_c5 = objective(h_initial)
    if result.success and time.time() - start_time < budget_s - 30:
