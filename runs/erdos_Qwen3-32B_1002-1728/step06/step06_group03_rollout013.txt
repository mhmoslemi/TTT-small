def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def process_h(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return h_clipped
        scaling_factor = target_sum / sum_h
        h_scaled = h_clipped * scaling_factor
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    def compute_c5(h_full):
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    n_particles = 30
    max_iterations = 30
    w, c1, c2 = 0.7, 1.2, 1.2
    perturbation_size = 0.05

    particles = [initial_h + np.random.randn(n_points) * perturbation_size for _ in range(n_particles)]
    particles = [process_h(p) for p in particles]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    pb_values = [compute_c5(p) for p in particles]
    global_best_idx = np.argmin(pb_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = pb_values[global_best_idx]

    iteration = 0
    while time.time() - start_time < budget_s - 60 and iteration < max_iterations:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            particles[i] = particles[i] + velocities[i]
            particles[i] = process_h(particles[i])
            value = compute_c5(particles[i])
            if value < pb_values[i]:
                personal_best[i] = particles[i].copy()
                pb_values[i] = value

        current_global_idx = np.argmin(pb_values)
        if pb_values[current_global_idx] < global_best_value:
            global_best = personal_best[current_global_idx].copy()
            global_best_value = pb_values[current_global_idx]
        iteration += 1

    top_candidates = [personal_best[i] for i in np.argsort(pb_values)[:3]]
    refined_values = []
    for candidate in top_candidates:
        def objective(x):
            x = np.clip(x, 0, 1)
            total = np.sum(x)
            if total == 0:
                return float('inf')
            scaled = (x / total) * target_sum
            scaled = np.clip(scaled, 0, 1)
            return compute_c5(scaled)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - target_sum})
        result = minimize(objective, candidate, method='SLSQP', bounds=[(0, 1)]*n_points,
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            refined = np.clip(result.x, 0, 1)
            refined_values.append((compute_c5(refined), refined))
        else:
            refined_values.append((np.inf, candidate))

    best_idx = np.argmin([rv[0] for rv in refined_values])
    best_c5 = refined_values[best_idx][0]
    best_h = process_h(refined_values[best_idx][1])
    return (best_h.tolist(), best_c5, n_points)
