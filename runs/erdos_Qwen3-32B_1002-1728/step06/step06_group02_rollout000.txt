def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_gradient(h, k_max, dx):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_max < n:
            grad[i] += (1 - h[i + k_max])
        if i - k_max >= 0:
            grad[i] -= h[i - k_max]
    grad *= dx
    return grad
