    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h, dx)
    current_tau = 1.0
    annealing_factor = 0.95
    max_inner_steps = 100

    current_h = best_h.copy()

    for step in range(max_inner_steps):
        if time.time() - start_time > budget_s - 30:
            break
        try:
            grad = finite_diff_gradient(current_h, dx, current_tau)
        except:
            continue
        learning_rate = 0.1
        current_h -= learning_rate * grad
        current_h = project_h(current_h, target_sum)
        if step % 10 == 0:
            current_c5 = evaluate_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        if step % 50 == 0:
            def objective(x):
                h_test = x.copy()
                h_test = project_h(h_test, target_sum)
                return evaluate_c5(h_test, dx)
            bounds = [(0, 1)]*n_points
            res = minimize(
                objective,
                x0=current_h,
                method='COBYLA',
                bounds=bounds,
                options={'maxiter': 200}
            )
            if res.success:
                refined_h = res.x.copy()
                refined_h = project_h(refined_h, target_sum)
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        current_tau *= annealing_factor
        current_tau = max(1e-5, current_tau)

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
