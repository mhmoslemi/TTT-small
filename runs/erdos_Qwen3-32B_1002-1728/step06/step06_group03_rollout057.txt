def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr_vals) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Use full asymmetric optimization
    h_candidate = h_full_initial.copy()
    current_sum = np.sum(h_candidate)
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        h_candidate *= (target_sum / current_sum)
        h_candidate = np.clip(h_candidate, 0, 1)
    best_c5 = objective(h_candidate)

    # Differential evolution with adaptive population size
    def de_objective(x_vars):
        h_var = np.array(x_vars)
        h_var_clipped = np.clip(h_var, 0, 1)
        h_var_clipped = h_var_clipped / np.sum(h_var_clipped) * target_sum
        h_var_clipped = np.clip(h_var_clipped, 0, 1)
        corr_vals = np.correlate(h_var_clipped, 1 - h_var_clipped, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    pop_size = min(15 + n_points // 5, 30)
    pops = [h_candidate * (np.random.rand(n_points) * 0.2 + 0.95) for _ in range(pop_size)]
    pops = [np.clip(p, 0, 1) for p in pops]
    de_result = differential_evolution(
        de_objective, bounds_per_var, strategy='rand1bin',
        maxiter=15, popsize=pop_size, tol=1e-4, mutation=(0.5, 1.0), recombination=0.8,
        seed=seed, workers=1, disp=False, init=pops
    )
    if de_result.success:
        de_h = np.clip(de_result.x, 0, 1)
        de_h = de_h / np.sum(de_h) * target_sum
        de_h = np.clip(de_h, 0, 1)
        c5_de = objective(de_h)
        if c5_de < best_c5:
            h_candidate = de_h
            best_c5 = c5_de

    # Local refinement with adaptive perturbations
    def refine(candidate):
        best = candidate.copy()
        for _ in range(10):
            perturb = np.random.randn(n_points) * 0.1 * np.random.rand()
            neighbor = candidate.copy() + perturb
            neighbor = np.clip(neighbor, 0, 1)
            scaled = neighbor / np.sum(neighbor) * target_sum
            scaled = np.clip(scaled, 0, 1)
            val = objective(scaled)
            if val < best_c5:
                best = scaled.copy()
                best_c5 = val
        return best

    # Annealing-like refinement within time budget
    while time.time() - start_time < budget_s - 10:
        step = max(0.01, (budget_s - (time.time() - start_time)) / budget_s * 0.1)
        perturb = np.random.randn(n_points) * step
        perturbed = h_candidate + perturb
        perturbed = np.clip(perturbed, 0, 1)

        val = objective(perturbed)

    if not np.isclose(final_sum, target_sum, atol=1e-6) and final_sum > 1e-8:

        h_candidate = np.clip(h_candidate, 0, 1)
