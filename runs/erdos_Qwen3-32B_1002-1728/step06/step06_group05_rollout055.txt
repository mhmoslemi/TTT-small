                final_candidates[idx] = optimized_h
                final_c5[idx] = optimized_c5

    if final_c5:
        best_idx = np.argmin(final_c5)
        best_h = final_candidates[best_idx]
        best_c5_val = final_c5[best_idx]
        if best_c5_val < best_c5:
            best_c5 = best_c5_val
            global_best = best_h

    h_full = np.clip(global_best, 0, 1)
    sum_full = np.sum(h_full)
    if sum_full != 0:
        h_full = (h_full / sum_full) * target_sum
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
