    best_c5 = compute_c5(best_binary)
    best_h = best_binary.copy()
    gen = 0
    max_gen = 300

    while time.time() - start_time < budget_s - 200 and gen < max_gen:
        gen += 1
        c5_list = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(c5_list)
        if c5_list[best_idx] < best_c5:
            best_c5 = c5_list[best_idx]
            best_h = population[best_idx].copy()

        # Selection
        n_parents = max(pop_size // 3, 10)
        parent_indices = np.argsort(c5_list)[:n_parents]
        parents = population[parent_indices]

        new_pop = [population[best_idx].copy()]
        while len(new_pop) < pop_size:
            p1, p2 = parents[np.random.choice(n_parents, 2, replace=False)]
            child = crossover(p1, p2)
            if np.random.rand() < 0.1:
                mutate(child)
            new_pop.append(child)
        population = np.array(new_pop)

    # Refinement
    bounds = [(0.0, 1.0)] * n_points
    constr = {'type': 'eq', 'fun': lambda x: np.sum(x) - T}
    result = minimize(compute_c5, x0=best_h, bounds=bounds, constraints=constr, method='SLSQP', 
                      options={'maxiter': 150, 'ftol': 1e-8, 'disp': False})
    if result.success:
        final_h = result.x
        final_c5 = result.fun
        best_h = final_h
        best_c5 = final_c5
    else:
        final_h = best_h.copy()
        final_c5 = best_c5

    return (final_h.tolist(), final_c5, n_points)
