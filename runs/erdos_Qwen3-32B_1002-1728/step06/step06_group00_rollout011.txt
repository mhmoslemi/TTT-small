            # Perturb best_h to escape local optima
            perturbed = best_h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = perturbed.sum()
            if perturbed_sum > 1e-8:
                perturbed = (perturbed / perturbed_sum) * target_sum
                perturbed = np.clip(perturbed, 0, 1)
            
            # Run SLSQP with constraints
            try:
                result = minimize(
                    compute_c5, x0=perturbed, method='SLSQP',
                    bounds=bounds_full, constraints=cons_full,
                    options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
                )
                if result.success:
                    candidate = np.clip(result.x, 0, 1)
                    candidate_sum = candidate.sum()
                    if abs(candidate_sum - target_sum) < 1e-6:
                        current_c5 = compute_c5(candidate)
                        if current_c5 < best_c5:
                            best_h = candidate
                            best_c5 = current_c5
            except:
                pass
            if time.time() - start_time > budget_s - 5:
                break

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-6:
        best_h = (best_h / best_h_sum) * target_sum
        best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), float(best_c5), n_points)
