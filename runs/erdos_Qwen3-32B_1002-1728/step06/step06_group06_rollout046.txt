import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)
    np.random.seed(seed)

    candidates = []
    num_starts = 30
    # Generate diversified starting points
    for _ in range(num_starts):
        if time.time() - start_time > budget_s - 30:
            break
        noise = np.random.normal(0, 0.1, n_points)
        c = h_initial + noise
        candidates.append(clip_and_rescale(c))

    best_c5 = float('inf')
    best_h = h_initial.copy()

    for candidate in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        def objective(x):
            x = np.reshape(x, -1)
            x = clip_and_rescale(x)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        # Feasibility constraint and bounds
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        bounds = [(0, 1)] * n_points
        result = minimize(
            objective, x0=candidate, method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': 200, 'gtol': 1e-8}
        )
        if result.success:
            x_opt = clip_and_rescale(result.x)
            c5 = np.max(np.correlate(x_opt, 1 - x_opt, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_opt.copy()

    # Final refinement on the best solution found
    if time.time() - start_time < budget_s - 10 and best_c5 < float('inf'):
        def refined_obj(x):
            x = np.reshape(x, -1)
            x = clip_and_rescale(x)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        result = minimize(
            refined_obj, x0=best_h, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result.success:
            x_opt = clip_and_rescale(result.x)
            c5 = np.max(np.correlate(x_opt, 1 - x_opt, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_opt

    return (best_h.tolist(), best_c5, n_points)
