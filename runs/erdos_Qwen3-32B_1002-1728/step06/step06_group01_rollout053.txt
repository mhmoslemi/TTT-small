import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum for half variables

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * dx

    # Initial symmetric setup
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = h_full_sym.sum()
    if abs(current_sum - (n_points / 2.0)) > 1e-6:
        scale = target_sum * 2.0 / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    # Define constraints for local optimization
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # PSO Setup and Execution
    np.random.seed(seed)
    n_particles = 30
    w, c1, c2 = 0.9, 2.0, 2.0
    positions = np.tile(h_half[None, :], (n_particles, 1))
    positions += 0.1 * np.random.rand(*positions.shape)
    positions = np.clip(positions, 0, 1)
    sum_pos = positions.sum(axis=1, keepdims=True)
    positions *= (target_sum / np.where(sum_pos == 0, 1, sum_pos))
    positions = np.clip(positions, 0, 1)
    velocities = np.zeros_like(positions)
    pbest_val = np.apply_along_axis(objective, 1, positions)
    pbest_pos = positions.copy()
    gbest_val, gbest_pos = pbest_val.min(), pbest_pos[pbest_val.argmin()].copy()

    while time.time() - start_time < budget_s - 60:
        r1, r2 = np.random.rand(n_particles, 1), np.random.rand(n_particles, 1)
        velocities = w * velocities + c1 * r1 * (pbest_pos - positions) + c2 * r2 * (gbest_pos - positions)
        positions += velocities
        positions = np.clip(positions, 0, 1)
        sum_pos = positions.sum(axis=1, keepdims=True)
        positions *= (target_sum / np.where(sum_pos == 0, 1, sum_pos))
        positions = np.clip(positions, 0, 1)
        new_vals = np.apply_along_axis(objective, 1, positions)
        for i in range(n_particles):
            if new_vals[i] < pbest_val[i]:
                pbest_val[i] = new_vals[i]
                pbest_pos[i] = positions[i].copy()
        current_gbest_idx = np.argmin(pbest_val)
        if pbest_val[current_gbest_idx] < gbest_val:
            gbest_val = pbest_val[current_gbest_idx]
            gbest_pos = pbest_pos[current_gbest_idx].copy()
        w *= 0.998

    best_h_half = gbest_pos.copy()
    best_c5 = gbest_val

    # Local refinement
    np.random.seed(seed)
    while time.time() - start_time < budget_s:
        n_attempt = int((budget_s - (time.time() - start_time)) / 10)
        noise_scale = 0.2 * (n_attempt / 100)
        for _ in range(min(3, n_attempt)):
            seed += 1
            idxs = np.random.choice(half_n, size=min(half_n, 10), replace=False)
            perturbed = best_h_half.copy()
            perturbed[idxs] += np.random.normal(0, noise_scale, len(idxs))
            perturbed = np.clip(perturbed, 0, 1)
            total = perturbed.sum()
            if total > 1e-6:
                perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
            method = ['SLSQP', 'L-BFGS-B', 'COBYLA'][np.random.randint(3)]
            result = minimize(
                fun=objective, x0=perturbed, method=method, bounds=None,
                constraints=cons, options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                new_h_half = np.clip(result.x, 0, 1)
                if abs(np.sum(new_h_half) - target_sum) < 1e-6:
                    new_c5 = objective(new_h_half)
                    if new_c5 < best_c5:
                        best_h_half = new_h_half
                        best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
