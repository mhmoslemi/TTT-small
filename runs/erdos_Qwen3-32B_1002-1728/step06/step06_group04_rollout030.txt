        for n in range(1, n_harm + 1):
            idx = 1 + 2*(n-1)
            a_n = theta[idx]
            b_n = theta[idx + 1]
            F += a_n * np.cos(np.pi * n * x_vals / 2)
            F += b_n * np.sin(np.pi * n * x_vals / 2)
        h = 1.0 / (1.0 + np.exp(-F))
        return h

        h = compute_h(theta, n_harm)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx
