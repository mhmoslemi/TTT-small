def crossover(parent1, parent2, dx, n_points, T):
    # Randomly generate a child by taking bits from parents
    # parent1 and parent2 are numpy arrays of 0 and 1.
    mask = np.random.rand(n_points) > 0.5
    child = np.where(mask, parent1, parent2).copy().astype(int)
    # Now, adjust to have T ones
    current_sum = np.sum(child)
    if current_sum == T:
        return child
    elif current_sum > T:
        # Need to flip (current_sum - T) ones to zeros
        num_to_flip = current_sum - T
        # Find positions where child is 1 and other parent is 0
        # Other parent is parent2 for the 1s in child
        flip_candidate_mask = (child == 1) & (parent2 == 0)
        indices_to_flip = np.where(flip_candidate_mask)[0]
        if len(indices_to_flip) < num_to_flip:
            # No such indices, fallback to any 1s
            indices_to_flip = np.where(child == 1)[0]
        # Randomly select num_to_flip indices to flip
        indices = np.random.choice(indices_to_flip, size=num_to_flip, replace=False)
        child[indices] = 0
    else:  # current_sum < T
        # Need to flip (T - current_sum) zeros to ones
        num_to_flip = T - current_sum
        # Find positions where child is 0 and parent2 is 1
        flip_candidate_mask = (child == 0) & (parent2 == 1)
        indices_to_flip = np.where(flip_candidate_mask)[0]
        if len(indices_to_flip) < num_to_flip:
            # No such indices, fallback to any 0s
            indices_to_flip = np.where(child == 0)[0]
        # Randomly select num_to_flip indices to flip
        indices = np.random.choice(indices_to_flip, size=num_to_flip, replace=False)
        child[indices] = 1
    return child

def mutate(individual, mutation_rate=0.01, n_points=None, T=None):
    if np.random.rand() < mutation_rate:
        # Choose two indices: one 1 and one 0
        ones_positions = np.where(individual == 1)[0]
        zeros_positions = np.where(individual == 0)[0]
        if len(ones_positions) == 0 or len(zeros_positions) == 0:
            return individual
        i1 = np.random.choice(ones_positions)
        i2 = np.random.choice(zeros_positions)
        # Swap
        individual[i1] = 0
        individual[i2] = 1
    return individual

def tournament_selection(population, fitness, tournament_size=3):
    # fitness is a list of (lower-better) values

    best_idx = indices[np.argmin(fitness[indices])]
    return population[best_idx]
