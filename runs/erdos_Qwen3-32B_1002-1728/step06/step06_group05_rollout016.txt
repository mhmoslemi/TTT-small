    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def enforce_feasibility(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        return x

    h_current = enforce_feasibility(h_initial.copy())
    best_h = h_current.copy()
    best_c5 = evaluate(best_h)

    T = 1.0
    alpha = 0.99
    max_iter = 1000
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 60:
            break
        h_new = h_current + np.random.normal(0, T, n_points)
        h_new = enforce_feasibility(h_new)
        c5_new = evaluate(h_new)
        delta = best_c5 - c5_new
        if delta < 0 or np.exp(delta / T) > np.random.rand():
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
        T *= alpha

    def objective(x):
        return evaluate(enforce_feasibility(x))

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    if result.success:
        optimized_h = enforce_feasibility(result.x)
        optimized_c5 = evaluate(optimized_h)
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            best_h = optimized_h

    return (best_h.tolist(), best_c5, n_points)
