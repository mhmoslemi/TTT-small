def compute_h(theta, n_points, N_harmonics):
    a0 = theta[0]
    a = theta[1:1 + N_harmonics]
    b = theta[1 + N_harmonics:1 + 2*N_harmonics]
    x = np.linspace(0, 2, n_points)
    h_cont = a0 * np.ones(n_points)
    for n in range(N_harmonics):
        h_cont += a[n] * np.cos((n+1) * np.pi * x) + b[n] * np.sin((n+1) * np.pi * x)
    h_sigm = 1.0 / (1.0 + np.exp(-h_cont))
    h_clipped = np.clip(h_sigm, 1e-6, 1 - 1e-6)
    target_sum = n_points / 2.0
    scaled_h = h_clipped * (target_sum / np.sum(h_clipped))
    return scaled_h

def objective(theta, n_points, N_harmonics, dx):
    h = compute_h(theta, n_points, N_harmonics)
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
