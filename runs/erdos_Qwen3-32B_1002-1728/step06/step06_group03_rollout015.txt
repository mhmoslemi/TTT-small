import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize and clip initial h
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = (h / current_sum) * target_sum
    h = np.clip(h, 0, 1)

    def compute_c5(h_vars):
        h_clipped = np.clip(h_vars, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    np.random.seed(seed)

    max_perturbations = 50
    perturbation_count = 0

    while (time.time() - start_time) < budget_s * 0.95 and perturbation_count < max_perturbations:
        perturbation_count += 1
        noise_std = 0.05 * best_h + 0.01
        noise = np.random.randn(n_points) * noise_std
        perturbed_h = best_h + noise
        new_h = np.clip(perturbed_h, 0, 1)
        total = np.sum(new_h)
        if total != 0:
            scaled_h = (new_h / total) * target_sum
        else:
            scaled_h = np.zeros_like(new_h)
        scaled_h = np.clip(scaled_h, 0, 1)

        bounds = [(0.0, 1.0)] * n_points
        cons_opt = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(
            compute_c5,
            x0=scaled_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_opt,
            options={'maxiter': 150, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if abs(refined_sum - target_sum) > 1e-6 and refined_sum != 0:

            refined_h = np.clip(refined_h, 0, 1)
            new_c5 = compute_c5(refined_h)
            if new_c5 < best_c5:

                best_c5 = new_c5
