    dx = 2.0 / n_points

    h_initial = clip_and_rescale(h_initial)

    best_candidates = []
    best_c5 = float('inf')

    np.random.seed(seed)
    n_initial = 30
    for _ in range(n_initial):
        if time.time() - start_time > budget_s - 40:
            break
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + noise
        h_perturbed = clip_and_rescale(h_perturbed)
        result = optimize_candidate(h_perturbed, n_points, dx)
        if result.success:
            h_opt = clip_and_rescale(result.x)
            c5 = evaluate_c5(h_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
            best_candidates.append((h_opt, c5))
        else:
            c5_perturbed = evaluate_c5(h_perturbed, dx)
            best_candidates.append((h_perturbed, c5_perturbed))

    best_candidates.sort(key=lambda x: x[1])
    top_k = best_candidates[:5]

    for candidate, c5 in top_k:
        if time.time() - start_time > budget_s - 20:
            break
        for _ in range(3):
            noise = np.random.normal(0, 0.2, n_points)
            h_perturbed = candidate + noise
            h_perturbed = clip_and_rescale(h_perturbed)
            result = optimize_candidate(h_perturbed, n_points, dx)
            if result.success:
                h_opt = clip_and_rescale(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt

    best_candidates.sort(key=lambda x: x[1])
    h_final = best_candidates[0][0] if best_candidates else h_initial
    c5_final = best_candidates[0][1] if best_candidates else evaluate_c5(h_initial, dx)

    return (h_final.tolist(), c5_final, n_points)
