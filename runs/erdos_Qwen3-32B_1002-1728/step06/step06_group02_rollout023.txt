import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_initial = np.array(initial_h_values).copy().astype(float)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is feasible
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-6:
        h_initial = h_initial * (target_sum / current_sum)
        h_initial = np.clip(h_initial, 0, 1)

    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Simulated Annealing parameters
    temp = 1.0
    temp_min = 1e-4
    cooling_rate = 0.998
    step_size = 0.2
    n_local_refine = 0

    while time.time() - start_time < budget_s - 10:
        # Metropolis-Hastings step
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, step_size * temp, 1)[0]
        new_h = best_h.copy()
        new_h[idx] += delta
        new_h = np.clip(new_h, 0, 1)
        current_sum_new = np.sum(new_h)
        if current_sum_new != 0:
            new_h *= target_sum / current_sum_new
        else:
            new_h[:] = target_sum / n_points
        new_h = np.clip(new_h, 0, 1)

        # Evaluate candidate
        current_c5 = evaluate_c5(best_h, dx)
        new_c5 = evaluate_c5(new_h, dx)
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / max(temp, 1e-8)):
            best_h = new_h.copy()
            if delta_c5 < 0:
                best_c5 = new_c5

        # Cooling schedule
        temp *= cooling_rate
        if temp < temp_min:
            break

        # Coordinate-wise local optimization periodically
        if (n_local_refine % 100 == 0) and (time.time() - start_time < budget_s - 60):
            def objective(x):
                x = np.clip(x, 0, 1).copy()
                current_sum = np.sum(x)
                x = x * (target_sum / current_sum) if current_sum != 0 else target_sum / n_points
                return evaluate_c5(x, dx)
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            res = minimize(objective, x0=best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 20})
            if res.success:
                x_opt = res.x
                x_opt = np.clip(x_opt, 0, 1)
                total = np.sum(x_opt)
                x_opt *= (target_sum / total) if total != 0 else target_sum / n_points
                x_opt = np.clip(x_opt, 0, 1)
                c5_opt = evaluate_c5(x_opt, dx)
                if c5_opt < best_c5:
                    best_h = x_opt.copy()
                    best_c5 = c5_opt
            n_local_refine += 1

    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    if abs(current_sum - target_sum) > 1e-6:
        best_h = best_h * (target_sum / current_sum)
    return (best_h.tolist(), best_c5, n_points)
