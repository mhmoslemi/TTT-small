import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    T = n_points / 2.0
    dx = 2.0 / n_points
    T_fixed = int(T)

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    def fix_sum(h_candidate):
        h = h_candidate.copy()
        current_sum = h.sum()
        if current_sum == T_fixed:
            return h
        elif current_sum < T_fixed:
            indices = np.where(h == 0)[0]
            needed = int(T_fixed - current_sum)
            h[indices[:needed]] = 1
        else:
            indices = np.where(h == 1)[0]
            needed = int(current_sum - T_fixed)
            h[indices[:needed]] = 0
        return h

    def generate_individual():
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:T_fixed]] = 1
        return h

    def tournament_selection(population, current_c5_list, size=3):
        indices = np.random.randint(0, len(population), size)
        best_index = indices[np.argmin(current_c5_list[indices])]
        return population[best_index].copy()

    h_initial_bin = fix_sum((h_full_initial >= 0.5).astype(float))
    population_size = 50
    population = [fix_sum(generate_individual()) for _ in range(population_size)]
    population[0] = h_initial_bin.copy()
    current_c5_list = [objective(ind) for ind in population]
    best_index = np.argmin(current_c5_list)
    best_c5 = current_c5_list[best_index]
    best_h = population[best_index].copy()

    while time.time() - start_time < budget_s - 30:
        next_population = [best_h.copy()]
        while len(next_population) < population_size:
            parent1 = tournament_selection(population, current_c5_list, size=3)
            parent2 = tournament_selection(population, current_c5_list, size=3)
            point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:point], parent2[point:]])
            child = fix_sum(child)
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                child[idx] = 1 - child[idx]
                child = fix_sum(child)
            next_population.append(child)
        population = next_population
        current_c5_list = [objective(ind) for ind in population]
        current_best_index = np.argmin(current_c5_list)
        if current_c5_list[current_best_index] < best_c5:
            best_c5 = current_c5_list[current_best_index]
            best_h = population[current_best_index].copy()

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        objective, x0=best_h, method='SLSQP', bounds=bounds,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result.success:
        new_h = clip_and_scale(result.x)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h

    best_h = clip_and_scale(best_h)
    assert abs(best_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (best_h.tolist(), best_c5, n_points)
