from scipy.optimize import minimize
import time

def enforce_feasibility(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        h_clipped += 1e-6
        h_sum += 1e-6
    h_scaled = h_clipped * (target_sum / h_sum)
    return np.clip(h_scaled, 0, 1)

def evaluate(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)

    dx = 2.0 / n_points

    best_h = h_current.copy()
    best_c5 = evaluate(best_h, n_points)

    T0 = 1.0
    alpha = 0.995
    T_min = 1e-5
