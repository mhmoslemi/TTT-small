        perturbed[idx] += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        try:
            result = minimize(
                objective_local, x0=perturbed, method='SLSQP',
                bounds=bounds_local, constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-6, 'disp': False}
            )
            if result.success:
                x_opt = result.x
                x_opt = np.clip(x_opt, 0, 1)
                new_c5 = np.max(np.correlate(x_opt, 1 - x_opt, mode='full')) * dx
                if new_c5 < best_c5:
                    best_h = x_opt.copy()
                    best_c5 = new_c5
        except:
            pass  # Continue on error
        time_remaining = budget_s - (time.time() - start_time)

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = best_h.sum()
    if abs(final_sum - target_sum) > 1e-6:
        best_h = (best_h / final_sum) * target_sum
        best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
