import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')[:n_points]
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum / 2}]

    # Initialize with symmetric average and scaling
    h_half = (h_full_initial[:half_n] + h_full_initial[-half_n:][::-1]) / 2
    scaling_factor = target_sum / (np.sum(h_half) + 1e-8)
    h_half = np.clip(h_half * scaling_factor, 0, 1)

    best_c5 = objective(h_half)
    best_h_half = h_half.copy()

    # Differential Evolution with adjusted parameters
    def de_objective(x):
        x_scaled = x / np.sum(x) * (target_sum / 2)
        x_clipped = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_clipped)
        corr = np.correlate(h_full, 1 - h_full, mode='full')[:n_points]
        return np.max(corr) * dx

    np.random.seed(seed)
    de_result = differential_evolution(de_objective, bounds_per_var, popsize=30, maxiter=40,
                                       mutation=(0.5, 1.0), strategy='best1bin', workers=1,
                                       tol=1e-5, disp=False)

    if de_result.success:
        de_x = np.clip(de_result.x, 0, 1)
        de_x_scaled = de_x / np.sum(de_x) * (target_sum / 2)
        de_x_scaled = np.clip(de_x_scaled, 0, 1)
        de_c5 = objective(de_x_scaled)
        if de_c5 < best_c5:
            best_h_half = de_x_scaled
            best_c5 = de_c5

    # Perturbation and local optimization
    np.random.seed(seed + 1)
    current_time = time.time()
    max_iter = int((budget_s - 10) / 5)
    for _ in range(max_iter):
        perturbation = np.random.normal(0, 0.15, half_n)
        perturbed = np.clip(best_h_half + perturbation, 0, 1)
        perturbed_scaled = perturbed / np.sum(perturbed) * (target_sum / 2)
        perturbed_scaled = np.clip(perturbed_scaled, 0, 1)

        # Select optimization method
        method = 'L-BFGS-B' if _ % 3 == 0 else 'SLSQP'
        result = minimize(objective, x0=perturbed_scaled, method=method, bounds=bounds_per_var,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-4})
        
        if result.success:
            candidate = np.clip(result.x, 0, 1)
            candidate_scaled = candidate / np.sum(candidate) * (target_sum / 2)
            candidate_scaled = np.clip(candidate_scaled, 0, 1)
            c5 = objective(candidate_scaled)
            if c5 < best_c5:
                best_h_half = candidate_scaled
                best_c5 = c5
                # Early exit if time is tight
                if time.time() - current_time < 10:
                    break

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - target_sum) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
