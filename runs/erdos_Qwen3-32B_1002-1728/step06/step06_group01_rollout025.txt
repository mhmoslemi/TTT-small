    coeffs = []
    current = x.copy()
    levels = int(np.log2(N))
    for level in range(levels):
        even = current[::2]
        odd = current[1::2]
        avg = 0.5 * (even + odd)
        diff = 0.5 * (even - odd)
        coeffs.extend(diff)
        current = avg
    coeffs.extend(current)
    return np.array(coeffs)

def inverse_haar_transform(coeffs, N):
    levels = int(np.log2(N))
    n = 1
    reconstructed = coeffs[-n:].copy()
    coeffs = coeffs[:-n]
    for level in reversed(range(levels)):
        n *= 2
        details = coeffs[-n//2:]
        coeffs = coeffs[:-n//2]
        expanded = np.zeros(n)
        for i in range(n // 2):
            a = reconstructed[i]
            d = details[i]
            expanded[2*i] = a + d
            expanded[2*i + 1] = a - d
        reconstructed = expanded
    return reconstructed

def make_feasible_h(theta, N, target_sum):
    h = inverse_haar_transform(theta, N)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    original_len = len(initial_h)
    N = 64  # Power of two
    x_original = np.linspace(0, 2, original_len)
    x_new = np.linspace(0, 2, N)
    initial_h_resampled = np.interp(x_new, x_original, initial_h)
    theta_initial = haar_transform(initial_h_resampled)
    target_sum = N / 2.0
    dx = 2.0 / N

    best_theta = theta_initial.copy()
    best_h = None
    best_c5 = np.inf

    def objective(theta):
        h = make_feasible_h(theta, N, target_sum)
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        top_p = max(1, int(0.2 * len(corr_vals)))
        worst_vals = np.sort(np.abs(corr_vals))[-top_p:]
        p = 20
        l_p = (np.sum(np.abs(worst_vals) ** p)) ** (1/p)
        l1 = 1e-3 * np.sum(np.abs(theta))
        return l_p + l1

    def callback(theta):
        nonlocal best_h, best_c5, best_theta
        h = make_feasible_h(theta, N, target_sum)
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
            best_theta = theta.copy()
        if time.time() - start_time > budget_s - 10:
            return True
        return False

    try:
        result = minimize(
            fun=objective,
            x0=best_theta,
            method='L-BFGS-B',
            jac=None,
            options={'maxiter': 300, 'ftol': 1e-10, 'disp': False},
            callback=callback
        )
    except:
        pass

    h_best = make_feasible_h(best_theta, N, target_sum)
    corr_vals = np.correlate(h_best, 1 - h_best, mode='full') * dx
    final_c5 = np.max(corr_vals)
    return (h_best.tolist(), final_c5, N)
