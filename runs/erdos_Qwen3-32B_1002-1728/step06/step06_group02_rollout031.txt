import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx

def compute_smooth_objective(h, dx, tau):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    scaled = corr_vals * dx
    exp_vals = np.exp(scaled / tau)
    return tau * np.log(np.sum(exp_vals))

def approximate_gradient(h, dx, tau, epsilon=1e-5):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus_clipped = np.clip(h_plus, 0, 1)
        sum_plus = np.sum(h_plus_clipped)
        h_plus_norm = h_plus_clipped * (n / 2.0 / sum_plus) if sum_plus != 0 else np.full(n, n / 2.0 / n)
        h_plus_norm = np.clip(h_plus_norm, 0, 1)
        smooth_plus = compute_smooth_objective(h_plus_norm, dx, tau)

        smooth_minus = compute_smooth_objective(h_minus_norm, dx, tau)

    return grad
