import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    T = n_points / 2.0
    T_int = int(T)
    dx = 2.0 / n_points
    np.random.seed(seed)

    def clip_and_scale_continuous(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    def evaluate(individual):
        corr_vals = np.correlate(individual, 1 - individual, mode='full')
        return float(np.max(corr_vals) * dx)

    # Initialize population
    def init_population():
        pop = []
        h_initial_binary = (h_full_initial > 0.5).astype(int)
        sum_init = h_initial_binary.sum()
        diff = T_int - sum_init
        if diff != 0:
            indices_zero = np.where(h_initial_binary == 0)[0]
            indices_one = np.where(h_initial_binary == 1)[0]
            if diff > 0:
                flip = np.random.choice(indices_zero, diff, replace=False)
                h_initial_binary[flip] = 1
            else:
                flip = np.random.choice(indices_one, -diff, replace=False)
                h_initial_binary[flip] = 0
        pop.append(h_initial_binary)
        for _ in range(49):
            indices = np.random.choice(n_points, T_int, replace=False)
            h = np.zeros(n_points, dtype=int)
            h[indices] = 1
            pop.append(h)
        return np.array(pop)

    population = init_population()
    pop_scores = np.array([evaluate(ind) for ind in population])
    best_idx = np.argmin(pop_scores)
    best_binary = population[best_idx].copy()
    best_c5 = pop_scores[best_idx]

    # Genetic Algorithm Loop
    while time.time() - start_time < budget_s - 300:
        # Selection
        parents = []
        for _ in range(50):
            a, b = np.random.choice(len(population), 2, replace=False)
            winner = a if pop_scores[a] < pop_scores[b] else b
            parents.append(population[winner])
        parents = np.array(parents)

        # Crossover & Repair
        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                cx_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:cx_point], p2[cx_point:]])
                child2 = np.concatenate([p2[:cx_point], p1[cx_point:]])
                # Repair to T_int
                def repair(h):
                    s = h.sum()

                    if diff == 0:
                        return h

                    indices1 = np.where(h == 1)[0]
                    if diff > 0:

                        h[sel] = 1

                        sel = np.random.choice(indices1, size=-diff, replace=False)
                        h[sel] = 0
                    return h

                offspring.extend([child1, child2])
            else:

        for child in offspring:
