            if optimized_c5 < best_c5:
                best_c5 = optimized_c5
                best_solution = result.x

    best_solution = np.clip(best_solution, 0, 1)
    sum_h = np.sum(best_solution)
    if sum_h != 0:
        best_solution = (best_solution / sum_h) * target_sum
    best_solution = np.clip(best_solution, 0, 1)
    return (best_solution.tolist(), best_c5, n_points)
