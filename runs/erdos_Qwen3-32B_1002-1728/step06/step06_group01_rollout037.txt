import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Since 2 * sum(h_half) = n_points / 2 => sum(h_half) = n_points / 4

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    def normalize_position(position, target_sum):
        clipped = np.clip(position, 0, 1)
        current_sum = np.sum(clipped)
        if current_sum == 0:
            return np.zeros_like(position)
        scale = target_sum / current_sum
        scaled = clipped * scale
        scaled_clipped = np.clip(scaled, 0, 1)
        return scaled_clipped

    # Initialize symmetric h_half from initial_h_values
    initial_h_half = h_full_initial[:half_n].copy()
    current_sum = np.sum(initial_h_half)
    if current_sum != 0:
        initial_h_half = (initial_h_half / current_sum) * target_sum_half
    initial_h_half = np.clip(initial_h_half, 0, 1)

    # PSO parameters
    n_particles = 30
    w = 0.729  # inertia weight
    c1 = 1.49445  # cognitive
    c2 = 1.49445  # social

    # Initialize particles and velocities
    particles = np.tile(initial_h_half, (n_particles, 1))
    perturb = np.random.uniform(-0.1, 0.1, (n_particles, half_n))
    particles += perturb
    for i in range(n_particles):
        particles[i] = normalize_position(particles[i], target_sum_half)
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, half_n))

    # Personal and global bests
    personal_best_positions = particles.copy()
    personal_best_values = [compute_c5(p) for p in particles]
    global_best_position = personal_best_positions[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)

    # PSO optimization loop
    while time.time() - start_time < budget_s - 30:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best_positions[i] - particles[i]) +
                c2 * r2 * (global_best_position - particles[i])
            )
            # Update position
            particles[i] = normalize_position(particles[i] + velocities[i], target_sum_half)

        # Update personal bests
        current_values = [compute_c5(p) for p in particles]
        for i in range(n_particles):
            if current_values[i] < personal_best_values[i]:
                personal_best_values[i] = current_values[i]
                personal_best_positions[i] = particles[i]

        # Update global best
        curr_best_idx = np.argmin(personal_best_values)
        if personal_best_values[curr_best_idx] < global_best_value:
            global_best_value = personal_best_values[curr_best_idx]
            global_best_position = personal_best_positions[curr_best_idx]

        if time.time() - start_time >= budget_s - 30:
            break

    # Local refinement
    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    result = minimize(
        fun=compute_c5,
        x0=global_best_position,
        method='SLSQP',
        bounds=bounds_per_var,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    final_h_half = result.x if result.success else global_best_position

    # Final validation
    h_full = create_full_h(final_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    best_c5 = compute_c5(final_h_half)

    return (h_full.tolist(), best_c5, n_points)
