    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(
        objective, global_best, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success:
        optimized = np.clip(result.x, 0, 1)
        s = np.sum(optimized)
        if s != 0:
            optimized = (optimized / s) * target_sum
        optimized = np.clip(optimized, 0, 1)
        if objective(optimized) < best_c5:
            best_c5 = objective(optimized)
            global_best = optimized
    return global_best, best_c5
