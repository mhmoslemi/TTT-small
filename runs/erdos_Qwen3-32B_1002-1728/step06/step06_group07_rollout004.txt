import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    best_c5 = objective(h_full_initial)
    best_h = h_full_initial.copy()

    # Fourier domain optimization
    n_terms = 20
    x = np.linspace(0, 2, n_points)
    M_cols = []
    for k in range(1, n_terms + 1):
        M_cols.append(np.cos(k * np.pi * x))
    M = np.column_stack(M_cols)
    initial_h = np.array(initial_h_values)
    a_initial = np.linalg.lstsq(M, initial_h - 0.5, rcond=None)[0]

    def fourier_obj(a):
        a = np.clip(a, -1.0, 1.0)
        h = 0.5 + M @ a
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # Constraints for Fourier optimization
    constraints = []
    for i in range(n_points):
        def cons_lower(a, idx=i):
            return 0.5 + M[idx] @ a
        def cons_upper(a, idx=i):
            return 1.0 - (0.5 + M[idx] @ a)
        constraints.append({'type': 'ineq', 'fun': cons_lower})
        constraints.append({'type': 'ineq', 'fun': cons_upper})

    bounds = [(-1.0, 1.0) for _ in range(n_terms)]
    result_fourier = minimize(
        fourier_obj, 
        x0=a_initial, 
        method='SLSQP', 
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-5}
    )

    if result_fourier.success:
        a_opt = result_fourier.x
        h_fft = 0.5 + M @ a_opt
        h_fft = clip_and_scale(h_fft)
        best_h = h_fft
        best_c5 = objective(best_h)

    # Differential evolution phase
    bounds_de = [(0.0, 1.0) for _ in range(n_points)]
    de_result = minimize(
        objective, 
        x0=best_h, 
        method='L-BFGS-B', 
        bounds=bounds_de,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if de_result.success:
        de_h = clip_and_scale(de_result.x)
        de_c5 = objective(de_h)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h

    # Final validation
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), best_c5, n_points)
