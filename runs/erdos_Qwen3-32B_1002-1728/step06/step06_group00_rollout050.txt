        h_full_initial = np.clip(h_full_initial, 0, 1)

    def create_full_h(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    def objective_feasibility(x, t):
        h_half = np.clip(x, 0, 1)
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, h_full, mode='full')
        violations = t - corr
        violations[violations < 0] = 0
        return np.max(violations)

    best_h_half = h_full_initial[:half_n].copy()
    best_t = 0.0
    t_low = 0.0
    t_high = 1.0

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    while time.time() - start_time < budget_s - 10:
        t_mid = (t_low + t_high) / 2
        def obj(x):
            return objective_feasibility(x, t_mid)
        result = minimize(obj, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            h_half_sol = np.clip(result.x, 0, 1)
            h_full_sol = create_full_h(h_half_sol)
            corr = np.correlate(h_full_sol, h_full_sol, mode='full')
            if np.min(corr) >= t_mid - 1e-6:
                best_t = t_mid
                best_h_half = h_half_sol
                t_low = t_mid
            else:
                t_high = t_mid

    h_final = create_full_h(best_h_half)
    h_final = np.clip(h_final, 0, 1)
    h_final_sum = h_final.sum()
    if abs(h_final_sum - target_sum_full) > 1e-6:
        h_final = (h_final / h_final_sum) * target_sum_full
        h_final = np.clip(h_final, 0, 1)

    corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_final.tolist(), c5_bound, n_points)
