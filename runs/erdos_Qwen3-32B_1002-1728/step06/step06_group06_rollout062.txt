    for _ in range(n_starts):
        perturbed = h_initial + np.random.normal(0, perturbation_size, n_points)
        initial_points.append(clip_and_rescale(perturbed))

    best_h = h_initial.copy()
    best_c5 = float('inf')

    for h_start in initial_points:
        if time.time() - start_time > budget_s - 60:
            break

        def objective(x):
            x = x.reshape(-1)
            overlaps = np.correlate(x, 1 - x, mode='full') * dx
            return np.max(overlaps)

        def gradient(x):
            x = x.reshape(-1)
            overlaps = np.correlate(x, 1 - x, mode='full') * dx
            max_idx = np.argmax(overlaps)
            grad = np.zeros(n_points)
            for i in range(n_points):
                term1 = (1.0 - x[i + max_idx]) * dx if i + max_idx < n_points else 0.0
                term2 = (-x[i - max_idx] * dx) if i - max_idx >= 0 else 0.0
                grad[i] = term1 + term2
            return grad

        constraints = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
        bounds = [(0, 1)] * n_points

        result = minimize(
            fun=objective,
            x0=h_start.copy().ravel(),
            method='SLSQP',
            jac=gradient,
            bounds=bounds,
            constraints=[constraints],
            options={
                'maxiter': 200,
                'gtol': 1e-10,
                'ftol': 1e-12,
                'eps': 1e-8,
                'disp': False
            }
        )
        if result.success:
            h_candidate = clip_and_rescale(result.x.reshape(-1))
            overlaps = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
            c5_candidate = np.max(overlaps)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate

    return (best_h.tolist(), best_c5, n_points)
