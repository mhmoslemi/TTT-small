import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # 35/2 = 17.5 for n=70
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def objective(y_vars):
        y = np.array(y_vars)
        sum_y = np.sum(y)
        if sum_y < target_sum_half:
            return float('inf')  # Enforce the sum constraint
        h_half = (target_sum_half / sum_y) * y
        h_full = create_full_h(h_half)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    def initial_y_from_h():
        h_half_initial = h_full_initial[:half_n]
        sum_h = np.sum(h_half_initial)
        if sum_h == 0:
            return np.zeros(half_n)
        return h_half_initial * (target_sum_half / sum_h)

    # Initial guess
    y_initial = initial_y_from_h()
    for i in range(5):
        y_initial[i] = max(y_initial[i], 0.01)
    constraints = ({'type': 'ineq', 'fun': lambda y: np.sum(y) - target_sum_half})

    # Differential Evolution with y_vars parameterization
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    np.random.seed(seed)
    de_result = differential_evolution(

        recombination=0.9, seed=seed, workers=1, disp=False
    )

    best_y = y_initial
    best_c5 = objective(best_y)
    if de_result.success and de_result.fun < best_c5:
        best_y = de_result.x
        best_c5 = de_result.fun

    # Local optimization from multiple perturbed starts
    np.random.seed(seed)

    n_attempts = 5
