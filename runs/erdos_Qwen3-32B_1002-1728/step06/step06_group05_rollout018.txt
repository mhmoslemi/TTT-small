        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx
    
    personal_best_coarse = particles_coarse.copy()
    personal_best_obj_coarse = [objective_coarse(p) for p in particles_coarse]
    global_best_idx_coarse = np.argmin(personal_best_obj_coarse)
    global_best_coarse = particles_coarse[global_best_idx_coarse].copy()
    best_c5_coarse = personal_best_obj_coarse[global_best_idx_coarse]
    
    # Coarse grid optimization
    w0_coarse = 0.9
    w_decay_coarse = (w0_coarse - 0.4) / max_iter_coarse
    
    for iter_coarse in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 120:
            break
        for i in range(pop_size_coarse):
            r1 = np.random.rand(n_coarse)
            r2 = np.random.rand(n_coarse)
            velocities_coarse[i] = (w0_coarse * velocities_coarse[i] +
                                  1.2 * r1 * (personal_best_coarse[i] - particles_coarse[i]) +
                                  1.2 * r2 * (global_best_coarse - particles_coarse[i]))
            particles_coarse[i] += velocities_coarse[i]
            particles_coarse[i] = np.clip(particles_coarse[i], 0, 1)
            sum_p = np.sum(particles_coarse[i])
            if sum_p != 0:
                particles_coarse[i] = (particles_coarse[i] / sum_p) * target_sum_coarse
            particles_coarse[i] = np.clip(particles_coarse[i], 0, 1)
        obj_vals_coarse = [objective_coarse(p) for p in particles_coarse]
        for i in range(pop_size_coarse):
            if obj_vals_coarse[i] < personal_best_obj_coarse[i]:
                personal_best_coarse[i] = particles_coarse[i].copy()
                personal_best_obj_coarse[i] = obj_vals_coarse[i]
        current_best_idx = np.argmin(personal_best_obj_coarse)
        if personal_best_obj_coarse[current_best_idx] < best_c5_coarse:
            global_best_coarse = personal_best_coarse[current_best_idx].copy()
            best_c5_coarse = personal_best_obj_coarse[current_best_idx]
        w0_coarse = max(w0_coarse - w_decay_coarse, 0.4)
    
    # Upsample to fine grid
    h_fine_initial = np.repeat(global_best_coarse, 2)
    target_sum_fine = n_points / 2
    sum_fine = np.sum(h_fine_initial)
    if sum_fine != 0:
        h_fine_initial = (h_fine_initial / sum_fine) * target_sum_fine
    h_fine_initial = np.clip(h_fine_initial, 0, 1)
    
    # Fine grid PSO
    n_fine = n_points
    pop_size_fine = 30
    max_iter_fine = 20
    particles_fine = np.zeros((pop_size_fine, n_fine))
    velocities_fine = np.random.uniform(-0.02, 0.02, (pop_size_fine, n_fine))
    particles_fine[0] = h_fine_initial.copy()
    
    for i in range(1, pop_size_fine):
        perturbation = np.random.normal(0, 0.05, n_fine) if i < 7 else np.random.uniform(-0.1, 0.1, n_fine)
        particles_fine[i] = np.clip(particles_fine[0] + perturbation, 0, 1)
        sum_p = np.sum(particles_fine[i])
        if sum_p != 0:
            particles_fine[i] = (particles_fine[i] / sum_p) * target_sum_fine
        particles_fine[i] = np.clip(particles_fine[i], 0, 1)
    
    def objective_fine(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum_fine
        x = np.clip(x, 0, 1)
        dx = 2.0 / n_fine
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx
    
    personal_best_fine = particles_fine.copy()
    personal_best_obj_fine = [objective_fine(p) for p in particles_fine]
    global_best_idx_fine = np.argmin(personal_best_obj_fine)
    global_best_fine = particles_fine[global_best_idx_fine].copy()
    best_c5_fine = personal_best_obj_fine[global_best_idx_fine]
    
    # Fine grid optimization
    w0_fine = 0.9
    w_decay_fine = (w0_fine - 0.4) / max_iter_fine
    
    for iter_fine in range(max_iter_fine):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(pop_size_fine):
            r1 = np.random.rand(n_fine)
            r2 = np.random.rand(n_fine)
            velocities_fine[i] = (w0_fine * velocities_fine[i] +
                                1.2 * r1 * (personal_best_fine[i] - particles_fine[i]) +
                                1.2 * r2 * (global_best_fine - particles_fine[i]))
            particles_fine[i] += velocities_fine[i]
            particles_fine[i] = np.clip(particles_fine[i], 0, 1)
            sum_p = np.sum(particles_fine[i])
            if sum_p != 0:
                particles_fine[i] = (particles_fine[i] / sum_p) * target_sum_fine
            particles_fine[i] = np.clip(particles_fine[i], 0, 1)
        obj_vals_fine = [objective_fine(p) for p in particles_fine]
        for i in range(pop_size_fine):
            if obj_vals_fine[i] < personal_best_obj_fine[i]:
                personal_best_fine[i] = particles_fine[i].copy()
                personal_best_obj_fine[i] = obj_vals_fine[i]
        current_best_idx = np.argmin(personal_best_obj_fine)
        if personal_best_obj_fine[current_best_idx] < best_c5_fine:
            global_best_fine = personal_best_fine[current_best_idx].copy()
            best_c5_fine = personal_best_obj_fine[current_best_idx]
        w0_fine = max(w0_fine - w_decay_fine, 0.4)
    
    # Final refinement
    h_opt = global_best_fine.copy()
    def func(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum_fine
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * (2.0 / n_fine)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum_fine
    
    if time.time() - start_time < budget_s - 10:
        result = minimize(
            fun=func,
            x0=h_opt,
            method='SLSQP',
            bounds=[(0,1) for _ in range(n_fine)],
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            sum_h = np.sum(h_opt)
            if sum_h != 0:
                h_opt = (h_opt / sum_h) * target_sum_fine
            
    h_opt = np.clip(h_opt, 0, 1)
    final_c5 = func(h_opt)
    
    return (h_opt.tolist(), final_c5, n_fine)
