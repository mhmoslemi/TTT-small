        bounds_coarse = [(0.0, 1.0)] * len(h_coarse)
        cons_coarse = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_coarse}]
        try:
            res = minimize(coarse_objective, h_coarse, bounds=bounds_coarse, constraints=cons_coarse, method='L-BFGS-B', options={'maxiter': 50})
            if res.success:
                h_coarse_opt = repair_h(res.x, target_coarse)
            else:
                h_coarse_opt = h_coarse
        except:
            h_coarse_opt = h_coarse

        h_fine_coarse = np.repeat(h_coarse_opt, 2)
        h_fine_coarse = repair_h(h_fine_coarse, target_sum_full)

        def fine_objective(x):
            h = repair_h(x.copy(), target_sum_full)
            corr_vals = np.correlate(h, 1 - h, mode='full')
            return np.max(corr_vals) * dx

        bounds_fine = [(0.0, 1.0)] * n_points
        cons_fine = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_full}]
        try:
            res_fine = minimize(fine_objective, h_fine_coarse, bounds=bounds_fine, constraints=cons_fine, method='SLSQP', options={'maxiter': 100, 'ftol': 1e-7})
            if res_fine.success:
                h_fine_opt = res_fine.x
                h_fine_opt = repair_h(h_fine_opt, target_sum_full)
                c5 = np.max(np.correlate(h_fine_opt, 1 - h_fine_opt, mode='full')) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_fine_opt
                candidates.append(h_fine_opt)
        except:
            pass

    if len(candidates) > 1 and (time.time() - start_time) < budget_s - 100:
        for i in range(len(candidates)):
            for j in range(i + 1, len(candidates)):
                if time.time() - start_time > budget_s - 50:
                    break
                h1 = candidates[i]
                h2 = candidates[j]
                for alpha in [0.25, 0.5, 0.75]:
                    blended = (1 - alpha) * h1 + alpha * h2
                    blended = repair_h(blended, target_sum_full)
                    def blend_obj(x):
                        h = repair_h(x, target_sum_full)
                        corr_vals = np.correlate(h, 1 - h, mode='full')
                        return np.max(corr_vals) * dx
                    try:
                        res_blend = minimize(blend_obj, blended, bounds=bounds_fine, constraints=cons_fine, method='L-BFGS-B', options={'maxiter': 50})
                        if res_blend.success:
                            h_blend_opt = res_blend.x
                            h_blend_opt = repair_h(h_blend_opt, target_sum_full)
                            c5 = np.max(np.correlate(h_blend_opt, 1 - h_blend_opt, mode='full')) * dx
                            if c5 < best_c5:
                                best_c5 = c5
                                best_h = h_blend_opt
                    except:
                        pass

    best_h = repair_h(best_h, target_sum_full)
    return (best_h.tolist(), best_c5, n_points)
