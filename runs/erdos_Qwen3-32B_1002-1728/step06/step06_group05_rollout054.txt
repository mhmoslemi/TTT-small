        w_current = w_max - (w_max - w_min) * (iter / max_pso_iter)
        for i in range(pop_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w_current * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = (particles[i] / sum_p) * target_sum
            particles[i] = np.clip(particles[i], 0, 1)

        obj_vals = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        current_best_idx = np.argmin(personal_best_obj)
        if personal_best_obj[current_best_idx] < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = personal_best_obj[current_best_idx]

    top_indices = np.argsort(personal_best_obj)[:K_refine]
    refined_c5 = best_c5

    for idx in top_indices:
        x0 = personal_best[idx].copy()
        res = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            x = res.x
            x = np.clip(x, 0, 1)
            total = np.sum(x)
            if total != 0:
                x = (x / total) * target_sum
            x = np.clip(x, 0, 1)
            val = objective(x)
            if val < refined_c5:
                refined_c5 = val
                global_best = x.copy()

    res = minimize(
        fun=objective,
        x0=global_best,
        method='SLSQP',
        bounds=[(0, 1)] * n_points,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if res.success:
        x = res.x
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        val = objective(x)
        if val < refined_c5:
            refined_c5 = val
            global_best = x

    h_final = global_best.copy()
    h_final = np.clip(h_final, 0, 1)
    total = np.sum(h_final)
    if total != 0:
        h_final = (h_final / total) * target_sum
    h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), refined_c5, n_points)
