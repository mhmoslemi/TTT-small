        de_h = np.clip(de_result.x, 0, 1.0)
        de_total = np.sum(de_h)
        if de_total != 0:
            de_h = (de_h / de_total) * (target_sum / 2.0)
        de_h = np.clip(de_h, 0, 1.0)
        c5_de = objective(de_h)
        if c5_de < best_c5:
            best_h_half = de_h
            best_c5 = c5_de

    # Hybrid refinement with multiple restarts and enhanced trust region
    def refine_candidate(x0, budget=200):
        x0 = np.clip(x0, 0, 1.0)
        x0_total = np.sum(x0)
        if x0_total != 0:
            x0 = (x0 / x0_total) * (target_sum / 2.0)
        x0 = np.clip(x0, 0, 1.0)
        for _ in range(int(budget / 40)):
            delta = np.random.randn(x0.size) * 0.1
            x_perturbed = np.clip(x0 + delta, 0, 1.0)
            res = minimize(
                objective,
                x_perturbed,
                method='L-BFGS-B',
                bounds=bounds_per_var,
                constraints=constraint_sum,
                options={'ftol': 1e-10, 'maxiter': 20}
            )
            if res.success:
                candidate = np.clip(res.x, 0, 1.0)
                c5_candidate = objective(candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h_half = candidate

    # Run refinement intensively until timer
    while time.time() - start_time < budget_s - 20:
        refine_candidate(best_h_half, budget=60)

    # Ensuring sum constraint and clipping
    h_best_full = create_full_h(np.clip(best_h_half, 0, 1.0))
    h_best_full = (h_best_full / np.sum(h_best_full)) * target_sum
    h_best_full = np.clip(h_best_full, 0, 1.0)

    return (h_best_full.tolist(), best_c5, n_points)
