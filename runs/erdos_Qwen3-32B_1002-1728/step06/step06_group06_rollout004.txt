import scipy.optimize as optimize

def project_h(h, target_sum):
    for _ in range(3):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.clip(target_sum / h.size, 0, 1)
        scale = target_sum / current_sum
        h = h_clipped * scale
    return np.clip(h, 0, 1)

def objective_function(x, target_sum, dx):
    x_proj = project_h(x, target_sum)
    corr_vals = np.correlate(x_proj, 1 - x_proj, mode='full')
    return np.max(corr_vals) * dx
