import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Repair initial solution
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = h_initial.sum()
    if current_sum != 0:
        h_initial = (h_initial / current_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    if abs(h_initial.sum() - target_sum) > 1e-6:
        h_initial = (h_initial / h_initial.sum()) * target_sum
        h_initial = np.clip(h_initial, 0, 1)

    # Objective with penalty for constraint violation
    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        penalty = 1e4 * (current_sum - target_sum)**2
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx + penalty

    bounds = [(0.0, 1.0)] * n_points
    np.random.seed(seed)

    # Global search with adaptive DE
    de_popsize = 12
    de_maxiter = 20
    de_budget = min(500, int(0.4 * budget_s))
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=de_maxiter, popsize=de_popsize,
            mutation=(0.5, 1.2), recombination=0.85,
            tol=1e-4, seed=seed, workers=1, polish=False
        )
        global_h = de_result.x
    except:
        global_h = h_initial.copy()

    # Local refinement with SLSQP
    def local_objective(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return h.sum() - target_sum

    cons = ({'type': 'eq', 'fun': constraint_sum})
    remaining_time = max(0, budget_s - (time.time() - start_time))
    maxiter_local = int(remaining_time / 0.1)
    maxiter_local = min(maxiter_local, 500)

    local_result = minimize(
        local_objective, x0=global_h, bounds=bounds,
        constraints=cons, method='SLSQP',
        options={'maxiter': maxiter_local, 'ftol': 1e-12}
    )

    # Final repair and evaluation
    if local_result.success:
        final_h = local_result.x
    else:
        final_h = global_h

    final_h = np.clip(final_h, 0, 1)
    final_sum = final_h.sum()
    if final_sum != 0:
        final_h = (final_h / final_sum) * target_sum
    final_h = np.clip(final_h, 0, 1)

    # Compute final C5
    corr_vals = np.correlate(final_h, 1 - final_h, mode='full')
    best_c5 = np.max(corr_vals) * dx

    return (final_h.tolist(), best_c5, n_points)
