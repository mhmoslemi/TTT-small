from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    h_initial = np.clip(h_initial, 0, 1)
    h_sum = h_initial.sum()
    if h_sum != 0:
        h_initial = (h_initial / h_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    h_sum = h_initial.sum()
    if abs(h_sum - target_sum) > 1e-6:
        h_initial = (h_initial / h_sum) * target_sum
        h_initial = np.clip(h_initial, 0, 1)

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = np.flip(h_half)
        h_full = np.clip(h_full, 0, 1)
        return h_full

    def objective(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - (target_sum / 2)})

    best_h_half = h_initial[:half_n].copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    popsize = 25
    maxiter = 15

    de_budget_s = budget_s * 0.5
    try:
        de_result = differential_evolution(
            lambda x: objective(x), bounds, strategy='best1bin',
            maxiter=maxiter, popsize=popsize, mutation=(0.6, 1.5),
            recombination=0.9, tol=1e-5, seed=seed, workers=1
        )
        if de_result.success and de_result.fun < best_c5:
            candidate = de_result.x.copy()
            candidate = np.clip(candidate, 0, 1)
            sum_candidate = candidate.sum()
            if sum_candidate != 0:
                candidate = (candidate / sum_candidate) * (target_sum / 2)
            candidate = np.clip(candidate, 0, 1)
            best_h_half = candidate
            best_c5 = objective(candidate)
    except:
        pass

    local_budget_s = budget_s * 0.4
    while time.time() - start_time < budget_s - 10:
        perturbed = best_h_half.copy()
        idxs = np.random.choice(half_n, size=int(half_n * 0.3), replace=False)
        noise = np.random.normal(0, 0.2, len(idxs))
        perturbed[idxs] = np.clip(perturbed[idxs] + noise, 0, 1)
        sum_pert = perturbed.sum()
        if sum_pert != 0:
            perturbed = (perturbed / sum_pert) * (target_sum / 2)
        perturbed = np.clip(perturbed, 0, 1)

        res = minimize(

            constraints=cons, options={'maxiter': 300, 'ftol': 1e-12}
