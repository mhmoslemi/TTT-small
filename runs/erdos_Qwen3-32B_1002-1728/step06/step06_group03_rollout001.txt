from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = np.clip(h, 0, 1)
        curr_corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(curr_corr)

    # Genetic Algorithm Setup
    np.random.seed(seed)
    pop_size = 24
    population = np.array([h_initial + 0.1 * np.random.randn(n_points) for _ in range(pop_size)])
    population = np.clip(population, 0, 1)
    for i in range(pop_size):
        s = np.sum(population[i])
        if s != 0:
            population[i] = (population[i] / s) * target_sum
        population[i] = np.clip(population[i], 0, 1)

    # Evolution Loop
    while time.time() - start_time < 0.7 * budget_s:
        fits = np.array([objective(p) for p in population])
        top_indices = np.argsort(fits)[:pop_size//2]
        parents = population[top_indices]

        new_population = parents.copy()
        for _ in range(pop_size - len(new_population)):
            p1, p2 = np.random.choice(len(parents), 2)
            child = 0.5 * parents[p1] + 0.5 * parents[p2]

                mut = np.random.randn(n_points) * 0.1
                child += mut
            child = np.clip(child, 0, 1)

                child = (child / s) * target_sum

    best_h = min(population, key=objective)

    # Local refinement
    def refine_func(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
