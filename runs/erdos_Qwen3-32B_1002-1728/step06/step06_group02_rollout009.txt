def find_max_k_and_corr(h, dx):
    n = len(h)
    max_corr = -float('inf')
    best_k = 0
    for k in range(n):
        corr = 0.0
        for i in range(n):
            j = (i + k) % n
            corr += h[i] * (1 - h[j]) * dx
        if corr > max_corr:
            max_corr = corr
            best_k = k
    return max_corr, best_k

def compute_overlap_gradient(h, current_k, dx):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        j = (i + current_k) % n
        m = (i - current_k) % n
        grad[i] = (1 - h[j]) * dx - h[m] * dx
