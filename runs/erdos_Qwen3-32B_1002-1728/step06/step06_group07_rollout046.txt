    dx = 2.0 / n_points
    T = n_points / 2.0  # Total sum of h

    # Set up Fourier basis parameters
    N = 20  # Number of Fourier terms
    x = np.arange(n_points) * dx  # Sample points

    # Precompute Fourier basis (cos and sin) for each term and position
    cos_basis = np.zeros((N, n_points))
    sin_basis = np.zeros((N, n_points))
    for k in range(1, N + 1):
        k_freq = k * np.pi / 2.0
        cos_basis[k-1, :] = np.cos(k_freq * x)
        sin_basis[k-1, :] = np.sin(k_freq * x)

    # Compute initial Fourier coefficients from h_initial (excluding a0)
    a0 = 0.5  # Fixed to satisfy ∫h(x)dx = 1
    a_opt_initial = np.zeros(N)
    b_opt_initial = np.zeros(N)
    for k in range(N):
        k_idx = k
        a_opt_initial[k] = np.sum(h_initial * cos_basis[k_idx]) * dx
        b_opt_initial[k] = np.sum(h_initial * sin_basis[k_idx]) * dx

    # Precompute coefficients for linear constraints
    A = np.array([np.sum(cos_basis[k]) for k in range(N)])
    B = np.array([np.sum(sin_basis[k]) for k in range(N)])

    def compute_h(ab):
        a = ab[:N]
        b = ab[N:]
        h = a0 * np.ones(n_points) + np.sum(a[:, np.newaxis] * cos_basis + b[:, np.newaxis] * sin_basis, axis=0)
        # Ensure numerical stability
        h = np.clip(h, 0, 1)
        return h

    def objective(ab):
        h = compute_h(ab)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr_vals) * dx)

    b_ub = (1 - a0) * np.ones(n_points)

    # Inequality constraints: h[i] >= 0 and <= 1
