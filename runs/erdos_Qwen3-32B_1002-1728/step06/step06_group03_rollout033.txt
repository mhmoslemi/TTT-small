            return np.max(np.correlate(x, 1 - x, mode='full')) * dx

        try:
            result = minimize(
                objective_opt, x0=new_h, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                optimized_h = np.clip(result.x, 0, 1)
                # Re-normalize to ensure constraint
                total = np.sum(optimized_h)
                if total != 0:
                    optimized_h = (optimized_h / total) * target_sum
                optimized_h = np.clip(optimized_h, 0, 1)
                optimized_c5 = objective_opt(optimized_h)
                if optimized_c5 < best_c5:
                    best_h = optimized_h.copy()
                    best_c5 = optimized_c5
        except:
            pass  # Skip failed optimizations

        # Early exit if time is running out
        if time.time() - start_time > budget_s * 0.85:
            break

    # Final validation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    total = np.sum(final_h)
    if total != 0:
        final_h = (final_h / total) * target_sum
    final_h = np.clip(final_h, 0, 1)
    assert np.isclose(np.sum(final_h), target_sum, atol=1e-6), "Constraint violation"

    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx

    return (final_h.tolist(), final_c5, n_points)
