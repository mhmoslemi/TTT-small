    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective_time(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    # Frequency domain optimization setup
    N_terms = 20
    x_vals = np.linspace(0, 2.0 - dx, n_points)
    a0 = 0.5

    # Compute initial Fourier coefficients
    features = []
    for x in x_vals:
        row = []
        for n in range(1, N_terms + 1):
            row.append(np.cos(n * np.pi * x))
            row.append(np.sin(n * np.pi * x))
        features.append(row)
    features_matrix = np.array(features)
    target_vals = h_init - a0
    coefs_freq, _ = np.linalg.lstsq(features_matrix, target_vals, rcond=None)

    # Objective function for frequency domain optimization
    def objective_freq(param_vec):
        h_freq = np.zeros(n_points)
        for i in range(n_points):
            xi = x_vals[i]
            val = a0
            idx = 0
            for n in range(1, N_terms + 1):
                a_n = param_vec[idx]
                b_n = param_vec[idx + 1]
                val += a_n * np.cos(n * np.pi * xi) + b_n * np.sin(n * np.pi * xi)
                idx += 2
            h_freq[i] = val
        h_processed = clip_and_scale(h_freq)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return np.max(corr_vals) * dx

    # Frequency domain optimization
    param0 = coefs_freq.reshape(-1)
    bounds_freq = [(-1.0, 1.0) for _ in range(2 * N_terms)]
    time_freq_start = time.time()
    result_freq = minimize(
        objective_freq, x0=param0, bounds=bounds_freq,
        method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-6}
    )

    # Convert optimized frequency domain to time domain
    h_freq_opt = np.zeros(n_points)
    for i in range(n_points):
        xi = x_vals[i]
        val = a0
        idx = 0
        for n in range(1, N_terms + 1):
            if idx + 1 >= len(result_freq.x):
                break
            a_n, b_n = result_freq.x[idx], result_freq.x[idx + 1]
            val += a_n * np.cos(n * np.pi * xi) + b_n * np.sin(n * np.pi * xi)
            idx += 2
        h_freq_opt[i] = val
    h_clipped = clip_and_scale(h_freq_opt)

    best_c5 = objective_time(h_clipped)
    best_h = h_clipped.copy()

    # Time domain refinement with SLSQP
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective_time, x0=best_h, method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = objective_time(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), best_c5, n_points)
