def run(seed=42, budget_s=1100, **kwargs):

    def repair(h_half):
        h_half = np.clip(h_half, 0, 1)
        current_sum = h_half.sum()
        if current_sum == 0:
            h_half[0] = 1e-8
            current_sum = 1e-8
        scale = target_sum_half / current_sum
        h_half = h_half * scale
        h_half = np.clip(h_half, 0, 1)
        return h_half

    # Create full symmetric h from half
    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    # C5 evaluation
    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population = [h_full_initial[:half_n].copy()]

        perturbed = repair(perturbed)

    best_c5 = evaluate_c5(population[0])
    best_h_half = population[0].copy()

    # Genetic algorithm phase
    n_generations = 200
    pop_size = 20
    tournament_size = 3
    mutation_rate = 0.1
    np.random.seed(seed)

    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 5:

        # Evaluate fitness
        fitness = [evaluate_c5(ind) for ind in population]

        # Generate next generation
        for _ in range(pop_size):

            participants = np.random.choice(pop_size, size=tournament_size, replace=False)
