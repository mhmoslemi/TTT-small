import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Initial validation and normalization
    h_initial = np.clip(h_full_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if current_sum != 0 and abs(current_sum - target_sum) > 1e-6:
        h_initial = (h_initial / current_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    # Perturb and refine iteratively
    num_perturb = 50
    for iter in range(num_perturb):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate perturbation with variable-dependent scale
        scale = 0.05 * best_h + 0.01
        perturbation = np.random.normal(0, 1, n_points) * scale
        candidate = best_h + perturbation
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        if current_sum != 0:
            candidate = (candidate / current_sum) * target_sum
        candidate = np.clip(candidate, 0, 1)

        # Define optimization constraints and bounds
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Perform local optimization with SLSQP
        result = minimize(
            compute_c5,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            optimized = np.clip(result.x, 0, 1)
            c5_candidate = compute_c5(optimized)
            if c5_candidate < best_c5:
                best_h = optimized.copy()
                best_c5 = c5_candidate

    # Final validation
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-6:
        best_h = (best_h / final_sum) * target_sum
    best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
