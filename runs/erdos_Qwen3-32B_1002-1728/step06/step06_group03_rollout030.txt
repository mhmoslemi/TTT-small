import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # sum(h_full) = n_points/2 => sum(h_half) = n_points/4

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def normalize_h_half(h_half):
        clipped = np.clip(h_half, 0, 1)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:
            return h_half
        scaled = (clipped / sum_clipped) * target_sum_half
        return np.clip(scaled, 0, 1)

    def evaluate_fitness(h_half):
        h_full = np.clip(create_full_h(h_half), 0, 1)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * (2.0 / n_points)

    n_particles = 30
    pos = [normalize_h_half(h_full_initial[:half_n] + np.random.randn(half_n)*0.1) for _ in range(n_particles)]
    vel = [np.random.uniform(-0.05, 0.05, half_n) for _ in range(n_particles)]
    pbest_pos, pbest_score = [p.copy() for p in pos], [evaluate_fitness(p) for p in pos]
    gbest_idx = np.argmin(pbest_score)
    gbest_pos, gbest_score = pbest_pos[gbest_idx].copy(), pbest_score[gbest_idx]

    np.random.seed(seed)
    while time.time() - start_time < budget_s * 0.9:
        phi_p, phi_g = 1.49, 1.49  # Standard PSO parameters
        for i in range(n_particles):
            r_p, r_g = np.random.rand(half_n), np.random.rand(half_n)
            vel[i] = 0.7 * vel[i] + phi_p * r_p * (pbest_pos[i] - pos[i]) + phi_g * r_g * (gbest_pos - pos[i])

    # Local refinement on top candidates
