import time
from scipy.optimize import minimize, OptimizeResult
import math

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is feasible
    h_initial = np.clip(h_initial, 0, 1)
    h_sum = np.sum(h_initial)
    if h_sum > 1e-8:
        h_initial = (h_initial / h_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Differential Evolution parameters
    pop_size = 30
    F = 0.8
    CR = 0.9
    max_iter = 200

    np.random.seed(seed)
    population = np.array([h_initial.copy() for _ in range(pop_size // 2)])
    # Perturb initial population for diversity
    for i in range(pop_size // 2, pop_size):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_initial + perturbation, 0, 1)
        h_sum_p = np.sum(h_perturbed)
        if h_sum_p > 1e-8:
            h_perturbed = (h_perturbed / h_sum_p) * target_sum
        h_perturbed = np.clip(h_perturbed, 0, 1)
        population = np.vstack([population, h_perturbed])

    def compute_c5(h_vec):
        h = np.clip(h_vec, 0, 1).copy()
        h_sum = np.sum(h)
        if h_sum > 1e-8:
            h = (h / h_sum) * target_sum
        h = np.clip(h, 0, 1)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # Evaluate initial population
    c5_values = np.array([compute_c5(ind) for ind in population])
    personal_best = population.copy()
    personal_best_c5 = c5_values.copy()
    global_best_idx = np.argmin(personal_best_c5)
    global_best = population[global_best_idx].copy()
    best_c5 = personal_best_c5[global_best_idx]

    # Differential Evolution loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break

            a, b, c = population[np.random.choice(idxs, 3, replace=False)]
            donor = a + F * (b - c)

            # Crossover
            trial = np.zeros_like(donor)
            crossover_mask = np.random.rand(n_points) < CR
            crossover_mask[np.argmin(crossover_mask)] = True  # Ensure at least one parameter is taken
            trial[crossover_mask] = donor[crossover_mask]
            trial[~crossover_mask] = population[i][~crossover_mask]

            # Constraint handling
            trial_clipped = np.clip(trial, 0, 1)

                trial_scaled = (trial_clipped / trial_sum) * target_sum
            else:
                trial_scaled = trial_clipped.copy()
            trial_scaled = np.clip(trial_scaled, 0, 1)

            return np.sum(x) - target_sum
