        h_values[i] = scaled_h
    
    # Final normalization and clipping
    h_values = np.clip(h_values, 0.0, 1.0)
    total = np.sum(h_values)
    if total > 1e-6:
        h_values = h_values / total * (n_points / 2.0)
    h_values = np.clip(h_values, 0.0, 1.0)
    
    # Final C5 evaluation
    corr_vals = np.correlate(h_values, 1.0 - h_values, mode='full') * dx
    c5_bound = np.max(corr_vals)
    
    # Budget check and validation
    if time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            # Rapid annealing-style perturbation if time remains
            perturbed_params = best_params * (1.0 + 0.1 * (2 * np.random.rand(7) - 1))
            perturbed_params[2] = np.clip(perturbed_params[2], 0.2, 1.8)
            perturbed_params[3:-2] = np.clip(perturbed_params[3:-2], 0.0001, 0.8)
            perturbed_params[-2:] = np.clip(perturbed_params[-2:], 1.0, 3.0)
            alt_c5 = parametric_objective(perturbed_params, n_points)
            if alt_c5 < c5_bound:
                a,b,c,w1,w2,l1,l2 = perturbed_params
                for i in range(n_points):
                    x = i * dx
                    if x < c:
                        d = abs(x - c)
                        h_left = a * (1.0 - (d / (w1 / 2.0))) if w1 > 0 else a
                        h_left = np.clip(min(1.0, h_left), 0.0, 1.0)
                        scaled_h = h_left * np.exp(-l1 * x)
                    else:
                        d = abs(x - c)
                        h_right = b * (1.0 - (d / (w2 / 2.0))) if w2 > 0 else b
                        h_right = np.clip(min(1.0, h_right), 0.0, 1.0)
                        scaled_h = h_right * np.exp(-l2 * (x - c))
                    h_values[i] = scaled_h
                h_values = np.clip(h_values, 0.0, 1.0)
                total = np.sum(h_values)
                if total > 1e-6:
                    h_values = h_values / total * (n_points / 2.0)
                h_values = np.clip(h_values, 0.0, 1.0)
                corr_vals = np.correlate(h_values, 1.0 - h_values, mode='full') * dx
                c5_bound = np.max(corr_vals)
    
    return (h_values.tolist(), c5_bound, n_points)
