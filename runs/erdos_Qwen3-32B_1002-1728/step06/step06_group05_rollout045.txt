import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = (h_initial / np.sum(h_initial)) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Differential Evolution parameters
    pop_size = 30
    F = 0.8
    CR = 0.9
    max_gen = 200

    # Initialize population
    np.random.seed(seed)
    population = np.zeros((pop_size, n_points))
    population[0] = h_initial.copy()
    for i in range(1, pop_size):
        perturbation = np.random.normal(0, 0.1, n_points)
        candidate = np.clip(h_initial + perturbation, 0, 1)
        candidate = (candidate / np.sum(candidate)) * target_sum
        candidate = np.clip(candidate, 0, 1)
        population[i] = candidate

    # Evaluate initial population
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    fitness = np.array([evaluate(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C5 = fitness[best_idx]

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 60:
            break  # Reserve 60s for final refinement
        for i in range(pop_size):
            # Mutation
            idxs = [j for j in range(pop_size) if j != i]
            a, b, c = population[np.random.choice(idxs, 3, replace=False)]
            mutant = a + F * (b - c)

            # Crossover
            cross_points = np.random.rand(n_points) < CR
            if not np.any(cross_points):  # Ensure at least one parameter is selected
                cross_points[np.random.randint(0, n_points)] = True
            trial = np.where(cross_points, mutant, population[i])

            # Constraint handling
            trial = np.clip(trial, 0, 1)
            total = np.sum(trial)
            if total != 0:
                trial = (trial / total) * target_sum
            trial = np.clip(trial, 0, 1)

            # Selection
            trial_fitness = evaluate(trial)
            if trial_fitness < fitness[i]:
                population[i] = trial.copy()
                fitness[i] = trial_fitness
                if trial_fitness < best_C5:
                    best_h = trial.copy()
                    best_C5 = trial_fitness

    # Final refinement using SLSQP on top candidates
    top_candidates = population[np.argsort(fitness)[:5]]
    for candidate in top_candidates:
        def objective(x):
            x = np.clip(x, 0, 1)
            total = np.sum(x)
            if total != 0:
                x = (x / total) * target_sum
            x = np.clip(x, 0, 1)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        def constraint_sum(x):
            return np.sum(x) - target_sum

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            refined = np.clip(result.x, 0, 1)
            refined_total = np.sum(refined)
            if refined_total != 0:
                refined = (refined / refined_total) * target_sum
            refined = np.clip(refined, 0, 1)
            refined_C5 = objective(refined)
            if refined_C5 < best_C5:
                best_h = refined
                best_C5 = refined_C5

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total != 0:
        best_h = (best_h / total) * target_sum
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), best_C5, n_points)
