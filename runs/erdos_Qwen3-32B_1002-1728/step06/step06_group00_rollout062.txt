    best_h_half = h_initial_final[:half_n].copy()

    def create_full_h(sym_vars):
        return np.r_[sym_vars, sym_vars[::-1]]

    def scale_and_clip(x):
        x_sum = np.sum(x)
        if x_sum == 0:
            return np.zeros_like(x)

        return np.clip(x_scaled, 0, 1)

    def objective(x):
