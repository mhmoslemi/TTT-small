        if sum_p != 0:
            particles_coarse[i] = (particles_coarse[i] / sum_p) * (n_coarse / 2)
    particles_coarse = np.clip(particles_coarse, 0, 1)
    velocities_coarse = np.random.uniform(-0.05, 0.05, (pop_coarse, n_coarse))

    def objective_coarse(x):
        x = np.clip(x, 0, 1).copy()
        sum_x = np.sum(x)
        if sum_x != 0:
            x = (x / sum_x) * (n_coarse / 2)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * (2.0 / n_coarse)

    personal_best_coarse = particles_coarse.copy()
    personal_best_obj_coarse = np.array([objective_coarse(p) for p in particles_coarse])
    global_best_idx_coarse = np.argmin(personal_best_obj_coarse)
    global_best_coarse = particles_coarse[global_best_idx_coarse].copy()
    best_c5_coarse = personal_best_obj_coarse[global_best_idx_coarse]

    for iter in range(iter_coarse):
        if time.time() - start_time > budget_s - 60:
            break
        w_current = w_coarse_start - ((w_coarse_start - w_coarse_end) * iter / iter_coarse)
        for i in range(pop_coarse):
            r1 = np.random.rand(n_coarse)
            r2 = np.random.rand(n_coarse)
            velocities_coarse[i] = (w_current * velocities_coarse[i] +
                                    c1_coarse * r1 * (personal_best_coarse[i] - particles_coarse[i]) +
                                    c2_coarse * r2 * (global_best_coarse - particles_coarse[i]))
            particles_coarse[i] += velocities_coarse[i]
            particles_coarse[i] = np.clip(particles_coarse[i], 0, 1)
            sum_p = np.sum(particles_coarse[i])
            if sum_p != 0:
                particles_coarse[i] = (particles_coarse[i] / sum_p) * (n_coarse / 2)
        particles_coarse = np.clip(particles_coarse, 0, 1)
        obj_vals_coarse = [objective_coarse(p) for p in particles_coarse]
        for i in range(pop_coarse):
            if obj_vals_coarse[i] < personal_best_obj_coarse[i]:
                personal_best_coarse[i] = particles_coarse[i].copy()
                personal_best_obj_coarse[i] = obj_vals_coarse[i]
        current_best_idx = np.argmin(personal_best_obj_coarse)
        current_best_c5 = personal_best_obj_coarse[current_best_idx]
        if current_best_c5 < best_c5_coarse:
            global_best_coarse = personal_best_coarse[current_best_idx].copy()
            best_c5_coarse = current_best_c5

    h_fine_upscaled = np.zeros(n_fine)
    for i in range(n_coarse):
        h_fine_upscaled[2 * i] = global_best_coarse[i]
        h_fine_upscaled[2 * i + 1] = global_best_coarse[i]

    # Fine grid PSO parameters
    pop_fine = 20
    iter_fine = 20
    w_fine_start = 0.8
    w_fine_end = 0.4
    c1_fine = 1.5
    c2_fine = 1.5

    np.random.seed(seed + 1)
    particles_fine = np.zeros((pop_fine, n_fine))
    velocities_fine = np.zeros((pop_fine, n_fine))
    particles_fine[0] = h_fine_upscaled.copy()

    for i in range(1, pop_fine):
        perturbation = np.random.normal(0, 0.05, n_fine)
        particles_fine[i] = np.clip(h_fine_upscaled + perturbation, 0, 1)
        sum_p = np.sum(particles_fine[i])
        if sum_p != 0:
            particles_fine[i] = (particles_fine[i] / sum_p) * target_sum_fine
    particles_fine = np.clip(particles_fine, 0, 1)
    velocities_fine = np.random.uniform(-0.02, 0.02, (pop_fine, n_fine))

    def objective_fine(x):
        x = np.clip(x, 0, 1).copy()
        sum_x = np.sum(x)
        if sum_x != 0:
            x = (x / sum_x) * target_sum_fine
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * (2.0 / n_fine)

    personal_best_fine = particles_fine.copy()
    personal_best_obj_fine = np.array([objective_fine(p) for p in particles_fine])
    global_best_idx_fine = np.argmin(personal_best_obj_fine)
    global_best_fine = particles_fine[global_best_idx_fine].copy()
    best_c5_fine = personal_best_obj_fine[global_best_idx_fine]

    for iter in range(iter_fine):
        if time.time() - start_time > budget_s - 60:
            break
        w_current = w_fine_start - ((w_fine_start - w_fine_end) * iter / iter_fine)
        for i in range(pop_fine):
            r1 = np.random.rand(n_fine)
            r2 = np.random.rand(n_fine)
            velocities_fine[i] = (w_current * velocities_fine[i] +
                                  c1_fine * r1 * (personal_best_fine[i] - particles_fine[i]) +
                                  c2_fine * r2 * (global_best_fine - particles_fine[i]))
            particles_fine[i] += velocities_fine[i]
            particles_fine[i] = np.clip(particles_fine[i], 0, 1)
            sum_p = np.sum(particles_fine[i])
            if sum_p != 0:
                particles_fine[i] = (particles_fine[i] / sum_p) * target_sum_fine
        particles_fine = np.clip(particles_fine, 0, 1)
        obj_vals_fine = [objective_fine(p) for p in particles_fine]
        for i in range(pop_fine):
            if obj_vals_fine[i] < personal_best_obj_fine[i]:
                personal_best_fine[i] = particles_fine[i].copy()
                personal_best_obj_fine[i] = obj_vals_fine[i]
        current_best_idx = np.argmin(personal_best_obj_fine)
        current_best_c5_fine = personal_best_obj_fine[current_best_idx]
        if current_best_c5_fine < best_c5_fine:
            global_best_fine = personal_best_fine[current_best_idx].copy()
            best_c5_fine = current_best_c5_fine

    def fun_fine(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x != 0:
            x = (x / sum_x) * target_sum_fine
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx_fine

    def constraint_sum_fine(x):
        return np.sum(x) - target_sum_fine

    result_fine = minimize(
        fun=fun_fine,
        x0=global_best_fine,
        method='SLSQP',
        bounds=[(0, 1)] * n_fine,
        constraints={'type': 'eq', 'fun': constraint_sum_fine},
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    if result_fine.success:
        h_opt = np.clip(result_fine.x, 0, 1)
        sum_opt = np.sum(h_opt)
        if sum_opt != 0:
            h_opt = (h_opt / sum_opt) * target_sum_fine
        h_opt = np.clip(h_opt, 0, 1)
        c5_opt = fun_fine(h_opt)
        if c5_opt < best_c5_fine:
            best_c5_fine = c5_opt
            global_best_fine = h_opt

    h_full = np.clip(global_best_fine, 0, 1)
    h_full_sum = np.sum(h_full)
    if h_full_sum != 0:
        h_full = (h_full / h_full_sum) * target_sum_fine
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5_fine, n_fine)
