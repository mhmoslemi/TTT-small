from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    def normalize_h(h):

        current_sum = np.sum(h_clipped)
        if current_sum < 1e-10:
            return h_clipped.copy()
        h_scaled = (h_clipped / current_sum) * target_sum
        return np.clip(h_scaled, 0, 1)

    # Genetic Algorithm parameters
    population_size = 30
    mutation_rate = 0.2
    np.random.seed(seed)

    # Initialize population with perturbations
    def generate_perturbed(h_base):
        perturbation = np.random.normal(0, 0.05, size=h_base.shape)
        h = h_base + perturbation
        return normalize_h(h)

    population = [generate_perturbed(h_initial) for _ in range(population_size)]

    best_h = None

    while time.time() - start_time < budget_s * 0.7:
        # Evaluate fitness
        fitnesses = [objective(ind) for ind in population]
        min_idx = np.argmin(fitnesses)

        # Selection (tournament style)
        selected_indices = np.argsort(fitnesses)[:population_size // 2]

        next_population = parents[:]

            parent1, parent2 = np.random.choice(parents, 2)
