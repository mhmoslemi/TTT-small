import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_vars):
        h_clipped = np.clip(h_vars, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        h_scaled = (h_clipped / total) * target_sum
        h_final = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_final, 1 - h_final, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    h_normalized = np.clip(h_initial.copy(), 0, 1)
    h_total = np.sum(h_normalized)
    if h_total != 0:
        h_normalized = (h_normalized / h_total) * target_sum
    h_initial = np.clip(h_normalized, 0, 1)

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    # Differential evolution with adjusted parameters
    de_result = differential_evolution(
        objective, bounds,
        strategy='best1bin',
        maxiter=40,
        popsize=25,
        mutation=(0.8, 1.5),
        recombination=0.95,
        seed=seed,
        workers=1,
        disp=False,
        polish=False
    )

    de_solution = np.clip(de_result.x, 0, 1)
    de_total = np.sum(de_solution)
    if de_total != 0:
        de_solution = (de_solution / de_total) * target_sum
    de_solution = np.clip(de_solution, 0, 1)
    de_c5 = objective(de_solution)
    if de_c5 < best_c5:
        best_h = de_solution.copy()
        best_c5 = de_c5

    # SLSQP local refinement
    def local_obj(x):
        h_clipped = np.clip(x, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        return np.max(np.correlate(
            (h_clipped / total) * target_sum,
            1 - (h_clipped / total) * target_sum,
            mode='full'
        )) * dx

    for _ in range(3):  # Multiple restarts with SLSQP
        if time.time() - start_time > budget_s - 15:
            break
        res = minimize(
            local_obj,
            x0=best_h + np.random.randn(n_points) * 0.05,
            method='SLSQP',
            bounds=bounds,
            options={
                'maxiter': 300,
                'ftol': 1e-10
            }
        )
        if res.success:
            refined = np.clip(res.x, 0, 1)
            refined_total = np.sum(refined)
            if refined_total != 0:
                refined = (refined / refined_total) * target_sum
            refined = np.clip(refined, 0, 1)
            refined_c5 = objective(refined)
            if refined_c5 < best_c5:
                best_h = refined.copy()
                best_c5 = refined_c5

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_total = np.sum(final_h)
    if abs(final_total - target_sum) > 1e-6:
        final_h = (final_h / final_total) * target_sum
    final_h = np.clip(final_h, 0, 1)

    return (final_h.tolist(), best_c5, n_points)
