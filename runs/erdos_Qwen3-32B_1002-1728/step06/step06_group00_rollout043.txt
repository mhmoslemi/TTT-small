    def create_full_h(sym_vars):

        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

        return x

    start_time = time.time()
