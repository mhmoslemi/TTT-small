import cvxpy as cp
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    # Frequency domain optimization
    N_terms = 20
    m_samples = 500
    x_samples = np.linspace(0, 2, m_samples)

    # Compute initial Fourier coefficients from h_full_initial
    x_grid = np.linspace(0, 2, n_points, endpoint=False)
    a_coeffs = np.zeros(N_terms)
    b_coeffs = np.zeros(N_terms)
    for k in range(1, N_terms + 1):
        sum_a = 0.0
        sum_b = 0.0
        for i in range(n_points):
            x = x_grid[i]
            sum_a += h_full_initial[i] * np.cos(np.pi * k * x)
            sum_b += h_full_initial[i] * np.sin(np.pi * k * x)
        a_coeffs[k-1] = sum_a * dx
        b_coeffs[k-1] = sum_b * dx

    # Set up CVXPY problem
    a_var = cp.Variable(N_terms, value=a_coeffs)
    b_var = cp.Variable(N_terms, value=b_coeffs)
    constraints = []
    for x in x_samples:
        h_expr = 0.5
        for k in range(N_terms):
            k_ = k + 1
            h_expr += a_var[k] * cp.cos(np.pi * k_ * x) + b_var[k] * cp.sin(np.pi * k_ * x)
        constraints.append(h_expr >= 0)
        constraints.append(h_expr <= 1)
    obj = cp.Minimize(cp.sum_squares(a_var) + cp.sum_squares(b_var))
    problem = cp.Problem(obj, constraints)

    # Solve
    try:
        problem.solve(solver=cp.MOSEK)
    except:
        problem.solve()

    # Generate h_fourier
    h_fourier = np.zeros(n_points)
    for i in range(n_points):
        x = x_grid[i]
        val = 0.5
        for k in range(N_terms):
            k_ = k + 1
            val += a_var.value[k] * np.cos(np.pi * k_ * x) + b_var.value[k] * np.sin(np.pi * k_ * x)
        h_fourier[i] = val

    # Clip and scale
    h_scaled = np.clip(h_fourier, 0, 1)
    sum_scaled = h_scaled.sum()
    if sum_scaled != 0:
        scale = T / sum_scaled
        h_scaled_clipped = np.clip(h_scaled * scale, 0, 1)
    else:
        h_scaled_clipped = h_scaled

    h_initial_opt = h_scaled_clipped

    # Define objective function for optimization
    def clip_and_scale(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_candidate)
        scale = T / current_sum
        scaled = np.clip(h_clipped * scale, 0, 1)
        return scaled

    def objective(h_candidate):
        h_processed = clip_and_scale(h_candidate)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return float(np.max(corr_vals) * dx)

    # Differential Evolution phase (reduced iterations)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial_opt.copy()
    best_c5 = objective(best_h)

    if time.time() - start_time < budget_s - 200:
        # Run DE
        de_result = differential_evolution(
            lambda x: objective(x), bounds, strategy='best1bin',
            maxiter=8, popsize=12, tol=1e-4, mutation=(0.5, 1),
            recombination=0.85, seed=seed, workers=1, disp=False
        )
        if de_result.success:
            de_h = clip_and_scale(de_result.x)
            de_c5 = objective(de_h)
            if de_c5 < best_c5:
                best_c5 = de_c5
                best_h = de_h

    # Local refinement
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective, x0=best_h, method='SLSQP',
            bounds=bounds, options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h = clip_and_scale(result.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Final validation
    final_h = clip_and_scale(best_h)
    assert abs(final_h.sum() - T) < 1e-6, "Sum constraint violated"
    return (final_h.tolist(), best_c5, n_points)
