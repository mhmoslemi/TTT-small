import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    total = h_clipped.sum()
    if abs(total) < 1e-10:
        h_clipped[:] = 0.5
        total = h_clipped.sum()
    return h_clipped * (target_sum / total)

def optimizer_objective(h_candidate, args):
    k_star_opt, dx_opt, lambda_opt, rho_opt, target_sum_opt = args
    corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
    overlap = corr[k_star_opt]
    sum_h = h_candidate.sum()
    constraint_term = lambda_opt * (sum_h - target_sum_opt) + (rho_opt / 2) * (sum_h - target_sum_opt)**2
    return overlap * dx_opt + constraint_term
