        if time.time() - start_time > budget_s - 5:
            break
        current_c5 = objective(best_h_half, n_points, dx)
        grad = np.zeros(half_n)
        for i in range(half_n):
            pert = np.zeros(half_n)
            pert[i] = eps
            h_pert = best_h_half + pert
            h_repair = repair_half(h_pert, target_sum_half)
            grad[i] = (objective(h_repair, n_points, dx) - current_c5) / eps
        best_h_half -= lr * grad
        best_h_half = repair_half(best_h_half, target_sum_half)
        new_c5 = objective(best_h_half, n_points, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
        lr *= 0.99

    h_full_best = create_full_h(best_h_half)
    return (h_full_best.tolist(), best_c5, n_points)
