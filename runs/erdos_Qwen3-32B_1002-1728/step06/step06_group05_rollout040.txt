import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce initial sum constraint
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h != 0:
        h_initial = (h_initial / sum_h) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Set up PSO
    np.random.seed(seed)
    pop_size = 30
    max_pso_iter = 30
    w_init = 0.9
    w_final = 0.4
    c1 = 2.0
    c2 = 2.0

    # Initialize particles
    particles = np.zeros((pop_size, n_points))
    # 50% from initial with perturbation
    for i in range(pop_size // 2):
        perturbation = np.random.normal(0, 0.3, n_points)
        perturbed = h_initial + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        if sum_p != 0:
            perturbed = (perturbed / sum_p) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        particles[i] = perturbed
    # 50% from random samples
    for i in range(pop_size // 2, pop_size):
        random_vals = np.random.uniform(0, 1, n_points)
        sum_rand = np.sum(random_vals)
        if sum_rand != 0:
            scaled = (random_vals / sum_rand) * target_sum
        else:
            scaled = np.zeros(n_points)
        scaled = np.clip(scaled, 0, 1)
        sum_clipped = np.sum(scaled)
        if sum_clipped != 0:
            scaled = (scaled / sum_clipped) * target_sum
        scaled = np.clip(scaled, 0, 1)
        particles[i] = scaled

    velocities = np.random.uniform(-0.05, 0.05, (pop_size, n_points))

    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    # Initialize personal best and global best
    personal_best = particles.copy()
    personal_best_obj = np.array([objective(p) for p in particles])
    global_best_idx = np.argmin(personal_best_obj)
    global_best = particles[global_best_idx].copy()
    best_c5 = personal_best_obj[global_best_idx]

    # PSO loop
    for iter in range(max_pso_iter):
        if time.time() - start_time > budget_s - 10:
            break
        # Update inertia weight
        w = w_init - (iter / max_pso_iter) * (w_init - w_final)
        # Update velocities and positions
        for i in range(pop_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = (particles[i] / sum_p) * target_sum
            particles[i] = np.clip(particles[i], 0, 1)
        # Evaluate new positions
        obj_vals = np.array([objective(p) for p in particles])
        for i in range(pop_size):
            if obj_vals[i] < personal_best_obj[i]:
                personal_best[i] = particles[i].copy()
                personal_best_obj[i] = obj_vals[i]
        # Update global best
        current_best_idx = np.argmin(personal_best_obj)
        current_best_c5 = personal_best_obj[current_best_idx]
        if current_best_c5 < best_c5:
            global_best = personal_best[current_best_idx].copy()
            best_c5 = current_best_c5

    # Localized Grid Search (LGS) on top 5 candidates
    top_indices = np.argsort(personal_best_obj)[:5]
    for idx in top_indices:
        candidate = personal_best[idx].copy()
        for _ in range(5):  # 5 perturbations per candidate
            if time.time() - start_time > budget_s - 10:
                break
            perturbation = np.random.normal(0, 0.05, n_points)
            new_candidate = candidate + perturbation

            if sum_p != 0:
                new_candidate = (new_candidate / sum_p) * target_sum
            new_candidate = np.clip(new_candidate, 0, 1)
            val = objective(new_candidate)
            if val < best_c5:
                best_c5 = val
                global_best = new_candidate.copy()

    # Final SLSQP optimization on the best candidate so far
    def fun(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    x0 = global_best
    result = minimize(
        fun=fun,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        optimized_h = np.clip(result.x, 0, 1)
        optimized_sum = np.sum(optimized_h)
        if optimized_sum != 0:
            optimized_h = (optimized_h / optimized_sum) * target_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_c5 = fun(optimized_h)
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            global_best = optimized_h

    # Final check and return
    h_full = np.clip(global_best, 0, 1)
    h_full_sum = np.sum(h_full)
    if h_full_sum != 0:
        h_full = (h_full / h_full_sum) * target_sum
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)

            global_best = personal_best[current_best_idx].copy()
            best_c5 = current_best_c5

    top_indices = np.argsort(personal_best_obj)[:5]
    for idx in top_indices:
        candidate = personal_best[idx].copy()
        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            perturbation = np.random.normal(0, 0.05, n_points)
            new_candidate = candidate + perturbation
            new_candidate = np.clip(new_candidate, 0, 1)
            sum_p = np.sum(new_candidate)
            if sum_p != 0:
                new_candidate = (new_candidate / sum_p) * target_sum
            new_candidate = np.clip(new_candidate, 0, 1)
            val = objective(new_candidate)
            if val < best_c5:
                best_c5 = val
                global_best = new_candidate.copy()

    def fun(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total != 0:
            x = (x / total) * target_sum
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    x0 = global_best
    result = minimize(
        fun=fun,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        optimized_h = np.clip(result.x, 0, 1)
        optimized_sum = np.sum(optimized_h)
        if optimized_sum != 0:
            optimized_h = (optimized_h / optimized_sum) * target_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_c5 = fun(optimized_h)
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            global_best = optimized_h

    h_full = np.clip(global_best, 0, 1)
    h_full_sum = np.sum(h_full)
    if h_full_sum != 0:
        h_full = (h_full / h_full_sum) * target_sum
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
