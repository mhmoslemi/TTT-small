import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points

        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Global optimization with Differential Evolution
    bounds_global = [(0.0, 1.0) for _ in range(n_points)]
    popsize = 20
    maxiter_global = 10
    de_time_ratio = 0.4
    de_end_time = start_time + de_budget(de_time_ratio, budget_s)

            recombination=0.9, seed=seed, maxiter=maxiter_global,
            disp=False, polish=False, workers=1
        )
        best_h = result_de.x
        best_c5 = compute_c5(best_h)
    except:
        best_h = h_full.copy()
        best_c5 = compute_c5(best_h)

    # Local refinement with L-BFGS-B and perturbation
    bounds_local = bounds_global

    early_stop = start_time + budget_s - 5

        def objective(x):
            return compute_c5(x)

        try:
            result_lbfgs = minimize(objective, x0=perturbed, method='L-BFGS-B',
                                   bounds=bounds_local, options={'maxiter': 200, 'ftol': 1e-10})
            if result_lbfgs.success:

    return (best_h.tolist(), best_c5, n_points)

def de_budget(ratio, budget_s):
    return ratio * budget_s
