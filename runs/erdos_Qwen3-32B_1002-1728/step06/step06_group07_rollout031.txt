    population.append(initial_binary.copy())
    for _ in range(n_population - 1):
        # Randomly choose T positions out of n_points
        indices = np.random.choice(n_points, size=int(T), replace=False)
        individual = np.zeros(n_points)
        individual[indices] = 1
        population.append(individual)

    best_overall = None
    best_score = float('inf')
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 60:
            break
        # Evaluate population
        fitness = []
        for ind in population:
            corr_vals = np.correlate(ind, 1 - ind, mode='full')
            score = np.max(corr_vals) * dx
            fitness.append(score)
        # Find current best
        min_idx = np.argmin(fitness)
        best_h = population[min_idx]
        if fitness[min_idx] < best_score or best_overall is None:
            best_score = fitness[min_idx]
            best_overall = best_h.copy()
        
        # Select parents (top 20%)
        n_parents = int(0.2 * n_population)
        parent_indices = np.argsort(fitness)[:n_parents]
        parents = [population[i] for i in parent_indices]
        
        # Create new population with elitism
        new_population = [best_h.copy()]
        while len(new_population) < n_population:
            parent = parents[np.random.choice(len(parents))]
            # Mutate: swap one 1 and one 0 in the parent
            ones = np.where(parent == 1)[0]
            zeros = np.where(parent == 0)[0]
            if not ones.size or not zeros.size:
                new_population.append(parent.copy())

            i1 = np.random.choice(ones)
            i2 = np.random.choice(zeros)

            child[i1] = 0
            child[i2] = 1
