    target_sum = n_points / 2.0

    def clip_and_rescale(h_vec):
        h = h_vec.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h.copy()
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)

    np.random.seed(seed)

    n_candidates = 20
    cand_starts = []
    for _ in range(n_candidates):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + noise
        perturbed = clip_and_rescale(perturbed)
        cand_starts.append(perturbed)

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    all_results = []

    for start in cand_starts:
        if time.time() - start_time > budget_s - 120:
            break

        start_flat = start.reshape(n_points)

        def constraint_sum(x):
            return x.sum() - target_sum

        constraints = {
            'type': 'eq',
            'fun': constraint_sum
        }

        result = minimize(
            fun=compute_c5,
            x0=start_flat,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8, 'disp': False}
        )

        if result.success:
            optimized = result.x
            c5 = compute_c5(optimized)
            all_results.append((optimized, c5))

    if not all_results:
        final_h = h_initial.copy()
        final_c5 = compute_c5(final_h)
        return (final_h.tolist(), final_c5, n_points)

    all_results.sort(key=lambda x: x[1])
    top_candidates = all_results[:5]

    for candidate, c5 in top_candidates:
        if time.time() - start_time > budget_s - 60:
            break

        result_refined = minimize(
            fun=compute_c5,
            x0=candidate,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10, 'disp': False}
        )

        if result_refined.success:
            refined_c5 = compute_c5(result_refined.x)
            if refined_c5 < c5:
                candidate[:] = result_refined.x
                c5 = refined_c5

    all_results.sort(key=lambda x: x[1])
    best_h_flat = all_results[0][0]
    best_h = clip_and_rescale(best_h_flat)
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n_points)
