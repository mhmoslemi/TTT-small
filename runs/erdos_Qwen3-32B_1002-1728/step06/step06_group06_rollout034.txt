import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    h_initial = clip_and_rescale(h_initial)
    np.random.seed(seed)

    n_initial = 20
    candidates = [h_initial + np.random.normal(0, 0.1, n_points) for _ in range(n_initial)]
    candidates = [clip_and_rescale(c) for c in candidates]
    candidates.append(h_initial)

    def compute_c5_and_k(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        max_c5 = corr[max_idx] * dx
        return max_c5, k

    def compute_gradient(h, k):
        n = len(h)
        idx1 = (np.arange(n) + k) % n
        idx2 = (np.arange(n) - k) % n
        return (1 - h[idx1]) - h[idx2]

    best_h = h_initial.copy()
    best_c5, _ = compute_c5_and_k(best_h)

    for candidate in candidates:
        if time.time() - start_time > budget_s - 10:
            break
        h = candidate.copy()
        step_size = 0.05
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            current_c5, k = compute_c5_and_k(h)
            grad = compute_gradient(h, k)
            h_new = h - step_size * grad
            h_new = clip_and_rescale(h_new)
            new_c5, _ = compute_c5_and_k(h_new)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
            step_size *= 0.99
        if compute_c5_and_k(h)[0] < best_c5:
            best_h = h.copy()
            best_c5 = compute_c5_and_k(h)[0]

    return (best_h.tolist(), best_c5, n_points)
