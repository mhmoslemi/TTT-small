def project(h, target_sum):
    for _ in range(3):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h_clipped * scale
    return np.clip(h, 0, 1)

def objective_nelder_mead(x, dx, target_sum):
    x = np.reshape(x, -1)
    x_proj = project(x, target_sum)
    corr_vals = np.correlate(x_proj, 1 - x_proj, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h_initial = project(h_initial, target_sum)

    np.random.seed(seed)

    n_particles = 40
    max_iter_pso = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5

    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    for _ in range(n_particles):
        pos = h_initial + np.random.normal(0, 0.1, n_points)
        pos = project(pos, target_sum)
        velocity = np.random.normal(0, 0.05, n_points)
        particles.append(pos)
        velocities.append(velocity)
        pb = pos.copy()
        pb_score = np.max(np.correlate(pb, 1 - pb, mode='full')) * dx
        personal_best_positions.append(pb)
        personal_best_scores.append(pb_score)

    global_best = h_initial.copy()
    global_best_score = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx
    for i in range(n_particles):
        if personal_best_scores[i] < global_best_score:
            global_best = personal_best_positions[i].copy()
            global_best_score = personal_best_scores[i]

    no_improve = 0
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s - 60:
            break
        prev_best_score = global_best_score
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best_positions[i] - particles[i]) +
                            social * r2 * (global_best - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = project(new_pos, target_sum)
            particles[i] = new_pos

            corr_vals = np.correlate(new_pos, 1 - new_pos, mode='full')
            current_score = np.max(corr_vals) * dx

            if current_score < personal_best_scores[i]:
                personal_best_positions[i] = new_pos.copy()
                personal_best_scores[i] = current_score
                if current_score < global_best_score:
                    global_best = new_pos.copy()
                    global_best_score = current_score
        if abs(prev_best_score - global_best_score) < 1e-8:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve >= 5:
            break

    top_indices = np.argsort(personal_best_scores)
    top_candidates = [personal_best_positions[i] for i in top_indices[:10]]

    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            fun=objective_nelder_mead,
            x0=candidate,
            method='Nelder-Mead',
            args=(dx, target_sum),
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 150, 'xatol': 1e-8, 'fatol': 1e-8, 'disp': False}
        )
        if result.success:
            x_opt = project(result.x, target_sum)
            corr_vals = np.correlate(x_opt, 1 - x_opt, mode='full')
            current_score = np.max(corr_vals) * dx
            if current_score < global_best_score:
                global_best = x_opt.copy()
                global_best_score = current_score

    return (global_best.tolist(), global_best_score, n_points)
