        method='SLSQP',
        bounds=bounds_fourier,
        options={'maxiter': 20, 'ftol': 1e-8, 'disp': False}
    )
    best_coeff = result_fourier.x if result_fourier.success else coeff_initial

    # Compute h from Fourier optimized coefficients
    a = best_coeff[:N]
    b = best_coeff[N:]
    h_fourier = 0.5 * np.ones(n_points)
    for i in range(N):
        h_fourier += a[i] * np.cos(np.pi * (i + 1) * x) + b[i] * np.sin(np.pi * (i + 1) * x)
    h_fourier_clipped = np.clip(h_fourier, 0, 1)
    h_fourier_sum = np.sum(h_fourier_clipped)
    h_fourier_scaled = h_fourier_clipped * (target_sum / h_fourier_sum) if h_fourier_sum != 0 else np.zeros(n_points)
    h_fourier_scaled = np.clip(h_fourier_scaled, 0, 1)

    # Time domain refinement
    def objective_time(h_candidate):
        if time.time() - start_time > budget_s - 10:
            return np.inf
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.inf
        h_scaled = h_clipped * (target_sum / current_sum)
        corr_vals = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr_vals) * dx

    bounds_time = [(0.0, 1.0) for _ in range(n_points)]
    result_time = minimize(
        objective_time,
        x0=h_fourier_scaled,
        method='L-BFGS-B',
        bounds=bounds_time,
        options={'maxiter': 50}
    )
    h_refined = result_time.x if result_time.success else h_fourier_scaled
    h_refined = np.clip(h_refined, 0, 1)
    h_refined_sum = np.sum(h_refined)
    h_refined_scaled = h_refined * (target_sum / h_refined_sum) if h_refined_sum != 0 else np.zeros(n_points)
    h_refined_scaled = np.clip(h_refined_scaled, 0, 1)

    # Final evaluation
    corr_vals = np.correlate(h_refined_scaled, 1 - h_refined_scaled, mode='full')
    c5_final = np.max(corr_vals) * dx
    return (h_refined_scaled.tolist(), c5_final, n_points)
