import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    h_full_sym = np.zeros(n_points)
    h_full_sym[:half_n] = h_full_initial[:half_n]
    h_full_sym[-half_n:] = h_full_initial[:half_n][::-1]
    h_full_sym = np.clip(h_full_sym, 0, 1)
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    def objective_de(x_vars):

        total = np.sum(x)
        if total != 0:

        else:

        h_full = create_full_h(x_clipped)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    bounds_de = [(0.0, 1.0)] * half_n
    np.random.seed(seed)
    try:
        de_result = differential_evolution(
            objective_de, bounds_de, strategy='best1bin',
