from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def create_h(m, w):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            dist = abs(x - m)
            if dist <= w / 2:
                h[i] = 1 - (dist / (w / 2))
            else:
                h[i] = 0.0
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum < 1e-10:
            h = np.zeros_like(h)
        else:
            scale = (n_points / 2.0) / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    def objective(params):
        m, w = params
        h = create_h(m, w)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_vals) * dx
        return c5

    # Initial guess
    m_initial = 1.0
    w_initial = 2.0

    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    best_h = h_initial

    # Optimization bounds
    bounds = [(0.0, 2.0), (1e-4, 4.0)]

    # Try multiple random restarts
    np.random.seed(seed)
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        m_rand = np.random.uniform(0.25, 1.75)
        w_rand = np.random.uniform(1.0, 3.5)
        res = minimize(
            fun=objective, x0=[m_rand, w_rand], bounds=bounds,
            method='L-BFGS-B', options={'maxiter': 200, 'ftol': 1e-8}

        if res.success:
            h_candidate = create_h(*res.x)
            c5_candidate = objective(res.x)
