        h_clipped = np.clip(h_clipped * scale, 0, 1)
        return h_clipped

    h_initial = clip_and_rescale(h_initial)

    n_candidates = 20
    np.random.seed(seed)
    candidates = []
    for _ in range(n_candidates):
        perturbation = np.random.normal(0, 0.1, n_points)
        candidate = h_initial + perturbation
        candidate = clip_and_rescale(candidate)
        candidates.append(candidate)

    best_c5 = np.inf
    best_h = h_initial.copy()

    for candidate in candidates:
        corr_vals = np.correlate(candidate, 1 - candidate, mode='full')
        c5 = np.max(corr_vals) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()

    max_proj_iters = 100
    for candidate in candidates:
        if time.time() - start_time > budget_s - 1:
            break
        current_h = candidate.copy()
        prev_c5 = np.inf
        for _ in range(max_proj_iters):
            current_h = clip_and_rescale(current_h)
            corr_vals = np.correlate(current_h, 1 - current_h, mode='full')
            c5 = np.max(corr_vals) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
            if np.abs(c5 - prev_c5) < 1e-6:
                break
            prev_c5 = c5
            if time.time() - start_time > budget_s - 1:
                break

    if (time.time() - start_time) < (budget_s - 50):
        def objective(x):
            x = x.reshape(-1)
            x = clip_and_rescale(x)
            corr_vals = np.correlate(x, 1 - x, mode='full')
            return np.max(corr_vals) * dx

        from scipy.optimize import minimize
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 150, 'gtol': 1e-8, 'disp': False}
        )
        if result.success:
            x_opt = clip_and_rescale(result.x)
            corr_vals = np.correlate(x_opt, 1 - x_opt, mode='full')
            c5_opt = np.max(corr_vals) * dx
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = x_opt

    return (best_h.tolist(), best_c5, n_points)
