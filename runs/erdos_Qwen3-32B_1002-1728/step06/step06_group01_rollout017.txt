        return np.max(corr_vals) * dx

    # Binary-inspired initialization with fixed sum
    np.random.seed(seed)
    target_sum = target_sum_half
    num_ones = int(np.floor(target_sum))
    num_half = 1 if target_sum > num_ones else 0

    half_vec = np.zeros(half_n)
    indices = np.random.permutation(half_n)
    ones_idx = indices[:num_ones]
    half_idx = indices[num_ones:num_ones+num_half]
    half_vec[ones_idx] = 1.0
    if len(half_idx) > 0:
        half_vec[half_idx[0]] = 0.5  # Symmetric fractional adjustment

    bounds_per_var = [(0, 1) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    best_h_half = half_vec.copy()
    best_c5 = objective(best_h_half)

    # Differential evolution with budget-aware population
    def objective_de(x_vars):
        x = np.clip(x_vars, 0, 1)
        total = np.sum(x)
        if total > 1e-9:
            x_scaled = x / total * target_sum_half
        else:
            x_scaled = x
        x_scaled = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_scaled)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def termination(t, iter, result):
        return (time.time() - start_time) > (budget_s - 20)

    try:
        popsize = max(10, int((half_n * 2) ** 0.5))
        de_result = differential_evolution(
            objective_de, bounds_per_var, 
            maxiter=20, popsize=popsize, workers=1, 
            mutation=(0.8, 1.2), recombination=0.9,
            seed=seed, tol=1e-4, callback=termination
        )
        if de_result.success:
            de_h = de_result.x.copy()
            de_h = np.clip(de_h / np.sum(de_h) * target_sum_half, 0, 1)
            c5 = objective(de_h)
            if c5 < best_c5:
                best_h_half = de_h
                best_c5 = c5
    except Exception:
        pass

    # Adaptive local search
    np.random.seed(seed)
    while time.time() - start_time < budget_s - 10:
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.5 * frac
        method_idx = int(frac * 3)
        methods = ['L-BFGS-B', 'COBYLA', 'SLSQP']

        idxs = np.random.choice(half_n, size=5 + int(half_n/10), replace=False)
        perturbed = best_h_half.copy()
        perturbed[idxs] += np.random.normal(0, noise_scale, len(idxs))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = np.clip(perturbed / np.sum(perturbed) * target_sum_half, 0, 1)

        res = minimize(objective, x0=perturbed, method=methods[method_idx],
                      bounds=bounds_per_var, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-6})

        if res.success:
            new_h = np.clip(res.x, 0, 1)
            if abs(np.sum(new_h) - target_sum_half) < 1e-5:
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_h_half = new_h
                    best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points/2) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
