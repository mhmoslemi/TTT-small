    x_vals = np.linspace(0, 2, n_points, endpoint=False)  # Or equally spaced

    # Fourier parameters

    a0 = 1.0 / 2.0  # Set this to ensure integral constraint

    # Construct the basis functions at each x_i
    cos_terms = []
    sin_terms = []
    for k in range(1, K+1):
        cos_term = np.cos(np.pi * k * x_vals / 2.0)
        sin_term = np.sin(np.pi * k * x_vals / 2.0)
        cos_terms.append(cos_term)
        sin_terms.append(sin_term)
    cos_terms = np.array(cos_terms)  # (K, n_points)
    sin_terms = np.array(sin_terms)  # (K, n_points)

    a = cp.Variable(K)

    h_i = a0 + cp.matmul(cos_terms.T, a) + cp.matmul(sin_terms.T, b)

    prob = cp.Problem(obj, constraints)

    # Get optimized coefficients
    a_opt = a.value
    b_opt = b.value

    # Compute the time-domain h values
    h_freq_opt = (a0 + np.dot(cos_terms.T, a_opt) + np.dot(sin_terms.T, b_opt)).flatten()

    final_h = clip_and_scale(best_h, T)
    if not np.allclose(final_h.sum(), T, atol=1e-6):
        # If sum is not correct, re-scale
        scale_factor = T / final_h.sum()
        final_h = np.clip(final_h * scale_factor, 0, 1)

    return (final_h.tolist(), best_c5, n_points)

4. The code uses K=20, n_points=70. The number of variables is 2*K=40, which is manageable.

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def clip_and_scale(h_candidate, T):
    h_clipped = np.clip(h_candidate, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return np.zeros_like(h_candidate)
    scale = T / current_sum
    scaled = np.clip(h_clipped * scale, 0, 1)
    return scaled

def objective(h_candidate, dx, T):
    h_processed = clip_and_scale(h_candidate, T)
    corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
    return float(np.max(corr_vals) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    x_vals = np.linspace(0, 2, n_points, endpoint=False)

    K = 20  # Number of Fourier terms
    a0 = 1.0 / 2.0

    # Construct basis functions
    cos_terms = []
    sin_terms = []
    for k in range(1, K + 1):
        cos_k = np.cos(np.pi * k * x_vals / 2.0)
        sin_k = np.sin(np.pi * k * x_vals / 2.0)
        cos_terms.append(cos_k)
        sin_terms.append(sin_k)
    cos_terms = np.array(cos_terms)
    sin_terms = np.array(sin_terms)

    a = cp.Variable(K)
    b = cp.Variable(K)
    h_i = a0 + cp.matmul(cos_terms.T, a) + cp.matmul(sin_terms.T, b)
    constraints = [h_i >= 0, h_i <= 1]
    objective_cp = cp.Minimize(cp.sum_squares(a) + cp.sum_squares(b))

    prob = cp.Problem(objective_cp, constraints)

    try:
        prob.solve(solver=cp.MOSEK, max_iter=1000, verbose=False)
    except:
        prob.solve(solver=cp.CVXOPT, verbose=False)

    a_opt = a.value or np.zeros(K)
    b_opt = b.value or np.zeros(K)

    h_freq_opt = (a0 + np.dot(cos_terms.T, a_opt) + np.dot(sin_terms.T, b_opt)).flatten()
    h_freq_opt = clip_and_scale(h_freq_opt, T)

    obj_for_opt = lambda h_candidate: objective(h_candidate, dx, T)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = obj_for_opt(h_freq_opt)
    best_h = h_freq_opt.copy()

    if time.time() - start_time < budget_s - 30:
        result = minimize(
            obj_for_opt, x0=best_h, method='SLSQP',
            bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h = clip_and_scale(result.x, T)
            new_c5 = obj_for_opt(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Final validation and adjustment
    final_h = clip_and_scale(best_h, T)
    final_h = np.clip(final_h, 0, 1.0)

    return (final_h.tolist(), best_c5, n_points)
