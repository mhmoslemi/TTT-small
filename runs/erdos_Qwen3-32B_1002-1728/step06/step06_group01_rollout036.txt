    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * dx

    # Symmetrize initial h
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n].copy()

    # PSO Setup
    np.random.seed(seed)
    pop_size = 20
    particles = []
    velocities = []
    for _ in range(pop_size):
        base = h_half.copy()
        noise = np.random.randn(half_n) * 0.1
        p = base + noise
        p = np.clip(p, 0, 1)
        total = np.sum(p)
        if total != 0:
            p = p * (target_sum_half / total)
        else:
            p = h_half.copy()
        p = np.clip(p, 0, 1)
        particles.append(p)
        velocities.append(np.random.randn(half_n) * 0.05)

    personal_best = [p.copy() for p in particles]
    fitness = [objective(p) for p in particles]
    best_idx = np.argmin(fitness)
    best_particle = particles[best_idx].copy()
    best_c5 = fitness[best_idx]

    w = 0.8
    c1 = 1.5
    c2 = 2.0

    while time.time() - start_time < budget_s - 30:
        for i in range(pop_size):
            p = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            new_v = w * v + c1 * r1 * (personal_best[i] - p) + c2 * r2 * (best_particle - p)
            new_p = p + new_v
            new_p = np.clip(new_p, 0, 1)
            total = np.sum(new_p)
            if total > 1e-9:
                new_p = new_p * (target_sum_half / total)
            else:
                new_p = h_half.copy()
            new_p = np.clip(new_p, 0, 1)
            velocities[i] = new_v
            particles[i] = new_p
            new_fit = objective(new_p)
            if new_fit < fitness[i]:
                personal_best[i] = new_p.copy()
                fitness[i] = new_fit
                if new_fit < best_c5:
                    best_particle = new_p.copy()
                    best_c5 = new_fit
        if time.time() - start_time >= budget_s - 30:
            break

    # Local refinement
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0, 1)] * half_n
    result = minimize(
        fun=objective, x0=best_particle, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    if result.success:
        refined = np.clip(result.x, 0, 1)
        total_refined = np.sum(refined)
        if total_refined != 0:
            refined = (refined / total_refined) * target_sum_half
        refined = np.clip(refined, 0, 1)
        c5_refined = objective(refined)
        if c5_refined < best_c5:
            best_particle = refined
            best_c5 = c5_refined

    h_full = create_full_h(best_particle)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
