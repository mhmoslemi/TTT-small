import numpy as np
from scipy.optimize import minimize
import time

def enforce_feasibility(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    total = np.sum(x)
    if total != 0.0:
        x = (x / total) * target_sum
    x = np.clip(x, 0.0, 1.0)
    return x

def evaluate(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_current = np.array(initial_h_values).copy()
    n_points = len(h_current)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Enforce feasibility on initial guess
    h_current = enforce_feasibility(h_current, target_sum)
    best_h = h_current.copy()
    best_c5 = evaluate(best_h, dx)
    T = 1.0
    alpha = 0.99
    max_iter = 3000

    np.random.seed(seed)

    # Simulated Annealing loop
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 30:
            break
        # Mutate current solution
        mutation = np.random.normal(0, T, n_points)
        h_new = h_current + mutation
        h_new = enforce_feasibility(h_new, target_sum)
        c5_new = evaluate(h_new, dx)
        # Metropolis criterion
        if c5_new < best_c5 or np.exp((best_c5 - c5_new) / T) > np.random.rand():
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        T *= alpha  # Cool down

    # Local refinement with SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    result = minimize(
        fun=lambda x: evaluate(x, dx),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        optimized_h = result.x.copy()
        optimized_h = enforce_feasibility(optimized_h, target_sum)
        optimized_c5 = evaluate(optimized_h, dx)
        if optimized_c5 < best_c5:
            best_h = optimized_h.copy()
            best_c5 = optimized_c5

    final_c5 = evaluate(best_h, dx)
