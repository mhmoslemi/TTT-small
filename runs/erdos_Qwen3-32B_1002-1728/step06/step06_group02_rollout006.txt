import numpy as np
import time

def evaluate_c5(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
    return np.max(corr_vals)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = np.array(initial_h_values).copy().astype(float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize h
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum != 0:
        h *= target_sum / current_sum
    else:
        h[:] = target_sum / n_points
    h = np.clip(h, 0, 1)
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5

    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_factor = 0.95
    temp = initial_temp
    steps_per_temp = 100

    while (time.time() - start_time) < budget_s - 30:
        for _ in range(steps_per_temp):
            if (time.time() - start_time) >= budget_s - 30:
                break
            i = np.random.randint(n_points)
            # Perturb and clip
            noise = np.random.normal(0, temp * 0.1)
            new_h = h.copy()
            new_h[i] += noise
            new_h[i] = np.clip(new_h[i], 0, 1)
            # Rescale to maintain sum constraint
            new_sum = new_h.sum()
            if new_sum != 0:
                new_h *= (target_sum / new_sum)
            else:
                new_h[:] = target_sum / n_points
            new_h = np.clip(new_h, 0, 1)
            # Compute new objective
            new_c5 = evaluate_c5(new_h, dx)
            # Acceptance criteria
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
                h = new_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            # End of step
        # Cool down
        temp *= cooling_factor

    # Final feasibility check
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    final_sum = h_final.sum()
    if final_sum != 0:
        h_final *= (target_sum / final_sum)
    else:
        h_final[:] = target_sum / n_points
    h_final = np.clip(h_final, 0, 1)
    assert abs(h_final.sum() - target_sum) < 1e-6

    return (h_final.tolist(), best_c5, n_points)
