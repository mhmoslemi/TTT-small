```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    np.random.seed(seed)
    n_initial = 30
    perturbations = []
    for _ in range(n_initial):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + noise
        perturbations.append(clip_and_rescale(perturbed))

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    optimized_candidates = []

    for h_candidate in perturbations:
        if time.time() - start_time > budget_s - 2 * 60:
            break
        objective = lambda x: np.max(np.correlate(clip_and_rescale(x), 
                                                  1 - clip_and_rescale(x), 
                                                  mode='full')) * dx
        x0 = h_candidate.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, x0=x0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'gtol': 1e-8, 'disp': False})
        if result.success:
            optimized = clip_and_rescale(result.x)
            c5_opt = np.max(np.correlate(optimized, 1 - optimized, mode='full')) * dx
            optimized_candidates.append((optimized, c5_opt))
            if c5_opt < best_c5:
                best_h = optimized.copy()
                best_c5 = c5_opt

    top_candidates = sorted(optimized_candidates, key=lambda x: x[1])[:5]
    
    for optimized, c5 in top_candidates:
        if time.time() - start_time > budget_s - 10:
            break
        noise_scale = 0.2 * optimized + 0.05
        perturbed = optimized + np.random.normal(0, noise_scale)
        perturbed = clip_and_rescale(perturbed)
        x0 = perturbed
        result = minimize(objective, x0=x0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'gtol': 1e-8, 'disp': False})
        if result.success:
            refined = clip_and_rescale(result.x)
            c5_refined = np.max(np.correlate(refined, 1 - refined, mode='full')) * dx
            if c5_refined < best_c5:
                best_h = refined.copy()
                best_c5 = c5_refined

    return (best_h.tolist(), best_c5, n_points)
```
