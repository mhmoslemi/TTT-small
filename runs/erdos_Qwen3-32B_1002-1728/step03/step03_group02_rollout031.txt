import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()

    constraints = [cp.sum(h) == target_sum]

    for k in range(-n_points + 1, n_points):
        start = max(0, -k)
        end = min(n_points, n_points - k)

        expr = cp.sum(h[valid_indices] * h[valid_indices + k])
        constraints.append(t >= dx * (target_sum - expr))

    problem = cp.Problem(cp.Minimize(t), constraints)
    problem.solve(solver="GUROBI", verbose=False, TimeLimit=budget_s - 30)

    best_h = h.value
    best_c5 = t.value

    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint violated"

    return (np.round(best_h).tolist(), best_c5, n_points)
