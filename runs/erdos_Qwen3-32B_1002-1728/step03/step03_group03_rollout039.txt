        delta_val = delta.value
        h_new = h + delta_val
        h_clipped = np.clip(h_new, 0.0, 1.0)

        # Adjust sum constraint
        current_sum = np.sum(h_clipped)
        if abs(current_sum - target_sum) > 1e-6:
            nsample = h.size
            diff = current_sum - target_sum
            idx = (np.arange(nsample) % 1).astype(int)
            h_clipped[idx[0]] = max(0.0, min(1.0, h_clipped[idx[0]] - diff))
        h = h_clipped

        # Update best solution
        current_c5 = objective_func(h)
        if current_c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = current_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
