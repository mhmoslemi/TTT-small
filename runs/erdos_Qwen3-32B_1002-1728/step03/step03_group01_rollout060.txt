import time
import numpy as np
from scipy.optimize import minimize

def evaluate_objective(h_vec, dx):
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 0.6
    c2 = 0.3
    max_v = 0.5
    max_iterations = 100

    np.random.seed(seed)
    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []

    for i in range(n_particles):
        if i == 0:
            pos = h_init.copy()
        else:
            pos = np.random.rand(n_points)
        pos = np.clip(pos, 0, 1)
        s = np.sum(pos)
        pos = pos * (target_sum / s)
        particles.append(pos)
        velocities.append(np.random.uniform(-max_v, max_v, n_points))
        score = evaluate_objective(pos, dx)
        personal_best_positions.append(pos.copy())
        personal_best_scores.append(score)

    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best_positions[global_best_idx].copy()

    iteration = 0
    while time.time() - start_time < budget_s * 0.9 and iteration < max_iterations:
        iteration += 1
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] 
                            + c1 * r1 * (personal_best_positions[i] - particles[i]) 
                            + c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_v, max_v)
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)

            if s > 1e-9:

            score = evaluate_objective(particles[i], dx)
            if score < personal_best_scores[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = score
                if score < evaluate_objective(global_best, dx):
                    global_best = particles[i].copy()

    # SLSQP refinement
    top_candidates = sorted(zip(personal_best_positions, personal_best_scores), key=lambda x: x[1])[:5]
