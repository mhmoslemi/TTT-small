import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Perturbation and optimization loop
    n_workers = 8
    std_levels = [0.05, 0.1, 0.15, 0.2, 0.02, 0.03, 0.07, 0.12]  # Varying perturbation strengths

    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.99:
            break
        rs = np.random.RandomState(seed + w)
        std = std_levels[w % len(std_levels)]  # Cycle through different stds
        # Generate perturbed initial guess
        perturbed = best_h + rs.normal(0, std, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        
        # Enforce integral constraint via iterative normalization and clipping
        for _ in range(5):  # Improves constraint satisfaction
            s = np.sum(perturbed)
            if abs(s) < 1e-9:
                break
            perturbed = np.clip(perturbed * (target_sum / s), 0, 1)
        
        # SLSQP optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            tol=1e-8,
            options={'maxiter': 400, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final constraint enforcement
            h_candidate = np.clip(h_candidate, 0, 1)
            for _ in range(3):  # Final refinement
                s = np.sum(h_candidate)
                if abs(s) < 1e-9:
                    break
                h_candidate = np.clip(h_candidate * (target_sum / s), 0, 1)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
