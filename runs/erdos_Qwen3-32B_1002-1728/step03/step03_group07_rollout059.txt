import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = h_init.copy()
    best_h = current_h.copy()
    best_energy = compute_energy(best_h)
    np.random.seed(seed)

    n_restarts = 5
    max_iters = 500
    perturb_sigma = 0.1

    for restart in range(n_restarts):
        if restart == 0:
            h_current = h_init.copy()
        else:
            h_current = best_h.copy()
            noise = perturb_sigma * (2 * np.random.rand(n_points) - 1)
            h_current += noise
        h_current = np.clip(h_current, 0, 1)
        total = np.sum(h_current)
        h_current *= target_sum / total if total != 0 else 1.0
        h_current = np.clip(h_current, 0, 1)
        current_energy = compute_energy(h_current)

        for iter in range(max_iters):
            if time.time() - start_time > budget_s * 0.99:
                break

            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            k_star = max_idx - (n_points - 1)

            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + k_star
                if 0 <= j < n_points:
                    grad[i] += (1 - h_current[j])
                    grad[j] -= h_current[i]
            grad *= dx

            alpha = 0.1 * 0.95 ** (iter // 100)
            h_new = h_current - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            total = np.sum(h_new)
            delta = target_sum - total
            if delta != 0:
                h_new += delta / n_points
                h_new = np.clip(h_new, 0, 1)

            new_energy = compute_energy(h_new)
            if new_energy < best_energy:
                best_energy = new_energy
                best_h[:] = h_new.copy()

            h_current[:] = h_new.copy()

            if iter % 200 == 0 and restart > 0:
                noise = 0.05 * (2 * np.random.rand(n_points) - 1)
                h_current += noise
                h_current = np.clip(h_current, 0, 1)
                total = np.sum(h_current)
                h_current *= target_sum / total if total != 0 else 1.0
                h_current = np.clip(h_current, 0, 1)

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    return (best_h.tolist(), best_energy, n_points)
