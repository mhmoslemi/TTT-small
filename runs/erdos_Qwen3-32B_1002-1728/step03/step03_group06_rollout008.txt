                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            if np.random.rand() < MUTATION_RATE:
                mutation = np.random.normal(0, 0.1, n_points)
                child += mutation
            child = repair(child, target_sum)
            new_population.append(child)
        population = np.array(new_population)

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
