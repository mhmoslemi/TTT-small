    pbest = np.zeros((N_PARTICLES, n_points))
    pbest_scores = np.full(N_PARTICLES, float('inf'))

        particles[i] = particles[i] / particles[i].sum() * target_sum

def score(h, dx):
    # Compute max correlation * dx
    h = h.reshape(-1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
