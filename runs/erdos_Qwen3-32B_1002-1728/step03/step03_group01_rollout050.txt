            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr) * dx
            error = np.sum(h) - target_sum
            return max_corr + 0.5 * rho * error**2 - lambd * error

        result = minimize(
            fun=objective_admm,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 40, 'ftol': 1e-7}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            h_candidate = np.clip(h_candidate, 0, 1)

            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx

            if c5_candidate < best_c5:
                best_h[:] = h_candidate
                best_c5 = c5_candidate

            # Update ADMM parameters
            new_error = np.sum(h_candidate) - target_sum
            lambd += rho * new_error
            if abs(new_error) > 1e-3:
                rho = min(rho * 1.2, 100.0)
            else:
                rho = max(rho / 1.2, 0.1)
        else:
            # Failed optimization, try modified initial point
            perturb = best_h.copy()
            perturb += (np.random.rand(n_points) - 0.5) * 0.1
            perturb = np.clip(perturb, 0, 1)
            perturb_sum = np.sum(perturb)
            if perturb_sum > 1e-9:
                perturb *= target_sum / perturb_sum
            
            result = minimize(
                fun=objective_admm,
                x0=perturb,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 40, 'ftol': 1e-7}
            )
            if result.success:
                h_candidate = result.x.reshape(n_points)
                h_candidate = np.clip(h_candidate, 0, 1)
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                
                if c5_candidate < best_c5:
                    best_h[:] = h_candidate
                    best_c5 = c5_candidate
                
                lambd += rho * (np.sum(h_candidate) - target_sum)

    # Final refinement under constraints
    def final_obj(h_vec):
        h = h_vec.reshape(n_points)
        return np.max(np.correlate(h, 1-h, mode='full')) * dx

    result = minimize(
        fun=final_obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}),
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    if result.success:
        h_candidate = np.clip(result.x.reshape(n_points), 0, 1)
        corr = np.correlate(h_candidate, 1-h_candidate, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h[:] = h_candidate
            best_c5 = c5

    # Final constraint validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
