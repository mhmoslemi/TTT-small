        h_perturbed = best_h.copy()
        h_perturbed += (rs.rand(n_points) - 0.5) * 0.2
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum > 1e-9:
            h_perturbed *= (target_sum / h_perturbed_sum)
        result = minimize(
            fun=evaluate,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            c5_candidate = evaluate(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate

    # Final validation
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.all((best_h >= 0) & (best_h <= 1 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
