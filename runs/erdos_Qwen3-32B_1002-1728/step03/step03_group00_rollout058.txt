        for i, idx in enumerate(replace_indices):

    h_full = create_full_h(best_h_vars)
    h_full = np.clip(h_full, 0, 1)
    assert np.isclose(np.sum(h_full), n_points / 2.0, atol=1e-6), "Sum constraint violated"
    
    return (h_full.tolist(), best_c5, n_points)
