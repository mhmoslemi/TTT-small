def run(seed=42, budget_s=1100, **kwargs):

    target_sum = n_points / 2.0

    n_particles = 50
    max_iter = 1000
    w = 0.9
    c1 = 1.5
    c2 = 1.5

    particles = np.zeros((n_particles, n_points))
    particles[0] = h_init.copy()
    for i in range(1, n_particles):
        h_rand = np.random.rand(n_points)
        sum_rand = h_rand.sum()
        particles[i] = np.clip((h_rand / sum_rand) * target_sum, 0, 1)

    velocities = 0.01 * np.random.rand(n_particles, n_points)

    pbest_positions = particles.copy()
    pbest_scores = np.array([np.max(np.correlate(p, 1 - p, "full")) * dx for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    def project_feasible(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0:
            return h_candidate
        scaling = target_sum / sum_clipped
        h_scaled = h_clipped * scaling
        h_scaled = np.clip(h_scaled, 0, 1)
        for _ in range(3):
            sum_scaled = h_scaled.sum()
            if sum_scaled == 0:
                break
            h_scaled = h_scaled * (target_sum / sum_scaled)
            h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        for i in range(n_particles):
            current_h = particles[i]
            corr = np.correlate(current_h, 1 - current_h, "full")
            current_score = np.max(corr) * dx
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = current_h.copy()
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = current_h.copy()

        for i in range(n_particles):

            velocities[i] = (w * velocities[i] +
