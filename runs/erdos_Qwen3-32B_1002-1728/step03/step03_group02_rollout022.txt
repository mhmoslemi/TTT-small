def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = int(n_points / 2)
    assert target_sum * 2 == n_points, "Need even number of points"

    def evaluate_c5(h_vec):
        # Compute the max of h * (1 - h) shifted
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def repair_binary(h_vec, target_sum):
        h_vec = np.copy(h_vec)
        current_sum = np.sum(h_vec)
        if current_sum == target_sum:
            return h_vec
        elif current_sum < target_sum:
            deficit = int(target_sum - current_sum)
            zeros = np.where(h_vec == 0)[0]
            np.random.shuffle(zeros)
            h_vec[zeros[:deficit]] = 1.0
        else:
            surplus = int(current_sum - target_sum)
            ones = np.where(h_vec == 1)[0]
            np.random.shuffle(ones)
            h_vec[ones[:surplus]] = 0.0
        return h_vec

    def tournament_selection(fitness, tournament_size=5):
        candidates = np.random.randint(0, len(fitness), tournament_size)
        best_i = np.argmin(fitness[candidates])
        return candidates[best_i]

    np.random.seed(seed)
    pop_size = 150
    population = []

    h_sorted = np.argsort(h_init)

    population.append(binary_initial)

        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:target_sum]] = 1.0
        population.append(h)

    population = np.array(population)
    best_h = None
    best_c5 = np.inf

    # GA parameters
    mutation_rate_initial = 0.05
    mutation_rate_final = 0.01
    tournament_size = 5

    while time.time() - start_time < budget_s * 0.90:
        # Evaluate population

        for i in range(pop_size):
            h_vec = population[i]
            c5 = evaluate_c5(h_vec)
            fitness.append(c5)
        fitness = np.array(fitness)

        # Update best
        idx = np.argmin(fitness)
        current_c5 = fitness[idx]
        current_h = population[idx].copy()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        progress_ratio = max(0.0, min(1.0, (budget_s * 0.90 - remaining_time) / (budget_s * 0.90)))
        mutation_rate = max(mutation_rate_final, mutation_rate_initial * (1 - progress_ratio))

        for _ in range(pop_size // 2):
            # Select parents
            p1_idx = tournament_selection(fitness, tournament_size)
            p2_idx = tournament_selection(fitness, tournament_size)
            parent1 = population[p1_idx]

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = int(n_points / 2)

    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def repair_binary(h_vec, target):
        h_vec = h_vec.copy()
        current_sum = np.sum(h_vec)
        if current_sum == target:
            return h_vec
        delta = int(target - current_sum)
        ones_idx = np.where(h_vec == 1)[0]
        zeros_idx = np.where(h_vec == 0)[0]
        np.random.shuffle(ones_idx)
        np.random.shuffle(zeros_idx)
        if delta > 0:
            h_vec[zeros_idx[:delta]] = 1.0
        else:
            h_vec[ones_idx[:abs(delta)]] = 0.0
        return h_vec

    def tournament_select(fitness, size=5):
        candidates = np.random.randint(0, len(fitness), size)
        best_i = np.argmin(fitness[candidates])
        return candidates[best_i]

    np.random.seed(seed)
    pop_size = 120
    population = []
    h_sorted = np.argsort(h_init)
    binary_initial = np.zeros(n_points)
    binary_initial[h_sorted[-target_sum:]] = 1.0
    population.append(binary_initial)

    for _ in range(pop_size - 1):
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:target_sum]] = 1.0
        population.append(h)

    population = np.array(population)
    best_h = None
    best_c5 = float('inf')

    mutation_rate_init = 0.05
    mutation_rate_min = 0.01
    tournament_size = 5

    while time.time() - start_time < budget_s * 0.85:
        fitness = np.array([evaluate_c5(p) for p in population])
        idx = np.argmin(fitness)
        current_c5 = fitness[idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[idx].copy()

        mutation_rate = max(mutation_rate_min, 
                           mutation_rate_init * (1 - (time.time() - start_time) / (budget_s * 0.85)))

        offspring = []
        for _ in range(pop_size // 2):
            p1 = population[tournament_select(fitness, tournament_size)]
            p2 = population[tournament_select(fitness, tournament_size)]
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, p1, p2)
            child = repair_binary(child, target_sum)

            m = int(mutation_rate * n_points / 2)
            if m > 0:
                ones_idx = np.where(child == 1)[0]
                zeros_idx = np.where(child == 0)[0]
                np.random.shuffle(ones_idx)
                np.random.shuffle(zeros_idx)
                child[ones_idx[:m]] = 0.0
                child[zeros_idx[:m]] = 1.0
            offspring.append(child)

        unique_offspring = []
        for ind in offspring:
            is_unique = True
            for can in unique_offspring:
                if np.array_equal(ind, can):
                    is_unique = False
                    break
            if is_unique:
                unique_offspring.append(ind)
        offspring = np.array(unique_offspring[:pop_size // 2])

        combined = np.vstack([population, offspring])
        new_fitness = [evaluate_c5(c) for c in combined]
        sorted_idx = np.argsort(new_fitness)
        population = combined[sorted_idx[:pop_size]]

    if best_h is None:
        idx = np.argmin([evaluate_c5(p) for p in population])
        best_h = population[idx]
        best_c5 = evaluate_c5(best_h)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
