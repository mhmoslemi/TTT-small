def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guesses based on periodic patterns and binary thresholds
    x_vals = np.linspace(0, 2, n_points)
    rs = np.random.RandomState(seed)
    
    # Sine-based structured patterns
    structured_initials = []
    for period in [2.0, 1.5, 3.0]:
        for threshold in [0.5, 0.4, 0.6]:
            wave = np.sin(2 * np.pi * x_vals / period)
            h = (wave > threshold).astype(float)
            # Adjust sum to target
            current_sum = np.sum(h)
            if current_sum < target_sum:
                diff = int(target_sum - current_sum)
                indices = np.argsort(wave)
                h[indices[:diff]] = 1.0
            elif current_sum > target_sum:
                diff = int(current_sum - target_sum)
                indices = np.argsort(wave)
                h[indices[:diff]] = 0.0
            structured_initials.append(h)

    # Include initial_h_values and random patterns
    initial_guesses = [h_init.copy()]
    for _ in range(2):
        idxs = rs.permutation(n_points)
        h = np.zeros(n_points)
        h[idxs[:n_points//2]] = 1.0
        initial_guesses.append(h)
    for s in structured_initials:
        initial_guesses.append(s)

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Surrogate objective with weighted sum of overlaps
    def surrogate_objective(h_vec):
        h = h_vec.reshape(n_points)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        weights = []
        for i in range(len(corr_vals)):
            shift = i - (n_points - 1)
            disp = abs(shift) * dx
            weights.append(np.exp(-0.75 * disp))  # Tuned hyperparameter alpha
        return np.sum(corr_vals * np.array(weights)) * dx

    bounds = Bounds(0.0, 1.0)
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    # Binary conversion helper
    def make_binary(h_vec):
        h = h_vec.reshape(n_points)
        sorted_indices = np.argsort(h)
        h_bin = np.zeros_like(h)
        target_1s = int(target_sum)
        h_bin[sorted_indices[-target_1s:]] = 1.0
        return h_bin

    all_seeds = initial_guesses.copy()
    while time.time() - start_time < budget_s * 0.90:
        if all_seeds:
            h_candidate = all_seeds.pop(0)
        else:
            # Perturb best solution
            perturbed = best_h + rs.normal(0, 0.04, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            if abs(np.sum(perturbed) - target_sum) > 1e-6:
                perturbed = (perturbed / np.sum(perturbed)) * target_sum
            h_candidate = perturbed

        # Hybrid optimization loop
        current_h = h_candidate / np.sum(h_candidate) * target_sum
        for _ in range(2):
            # Continuous optimization
            result = minimize(
                fun=surrogate_objective,
                x0=current_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 80, 'ftol': 1e-8}
            )
            if result.success:
                current_h = result.x.reshape(n_points)
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h / np.sum(current_h) * target_sum

                # Evaluate raw constraint violation first
                corr = np.correlate(current_h, 1 - current_h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5

                # Binary refine
                h_bin = make_binary(current_h)
                result_bin = minimize(
                    fun=surrogate_objective,
                    x0=h_bin,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                if result_bin.success:
                    h_bin_opt = result_bin.x.reshape(n_points)
                    corr_bin = np.correlate(h_bin_opt, 1 - h_bin_opt, mode='full')
                    c5_bin = np.max(corr_bin) * dx
                    if c5_bin < best_c5:
                        best_h = h_bin_opt.copy()
                        best_c5 = c5_bin
            else:
                current_h = current_h / np.sum(current_h) * target_sum

            if time.time() - start_time >= budget_s * 0.90:
                break

    # Final constraint enforcement
    best_h = best_h / np.sum(best_h) * target_sum
    best_h = np.clip(best_h, 0, 1)
    
    assert abs(np.sum(best_h) - target_sum) < 1e-6, f"Mass constraint failed: {np.sum(best_h)}"
    assert np.all((best_h >= -1e-7) & (best_h <= 1.0001)), f"Value bounds failed"
    
    return (best_h.tolist(), best_c5, n_points)
