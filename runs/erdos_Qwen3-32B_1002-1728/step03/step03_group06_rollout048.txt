import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_renormalize(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        return (h_clipped / h_sum) * target_sum if h_sum > 1e-10 else h_clipped

    # Initialize swarm
    np.random.seed(seed)
    swarm_size = 100
    positions = []
    if len(h_init) > 0:
        positions.append(h_init.copy())
    for _ in range(swarm_size - 1):
        base = 0.5 + 0.1 * np.random.randn(n_points)
        base = clip_and_renormalize(base)
        positions.append(base)
    positions = np.array(positions)
    velocities = 0.01 * np.random.randn(swarm_size, n_points)

    # Initialize personal and global best
    pbest_positions = positions.copy()
    pbest_scores = np.array([objective(p) for p in positions])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    inertia = 0.7
    inertia_decay = 0.995
    cognitive = 1.2
    social = 1.2
    velocity_clamp = 0.1

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest_positions[i] - positions[i]) +
                             social * r2 * (gbest_position - positions[i]))
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)

            # Update position
            positions[i] = positions[i] + velocities[i]
            positions[i] = clip_and_renormalize(positions[i])

            # Evaluate new position
            score = objective(positions[i])
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = positions[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest_position = positions[i].copy()

        # Decay inertia for convergence
        inertia = max(0.4, inertia * inertia_decay)

    best_h = gbest_position
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), gbest_score, n_points)
