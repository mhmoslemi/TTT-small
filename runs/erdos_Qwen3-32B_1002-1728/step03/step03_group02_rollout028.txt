import time
import numpy as np
from scipy.optimize import minimize

def generate_structured_initial(n_points, period, threshold):
    x_vals = np.linspace(0, 2, n_points)
    sine_vals = np.sin(np.pi * x_vals / period)
    h = (sine_vals > threshold).astype(float)
    h = h / h.sum() * (n_points / 2.0)
    return np.clip(h, 0, 1)

def compute_weights(n_points, alpha):
    k_vals = np.arange(-n_points + 1, n_points)
    return np.exp(-alpha * np.abs(k_vals))

def surrogate_objective(h_vec, n_points, target_sum, weights, dx):
    h = h_vec.reshape(n_points)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / (h_sum + 1e-12) * target_sum
    corr = np.correlate(h, 1 - h, mode='full')
    return np.sum(corr * weights) * dx

def evaluate_c5(h_vec, n_points, dx):
    h = h_vec.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured initial guesses
    structured_inits = [
        generate_structured_initial(n_points, 2.0, 0.5),
        generate_structured_initial(n_points, 2.5, 0.4),
        generate_structured_initial(n_points, 1.5, 0.6),
        h_init.copy()
    ]
    np.random.RandomState(seed).shuffle(structured_inits)

    best_c5 = float('inf')
    best_h = h_init.copy()

    alphas = [0.1, 0.2, 0.05]
    for idx, h_initial in enumerate(structured_inits):
        if time.time() - start_time > budget_s * 0.98:

        bounds = [(0.0, 1.0)] * n_points

        result = minimize(

            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            args=(n_points, target_sum, weights, dx),
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
        )

        result = minimize(

            x0=perturbed,
