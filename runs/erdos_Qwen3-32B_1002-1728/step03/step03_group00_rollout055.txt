import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Because sum(h) * dx = 1 => sum(h) = n_points / 2.0

    def objective_full(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_full):
        return np.sum(h_full) - target_sum

    def normalize_perturbation(perturbed, target_sum, n_points):
        perturbed_clipped = np.clip(perturbed, 0.0, 1.0)
        s = np.sum(perturbed_clipped)
        if abs(s) < 1e-10:
            return np.zeros(n_points)
        perturbed_scaled = perturbed_clipped * (target_sum / s)
        return np.clip(perturbed_scaled, 0.0, 1.0)

    best_h = h_full_initial.copy()
    best_c5 = objective_full(best_h)

    np.random.seed(seed)
    n_attempt = 0
    time_per_iter = 0.0
    start_iter_time = 0.0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        subseed = seed + n_attempt

        if n_attempt == 1 or (elapsed - start_iter_time) < 10:
            perturbation_scale = 0.15
        else:
            perturbation_scale = 0.05

        rnd = np.random.RandomState(subseed)
        perturbed = best_h + rnd.normal(0, perturbation_scale, n_points)
        perturbed = normalize_perturbation(perturbed, target_sum, n_points)
        
        bounds_full = [(0.0, 1.0)] * n_points
        cons = ({'type': 'eq', 'fun': constraint_sum})
        
        result = minimize(
            fun=objective_full,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_full,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:

            new_c5 = objective_full(new_h)
            if np.isfinite(new_c5) and new_c5 < best_c5:

                start_iter_time = elapsed

        # Adaptive scale if progress stalls

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
