def repair_candidate(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-6:
        return h_clipped
    delta = target_sum - current_sum
    if delta > 0:
        indices = np.where(h_clipped < 1.0)[0]
        if len(indices) == 0:
            return h_clipped
        amount = delta / len(indices)
        for i in indices:
            h_clipped[i] = min(h_clipped[i] + amount, 1.0)
    else:
        indices = np.where(h_clipped > 0.0)[0]

            return h_clipped
        amount = -delta / len(indices)
        for i in indices:
            h_clipped[i] = max(h_clipped[i] - amount, 0.0)
    return h_clipped

def repair_candidate(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-6:
        return h_clipped
    delta = target_sum - current_sum
    if delta > 0:
        indices = np.where(h_clipped < 1.0)[0]
        if len(indices) == 0:
            return h_clipped
        amount = delta / len(indices)
        for i in indices:
            h_clipped[i] = min(h_clipped[i] + amount, 1.0)
    else:
        indices = np.where(h_clipped > 0.0)[0]
        if len(indices) == 0:
            h_clipped.fill(target_sum / len(h_clipped))
            return h_clipped
        amount = -delta / len(indices)
        for i in indices:
            h_clipped[i] = max(h_clipped[i] - amount, 0.0)
