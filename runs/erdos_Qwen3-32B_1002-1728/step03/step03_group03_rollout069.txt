import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h_flat = h_vec.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    best_h = h.copy()
    best_c5 = objective(h)

    while time.time() - start_time < budget_s * 0.95:
        # Perturb and normalize
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-6:
            perturbed = (perturbed / perturbed_sum) * target_sum
        
        # Optimize with tighter tolerance and more iterations
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        
        if result.success:
            h_candidate = np.clip(result.x, 0.0, 1.0)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6:
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
