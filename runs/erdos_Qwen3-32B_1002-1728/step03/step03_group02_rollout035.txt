        zeros = np.where(current_h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i, j = rs.choice(ones), rs.choice(zeros)
        new_h = current_h.copy()
        new_h[i], new_h[j] = 0.0, 1.0

        # Evaluate new candidate
        new_energy = objective_continuous(new_h)
        delta_energy = new_energy - current_energy

        # Acceptance criteria
        if delta_energy < 0 or rs.rand() < np.exp(-delta_energy/T):
            current_h, current_energy = new_h, new_energy

        if new_energy < best_energy:
            best_h, best_energy = new_h, new_energy

        T *= alpha  # Cooling

    # Final validation
    assert np.allclose(np.sum(best_h), target_sum), "Integral constraint violated"
    assert np.all((best_h >= 0) & (best_h <= 1)), "Value bounds violated"

    return (best_h.tolist(), best_energy, n_points)
