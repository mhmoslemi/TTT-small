import numpy as np
from scipy.optimize import minimize

def binary_h(h_vec, target_sum):
    n_points = len(h_vec)
    h_sorted = np.argsort(h_vec)
    n_ones = int(target_sum)
    h_binary = np.zeros(n_points)
    h_binary[h_sorted[-n_ones:]] = 1.0
    return h_binary

        for t in thresholds:
