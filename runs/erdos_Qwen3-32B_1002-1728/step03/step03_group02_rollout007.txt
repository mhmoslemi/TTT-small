        checkerboard = np.kron([1, 0], np.ones(n_points//2))[:n_points]
        structured.append(checkerboard)
        return structured

    rs = np.random.RandomState(seed)

    # Initialize population with diverse candidates
    population = [h_init.copy()]
    for k in generate_structured():
        population.append(k)
    for _ in range(5):
        indices = rs.permutation(n_points)
        h_r = np.zeros(n_points)
        h_r[indices[:n_points//2]] = 1
        population.append(h_r)
    for _ in range(6):
        perturbed = h_init + rs.normal(0, 0.03, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed /= np.sum(perturbed) * target_sum
        population.append(perturbed)

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    # Evaluate initial population
    for h in population:
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8}
        )
        if result.success:
            val = objective(result.x)
            if val < best_c5:
                best_h = result.x
                best_c5 = val

    # Main population-driven loop
    population = [minimize(objective, x0=h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 60}).x for h in population]

    while time.time() - start_time < 0.85 * budget_s and len(population) < 150:
        for k in range(min(5, len(population))):
            h_candidate = population[k].copy()
            # Perturb continuous
            perturbed = h_candidate + rs.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed /= np.sum(perturbed) * target_sum
            result = minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 90, 'ftol': 1e-8}
            )
            if result.success:
                val = objective(result.x)
                if val < best_c5:
                    best_h = result.x
                    best_c5 = val
                    population.append(result.x)

            # Perturb binary
            bin_h = binary_h(h_candidate, target_sum)
            idxs1 = np.argwhere(bin_h > 0.5).ravel()
            idxs0 = np.argwhere(bin_h < 0.5).ravel()
            for _ in range(3):
                if len(idxs1) and len(idxs0):
                    i, j = rs.choice(len(idxs1)), rs.choice(len(idxs0))
                    bin_h[[idxs1[i], idxs0[j]]] = 1 - bin_h[[idxs1[i], idxs0[j]]]
            result = minimize(
                fun=objective,
                x0=bin_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if result.success:
                val = objective(result.x)
                if val < best_c5:
                    best_h = result.x
                    best_c5 = val
                    population.append(result.x)

        population.sort(key=lambda x: objective(x))
        population = population[:100]  # Limit population size

    if time.time() - start_time < 0.9 * budget_s:
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-10}
        )
        if result.success:
            val = objective(result.x)
            if val < best_c5:
                best_h = result.x
                best_c5 = val

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    return (best_h.tolist(), best_c5, n_points)
