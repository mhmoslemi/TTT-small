from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to meet the integral constraint
    current_sum = np.sum(h_full_initial)
    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        h_full_initial = h_full_initial * (target_sum / current_sum)
    h_full_initial = np.clip(h_full_initial, 0, 1)
    best_h = h_full_initial.copy()
    best_c5 = maximize_correlation(best_h, dx)

    def objective_func(h):
        return maximize_correlation(h, dx)

    np.random.seed(seed)
    attempts = 0
    max_attempts = 100  # Example upper bound; limited by time

    while time.time() - start_time < budget_s - 10:
        attempts += 1
        if attempts > max_attempts:
            break
        # Generate perturbed guess with larger step size
        perturbed = best_h + np.random.normal(0, 0.2, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        
        # Optimize
        result = minimize(

            x0=perturbed,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],

        if result.success:

                new_c5 = objective_func(new_h)

    # Final checks

def maximize_correlation(h, dx):
