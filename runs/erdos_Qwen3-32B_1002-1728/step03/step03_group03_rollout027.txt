        h = h / h.sum() * target_sum  # Ensure sum constraint

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final adjustment to ensure constraint
    best_h = best_h / best_h.sum() * target_sum
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h.tolist(), best_c5, n_points)
