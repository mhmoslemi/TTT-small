                    perturbed1 = np.clip(perturbed1, 0, 1)
                    perturbed1_sum = np.sum(perturbed1)
                    if abs(perturbed1_sum - target_sum) > 1e-6:
                        perturbed1 = (perturbed1 / perturbed1_sum) * target_sum
                    additional_initials.append(perturbed1.copy())
                for _ in range(2):
                    perturbed2 = best_h + rs.normal(0, 0.01, n_points)
                    perturbed2 = np.clip(perturbed2, 0, 1)
                    perturbed2_sum = np.sum(perturbed2)
                    if abs(perturbed2_sum - target_sum) > 1e-6:
                        perturbed2 = (perturbed2 / perturbed2_sum) * target_sum
                    additional_initials.append(perturbed2.copy())

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
