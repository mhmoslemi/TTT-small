def evaluate_c5(h, dx):
    n_points = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_binary(child, m):
    current_sum = int(np.sum(child))
    if current_sum == m:
        return child.copy()
    if current_sum < m:
        delta = m - current_sum
        zeros = np.where(child == 0)[0]
        if len(zeros) >= delta:
            child[zeros[:delta]] = 1.0
    else:
        delta = current_sum - m
        ones = np.where(child == 1)[0]
        if len(ones) >= delta:
            child[ones[:delta]] = 0.0
    return child.copy()

def mutate_binary(child, m, n_swaps):
    child = child.copy()
    for _ in range(n_swaps):
        ones = np.where(child == 1.0)[0]
        zeros = np.where(child == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        idx1 = np.random.choice(ones)
        idx2 = np.random.choice(zeros)
        child[[idx1, idx2]] = [0.0, 1.0]
    return child

    pop_size = 50
    population = [h_binary.copy()]
    for _ in range(pop_size - 1):
        indices = np.random.choice(n_points, size=target_sum, replace=False)
        new_h = np.zeros(n_points)
        new_h[indices] = 1.0
        population.append(new_h)

    best_c5 = float('inf')
    best_h = h_binary.copy()

    while time.time() - start_time < budget_s * 0.95:
        c5_values = np.array([evaluate_c5(ind, dx) for ind in population])
        min_c5_idx = np.argmin(c5_values)
        cur_min_c5 = c5_values[min_c5_idx]
        if cur_min_c5 < best_c5:
            best_h = population[min_c5_idx].copy()
            best_c5 = cur_min_c5

        selected_indices = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, 5, replace=False)
            tournament_values = c5_values[tournament]
            winner_idx = tournament[np.argmin(tournament_values)]
            selected_indices.append(winner_idx)

        new_population = []
        for i in range(0, pop_size, 2):
            if i+1 < pop_size:
                parent1 = population[selected_indices[i]]
                parent2 = population[selected_indices[i+1]]
            else:
                parent1 = population[selected_indices[i]]
                parent2 = population[selected_indices[i-1]]
            child = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child[j] = parent1[j]
                else:
                    child[j] = parent2[j]
            child = repair_binary(child, target_sum)
            if np.random.rand() < 0.1:
                child = mutate_binary(child, target_sum, n_swaps=2)
            new_population.append(child)

        population = new_population[:pop_size]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = minimize(
        fun=lambda h: evaluate_c5(h, dx),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if result.success:
        h_opt = result.x
        sum_opt = np.sum(h_opt)
        if abs(sum_opt - target_sum) < 1e-6:
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
