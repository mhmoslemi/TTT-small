        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    best_c5 = objective(best_h)
    best_result = None

    while time.time() - start_time < budget_s:

        # Try different optimizers with early stopping
