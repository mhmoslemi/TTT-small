    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed + int(time.time()))
        if rs.rand() < 0.8:
            # Binary-based perturbation with enhanced exploration
            m = int(round(target_sum))
            indices = np.argpartition(h_init, -m)[-m:]
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, 0.07, n_points)
        else:
            # Continuous perturbation with widened noise
            perturbed = best_h + rs.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Remove the problematic scaling step
        # and rely on optimizer constraint enforcement
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.98:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
