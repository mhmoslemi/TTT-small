        # Generate offspring
        for _ in range(population_size - elite_count):
            # Tournament selection
            def select():
                return population[ranks[random.choice(range(len(population)))]]
            p1 = select()
            p2 = select()
            
            # One-point crossover
            cp = np.random.randint(1, n_points)
            child = np.concatenate([p1[:cp], p2[cp:]])
            
            # Mutation
            if random.random() < 0.2:
                mut_points = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                noise = np.random.normal(0, 0.1, len(mut_points))
                child[mut_points] += noise
            
            child = repair_candidate(child)
            new_pop.append(child)
        
        population = new_pop

    assert best_h is not None
    return (best_h.tolist(), best_c5, n_points)
