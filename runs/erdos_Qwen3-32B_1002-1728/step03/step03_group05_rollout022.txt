import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def de_objective(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h < 1e-9:
            return float('inf')
        h_normalized = h_clipped * (target_sum / sum_h)
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Differential Evolution with time guard
    if time.time() - start_time < budget_s * 0.8:
        try:
            de_result = differential_evolution(
                de_objective,
                bounds,
                seed=seed,
                popsize=12,
                mutation=(0.5, 1.0),
                recombination=0.85,
                strategy='best1bin',
                tol=1e-6,
                maxiter=60,
                workers=1
            )
            if de_result.success:
                h_candidate = de_result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                sum_h = np.sum(h_candidate)
                if sum_h > 1e-9:
                    h_candidate *= target_sum / sum_h
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass

    # Local refinement
    def opt_objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Multi-start local search
    n_trials = 3
    for i in range(n_trials):
        if time.time() - start_time > budget_s * 0.95:
            break
        if i == 0:
            x0 = best_h
        else:
            x0 = np.random.RandomState(seed + i).uniform(0, 1, n_points)
            x0 = np.clip(x0, 0, 1)
            sum_x0 = np.sum(x0)
            if sum_x0 > 1e-6:
                x0 *= target_sum / sum_x0
            else:
                x0[0] = target_sum
        result = minimize(
            opt_objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200}
        )
        if result.success:
            h_candidate = result.x
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    return (best_h.tolist(), best_c5, n_points)
