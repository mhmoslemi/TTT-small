        h = repair(h)
        population.append(h)

    best_c5 = np.inf
    best_h = h_init.copy()

    while time.time() - time_start < budget_s:
        fitness = evaluate(population)
        # Update best
        for i in range(len(population)):
            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = population[i].copy()
        # Create next generation
        new_pop = []
        # Elites
        elite_indices = np.argsort(fitness)[:elites_num]
        for idx in elite_indices:
            new_pop.append(population[idx].copy())

        # Generate fillers
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_sigma, n_points)
                child += noise
            child = repair(child)
            new_pop.append(child)
        population = new_pop

    # Final checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
