import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_with_penalty(h_vec):
        h = h_vec.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5 = np.max(corr) * dx
        sum_penalty = 1000.0 * (current_sum - target_sum) ** 2
        return c5 + sum_penalty

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 1e-9:
            h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initializations
    initializations = [h_init.copy()]
    rs = np.random.RandomState(seed)
    for _ in range(10):  # Random binary patterns
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)

    # Sinusoidal patterns
    for phase in [0.0, np.pi / 2, np.pi]:
        x = 2 * np.pi * np.arange(n_points) / n_points + phase
        h = 0.5 + 0.5 * np.sin(x)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        initializations.append(h)

    # Linear ramp patterns
    x = np.linspace(0, 1, n_points)
    x_rev = np.linspace(1, 0, n_points)
    h_ramp = x * (target_sum / np.sum(x))
    h_ramp_rev = x_rev * (target_sum / np.sum(x_rev))
    initializations.append(h_ramp)
    initializations.append(h_ramp_rev)

    # Flatten initializations
    initializations = [i.ravel() for i in initializations]

    # Nelder-Mead global search phase
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    for h0 in initializations:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            fun=objective_with_penalty,
            x0=h0,
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 50, 'xatol': 1e-4, 'fatol': 1e-4}

        h_candidate = result.x.copy()
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:

            best_c5 = c5_candidate

    # SLSQP refinement phase
    while time.time() - start_time < budget_s * 0.95:
        perturb = best_h.copy()
        perturb += (rs.rand(n_points) - 0.5) * 0.1
        perturb = np.clip(perturb, 0, 1)
        cur_sum = np.sum(perturb)
        if cur_sum > 1e-9:
            perturb *= (target_sum / cur_sum)
        result = minimize(
            fun=compute_c5,

            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],

    best_h = np.clip(best_h, 0, 1)

    # Validate constraints
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
