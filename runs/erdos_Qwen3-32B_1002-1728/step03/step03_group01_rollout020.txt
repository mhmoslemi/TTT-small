    population.append(1 - h_ramp)

    # Normalize population members
    pop_norm = []
    for h in population:
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s < 1e-9:
            continue
        h_norm = h_clipped * target_sum / s
        pop_norm.append(h_norm)

    # Run DE with constraints via normalization
    de_pop_size = 20
    de_args = {
        'strategy': 'best1bin',
        'mutation': (0.5, 1),
        'recombination': 0.7,
        'seed': seed,
        'maxiter': 25,
        'popsize': de_pop_size,
        'tol': 1e-6,
        'polish': False,
        'init': pop_norm,
        'workers': -1,  # Use all threads if available (but only 1 CPU is specified)
    }

    de_result = differential_evolution(
        objective_de,
        bounds,
        **de_args
    )

    best_h_de = de_result.x
    h_clipped = np.clip(best_h_de, 0, 1)
    s = np.sum(h_clipped)
    if s < 1e-9:
        best_h_de_norm = h_clipped.copy()
    else:
        best_h_de_norm = h_clipped * target_sum / s

    # SLSQP refinement on best DE solution
    result = minimize(
        fun=objective_slsqp,
        x0=best_h_de_norm,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    best_h = best_h_de_norm.copy()
    best_c5 = de_result.fun

    if result.success:
        candidate_h = result.x
        candidate_c5 = objective_slsqp(candidate_h)
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5

    # Local refinements with perturbations
    while time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s * 0.97:
            break
        perturbed = best_h.copy()
        perturbed += (rs.rand(n_points) - 0.5) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if s < 1e-9:
            continue
        perturbed *= target_sum / s
        result = minimize(
            fun=objective_slsqp,
            x0=perturbed,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            method='SLSQP',
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            candidate_h = result.x
            candidate_c5 = objective_slsqp(candidate_h)
            if candidate_c5 < best_c5:
                best_h[:] = candidate_h[:]
                best_c5 = candidate_c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
