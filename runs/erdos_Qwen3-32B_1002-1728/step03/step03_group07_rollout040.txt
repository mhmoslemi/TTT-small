        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        C5 = corr[max_idx] * dx
        k_star = int(max_k)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i - k_star
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
        return C5, grad

    def project(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum, atol=1e-8):
            h *= target_sum / current_sum
            h = np.clip(h, 0, 1)
        return h.flatten()

    def objective(h_vec):
        h = project(h_vec)
        C5, _ = compute_C5_grad(h)
        return C5

    def gradient(h_vec):
        h = project(h_vec)
        _, grad = compute_C5_grad(h)
        return grad

    best_C5 = np.inf
    best_h = h0.copy()

    np.random.seed(seed)
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.99:
            break
        sigma = 0.1 * (0.5 ** restart)
        if restart == 0:
            current_init = h0.copy()
        else:
            pert = np.random.normal(0, sigma, n_points)
            current_init = np.clip(h0 + pert, 0, 1)
            current_init = current_init / current_init.sum() * target_sum

        result = opt.minimize(
            objective,
            current_init,
            method='L-BFGS-B',
            jac=gradient,
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        if result.success:
            new_h = result.x
            new_C5 = objective(new_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()

        if restart % 2 == 0 and restart > 0:
            pert = np.random.uniform(-0.05, 0.05, best_h.shape)
            best_h = np.clip(best_h + pert, 0, 1)
            best_h = best_h / best_h.sum() * target_sum

    best_h = project(best_h)
    best_C5 = objective(best_h)

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all(np.isfinite(best_h))
    return (best_h.tolist(), best_C5, n_points)
