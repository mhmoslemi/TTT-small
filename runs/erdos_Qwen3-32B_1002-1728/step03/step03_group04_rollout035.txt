import time
import numpy as np
from scipy.optimize import basinhopping

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        h_projected = h_clipped * (target_sum / sum_h)
        corr = np.correlate(h_projected, 1 - h_projected, mode='full')

    def projection(x):
        h = x.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        h_projected = h_clipped * (target_sum / sum_h)
        return h_projected.flatten()

    local_min_args = {
        'method': 'L-BFGS-B',
        'bounds': [(0, 1)] * n_points,
        'options': {'ftol': 1e-8, 'maxfun': 150}
    }

    remaining_time = budget_s * 0.95
    current_h = h_init.copy()
    step_size = 0.25

    while remaining_time > 10:  # Reserve time for final checks
        max_iter = int(remaining_time / 1.5)
        if max_iter < 5:
            break
        result = basinhopping(
            objective,
            current_h,
            niter=min(max_iter, 30),
            T=step_size,
            minimizer_kwargs=local_min_args,
            take_step=lambda x: x + np.random.uniform(-step_size, step_size, len(x)),
            interval=50,
            seed=seed
        )
        current_h = result.x
        step_size *= 0.95
        remaining_time = budget_s * 0.95 - (time.time() - start_time)

    # Final projection and validation
    h = current_h.reshape(n_points)
    h = projection(h)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx

    assert abs(np.sum(h) - target_sum) < 1e-6
    assert np.all((h >= -1e-6) & (h <= 1.0 + 1e-6))
    return (h.tolist(), c5, n_points)
