    def binary_h(h_vec):
        h = h_vec.copy()
        h_sorted = np.argsort(h)
        n_ones = int(target_sum)
        h_binary = np.zeros(n_points)
        h_binary[h_sorted[-n_ones:]] = 1.0
        return h_binary

    alphas = [0.1, 0.05, 0.2]  # Try different alphas for the surrogate

    def optimize_with_alpha(h0, alpha_val):
        result = minimize(
            fun=lambda h: surrogate_objective(h, alpha_val),
            x0=h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            return result.x
        return h0

    all_initials = [h_init] + additional_initials

    while time.time() - start_time < budget_s * 0.90:
        if not all_initials:
            # Perturb the best solution
            perturbed = best_h + rs.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if abs(perturbed_sum - target_sum) > 1e-6:
                perturbed = (perturbed / perturbed_sum) * target_sum
            all_initials.append(perturbed)
        current_initial = all_initials.pop(0)
        current_h = current_initial.copy()

        for alpha in alphas:
            # Continuous optimization with surrogate
            optimized = optimize_with_alpha(current_h, alpha)
            if optimized is not None:
                current_h = optimized
            # Binarize
            bin_h = binary_h(current_h)
            result_bin = minimize(
                fun=lambda h: surrogate_objective(h, alpha),
                x0=bin_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}

            if result_bin.success:
                bin_optimized = result_bin.x
                if np.all((bin_optimized >= -1e-6) & (bin_optimized <= 1 + 1e-6)):
                    current_h = bin_optimized

        # Compute actual C5 for the final h
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx

    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6)), "Bounds violated"
    return (best_h.tolist(), final_c5, n_points)
