import time
import numpy as np
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_template = np.array(initial_h_values)
    n_points = len(h_template)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize binary solution with correct sum
    h_binary = np.zeros(n_points)
    top_indices = np.argsort(h_template)[-int(target_sum):]
    h_binary[top_indices] = 1.0

    best_h = h_binary.copy()
    corr = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Set up simulated annealing
    np.random.seed(seed)
    temp = 15.0
    cooling_rate = 0.9995
    iteration = 0

    while time.time() - start_time < budget_s:
        # Generate neighbor by swapping 0 and 1
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_candidate = best_h.copy()
        h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]

        # Compute new C5
        corr_candidate = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx

        # Metropolis criterion
        delta = c5_candidate - best_c5
        if delta <= 0 or np.random.rand() < math.exp(-delta / temp):
            best_h = h_candidate
            best_c5 = c5_candidate

        temp *= cooling_rate
        iteration += 1

        # Reset temperature occasionally
        if iteration % 300 == 0:
            temp = max(temp * 0.3, 0.2)

    # Final constraint checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
