import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Compute initial overlaps
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return corr * dx

    def get_worst_case(h_vec):
        overlaps = compute_overlap(h_vec)
        valid_indices = np.arange(1, len(overlaps) - 1)
        return valid_indices[np.argmax(overlaps[valid_indices])]

    best_h = h.copy()
    best_c5 = np.max(compute_overlap(h)) * dx
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Identify worst-case lag
        k_max = get_worst_case(best_h)
        k_shift = int(k_max)
        
        # Compute gradient at h: g(x) = 1 - h(x + k)
        # Need to handle shift carefully
        masked_shift = int(np.round(2 * (n_points - k_shift) / 2))
        g = 1 - np.roll(best_h, k_shift)[n_points-1:]
        g = np.concatenate([g, np.zeros(k_shift)])  # Align with forward shift

        # Linearized LP optimization
        delta = cp.Variable(n_points)
        constraints = [
            cp.sum(delta) == 0,
            0 <= best_h + delta, 
            best_h + delta <= 1
        ]
        objective = cp.Minimize(delta @ g)
        prob = cp.Problem(objective, constraints)

        try:
            result = prob.solve(solver='ECOS', verbose=False)
            if result is not None and delta.value is not None:
                new_h = best_h + delta.value
                new_h = np.clip(new_h, 0, 1)
                
                # Renormalize if needed
                current_sum = np.sum(new_h)
                if not np.isclose(current_sum, target_sum):
                    scale = target_sum / current_sum
                    new_h = np.clip(new_h * scale, 0, 1)
                
                # Evaluate candidate
                c5_candidate = np.max(compute_overlap(new_h)) * dx
                if c5_candidate < best_c5:
