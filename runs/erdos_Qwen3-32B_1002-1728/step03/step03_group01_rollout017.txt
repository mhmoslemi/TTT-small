import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    original_h = np.array(initial_h_values)
    original_n = len(original_h)
    n_points = 100
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    x_orig = np.linspace(0, 2, original_n, endpoint=False)
    x_new = np.linspace(0, 2, n_points, endpoint=False)
    idx = np.searchsorted(x_orig, x_new, side='right') - 1
    idx = np.clip(idx, 0, original_n - 1)
    h_init = original_h[idx]

    def process_params(h_vec, target_sum=target_sum):
        h = np.clip(h_vec, 0.0, 1.0)
        current_sum = h.sum()
        if current_sum < 1e-9:
            h = np.full_like(h, target_sum / n_points)
        else:
            h *= (target_sum / current_sum)
            h = np.clip(h, 0.0, 1.0)
        return h

    def objective_de(x):
        h_processed = process_params(x)
        corr = np.correlate(h_processed, 1 - h_processed, mode='full')
        return np.max(corr) * dx

    rs = np.random.RandomState(seed)
    initial_population = [h_init.copy()]
    for _ in range(5):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initial_population.append(h)

    for _ in range(2):
        freq = rs.uniform(0.1, 0.5)
        phase = rs.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
        h = np.clip(h, 0.0, 1.0)
        initial_population.append(h)

    ramp = np.linspace(0, 1, n_points // 2)
    ramp = np.concatenate((ramp, np.linspace(1, 0, n_points - n_points // 2)))
    initial_population.append(ramp)
    ramp_linear = np.linspace(0, 1, n_points)
    initial_population.append(ramp_linear)

    initial_candidates = [process_params(h.copy()) for h in initial_population]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result_de = differential_evolution(
        func=objective_de,
        bounds=bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-4,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=rs,
        init=np.array(initial_candidates),
        maxiter=int(200),
        disp=False
    )

    best_de = process_params(result_de.x)
    c5_de = objective_de(result_de.x)

    best_h = best_de.copy()
    best_c5 = c5_de

    def objective_slsqp(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    bounds_sl = [(0.0, 1.0) for _ in range(n_points)]
    cons_sl = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    if time.time() - start_time < budget_s:
        result_slsqp = minimize(
            fun=objective_slsqp,
            x0=best_de,
            method='SLSQP',
            bounds=bounds_sl,
            constraints=cons_sl,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result_slsqp.success:
            candidate_h = result_slsqp.x
            corr = np.correlate(candidate_h, 1 - candidate_h, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = candidate_h
                best_c5 = c5_candidate

    while time.time() - start_time < budget_s * 0.97:
        perturbed = best_h.copy()
        perturbed += (rs.rand(n_points) - 0.5) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-9:
            perturbed *= target_sum / perturbed_sum
        result = minimize(
            fun=objective_slsqp,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_sl,
            constraints=cons_sl,
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.99:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
