import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_c5 = np.inf
    best_h = h.copy()

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Initialize best_c5 with the current value
    best_c5 = compute_c5(h)
    max_iter = 1000  # Max number of iterations to try
    no_improve_count = 0
    epsilon = 0.1  # Perturbation bound

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute current C5 and find k_max index
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        # Convert to displacement d
        d = max_index - (n_points - 1)

        # Compute gradient for displacement d
        g = np.zeros(n_points)
        for i in range(n_points):
            j = i + d
            if 0 <= j < n_points:
                g[i] += 1.0 - h[j]
        for j in range(n_points):
            i = j - d
            if 0 <= i < n_points:
                g[j] -= h[i]

        # Setup LP problem
        delta = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(g * delta))
        constraints = [
            delta >= -h,  # delta[i] >= -h[i] → h[i] + delta[i] ≥ 0
            delta <= 1 - h,  # delta[i] ≤ 1 - h[i] → h[i] + delta[i] ≤ 1
            cp.sum(delta) == 0,  # keep the sum the same
            cp.norm(delta, 'inf') <= epsilon
        ]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            new_h = h + delta.value
            new_sum = np.sum(new_h)
            assert abs(new_sum - target_sum) < 1e-6, "Sum constraint failed"
            h = new_h
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve_count = 0
                epsilon = 0.1  # Reset epsilon
            else:
                # No improvement, decrease epsilon
                epsilon *= 0.9
                no_improve_count += 1
                if no_improve_count > 20 and epsilon > 1e-3:
                    # Perturb randomly after 20 iterations without improvement
                    np.random.seed(seed)
                    perturbation = (np.random.rand(n_points) - 0.5) * 0.1
                    perturbation = np.clip(perturbation, -h, 1 - h)
                    perturbation_sum = np.sum(perturbation)
                    perturbation -= perturbation_sum / n_points
                    h = np.clip(h + perturbation, 0, 1)
                    # Ensure sum constraint
                    scale_factor = target_sum / np.sum(h)
                    h = h * scale_factor
                    epsilon = 0.1
                    no_improve_count = 0
        except:
            # Solver error: reduce epsilon and continue
            epsilon *= 0.5
            no_improve_count += 1
            continue

    # Final validation
    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6)), "Values out of bounds"
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    return (best_h.tolist(), best_c5, n_points)
