        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate multiple initial points
    h0_list = [h_init]
    h_uni = np.ones(n_points) * 0.5
    h0_list.append(h_uni)
    h_tri = create_triangle(n_points)
    h0_list.append(h_tri)

    best_c5 = float('inf')
    best_h = h_init.copy()

    for h0 in h0_list:
        h0 = np.array(h0).copy()
        # Normalize to target_sum
        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        if abs(current_sum - target_sum) > 1e-6:
            scaling_factor = target_sum / current_sum
            h0 = np.clip(h0 * scaling_factor, 0, 1.0)
        # Run optimization
        result = minimize(
            fun=objective,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-7},
            jac=None
        )
        if result.success:
            h_candidate = result.x
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate

    # Perturbations and further optimization
    while time.time() - start_time < budget_s * 0.90:
        rs = np.random.RandomState(seed)
        seed += 1
        perturb_factor = 0.05
        h_perturbed = best_h + rs.normal(0, perturb_factor, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum != 0:
            h_perturbed = (h_perturbed / current_sum) * target_sum
        else:
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = (h_perturbed / h_perturbed.sum()) * target_sum
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-7},
            jac=None
        )
        if result.success:
            h_candidate = result.x
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate

    # Final assertion checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
