import time
import numpy as np

def compute_smoothed_objective(h, dx, smoothing_eps):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    exp_vals = np.exp((corr - max_val) / smoothing_eps)
    sum_exp = np.sum(exp_vals)
    return (max_val + smoothing_eps * np.log(sum_exp)) * dx

def compute_gradient(h, dx, smoothing_eps):
    delta = 1e-5
    n = len(h)
    gradient = np.zeros(n)
    for i in range(n):
        h_up = h.copy()
        h_up[i] += delta
        obj_up = compute_smoothed_objective(h_up, dx, smoothing_eps)
        h_down = h.copy()
        h_down[i] -= delta
        obj_down = compute_smoothed_objective(h_down, dx, smoothing_eps)
        gradient[i] = (obj_up - obj_down) / (2 * delta)
    return gradient

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
