import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Function to perturb h
    def perturb_h(h, seed=None):
        rng = np.random.RandomState(seed)
        h_perturbed = h.copy()
        # Apply small random perturbations
        perturbation = rng.uniform(-0.1, 0.1, size=n_points)
        h_perturbed += perturbation
        # Clip to [0, 1]
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Rescale to meet sum constraint
        current_sum = np.sum(h_perturbed)
        if current_sum < 1e-6:
            return h.copy()
        h_perturbed = h_perturbed * target_sum / current_sum
        return h_perturbed

    # Objective function
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        # Add penalty for sum constraint
        current_sum = np.sum(h)
        penalty = 1.0e4 * (current_sum - target_sum)**2
        return c5 + penalty

    # Gradient function
    def gradient(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        k_shift = max_corr_index - (n_points - 1)
        # Compute gradient for the dominant shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_shift < n_points:
                grad[i] += 1 - h[i + k_shift]
            if i - k_shift >= 0:
                grad[i] -= h[i - k_shift]
        grad *= dx
        # Add gradient of the penalty term
        current_sum = np.sum(h)
        grad += 2 * 1e4 * (current_sum - target_sum)
        return grad.flatten()

    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_energy = np.inf
    best_h = h_init.copy()

    # Run multiple optimizations
    np.random.seed(seed)
    num_starts = 5
    for start_idx in range(num_starts):
        # Generate perturbed start
        perturbation_seed = seed + start_idx
        h_start = perturb_h(h_init, seed=perturbation_seed)

        # Run optimization
        result = minimize(
            fun=objective,
            x0=h_start,
            jac=gradient,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'gtol': 1e-6}
        )
        h_opt = result.x.reshape(n_points)
        current_energy = objective(h_opt)

        # Compute actual C5 without penalty
        current_sum = np.sum(h_opt)
        true_energy = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx

        if true_energy < best_energy:
            best_energy = true_energy
            best_h = h_opt.copy()

        # Check if time is up
        if time.time() - start_time > budget_s * 0.98:  # leave some buffer
            break

    # Final projection to meet sum constraint
    best_h = best_h * (target_sum / np.sum(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)

    # Recompute the C5 after projection
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), final_c5, n_points)

This code attempts to run 5 optimizations with perturbed initializations and uses L-BFGS-B with gradient-based optimization. The penalty coefficient was adjusted from 1e6 to 1e4 to prevent the penalty from overwhelming the C5 objective during optimization. It also includes a projection step after optimization to ensure the sum constraint is met.

Note that the code may require adjustments to the penalty coefficient or the perturbation magnitude to balance between the C5 objective and the constraint. Also, the number of starts may be adjusted according to the time budget.

This approach is different from the previous simulated annealing method and leverages gradient-based optimization, which should be more efficient and potentially find better solutions.
