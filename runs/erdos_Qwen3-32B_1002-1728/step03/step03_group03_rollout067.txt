import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h_flat = np.clip(h_vec, 0.0, 1.0)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    def normalize(h_vec):
        h_clipped = np.clip(h_vec, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0.0, 1.0)

    h = normalize(h)

    # Initial optimization
    for _ in range(2):
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8}
        )
        if result.success:
            h = normalize(result.x)
        if time.time() - start_time > budget_s * 0.75:
            break

    best_h = h.copy()
    best_c5 = objective(best_h)

    # Perturbation loop
    max_perturbations = 50
    perturbation_count = 0
    while time.time() - start_time < budget_s * 0.98 and perturbation_count < max_perturbations:
        perturbation_count += 1
        np.random.seed(int(time.time() % 1000) ^ seed ^ perturbation_count)
        noise_scale = np.random.uniform(0.02, 0.1)
        perturbed = best_h + np.random.normal(0, noise_scale, n_points)
        perturbed = normalize(perturbed)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-8}
        )
        if result.success:
            candidate = normalize(result.x)
            c5_candidate = objective(candidate)
            if c5_candidate < best_c5:
                best_h = candidate
                best_c5 = c5_candidate

    # Final refinement
    for _ in range(3):
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            best_h = normalize(result.x)
            best_c5 = objective(best_h)

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
