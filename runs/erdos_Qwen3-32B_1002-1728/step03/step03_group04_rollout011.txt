def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np
    from scipy.optimize import minimize
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    rs = np.random.RandomState(seed)

    while time.time() - start_time < budget_s * 0.95:
        current_time = time.time() - start_time
        frac = current_time / (budget_s * 0.95)
        noise_std = max(0.01, 0.3 - 0.25 * frac)

        choice = rs.choice(3)
        if choice == 0:
            # Continuous perturbation
            perturbed = best_h + rs.normal(0, noise_std, n_points)
        elif choice == 1:
            # Binary perturbation with random indices
            m = int(round(target_sum))
            indices = rs.choice(n_points, size=m, replace=False)
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, noise_std, n_points)
        else:
            # Shifted perturbation
            delta = rs.randint(1, n_points // 2)
            shifted = np.roll(best_h, delta)
            perturbed = shifted + rs.normal(0, noise_std, n_points)

        # Ensure feasibility
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-9:
            perturbed = (perturbed / perturbed_sum) * target_sum

        # Local optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):

    return (best_h.tolist(), best_c5, n_points)

import time

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    h_init = np.array(initial_h_values)
    n_points = len(h_init)

    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    rs = np.random.RandomState(seed)

    while time.time() - start_time < budget_s * 0.95:

        noise_std = max(0.05, 0.25 - 0.2 * frac)

        choice = rs.choice(3)

            m = int(np.round(target_sum))

            shifted = np.roll(best_h, delta)

    return (best_h.tolist(), best_c5, n_points)
