import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Increase number of workers and perturbation scale
    n_workers = 10
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
        rs = np.random.RandomState(seed + w)
        perturbed = best_h + rs.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if abs(s) > 1e-9:
            perturbed = perturbed * (target_sum / s)
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
                if np.max(corr_candidate) < best_c5:
                    best_h = h_candidate
                    best_c5 = np.max(corr_candidate)

    # Final gradient descent refinement
    result_refine = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-10,
        options={'maxiter': 500, 'ftol': 1e-10}
    )
    if result_refine.success:
        final_h = result_refine.x
        final_sum = np.sum(final_h)
        if abs(final_sum - target_sum) < 1e-6 and np.all((final_h >= -1e-6) & (final_h <= 1.0 + 1e-6)):
            return (final_h.tolist(), np.max(np.correlate(final_h, 1 - final_h, 'full') * dx), n_points)

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
