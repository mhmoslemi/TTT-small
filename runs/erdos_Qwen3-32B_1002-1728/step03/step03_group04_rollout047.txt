import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Perturbation loop with hybrid approaches
    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed)
        elapsed = time.time() - start_time
        noise_std = 0.1 * (1.0 - elapsed / (budget_s * 0.95))  # Noise decays over time
        perturbed = np.copy(best_h)
        
        if rs.rand() < 0.33:
            # Continuous perturbation
            perturbed += rs.normal(0, noise_std, n_points)
        elif rs.rand() < 0.5:
            # Binary pattern perturbation
            m = int(round(target_sum))
            indices = rs.choice(n_points, m, replace=False)
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, noise_std, n_points)
        else:
            # Shift perturbation
            shift = rs.randint(n_points)
            perturbed = np.roll(best_h, shift) + rs.normal(0, noise_std, n_points)

        # Clip and normalize to constraints
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * target_sum

        # Local optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= -1e-6) & (h_candidate <= 1.0 + 1e-6)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        seed += 1  # Ensure next perturbation has different randomness

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
