import time
import numpy as np
import math

def evaluate_c5(h_vec, dx):
    h = h_vec.reshape(-1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    assert target_sum.is_integer(), "n_points must be even"
    target_sum = int(target_sum)

    # Genetic Algorithm setup
    POP_SIZE = 80
    NUM_GENERATIONS = 1000
    MUTATION_PROB = 0.1

    # Initialize population with initial solution and random binvecs
    pop = np.zeros((POP_SIZE, n_points))
    pop[0] = initial_h  # Use the initial construction
    for i in range(1, POP_SIZE):
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:target_sum]] = 1.0
        pop[i] = h

    best_c5 = math.inf
    best_h = None

    # Evaluate initial population
    c5_values = np.zeros(POP_SIZE)
    for i in range(POP_SIZE):
        c5 = evaluate_c5(pop[i], dx)
        c5_values[i] = c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = pop[i].copy()

    # Genetic Algorithm loop
    for gen in range(NUM_GENERATIONS):
        if time.time() - start_time > budget_s * 0.98:
            break

        # Selection (tournament)
        parent_indices = np.zeros(2 * POP_SIZE // 5, dtype=int)  # Top 40% of population size
        for i in range(len(parent_indices)):
            samples = np.random.choice(POP_SIZE, 5)
            parent_indices[i] = samples[np.argmin(c5_values[samples])]

        # Crossover
        children = []
        for i in range(0, len(parent_indices), 2):
            p1 = parent_indices[i]
            p2 = parent_indices[i + 1] if i + 1 < len(parent_indices) else p1
            child = pop[p1].copy()
            mask = np.random.randint(2, size=n_points).astype(bool)
            child[mask] = pop[p2, mask]
            children.append(child)

        # Mutation
        for child in children:
            if np.random.rand() < MUTATION_PROB:
                # Swap a random 1 and 0 in the child
                indices_1 = np.where(child == 1)[0]
                indices_0 = np.where(child == 0)[0]
                if len(indices_1) > 0 and len(indices_0) > 0:
                    # Correct sum by swapping to maintain target_sum
                    while np.sum(child) != target_sum:
                        if np.sum(child) < target_sum:
                            idx0 = np.random.choice(indices_0)
                            idx1 = np.random.choice(indices_1)
                            child[idx0], child[idx1] = 1, 0
