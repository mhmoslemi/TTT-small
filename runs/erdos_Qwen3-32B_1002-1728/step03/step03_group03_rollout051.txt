    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h_flat = h_vec.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s * 0.95:
        np.random.seed(int(time.time() % 1000) ^ seed)
        pert = np.random.normal(0, 0.1, n_points)
        perturbed = best_h + pert
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * target_sum
        else:
            perturbed[:] = target_sum / n_points
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            candidate = result.x.reshape(n_points)
            candidate_sum = np.sum(candidate)
            if abs(candidate_sum - target_sum) < 1e-6 and np.all((candidate >= -1e-6) & (candidate <= 1.0 + 1e-6)):
                candidate_c5 = objective(candidate)
                if candidate_c5 < best_c5:
                    best_h = candidate
                    best_c5 = candidate_c5

    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    if result_final.success:
        final_candidate = result_final.x.reshape(n_points)
        final_sum = np.sum(final_candidate)
        if abs(final_sum - target_sum) < 1e-6 and np.all((final_candidate >= -1e-6) & (final_candidate <= 1.0 + 1e-6)):
            final_c5 = objective(final_candidate)
            if final_c5 < best_c5:
                best_h = final_candidate
                best_c5 = final_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
