    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - m})

        objective,
        x0=h_best,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    h_refined = result.x
    corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    c5 = np.max(corr) * dx

    h_final = h_refined if c5 < best_c5 else h_best
    return (h_final.tolist(), c5, n_points)
