    best_bin = population[best_idx]

    # Local optimization
    target_sum = n_points / 2
    bounds = [(0, 1)] * n_points
    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    result = minimize(
        fun=lambda x: np.max(np.correlate(x, 1 - x, 'full')) * dx,
        x0=best_bin,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100}
    )

    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, 'full')) * dx

    # Final checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all(best_h >= -1e-6) and np.all(best_h <= 1 + 1e-6), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
