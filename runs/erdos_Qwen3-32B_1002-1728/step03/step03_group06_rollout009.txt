import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 50
    mutation_rate = 0.2
    mutation_std = 0.05
    elite_ratio = 0.2

    def repair_individual(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h != 0:
            h_scaled = (h_clipped / sum_h) * target_sum
        else:
            h_scaled = target_sum / n_points * np.ones_like(h_clipped)
        h_repaired = np.clip(h_scaled, 0, 1)
        return h_repaired

    # Initialize population
    population = []
    population.append(h_init.copy())
    for _ in range(pop_size - 1):
        base = h_init.copy()
        perturbed = base + np.random.normal(0, 0.2, n_points)
        population.append(repair_individual(perturbed))
    population = np.array([repair_individual(p) for p in population])

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s * 0.95:
        fitness = []
        for indiv in population:
            corr = np.correlate(indiv, 1 - indiv, mode='full')
            current_c5 = np.max(corr) * dx
            fitness.append(current_c5)
        fitness = np.array(fitness)

        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Tournament selection
        parent_indices = []
        tournament_size = 3
        num_parents = pop_size // 2
        for _ in range(num_parents):
            contestants = np.random.choice(pop_size, tournament_size, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parent_indices.append(winner)
        parent_indices = np.array(parent_indices)

        # Generate offspring
        offspring = []
        for i in range(0, len(parent_indices), 2):
            if i + 1 >= len(parent_indices):
                break
            p1 = population[parent_indices[i]]
            p2 = population[parent_indices[i + 1]]
            alpha = np.random.uniform(0.4, 0.6, n_points)
            child1 = alpha * p1 + (1 - alpha) * p2
            child2 = alpha * p2 + (1 - alpha) * p1

            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, mutation_std, n_points)
                    child += noise
                offspring.append(repair_individual(child))

        # Elitism and population update
        elite_count = int(pop_size * elite_ratio)
        elite_indices = np.argsort(fitness)[:elite_count]
        elite_population = population[elite_indices]
        new_population = np.vstack([elite_population, offspring[:pop_size - elite_count]])

        # Fill remaining slots if needed
        remaining = pop_size - new_population.shape[0]
        if remaining > 0:
            filler = population[np.random.choice(pop_size, remaining, replace=False)]
            new_population = np.vstack([new_population, filler])
        population = new_population

    # Final repair to ensure constraints
    best_h = repair_individual(best_h)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
