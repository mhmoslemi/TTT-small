import numpy as np
from numpy.random import uniform, normal, choice

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)
    
    def initialize_population():
        pop = [h_init.copy()]
        for _ in range(49):
            h = uniform(0, 1, n_points)
            h = (h / h.sum()) * target_sum
            pop.append(h)
        return np.array(pop)

    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, fitness, tournament_size=5):
        indices = np.random.choice(len(pop), tournament_size, replace=False)
        winner_idx = indices[np.argmin(fitness[indices])]
        return pop[winner_idx].copy()

    def repair_solution(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale_factor = target_sum / (current_sum + 1e-9)
        return h * scale_factor

    def crossover(parent1, parent2):
        alpha = uniform(0.4, 0.6)
        child = alpha * parent1 + (1 - alpha) * parent2
        return repair_solution(child)

    def mutate(solution, mut_strength=0.05):
        mutation = normal(0, mut_strength, n_points)
        mutated = solution + mutation
        return repair_solution(mutated)

    population = initialize_population()
    best_h = h_init
    best_c5 = evaluate_fitness(best_h)
    patience_counter = 0

    while time.time() - start_time < budget_s:

        # Elitism: carry over top 20%
