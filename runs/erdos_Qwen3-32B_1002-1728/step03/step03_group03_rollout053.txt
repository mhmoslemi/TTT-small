def project_to_feasible(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-8:
        return h_clipped
    delta = target_sum - current_sum
    idx_not_at_bounds = np.where((h_clipped > 0.0) & (h_clipped < 1.0))[0]
    if len(idx_not_at_bounds) == 0:
        return h_clipped
    if delta > 0:
        max_increments = 1.0 - h_clipped[idx_not_at_bounds]
        total_increments = np.sum(max_increments)
        if total_increments <= 0:
            return h_clipped
        increment = delta * (max_increments / total_increments)
        h_clipped[idx_not_at_bounds] += increment
    else:
