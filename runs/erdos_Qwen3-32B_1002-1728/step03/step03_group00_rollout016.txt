import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        g = 1.0 - h_full
        n = len(h_full)
        corr_len = 2 * n - 1
        fg = np.fft.fft(h_full, corr_len)
        fg2 = np.fft.fft(g, corr_len)
        cross_corr = np.fft.ifft(fg * np.conj(fg2)).real
        return np.max(cross_corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Create symmetric initial guess
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    n_attempt = 0
    perturbation_std = 0.1

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        # Reduce perturbation as we approach time limit
        time_remaining = budget_s - (time.time() - start_time)
        t_ratio = time_remaining / budget_s
        current_std = perturbation_std * t_ratio

        # Create a new random state for this attempt
        rnd = np.random.RandomState(seed + n_attempt)
        perturbed = best_h_half + rnd.normal(0, current_std, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
