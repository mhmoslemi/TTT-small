    best_c5 = compute_c5_fast(create_full_h(best_h_half))

    np.random.seed(seed)
    delta_steps = np.linspace(-0.1, 0.1, num=5)

    while time.time() - start_time < budget_s - 30:
        for i in range(half_n):
            current_value = best_h_half[i]
            for delta in delta_steps:
                new_value = current_value + delta
                new_value = np.clip(new_value, 0.0, 1.0)
                new_h_half = best_h_half.copy()
                new_h_half[i] = new_value
                current_sum = np.sum(new_h_half)
                if current_sum < 1e-6:
                    continue
                scaling_factor = target_sum_half / current_sum
                new_h_half_scaled = new_h_half * scaling_factor
                new_h_half_scaled = np.clip(new_h_half_scaled, 0, 1)
                # Check sum again after clamping
                if np.sum(new_h_half_scaled) < 1e-6:
                    continue
                # Compute C5 for this candidate
                h_full_candidate = create_full_h(new_h_half_scaled)
                c5_candidate = compute_c5_fast(h_full_candidate)
                if c5_candidate < best_c5:
                    # Run optimize
                    result = minimize(
                        fun=compute_c5_fast,
                        x0=new_h_half_scaled,
                        method='SLSQP',
                        bounds=bounds_per_var,
                        constraints=cons,
                        options={'maxiter': 50, 'ftol': 1e-6, 'disp': False}
                    )
                    if result.success:
                        improved_h_half = result.x
                        improved_h_full = create_full_h(improved_h_half)
                        improved_c5 = compute_c5_fast(improved_h_full)
                        if improved_c5 < best_c5 and np.all(improved_h_half >= 0) and np.all(improved_h_half <= 1):
                            best_h_half = improved_h_half
                            best_c5 = improved_c5

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
