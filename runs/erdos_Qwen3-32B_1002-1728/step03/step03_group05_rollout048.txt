    return np.max(corr) * dx

    n_points = len(h_init)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    lambda_ = 20
    sigma = 0.2
    max_generations = 100
    mu = 5
    current_mean = h_init.copy()
    best_h = current_mean.copy()
    best_c5 = evaluate_c5(best_h, dx, n_points)
