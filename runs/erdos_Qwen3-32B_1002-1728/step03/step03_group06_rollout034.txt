import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial optimization with SLSQP
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    best_h = result.x.copy()
    best_c5 = objective(best_h)

    # Perturbation loop with adaptive strategy
    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed)
        perturb_factor = max(0.02, 0.05 / np.sqrt(1 + (time.time() - start_time) / 300))
        perturbed = best_h + rs.normal(0, perturb_factor, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = (perturbed / perturbed_sum) * target_sum
        
        # Alternating optimization methods for diversity
        if seed % 3 == 0:
            method = 'SLSQP'
            maxiter = 600
        elif seed % 3 == 1:
            method = 'Nelder-Mead'
            maxiter = 300
        else:
            method = 'L-BFGS-B'
            maxiter = 400
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method=method,
            bounds=bounds,
            constraints=cons if method in ['SLSQP'] else None,
            options={'maxiter': maxiter, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.copy()
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h[:] = h_candidate[:]
                    best_c5 = c5_candidate
        seed += 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
