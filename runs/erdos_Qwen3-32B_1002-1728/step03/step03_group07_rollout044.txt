import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_energy = compute_energy(h_init)
    best_h = h_init.copy()
    rs = np.random.RandomState(seed)

    num_initializations = 5
    max_iterations = 2000
    top_shift_count = 3
    alpha = 0.1
    alpha_decay = 0.95
    perturb_std = 0.1

    initializations = [h_init.copy()]
    for _ in range(num_initializations - 1):
        noise = rs.normal(scale=perturb_std, size=n_points)
        h_perturbed = np.clip(h_init + noise, 0, 1)
        h_perturbed *= target_sum / np.sum(h_perturbed)
        initializations.append(h_perturbed)

    for h0 in initializations:
        h = h0.copy()
        current_energy = compute_energy(h)
        for iter in range(max_iterations):
            if time.time() - start_time > budget_s * 0.99:
                break
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            shifts = np.arange(len(corr)) - (n_points - 1)
            valid_shifts = (shifts >= 0) & (shifts < n_points)
            valid_corr = corr[valid_shifts]
            valid_shifts_values = shifts[valid_shifts]

            if np.any(valid_corr):
                top_indices = np.argpartition(-valid_corr, top_shift_count)[:, :top_shift_count]
                k_values = valid_shifts_values[top_indices]
                if k_values.size == 0:
                    continue
                grad_avg = np.zeros(n_points)
                for k in k_values[:top_shift_count]:
                    grad = np.zeros(n_points)
                    for i in range(n_points):
                        j = (i + k) % n_points
                        grad[i] += (1 - h[j])
                        grad[j] -= h[i]
                    grad *= dx
                    grad_avg += grad / top_shift_count
                h_update = h - alpha * grad_avg
                h_update = np.clip(h_update, 0, 1)
                h_sum = np.sum(h_update)
                if h_sum > 1e-8:
                    h_update *= (target_sum / h_sum)
                else:
                    h_update[:] = target_sum / n_points
                energy_update = compute_energy(h_update)
                if energy_update < current_energy:
                    h[:] = h_update
                    current_energy = energy_update
                    if current_energy < best_energy:
                        best_energy = current_energy
                        best_h[:] = h.copy()
                        alpha = 0.1 * (alpha_decay ** (iter // 200))
                else:
                    alpha *= 0.9
        if current_energy < best_energy:
            best_energy = current_energy
            best_h[:] = h.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
