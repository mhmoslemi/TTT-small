            init_h = np.clip(init_h, 0, 1)

        # Define penalized objective
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            obj = max_corr * dx
            # Penalize deviation from target sum
            sum_h = np.sum(h)
            penalty = 1e6 * (sum_h - target_sum)**2
            return obj + penalty

        # Run optimization
        result = minimize(
            fun=objective,
            x0=init_h,
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-8}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_candidate = np.sum(h_candidate)
            # Final projection to constraint
            if not np.isclose(sum_candidate, target_sum, atol=1e-5):
                h_candidate = (h_candidate / sum_candidate) * target_sum
                h_candidate = np.clip(h_candidate, 0, 1)
            c5_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

        # Early stopping
        if time.time() - start_time >= budget_s * 0.99:
            break

    if best_h is None:
        best_h = h_init
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
