from scipy.optimize import minimize

    corr = np.correlate(h, 1 - h, mode='full')

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    population_size = 30
    w = 0.7
    c1 = 0.6
    c2 = 0.3
    max_v = 0.5
    max_iterations_pso = 100

    # Initialize particles and velocities
    particles = [np.clip(h_init.copy(), 0, 1)]
    velocities = [1e-4 * (np.random.rand(n_points) - 0.5)]
    
    # Normalize initial particle
    if np.sum(particles[0]) > 1e-9:
        particles[0] = particles[0] * target_sum / np.sum(particles[0])
    particles[0] = np.clip(particles[0], 0, 1)

    # Personal best data
    personal_best_positions = [particles[0].copy()]
    personal_best_values = [float('inf')]

    # Generate remaining particles
    rs = np.random.RandomState(seed)
    for _ in range(population_size - 1):
        h = rs.rand(n_points)
        if np.sum(h) > 1e-9:
            h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(1e-4 * (rs.rand(n_points) - 0.5))
        personal_best_positions.append(h.copy())
        # Compute initial C5 value
        corr = np.correlate(h, 1 - h, mode='full')
        pb_value = np.max(corr) * dx
        personal_best_values.append(pb_value)

    # Set global best
    global_idx = np.argmin(personal_best_values)
    global_best = personal_best_positions[global_idx].copy()
    global_best_value = personal_best_values[global_idx]

    # PSO iterations
    for iteration in range(max_iterations_pso):
        if time.time() - start_time > budget_s * 0.98:
            break
        for i in range(population_size):
            r1, r2 = rs.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best_positions[i] - particles[i]) + \
                           c2 * r2 * (global_best - particles[i])
            velocities[i] = np.clip(velocities[i], -max_v, max_v)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = np.sum(particles[i])
            if s > 1e-9:
                particles[i] = particles[i] * target_sum / s
            particles[i] = np.clip(particles[i], 0, 1)

            # Update personal best
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_value = np.max(corr) * dx
            if current_value < personal_best_values[i]:
                personal_best_positions[i] = h.copy()
                personal_best_values[i] = current_value

        # Update global best
        global_idx = np.argmin(personal_best_values)
        if personal_best_values[global_idx] < global_best_value:

    # Post-PSO refinement with SLSQP
    best_h = global_best.copy()
    best_c5 = global_best_value

    # Use top 5 for refinement

                best_c5 = candidate_c5

    return (best_h.tolist(), best_c5, n_points)
