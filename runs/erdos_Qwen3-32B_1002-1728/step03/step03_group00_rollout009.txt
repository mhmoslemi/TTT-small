        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    # Create symmetric initial guess
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_initial = h_full_sym[0:half_n]

    best_h_half = h_half_initial.copy()
    best_c5 = np.max(np.correlate(create_full_h(best_h_half), 1 - create_full_h(best_h_half), mode='full')) * dx

    bounds_half = [(0.0, 1.0) for _ in range(half_n)]

    def objective_function(x):
        x = np.array(x)
        sum_x = np.sum(x)
        if sum_x == 0 or np.any(x < -1e-8) or np.any(x > 1.0 + 1e-8):
            return np.inf

        h_full = create_full_h(x_scaled)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    remaining_time = budget_s - (time.time() - start_time) - 60  # Reserve 60s for post-processing
    if remaining_time > 0:
        # Configure DE parameters to fit time budget
        popsize = 15  # Adjust population size based on problem scale

        maxiter = max(1, min(maxiter, 500))  # Cap iterations

        result_de = differential_evolution(
            objective_function,
            bounds=bounds_half,
            x0=h_half_initial,

    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
