def objective_nelder(x, *args):
    n_points, dx, target_sum = args
    h = x.reshape(n_points)
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum < 1e-10:
        return float('inf')
    scale = target_sum / current_sum
    h_normalized = h_clipped * scale
    corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
    return np.max(corr) * dx

    h = h_vec.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full')
