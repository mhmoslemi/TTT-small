                hb_sum = np.sum(hb)
                if hb_sum > 1e-6: hb = hb * target_sum / hb_sum
                hb = np.clip(hb, 0, 1)
                corrb = np.correlate(hb, 1 - hb, 'full')
                c5b = np.max(corrb) * dx
                if c5b < best_c5: best_c5, best_h = c5b, hb

        pert = best_h + np.random.RandomState(seed).normal(0, .02, n_points)
        pert = np.clip(pert, 0, 1)
        pert_sum = np.sum(pert)
        if pert_sum > 1e-6: pert = pert * target_sum / pert_sum
        res_p = minimize(lambda x: obj_surf(x, alpha), pert,
                         method='L-BFGS-B', bounds=[(0,1)]*n_points,
                         options={'maxiter': 30})
        if res_p.success:
            hp = res_p.x
            hp_sum = np.sum(hp)
            if hp_sum > 1e-6: hp = hp * target_sum / hp_sum
            hp = np.clip(hp, 0, 1)
            corrp = np.correlate(hp, 1 - hp, 'full')
            c5p = np.max(corrp) * dx
            if c5p < best_c5: best_c5, best_h = c5p, hp

    if best_h is None:
        best_h, best_c5 = h_init, np.max(np.correlate(h_init, 1 - h_init, 'full')) * dx

    best_h_sum = np.sum(best_h)
    best_h = np.clip(best_h, 0, 1) * target_sum / best_h_sum
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), np.max(np.correlate(best_h, 1 - best_h, 'full')) * dx, n_points)
