import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_perturbations(base_h, num_perts, rng):
        perts = []
        for _ in range(num_perts):
            h_pert = base_h.copy()
            indices = rng.permutation(n_points)
            for i in range(0, len(indices), 2):
                if i + 1 >= len(indices):
                    break
                a, b = indices[i], indices[i + 1]
                max_move = min(1 - h_pert[a], h_pert[b])
                move = rng.uniform(-0.2, 0.2) * min(max_move, 0.1)
                h_pert[a] += move
                h_pert[b] -= move
                h_pert = np.clip(h_pert, 0, 1)
            perts.append(h_pert)
        return perts

    bounds = Bounds(0.0, 1.0)
    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    rs = np.random.RandomState(seed)

    best_energy = compute_energy(h_init)
    best_h = h_init.copy()

    max_perts = 25
    perturbed_starts = generate_perturbations(h_init, max_perts, rs)

    for idx, h_start in enumerate(perturbed_starts):
        if time.time() - start_time >= budget_s * 0.99:
            break

        result = minimize(
            compute_energy,
            h0=h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 50, 'ftol': 1e-6}
        )

        if result.success and compute_energy(result.x) < best_energy:
            best_h = result.x.copy()
            best_energy = compute_energy(best_h)

    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds"

    return (best_h.tolist(), best_energy, n_points)
