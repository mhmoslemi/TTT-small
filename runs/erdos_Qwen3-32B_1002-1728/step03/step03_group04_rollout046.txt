import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial optimization
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if result.success:
        best_h = result.x.reshape(n_points)
        best_c5 = objective(best_h)

    iteration = 0
    while time.time() - start_time < budget_s * 0.99:
        rs = np.random.RandomState(seed + int(time.time()) + iteration)
        elapsed = time.time() - start_time
        remaining_time_ratio = max(0.01, 1.0 - (elapsed / budget_s))
        current_sigma = 0.2 * remaining_time_ratio

        # Choose perturbation type
        choice = rs.rand()
        if choice < 1/3:
            # Continuous perturbation
            perturbed = best_h + rs.normal(0, current_sigma, n_points)
        elif choice < 2/3:
            # Binary sampling
            m = int(round(target_sum))
            indices = np.argpartition(best_h, -m)[-m:]
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, current_sigma, n_points)
        else:
            # Shift perturbation
            n_shift = rs.randint(1, n_points // 2)
            h_shifted = np.roll(best_h, n_shift)
            perturbed = h_shifted + rs.normal(0, current_sigma * 0.5, n_points)

        # Enforce bounds and adjust sum
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if abs(perturbed_sum) < 1e-6:
            perturbed = best_h.copy()
        else:
            perturbed = (perturbed / perturbed_sum) * target_sum

        # Local optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-6}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1.0)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate

        iteration += 1
        if iteration % 100 == 0:

    assert np.all((best_h >= -1e-6) and (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_init)

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    best_c5 = objective(best_h)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if result.success:
        best_h = result.x.reshape(n_points)
        best_c5 = objective(best_h)

    while time.time() - start_time < budget_s * 0.99:
        rs = np.random.RandomState(seed + int(time.time()))
