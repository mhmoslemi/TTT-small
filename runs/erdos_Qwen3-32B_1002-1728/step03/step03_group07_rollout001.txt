    # Attempt multiple initializations with perturbations
    for _ in range(5):
        h0 = h_init.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0 + noise, 0, 1)

        # Project to maintain sum constraint
        current_sum = h0.sum()
        if abs(current_sum) > 1e-8:
            h0 *= target_sum / current_sum
        h0 = np.clip(h0, 0, 1)

        def f_obj(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def f_grad(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            k_star = max_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                ipk = (i + k_star) % n_points
                imk = (i - k_star) % n_points
                grad[i] = (1 - h[ipk]) - h[imk]
            return (grad * dx).flatten()

        # Sum constraint
        def constraint_sum(h):
            return np.sum(h) - target_sum

        # Bounds and constraints

        constraints = [{'type': 'eq', 'fun': constraint_sum}]
