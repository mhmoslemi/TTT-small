        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            h_corrected = np.zeros_like(h_clipped)
            h_corrected[0] = target_sum
            return h_corrected
        else:
            return h_clipped * (target_sum / sum_h)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    for i in range(n_particles):
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + perturbation
        particles[i] = normalize_h(perturbed)
    velocities = np.random.rand(n_particles, n_points) * 0.05

    personal_best_positions = particles.copy()
    personal_best_values = np.array([compute_c5(p) for p in particles])
    best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[best_idx].copy()
    global_best_value = personal_best_values[best_idx]

    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.7:
            break
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (
                inertia * velocities[i]
                + cognitive * r1 * (personal_best_positions[i] - particles[i])
                + social * r2 * (global_best_position - particles[i])
            )
            particles[i] += velocities[i]
            particles[i] = normalize_h(particles[i])
            current_c5 = compute_c5(particles[i])
            if current_c5 < personal_best_values[i]:
                personal_best_values[i] = current_c5
                personal_best_positions[i] = particles[i].copy()
                if current_c5 < global_best_value:
                    global_best_value = current_c5
                    global_best_position = particles[i].copy()
        if time.time() - start_time > budget_s * 0.7:
            break

    best_h = global_best_position.copy()

    def objective(h_vec):
        h_flat = h_vec.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success:
        best_h = result.x.reshape(n_points)

    current_best_c5 = objective(best_h)

    while time.time() - start_time < budget_s * 0.95:
        np.random.seed(int(time.time() % 1000) ^ seed)
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * target_sum
        result_perturb = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result_perturb.success:
            h_candidate = result_perturb.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1 + 1e-6)):
                c5_candidate = objective(h_candidate)
                if c5_candidate < current_best_c5:
                    best_h = h_candidate
                    current_best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.95:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), current_best_c5, n_points)
