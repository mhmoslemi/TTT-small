def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    np.random.seed(seed)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    while time.time() - start_time < budget_s:
        # Compute current overlap integrals
        corr_full = np.correlate(h, 1 - h, mode='full')
        i_max = np.argmax(corr_full)
        k_phys_index = i_max - (n_points - 1)
        
        # Construct gradient vector for worst lag
        g = np.zeros(n_points)
        for i in range(n_points):
            shifted = i + k_phys_index
            if 0 <= shifted < n_points:
                g[i] = 1.0 - h[shifted]
            else:
                g[i] = 1.0
        
        # Formulate and solve LP
        delta = cp.Variable(n_points)
        objective = cp.Minimize(g @ delta)
        constraints = [cp.sum(delta) == 0]
        for i in range(n_points):
            constraints += [delta[i] >= -h[i]]
            constraints += [delta[i] <= 1 - h[i]]
        problem = cp.Problem(objective, constraints)
        
        try:

        except:
            continue
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h + delta.value
