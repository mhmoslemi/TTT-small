            grad[i] = (f_p - f_n) / (2 * delta)

        current_h = current_h - step_size * grad
        current_h = project(current_h, target_sum)

        c5 = evaluate_c5(current_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint failed"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
