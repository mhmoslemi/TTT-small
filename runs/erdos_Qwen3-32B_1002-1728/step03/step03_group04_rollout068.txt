import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum == 0:
            h_proj = np.zeros_like(h_clipped)
        else:
            h_proj = h_clipped * (target_sum / h_sum)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    # Initial optimization with Nelder-Mead
    result = minimize(
        fun=objective,
        x0=h_init,
        method='Nelder-Mead',
        bounds=bounds,
        options={'xatol': 1e-6, 'maxiter': 300, 'fatol': 1e-8}
    )

    best_h = result.x.copy()
    best_c5 = result.fun

    # Perturbation-based restarts
    restarts = 0
    while time.time() - start_time < budget_s * 0.9:

            sub_x0 = np.zeros(n_points)

        # Sub-optimization
        sub_result = minimize(
            fun=objective,
