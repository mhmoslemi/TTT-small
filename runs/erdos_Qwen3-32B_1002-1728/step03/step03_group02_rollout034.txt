        h_part = h[indices]

        overlap = cp.sum(h_part * (1 - shifted))
        constraints.append(overlap * dx <= t)

    problem = cp.Problem(cp.Minimize(t), constraints)
    problem.solve(solver=cp.GLPK_MI, verbose=False, maxiters=100000, time_limit=budget_s)

import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points

    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
