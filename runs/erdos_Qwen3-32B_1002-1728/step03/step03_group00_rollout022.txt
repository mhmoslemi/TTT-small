import numpy as np
import time
from numpy.random import RandomState

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_full) = n_points / 2 => sum(h_half) = n_points / 4

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def compute_C5(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_individual(individual):
        repaired = np.clip(individual, 0.0, 1.0)
        current_sum = repaired.sum()
        if current_sum != 0:
            repaired = (repaired / current_sum) * target_sum_half
        repaired = np.clip(repaired, 0.0, 1.0)
        return repaired

    population_size = 50
    population = []

    # Create symmetric initial h
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = h_full_sym.sum()
    if current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[:half_n]
    best_h_half = h_half.copy()
    best_c5 = compute_C5(best_h_half)
    population.append(h_half.copy())

    # Initialize population with perturbations
    rs = RandomState(seed)
    for _ in range(population_size - 1):
        perturbed = h_half + rs.normal(0, 0.1, size=half_n)
        perturbed = repair_individual(perturbed)
        population.append(perturbed)

    while time.time() - start_time < budget_s - 60:
        fitnesses = [compute_C5(ind) for ind in population]
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h_half = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
        sorted_indices = np.argsort(fitnesses)
        top_k = int(population_size * 0.2)
        selected_parent_indices = sorted_indices[:top_k]

        offspring = []
        while len(offspring) < population_size - top_k:
            # Tournament selection
            i1 = rs.choice(selected_parent_indices)
            i2 = rs.choice(selected_parent_indices)
            parent1 = population[i1]
            parent2 = population[i2]

            # Crossover
            if rs.rand() < 0.7:
                cut = rs.randint(1, half_n)
                child1 = np.concatenate([parent1[:cut], parent2[cut:]])
                child2 = np.concatenate([parent2[:cut], parent1[cut:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            # Mutation
            for child in [child1, child2]:
                if rs.rand() < 0.1:
                    mask = rs.rand(half_n) < 0.1
                    noise = rs.normal(0, 0.05, half_n)
                    child[mask] += noise[mask]
                child = repair_individual(child)
                if len(offspring) < population_size - top_k:
                    offspring.append(child)

        new_population = [population[i] for i in selected_parent_indices] + offspring
        if len(new_population) > population_size:
            new_population = new_population[:population_size]

    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
