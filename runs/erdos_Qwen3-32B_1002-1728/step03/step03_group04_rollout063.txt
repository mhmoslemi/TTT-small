def generate_perturbation(h, n_pairs=5, delta_range=0.2):
    h_perturbed = h.copy()
    n = len(h)
    for _ in range(n_pairs):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:

        max_delta_i = h_perturbed[i]

            h_perturbed[j] -= delta

            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1.0 + 1e-6)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h[:] = h_candidate[:]
                    best_c5 = c5_candidate
                    success_count = 0
                    current_delta *= 0.95
                else:
                    success_count += 1
        if success_count >= 15:
            current_delta = min(0.5, current_delta * 1.1)
            success_count = 0
        if time.time() - start_time > budget_s * 0.95:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
