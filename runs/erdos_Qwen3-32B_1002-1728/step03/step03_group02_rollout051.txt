import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points

    h = cp.Variable(n_points, boolean=True)

    constraints = [cp.sum(h) == target_sum]

        valid_x = []
        for x in range(n_points):

                valid_x.append(x)

        if len(valid_indices) > 0:
            terms = h[valid_indices] * (1 - h[valid_indices + k])
            constraints.append(t >= cp.sum(terms) * dx)

    problem = cp.Problem(cp.Minimize(t), constraints)
    time_left = max(budget_s - (time.time() - start_time), 0.1)
    result = problem.solve(solver=cp.GLPK_MI, verbose=False, time_limit=time_left)

    best_h = h.value
    best_c5 = t.value if problem.status in ["optimal", "optimal_inaccurate"] else np.inf

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert all(0 <= x <= 1 for x in best_h)
    return (best_h.tolist(), best_c5, n_points)
