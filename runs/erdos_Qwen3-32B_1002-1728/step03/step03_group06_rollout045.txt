import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair(h_vec, target_sum):
        h = np.copy(h_vec)
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        if abs(current_sum - target_sum) < 1e-8:
            return h
        delta = current_sum - target_sum
        if delta > 0:
            indices = np.argsort(h)
            for idx in indices:
                amount = min(h[idx], delta)
                h[idx] -= amount
                delta -= amount
                if delta <= 1e-8:
                    break
        else:
            delta = -delta
            indices = np.argsort(-h)
            for idx in indices:
                amount = min(1.0 - h[idx], delta)
                h[idx] += amount
                delta -= amount
                if delta <= 1e-8:
                    break
        h = np.clip(h, 0.0, 1.0)
        return h

    POPULATION = 50
    ELITE_RATE = 0.2
    MUTATION_RATE = 0.2
    CROSSOVER_RATE = 0.8
    SIGMA = 0.1

    np.random.seed(seed)
    population = [h_init.copy()]
    for _ in range(POPULATION - 1):
        h = h_init.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = repair(h, target_sum)
        population.append(h)
    population = np.array(population)

    def get_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    fitness = np.array([get_fitness(h) for h in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    while time.time() - start_time < budget_s:
        new_population = []
        elites = int(POPULATION * ELITE_RATE)
        elite_indices = np.argsort(fitness)[:elites]
        for i in elite_indices:
            new_population.append(population[i].copy())

        while len(new_population) < POPULATION:
            idx1 = np.random.randint(0, POPULATION)
            idx2 = np.random.randint(0, POPULATION)
            p1 = population[idx1].copy()
            p2 = population[idx2].copy()
            if np.random.rand() < CROSSOVER_RATE:
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy() if np.random.rand() < 0.5 else p2.copy()

            for i in range(n_points):
                if np.random.rand() < MUTATION_RATE:
                    child[i] += np.random.normal(0, SIGMA)
            child = repair(child, target_sum)
            new_population.append(child)

        new_population = np.array(new_population)
        new_fitness = np.array([get_fitness(h) for h in new_population])
        population = new_population
        fitness = new_fitness
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)].copy()
        if time.time() - start_time > budget_s * 0.99:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
