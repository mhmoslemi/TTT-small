        if abs(perturbed_sum) > 1e-6:
            perturbed = (perturbed / perturbed_sum) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
            initializations.append(perturbed)

    for init in initializations:
        if time.time() - start_time > budget_s * 0.95:
            break
        current_init = init.copy()
        result = minimize(
            fun=objective,
            x0=current_init,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 250, 'ftol': 1e-9}
        )
        if result.success:
            h_candidate = result.x
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= -1e-6) & (h_candidate <= 1.0 + 1e-6)):
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
