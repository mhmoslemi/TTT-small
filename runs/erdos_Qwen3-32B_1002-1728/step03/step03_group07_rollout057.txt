        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                grad[i] += 1.0 - x[i + k_star]
            if i - k_star >= 0:
                grad[i] -= x[i - k_star]
        grad *= dx
        return (float(np.max(corr) * dx), grad)

    best_h = h.copy()
    best_energy = compute_energy(best_h)
    seeds = [42, 123, 456, 789, 101]
    budget_per_run = budget_s / len(seeds)

    for seed_val in seeds:
        np.random.seed(seed_val)
        # Perturb initial h
        perturbation = 0.1 * np.random.uniform(-1, 1, size=n_points)
        h_pert = h + perturbation
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = h_pert.sum()
        h_pert = (h_pert * target_sum / current_sum).clip(0, 1)
        
        # Optimize
        bounds = scipy.optimize.Bounds(0, 1)
        result = scipy.optimize.minimize(
            objective,
            h_pert, 
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': 200}
        )
        optimized_h = np.clip(result.x.reshape(n_points), 0, 1)
        current_sum = optimized_h.sum()
        optimized_h = (optimized_h * target_sum / current_sum).clip(0, 1)
        
        energy = compute_energy(optimized_h)
        if energy < best_energy:
            best_h = optimized_h.copy()
            best_energy = energy

        if time.time() - start_time > budget_s:
            break

    # Final constraint check
    current_sum = best_h.sum()
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h * target_sum / current_sum).clip(0, 1)
    assert abs(best_h.sum() - target_sum) < 1e-6
    return (best_h.tolist(), best_energy, n_points)
