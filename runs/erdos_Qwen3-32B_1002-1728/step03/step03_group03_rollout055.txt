import time
import numpy as np
from scipy.optimize import minimize

def objective(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):

    max_iter = 500

    c1, c2 = 2.0, 2.0
    particles = []
    velocities = []

    # Initialize particles around initial_h
