def compute_overlaps(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    start = n - 1
    end = start + n
    return corr[start:end]

    overlaps = compute_overlaps(h)
    return np.max(overlaps) * dx
