import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k_star):
        grad = np.zeros_like(h)
        for j in range(n_points):
            shifted_minus = j - k_star
            term1 = 1.0
            if 0 <= shifted_minus < n_points:
                term1 = 1.0 - h[shifted_minus]
            shifted_plus = j + k_star
            term2 = 0.0
            if 0 <= shifted_plus < n_points:
                term2 = h[shifted_plus]
            grad[j] = term1 - term2
        return grad

    def project_to_constraints(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        delta = target_sum - sum_h
        h_proj = h_clipped + delta / n_points
        return np.clip(h_proj, 0, 1)

    np.random.seed(seed)
    start_time = time.time()
    best_energy = compute_energy(h_init)
    best_h = h_init.copy()

    # Generate multiple initializations
    n_initializations = 5
    for _ in range(n_initializations):
        # Perturb initialization
        noise = 0.1 * np.random.rand(n_points)
        init = h_init.copy() + noise
        init = np.clip(init, 0, 1)
        init_sum = np.sum(init)
        init *= target_sum / init_sum
        init = np.clip(init, 0, 1)
        current_h = init.copy()
        alpha = 0.1
        for iter in range(500):
            if time.time() - start_time > budget_s:
                break
            # Compute max correlation and corresponding k*
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            pos = np.argmax(corr)
            k_star = pos - (n_points - 1)
            # Compute gradient
            grad = compute_gradient(current_h, k_star)
            # Update step
            current_h -= alpha * grad
            current_h = project_to_constraints(current_h)
            # Update energy
            current_energy = compute_energy(current_h)
            # Periodic perturbation
            if iter % 200 == 0 and iter > 0:
                current_h += 0.02 * np.random.randn(n_points)
                current_h = project_to_constraints(current_h)
            # Manage learning rate
            if current_energy < compute_energy(best_h):
                alpha *= 1.02
            else:
                alpha *= 0.95
            # Update best solution
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = current_h.copy()
        if time.time() - start_time > budget_s:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
