import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    n_workers = 15
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.95:
            break
        rs = np.random.RandomState(seed + w)
        if w % 2 == 0:
            # Sinusoidal perturbation
            freq = rs.randint(1, 10)
            phase = rs.uniform(0, 2 * np.pi)
            sin_pert = 0.1 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
            perturbed = best_h.copy() + sin_pert
        else:
            # Random perturbation
            perturbed = best_h.copy() + rs.normal(0, 0.05, n_points)
        # Enforce bounds and constraint
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if abs(s) > 1e-9:
            perturbed = perturbed * (target_sum / s)
        # Optimize from this point
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final check for constraints
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    # Final sanity checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
