def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    h_clipped = np.clip(h_vec, 0.0, 1.0)
    current_sum = h_clipped.sum()
    if np.isclose(current_sum, 0):
        return h_clipped
