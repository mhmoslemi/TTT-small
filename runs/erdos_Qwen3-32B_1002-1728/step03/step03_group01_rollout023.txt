            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = slsqp_h.copy()

    # Final constraint validation
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6), "Constraint violation"
    assert np.all(0 <= best_h <= 1), "Out-of-bounds values"
    return (best_h.tolist(), best_c5, n_points)
