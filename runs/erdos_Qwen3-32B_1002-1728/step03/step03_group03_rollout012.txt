import time
import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if current_sum == 0:
        h[:] = target_sum / h.size
        return h
    scale_fac = target_sum / current_sum
    h *= scale_fac
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-8:
        idx = np.random.randint(h.size)
        adjustment = target_sum - current_sum
        h[idx] += adjustment
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    
    for i in range(1, population_size):
        while True:
            h_rand = np.random.rand(n_points)
            cs = np.sum(h_rand)
            if cs != 0:
                h_scaled = h_rand * (target_sum / cs)
                h_repaired = repair(h_scaled, target_sum).copy()
                population[i] = h_repaired
                break
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    
    while time.time() - start_time < budget_s:
        c5_values = [compute_c5(ind, n_points) for ind in population]
        idx_sorted = np.argsort(c5_values)
        population = [population[i] for i in idx_sorted]
        current_best = c5_values[idx_sorted[0]]
        if current_best < best_c5:
            best_h = population[0].copy()
            best_c5 = current_best
        
        n_elite = max(1, int(0.2 * population_size))
        elites = population[:n_elite]
        new_population = elites.copy()
        
        for _ in range(population_size - n_elite):
            p1_idx = np.random.choice(len(population), 3, replace=False)
            p2_idx = np.random.choice(len(population), 3, replace=False)
            p1 = population[np.argmin([c5_values[i] for i in p1_idx])]
            p2 = population[np.argmin([c5_values[i] for i in p2_idx])]
            
            crossover_point = np.random.randint(1, n_points)
