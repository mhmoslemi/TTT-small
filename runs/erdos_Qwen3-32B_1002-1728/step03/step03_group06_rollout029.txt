def normalize_h(h, target_sum):
    sum_h = np.sum(h)
    if sum_h == 0:
        return np.zeros_like(h)
    # scale to target_sum
    scaled = h * (target_sum / sum_h)
    clipped = np.clip(scaled, 0, 1)
    sum_clipped = np.sum(clipped)
    if sum_clipped == 0:
        return np.zeros_like(h)
    # renormalize
    normalized = clipped * (target_sum / sum_clipped)
    normalized = np.clip(normalized, 0, 1)
    return normalized
