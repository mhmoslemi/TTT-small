    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    rs = np.random.RandomState(seed)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    # Generate initial population
    population = []
    population.append(h_init.copy())
    for _ in range(50):
        perturbed = h_init + rs.normal(0, 0.03, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if not np.all((perturbed >= 0) & (perturbed <= 1)):
            perturbed = np.clip(perturbed, 0, 1)
        if not np.isclose(perturbed_sum, target_sum, rtol=1e-5):
            perturbed = (perturbed / np.sum(perturbed)) * target_sum
        population.append(perturbed)

    even = np.zeros(n_points)
    even[::2] = 1
    population.append(even)
    odd = np.zeros(n_points)
    odd[1::2] = 1
    population.append(odd)
    checkerboard = np.kron([1, 0], np.ones(n_points // 2))[:n_points]
    population.append(checkerboard)

    for _ in range(95):
        indices = rs.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:int(target_sum)]] = 1.0
        population.append(h)

    best_c5 = float('inf')
    best_h = h_init.copy()

    while time.time() - start_time < budget_s * 0.90:
        scores = []
        for h in population:
            h_sum = np.sum(h)
            if not np.isclose(h_sum, target_sum, rtol=1e-5):
                h = (h / h_sum) * target_sum
            corr = np.correlate(h, 1 - h, mode='full')
            scores.append(np.max(corr) * dx)
        sorted_indices = np.argsort(scores)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_scores = [scores[i] for i in sorted_indices]

        current_best_c5 = sorted_scores[0]
        current_best_h = sorted_pop[0].copy()
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h

        elites_count = max(1, int(0.1 * len(sorted_pop)))
        elites = sorted_pop[:elites_count]
        new_candidates = []

        for elite in elites:
            result = minimize(
                fun=lambda x: objective(x, dx),
                x0=elite,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                h_opt = result.x.reshape(n_points)
                if not np.isclose(np.sum(h_opt), target_sum, rtol=1e-5):
                    h_opt = (h_opt / np.sum(h_opt)) * target_sum
                new_candidates.append(h_opt)
                h_binary = binary_threshold(h_opt, target_sum)
                new_candidates.append(h_binary)
                h_perturbed = perturb_binary(h_binary, n_flip=2, rs=rs)
                new_candidates.append(h_perturbed)

        while len(new_candidates) < 10:
            indices = rs.permutation(n_points)
            h = np.zeros(n_points)
            h[indices[:int(target_sum)]] = 1.0
            new_candidates.append(h)

        combined = elites + new_candidates
        scores_combined = []
        for h in combined:
            h_sum = np.sum(h)
            if not np.isclose(h_sum, target_sum, rtol=1e-5):
                h = (h / h_sum) * target_sum
            corr = np.correlate(h, 1 - h, mode='full')
            scores_combined.append(np.max(corr) * dx)
        sorted_combined_indices = np.argsort(scores_combined)
        new_population = [combined[i] for i in sorted_combined_indices][:150]
        population = new_population

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
