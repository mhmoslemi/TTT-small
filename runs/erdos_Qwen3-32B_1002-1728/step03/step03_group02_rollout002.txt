import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    def objective_continuous(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Continuous optimization phase
    result_cont = minimize(
        fun=objective_continuous,
        x0=h_init,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    h_cont = result_cont.x

    # Binarize continuous solution
    sorted_indices = np.argsort(-h_cont)
    h_bin = np.zeros(n_points)
    h_bin[sorted_indices[:int(target_sum)]] = 1.0
    best_h = h_bin.copy()
    best_energy = objective_continuous(best_h)

    # Simulated Annealing phase
    np.random.seed(seed)
    T = 1.0
    alpha = 0.99
    T_min = 1e-3
    current_h = h_bin.copy()
    current_energy = best_energy

    while T > T_min and time.time() - start_time < budget_s * 0.95:
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i, j = np.random.choice(ones), np.random.choice(zeros)
        new_h = current_h.copy()
        new_h[i], new_h[j] = 0.0, 1.0
        new_energy = objective_continuous(new_h)
        delta_E = new_energy - current_energy
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h, current_energy = new_h, new_energy
            if current_energy < best_energy:
                best_h, best_energy = current_h.copy(), current_energy
        T *= alpha

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
