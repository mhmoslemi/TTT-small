            m = int(round(target_sum))
            indices = np.argpartition(best_h, -m)[-m:]
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, perturb_strength, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = (perturbed / perturbed_sum) * target_sum
        current_h = perturbed.copy()
        for _ in range(inner_opt_steps):
            corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
            kstar = np.argmax(corr)
            def objective_k(h_vec):
                h = h_vec.reshape(n_points)
                c = np.correlate(h, 1 - h, mode='full')
                return c[kstar] * dx
            result = minimize(
                fun=objective_k,
                x0=current_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            if result.success:
                new_h = result.x.reshape(n_points)
                if abs(np.sum(new_h) - target_sum) < 1e-6 and np.all(new_h <= 1 + 1e-6):
                    current_h = new_h
                else:
                    current_h = np.clip(current_h, 0, 1)
                    current_h *= target_sum / np.sum(current_h)
            else:
                break
        new_c5 = compute_c5(current_h)
        if new_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = new_c5
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
