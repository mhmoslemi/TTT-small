import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate all possible shifts
    shifts = list(range(-n_points + 1, n_points))
    
    # Define optimization variables
    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()

    # Constraints
    constraints = [cp.sum(h) == target_sum]

    # Add constraints for all valid shifts
    for k in shifts:
        start = max(0, -k)
        end = min(n_points - 1, n_points - 1 - k)
        if start <= end:
            indices = np.arange(start, end + 1)
            terms = h[indices] * (1 - h[indices + k])
            sum_terms = cp.sum(terms)
            constraints.append(sum_terms * dx <= t)

    # Objective: minimize t
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Solve the MILP problem with time budget
    remaining_time = max(1, budget_s - (time.time() - start_time))
    problem.solve(solver=cp.GLPK_MI, verbose=False, time_limit=remaining_time)

    # Ensure solution is valid

    return (best_h.tolist(), best_c5, n_points)
