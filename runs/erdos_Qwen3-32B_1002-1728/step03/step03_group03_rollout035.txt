        perturbed = h + np.random.normal(0, 0.1, n_points)
        particles.append(perturbed)
        velocities.append(np.random.normal(0, 0.01, n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    for i in range(n_particles):
        particles[i] = project_position(particles[i], target_sum)

    pbest_pos = particles.copy()
    pbest_obj = np.array([objective(p) for p in particles])
    gbest_idx = pbest_obj.argmin()
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_obj = pbest_obj[gbest_idx]

    for iter in range(max_iterations_pso):
        if time.time() - start_time > budget_s * 0.92:
            break
        w_current = 0.9 - (0.9 - 0.4) * iter / max_iterations_pso

        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            cognition = w_cognitive * r1 * (pbest_pos[i] - particles[i])
            social = w_social * r2 * (gbest_pos - particles[i])
            velocities[i] = w_current * velocities[i] + cognition + social
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            particles[i] += velocities[i]
            particles[i] = project_position(particles[i], target_sum)

            obj = objective(particles[i])
            if obj < pbest_obj[i]:
                pbest_pos[i] = particles[i].copy()
                pbest_obj[i] = obj

        current_best_idx = pbest_obj.argmin()
        if pbest_obj[current_best_idx] < gbest_obj:
            gbest_pos = pbest_pos[current_best_idx].copy()
            gbest_obj = pbest_obj[current_best_idx]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result_slsqp = minimize(
        fun=objective,
        x0=gbest_pos,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    best_h = result_slsqp.x
    best_c5 = objective(best_h)

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
