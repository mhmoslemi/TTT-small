import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = np.inf
    best_h = h.copy()

    def objective_with_penalty(h_flat, lambda_penalty):
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        max_corr = np.max(corr) * dx
        sum_h = np.sum(h_flat)
        sum_penalty = lambda_penalty * (sum_h - target_sum) ** 2
        return max_corr + sum_penalty

    # Initialize with penalty coefficient
    lambda_penalty = 100.0

    while time.time() - start_time < budget_s * 0.9:
        res = minimize(
            lambda x: objective_with_penalty(x, lambda_penalty),
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if res.success:
            h_candidate = res.x.copy()
            # Clip and project
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            sum_h = np.sum(h_candidate)
            if sum_h != 0:
                h_candidate = (h_candidate / sum_h) * target_sum
            # Re-check constraints
            sum_h = np.sum(h_candidate)
            c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if abs(sum_h - target_sum) < 1e-6:
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate.copy()
            # Perturb and increase penalty
            perturbed = h_candidate + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            sum_h = np.sum(perturbed)
            if sum_h != 0:
                perturbed = (perturbed / sum_h) * target_sum
            best_h = perturbed
            lambda_penalty = min(lambda_penalty * 1.2, 1e5)
        if time.time() - start_time > budget_s * 0.98:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
