        winner_idx = indices[np.argmin(fitnesses[indices])]
        return population[winner_idx].copy()

        if delta > 0:
            ones = np.where(child == 1)[0]
            flip = np.random.choice(ones, delta, replace=False)

            zeros = np.where(child == 0)[0]
