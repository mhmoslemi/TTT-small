import time
import numpy as np
from scipy.optimize import minimize

def adjust_to_sum(h, target_sum, epsilon=1e-8):
    h = h.copy()
    s = h.sum()
    if abs(s - target_sum) < epsilon:
        return h
    if s < target_sum:
        sorted_indices = np.argsort(h)
        h_available = 1.0 - h[sorted_indices]
        delta_needed = target_sum - s
        i = 0
        while delta_needed > 0 and i < len(h):
            add = min(delta_needed, h_available[i])
            h[sorted_indices[i]] += add
            delta_needed -= add
            i += 1
    else:
        sorted_indices = np.argsort(-h)
        h_available = h[sorted_indices]
        delta_needed = s - target_sum
        i = 0
        while delta_needed > 0 and i < len(h):
            remove = min(delta_needed, h_available[i])
            h[sorted_indices[i]] -= remove
            delta_needed -= remove
            i += 1
    return h

def compute_objective_single(h_vec, dx):
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def compute_objective(positions, dx):
    results = []
    for pos in positions:
        corr = np.correlate(pos, 1 - pos, mode='full')
        results.append(np.max(corr) * dx)
    return np.array(results)
