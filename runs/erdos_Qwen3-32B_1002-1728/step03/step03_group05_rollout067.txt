def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_vec):

        corr = np.correlate(h, 1 - h, mode='full')

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    n_workers = 15
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
