import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds_l, bounds_u = 0.0, 1.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize_h(h):
        h_clipped = np.clip(h, bounds_l, bounds_u)
        current_sum = h_clipped.sum()
        if abs(current_sum) < 1e-9:
            return np.zeros_like(h)
        return h_clipped * (target_sum / current_sum)

    # Initialize PSO population
    np.random.seed(seed)
    n_particles = 30
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []

    for _ in range(n_particles):
        if _ == 0:
            h = h_init.copy()
        else:
            # Random binary initialization
            idx = np.random.choice(n_points, int(target_sum), replace=False)
            h = np.zeros(n_points)
            h[idx] = 1.0
        h = normalize_h(h)
        particles.append(h.copy())
        velocities.append(0.01 * np.random.randn(n_points))
        personal_best.append(h.copy())
        personal_best_c5.append(compute_c5(h))

    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]

    # PSO parameters
    w = 0.7  # inertia
    c1, c2 = 0.6, 0.3  # cognitive/social
    max_v = 0.5

    iter_count = 0
    no_improvement = 0
    max_no_imp = 10

    while time.time() - start_time < budget_s * 0.8:
        iter_count += 1
        for i in range(n_particles):
            if time.time() - start_time > budget_s * 0.8:
                break
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_cognitive = c1 * r1 * (personal_best[i] - particles[i])
            vel_social = c2 * r2 * (global_best - particles[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            np.clip(velocities[i], -max_v, max_v, out=velocities[i])
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = normalize_h(particles[i])
            
            # Evaluate and update
            c5 = compute_c5(particles[i])
            if c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = c5
                if c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = c5
                    no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= max_no_imp:
                break
                
    # SLSQP refinement of top candidates
    top_candidates = np.argsort(personal_best_c5)[:5]
    final_best = global_best.copy()
    final_c5 = global_best_c5

    for idx in top_candidates:
        if time.time() - start_time > budget_s * 0.98:
            break
        h0 = personal_best[idx].copy()
        result = minimize(
            fun=compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = normalize_h(result.x)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < final_c5:
                final_best = h_candidate.copy()
                final_c5 = c5_candidate

    # Final validation
    assert abs(np.sum(final_best) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((final_best >= -1e-6) & (final_best <= 1.0 + 1e-6)), "Value bounds violated"
    return (final_best.tolist(), final_c5, n_points)
