import time
import numpy as np

def evaluate_objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair(h, target_sum):
    h = np.clip(h, 0, 1.0)
    h_sum = h.sum()
    if h_sum == 0:
        h = np.full(len(h), target_sum / len(h))
    else:
        h = h * (target_sum / h_sum)
    h = np.clip(h, 0, 1.0)
    return h

def mutate(h, target_sum, mutation_strength=0.1):
    if np.random.rand() < 0.2:
        mask = np.random.rand(len(h)) < mutation_strength
        h[mask] += np.random.normal(0, 0.1, size=len(h))[mask]
    return repair(h, target_sum)

def tournament_selection(population, fitness, tournament_size=5, num_parents=10):
    selected = []
    for _ in range(num_parents):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament = population[indices]
        tournament_fitness = fitness[indices]
        winner_idx = np.argmin(tournament_fitness)
        selected.append(tournament[winner_idx])
    return np.array(selected)

def generate_random_h(n_points, target_sum):

    h_sum = h.sum()
    if h_sum != 0:
        h = h * (target_sum / h_sum)
    h = np.clip(h, 0, 1.0)
    return h
