            perturb = np.random.normal(0, 0.1, n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            h_sum = h.sum()
            h = (h / h_sum) * target_sum
            no_improvement_counter = 0

    return (best_h.tolist(), best_c5, n_points)
