        h_clipped = h_clipped * (target_sum / total)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        k_vals = [(i - n_points + 1) * dx for i in range(len(corr))]
        weights = np.exp(-alpha * np.abs(k_vals))
        return np.sum(weights * corr * dx)

    def binary_threshold(h, n_samples):
        h_sorted = np.argsort(h)
        h_binary = np.zeros_like(h)
        h_binary[h_sorted[-n_samples:]] = 1.0
        return h_binary

    best_c5 = np.inf
    best_h = h_init.copy()
    bounds = [(0.0, 1.0)] * n_points

    # Solve each initialization
    for h in all_initials:
        h_flat = h.reshape(-1)
        result = minimize(
            fun=surrogate_obj,
            x0=h_flat,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            total = h_candidate.sum()
            h_candidate = h_candidate * (target_sum / total)
            c5_val = compute_c5(h_candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()

            # Refine with binary rounding
            h_binary = binary_threshold(h_candidate, n_points//2)
            result_binary = minimize(
                fun=compute_c5,
                x0=h_binary,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 80, 'ftol': 1e-8}
            )
            if result_binary.success:
                c5_binary = compute_c5(result_binary.x)
                if c5_binary < best_c5:
                    best_c5 = c5_binary
                    best_h = result_binary.x.copy()

    # Perturb and continue
    perturbation_seed = rs.permutation(n_points // 5)
    while time.time() - start_time < budget_s * 0.9 and perturbation_seed.size:
        i = perturbation_seed[0]
        perturbed = best_h + np.random.normal(0.0, 0.03, (n_points,))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if abs(perturbed_sum - target_sum) > 1e-6:
            perturbed = perturbed * (target_sum / perturbed_sum)
        result = minimize(
            fun=surrogate_obj,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            c5_val = compute_c5(h_candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()
        perturbation_seed = perturbation_seed[1:]

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint failed"
    assert np.all((best_h >= -1e-12) & (best_h <= 1 + 1e-12)), "Bound constraint failed"
    return (best_h.tolist(), best_c5, n_points)
