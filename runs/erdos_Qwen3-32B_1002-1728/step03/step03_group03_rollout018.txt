import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)
    n_particles = 50
    positions = []
    for _ in range(n_particles):
        pert = np.random.normal(0, 0.1, n_points)
        h = np.clip(h_initial + pert, 0, 1)
        h = (h / h.sum()) * target_sum if h.sum() != 0 else h_initial.copy()
        positions.append(h)
    positions = np.array(positions)
    velocities = np.zeros_like(positions)
    pbest_pos = positions.copy()
    pbest_obj = np.array([np.max(np.correlate(p, 1 - p, 'full')) * dx for p in positions])
    gbest_idx = np.argmin(pbest_obj)
    gbest_pos, gbest_obj = pbest_pos[gbest_idx].copy(), pbest_obj[gbest_idx]

    cognitive, social = 2.0, 2.0
    w_max, w_min = 0.9, 0.4
    max_iter = 500

    for iter in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
        w = w_max - (w_max - w_min) * iter / max_iter
        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)
        velocities = w * velocities + cognitive * r1 * (pbest_pos - positions) + social * r2 * (gbest_pos - positions)
        positions += velocities
        for i in range(n_particles):
            h = positions[i]
            h_clipped = np.clip(h, 0, 1)
            h_sum = h_clipped.sum()
            if h_sum != 0:
                h = (h_clipped / h_sum) * target_sum
            positions[i] = h

        current_objs = np.array([np.max(np.correlate(p, 1 - p, 'full')) * dx for p in positions])
        better = current_objs < pbest_obj
        pbest_pos[better] = positions[better].copy()
        pbest_obj[better] = current_objs[better]
        if np.min(current_objs) < gbest_obj:
            gbest_idx = np.argmin(current_objs)
            gbest_obj = pbest_obj[gbest_idx]
            gbest_pos = positions[gbest_idx].copy()

    best_candidates = list(positions[[np.argmin(pbest_obj), gbest_idx, *np.random.choice(n_particles, 2, replace=False)]])
    best_h = gbest_pos.copy()
    best_c5 = gbest_obj
    for candidate in best_candidates:
        if time.time() - start_time > budget_s * 0.98:
            break
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, 'full')) * dx,
            x0=candidate,
            bounds=[(0, 1)]*n_points,
            constraints={'type': 'eq', 'fun': lambda x: x.sum() - target_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            h = res.x.clip(0, 1)
            h = (h / h.sum()) * target_sum if h.sum() != 0 else h
            c5 = np.max(np.correlate(h, 1 - h, 'full')) * dx
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / best_h.sum()) * target_sum if best_h.sum() != 0 else best_h
    assert np.allclose(best_h.sum(), target_sum, atol=1e-6)
    return (best_h.tolist(), best_c5, n_points)
