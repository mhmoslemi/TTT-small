        # Penalty for constraint
        sum_h = np.sum(h)
        sum_penalty = 1e3 * (sum_h - target_sum)**2
        return max_c + sum_penalty

    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result_de = differential_evolution(
        func=objective,
        bounds=bounds,
        seed=seed,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7
    )

    # Extract best solution
    h_de = result_de.x.reshape(n_points)
    # Adjust sum to be exactly target_sum
    current_sum = np.sum(h_de)
    correction_factor = target_sum / current_sum
    h_corrected = h_de * correction_factor

    # Clip to [0,1] in case of overshoot
    h_corrected = np.clip(h_corrected, 0, 1)

    # Compute C5 for corrected solution
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    c5_corrected = compute_c5(h_corrected)

    # Now run Nelder-Mead from this corrected solution
    result_nelder = minimize(
        fun=compute_c5,
        x0=h_corrected,
        method='Nelder-Mead',
        bounds=bounds,
        options={'maxiter': 200, 'xatol': 1e-6}
    )

    best_h = result_nelder.x.reshape(n_points)
    best_c5 = compute_c5(best_h)

    # Perturb and try again
    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed)
        perturbed = best_h + rs.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum < target_sum:
            # Adjust by adding to samples
            perturbed = (perturbed / perturbed_sum) * target_sum
        else:
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            perturbed = (perturbed / perturbed_sum) * target_sum

        result = minimize(
            fun=compute_c5,
            x0=perturbed,
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 150, 'xatol': 1e-6}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        seed += 1

    # Final assertions
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)

This code combines differential evolution to find a good initial solution, followed by Nelder-Mead refinement. The differential evolution helps escape local optima, while the Nelder-Mead fine-tunes the result. 

The differential evolution parameters are chosen with some reasonable values: maxiter=100, popsize=15. These can be adjusted, but given the time limit, they might be sufficient.

Another improvement is to perform multiple restarts of differential evolution with different seeds in the available time, but the code as written uses one run of DE followed by Nelder-Mead and some perturbations.

This approach might yield a better C₅ bound than the previous one, as the global search could find better starting points.

However, differential evolution with a popsize of 15 and 100 iterations would require 15*100=1500 function evaluations, each of which involves the O(n²) operation. For n=70, this is about 1500 * 70² = 7.35e6 operations. Considering Python's speed, this might take time, but within 1100 seconds.

The perturbation loop adds more evaluations, but the code is designed to spend 95% of the time budget on this.

import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective_with_penalty(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_c = np.max(corr) * dx
        sum_h = np.sum(h)
        sum_penalty = 1e4 * (sum_h - target_sum)**2
        return max_c + sum_penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result_de = differential_evolution(
        func=objective_with_penalty,
        bounds=bounds,
        seed=seed,
        strategy='best1bin',
        maxiter=75,
        popsize=12,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8
    )

    h_de = result_de.x.reshape(n_points)
    current_sum = np.sum(h_de)
    scaling = target_sum / current_sum
    h_scaled = np.clip(h_de * scaling, 0, 1)

    result_nelder = minimize(
        fun=compute_c5,
        x0=h_scaled,
        method='Nelder-Mead',
        bounds=bounds,
        options={'maxiter': 200, 'xatol': 1e-6}
    )

    best_h = result_nelder.x.reshape(n_points)
    best_c5 = compute_c5(best_h)

    def adjust_solution(h_vec):
        h = h_vec.copy()
        h_sum = np.sum(h)
        diff = h_sum - target_sum
        if abs(diff) < 1e-6:
            return h
        indices = np.argsort(np.abs(h - 0.5))[::-1]
        for i in indices:
            delta = min(max(diff, -h[i]), 0)
            h[i] += delta
            diff += delta
            if abs(diff) < 1e-6:
                break
        return np.clip(h, 0, 1)

    def refine_and_check(h_vec):
        h = adjust_solution(h_vec)
        result = minimize(
            fun=compute_c5,
            x0=h,
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 150, 'xatol': 1e-6}
        )
        if result.success:
            c5_candidate = compute_c5(result.x)
            if c5_candidate < best_c5:
                return result.x, c5_candidate
        return None, None

    rs = np.random.RandomState(seed)
    perturb_magnitude = 0.3
    seed_seq = np.fromiter((rs.randint(0, 10000) for _ in range(30)), dtype=int)

    for i, s in enumerate(seed_seq):
        if time.time() - start_time > budget_s * 0.95:
            break
        noise = rs.normal(0, perturb_magnitude, n_points)
        candidate = best_h + noise
        candidate = np.clip(candidate, 0, 1)
        candidate[0] = min(max(candidate[0], 0.5 - perturb_magnitude), 0.5 + perturb_magnitude)
        candidate = adjust_solution(candidate)
        result = minimize(
            fun=compute_c5,
            x0=candidate,
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 120, 'xatol': 1e-6}
        )
        if result.success:
            c5_candidate = compute_c5(result.x)
            if c5_candidate < best_c5:
                best_h = result.x
                best_c5 = c5_candidate

    final_best = adjust_solution(best_h)
    final_c5 = compute_c5(final_best)

    assert abs(np.sum(final_best) - target_sum) < 1e-5
    assert np.all((final_best >= -1e-6) & (final_best <= 1 + 1e-6))
    return (final_best.tolist(), final_c5, n_points)
