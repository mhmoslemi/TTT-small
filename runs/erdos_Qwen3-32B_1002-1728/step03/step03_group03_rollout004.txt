import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_h = h.copy()
    current_c5 = best_c5

    while time.time() - start_time < budget_s:
        # Compute current worst overlap
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        k_samples = max_idx - (n_points - 1)

        # Compute gradient for LP
        g = np.ones(n_points)
        for i in range(n_points):
            shifted_i = i + k_samples
            if 0 <= shifted_i < n_points:
                g[i] = 1.0 - current_h[shifted_i]

        # Solve LP for perturbation
        delta = cp.Variable(n_points)
        constraints = [
            delta >= -current_h,
            delta <= 1 - current_h,
            cp.sum(delta) == 0
        ]
        objective = cp.Minimize(cp.sum(g * delta))
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.GLOP, verbose=False)
            if problem.status != 'optimal':
                continue

            h_candidate = current_h + delta.value
            h_candidate = np.clip(h_candidate, 0, 1)
            current_sum = np.sum(h_candidate)

            # Renormalize if needed
            if abs(current_sum - target_sum) > 1e-6:
                scale = target_sum / current_sum
                h_candidate = np.clip(h_candidate * scale, 0, 1)

            c5_candidate = compute_c5(h_candidate)

            if c5_candidate < current_c5:
                current_h = h_candidate.copy()
                current_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = current_h.copy()
        except:
