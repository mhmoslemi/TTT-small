import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce initial constraints
    h = h_initial.copy()
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        scale = target_sum / current_sum
        h = h * scale
    h = np.clip(h, 0, 1)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    np.random.seed(seed)
    attempts = 0

    while time.time() - start_time < budget_s - 30:
        attempts += 1
        # Generate perturbed vector
        rseed = int(1000 * np.random.rand()) if attempts > 1 else seed
        np.random.seed(rseed)
        perturbation = np.random.randn(n_points) * 0.1
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.05)  # Allow small overshoot
        # Normalize sum
        s = np.sum(perturbed)
        if s != 0:
            perturbed = (perturbed / s) * target_sum
        # Clip again
        perturbed = np.clip(perturbed, 0, 1)
        # Post pseudonormalization to handle clipping
        s2 = np.sum(perturbed)
        if abs(s2 - target_sum) > 1e-3:
            perturbed = perturbed * (target_sum / s2)
            perturbed = np.clip(perturbed, 0, 1)
        # Optimize
        res = minimize(
            fun=lambda x: compute_c5(x),
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if not res.success:
            continue
        h_new = res.x
        c5_new = compute_c5(h_new)
        # Check constraints
        s_new = np.sum(h_new)
        if abs(s_new - target_sum) < 1e-5 and np.all((h_new >= -1e-8) & (h_new <= 1+1e-8)):
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
