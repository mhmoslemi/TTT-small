import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    # Differential Evolution with feasibility projection
    def de_objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    try:
        result_de = differential_evolution(
            de_objective,
            bounds,
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            tol=1e-4,
            workers=1,
            seed=seed
        )
        h_de = result_de.x.reshape(n_points)
    except:
        h_de = h_init.copy()

    # Feasibility projection
    h_proj = np.clip(h_de, 0, 1)
    s = np.sum(h_proj)
    if abs(s) > 1e-9:
        h_proj *= (target_sum / s)
    else:
        h_proj[:] = 0.5

    # Refine with SLSQP
    result_refine = minimize(
        objective,
        h_proj,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    if result_refine.success:
        h_candidate = result_refine.x
        if abs(np.sum(h_candidate) - target_sum) < 1e-6:
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    # Perturbed local searches
    n_workers = 5
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
        rs = np.random.RandomState(seed + w)
        perturbed = h_de + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if abs(s) > 1e-9:
            perturbed *= (target_sum / s)
        else:
            perturbed[:] = 0.5
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    # Post-check constraints
    if abs(np.sum(best_h) - target_sum) > 1e-6:
        best_h = h_init.copy()
        best_c5 = objective(best_h)

    return (best_h.tolist(), best_c5, n_points)
