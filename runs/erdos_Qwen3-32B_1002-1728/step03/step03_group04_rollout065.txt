import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    h_init = np.array(initial_h_values)
    n_points = len(h_init)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    perturbation_std_initial = 0.2
    perturbation_decay = 0.99
