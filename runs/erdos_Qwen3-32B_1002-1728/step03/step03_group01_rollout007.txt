import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # PSO setup
    np.random.seed(seed)
    n_particles = 30
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    personal_best = np.zeros((n_particles, n_points))
    personal_best_c5 = np.full(n_particles, np.inf)
    
    # Initialize particles
    particles[0] = h_init.copy()
    particles[0] = np.clip(particles[0], 0, 1) * (target_sum / np.sum(particles[0]))
    rs = np.random.RandomState(seed)
    for i in range(1, n_particles):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        h += (rs.rand(n_points) - 0.5) * 0.2
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)
        particles[i] = h

    # Initialize personal best
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        h = particles[i]
        h_sum = np.sum(h)
        if h_sum > 1e-9:
            h *= target_sum / h_sum
        particles[i] = h
        c5 = objective(h)
        personal_best[i] = h.copy()
        personal_best_c5[i] = c5

    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    best_c5 = personal_best_c5[global_best_idx]

    # PSO parameters
    w = 0.7
    c1 = 0.6
    c2 = 0.3
    max_vel = 0.5
    pso_iter = 0
    max_pso_iter = 100

    # PSO main loop
    while time.time() - start_time < budget_s * 0.8 and pso_iter < max_pso_iter:
        for i in range(n_particles):
            if time.time() - start_time > budget_s * 0.8:
                break
            r1, r2 = rs.rand(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)

        # Normalize to maintain constraint
        for i in range(n_particles):
            h = particles[i]
            h_sum = np.sum(h)
            if h_sum > 1e-9:
                h *= target_sum / h_sum
            particles[i] = h

        # Update personal and global best
        for i in range(n_particles):
            h = particles[i]
            c5_candidate = objective(h)
            if c5_candidate < personal_best_c5[i]:
                personal_best[i] = h.copy()
                personal_best_c5[i] = c5_candidate
            if c5_candidate < best_c5:
                global_best[:] = h.copy()
                best_c5 = c5_candidate

        pso_iter += 1

    # SLSQP refinement on top candidates
    top_indices = np.argsort(personal_best_c5)[:5]
    for i in top_indices:
        if time.time() - start_time > budget_s * 0.95:
            break
        h0 = personal_best[i].copy()
        result = minimize(
            fun=objective,
            x0=h0,
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            method='SLSQP',
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                global_best[:] = h_candidate
                best_c5 = c5_candidate

    # Final validation
    assert abs(np.sum(global_best) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((global_best >= -1e-6) & (global_best <= 1.0 + 1e-6)), "Value bounds violated"
    return (global_best.tolist(), best_c5, n_points)
