    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # Optimization parameters
    epsilon = 0.01  # Smoothing parameter
    delta = 1e-5    # Perturbation step
    step_size = 0.1
    current_h = h.copy()

    def compute_corr(h_vec):
        return np.correlate(h_vec, 1 - h_vec, mode='full') * dx

    def smooth_obj(h_vec):
        corr = compute_corr(h_vec)
        max_corr = np.max(corr)
        centered = corr - max_corr
        exp_vals = np.exp(centered / epsilon)
        sum_vals = exp_vals.sum()
        return epsilon * np.log(sum_vals) + max_corr

    while time.time() - start_time < budget_s * 0.98:
        # Compute gradient via forward differences
        grad = np.zeros(n_points)
        L0 = smooth_obj(current_h)
        for i in range(n_points):
            h_pert = current_h.copy()
            h_pert[i] += delta
            L_pert = smooth_obj(h_pert)
            grad[i] = (L_pert - L0) / delta

        # Gradient descent with projection
        new_h = current_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_sum = new_h.sum()

        if abs(new_sum) > 1e-10:
            new_h *= (target_sum / new_sum)
        else:
            new_h[:] = 0.5 * target_sum / n_points

        current_h = np.clip(new_h, 0, 1)

        # Update best solution
        current_corr = compute_corr(current_h)
        current_c5 = np.max(current_corr)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    # Final constraint enforcement
    best_h = np.clip(best_h, 0, 1)
    best_h *= target_sum / best_h.sum()
    return (best_h.tolist(), best_c5, n_points)
