                    child = (child / total) * target_sum_half
                child = np.clip(child, 0, 1)
            offspring.append(child)

        population = parents + offspring

    h_full = create_full_h(best_h_vars)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Bounds violated"
    return (h_full.tolist(), best_c5, n_points)
