import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize_perturbation(h_perturbed, target_sum):
        h_clipped = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h_perturbed)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    best_h = np.array(h_init)
    best_c5 = objective(best_h)
    max_iter = 200  # SLSQP iterations for main run
    local_iter = 100  # SLSQP iterations for perturbations
    temps = np.linspace(0.1, 0.01, 300)  # Cooling schedule

    for t_idx in range(len(temps)):
        if time.time() - start_time > budget_s:
            break
        temp = temps[t_idx]
        rs = np.random.RandomState(seed + t_idx)
        if rs.rand() < 0.5:
            # Continuous perturbation with zero-mean noise
            noise = rs.normal(0, temp, n_points)
            noise -= np.mean(noise)
            perturbed = best_h + noise
        else:
            # Binarize with noise and permute
            m = int(round(target_sum))
            noisy_vals = best_h + rs.normal(0, 0.02, n_points)
            indices = np.argpartition(noisy_vals, -m)[-m:]
            bin_perturbed = np.zeros(n_points)
            bin_perturbed[indices] = 1.0
            perturbed = bin_perturbed + rs.normal(0, 0.03, n_points)
        # Normalize while respecting bounds
        perturbed = normalize_perturbation(perturbed, target_sum)
        # Local refine
        result = minimize(
            fun=objective,
            x0=perturbed,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': local_iter if t_idx < 150 else 50}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final safety normalization
            h_candidate = normalize_perturbation(h_candidate, target_sum)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:

    # Final check
    best_h = normalize_perturbation(best_h, target_sum)
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
