    while time.time() - start_time < budget_s * 0.95:
        gen_c5 = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            gen_c5.append(np.max(corr) * dx)
        idx = np.argmin(gen_c5)
        candidate_c5 = np.min(gen_c5)
        if candidate_c5 < best_c5:
            best_h = population[idx].copy()
            best_c5 = candidate_c5

        new_pop = []
        for _ in range(len(population)):
            # Tournament select 2 parents
            parents = [population[np.random.choice(len(population))] for _ in range(5)]
            parent1 = min(parents, key=lambda x: np.max(np.correlate(x, 1 - x, 'full')) * dx)
            parents2 = [population[np.random.choice(len(population))] for _ in range(5)]
            parent2 = min(parents2, key=lambda x: np.max(np.correlate(x, 1 - x, 'full')) * dx)

            # Crossover
            child = parent1.copy()
            diff = np.where(parent1 != parent2)[0]
            if len(diff) > 0:
                n_swap = max(1, int(0.1 * len(diff)))
                indices = np.random.choice(diff, n_swap, replace=False)
                child[indices] = 1 - child[indices]

            # Mutation
            if np.random.rand() < 0.1:
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                i1, i2 = np.random.choice(ones), np.random.choice(zeros)
                child[[i1, i2]] = child[[i2, i1]]
            new_pop.append(child)
        population = new_pop
        gen += 1

        # Local optimization every 10 generations
        if gen % 10 == 0 and time.time() - start_time < budget_s * 0.85:
            def obj_func(h_vec):
                h = h_vec.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx

            res = minimize(
                obj_func,
                x0=best_h.copy(),
                bounds=[(0, 1)] * n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - m},
                method='SLSQP',
                options={'maxiter': 100}
            )
            if res.success:
                h_opt = res.x.reshape(n_points)
                sum_opt = np.sum(h_opt)
                if abs(sum_opt - m) < 1e-6:
                    corr = np.correlate(h_opt, 1 - h_opt, 'full')
                    c5 = np.max(corr) * dx
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
                        population.append(binarize(best_h))

        if time.time() - start_time > budget_s * 0.98:
            break

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
