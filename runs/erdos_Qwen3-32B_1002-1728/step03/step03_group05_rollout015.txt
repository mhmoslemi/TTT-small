import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Generate diverse starting points with structured and random perturbations
    n_workers = 12
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
        rs = np.random.RandomState(seed + w)
        # Generate perturbed initial guess with structured or random pattern
        if w % 2 == 0:
            # Structured sine wave perturbation
            freq = (w // 2) + 1  # Frequency variation
            phase = rs.rand() * 2 * np.pi  # Random phase
            perturbation = 0.05 * np.sin(
                2 * np.pi * freq * np.arange(n_points) / n_points + phase
            )
        else:
            # Random noise perturbation
            perturbation = rs.normal(0, 0.05, n_points)
        
        # Apply perturbation and enforce bounds
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        # Normalize to preserve integral constraint
        s = np.sum(perturbed)
        if abs(s) > 1e-9:
            perturbed = perturbed * (target_sum / s)
        
        # Optimize with SLSQP using reduced maxiter for faster iteration
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final constraint check
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
