        scale_factor = target_sum / sum_de
        scaled_de = h_de_clip * scale_factor
        h_feas_de = np.clip(scaled_de, 0.0, 1.0)
    else:
        h_feas_de = h_init.copy()

    # Refine with SLSQP
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result_feas = minimize(fun=objective, x0=h_feas_de, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 300, 'ftol': 1e-8})
    best_h = result_feas.x
    best_c5 = result_feas.fun if result_feas.success else 0.5

    # Perturbation phase
    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed + int(time.time()))
        if rs.rand() < 0.5:
            perturbed = best_h + rs.normal(0, 0.03, n_points)
        else:
            m = int(round(target_sum))
            indices = np.argpartition(best_h, -m)[-m:]
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, 0.03, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-6:
            perturbed = (perturbed / perturbed_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        result = minimize(fun=objective, x0=perturbed, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            h_candidate = result.x
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= -1e-6) & (h_candidate <= 1.0 + 1e-6)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.95:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
