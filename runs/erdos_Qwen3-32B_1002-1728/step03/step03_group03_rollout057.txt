import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        max_index = np.argmax(overlaps)
        k_steps = max_index - (n_points - 1)

        # Compute gradient g
        g = np.zeros(n_points)
        if k_steps >= 0 and k_steps < n_points:
            g[:n_points - k_steps] = 1 - h[k_steps:]
        elif k_steps < 0:
            k_steps_abs = -k_steps
            if k_steps_abs < n_points:
                g[k_steps_abs:] = 1 - h[:-k_steps_abs]

        # Set up and solve LP
        delta = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(g * delta))
        constraints = [
            cp.sum(delta) == 0,
            delta <= 0.05,
            delta >= -0.05
        ]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.GLPK, max_iter=100, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                delta_h = delta.value
                new_h = h + delta_h
                new_h_clipped = np.clip(new_h, 0, 1)
                s = np.sum(new_h_clipped)
                if s > 1e-8:
                    scaling = target_sum / s
                    new_h_scaled = new_h_clipped * scaling
                    new_h_scaled = np.clip(new_h_scaled, 0, 1)
                    candidate_c5 = np.max(np.correlate(new_h_scaled, 1 - new_h_scaled, mode='full') * dx)
                    if candidate_c5 < best_c5:
                        h = new_h_scaled
                        best_c5 = candidate_c5
                        best_h = h.copy()
        except Exception:
            pass

        if time.time() - start_time >= budget_s:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
