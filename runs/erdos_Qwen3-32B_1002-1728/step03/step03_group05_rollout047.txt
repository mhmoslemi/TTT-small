import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points

    de_result = differential_evolution(
        func=wrapped_objective,
        bounds=bounds,
        maxiter=15,
        popsize=10,
