    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Binarize and optimize
    h_binary = binary_h(best_h, target_sum)
    result_bin = minimize(
        fun=objective,
        x0=h_binary,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result_bin.success:
        h_candidate = result_bin.x.reshape(n_points)
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    all_initials = additional_initials.copy()
    while time.time() - start_time < budget_s * 0.90:
        if all_initials:
            next_initial = all_initials.pop(0)
        else:
            # Perturb the best_h and continue
            perturbed = best_h + rs.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if abs(perturbed_sum - target_sum) > 1e-6:
                perturbed = (perturbed / perturbed_sum) * target_sum
            next_initial = perturbed

        result = minimize(
            fun=objective,
            x0=next_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

                # Try binary version
                h_bin = binary_h(h_candidate, target_sum)
                result_binary = minimize(
                    fun=objective,
                    x0=h_bin,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 80, 'ftol': 1e-8}
                )
                if result_binary.success:
                    corr_binary = np.correlate(result_binary.x, 1 - result_binary.x, mode='full')
                    c5_binary = np.max(corr_binary) * dx
                    if c5_binary < best_c5:
                        best_h = result_binary.x
                        best_c5 = c5_binary

    # Final checks and return
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
