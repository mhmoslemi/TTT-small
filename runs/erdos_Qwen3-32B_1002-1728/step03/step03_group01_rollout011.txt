import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Generate diverse initial solutions
    initializations = [h_init.copy()]
    rs = np.random.RandomState(seed)
    for _ in range(14):  # Total 15 initial solutions
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)

    best_c5 = float('inf')
    best_h = h_init.copy()
    valid_candidates = []

    # SA parameters
    T_initial = 100.0
    T_min = 1e-5
    alpha_cool = 0.999
    max_temp_iterations = 60  # Conservative upper bound per temperature step

    for h_sa in initializations:
        if time.time() - start_time > budget_s * 0.9:
            break
        current_h = h_sa.copy()
        current_energy = objective(current_h)
        T = T_initial

        while T > T_min and time.time() - start_time < budget_s * 0.9:
            for _ in range(max_temp_iterations):
                if time.time() - start_time > budget_s * 0.9:
                    break
                # Perturb and normalize
                perturbation = rs.normal(0, T / 10.0, n_points)
                new_h = np.clip(current_h + perturbation, 0, 1)
                h_sum = np.sum(new_h)
                if h_sum < 1e-9:
                    new_h = np.ones_like(new_h) * (target_sum / n_points)
                else:
                    new_h *= target_sum / h_sum
                new_energy = objective(new_h)

                # Metropolis acceptance
                delta = new_energy - current_energy
                if delta < 0 or rs.rand() < np.exp(-delta / T):
                    current_h, current_energy = new_h, new_energy
                    if current_energy < best_c5:
                        best_c5 = current_energy
                        best_h = current_h.copy()
                        valid_candidates.append((best_h.copy(), best_c5))
            T *= alpha_cool

    # SLSQP refinement on top candidates
    if valid_candidates and time.time() - start_time < budget_s * 0.97:
        if len(valid_candidates) > 5:
            # Select top 5 candidates based on C5
            valid_candidates.sort(key=lambda x: x[1])
            top_candidates = [c[0] for c in valid_candidates[:5]]
        else:
            top_candidates = [best_h]
        for candidate in top_candidates:
            if time.time() - start_time > budget_s * 0.98:
                break
            result = minimize(
                fun=objective,
                x0=candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            if result.success:
                h_candidate = result.x.reshape(n_points)
                c5 = objective(h_candidate)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate.copy()

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Constraint violation"
    return (best_h.tolist(), best_c5, n_points)
