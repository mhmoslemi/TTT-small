import time
import numpy as np

def compute_c5(h_vec, dx):
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    num_particles = 100
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    particles[0] = h_init.copy()
    velocities[0] = np.random.normal(0, 0.01, n_points)

    for i in range(1, num_particles):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        particles[i] = h_rand
        velocities[i] = np.random.normal(0, 0.01, n_points)

    pbest_positions = particles.copy()
    pbest_scores = np.array([compute_c5(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    inertia_initial, inertia_final = 0.7, 0.4
    cognitive, social = 1.2, 1.2
    velocity_clamp = 0.1
    velocity_decay = 0.99

    iter_count = 0
    no_improve = 0

    while time.time() - start_time < budget_s * 0.99:
        iter_count += 1
        elapsed = time.time() - start_time
        time_fraction = min(elapsed / (budget_s * 0.99), 1.0)
        inertia = inertia_initial + (inertia_final - inertia_initial) * time_fraction

        for i in range(num_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest_positions[i] - particles[i]) +
                social * r2 * (gbest_position - particles[i]) 
            ) * velocity_decay
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h == 0:
                particles[i].fill(0.5)
                sum_h = np.sum(particles[i])
            particles[i] = particles[i] * (target_sum / sum_h)

        new_scores = np.array([compute_c5(p, dx) for p in particles])
        updated = np.where(new_scores < pbest_scores)[0]
        pbest_scores[updated] = new_scores[updated]
        pbest_positions[updated] = particles[updated]

        current_best_idx = np.argmin(pbest_scores)
        if pbest_scores[current_best_idx] < gbest_score:
            gbest_score = pbest_scores[current_best_idx]
            gbest_position = pbest_positions[current_best_idx].copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            cognitive = 1.2
            social = 1.2
            for i in range(num_particles):
                if np.random.rand() < 0.2:
                    h_rand = np.random.rand(n_points)
                    h_rand = np.clip(h_rand, 0, 1)
                    h_rand = h_rand * (target_sum / np.sum(h_rand))
                    particles[i] = h_rand
                    velocities[i] = np.random.normal(0, 0.01, n_points)
            no_improve = 0

    return (best_h.tolist(), best_c5, n_points)
