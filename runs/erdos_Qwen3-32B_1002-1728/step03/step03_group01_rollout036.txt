        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-9:
            perturbed *= target_sum / perturbed_sum
        else:
            idx = rs.choice(n_points, int(target_sum), replace=False)
            perturbed = np.zeros(n_points)
            perturbed[idx] = 1.0
        res = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            h_candidate = res.x.reshape(n_points)
            score = evaluate(h_candidate)
            if score < best_score:
                best_score = score
                best_h = h_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Values out of bounds"
    return (best_h.tolist(), best_score, n_points)
