import time
import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize PSO parameters
    n_particles = 100
    max_iter = 500
    inertia = 0.7
    cognitive = 1.2
    social = 1.2
    vel_decay = 0.99
    vel_clamp = 0.1

    # Initialize particles
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    particles[0] = h_init.copy()
    particles[1:] = 0.5 + np.random.uniform(-0.1, 0.1, (n_particles - 1, n_points))
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        s = particles[i].sum()
        particles[i] = (particles[i] / s) * target_sum

    pbest_pos = particles.copy()
    pbest_cost = np.array([objective(p) for p in particles])
    gbest_idx = np.argmin(pbest_cost)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_cost = pbest_cost[gbest_idx]

    start_time = time.time()

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
        # Update parameters
        inertia *= 0.995
        cognitive *= 0.999
        social *= 0.999

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest_pos[i] - particles[i]) +
                             social * r2 * (gbest_pos - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_clamp, vel_clamp)
            velocities[i] *= vel_decay

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = particles[i].sum()
            particles[i] = (particles[i] / s) * target_sum

        # Evaluate and update personal(best) and global(best)
        current_costs = np.array([objective(p) for p in particles])
        improved = current_costs < pbest_cost
        pbest_cost[improved] = current_costs[improved]
        pbest_pos[improved] = particles[improved].copy()
        current_gbest_idx = np.argmin(pbest_cost)
        if pbest_cost[current_gbest_idx] < gbest_cost:
            gbest_cost = pbest_cost[current_gbest_idx]
            gbest_pos = pbest_pos[current_gbest_idx].copy()

        if iteration % 10 == 0:
            print(f"Iter {iteration}: C5={gbest_cost:.8f}")

    best_h = gbest_pos
    best_c5 = gbest_cost
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
