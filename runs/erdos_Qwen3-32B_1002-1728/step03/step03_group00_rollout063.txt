import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = total/2 * 2

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Create symmetric initial guess
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        # Alternate perturbation scales
        if n_attempt % 3 == 0:
            scale = 0.15
        elif n_attempt % 2 == 0:
            scale = 0.1
        else:
            scale = 0.2
        rnd = np.random.RandomState(seed + n_attempt)
        perturbed = best_h_half + rnd.normal(0, scale * np.ones(half_n), half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        # Relaxed method and options
        result_slsqp = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        result_cobyla = minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds_per_var,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
            options={'maxiter': 200, 'tol': 1e-8, 'disp': False}
        )

        results = [result_slsqp, result_cobyla]
        for result in results:
            if result.success:
                new_h_half = result.x.reshape(half_n)
                new_corr = np.correlate(create_full_h(new_h_half), 1 - create_full_h(new_h_half), mode='full')
                new_c5 = np.max(new_corr) * dx
                if (abs(np.sum(new_h_half) - target_sum_half) < 1e-6 and
                    np.all((new_h_half >= -1e-6) & (new_h_half <= 1.0 + 1e-6))):
                    if new_c5 < best_c5:
                        best_h_half = new_h_half
                        best_c5 = new_c5

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
