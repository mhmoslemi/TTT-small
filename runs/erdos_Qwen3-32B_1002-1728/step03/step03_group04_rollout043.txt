def adjust_vector(perturbed, target_sum, eps=1e-8):
    n_points = len(perturbed)
    perturbed = np.clip(perturbed, 0.0, 1.0)
    perturbed_sum = np.sum(perturbed)
    if abs(perturbed_sum - target_sum) < eps:
        return perturbed
    delta = target_sum - perturbed_sum
    indices = np.argsort(perturbed)

        indices = indices[::-1]

            addable = 1.0 - perturbed[i]

            delta -= added

                break

                break
