                pbest_values[i] = val
                if val < gbest_value:
                    gbest_value = val
                    gbest_position = new_pos.copy()

    all_candidates = np.vstack([particles, pbest_positions])
    _, idx = np.unique(np.round(all_candidates, 8), axis=0, return_index=True)
    candidates = all_candidates[idx].tolist()
    candidates.sort(key=objective)
    top_candidates = candidates[:5]

    best_c5 = gbest_value
    best_h = gbest_position.copy()

    for h0 in top_candidates:
        if time.time() - start_time > budget_s * 0.98:
            break
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        res = minimize(objective, h0=h0, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 120, 'ftol': 1e-8})
        if res.success:
            h_candidate = res.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
            c5 = objective(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    assert np.all(best_h <= 1 + 1e-6) and np.all(best_h >= -1e-6)
    c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), c5_bound, n_points)
