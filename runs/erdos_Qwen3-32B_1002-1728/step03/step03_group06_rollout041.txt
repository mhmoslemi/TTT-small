import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_triangular(n):
        x = np.linspace(0, 2, n)
        h = 1 - np.abs(x - 1.0)
        h = np.clip(h, 0, 1)
        h = (h / np.sum(h)) * target_sum
        return h

    def make_step(n):
        h = np.zeros(n)
        h[:n//2] = 1.0
        return h

    initial_guesses = [
        h_init,
        make_triangular(n_points),
        make_step(n_points),
        0.5 * np.ones(n_points),
    ]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for guess in initial_guesses:
        # Normalize guess to target_sum and clip
        guess = np.clip(guess, 0, 1)

        result = minimize(
            fun=lambda h: np.max(np.correlate(h, 1-h, 'full')) * dx,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = np.clip(result.x, 0, 1)

                if c5_candidate < best_c5:

                    best_c5 = c5_candidate
