import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    rs = np.random.RandomState(seed)
    
    # Generate initial population
    population = [h_init.copy()]
    for _ in range(5):
        h_pert = h_init + rs.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert_sum = np.sum(h_pert)
        if abs(h_pert_sum - target_sum) > 1e-6:
            h_pert = (h_pert / h_pert_sum) * target_sum
        population.append(h_pert)
    for _ in range(4):
        indices = rs.permutation(n_points)
        h_binary = np.zeros(n_points)
        h_binary[indices[:n_points//2]] = 1.0
        population.append(h_binary)
    even = np.zeros(n_points)
    even[::2] = 1.0
    population.append(even)
    odd = np.zeros(n_points)
    odd[1::2] = 1.0
    population.append(odd)
    checkerboard = np.kron([1, 0], np.ones(n_points//2)).flatten()[:n_points]
    population.append(checkerboard)
    
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    alpha_surrogate = 0.1
    
    def surrogate(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        sorted_corr = np.sort(corr)[::-1]
        weights = np.exp(-alpha_surrogate * np.arange(len(sorted_corr)))
        return np.sum(weights * sorted_corr * dx)
    
    def threshold(h_vec):
        h = h_vec.reshape(n_points)
        sorted_indices = np.argsort(h)
        h_binary = np.zeros(n_points)
        h_binary[sorted_indices[-n_points//2:]] = 1.0
        return h_binary
    
    def perturb_binary(h_binary, p=0.03):
        n_flip = max(1, int(p * n_points))
        to_flip = rs.choice(n_points, size=n_flip, replace=False)
        h_pert = h_binary.copy()
        for idx in to_flip:
            h_pert[idx] = 1 - h_pert[idx]
        if np.sum(h_pert) != target_sum:
            if np.sum(h_pert) < target_sum:
                idx_zeros = np.where(h_pert == 0)[0]
                h_pert[np.random.choice(idx_zeros)] = 1.0
            else:
                idx_ones = np.where(h_pert == 1)[0]
                h_pert[np.random.choice(idx_ones)] = 0.0
        return h_pert
    
    while time.time() - start_time < budget_s * 0.9:
        for candidate in list(population):
            if not np.isnan(candidate).any():
                result = minimize(
                    fun=surrogate,
                    x0=candidate,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-6}
                )
                if result.success:
                    h_opt = result.x
                    corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
                    c5_opt = np.max(corr_opt) * dx
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                    h_binary = threshold(h_opt)
                    result_bin = minimize(
                        fun=surrogate,
                        x0=h_binary,
                        method='L-BFGS-B',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 30, 'ftol': 1e-6}
                    )
                    if result_bin.success:
                        corr_bin = np.correlate(result_bin.x, 1 - result_bin.x, mode='full')
                        c5_bin = np.max(corr_bin) * dx
                        if c5_bin < best_c5:
                            best_h = result_bin.x
                            best_c5 = c5_bin
                    h_pert = perturb_binary(h_binary)
                    population.append(h_pert)
        if len(population) < 30:
            new = best_h + rs.normal(0, 0.05, n_points)
            new = np.clip(new, 0, 1)
            new_sum = np.sum(new)
            if abs(new_sum - target_sum) > 1e-6:
                new = (new / new_sum) * target_sum
            population.append(new)
        population = population[-150:] if len(population) > 150 else population
    
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
