    if best_sum < 1e-10:
        best_h_scaled = np.zeros(n_points)
    else:
        best_h_scaled = best_h_clipped * (m / best_sum)
    corr = np.correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
    best_c5 = np.max(corr) * dx

    # Final checks
    assert abs(np.sum(best_h_scaled) - m) < 1e-6, "Sum constraint violated"
    assert np.all((best_h_scaled >= -1e-6) & (best_h_scaled <= 1.0 + 1e-6)), "Bounds violated"
    return (best_h_scaled.tolist(), best_c5, n_points)
