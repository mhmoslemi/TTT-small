import numpy as np
from scipy.optimize import minimize, differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    even = np.zeros(n_points)
    even[np.arange(0, n_points, 2)] = 1.0
    odd = np.zeros(n_points)
    odd[np.arange(1, n_points, 2)] = 1.0
    checkerboard = np.kron([1, 0], np.ones(n_points//2))[:n_points]
    rs = np.random.RandomState(seed)
    indices = rs.permutation(n_points)
    random1 = np.zeros(n_points)
    random1[indices[:n_points//2]] = 1.0
    additional_initials = [even, odd, checkerboard, random1]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def binary_h(h_vec, target_sum_bin):
        h = h_vec.reshape(n_points)
        n_points = len(h)
        h_sorted = np.argsort(h)
        n_ones = int(target_sum_bin)
        h_binary = np.zeros(n_points)
        h_binary[h_sorted[-n_ones:]] = 1.0
        return h_binary

    result_init = minimize(fun=objective, x0=h_init, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-8})
    best_h = result_init.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Differential Evolution with penalty
    def penalized_objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e4 * (sum_h - target_sum) ** 2
        return max_corr + penalty

    if time.time() - start_time < budget_s * 0.8:
        de_bounds = [(0.0, 1.0)] * n_points
        de_result = differential_evolution(penalized_objective, de_bounds, strategy='best1bin',
                                           maxiter=50, popsize=10, tol=1e-5,

        h_de = de_result.x.reshape(n_points)

            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    result_bin = minimize(fun=objective, x0=h_binary, method='SLSQP', bounds=bounds, constraints=cons,

    if result_bin.success:

        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')

            next_initial = perturbed
