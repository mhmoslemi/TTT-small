import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Seed for randomness
    np.random.seed(seed)
    iteration = 0
    no_improvement = 0
    max_no_improvement = 50

    while time.time() - start_time < budget_s * 0.98:

        elapsed = time.time() - start_time
        time_frac = max(0, min(1.0, elapsed / (budget_s * 0.98)))
        noise_std = 0.1 * (1 - time_frac) + 0.01 * time_frac
        rs = np.random.RandomState(seed + iteration)

        if rs.rand() < 0.5:
            # Continuous perturbation with adaptive noise
            perturbed = best_h + rs.normal(0, noise_std, n_points)
        else:

            m = int(round(target_sum))
            # Add noise to current h for index selection
            noisy_h = best_h + rs.normal(0, 0.1, n_points)
            indices = np.argpartition(noisy_h, -m)[-m:]

            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, 0.05, n_points)

        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if abs(perturbed_sum) > 1e-10:
            perturbed = (perturbed / perturbed_sum) * target_sum

        # Local refinement
        inner_result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 250, 'ftol': 1e-8}
        )

        if inner_result.success:
            h_candidate = inner_result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):

                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    no_improvement = 0
                else:
                    no_improvement += 1

        iteration += 1

        # Early stopping if no improvement in recent iterations
        if no_improvement > max_no_improvement:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
