import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    max_iter = 500
    convergence_norm = 1e-6
    iter_count = 0

    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)

        # Construct gradient for the worst lag
        g = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                g[i] = 1 - h[j]

        # Solve LP to minimize the weighted delta_h
        delta_h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(g * delta_h_var))
        constraints = [
            cp.sum(delta_h_var) == 0,
            delta_h_var <= 1 - h,
            delta_h_var >= -h
        ]
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.SCIP, verbose=False)
        except Exception:
            pass

        if delta_h_var.value is not None and problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            delta = delta_h_var.value
            norm_delta = np.linalg.norm(delta)
            if norm_delta < convergence_norm:
                break
            h += delta
            h = np.clip(h, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            iter_count += 1

    # Validate final constraints
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
