            if i + 1 >= len(parents):

            p1 = parents[i]

            # Uniform crossover
            child = np.zeros(n_points, dtype=np.float64)
            for j in range(n_points):
                if rs.rand() < 0.5:
                    child[j] = p1[j]
                else:
                    child[j] = p2[j]
            # Repair sum
