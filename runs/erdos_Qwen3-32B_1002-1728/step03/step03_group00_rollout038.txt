    for i in range(1, population_size):
        population[i] = (initial_h_half + rng.normal(0, 0.1, half_n)).copy()
        population[i] = np.clip(population[i], 0, 1)
        s = np.sum(population[i])
        if s != 0:
            population[i] = (population[i] / s) * target_sum_half
        population[i] = np.clip(population[i], 0, 1)

    best_h_half = initial_h_half.copy()
    best_c5 = evaluate(best_h_half)

    while time.time() - start_time < budget_s - 30:
        # Evaluate all
        fitness = np.array([evaluate(ind) for ind in population])
        current_idx = np.argmin(fitness)
        if fitness[current_idx] < best_c5:
            best_h_half = population[current_idx].copy()
            best_c5 = fitness[current_idx]

        # Selection via tournament
        selected_indices = []
        for _ in range(population_size // 2):
            idxs = rng.choice(population_size, size=3, replace=False)
            best_inxs = idxs[np.argmin(fitness[idxs])]
            selected_indices.append(best_inxs)
        selected_parents = population[selected_indices]

        # Crossover and mutation
        offspring = []
        for i in range(0, len(selected_parents), 2):
            if i + 1 >= len(selected_parents):
                continue
            p1, p2 = selected_parents[i], selected_parents[i + 1]
            # Mean crossover
            child1 = (p1 + p2) * 0.5
            # Single-point crossover
            crossover_point = rng.integers(1, half_n - 1)
            child2 = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
            for child in [child1, child2]:
                # Mutation
                if rng.random() < 0.1:  # mutation rate
                    child += rng.normal(0, 0.05, half_n)
                child = np.clip(child, 0, 1)
                s = np.sum(child)
                if s != 0:
                    child = (child / s) * target_sum_half
                child = np.clip(child, 0, 1)
                offspring.append(child)

        # Update population with elitism and offspring
        new_population = [best_h_half]
        while len(new_population) < population_size:
            if offspring:
                new_population.append(offspring.pop(0))
            else:
                new_population.append(population[rng.integers(population_size)])
        population = np.array(new_population)

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
