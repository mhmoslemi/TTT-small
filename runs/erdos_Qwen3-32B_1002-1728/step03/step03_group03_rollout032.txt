import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h_flat = h_vec.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    # Continuously optimize from different perturbed states
    h = np.array(initial_h_values)
    # First round of optimization
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-10}
    )
    if result.success:
        h = result.x
    else:
        print("Initial optimization failed, using original guess")

    # Perturbation and local search
    best_h = h.copy()
    best_c5 = objective(h)

    # Try multiple random restarts within time budget
    while time.time() - start_time < budget_s:
        np.random.seed(int((time.time() % 1000) ^ seed))
        noise = np.random.normal(0, 0.05, n_points) * (1 - h * 2)  # Keep noise symmetric
        candidate = np.clip(h + noise, 0, 1)
        # Renormalize to maintain constraint
        if np.sum(candidate) != 0:
            candidate = (candidate / np.sum(candidate)) * target_sum
        # Optimize candidate
        result_restart = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result_restart.success and np.all(np.isfinite(result_restart.x)):
            h_candidate = result_restart.x
            # Validate constraints
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = (h_candidate / np.sum(h_candidate)) * target_sum
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        # Conservative time monitoring
        if time.time() - start_time > budget_s * 0.95:
            break

    # Final cleaning pass
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * target_sum

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
