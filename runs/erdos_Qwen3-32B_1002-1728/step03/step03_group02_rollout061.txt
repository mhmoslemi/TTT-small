import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    np.random.seed(seed)
    population_size = 80
    population = [h_init.copy()]
    
    # Generate diverse initial population
    for _ in range(population_size - 1):
        if np.random.rand() < 0.5:
            # Perturbed initial
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        else:
            # Random binary
            idx = np.random.permutation(n_points)
            h = np.zeros(n_points)
            h[idx[:int(target_sum)]] = 1.0
        population.append(h)
    
    best_h = h_init.copy()
    best_c5 = np.inf
    
    def surrogate_objective(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full')
        weights = np.exp(-0.1 * np.sort(-overlaps))
        return np.sum(overlaps * weights) * dx
    
    def constrained_objective(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full')
        return np.max(overlaps) * dx
    
    def binary_threshold(h_vec):
        h_sorted = np.argsort(h_vec)
        n_ones = int(target_sum)
        h_binary = np.zeros(n_points)
        h_binary[h_sorted[-n_ones:]] = 1.0
        return h_binary
    
    def perturb_binary(h_vec, rate=0.03):
        h = h_vec.copy()
        n_flip = int(n_points * rate)
        idx1 = np.random.choice(n_points, n_flip, replace=False)[h > 0.5]
        idx0 = np.random.choice(n_points, n_flip, replace=False)[h < 0.5]
        h[idx1] = 0.0
        h[idx0] = 1.0
        return h
    
    while time.time() - start_time < budget_s * 0.98:
        np.random.shuffle(population)
        for i in range(0, min(len(population), 40)):
            if time.time() - start_time > budget_s:
                break
            h_candidate = population[i].copy()
            
            # Continuous optimization with surrogate
            result_cont = minimize(
                fun=surrogate_objective,
                x0=h_candidate,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 50}
            )
            if result_cont.success:
                h_cont = result_cont.x.reshape(n_points)
                h_cont_sum = np.sum(h_cont)
                if abs(h_cont_sum - target_sum) > 1e-6:
                    h_cont = h_cont * target_sum / h_cont_sum
                population[i] = h_cont
                
                # Binary optimization
