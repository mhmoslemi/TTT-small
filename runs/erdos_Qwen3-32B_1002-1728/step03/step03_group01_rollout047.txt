import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initializations
    initializations = [h_init.copy()]
    rs = np.random.RandomState(seed)
    for _ in range(14):  # 15 total initializations
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)

    best_c5 = float('inf')
    best_h = h_init.copy()

    # Simulated Annealing Parameters
    T_initial = 100.0
    alpha = 0.999
    max_steps_per_SA = 1000
    T_min = 1e-5

    for h0 in initializations:
        if time.time() - start_time > 0.9 * budget_s:
            break
        current_h = h0.copy().astype(float)
        current_energy = objective(current_h)
        best_SA_h = current_h.copy()
        best_SA_energy = current_energy
        T = T_initial

        for step in range(max_steps_per_SA):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # Perturb solution
            new_h = current_h + np.random.normal(0, T * 0.1, n_points)
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum < 1e-9:
                continue  # Avoid division by zero
            new_h = new_h * target_sum / new_sum
            new_energy = objective(new_h)
            delta_energy = new_energy - current_energy

            if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):
                current_h[:] = new_h
                current_energy = new_energy
                if current_energy < best_SA_energy:
                    best_SA_h[:] = current_h
                    best_SA_energy = current_energy

            # Cool down
            T *= alpha
            if T < T_min:
                break

        # Update global best
        if best_SA_energy < best_c5:
            best_c5 = best_SA_energy
            best_h[:] = best_SA_h

    # Post-SA refinement with SLSQP
    time_left = budget_s - (time.time() - start_time)
    if time_left > 50:
        top_perturbed = [best_h.copy()]
        for _ in range(2):
            perturbed = best_h.copy()
            perturbed += np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum > 1e-9:
                perturbed *= target_sum / perturbed_sum
            top_perturbed.append(perturbed)

        for candidate in top_perturbed:
            if time.time() - start_time > budget_s - 20:  # Final safety margin
                break
            result = minimize(
                fun=objective,
                x0=candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 120, 'ftol': 1e-8}
            )
            if result.success:
                h_candidate = result.x.reshape(n_points)
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h[:] = h_candidate
                    best_c5 = c5

    # Final constraint validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
