    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    if (time.time() - start_time) < budget_s * 0.9:  # 90% time budget for global search
        try:
            result_de = differential_evolution(
                de_objective,
                bounds=bounds,
                args=(n_points, target_sum, dx),
                strategy='best1bin',
                mutation=(0.5, 1.0),
                recombination=0.8,
                popsize=15,
                tol=1e-6,
                maxiter=40,
                polish=False,
                init='sobol',
                seed=seed,
                workers=1
            )
            h_de = result_de.x
            h_de_clipped = np.clip(h_de, 0, 1)
            s_de = np.sum(h_de_clipped)
            if s_de > 1e-12:  # Ensure non-zero sum
                h_de_scaled = h_de_clipped * (target_sum / s_de)
            else:
                h_de_scaled = h_de_clipped
            h_de_scaled = np.clip(h_de_scaled, 0, 1)
            corr_de = np.correlate(h_de_scaled, 1 - h_de_scaled, mode='full')
            c5_de = np.max(corr_de) * dx
            if c5_de < best_c5:
                best_h = h_de_scaled
                best_c5 = c5_de
        except:
            pass

    if (time.time() - start_time) < budget_s * 0.95:
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        bounds_local = bounds

        def objective_local(h_vec):
            h = h_vec.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        result = minimize(
            fun=objective_local,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
