import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_full(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def clip_and_rescale(h):
        h_clipped = np.clip(h.copy(), 0.0, 1.0)
        total = np.sum(h_clipped)
        if abs(total - target_sum) > 1e-6 and total != 0:
            h_clipped = h_clipped * (target_sum / total)
        h_clipped = np.clip(h_clipped, 0.0, 1.0)
        return h_clipped

    h_clipped = clip_and_rescale(h_initial)

    bounds_full = [(0.0, 1.0)] * n_points
    constraints_full = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    np.random.seed(seed)
