        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        return h_scaled

    def evaluate_fitness(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    pop_size = 60
    generations = 80
    tournament_size = 5
    mutation_rate = 0.2
    elites = 5

    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = clip_and_scale(h_rand)
        population.append(h_rand)
    population = np.array(population)

    best_h = h_init.copy()
    best_c5 = evaluate_fitness(best_h)

    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.98:
            break
        fitness = np.array([evaluate_fitness(ind) for ind in population])
        elite_indices = np.argsort(fitness)[:elites]
        elite_pop = population[elite_indices]

        parents = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(pop_size, tournament_size)
            tournament_fitness = fitness[tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            parents.append(population[winner_idx].copy())
        parents = np.array(parents)

        offspring = []
        for i in range(len(parents)):
            if np.random.rand() < 0.8:
                p1, p2 = parents[np.random.choice(len(parents), 2)]
                cp = np.random.randint(1, n_points)
                child = np.concatenate([p1[:cp], p2[cp:]])
            else:
                child = parents[np.random.choice(len(parents))].copy()
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
                for j in mut_idx:
                    if np.random.rand() < 0.5:
                        child[j] += np.random.uniform(-0.1, 0.1)
                    else:
                        child[j] = 1 - child[j]
            child = clip_and_scale(child)
            offspring.append(child)
        offspring = np.array(offspring)

        combined = np.concatenate([elite_pop, parents, offspring])
        n_combined = combined.shape[0]
        if n_combined > pop_size:
            scores = [evaluate_fitness(c) for c in combined]
            idxs = np.argsort(scores)
            population = combined[idxs[:pop_size]]
        else:
            population = combined

        candidate_scores = np.array([evaluate_fitness(p) for p in population])
        generation_best = np.min(candidate_scores)
        if generation_best < best_c5:
            best_c5 = generation_best
            best_h = population[np.argmin(candidate_scores)].copy()

    final_h = clip_and_scale(best_h)
    final_c5 = evaluate_fitness(final_h)

    def refine():
        bounds = [(0.0, 1.0)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        result = minimize(
            fun=lambda h: evaluate_fitness(h),
            x0=final_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = clip_and_scale(result.x)
            c5_candidate = evaluate_fitness(h_candidate)
            return h_candidate.tolist(), c5_candidate
        return (final_h.tolist(), final_c5)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        final_h, final_c5 = refine()

    assert abs(np.sum(final_h) - target_sum) < 1e-6
    assert np.all((0 <= final_h) & (final_h <= 1))
    return (final_h, final_c5, n_points)
