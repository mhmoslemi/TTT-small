        return np.max(corr) * dx

            initializations.append(h_scaled)

    best_c5 = np.inf
    best_h = h_init.copy()

        h0 = h0.copy()
        h0_sum = np.sum(h0)
        if not np.isclose(h0_sum, target_sum):

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    rs = np.random.RandomState(seed)
    initializations = [h_init.copy()]

    # Generate diverse initializations
    for _ in range(10):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)

    for _ in range(3):
        x_vals = np.linspace(0, 2, n_points)
        freq = rs.uniform(1, 4)
        amp = rs.uniform(0.2, 0.4)
        h_base = 0.5 + amp * np.sin(freq * np.pi * x_vals)
        h_base = np.clip(h_base, 0, 1)
        sum_h = np.sum(h_base)
        if sum_h > 1e-9:
            h_scaled = h_base * target_sum / sum_h
            h_scaled = np.clip(h_scaled, 0, 1)
            initializations.append(h_scaled)

    # Global search with Nelder-Mead
    best_c5 = float('inf')
    best_h = h_init.copy()
    candidates = []

    for h0 in initializations:
        if time.time() - start_time > budget_s * 0.9:
            break
        h0 = h0.copy()
        h0_sum = np.sum(h0)
        if not np.isclose(h0_sum, target_sum):
            h0 *= target_sum / h0_sum
            h0 = np.clip(h0, 0, 1)

        nm_result = minimize(
            fun=objective,
            x0=h0,
            method='Nelder-Mead',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 100, 'xatol': 1e-5, 'fatol': 1e-5}
        )

        if nm_result.success:
            h_candidate = nm_result.x.copy()
            h_sum = np.sum(h_candidate)
            if not np.isclose(h_sum, target_sum):
                if h_sum > 1e-6:
                    h_candidate *= target_sum / h_sum
                h_candidate = np.clip(h_candidate, 0, 1)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            candidates.append((c5_candidate, h_candidate))
        else:
            corr = np.correlate(h0, 1 - h0, mode='full')
            c5_candidate = np.max(corr) * dx
            candidates.append((c5_candidate, h0))

    candidates.sort()
    top_fraction = max(1, len(candidates) // 10)
    refined = []

    # Local refinement with SLSQP
    for c5, h_start in candidates[:top_fraction]:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            fun=objective,
            x0=h_start,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.copy()
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            refined.append((c5_candidate, h_candidate))

        if perturbed_sum > 1e-9:

            perturbed = np.clip(perturbed, 0, 1)

        result = minimize(

            x0=perturbed,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,

            options={'maxiter': 150, 'ftol': 1e-8}
