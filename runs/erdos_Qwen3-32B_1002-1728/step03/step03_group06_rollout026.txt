import time
import numpy as np
import random
from copy import deepcopy

def tournament_selection(population, fits, tournament_size):
    indices = np.random.choice(len(population), tournament_size)
    selected_idx = indices[np.argmin([fits[i] for i in indices])]
    return population[selected_idx].copy()

def crossover(parent1, parent2):
    alpha = np.random.rand()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(child, mut_prob, mut_std, target_sum):
    if np.random.rand() < mut_prob:
        mutation = np.random.normal(0, mut_std, child.shape)
        child += mutation
        child = np.clip(child, 0, 1)
        child *= target_sum / np.sum(child)
    return child

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 50
    tournament_size = 3
    elitism_count = 10
    mut_std = 0.1
    mut_prob = 0.2
    crossover_prob = 0.8

    np.random.seed(seed)
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= target_sum / h.sum()
        population.append(h)

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s * 0.99:
        fits = []
        for ind in population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            max_corr = np.max(corr) * dx
            fits.append(max_corr)

        idx = np.argmin(fits)
        current_best = fits[idx]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[idx].copy()

        new_population = [population[idx].copy() for _ in range(elitism_count)]

        while len(new_population) < pop_size:
            p1 = tournament_selection(population, fits, tournament_size)
            p2 = tournament_selection(population, fits, tournament_size)
            c1, c2 = p1.copy(), p2.copy()
            if random.random() < crossover_prob:
                c1, c2 = crossover(p1, p2)
            c1 = mutate(c1, mut_prob, mut_std, target_sum)
            c2 = mutate(c2, mut_prob, mut_std, target_sum)
            if len(new_population) < pop_size:
                new_population.append(c1)
            if len(new_population) < pop_size:
                new_population.append(c2)

        population = new_population

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
