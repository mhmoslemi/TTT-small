        new_energy = compute_energy(current_h)
        current_energy = compute_energy(current_h)

        # Acceptance probability
        delta_E = current_energy - new_energy
        if delta_E > 0 or np.random.rand() < np.exp(delta_E / T):
            # Accept the move
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = current_h.copy()
        else:
            # Reject the move, swap back
            current_h[i], current_h[j] = current_h[j], current_h[i]

        T *= alpha
        if T < T_min:
            T = T_min

    assert abs(np.sum(best_h) - target_sum) < 1e-5, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
