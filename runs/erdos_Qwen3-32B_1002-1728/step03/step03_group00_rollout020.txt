    h_full = best_h_flat.reshape(n_points)
    h_clipped = np.clip(h_full, 0, 1)
    s = np.sum(h_clipped)
    scale = target_sum / s if s > 1e-6 else 1.0
    h_scaled = np.clip(h_clipped * scale, 0, 1)
    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
    final_c5 = np.max(corr) * dx

    assert abs(np.sum(h_scaled) - target_sum) < 1e-5, "Integral constraint violated"
    assert np.all((h_scaled >= -1e-6) & (h_scaled <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_scaled.tolist(), final_c5, n_points)
