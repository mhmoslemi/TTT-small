import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rs = np.random.RandomState(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def continuous_objective(h_vec):
        return compute_energy(h_vec.reshape(n_points))

    # Phase 1: Continuous optimization
    bounds_continuous = [(0.0, 1.0)] * n_points
    cons_continuous = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    result_continuous = minimize(
        continuous_objective,
        x0=h_init,
        method='L-BFGS-B',
        bounds=bounds_continuous,
        constraints=cons_continuous,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    if not result_continuous.success:
        best_h = np.copy(h_init)
        best_energy = compute_energy(h_init)
    else:
        h_continuous = result_continuous.x.reshape(n_points)
        best_h = np.copy(h_continuous)
        best_energy = compute_energy(h_continuous)

    # Phase 2: Binary optimization with simulated annealing
    def threshold_h(h):
        indices = np.argsort(h)
        h_binary = np.zeros(n_points)
        h_binary[indices[-int(target_sum):]] = 1.0
        # Fix tie-breaking rounding errors
        while np.sum(h_binary) < target_sum:
            h_binary[np.argmin(h_binary)] = 1.0
        while np.sum(h_binary) > target_sum:
            h_binary[np.argmax(h_binary)] = 0.0
        return h_binary

    best_h = threshold_h(best_h)
    best_energy = compute_energy(best_h)

    # SA parameters
    T_initial = 1.0
    alpha = 0.99
    T = T_initial
    T_min = 1e-3

    accepted = 0
    rejected = 0

    while T > T_min and time.time() - start_time < budget_s * 0.98:
        i, j = rs.choice(n_points, size=2, replace=False)
        if best_h[i] == best_h[j]:
            continue

        # Swap values and maintain mass constraint
        proposed = best_h.copy()
        proposed[i], proposed[j] = proposed[j], proposed[i]
        proposed_energy = compute_energy(proposed)

        delta_E = proposed_energy - best_energy

        if delta_E < 0 or rs.rand() < np.exp(-delta_E / T):
            best_h = proposed
            best_energy = proposed_energy
            accepted += 1
        else:
            rejected += 1

        # Adaptive cooling
        if accepted > rejected:
            T *= alpha ** 2  # Faster cooling if effective
        else:
            T *= alpha

    # Final verification
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Mass constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_energy, n_points)
