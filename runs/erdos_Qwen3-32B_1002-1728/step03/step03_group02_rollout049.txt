        ones = np.where(h_vec == 1)[0]
        zeros = np.where(h_vec == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return h_vec.copy()
        i = random_state.choice(ones)
        j = random_state.choice(zeros)
        new_h = h_vec.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return new_h

    while time.time() - start_time < budget_s * 0.98:
        new_h = propose_move(current_h, rs)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or rs.rand() < np.exp(-delta / temperature):
            current_h = new_h
            current_c5 = new_c5

        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

        temperature *= cooling_rate

        # Restart with random solution if no progress for 1e4 steps
        if (time.time() - start_time) > 300 and not hasattr(run, 'restart_count'):
            run.restart_count = 0
            new_h = np.zeros(n_points)
            indices = rs.permutation(n_points)
            new_h[indices[:int(target_sum)]] = 1.0
            current_h = new_h
            current_c5 = compute_c5(current_h)
            temperature = max(temperature, initial_temperature)
            run.restart_count += 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
