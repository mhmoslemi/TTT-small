import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 100  # Increased resolution for more precise patterns
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Interpolate from initial_h_values to new resolution
    h_init = np.array(initial_h_values)
    x0 = np.linspace(0, 2, len(h_init))
    x = np.linspace(0, 2, n_points)
    h_interp = np.interp(x, x0, h_init)
    h_interp *= target_sum / h_interp.sum()
    h_base = np.clip(h_interp, 0, 1)

    # Normalize to constraint
    h_base = h_base * (target_sum / h_base.sum())

    # Diversity-initial population (pop_size=30)
    pop_size = 30
    rs = np.random.RandomState(seed)
    population = [h_base.copy()]

    # Binary patterns
    for _ in range(10):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        population.append(h)

    # Sinusoidal patterns
    for phase in np.linspace(0, 2*np.pi, 6, endpoint=False):
        for freq in [1, 2.5, 5, 8]:
            h = 0.5 + 0.5*np.sin(2*np.pi*freq*x + phase)
            h = np.clip(h, 0, 1)
            h *= target_sum / h.sum()
            population.append(h)

    # Triangular ramps
    for dir in [1, -1]:
        h = (dir * (x - 1) + 1.0) / 2
        h = np.clip(h, 0, 1)
        h *= target_sum / h.sum()
        population.append(h)

    # Ensure population size
    population = population[:pop_size]
    population = [np.clip(np.array(p), 0, 1) for p in population]

    # Constraint enforcement function
    def rescale(x):
        s = x.sum()
        if abs(s) < 1e-12:
            idx = rs.choice(n_points, int(target_sum), replace=False)
            return np.array([1.0 if i < int(target_sum) else 0 for i in idx] + [0]*int(n_points - int(target_sum))) * (target_sum / int(target_sum))
        return np.clip(x, 0, 1) * (target_sum / s)

    # Evaluation function
    def evaluate(x):
        h = np.reshape(x, (-1,))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Differential evolution with rescaling constraints
    best_h = h_base.copy()
    best_c5 = evaluate(h_base)

    F = 0.9  # Scaling factor
    CR = 0.85  # Crossover rate

    # Main DE loop
    generation = 0
    while time.time() - start_time < budget_s * 0.85 and generation < 150:
        new_population = population[:]
        for i in range(pop_size):
            a, b, c = rs.choice(pop_size, 3, replace=False)
            mutant = np.clip(population[a] + F * (population[b] - population[c]), -1, 2)
            trial = population[i].copy()
            mask = rs.rand(n_points) < CR
            trial[mask] = mutant[mask]
            trial = rescale(trial)
            trial_c5 = evaluate(trial)
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5
            if trial_c5 < evaluate(population[i]):
                new_population[i] = trial
        population = new_population
        generation += 1

    # Local search refinement
    if time.time() - start_time < budget_s * 0.97:
        for _ in range(5):
            perturbed = best_h.copy()
            perturbed += np.random.uniform(-0.1, 0.1, n_points)
            perturbed = rescale(perturbed)
            res = minimize(evaluate, perturbed, bounds=[(0,1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
                          method='L-BFGS-B' if 'L-BFGS-B' in minimize.methods else 'TNC',
                          options={'ftol': 1e-10})
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun

    # Final constraint checking
    h_final = np.clip(best_h, 0, 1)
    h_final *= target_sum / h_final.sum()
    final_c5 = evaluate(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
