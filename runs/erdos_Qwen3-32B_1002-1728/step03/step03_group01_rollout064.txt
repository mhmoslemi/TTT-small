import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Generate diverse initializations
    rs = np.random.RandomState(seed)
    initializations = [h_init.copy()]
    for _ in range(5):  # Binary patterns
        idx = rs.choice(n_points, n_points//2, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)
    for _ in range(5):  # Sinusoidal patterns
        freq = 0.1 + 0.5 * rs.rand()
        x = np.linspace(0, 2 * np.pi, n_points)
        h = 0.5 * np.sin(freq * x) + 0.5
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum > 1e-9:
            h *= target_sum / h_sum
        else:
            h = np.random.rand(n_points)
            h *= target_sum / h.sum()
        initializations.append(h)

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    # Simulated Annealing
    T_initial = 100.0
    alpha = 0.999
    min_T = 1e-5

    for h_start in initializations:
        if time.time() - start_time > budget_s * 0.9:
            break
        h = h_start.copy()
        T = T_initial
        current_energy = objective(h)
        while T > min_T and (time.time() - start_time) < budget_s * 0.95:
            # Perturb
            h_new = h + rs.normal(0, scale=T, size=n_points)
            h_new = np.clip(h_new, 0, 1)
            h_sum = h_new.sum()
            if h_sum > 1e-9:
                h_new *= target_sum / h_sum
            new_energy = objective(h_new)
            delta = new_energy - current_energy
            if delta < 0 or rs.rand() < np.exp(-delta / T):
                h = h_new.copy()
                current_energy = new_energy
                if current_energy < best_c5:
                    best_h = h.copy()
                    best_c5 = current_energy
            T *= alpha

    # Final SLSQP refinement
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.97:
            break
        perturbed = best_h.copy()
        perturbed += (rs.rand(n_points) - 0.5) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-9:
            perturbed *= target_sum / perturbed_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            candidate = result.x.reshape(n_points)
            c5_candidate = objective(candidate)
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
