def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)

    smoothing_eps = 0.001
    learning_rate = 0.05
    delta = 1e-4
    best_h = h.copy()

        exp_vals = np.exp((scaled - max_val) / smoothing_eps)
        return max_val + smoothing_eps * np.log(np.sum(exp_vals))

            sum_clamped = np.sum(clamped)
