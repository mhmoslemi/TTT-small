import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Objective function to minimize C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Repair function to maintain constraints
    def repair_h(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        sum_current = np.sum(h_clipped)
        if abs(sum_current) < 1e-10:
            return h_clipped
        h_scaled = h_clipped * (target_sum / sum_current)
        h_repaired = np.clip(h_scaled, 0, 1)
        return h_repaired

    # Tournament selection
    def tournament_selection(population, fitness, size=5):
        contestants = np.random.choice(len(population), size=size, replace=False)
        best_idx = contestants[0]
        for idx in contestants[1:]:
            if fitness[idx] < fitness[best_idx]:
                best_idx = idx
        return population[best_idx].copy()

    # Initialize population with perturbation and random candidates
    population = []
    for _ in range(100):
        if _ < 50:
            perturbed = h_init + np.random.normal(0, 0.05, n_points)
        else:
            perturbed = np.random.rand(n_points)
        population.append(repair_h(perturbed))
    fitness = [compute_c5(h) for h in population]

    # Genetic Algorithm with time tracking
    max_generations = 100
    for gen in range(max_generations):
        if time.time() - start_time > budget_s * 0.95:
            break
        new_population = []
        while len(new_population) < 100 and (time.time() - start_time) < budget_s * 0.98:
            # Parent selection
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            
            # Crossover (swap a random segment)
            start = np.random.randint(0, n_points)
            length = np.random.randint(int(0.2 * n_points), int(0.5 * n_points))
            end = min(start + length, n_points)
            child = p1.copy()
            child[start:end] = p2[start:end]
            
            # Mutation (10% of elements)
            if np.random.rand() < 0.1:
                indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
                for i in indices:
                    if np.random.rand() < 0.5:
                        child[i] += np.random.uniform(-0.1, 0.1)
                    else:
                        child[i] = 1 - child[i]
            child = repair_h(child)
            new_population.append(child)
        
        # Evaluate new population
        new_fitness = [compute_c5(h) for h in new_population]
        combined = population + new_population
        combined_fitness = np.array(fitness + new_fitness)
        idxs = np.argsort(combined_fitness)
        population = [combined[i] for i in idxs[:100]]
        fitness = list(combined_fitness[idxs[:100]])

    # Local refinement on top candidates
    best_h = None
    best_c5 = float('inf')
    for h in population[:5]:
        if time.time() - start_time > budget_s * 0.99:
            break
        result = minimize(
            fun=compute_c5,
            x0=h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            options={'maxiter': 200}
        )
        if result.success:
            refined_h = repair_h(result.x)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h

    # Final validation
    if best_h is None:
        best_h = population[np.argmin(fitness)].copy()
        best_c5 = min(fitness)
    
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
