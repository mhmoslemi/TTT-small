import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    # Continuous relaxation phase
    def objective_relax(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result_relax = minimize(
        fun=objective_relax,
        x0=h_init,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8}
    )
    h_relaxed = result_relax.x.reshape(n_points)

    # Binary thresholding
    indices = np.argsort(h_relaxed)
    n_ones = int(target_sum)
    h_binary = np.zeros(n_points)
    h_binary[indices[-n_ones:]] = 1.0

    # SA parameters
    rs = np.random.RandomState(seed)
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_binary.copy()
    best_c5 = compute_c5(best_h)
    ones_list = list(np.where(best_h)[0])
    zeros_list = list(np.where(1 - best_h)[0])
    
    T = 1.0
    alpha = 0.995
    T_min = 1e-4
    iter_count = 0

    while time.time() - start_time < budget_s * 0.95:
        if not ones_list or not zeros_list:
            break
        if T < T_min:
            T = 1.0  # Reset temperature
            alpha = 0.99  # Cooler rate for late stages

        # Propose a move (1-bit to 0 and 0-bit to 1)
        i = rs.choice(ones_list)
