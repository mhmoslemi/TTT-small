import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(x):
        h_full = create_full_h(x)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    # Symmetric initial guess from h_full_initial
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    total = h_full_sym.sum()
    if total != 0:
        scale = (n_points / 2.0) / total
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    best_h_half = h_full_sym[:half_n]
    best_obj = objective(best_h_half)

    np.random.seed(seed)
    current_seed = seed

    max_iter_per_run = 200
    while time.time() - start_time < budget_s - 10:
        current_seed += 1
        np.random.seed(current_seed)
        # Random initial h_vars scaled to target sum
        h_rand = np.random.rand(half_n)
        h_sum = h_rand.sum()
        if h_sum > 1e-6:
            h_rand *= (target_sum_half / h_sum)
        h_rand = np.clip(h_rand, 0, 1)

        result = minimize(
            fun=objective,
            x0=h_rand,
            method='COBYLA',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': max_iter_per_run, 'ftol': 1e-6}
        )

        if result.success:
            sol = result.x
            h_full = create_full_h(sol)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            new_obj = np.max(corr) * dx
            if new_obj < best_obj:
                best_h_half = sol.copy()
                best_obj = new_obj

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), best_obj, n_points)
