import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h_vec, dx):
    h = h_vec
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # PSO parameters
    num_particles = 30
    max_iter = 100
    w = 0.7
    c1 = 0.6
    c2 = 0.3
    max_vel = 0.5
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initialize particles
    rs = np.random.RandomState(seed)
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    for _ in range(num_particles):
        if _ == 0:
            pos = h_init.copy()
        else:
            # Random binary perturbation
            idx = rs.choice(n_points, int(target_sum), replace=False)
            p_binary = np.zeros(n_points)
            p_binary[idx] = 1.0
            pos = p_binary + rs.uniform(-0.1, 0.1, n_points)
            pos = np.clip(pos, 0, 1)
        # Normalize sum constraint
        pos_sum = np.sum(pos)
        pos = pos * (target_sum / pos_sum)
        particles.append(pos)
        velocities.append(rs.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(pos.copy())
        pbest_scores.append(compute_c5(pos, dx))
    
    gbest_idx = np.argmin(pbest_scores)
    gbest_pos = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO Main Loop
    for iter_pso in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
        for i in range(num_particles):
            current_pos = particles[i]
            current_vel = velocities[i]
            r1, r2 = rs.random(), rs.random()
            cognitive = c1 * r1 * (pbest_positions[i] - current_pos)
            social = c2 * r2 * (gbest_pos - current_pos)
            velocities[i] = w * current_vel + cognitive + social
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = current_pos + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            cur_sum = np.sum(particles[i])

            score = compute_c5(particles[i], dx)

            if score < pbest_scores[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_scores[i] = score
                # Update global best
