import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    rs = np.random.RandomState(seed)

    while time.time() - start_time < budget_s * 0.95:
        # Randomly choose between binary-based perturbation (80%) and continuous (20%)
        if rs.rand() < 0.8:
            m = int(round(target_sum))
            indices = rs.choice(n_points, size=m, replace=False)
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, 0.05 + best_c5 * 10, n_points)
        else:
            perturbed = best_h + rs.normal(0, 0.05, n_points)

        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = (perturbed / perturbed_sum) * target_sum

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
