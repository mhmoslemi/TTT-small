    h_init = np.array(initial_h_values)

    rho = 1.0
    dual_var = 0.0
    max_admm = 100

    while admm_it < max_admm and time.time() - start_time < budget_s * 0.7:

        def objective_aug(h_vec):

            corr = np.correlate(h, 1 - h, mode='full')

        result = minimize(
