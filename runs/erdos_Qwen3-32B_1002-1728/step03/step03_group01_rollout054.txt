    while len(population) < pop_size:
        h = rs.rand(new_n)
        h *= target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    population = np.array(population)
    best_c5 = float('inf')
    best_h = h_init_high.copy()

    F, CR = 0.8, 0.9
    max_gen = 50

    for generation in range(max_gen):
        if time.time() - start_time > budget_s * 0.9:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            indices = rs.choice(candidates, 3, replace=False)
            a, b, c = population[indices]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            if mutant_sum > 1e-9:
                mutant = mutant * target_sum / mutant_sum
            else:
                mutant = np.zeros_like(mutant)
            mutant = np.clip(mutant, 0, 1)
            target = population[i]
            trial = target.copy()
            cross_points = rs.rand(new_n) < CR
            if not np.any(cross_points):
                cross_points[rs.randint(new_n)] = True
            trial[cross_points] = mutant[cross_points]
            trial_obj = objective(trial)
            if trial_obj < objective(target):
                population[i] = trial
                if trial_obj < best_c5:
                    best_c5, best_h = trial_obj, trial.copy()

    top_candidates = [best_h.copy()]
    scores = np.array([objective(ind) for ind in population])
    top_indices = np.argsort(scores)[:5]
    for idx in top_indices:
        top_candidates.append(population[idx].copy())

    for h0 in top_candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 150, 'ftol': 1e-8})
        if result.success:
            h_candidate = result.x.reshape(new_n)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5, best_h = c5_candidate, h_candidate

    while time.time() - start_time < budget_s * 0.97:
        perturbed = best_h.copy()
        perturbed += (rs.rand(new_n) - 0.5) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-9:
            perturbed *= target_sum / perturbed_sum
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            h_candidate = result.x.reshape(new_n)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5, best_h = c5_candidate, h_candidate

    best_h = np.clip(best_h, 0, 1)
    best_h *= target_sum / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, new_n)
