        population = np.array(offspring)

    final_candidates = [best_h.copy()]
    if len(population) > 0:
        candidates = [pop for pop in population]
        candidate_c5 = [compute_c5(c) for c in candidates]
        top_indices = np.argsort(candidate_c5)[:4]
        for idx in top_indices:
            final_candidates.append(candidates[idx].copy())
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    
    for h_candidate in final_candidates:
        if time.time() - start_time > budget_s * 0.995:
            break
        result = minimize(
            fun=lambda x: compute_c5(x),
            x0=h_candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200}
        )
        if result.success:
            prop_h = result.x.copy()
            prop_sum = np.sum(prop_h)
            if abs(prop_sum - target_sum) < 1e-6 and np.all((prop_h >= -1e-6) & (prop_h <= 1.0 + 1e-6)):
                prop_c5 = compute_c5(prop_h)
                if prop_c5 < best_c5:
                    best_h = prop_h
                    best_c5 = prop_c5

    assert best_h is not None
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
