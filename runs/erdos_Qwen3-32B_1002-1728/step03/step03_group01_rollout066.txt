        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5

    # Local search refinement with larger perturbations
    while time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s * 0.98:
            break
        perturbed = best_h.copy()
        # Add larger noise and ensure constraint adherence
        noise = (np.random.rand(n_points) - 0.5) * 0.2  # Doubled magnitude
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-9:
            perturbed *= target_sum / perturbed_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h[:] = h_candidate[:]
                best_c5 = c5_candidate

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
