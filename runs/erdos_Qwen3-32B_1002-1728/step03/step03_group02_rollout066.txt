import time
import numpy as np
from scipy.optimize import minimize, L_BFGS_B

def surrogate_objective(h_vec, alpha, dx, n_points):
    h = h_vec.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full')
    shift_values = np.arange(len(corr)) - (n_points - 1)
    shift_k = shift_values * dx
    weights = np.exp(-alpha * np.abs(shift_k))
    return np.sum(weights * corr) * dx

def binary_h(h_vec, target_sum_bin):
    h_sorted = np.argsort(h_vec)
    n_ones = int(target_sum_bin)
    h_binary = np.zeros_like(h_vec)
    h_binary[h_sorted[-n_ones:]] = 1.0
    return h_binary

def constraint_sum(h_vec, target_sum):
    return np.sum(h_vec) - target_sum

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate additional structured initial guesses
    even = np.zeros(n_points)
    even[np.arange(0, n_points, 2)] = 1.0
    odd = np.zeros(n_points)
    odd[np.arange(1, n_points, 2)] = 1.0
    checkerboard = np.kron([1, 0], np.ones(n_points//2))[:n_points]
    rs = np.random.RandomState(seed)
    indices = rs.permutation(n_points)
    random1 = np.zeros(n_points)
    random1[indices[:n_points//2]] = 1.0

    # Sine-based patterns
    x_vals = np.linspace(0, 2, n_points)
    additional_initials = [
        even, odd, checkerboard, random1,
    ]
    for period in [1.5, 2.0, 2.5]:
        for thres in [0.4, 0.5, 0.6]:
            h_sine = (np.sin(np.pi * x_vals / period) > thres).astype(float)
            h_sine = h_sine * (target_sum / np.sum(h_sine))
            h_sine = np.clip(h_sine, 0, 1)
            additional_initials.append(h_sine)

    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    all_initials = additional_initials.copy()

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (target_sum,)}]

    while time.time() - start_time < budget_s * 0.95:
        if not all_initials:
            perturbed = best_h + rs.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if np.abs(perturbed_sum - target_sum) > 1e-6:
                perturbed = (perturbed / perturbed_sum) * target_sum
            next_initial = perturbed
        else:
            next_initial = all_initials.pop(0)

        for alpha in [0.05, 0.1, 0.2]:
            # Normalize initial guess to meet constraint
            x0 = next_initial.copy()
            x0_sum = np.sum(x0)
            if np.abs(x0_sum - target_sum) > 1e-6:
                x0 = x0 * (target_sum / x0_sum)

            result = minimize(
                fun=surrogate_objective,
                x0=x0,
                args=(alpha, dx, n_points),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 120, 'ftol': 1e-8}
            )
            if result.success:
                h_candidate = result.x.reshape(n_points)
                # Check and update best solution
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

                # Binarize and refine
                h_bin = binary_h(h_candidate, target_sum)
                result_bin = minimize(
                    fun=surrogate_objective,
                    x0=h_bin,
                    args=(alpha, dx, n_points),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 80, 'ftol': 1e-8}
                )
                if result_bin.success:
                    h_candidate_bin = result_bin.x
                    corr_candidate_bin = np.correlate(h_candidate_bin, 1 - h_candidate_bin, mode='full')
                    c5_candidate_bin = np.max(corr_candidate_bin) * dx
                    if c5_candidate_bin < best_c5:
                        best_h = h_candidate_bin
                        best_c5 = c5_candidate_bin

    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.all(np.isclose(best_h, best_h.clip(0, 1)))
    return (best_h.tolist(), best_c5, n_points)
