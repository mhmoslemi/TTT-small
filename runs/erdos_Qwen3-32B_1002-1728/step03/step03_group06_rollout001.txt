def compute_smooth_max(corr, epsilon):
    # Avoid overflow by subtracting max
    shift = np.max(corr)
    shifted = corr - shift
    sum_exp = np.sum(np.exp(shifted / epsilon))
    # Handle possible underflow (sum_exp approaches zero)
    if sum_exp < 1e-20:

    return shift + epsilon * np.log(sum_exp)

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project_h(h_vec, target_sum):
        h_clipped = np.clip(h_vec, 0.0, 1.0)
        current_sum = h_clipped.sum()
        if current_sum == 0 or np.isnan(current_sum) or np.isinf(current_sum):
            return h_clipped
        h_normalized = (h_clipped / current_sum) * target_sum
        return h_normalized

    h = project_h(h_init, target_sum)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()

    learning_rate = 0.1
    smoothing_epsilon = 1e-4
    max_iterations = 10000
    iteration = 0

    def compute_smooth_max(corr, epsilon):
        shift = np.max(corr)
        shifted = corr - shift
        sum_exp = np.sum(np.exp(shifted / epsilon))
        if sum_exp < 1e-20:
            sum_exp = 1e-20
        return shift + epsilon * np.log(sum_exp)

    while time.time() - start_time < budget_s * 0.98 and iteration < max_iterations:
        h_feasible = project_h(h, target_sum)
        n = len(h_feasible)
        eps_g = 1e-4
        grad = np.zeros(n)
        for i in range(n):
            # Perturb up
            h_perturbed = h_feasible.copy()
            h_perturbed[i] += eps_g
            h_p = project_h(h_perturbed, target_sum)
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')
            obj_p = compute_smooth_max(corr_p, smoothing_epsilon) * dx

            # Perturb down
            h_perturbed = h_feasible.copy()
            h_perturbed[i] -= eps_g
            h_m = project_h(h_perturbed, target_sum)
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')
            obj_m = compute_smooth_max(corr_m, smoothing_epsilon) * dx

            grad[i] = (obj_p - obj_m) / (2 * eps_g)

        h_proposal = h_feasible - learning_rate * grad
        h_proposal = project_h(h_proposal, target_sum)
        current_corr = np.correlate(h_proposal, 1 - h_proposal, mode='full')
        current_c5 = np.max(current_corr) * dx

        if current_c5 < best_c5:
            best_h = h_proposal.copy()
            best_c5 = current_c5
            learning_rate = 0.1
        else:
            learning_rate *= 0.9

        h = h_proposal.copy()
        iteration += 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
