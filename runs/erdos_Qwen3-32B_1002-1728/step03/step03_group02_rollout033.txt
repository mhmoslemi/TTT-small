import time
import numpy as np
import cvxpy as cp

def binary_h(h_vec, target_sum_bin):
    n_points = len(h_vec)
    h_sorted = np.argsort(h_vec)
    n_ones = int(target_sum_bin)
    h_binary = np.zeros(n_points)
    h_binary[h_sorted[-n_ones:]] = 1.0
    return h_binary

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial binary solution from h_init
    h_init_binary = binary_h(h_init, target_sum)

    # Precompute all possible shifts
    shifts = [k for k in range(-n_points + 1, n_points) if abs(k) <= n_points - 1]

    # MILP formulation
    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()

    constraints = [cp.sum(h) == target_sum]

    for k in shifts:
        valid_indices = np.arange(n_points - abs(k))
        if k > 0:
            lhs = cp.sum(h[valid_indices] * (1 - h[valid_indices + k]))
        else:
            lhs = cp.sum(h[valid_indices + k] * (1 - h[valid_indices]))
        constraints.append(lhs * dx <= t)

    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Set initial guess
    h.value = h_init_binary
    problem.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s - 10)

    best_h = np.round(h.value).flatten()
    best_c5 = t.value

    # Ensure feasibility
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
