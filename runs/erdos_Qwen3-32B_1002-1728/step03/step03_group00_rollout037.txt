    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)

    best_h = h_full_initial.copy()
    best_c5 = objective_full(best_h)

    np.random.seed(seed)
    perturbation_width = 0.2
    maxiter = 100

    while time.time() - start_time < budget_s - 30:
        # Create perturbed initial guess
        perturbation = np.random.randn(n_points) * perturbation_width
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum != 0:
            perturbed = (perturbed / current_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        # Constraints and bounds
        def constraint_sum(x):
            return np.sum(x) - target_sum

        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0.0, 1.0)] * n_points

        result = minimize(
            fun=objective_full,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-8}
        )

        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_sum = np.sum(new_h)
            if abs(new_sum - target_sum) > 1e-6:
                new_h += (target_sum - new_sum) / n_points
                new_h = np.clip(new_h, 0, 1)
            new_c5 = objective_full(new_h)
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6)) and new_c5 < best_c5):
                best_h = new_h
                best_c5 = new_c5

        # Reduce perturbation width over time
        perturbation_width *= 0.999

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
