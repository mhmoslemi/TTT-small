import time
import numpy as np
import random
from scipy.optimize import minimize

def crossover(parent1, parent2):
    diff_indices = np.where(parent1 != parent2)[0]
    if len(diff_indices) == 0:
        return parent1.copy()
    n_swap = np.random.randint(1, len(diff_indices) // 2 + 1)
    swap_indices = np.random.choice(diff_indices, n_swap, replace=False)
    child = parent1.copy()
    child[swap_indices] = 1 - child[swap_indices]
    return child

def mutate(individual):
    try:
        idx1 = np.random.choice(np.where(individual == 1)[0])
        idx0 = np.random.choice(np.where(individual == 0)[0])
        new_individual = individual.copy()
        new_individual[idx1] = 0
        new_individual[idx0] = 1
        return new_individual
    except:  # Fallback if no 0s or 1s to swap
        return individual.copy()

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    m = int(np.round(n_points / 2))  # Ensure integral constraint is satisfied
    dx = 2.0 / n_points
    
    def binarize(h_vec):
        indices = np.argpartition(h_vec, -m)[-m:]
        bin_h = np.zeros_like(h_vec)
        bin_h[indices] = 1.0
        return bin_h
    
    def evaluate(individual):
        corr = np.correlate(individual, 1 - individual, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 50
    init_binary = binarize(h_init)
    population = [init_binary]
    for _ in range(population_size - 1):

    best_idx = np.argmin(fitness)

    # Run genetic algorithm
    while time.time() - start_time < budget_s * 0.85:

        for _ in range(population_size):

            # Mutation

            best_c5 = fitness[current_best_idx]

            break
    
    # Local refinement of best binary solution
    target_sum = m

        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
