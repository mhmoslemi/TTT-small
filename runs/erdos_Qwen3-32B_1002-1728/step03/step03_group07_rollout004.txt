import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_energy = np.inf
    best_h = h_init.copy()
    rng = np.random.Generator(np.random.PCG64(seed))

    # Multiple restarts: using initial_h and 4 perturbed ones
    for restart in range(5):
        if restart > 0:
            # Perturb the current best h
            perturbation = 0.1 * (0.5 ** (restart - 1))
            h_current = best_h + perturbation * rng.normal(size=n_points)
            h_current = np.clip(h_current, 0, 1)
            current_sum = np.sum(h_current)
            if current_sum != 0:
                h_current *= target_sum / current_sum
            else:
                h_current = best_h.copy()
        else:
            h_current = h_init.copy()

        # Optimization with SLSQP
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def objective_func(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            return np.max(corr) * dx

        def gradient_func(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k_star = max_idx - (n_points - 1)
            grad = np.zeros_like(h)
            if k_star >= 0:
                for i in range(n_points - k_star):
                    grad[i] += 1.0 - h[i + k_star]
            else:
                k = -k_star
                for i in range(k, n_points):
                    grad[i] += 1.0 - h[i - k]
            return grad * dx

        h_flat = h_current.flatten().copy()
        result = scipy.optimize.minimize(
            fun=objective_func,
            x0=h_flat,
            method='SLSQP',
            jac=gradient_func,
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-8}
        )

        if result.success:
            energy = result.fun
            h_new = result.x
        else:

            h_new = best_h.copy()
