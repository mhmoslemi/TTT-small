from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)

    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    def make_uniform():
        return np.ones(n_points) * 0.5

    def make_triangle():
        i_vals = np.arange(n_points)
        mid = n_points // 2
        triangle = np.zeros(n_points)
        triangle[:mid] = i_vals[:mid] / (mid - 1 + 1e-10)
        triangle[mid:] = 1.0 - (i_vals[mid:] - mid) / (mid + 1e-10)
        return np.clip(triangle, 0, 1)

    candidates = [
        h_init,
        make_uniform(),
        make_triangle(),
    ]

    def generate_candidate():

        seed += 1
        rs = np.random.RandomState(seed)
        h = rs.rand(n_points)
        h = (h / h.sum()) * target_sum
        return np.clip(h, 0, 1)

    while time.time() - start_time < budget_s * 0.95:
        for cand in [generate_candidate() for _ in range(3)] + candidates:
            if time.time() - start_time >= budget_s * 0.95:
                break
            try:
                result = minimize(
                    fun=objective,
                    x0=cand,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 100, 'ftol': 1e-8}
