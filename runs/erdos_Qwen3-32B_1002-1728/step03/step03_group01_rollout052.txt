    def objective_function(h):
        # Compute C₅ for h
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = h_clipped.sum()
        if abs(current_sum) < 1e-10:
            # Fallback to uniform distribution
            return np.ones_like(h) * 0.5
        scaling_factor = target_sum / current_sum
        h_rescaled = h_clipped * scaling_factor
        # Clip again in case scaling pushed beyond [0,1]
        h_final = np.clip(h_rescaled, 0.0, 1.0)
        return h_final

    # Generate initial population
    population = []
    population.append(h_init.copy())
    rs = np.random.RandomState(seed)
    # Add random binary patterns
    for _ in range(5):

        c5 = objective_function(candidate)
        if c5 < best_c5:
            best_h = candidate
            best_c5 = c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
