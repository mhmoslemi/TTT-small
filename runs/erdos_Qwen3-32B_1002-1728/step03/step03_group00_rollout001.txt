    best_c5 = objective(best_h_full)

    current_sum = np.sum(best_h_full)
    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        best_h_full = best_h_full * (target_sum / current_sum)
        best_h_full = np.clip(best_h_full, 0, 1)
    best_c5 = objective(best_h_full)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        perturb_sigma = 0.1 if n_attempt < 50 else 0.25
        seed_offset = n_attempt // 3

        rnd = np.random.RandomState(seed + seed_offset)
        perturbed = best_h_full + rnd.normal(0, perturb_sigma, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP' if n_attempt % 2 else 'L-BFGS-B',
            bounds=bounds_full,
            constraints=cons_sum,
            options={'maxiter': 120, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            new_h_full = result.x
            new_corr = np.correlate(new_h_full, 1 - new_h_full, mode='full')
            new_c5 = np.max(new_corr) * dx
            if (abs(np.sum(new_h_full) - target_sum) < 1e-6 and
                np.all((new_h_full >= -1e-6) & (new_h_full <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h_full = new_h_full
                    best_c5 = new_c5

    assert abs(np.sum(best_h_full) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h_full >= -1e-6) & (best_h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h_full.tolist(), best_c5, n_points)
