        current_energy = compute_energy(current_h)
        step_size = 0.1
        no_improvement = 0

        while time.time() - start_time < budget_s * 0.99:
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            max_corr = np.max(corr)
            max_idx = np.argmax(corr)
            lags = np.arange(len(corr)) - (n_points - 1)
            max_lag = lags[max_idx]

            grad = np.zeros_like(current_h)
            if max_lag != 0:
                for i in range(n_points):
                    if 0 <= i + max_lag < n_points:
                        grad[i] += 1 - current_h[i + max_lag]
                    if 0 <= i - max_lag < n_points:
                        grad[i] -= current_h[i - max_lag]

            # Normalization for stability
            norm = np.linalg.norm(grad)
            if norm > 0:
                grad = grad / norm

            # Step and clip
            candidate_h = current_h - step_size * grad
            candidate_h = np.clip(candidate_h, 0, 1)
            # Renormalize
            candidate_sum = np.sum(candidate_h)
            if candidate_sum > 0:
                candidate_h *= target_sum / candidate_sum

            candidate_energy = compute_energy(candidate_h)
            if candidate_energy < current_energy:
                current_energy = candidate_energy
                current_h = candidate_h.copy()
                step_size = 0.1
                no_improvement = 0
            else:
                no_improvement += 1
                step_size *= 0.95

            if no_improvement > 20 and step_size > 1e-5:
                # Perturb and restart
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                current_h = current_h * 0.9 + perturbation * 0.1
                current_h = np.clip(current_h, 0, 1)
                current_h *= (target_sum / current_h.sum())
                step_size = 0.1

            if current_energy < best_energy:
                best_energy = current_energy
                best_h = current_h.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_energy, n_points)
