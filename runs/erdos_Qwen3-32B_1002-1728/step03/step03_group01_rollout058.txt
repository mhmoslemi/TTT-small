import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    np.random.seed(seed)
    initializations = [h_init.copy()]
    
    # Random binary initializations
    rs = np.random.RandomState(seed)
    for _ in range(10):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)
    
    # Structured initializations
    positions = np.linspace(0, 2, n_points)
    step_mid = np.zeros(n_points)
    step_mid[positions >= 1.0] = 1.0
    initializations.append(step_mid)
    
    linear_ramp = positions / 2.0
    linear_ramp_clipped = np.clip(linear_ramp, 0, 1)
    initializations.append(linear_ramp_clipped)
    
    inverse_ramp = 1.0 - linear_ramp
    inverse_ramp_clipped = np.clip(inverse_ramp, 0, 1)
    initializations.append(inverse_ramp_clipped)
    
    sin_pattern = 0.5 + 0.5 * np.sin(np.pi * (positions - 1))
    sin_pattern_clipped = np.clip(sin_pattern, 0, 1)
    initializations.append(sin_pattern_clipped)
    
    cos_pattern = 0.5 + 0.5 * np.cos(np.pi * positions)
    cos_pattern_clipped = np.clip(cos_pattern, 0, 1)
    initializations.append(cos_pattern_clipped)
    
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Optimization loop
    for idx, h0 in enumerate(initializations):
        if time.time() - start_time > budget_s * 0.9:
            break
        h0_clipped = np.clip(h0, 0, 1)
        result = minimize(
            fun=objective,
            x0=h0_clipped,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5
    
    # Local refinement
    while time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s * 0.98:
            break
        perturbed = best_h.copy()
        perturbed += (np.random.rand(n_points) - 0.5) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum < 1e-9:
            continue
        perturbed *= target_sum / perturbed_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
