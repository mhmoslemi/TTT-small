def run(seed=42, budget_s=1100, **kwargs):

    h_init = np.array(initial_h_values)
    n_points = len(h_init)

    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Use more workers
    n_workers = 10
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
        rs = np.random.RandomState(seed + w)
        # Generate different perturbations for each worker
        std_dev = 0.05 if w < 6 else 0.1  # Mix of smaller and larger perturbations
        perturbed = best_h + rs.normal(0, std_dev, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Project to feasible set
        perturbed = project(perturbed, target_sum)
        # Optimize from this point
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            # Reduce maxiter for efficiency
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final check for constraints
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx

                    best_h = h_candidate
                    best_c5 = c5_candidate

    # Final sanity checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"

def project(h, target_sum, max_iter=100, tolerance=1e-8):
    h = np.copy(h)

        current_sum = np.sum(h)
        delta = current_sum - target_sum
        if abs(delta) < tolerance:
            break

            m = np.sum(eligible)
