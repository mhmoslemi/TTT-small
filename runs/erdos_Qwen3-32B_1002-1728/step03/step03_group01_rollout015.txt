            corr = np.correlate(candidate, 1 - candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h[:] = candidate[:]
                best_c5 = c5

    # Aggressive local search with perturbations
    while time.time() - start_time < budget_s * 0.95:
        perturbed = best_h.copy()
        idxs = rs.choice(n_points, size=int(n_points * 0.1), replace=False)
        perturbed[idxs] += (rs.rand(len(idxs)) - 0.5) * 0.2
        perturbed = np.clip(perturbed, 0, 1)
        s = perturbed.sum()
        if s > 1e-9:
            perturbed *= target_sum / s
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8}
        )
        if result.success:
            candidate = result.x.reshape(n_points)
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h[:] = candidate[:]
                best_c5 = c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
