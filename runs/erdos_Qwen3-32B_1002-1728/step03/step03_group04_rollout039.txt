import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    m = int(round(target_sum))

    # Compute initial binary representation
    indices = np.argpartition(h_init, -m)[-m:]
    h_binary = np.zeros(n_points)
    h_binary[indices] = 1.0

    def compute_c5(individual):
        h_inv = 1.0 - individual
        corr = np.correlate(individual, h_inv, mode='full')
        return np.max(corr) * dx

    def select_parent(population, fitnesses, tournament_size=5):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament = [population[i] for i in indices]
        fitness_tournament = [fitnesses[i] for i in indices]
        best_idx = np.argmin(fitness_tournament)
        return tournament[best_idx]

    def repair(individual, m):
        current_sum = int(np.sum(individual))
        if current_sum == m:
            return individual
        elif current_sum > m:
            ones = np.where(individual == 1)[0]
            np.random.shuffle(ones)
            excess = current_sum - m
            individual[ones[:excess]] = 0
        else:
            zeros = np.where(individual == 0)[0]
            np.random.shuffle(zeros)
            deficit = m - current_sum
            individual[zeros[:deficit]] = 1
        return individual

    def mutate(individual, n_swaps):
        idxs = np.random.choice(n_points, size=2 * n_swaps, replace=False)
        for j in range(n_swaps):
            a, b = idxs[2 * j], idxs[2 * j + 1]
            if individual[a] == 1 and individual[b] == 0:
                individual[a], individual[b] = 0, 1
            elif individual[a] == 0 and individual[b] == 1:
                individual[a], individual[b] = 1, 0
        return individual

    # Initialize population
    pop_size = 40
    population = [h_binary.copy()]
    for _ in range(pop_size - 1):
        new_ind = h_binary.copy()
        swaps = int(0.05 * n_points)
        if swaps < 1:
            swaps = 1
        mutate(new_ind, swaps)
        population.append(new_ind)

    # GA parameters
    max_generations = 500
    pop_best = h_binary.copy()
    best_c5 = compute_c5(pop_best)

    # Begin GA optimization
    generation = 0
    while time.time() - start_time < budget_s * 0.85 and generation < max_generations:
        # Evaluate current population
        fitnesses = [compute_c5(ind) for ind in population]
        current_best_idx = np.argmin(fitnesses)
        current_c5 = fitnesses[current_best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            pop_best = population[current_best_idx].copy()

        # Generate new population
        new_population = []
        while len(new_population) < pop_size:
            # Parent selection (tournament)
            parent1 = select_parent(population, fitnesses)
            parent2 = select_parent(population, fitnesses)

            # Crossover
            mask = np.random.rand(n_points) < 0.5
            child = np.zeros(n_points)
            child[mask] = parent1[mask]
            child[~mask] = parent2[~mask]
            child = repair(child, m)

            # Mutation
            swaps = int(0.05 * n_points)
            if swaps < 1:
                swaps = 1
            mutate(child, swaps)
            new_population.append(child)

        population = new_population
        generation += 1

    # Local optimization of best solution
    h_cont = pop_best.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    result = minimize(
        fun=compute_c5,
        x0=h_cont,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    final_c5 = compute_c5(best_h)

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), final_c5, n_points)
