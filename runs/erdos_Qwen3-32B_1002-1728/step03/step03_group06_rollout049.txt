            break
        if m % 2 != 0:
            m += 1
        dx = 2.0 / m
        # Precompute overlap matrix for all shifts k
        A = []
        for k in range(1, m):
            shift = k * dx
            a_km = np.zeros((m, m))
            for i in range(m):
                s_i = i * dx
                e_i = (i + 1) * dx
                for j in range(m):
                    s_j = j * dx
                    x_start = s_j - shift
                    x_end = (j + 1) * dx - shift
                    x_overlap_min = max(s_i, x_start)
                    x_overlap_max = min(e_i, x_end)
                    if x_overlap_min < x_overlap_max:
                        a_km[i, j] = x_overlap_max - x_overlap_min
            A.append(a_km)
        if not A:
            continue

        # Optimization variables
        b = cp.Variable(m, boolean=True)
        t = cp.Variable()
        z = cp.Variable((m, m), boolean=True)
        constraints = [cp.sum(b) == m / 2]

                    z[i, j] <= b[i],

                ]

            constraints += [t >= expr]
        prob = cp.Problem(cp.Minimize(t), constraints)
