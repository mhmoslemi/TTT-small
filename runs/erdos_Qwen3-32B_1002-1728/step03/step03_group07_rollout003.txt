import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair_h(h_flat):
        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if abs(current_sum) < 1e-10:
            return np.zeros_like(h_flat)
        scale_factor = target_sum / current_sum
        h_scaled = h_clipped * scale_factor
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        return h_scaled

    def objective(h_flat):
        h_repaired = repair_h(h_flat).reshape(n_points)
        cross_corr = np.correlate(h_repared, 1 - h_repaired, mode='full')
        return np.max(cross_corr) * dx

    def gradient(h_flat):
        h_repaired = repair_h(h_flat).reshape(n_points)
        cross_corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        k_index = np.argmax(cross_corr)
        m = k_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:
                grad[i] += (1.0 - h_repaired[j]) * dx
                grad[j] -= h_repaired[i] * dx
        return grad.flatten()

    best_h = None
    best_energy = float('inf')
    seeds = [42, 0, 1, 2, 3]

        h_pert = repair_h(h_pert)
