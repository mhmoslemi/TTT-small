import time
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_energy = float('inf')
    best_h = h_init.copy()

    np.random.seed(seed)
    for _ in range(5):  # Multi-start with perturbations
        # Generate perturbed initial guess
        if _ == 0:
            h0 = h_init.copy()
        else:
            perturbation = np.random.rand(n_points) * 0.2
            h0 = h_init + perturbation
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (target_sum / h0.sum())

        bounds = Bounds(0.0, 1.0)
        constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def gradient(h):
            corr = correlate(h, 1 - h, mode='full')
            kstar = np.argmax(corr)
            s = kstar - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
                    grad[j] += -h[i]
            return grad * dx

        # Rescale initial guess to ensure constraint
        h0 = h0 * (target_sum / np.sum(h0))

        # Optimization with SLSQP
        res = minimize(
            objective,
            h0,
            method='SLSQP',
            jac=gradient,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        current_h = np.clip(res.x, 0, 1)
        current_h = current_h * (target_sum / np.sum(current_h))
        current_energy = objective(current_h)

        if current_energy < best_energy:
            best_energy = current_energy
            best_h = current_h

        if time.time() - start_time > budget_s * 0.99:
            break

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_energy, n_points)
