    h_adjusted = np.clip(h_adjusted, 0.0, 1.0)
    current_sum = np.sum(h_adjusted)
    iterations = 0
    while abs(current_sum - target_sum) > 1e-5 and iterations < 10:
        delta = (target_sum - current_sum) / n_points
        h_adjusted += delta
        h_adjusted = np.clip(h_adjusted, 0.0, 1.0)
        current_sum = np.sum(h_adjusted)
        iterations += 1
    return h_adjusted

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def surrogate_objective(h_vec, alpha=0.1):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        weights = np.array([np.exp(-alpha * abs((j - n_points + 1) * dx)) for j in range(len(corr))])
        return np.sum(corr * weights) * dx

    def objective_max(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def binary_h(h_vec):
        h = h_vec.copy()
        n_ones = int(target_sum)
        h_sorted = np.argsort(h)
        h_binary = np.zeros_like(h)
        h_binary[h_sorted[-n_ones:]] = 1.0
        if n_ones < target_sum:
            h_binary[h_sorted[n_ones]] += target_sum - n_ones
        return h_binary

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    structured_initials = []
    for period in [1.5, 2.0, 2.5, 3.0]:
        for threshold in [0.4, 0.5, 0.6]:
            h_candidate = generate_periodic_h(n_points, period, threshold)
            structured_initials.append(h_candidate)
    structured_initials.append(h_init)

    best_h = h_init.copy()
    best_c5 = objective_max(best_h)

    rs = np.random.RandomState(seed)
    for initial in structured_initials:
        if time.time() - start_time > budget_s:
            break
        perturbed = initial + rs.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = enforce_sum_constraint(perturbed, target_sum)

        result_surrogate = minimize(
            fun=surrogate_objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result_surrogate.success:
            h_candidate = result_surrogate.x.reshape(n_points)
            result_max = minimize(
                fun=objective_max,
                x0=h_candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if result_max.success:
                h_candidate = result_max.x.reshape(n_points)
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h = h_candidate
                    best_c5 = c5
                    h_bin = binary_h(h_candidate)
                    result_binary = minimize(
                        fun=objective_max,
                        x0=h_bin,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 30, 'ftol': 1e-6}
                    )
                    if result_binary.success:
                        corr_bin = np.correlate(result_binary.x, 1 - result_binary.x, mode='full')
                        c5_bin = np.max(corr_bin) * dx
                        if c5_bin < best_c5:
                            best_h = result_binary.x
                            best_c5 = c5_bin

    if time.time() - start_time < budget_s * 0.95:
        for _ in range(10):
            perturbed = best_h + rs.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = enforce_sum_constraint(perturbed, target_sum)
            if np.allclose(np.sum(perturbed), target_sum):
                result = minimize(
                    fun=objective_max,
                    x0=perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 40}
                )
                if result.success:
                    h_candidate = result.x.reshape(n_points)
                    corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                    c5 = np.max(corr) * dx
                    if c5 < best_c5:
                        best_h = h_candidate
                        best_c5 = c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
