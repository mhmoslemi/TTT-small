    def objective(h_vec):

        corr = np.correlate(h, 1 - h, mode='full')

    def project(h, target_sum):

        if abs(s) > 1e-9:
            h = h * (target_sum / s)
        return h

    # Hybrid initialization: structured and random perturbations
