import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    N_particles = 100
    max_iter = 500
    inertia_initial = 0.7
    inertia_final = 0.4
    cognitive_initial = 1.2
    social_initial = 1.2
    velocity_clamp = 0.1
    eps = 1e-6

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    particles = [h_init.copy()]
    velocities = [np.zeros(n_points)]
    pbest_pos = [h_init.copy()]
    pbest_fit = [compute_c5(h_init)]
    gbest_pos = h_init.copy()
    gbest_fit = pbest_fit[0]

    # Initialize remaining particles
    np.random.seed(seed)

            s = rand_h.sum()
            if abs(s) < eps:
                continue
            scaled = rand_h * (target_sum / s)
            clamped = np.clip(scaled, 0, 1)
            if abs(clamped.sum() - target_sum) < 1e-3:
                break
        particles.append(clamped)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        fit = compute_c5(clamped)

    iteration = 0

        inertia = inertia_initial - (inertia_initial - inertia_final) * iteration / max_iter
        cognitive = max(0.1, cognitive_initial - 0.01 * iteration)

            if abs(s) > eps:

            new_pos = np.clip(new_pos, 0, 1)
