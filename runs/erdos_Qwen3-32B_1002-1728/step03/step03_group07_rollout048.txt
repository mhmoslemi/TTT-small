import time
import numpy as np

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    sum_h = np.sum(h_clipped)
    if abs(sum_h - target_sum) < 1e-6:
        return h_clipped
    # Scale to target_sum, while keeping h in [0,1]
    # But scaling may not work if some values are already at bounds
    # So adjust the scaling to distribute the excess/deficit
    # For now, a simple approach:
    h_scaled = h_clipped * (target_sum / sum_h)
    return np.clip(h_scaled, 0, 1)

def compute_energy(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    target_k = idx_max - (n_points - 1)
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + target_k
        if 0 <= j < n_points:
            grad[i] = -(1 - h[j])
        else:
            grad[i] = 0.0
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)

    # Generate initial directions and candidates
    directions = []
    for _ in range(10):  # Number of directions to generate
        i, j = np.random.choice(n_points, size=2, replace=False)
        # Create a direction vector that has +1 at i, -1 at j
        dir_vec = np.zeros(n_points)
        dir_vec[i] = 1
        dir_vec[j] = -1
        dir_vec /= np.sqrt(2)  # Normalize the direction
        directions.append(dir_vec)

    initial_candidates = []
    for dir in directions:
        # Perturb both ways
        alpha = 0.05  # Small step size
        candidate_plus = project(h_init + alpha * dir, target_sum)
        candidate_minus = project(h_init - alpha * dir, target_sum)
        initial_candidates.append(candidate_plus)
        initial_candidates.append(candidate_minus)
    initial_candidates.append(h_init.copy())  # Original as a candidate

    best_h = h_init.copy()
    best_energy = compute_energy(best_h, dx)

    # For each candidate, perform local optimization
    for candidate in initial_candidates:
        current_h = candidate.copy()
        current_energy = compute_energy(current_h, dx)
        for _ in range(50):  # 50 steps of gradient descent
            if time.time() - start_time > budget_s - 10:  # leave some buffer
                break
            grad = compute_gradient(current_h, n_points)
            # Normalize gradient
            norm = np.sqrt(np.dot(grad, grad))
            if norm == 0:

            new_h = current_h - step_size * grad_unit

            # Compute new energy
            new_energy = compute_energy(new_h, dx)
            # Accept if better
            if new_energy < current_energy:
                current_h = new_h
                current_energy = new_energy
            # Else, reduce step size? Or continue
        if current_energy < best_energy:
            best_h = current_h.copy()
            best_energy = current_energy

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"

import time
import numpy as np

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    sum_h = np.sum(h_clipped)
    if abs(sum_h - target_sum) < 1e-6:
        return h_clipped
    scale = target_sum / sum_h
    h_scaled = h_clipped * scale
    return np.clip(h_scaled, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)

    # Generate initial directions and candidates
    directions = []
    for _ in range(15):  # Generate 15 directions
        i, j = np.random.choice(n_points, size=2, replace=False)
        dir_vec = np.zeros(n_points)
        dir_vec[i] = 1.0
        dir_vec[j] = -1.0
        dir_vec /= np.linalg.norm(dir_vec)
        directions.append(dir_vec)

    initial_candidates = [h_init.copy()]
    alpha = 0.03  # Perturbation step
    for dir in directions:
        candidate_plus = project(h_init + alpha * dir, target_sum)
        candidate_minus = project(h_init - alpha * dir, target_sum)
        initial_candidates.append(candidate_plus)
        initial_candidates.append(candidate_minus)

    best_h = h_init.copy()
    best_energy = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Local optimization for each candidate
    for candidate in initial_candidates:
        current_h = candidate.copy()
        current_best = current_h.copy()
        energy = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        for _ in range(100):  # 100 local steps
            if time.time() - start_time > budget_s - 20:  # Early exit
                break
            # Compute gradient at current_h
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            idx_max = np.argmax(corr)
            k = idx_max - (n_points - 1)  # Relative shift (lag)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] = -(1 - current_h[j])
                else:
                    grad[i] = 0.0
            if np.linalg.norm(grad) < 1e-8:
                continue

            # Normalize gradient and take step
            grad_unit = grad / np.linalg.norm(grad)
            step_size = 0.02  # Fixed step
            new_h = current_h - step_size * grad_unit
            new_h = project(new_h, target_sum)

            # Compute new energy
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_energy = np.max(new_corr) * dx

            # Accept if improved
            if new_energy < energy:
                current_h = new_h
                energy = new_energy
                if energy < np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx:
                    best_h = current_h.copy()
                    best_energy = energy

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
