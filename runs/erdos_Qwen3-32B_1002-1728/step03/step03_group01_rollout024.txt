        if result.success:
            current_best = project_to_feasible(result.x)
            c5_val = objective(current_best)
            if c5_val < best_c5:
                best_h[:] = current_best[:]
                best_c5 = c5_val

    # Final validation
    best_h = project_to_feasible(best_h)
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds error"
    return (best_h.tolist(), best_c5, n_points)
