                    term2 = -current_h[i - k]
                grad[i] = term1 + term2
            grad *= dx  # Scale by dx
            
            # Line search with backtracking
            for _ in range(5):
                candidate = current_h - step_size * grad
                candidate = np.clip(candidate, 0, 1)
                current_sum = candidate.sum()
                if current_sum > 1e-6:
                    candidate *= target_sum / current_sum
                candidate = np.clip(candidate, 0, 1)
                energy = compute_energy(candidate)
                if energy < current_energy:
                    current_energy = energy
                    current_h = candidate
                    break
                step_size *= 0.5
            
            # Periodic restart with perturbation
            if iter_count % 100 == 0:
                if rs.rand() < 0.1:
                    perturbation = rs.uniform(-0.05, 0.05, size=n_points)
                    current_h += perturbation
                    current_h = np.clip(current_h, 0, 1)
                    current_sum = current_h.sum()
                    if current_sum > 1e-6:
                        current_h *= target_sum / current_sum
                    current_h = np.clip(current_h, 0, 1)
                    current_energy = compute_energy(current_h)
            
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = current_h.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
