from scipy.optimize import minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, c5_list, tournament_size=5):
    contestants = np.random.choice(len(population), size=tournament_size, replace=False)
    c5s = [c5_list[i] for i in contestants]
    winner_idx = contestants[np.argmin(c5s)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, n_points):
    l = np.random.uniform(0.2, 0.5) * n_points
    l = max(1, int(l))
    start = np.random.randint(0, n_points - l)
    end = start + l
    child = parent1.copy()
    child[start:end] = parent2[start:end]
    return child

def mutate(individual, n_points, target_sum):
    if np.random.rand() < 0.1:
        num_mutate = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=num_mutate, replace=False)
        for idx in indices:
            if np.random.rand() < 0.5:
                individual[idx] += np.random.uniform(-0.1, 0.1)
            else:
                individual[idx] = 1.0 - individual[idx]
            individual[idx] = np.clip(individual[idx], 0.0, 1.0)
    s = np.sum(individual)
    if s != 0.0:
        individual = individual * (target_sum / s)
    individual = np.clip(individual, 0.0, 1.0)
    return individual
