import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Genetic Algorithm parameters
    population_size = 50
    num_generations = 80
    mutation_rate = 0.1
    elitism_size = 10

    # Initialize population
    rs = np.random.RandomState(seed)
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = rs.rand(n_points)
        h = h * (target_sum / h.sum()) if h.sum() > 1e-9 else 0.5
        h = np.clip(h, 0, 1)
        population.append(h)
    population = np.array(population)

    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Genetic Algorithm
    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.9:
            break

        # Evaluate fitness
        fitness = np.array([calc_c5(population[i]) for i in range(population_size)])
        indices = np.argsort(fitness)

        # Elite preservation
        elite = population[indices[:elitism_size]]

        # Selection via tournament
        selected = []
        for _ in range(population_size - elitism_size):
            candidates = rs.choice(population_size, 5)
            selected.append(population[indices[np.argmin(fitness[candidates])]])

        # Crossover and mutation
        offspring = []
        for _ in range(population_size - elitism_size):
            p1, p2 = rs.choice(selected, 2)
            child = p1.copy()
            # Segment crossover
            start = rs.randint(0, n_points)
            length = rs.randint(1, n_points - start)
            child[start:start+length] = p2[start:start+length]
            # Mutation
            if rs.rand() < mutation_rate:
                mutate_at = rs.choice(n_points, size=int(n_points * 0.15), replace=False)
                for i in mutate_at:
                    if rs.rand() < 0.5:
                        child[i] += rs.uniform(-0.1, 0.1)
                    else:
                        child[i] = 1 - child[i]
            # Repair constraints
            child = np.clip(child, 0, 1)
            s = child.sum()
            if s > 1e-9:
                child *= (target_sum / s)
            else:
                child[:] = 0.5
            offspring.append(child)

        population = np.concatenate((elite, np.array(offspring)))

    # Local optimization
    best_h = population[np.argmin([calc_c5(h) for h in population])]
    result = minimize(
        lambda h: calc_c5(h.reshape(n_points)),
        x0=best_h,
        bounds=[(0, 1)] * n_points,
        constraints={'type': 'eq', 'fun': lambda h: h.sum() - target_sum},
        method='SLSQP',
        options={'maxiter': 200}
    )
    if result.success:
        final_h = np.clip(result.x.reshape(n_points), 0, 1)
        s = final_h.sum()
        if s > 1e-9:
            final_h *= (target_sum / s)
    else:
        final_h = best_h

    assert abs(final_h.sum() - target_sum) < 1e-6
    assert np.all((final_h >= -1e-6) & (final_h <= 1.0 + 1e-6))
    return (final_h.tolist(), calc_c5(final_h), n_points)
