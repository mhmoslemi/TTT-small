    n_points = len(h_init)

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Global exploration with DE
    result_de = differential_evolution(
        func=objective_de,
        bounds=bounds,
        popsize=5,  # Reduced population size for time efficiency

        maxiter=20,

        seed=seed,

        workers=1,

            best_c5 = 1.0
        else:
            h_best = h_best * (target_sum / s)

    h_candidate = np.clip(h_candidate, 0, 1)
