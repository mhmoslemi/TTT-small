import time

import cvxpy as cp

def run():
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    budget_seconds = 1100

    # Define the maximum number of iterations and epsilon for LP
    max_iterations = 500
    epsilon = 0.1

    # Precompute 1 - h for possible reuse
    for _ in range(max_iterations):
        # Compute the overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_value = np.max(corr)
        max_index = np.argmax(corr)
        k_steps = max_index - (n_points - 1)
        
        # Compute gradient vector g
        g = np.array([1.0 if (i + k_steps < 0 or i + k_steps >= n_points) else 1.0 - h[i + k_steps] for i in range(n_points)])

        # Formulate LP problem
        delta_h = cp.Variable(n_points)
        constraints = [
            cp.sum(delta_h) == 0,
            delta_h <= epsilon,
            delta_h >= -epsilon
        ]
        objective = cp.Minimize(np.dot(g, delta_h))
        problem = cp.Problem(objective, constraints)
        
        # Solve the problem
        problem.solve(solver=cp.CVXOPT)
        
        if problem.status != cp.OPTIMAL:
            break  # Exit if LP not solved

        # Update h
        h += delta_h.value
        h = np.clip(h, 0.0, 1.0)
        
        # Renormalize
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6:
            adjust = (target_sum - current_sum) / n_points
            h += adjust
            h = np.clip(h, 0.0, 1.0)
        
        # Check if time is up
        if time.time() - start_time > budget_seconds * 0.99:
            break

    # Compute final C5
    corr = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx

    assert abs(np.sum(h) - target_sum) < 1e-6
    return (h.tolist(), c5_bound, n_points)
