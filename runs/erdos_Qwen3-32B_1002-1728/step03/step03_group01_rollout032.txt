import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initializations
    initializations = [h_init.copy()]
    rs = np.random.RandomState(seed)
    for _ in range(10):  # Increased from 5 to 10 for more exploration
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)
    
    best_c5 = float('inf')
    best_h = h_init.copy()

    # Evaluate initializations with SLSQP
    for idx, h0 in enumerate(initializations):
        if time.time() - start_time > budget_s * 0.85:
            break
        result = minimize(
            fun=objective,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1-h_candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5

    # Local search refinement with adaptive perturbation
    while time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s * 0.98:
            break
        # Vary perturbation intensity
        intensity = max(0.05, (budget_s - time.time() + start_time)/1100 * 0.2)
        perturbed = best_h.copy()
        noise = (np.random.rand(n_points) - 0.5) * intensity
        perturbed += noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-9:
            perturbed *= target_sum / perturbed_sum
        result = minimize(

            x0=perturbed,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',

        if result.success:
