from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum of first half to satisfy full sum = 1

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective_de(x):
        x_total = np.sum(x)
        if x_total <= 1e-10:
            return float('inf')
        x_scaled = x * (target_sum_half / x_total)
        x_clipped = np.clip(x_scaled, 0, 1)
        h_full = create_full_h(x_clipped)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective_sl(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Symmetric initialization
    h_full_sym = h_full_initial.copy()
    h_full_sym[:half_n] = h_full_sym[-half_n:] = (h_full_initial[:half_n] + h_full_initial[-half_n:])/2
    h_full_sym *= n_points / 2 / np.sum(h_full_sym)
    best_h_half = np.clip(h_full_sym[:half_n], 0, 1)
    best_c5 = objective_sl(best_h_half)

    # Differential evolution phase
    if time.time() - start_time < budget_s - 120 and n_points > 20:
        try:
            de_result = differential_evolution(
                objective_de, 
                bounds_per_var,
                strategy='best1bin',
                maxiter=30,
                popsize=10,
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.8,
                polish=False,
                workers=1
            )
            if de_result.success:
                de_best_half = de_result.x
                total = np.sum(de_best_half)
                if total > 1e-10:
                    de_best_half = np.clip(de_best_half * (target_sum_half/total), 0, 1)
                    sl_result = minimize(
                        objective_sl, de_best_half,
                        method='SLSQP', bounds=bounds_per_var,
                        constraints=cons, options={'maxiter': 150, 'ftol': 1e-9}
                    )
                    if sl_result.success:
                        sl_h_half = sl_result.x
                        sl_c5 = objective_sl(sl_h_half)
                        if sl_c5 < best_c5:
                            best_h_half = sl_h_half
                            best_c5 = sl_c5
        except Exception as e:
            pass

    np.random.seed(seed)
    n_attempt = 0
    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        perturbed = best_h_half + np.random.normal(0, 0.12 * np.ones(half_n), half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total > 1e-10:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            objective_sl, perturbed,
            method='SLSQP', bounds=bounds_per_var,
            constraints=cons, options={'maxiter': 150, 'ftol': 1e-9}
        )

        if result.success:
            new_h_half = result.x
            new_c5 = objective_sl(new_h_half)
            if abs(np.sum(new_h_half) - target_sum_half) < 1e-6 and np.all((new_h_half >= -1e-6) & (new_h_half <= 1.0 + 1e-6)):
                if new_c5 < best_c5:
                    best_h_half = new_h_half
                    best_c5 = new_c5

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
