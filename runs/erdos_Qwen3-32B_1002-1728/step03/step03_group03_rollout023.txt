def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

    iter_count = 0
    max_iter = 1000  # a large enough number, but time is the limiting factor

    while time.time() - start_time < budget_s:
        iter_count +=1
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        current_c5 = max_overlap * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Find the index of maximum overlap
        max_idx = np.argmax(corr)
        # Compute lag k_max in sample steps
        k_lag = max_idx - (n_points -1)

        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_lag
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
            else:
                grad[i] = 0.0

        # Solve the LP
        delta_h = cp.Variable(n_points)
        objective = cp.Minimize(grad @ delta_h)
        constraints = [
            cp.sum(delta_h) == 0,
            delta_h >= -h,  # so h + delta_h >= 0
            delta_h <= 1 - h  # h + delta_h <=1
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status == cp.OPTIMAL:
            h = h + delta_h.value
            h = np.clip(h, 0.0, 1.0)  # in case of numerical issues
        else:
            # If LP not solved, perhaps perturb and try again or break
            pass

        # Early stopping if no improvement
        if iter_count % 10 == 0:
            # Re-evaluate overlaps
            corr_check = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(corr_check) * dx
            if abs(new_c5 - best_c5) < 1e-6:
                break

    # Final check
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return best_h.tolist(), final_c5, n_points

import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    while time.time() - start_time < budget_s:
        # Compute current overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Find the worst lag
        max_idx = np.argmax(corr)
        k_lag = max_idx - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_lag
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
            else:
                grad[i] = 0.0

        # Set up LP
        delta_h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(grad, delta_h)))

            delta_h >= -h,
            delta_h <= 1 - h

        problem = cp.Problem(objective, constraints)

        # Solve
        try:
            problem.solve(solver=cp.GLPK, verbose=False)
            if problem.status == cp.OPTIMAL:
                h = h + delta_h.value
                h = np.clip(h, 0.0, 1.0)
            # else, do nothing
        except:
            pass  # solver error, skip this iteration
