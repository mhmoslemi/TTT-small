import time
import numpy as np
from scipy.optimize import differential_evolution, minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair_h(h_vec):
        h_clipped = np.clip(h_vec, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum <= 1e-12:
            return np.zeros_like(h_clipped)
        scale = target_sum / h_sum
        return np.clip(h_clipped * scale, 0.0, 1.0)

    def objective(h_vec):
        h_repaired = repair_h(h_vec)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full') * dx
        return np.max(corr)

    def objective_de(h_vec):
        return objective(h_vec)

    bounds = Bounds(0.0, 1.0)
    x0 = h.copy()

    # Differential Evolution with time limit
    de_time_limit = budget_s * 0.6
    de_start = time.time()
    result_de = differential_evolution(
        objective_de, 
        [(0.0, 1.0)] * n_points, 
        maxiter=50, 
        popsize=5,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: time.time() - start > de_time_limit,
        polish=False,
        tol=1e-6
    )
    x_de = result_de.x
    h_de = repair_h(x_de)

    # Local optimization
    def objective_local(x):
        return objective(x)

    bounds_local = Bounds(0.0, 1.0)
    result_local = minimize(
        objective_local, 
        x0=h_de,
        method='L-BFGS-B',
        bounds=bounds_local,
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if not result_local.success:
        result_local = minimize(
            objective_local, 
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds_local,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
    best_h = repair_h(result_local.x)
    best_c5 = objective(best_h)

    # Final repair
    h_best = best_h
    assert abs(np.sum(h_best) - target_sum) < 1e-6
    return (h_best.tolist(), best_c5, n_points)
