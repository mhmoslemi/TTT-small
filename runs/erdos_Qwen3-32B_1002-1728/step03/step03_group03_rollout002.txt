import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = initial_h_values
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    n_particles = 50
    max_iter = 300
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0

    np.random.seed(seed)
    particles = np.tile(h_initial, (n_particles, 1)).copy()
    # Initial perturbation
    noise = 0.1 * np.random.normal(size=(n_particles, n_points))
    particles += noise
    # Clip and scale to constraints

    row_sums = np.sum(particles_clipped, axis=1, keepdims=True)
    particles = (particles_clipped * (target_sum / np.where(row_sums == 0, 1, row_sums))).astype(np.float64)

    velocities = 0.01 * np.random.normal(size=(n_particles, n_points))

    def evaluate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Initialize personal and global bests
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([evaluate(p) for p in particles])
    valid_indices = np.isfinite(personal_best_scores)
    personal_best_scores[~valid_indices] = float('inf')
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = particles[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        time_remaining = budget_s - (time.time() - start_time)
        fraction = max(0, (time_remaining / budget_s))
        w = w_max - (w_max - w_min) * (i / max_iter)

        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        velocities = w * velocities + c1 * r1 * (personal_best_positions - particles) + c2 * r2 * (global_best_position - particles)
        particles += velocities
        particles_clipped = np.clip(particles, 0.0, 1.0)
        row_sums = np.sum(particles_clipped, axis=1, keepdims=True)
        particles = np.where(row_sums[:, :, None] == 0, 0.5, (particles_clipped * (target_sum / row_sums)))

        current_scores = np.array([evaluate(p) for p in particles])
        improved = current_scores < personal_best_scores
        personal_best_positions[improved] = particles[improved]
        personal_best_scores[improved] = current_scores[improved]
        current_global_min = np.min(personal_best_scores)
        if current_global_min < global_best_score:

            global_best_score = current_global_min
