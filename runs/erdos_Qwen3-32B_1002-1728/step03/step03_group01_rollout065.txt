import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.copy().reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initial solutions
    initializations = [h_init.copy()]
    rs = np.random.RandomState(seed)
    for _ in range(14):  # 14 additional random patterns
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = h_init.copy()
    best_energy = objective(best_h)

    # Simulated Annealing parameters
    T_initial = 100.0
    alpha = 0.9995
    T_min = 1e-5
    cool_interval = 50  # Number of iterations per temperature

    for start_h in initializations:
        if time.time() - start_time > budget_s * 0.9:
            break
        current_h = start_h.copy()
        current_energy = objective(current_h)
        if current_energy < best_energy:
            best_h[:] = current_h[:]
            best_energy = current_energy

        temperature = T_initial
        while temperature > T_min and time.time() - start_time < budget_s * 0.9:
            for _ in range(cool_interval):
                perturbed = current_h + 0.3 * temperature * rs.randn(n_points)
                perturbed = np.clip(perturbed, 0, 1)
                sum_perturbed = np.sum(perturbed)
                if sum_perturbed > 1e-9:
                    perturbed *= target_sum / sum_perturbed

                new_energy = objective(perturbed)
                delta = new_energy - current_energy

                if (delta < 0 or np.exp(-delta / temperature) > rs.rand()):
                    current_h = perturbed.copy()
                    current_energy = new_energy
                    if current_energy < best_energy:
                        best_h[:] = current_h[:]
                        best_energy = current_energy

            temperature *= alpha

    # Final refinement with SLSQP on best candidate
    if time.time() - start_time < budget_s * 0.95:
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x.reshape(n_points)
            final_energy = objective(final_h)
            if final_energy < best_energy:
                best_h[:] = final_h[:]

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_energy, n_points)
