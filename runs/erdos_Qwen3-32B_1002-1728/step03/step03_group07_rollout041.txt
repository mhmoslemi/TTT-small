import numpy as np

def compute_energy_and_gradient(h, dx, n_points):
    current_corr = np.correlate(h, 1 - h, mode='full')
    max_corr_val = np.max(current_corr)
    max_corr_idx = np.argmax(current_corr)
    k_star = max_corr_idx - (n_points - 1)
    idxs = np.arange(n_points)
    shifted1 = idxs + k_star
    valid1 = (shifted1 >= 0) & (shifted1 < n_points)
    term1 = np.where(valid1, 1.0 - h[shifted1], 0.0)

    term2 = np.where(valid2, h[shifted2], 0.0)
    grad = (term1 - term2) * dx
