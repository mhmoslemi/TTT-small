import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # MILP formulation
    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    constraints = [cp.sum(h) == target_sum]
    
    # Iterate over all possible shifts
    for k in range(-n_points + 1, n_points):
        start = max(0, -k)
        end = min(n_points - 1, n_points - 1 - k)
        if start > end:
            continue
        valid_indices = np.arange(start, end + 1)
        shifted_indices = valid_indices + k
        lhs = cp.sum(h[valid_indices] * (1 - h[shifted_indices]))
        constraints.append(lhs * dx <= t)
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Use GLPK_MI with time limit
    problem.solve(solver='GLPK_MI', verbose=False, time_limit=budget_s - 10)
    
    best_h = h.value
    best_c5 = t.value
    
    # Ensure feasibility and round if necessary
    best_h = np.round(best_h).astype(float)
    if abs(np.sum(best_h) - target_sum) > 1e-6:
        # Rebalance if rounding caused sum mismatch (shouldn't happen with MILP)
        indices = np.argsort(best_h)
        best_h[indices[:int(target_sum)]] = 1.0
        best_h[indices[int(target_sum):]] = 0.0
    
    # Final check of constraints
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all(np.array([0.0, 1.0]) >= best_h) and np.all(best_h >= np.array([0.0, 1.0])), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
