    if not np.allclose(np.sum(h_de_candidate), target_sum):
        h_slsqp_initial = h_init.copy()
    else:
        h_slsqp_initial = h_de_candidate

    result_slsqp = minimize(objective_slsqp, 
                            x0=h_slsqp_initial,
                            method='SLSQP',
                            bounds=bounds_slsqp,
                            constraints=cons_slsqp,
                            tol=1e-8,
                            options={'maxiter': 300, 'ftol': 1e-8})

    best_h = h_slsqp_initial if not result_slsqp.success else result_slsqp.x
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr_best) * dx

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
