    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    m = 10  # Hidden units in neural network
    param_count = 3 * m + 1

    # Seed the random number generator for reproducibility
    np.random.seed(seed)
    # Small initial parameters to avoid saturation in sigmoid
    initial_params = np.random.randn(param_count) * 0.05

    def objective(params):
        h = compute_h(params, n_points, m)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(params):
        h = compute_h(params, n_points, m)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(-1e4, 1e4) for _ in range(param_count)]  # Loose param bounds

    # Optimize using gradient-based method
    result = minimize(
        fun=objective,
        x0=initial_params,
        method='SLSQP',
        constraints=cons,
        bounds=bounds,
        options={'maxiter': 300, 'ftol': 1e-6}
    )

    h_opt = compute_h(result.x, n_points, m)

    # Correct any small deviations from sum constraint
    sum_h = np.sum(h_opt)
    if not np.isclose(sum_h, target_sum, rtol=1e-5):
        correction = (target_sum - sum_h) / n_points
        h_opt += correction
    h_opt = np.clip(h_opt, 0.0, 1.0)

    # Compute final C₅ bound
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx

    # Final validation
    assert np.isclose(sum(h_opt), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((h_opt >= 0) & (h_opt <= 1)), "Value bounds violated"

    return (h_opt.tolist(), c5_bound, n_points)
