import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    pop_size = 30
    w = 0.7
    c1 = 0.6
    c2 = 0.3
    max_velocity = 0.5

    # Initialize particles
    particles = [h_init.copy()]
    velocities = [(np.random.rand(n_points) - 0.5) * max_velocity]
    # Evaluate initial
    corr = np.correlate(particles[0], 1 - particles[0], mode='full')
    pbest_values = [np.max(corr) * dx]
    pbest_positions = [particles[0].copy()]

    rs = np.random.RandomState(seed)
    for _ in range(pop_size - 1):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        particles.append(h)
        velocities.append((rs.rand(n_points) - 0.5) * max_velocity)
        corr = np.correlate(h, 1 - h, mode='full')
        pbest_values.append(np.max(corr) * dx)
        pbest_positions.append(h.copy())

    gbest_value = min(pbest_values)
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()

    # PSO Iteration Loop
    while time.time() - start_time < budget_s * 0.9:
        for i in range(pop_size):
            current_pos = particles[i].copy()
            current_vel = velocities[i].copy()

            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)

            vel_cognitive = c1 * r1 * (pbest_positions[i] - current_pos)
            vel_social = c2 * r2 * (gbest_position - current_pos)
            new_vel = w * current_vel + vel_cognitive + vel_social
            new_vel = np.clip(new_vel, -max_velocity, max_velocity)
            velocities[i] = new_vel

            new_pos = current_pos + new_vel
            new_pos = np.clip(new_pos, 0, 1)
            s = np.sum(new_pos)
            if s > 1e-8:
                new_pos *= target_sum / s
            else:
                new_pos = particles[i].copy()
            particles[i] = new_pos

            corr = np.correlate(new_pos, 1 - new_pos, mode='full')
            current_value = np.max(corr) * dx
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = new_pos.copy()

        # Update global best
        for i in range(pop_size):
            if pbest_values[i] < gbest_value:
                gbest_value = pbest_values[i]
                gbest_position = pbest_positions[i].copy()
        if time.time() - start_time > budget_s * 0.9:
            break

    # Prepare candidates for refinement
    all_cands = list(zip(pbest_values, pbest_positions))
    all_cands.sort()
    top_cands = [x[1] for x in all_cands[:5]]

    best_h = gbest_position.copy()
    best_c5 = gbest_value

    # SLSQP refinement
    bounds_min = np.zeros(n_points)
    bounds_max = np.ones(n_points)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    for h0 in top_cands:
        if time.time() - start_time > budget_s * 0.95:
            break
        result = minimize(
            fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            x0=h0,
            bounds=list(zip(bounds_min, bounds_max)),
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_candidate
                best_c5 = c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
