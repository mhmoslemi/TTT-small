import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Initialize with multiple diverse starting points using structured perturbations
    n_workers = 10
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
        # Generate a structured perturbation using sine waves
        f = w + 1
        phase = (w % 4) * np.pi / 4.0
        x = np.arange(n_points)
        sine_pert = 0.1 * np.sin(2 * np.pi * f * x / n_points + phase)
        perturbed = best_h + sine_pert
        perturbed = np.clip(perturbed, 0, 1)
        # Enforce integral constraint via normalization after clipping
        s = np.sum(perturbed)
        if abs(s) > 1e-9:
            perturbed = perturbed * (target_sum / s)
        # Optimize from this point
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final check for constraints
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:

                c5_candidate = np.max(corr_candidate) * dx
