def tournament_selection(evaluations, k=3):
    # Select k random indices, return the one with lowest c5
    contestants = random.sample(evaluations, k)
    contestants.sort(key=lambda x: x[0])
    return contestants[0]

def crossover(p1, p2):
    # 1-point crossover
    n = len(p1)
    point = np.random.randint(1, n)
    child = np.concatenate( [p1[:point], p2[point:]] )
    return child.copy()

def mutate(h, n_points):
    # Perturb 10% of elements with Gaussian noise
    n_mutate = int(0.1 * n_points)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    h = h.copy()
    h[indices] += np.random.normal(0, 0.1, size=n_mutate)
    h = np.clip(h, 0, 1)
    return h

def repair(h, n_points, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum == 0:
        # Rare case, assign a small value
        h += 1e-6
        current_sum = h.sum()
    h = h * (target_sum / current_sum)
    return h

def objective(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
