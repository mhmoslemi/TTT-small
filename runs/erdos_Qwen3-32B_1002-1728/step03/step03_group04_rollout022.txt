def uniform_crossover(parent1, parent2):
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def fix_sum(child, m):
    current_sum = np.sum(child)
    if current_sum == m:
        return child.copy()
    elif current_sum > m:
        diff = int(current_sum - m)
        ones_idx = np.where(child == 1)[0]
        np.random.shuffle(ones_idx)
        to_flip = ones_idx[:diff]
        new_child = child.copy()
        new_child[to_flip] = 0.0
        return new_child
    else:
        diff = int(m - current_sum)
        zeros_idx = np.where(child == 0)[0]
        np.random.shuffle(zeros_idx)
        to_flip = zeros_idx[:diff]
        new_child = child.copy()
        new_child[to_flip] = 1.0
        return new_child
