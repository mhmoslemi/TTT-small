            perturbed = best_h + rs.normal(0, current_std, n_points)
        elif option == 1:
            # Binarize and perturb
            m = int(round(target_sum))
            indices = np.argpartition(best_h, -m)[-m:]
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, current_std, n_points)
        else:
            # Shift perturbation
            s = rs.randint(1, n_points // 2)
            shifted = np.roll(best_h, s)
            perturbed = shifted + rs.normal(0, current_std, n_points)

        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum < 1e-10:
            perturbed = np.zeros(n_points)
            perturbed_sum = 0.0
        else:
            perturbed = (perturbed / perturbed_sum) * target_sum

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.95:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
