import time
import numpy as np

def repair(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum) < 1e-10:
        return h_clipped.copy()
    h_scaled = h_clipped * (target_sum / current_sum)
    h_scaled = np.clip(h_scaled, 0, 1)
    current_sum = np.sum(h_scaled)
    if abs(current_sum - target_sum) < 1e-8:
        return h_scaled.copy()
    delta = target_sum - current_sum
    indices = np.argsort(h_scaled)
    for i in indices:
        if h_scaled[i] < 1.0:
            add = min(delta, 1.0 - h_scaled[i])
            h_scaled[i] += add
            delta -= add
            if abs(delta) < 1e-8:
                break
    return h_scaled.copy()

def crossover(parent1, parent2):
    if np.random.rand() < 0.5:
        n = len(parent1)
        point = np.random.randint(n)
        child = np.concatenate([parent1[:point], parent2[point:]])
    else:
        child = np.where(np.random.rand(len(parent1)) < 0.5, parent1, parent2)
    return child

def mutate(h, mutation_strength=0.1):
    noise = np.random.normal(0, mutation_strength, len(h))
    return np.clip(h + noise, 0, 1)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    correl = np.correlate(h, 1 - h, mode='full')
    return np.max(correl) * dx
