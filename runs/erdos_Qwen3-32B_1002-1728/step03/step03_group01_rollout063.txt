import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    rs = np.random.RandomState(seed)
    # Generate diverse initial points
    initial_candidates = [h.copy()]
    for _ in range(3):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h0 = np.zeros(n_points)
        h0[idx] = 1.0
        initial_candidates.append(h0)

    T_init = 100.0
    alpha = 0.999

    for candidate in initial_candidates:
        if time.time() - start_time > budget_s * 0.89:
            break
        curr_h = candidate.copy()
        curr_c5 = compute_c5(curr_h)
        T = T_init
        while time.time() - start_time < budget_s * 0.89:
            perturbation = rs.normal(0, 1, size=n_points) * T * 0.1
            new_h = np.clip(curr_h + perturbation, 0, 1)
            new_h_sum = new_h.sum()
            if new_h_sum > 1e-9:
                new_h *= target_sum / new_h_sum
            new_c5 = compute_c5(new_h)
            delta = new_c5 - curr_c5
            if delta < 0 or np.exp(-delta / T) > rs.rand():
                curr_h[:] = new_h
                curr_c5 = new_c5
                if curr_c5 < best_c5:
                    best_h[:] = curr_h
                    best_c5 = curr_c5
            T *= alpha

    # Post-SA refinement
    while time.time() - start_time < budget_s * 0.99:
        result = minimize(
            fun=compute_c5,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-10,
            options={'maxiter': 180, 'ftol': 1e-10}
        )
        if result.success:
            candidate = result.x
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_h[:] = candidate
                best_c5 = c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
