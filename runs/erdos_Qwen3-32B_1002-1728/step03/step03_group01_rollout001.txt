            best_c5 = c5_candidate
            best_h = h_candidate

    # Perturbation and refinement
    while time.time() - start_time < budget_s * 0.95:
        perturbed = best_h.copy()
        perturbed += np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if s > 1e-12:
            perturbed *= T / s
        perturbed = np.clip(perturbed, 0, 1)
        result = minimize(
            fun=lambda h: np.max(np.correlate(h, 1-h, 'full')) * dx,
            x0=perturbed, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-8}
        )
        if result.success:
            h_new = np.clip(result.x, 0, 1)
            c5_new = np.max(np.correlate(h_new, 1 - h_new, 'full')) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h[:] = h_new[:]
        if time.time() - start_time > budget_s - 60:
            break

    assert abs(np.sum(best_h) - T) < 1e-6, "Sum constraint violated"
    assert all(0 <= x <= 1 for x in best_h), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
