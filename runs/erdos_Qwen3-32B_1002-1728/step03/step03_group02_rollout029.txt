                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate
        except Exception as e:
            pass

    rs = np.random.RandomState(seed * 2)
    while time.time() - start_time < budget_s:
        perturbation = rs.normal(0, 0.05, n_points)
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-6:
            perturbed = (perturbed / perturbed_sum) * target_sum
        res_perturb = minimize(
            compute_surrogate,
            x0=perturbed,
            args=(surrogate_alpha, n_points, dx, target_sum),
            method='L-BFGS-B',
            bounds=[(0, 1)]*n_points,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if res_perturb.success:
            h_candidate = res_perturb.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
        if time.time() - start_time > budget_s * 0.9:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
