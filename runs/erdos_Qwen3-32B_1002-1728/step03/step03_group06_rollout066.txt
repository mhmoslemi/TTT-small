    s = h_candidate.sum()
    if abs(s - target_sum) < 1e-6:
        return h_candidate
    elif s < target_sum:
        # Need to increase sum by delta = target_sum - s.
        delta = target_sum - s
        # Find the values that can be increased (currently <1)
        indices_increasing = np.where(h_candidate < 1)[0]
        if len(indices_increasing) == 0:
            # All are at 1. Not possible since s < target_sum
            return h_candidate  # should not happen
        # Sort in order of how much they can increase before hitting 1
        # We can increase each by at most 1 - h_candidate[i]
        # Distribute delta as much as possible.
        delta_left = delta
        highest_increase = min(1 - h_candidate[indices_increasing], delta_left / len(indices_increasing))
        # Increase all by the same amount
        h_candidate[indices_increasing] += highest_increase
        delta_left -= highest_increase * len(indices_increasing)
        # Remaining delta can be distributed to some indices
        # For the remaining delta_left, increase the first (indices_increasing) entries that can accommodate it.
        while delta_left > 0 and len(indices_increasing) > 0:
            highest_increase = 1 - h_candidate[indices_increasing[0]]
            if highest_increase > 0:
                increment = min(highest_increase, delta_left)
                h_candidate[indices_increasing[0]] += increment
                delta_left -= increment
                indices_increasing = indices_increasing[1:]
            else:
                indices_increasing = indices_increasing[1:]

    else:

        delta = s - target_sum
        indices_decreasing = np.where(h_candidate > 0)[0]
        if len(indices_decreasing) == 0:

                delta_left -= decrement
                indices_decreasing = indices_decreasing[1:]

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Validate initial_h_values
    assert abs(np.sum(h_init) - target_sum) < 1e-6, "Initial h violates sum constraint"
    assert np.all((h_init >= 0) & (h_init <= 1)), "Initial h out of bounds"

    # PSO parameters
    n_particles = 100
    max_iter = 500
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_start = 1.2
    social_start = 1.2
    velocity_decay = 0.99
    velocity_clamp = 0.1

    # Initialize particles
    particles = [h_init.copy()]
    velocities = [np.zeros(n_points)]
    pbest = [h_init.copy()]
    pbest_values = [np.max(np.correlate(pbest[0], 1 - pbest[0], mode='full')) * dx]

    # Generate other particles
    for _ in range(n_particles - 1):
        m = int(np.round(target_sum))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=m, replace=False)
        h[indices] = 1.0
        noise = (np.random.rand(n_points) - 0.5) * 0.2
        h_perturbed = np.clip(h + noise, 0, 1)
        h_sum = h_perturbed.sum()
        if h_sum > 1e-6:
            h_scaled = h_perturbed * (target_sum / h_sum)
            h_scaled = np.clip(h_scaled, 0, 1)
        else:
            h_scaled = np.zeros(n_points)
        particles.append(h_scaled)
        velocities.append(velocity_clamp * (2 * np.random.rand(n_points) - 1))
        pbest.append(h_scaled.copy())
        pbest_values.append(np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx)

    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    best_c5 = pbest_values[gbest_idx]

    iteration = 0
    patience = 0
    max_patience = 100

    while time.time() - start_time < budget_s * 0.99 and iteration < max_iter:
        iteration += 1
        t_ratio = iteration / max_iter if max_iter > 0 else 0
        inertia = inertia_start - t_ratio * (inertia_start - inertia_end)
        cognitive = max(0, cognitive_start - 0.01 * iteration)
        social = max(0, social_start - 0.01 * iteration)

        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(2, n_points)

            # Update velocity
            v_new = (inertia * v +
                     cognitive * r1 * (pbest[i] - h) +
                     social * r2 * (gbest - h))
            v_new = np.clip(v_new, -velocity_clamp, velocity_clamp) * velocity_decay
            velocities[i] = v_new

            # Update position
            h_new = np.clip(h + v_new, 0, 1)
            h_sum = h_new.sum()
            if h_sum > 1e-6:
                h_scaled = h_new * (target_sum / h_sum)
                h_scaled = np.clip(h_scaled, 0, 1)
                particles[i] = h_scaled
            else:
                particles[i] = np.zeros(n_points)

            # Evaluate and update pbest
            current_c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
                if current_c5 < best_c5:
                    gbest = pbest[i].copy()
                    best_c5 = current_c5
                    patience = 0
            else:
                patience += 1

        if patience >= max_patience:
            # Reset 20% of particles
            for _ in range(int(0.2 * n_particles)):
                idx = np.random.randint(n_particles)
                m = int(np.round(target_sum))
                h = np.zeros(n_points)
                indices = np.random.choice(n_points, m, replace=False)
                h[indices] = 1.0
                noise = (np.random.rand(n_points) - 0.5) * 0.2
                h_perturbed = np.clip(h + noise, 0, 1)
                h_sum = h_perturbed.sum()
                if h_sum > 1e-6:

                    h_scaled = np.clip(h_scaled, 0, 1)

                    h_scaled = np.zeros(n_points)
                particles[idx] = h_scaled

                pbest[idx] = h_scaled.copy()
                pbest_values[idx] = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
            patience = 0

    assert abs(np.sum(gbest) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((gbest >= -1e-6) & (gbest <= 1.0 + 1e-6)), "Value bounds violated"
    return (gbest.tolist(), best_c5, n_points)
