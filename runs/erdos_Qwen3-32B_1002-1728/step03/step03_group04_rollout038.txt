import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    m = n_points // 2
    dx = 2.0 / n_points
    population_size = 50

    def evaluate_c5(individual):
        h = individual
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # Binarize initial solution
    top_indices = np.argpartition(h_init, -m)[-m:]
    initial_binary = np.zeros(n_points, dtype=np.float64)
    initial_binary[top_indices] = 1.0

    # Initialize population
    population = [initial_binary]
    for _ in range(population_size - 1):
        indices = np.random.choice(n_points, size=m, replace=False)
        indiv = np.zeros(n_points)
        indiv[indices] = 1.0
        population.append(indiv)

    best_c5 = np.inf
    best_h = None

    while time.time() - start_time < budget_s * 0.9:
        # Evaluate fitness
        fitness = [evaluate_c5(indiv) for indiv in population]
        current_best_index = np.argmin(fitness)
        if fitness[current_best_index] < best_c5:
            best_c5 = fitness[current_best_index]
            best_h = population[current_best_index].copy()

        # Elitism: carry forward the best individual
        best_index = np.argmin(fitness)
        elite = population[best_index].copy()
        new_population = [elite]

        # Generate remaining population via crossover and mutation
        for _ in range(population_size - 1):
            # Tournament selection
            def tournament(pop, fit, size=5):
                indices = np.random.choice(len(pop), size=size, replace=False)
                return indices[np.argmin([fit[i] for i in indices])]

            p1_idx = tournament(population, fitness)
            p2_idx = tournament(population, fitness)
            p1 = population[p1_idx]
            p2 = population[p2_idx]

            # Crossover
            child = p1.copy()
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p2[i]
            # Correct sum
            current_sum = np.sum(child)
            diff = int(np.round(current_sum - m))
            if diff > 0:
                ones = np.where(child == 1)[0]
                indices = np.random.choice(ones, size=diff, replace=False)
                child[indices] = 0.0
            elif diff < 0:
                zeros = np.where(child == 0)[0]
                indices = np.random.choice(zeros, size=-diff, replace=False)
                child[indices] = 1.0

            # Mutation
            if np.random.rand() < 0.1:
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i1 = np.random.choice(ones)
                    i2 = np.random.choice(zeros)
                    child[i1], child[i2] = 0.0, 1.0

            new_population.append(child)

        population = new_population

        # Local refinement
        if best_h is not None and time.time() - start_time < budget_s * 0.98:
            target_sum = n_points / 2.0
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
            def objective(x):
                h = x.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            result = minimize(
                fun=objective,
                x0=best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:

                if np.all((h_candidate >= 0) & (h_candidate <= 1)) and abs(np.sum(h_candidate) - target_sum) < 1e-6:
                    c5_candidate = evaluate_c5(h_candidate)

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
