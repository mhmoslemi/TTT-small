def run(seed=42, budget_s=1100, **kwargs):

    # Generate more diverse initial guesses
    np.random.seed(seed)
    additional_initials = []
    for _ in range(5):
        indices = np.random.permutation(n_points)
        random_h = np.zeros(n_points)
        random_h[indices[:n_points//2]] = 1.0
        additional_initials.append(random_h)
    
    # Add structured patterns
    even = np.zeros(n_points)
    even[::2] = 1.0
    odd = np.zeros(n_points)
    odd[1::2] = 1.0
    checkerboard1 = np.kron([1, 0], np.ones((n_points + 1) // 2))[:n_points]
    checkerboard2 = np.kron([0, 1], np.ones((n_points + 1) // 2))[:n_points]
    additional_initials.extend([even, odd, checkerboard1, checkerboard2])

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result_init = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    best_h = result_init.x.reshape(n_points)
    best_c5 = objective(best_h)

    all_initials = additional_initials.copy()

    while time.time() - start_time < budget_s * 0.90:
        if all_initials:
            next_initial = all_initials.pop(0)
        else:
            perturbed = best_h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if abs(perturbed_sum - target_sum) > 1e-6:
                perturbed = (perturbed / perturbed_sum) * target_sum
            next_initial = perturbed

        methods = ['SLSQP', 'Nelder-Mead']
        for method in methods:
            if time.time() - start_time >= budget_s * 0.90:

            result = minimize(
