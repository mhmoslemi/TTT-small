import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h_flat = h_vec
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    def repair(h, target_sum):
        h = h.copy()
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-8:
            return h
        indices = np.argsort(h)
        h_sorted = h[indices]
        n = len(h_sorted)
        delta = target_sum - current_sum
        if delta > 0:
            for i in reversed(range(n)):
                add = min(1.0 - h_sorted[i], delta)
                h_sorted[i] += add
                delta -= add
                if delta <= 1e-8:
                    break
            if delta > 1e-8:
                h_sorted += delta / n
        else:
            delta = -delta
            for i in range(n):
                subtract = min(h_sorted[i], delta)
                h_sorted[i] -= subtract
                delta -= subtract
                if delta <= 1e-8:
                    break
            if delta > 1e-8:
                h_sorted -= delta / n
        h[indices] = h_sorted
        return h

    def select_tournament(population, fitnesses):
        tournament_size = 3
        indices = np.random.choice(len(population), tournament_size)
        selected_idx = indices[np.argmin(fitnesses[indices])]
        return population[selected_idx].copy()

    # Initialize population
    population = [h_init.copy()]
    np.random.seed(seed)
    for _ in range(49):
        h = np.random.rand(n_points)
        h = repair(h, target_sum)
        population.append(h)
    population = np.array(population)

    best_c5 = float('inf')
    best_h = h_init.copy()
    last_improvement_time = start_time

    while time.time() - start_time < budget_s:
        current_fitness = np.array([objective(ind) for ind in population])
        elite_count = 10
        elite_indices = np.argsort(current_fitness)[:elite_count]
        elites = population[elite_indices]
        current_best_index = elite_indices[0]
        current_best_c5 = current_fitness[current_best_index]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_index].copy()
            last_improvement_time = time.time()

        if time.time() - last_improvement_time > 200:
            break

        next_generation = elites.copy()
        for _ in range(len(population) - elite_count):
            parent1 = select_tournament(population, current_fitness)
            parent2 = select_tournament(population, current_fitness)
            if np.random.rand() < 0.8:
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:
                child = parent1.copy() if np.random.rand() < 0.5 else parent2.copy()
            child = repair(child, target_sum)
            if np.random.rand() < 0.2:
                num_mutate = max(1, int(0.1 * n_points))
                mut_indices = np.random.choice(n_points, size=num_mutate, replace=False)
                child[mut_indices] += np.random.normal(0, 0.1, size=num_mutate)
                np.clip(child, 0.0, 1.0, out=child)
                child = repair(child, target_sum)
            next_generation = np.vstack((next_generation, child))
            if time.time() - start_time > budget_s:
                break
        population = next_generation

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
