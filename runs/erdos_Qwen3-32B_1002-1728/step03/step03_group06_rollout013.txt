import time
import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    N = 100  # Population size
    max_iter = 1000
    inertia_start, inertia_end = 0.9, 0.4
    cognitive_start, social_start = 1.2, 1.2
    vel_clamp = 0.1
    decay_vel = 0.99
    rs = np.random.RandomState(seed)

    # Initialize particles
    particles = []
    velocities = []
    pbest_pos = []
    pbest_val = []

    # First particle: use initial_h_values
    h = h_init.copy()
    particles.append(h)
    velocities.append(rs.uniform(-0.01, 0.01, n_points))
    pbest_pos.append(h)
    pbest_val.append(np.inf)

    # Remaining particles: random around 0.5
    for _ in range(N-1):
        h = rs.uniform(0.4, 0.6, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        particles.append(h)
        velocities.append(rs.uniform(-0.01, 0.01, n_points))
        pbest_pos.append(h)
        pbest_val.append(np.inf)

    particles = np.array(particles)
    velocities = np.array(velocities)
    pbest_pos = np.array(pbest_pos)
    pbest_val = np.array(pbest_val)
    gbest_idx = np.argmin(pbest_val)
    gbest_pos = pbest_pos[gbest_idx].copy()

    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break

        # Adaptive weights
        inertia = inertia_start - (inertia_start - inertia_end) * iter_ / max_iter
        cognitive = max(cognitive_start - 0.01 * iter_, 0.5)
        social = max(social_start - 0.01 * iter_, 0.5)

        for i in range(N):
            # Update velocity
            r1, r2 = rs.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest_pos[i] - particles[i]) +
                             social * r2 * (gbest_pos - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_clamp, vel_clamp)
            velocities[i] *= decay_vel

            # Update position
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            current_sum = np.sum(particles[i])
            if current_sum > 1e-8:  # Avoid division by zero
                particles[i] = particles[i] / current_sum * target_sum

        # Evaluate fitness
        def compute_C5(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        current_vals = np.array([compute_C5(p) for p in particles])
        improved = current_vals < pbest_val
        pbest_val[improved] = current_vals[improved]
        pbest_pos[improved] = particles[improved].copy()
        if np.min(current_vals) < np.min(pbest_val):
            gbest_idx = np.argmin(pbest_val)
            gbest_pos = pbest_pos[gbest_idx].copy()

    best_h = gbest_pos
    best_c5 = np.min(pbest_val)

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
