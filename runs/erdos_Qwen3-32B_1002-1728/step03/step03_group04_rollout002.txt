import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
