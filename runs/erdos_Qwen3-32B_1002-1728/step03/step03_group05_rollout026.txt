            corr = np.correlate(h_fine, 1 - h_fine, mode='full')
            final_c5 = np.max(corr) * dx
            if final_c5 < best_c5:
                best_h, best_c5 = h_fine, final_c5
    except Exception:
        pass

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
