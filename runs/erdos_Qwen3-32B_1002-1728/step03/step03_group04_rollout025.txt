import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    target_sum_tol = 1e-6

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_de(dh):
        h = dh.reshape(n_points)
        sum_violation = (np.sum(h) - target_sum) ** 2
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx + 1e6 * sum_violation

    def correct_sum(h_vec):
        h = h_vec.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            h_scaled = np.zeros_like(h_clipped)
        else:
            h_scaled = h_clipped * (target_sum / s)
        return np.clip(h_scaled, 0, 1).flatten()

    # Phase 1: Differential Evolution
    de_result = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=8,
        mutation=(0.5, 0.95),
        recombination=0.8,
        seed=seed,
        workers=1,
        polish=False,
        tol=1e-4
    )
    best_de = correct_sum(de_result.x)
    corr = np.correlate(best_de, 1 - best_de, mode='full')
    best_c5 = np.max(corr) * dx

    # Phase 2: Local refinement
    def objective_refine(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = minimize(
        objective_refine,
        x0=best_de,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    h_best = result.x.reshape(n_points)

    # Final verification
    assert abs(np.sum(h_best) - target_sum) < target_sum_tol, "Sum constraint violated"
    assert np.all((h_best >= -1e-6) & (h_best <= 1.0 + 1e-6)), "Bounds violated"
    corr = np.correlate(h_best, 1 - h_best, mode='full')
    final_c5 = np.max(corr) * dx

    return (h_best.tolist(), final_c5, n_points)
