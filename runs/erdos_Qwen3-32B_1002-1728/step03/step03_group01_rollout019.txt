import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair_candidate(candidate):
        x = np.clip(candidate, 0.0, 1.0)
        s = np.sum(x)
        if s < 1e-9:
            x = np.zeros_like(x)
            x[0] = target_sum
            return x
        scale = target_sum / s
        x *= scale
        x = np.clip(x, 0.0, 1.0)
        return x

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    rs = np.random.RandomState(seed)
    population = [h_init.copy()]

    # Binary patterns
    for _ in range(5):
        indices = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        h_repaired = repair_candidate(h)
        population.append(h_repaired)

    # Sinusoidal patterns
    for _ in range(5):
        i = np.arange(n_points)
        f = rs.rand() * 2 + 1
        phase = rs.rand() * 2 * np.pi
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * (i + phase) / n_points)
        h_repaired = repair_candidate(h)
        population.append(h_repaired)

    # Ramp patterns
    for _ in range(5):
        i = np.arange(n_points)
        slope = (rs.rand() - 0.5) * 4
        h = slope * i + rs.rand()
        h_repaired = repair_candidate(h)
        population.append(h_repaired)

    # Fill to 20-member population
    while len(population) < 20:
        indices = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        h_repaired = repair_candidate(h)
        population.append(h_repaired)
    population = np.array(population)

    # Differential evolution
    F = 0.8
    CR = 0.9
    while time.time() - start_time < budget_s * 0.9:
        idxs = rs.choice(20, 3, replace=False)
        a, b, c = population[idxs]
        mutant = a + F * (b - c)
        i = rs.randint(20)
        target = population[i]
        crossover_mask = rs.rand(n_points) < CR
        trial = np.where(crossover_mask, mutant, target)
        trial_repaired = repair_candidate(trial)
        if objective(trial_repaired) < objective(target):
            population[i] = trial_repaired
        if time.time() - start_time > budget_s * 0.97:
            break

    # Extract best from population
    evals = [objective(ind) for ind in population]
    best_de_idx = np.argmin(evals)
    best_h = population[best_de_idx].copy()
    best_c5 = evals[best_de_idx]

    # Gradient refinement
    while time.time() - start_time < budget_s * 0.98:
        perturbed = best_h.copy()
        perturbed += (rs.rand(n_points) - 0.5) * 0.1
        perturbed = repair_candidate(perturbed)
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h[:] = h_candidate

    # Final validation
    assert np.all(np.abs(np.sum(best_h) - target_sum) < 1e-6), "Constraint violation"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds exceeded"
    return (best_h.tolist(), best_c5, n_points)
