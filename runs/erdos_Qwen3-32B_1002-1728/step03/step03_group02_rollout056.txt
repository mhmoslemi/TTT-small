    ones = idxs[child == 1]
    if diff > 0:
        n_flip = min(diff, len(zeros))
        child[zeros[:n_flip]] = 1
    elif diff < 0:
        n_flip = min(-diff, len(ones))
        child[ones[:n_flip]] = 0
    return child

def mutate(child, mutation_rate, target_sum):
    child = child.copy()
    n_mut = max(1, int(mutation_rate * len(child)))
    one_idx = np.where(child == 1)[0]
    zero_idx = np.where(child == 0)[0]
    if len(one_idx) * len(zero_idx) == 0:
        return child
    n_flip = min(n_mut, len(one_idx), len(zero_idx))
    one_flip = np.random.choice(one_idx, n_flip, replace=False)
    zero_flip = np.random.choice(zero_idx, n_flip, replace=False)
    child[one_flip] = 0
    child[zero_flip] = 1
    return child

def tournament_selection(population, fitnesses, tournament_size=5):
    best_idx = np.random.randint(len(population))
    for _ in range(tournament_size - 1):
        idx = np.random.randint(len(population))
        if fitnesses[idx] < fitnesses[best_idx]:
            best_idx = idx
    return best_idx

def calculate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points // 2
    dx = 2.0 / n_points
    pop_size = 150

    population = []
    # Generate from initial solution
    for _ in range(max(1, int(0.1*pop_size))):
        idxs = np.argsort(h_init)[::-1]
        h = np.zeros(n_points)
        h[idxs[:target_sum]] = 1
        population.append(h)
    # Generate random solutions
    for _ in range(pop_size - len(population)):
        h = np.zeros(n_points)
        idxs = np.random.permutation(n_points)
        h[idxs[:target_sum]] = 1
        population.append(h)

    best_idx = 0
    best_c5 = calculate_c5(population[0], dx)
    for i in range(pop_size):
        c5 = calculate_c5(population[i], dx)
        if c5 < best_c5:
            best_c5 = c5
            best_idx = i
    best_h = population[best_idx].copy()

    start_time = time.time()
    while time.time() - start_time < budget_s - 20:
        new_pop = []
        # Generate offspring
        for _ in range(pop_size):
            p1_idx = tournament_selection(population, [calculate_c5(h, dx) for h in population], 5)
            p2_idx = tournament_selection(population, [calculate_c5(h, dx) for h in population], 5)
            parent1 = population[p1_idx]
            parent2 = population[p2_idx]
            child1, child2 = crossover(parent1, parent2)
            child1 = repair(child1, target_sum)
            child2 = repair(child2, target_sum)
            child1 = mutate(child1, 0.05, target_sum)
            child2 = mutate(child2, 0.05, target_sum)
            new_pop.extend([child1, child2])
        
        # Merge and select best
        combined = population + new_pop
        combined_c5 = [calculate_c5(h, dx) for h in combined]
        sorted_idx = np.argsort(combined_c5)
        population = [combined[i] for i in sorted_idx[:pop_size]]
        
        # Update best
        current_c5 = min(combined_c5[:pop_size])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0].copy()

        # Maintain diversity
        if time.time() - start_time > 10 and np.random.rand() < 0.1:
            h = np.zeros(n_points)
            idxs = np.random.permutation(n_points)
            h[idxs[:target_sum]] = 1
            population[np.random.randint(pop_size)] = h

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
