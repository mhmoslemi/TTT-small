import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = tuple([(0.0, 1.0)] * n_points)

    # CMA-ES configuration
    cma_options = {
        'popsize': 30,
        'sigma0': 0.2,
        'bounds': bounds,
        'seed': seed,
        'maxiter': 50
    }

    result_cma = minimize(
        fun=objective,
        x0=h_init,
        method='CMAES',
        bounds=bounds,
        options=cma_options
    )

    best_h = result_cma.x.reshape(n_points)
    best_c5 = objective(best_h)

    # Gradient refinement
    lr = 0.05
    max_local_steps = 500
    h = best_h.copy()

    for step in range(max_local_steps):
        if time.time() - start_time > budget_s * 0.98:
            break
        grad = np.zeros_like(h)
        eps = 1e-5

            h_up = h.copy()

            up = np.clip(h_up, 0, 1)
