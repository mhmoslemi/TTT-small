import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initializations
    rs = np.random.RandomState(seed)
    population = []

    # Uniform initialization
    h_uniform = np.full(n_points, 0.5)
    population.append(h_uniform)

    # Binary random patterns
    for _ in range(4):
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        population.append(h)

    # Sinusoidal patterns
    for freq in [0.25, 0.5, 1.0]:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * freq * x)
        h = np.clip(h, 0.0, 1.0)
        population.append(h)

    # Ramp patterns
    ramp = np.linspace(0, 1, n_points)
    h_ramp = 1.0 * (ramp > 1.0)
    population.append(h_ramp)
    h_ramp2 = 0.5 * (ramp - 0.5)
    h_ramp2 = np.clip(h_ramp2, 0.0, 1.0)
    population.append(h_ramp2)

    # Step function pattern
    h_step = np.zeros(n_points)
    h_step[n_points//3:2*n_points//3] = 1.0
    population.append(h_step)

    while len(population) < 30:
        population.append(population[rs.randint(len(population))].copy())
    population = np.array(population)

    # Differential Evolution parameters
    F = 0.8
    CR = 0.9
    max_generations = 100

    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # DE loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
        for i in range(len(population)):
            # Select 3 random distinct indices
            indices = np.random.choice(len(population), 4, replace=False)
            a, b, c = indices[1], indices[2], indices[3]
            # Mutate
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            # Crossover
            trial = population[i].copy()
            j_rand = rs.randint(n_points)
            for j in range(n_points):
                if rs.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
            # Repair constraints
            trial_sum = trial.sum()
            if trial_sum > 1e-9:
                trial = trial * (target_sum / trial_sum)
            else:
                idx = rs.choice(n_points, int(target_sum), replace=False)
                trial = np.zeros(n_points)
                trial[idx] = 1.0
            # Evaluate
            trial_c5 = objective(trial)
            current_c5 = objective(population[i])
            if trial_c5 < current_c5:
                population[i] = trial

    # Evaluate population after DE
    c5_population = [objective(h) for h in population]
    best_de_idx = np.argmin(c5_population)
    best_h = population[best_de_idx].copy()
    best_c5 = c5_population[best_de_idx]

    # Gradient refinement on top candidates
    top_candidates = np.argsort(c5_population)[:5]
    for idx in top_candidates:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            fun=objective,
            x0=population[idx],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate

    # Final local search
    while time.time() - start_time < budget_s * 0.98:
        perturbed = best_h.copy()
        perturbed += (rs.rand(n_points) - 0.5) * 0.2
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-9:
            perturbed = perturbed * (target_sum / perturbed_sum)
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8}
        )

            h_candidate = result.x.reshape(n_points)
