import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    rs = np.random.RandomState(seed)

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def binary_threshold(h_vec):
        h = h_vec.reshape(n_points)
        idx = np.argsort(h)
        h_bin = np.zeros(n_points)
        h_bin[idx[-int(target_sum):]] = 1.0
        return h_bin

    def normalize_sum(h_vec):
        h = h_vec.reshape(n_points)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0.0, 1.0)
        h += (target_sum - np.sum(h)) / n_points
        return h

    def generate_binary():
        idx = rs.permutation(n_points)
        h = np.zeros(n_points)
        h[idx[:int(target_sum)]] = 1.0
        return h

    population = []
    # Perturbed initial solutions
    for _ in range(5):
        perturbed = h_init + rs.normal(0, 0.1, n_points)
        perturbed = normalize_sum(perturbed)
        population.append(perturbed)
    # Binarized initial
    population.append(binary_threshold(h_init))
    # Structured patterns
    even = np.kron([1.0, 0.0], np.ones(n_points // 2))[:n_points]
    odd = np.kron([0.0, 1.0], np.ones(n_points // 2))[:n_points]
    checkerboard = np.kron([1.0, 0.0, 0.0, 1.0], np.ones(n_points // 4))[:n_points]
    population.extend([even, odd, checkerboard])
    # Random binaries
    for _ in range(10):
        population.append(generate_binary())

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    best_h = None

    while time.time() - start_time < budget_s * 0.9:
        # Evaluate population
        valid_population = []

            valid_population.append((c5, h))
