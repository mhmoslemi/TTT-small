        max_idx = np.argmax(corr)
        lag_k = max_idx - (n_points - 1)

        # Compute gradient for current lag_k
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + lag_k
            if 0 <= j < n_points:
                grad[i] += 1 - h[j]
            m = i - lag_k
            if 0 <= m < n_points:
                grad[i] -= h[m]

        # Gradient step with projection
        h_proposed = h - step_size * grad
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed *= target_sum / np.sum(h_proposed)

        c5_proposed = objective(h_proposed)
        if c5_proposed < best_c5:
            h = h_proposed.copy()
            best_c5 = c5_proposed
            best_h = h.copy()

    # Refine with adaptive local search
    rs = np.random.RandomState(seed)
    for _ in range(3):  # 3 diverse perturbations
        if time.time() - start_time > budget_s * 0.9:
            break
        perturbation = rs.rand(n_points) * 0.2
        x0 = (best_h * (1 - 0.1) + perturbation * 0.1) % 1.0
        x0 *= target_sum / np.sum(x0)
        
        result = minimize(
            fun=objective,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            candidate = result.x.reshape(n_points)
            c5 = objective(candidate)
            if c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = c5

    # Final verification
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
