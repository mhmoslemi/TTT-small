            h_np = h_output.detach().numpy()
            h_clipped = np.clip(h_np, 0.0, 1.0)
            h_scaled = h_clipped * (target_sum / h_clipped.sum())
            h_scaled = np.clip(h_scaled, 0.0, 1.0)
            current_c5 = evaluate_c5(h_scaled, dx)
            if current_c5 < best_c5:
                best_h = h_scaled.copy()
                best_c5 = current_c5

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
