import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_energy = compute_energy(h)
    current_h = best_h.copy()
    current_energy = best_energy

    np.random.seed(seed)
    max_line_search_steps = 10
    perturb_interval = 300  # seconds
    last_perturb = start_time

    while time.time() - start_time < budget_s:
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_index = np.argmax(corr)
        k_star = max_index - (n_points - 1)

        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            term1 = 0.0
            if 0 <= j + k_star < n_points:
                term1 = 1.0 - current_h[j + k_star]
            term2 = 0.0
            if 0 <= j - k_star < n_points:
                term2 = -current_h[j - k_star]
            gradient[j] = dx * (term1 + term2)

        # Line search
        step_size = 0.1
        improved = False
        for _ in range(max_line_search_steps):
            proposed_h = current_h - step_size * gradient
            proposed_h = np.clip(proposed_h, 0.0, 1.0)
            current_sum = proposed_h.sum()
            if not np.isclose(current_sum, 0.0):
                scale = target_sum / current_sum
                proposed_h *= scale
                proposed_energy = compute_energy(proposed_h)
                if proposed_energy < current_energy:
                    current_h = proposed_h
                    current_energy = proposed_energy
                    improved = True
                    break
            step_size *= 0.5

        # Update best solution
        if current_energy < best_energy:
            best_energy = current_energy
            best_h[:] = current_h.copy()

        # Perturb periodically
        if time.time() - last_perturb > perturb_interval:
            perturbation = 0.05 * np.random.randn(n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            current_sum = perturbed_h.sum()
            if not np.isclose(current_sum, 0.0):
                scale = target_sum / current_sum
                perturbed_h *= scale
                current_h[:] = perturbed_h
            last_perturb = time.time()

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_energy, n_points)
