import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds_continuous = [(0.0, 1.0)] * n_points
    cons_continuous = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    result_continuous = minimize(
        fun=evaluate_c5,
        x0=h_init,
        method='L-BFGS-B',
        bounds=bounds_continuous,
        constraints=cons_continuous,
        options={'maxiter': 300}
    )
    h_continuous = result_continuous.x.reshape(n_points)

    indices = np.argsort(h_continuous)[::-1]
    h_bin = np.zeros(n_points)

        return np.where(h == 1)[0], np.where(h == 0)[0]
