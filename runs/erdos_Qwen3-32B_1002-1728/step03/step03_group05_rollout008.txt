    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 50
    generations = 50
    tournament_size = 5
    mutation_rate = 0.1
    top_candidates_for_local = 5
    np.random.seed(seed)

    # Generate initial population
    population = []
    for _ in range(population_size):
        h = h_init.copy()
        h += np.random.normal(0, 0.1, h.shape)
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s != 0:
            h *= target_sum / s
        else:
            h = np.full_like(h, 0.5)
        population.append(h)
    population = np.array(population)

    best_c5 = evaluate_c5(population[0], dx)
    best_individual = population[0].copy()

    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.98:
            break
        fitness = np.array([evaluate_c5(ind, dx) for ind in population])
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_individual = population[current_best_idx].copy()
            best_c5 = current_best_c5
        # Tournament selection
        parents = []
        for _ in range(population_size):
            candidates = np.random.choice(population_size, tournament_size)
            parent_idx = candidates[np.argmin(fitness[candidates])]
            parents.append(population[parent_idx])
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 >= population_size:
                break
            p1 = parents[i]
            p2 = parents[i + 1]
            # Crossover
            if np.random.rand() < 0.8:
                start = np.random.randint(0, n_points)
                length = np.random.randint(int(n_points * 0.2), int(n_points * 0.5))
                end = min(start + length, n_points)
                child1 = p1.copy()
                child2 = p2.copy()
                child1[start:end], child2[start:end] = p2[start:end], p1[start:end]
            else:
                child1, child2 = p1.copy(), p2.copy()
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    n_mut = max(1, int(n_points * 0.1))
                    mut_indices = np.random.choice(n_points, n_mut, replace=False)
                    for idx in mut_indices:
                        if np.random.rand() < 0.5:
                            child[idx] = np.clip(child[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
                        else:
                            child[idx] = 1 - child[idx]
                child = np.clip(child, 0, 1)
                s = np.sum(child)
                if s != 0:
                    child *= target_sum / s
                else:
                    child = np.full_like(child, 0.5)
                offspring.append(child)
        # Truncate and complete population
        if len(offspring) > population_size:
            offspring = offspring[:population_size]
        else:
            while len(offspring) < population_size:
                offspring.append(best_individual.copy())
        population = np.array(offspring)

    # Local optimization
    fitness = np.array([evaluate_c5(ind, dx) for ind in population])
    top_indices = np.argsort(fitness)[:top_candidates_for_local]
    for idx in top_indices:
        if time.time() - start_time > budget_s:
            break
        h0 = population[idx].copy()
        result = minimize(
            fun=lambda h: evaluate_c5(h, dx),
            x0=h0,
            method='SLSQP',
            bounds=[(0,1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 200},
            tol=1e-8
        )
        if result.success:
            h_candidate = result.x
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                c5_candidate = evaluate_c5(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_individual = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_individual) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_individual >= -1e-6) & (best_individual <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_individual.tolist(), best_c5, n_points)
