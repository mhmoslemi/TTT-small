import time
import numpy as np
from numpy.random import choice, permutation

def binary_h(h_vec, target_sum):
    n_points = len(h_vec)
    indices = np.argsort(h_vec)
    h_binary = np.zeros(n_points)
    h_binary[indices[-int(target_sum):]] = 1.0
    return h_binary

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def generate_random_individual(n_points, required_sum):
    h = np.zeros(n_points)
    indices = np.random.choice(n_points, required_sum, replace=False)
    h[indices] = 1.0
    return h

def repair_individual(h, required_sum):
    current_sum = int(np.sum(h))
    if current_sum == required_sum:
        return h.copy()
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    diff = required_sum - current_sum
    if diff > 0:
        flip_indices = zeros_idx[:diff]
        h[flip_indices] = 1.0
    else:
        flip_indices = ones_idx[: -diff]
        h[flip_indices] = 0.0
    return h

def mutate_individual(h, required_sum, n_points, mut_rate):
    h = h.copy()
    num_flips = int(n_points * mut_rate / 2)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(num_flips):
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = np.random.choice(ones)
        i2 = np.random.choice(zeros)
        h[i1] = 0.0
        h[i2] = 1.0
        ones = np.setdiff1d(ones, i1)
        zeros = np.setdiff1d(zeros, i2)
    return h

def crossover_parents(parent1, parent2, required_sum):
    mask = np.random.rand(len(parent1)) < 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(~mask, parent1, parent2)
    child1 = repair_individual(child1, required_sum)
    child2 = repair_individual(child2, required_sum)
    return child1, child2

def tournament_selection(population, tourn_size, dx):
    indices = np.random.choice(len(population), tourn_size, replace=False)
    candidates = [population[i] for i in indices]
    scores = [compute_c5(candidate, dx) for candidate in candidates]
    best_idx = np.argmin(scores)
    return candidates[best_idx]
