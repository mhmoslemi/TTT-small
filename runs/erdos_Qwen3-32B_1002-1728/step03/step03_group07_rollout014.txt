        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale = target_sum / np.sum(h_perturbed)
        h_perturbed *= scale
        h_perturbed = np.clip(h_perturbed, 0, 1)
        candidates.append(h_perturbed)

    for candidate in candidates:
        current_h = candidate.copy()
        current_energy = compute_energy(current_h, n, dx)
        iter_since_improve = 0
        for iter in range(max_iter):
            if time.time() - start_time > budget_s * 0.99:
                break
            # Compute max overlap and shift
            corr = np.correlate(current_h, 1.0 - current_h, mode='full')
            m = np.argmax(corr)
            s_array = m - (n - 1)
            # Compute gradient
            grad = np.zeros(n)
            for i in range(n):
                j = i + s_array
                if 0 <= j < n:
                    grad[i] += 1.0 - current_h[j]
                    grad[j] -= current_h[i]
            # Normalize gradient
            grad -= np.mean(grad)
            grad *= 1.0 / (1e-8 + np.max(np.abs(grad)))
            # Line search
            step_size = 0.05
            best_step = 0.0
            best_energy = current_energy
            for _ in range(5):
                test_h = current_h - step_size * grad
                test_h = np.clip(test_h, 0, 1)
                scale = target_sum / np.sum(test_h)
                test_h *= scale
                test_h = np.clip(test_h, 0, 1)
                test_energy = compute_energy(test_h, n, dx)
                if test_energy < best_energy:
                    best_energy = test_energy
                    best_step = step_size
                    current_h = test_h.copy()
                    current_energy = test_energy
                    iter_since_improve = 0
                    break  # Accept first improvement
                step_size *= 0.5
            iter_since_improve += 1
            # Check for perturbation
            if iter_since_improve > perturb_interval:
                current_h += np.random.uniform(-perturb_strength, perturb_strength, n)
                current_h = np.clip(current_h, 0, 1)
                scale = target_sum / np.sum(current_h)
                current_h *= scale
                current_h = np.clip(current_h, 0, 1)
                iter_since_improve = 0
            # Update global best
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = current_h.copy()
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_energy, n)
