def project(h, target_sum):
    # Minimize squared error with constraints
    n_points = len(h)
    def objective_proj(x):
        return 0.5 * np.sum((x - h) ** 2)
    def constraint_proj(x):
        return np.sum(x) - target_sum
    # Bounds
    bounds_proj = [(0, 1) for _ in range(n_points)]
    # Initial guess is h
    x0 = h.copy()
    # Constraints
    constraints_proj = {'type': 'eq', 'fun': constraint_proj}
    # Run optimization
    result = scipy.optimize.minimize(objective_proj, x0=x0, method='SLSQP', bounds=bounds_proj, constraints=constraints_proj)
    if result.success:
        return result.x
    else:
        # Fallback: clip and scale
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            return h_clipped
        scale = target_sum / s
        h_scaled = h_clipped * scale
        # Clip again in case out of bounds
        return np.clip(h_scaled, 0, 1)
