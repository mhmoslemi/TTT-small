            fun=objective,
            x0=perturbed,
            method='Nelder-Mead',
            options={'maxiter': 100},
            bounds=[(0, 1)] * n_points
        )
        if res.success:
            candidate_h = project(res.x)
            candidate_c5 = objective(candidate_h)
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
    
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
