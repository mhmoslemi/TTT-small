import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    num_particles = 50
    max_iterations = 500
    inertia_start, inertia_end = 0.9, 0.4
    cognitive, social = 2.0, 2.0

    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    personal_best = np.zeros((num_particles, n_points))
    personal_best_scores = np.full(num_particles, np.inf)
    global_best = np.zeros(n_points)
    global_best_score = np.inf

    # Initialize particles around initial_h_values
    for i in range(num_particles):
        perturbation = np.random.normal(0, 0.1, n_points)
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        sum_new_h = np.sum(new_h)
        if sum_new_h != 0:
            new_h = new_h * (target_sum / sum_new_h)
        particles[i] = new_h
        score = objective(new_h)
        if score < personal_best_scores[i]:
            personal_best_scores[i] = score
            personal_best[i] = new_h.copy()
            if score < global_best_score:
                global_best_score = score
                global_best = new_h.copy()

    # PSO loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.9:
            break
        inertia = max(inertia_end, inertia_start - (inertia_start - inertia_end) * iteration / max_iterations)
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best[i] - particles[i]) +
                             social * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] *= (target_sum / sum_p)
            else:
                particles[i] = target_sum / n_points

            score = objective(particles[i])
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best = particles[i].copy()

    # Local search on top 10 candidates
    best_h = global_best.copy()
    best_c5 = global_best_score
    top_indices = np.argsort(personal_best_scores)[:10]

    for idx in top_indices:
        if time.time() - start_time > budget_s * 0.95:
            break
        h_candidate = personal_best[idx].copy()
        result = minimize(
            fun=lambda h: objective(h),
            x0=h_candidate,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_local = np.clip(result.x, 0, 1)
            sum_h = np.sum(h_local)
            if sum_h != 0:
                h_local *= (target_sum / sum_h)
            c5 = objective(h_local)
            if c5 < best_c5:
                best_h = h_local
                best_c5 = c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
