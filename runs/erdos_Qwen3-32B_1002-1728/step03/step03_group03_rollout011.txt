            candidates2 = np.random.choice(len(population), 3, replace=False)
            parent2_idx = candidates2[np.argmin(c5_values[candidates2])]
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]

            # Crossover
            split = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:split], parent2[split:]])

            # Mutation
            mutation_mask = np.random.rand(n_points) < 0.1
            mutation = np.random.normal(0, 0.15, n_points)
            child[mutation_mask] += mutation[mutation_mask]
            child = np.clip(child, 0.0, 1.0)
            s = child.sum()
            if s != 0:
                child = (child * (target_sum / s)).clip(0.0, 1.0)
            else:
                child = np.random.rand(n_points)
                child = np.clip(child, 0.0, 1.0)
                s2 = child.sum()
                if s2 != 0:
                    child = (child * (target_sum / s2)).clip(0.0, 1.0)
                else:
                    child = np.ones(n_points) * (target_sum / n_points)
                    child.clip(0.0, 1.0)
            new_population.append(child)
        population = np.array(new_population)
        if time.time() - start_time > budget_s * 0.98:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
