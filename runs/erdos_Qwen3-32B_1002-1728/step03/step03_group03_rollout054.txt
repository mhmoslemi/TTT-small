import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    n_particles = 50
    c1, c2 = 2.0, 2.0
    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros_like(positions)
    pbest_positions = np.zeros_like(positions)
    pbest_scores = np.full(n_particles, np.inf)
    gbest_position = np.zeros(n_points)
    gbest_score = np.inf

    # Initialize particles around initial_h_values
    for i in range(n_particles):
        np.random.seed(seed + i)
        perturbation = np.random.normal(0, 0.1, n_points)
        noisy = h + perturbation
        noisy = np.clip(noisy, 0.0, 1.0)
        current_sum = np.sum(noisy)
        noisy = (noisy / np.clip(current_sum, 1e-8, None)) * target_sum
        positions[i] = noisy
        pbest_positions[i] = noisy.copy()
        score = np.max(np.correlate(noisy, 1 - noisy, mode='full')) * dx
        pbest_scores[i] = score
        if score < gbest_score:
            gbest_score = score
            gbest_position[:] = noisy.copy()

    # Run PSO until 80% of time budget
    while time.time() - start_time < budget_s * 0.8:
        for i in range(n_particles):
            if i == 0:
                elapsed = time.time() - start_time
                w = max(0.4, 0.9 - (0.9 - 0.4) * (elapsed / (0.8 * budget_s)))
            else:
                w = max(0.4, 0.9 - (0.9 - 0.4) * ((time.time() - start_time) / (0.8 * budget_s)))
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) +
                            c2 * r2 * (gbest_position - positions[i]))
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0.0, 1.0)
            current_sum = np.sum(positions[i])
            if current_sum < 1e-8:
                positions[i] = np.full(n_points, target_sum / n_points)
            else:
                positions[i] = (positions[i] / current_sum) * target_sum
        for i in range(n_particles):
            current_score = np.max(np.correlate(positions[i], 1 - positions[i], mode='full')) * dx
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = positions[i].copy()
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position[:] = positions[i].copy()
        if time.time() - start_time > budget_s * 0.8:
            break

    # Local refinement with SLSQP
    def optimize_constrained(h_vec):
        result = minimize(
            fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            x0=h_vec,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            bounds=[(0.0, 1.0)] * n_points,
            method='SLSQP',
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        return result

    def ensure_feasible(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        h_vec = (h_vec / np.sum(h_vec)) * target_sum
        return h_vec

    best_h = ensure_feasible(gbest_position)
    best_score = gbest_score
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:

                refined = ensure_feasible(refined)
                refined_score = np.max(np.correlate(refined, 1 - refined, mode='full')) * dx
                if refined_score < best_score:
                    best_h = refined
                    best_score = refined_score

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_score, n_points)
