import time
import numpy as np
from numpy.random import RandomState

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    num_particles = 100
    max_iter = 1000
    rs = RandomState(seed)

    # Particle initialization with constraints
    def clamp_and_normalize(pos):
        pos = np.clip(pos, 0, 1)
        pos_sum = pos.sum()
        if pos_sum == 0:
            return np.full(n_points, target_sum / n_points)
        return pos * (target_sum / pos_sum)

    positions = np.zeros((num_particles, n_points))
    positions[0] = h_init.copy()
    for i in range(1, num_particles):
        while True:
            h_rand = rs.rand(n_points)
            h_sum = h_rand.sum()
            if h_sum >= target_sum:
                positions[i] = h_rand * (target_sum / h_sum)
                break

    velocities = 0.01 * rs.randn(num_particles, n_points)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Evaluate initial fitness
    fitness = np.array([objective(p) for p in positions])
    personal_best_pos = positions.copy()
    personal_best_fit = fitness.copy()
    global_idx = np.argmin(fitness)
    global_pos = positions[global_idx].copy()
    best_c5 = fitness[global_idx]

    # Hyperparameters
    inertia_decay = 0.995
    inertia = 0.7
    cognitive = 1.1
    social = 1.1

    # Core optimization loop
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1 = rs.rand()
            r2 = rs.rand()
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best_pos[i] - positions[i]) +
                            social * r2 * (global_pos - positions[i]))
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            positions[i] = clamp_and_normalize(positions[i] + velocities[i])
            
        # Update fitness
        updated = []
        for i in range(num_particles):
            f = objective(positions[i])
            if f < personal_best_fit[i]:
                personal_best_fit[i] = f
                personal_best_pos[i] = positions[i].copy()
                updated.append((i, f))
        
        # Update global best
        for i, f in updated:
            if f < best_c5:
                best_c5 = f
                global_pos = positions[i].copy()

        inertia *= inertia_decay

    # Final constraint verification
    assert abs(np.sum(global_pos) - target_sum) < 1e-6
    assert np.all((global_pos >= -1e-6) & (global_pos <= 1.0 + 1e-6)), f"Values out of range: {global_pos.min()}, {global_pos.max()}"
    
    return (global_pos.tolist(), best_c5, n_points)
