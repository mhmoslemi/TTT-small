import time
import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scaling_factor = target_sum / current_sum if current_sum != 0 else 1.0
        h = h * scaling_factor
        return np.clip(h, 0, 1)  # Final clip in case of scaling edge cases

    def create_individual():
        if random.random() < 0.5:
            return h_init.copy()
        h = np.random.rand(n_points) * random.uniform(0.2, 0.8)
        return repair(h)

    population = [create_individual() for _ in range(50)]
    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s * 0.95:
        fitness = [evaluate(ind) for ind in population]
        population = [population[i] for i in np.argsort(fitness)]
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5

        next_gen = []
        elites = int(0.2 * len(population))
        next_gen.extend(population[:elites])

        while len(next_gen) < len(population):
            parent1, parent2 = [population[random.choice(range(int(0.1 * len(population))))] for _ in range(2)]
            if random.random() < 0.8:
                alpha = random.random()
                child1 = repair(alpha * parent1 + (1 - alpha) * parent2)
                child2 = repair(alpha * parent2 + (1 - alpha) * parent1)
                next_gen.extend([child1, child2])
            else:
                for parent in [parent1, parent2]:
                    if random.random() < 0.2:
                        mutation = np.random.randn(n_points) * 0.1
                        mutated = parent + mutation
                        next_gen.append(repair(mutated))
                    else:
                        next_gen.append(parent.copy())

        population = next_gen

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
