import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    np.random.seed(seed)

    for iter_idx in range(500):
        if time.time() - start_time > budget_s:
            break

        current_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(current_corr)
        k_shift = max_idx - (n_points - 1)
        if k_shift < 0:
            continue

        g = np.zeros(n_points)
        for i in range(n_points - k_shift):
            if i + k_shift < n_points:
                g[i] = 1.0 - h[i + k_shift]

        delta = cp.Variable(n_points)
        objective = cp.Minimize(g @ delta)
        constraints = [
            cp.sum(delta) == 0,
            delta <= 1.0 - h,
            delta >= -h
        ]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except:
            continue

        if problem.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
            continue

        h_update = h + delta.value
        h_update = np.clip(h_update, 0.0, 1.0)
        current_sum = np.sum(h_update)
        if not np.isclose(current_sum, target_sum, atol=1e-6):
            delta_sum = target_sum - current_sum
            if abs(delta_sum) > 1e-8:
                h_update += delta_sum / n_points
                h_update = np.clip(h_update, 0.0, 1.0)

        new_c5 = compute_c5(h_update)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_update.copy()
            h = h_update.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
