import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_binary(h):
        h = h.copy()
        idx = np.argsort(h)
        n_ones = int(target_sum)
        h_bin = np.zeros_like(h)
        h_bin[idx[-n_ones:]] = 1.0
        return h_bin

    def repair(h):
        h[h < 0.5] = 0.0
        h[h >= 0.5] = 1.0
        return make_binary(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def create_population(size):
        pop = []
        rs = np.random.RandomState(seed + 42)
        # Include initial
        if len(h_init) == n_points:
            pop.append(h_init)
        # Add random samples
        for _ in range(size):
            idx = rs.permutation(n_points)
            h = np.zeros(n_points)
            h[idx[:int(target_sum)]] = 1.0
            pop.append(h)
        return np.array(pop)

    def tournament_select(pop, fitness, size=5):
        idx = np.random.randint(len(pop), size=size)
        best_idx = idx[np.argmin(fitness[idx])]
        return pop[best_idx].copy()

    def crossover(p1, p2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, p1, p2)
        return repair(child)

    def mutate(h, mutation_rate):
        h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        n_flip = int(len(ones) * mutation_rate)
        if n_flip > 0:
            h[ones[:n_flip]] = 0
            h[zeros[:n_flip]] = 1
        return h

    population_size = 150
    max_generations = 600
    mutation_rate = 0.05

    pop = create_population(population_size)
    fit = np.array([evaluate(ind) for ind in pop])
    best_idx = np.argmin(fit)
    best = pop[best_idx].copy()
    best_c5 = fit[best_idx]

    rs = np.random.RandomState(seed)
    gen = 0
    while time.time() - start_time < budget_s * 0.9 and gen < max_generations:
        new_pop = []
        new_fit = []
        for _ in range(population_size // 2):
            # Selection
            p1 = tournament_select(pop, fit)
            p2 = tournament_select(pop, fit)
            # Crossover
            child = crossover(p1, p2)
            # Mutation
            child = mutate(child, mutation_rate * (1 - gen / max_generations)))
            new_pop.append(child)
        # Evaluate new population
        new_pop = np.array(new_pop)
        new_fit = np.array([evaluate(ind) for ind in new_pop])
        # Merge with existing population
        pop = np.concatenate([pop, new_pop])
        fit = np.concatenate([fit, new_fit])
        # Select survivors (elitism + rest)
        all_idx = np.argsort(fit)
        pop = pop[all_idx[:population_size]]
        fit = fit[all_idx[:population_size]]
        # Update best
        current_best = pop[np.argmin(fit)]
        current_c5 = fit[np.argmin(fit)]
        if current_c5 < best_c5:
            best = current_best
            best_c5 = current_c5
        gen += 1

    # Final repair pass
    best = repair(best)
    final_c5 = evaluate(best)
    best_c5 = min(best_c5, final_c5)

    assert abs(np.sum(best) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best >= -1e-6) & (best <= 1.0 + 1e-6)), "Value bounds violated"
    return (best.tolist(), best_c5, n_points)
