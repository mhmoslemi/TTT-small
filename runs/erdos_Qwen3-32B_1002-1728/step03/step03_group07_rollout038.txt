import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h_proj = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_proj)
        if np.isclose(current_sum, target_sum):
            return h_proj
        scale = target_sum / current_sum
        h_proj *= scale
        h_proj = np.clip(h_proj, 0.0, 1.0)
        return h_proj

    def objective_with_penalty(h_flat):
        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        sum_penalty = 1e4 * (current_sum - target_sum) ** 2
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + sum_penalty

    def gradient_with_penalty(h_flat):
        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        max_index = np.argmax(corr)
        k_star = max_index - (n_points - 1)
        grad = np.zeros(n_points)

        for i in range(n_points):
            j = i + k_star
            if 0 <= j < n_points:
                grad[i] += 1 - h_clipped[j]
                grad[j] += -h_clipped[i]
        grad *= dx

        grad_penalty = 2 * 1e4 * (current_sum - target_sum)
        grad += grad_penalty
        return grad.flatten()

    best_energy = float('inf')
    best_h = h_init.copy()

    np.random.seed(seed)
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.99:
            break
        perturbation = 0.05 * (2 * np.random.rand(n_points) - 1)
        h0 = (project(h_init) + perturbation)  # Perturb around initial solution
        h0 = project(h0)
        result = minimize(
            objective_with_penalty,
            h0.flatten(),
            method='L-BFGS-B',
            jac=gradient_with_penalty,
            bounds=[(0.0, 1.0)] * n_points,
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        if result.success:
            current_h = result.x.reshape(n_points)
            current_energy = result.fun - 1e4 * (np.sum(project(current_h)) - target_sum)**2
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = project(current_h)

    final_h = project(best_h)
    assert abs(np.sum(final_h) - target_sum) < 1e-6
    assert np.all((final_h >= -1e-6) & (final_h <= 1.0 + 1e-6))
    return (final_h.tolist(), best_energy, n_points)
