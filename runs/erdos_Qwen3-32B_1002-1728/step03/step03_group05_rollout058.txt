    best_c5_values = [objective_func(h_init.copy())]

    best_solutions.append(best_h_de.copy())
    best_c5_values.append(objective_func(best_h_de.copy()))

    rs = np.random.RandomState(seed)
    n_perturbs = 5
    for i in range(n_perturbs):
        if time.time() - start_time > budget_s * 0.98:
            break
        perturbed = best_h_de.copy()
        perturbed += rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        s = np.sum(perturbed)
        if s != 0:
            perturbed = perturbed * (target_sum / s)
            perturbed = np.clip(perturbed, 0.0, 1.0)
        best_solutions.append(perturbed)
        best_c5_values.append(objective_func(perturbed))

    best_h = best_solutions[0].copy()
    best_c5 = best_c5_values[0]

    for i, candidate in enumerate(best_solutions):
        if time.time() - start_time > budget_s * 0.98:
            break
        result_local = minimize(
            fun=objective_func,
            x0=candidate,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result_local.success:
            h_candidate = result_local.x.reshape(n_points)
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
