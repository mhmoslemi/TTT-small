import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initializations
    rs = np.random.RandomState(seed)
    initializations = [h_init.copy()]
    for _ in range(10):
        # Binary patterns
        idx = rs.choice(n_points, int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        initializations.append(h)
    for freq in [1, 2, 3]:
        # Sinusoidal patterns
        h = (np.sin(2 * np.pi * freq * np.arange(n_points) / n_points) + 1) / 2
        initializations.append(h)

    best_c5 = float('inf')
    best_h = h_init.copy()

    # Simulated Annealing parameters
    T_initial = 100.0
    T_min = 1e-5
    alpha = 0.999
    for init in initializations:
        if time.time() - start_time > budget_s * 0.9:
            break
        current_h = init.copy()
        current_energy = objective(current_h)
        if current_energy < best_c5:
            best_c5 = current_energy
            best_h = current_h.copy()

        T = T_initial
        while T > T_min and time.time() - start_time < budget_s * 0.9:
            # Generate perturbed solution
            noise = rs.normal(0, 0.1 * T, n_points)
            new_h = current_h + noise
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new < 1e-9:
                new_h = np.zeros_like(new_h)
            else:
                new_h = new_h * (target_sum / sum_new)
            new_energy = objective(new_h)
            delta = new_energy - current_energy

            # Accept or reject
            if delta < 0 or rs.rand() < np.exp(-delta / T):
                current_h = new_h.copy()
                current_energy = new_energy
                if current_energy < best_c5:
                    best_c5 = current_energy
                    best_h = current_h.copy()
            T *= alpha

    # Local refinement
    if time.time() - start_time < budget_s * 0.95:
        result = minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            c5 = objective(h_candidate)
            if c5 < best_c5:
                best_h = h_candidate
                best_c5 = c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
