import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_full(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        half_n = len(h_half)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    # Initial best full h
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        h_initial = (h_initial / current_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    best_full = h_initial.copy()
    best_c5 = objective_full(best_full)

    np.random.seed(seed)
    n_attempt = 0

    # Bounds for full variables
    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    cons_full = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    half_n = n_points // 2

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        rnd = np.random.RandomState(seed + n_attempt)
        use_symmetric = rnd.rand() < 0.5

        if use_symmetric:
            half_vars = best_full[:half_n]
            perturbed_half = half_vars + rnd.normal(0, 0.1, half_n)
            full_perturbed = create_full_h(perturbed_half)
        else:
            full_perturbed = best_full + rnd.normal(0, 0.05, n_points)

        # Rescale sum
        total_perturbed = np.sum(full_perturbed)
        if total_perturbed != 0:
            full_perturbed = (full_perturbed / total_perturbed) * target_sum
        full_perturbed = np.clip(full_perturbed, 0, 1)

        result = minimize(
            fun=objective_full,
            x0=full_perturbed,
            method='SLSQP',
            bounds=bounds_full,
            constraints=cons_full,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            new_full = result.x
            if abs(np.sum(new_full) - target_sum) < 1e-6 and np.all((new_full >= -1e-6) & (new_full <= 1.0 + 1e-6)):
                new_corr = np.correlate(new_full, 1 - new_full, mode='full')
                new_c5 = np.max(new_corr) * dx
                if new_c5 < best_c5:
                    best_full = new_full.copy()
                    best_c5 = new_c5

    assert np.all((best_full >= -1e-6) & (best_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_full.tolist(), best_c5, n_points)
