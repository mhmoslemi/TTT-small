import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_c5 = np.inf
    best_h = h.copy()

    max_iter = 500
    eps = 0.1  # Step size for perturbations
    solver_options = {'numeric': False}

    # Store the best solution whenever improvement is found
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute overlap values across all lags
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Identify the worst overlap's lag
        k_max = np.argmax(corr)
        k_lag = k_max - (n_points - 1)

        # Compute gradient for this lag
        g = np.zeros(n_points)
        for x in range(n_points):
            x_shifted = x + k_lag
            if 0 <= x_shifted < n_points:
                g[x] = 1.0 - h[x_shifted]

        # Solve LP to minimize the linearized overlap
        delta_h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(g * delta_h))
        constraints = [
            cp.sum(delta_h) == 0,
            delta_h >= -eps,
            delta_h <= eps
        ]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, **solver_options)
        except:
            continue  # Skip if solver fails

        # Update solution
        h_update = h + delta_h.value
        h_update = np.clip(h_update, 0.0, 1.0)  # Enforce bounds
        h_update = h_update / h_update.sum() * target_sum  # Normalize sum

        # Avoid small numerical errors violating bounds
        h_update = np.clip(h_update, 0.0, 1.0)
        h = h_update

    # Ensure final solution meets constraints
    assert abs(h.sum() - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((h >= -1e-6) & (h <= 1.0 + 1e-6)), "Value bounds violated"
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    return (h.tolist(), final_c5, n_points)
