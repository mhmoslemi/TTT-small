import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial check
    assert abs(np.sum(h) - target_sum) < 1e-6, "Initial h does not satisfy sum constraint"

    # Parameters
    max_iterations = 500
    epsilon = 0.1
    iterations_without_improvement = 0
    best_c5 = None
    best_h = h.copy()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        max_corr_value = corr[max_corr_index]
        current_c5 = max_corr_value * dx

        # Update best solution
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1

        # If no improvement for a while, reduce epsilon
        if iterations_without_improvement > 20:
            epsilon *= 0.5
            iterations_without_improvement = 0
            if epsilon < 1e-5:
                break

        # Determine k_shift
        n = n_points
        k_shift = max_corr_index - (n - 1)

        # Compute gradient
        gradient = np.zeros(n)
        for i in range(n):
            if i + k_shift < n:
                gradient[i] += 1.0 - h[i + k_shift]
            if i - k_shift >= 0:
                gradient[i] -= h[i - k_shift]

        # Solve LP for delta
        delta = cp.Variable(n)
        objective = cp.Minimize(gradient @ delta)
        constraints = [
            cp.sum(delta) == 0,
            delta >= -epsilon,
            delta <= epsilon,
            delta >= -h,  # Ensure h + delta >= 0
            delta <= 1.0 - h  # Ensure h + delta <= 1
        ]
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.GLOP, verbose=False)  # GLOP for LP
        except:
            # In case of solver error, skip this iteration
            continue

        if problem.status != cp.OPTIMAL:
            # Could not find optimal solution; skip
            continue

        # Update h with the solution
        delta_value = delta.value
        h += delta_value

        # Clip h to [0, 1] (numerical errors?) but LP should have prevented this
        h = np.clip(h, 0.0, 1.0)

        # Ensure sum constraint (should be maintained by LP)
        # Slight adjustments due to numerical precision
        h_sum = np.sum(h)
        correction = (target_sum - h_sum) / n_points
        h += correction
        h = np.clip(h, 0.0, 1.0)  # in case correction pushed any out of bounds

    # After loop, return the best solution
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
