    current_sum = np.sum(h_clipped)
    # If current_sum is already correct, return
    if abs(current_sum - target_sum) < 1e-8:
        return h_clipped
    # Compute how much to add to each element to reach target_sum
    # Distribute the needed delta equally
    delta = (target_sum - current_sum) / len(h)
    # Apply delta, but ensure the new values stay within [0, 1]
    # First add delta to all elements
    h_proj = h_clipped + delta
    # Clip again to ensure bounds
    h_proj_clipped = np.clip(h_proj, 0.0, 1.0)
    # Compute how much was adjusted (difference between target_sum and new sum)
    new_sum = np.sum(h_proj_clipped)

    # But this may require multiple iterations. Given time constraints, we'll proceed once.
    # Note: this approach may not always reach the exact target_sum, but it's a reasonable approximation
