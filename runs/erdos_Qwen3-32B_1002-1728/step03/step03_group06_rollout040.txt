    # Perturbed versions
    for i in range(3):
        rs = np.random.RandomState(seed + i)
        perturbation = rs.normal(0, 0.1, n_points)
        perturbed = h_init + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if abs(perturbed_sum) < 1e-10:
            perturbed_scaled = np.full(n_points, 0.5)
        else:
            perturbed_scaled = perturbed * (target_sum / perturbed_sum)
        # Ensure values are clamped again after scaling (in case of overflow)
        perturbed_scaled = np.clip(perturbed_scaled, 0.0, 1.0)
        # Ensure sum is correct after clamping (may need to normalize again, but simplified here)
        initial_guesses.append(perturbed_scaled)

    # Run optimization for each initial guess within time budget
    for guess in initial_guesses:
        if time.time() - start_time >= budget_s * 0.98:
            break
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Ensure the solution meets the constraints due to possible numerical errors
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            if h_candidate_sum < target_sum * 0.9 or h_candidate_sum > target_sum * 1.1:
                # If sum is too off, try to re-scale
                h_candidate = h_candidate * (target_sum / h_candidate_sum)
                h_candidate = np.clip(h_candidate, 0, 1)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    # Final check to ensure constraints
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all(best_h >= -1e-6) and np.all(best_h <= 1.0 + 1e-6)
    return (best_h.tolist(), best_c5, n_points)

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    best_c5 = objective(best_h)
    np.random.seed(seed + 1)

    # Generate structured and perturbed initial guesses
    initial_guesses = [
        h_init,
        np.full(n_points, 0.5),
        np.r_[np.ones(n_points // 2), np.zeros(n_points - n_points // 2)]
    ]

    for i in range(3):  # Add 3 perturbed initializations
        rs = np.random.RandomState(seed + i + 10)
        perturbed = h_init + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        perturbed = np.clip(perturbed * (target_sum / s), 0, 1)
        initial_guesses.append(perturbed)

            break
        result = minimize(
            fun=objective,
            x0=guess,
