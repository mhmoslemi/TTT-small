def generate_random_binary(n_points: int, target_sum: int) -> NDArray:

    indices = np.random.permutation(n_points)
    h[indices[:target_sum]] = 1.0
    return h

def repair_binary(h: NDArray, target_sum: int) -> NDArray:
    h = h.copy()
    current_sum = int(np.sum(h))
    if current_sum == target_sum:
        return h
    elif current_sum < target_sum:
        needed = target_sum - current_sum
        zero_indices = np.where(h <= 0.5)[0]
        selected = np.random.choice(zero_indices, needed, replace=False)
        h[selected] = 1.0
    else:
        needed = current_sum - target_sum
        one_indices = np.where(h > 0.5)[0]
        selected = np.random.choice(one_indices, needed, replace=False)
        h[selected] = 0.0
    return h

def mutate(h: NDArray, mutation_rate: float, target_sum: int) -> NDArray:
    h = h.copy()
    n_points = len(h)
    m = int(mutation_rate * n_points)
    if m == 0:
        return h
    flip_0_to_1 = m // 2
    flip_1_to_0 = m // 2
    if m % 2 == 1:
        flip_0_to_1 += 1
    one_indices = np.where(h > 0.5)[0]
    zero_indices = np.where(h <= 0.5)[0]
    selected_0 = np.random.choice(zero_indices, flip_0_to_1, replace=False)
    selected_1 = np.random.choice(one_indices, flip_1_to_0, replace=False)
    h[selected_0] = 1.0
    h[selected_1] = 0.0
    return repair_binary(h, target_sum)

def crossover(parent_a: NDArray, parent_b: NDArray) -> NDArray:
    mask = np.random.rand(len(parent_a)) < 0.5
    child = np.where(mask, parent_a, parent_b)
    return child

def evaluate_individual(h: NDArray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full')

    selected_pop = population[indices]
    selected_scores = scores[indices]
    best_index = np.argmin(selected_scores)
