def local_optimize(h, n_points, dx, max_iterations=100, step_size=0.05):
    current_energy = None
    best_energy = compute_energy(h)
    for _ in range(max_iterations):
        original_h = h.copy()
        # Find k_max and overlapping pairs
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        # For this code, assuming lag is positive and valid
        if lag < 0 or lag >= n_points:
            continue  # invalid lag, skip
        best_pair = None
        best_move = 0.0
        for i in range(n_points - lag):
            j = i + lag
            sum_ij = h[i] + h[j]
            if sum_ij < 1:
                # move from i to j
                gain = (1 - sum_ij) * step_size
                best_move = (i, j, 'to_j')
                best_pair = (i, j, gain)
            elif sum_ij > 1:
                # move from j to i
                gain = (1 - sum_ij) * (-step_size)
                best_move = (j, i, 'to_j')  # since moving from j to i is same as i,j to j,i
                best_pair = (j, i, gain)
        if best_pair is None:
            # no pair found, break
            break
        else:
            # apply the move
            i, j, _ = best_pair
            sum_ij = h[i] + h[j]
            if sum_ij < 1:
                delta = min(step_size, h[i])  # Ensure we don't go negative
                h[i] -= delta
                h[j] += delta
            else:
                delta = min(step_size, h[j])
                h[i] += delta
                h[j] -= delta
        # Ensure the sum is maintained
        total = h.sum()
        h *= (n_points / 2.0) / total  # normalize in case of floating point errors
        # Check if the energy improved
        new_energy = compute_energy(h)
        if new_energy >= best_energy:
            h = original_h.copy()  # roll back
            break
        best_energy = new_energy
    return h
