import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    n_particles = 80
    particles = []
    particles.append(np.array(h_init).copy())
    for _ in range(n_particles - 1):
        perturbed = 0.5 + np.random.randn(n_points) * 0.05
        perturbed = np.clip(perturbed, 0, 1)
        s = perturbed.sum()
        perturbed *= (target_sum / s)
        particles.append(perturbed)
    particles = np.array(particles)

    velocities = 0.02 * np.random.randn(n_particles, n_points)
    pbest_positions = particles.copy()
    pbest_scores = np.array([compute_c5(p) for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    max_iter = 1000
    inertia = 0.9
    cognitive = 1.2
    social = 1.2

    for iter in range(max_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest_positions[i] - particles[i]) +
                             social * r2 * (gbest_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = particles[i].sum()
            if s < 1e-6:
                particles[i].fill(0)
            else:
                particles[i] *= (target_sum / s)
            score = compute_c5(particles[i])
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest_position = particles[i].copy()
        # Decay parameters
        inertia = 0.9 - iter * (0.5 / max_iter)
        cognitive = max(0.4, 1.2 - 0.002 * iter)
        social = max(0.4, 1.2 - 0.002 * iter)

    assert abs(np.sum(gbest_position) - target_sum) < 1e-6
    assert np.all((gbest_position >= -1e-6) & (gbest_position <= 1.0 + 1e-6))
    return (gbest_position.tolist(), gbest_score, n_points)
