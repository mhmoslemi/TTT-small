    best_c5 = np.max(compute_corrs(h))

    while time.time() - start_time < budget_s * 0.99:
        # Compute worst k_lag
        corrs = compute_corrs(h)
        k_idx_max = np.argmax(corrs)
        k_lag = k_idx_max - (n_points - 1)

        # Construct gradient for worst k_lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_lag
            if 0 <= j < n_points:
                gradient[i] = 1.0 - h[j]

        # Solve LP to minimize overlap
        delta = cp.Variable(n_points)
        constraints = [
            cp.sum(delta) == 0,
            h + delta <= 1.0,
            h + delta >= 0.0
        ]
        try:
            prob = cp.Problem(cp.Minimize(gradient @ delta), constraints)
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            continue

        if prob.status != cp.OPTIMAL:
            continue

        h_candidate = np.clip(h + delta.value, 0.0, 1.0)
        current_sum = np.sum(h_candidate)
        if abs(current_sum - target_sum) > 1e-6:
            delta_sum = target_sum - current_sum
            h_candidate += delta_sum / n_points

        h = h_candidate

        # Update best solution
        current_c5 = np.max(compute_corrs(h))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
