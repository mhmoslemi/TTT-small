import time

import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Setup for convex optimization
    delta_h = cp.Variable(n_points)
    epsilon = 0.1  # Initial perturbation step

    while time.time() - start_time < budget_s * 0.98:
        # Compute overlaps and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        corr_values = corr * dx
        max_index = np.argmax(corr_values)
        max_corr = corr_values[max_index]

        # Determine lag k corresponding to max_index
        n = len(h)
        k = max_index - (n - 1)

        # Calculate gradient for this k
        g = np.zeros(n)
        for x in range(n):
            if 0 <= x + k < n:
                g[x] = (1.0 - h[x + k]) * dx

        # Solve LP to minimize g·delta_h with constraints
        objective = cp.Minimize(delta_h @ g)
        constraints = [
            cp.sum(delta_h) == 0,
            delta_h >= -epsilon,
            delta_h <= epsilon
        ]
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.GLOP)

        if problem.status != cp.OPTIMAL:
            # Backoff if infeasible
            epsilon *= 0.7
            if epsilon < 1e-6:
                break
            continue

        # Update h, ensuring constraints
        new_h = h + delta_h.value
        new_h = np.clip(new_h, 0.0, 1.0)

        # Normalize to target_sum if necessary
        current_sum = np.sum(new_h)
        if abs(current_sum - target_sum) > 1e-6:
            new_h = (new_h / current_sum) * target_sum
        h = np.clip(new_h, 0.0, 1.0)

        # Check if improved
