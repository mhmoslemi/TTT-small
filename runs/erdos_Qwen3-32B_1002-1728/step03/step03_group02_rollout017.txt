    initial_c5 = np.max(np.correlate(h_binary, 1 - h_binary, mode='full')) * dx
    best_c5 = initial_c5
    best_h = h_binary.copy()

    current_h = h_binary.copy()
    current_c5 = initial_c5
    T = 1.0
    alpha = 0.995
    T_min = 1e-4
    np.random.seed(seed)

    while time.time() - start_time < budget_s * 0.98:
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i, j = np.random.choice(ones_idx), np.random.choice(zeros_idx)
        new_h = current_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h, current_c5 = new_h, new_c5
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5
        T *= alpha
        if T < T_min:
            T = 1.0

    final_h = best_h
    final_c5 = best_c5

    assert abs(np.sum(final_h) - target_sum) < 1e-6
    assert np.all((final_h >= -1e-6) & (final_h <= 1.0 + 1e-6))
    return (final_h.tolist(), final_c5, n_points)
