import time
import numpy as np
from scipy.optimize import minimize

def evaluate_energy(h_vec, dx):
    h = h_vec.flatten()
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Diverse initialization
    population = [h_init.copy()]
    np.random.seed(seed)
    for _ in range(15):
        if _ < 5:
            phase = np.linspace(0, 2 * np.pi, n_points)
            h = 0.5 + 0.5 * np.sin(phase)
        else:
            indices = np.random.choice(n_points, int(target_sum), replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
        h_sum = h.sum()
        if h_sum > 0:
            h *= target_sum / h_sum
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_energy = float('inf')
    best_h = h_init.copy()
    
    # Evaluate initial population
    for h in population:
        energy = evaluate_energy(h, dx)
        if energy < best_energy:
            best_energy = energy
            best_h[:] = h[:]
    
    # Simulated Annealing parameters
    T = 100.0
    T_min = 1e-4
    alpha = 0.995
    
    # Main SA loop
    while time.time() - start_time < budget_s * 0.9 and T > T_min:
        for i in range(len(population)):
            current = population[i].copy()
            noise = np.random.randn(n_points) * (T / 100)
            perturbed = np.clip(current + noise, 0, 1)
            p_sum = perturbed.sum()
            if p_sum > 1e-9:
                perturbed *= target_sum / p_sum
                perturbed = np.clip(perturbed, 0, 1)
                old_e = evaluate_energy(current, dx)
                new_e = evaluate_energy(perturbed, dx)
                if new_e < old_e or np.random.rand() < np.exp((old_e - new_e)/T):
                    population[i][:] = perturbed[:]
                    if new_e < best_energy:
                        best_energy = new_e
                        best_h[:] = perturbed[:]
        T *= alpha
    
    # Final optimization
    cons = ({'type': 'eq', 'fun': lambda h: h.sum() - target_sum})
    result = minimize(
        lambda h: evaluate_energy(h, dx),
        x0=best_h,
        bounds=[(0,1)]*n_points,
        constraints=cons,

    )

        best_energy = result.fun
        best_h[:] = result.x
    
    assert abs(best_h.sum() - target_sum) < 1e-6
    return (best_h.tolist(), best_energy, n_points)
