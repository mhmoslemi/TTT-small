    x_values = np.linspace(0, 2, n_points, endpoint=False)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    num_basis = 8  # Number of basis functions (Chebyshev polynomials)
    x_normalized = (x_values - 1.0) / 1.0  # Scale [0, 2] to [-1, 1]

    # Precompute Chebyshev basis matrix (shape: num_basis x n_points)
    T = []
    T.append(np.ones(n_points))  # T0(x) = 1
    if num_basis > 1:
        T1 = 2 * x_normalized
        T.append(T1)
    for i in range(2, num_basis):
        Ti = 2 * x_normalized * T[i - 1] - T[i - 2]
        T.append(Ti)
    T = np.stack(T, axis=0)

    # Initial coefficients: fit to logit(initial_h_values) using pseudo-inverse
    y = np.clip(h, 1e-5, 1 - 1e-5)  # Avoid logit issues at 0/1
    y_logit = np.log(y / (1 - y))
    try:
        T_pinv = np.linalg.pinv(T.T @ T + 1e-6 * np.eye(num_basis)) @ T.T
        coeffs_initial = T_pinv @ y_logit
    except:
        coeffs_initial = np.random.randn(num_basis)

    # Define forward function to compute h from coefficients
    def forward(coeffs):
        linear_comb = np.dot(coeffs, T)  # Shape: (n_points,)
        return 1 / (1 + np.exp(-linear_comb))  # Sigmoid applied

    # Objective function: C5 + penalty for sum constraint
    def objective(coeffs):
        h_current = forward(coeffs)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_corr = np.max(corr)
        sum_h = np.sum(h_current)
        penalty = 10000 * (sum_h - target_sum)**2
        return max_corr * dx + penalty

    # Optimization
    result = minimize(
        fun=objective,
        x0=coeffs_initial,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-12},
        bounds=[(-50, 50)] * num_basis
    )
    h_optimized = forward(result.x)
    corr_result = np.correlate(h_optimized, 1 - h_optimized, mode='full')
    c5_bound = np.max(corr_result) * dx

    # Validate constraints
    assert abs(np.sum(h_optimized) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((h_optimized >= 0) & (h_optimized <= 1 + 1e-6)), "Bounds violated"
    
    return (h_optimized.tolist(), c5_bound, n_points)
