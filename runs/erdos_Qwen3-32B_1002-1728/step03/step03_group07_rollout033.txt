                        grad[i] -= current_h[m]

                # Backtracking line search
                step_size = 0.1
                found = False
                for _ in range(10):
                    h_candidate = current_h - step_size * grad
                    h_clipped = np.clip(h_candidate, 0, 1)
                    s = h_clipped.sum()
                    if s > 1e-6:
                        h_candidate = h_clipped * (target_sum / s)
                    else:
                        h_candidate = h_clipped
                    new_energy = compute_energy(h_candidate)
                    if new_energy < current_energy:
                        current_h[:] = h_candidate
                        current_energy = new_energy
                        found = True
                        break
                    step_size *= 0.5

                # Update best
                if current_energy < best_energy:
                    best_energy = current_energy
                    best_h[:] = current_h.copy()

                # Perturb occasionally to escape local minima
                if rs.rand() < 0.05:
                    noise = rs.normal(0, 0.02, n_points)
                    current_h += noise
                    current_h = np.clip(current_h, 0, 1)
                    s = current_h.sum()
                    if s > 1e-6:
                        current_h *= target_sum / s

        # Perturb best solution and continue
        if time.time() - start_time < budget_s * 0.98:
            perturbed = best_h.copy()
            noise = rs.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed + noise, 0, 1)
            s = perturbed.sum()
            if s > 1e-6:
                perturbed *= target_sum / s
            best_h = perturbed

    # Final feasibility check
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= 0) & (best_h <= 1)), "Bounds violated"
    return (best_h.tolist(), best_energy, n_points)
