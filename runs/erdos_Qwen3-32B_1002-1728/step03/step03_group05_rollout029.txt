import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Enhanced search configuration
    n_workers = 12  # Increased number of search directions
    max_workers_time = int(budget_s * 0.95)
    
    for w in range(n_workers):
        if time.time() - start_time > budget_s:
            break
        # Generate diverse initial guesses with more aggressive perturbation
        rs = np.random.RandomState(seed + w)
        if w < n_workers // 2:
            # First half: perturb initial solution
            perturbed = best_h + rs.normal(0, 0.15, n_points)
        else:
            # Second half: random sampling within bounds
            perturbed = rs.uniform(0, 1, n_points)
        
        # Ensure valid constraint via projection
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if abs(s) > 1e-9:
            perturbed = perturbed * (target_sum / s)
        perturbed = np.clip(perturbed, 0, 1)
        
        # Optimizer configuration with tighter tolerances
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={
                'maxiter': 200,    # Balanced exploration-exploitation
                'ftol': 1e-10,
                'disp': False
            }
        )
        
        if result.success:
            h_candidate = result.x.reshape(n_points)
            # Final constraint validation
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    # Final post-processing: ensure solution validity
    best_h = np.clip(best_h, 0, 1)
    normalization_factor = target_sum / np.sum(best_h)
    best_h = best_h * normalization_factor
    best_h = np.clip(best_h, 0, 1)
    
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
