                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy() if np.random.rand() < 0.5 else p2.copy()
            
            # Mutation
            if np.random.rand() < MUTATION_RATE:
                delta = np.random.normal(0, MUTATION_MAGNITUDE, half_n)
                child += delta
                child = np.clip(child, 0, 1)
                s = np.sum(child)
                if s != 0:
                    child = (child / s) * target_sum_half

            offspring.append(child)
            
        # Form new population
        population = [population[i] for i in idxs[:NUM_PARENTS]] + offspring

    h_full = create_full_h(best_h_half)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0+1e-6))
    assert abs(np.sum(h_full) - n_points/2.0) < 1e-6
    
    return (h_full.tolist(), best_c5, n_points)
