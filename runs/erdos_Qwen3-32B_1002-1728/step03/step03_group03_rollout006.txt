import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    iteration = 0
    max_iterations = 500

    while time.time() - start_time < budget_s and iteration < max_iterations:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_max = max_index - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = i + k_max
            if 0 <= shifted_i < n_points:
                gradient[i] = dx * (1 - h[shifted_i])
            else:
                gradient[i] = 0.0

        delta_h = cp.Variable(n_points)
        objective = cp.Minimize(gradient @ delta_h)
        constraints = [
            h + delta_h <= 1.0,
            h + delta_h >= 0.0,
            cp.sum(delta_h) == 0,
            delta_h <= 0.1,
            delta_h >= -0.1
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h + delta_h.value
            h_new_clipped = np.clip(h_new, 0.0, 1.0)
            difference = target_sum - np.sum(h_new_clipped)
            h = h_new_clipped + difference / n_points
            iteration += 1
        else:
            break

    assert np.all(best_h >= 0.0) and np.all(best_h <= 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
