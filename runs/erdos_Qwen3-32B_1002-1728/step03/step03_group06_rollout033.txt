        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def round_h(h_vec, target_sum):
        m = int(target_sum)
        indices = np.argsort(h_vec)[::-1]
        h_binary = np.zeros_like(h_vec)
        h_binary[indices[:m]] = 1.0
