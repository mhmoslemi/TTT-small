import numpy as np
from scipy.optimize import minimize

def generate_sinusoidal_pattern(n_points, target_sum):
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x / 2))
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum < 1e-6:
        return h * 0.0
    scaling = target_sum / h_sum
    h *= scaling
    h = np.clip(h, 0, 1)
    return h

def generate_linear_ramp(n_points, target_sum):
    mid = n_points // 2
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1.0 - (abs(i - mid) / mid * 2) if mid != 0 else 0
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum < 1e-6:
        return h * 0.0
    scaling = target_sum / h_sum
    h *= scaling
    h = np.clip(h, 0, 1)
    return h
