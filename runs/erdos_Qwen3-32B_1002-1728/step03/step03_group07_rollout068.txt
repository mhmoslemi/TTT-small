Another possible optimization: using vectorization for the gradient computation.

For example, for a given lag k, the valid indices are where i ranges from max(0, -k) to min(n_points-1, n_points-1 -k). So for k >=0: i ranges from 0 to n_points -k -1. For k <0: i ranges from -k to n_points -1.

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy_and_lag(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_energy = np.max(corr) * dx
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        return max_energy, lag

    def perturb_h(h, seed=None):
        rs = np.random.RandomState(seed)
        h_perturbed = h.copy()
        noise = rs.uniform(-0.05, 0.05, h.shape)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = h_perturbed.sum()
        scaling = target_sum / current_sum
        h_perturbed *= scaling
        return h_perturbed

    inits = [h_init]
    for i in range(3):
        inits.append(perturb_h(h_init, seed + i))

    best_h = h_init.copy()
    best_energy, _ = compute_energy_and_lag(best_h)
    max_iterations = 200
    max_line_search_steps = 20
    beta = 0.5
    remaining_time = budget_s * 0.99

    rs = np.random.RandomState(seed)
    for init in inits:
        if time.time() - start_time > budget_s * 0.99:
            break
        current_h = init.copy()
        current_energy, current_lag = compute_energy_and_lag(current_h)
        for iter_idx in range(max_iterations):
            if time.time() - start_time > budget_s * 0.99:
                break
            energy, lag = compute_energy_and_lag(current_h)
            # Compute gradient vector using vectorized indices
            gradient = np.zeros(n_points)
            k = lag
            mask = (0 <= np.arange(n_points) + k) & (np.arange(n_points) + k < n_points)
            valid_i = np.arange(n_points)[mask]
            gradient[valid_i] = 1 - current_h[valid_i + k]
            # Line search
            step_size = 0.1
            improved = False
            for _ in range(max_line_search_steps):
                h_candidate = current_h - step_size * gradient
                h_candidate_clip = np.clip(h_candidate, 0, 1)
                current_sum = h_candidate_clip.sum()
                if current_sum != 0:
                    scaling = target_sum / current_sum
                else:
                    scaling = 1.0
                h_candidate_scaled = h_candidate_clip * scaling
                new_energy, _ = compute_energy_and_lag(h_candidate_scaled)
                if new_energy < current_energy:
                    improved = True
                    break
                else:
                    step_size *= beta
                    if step_size < 1e-6:
                        break
            if improved:
                current_h = h_candidate_scaled
                current_energy = new_energy
                if current_energy < best_energy:
                    best_h = current_h.copy()
                    best_energy = current_energy
        # Perturb after processing this init
        if time.time() - start_time < budget_s * 0.99:
            perturbed = current_h + rs.uniform(-0.02, 0.02, current_h.shape)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = perturbed.sum()
            if current_sum > 1e-9:
                perturbed *= (target_sum / current_sum)
            if compute_energy_and_lag(perturbed)[0] < best_energy:
                best_h = perturbed
                best_energy = compute_energy_and_lag(best_h)[0]

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
