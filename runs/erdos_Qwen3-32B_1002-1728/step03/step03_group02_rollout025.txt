import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    rs = np.random.RandomState(seed)

    def generate_binary():
        indices = rs.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:int(target_sum)]] = 1.0
        return h

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        return evaluate(h)

    population_size = 150
    population = [h_init]
    for _ in range(population_size - 1):
        if _ % 3 == 0:
            population.append(generate_binary())
        else:
            pattern = np.kron([1, 0, 0.5], np.ones(n_points//3)).flatten()[:n_points]
            pattern = (pattern / np.sum(pattern)) * target_sum
            population.append(pattern)

    for i in range(len(population)):
        population[i] /= np.sum(population[i]) * 2.0
        population[i] = np.clip(population[i], 0.0, 1.0)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    
    best_h = None
    best_c5 = float('inf')
    while time.time() - start_time < budget_s * 0.9:
        population.sort(key=fitness)
        elites = population[:population_size//10]

        new_candidates = []
        for elite in elites:
            if np.random.rand() < 0.5:
                perturbation = rs.normal(0, 0.05, n_points)
                perturb_h = np.clip(elite + perturbation, 0, 1)
                perturb_h = perturb_h * (target_sum / np.sum(perturb_h))
            else:
                indices = np.argsort(elite)
                bin_core = np.zeros(n_points)
                bin_core[indices[-int(target_sum):]] = 1.0
                perturb_h = bin_core + rs.uniform(-0.05, 0.05, n_points)
                perturb_h = np.clip(perturb_h, 0, 1)
                perturb_h = perturb_h * (target_sum / np.sum(perturb_h))

            def obj(h):
                return evaluate(h)
                
            result = minimize(obj, perturb_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 60})
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                new_candidates.append(h_opt)

        population += new_candidates
        population.sort(key=fitness)
        population = population[:population_size]

        if population[0].shape[0]:
            current_c5 = fitness(population[0])

    if best_h is None:

        best_c5 = evaluate(best_h)

    return (best_h.tolist(), best_c5, n_points)
