        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Continuous Relaxation
    result_cont = minimize(
        fun=objective_cont,
        x0=h_cont,
        method='L-BFGS-B',
        bounds=bounds_cont,
        constraints=cons_cont,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    h_best = result_cont.x
    c5_best = result_cont.fun

    # Binarization
    h_bin = np.zeros(n_points)
    h_sorted_indices = np.argsort(h_best)[::-1]

    # Simulated Annealing
    rs = np.random.RandomState(seed)
    T_initial = 1.5
    alpha = 0.99
    T_current = T_initial

        return np.max(corr) * dx
