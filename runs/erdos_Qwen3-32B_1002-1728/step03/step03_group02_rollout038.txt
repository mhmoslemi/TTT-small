import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds_continuous = [(0.0, 1.0) for _ in range(n_points)]
    cons_continuous = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    def objective_continuous(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result_continuous = minimize(
        fun=objective_continuous,
        x0=h_init,
        method='SLSQP',
        bounds=bounds_continuous,
        constraints=cons_continuous,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result_continuous.success:
        h_continuous = result_continuous.x
    else:
        h_continuous = h_init

    def binary_h(h_vec, target_sum_bin):
        n_points = len(h_vec)
        h_sorted = np.argsort(h_vec)
        n_ones = int(target_sum_bin)
        h_binary = np.zeros(n_points)
        h_binary[h_sorted[-n_ones:]] = 1.0
        return h_binary

    h_binary = binary_h(h_continuous, target_sum)

    T_initial = 1.0
    alpha = 0.99
    T = T_initial
    T_min = 1e-4
    rs = np.random.RandomState(seed)

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_binary.copy()
    current_h = h_binary.copy()
    best_energy = compute_energy(current_h)

    while T > T_min and (time.time() - start_time) < budget_s * 0.95:
        for _ in range(100):
            if (time.time() - start_time) >= budget_s * 0.99:
                break
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rs.choice(ones)
            j = rs.choice(zeros)
            new_h = current_h.copy()
            new_h[i], new_h[j] = 0.0, 1.0
            current_energy = compute_energy(current_h)
            new_energy = compute_energy(new_h)
            delta_E = new_energy - current_energy
            if delta_E < 0 or np.exp(-delta_E / T) > rs.rand():
                current_h[:] = new_h[:]
                if new_energy < best_energy:
                    best_energy = new_energy
                    best_h[:] = new_h[:]
        T *= alpha

    final_energy = compute_energy(best_h)
    return (best_h.tolist(), final_energy, n_points)
