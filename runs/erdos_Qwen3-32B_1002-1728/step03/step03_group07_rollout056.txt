import time
import numpy as np
from scipy.optimize import line_search

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h_init.copy()
    best_energy = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_energy(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        k_star = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            term1 = 0.0
            if 0 <= m + k_star < n_points:
                term1 = (1.0 - h[m + k_star])
            term2 = 0.0
            if 0 <= m - k_star < n_points:
                term2 = h[m - k_star]
            grad[m] = (term1 - term2) * dx
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        delta = (target_sum - current_sum) / n_points
        h += delta
        return np.clip(h, 0, 1)

    np_random = np.random.RandomState(seed)
    restarts = 5
    max_iter = 200
    perturbation_sigma = 0.1
    for r in range(restarts):
        if r == 0:
            h = best_h.copy()
        else:
            perturbation = np_random.rand(n_points) * 2 - 1
            h = np.clip(best_h + perturbation_sigma * perturbation, 0, 1)
            h = project(h)
        h = project(h)
        current_energy = compute_energy(h)
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.99:
                break
            grad = compute_gradient(h)
            if np.isnan(grad).any():
                continue  # Skip invalid gradients
            h_proposal = h - 0.1 * grad
            h_proposal = project(h_proposal)
            energy_proposal = compute_energy(h_proposal)
            if energy_proposal < best_energy:
                h = h_proposal.copy()
                if energy_proposal < current_energy:
                    current_energy = energy_proposal
                    if energy_proposal < best_energy:
                        best_h = h.copy()

            if i % 50 == 0:

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_energy, n_points)
