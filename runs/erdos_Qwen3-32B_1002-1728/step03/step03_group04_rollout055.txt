import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    total_budget = budget_s
    while time.time() - start_time < total_budget * 0.95:
        rs = np.random.RandomState(seed + int((time.time() - start_time) * 1000))
        elapsed = time.time() - start_time
        progress = elapsed / total_budget
        if progress < 0.5:
            perturbation_type = 'early'
            perturb_scale = max(0.01, 0.1 * (1.0 - progress))
            slsqp_iter = 150
        else:
            perturbation_type = 'late'
            perturb_scale = max(0.01, 0.05 * (1.0 - progress))
            slsqp_iter = 200

        if rs.rand() < 0.5:
