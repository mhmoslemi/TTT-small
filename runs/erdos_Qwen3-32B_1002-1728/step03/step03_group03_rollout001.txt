import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx

    # Feasibility projection
    def project_h(h_vec):
        h_clipped = np.clip(h_vec, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum < 1e-9:
            h_proj = np.zeros_like(h_clipped)
            h_proj[np.random.randint(n_points)] = target_sum
            return h_proj
        scaling = target_sum / current_sum
        return np.clip(h_clipped * scaling, 0.0, 1.0)

    # PSO initialization
    n_particles = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive = 2.0
    social = 2.0
    max_iter = 200

    particles = np.array([project_h(h + np.random.normal(0, 0.1, n_points)) for _ in range(n_particles)])
    velocities = np.random.normal(0, 0.01, (n_particles, n_points))
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([objective(p) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # PSO loop
    for iter in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        t = iter / max_iter
        inertia = inertia_start - t * (inertia_start - inertia_end)
        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        velocities = (inertia * velocities +
                      cognitive * r1 * (personal_best_positions - particles) +
                      social * r2 * (global_best_position - particles))
        particles = particles + velocities
        for i in range(n_particles):
            particles[i] = project_h(particles[i])
            score = objective(particles[i])
            if score < personal_best_scores[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = score
        current_best_idx = np.argmin(personal_best_scores)
        if personal_best_scores[current_best_idx] < global_best_score:
            global_best_position = personal_best_positions[current_best_idx].copy()
            global_best_score = personal_best_scores[current_best_idx]

    # Final refinement
    final_h = project_h(global_best_position)
    if time.time() - start_time < budget_s * 0.98:
        result = minimize(
            fun=objective,
            x0=final_h,
            method='SLSQP',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            final_h = project_h(result.x)
            final_score = objective(final_h)
            if final_score < global_best_score:
                global_best_position = final_h
                global_best_score = final_score

    assert abs(np.sum(global_best_position) - target_sum) < 1e-6
    assert np.all((global_best_position >= -1e-6) & (global_best_position <= 1.0 + 1e-6))
    return (global_best_position.tolist(), global_best_score, n_points)
