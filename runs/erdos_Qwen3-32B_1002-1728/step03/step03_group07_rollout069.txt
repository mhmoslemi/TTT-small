import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    def compute_gradient(h, k_star):
        n = len(h)
        indices = np.arange(n)
        j = indices + k_star
        valid_j = np.logical_and(j >= 0, j < n)
        term1 = np.zeros_like(h)
        term1[valid_j] = 1.0 - h[j[valid_j]]
        k = indices - k_star
        valid_k = np.logical_and(k >= 0, k < n)
        term2 = np.zeros_like(h)
        term2[valid_k] = h[k[valid_k]]
        return term1 - term2

    best_h = h.copy()
    best_energy = compute_energy(h)
    current_h = h.copy()
    np.random.seed(seed)

    iteration = 0
    while time.time() - start_time < budget_s * 0.99:
        # Update max and k*
        corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
        max_index = np.argmax(corr)
        k_star = max_index - (n_points - 1)
        grad = compute_gradient(current_h, k_star)

        # Line search
        best_step_energy = float('inf')
        best_step_h = None
        for step in [0.1, 0.05, 0.025, 0.01, 0.005]:
            candidate = current_h - step * grad
            candidate = np.clip(candidate, 0.0, 1.0)
            current_sum = np.sum(candidate)
            if current_sum < 1e-10:
                continue
            scale = target_sum / current_sum
            candidate *= scale
            energy = compute_energy(candidate)
            if energy < best_step_energy:
                best_step_energy = energy
                best_step_h = candidate.copy()

        # Accept best step
        if best_step_h is not None and best_step_energy < compute_energy(current_h):
            current_h = best_step_h

        # Update best
        current_energy = compute_energy(current_h)
        if current_energy < best_energy:
            best_energy = current_energy
            best_h = current_h.copy()

        # Periodic perturbations
        if iteration % 500 == 0:
            perturbation = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = np.clip(current_h + perturbation, 0.0, 1.0)
            sum_perturbed = np.sum(perturbed)
            if sum_perturbed > 1e-10:
                scale = target_sum / sum_perturbed
                perturbed *= scale
                if compute_energy(perturbed) < best_energy:
                    current_h = perturbed.copy()
        iteration += 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
