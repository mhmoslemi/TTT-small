import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize PSO parameters
    np.random.seed(seed)
    random.seed(seed)  # For random() in velocity update
    n_particles = 30
    inertia = 0.7
    c1 = 0.6  # Cognitive learning rate
    c2 = 0.3  # Social learning rate
    max_vel = 0.5  # Maximum velocity step

    # Initialize particles
    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, size=n_points))
    pbest = h_init.copy()
    # Calculate initial score for the first particle
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    pbest_score = compute_c5(pbest)
    personal_best_positions.append(pbest.copy())
    personal_best_scores.append(pbest_score)
    
    # Generate remaining particles
    for _ in range(n_particles - 1):
        # Binary initialization for diversity
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        particles.append(h)
        # Initial velocities
        velocities.append(np.random.uniform(-0.1, 0.1, size=n_points))
        # Compute personal best
        pbest = h.copy()
        pbest_score = compute_c5(pbest)
        personal_best_positions.append(pbest.copy())
        personal_best_scores.append(pbest_score)
    
    # Set global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    while time.time() - start_time < budget_s * 0.9:
        for i in range(n_particles):
            # Break early if needed
            if time.time() - start_time > budget_s * 0.9:
                break
            
            # Get current particle info
            current_position = particles[i]
            current_velocity = velocities[i]
            
            # Update velocity
            r1 = random.random()
            r2 = random.random()
            cognitive = c1 * r1 * (personal_best_positions[i] - current_position)
            social = c2 * r2 * (global_best_position - current_position)
            current_velocity = inertia * current_velocity + cognitive + social
            current_velocity = np.clip(current_velocity, -max_vel, max_vel)
            
            # Update position
            current_position += current_velocity
            
            # Apply constraints: clip to [0, 1], then rescale to sum
            current_position = np.clip(current_position, 0.0, 1.0)
            current_sum = np.sum(current_position)
            if current_sum == 0:
                # Improbably case, but handle it
                current_position = np.zeros(n_points)
                current_position[:int(target_sum)] = 1.0
                current_position = np.random.permutation(current_position)
            else:
                current_position *= (target_sum / current_sum)
                current_position = np.clip(current_position, 0.0, 1.0)
            
            # Update the particle's position
            particles[i] = current_position
            
            # Compute new score
            current_score = compute_c5(current_position)
            
            # Update personal best if better
            if current_score < personal_best_scores[i]:
                personal_best_positions[i] = current_position.copy()
                personal_best_scores[i] = current_score
                
                # Update global best if necessary
                if current_score < global_best_score:
                    global_best_position = current_position.copy()
                    global_best_score = current_score
        
        # End iteration
        
    # PSO loop finished
    
    # Now take top candidates and refine with SLSQP
    best_candidates = np.argsort(personal_best_scores)[:5]
    best_c5 = global_best_score
    best_h = global_best_position.copy()
    
    for idx in best_candidates:
        pbest_pos = personal_best_positions[idx].copy()
        result = minimize(
            fun=lambda h: compute_c5(h),
            x0=pbest_pos,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            candidate = result.x.reshape(n_points)
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
        # Check time budget
        if time.time() - start_time > budget_s * 0.97:

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

import time
import numpy as np
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    np.random.seed(seed)
    random.seed(seed)
    n_particles = 30
    inertia = 0.7
    c1 = 0.6
    c2 = 0.3
    max_vel = 0.5

    # Initialize particles
    particles = [h_init.copy()]
    velocities = [np.random.uniform(-0.1, 0.1, n_points)]
    personal_best_positions = [h_init.copy()]
    personal_best_scores = [np.max(np.correlate(particles[0], 1 - particles[0], mode='full')) * dx]
    for _ in range(n_particles - 1):
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_score = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        personal_best_positions.append(h.copy())
        personal_best_scores.append(p_score)

    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    while time.time() - start_time < budget_s * 0.9:
        for i in range(n_particles):
            if time.time() - start_time > budget_s * 0.92:
                break

            current_pos = particles[i].copy()
            velocity = velocities[i].copy()

            # Velocity update
            r1, r2 = random.random(), random.random()
            vel_cognitive = c1 * r1 * (personal_best_positions[i] - current_pos)
            vel_social = c2 * r2 * (global_best - current_pos)
            velocity = inertia * velocity + vel_cognitive + vel_social
            velocity = np.clip(velocity, -max_vel, max_vel)

            # Position update
            current_pos += velocity
            current_pos = np.clip(current_pos, 0.0, 1.0)
            current_sum = np.sum(current_pos)
            if current_sum > 1e-8:
                scaling = target_sum / current_sum
                current_pos *= scaling
                current_pos = np.clip(current_pos, 0.0, 1.0)
                particles[i] = current_pos.copy()

                personal_best_scores[i] = current_score
                # Update global best

                    global_best_score = current_score

    best_c5 = global_best_score
    top_indices = np.argsort(personal_best_scores)[:min(5, len(personal_best_scores))]

            break

            x0=pbest,

    return (best_h.tolist(), best_c5, n_points)
