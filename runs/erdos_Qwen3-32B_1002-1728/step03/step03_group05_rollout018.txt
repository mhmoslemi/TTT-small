import time
import numpy as np
from scipy.optimize import minimize

def project_to_sum(h_vec, target_sum, tolerance=1e-6, max_iter=100):
    h = h_vec.astype(float).copy()
    for _ in range(max_iter):
        current_sum = h.sum()
        delta = target_sum - current_sum
        if abs(delta) < tolerance:
            break
        free = (h > 1e-6) & (h < 1 - 1e-6)
        free_indices = np.where(free)[0]
        if not free_indices.size:
            break
        if delta > 0:
            amount = delta / free_indices.size
            h[free_indices] += amount
            h[h > 1.0] = 1.0
        else:
            amount = -delta / free_indices.size
            h[free_indices] -= amount
            h[h < 0.0] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    n_workers = 8
    for w in range(n_workers):
        if time.time() - start_time > budget_s * 0.98:
            break
        rs = np.random.RandomState(seed + w)
        perturbed = best_h + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = project_to_sum(perturbed, target_sum)
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            if abs(np.sum(h_candidate) - target_sum) < 1e-6:
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
