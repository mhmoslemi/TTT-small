        h = (sine + 1) / 2.0
        h_sum = h.sum()
        if h_sum == 0:
            return h.copy()
        scale = target_sum / h_sum
        h_scaled = np.clip(h * scale, 0, 1)

    even = np.zeros(n_points)

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured initial guesses
    def generate_sine(period):
        x = np.linspace(0, 2, n_points)
        omega = 2 * np.pi / period
        sine = np.sin(omega * x) + 1
        h = sine / 2.0
        h_sum = h.sum()
        if h_sum == 0:
            return h.copy()
        h = np.clip(h * (target_sum / h_sum), 0, 1)
        return h

    # Additional initial guesses
    even = np.zeros(n_points)
    even[::2] = 1.0
    odd = np.zeros(n_points)
    odd[1::2] = 1.0
    checkerboard = np.kron([1, 0], np.ones(n_points // 2))[:n_points]
    rs = np.random.RandomState(seed)
    indices = rs.permutation(n_points)
    random1 = np.zeros(n_points)
    random1[indices[:n_points//2]] = 1.0
    sine2_period = generate_sine(2.0)
    sine15_period = generate_sine(1.5)
    sine25_period = generate_sine(2.5)

    all_initials = [
        h_init, even, odd, checkerboard, random1,
        sine2_period, sine15_period, sine25_period
    ]

    # Perturbation function
    def perturb(h, seed=None):
        h_pert = h.copy()
        if seed is not None:
            rs = np.random.RandomState(seed)
        else:
            rs = rs
        h_pert += rs.normal(0, 0.04, h_pert.shape)
        h_pert = np.clip(h_pert, 0, 1)
        sum_h = h_pert.sum()
        if sum_h == 0:
            return h.copy()
        h_pert = h_pert * (target_sum / sum_h)
        h_pert = np.clip(h_pert, 0, 1)
        return h_pert

    # Surrogate objective
    def surrogate_objective(h_vec, alpha):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(len(corr)) - (n_points - 1)
        distances = lags * dx
        weights = np.exp(-alpha * np.abs(distances))
        return np.sum(corr * weights) * dx

    # Binary thresholding
    def binary_threshold(h_vec):
        h = h_vec.reshape(n_points)
        n_ones = int(target_sum)
        indices = np.argsort(h)
        h_bin = np.zeros_like(h)
        h_bin[indices[-n_ones:]] = 1.0
        return h_bin

    best_c5 = np.inf
    best_h = h_init.copy()
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0, 1)] * n_points

        if time.time() - start_time > budget_s * 0.9: 
            break
        for alpha in [0.1, 0.2]:

                return surrogate_objective(h_vec, alpha)

            result = minimize(

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds"
    return (best_h.tolist(), best_c5, n_points)
