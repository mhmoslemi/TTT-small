import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    while time.time() - start_time < budget_s * 0.99:
        elapsed = time.time() - start_time
        time_fraction = elapsed / budget_s
        std_continuous = max(0.005, 0.1 * (1 - time_fraction))
        std_other = 0.05 * (1 - time_fraction)
        rs = np.random.RandomState(seed + int(elapsed * 1000) % 2**32)
        if rs.rand() < 1/3:
            # Continuous perturbation
            perturbed = best_h + rs.normal(0, std_continuous, n_points)
        elif rs.rand() < 2/3:
            # Shift perturbation
            shift_amount = rs.randint(1, n_points // 4 + 1)
            perturbed = np.roll(best_h, shift_amount)
            perturbed += rs.normal(0, std_other, n_points)
        else:
            # Binarize and perturb
            m = int(round(target_sum))
            indices = np.argpartition(best_h, -m)[-m:]
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, std_other, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-9:
            perturbed = (perturbed / perturbed_sum) * target_sum
        else:
            m = int(round(target_sum))
            indices = rs.choice(n_points, size=m, replace=False)
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            if np.all((h_candidate >= 0) & (h_candidate <= 1.0 + 1e-6)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.99:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
