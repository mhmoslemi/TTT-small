            options={'maxiter': 100, 'xatol': 1e-4, 'fatol': 1e-4}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            h_clipped = np.clip(h_candidate, 0, 1)
            h_sum = h_clipped.sum()
            if h_sum < 1e-9:
                continue
            h_normalized = h_clipped * (target_sum / h_sum)
            corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_normalized.copy()
                best_c5 = c5

    # SLSQP refinement
    candidates = [best_h.copy()]
    for cid, h_candidate in enumerate(candidates):
        if time.time() - start_time > budget_s * 0.89:
            break
        result = minimize(
            fun=objective_slsqp,
            x0=h_candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_refined = result.x.reshape(n_points)
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5

    # Local search with perturbations
    while time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s * 0.97:
            break
        perturbed = best_h.copy() + (np.random.rand(n_points) - 0.5) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        psum = perturbed.sum()
        if psum < 1e-9:
            continue
        perturbed = perturbed * (target_sum / psum)
        result = minimize(
            fun=objective_slsqp,
            x0=perturbed,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            method='SLSQP',
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
