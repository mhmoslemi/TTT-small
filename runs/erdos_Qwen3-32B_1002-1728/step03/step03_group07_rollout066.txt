import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_energy_and_kmax(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        k_max_idx = np.argmax(corr)
        return max_val * dx, k_max_idx

    def compute_gradient(h, k_max_idx):
        n = len(h)
        m = k_max_idx - (n - 1)  # Convert correlation index to shift
        gradient = np.zeros(n)
        if 0 <= m < n:
            for i in range(n - m):
                j = i + m
                gradient[i] += (1 - h[j])
                gradient[j] -= h[i]
        return gradient

    def project_to_feasible(h, target_sum=target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-8:
            return h
        diff = target_sum - current_sum
        h += diff / n_points
        h = np.clip(h, 0, 1)
        return h

    grid_size = 50
    np.random.seed(seed)
    grid_points = [h_init.copy()]
    for _ in range(grid_size - 1):
        h_new = h_init.copy()
        while True:
            i, j = np.random.choice(n_points, 2, replace=False)
            if h_new[i] < 1 and h_new[j] > 0:
                break
        delta = np.random.uniform(0.05, 0.2)
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_to_feasible(h_new)
        grid_points.append(h_new)

    best_h = h_init.copy()
    best_energy, _ = compute_energy_and_kmax(best_h)

    for candidate in grid_points:
        if time.time() - start_time > budget_s * 0.99:
            break
        h_opt = candidate.copy()
        energy_current, k_max_idx = compute_energy_and_kmax(h_opt)
