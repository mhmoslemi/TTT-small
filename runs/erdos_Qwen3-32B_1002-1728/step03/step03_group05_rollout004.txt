import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # CMA-ES parameters
    lambda_ = 30  # Population size
    mu = 5        # Number of top candidates for selection
    sigma = 0.3   # Initial step size
    generations = 60  # Maximum generations
    mean = h_init.copy()

    best_c5 = np.inf
    best_h = h_init.copy()

    # Global exploration: Population-based CMA-ES approximation
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.98:
            break

        population = []
        for _ in range(lambda_):
            # Perturb current mean
            perturb = np.random.default_rng(seed + gen).normal(0, sigma, n_points)
            h_candidate = mean + perturb
            # Clip to [0, 1]
            h_clipped = np.clip(h_candidate, 0, 1)
            # Normalize to satisfy sum constraint
            s = np.sum(h_clipped)
            if s > 1e-9:
                h_clipped *= (target_sum / s)
            else:
                h_clipped[:] = target_sum / n_points  # fallback if all values are zero
            population.append(h_clipped)

        # Evaluate population
        candidates = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            candidates.append((c5, h))

        # Sort and select top μ
        candidates.sort()
        top_mu = [c[1] for c in candidates[:mu]]

        # Update mean
        mean = np.mean(top_mu, axis=0)

        # Update best solution
        if candidates[0][0] < best_c5:
            best_c5 = candidates[0][0]
            best_h = top_mu[0]

    # Local refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Final local optimization on the best candidate
    try:
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-10}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            s = np.sum(h_candidate)
            if abs(s - target_sum) > 1e-9:
                h_candidate *= (target_sum / s)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')

            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
    except:

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
