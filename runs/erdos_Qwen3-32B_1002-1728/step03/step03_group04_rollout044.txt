import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_best = np.array(initial_h_values)
    n_points = len(h_best)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=compute_c5,
        x0=h_best,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-6}
    )

    if result.success:
        h_best = result.x
        best_c5 = compute_c5(h_best)
    else:
        best_c5 = compute_c5(h_best)

    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed + int(time.time()))
        m = int(round(target_sum))
        if rs.rand() < 0.8:  # More binary-based starts
            indices = np.random.choice(n_points, size=m, replace=False)
            h_bin = np.zeros(n_points)
            h_bin[indices] = 1.0
            perturbed = h_bin + rs.normal(0, 0.1, n_points)
        else:
            perturbed = h_best + rs.normal(0, 0.1, n_points)

        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-6:
            perturbed = (perturbed / perturbed_sum) * target_sum
            result = minimize(
                fun=compute_c5,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            if result.success:
                h_candidate = result.x
                sum_candidate = np.sum(h_candidate)
                if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                    new_c5 = compute_c5(h_candidate)
                    if new_c5 < best_c5:
                        h_best = h_candidate
                        best_c5 = new_c5
        if time.time() - start_time > budget_s * 0.95:

    assert abs(np.sum(h_best) - target_sum) < 1e-6, "Integral constraint violated"

    return (h_best.tolist(), best_c5, n_points)
