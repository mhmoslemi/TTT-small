import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    best_c5 = objective(best_h)
    
    # First optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success and result.fun < best_c5:
        best_h = result.x.reshape(n_points)
        best_c5 = result.fun

    # Perturbation and multi-start optimization
    while time.time() - start_time < budget_s * 0.98:
        # Generate perturbation
        perturbation = np.random.normal(0, 0.1, n_points)
        x0 = best_h + perturbation
        x0 = np.clip(x0, 0, 1)
        x0_sum = x0.sum()
        if x0_sum > 1e-6:
            x0 = (x0 / x0_sum) * target_sum
            x0 = np.clip(x0, 0, 1)
        else:
            x0 = np.zeros_like(x0)
            x0[0] = target_sum
        
        # Optimization
        local_result = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if local_result.success:
            local_h = local_result.x.reshape(n_points)
            local_c5 = local_result.fun
            if local_c5 < best_c5:
                best_c5 = local_c5
                best_h = local_h

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
