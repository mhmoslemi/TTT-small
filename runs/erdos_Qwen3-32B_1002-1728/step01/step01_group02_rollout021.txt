def evaluate_exact_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def evaluate_smoothed_C5(h_vector, beta, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full') * dx
    max_val = np.max(cross)
    exp_vals = np.exp(beta * (cross - max_val))
    total = np.sum(exp_vals)
    return (1.0 / beta) * np.log(total) + max_val
