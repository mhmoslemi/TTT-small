import numpy as np
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        old_n = len(h)
        old_x = np.linspace(0, 2, old_n)
        new_x = np.linspace(0, 2, n_points)
        h = np.interp(new_x, old_x, h)
        h *= (n_points / 2) / h.sum()
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-5:mid+5] = 0.5
        h *= (n_points / 2) / h.sum()
        h = np.clip(h, 0, 1)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    best_h, best_c5 = h.copy(), compute_c5(h)

    # Simulated Annealing
    T = 1.0
    alpha = 0.995
    sigma = 0.25
    current_h = h.copy()
    current_c5 = best_c5

    while time.time() - start_time < budget_s * 0.9:
        # Perturb
        perturbation = sigma * T * np.random.randn(n_points)
        new_h = np.clip(current_h + perturbation, 0, 1)
        new_h *= (n_points / 2) / new_h.sum()
        new_c5 = compute_c5(new_h)

        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / max(T, 1e-8)) > np.random.rand():
            current_h, current_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5

        T *= alpha
        if T < 1e-8:
            break

    # Local refinement with SLSQP
    def objective(x):
        x = np.clip(x, 0, 1)
        x *= (n_points / 2) / x.sum()
        return compute_c5(x)

    bounds = Bounds(0, 1)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})

    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_h *= (n_points / 2) / refined_h.sum()
        refined_c5 = compute_c5(refined_h)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    return best_h, best_c5, n_points
