        # Solve
        prob = cp.Problem(cp.Minimize(C), constraints)

        # Use a suitable solver
        try:
            prob.solve(solver=cp.CSVM, verbose=False)

            pass  # handle solver errors?

        if h.value is not None and C.value is not None:

            sum_h = np.sum(h_candidate)
            if abs(sum_h - n / 2.0) > 1e-6:
                h_candidate *= (n / 2.0) / sum_h
            new_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if new_c5 < best_c5:
                best_h, best_c5 = h_candidate, new_c5
        h_current = best_h

    return best_h, best_c5, n
