import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    import time
    
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Define bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Repair function to ensure constraints are met
    def repair(x):
        x_clipped = np.clip(x, 0, 1)
        current_sum = np.sum(x_clipped)
        if current_sum == 0:
            return np.full(n_points, target_sum / n_points)
        scale_factor = target_sum / current_sum
        x_scaled = x_clipped * scale_factor
        x_scaled_clipped = np.clip(x_scaled, 0, 1)
        return x_scaled_clipped
    
    def objective(x):
        x_repaired = repair(x)
        corr = np.correlate(x_repaired, 1 - x_repaired, mode='full')
        return np.max(corr) * dx
    
    # Run differential evolution
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=30,  # Tuned to fit within time budget
        popsize=15,  # Balanced exploration and computation
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=False,
        init='random'
    )
    
    optimized_h = result.x
    optimized_h = repair(optimized_h)
    
    corr = np.correlate(optimized_h, 1 - optimized_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Secondary optimization if time remains
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.6:
        # Fine-tune with smaller mutation
        result2 = differential_evolution(
            func=objective,
            bounds=bounds,
            strategy='rand1bin',
            maxiter=10,
            popsize=10,
            mutation=0.2,
            recombination=0.9,
            seed=seed + 1,
            polish=False
        )
        h2 = repair(result2.x)
        corr2 = np.correlate(h2, 1 - h2, mode='full')
        c52 = np.max(corr2) * dx
        if c52 < best_c5:
            optimized_h = h2
            best_c5 = c52
    
    # Final constraint validation
    assert abs(np.sum(optimized_h) - target_sum) < 1e-5, "Sum constraint not satisfied"
    assert np.all(optimized_h >= -1e-6) and np.all(optimized_h <= 1.0 + 1e-6), "Bounds violated"
    
    return (optimized_h.tolist(), best_c5, n_points)
