    np.random.seed(seed)
    h_best = np.array(initial_h_values)
    n_points = len(h_best)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    obj_partial = partial(objective, n_points=n_points, dx=dx)
    cons = [{'type': 'eq', 'fun': partial(constraint_eq, target_sum=target_sum)}]

    h_best_c5 = compute_c5(h_best, n_points, dx)

    while time.time() - start_time < budget_s * 0.98:
        h_pert = perturb(h_best, target_sum, n_points)
        result = minimize(
            fun=obj_partial,
            x0=h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_new = result.x.reshape(n_points)
            if abs(np.sum(h_new) - target_sum) < 1e-6 and np.all((h_new >= 0) & (h_new <= 1)):
                new_c5 = compute_c5(h_new, n_points, dx)
                if new_c5 < h_best_c5:
                    h_best = h_new
                    h_best_c5 = new_c5

    corr = np.correlate(h_best, 1 - h_best, mode='full')
    best_c5 = np.max(corr) * dx
    return (h_best.tolist(), best_c5, n_points)
