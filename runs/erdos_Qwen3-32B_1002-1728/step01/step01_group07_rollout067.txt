        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if np.isclose(sum_clipped, 0):
            return float('inf')
        h_scaled = h_clipped * (target_sum / sum_clipped)
        h_final = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = differential_evolution(
        func=objective_de,
        bounds=bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=10,
        tolerance=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        polish=False,
        init='sobol'
    )
    
    optimized_h_de = result.x.reshape(n_points)
    
    # Post-process to ensure constraints are met
    h_clipped = np.clip(optimized_h_de, 0, 1)
    sum_clipped = np.sum(h_clipped)
    if np.isclose(sum_clipped, 0):
        h_scaled = h_clipped.copy()
    else:
        h_scaled = h_clipped * (target_sum / sum_clipped)
    optimized_h = np.clip(h_scaled, 0, 1)
    
    corr = np.correlate(optimized_h, 1 - optimized_h, mode='full')
    best_c5 = np.max(corr) * dx
