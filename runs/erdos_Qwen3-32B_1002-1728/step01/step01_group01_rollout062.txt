    start_time = time.time()
    n_points = len(initial_h_values)
    h_best = initial_h_values.copy()
    best_c5, _ = evaluate_erdos_solution(h_best)
    
    dx = 2.0 / n_points
    
    while time.time() - start_time < budget_s - 10:  # Leave buffer for final evaluation
        h = h_best.copy()
        h_var = cp.Variable(n_points)
        c = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        for k in range(1, n_points):  # All non-zero shifts
            valid_i = np.arange(n_points - k)
            if len(valid_i) == 0:
                continue
            sum_expr = 0.0
            for i in valid_i:
                j = i + k
                hi_old = h[i]
                hj_old = h[j]
                # Linearized term for h[i](1 - h[j])
                sum_expr += (1 - hj_old) * h_var[i] - hi_old * h_var[j] + hi_old * hj_old
            constraints.append(sum_expr <= c * n_points / 2)
        
        objective = cp.Minimize(c)
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, max_iter=1000, verbose=False)
        except Exception:
            continue
        
        if h_var.value is None:
            continue
        
        h_new = h_var.value.copy()
        h_new = np.clip(h_new, 0, 1)
        scaling = (n_points / 2) / (np.sum(h_new) + 1e-12)  # Avoid division by zero
        h_new *= scaling
        
        current_c5, _ = evaluate_erdos_solution(h_new)
        
        if current_c5 < best_c5:
            h_best = h_new
            best_c5 = current_c5
        
        # Prevent infinite loop with no progress
        if time.time() - start_time > budget_s - 5:
            break
    
    return h_best, best_c5, n_points
