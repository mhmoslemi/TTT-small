def repair_h(h_candidate, target_sum):
    h_clipped = np.clip(h_candidate, 0, 1)
    current_sum = np.sum(h_clipped)
    delta = target_sum - current_sum

    if delta == 0:
        return h_clipped

    indices = np.arange(len(h_candidate))
    # Determine which elements can be adjusted
    if delta > 0:
        # Need to increase elements from 0 to 1
        # Sort indices by current value in ascending order
        idx_sorted = indices[np.argsort(h_clipped)]
        remaining = delta
        for i in idx_sorted:
            amount = max(0, min(remaining, 1 - h_clipped[i]))
            h_clipped[i] += amount
            remaining -= amount
            if remaining <= 1e-10:
                break
    else:

        idx_sorted = indices[np.argsort(-h_clipped)]
        remaining = -delta  # now positive

            amount = max(0, min(remaining, h_clipped[i]))
            h_clipped[i] -= amount
            remaining -= amount
            if remaining <= 1e-10:
                break

    return h_clipped
