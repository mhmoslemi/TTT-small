import numpy as np

def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return np.zeros_like(h)
    h_scaled = h_clipped * (target_sum / current_sum)
    return np.clip(h_scaled, 0, 1)

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
