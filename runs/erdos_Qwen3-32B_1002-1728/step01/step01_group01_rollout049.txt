            idx = np.where(h_clipped < 1)[0]
            if len(idx) == 0:
                return h_clipped
            h_clipped[idx] += delta / len(idx)
        else:
            delta = s - target_sum
            idx = np.where(h_clipped > 0)[0]
            if len(idx) == 0:
                return h_clipped
            h_clipped[idx] -= delta / len(idx)
        return np.clip(h_clipped, 0, 1)

    population_size = 50
    np.random.seed(seed)
    population = [project(initial_h_values + np.random.normal(0, 0.05, n_points)) for _ in range(population_size // 2)]
    for _ in range(population_size // 2):
        h = np.random.rand(n_points)
        population.append(project(h))
    population = np.array(population)
    best_h = population[0].copy()
    best_c5 = objective_function(best_h)

    F = 0.8
    CR = 0.9

    while time() - start_time < budget_s - 10:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            mask = np.random.rand(n_points) < CR
            trial = np.where(mask, mutant, trial)
            trial = project(trial)
            current_c5 = objective_function(population[i])
            trial_c5 = objective_function(trial)
            if trial_c5 < current_c5:
                population[i] = trial
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5

    return best_h, best_c5, n_points
