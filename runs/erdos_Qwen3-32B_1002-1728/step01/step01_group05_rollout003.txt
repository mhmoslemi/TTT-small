        if x0_sum > 1e-9:
            x0 = x0 * target_sum / x0_sum
        x0 = np.clip(x0, 0.0, 1.0)
            
        loc_result = minimize(lambda p: compute_c5(p, dx), x0=x0, method='SLSQP',
                             bounds=bounds_local, constraints=constraints,
                             options={'maxiter': 150, 'ftol': 1e-8})
        if loc_result.success:
            new_h = loc_result.x
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
    
    # Perturbation and restarts
    perturb_amp = 0.2
    perturb_decay = 0.85
    trials = 0
    while time.time() - start_time < budget_s * 0.9:
        perturbation = np.random.uniform(-perturb_amp, perturb_amp, n_points)
        perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if abs(perturbed_sum) > 1e-9:
            perturbed *= target_sum / perturbed_sum
        perturbed = np.clip(perturbed, 0.0, 1.0)
        
        loc_result = minimize(lambda p: compute_c5(p, dx), x0=perturbed, method='SLSQP',
                             bounds=bounds_local, constraints=constraints,
                             options={'maxiter': 100, 'ftol': 1e-6})
        
        if loc_result.success:
            new_h = loc_result.x
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        trials += 1
        perturb_amp *= perturb_decay
    
    # Final feasibility check
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-6:
        best_h = np.clip(best_h * (target_sum / best_h_sum), 0.0, 1.0)
    
    return (best_h, best_c5, n_points)
