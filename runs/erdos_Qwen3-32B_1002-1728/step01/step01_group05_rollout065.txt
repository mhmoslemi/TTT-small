import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds

def project_to_feasible(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, 0.0, atol=1e-9):
        return np.zeros_like(h_clipped)
    h_scaled = h_clipped * (target_sum / current_sum)
    return h_scaled

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def de_objective(x, dx, target_sum):
    h = x.reshape(-1)
    h_proj = project_to_feasible(h, target_sum)
    return compute_c5(h_proj, dx)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)
    
    # Differential Evolution phase (global search)
    if 'initial_h_values' in globals():
        de_bounds = [(0, 1) for _ in range(n_points)]
        pop_size = 10
        de_args = (dx, target_sum)
        try:
            de_result = differential_evolution(
                de_objective,
                de_bounds,
                args=de_args,
                strategy='best1bin',
                maxiter=15,
                popsize=pop_size,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                workers=1,
                polish=False,
                tol=1e-5,
                disp=False,
                updating='deferred'
            )
            if de_result.success:
                de_h = project_to_feasible(de_result.x, target_sum)
                de_c5 = compute_c5(de_h, dx)
                if de_c5 < best_c5:
                    best_h = de_h.copy()
                    best_c5 = de_c5
        except Exception:
            pass
        
    # Local refinement with SLSQP
    result = minimize(
        lambda x: compute_c5(x, dx),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 150, 'ftol': 1e-8}
    )
    if result.success:
        slsqp_h = result.x
        slsqp_c5 = compute_c5(slsqp_h, dx)
        if slsqp_c5 < best_c5:
            best_h = slsqp_h.copy()
            best_c5 = slsqp_c5
            
    # Time-managed iterations
    current_time = time.time()
    remaining = budget_s - (current_time - start_time)

        while perturbations < max_perturbations and (time.time() - start_time) < budget_s:
            perturbed = best_h.copy()
