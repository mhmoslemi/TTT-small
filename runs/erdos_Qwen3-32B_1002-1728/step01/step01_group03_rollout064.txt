def downsample_h(h, new_n):
    old_n = len(h)
    ratio = old_n // new_n
    h_coarse = []
    for i in range(new_n):
        start = i * ratio
        end = start + ratio
        group_sum = np.sum(h[start:end])
        avg = group_sum / ratio
        h_coarse.append(avg)
    return np.array(h_coarse)

def upsample_h(h, new_n):

    h_fine = []
