        h_de = np.clip(result_de.x, 0, 1)
        s = np.sum(h_de)
        h_scaled = h_de * (target_sum / s) if s > 1e-10 else np.copy(h_de)
        h_projected = np.clip(h_scaled, 0, 1)
        if compute_c5(h_projected, dx) < best_c5:
            best_h[:] = h_projected.copy()
            best_c5 = compute_c5(best_h, dx)
    except:
        pass

    # Local refinement with SLSQP
    bounds_sl = Bounds(0.0, 1.0)
    const_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    max_local_tries = 3
    for _ in range(max_local_tries):
        if time.time() - start_time > budget_s * 0.98:
            break
        
        # Initial perturbation
        perturbed = np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = np.clip(perturbed * (target_sum / perturbed_sum), 0, 1) if perturbed_sum > 1e-10 else np.copy(perturbed)
        
        try:
            result_local = scipy_minimize(
                lambda x: compute_c5(x, dx),
                x0=perturbed,
                method='SLSQP',
                bounds=bounds_sl,
                constraints=[const_sum],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result_local.success:
                new_h = result_local.x
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_h[:] = new_h.copy()
                    best_c5 = new_c5
        except:
            pass

    return (best_h, best_c5, n_points)
