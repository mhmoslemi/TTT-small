        freqs = np.arange(1, (p + 1) // 2)
        angles = 2 * np.pi * freqs[:, None] * np.arange(n_points) / p
        real_parts = np.cos(angles)
        avg_real = np.mean(real_parts, axis=0)
        h_unity = 0.5 + 0.5 * avg_real
        h_unity = h_unity * (target_sum / np.sum(h_unity))
        h_unity = np.clip(h_unity, 0.0, 1.0)
        candidates.append(h_unity)

    # Include the initial_h_values
    candidates.append(h0.copy())

    # Evaluate candidates
    best_candidate = None
    best_c5 = float('inf')
    for cand in candidates:
        c5 = compute_c5(cand, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_candidate = cand.copy()

    # Optimization setup
    best_h = best_candidate.copy()
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)

    perturb_amp = 0.1
    perturb_decay = 0.95
    end_time = start_time + budget_s

    while time.time() < end_time:
        perturbation = np.random.uniform(-perturb_amp, perturb_amp, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        if not np.isclose(np.sum(perturbed_h), 0):
            perturbed_h = perturbed_h * (target_sum / np.sum(perturbed_h))
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)

        try:
            result = minimize(lambda x: compute_c5(x, dx), x0=perturbed_h, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 50, 'ftol': 1e-6})
        except Exception:
            continue

        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0.0, 1.0)
            if not np.isclose(np.sum(new_h), 0):
                new_h = new_h * (target_sum / np.sum(new_h))
            new_h = np.clip(new_h, 0.0, 1.0)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

        perturb_amp *= perturb_decay

    return (best_h, best_c5, n_points)
