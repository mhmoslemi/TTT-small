import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.optimize import Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_to_constraints(h_unscaled, target_sum, n_points):
    h_clipped = np.clip(h_unscaled, 0, 0.999999)
    current_sum = np.sum(h_clipped)
    if current_sum < 1e-9:
        return np.clip(h_unscaled, 0, 1)
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    global initial_h_values
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Perturb initial guess
    perturb_indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
    h_initial = h0.copy()
    h_initial[perturb_indices] = np.clip(h_initial[perturb_indices] + np.random.uniform(-0.2, 0.2, size=len(perturb_indices)), 0, 1)
    h_initial = project_to_constraints(h_initial, target_sum, n_points)

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Global search with differential evolution
    de_result = differential_evolution(
        lambda x: compute_c5(project_to_constraints(x, target_sum, n_points), dx),
        bounds,
        strategy='best1bin',
        maxiter=20,
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=False,
        disp=False
    )

    best_de_h = project_to_constraints(de_result.x, target_sum, n_points)
    c5_de = compute_c5(best_de_h, dx)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 0:
        return (best_de_h, c5_de, n_points)

    # Local refinement with SLSQP
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsp = Bounds(0.0, 1.0)
    result = minimize(
        lambda x: compute_c5(x, dx),
        x0=best_de_h,
        method='SLSQP',
        bounds=bounds_slsp,
        constraints=constraints,
        options={
            'maxiter': 100,
            'ftol': 1e-6
        }
    )

    best_h = result.x if result.success else best_de_h
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n_points)
