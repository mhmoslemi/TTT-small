    c1 = 2.0   # cognitive
    c2 = 2.0   # social
    v_clamp = 0.1  # velocity bound

    while time.time() - start_time < budget_s - 5:
        current_scores = np.array([compute_c5(particles[i]) for i in range(n_particles)])
        for i in range(n_particles):
            if current_scores[i] < personal_best_score[i]:
                personal_best_pos[i] = particles[i].copy()
                personal_best_score[i] = current_scores[i]
                if current_scores[i] < gbest_score:
                    gbest_pos = particles[i].copy()
                    gbest_score = current_scores[i]

        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel_cognitive = c1 * r1 * (personal_best_pos[i] - particles[i])
            vel_social = c2 * r2 * (gbest_pos - particles[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            velocities[i] = np.clip(velocities[i], -v_clamp, v_clamp)

            # Update position and maintain constraints
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = particles[i].sum()
            if not np.isclose(current_sum, target_sum, atol=1e-6):
                scale = target_sum / current_sum
                particles[i] *= scale
                particles[i] = np.clip(particles[i], 0, 1)

    return gbest_pos, gbest_score, n_points
