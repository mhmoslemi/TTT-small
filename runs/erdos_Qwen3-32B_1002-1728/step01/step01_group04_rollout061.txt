import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    def rescale_to_sum(h_vec):
        current_sum = np.sum(h_vec)
        if not np.isclose(current_sum, 0):
            scaled = h_vec * (target_sum / current_sum)
            scaled = np.clip(scaled, 0, 1)
        else:
            scaled = np.zeros_like(h_vec)
        return scaled

    best_h = h.copy()
    best_c5 = compute_c5(h)

    lambda_val = 1.0
    lambda_cooling = 0.995
    iterations = 300
    step_size = 0.1
    perturbation = 0.02

    for _ in range(iterations):
        if time.time() - start_time > budget_s - 5:
            break

        x = h - 0.5
        grad = np.zeros_like(x)
        for i in range(n_points):
            xp = x.copy()
            xp[i] += perturbation
            h_p = rescale_to_sum(1 / (1 + np.exp(-lambda_val * xp)))
            cp = compute_c5(h_p)

            xn = x.copy()
            xn[i] -= perturbation
            h_n = rescale_to_sum(1 / (1 + np.exp(-lambda_val * xn)))
            cn = compute_c5(h_n)

            grad[i] = (cp - cn) / (2 * perturbation)

        x = np.clip(x - step_size * grad, -1, 1)
        h = rescale_to_sum(1 / (1 + np.exp(-lambda_val * x)))
        lambda_val *= lambda_cooling

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    def objective(x_vec):
        x = x_vec.reshape(n_points)
        h = rescale_to_sum(1 / (1 + np.exp(-lambda_val * x)))
        return compute_c5(h)

    x_flat = (best_h - 0.5).flatten()
    result = scipy.optimize.minimize(
        objective,
        x0=x_flat,
        bounds=[(-1, 1)] * n_points,
        method='L-BFGS-B',
        options={'maxiter': 20, 'ftol': 1e-6}

    final_x = result.x.reshape(n_points)
    final_h = rescale_to_sum(1 / (1 + np.exp(-lambda_val * final_x)))
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    return best_h, best_c5, n_points
