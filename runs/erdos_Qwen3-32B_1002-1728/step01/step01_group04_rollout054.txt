import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # PSO Parameters
    n_particles = 20
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    velocity_clamp = 0.1

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    def perturb(h0, noise_scale=0.05):
        h = h0 + np.random.normal(0, noise_scale, size=h0.shape)
        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        if s < 1e-10:
            return np.clip(h0 + np.random.normal(0, 0.1, h0.shape), 0, 1)
        h_scaled = h_clipped * (target_sum / s)
        return np.clip(h_scaled, 0, 1)

    def create_feasible(n_points, target_sum):
        h = np.random.rand(n_points)
        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        if s < 1e-10:
            h_scaled = np.zeros(n_points)
            h_scaled[n_points // 2] = target_sum
        else:
            h_scaled = h_clipped * (target_sum / s)
        return np.clip(h_scaled, 0, 1)

    # Initialize swarm
    positions = [perturb(h_initial, 0.05)]
    velocities = [np.zeros(n_points)]
    for _ in range(n_particles - 1):
        h = create_feasible(n_points, target_sum)
        positions.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    personal_best_positions = [p.copy() for p in positions]
    personal_best_values = [compute_c5(p, dx) for p in positions]
    global_best_idx = np.argmin(personal_best_values)
    global_best_value = personal_best_values[global_best_idx]
    global_best_position = personal_best_positions[global_best_idx].copy()

    while time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            current = positions[i].copy()
            current_c5 = compute_c5(current, dx)

            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best_values[i] = current_c5
                personal_best_positions[i] = current.copy()
                if current_c5 < global_best_value:
                    global_best_value = current_c5
                    global_best_position = current.copy()

            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (personal_best_positions[i] - current) +
                social * r2 * (global_best_position - current)
            )
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)

            # Position update
            positions[i] = np.clip(current + velocities[i], 0, 1)
            s = positions[i].sum()
            if s < 1e-10:
                positions[i] = np.zeros(n_points)
                positions[i][n_points // 2] = target_sum
                positions[i] = np.clip(positions[i], 0, 1)
            else:
                positions[i] = np.clip(positions[i] * (target_sum / s), 0, 1)

    return global_best_position, global_best_value, n_points
