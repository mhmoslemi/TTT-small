import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_best = initial_h_values.copy()
    np.random.seed(seed)
    best_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
    
    alpha = 0.05  # learning rate
    patience = 0
    max_patience = 100  # reset learning rate and perturb if no improvement
    perturbation_scale = 0.05  # for injected noise

    while time.time() - start_time < budget_s - 10:
        # Compute current overlaps
        correlate_vals = np.correlate(h_best, 1 - h_best, mode='full')
        current_c5 = np.max(correlate_vals) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            patience = 0
            alpha = min(0.1, alpha * 1.1)  # cautiously increase learning rate
        else:
            patience += 1
            alpha *= 0.99  # decay learning rate

        if patience > max_patience:
            # Perturb to escape local minima
            h_best = np.clip(h_best + np.random.normal(0, perturbation_scale, n_points), 0, 1)
            h_best = h_best / np.sum(h_best) * (n_points / 2)
            patience = 0
            alpha = 0.05  # reset learning rate

        # Select top 5% of shifts or 1 shift if empty
        top_quantile = np.quantile(correlate_vals, 0.95)
        top_indices = np.where(correlate_vals >= top_quantile)[0]
        if len(top_indices) == 0:
            top_indices = [np.argmax(correlate_vals)]

        # Analytical gradient from top shifts
        gradient = np.zeros(n_points)
        for idx in top_indices:
            k_real = idx - (n_points - 1)
            for m in range(n_points):
                term1 = 0.0
                if 0 <= (m + k_real) < n_points:
                    term1 = 1 - h_best[m + k_real]
                term2 = 0.0
                if 0 <= (m - k_real) < n_points:
                    term2 = -h_best[m - k_real]
                gradient[m] += term1 + term2
        
        if len(top_indices) > 0:
            gradient /= len(top_indices)

        # Gradient step + projection
        h_proposed = h_best - alpha * gradient
        h_proposed = np.clip(h_proposed, 0.0, 1.0)
        curr_sum = np.sum(h_proposed)
        if abs(curr_sum) > 1e-10:
            h_proposed *= (n_points / 2.0) / curr_sum
        else:
            h_proposed[:] = 0.5

        # Validate and update
        h_best = h_proposed

        if time.time() - start_time > budget_s - 5:
            break

    final_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
    return h_best, final_c5, n_points
