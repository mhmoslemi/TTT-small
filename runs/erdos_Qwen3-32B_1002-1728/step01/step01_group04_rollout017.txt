import numpy as np
import time

def compute_c5(h_vec, dx_val):
    corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
    return np.max(corr)

def clip_and_scale(h, target_sum):
    clipped = np.clip(h, 0, 1)
    current_sum = clipped.sum()
    if current_sum == 0:
        scaled = np.zeros_like(h)
    else:
        scaled = clipped * (target_sum / current_sum)
    scaled_clipped = np.clip(scaled, 0, 1)
    return scaled_clipped
