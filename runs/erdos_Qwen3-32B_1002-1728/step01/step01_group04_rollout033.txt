import numpy as np
import scipy.optimize

def run(seed=42, budget_s=1100):

    h0 = initial_h_values
    n_initial = len(h0)
    dx_initial = 2.0 / n_initial

    K = 4
    n_per = n_initial // K
    remainder = n_initial % K
    start_idx = 0
    s_list = []
    e_list = []
    a_list = []
    for i in range(K):
        n_points = n_per + 1 if i < remainder else n_per
        end_idx = start_idx + n_points
        segment = h0[start_idx:end_idx]
        a_i = np.mean(segment)
        s_i = start_idx * dx_initial
        e_i = end_idx * dx_initial
        s_list.append(s_i)
        e_list.append(e_i)
        a_list.append(a_i)
        start_idx = end_idx
    variables = []
    for i in range(K):
        variables.extend([s_list[i], e_list[i], a_list[i]])
    variables = np.array(variables)

    def objective(x):
        h = np.zeros(n_initial)
        for i in range(K):
            s_i = x[3 * i]
            e_i = x[3 * i + 1]
            a_i = x[3 * i + 2]
            left = int(np.floor(s_i / dx_initial))
            right = int(np.ceil(e_i / dx_initial)) + 2
            for j in range(left, min(right, n_initial)):
                x_j = j * dx_initial
                if s_i <= x_j < e_i:
                    h[j] = a_i
        corr = np.correlate(h, 1 - h, mode='full') * dx_initial
        return np.max(corr)

    constraints = []
    for i in range(K - 1):
        def cons_func(x, idx=i):
            return x[3 * idx + 1] - x[3 * (idx + 1)]
        constraints.append({'type': 'ineq', 'fun': cons_func})
    def e_last(x):
        return x[3 * (K - 1) + 1] - 2.0
    constraints.append({'type': 'ineq', 'fun': e_last})
    def area(x):
        total = 0.0
        for i in range(K):
            s, e, a = x[3*i:3*i+3]
            total += a * (e - s)
        return total - 1.0
    constraints.append({'type': 'eq', 'fun': area})

    bds = []
    for _ in range(K):
        bds.extend([(0.0, 2.0), (0.0, 2.0), (0.0, 1.0)])

    best_c5 = np.inf
    best_h = None

    result = scipy.optimize.minimize(

        method='SLSQP',
