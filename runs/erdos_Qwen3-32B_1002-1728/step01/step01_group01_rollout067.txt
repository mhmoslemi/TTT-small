import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import time

    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_best = initial_h_values.copy()
    
    alpha = 0.01  # Step size
    max_iterations = 2000
    max_time = budget_s - 10
    iterations_without_progress = 0
    max_no_progress = 100
    alpha_decay = 0.95
    alpha_increase = 1.1
    perturb_std = 0.05

    def compute_worst_shift(h):
        best_c5 = -1.0
        best_k = 0
        n = len(h)
        for k in range(n):
            shifted = np.roll(h, -k)
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > best_c5:
                best_c5 = overlap
                best_k = k
        return best_k, best_c5
    
    def compute_gradient(h, k):
        shifted_forward = np.roll(h, -k)
        shifted_backward = np.roll(h, k)
        return (1 - shifted_forward) - shifted_backward

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        target = n_points / 2.0
        delta = (target - current_sum) / n_points
        h_clipped += delta
        return np.clip(h_clipped, 0, 1)

    best_k, best_c5 = compute_worst_shift(h_best)

    for _ in range(max_iterations):
        if time.time() - start_time > max_time:
            break

        k, worst_c5 = compute_worst_shift(h_best)
        grad = compute_gradient(h_best, k)
        h_candidate = h_best - alpha * grad
        h_projected = project(h_candidate)
        new_k, new_c5 = compute_worst_shift(h_projected)

        if new_c5 < best_c5:
            h_best = h_projected
            best_c5 = new_c5
            iterations_without_progress = 0
            alpha = min(alpha * alpha_increase, 0.2)
        else:
            iterations_without_progress += 1
            alpha *= alpha_decay

        if iterations_without_progress >= max_no_progress:
            h_best = np.clip(h_best + np.random.normal(0, perturb_std, n_points), 0, 1)
            h_best = project(h_best)
            iterations_without_progress = 0
            alpha = 0.01

    return h_best, best_c5, n_points
