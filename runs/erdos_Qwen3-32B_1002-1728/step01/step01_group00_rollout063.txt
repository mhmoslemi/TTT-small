def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    population_size = 50
    population = []
    for _ in range(population_size):
        candidate = h_initial.copy()
        noise = np.random.normal(0.0, 0.05, n_points)
        candidate += noise
        candidate = np.clip(candidate, 0, 1)
        current_sum = candidate.sum()
        required_sum = n_points / 2
        delta = required_sum - current_sum
        if abs(delta) > 1e-10:
            candidate[0] += delta
            candidate = np.clip(candidate, 0, 1)
        population.append(candidate)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()

    elite_size = 5
    while time.time() - start_time < budget_s - 10:
        scores = [compute_c5(h, n_points) for h in population]
        indexed = list(enumerate(scores))
        indexed.sort(key=lambda x: x[1])
        sorted_indices = [idx for idx, _ in indexed]
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_scores = [scores[i] for i in sorted_indices]

        if sorted_scores[0] < best_c5:
            best_c5 = sorted_scores[0]
            best_h[:] = sorted_pop[0].copy()

        new_pop = sorted_pop[:elite_size]
        remaining = population_size - elite_size

        while len(new_pop) < population_size:
            p1 = tournament_selection(sorted_pop, sorted_scores, k=5)
            p2 = tournament_selection(sorted_pop, sorted_scores, k=5)
            child = crossover(p1, p2, n_points)
            child = mutate(child, n_points)
            new_pop.append(child)

        population = new_pop.copy()

    return best_h, best_c5, n_points
