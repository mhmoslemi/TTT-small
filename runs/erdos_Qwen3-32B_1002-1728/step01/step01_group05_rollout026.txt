            x_proj = best_h
        
        # Local refinement with SLSQP
        def local_obj(x): return compute_c5(x, dx)
        def sum_constraint(x): return np.sum(x) - target_sum
        bounds_local = Bounds(0.0, 1.0)
        constraints = {'type': 'eq', 'fun': sum_constraint}
        
        perturb_amp = 0.1
        perturb_decay = 0.95
        max_local_trials = int(budget_s // 20)
        
        for trials in range(max_local_trials):
            if trials > 0 and trials % 10 == 0:
                perturbation = np.random.uniform(-perturb_amp, perturb_amp, n_points)
                x_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
                x_perturbed = project(x_perturbed, target_sum)
                result = minimize(local_obj, x_perturbed, method='SLSQP',
                                  bounds=bounds_local, constraints=constraints,
                                  options={'maxiter': 100, 'ftol': 1e-6})
            else:
                result = minimize(local_obj, best_h, method='SLSQP',
                                  bounds=bounds_local, constraints=constraints,
                                  options={'maxiter': 100, 'ftol': 1e-6})
            
            if result.success:
                new_h = result.x
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
            perturb_amp *= perturb_decay
            
            if time.time() - start_time > budget_s:
                break
        
    except Exception as e:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = project(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h, final_c5, n_points)
