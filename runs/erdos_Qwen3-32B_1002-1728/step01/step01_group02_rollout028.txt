import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h_vector, n_points):
    dx = 2.0 / n_points
    h = h_vector.reshape(n_points)
    ones_h = 1 - h
    cross = np.correlate(h, ones_h, mode='full')
    return np.max(cross * dx)

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.full(n_points, 0.5, dtype=np.float64)

    perturbations = []
    for _ in range(5):
        h_pert = h_initial.copy()
        h_pert += np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert_sum = h_pert.sum()
        if h_pert_sum > 1e-8:
            h_pert *= target_sum / h_pert_sum
            h_pert = np.clip(h_pert, 0, 1)
        perturbations.append(h_pert)

    best_c5 = evaluate_C5(h_initial, n_points)
    best_h = h_initial.copy()

    for pert in perturbations:
        h = pert.copy()
        lambda_ = 0.0
        rho = 1e-4
        beta_progression = [2**i for i in range(11)]
        
        for beta in beta_progression:
            if time.time() - start_time > budget_s - 10:
                break

            def objective(x):
                x = x.reshape(n_points)
                cross = np.correlate(x, 1 - x, mode='full')
                cross_scaled = cross * dx
                log_sum_exp = np.log(np.exp(beta * cross_scaled).sum()) / beta
                sum_x = x.sum()
                constraint = sum_x - target_sum
                return log_sum_exp + lambda_ * constraint + 0.5 * rho * constraint**2

            bounds = [(0., 1.) for _ in range(n_points)]
            result = minimize(objective, h, method='trust-constr', bounds=bounds, 
                             options={'maxiter': 30, 'xtol': 1e-6})
            
            if result.success:
                h = result.x.reshape(n_points)
                h = np.clip(h, 0, 1)
                h_sum = h.sum()
                if abs(h_sum - target_sum) > 1e-8:
                    h = (h / h_sum) * target_sum
                    h = np.clip(h, 0, 1)
                
                current_c5 = evaluate_C5(h, n_points)
                if current_c5 < best_c5:
                    best_c5, best_h = current_c5, h.copy()
                
                violation = h.sum() - target_sum
                lambda_ += rho * violation
                if violation > 1e-4:
                    rho *= 2.0

    return best_h, best_c5, n_points
