                    population[i] = rand_h
                    current_c5[i] = rand_c5

        # Update best solution
        idx = np.argmin(current_c5)
        if current_c5[idx] < best_c5:
            best_h = population[idx].copy()
            best_c5 = current_c5[idx]

    # Final checks
    assert np.all((best_h >= 0) & (best_h <= 1)), "Values out of [0, 1]!"
    assert np.isclose(best_h.sum(), n_points / 2), "Sum constraint violated!"
    
    return best_h, best_c5, n_points
