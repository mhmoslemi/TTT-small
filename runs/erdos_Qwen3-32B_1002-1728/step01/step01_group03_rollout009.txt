            corr = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(corr)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        current_best = best_h.copy()
        current_c5 = objective(current_best)
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining < 5:
            break
        
        for _ in range(5):
            if time.time() - start_time >= budget_s:
                break
            x0 = get_perturbed(current_best, target_sum, noise=0.15 if n_points == 45 else 0.05)
            res = minimize(
                objective, x0=x0, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            if res.success:
                new_h = res.x.copy()
                new_c5 = objective(new_h)
                if new_c5 < current_c5:
                    current_best = new_h.copy()
                    current_c5 = new_c5
        
        if best_c5 is None or current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    final_c5 = evaluate_erdos_solution(best_h)[0]
    return (best_h, final_c5, len(best_h))
