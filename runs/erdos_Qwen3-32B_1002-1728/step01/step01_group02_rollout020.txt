def evaluate_objective_log_sum_exp(h_vector, n_points, dx, beta):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    overlaps = cross * dx
    exp_vals = np.exp(beta * overlaps)
    return np.log(np.sum(exp_vals)) / beta
