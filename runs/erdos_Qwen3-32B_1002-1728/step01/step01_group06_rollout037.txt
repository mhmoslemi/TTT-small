import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Use initial_h_values directly if available
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        n = len(h0)
        # Ensure constraint: sum(h) = n/2
        h0 = np.clip(h0, 0, 1)
        h0 *= (n / 2) / h0.sum()
    else:
        n = 280
        h0 = np.zeros(n)
        h0[n//2-10:n//2+10] = 0.25
        h0 *= (n / 2) / h0.sum()
        h0 = np.clip(h0, 0, 1)

    # PSO parameters
    swarm_size = 50
    dims = n
    max_iter = 500
    w = 0.9  # Inertia
    c1, c2 = 1.5, 1.5
    velocities = 0.1 * np.random.randn(swarm_size, dims)
    positions = np.array([perturb_h(h0) for _ in range(swarm_size)])
    personal_best_positions = positions.copy()
    personal_best_scores = np.array([compute_c5(h) for h in positions])
    gbest_idx = personal_best_scores.argmin()
    gbest_position = personal_best_positions[gbest_idx]
    gbest_score = personal_best_scores[gbest_idx]

    # Velocity bounds
    v_min, v_max = -0.5, 0.5

    for iteration in range(max_iter):
        if time.time() - start_time > max_time * 0.98:
            break

        # Decay inertia weight
        w = 0.9 - (0.9 - 0.4) * iteration / max_iter

        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(dims), np.random.rand(dims)
            cognitive = c1 * r1 * (personal_best_positions[i] - positions[i])
            social = c2 * r2 * (gbest_position - positions[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], v_min, v_max)

            # Update position
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)

            # Normalize to meet sum constraint
            positions[i] *= (n / 2) / positions[i].sum()
            positions[i] = np.clip(positions[i], 0, 1)

        # Evaluate all particles
        scores = np.array([compute_c5(h) for h in positions])

        # Update personal bests
        improved = scores < personal_best_scores
        personal_best_scores[improved] = scores[improved]
        personal_best_positions[improved] = positions[improved].copy()

        # Update global best
        current_gbest_idx = scores.argmin()
        if scores[current_gbest_idx] < gbest_score:
            gbest_score = scores[current_gbest_idx]
            gbest_position = positions[current_gbest_idx].copy()

    return gbest_position, gbest_score, n

def perturb_h(h, noise_scale=0.05):
    """Create a small noisy perturbation of h while preserving basic properties"""
    noise = noise_scale * (2 * np.random.rand(*h.shape) - 1)
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new *= (len(h) / 2) / h_new.sum()
    return h_new

def compute_c5(h):
    """Calculate C5 for a discretized h vector"""
    c5 = correlate(h, 1 - h, mode='full') * (2.0 / len(h))
    return np.max(c5)
