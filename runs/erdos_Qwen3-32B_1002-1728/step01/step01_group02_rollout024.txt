import numpy as np
import time
from scipy.optimize import minimize

def repair_h(h, required_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - required_sum) < 1e-8:
        return h_clipped
    scale = required_sum / current_sum
    adjusted = h_clipped * scale
    return np.clip(adjusted, 0.0, 1.0)

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.zeros(n_points).astype(np.float64)
        h_initial[:] = 0.5

    h_initial = repair_h(h_initial, required_sum)

    n_candidates = 5
    candidates = []
    for _ in range(n_candidates):
        perturbed = h_initial.copy()
        perturbed += np.random.normal(0.0, 0.1, n_points)
        perturbed = repair_h(perturbed, required_sum)
        candidates.append(perturbed)

    best_c5 = evaluate_C5(h_initial, n_points, dx)
    best_h = h_initial.copy()

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - required_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    beta_sequence = [2**i for i in range(0, 11)]

    for candidate in candidates:
        if time.time() - start_time > budget_s - 20:
            break
        current_h = candidate.copy()
        for beta in beta_sequence:
            def smoothed_obj(h_vec):
                h = h_vec.reshape(n_points)
                omh = 1.0 - h
                cross = np.correlate(h, omh, mode='full')
                c = cross * dx
                exp_beta_c = np.exp(beta * c)
                return (1.0 / beta) * np.log(np.sum(exp_beta_c))
            result = minimize(smoothed_obj, current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if not result.success:
                current_h = repair_h(result.x, required_sum)
            else:
                current_h = repair_h(result.x, required_sum)
            current_c5 = evaluate_C5(current_h, n_points, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

    n_fine_candidates = 2
    for _ in range(n_fine_candidates):
        if time.time() - start_time > budget_s - 5:
            break
        perturbed = best_h.copy()
        perturbed += np.random.normal(0.0, 0.05, n_points)
        perturbed = repair_h(perturbed, required_sum)
        def exact_obj(h_vec):
            return evaluate_C5(h_vec, n_points, dx)
        result = minimize(exact_obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            current_h = repair_h(result.x, required_sum)
            current_c5 = evaluate_C5(current_h, n_points, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

    return best_h, best_c5, n_points
