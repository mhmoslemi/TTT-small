        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            return float('inf')
        h_clipped *= target_sum / s
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Repair the initial solution and use as part of initial population
    h0 = np.clip(h_init, 0, 1)
    s = h0.sum()
    h0 *= target_sum / s
    
    # Create initial population
    pop_size = 20
    init_pop = np.zeros((pop_size, n_points))
    init_pop[0] = h0
    for i in range(1, pop_size):
        init_pop[i] = np.random.uniform(0, 1, n_points)
    
    # Track time to adjust iterations if needed
    maxiter = 100
    start_time = time.time()
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=pop_size,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=maxiter,
        polish=False,
        disp=False,
        tol=1e-6,
        seed=seed,
        init=init_pop
    )
    
    elapsed = time.time() - start_time
    
    # Final repair
    h_opt = result.x.reshape(n_points)
    h_clipped = np.clip(h_opt, 0, 1)
    s = h_clipped.sum()
    if s == 0:
        h_clipped = np.zeros(n_points)
    else:
        h_clipped *= target_sum / s
    best_c5 = objective(h_clipped)
    
    return (h_clipped, best_c5, n_points)
