import numpy as np

def project_sum(h, target_sum, n_points):
    """Project vector h to [0,1]^n with sum constraint"""
    h = h.copy()
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < 1e-9:
        return h

    indices = np.arange(n_points)
    
    if delta > 0:
        # Need to increase values in h by delta
        
        # Sort indices by current h value (ascending)
        indices = indices[np.argsort(h)]
        for i in indices:
            if h[i] >= 1.0:
                continue
            available = 1.0 - h[i]
            add = min(delta, available)
            h[i] += add
            delta -= add
            if delta < 1e-9:
                break
    else:
        # Need to decrease values in h by |delta|
        delta = -delta
        
        # Sort indices by current h value (descending)
        indices = indices[np.argsort(-h)]
        for i in indices:
            if h[i] <= 0.0:
                continue
            available = h[i]
            subtract = min(delta, available)
            h[i] -= subtract
            delta -= subtract
            if delta < 1e-9:
                break
    return h
