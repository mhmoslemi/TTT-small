import time
from scipy.optimize import minimize

# Global state variables for the objective function
beta = 0.0
n_points = 0
dx = 0.0

def objective_lse(h_vec):
    global beta, n_points, dx
    h = h_vec.reshape(n_points)
    ones_h = 1.0 - h
    cross = np.correlate(h, ones_h, mode='full')
    c_k = cross * dx
    exp_vals = np.exp(beta * c_k)
    total = np.sum(exp_vals)
    if beta == 0.0:
        return np.max(c_k)
    return (1.0 / beta) * np.log(total)

    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)
