import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    import time
    
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    bounds = tuple((0.0, 1.0) for _ in range(n_points))
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    
    # Project h into feasible space respecting constraints
    def project_h(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        total = max(1e-9, float(np.sum(h)))
        h = (h / total) * target_sum
        return h
    
    best_h = project_h(best_h)
    best_c5 = objective(best_h)
    
    while True:
        # Generate perturbed initial guess
        rs = np.random.RandomState(seed)
        seed += 1
        perturbed = best_h + rs.normal(0, 0.1, n_points)
        perturbed = project_h(perturbed)
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            candidate_h = project_h(result.x)
            candidate_c5 = objective(candidate_h)
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
        
        if time.time() - start_time > budget_s - 0.1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
