import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        full_corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(full_corr)

    population_size = 50
    F = 0.8
    crossover_rate = 0.9

    # Initialize population with initial_h_values and random feasible candidates
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.uniform(0, 1, n_points)
        h_dictate_sum = h_rand * (target_sum / h_rand.sum())
        h_clipped = np.clip(h_dictate_sum, 0, 1)
        population.append(h_clipped)
    population = np.array(population)

    # Evaluate fitness of all population members
    fitness = np.array([compute_c5(h, dx) for h in population])

    best_index = np.argmin(fitness)
    best_h, best_c5 = population[best_index].copy(), fitness[best_index]

    while time.time() - start_time < budget_s - 5:
        for i in range(population_size):
            # Select 3 unique individuals different from i
            indices = np.random.choice([j for j in range(population_size) if j != i], 3, replace=False)
            a, b, c = indices
            x_a = population[a]
            x_b = population[b]
            x_c = population[c]
            mutant = x_a + F * (x_b - x_c)
            trial = np.copy(population[i])
            mask = np.random.rand(n_points) < crossover_rate
            trial[mask] = mutant[mask]

            # Enforce constraints
            trial_clipped = np.clip(trial, 0, 1)
            trial_sum = trial_clipped.sum()
            if trial_sum == 0:
                continue
            trial_scaled = trial_clipped * (target_sum / trial_sum)
            trial_final = np.clip(trial_scaled, 0, 1)

            # Selection
            trial_c5 = compute_c5(trial_final, dx)
            if trial_c5 < fitness[i]:
                population[i] = trial_final
                fitness[i] = trial_c5
                if trial_c5 < best_c5:
                    best_h, best_c5 = trial_final.copy(), trial_c5

    # Final constraint check
    final_best_sum = best_h.sum()
    if not np.isclose(final_best_sum, target_sum, atol=1e-6):
        best_h = best_h * (target_sum / final_best_sum)
        best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5, n_points
