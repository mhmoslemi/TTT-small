def run(seed=42, budget_s=1100):

                s = h_clipped.sum()

                else:
                    h_scaled = h_clipped * (target_sum / s)
                h_scaled = np.clip(h_scaled, 0, 1)

                break

    global_best_fitness = np.min(personal_best_fitness)
    global_best_pos = personal_best_pos[personal_best_fitness.argmin()].copy()

    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive (personal) coefficient
    c2 = 2.0  # social (global) coefficient

            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = c1 * r1 * (personal_best_pos[i] - particles[i])

    return global_best_pos, global_best_fitness, n_points

        personal_best_pos[improved] = particles[improved]
        personal_best_fitness[improved] = new_fitness[improved]
        current_min = np.min(personal_best_fitness)
        if current_min < global_best_fitness:
            global_best_fitness = current_min
            global_best_pos = personal_best_pos[personal_best_fitness.argmin()].copy()

    return global_best_pos, global_best_fitness, n_points
