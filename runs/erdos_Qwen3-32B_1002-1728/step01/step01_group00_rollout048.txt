    scores = [fitness[i] for i in indices]
    winner_idx = np.argmin(scores)
    return candidates[winner_idx]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    population_size = 50
    mutation_rate = 0.1
    noise_std = 0.05
    tournament_size = 5
    elitism_count = 5

    # Initialize population with small perturbations around initial solution
    population = []
    for _ in range(population_size):
        h = h_initial.copy()
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        # Clip and adjust sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        h[0] += delta
        h = np.clip(h, 0, 1)
        population.append(h)

    best_c5 = np.inf
    best_h = h_initial.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        # Evaluate population
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Select elites
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:elitism_count]]

        new_population = elites.copy()
        while len(new_population) < population_size:
            # Select parents via tournament
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)

            # Crossover with blend (alpha=0.5)
            alpha = 0.5
            child = alpha * parent1 + (1 - alpha) * parent2

            # Mutation: perturb 10% of elements
            mask = np.random.rand(n_points) < mutation_rate
            child += mask * np.random.normal(0, noise_std, n_points)
            child = np.clip(child, 0, 1)

            # Adjust sum constraint
            current_sum = np.sum(child)
            delta = target_sum - current_sum
            child[0] += delta
            child = np.clip(child, 0, 1)
            new_population.append(child)

        population = new_population

    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr)
    return best_h, c5_bound, n_points
