import numpy as np
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def perturb(h, noise=0.15):
        new_h = h.copy()
        new_h += np.random.normal(0, noise, len(h))
        new_h = np.clip(new_h, 0, 1)
        total = np.sum(new_h)
        if total != 0:
            new_h *= (target_sum / total)
        new_h = np.clip(new_h, 0, 1)
        return new_h
    
    def crossover(parent1, parent2):
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        total = child.sum()
        if total != 0:
            child *= (target_sum / total)
        child = np.clip(child, 0, 1)
        return child
    
    population_size = 20
    population = [perturb(h_init, noise=0.2) for _ in range(population_size)]
    
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    start_time = time.time()
    max_generations = 100
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 60:
            break
        evaluations = [(compute_c5(ind), ind) for ind in population]
        evaluations.sort()
        if evaluations[0][0] < best_c5:
            best_c5 = evaluations[0][0]
            best_h = evaluations[0][1].copy()
        
        top_half = [e[1] for e in evaluations[:len(evaluations)//2]]
        new_population = top_half.copy()
        
        while len(new_population) < population_size:
            p1 = random.choice(top_half)
            p2 = random.choice(top_half)
            child = crossover(p1, p2)
            if random.random() < 0.2:
                child = perturb(child)
            new_population.append(child)
        population = new_population
    
    def objective_flat(x):
        return compute_c5(x)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    if time.time() - start_time < budget_s - 30:
        with np.errstate(all='ignore'):
            res = minimize(objective_flat, x0=best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
        if res.success:
            refined = res.x
            if compute_c5(refined) < best_c5:
                best_h, best_c5 = refined, compute_c5(refined)
    
    return (best_h, best_c5, n_points)
