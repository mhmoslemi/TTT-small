            recombination=0.7,
            polish=False,
            updating='deferred',
            seed=seed
        )
        if result_de.success:
            x_de = project_to_feasible(result_de.x, target_sum)
            c5_de = compute_c5(x_de, dx)
            if c5_de < best_c5:
                best_h, best_c5 = x_de, c5_de
    except Exception:
        pass  # Handle failure gracefully

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    if remaining > 10:
        # Local refinement with SLSQP and perturbations
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds_sq = Bounds(lb=0.0, ub=1.0)
        perturb_amp = 0.1
        perturb_decay = 0.95
        trials = 0
        max_trials = 50

        while time.time() - start_time < budget_s and trials < max_trials:
            if trials > 0 and trials % 5 == 0:
                perturbation = np.random.uniform(-perturb_amp, perturb_amp, n_points)
                perturbed_x = np.clip(best_h + perturbation, 0.0, 1.0)
                perturbed_sum = np.sum(perturbed_x)
                if perturbed_sum > 1e-9:
                    perturbed_x = perturbed_x * (target_sum / perturbed_sum)
                perturbed_x = np.clip(perturbed_x, 0.0, 1.0)
                x0 = perturbed_x
            else:
                x0 = best_h

            try:
                result = minimize(
                    lambda x: compute_c5(x, dx),
                    x0=x0,
                    method='SLSQP',
                    bounds=bounds_sq,
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-6}
                )
                if result.success:
                    new_h = result.x
                    new_h = np.clip(new_h, 0.0, 1.0)  # Enforce bounds
                    new_c5 = compute_c5(new_h, dx)
                    if new_c5 < best_c5:
                        best_h, best_c5 = new_h, new_c5
            except Exception:
                pass  # Skip failed optimizations
            trials += 1
            perturb_amp *= perturb_decay

    return (best_h, best_c5, n_points)
