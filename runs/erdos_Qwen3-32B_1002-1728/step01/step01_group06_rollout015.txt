```python
import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    EPS = 1e-8

    # Initialize with provided initial_h_values
    h = initial_h_values.copy()
    n = len(h)
    target_sum = n / 2

    # Ensure h is valid
    h = np.clip(h, 0, 1)
    h *= target_sum / h.sum()
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Evolutionary Strategy parameters
    sigma = 0.1
    lambda_ = 10
    no_improve = 0

    while time.time() - start_time < max_time * 0.98:
        offspring_list = []
        for _ in range(lambda_):
            # Mutate
            child = np.clip(best_h + sigma * np.random.randn(n), 0, 1)
            # Repair: enforce sum constraint
            sum_child = np.sum(child)
            if abs(sum_child) < EPS:
                sum_child = EPS
            child = target_sum / sum_child * child
            child = np.clip(child, 0, 1)
            # Evaluate
            c5 = compute_c5(child)
            offspring_list.append((child, c5))
        
        # Select best offspring
        offspring_list.sort(key=lambda x: x[1])
        best_offspring, best_offspring_c5 = offspring_list[0]

        if best_offspring_c5 < best_c5:
            best_h[:] = best_offspring
            best_c5 = best_offspring_c5
            no_improve = 0
        else:
            no_improve += 1

        # Adapt mutation strength
        if no_improve >= 50:
            sigma = max(sigma / 2, 1e-5)
            no_improve = 0

    return best_h, best_c5, n
```
