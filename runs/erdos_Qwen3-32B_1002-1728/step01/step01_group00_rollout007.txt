        velocities.append(0.05 * np.random.randn(n_points))

    particles = np.array(particles)
    velocities = np.array(velocities, dtype=float)
    pbest_positions = particles.copy()
    pbest_values = np.array([compute_c5(np.clip(p, 0, 1)) for p in particles])
    gbest_idx = np.argmin(pbest_values)
    gbest_position = np.clip(pbest_positions[gbest_idx], 0, 1)
    gbest_value = pbest_values[gbest_idx]

    start_time = time.time()
    max_time = budget_s

    c1 = 2.0
    c2 = 2.0
    w_max = 0.9
    w_min = 0.4

    while time.time() - start_time < max_time:
        elapsed = time.time() - start_time
        progress_ratio = min(elapsed / max_time, 1.0)
        w = w_max - progress_ratio * (w_max - w_min)

        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)

        cognitive = c1 * r1 * (pbest_positions - particles)
        social = c2 * r2 * (np.expand_dims(gbest_position, 0) - particles)
        velocities = w * velocities + cognitive + social

        particles += velocities

        for i in range(n_particles):
            particles[i] = project(particles[i], target_sum)

        new_values = np.array([compute_c5(p) for p in particles])
        improved = new_values < pbest_values
        pbest_values[improved] = new_values[improved]
        pbest_positions[improved] = particles[improved].copy()

        current_gbest = np.min(pbest_values)
        if current_gbest < gbest_value:
            gbest_idx = np.argmin(pbest_values)
            gbest_value = current_gbest
            gbest_position = particles[gbest_idx].copy()

    final_corr = np.correlate(gbest_position, 1 - gbest_position, mode='full') * dx
    return gbest_position, np.max(final_corr), n_points
