import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    n_particles = 20
    particles = [np.random.rand(n_points) for _ in range(n_particles)]

    corr = correlate(gbest_position, 1 - gbest_position, mode='full')
    gbest_score = np.max(corr) * dx

    start_time = time.time()

    w_max, w_min = 0.9, 0.4

    while time.time() - start_time < budget_s - 0.1:
        progress = (time.time() - start_time) / budget_s
        w = w_max - progress * (w_max - w_min)
        if w < w_min:
            w = w_min

        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            particles[i] += velocities[i]
            particles[i] = _project(particles[i], target_sum, n_points)

            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            score = np.max(corr) * dx

                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()

                    gbest_position = particles[i].copy()

    corr_values = np.correlate(gbest_position, 1 - gbest_position, mode='full') * dx

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    temp = clamped * (target_sum / clamped_sum)
