import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()

    start_time = time.time()
    best_c5 = None
    best_h = h_initial.copy()

    while time.time() - start_time < budget_s - 30:
        # Generate perturbed initial guess
        if best_c5 is not None:
            x0 = best_h.copy()
            x0 += np.random.normal(0, 0.02, n_points)
            x0 = np.clip(x0, 0, 1)
            offset = np.sum(x0) - target_sum
            x0 -= offset / n_points
            x0 = np.clip(x0, 0, 1)
        else:
            x0 = h_initial.copy()

        # Optimization step
        result = minimize(
            objective,
            x0=x0,
            method='COBYLA',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 300, 'ftol': 1e-6}
        )

        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        offset = np.sum(curr_h) - target_sum
        curr_h -= offset / n_points
        curr_h = np.clip(curr_h, 0, 1)

        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)

        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()

        if time.time() - start_time > budget_s - 30:
            break

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    offset = np.sum(best_h) - target_sum
    best_h -= offset / n_points
    best_h = np.clip(best_h, 0, 1)

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
