def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    max_iter = 1000
    cognitive = 2.0
    social = 2.0
    v_max = 0.1

    np.random.seed(seed)
    particles = []
    velocities = []
    pbest_h = []
    pbest_scores = []

        particles.append(h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest_h.append(h.copy())
        pbest_scores.append(np.inf)
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    pbest_h = np.array(pbest_h)
    pbest_scores = np.array(pbest_scores)
    
    gbest_score = np.inf
    gbest_h = particles[0].copy()

    while time.time() - start_time < budget_s:
        gamma = 0.9 - (time.time() - start_time) / budget_s * (0.9 - 0.4)
        r1 = np.random.rand(num_particles, 1)

        velocities = gamma * velocities + cognitive * r1 * (pbest_h - particles) + social * r2 * (gbest_h[None, :] - particles)
