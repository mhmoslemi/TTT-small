            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        offset = np.sum(curr_h) - n_points/2
        if abs(offset) > 1e-8:
            curr_h[0] -= offset
            curr_h = np.clip(curr_h, 0, 1)
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()

    # Perturbation-based restarts
    max_restart = 3
    for _ in range(max_restart):
        if time.time() - start_time > budget_s - 10:
            break
        x0 = best_h.copy()
        x0 += np.random.normal(0, 0.02, n_points)
        x0 = np.clip(x0, 0, 1)
        offset = np.sum(x0) - n_points/2
        x0[0] -= offset
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
