import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result_initial = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 80, 'ftol': 1e-8}
    )
    best_h = result_initial.x
    best_c5 = objective(best_h)

    rng = np.random.RandomState(seed)
    max_attempts = 200

    for _ in range(max_attempts):
        if time.time() - start_time > budget_s - 5:
            break
        perturbed = best_h.copy() + rng.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_pert = np.sum(perturbed)
        if sum_pert < 1e-8:
            perturbed = best_h.copy()
        else:
            perturbed = perturbed * (target_sum / sum_pert)
        result_perturb = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-8}
        )
        h_perturb = result_perturb.x
        if abs(np.sum(h_perturb) - target_sum) > 1e-6:
            continue
        if np.any(h_perturb < -1e-6) or np.any(h_perturb > 1.0 + 1e-6):
            continue
        current_c5 = objective(h_perturb)
        if current_c5 < best_c5:
            best_h = h_perturb
            best_c5 = current_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
