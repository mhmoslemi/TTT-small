import numpy as np
import time
import random

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair_candidate(c):
        c_clipped = np.clip(c, 0, 1)
        s = np.sum(c_clipped)
        if s == 0:
            return np.clip(c, 0, 1)
        c_repaired = c_clipped * (target_sum / s)
        return np.clip(c_repaired, 0, 1)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    pop_size = 40
    population = [h_init]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair_candidate(h)
        population.append(h)
    population = [repair_candidate(h) for h in population]

    best_c5 = float('inf')
    best_h = h_init.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        current_pop = np.array(population)
        fitness = np.array([compute_c5(h) for h in current_pop])
        elite_idx = np.argmin(fitness)
        elite = current_pop[elite_idx].copy()
        if fitness[elite_idx] < best_c5:
            best_c5 = fitness[elite_idx]
            best_h = elite.copy()

        parents = []
        for _ in range(pop_size // 2):
            # Tournament selection
            contestants = random.sample(range(pop_size), 3)
            parent_idx = min(contestants, key=lambda i: fitness[i])
            parents.append(current_pop[parent_idx].copy())

        offspring = parents.copy()
        for _ in range(pop_size):
            idxs = random.sample(range(len(parents)), 2)
            p1, p2 = parents[idxs[0]], parents[idxs[1]]
            cross_point = random.randint(1, n_points - 1)
            child = np.concatenate((p1[:cross_point], p2[cross_point:]))
            child = repair_candidate(child)
            # Mutation
            if random.random() < 0.2:
                indices = np.random.choice(n_points, size=3, replace=False)
                pert = np.random.normal(0, 0.1, size=3)
                child[indices] += pert
                child = repair_candidate(child)
            offspring.append(child)

        # Keep elite and top offspring
        offspring.append(elite)
        offspring = [repair_candidate(h) for h in offspring]
        scores = [(compute_c5(h), h) for h in offspring]
        scores.sort(key=lambda x: x[0])
        top_offspring = [h for (s, h) in scores[:pop_size]]
        population = [elite] + top_offspring[:pop_size - 1]

    return best_h, best_c5, n_points
