import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # PSO parameters
    n_particles = 20
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1

    # Initialize particles and velocities
    particles = []
    velocities = []
    for _ in range(n_particles):
        pert = h_initial + np.random.normal(0, 0.01, n_points)
        clamped = np.clip(pert, 0, 1)
        s = clamped.sum()
        if s > 1e-10:
            clamped *= (n_points / 2) / s
        else:
            clamped = np.ones(n_points) * 0.5
        particles.append(clamped)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    # Compute initial personal bests and global best
    pbest_positions = particles.copy()
    pbest_scores = np.array([np.max(np.correlate(p, 1 - p, 'full')) * dx for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        t_elapsed = time.time() - start_time
        w = 0.9 - (t_elapsed / budget_s) * 0.8
        w = np.clip(w, 0.1, 0.9)

        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = w * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_position - particles[i])
            vel_update = np.clip(vel_update, -max_vel, max_vel)
            velocities[i] = vel_update
            # Update position
            particles[i] += vel_update
            # Project to feasible
            clamped = np.clip(particles[i], 0, 1)
            s = clamped.sum()
            if s > 1e-10:
                clamped *= (n_points / 2) / s
            else:
                clamped = np.ones(n_points) * 0.5
            particles[i] = clamped

            # Compute score
            corr = np.correlate(clamped, 1 - clamped, 'full')
            score = np.max(corr) * dx
            # Update pbest
            if score < pbest_scores[i]:
                pbest_positions[i] = clamped.copy()
                pbest_scores[i] = score
                # Update gbest
                if score < gbest_score:
                    gbest_score = score
                    gbest_position = clamped.copy()

    return gbest_position, gbest_score, n_points

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # PSO parameters
    n_particles = 20
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.15

    # Initialize particles and velocities
    particles = []
    velocities = []
    for _ in range(n_particles):
        pert = h_initial + np.random.normal(0, 0.01, n_points)
        clamped = np.clip(pert, 0, 1)
        s = clamped.sum()
        if s > 1e-10:
            clamped *= (n_points / 2) / s
        else:
            clamped = np.ones(n_points) * 0.5
        particles.append(clamped)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    # Compute initial personal bests and global best
    pbest_positions = particles.copy()
    pbest_scores = np.array([np.max(np.correlate(p, 1 - p, 'full')) * dx for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        t_elapsed = time.time() - start_time
        w = 0.9 - (t_elapsed / budget_s) * 0.8
        w = np.clip(w, 0.1, 0.9)

        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = w * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_position - particles[i])
            vel_update = np.clip(vel_update, -max_vel, max_vel)
            velocities[i] = vel_update
            # Update position
            particles[i] += vel_update
            # Project to feasible
            clamped = np.clip(particles[i], 0, 1)
            s = clamped.sum()
            if s > 1e-10:
                clamped *= (n_points / 2) / s
            else:
                clamped = np.ones(n_points) * 0.5
            particles[i] = clamped

            # Compute score
            corr = np.correlate(clamped, 1 - clamped, 'full')
            score = np.max(corr) * dx
            # Update pbest
