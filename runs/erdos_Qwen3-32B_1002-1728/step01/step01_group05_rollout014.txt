                adj_perturbed_h = perturbed_h * (target_sum / perturbed_sum)
                adj_perturbed_h = np.clip(adj_perturbed_h, 0.0, 1.0)
            result = minimize(lambda x: compute_c5(x, dx), x0=adj_perturbed_h, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 100, 'ftol': 1e-6})
        else:
            result = minimize(lambda x: compute_c5(x, dx), x0=best_h.copy(), method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 100, 'ftol': 1e-6})
        
        if result.success:
            new_h = result.x
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        trials += 1
        perturb_amp *= perturb_decay
    
    return (best_h, best_c5, n_points)
