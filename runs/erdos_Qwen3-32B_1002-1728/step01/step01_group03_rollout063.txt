        h_pert *= target_sum / np.sum(h_pert)
        return h_pert
    
    if len(h_prev) != n_init and not np.array_equal(h_prev, h_init):
        upsampled = h_prev.copy()
        while len(upsampled) < n_init:
            factor = n_init // len(upsampled)
            extra = n_init - factor * len(upsampled)
            upsampled = np.tile(upsampled, factor)
            if extra > 0:
                upsampled = np.append(upsampled, upsampled[:extra])
        upsampled *= target_sum / np.sum(upsampled)
        h_candidate = np.clip(upsampled, 0, 1)
    else:
        h_candidate = h_prev if len(h_prev) == n_init else h_init
    
    best_h = h_candidate.copy()
    best_c5 = objective_final(best_h)
    start_time = time.time()
    
    tries = 0
    while time.time() - start_time < budget_s and tries < 20:
        tries += 1
        x0 = get_perturbed(best_h) if tries % 3 else h_candidate.copy()
        res = minimize(objective_final, x0=x0, method='Nelder-Mead', bounds=[(0,1)]*n_init, options={'maxiter': 50})
        if res.success:
            val = objective_final(res.x)
            if val < best_c5:
                best_h = res.x.copy()
                best_c5 = val
    
    return (best_h, best_c5, n_init)
