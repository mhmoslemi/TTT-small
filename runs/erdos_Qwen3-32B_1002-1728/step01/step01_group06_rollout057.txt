import numpy as np
import time

def compute_c5(h_vals, n):
    corr = np.correlate(h_vals, 1 - h_vals, mode='full') * (2 / n)
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_generations = 1000
    lambda_offspring = 10
    adapt_interval = 50

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        current_sum = h.sum()
        h = h * (n / 2) / current_sum
        h = np.clip(h, 0, 1)
    else:
        n = 280
        h = np.zeros(n)
        h[135:145] = 0.5  # Central pulse
        h *= (n / 2) / h.sum()
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    sigma = 0.1
    prev_c5 = best_c5

    for generation in range(max_generations):
        if time.time() - start_time > budget_s * 0.98:
            break

        offspring = []
        c5_values = []
        for _ in range(lambda_offspring):
            candidate = best_h + sigma * np.random.randn(n)
            candidate = np.clip(candidate, 0, 1)
            current_sum = candidate.sum()
            if current_sum > 1e-10:
                candidate *= (n / 2) / current_sum
            else:
                continue
            offspring.append(candidate)
            c5_values.append(compute_c5(candidate, n))

        if offspring:
            best_idx = np.argmin(c5_values)
            new_candidate = offspring[best_idx]
            new_c5 = c5_values[best_idx]
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_candidate

        if generation % adapt_interval == 0 and generation > 0:
            if best_c5 < prev_c5:
                sigma *= 0.95
                prev_c5 = best_c5
            else:
                sigma *= 0.7

    return best_h, best_c5, n
