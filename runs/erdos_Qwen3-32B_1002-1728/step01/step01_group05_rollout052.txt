def compute_c5(h, dx):

    one_minus_h = 1.0 - h_array
    corr = np.correlate(h_array, one_minus_h, mode='full')
    corr *= dx
    return np.max(corr)

def project_to_feasible(h, target_sum, n_points):

    if current_sum < 1e-12:
        h[:] = 0.5
    else:
        h *= target_sum / current_sum
        h = np.clip(h, 0.0, 1.0)
    return h
