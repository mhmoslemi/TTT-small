            objective,
            x0=x0_perturbed,
            method='COBYLA',
            bounds=bounds,
            options={'maxiter': 100, 'rhobeg': 0.1, 'tol': 1e-6}
        )
        
        curr_h = result.x.copy()
        curr_h = repair_feasibility(curr_h, n_points)
        
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
    
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    
    return best_h, c5_bound, n_points
