import time

def run(seed=42, budget_s=1100, **kwargs):

    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    num_particles = 50
    v_max = 0.1
    c1 = 2.0
    c2 = 2.0
    max_inertia_steps = 500

    np.random.seed(seed)
    particles = []
    for _ in range(num_particles):
        perturbation = np.random.randn(n_points) * 0.05
        h = h_init + perturbation
        h = np.clip(h, 0, 1.0)
        h = (h / h.sum()) * target_sum
        particles.append(h)
    particles = np.array(particles)

    velocities = np.random.uniform(-v_max, v_max, (num_particles, n_points))

    fitness = np.zeros(num_particles)
    for i in range(num_particles):
        h = particles[i]
        corr = np.correlate(h, 1 - h, mode='full')
        fitness[i] = np.max(corr) * dx
    pbest_positions = particles.copy()
    pbest_fitness = fitness.copy()

    gbest_idx = np.argmin(fitness)
    gbest_position = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        step_ratio = min(iteration / max_inertia_steps, 1.0)
        w = 0.9 - step_ratio * (0.9 - 0.4)

        r1 = np.random.rand(num_particles, 1)
        r2 = np.random.rand(num_particles, 1)

        velocities = w * velocities + vel_cognitive + vel_social
        velocities = np.clip(velocities, -v_max, v_max)

        row_sums = particles.sum(axis=1, keepdims=True)
        particles *= target_sum / (row_sums + 1e-10)
