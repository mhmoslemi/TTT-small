import numpy as np

def tournament_selection(population, fitness, tournament_size=5):
    indices = np.random.choice(len(population), tournament_size)
    candidates = [population[i] for i in indices]
    fitness_vals = [fitness[i] for i in indices]
    winner_idx = np.argmin(fitness_vals)
    return candidates[winner_idx]

def crossover(parent1, parent2):
    alpha = np.random.rand()
    return alpha * parent1 + (1 - alpha) * parent2

def mutate(candidate, mutation_rate=0.1, noise_std=0.02):
    candidate = candidate.copy()
    for i in range(len(candidate)):
        if np.random.rand() < mutation_rate:
            delta = np.random.normal(0, noise_std)
            candidate[i] = np.clip(candidate[i] + delta, 0, 1)
    return candidate

def enforce_constraints(candidate, n_points):
    desired_sum = n_points / 2
    candidate = np.clip(candidate, 0, 1)
    current_sum = candidate.sum()
    if abs(current_sum - desired_sum) > 1e-8:
        candidate[0] += (desired_sum - current_sum)
        candidate = np.clip(candidate, 0, 1)
    return candidate

        candidate += noise
        candidate = enforce_constraints(np.clip(candidate, 0, 1), n_points)
