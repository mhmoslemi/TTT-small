def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Define population
    np.random.seed(seed)
    population_size = 50
    population = []
    # Add 5 copies of initial_h
    for _ in range(5):
        population.append(h_init.copy())
    # Add 45 perturbed copies
    for _ in range(45):
        h_pert = h_init.copy() + np.random.normal(0, 0.05, n_points)
        h_clipped = np.clip(h_pert, 0, 1)
        current_sum = h_clipped.sum()
        if np.isclose(current_sum, 0):
            population.append(np.zeros_like(h_pert))
        else:
            h_scaled = h_clipped * (target_sum / current_sum)
            h_scaled_clipped = np.clip(h_scaled, 0, 1)
            population.append(h_scaled_clipped)
    
    best_h = None
    best_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 10:  # leave 10s for final checks
        # Evaluate population
        c5_list = []
        for h_candidate in population:
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            c5_list.append(c5_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        
        # Selection: select top 50% as parents
        indices = np.argsort(c5_list)
        top_indices = indices[:population_size//2]
        parents = [population[i] for i in top_indices]
        
        # Generate new population: 50% from top, 50% from cross and mutate
        new_population = parents.copy()
        for _ in range(population_size - len(new_population)):
            # Crossover: blend two parents
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            child = 0.5 * parent1 + 0.5 * parent2
            # Mutation: small random noise
            if np.random.rand() < 0.1:  # 10% mutation rate
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
            # Repair
            current_sum_child = np.sum(child)
            if np.isclose(current_sum_child, 0):
                child_repaired = np.zeros_like(child)
            else:
                h_clipped = np.clip(child, 0, 1)
                h_scaled = h_clipped * (target_sum / h_clipped.sum())
                child_repaired = np.clip(h_scaled, 0, 1)
            new_population.append(child_repaired)
        
        population = new_population
