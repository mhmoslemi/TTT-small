import numpy as np
import time
from scipy.optimize import minimize, Bounds

def project_to_feasible(x, target_sum):
    x_clipped = np.clip(x, 0.0, 1.0)
    s = x_clipped.sum()
    if np.isclose(s, 0.0):
        return np.full_like(x, 0.5)
    x_scaled = x_clipped * (target_sum / s)
    x_scaled = np.clip(x_scaled, 0.0, 1.0)
    return x_scaled

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.arange(n_points)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    candidates = []
    # Sinusoidal patterns
    for period in [4, 5, 6]:
        T = n_points / period
        sin_signal = (1 + np.sin(2 * np.pi * x / T)) / 2.0
        candidates.append(project_to_feasible(sin_signal, target_sum))
    # Triangular patterns
    for period in [20, 25, 30]:
        x_period = x % period
        wave = 2 * (x_period / period - 0.5)
        mask = x_period >= period / 2
        wave[mask] = 2 * (0.5 - (x_period[mask] - period / 2) / period)
        triangle = (wave + 1) / 2
        candidates.append(project_to_feasible(triangle, target_sum))
    # Cos-based "roots" patterns
    for roots in [5, 8, 10]:
        cos_signal = 0.5 * (1 + np.cos(2 * np.pi * roots * x / (2 * n_points)))
        candidates.append(project_to_feasible(cos_signal, target_sum))
    # Include initial h as candidate
    candidates.append(h0)

    # Evaluate all candidates
    for h in candidates:
        if time.time() - start_time > budget_s * 0.9:
            break
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Optimize best candidate
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)
    result = minimize(lambda x: compute_c5(x, dx), x0=best_h, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-8})
    if result.success:
        final_h = result.x
        final_c5 = compute_c5(final_h, dx)
        best_h, best_c5 = (final_h, final_c5) if final_c5 < best_c5 else (best_h, best_c5)

    return (best_h.clip(0, 1), best_c5, n_points)
