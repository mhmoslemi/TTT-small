def generate_periodic_h(p, n_points):
    # Generate base pattern for period p
    k = int(0.5 * p)
    s = 0.5 * p  # required sum per period
    base = np.zeros(p)

    if remaining > 0:

    # Now replicate to n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        period_idx = i % p
        h[i] = base[period_idx]
    return h

import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)

    # Generate structured initial guesses
    periods = [3, 4, 5, 6, 7, 8, 9, 10, 12]
    initial_guesses = []
    for p in periods:
        base = np.zeros(p)
        k = int(0.5 * p)
        s = 0.5 * p
        base[:k] = 1.0
        remaining = p - k
        if remaining > 0:
            base[k:] = (s - k) / remaining
        h = np.zeros(n_points)
        for i in range(n_points):
            period_idx = i % p
            h[i] = base[period_idx]
        initial_guesses.append(h)
    # Add the provided initial guess
    initial_guesses.append(h_initial.copy())

    best_c5 = None
    best_h = h_initial.copy()
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        x0 = guess.copy()

        # Run optimization
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        curr_h = result.x.copy()
        # Enforce constraints
        curr_h = np.clip(curr_h, 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset  # Adjust to meet the sum constraint

        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)

        if (best_c5 is None) or (curr_c5 < best_c5):
            best_c5 = curr_c5
            best_h = curr_h.copy()

    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)

    return best_h, c5_bound, n_points
