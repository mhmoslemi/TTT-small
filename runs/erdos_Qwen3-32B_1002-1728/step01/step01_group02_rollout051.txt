    return np.max(cross * dx)

import numpy as np
import cvxpy as cp
import time

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

    dx = 2.0 / n_points

    best_c5 = evaluate_C5(h_current, n_points, dx)
    best_h = h_current.copy()

    while time.time() - start_time < budget_s - 20:
        # Compute worst overlap
        cross = np.correlate(h_current, 1 - h_current, mode='full')
        overlaps = cross * dx
        idx_max = np.argmax(overlaps)
        k_shift = idx_max - (n_points - 1)

        # Generate valid indices for k_shift
        valid_i = []
        for i in range(n_points):

        h_var = cp.Variable(n_points)
        t_var = cp.Variable()
        expr = 0.0

            term = (1.0 - h_current[i + k_shift]) * h_var[i] - h_current[i] * h_var[j]
            expr += term

            h_var <= 1,
            overlap_current + expr * dx <= t_var
        ]
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
