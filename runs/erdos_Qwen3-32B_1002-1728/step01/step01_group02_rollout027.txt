def objective_smooth(h_vec, beta, n_points, dx):
    h = h_vec.reshape(n_points)
    ones_minus_h = 1 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    cross_values = cross * dx
    exp_values = np.exp(beta * cross_values)
    sum_exp = np.sum(exp_values)
    if sum_exp <= 1e-12 or beta == 0:
        return np.max(cross_values)
    smoothed_c5 = (1.0 / beta) * math.log(sum_exp)
    return smoothed_c5

def objective_exact(h_vec, n_points, dx):
    h = h_vec.reshape(n_points)
    ones_minus_h = 1 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def generate_perturbed(h_base, n=3, noise_scale=0.1, n_points=None):
    assert n_points is not None
    candidates = []
    for _ in range(n):
        pert = np.random.normal(0, noise_scale, n_points)
        perturbed = h_base.copy() + pert
        perturbed_clipped = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed_clipped)
        if current_sum < 1e-8:
            candidates.append(perturbed_clipped.copy())
            continue
        scaling = (n_points / 2) / current_sum
        perturbed_scaled = perturbed_clipped * scaling
        perturbed_final = np.clip(perturbed_scaled, 0, 1)
        candidates.append(perturbed_final.copy())
    return candidates

def run():
    start_time = time.time()
    budget_s = 1100
    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.ones(n_points).astype(np.float64) * 0.5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    betas = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    candidates = [h_initial] + generate_perturbed(h_initial, n=4, noise_scale=0.15, n_points=n_points)

    best_h = h_initial.copy()
    best_c5 = np.inf

    for candidate in candidates:
        h = candidate.copy()
        for beta in betas:
            if time.time() - start_time >= budget_s - 20:
                break

            cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_points,)}]
            args = (beta, n_points, dx)

            result = minimize(
                objective_smooth,
                h,
                args=args,
                method='SLSQP',
                bounds=bounds,
                constraints=cons
            )

            if result.success:
                h_opt = result.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                sum_opt = np.sum(h_opt)
                if abs(sum_opt - n_points / 2) > 1e-8:
                    scaling = (n_points / 2) / sum_opt
                    h_opt = np.clip(h_opt * scaling, 0, 1)
                h = h_opt

                current_c5 = evaluate_C5(h, n_points, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            h = np.clip(h, 0, 1)
        # Fine-tuning after beta progression
        if not math.isinf(best_c5) and time.time() - start_time < budget_s - 10:
            result_fine = minimize(
                objective_exact,
                h,
                args=(n_points, dx),
                method='SLSQP',
                bounds=bounds,
                constraints=cons
            )
            if result_fine.success:
                h_fine = result_fine.x.copy()
                h_fine = np.clip(h_fine, 0, 1)
                sum_fine = np.sum(h_fine)
                if abs(sum_fine - n_points / 2) > 1e-8:
                    scaling = (n_points / 2) / sum_fine
                    h_fine = np.clip(h_fine * scaling, 0, 1)
                final_c5 = evaluate_C5(h_fine, n_points, dx)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = h_fine.copy()
    return best_h, best_c5, n_points
