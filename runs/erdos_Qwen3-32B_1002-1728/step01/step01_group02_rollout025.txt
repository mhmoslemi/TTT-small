import numpy as np
import time
from scipy.optimize import minimize

def run():
    start_time = time.time()
    budget_s = 1100
    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.full(n_points, 0.5, dtype=np.float64)

    candidates = [h_initial.copy()]
    for _ in range(4):
        perturbed = h_initial + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        repair_scale = (n_points / 2) / np.sum(perturbed)
        perturbed = np.clip(perturbed * repair_scale, 0, 1)
        candidates.append(perturbed)

    best_c5 = float('inf')
    best_h = h_initial.copy()
    beta_steps = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    for candidate in candidates:
        h_current = candidate.copy()
        for beta in beta_steps:
            if time.time() - start_time > budget_s - 20:
                break

            def obj_smoothed(h_vec):
                h = h_vec.reshape(n_points)
                cross = np.correlate(h, 1 - h, mode='full')
                return np.log(np.sum(np.exp(beta * cross * dx))) / beta

            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1)] * n_points
            result = minimize(obj_smoothed, h_current, method='SLSQP', bounds=bounds, constraints=cons)

            if result.success:
                h_new = np.clip(result.x, 0, 1)
                h_sum = np.sum(h_new)
                if abs(h_sum - n_points/2) > 1e-8:
                    h_new = np.clip(h_new * (n_points/2) / h_sum, 0, 1)

                cross = np.correlate(h_new, 1 - h_new, mode='full')
                c5 = np.max(cross) * dx
                if c5 < best_c5:
                    best_c5, best_h = c5, h_new.copy()
                h_current = h_new

    if time.time() - start_time < budget_s - 20:
        def exact_obj(h_vec):
            return np.max(np.correlate(h_vec.reshape(n_points), 1 - h_vec.reshape(n_points), mode='full')) * dx

        result_final = minimize(exact_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons)
        if result_final.success:
            h_final = np.clip(result_final.x, 0, 1)
            h_sum = np.sum(h_final)
            if abs(h_sum - n_points/2) > 1e-8:
                h_final = np.clip(h_final * (n_points/2) / h_sum, 0, 1)
            c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
            if c5_final < best_c5:
                best_c5, best_h = c5_final, h_final.copy()

    return best_h, best_c5, n_points
