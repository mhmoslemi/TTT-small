import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Objective function (C5 value)
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Parameters for PSO
    n_particles = 50
    max_iter = 1000
    alpha = 2.0  # cognitive coefficient
    beta = 2.0   # social coefficient
    v_max = 0.1  # maximum velocity step

    # Initialize particles with perturbations
    particles = np.zeros((n_particles, n_points))
    for i in range(n_particles):
        particle = h_init + np.random.normal(0, 0.05, size=n_points)
        particle = np.clip(particle, 0.0, 1.0)
        s = np.sum(particle)
        if s != 0:
            particle = particle * (target_sum / s)
        particles[i] = particle

    # Initialize velocities
    velocities = np.random.uniform(-v_max, v_max, size=(n_particles, n_points))

    # Evaluate initial objectives
    current_objs = np.array([compute_c5(p) for p in particles])
    pbest = particles.copy()
    pbest_obj = current_objs.copy()
    gbest_idx = np.argmin(pbest_obj)
    gbest = pbest[gbest_idx].copy()
    gbest_obj = pbest_obj[gbest_idx]

    # PSO loop
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1.0:
            break  # Leave 1s buffer to complete return

        # Update inertia weight
        gamma = max(0.4, 0.9 - (0.9 - 0.4) * iter_count / max_iter)

        # Update velocities and positions
        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)
        velocities = gamma * velocities + alpha * r1 * (pbest - particles) + beta * r2 * (gbest - particles)
        velocities = np.clip(velocities, -v_max, v_max)
        particles += velocities

        # Clip values to [0, 1]
        particles = np.clip(particles, 0.0, 1.0)

        # Scale to maintain sum constraint
        sums = np.sum(particles, axis=1)
        valid_sums = sums != 0
        particles[valid_sums] *= target_sum / sums[valid_sums][:, np.newaxis]

        # After scaling, clip again to ensure [0, 1]
        particles = np.clip(particles, 0.0, 1.0)

        # Evaluate new objectives
        current_objs = np.array([compute_c5(p) for p in particles])

        # Update pbest and gbest
        for i in range(n_particles):
