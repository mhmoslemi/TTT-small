    start_time = time.time()
    target_n = 280
    sigma_start = 0.1
    lambda_offspring = 10
    adapt_interval = 50
    max_generations = 1000

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
        if n_points != target_n:
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, target_n)
            h = np.interp(new_x, old_x, h)
            h *= (target_n / 2) / h.sum()
            h = np.clip(h, 0, 1)
    else:
        h = np.zeros(target_n)
        mid = target_n // 2
        h[mid-5:mid+5] = 0.5
        h *= (target_n / 2) / h.sum()
    h = np.clip(h, 0, 1)
    
    def compute_c5(_h):
        _1m_h = 1 - _h
        corr = np.correlate(_h, _1m_h, mode='full')
        return np.max(corr * (2 / target_n))

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_best = h.copy()
    current_c5 = best_c5
    sigma = sigma_start
    no_improvement_count = 0

    for gen in range(max_generations):
        if time.time() - start_time > budget_s * 0.99:
            break

        offsprings = []
        for _ in range(lambda_offspring):
            # Mutate and repair
            perturbed = current_best + sigma * np.random.randn(target_n)
            clipped = np.clip(perturbed, 0, 1)
            sum_clipped = clipped.sum()
            h_scaled = np.zeros_like(clipped)
            if sum_clipped > 0:
                scaling = (target_n / 2) / sum_clipped
                h_scaled = np.clip(clipped * scaling, 0, 1)
            else:
                # Fallback for invalid repair
                h_scaled = best_h.copy()
            offsprings.append((h_scaled, compute_c5(h_scaled)))

        # Evaluate candidates
        candidates = [(current_best, current_c5)] + [(h, c) for h, c in offsprings]
        candidates.sort(key=lambda x: x[1])
        best_in_gen, best_in_gen_c = candidates[0]

        if best_in_gen_c < best_c5:
            best_c5 = best_in_gen_c
            best_h = best_in_gen.copy()
            current_best = best_in_gen.copy()
            current_c5 = best_in_gen_c
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Avoid sigma vanishing
            if no_improvement_count >= adapt_interval:
                sigma *= 0.5
                no_improvement_count = 0

    return best_h, best_c5, target_n
