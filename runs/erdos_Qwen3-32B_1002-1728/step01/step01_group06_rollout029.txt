import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Interpolate initial solution to 280 points
    h = initial_h_values.copy()
    n = 280
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, n)
    h = np.clip(np.interp(new_x, old_x, h), 0, 1)
    h *= (n / 2) / h.sum()
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    dx = 2.0 / n
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, 'full') * dx)
    current_h = h.copy()
    current_c5 = best_c5

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    sigma = 0.1
    
    while time.time() - start_time < budget_s * 0.95:
        # Generate proposal
        h_proposal = current_h + sigma * T * np.random.randn(n)
        h_proposal = np.clip(h_proposal, 0, 1)
        h_proposal *= (n / 2) / h_proposal.sum()
        h_proposal = np.clip(h_proposal, 0, 1)

        # Compute objective
        corr = np.correlate(h_proposal, 1 - h_proposal, 'full')
        proposed_c5 = np.max(corr) * dx

        # Acceptance probability
        delta = proposed_c5 - current_c5
        if delta < 0:
            current_h = h_proposal.copy()
            current_c5 = proposed_c5
            if proposed_c5 < best_c5:
                best_h = h_proposal.copy()
                best_c5 = proposed_c5
                T *= 0.8  # Accelerated cooling on improvement
        else:
            if np.random.rand() < np.exp(-delta / max(T, 1e-8)):
                current_h = h_proposal.copy()
                current_c5 = proposed_c5

        T *= alpha

    # SLSQP refinement
    def objective(x):
        x = x.reshape(-1)
        corr = np.correlate(x, 1 - x, 'full') * dx
        return np.max(corr)

    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=Bounds(0, 1),
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}],
        options={'maxiter': 250, 'ftol': 1e-8}

        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final constraint enforcement
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n / 2) / best_h.sum()
