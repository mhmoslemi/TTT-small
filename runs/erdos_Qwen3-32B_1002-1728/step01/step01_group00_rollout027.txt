import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    n_particles = 20
    np.random.seed(seed)

    # Initialize particles with Gaussian perturbations
    particles = [np.random.normal(h_initial, 0.05, n_points) for _ in range(n_particles)]
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        scale_factor = (n_points / 2) / particles[i].sum()
        particles[i] *= scale_factor

    velocities = [0.1 * np.random.randn(n_points) for _ in range(n_particles)]

    # Feasibility projection function
    def project(p):
        p = np.clip(p, 0, 1).copy()
        scale = (n_points / 2) / max(p.sum(), 1e-8)
        return p * scale

    # Objective function
    def compute_c5(h):
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # Initialize personal best and global best
    pbest = [p.copy() for p in particles]
    pbest_scores = [compute_c5(p) for p in pbest]
    gbest = pbest[np.argmin(pbest_scores)].copy()
    best_score = min(pbest_scores)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        elapsed = time.time() - start_time
        inertia = 0.9 - (0.8 * elapsed / budget_s)
        w = np.clip(inertia, 0.1, 0.9)

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                2.0 * r1 * (pbest[i] - particles[i]) +
                2.0 * r2 * (gbest - particles[i])
            )

            # Update position and project
            particles[i] += velocities[i]
            particles[i] = project(particles[i])

            # Update personal best
            current_score = compute_c5(particles[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest[i] = particles[i].copy()

                # Update global best
                if current_score < best_score:
                    best_score = current_score
                    gbest = particles[i].copy()

    return gbest, best_score, n_points
