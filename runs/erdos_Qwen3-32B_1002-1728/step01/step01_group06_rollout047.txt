import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    n_points = 280

    # Initialize h using initial_h_values or default
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        old_n = len(h)
        old_x = np.linspace(0, 2, old_n)
        new_x = np.linspace(0, 2, n_points)
        h = np.interp(new_x, old_x, h)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-5:mid+5] = 0.5
        h *= (n_points / 2) / h.sum()
        h = np.clip(h, 0, 1)

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Simulated Annealing parameters
    T_initial = 1.0
    alpha_normal = 0.995
    sigma = 0.1
    T = T_initial

    for i in range(10000):
        if time.time() - start_time > max_time * 0.95:
            break
        # Generate perturbed proposal
        proposal = current_h + sigma * T * np.random.randn(n_points)
        proposal = np.clip(proposal, 0, 1)
        total = proposal.sum()
        if total > 1e-10:
            proposal = proposal * (n_points / 2) / total
        else:
            proposal[:] = 0.0

        new_c5 = compute_c5(proposal)
        delta = new_c5 - current_c5

        # Accept if improvement or probabilistically
        if delta < 0 or np.exp(-delta / max(T, 1e-6)) > np.random.rand():
            current_h[:] = proposal
            current_c5 = new_c5
            # Check for best solution
            if new_c5 < best_c5:
                best_h[:] = proposal
                best_c5 = new_c5
                alpha_fast = 0.8 if delta < 0 else alpha_normal
            else:
                alpha_fast = alpha_normal
        else:
            alpha_fast = alpha_normal

        # Cool the system
        T *= alpha_fast

    # SLSQP refinement
    def objective(x):
        x = x.flatten()
        return compute_c5(x)

    bounds = Bounds(0, 1)
    constraint_sum = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint_sum,
                      options={'maxiter': 200, 'ftol': 1e-6})

    if result.success:
        x = np.clip(result.x, 0, 1)
        x = x / x.sum() * (n_points / 2)
        x = np.clip(x, 0, 1)
        new_c5 = compute_c5(x)
        if new_c5 < best_c5:
            best_h = x
            best_c5 = new_c5

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * (n_points / 2)
    return best_h, best_c5, n_points
