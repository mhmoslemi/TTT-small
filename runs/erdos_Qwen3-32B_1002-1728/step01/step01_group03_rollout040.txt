import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    popsize = 20
    
    # Generate initial population around h_init
    initial_population = []
    for _ in range(popsize):
        perturbed = h_init + np.random.normal(0, 0.1, n_points)
        perturbed_clipped = np.clip(perturbed, 0, 1)
        s = perturbed_clipped.sum()
        if s == 0:
            vec = np.full(n_points, 0.5)
        else:
            vec = perturbed_clipped * (target_sum / s)
        initial_population.append(vec)
    initial_population = np.array(initial_population)
    
    # Objective function with constraint repair
    def objective(h_flat):
        h = h_flat.copy()
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum == 0:
            return np.inf
        h_scaled = h_clipped * (target_sum / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
        return np.max(corr)
    
    result = differential_evolution(
        objective, bounds, init=initial_population,
        strategy='best1bin', maxiter=150,
        mutation=(0.5, 1), recombination=0.8,

    )
    
    best_h = result.x.reshape(n_points)
    best_c5 = result.fun
