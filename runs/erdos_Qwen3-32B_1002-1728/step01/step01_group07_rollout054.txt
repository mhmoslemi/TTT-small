import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x.copy()
    best_c5 = objective(best_h)

    # Multiple perturbation runs
    attempt = 0
    while time.time() - start_time < budget_s * 0.99:
        if attempt >= 50:
            break
        attempt += 1
        current_seed = seed + attempt
        rs = np.random.RandomState(current_seed)
        perturbed = best_h + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = perturbed.sum()
        if current_sum < 1e-6:
            perturbed.fill(0.0)
            perturbed[0] = target_sum
        else:
            perturbed = (perturbed / current_sum) * target_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate

    return (best_h.tolist(), best_c5, n_points)
