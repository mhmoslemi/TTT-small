import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def generate_random_h():
        h = np.random.rand(n_points)
        s = h.sum()
        if s == 0:
            return h.copy()
        h = np.clip(h * (target_sum / s), 0, 1)
        s2 = h.sum()
        if not np.isclose(s2, target_sum, atol=1e-6):
            h = np.clip(h * (target_sum / s2), 0, 1)
        return h

    population_size = 50
    population = [h_initial.copy()]
    population[0] = np.clip(population[0], 0, 1)
    for _ in range(population_size - 1):
        population.append(generate_random_h())

    best_c5 = float('inf')
    best_h = None
    F = 0.8
    CR = 0.9

    while time.time() - start_time < budget_s - 5:
        for idx in range(population_size):
            p = population[idx].copy()
            others = [i for i in range(population_size) if i != idx]
            a, b, c = np.random.choice(others, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.where(np.random.rand(n_points) < CR, mutant, p)
            trial = np.clip(trial, 0, 1)
            s = trial.sum()
            if not np.isclose(s, target_sum, atol=1e-6):
                trial = np.clip(trial * (target_sum / s), 0, 1)
            trial_c5 = compute_c5(trial)
            p_c5 = compute_c5(p)
            if trial_c5 < p_c5:
                population[idx] = trial
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()

    if best_h is None:
        best_c5 = min(compute_c5(ind) for ind in population)
        best_h = population[np.argmin([compute_c5(ind) for ind in population])]

    # Enforce constraints one final time
    best_h = np.clip(best_h, 0, 1)
    s = best_h.sum()
    if not np.isclose(s, target_sum, atol=1e-6):
        best_h = np.clip(best_h * (target_sum / s), 0, 1)
    return best_h, best_c5, n_points
