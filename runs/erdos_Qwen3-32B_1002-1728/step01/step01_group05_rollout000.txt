import numpy as np
import time
from scipy.optimize import differential_evolution, Bounds, minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if np.isclose(current_sum, 0):
        return np.full_like(h, 0.5)
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_final = np.clip(h_scaled, 0, 1)
    return h_final

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    # Perturb initial guess for global search
    perturbation = np.random.normal(0, 0.1, size=n_points)
    h_perturbed = h0 + perturbation
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed_projected = project_h(h_perturbed, target_sum)
    best_h = h_perturbed_projected
    best_c5 = compute_c5(best_h, dx)
    
    # Differential evolution (global exploration)
    bounds = [(0, 1)] * n_points
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Projected objective for DE
    def de_objective(x):
        return compute_c5(project_h(x, target_sum), dx)
    
    if remaining > 0:
        result_de = differential_evolution(
            de_objective,
            bounds,
            strategy='best1bin',
            maxiter=20,
            popsize=15,
            tol=1e-5,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            polish=False,
            init='sobol'
        )
        h_de = result_de.x
        h_de = project_h(h_de, target_sum)
        current_c5 = compute_c5(h_de, dx)
        if current_c5 < best_c5:
            best_h = h_de
            best_c5 = current_c5
    
    # Local refinement with SLSQP
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_local = Bounds(lb=0.0, ub=1.0)
    
    if not np.isclose(np.sum(best_h), target_sum, atol=1e-6):
        best_h = project_h(best_h, target_sum)
    
    if time.time() - start_time < budget_s:
        # SLSQP with trust-like parameter
        result = minimize(
            lambda x: compute_c5(x, dx),
            x0=best_h,
            method='SLSQP',
            bounds=bounds_local,
            constraints=constraints,
            options={
                'maxiter': 100,
                'ftol': 1e-8
            }
        )
        if result.success:
            new_h = result.x
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
    
    # Final constraint check
    best_h = np.clip(best_h, 0.0, 1.0)
    if not np.isclose(np.sum(best_h), target_sum, atol=1e-6):
        best_h = project_h(best_h, target_sum)
    
    return (best_h, best_c5, n_points)
