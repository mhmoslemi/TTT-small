import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2
    h_best = h.copy()
    best_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
    
    # Parameters
    alpha = 0.05
    alpha_decay_factor = 0.95
    no_improve_threshold = 50
    perturb_strength = 0.1
    
    # Track improvements
    no_improve_count = 0
    alpha_curr = alpha
    
    while time.time() - start_time < budget_s:
        # Compute current overlap max
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        # Compute gradient for current k
        g = np.zeros(n_points)
        if k >= 0:
            if k < n_points:
                indices = np.arange(n_points - k)
                g[indices] += 1 - h[indices + k]
                g[indices + k] -= h[indices]
        else:
            k_val = -k
            if k_val < n_points:
                indices = np.arange(k_val, n_points)
                g[indices] += 1 - h[indices - k_val]
                g[indices - k_val] -= h[indices]
        
        # Gradient step
        h = h - alpha_curr * g
        
        # Clip and scale
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped > 1e-10:
            h_scaled = h_clipped * (desired_sum / sum_clipped)
            h_clipped = np.clip(h_scaled, 0, 1)
        h = h_clipped
        
        # Evaluate improvement
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_current) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = h.copy()
            no_improve_count = 0
            alpha_curr = min(alpha_curr * 1.05, alpha * 3)

                h = np.clip(h + perturbation, 0, 1)
