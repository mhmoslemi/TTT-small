import numpy as np
import time

def tournament_selection(population, fitness, tournament_size=3):
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[0]
    for idx in indices[1:]:
        if fitness[idx] < fitness[best_idx]:  # lower is better
            best_idx = idx
    return population[best_idx]

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    def repair(individual, target_sum):
        clipped = np.clip(individual, 0, 1)
        s = np.sum(clipped)
        if s == 0:
            return initial_h_values.copy()
        scaled = clipped * (target_sum / s)
        new_clipped = np.clip(scaled, 0, 1)
        new_s = np.sum(new_clipped)
        if new_s == 0:
            return initial_h_values.copy()
        final_scaled = new_clipped * (target_sum / new_s)
        final_scaled = np.clip(final_scaled, 0, 1)
        s = np.sum(final_scaled)
        if not np.isclose(s, target_sum, atol=1e-6):
            final_scaled = final_scaled * (target_sum / s)
        return final_scaled

    def mutate(individual, mutation_strength=0.05):
        noise = np.random.normal(0, mutation_strength, size=individual.shape)
        mutated = individual + noise
        return repair(mutated, target_sum)

    pop_size = 50
    elite_pct = 0.1
    mutation_strength = 0.05
    tournament_size = 3

    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        candidate = repair(h_rand, target_sum)
        population.append(candidate)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    while time.time() - start_time < budget_s - 5:
        fitness = [compute_c5(ind, dx) for ind in population]
        current_min_c5 = min(fitness)
        if current_min_c5 < best_c5:
            best_c5 = current_min_c5
            best_h = population[fitness.index(current_min_c5)].copy()

        elite_count = int(pop_size * elite_pct)
        elite_indices = np.argsort(fitness)[:elite_count]
        elite_individuals = [population[i] for i in elite_indices]

        new_population = elite_individuals.copy()
        while len(new_population) < pop_size:
