    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for seed in range(n_perturbations):
        if time.time() - start_time > budget_s - 20:
            break

        np.random.seed(42 + seed)
        perturbation = np.random.normal(0.0, 0.1, n_points)
        new_h = np.clip(h_initial + perturbation, 0.0, 1.0)
        current_sum = np.sum(new_h)
        if not np.isclose(current_sum, n_points / 2, atol=1e-8):
            new_h = np.clip(new_h * (n_points / 2) / current_sum, 0.0, 1.0)

        current_h = new_h.copy()

        for beta in beta_sequence:
            if time.time() - start_time > budget_s - 20:
                break

            def objective_with_beta(h_vec):
                h = h_vec.reshape(n_points)
                ones_minus_h = 1.0 - h
                cross_corr = np.correlate(h, ones_minus_h, mode='full') * dx
                return np.log(np.sum(np.exp(beta * cross_corr)))

            result = minimize(objective_with_beta, current_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
            if result.success:
                current_h = result.x.copy()

        if time.time() - start_time < budget_s - 20:
            def objective_fine(h_vec):
                h = h_vec.reshape(n_points)
                ones_minus_h = 1.0 - h
                cross_corr = np.correlate(h, ones_minus_h, mode='full') * dx
                return np.max(cross_corr)

            result_fine = minimize(objective_fine, current_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
            if result_fine.success:
                current_h = result_fine.x.copy()

        current_c5 = evaluate_erdos_solution.evaluate_C5(current_h, n_points, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    return best_h, best_c5, n_points
