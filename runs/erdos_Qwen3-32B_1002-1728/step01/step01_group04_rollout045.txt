import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def adjust_candidate(candidate):
        candidate_clipped = np.clip(candidate, 0, 1)
        s = candidate_clipped.sum()
        if s == 0:
            return candidate.copy()
        candidate_scaled = candidate_clipped * (target_sum / s)
        return np.clip(candidate_scaled, 0, 1)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    # Initialize population
    population = [h_initial.copy()]
    for _ in range(49):
        noise = np.random.normal(0, 0.1, size=h_initial.shape)
        new_candidate = h_initial + noise
        population.append(adjust_candidate(new_candidate))

    fitness = [compute_c5(ind) for ind in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()
    best_c5 = fitness[best_index]

    F = 0.8
    CR = 0.9
    population_size = len(population)

    while time.time() - start_time < budget_s - 5:
        for i in range(population_size):
            indices = list(range(population_size))
            np.random.shuffle(indices)
            a, b, c = indices[:3]
            mutant = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            j_rand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
            trial_adjusted = adjust_candidate(trial)
            trial_c5 = compute_c5(trial_adjusted)
            if trial_c5 < fitness[i]:
                population[i] = trial_adjusted.copy()

                    best_c5 = trial_c5
