import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    
    initials = [h_initial.copy()]

    for k in [3, 4, 5, 7, 9]:  # Varying periodic patterns

        x_val = 0.4
        for i in range(n_points):
            if i % k == 0:
                h_pat[i] = 0.5 + x_val
            else:
                h_pat[i] = 0.5 - x_val
        h_pat = np.clip(h_pat, 0, 1)
        initials.append(h_pat)
    
    best_c5 = None
    best_h = h_initial.copy()

    for x0 in initials:
