import numpy as np
import time


def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def adjust_sum(h, target):
        current_sum = h.sum()
        delta = current_sum - target
        if np.isclose(delta, 0):
            return h
        elif delta > 0:
            # Reduce the sum: target < current_sum
            eligible = h > 0
            count = eligible.sum()
            if count == 0:
                return h
            per_elem = delta / count
            h[eligible] -= per_elem
            h = np.clip(h, 0, 1)
        else:
            # Increase the sum: target > current_sum
            delta = -delta
            eligible = h < 1
            count = eligible.sum()
            if count == 0:
                return h
            per_elem = delta / count
            h[eligible] += per_elem
            h = np.clip(h, 0, 1)
        return h

    np.random.seed(seed)
    h = h_initial.copy()
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        # Project onto [0, 1]
        h = np.clip(h, 0, 1)

        # Correct sum constraint
        h = adjust_sum(h, target_sum)

        # Perturb with noise relative to current h
        noise = np.random.normal(0, 0.02, n_points)
        h += noise
        h = np.clip(h, 0, 1)

        # Evaluate current C5
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)

        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()

    # Final validation of constraints

    best_h = adjust_sum(best_h, target_sum)
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr)

    return best_h, c5_bound, n_points
