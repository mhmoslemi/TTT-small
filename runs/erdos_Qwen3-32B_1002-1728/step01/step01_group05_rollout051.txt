    # Add roots of unity patterns
    for p in [4, 5, 6, 8, 12]:
        h = generate_roots(n_points, p)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        h = np.clip(h, 0, 1)
        initial_candidates.append(h)
    
    # Add perturbed initializations
    for _ in range(3):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h0 + perturbation, 0.0, 1.0)
        h_sum = h.sum()
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        h = np.clip(h, 0.0, 1.0)
        initial_candidates.append(h)
    
    # Optimize each candidate
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)
    
    for candidate in initial_candidates:
        if time.time() - start_time >= budget_s:
            break
        # Ensure candidate is feasible
        h = np.clip(candidate, 0, 1).copy()
        h_sum = h.sum()
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        h = np.clip(h, 0, 1)
        # Run optimization
        result = minimize(
            lambda x: compute_c5(x, dx),
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success:
            new_h = result.x
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
    
    return (best_h, best_c5, n_points)
