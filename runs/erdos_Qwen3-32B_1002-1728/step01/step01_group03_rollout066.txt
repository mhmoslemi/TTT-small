        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()

        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        if abs(current_sum) < 1e-10:
            return h_pert  # to avoid division by zero, though unlikely

import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, noise=0.2):
        h_pert = h.copy()
        r = np.random.uniform(-1, 1, n_points)
        r -= np.mean(r)
        h_pert += noise * r
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        if abs(current_sum) < 1e-10:
            return h_pert
        h_pert *= target_sum / current_sum
        return h_pert
    
    current_noise = 0.2
    improvement_count = 0
    cooldown_since = start_time
    cooldown_time = 10
    max_tries = 100
    
    while time.time() - start_time < budget_s - 30:
        x0 = get_perturbed(best_h, current_noise)
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5 - 1e-5:  # Require noticeable improvement
                best_h[:] = res.x
                best_c5 = curr_val
                cooldown_since = time.time()
                improvement_count = 0
                current_noise = min(current_noise * 1.2, 0.5)
            else:
                improvement_count += 1
                if improvement_count > 3:  # Reduce noise after 3 failed tries
                    current_noise *= 0.8
                    improvement_count = 0
        # Enforce cooldown to avoid getting stuck
        if time.time() - cooldown_since > cooldown_time:
            current_noise = min(current_noise * 1.5, 0.5)
            cooldown_since = time.time()
    
    return (best_h, best_c5, n_points)
