import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    n_particles = 50
    max_iter = 10000
    c1, c2 = 2.0, 2.0
    w_max, w_min = 0.9, 0.4
    v_max = 0.1

    rng = np.random.RandomState(seed)

    # Initialize positions
    positions = np.clip(rng.normal(loc=h_init, scale=0.05, size=(n_particles, n_points)), 0, 1)
    for i in range(n_particles):
        s = positions[i].sum()
        if s != 0:
            positions[i] *= target_sum / s
        else:
            positions[i] = h_init.copy()

    velocities = rng.uniform(-v_max, v_max, size=positions.shape)

    pbest_pos = positions.copy()
    pbest_obj = np.array([objective(p, dx, n_points) for p in positions])
    gbest_idx = np.argmin(pbest_obj)
    gbest = pbest_pos[gbest_idx].copy()
    best_c5 = pbest_obj[gbest_idx]

    while time.time() - start_time < budget_s:
        t_remaining = budget_s - (time.time() - start_time)
        w = w_max - (w_max - w_min) * (budget_s - t_remaining) / budget_s
        w = np.clip(w, w_min, w_max)

        for i in range(n_particles):
            r1, r2 = rng.rand(), rng.rand()
            vel_cog = c1 * r1 * (pbest_pos[i] - positions[i])
            vel_soc = c2 * r2 * (gbest - positions[i])
            velocities[i] = w * velocities[i] + vel_cog + vel_soc
            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)

            s = positions[i].sum()
            if s != 0:
                positions[i] *= target_sum / s
            else:
                positions[i] = h_init.copy()

        current_obj = np.array([objective(p, dx, n_points) for p in positions])

        for i in range(n_particles):
            if current_obj[i] < pbest_obj[i]:
                pbest_pos[i] = positions[i].copy()
                pbest_obj[i] = current_obj[i]

        current_best = np.argmin(pbest_obj)
        if pbest_obj[current_best] < best_c5:
            gbest = pbest_pos[current_best].copy()
            best_c5 = pbest_obj[current_best]

    assert abs(np.sum(gbest) - target_sum) < 1e-6
    assert np.all((gbest >= -1e-6) & (gbest <= 1.0 + 1e-6))
    return (gbest.tolist(), best_c5, n_points)

def objective(h_vec, dx, n_points):
    h = h_vec.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
