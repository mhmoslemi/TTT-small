                          options={'maxiter': 100, 'ftol': 1e-6})
        if result.success:
            new_h = result.x
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        trials += 1
        perturb_amp *= perturb_decay

    return (best_h, best_c5, n_points)
