import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    import time
    
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1).copy()
        sum_h = h.sum()
        if sum_h == 0:
            return float('inf')
        scaling = target_sum / sum_h
        h = np.clip(h * scaling, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constrain_h(h):
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h == 0:
            scaling = 1.0
        else:
            scaling = target_sum / sum_h
        return np.clip(h * scaling, 0, 1)
    
    result = differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        polish=True,
        init='sobol',
        atol=1e-6
    )
    
    h_opt = constrain_h(result.x)
    c5 = objective(h_opt)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.8:
        perturbed = result.x + np.random.default_rng(seed).normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        result2 = differential_evolution(
            objective,
            bounds,
            strategy='rand1bin',
            maxiter=50,
            popsize=20,
            seed=seed + 1,
            init=perturbed.reshape((1, -1))
        )
        h2 = constrain_h(result2.x)
        c5_2 = objective(h2)
        if c5_2 < c5:
            h_opt, c5 = h2, c5_2
    
    assert abs(h_opt.sum() * dx - 1.0) < 1e-6, "Integral constraint violated"
    assert np.allclose(h_opt, np.clip(h_opt, 0, 1)), "Value bounds violated"
    
    return (h_opt.tolist(), c5, n_points)
