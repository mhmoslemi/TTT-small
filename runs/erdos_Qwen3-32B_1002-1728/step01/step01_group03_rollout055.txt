import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, noise):
        h_pert = h.copy()
        perturbation = np.random.uniform(-noise, noise, h_pert.shape)
        h_pert += perturbation
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = h_pert.sum()
        if current_sum != 0:
            scale = target_sum / current_sum
            h_pert *= scale
            h_pert = np.clip(h_pert, 0, 1)  # Clip after scaling
        return h_pert
    
    # Perturbation phases with decreasing noise and increasing attempts
    pert_phases = [
        {'noise': 0.2, 'attempts': 40},
        {'noise': 0.15, 'attempts': 35},
        {'noise': 0.1, 'attempts': 30},
        {'noise': 0.05, 'attempts': 25},
        {'noise': 0.02, 'attempts': 20},
    ]
    
    for phase in pert_phases:
        noise = phase['noise']
        max_attempts = phase['attempts']
        attempts = 0

            if x0_sum != target_sum:
                x0 = x0 * (target_sum / x0_sum)
