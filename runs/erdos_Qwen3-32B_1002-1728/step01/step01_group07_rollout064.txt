import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = np.array(initial_h_values).flatten()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_individual(h):
        h = np.clip(h, 0, 1)
        h_scale = h / h.sum() * target_sum
        return h_scale

    # Initialize population
    pop_size = 50
    pop = [h_init.copy() for _ in range(pop_size // 5)]
    for _ in range(pop_size - pop_size // 5):
        perturbed = repair_individual(h_init + np.random.normal(0, 0.05, n_points))
        pop.append(perturbed)
    pop = np.array(pop)

    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fit = np.array([compute_c5(ind) for ind in pop])
        if np.min(fit) < best_c5:
            best_idx = np.argmin(fit)
            best_h = pop[best_idx].copy()
            best_c5 = fit[best_idx]

        # Select parents (best half)
        sorted_idx = np.argsort(fit)
        parent_count = pop_size // 2
        parents = pop[sorted_idx[:parent_count]]

        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - parent_count:
            # Crossover
            if np.random.rand() < 0.8:
                a, b = parents[np.random.choice(parent_count, 2)]
                beta = 0.5
                child = (1 - beta) * a + beta * b

                a = parents[np.random.choice(parent_count)]

            # Repair child
            repaired = repair_individual(child)
            offspring.append(repaired)

        # Replace worst half
        pop[sorted_idx[parent_count:]] = offspring
        # Elitism: enforce always keeping the best
        pop[0] = repair_individual(best_h)

    return (best_h.tolist(), best_c5, n_points)
