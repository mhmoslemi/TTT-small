                else:

                    grad[i] += (1 - 2*h[i])
                count +=1

                    grad[i] += (1 - h[j])
                    grad[j] += (-h[i])
                count +=1

                valid = n_points - l
                for i in range(l, n_points):
                    j = i - l
                    grad[i] += (1 - h[j])
                    grad[j] += (-h[i])
                count +=1
        if count >0:
            grad /= count
        return grad

        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    best_c5 = np.max(compute_all_overlaps(h_best))

    step_size = 0.1
    iterations =0
    while time.time() - start_time < budget_s - 5:
        iterations +=1

            h_best = project(h_best)

        # Check time
        if time.time() - start_time > budget_s - 5:
            break

    return h_best, best_c5, n_points
