                      constraints=constr, options={'maxiter': 100, 'ftol': 1e-6})
    if result.success and (time.time() - start_time < budget_s):
        final_x = result.x
        final_c5 = refined_objective(final_x)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_x = final_x.copy()

    # Final validation
    final_c5 = compute_c5(project_h(best_x, n_points), dx, n_points)
    return (best_x, final_c5, n_points)
