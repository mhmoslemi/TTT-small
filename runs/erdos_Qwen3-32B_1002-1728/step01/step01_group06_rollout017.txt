import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Initialize h using the provided initial_h_values or default
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h_sum = np.sum(h)
        h *= (n / 2) / h_sum
        h = np.clip(h, 0, 1)
    else:
        n = 280

        mid = n // 2
        h[mid-5:mid+5] = 0.5
        h *= (n / 2) / h.sum()
        h = np.clip(h, 0, 1)

    best_h = h.copy()

    lambda_ = 10
