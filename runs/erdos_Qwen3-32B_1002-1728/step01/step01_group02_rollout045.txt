            optimized_h = repair_h(optimized_h, n_points)
            current_c5 = evaluate_C5(optimized_h, n_points, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h.copy()
    
    best_h = repair_h(best_h, n_points)
    return best_h, best_c5, n_points
