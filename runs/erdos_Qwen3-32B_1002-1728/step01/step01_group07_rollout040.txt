    import numpy as np

    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    np.random.seed(seed)

    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    for i in range(n_particles):
        perturbed = h_init + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        perturbed = perturbed * (target_sum / sum_perturbed)
        particles[i] = perturbed

    velocities = 0.1 * np.random.uniform(-1, 1, (n_particles, n_points))

    pbest = particles.copy()
    pbest_c5 = np.zeros(n_particles)
    for i in range(n_particles):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        pbest_c5[i] = np.max(corr) * dx
    gbest = particles[pbest_c5.argmin()].copy()
    gbest_c5 = np.min(pbest_c5)

    v_max = 0.1
    cognitive = 2.0
    social = 2.0

    while time.time() - start_time < budget_s * 0.95:
        elapsed = time.time() - start_time
        time_ratio = elapsed / budget_s
        inertia = max(0.4, 0.9 - time_ratio * 0.5)

        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)

        for i in range(n_particles):
            cog_comp = cognitive * r1[i] * (pbest[i] - particles[i])
            soc_comp = social * r2[i] * (gbest - particles[i])
            velocities[i] = inertia * velocities[i] + cog_comp + soc_comp
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
