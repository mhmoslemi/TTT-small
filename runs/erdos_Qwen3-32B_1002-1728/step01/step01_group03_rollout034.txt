import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    def get_perturbed(h, noise_strength=0.1):
        h_pert = h.copy()
        h_pert += np.random.uniform(-noise_strength, noise_strength, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert *= target_sum / current_sum
        return h_pert
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Adaptive noise schedule for exploration/exploitation
    noise_schedule = []
    noise_levels = [0.3, 0.25, 0.2, 0.15, 0.1, 0.05]
    for noise in noise_levels:

    for noise_strength in noise_schedule:

        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()

    return (best_h, best_c5, n_points)
