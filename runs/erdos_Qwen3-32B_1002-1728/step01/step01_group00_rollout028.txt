import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.ascontiguousarray(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament_selection(population, c5_list, tournament_size=5):
        indices = np.random.choice(len(population), size=tournament_size, replace=False)
        best_idx = indices[np.argmin([c5_list[i] for i in indices])]
        return population[best_idx].copy()

    def ensure_feasible(candidate):
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        if current_sum > 1e-10:
            scaling_factor = (n_points / 2) / current_sum
            candidate *= scaling_factor
            candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) != 0:
            offset = (np.sum(candidate) - n_points / 2)
            candidate[0] -= offset
        return candidate

    np.random.seed(seed)
    population_size = 50
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        perturbed = h_initial.copy() + np.random.normal(0, 0.05, n_points)
        population.append(ensure_feasible(perturbed))

    c5_list = [compute_c5(p) for p in population]
    best_idx = np.argmin(c5_list)
    best_h = population[best_idx].copy()
    best_c5 = c5_list[best_idx]

    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1
        new_population = []
        # Elitism: carry over top individuals
        elite_count = 5
        elites = [population[i] for i in np.argsort(c5_list)[:elite_count]]
        new_population.extend(elites)

        while len(new_population) < population_size:
            parent1 = tournament_selection(population, c5_list)
            parent2 = tournament_selection(population, c5_list)
            # Crossover: blend
            alpha = 0.5
            child = alpha * parent1 + (1 - alpha) * parent2
            # Mutation
            if np.random.rand() < 0.1:
                mutation_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                noise = np.random.normal(0, 0.05, size=len(mutation_indices))
                child[mutation_indices] += noise
            # Ensure feasible
            child = ensure_feasible(child)
            new_population.append(child)

        population = new_population
        c5_list = [compute_c5(p) for p in population]
        current_best = np.min(c5_list)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(c5_list)].copy()

        # Check time budget periodically
        if time.time() - start_time + 10 > budget_s:
            break

    # Final validation
    return best_h, best_c5, n_points
