import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    def generate_feasible_candidate():
        while True:
            new_h = np.random.rand(n_points)
            new_h = np.clip(new_h, 0, 1)
            s = new_h.sum()
            if s == 0:
                continue
            new_h = new_h * (target_sum / s)
            new_h = np.clip(new_h, 0, 1)
            return new_h

    pop_size = 50
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        population.append(generate_feasible_candidate())
    population = np.array(population)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    F = 0.8
    CR = 0.9

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            current_h = population[i].copy()
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            # Crossover
            cross_points = np.random.rand(n_points) < CR
            cross_points[np.random.randint(n_points)] = True  # Ensure at least one parameter is mutated
            trial = np.where(cross_points, mutant, current_h)
            # Feasibility adjustment
            trial = np.clip(trial, 0, 1)
            s = trial.sum()
            if s == 0:
                continue
            trial = trial * (target_sum / s)
            trial = np.clip(trial, 0, 1)

            trial_c5 = compute_c5(trial, dx)
            current_c5 = compute_c5(current_h, dx)

            if trial_c5 < current_c5:
                population[i] = trial
                if trial_c5 < best_c5:
                    best_h, best_c5 = trial.copy(), trial_c5
        # Early exit if timeout near
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points
