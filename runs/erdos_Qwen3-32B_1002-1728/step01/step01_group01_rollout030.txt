def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Define the objective function
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Projection function to enforce constraints
    def project(h_vec):
        target_sum = n_points / 2
        h = h_vec.copy()
        for _ in range(3):  # Three iterations of clip and scale
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            if s == 0:
                return h  # This shouldn't happen
            scale = target_sum / s
            h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    # Initialize population
    np.random.seed(seed)
    pop_size = 50
    population = []
    # First half: perturbations of initial_h_values
    for _ in range(pop_size // 2):
        h = initial_h_values + 0.1 * np.random.randn(n_points)
        population.append(project(h))
    # Second half: random vectors
    for _ in range(pop_size // 2):
        h = np.random.rand(n_points)
        population.append(project(h))
    population = np.array(population)
    
    # Track best solution
    best_idx = np.argmin([evaluate(ind) for ind in population])
    best_h = population[best_idx].copy()
    best_c5 = evaluate(best_h)
    
    F = 0.8
    CR = 0.9
    
    # Main DE loop
    while time.time() - start_time < budget_s - 10:
        for i in range(pop_size):
            # Select three distinct random individuals
            candidates = [j for j in range(pop_size) if j != i]
            a, b, c = population[np.random.choice(candidates, 3, replace=False)]
            mutant = a + F * (b - c)
            # Crossover
            trial = np.copy(population[i])
            idxs = np.random.rand(n_points) < CR
            if not np.any(idxs):
                idxs[np.random.randint(n_points)] = True
            trial[idxs] = mutant[idxs]
            # Project to feasible set
            trial_projected = project(trial)
            # Evaluate trial
            trial_c5 = evaluate(trial_projected)
            # Replace if better
            if trial_c5 < evaluate(population[i]):
                population[i] = trial_projected
                # Update best solution
                if trial_c5 < best_c5:
                    best_h = trial_projected.copy()
                    best_c5 = trial_c5
        # Check if time is up
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, n_points

Now, in the objective function, the code uses np.correlate in full mode. This returns the cross-correlation for all lags. import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()
    
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project(h_vec):
        h = h_vec.copy()
        for _ in range(3):
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            if s == 0:
                return h
            scale = target_sum / s
            h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    np.random.seed(seed)
    pop_size = 50
    population = []
    for _ in range(pop_size // 2):
        h = initial_h_values + 0.1 * np.random.randn(n_points)
        population.append(project(h))
    for _ in range(pop_size // 2):
        h = np.random.rand(n_points)
        population.append(project(h))
    population = np.array(population)
    
    best_h = population[np.argmin([evaluate(ind) for ind in population])].copy()
    best_c5 = evaluate(best_h)
    F = 0.8
    CR = 0.9
    
    while time() - start_time < budget_s - 5:
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            a, b, c = population[np.random.choice(candidates, 3, replace=False)]
            mutant = a + F * (b - c)
            trial = population[i].copy()
            mask = np.random.rand(n_points) < CR
            if not np.any(mask):
                mask[np.random.randint(n_points)] = True
            trial[mask] = mutant[mask]
            trial_projected = project(trial)
            trial_c5 = evaluate(trial_projected)
            if trial_c5 < evaluate(population[i]):
                population[i] = trial_projected
                if trial_c5 < best_c5:
                    best_h[:] = trial_projected
                    best_c5 = trial_c5
        if time() - start_time > budget_s - 5:
            break
    
    return best_h, best_c5, n_points
