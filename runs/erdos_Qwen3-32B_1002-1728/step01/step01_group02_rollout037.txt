                args=(n_points, dx, beta),
                method='SLSQP',
                bounds=bounds,
                constraints=cons
            )
            if result.success:
                h_current = result.x.copy()
                current_c5 = evaluate_C5(h_current, n_points, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    h_best = h_current.copy()

    return h_best, best_c5, n_points
