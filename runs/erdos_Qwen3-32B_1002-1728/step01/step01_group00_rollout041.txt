import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    best_h = h_initial.copy()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    current_h = best_h.copy()
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps and identify worst k
        corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
        max_idx = np.argmax(corr)
        k_max = max_idx
        
        # Compute gradient for worst k
        def compute_grad(h_vec, k):
            n = len(h_vec)
            grad = np.zeros(n)
            indices = np.arange(n)
            forward_mask = indices + k < n
            grad[forward_mask] = 1 - h_vec[indices[forward_mask] + k]
            backward_mask = indices - k >= 0
            grad[backward_mask] -= h_vec[indices[backward_mask] - k]
            grad *= dx  # Scale by dx
            return grad
        
        grad = compute_grad(current_h, k_max)
        
        # Gradient step with adaptive learning rate
        learning_rate = 0.2 if (time.time() - start_time) < 300 else 0.05
        current_h -= grad * learning_rate * (1.0 + 0.1 * np.random.rand())
        
        # Project onto feasible region
        current_h = np.clip(current_h, 0, 1)
        target_sum = n_points / 2
        current_sum = current_h.sum()

        if np.abs(delta) > 1e-8:
            # Adjust using median non-saturated element
