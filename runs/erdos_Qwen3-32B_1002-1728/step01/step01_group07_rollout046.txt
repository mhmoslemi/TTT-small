import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)
    
    population_size = 50
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        pert = h_init + np.random.normal(0, 0.05, n_points)
        population.append(repair(pert))
    population = np.array(population)
    
    best_c5 = float('inf')
    best_h = None
    
    while time.time() - start_time < budget_s:
        # Evaluate all individuals
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Track best
        idx_best = np.argmin(fitness)
        current_best = fitness[idx_best]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[idx_best].copy()
        
        # Select parents (top 50%)
        ranks = np.argsort(fitness)
        parents = population[ranks[:population_size//2]]
        
        # Create offspring
        offspring = []
        for _ in range(population_size - len(parents)):
            # Blend crossover
            p1, p2 = np.random.choice(len(parents), 2)
            child = 0.5 * parents[p1] + 0.5 * parents[p2]
            # Mutation
            if np.random.rand() < 0.1:
                child += np.random.normal(0, 0.02, n_points)
            child = repair(child)
            offspring.append(child)
        
        # Replace worst 50%
        population = parents
        population = np.vstack([population, np.array(offspring)])
    
    return (best_h.tolist(), best_c5, n_points)
