    while time.time() - start_time < budget_s - 5:
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            a_idx, b_idx, c_idx = idxs
            a, b, c = population[a_idx], population[b_idx], population[c_idx]
            mutant = a + 0.8 * (b - c)
            cross_mask = np.random.rand(n_points) < 0.9
            trial = np.where(cross_mask, mutant, population[i])
            trial_repaired = repair_vector(trial, target_sum)
            trial_c5 = compute_c5(trial_repaired, dx)
            current_c5 = compute_c5(population[i], dx)
            if trial_c5 < current_c5:
                population[i] = trial_repaired
                if trial_c5 < best_c5:
                    best_h = trial_repaired.copy()
                    best_c5 = trial_c5

    return best_h, best_c5, n_points
