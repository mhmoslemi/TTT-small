import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)

    population_size = 50
    population = []
    # Add 5 copies of initial_h_values
    for _ in range(5):
        population.append(h_init.copy())
    # Add 45 perturbed variants
    for _ in range(45):
        perturbed = h_init + np.random.normal(0, 0.05, n_points)
        perturbed = repair(perturbed, target_sum)
        population.append(perturbed)
    population = np.array(population)

    best_c5 = float('inf')
    best_h = None

    def repair(h_vec, target_sum):
        h = np.clip(h_vec, 0, 1)
        total = h.sum()
        if total == 0:
            return h.copy()
        h = h * (target_sum / total)
        return h

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Genetic Algorithm loop
    while time.time() - start_time < budget_s:
        c5_values = np.array([compute_c5(ind) for ind in population])
        current_best_idx = np.argmin(c5_values)
        current_best_c5 = c5_values[current_best_idx]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()

        # Sort population by C5
        sorted_indices = np.argsort(c5_values)
        sorted_pop = population[sorted_indices]

        # Select top 50% as parents
        parents = sorted_pop[:25]

        # Generate 25 new offspring
        new_offspring = []
        for _ in range(25):
            # Select parents
            i1, i2 = np.random.choice(25, size=2)
            p1, p2 = parents[i1], parents[i2]
            child = (p1 + p2) / 2.0

            # Mutation
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                noise = np.random.normal(0, 0.02)
                child[idx] += noise
                child[idx] = np.clip(child[idx], 0, 1)

            # Repair
            child = repair(child, target_sum)
            new_offspring.append(child)

        new_offspring = np.array(new_offspring)
        population = np.concatenate([parents, new_offspring])

    # Final checks
    assert np.all(0 <= best_h) and np.all(best_h <= 1), "Value bounds violated"
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"

    return (best_h.tolist(), best_c5, n_points)
