import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr) * dx)

    def project_h(h_vec, target_sum=n_points / 2):
        h = np.clip(h_vec, 0, 1)
        current_sum = float(np.sum(h))
        if abs(current_sum - target_sum) < 1e-8:
            return h
        if current_sum <= 1e-8:
            return np.full_like(h, target_sum / n_points)
        scale_factor = target_sum / current_sum
        h_scaled = h * scale_factor
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    np.random.seed(int(1000 * (start_time - int(start_time))) + seed)
    particles = []
    velocities = []

    # Initialize half from initial, half random
    for _ in range(n_points // 2):
        base = initial_h_values + np.random.normal(0, 0.05, n_points)
        base = np.clip(base, 0, 1)
        base = project_h(base)
        particles.append(base)
    for _ in range(n_points // 2):
        rand_h = np.random.uniform(0, 1, n_points)
        rand_h = project_h(rand_h)
        particles.append(rand_h)

    particles = np.array(particles)
    velocities = np.random.uniform(-0.1, 0.1, (len(particles), n_points))

    pbest = particles.copy()
    pbest_scores = np.array([objective(p) for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    c1, c2 = 1.42, 1.42

    while time.time() - start_time < budget_s - 5:
        t_frac = (time.time() - start_time) / budget_s
        w = max(0.4, 0.8 - 0.4 * t_frac)  # Inertia decay

        for i in range(len(particles)):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +

        current_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[current_gbest_idx] < gbest_score:
