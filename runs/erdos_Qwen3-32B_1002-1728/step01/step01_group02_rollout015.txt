        h_prev *= scaling_factor
        h_prev = np.clip(h_prev, 0, 1)

    best_c5 = evaluate_C5(h_prev, n_points, dx)
    best_h = h_prev.copy()

    # Precompute valid shifts for each k
    valid_shifts = []
    for k in range(- (n_points - 1), n_points):
        i_start = max(0, -k)
        i_end = min(n_points - 1, n_points - 1 - k)
        if i_start <= i_end:
            valid_shifts.append(k)

    while time.time() - start_time < budget_s - 5:
        # Create variables and constraints for LP
        h = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1,
        ]
        for k in valid_shifts:
            expr = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    expr += (1 - h_prev[j]) * h[i] - h_prev[i] * h[j]
            expr *= dx
            constraints.append(t >= expr)

        problem = cp.Problem(cp.Minimize(t), constraints)
        problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status != cp.OPTIMAL:
            # Perturb and continue
            perturbation = np.random.normal(0, 0.01, n_points)
            h_prev = np.clip(h_prev + perturbation, 0, 1)
            sum_h = np.sum(h_prev)
            if abs(sum_h - (n_points / 2)) > 1e-8:
                scaling_factor = (n_points / 2) / sum_h
                h_prev *= scaling_factor
                h_prev = np.clip(h_prev, 0, 1)
            continue

        current_h = np.clip(h.value, 0, 1)
        sum_h = np.sum(current_h)
        if abs(sum_h - (n_points / 2)) > 1e-8:
            scaling_factor = (n_points / 2) / sum_h
            current_h *= scaling_factor
            current_h = np.clip(current_h, 0, 1)

        h_prev = current_h.copy()

        current_c5 = evaluate_C5(h_prev, n_points, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_prev.copy()

        # Perturbation for exploration
        perturbation = np.random.normal(0, 0.01, n_points)
        h_prev = np.clip(h_prev + perturbation, 0, 1)
        sum_h = np.sum(h_prev)
        if abs(sum_h - (n_points / 2)) > 1e-8:
            scaling_factor = (n_points / 2) / sum_h
            h_prev *= scaling_factor
            h_prev = np.clip(h_prev, 0, 1)

    return best_h, best_c5, n_points
