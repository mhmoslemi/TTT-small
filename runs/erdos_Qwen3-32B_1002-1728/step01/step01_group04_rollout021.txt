import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def normalize_and_clip(x):
        x_clipped = np.clip(x, 0, 1)
        s = x_clipped.sum()
        if s == 0:
            x_clipped[:] = 0.5
        else:
            x_scaled = x_clipped * (target_sum / s)
            x_clipped = np.clip(x_scaled, 0, 1)
        return x_clipped

    def tournament_selection(population, fitness):
        indices = np.random.choice(len(population), size=3)
        candidates = [population[i] for i in indices]
        scores = [fitness[i] for i in indices]
        return candidates[np.argmin(scores)]

    pop_size = 50
    elite_size = 5
    mutation_rate = 0.1
    mutation_strength = 0.05

    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        temp = np.random.rand(n_points)
        population.append(normalize_and_clip(temp))

    fitness = [evaluate(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    while time.time() - start_time < budget_s - 5:
        new_pop = []
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_pop.append(population[idx].copy())
        # Generate remaining
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            child = normalize_and_clip(child)
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child += noise
                child = normalize_and_clip(child)
            new_pop.append(child)
        population = new_pop
        fitness = [evaluate(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]

    return best_h, best_c5, n_points
