import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def project_h(h):
        h = np.clip(h, 0, 1).copy()
        s = h.sum()
        if np.isclose(s, target_sum):
            return h
        while not np.isclose(s, target_sum, atol=1e-6):
            if s < target_sum:
                delta = target_sum - s
                h += delta / n_points
            else:
                delta = s - target_sum
                h -= delta / n_points
            h = np.clip(h, 0, 1)
            s = h.sum()
        return h

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    n_particles = 50
    positions = []
    for _ in range(n_particles // 2):
        pert = np.random.normal(0, 0.05, n_points)
        new_h = project_h(np.clip(initial_h_values + pert, 0, 1))
        positions.append(new_h)
    for _ in range(n_particles // 2):
        new_h = project_h(np.random.uniform(0, 1, n_points))
        positions.append(new_h)
    velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(n_particles)]
    personal_best = positions.copy()
    pb_scores = [objective(p) for p in positions]
    gbest_idx = np.argmin(pb_scores)
    gbest = personal_best[gbest_idx].copy()
    gbest_score = pb_scores[gbest_idx]

    w_start, w_end, c1, c2 = 0.8, 0.4, 1.42, 1.42

    while time() - start_time < budget_s - 10:
        elapsed = time() - start_time
        w = w_start - (w_start - w_end) * (elapsed / (budget_s - 10))
        for i in range(n_particles):
            pos = positions[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            pbest = personal_best[i]
            gbest_diff = (c1 * r1 * (pbest - pos) + c2 * r2 * (gbest - pos))
            velocities[i] = w * vel + gbest_diff
            new_pos = project_h(pos + velocities[i])
            positions[i] = new_pos
            score = objective(new_pos)
            if score < pb_scores[i] - 1e-8:
                personal_best[i] = new_pos.copy()
                pb_scores[i] = score
                if score < gbest_score:
                    gbest[:] = new_pos
                    gbest_score = score
        if time() - start_time >= budget_s - 10:
            break

    return gbest, gbest_score, n_points
