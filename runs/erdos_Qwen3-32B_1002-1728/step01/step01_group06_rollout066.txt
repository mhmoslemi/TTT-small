import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initialize h using initial_h_values
    if 'initial_h_values' in globals():

    h = np.clip(h, 0, 1)
    
    best_h = h.copy()

    sigma = 0.1
    lambda_ = 10
    no_improvement = 0
    adaptation_interval = 50
    
    while time.time() - start_time < budget_s - 0.1:
        offspring = []
        for _ in range(lambda_):
            # Mutate and repair
            new_h = best_h + sigma * np.random.randn(n)
            new_h_clipped = np.clip(new_h, 0, 1)

                new_scaled = np.zeros_like(new_h_clipped)
