    best_idx = indices[np.argmin(fitness[indices])]
    return population[best_idx].copy()

def mutate_h(h, mutation_rate, n_points, target_sum):
    mutation = np.random.normal(0, mutation_rate, n_points)
    h_mutated = h + mutation
    h_clipped = np.clip(h_mutated, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return h.copy()
    scaling = target_sum / current_sum
    h_scaled = h_clipped * scaling
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    POP_SIZE = 40
    MUTATION_RATE = 0.1
    TOURNAMENT_SIZE = 3
    ELITISM_COUNT = 2

    population = [h_init]
    for _ in range(POP_SIZE - 1):
        while True:
            h = np.random.uniform(0, 1, n_points)
            current_sum = h.sum()
            if current_sum == 0:
                continue
            scale_factor = target_sum / current_sum
            h_scaled = h * scale_factor
            h_clipped = np.clip(h_scaled, 0, 1)
            new_sum = h_clipped.sum()
            if np.isclose(new_sum, target_sum, rtol=1e-5):
                population.append(h_clipped)
                break
            diff = target_sum - new_sum
            h_clipped += diff / n_points
            h_clipped = np.clip(h_clipped, 0, 1)
            if np.isclose(h_clipped.sum(), target_sum, rtol=1e-5):
                population.append(h_clipped)
                break
    population = np.array(population)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)

    while time.time() - start_time < budget_s - 1:
        fitness = np.array([compute_c5(h, dx) for h in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h, best_c5 = population[current_best_idx].copy(), fitness[current_best_idx]

        elite = population[np.argmin(fitness)]
        parent_indices = np.argsort(fitness)[:POP_SIZE//2]

        new_population = [elite]
        while len(new_population) < POP_SIZE:
            parent1 = tournament_selection(population, fitness, TOURNAMENT_SIZE)
            parent2 = tournament_selection(population, fitness, TOURNAMENT_SIZE)
            cross_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            child2 = np.concatenate([parent2[:cross_point], parent1[cross_point:]])
            child1 = mutate_h(child1, MUTATION_RATE, n_points, target_sum)
            child2 = mutate_h(child2, MUTATION_RATE, n_points, target_sum)
            new_population.append(child1)
            if len(new_population) < POP_SIZE:
                new_population.append(child2)
        population = np.array(new_population)

    return best_h, best_c5, n_points
