            old_n = best_n
            old_x = np.linspace(0, 2, old_n)
            new_x = np.linspace(0, 2, grid)
            h_resampled = np.interp(new_x, old_x, best_h)
            h_resampled = np.clip(h_resampled, 0, 1)
            total = np.sum(h_resampled)
            if total != 0:
                h_resampled *= (grid / 2) / total
            h_resampled = np.clip(h_resampled, 0, 1)
            current_h = h_resampled
        else:
            if 'initial_h_values' in globals():
                old_n = len(initial_h_values)
                old_x = np.linspace(0, 2, old_n)
                new_x = np.linspace(0, 2, grid)
                h_resampled = np.interp(new_x, old_x, initial_h_values)
                h_resampled = np.clip(h_resampled, 0, 1)
                total = np.sum(h_resampled)
                if total != 0:
                    h_resampled *= (grid / 2) / total
                h_resampled = np.clip(h_resampled, 0, 1)
                current_h = h_resampled
            else:
                h = np.zeros(grid)
                mid = grid // 2
                h[mid-5:mid+5] = 0.5
                total = np.sum(h)
                if total != 0:
                    h *= (grid / 2) / total
                h = np.clip(h, 0, 1)
                current_h = h

        if grid == 70:
            pop_size = 30
            max_iterations = 300
            pa = 0.25
        else:
            pop_size = 50
            max_iterations = 200
            pa = 0.25

        nests = [repair(current_h + np.random.normal(0, 0.05, grid), grid) for _ in range(pop_size)]
        nests = np.array(nests)
        c5_values = [evaluate_c5(n, grid) for n in nests]
        best_idx = np.argmin(c5_values)
        best_in_grid = nests[best_idx].copy()
        best_c5_in_grid = c5_values[best_idx]

        for it in range(max_iterations):
            beta = 1.5
            sigma = (math.gamma(1 + beta) * math.sin(math.pi * beta / 2) / (math.gamma((1 + beta) / 2) * 2**((beta-1)/2))) ** (1/beta)
            for i in range(pop_size):
                step_size = 0.1 * (1 - it / max_iterations)
                u = np.random.normal(0, sigma, grid)
                v = np.random.normal(0, 1, grid)
                step = u / (abs(v) ** (1 / beta))
                new_nest = nests[i] + step_size * step
                new_nest = repair(new_nest, grid)
                new_c5 = evaluate_c5(new_nest, grid)
                if new_c5 < c5_values[i]:
                    nests[i] = new_nest
                    c5_values[i] = new_c5

            num_abandon = int(pa * pop_size)
            worst_indices = np.argsort(c5_values)[-num_abandon:]
            for idx in worst_indices:
                random_nest = np.random.rand(grid)
                random_nest = repair(random_nest, grid)
                c5 = evaluate_c5(random_nest, grid)
                nests[idx] = random_nest
                c5_values[idx] = c5

            current_best_idx = np.argmin(c5_values)
            if c5_values[current_best_idx] < best_c5_in_grid:
                best_in_grid = nests[current_best_idx].copy()
                best_c5_in_grid = c5_values[current_best_idx]

            if time.time() - start_time > max_time * 0.98:
                break

        if best_c5_in_grid < best_c5:
            best_c5 = best_c5_in_grid
            best_h = best_in_grid
            best_n = grid

    return best_h, best_c5, best_n
