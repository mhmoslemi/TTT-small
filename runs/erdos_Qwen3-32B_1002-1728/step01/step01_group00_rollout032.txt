import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)
    best_c5 = None
    best_h = h_initial.copy()
    start_time = time.time()

    # Generate periodic initial guesses with various periods
    periods = [3,4,5,6,7,8,9,10]
    initial_guesses = [h_initial.copy()]
    for p in periods:
        pattern = np.zeros(n_points)
        for i in range(n_points // p):
            idx = i * p
            if idx >= n_points:
                break
            pattern[idx] = 1.0
        current_sum = np.sum(pattern)
        n_zeros = n_points - current_sum
        if n_zeros == 0:
            x = 0.5
        else:
            x = (target_sum - current_sum) / n_zeros
        pattern = np.where(pattern == 1, 1.0, x)
        initial_guesses.append(pattern)

    # Optimize each initial guess
    for guess in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        x0 = guess.copy()
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 150, 'ftol': 1e-6}
        )
        curr_h = result.x.copy()
        # Ensure solution is feasible
        curr_h = np.clip(curr_h, 0, 1)
        actual_sum = np.sum(curr_h)
        if not np.isclose(actual_sum, target_sum, atol=1e-8):
            # Adjust to meet the sum constraint
            offset = actual_sum - target_sum
            # Distribute the offset to first element
            curr_h[0] -= offset
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
        # Perturb for next iteration
        x0 = best_h.copy()
        x0 += np.random.normal(0, 0.01, n_points)

        x0[0] -= np.sum(x0) - target_sum

        if time.time() - start_time > budget_s - 5:
            break
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        actual_sum = np.sum(curr_h)
        if not np.isclose(actual_sum, target_sum, atol=1e-8):
            offset = actual_sum - target_sum
            curr_h[0] -= offset
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if curr_c5 < best_c5:
