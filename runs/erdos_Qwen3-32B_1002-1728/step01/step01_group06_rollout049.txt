            child = repair(child)
            child_c5 = compute_c5(child)
            offspring.append( (child, child_c5) )
        
        # Find best offspring
        offspring.sort(key=lambda x: x[1])
        best_offspring, best_offspring_c5 = offspring[0]
        
        # Update if improvement
        if best_offspring_c5 < best_c5:
            h = best_offspring.copy()
            best_c5 = best_offspring_c5
            best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1
        
        # Adapt mutation strength
        if no_improvement_counter >= 50:
            sigma *= 0.5
            no_improvement_counter = 0
            # Avoid sigma getting too small
            if sigma < 1e-5:
                sigma = 1e-5

    return best_h, best_c5, grid_size
