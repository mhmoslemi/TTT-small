import numpy as np
import time
from scipy.optimize import minimize, Bounds
from functools import partial

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def generate_sinusoidal(p, n_points):
    x = np.arange(n_points)
    period = n_points / p
    angle = 2 * np.pi * x / period
    h = 0.5 * (1 + np.sin(angle))
    return h

def generate_triangular(p, n_points):
    x = np.arange(n_points)
    period = n_points / p
    frac = (x / period) % 2
    h = np.where(frac < 1, frac, 2 - frac)
    return h

def generate_roots_of_unity(p, n_points):
    angles = 2 * np.pi * np.arange(p) / p
    real_parts = np.cos(angles)
    h = (real_parts + 1) / 2
    repeats = n_points // p
    remainder = n_points % p
    h_pattern = np.tile(h, repeats)
    if remainder > 0:
        h_pattern = np.append(h_pattern, h[:remainder])
    return h_pattern

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    bounds = Bounds(lb=0.0, ub=1.0)
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def project(h):
        h_proj = np.clip(h, 0, 1)
        h_proj_sum = np.sum(h_proj)
        if h_proj_sum == 0:
            return np.zeros_like(h)
        h_proj_scaled = h_proj * (target_sum / h_proj_sum)
        h_proj_scaled = np.clip(h_proj_scaled, 0, 1)
        return h_proj_scaled

    patterns = [
        generate_sinusoidal(p, n_points) for p in [3, 4, 5, 6, 7]
    ] + [
        generate_triangular(p, n_points) for p in [3, 4, 5]
    ] + [
        generate_roots_of_unity(p, n_points) for p in [8, 10, 16]
    ] + [h0]

    candidates = [project(p) for p in patterns]

    for candidate in candidates:
        if time.time() - start_time > budget_s - 10:
            break
        current_c5 = compute_c5(candidate, dx)
        if current_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = current_c5

        result = minimize(lambda x: compute_c5(x, dx), x0=candidate, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-6})
        if result.success:
            new_c5 = compute_c5(result.x, dx)
            if new_c5 < best_c5:
                best_h = result.x.copy()
                best_c5 = new_c5

    return (best_h, best_c5, n_points)
