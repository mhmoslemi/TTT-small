            grad[valid - abs_shift] -= h_best[valid]
        
        # Update step
        h_proposal = np.clip(h_best - alpha * grad, 0, 1)
        
        # Normalize
        total = np.sum(h_proposal)
        scale = (n_points / 2.0) / total if total > 1e-9 else n_points / 2.0
        h_proposal = np.clip(h_proposal * scale, 0, 1)
        
        # Ensure exact normalization
        total = np.sum(h_proposal)
        if abs(total - n_points/2.0) > 1e-6:
            h_proposal = h_proposal / total * (n_points/2.0)
            h_proposal = np.clip(h_proposal, 0, 1)
        
        new_c5 = compute_c5(h_proposal)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_proposal.copy()
            no_improve = 0
            alpha = min(alpha * 1.3, 0.1)
        else:
            no_improve += 1
            alpha = max(alpha * 0.9, 1e-4)
            h_best = h_proposal.copy()
        
        if no_improve > patience:
            # Perturb and reset
            perturbation = np.random.normal(0, 0.02, n_points)
            h_best = np.clip(best_h + perturbation, 0, 1)
            total = np.sum(h_best)
            h_best = np.clip(h_best * (n_points/2.0) / total, 0, 1)
            current_c5 = compute_c5(h_best)
            no_improve = 0
            alpha = 0.01
        
        if iter_count % perturb_freq == 0 and iter_count > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h_best = np.clip(h_best + perturbation, 0, 1)
            total = np.sum(h_best)
            h_best = np.clip(h_best * (n_points/2.0) / total, 0, 1)
    
    return best_h, best_c5, n_points
