        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    try:
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        correlations = []
        for k in range(1, n_points):  # Positive shifts only
            valid_i_count = n_points - k
            if valid_i_count <= 0:
                continue
            h_part = h_var[0:valid_i_count]
            h_shifted = h_var[k:k+valid_i_count]
            corr = cp.sum(h_part * (1 - h_shifted)) * dx
            correlations.append(corr)
        t_var = cp.Variable()
        objective = cp.Minimize(t_var)
        constraints.extend([corr <= t_var for corr in correlations])
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False, max_iter=500)
        candidate = np.clip(h_var.value, 0, 1)
        candidate_sum = candidate.sum()
        if not np.isclose(candidate_sum, target_sum):
            candidate *= target_sum / candidate_sum
            candidate = np.clip(candidate, 0, 1)
        current_c5 = compute_c5(candidate, dx)
        if current_c5 < best_c5:
            best_h, best_c5 = candidate, current_c5
    except:
        pass

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    if not np.isclose(best_h_sum, target_sum):
        best_h *= target_sum / best_h_sum
        best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5, n_points
