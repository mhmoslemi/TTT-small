    while time.time() - start_time < 0.9 * budget_s:
        # Generate perturbed initial guess with varying seed
        perturbed = optimized_h + np.random.RandomState(seed + perturbation_count).normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum > 1e-6:
            perturbed = (perturbed / current_sum) * target_sum
        else:
            perturbed = np.zeros_like(perturbed)
        perturbed = perturbed.reshape(n_points)
        
        # Run faster optimization with smaller maxiter
        secondary_result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        
        h_secondary = secondary_result.x.reshape(n_points)
        corr_secondary = np.correlate(h_secondary, 1 - h_secondary, mode='full')
        current_c5 = np.max(corr_secondary) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            optimized_h = h_secondary
        
        perturbation_count += 1
    
    # Final checks
    assert abs(np.sum(optimized_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((optimized_h >= -1e-6) & (optimized_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (optimized_h.tolist(), best_c5, n_points)
