    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - desired_sum}]

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.zeros(n_points).astype(np.float64)
        h_initial[:] = 0.5  # Simple initialization

    candidates = [h_initial.copy()]
    for _ in range(4):
        perturbed = np.random.normal(0.0, 0.1, n_points)
        new_h = h_initial.copy() + perturbed
        new_h = np.clip(new_h, 0.0, 1.0)
        current_sum = np.sum(new_h)
        if abs(current_sum - desired_sum) > 1e-8:
            new_h = np.clip(new_h * (desired_sum / current_sum), 0.0, 1.0)
        candidates.append(new_h)

    all_candidates_coarse = []
    beta_sequence = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    for candidate in candidates:
        current_h = candidate.copy()
        for beta in beta_sequence:
            if time.time() - start_time > budget_s - 20:
                break
            def objective_func(h_vec, beta=beta):
                h = h_vec.reshape(n_points)
                ones_minus_h = 1.0 - h
                cross = np.correlate(h, ones_minus_h, mode='full')
                cross *= dx
                exp_sum = np.sum(np.exp(beta * cross))
                return (1.0 / beta) * np.log(exp_sum)
            result = minimize(objective_func, current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                optimized_h = result.x.copy()
                optimized_h = np.clip(optimized_h, 0.0, 1.0)
                current_sum = np.sum(optimized_h)
                if abs(current_sum - desired_sum) > 1e-8:
                    optimized_h = np.clip(optimized_h * (desired_sum / current_sum), 0.0, 1.0)
                c5_current = evaluate_C5(optimized_h, n_points, dx)
                all_candidates_coarse.append((optimized_h, c5_current))
                current_h = optimized_h.copy()
        if time.time() - start_time > budget_s - 20:
            break

    if all_candidates_coarse:
        top_coarse = sorted(all_candidates_coarse, key=lambda x: x[1])[:2]
    else:
        top_coarse = [(h_initial, evaluate_C5(h_initial, n_points, dx))]

    for h_candidate, _ in top_coarse:
        for _ in range(3):
            if time.time() - start_time > budget_s - 5:
                break
            def objective_func_fine(h_vec):
                h = h_vec.reshape(n_points)
                ones_minus_h = 1 - h
                cross = np.correlate(h, ones_minus_h, mode='full')
                return np.max(cross * dx)
            result = minimize(objective_func_fine, h_candidate, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                optimized_h = result.x.copy()
                optimized_h = np.clip(optimized_h, 0.0, 1.0)
                current_sum = np.sum(optimized_h)
                if abs(current_sum - desired_sum) > 1e-8:
                    optimized_h = np.clip(optimized_h * (desired_sum / current_sum), 0.0, 1.0)
                c5_current = evaluate_C5(optimized_h, n_points, dx)
                all_candidates_coarse.append((optimized_h, c5_current))

    best_h = h_initial.copy()
    best_c5 = evaluate_C5(best_h, n_points, dx)
    if all_candidates_coarse:
        best_info = min(all_candidates_coarse, key=lambda x: x[1])
        best_h, best_c5 = best_info

    return best_h, best_c5, n_points
