import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, noise=0.2):
        h_pert = h.copy()
        h_pert += np.random.uniform(-noise, noise, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert *= target_sum / current_sum
        h_pert = np.clip(h_pert, 0, 1)
        return h_pert
    
    tries = 0
    max_tries = 40
    noise_level = 0.2
    
    while time.time() - start_time < budget_s - 60 and tries < max_tries:
        tries += 1
        x0 = get_perturbed(best_h, noise=noise_level)
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-6}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val
                # Improve: reduce noise
                noise_level = max(0.05, noise_level - 0.05)
            else:
                # No improvement, increase noise
                noise_level = min(0.5, noise_level + 0.05)
        else:
            # Optimization failed, increase noise
            noise_level = min(0.5, noise_level + 0.05)
    
    return (best_h, best_c5, n_points)
