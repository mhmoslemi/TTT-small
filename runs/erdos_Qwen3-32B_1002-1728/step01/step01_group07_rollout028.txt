import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if s == 0:
            return h
        return h * (target_sum / s)

    np.random.seed(int(seed) % (2**32))
    pop_size = 50
    pop = []
    for _ in range(5):
        pop.append(h_init.copy())
    for _ in range(45):
        perturbed = h_init.copy() + np.random.normal(0, 0.05, n_points)
        pop.append(repair(perturbed))

    best_h = np.array(h_init)
    best_c5 = np.inf

    while time.time() - start_time < budget_s:
        fitness = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        sorted_idx = np.argsort(fitness)
        top_half = [pop[i] for i in sorted_idx[:25]]

        offspring = []
        for _ in range(20):
            p1, p2 = np.random.choice(top_half, 2)
            alpha = 0.5
            child = alpha * p1 + (1 - alpha) * p2
            child = repair(child)
            offspring.append(child)
        for _ in range(5):
            parent = np.random.choice(top_half)
            mutated = parent.copy() + np.random.normal(0, 0.02, n_points)
            mutated = repair(mutated)
            offspring.append(mutated)

        pop = top_half + offspring
        if len(pop) > pop_size:
            pop = [pop[i] for i in np.random.permutation(len(pop))[:pop_size]]

    return (best_h.tolist(), best_c5, n_points)
