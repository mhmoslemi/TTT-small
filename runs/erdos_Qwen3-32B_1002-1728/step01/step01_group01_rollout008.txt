def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * dx

    while time.time() - start_time < budget_s:
        # Formulate convex problem with current h_prev
        x = cp.Variable(n_points)
        C = cp.Variable()
        
        sum_constraint = cp.sum(x) == n_points / 2.0
        bounds_constraints = [0 <= x, x <= 1]
        constraints = [sum_constraint] + bounds_constraints
        
        for k in range(n_points):  # Process all k (0 <= k < n_points)
            overlap_len = n_points - k
            if overlap_len <= 0:
                continue

            indices_i_pos = np.arange(overlap_len)
            indices_j_pos = indices_i_pos + k
            linearized_sum_pos = 0
            for i, j in zip(indices_i_pos, indices_j_pos):
                x_i = h_prev[i]
                x_j = h_prev[j]
                linearized_sum_pos += x[i] * (1 - x_j) - x_i * x[j]
            constraint_pos = linearized_sum_pos * dx <= C
            constraints.append(constraint_pos)
