import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    h_current = result.x if result.success else h_init
    current_c5 = objective(h_current)

    best_h = h_current.copy()
    best_c5 = current_c5

    # Perturbation loop with seed-based diversity
    def generate_perturbation(h):
        rng = np.random.RandomState(seed + n_points + int(time.time()))
        noise = rng.normal(0, 0.05, n_points)
        h_perturbed = h.copy() + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = h_perturbed.sum()
        if sum_perturbed < 1e-9:
            return h.copy()
        h_perturbed = h_perturbed * (target_sum / sum_perturbed)
        return h_perturbed

    while time.time() - start_time < budget_s * 0.99:
        h_perturbed = generate_perturbation(best_h)
        result_perturb = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result_perturb.success:
            h_new = result_perturb.x
            new_c5 = objective(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5

    # Final validation
    h = best_h.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx

    assert abs(np.sum(h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((h >= -1e-6) & (h <= 1.0 + 1e-6)), "Value bounds violated"

    return (h.tolist(), final_c5, n_points)
