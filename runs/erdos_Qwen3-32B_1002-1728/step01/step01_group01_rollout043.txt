def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np
    import cvxpy as cp

    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_best = initial_h_values.copy()
    target_sum = n_points / 2
    h_curr = h_best.copy()
    best_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx

    # Precompute valid k and pairs
    valid_k_list = []
    valid_pairs_list = []
    for k_candidate in range(-n_points + 1, n_points):
        pairs = []
        for i in range(n_points):
            j = i + k_candidate
            if 0 <= j < n_points:
                pairs.append( (i, j) )
        if len(pairs) > 0:
            valid_k_list.append(k_candidate)
            valid_pairs_list.append(pairs)

    # SCP iterations
    while time.time() - start_time < budget_s - 10:
        # Formulate convex problem
        h = cp.Variable(n_points)
        C_total = cp.Variable()
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]

        # Add constraints for all k
        for k_idx in range(len(valid_pairs_list)):
            pairs = valid_pairs_list[k_idx]
            # Compute sum of constants and linear parts for this k's pairs
            constants_sum = 0.0
            linear_expr = 0.0
            for i, j in pairs:
                constants_sum += h_curr[i] * (1 - h_curr[j])
                linear_expr += (1 - h_curr[j]) * h[i] - h_curr[i] * h[j]
            # Add constraint
            constraints.append( linear_expr + constants_sum <= C_total / dx )

        # Objective is to minimize C_total
        prob = cp.Problem(cp.Minimize(C_total), constraints)
        # Solve with cvxpy and a solver (try CVXOPT first)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)  # can also try GLOP, but CVXOPT is more for convex
        except:
            # fallback to another solver if needed
            prob.solve(solver=cp.SCS, verbose=False)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h.value
            # Clip and project
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            if sum_h != 0:
                h_new = h_new * (target_sum / sum_h)
            else:
                # Fallback if sum is zero
                h_new = np.zeros_like(h_new)
            # Compute current c5
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                h_best = h_new
                best_c5 = current_c5
            h_curr = h_new
        else:
            # Problem not solved, break or skip?
            break

        # Check time
        if time.time() - start_time > budget_s - 10:
            break

import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h_curr = initial_h_values.copy()
    h_best = h_curr.copy()
    
    # Compute initial C5
    corr = np.correlate(h_best, 1 - h_best, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Precompute valid shifts and pairs
    valid_pairs_list = []
    for k in range(-n_points + 1, n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append((i, j))
        if len(pairs) > 0:
            valid_pairs_list.append(pairs)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 10:
        h = cp.Variable(n_points)
        C_total = cp.Variable()
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        for pairs in valid_pairs_list:
            linear_expr = 0.0
            const_sum = 0.0
            for i, j in pairs:
                const_sum += h_curr[i] * (1 - h_curr[j])
                linear_expr += (1 - h_curr[j]) * h[i] - h_curr[i] * h[j]
            constraints.append(linear_expr + const_sum <= C_total / dx)
        
        prob = cp.Problem(cp.Minimize(C_total), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
        except:
            try:
                prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
            except:
                try:
                    prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
                except:
                    pass
        
        if prob.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
            h_new = np.clip(h.value, 0, 1)
            total = np.sum(h_new)
            if total > 1e-8:
                h_new *= target_sum / total
            
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                h_best = h_new
                best_c5 = current_c5
            h_curr = h_new
        else:
            break  # Exit early if solve fails
        
        iter_count += 1
        if iter_count > 50 or time.time() - start_time > budget_s - 10:
            break
    
    return h_best, best_c5, n_points
