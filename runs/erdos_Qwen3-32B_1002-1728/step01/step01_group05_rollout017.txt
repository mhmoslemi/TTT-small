        h_perturbed = np.clip(h_perturbed, 0, 1)
        result = minimize(
            lambda x: compute_c5(x), 
            x0=h_perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': max_iter_per, 'ftol': 1e-6}
        )
        if result.success:
            new_h = result.x
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    return (best_h, best_c5, n_points)
