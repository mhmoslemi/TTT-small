    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = np.zeros((n_particles, n_points))
    personal_best_fitness = np.zeros(n_particles)
    global_best = np.zeros(n_points)
    global_best_fitness = float('inf')

    # Initialize half of the particles around initial_h_values
    half = n_particles // 2
    for i in range(half):
        perturbation = np.random.normal(0, 0.1, n_points)
        particles[i] = initial_h_values + perturbation
    for i in range(half, n_particles):
        particles[i] = np.random.uniform(0, 1, n_points)

    # Project and evaluate initial particles
    for i in range(n_particles):
        particles[i] = project_h(particles[i], target_sum)
        personal_best[i] = particles[i].copy()
        current_fitness = objective(particles[i], dx)
        personal_best_fitness[i] = current_fitness
        if current_fitness < global_best_fitness:
            global_best = personal_best[i].copy()
            global_best_fitness = current_fitness

    # Main PSO loop
    while time.time() - start_time < budget_s - 10:
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            # Update position
            particles[i] += velocities[i]
            # Project to feasible region
            particles[i] = project_h(particles[i], target_sum)
            # Evaluate fitness
            current_fitness = objective(particles[i], dx)
            # Update personal best
            if current_fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = current_fitness
                if current_fitness < global_best_fitness:
                    global_best = personal_best[i].copy()
                    global_best_fitness = current_fitness
        # Inertia weight decay
        w = max(0.4, w - 0.01)
        if time.time() - start_time >= budget_s - 10:
            break

    return global_best, global_best_fitness, n_points

def objective(h_vec, dx):
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return h_clipped.copy()
    delta = target_sum - current_sum
    if delta > 0:
        # Increase h to reach target_sum
        indices = h_clipped < 1.0
        if np.any(indices):
            distrib = indices.astype(float)
            h_clipped += delta * distrib / distrib.sum()
    else:
        # Decrease h to reach target_sum
        indices = h_clipped > 0.0
        if np.any(indices):
            distrib = indices.astype(float)
            h_clipped += delta * distrib / distrib.sum()
    return np.clip(h_clipped, 0, 1)
