import numpy as np
import time
import copy

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    pop_size = 50
    num_elites = 5
    mutation_sigma = 0.02
    tournament_k = 5
    time_budget = budget_s

    def perturb_and_feasibilize(h_vec, sigma):
        h = h_vec.copy()
        h += np.random.normal(0, sigma, h.shape)
        h = np.clip(h, 0, 1)
        delta = (n_points / 2) - h.sum()
        h[0] += delta
        h[0] = np.clip(h[0], 0, 1)
        return h

    def crossover(p1, p2):
        alpha = 0.5
        c1 = alpha * p1 + (1 - alpha) * p2
        c2 = (1 - alpha) * p1 + alpha * p2
        return c1, c2

    def mutate(child, sigma):
        ch = child.copy()
        num_mutate = max(1, int(0.1 * len(ch)))
        idxs = np.random.choice(len(ch), num_mutate, replace=False)
        ch[idxs] += np.random.normal(0, sigma, num_mutate)
        ch = np.clip(ch, 0, 1)
        delta = (n_points / 2) - ch.sum()
        ch[0] += delta
        ch[0] = np.clip(ch[0], 0, 1)
        return ch

    def evaluate(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full') * dx)

    # Initialize population
    population = [perturb_and_feasibilize(h_initial, mutation_sigma) for _ in range(pop_size)]
    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        # Tournament selection
        selected = []
        for _ in range(pop_size // 2):
            contestants = np.random.choice(population, tournament_k, replace=False)
            fit_contestants = [evaluate(c) for c in contestants]
            selected.append(contestants[np.argmin(fit_contestants)])

        # Crossover and mutation
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                p1, p2 = selected[i], selected[i + 1]
                c1, c2 = crossover(p1, p2)
                offspring.append(mutate(c1, mutation_sigma))
                offspring.append(mutate(c2, mutation_sigma))
            else:
                offspring.append(selected[i].copy())

        # Elitism and new population
        elite_indices = np.argsort(fitness)[:num_elites]
        elites = [population[i] for i in elite_indices]
        new_population = elites + offspring
        if len(new_population) < pop_size:
            new_population += [perturb_and_feasibilize(h_initial, mutation_sigma) for _ in range(pop_size - len(new_population))]
        elif len(new_population) > pop_size:
            new_population = new_population[:pop_size]
        population = new_population

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr_values), n_points
