import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1

    particles = []
    velocities = []
    for i in range(n_particles):
        if i == 0:
            p = h_initial.copy()
        else:
            if i % 3 == 0:
                p = np.random.uniform(0, 1, n_points)
                s = p.sum()
                if s == 0:
                    p = h_initial.copy()
                else:
                    p = p * (target_sum / s)
                    p = np.clip(p, 0, 1)
            else:
                p = h_initial.copy() + np.random.normal(0, 0.05, n_points)
                p = np.clip(p, 0, 1)
                s = p.sum()
                if s == 0:
                    p = h_initial.copy()
                else:
                    p = p * (target_sum / s)
                    p = np.clip(p, 0, 1)
        particles.append(p)
        velocities.append(np.zeros_like(p))

    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [compute_c5(p) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    gb_pos = personal_best_positions[global_best_idx].copy()
    gb_c5 = personal_best_scores[global_best_idx]

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        for i in range(n_particles):
            curr_pos = particles[i]
            curr_vel = velocities[i]
            pb_pos = personal_best_positions[i]
            r1, r2 = np.random.rand(2)
            cog = c1 * r1 * (pb_pos - curr_pos)
            soc = c2 * r2 * (gb_pos - curr_pos)
            new_vel = np.clip(
                w * curr_vel + cog + soc,
                -max_velocity,
                max_velocity
            )
            new_pos = curr_pos + new_vel
            new_clipped = np.clip(new_pos, 0, 1)
            s = new_clipped.sum()
            if s != 0:
                scale = target_sum / s
                new_scaled = new_clipped * scale
                new_pos_final = np.clip(new_scaled, 0, 1)
            else:
                new_pos_final = new_clipped.copy()
            # Update state
            particles[i] = new_pos_final
            velocities[i] = new_vel
            # Evaluate and update
            new_c5 = compute_c5(new_pos_final)
            if new_c5 < personal_best_scores[i]:
                personal_best_positions[i] = new_pos_final.copy()
                personal_best_scores[i] = new_c5
                if new_c5 < gb_c5:
                    gb_pos[:] = new_pos_final
                    gb_c5 = new_c5
    return gb_pos, gb_c5, n_points
