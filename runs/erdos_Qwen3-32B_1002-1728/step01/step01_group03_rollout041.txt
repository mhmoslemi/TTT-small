import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h *= target_sum / current_sum
        return h
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    POP_SIZE = 20
    NOISE_SCALE = 0.15
    MAX_ITER = budget_s // 20  # Conservative iteration estimate
    
    # Initialize population with diverse samples
    population = [repair(h_init + np.random.normal(0, 0.05, n_points)) for _ in range(POP_SIZE)]
    fitness = [objective(ind) for ind in population]
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        new_population = []
        
        # Select top performers for mutation
        top_indices = np.argsort(fitness)[:POP_SIZE//2]
        top_individuals = [population[i] for i in top_indices]
        
        # Generate new candidates through mutation
        for ind in top_individuals:
            mutated = ind + np.random.normal(0, NOISE_SCALE, n_points)
            mutated = repair(mutated)
            new_population.append(mutated)
        
        # Evaluate new candidates
        new_fitness = [objective(ind) for ind in new_population]
        
        # Combine populations
        combined = population + new_population
        combined_scores = fitness + new_fitness
        
        # Select best POP_SIZE candidates
        best_indices = np.argsort(combined_scores)[:POP_SIZE]
        population = [combined[i] for i in best_indices]
        fitness = [combined_scores[i] for i in best_indices]
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Adaptive time management
        if iter_count % 5 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
    
    return (best_h, best_c5, n_points)
