        new_h = np.clip(new_h, 0, 1)
        s = new_h.sum()
        if s == 0:
            continue
        new_h *= (target_sum) / s
        new_c5 = objective(new_h)

        # Acceptance probability
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                T *= 0.9  # Faster cooling if improvement
        else:
            T *= alpha  # Normal cooling

        T *= alpha  # Base cooling

    # SLSQP refinement
    if time.time() - start_time < budget_s * 0.98:
        bounds = Bounds(0, 1)
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_eq}, options={'maxiter': 250})
        if result.success:
            optimized_h = np.clip(result.x, 0, 1)
            s = optimized_h.sum()
            if s != 0:
                optimized_h *= (target_sum) / s
            final_c5 = objective(optimized_h)
            if final_c5 < best_c5:
                best_h = optimized_h
                best_c5 = final_c5

    # Final check for constraints
    best_h = np.clip(best_h, 0, 1)
    s = best_h.sum()
    if s != 0:
        best_h *= (target_sum) / s

    return best_h, best_c5, n_points
