import numpy as np
import time

def objective(h_vec):
    n_points = len(h_vec)
    dx = 2.0 / n_points
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    target_h = initial_h_values.copy()
    
    # Initialize population
    pop_size = 50
    population = []
    
    # 50% perturbed initial
    for _ in range(pop_size // 2):
        perturb = np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(target_h + perturb, 0, 1)
        h_pert = h_pert / h_pert.sum() * target_sum
        population.append(h_pert)
    
    # 50% random
    for _ in range(pop_size // 2):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / h_rand.sum() * target_sum
        population.append(h_rand)
    
    population = np.array(population)
    
    # Differential Evolution parameters
    F = 0.8
    CR = 0.9
    best_idx = objective(population).argmin()
    best_solution = population[best_idx].copy()
    best_c5 = objective(best_solution)
    
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            idxs = np.setdiff1d(np.arange(pop_size), i)
            a, b, c = population[np.random.choice(idxs, 3, replace=False)]
            mutant = a + F * (b - c)
            
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            
            # Projection: [0,1] clip + normalize
            clipped = np.clip(trial, 0, 1)
            trial_sum = clipped.sum()
            if trial_sum == 0:
                scaled = np.zeros_like(clipped)
            else:
                scaled = clipped * (target_sum / trial_sum)
            final = np.clip(scaled, 0, 1)
            
            # Final normalization step
            final_sum = final.sum()
            if final_sum > 1e-8:
                final = final / final_sum * target_sum
            
            current_val = objective(population[i])
            trial_val = objective(final)
            
            if trial_val < current_val:
                population[i] = final
        
        # Update best solution
        current_all_vals = [objective(ind) for ind in population]
        current_best_idx = np.argmin(current_all_vals)
        if current_all_vals[current_best_idx] < best_c5:
            best_solution = population[current_best_idx].copy()
            best_c5 = current_all_vals[current_best_idx]
            
            # Perturb best solution to escape local minima
            perturb = np.random.normal(0, 0.05, n_points)
            best_solution = np.clip(best_solution + perturb, 0, 1)
            best_solution = best_solution / best_solution.sum() * target_sum
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
            
    return best_solution, best_c5, n_points
