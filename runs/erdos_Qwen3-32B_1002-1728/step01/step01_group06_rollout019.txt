def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Determine n_points
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        old_n = len(h_initial)
        # Resample to n_points
        old_x = np.linspace(0, 2, old_n)
        new_x = np.linspace(0, 2, n_points)
        h = np.interp(new_x, old_x, h_initial)
    else:
        # Default initialization: central pulse
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-10:mid+10] = 0.5  # choose a range to spread the initial h
        h *= target_sum / h.sum()

    h = np.clip(h, 0, 1)
    h = repair(h, target_sum)

    # PSO parameters
    num_particles = 50
    num_dimensions = n_points
    cognitive = 1.5  # cognitive coefficient (phi_p)
    social = 1.5  # social coefficient (phi_g)
    inertia_min = 0.4
    inertia_max = 0.9
    vel_max = 0.5  # maximum velocity change
    max_iter = 200  # max number of iterations
    early_stop_patience = 50  # stop if no improvement in this many iterations
    early_stop_improvement_threshold = 1e-4

    velocities = np.zeros_like(particles)
    personal_best_positions = np.zeros_like(particles)
    personal_best_scores = np.full(num_particles, float('inf'))
    global_best_position = h.copy()
    global_best_score = compute_c5(h, n_points)

            global_best_score = score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    n_points = 280

    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-10:mid+10] = 0.5
        h *= target_sum / h.sum()

    h = np.clip(h, 0, 1)
    h = repair(h, target_sum)

    # PSO parameters
    num_particles = 50
    cognitive = 1.5
    social = 1.5
    inertia_min = 0.4
    inertia_max = 0.9
    vel_max = 0.5
    max_iter = 200
    patience = 30

    # Particle initialization
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros_like(particles)
    personal_best = np.repeat(h[np.newaxis], num_particles, axis=0)
    pbest_scores = np.full(num_particles, np.inf)
    gbest = h.copy()
    gbest_score = compute_c5(h, n_points)
    no_improvement = 0

    # Initialize particles with perturbations
    for i in range(num_particles):
        perturbation = 0.2 * np.random.randn(n_points)
        particles[i] = h + perturbation
        particles[i] = repair(particles[i], target_sum)
        current_score = compute_c5(particles[i], n_points)
        if current_score < pbest_scores[i]:
            personal_best[i] = particles[i].copy()
            pbest_scores[i] = current_score
        if current_score < gbest_score:
            gbest[:] = particles[i].copy()
            gbest_score = current_score

    # PSO main loop
    for iter in range(max_iter):
        inertia = inertia_max - (inertia_max - inertia_min) * iter / max_iter

        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best[i] - particles[i]) +
                            social * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            particles[i] += velocities[i]
            particles[i] = repair(particles[i], target_sum)

            # Update personal best

                pbest_scores[i] = current_score

            # Update global best

            else:
