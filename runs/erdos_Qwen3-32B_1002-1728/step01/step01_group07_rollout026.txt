        perturbed = (perturbed / np.sum(perturbed)) * target_sum
        return perturbed
    
    np.random.seed(seed)
    max_tries = 50
    num_tries = 0
    
    while time.time() - start_time < budget_s * 0.95:
        h0 = perturb_h(h_current, int((time.time() % 1) * 1e6))
        result = minimize(
            fun=objective,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_opt = result.x
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                h_current = h_opt.copy()
                best_c5 = c5_opt
        num_tries += 1
        if num_tries >= max_tries:
            break
    
    assert abs(np.sum(h_current) - target_sum) < 1e-4, "Integral constraint violated"
    assert np.all((h_current >= -1e-6) & (h_current <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_current.tolist(), best_c5, n_points)
