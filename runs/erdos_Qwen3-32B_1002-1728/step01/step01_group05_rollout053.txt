    h_clipped = np.clip(h_scaled, 0, 1)
    new_sum = np.sum(h_clipped)
    if abs(new_sum) < 1e-9:
        return np.zeros_like(h_clipped)
    h_final = h_clipped * target_sum / new_sum
    h_final = np.clip(h_final, 0, 1)
    return h_final

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    # Perturb initial_h_values
    perturb_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
    perturbation = np.random.uniform(-0.1, 0.1, len(perturb_indices))
    h_current = h0.copy()
    h_current[perturb_indices] += perturbation
    h_current = np.clip(h_current, 0, 1)
    h_current = project_to_feasible(h_current, n_points)

    # Define bounds for global search
    bounds = [(0, 1) for _ in range(n_points)]

    # Objective for global search
    def objective_global(h_flat):
        h_projected = project_to_feasible(h_flat, n_points)
        return compute_c5(h_projected, dx)

    # Run differential evolution for global search
    result_global = differential_evolution(
        func=objective_global,
        bounds=bounds,
        seed=seed,
        maxiter=100,
        popsize=15,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-4,
        updating='immediate',
        workers=1
    )

    h_global = result_global.x if result_global.success else h_current
    h_global_projected = project_to_feasible(h_global, n_points)

    current_time = time.time()
    remaining_budget = budget_s - (current_time - start_time)

    # Local refinement using SLSQP
    def objective_local(x):
        return compute_c5(x, dx)

    constraints_local = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_local = Bounds(0, 1)

    # Try perturbations and refine
    h_best = h_global_projected.copy()
    best_c5 = compute_c5(h_best, dx)
    start_local = time.time()

    perturb_amp = 0.05
    trials = 0
    while time.time() - start_local < remaining_budget and trials < 5:
        h_perturbed = h_best + np.random.uniform(-perturb_amp, perturb_amp, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = project_to_feasible(h_perturbed, n_points)

        result = minimize(
            fun=objective_local,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds_local,
            constraints=constraints_local,
            options={'maxiter': 50, 'ftol': 1e-6}
        )

        if result.success:
            h_candidate = result.x
            h_candidate = project_to_feasible(h_candidate, n_points)
            c5_candidate = compute_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                h_best = h_candidate.copy()
                best_c5 = c5_candidate

        trials += 1

    return (h_best, best_c5, n_points)
