import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, required_sum):
    h_clamped = np.clip(h, 0, 1)
    sum_clamped = np.sum(h_clamped)
    delta = required_sum - sum_clamped
    if delta != 0:
        h_clamped[0] += delta
        h_clamped = np.clip(h_clamped, 0, 1)
    return h_clamped

    for i in range(1, n_particles):
        particles[i] = h_initial.copy() + np.random.normal(0, 0.05, n_points)
    for i in range(n_particles):

    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
