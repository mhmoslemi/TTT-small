import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    n_points = len(initial_h_values)
    h_current = initial_h_values.copy()
    best_h = h_current.copy()
    dx = 2.0 / n_points
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Precompute shifts and their valid (i, j) indices
    shifts_info = []
    for k in range(-n_points + 1, n_points):
        valid = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid.append((i, j))
        if valid:
            shifts_info.append((k, valid))
    
    while time.time() - start_time < budget_s - 5:
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        for k, indices in shifts_info:
            sum_current_val = 0.0
            for i, j in indices:
                sum_current_val += h_current[i] * (1 - h_current[j])
            
            linear_expr = 0.0
            for i, j in indices:
                linear_expr += (1 - h_current[j]) * h[i] - h_current[i] * h[j]
            
            constraints.append(linear_expr <= C * (n_points / 2) - sum_current_val)
        
        objective = cp.Minimize(C)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.CVXOPT, verbose=False, maxiters=100)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h.value
                new_h = np.clip(new_h, 0, 1)
                new_sum = np.sum(new_h)
                if not np.isclose(new_sum, n_points / 2, atol=1e-6):
                    new_h *= (n_points / 2) / new_sum
                corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                h_current = new_h.copy()
            else:
                print("Subproblem not solved optimally, skipping.")
        except Exception as e:
            print(f"Solver error: {e}, skipping iteration.")
        
        if time.time() - start_time > budget_s - 5:
            break
    
    return best_h, best_c5, n_points
