import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize
import time

def project(h_vec, target_sum):
    h_clipped = np.clip(h_vec, 0.0, 1.0)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return np.zeros_like(h_vec)
    scaling_factor = target_sum / h_sum
    h_scaled = h_clipped * scaling_factor
    h_projected = np.clip(h_scaled, 0.0, 1.0)
    return h_projected

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def de_objective(x, target_sum, dx):
    h_proj = project(x, target_sum)
    return compute_c5(h_proj, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_perturbed = initial_h.copy() + np.random.normal(0, 0.1, size=n_points)
    h_perturbed = project(h_perturbed, target_sum)

    bounds = [(0.0, 1.0)] * n_points
    result_de = differential_evolution(
        func=lambda x: de_objective(x, target_sum, dx),
        bounds=bounds,
        popsize=20,
        strategy='best1bin',
        maxiter=50,
        polish=False,
        recombination=0.9,
        mutation=(0.5, 1.0),
        seed=seed,
        workers=-1,
        tol=1e-4
    )

    best_h = project(result_de.x, target_sum)
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    perturb_amp = 0.1
    decay_rate = 0.95
    trials = 0
    max_trials = 50

    while time.time() - start_time < budget_s and trials < max_trials:
        if trials > 0 and trials % 10 == 0:
            perturbation = np.random.uniform(-perturb_amp, perturb_amp, size=n_points)
            new_h = np.clip(best_h + perturbation, 0.0, 1.0)
            new_h = project(new_h, target_sum)
            x0 = new_h.copy()
        else:
            x0 = best_h.copy()
        result = minimize(
            lambda x: compute_c5(x, dx),
            x0=x0,
            method='SLSQP',
            bounds=Bounds(0.0, 1.0),
            constraints=({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}),
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success:
            candidate_h = result.x
            candidate_h = project(candidate_h, target_sum)
            candidate_c5 = compute_c5(candidate_h, dx)
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
        trials += 1
        perturb_amp *= decay_rate

    return (best_h, best_c5, n_points)
