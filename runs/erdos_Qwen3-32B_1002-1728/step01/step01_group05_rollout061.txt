from scipy.optimize import differential_evolution, Bounds, minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):

    return (best_h, best_c5, n_points)

import numpy as np
import time
from scipy.optimize import differential_evolution, Bounds, minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = Bounds(lb=0.0, ub=1.0)
    
    def repair(x):
        h = np.clip(x.copy(), 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.ones_like(h) * 0.5
        else:
            h = h * (target_sum / sum_h)
            h = np.clip(h, 0.0, 1.0)
        return h
    
    def objective(x):
        h_repaired = repair(x)
        return compute_c5(h_repaired, dx)
    
    de_bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_total = time.time()
    
    # Global Exploration with Differential Evolution
    result_de = differential_evolution(
        objective, de_bounds,
        strategy='rand1bin', maxiter=50, popsize=20,
        recombination=0.7, mutation=(0.5, 1.0),
        polish=False, workers=1
    )
    best_h = repair(result_de.x)
    best_c5 = compute_c5(best_h, dx)
    elapsed_global = time.time() - start_total
    
    # Local Refinement with SLSQP and Perturbations
    def slsqp_obj(x):
        return compute_c5(x, dx)
    
    slsqp_constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Perform multiple perturbed restarts
    best_h_final = best_h.copy()
    best_c5_final = best_c5
    remaining_time = budget_s - elapsed_global
    
    if remaining_time > 30:
        restarts = 0
        perturb_decay = 0.95
        perturb_amp = 0.05
        while remaining_time > 30:
            perturbation = np.random.normal(0.0, perturb_amp, size=n_points)
            perturbed = np.clip(best_h_final + perturbation, 0.0, 1.0)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum == 0:
                perturbed = np.ones_like(perturbed) * 0.5
            else:
                perturbed = perturbed * (target_sum / perturbed_sum)
                perturbed = np.clip(perturbed, 0.0, 1.0)
            
            result_slsqp = minimize(
                slsqp_obj, x0=perturbed, method='SLSQP',
                bounds=bounds, constraints=slsqp_constraints,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if result_slsqp.success:
                candidate_c5 = compute_c5(result_slsqp.x, dx)
                if candidate_c5 < best_c5_final:
                    best_h_final = result_slsqp.x
                    best_c5_final = candidate_c5
            perturb_amp *= perturb_decay
            remaining_time = budget_s - (time.time() - start_total)
            restarts += 1
            if restarts >= 3:
                break
    
    return (best_h_final, best_c5_final, n_points)
