def adjust_sum(h, target_sum):
    h = np.array(h, copy=True)
    current = np.sum(h)
    if abs(current - target_sum) < 1e-8:
        return h
    n = len(h)

        indices = np.argsort(h)
        remaining = delta

                continue

            remaining -= add

        remaining = -delta

                continue

            h[i] -= subtract
            remaining -= subtract
