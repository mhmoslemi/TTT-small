import numpy as np
import time
from scipy.optimize import minimize
from functools import partial

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    cross = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(cross)

def augmented_objective(x, beta, lam, rho, n_points, dx, target_sum):
    h = x.reshape(n_points)
    cross = np.correlate(h, 1 - h, mode='full') * dx
    exp_vals = np.exp(beta * cross)
    sum_exp = np.sum(exp_vals)
    sum_exp = np.clip(sum_exp, 1e-8, None)
    log_sum_exp = np.log(sum_exp) / beta
    constraint = np.sum(h) - target_sum
    return log_sum_exp + lam * constraint + 0.5 * rho * (constraint ** 2)

def repair_bounds(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-8:
        h *= target_sum / np.clip(current_sum, 1e-8, None)
        h = np.clip(h, 0.0, 1.0)
    return h

def run():
    start_time = time.time()
    budget_s = 1100
    
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from provided initial_h_values
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy().astype(np.float64)
    else:
        h_init = np.zeros(n_points, dtype=np.float64)
        h_init[:] = 0.5
    
    h_init = repair_bounds(h_init, target_sum)
    best_c5 = None
    best_h = h_init.copy()
    
    n_restarts = 3
    beta_values = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 10:
            break
        
        h = h_init.copy()
        perturbation = np.random.normal(0.0, 0.05, n_points)
        h += perturbation
        h = repair_bounds(h, target_sum)
        
        lam = 0.0
        rho = 1e-4
        h_current = h.copy()
        
        for beta in beta_values:
            if time.time() - start_time > budget_s - 10:
                break

                               n_points=n_points, dx=dx, target_sum=target_sum)

                              bounds=[(0.0, 1.0)] * n_points,

            if result.success:
                h_current = repair_bounds(result.x.copy(), target_sum)
                constraint = np.sum(h_current) - target_sum
                lam += rho * constraint
                rho = min(rho * 2, 1e4)

                if best_c5 is None or c5 < best_c5:

                    best_h = h_current.copy()
    return best_h, best_c5, n_points
