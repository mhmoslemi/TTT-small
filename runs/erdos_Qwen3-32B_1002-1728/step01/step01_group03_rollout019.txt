import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, noise_level):
        h_pert = h.copy()
        h_pert += np.random.uniform(-noise_level, noise_level, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        factor = target_sum / current_sum
        h_pert *= factor
        h_pert = np.clip(h_pert, 0, 1)
        return h_pert
    
    # Adaptive noise and diversification strategy
    noise_levels = [0.2, 0.15, 0.1, 0.08]
    tries_per_level = 5
    current_level = 0
    best_improvement_time = start_time
    
    while time.time() - start_time < budget_s - 20:
        # Restart perturbation level if no improvement for 200s
        if time.time() - best_improvement_time > 200:
            current_level = 0 if current_level < len(noise_levels) - 1 else current_level + 1
            best_improvement_time = time.time()
        
        noise = noise_levels[current_level]
        for _ in range(tries_per_level):
            if time.time() - start_time > budget_s - 20:
                break
                
            x0 = get_perturbed(best_h, noise)
            res = minimize(
                objective, 
                x0=x0, 
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 120, 'ftol': 1e-6}
            )
            
            if res.success:
                curr_val = objective(res.x)
                if curr_val < best_c5:
                    best_h = res.x.copy()
                    best_c5 = curr_val
                    best_improvement_time = time.time()
        
        # Progressively lower noise as time remains to polish
        if current_level < len(noise_levels) - 1 and time.time() - start_time < budget_s - 300:
            current_level += 1
    
    return (best_h, best_c5, n_points)
