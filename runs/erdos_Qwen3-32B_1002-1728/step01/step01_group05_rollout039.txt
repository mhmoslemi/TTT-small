import numpy as np
import time
from scipy.optimize import minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()

    start_time = time.time()

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)

    perturb_amp = 0.3
    perturb_decay = 0.95
    trials = 0
    max_trials = 150

    while time.time() - start_time < budget_s and trials < max_trials:
        # Create a perturbed candidate
        perturbation = np.random.uniform(-perturb_amp, perturb_amp, size=n_points)
        perturbed_h = best_h + perturbation

        perturbed_sum = np.sum(perturbed_h)
        if abs(perturbed_sum) < 1e-9:
            adj_perturbed_h = np.zeros_like(perturbed_h)
        else:
            adj_perturbed_h = perturbed_h * (target_sum / perturbed_sum)
            adj_perturbed_h = np.clip(adj_perturbed_h, 0.0, 1.0)

        # Optimize with SLSQP using tighter tolerance and more iterations
