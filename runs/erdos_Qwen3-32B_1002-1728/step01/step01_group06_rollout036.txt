import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    max_time = budget_s

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 280
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-10:mid+10] = 0.25
        h *= (n_points / 2) / h.sum()

    population_size = 50
    max_iter = 500
    c1, c2 = 1.5, 1.5
    w_max, w_min = 0.9, 0.4
    max_velocity = 0.5

    particles = []
    velocities = []
    for _ in range(population_size):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        s = new_h.sum()
        if s == 0:
            new_h = h.copy()
        else:
            new_h *= (n_points / 2) / s
        particles.append(new_h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    pbest_pos = particles.copy()
    pbest_obj = np.array([compute_c5(particles[i], n_points) for i in range(population_size)])
    gbest_idx = np.argmin(pbest_obj)
    gbest_pos, gbest_obj = pbest_pos[gbest_idx].copy(), pbest_obj[gbest_idx]
