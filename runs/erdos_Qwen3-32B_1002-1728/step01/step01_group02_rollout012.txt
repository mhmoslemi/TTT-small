import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h_values, n_points, dx):
    ones_minus_h = 1.0 - h_values.reshape(n_points)
    cross = np.correlate(ones_minus_h, h_values.reshape(n_points), mode='full')
    return np.max(cross) * dx

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points

    def clip_sum(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if np.isclose(h_sum, 0):
            h[:] = 0.5
        else:
            h *= (n_points / 2) / h_sum
        return np.clip(h, 0, 1)

    if 'initial_h_values' in globals():
        h = initial_h_values.astype(np.float64)
    else:
        h = np.full(n_points, 0.5, dtype=np.float64)
    h = clip_sum(h)

    best_c5 = evaluate_C5(h, n_points, dx)
    best_h = h.copy()

    perturbs = [h.copy()]
    for _ in range(5):
        p = h.copy()
        noise = np.random.normal(0.0, 0.05, n_points)
        p += noise
        p = clip_sum(p)
        perturbs.append(p)

    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    for start in perturbs:
        if time.time() - start_time > budget_s - 20:
            break
        h_current = start.copy()
        for beta in beta_schedule:
            def objective(h_vec):
                h = h_vec.reshape(n_points)
                corr = np.correlate(h, 1-h, mode='full')
                corr *= dx
                scaled = beta * corr
                max_scaled = np.max(scaled)
                exp_scaled = np.exp(scaled - max_scaled)
                sum_exp = np.sum(exp_scaled)
                return (max_scaled + np.log(sum_exp)) / beta
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = minimize(objective, h_current, method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}],
                              options={'ftol': 1e-6, 'maxiter': 20})
            if result.success:
                h_current = clip_sum(result.x)
            else:
                h_current = start.copy()
            if time.time() - start_time > budget_s - 10:
                break
        current_c5 = evaluate_C5(h_current, n_points, dx)
        if current_c5 < best_c5:
            best_c5, best_h = current_c5, h_current.copy()
    
    return best_h, best_c5, n_points
