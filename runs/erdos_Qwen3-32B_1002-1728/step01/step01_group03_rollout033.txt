import numpy as np
import time
from scipy.optimize import minimize

def get_perturbed(h, noise_strength, num_perturb, target_sum):
    h_pert = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, size=int(num_perturb), replace=False)
    perturbation = np.random.uniform(-noise_strength, noise_strength, size=len(indices))
    h_pert[indices] += perturbation
    h_pert = np.clip(h_pert, 0, 1)
    current_sum = np.sum(h_pert)
    if current_sum != 0:
        h_pert *= target_sum / current_sum
    h_pert = np.clip(h_pert, 0, 1)
    return h_pert

    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
