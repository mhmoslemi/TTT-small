import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat, n, dx):
        h = h_flat.reshape(n)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    def downsample_h(h, factor):
        n = len(h)
        n_new = n // factor
        h_down = np.zeros(n_new)
        for i in range(n_new):
            start = i * factor
            end = start + factor
            h_down[i] = np.mean(h[start:end])
        return h_down
    
    def upsample_h(h_coarse, factor):
        return np.kron(h_coarse, np.ones(factor, dtype=int))
    
    # Coarse optimization
    ds_factor = 5
    n_coarse = n_points // ds_factor
    h_coarse_init = downsample_h(h_init, ds_factor)
    target_sum_coarse = n_coarse / 2
    
    def objective_coarse(h_flat):
        return objective(h_flat, n_coarse, 2.0 / n_coarse)
    
    cons_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_coarse}]
    bounds_coarse = [(0.0, 1.0) for _ in range(n_coarse)]
    
    best_c5_coarse = objective_coarse(h_coarse_init)
    best_h_coarse = h_coarse_init.copy()
    
    start_time = time.time()
    time_budget_coarse = min(300, budget_s - 60)
    
    while time.time() - start_time < time_budget_coarse:
        h_pert = best_h_coarse.copy()
        h_pert += np.random.uniform(-0.1, 0.1, n_coarse)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert *= target_sum_coarse / current_sum
        res = minimize(
            objective_coarse, x0=h_pert, method='L-BFGS-B',
            bounds=bounds_coarse, constraints=cons_coarse,
            options={'maxiter': 30, 'ftol': 1e-6}
        )
        if res.success:
            curr_val = objective_coarse(res.x)
            if curr_val < best_c5_coarse:
                best_c5_coarse = curr_val
                best_h_coarse[:] = res.x
    
    h_upsampled = upsample_h(best_h_coarse, ds_factor)
    h_upsampled *= target_sum / np.sum(h_upsampled)
    
    # Final optimization at full resolution
    n_points = len(h_upsampled)
    dx = 2.0 / n_points
    best_h = h_upsampled.copy()
    best_c5 = objective(best_h, n_points, dx)
    
    start_time_final = time.time()
    time_budget_final = max(budget_s - (time.time() - start_time), 60)

        h_pert *= target_sum / current_sum

            if curr_val < best_c5:
                best_c5 = curr_val
                best_h[:] = res.x
    
    return (best_h, best_c5, n_points)
