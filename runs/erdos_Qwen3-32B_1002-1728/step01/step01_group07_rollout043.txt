import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import time
    
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8}
    )
    h_best = result.x.copy()
    c5_best = objective(h_best)
    
    # Perturbation and additional optimizations
    rs = np.random.RandomState(seed)
    elapsed = time.time() - start_time
    max_attempts = int((budget_s - elapsed) / 90) + 1  # Heuristic based on ~90s per attempt
    
    for attempt in range(max_attempts):
        if time.time() - start_time > budget_s - 15:
            break
        # Perturb around previous solution
        perturbed = h_best + rs.normal(0, 0.08, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Renormalize to maintain constraint
        perturbed *= (target_sum / np.clip(np.sum(perturbed), 1e-8, n_points))
        
        # Adaptive maxiter based on time
        remaining = budget_s - (time.time() - start_time)
        maxiter = min(150, int(remaining // 10 * 1.5))
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP' if attempt % 3 != 0 else 'L-BFGS-B',
            bounds=bounds,
            constraints=cons if attempt % 3 != 0 else None,
            options={
                'maxiter': maxiter,
                'ftol': 1e-8 if attempt % 2 else 1e-7,
                'eps': 1e-4
            }
        )
        
        if result.success:
            h_candidate = np.clip(result.x, 0, 1)
            # Enforce constraint in case of solver error
            h_candidate *= (target_sum / np.sum(h_candidate))
            c5_candidate = objective(h_candidate)
            if c5_candidate < c5_best:
                h_best = h_candidate.copy()
                c5_best = c5_candidate
    
    # Final sanitization
    h_best = np.clip(h_best, 0, 1)
    h_best *= (target_sum / np.sum(h_best))
    
    return (h_best.tolist(), c5_best, n_points)
