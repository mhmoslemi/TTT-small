    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Ensure feasibility: clip and scale
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s != 0:
            h *= target_sum / s
            h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0)] * n_points
    
    def take_step(x):
        noise = np.random.uniform(-0.08, 0.08, len(x))
        x += noise
        x = np.clip(x, 0, 1)
        s = x.sum()
        if s != 0:
            x *= target_sum / s
            x = np.clip(x, 0, 1)
        return x
    
    minimizer_kwargs = {
        'method': 'SLSQP',
        'bounds': bounds,
        'options': {'maxiter': 50, 'ftol': 1e-6}
    }
    
    bh = basinhopping(
        objective,
        x0=h_init,
        minimizer_kwargs=minimizer_kwargs,
        take_step=take_step,
        niter=1,  # Will expand in loop
        T=2.0,
        stepsize=0.05
    )
    
    start_time = time.time()
    while time.time() - start_time < budget_s - 60:
        remaining = budget_s - 60 - (time.time() - start_time)
        if remaining < 1:
            break
        nit = max(1, int(remaining / 5))  # Adaptive based on 5s per iteration
        bh.run(nit)
    
    best_h = bh.x.copy()
    # Final repair to strictly satisfy constraints
    best_h = np.clip(best_h, 0, 1)
    s = best_h.sum()
    if s != 0:
        best_h *= target_sum / s
        best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
