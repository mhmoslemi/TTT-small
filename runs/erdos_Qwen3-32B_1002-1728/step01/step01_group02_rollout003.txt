import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5_aux(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return cross * dx

def run():
    start_time = time.time()
    budget_s = 1100
    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.full(n_points, 0.5, dtype=np.float64)

    best_h = h_initial.copy()
    overlaps = evaluate_C5_aux(best_h, n_points, dx)
    best_c5 = np.max(overlaps)

    def repair(h_vec):
        h = h_vec.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        total = h_clipped.sum()
        if abs(total - n_points / 2) > 1e-8:
            scale = (n_points / 2) / total
            h_clipped = np.clip(h_clipped * scale, 0.0, 1.0)
        return h_clipped

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    beta_seq = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    def objective_fine(h_vec):
        return np.max(evaluate_C5_aux(h_vec, n_points, dx))

    num_starts = 5
    starts = [h_initial.copy()]
    for i in range(num_starts):
        np.random.seed(i + int((time.time()%1000)*1000))
        noise = np.random.normal(0, 0.1, n_points)
        new_h = (h_initial + noise).copy()
        new_h = repair(new_h)
        starts.append(new_h)

    for start in starts:
        if time.time() - start_time >= budget_s - 10:
            break
        current_h = start.copy()
        for beta in beta_seq:
            if time.time() - start_time >= budget_s - 10:
                break

            def objective(h_vec, b=beta):
                h = h_vec.reshape(n_points)
                ones_h = 1.0 - h
                cross_corr = np.correlate(h, ones_h, mode='full')
                overlaps = cross_corr * dx
                max_val = np.max(overlaps)
                shifted = np.exp(b * (overlaps - max_val))
                sum_exp = shifted.sum()
                return max_val + (1 / b) * np.log(sum_exp)

            res = minimize(objective, current_h, method='SLSQP', bounds=bounds, constraints=cons, jac=None,

            if res.success:
                improved = repair(res.x)
                c5 = np.max(evaluate_C5_aux(improved, n_points, dx))
