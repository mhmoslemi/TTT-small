import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n = 280
        h = np.zeros(n)
        h[n//2-5:n//2+5] = 0.5
        h *= n / 2 / h.sum()
    n = len(h)
    
    # Feasible initial point
    h = np.clip(h, 0, 1)
    h *= n / 2 / h.sum()
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, 'full') * (2 / n))
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    sigma = 0.15
    
    while time.time() - start_time < budget_s * 0.98:
        # Generate proposal
        noise = np.random.normal(size=n)
        h_proposal = h + sigma * T * noise
        h_proposal = np.clip(h_proposal, 0, 1)
        proposal_sum = h_proposal.sum()
        if proposal_sum > 1e-9:
            h_proposal *= (n / 2) / proposal_sum
        
        # Compute objectives
        current_obj = np.max(np.correlate(h, 1-h, 'full') * (2 / n))
        proposed_obj = np.max(np.correlate(h_proposal, 1-h_proposal, 'full') * (2 / n))
        
        if proposed_obj < current_obj:
            h = h_proposal.copy()
            if proposed_obj < best_c5:
                best_h = h.copy()
                best_c5 = proposed_obj
                T *= 0.9  # Faster cooling on improvement
        else:
            delta = proposed_obj - current_obj
            if np.random.rand() < np.exp(-delta / T):
                h = h_proposal.copy()
        T *= alpha
    
    # Local optimization refinement
    def objective(x):
        return np.max(np.correlate(x, 1-x, 'full') * (2 / n))
    
    bounds = Bounds(0, 1)
    constraint = {'type': 'eq', 'fun': lambda x: x.sum() - n/2}
    
    result = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    
    if result.success:
        optimized = result.x
        optimized_c5 = result.fun
        if optimized_c5 < best_c5:
            best_h = optimized
            best_c5 = optimized_c5
    
    return best_h, best_c5, n
