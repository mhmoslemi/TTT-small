import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    np.random.seed(seed)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        current_h = best_h.copy()
        # Feasibility projection and perturbation
        current_h = np.clip(current_h, 0, 1)
        current_sum = np.sum(current_h)
        delta = (n_points / 2.0) - current_sum
        current_h[0] += delta
        current_h[0] = np.clip(current_h[0], 0, 1)
        # Add random perturbation
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        current_h = np.clip(current_h + perturb, 0, 1)
        # Re-project to sum constraint
        current_sum = np.sum(current_h)
        delta = (n_points / 2.0) - current_sum
        current_h[0] += delta
        current_h[0] = np.clip(current_h[0], 0, 1)
        # Objective check
        corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
        new_c5 = np.max(corr)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = current_h

    # Final projection to ensure strict feasibility
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    delta = (n_points / 2.0) - current_sum
    best_h[0] += delta
    best_h[0] = np.clip(best_h[0], 0, 1)

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
