import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps(h, dx):
    h = h.reshape(-1)
    ones_minus_h = 1 - h
    cross = np.correlate(h, ones_minus_h, mode='full') * dx
    return cross

def evaluate_C5(h_vec, n_points, dx):
    return np.max(compute_overlaps(h_vec, dx))

def objective_coarse(h_vec, beta, n_points, dx):
    cross = compute_overlaps(h_vec, dx)
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * cross)))

def run():
    np.random.seed(42)
    start_time = time.time()
    budget_s = 1100

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
        n_points = h_initial.size
    else:
        n_points = 45
        h_initial = np.full(n_points, 0.5, dtype=np.float64)

    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = evaluate_C5(best_h, n_points, dx)

    candidates = [h_initial.copy()]
    for _ in range(4):
        perturbed = h_initial.copy() + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if not np.isclose(perturbed.sum(), n_points / 2, atol=1e-8):
            scale = (n_points / 2) / perturbed.sum()
            perturbed = np.clip(perturbed * scale, 0, 1)
        candidates.append(perturbed)

    beta_seq = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1)] * n_points

    for candidate in candidates:

        for beta in beta_seq:
            def obj(h_vec):
