import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    pop_size = 50
    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover rate
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i < pop_size // 2:
            # Perturb initial_h_values
            h = initial_h_values.copy() + np.random.normal(0, 0.1, n_points)
        else:
            # Random init
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        population.append(h)
    population = np.array(population)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s - 5:
        for idx in range(pop_size):
            # Select 3 distinct indices
            candidates = np.delete(np.arange(pop_size), idx)
            a, b, c = population[np.random.choice(candidates, 3, replace=False)]
            # Mutation
            mutant = a + F * (b - c)
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[idx])
            # Ensure at least one parameter from mutant
            if not np.any(np.random.rand(n_points) < CR):
                mut_idx = np.random.randint(n_points)
                trial[mut_idx] = mutant[mut_idx]
            
            # Clip and normalize
            trial = np.clip(trial, 0, 1)
            current_sum = trial.sum()
            current_sum = max(current_sum, 1e-6)  # Prevent division by zero
            trial = trial / current_sum * (n_points / 2)
            
            # Evaluate and update
            trial_c5 = objective(trial)
            target_c5 = objective(population[idx])
            if trial_c5 < target_c5:
                population[idx] = trial
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5
                    
        # Perturb best occasionally to prevent premature convergence
        if np.random.rand() < 0.1:
            best_h = np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1)
            best_h = best_h / best_h.sum() * (n_points / 2)
    
    return best_h, best_c5, n_points
