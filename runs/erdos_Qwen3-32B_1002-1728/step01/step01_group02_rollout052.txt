import numpy as np
import time
from scipy.optimize import minimize

def constraint_sum(x, args):
    n_points = args[0]
    return np.sum(x) - n_points / 2

def objective_log_sum_exp(x, args):
    n_points, dx, beta = args
    h = x.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    overlaps = cross * dx
    exp_overlaps = np.exp(beta * overlaps)
    sum_exp = np.sum(exp_overlaps)
    return (1.0 / beta) * np.log(sum_exp)

def evaluate_C5(x, args):
    n_points, dx = args
    h = x.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def run():

    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h_initial[:] = 0.5

    best_c5 = evaluate_C5(h_initial, (n_points, dx))
    best_h = h_initial.copy()

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_points,)}]

    perturbed_candidates = []

        seed = int((time.time() % 1000) * 1000)
        perturbed = np.random.default_rng(seed).normal(0.0, 0.1, n_points)
        new_h = np.clip(h_initial + perturbed, 0.0, 1.0)
