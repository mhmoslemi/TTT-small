import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time()
    alpha = 0.1
    alpha_decay = 0.99
    max_no_improvement = 100
    no_improvement = 0
    perturb_std = 0.05

    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    while time() - start_time < budget_s - 10:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        # Compute gradient for current k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0
            if 0 <= i + k < n_points:
                term1 = 1 - h[i + k]
            term2 = 0
            if 0 <= i - k < n_points:
                term2 = -h[i - k]
            gradient[i] = term1 + term2

        # Gradient step
        h -= alpha * gradient

        # Project onto box constraint and sum constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale_factor = (n_points / 2) / (h_sum + 1e-10)
        h *= scale_factor
        # Clip again after scaling and re-scale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale_factor = (n_points / 2) / (h_sum + 1e-10)
        h *= scale_factor

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h
            no_improvement = 0
            alpha *= alpha_decay
        else:
            no_improvement += 1
            if no_improvement >= max_no_improvement:
                # Perturb and reset
                h = best_h.copy()
                noise = np.random.normal(0, perturb_std, n_points)
                h = np.clip(h + noise, 0.0, 1.0)
                h_sum = np.sum(h)
                h *= (n_points / 2) / h_sum
                alpha *= 0.5
                no_improvement = 0
        alpha = max(alpha, 1e-5)

    return best_h, best_c5, n_points
