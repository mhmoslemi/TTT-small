import numpy as np
import time
from scipy.optimize import minimize, differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        current_sum = np.sum(h)
        if current_sum == 0:
            return float('inf')
        h_scaled = h * (target_sum / current_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result_de = differential_evolution(
        objective_func, bounds,
        strategy='best1bin',
        maxiter=5,
        popsize=5,
        tol=1e-5,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=False)
    best_h = result_de.x
    best_c5 = objective_func(best_h)
    
    # Refine with local search
    def objective_slspq(h_flat):
        return objective_func(h_flat)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    res = minimize(
        objective_slspq, x0=best_h, 
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if res.success:
        curr_val = objective_func(res.x)
        if curr_val < best_c5:
            best_h = res.x
            best_c5 = curr_val
    
    # Additional perturbation & optimization
    while time.time() - start_time < budget_s - 30:
        x0 = best_h + np.random.normal(0, 0.05, n_points)
        x0 = np.clip(x0, 0, 1)
        sum_pert = np.sum(x0)
        x0 = x0 * (target_sum / sum_pert) if sum_pert != 0 else x0
        subres = minimize(
            objective_slspq, x0=x0, 
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50}
        )
        if subres.success:
            val = objective_func(subres.x)
            if val < best_c5:
                best_h = subres.x
                best_c5 = val
    
    return (best_h, best_c5, n_points)
