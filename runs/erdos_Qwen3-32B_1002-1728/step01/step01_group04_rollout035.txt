    positions = np.zeros((n_particles, n_points))
    positions[0] = h.copy()
    for i in range(1, n_particles):
        h_rand = np.random.uniform(0, 1, n_points)
        h_sum = h_rand.sum()
        if h_sum == 0:
            h_scaled = np.zeros(n_points)
        else:
            h_scaled = h_rand * (target_sum / h_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        h_clipped_sum = h_clipped.sum()
        if h_clipped_sum == 0:
            h_scaled = np.zeros(n_points)
        else:
            h_scaled = h_clipped * (target_sum / h_clipped_sum)
        h_scaled = np.clip(h_scaled, 0, 1)
        positions[i] = h_scaled
    
    velocities = np.zeros_like(positions)
    pbest_positions = np.copy(positions)
    pbest_fitness = np.array([np.max(np.correlate(p, 1 - p, mode='full') * dx) for p in positions])
    gbest_idx = np.argmin(pbest_fitness)
    gbest_position = np.copy(pbest_positions[gbest_idx])
    gbest_fitness = pbest_fitness[gbest_idx]
    
    while time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - positions[i]) +
                c2 * r2 * (gbest_position - positions[i])
            )
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            positions[i] += velocities[i]
            
            h_clipped = np.clip(positions[i], 0, 1)
            h_sum = h_clipped.sum()
            if h_sum != 0:
                h_scaled = h_clipped * (target_sum / h_sum)
                h_clipped = np.clip(h_scaled, 0, 1)
            positions[i] = h_clipped
            
            current_fitness = np.max(np.correlate(positions[i], 1 - positions[i], mode='full') * dx)
            
            if current_fitness < pbest_fitness[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_fitness[i] = current_fitness
                if current_fitness < gbest_fitness:
                    gbest_position = positions[i].copy()
                    gbest_fitness = current_fitness
    
    return gbest_position, gbest_fitness, n_points
