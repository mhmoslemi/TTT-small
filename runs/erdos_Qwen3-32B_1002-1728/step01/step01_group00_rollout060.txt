import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corrs = np.correlate(h, 1 - h, mode='full')
        return np.max(corrs) * dx

    def project(h_vec, target_sum):
        h_clamped = np.clip(h_vec, 0.0, 1.0)
        current_sum = np.sum(h_clamped)
        if current_sum == 0:
            h_clamped[0] = 1.0
            current_sum = 1.0
        scale = target_sum / current_sum
        h_scaled = h_clamped * scale
        h_proj = np.clip(h_scaled, 0.0, 1.0)
        return h_proj

    n_particles = 20
    particles = []
    for _ in range(n_particles):
        h = h_initial.copy() + np.random.normal(0, 0.01, n_points)
        particles.append(project(h, target_sum))
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]

    pbest_positions = [p.copy() for p in particles]
    pbest_scores = [compute_c5(p) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        current_time = time.time() - start_time
        elapsed_fraction = current_time / budget_s
        w = 0.9 - 0.8 * elapsed_fraction

        for i in range(n_particles):
