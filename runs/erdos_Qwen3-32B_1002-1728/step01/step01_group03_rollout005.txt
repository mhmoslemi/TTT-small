        x_scaled = np.clip(x_scaled, 0.0, 1.0)
        exceeded = x_scaled > 1.0
        if np.any(exceeded):
            x_scaled[exceeded] = 1.0
            new_sum = np.sum(x_scaled)
            if new_sum < target_sum - 1e-8:
                extra = target_sum - new_sum
                available = (1.0 - x_scaled) * ~exceeded
                added = extra / (np.sum(available) + 1e-8) * available
                x_scaled += added
                x_scaled = np.clip(x_scaled, 0.0, 1.0)
        return x_scaled
    
    def objective(x):
        h = repair(x)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()
    max_time = budget_s - 60  # Reserve 60s for final steps
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        init='sobol',
        updating='deferred',
        workers=1
    )
    
    # Time-based termination
    while time.time() - start_time < max_time:
        pass  # Allow for parallel/async execution tracking here if needed
    
    if result.success:
        h_opt = repair(result.x)
    else:
        h_opt = h_init.copy()
    
    final_c5 = objective(h_opt)
    return (h_opt, final_c5, n_points)
