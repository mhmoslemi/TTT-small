import time
from scipy.optimize import minimize, Bounds
from cma import CMAEvolutionStrategy

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx

def project_to_simplex(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

    if s == 0:
        return np.zeros_like(h)
    return h * (target_sum / s)
