import numpy as np
import time
import random

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    def repair(h_candidate, target_sum=target_sum):
        h_candidate = np.clip(h_candidate, 0, 1)
        s = h_candidate.sum()
        if s == 0:
            return np.clip(h_init, 0, 1)
        scaling = target_sum / s
        h_candidate *= scaling
        h_candidate = np.clip(h_candidate, 0, 1)
        return h_candidate

    def perturb(h, noise_scale=0.03):
        h = h.copy()
        h += np.random.normal(0, noise_scale, h.shape)
        return repair(h)

    def tournament_selection(population, fitnesses, tournament_size=4):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin(fitnesses[indices])]
        return population[best_idx].copy()

    # Generate initial population with repair
    pop_size = 50
    population = [h_init]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        population.append(repair(h))
    population = np.array(population)
    fitness = np.array([compute_c5(ind, dx) for ind in population])
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = np.min(fitness)

    # Main GA loop with adaptive replacement
    while time.time() - start_time < budget_s - 10:
        parents = []
        for _ in range(pop_size // 2):
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            # Crossover with probabilistic point
            if np.random.rand() > 0.2:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                c1 = np.where(mask, p1, p2)
                c2 = np.where(mask, p2, p1)
            else:
                # Single-point crossover
                cp = np.random.randint(1, n_points)
                c1 = np.concatenate((p1[:cp], p2[cp:]))
                c2 = np.concatenate((p2[:cp], p1[cp:]))
            # Apply mutation
            c1 = perturb(c1, 0.05 if np.random.rand() < 0.2 else 0.01)
            c2 = perturb(c2, 0.05 if np.random.rand() < 0.2 else 0.01)
            parents.append((c1, c2))

        offspring = np.array([c for pair in parents for c in pair])
        offspring_fitness = np.array([compute_c5(c, dx) for c in offspring])
        combined = np.concatenate((population, offspring))
        combined_fitness = np.concatenate((fitness, offspring_fitness))
        
        # Select new population with diversity pressure
        new_population = []
        # Carry elites
        elite_count = int(pop_size * 0.05)
        elite_idx = np.argsort(combined_fitness)[:elite_count]
        new_population.extend(combined[elite_idx])

        # Fill rest with tournament
        while len(new_population) < pop_size:
            candidate = tournament_selection(combined, combined_fitness)
            new_population.append(candidate)
        population = np.array(new_population)
        fitness = np.array([compute_c5(ind, dx) for ind in population])
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_h, best_c5 = population[np.argmin(fitness)].copy(), current_best

            best_h, best_c5 = neighbor, neighbor_c5

    return best_h, best_c5, n_points
