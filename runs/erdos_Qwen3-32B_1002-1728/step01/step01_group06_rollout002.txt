import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    max_time = budget_s
    np.random.seed(seed)
    
    n_points = 280
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        old_n = len(initial_h)
        if old_n != n_points:
            old_x = np.linspace(0, 2, old_n)
            new_x = np.linspace(0, 2, n_points)
            h_init = np.interp(new_x, old_x, initial_h)
            h_init *= (n_points / 2) / h_init.sum()
            h_init = np.clip(h_init, 0, 1)
        else:
            h_init = initial_h.copy()
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[mid-5:mid+5] = 1.0
        h_init *= (n_points / 2) / h_init.sum()

    n_particles = 50
    c1, c2 = 1.5, 1.5
    w_max, w_min = 0.9, 0.4
    velocity_clamp = 0.5
    max_iter = 500

    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    for i in range(n_particles):
        pos = h_init.copy() + np.random.uniform(-0.1, 0.1, n_points)
        pos = np.clip(pos, 0, 1)
        pos *= (n_points / 2) / pos.sum()
        positions[i] = pos
        velocities[i] = np.random.uniform(-0.05, 0.05, n_points)

    def objective(h):
        corr = np.correlate(h, 1-h, 'full')
        return np.max(corr) * (2.0 / n_points)

    scores = np.array([objective(p) for p in positions])
    pbest_pos = positions.copy()
    pbest_scores = scores.copy()
    gbest_idx = np.argmin(pbest_scores)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    no_improve = 0

    for iter in range(max_iter):
        if time.time() - start_time > max_time * 0.98:
            break
        w = w_max - (w_max - w_min) * iter / max_iter
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            cog = c1 * r1 * (pbest_pos[i] - positions[i])
            soc = c2 * r2 * (gbest_pos - positions[i])
            new_vel = w * velocities[i] + cog + soc
            new_vel = np.clip(new_vel, -velocity_clamp, velocity_clamp)
            velocities[i] = new_vel
            new_pos = np.clip(positions[i] + new_vel, 0, 1)
            new_pos *= (n_points / 2) / new_pos.sum()
            positions[i] = new_pos

        scores = np.array([objective(p) for p in positions])
        for i in range(n_particles):
            if scores[i] < pbest_scores[i]:
                pbest_scores[i] = scores[i]
                pbest_pos[i] = positions[i].copy()
        
        curr_gbest_idx = np.argmin(pbest_scores)
        curr_gbest_score = pbest_scores[curr_gbest_idx]
        if curr_gbest_score < gbest_score:
            gbest_score = curr_gbest_score
            gbest_pos = pbest_pos[curr_gbest_idx].copy()
            no_improve = 0
        else:

                break

    return gbest_pos, gbest_score, n_points
