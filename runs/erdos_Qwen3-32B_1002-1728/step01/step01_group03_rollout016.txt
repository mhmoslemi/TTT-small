import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    current_noise = 0.2
    
    def get_perturbed(h, noise):
        h_pert = h.copy()
        h_pert += np.random.uniform(-noise, noise, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert *= target_sum / current_sum
        return h_pert
    
    tries = 0
    while time.time() - start_time < budget_s - 60:
        tries += 1
        x0 = get_perturbed(best_h, current_noise)
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        new_c5 = objective(res.x) if res.success else np.inf
        if res.success and new_c5 < best_c5:
            best_h = res.x.copy()
            best_c5 = new_c5
            # Reduce noise if improvement
            current_noise = max(0.05, current_noise * 0.9)
        else:
            # Increase noise if no improvement
            current_noise = min(0.5, current_noise * 1.1)
        # Limit tries to prevent endless loop (but primarily time-based)
        if tries > 50:
            break
    return (best_h, best_c5, n_points)
