    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    # Global search with differential evolution
    def objective(global_h):
        global_h = global_h.reshape(n_points)
        clipped = np.clip(global_h, 0, 1)
        scaled = clipped * (target_sum / np.sum(clipped)) if np.sum(clipped) != 0 else np.zeros_like(clipped)
        scaled = np.clip(scaled, 0, 1)
        return compute_c5(scaled, dx)
    
    bounds_de = [(0, 1)] * n_points
    de_start = time.time()
    de_result = differential_evolution(
        objective, bounds_de, strategy='best1bin',
        maxiter=30, popsize=20, tol=1e-5,
        mutation=(0.5, 1.0), recombination=0.7,
        seed=seed, workers=1, updating='deferred'
    )
    de_end = time.time()
    
    h_de = de_result.x.reshape(n_points)
    h_de = np.clip(h_de, 0, 1)
    if np.sum(h_de) != 0:
        h_de = h_de * (target_sum / np.sum(h_de))
        h_de = np.clip(h_de, 0, 1)
    
    if compute_c5(h_de, dx) < best_c5:
        best_h = h_de.copy()
        best_c5 = compute_c5(best_h, dx)
    
    # Local refinement with SLSQP
    bounds = Bounds(0.0, 1.0)
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    def slsqp_objective(h_vec):
        return compute_c5(h_vec, dx)
    
    for initial_guess in [best_h, h0]:
        result = minimize(
            slsqp_objective, x0=initial_guess,
            method='SLSQP', bounds=bounds,
            constraints=constraint, options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success:
            c5_new = result.fun
            if c5_new < best_c5:
                best_h = result.x.copy()
                best_c5 = c5_new
    
    # Perturbation-driven local search
    perturb_amp = 0.2
    max_perturb_steps = 150
    trials = 0
    
    while time.time() - start_time < budget_s and trials < max_perturb_steps:
        if trials % 5 == 0:
            base = best_h.copy()
            perturbation = np.random.uniform(-perturb_amp, perturb_amp, n_points)
            perturbed = np.clip(base + perturbation, 0, 1)
            val_sum = np.sum(perturbed)
            if val_sum != 0:
                perturbed = perturbed * (target_sum / val_sum)
                perturbed = np.clip(perturbed, 0, 1)
            result = minimize(
                slsqp_objective, x0=perturbed, method='SLSQP',
                bounds=bounds, constraints=constraint,
                options={'maxiter': 50}
            )
        else:
            result = minimize(
                slsqp_objective, x0=best_h, method='SLSQP',
                bounds=bounds, constraints=constraint,
                options={'maxiter': 50}
            )
        
        if result.success:
            c5_new = result.fun
            if c5_new < best_c5:
                best_h = result.x.copy()
                best_c5 = c5_new
        perturb_amp *= 0.98
        trials += 1
    
    return (best_h, best_c5, n_points)
