from scipy.optimize import differential_evolution, minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)

        h = x.reshape(n_points)

            sum_h = 0.5 * n_points
            h_clipped = np.ones_like(h_clipped) * 0.5

    bounds = [(0, 1)] * n_points
