import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_budget = budget_s

    grid_size = 280
    target_sum = grid_size / 2  # 140.0 for 280 points

    # Initialize h using initial_h_values or default
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        old_n = len(initial_h)
        if old_n != grid_size:
            old_x = np.linspace(0, 2, old_n)
            new_x = np.linspace(0, 2, grid_size)
            h = np.interp(new_x, old_x, initial_h)
        else:
            h = initial_h.copy()
        h *= target_sum / h.sum()
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(grid_size)
        mid = grid_size // 2
        h[mid-5:mid+5] = 0.5
        h *= target_sum / h.sum()
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h, grid_size)

    # Simulated Annealing parameters
    temperature = 1.0
    cooling_rate = 0.995
    sigma = 0.1
    alpha_improve = 0.9
    max_iterations = 10000  # Tuned for early termination if needed

    for _ in range(max_iterations):
        if time.time() - start_time > max_budget * 0.98:
            break

        # Generate perturbation and repair constraints
        perturbation = sigma * temperature * np.random.randn(grid_size)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        scale = target_sum / new_h.sum()
        new_h *= scale
        new_h = np.clip(new_h, 0, 1)

        # Evaluate new candidate
        new_c5 = compute_c5(new_h, grid_size)
        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / temperature) > np.random.rand():
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
                temperature *= alpha_improve  # Faster cooling on improvement
            # Accept in current state
            best_h, best_c5 = new_h.copy(), new_c5
        temperature *= cooling_rate  # Regular cooling

    # Final SLSQP refinement within remaining budget
    if time.time() - start_time < max_budget * 0.98:
        def objective(x):
            x = np.clip(x, 0, 1)
            return compute_c5(x, grid_size)
        bounds = Bounds(0, 1)
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(
            objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        if result.success:
            refined_h = np.clip(result.x, 0, 1)

            final_c5 = compute_c5(refined_h, grid_size)
            if final_c5 < best_c5:
                best_h, best_c5 = refined_h, final_c5

    return best_h, best_c5, grid_size

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
