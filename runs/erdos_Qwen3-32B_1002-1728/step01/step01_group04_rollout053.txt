import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    def project_to_constraints(h, target_sum_val):
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum == 0:
            h_clipped[:] = target_sum_val / n_points
        else:
            factor = target_sum_val / h_sum
            h_scaled = h_clipped * factor
            h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1

    particles = np.zeros((n_particles, n_points))
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    pbest_positions = particles.copy()
    pbest_scores = np.full(n_particles, np.inf)

    particles[0] = h.copy()
    for i in range(1, n_particles):
        raw = np.random.rand(n_points)
        particles[i] = project_to_constraints(raw.copy(), target_sum)

    for i in range(n_particles):
        particles[i] = project_to_constraints(particles[i].copy(), target_sum)
        pbest_scores[i] = compute_c5(particles[i], dx)
        pbest_positions[i] = particles[i].copy()

    gbest_score = min(pbest_scores)
    gbest_index = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_index].copy()

    while time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = project_to_constraints(particles[i].copy(), target_sum)
            current_score = compute_c5(particles[i], dx)
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i].copy()

    gbest_position = project_to_constraints(gbest_position.copy(), target_sum)
    final_score = compute_c5(gbest_position, dx)
    return gbest_position, final_score, n_points
