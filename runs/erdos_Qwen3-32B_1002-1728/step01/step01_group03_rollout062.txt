import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def generate_random_h():
        while True:
            h = np.random.uniform(0, 1, n_points)
            h_sum = h.sum()
            h *= target_sum / h_sum
            h = np.clip(h, 0, 1)
            if abs(h.sum() - target_sum) < 1e-6:
                return h
    
    def get_perturbed(h, noise):
        h_pert = h.copy()
        h_pert += np.random.uniform(-noise, noise, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert *= target_sum / h_pert.sum()
        return h_pert
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    total_tries = 0
    while time.time() - start_time < budget_s - 30 and total_tries < 20:
        total_tries += 1
        
        # Stochastic starting point selection
        if np.random.rand() < 0.2:
            x0 = generate_random_h()
        else:
            # Dynamic noise scaling
            noise_level = max(0.1, 0.25 - 0.01 * total_tries)
            x0 = get_perturbed(best_h, noise_level)
        
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-7, 'disp': False}
        )
        
        if res.success:
            current_val = objective(res.x)
            if current_val < best_c5:
                best_h = res.x.copy()
                best_c5 = current_val
    
    return (best_h, best_c5, n_points)
