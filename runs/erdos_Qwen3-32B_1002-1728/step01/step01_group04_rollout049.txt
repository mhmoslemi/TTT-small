        end_idx = n_points
        start = start_idx * dx_val
        end = end_idx * dx_val
        intervals.append((start, end, current_val))
        return intervals

    initial_intervals = parse_intervals(h, n_points, dx)
    K = len(initial_intervals)

    def intervals_to_vars(intervals):
        vars = []
        for (s, e, a) in intervals:
            vars.extend([s, e, a])
        return np.array(vars)

    x0 = intervals_to_vars(initial_intervals)

    bounds = []

        bounds.append((0.0, 2.0))  # s

            return x[3*i + 1] - x[3*(i+1)]

        return total - target_sum

    constraints.append({'type': 'eq', 'fun': sum_constraint})
