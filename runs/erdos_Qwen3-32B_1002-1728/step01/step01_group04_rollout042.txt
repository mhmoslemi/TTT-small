            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if not result.success:
            continue
        candidate = np.clip(result.x, 0, 1)
        candidate_sum = candidate.sum()
        if not np.isclose(candidate_sum, target_sum, atol=1e-6):
            candidate *= target_sum / candidate_sum
            candidate = np.clip(candidate, 0, 1)
        current_c5 = compute_c5(candidate, dx)
        if current_c5 < best_c5:
            best_h, best_c5 = candidate, current_c5

    return best_h, best_c5, n_points
