import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h_vec, target_sum):
        h = h_vec.copy()
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        if abs(current_sum) < 1e-8:
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    population = []
    for _ in range(pop_size // 2):
        perturbed = initial_h_values.copy() + np.random.normal(0, 0.05, n_points)
        population.append(project(perturbed, target_sum))
    for _ in range(pop_size // 2):
        rand_h = np.random.rand(n_points)
        population.append(project(rand_h, target_sum))
    population = np.array(population)
    obj_values = np.array([objective(p) for p in population])
    best_idx = np.argmin(obj_values)
    best_h = population[best_idx].copy()
    best_c5 = obj_values[best_idx]

    start_time = time.time()
    F = 0.8
    CR = 0.9

    while time.time() - start_time < budget_s - 10:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            r = np.random.choice(idxs, 3, replace=False)
            r1, r2, r3 = r
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = project(trial, target_sum)
            trial_c5 = objective(trial)
            if trial_c5 < obj_values[i]:
                population[i] = trial.copy()
                obj_values[i] = trial_c5
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5
        if np.random.rand() < 0.1:
            perturbed = best_h + np.random.normal(0, 0.05, n_points)
            perturbed = project(perturbed, target_sum)
            trial_c5 = objective(perturbed)
            if trial_c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = trial_c5
        if time.time() - start_time > budget_s - 10:
            break
    return best_h, best_c5, n_points
