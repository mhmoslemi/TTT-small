import numpy as np
import time
from scipy.optimize import differential_evolution, Bounds, minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    def project_to_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if abs(current_sum) < 1e-9:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h *= scale
        return np.clip(h, 0.0, 1.0)
    
    # Initial perturbation of h0
    perturb_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
    perturbed = h0.copy()
    perturbed[perturb_indices] += np.random.uniform(-0.2, 0.2, size=len(perturb_indices))
    h_feasible = project_to_feasible(perturbed)
    candidate_c5 = compute_c5(h_feasible, dx)
    if candidate_c5 < best_c5:
        best_h = h_feasible.copy()
        best_c5 = candidate_c5
    
    # Global search with DE
    def obj_de(x):
        h = x.reshape(n_points)
        h_feas = project_to_feasible(h)
        return compute_c5(h_feas, dx)
    
    bounds_de = [(0, 1) for _ in range(n_points)]
    while time.time() - start_time < budget_s * 0.3 and time.time() - start_time < 300:
        result_de = differential_evolution(obj_de, bounds_de, maxiter=20, popsize=15, strategy='best1bin', mutation=(0.5, 1), recombination=0.8, tol=1e-6)
        if result_de.success:

    slsqp_bounds = Bounds(lb=0.0, ub=1.0)
