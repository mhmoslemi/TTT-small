from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:

            return h_adj
        total_possible = h_adj[can_decr].sum()
