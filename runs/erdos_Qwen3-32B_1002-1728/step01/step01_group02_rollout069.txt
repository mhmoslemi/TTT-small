import numpy as np
import cvxpy as cp
import time

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def compute_gradient(h, k_shift, dx):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        term1 = 0.0
        j_plus_k = j + k_shift
        if 0 <= j_plus_k < n:
            term1 = 1.0 - h[j_plus_k]
        term2 = 0.0
        j_minus_k = j - k_shift
        if 0 <= j_minus_k < n:
            term2 = -h[j_minus_k]
        grad[j] = dx * (term1 + term2)
    return grad

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy().astype(np.float64)
    else:
        h = np.ones(n_points) * 0.5
        h = np.clip(h, 0.0, 1.0)

    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum

    best_c5 = evaluate_C5(h.copy(), n_points, dx)
    best_h = h.copy()

    while time.time() - start_time < budget_s - 10:
        cross = np.correlate(h, 1 - h, mode='full') * dx
        max_over_c = np.max(cross)
        index_c = np.argmax(cross)
        k_shift = index_c - (n_points - 1)

        grad = compute_gradient(h, k_shift, dx)
        overlap_current = cross[index_c]

        h_var = cp.Variable(n_points)
        t = cp.Variable()

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            t >= overlap_current + cp.sum(cp.multiply(grad, (h_var - h)))
        ]
        objective = cp.Minimize(t)

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.SCS, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.array(h_var.value).flatten()
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            if abs(new_sum - n_points / 2) > 1e-8:
                new_h = new_h * (n_points / 2) / (new_sum + 1e-8)
            current_c5 = evaluate_C5(new_h, n_points, dx)
            if current_c5 < best_c5:
                best_c5, best_h = current_c5, new_h.copy()
            h = new_h
        else:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)

    return best_h, best_c5, n_points
