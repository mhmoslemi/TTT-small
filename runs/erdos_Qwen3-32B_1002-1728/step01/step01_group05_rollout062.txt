import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds

def project(x, target_sum):
    x_clipped = np.clip(x, 0.0, 1.0)
    s = np.sum(x_clipped)
    if s < 1e-12:
        return np.zeros_like(x)
    scaling = target_sum / s
    return x_clipped * scaling

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project(h0, target_sum)
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(x):
        return compute_c5(project(x, target_sum), dx)

    try:
        global_result = differential_evolution(
            objective,
            bounds=bounds,
            maxiter=10,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            workers=1,
            strategy='best1bin',
            polish=False,
            disp=False
        )
        global_h = project(global_result.x, target_sum)
        global_c5 = compute_c5(global_h, dx)
        if global_c5 < best_c5:
            best_h = global_h.copy()
            best_c5 = global_c5
    except Exception:
        pass

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = Bounds(0.0, 1.0)
    max_local_trials = 15
