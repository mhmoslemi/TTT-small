import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_current = initial_h_values.copy()
    best_h = h_current.copy()
    best_c5 = np.inf

    # Precompute valid (i,j) pairs for all shifts
    shifts = []
    for k in range(-n_points + 1, n_points):
        valid_i = []
        if k >= 0:
            for i in range(0, n_points - k):
                j = i + k
                valid_i.append((i, j))
        else:
            k_abs = -k
            for i in range(k_abs, n_points):
                j = i + k
                valid_i.append((i, j))
        if valid_i:
            shifts.append((k, valid_i))

    while time.time() - start_time < budget_s:
        # Form convex subproblem
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [cp.sum(h) == n_points / 2.0, h >= 0, h <= 1]
        for k_info in shifts:
            _, pairs = k_info
            sum_linear = 0.0
            sum_terms = 0.0
            for i, j in pairs:
                hc_i = h_current[i]
                hc_j = h_current[j]
                sum_terms += hc_i * (1 - hc_j)
                sum_linear += (1 - hc_j) * h[i] - hc_i * h[j]
            constraint = sum_linear + sum_terms <= C / dx
            constraints.append(constraint)
        
        objective = cp.Minimize(C)
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.CVXOPT, max_iters=100, verbose=False, abstol=1e-6, reltol=1e-6)
            if problem.status != cp.OPTIMAL:
                h_current = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
                h_current = h_current / h_current.sum() * (n_points / 2.0)
                continue
            new_h = np.array(h.value)
            new_h = np.clip(new_h, 0, 1)
            scaling = (n_points / 2.0) / new_h.sum()
            new_h *= scaling
            # Evaluate current solution
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
                h_current = new_h.copy()
        except Exception as e:
            h_current = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
            h_current = h_current / h_current.sum() * (n_points / 2.0)
    
    return best_h, best_c5, n_points
