import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    def repair_solution(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        s = h_clipped.sum()
        if s == 0:
            return h_init.copy()
        h_scaled = h_clipped * (target_sum / s)
        return h_scaled

    def tournament_selection(population, fitness, size=3):
        indices = np.random.randint(0, len(population), size)
        idx = np.argmin(fitness[indices])
        return population[indices[idx]], fitness[indices[idx]]

    population_size = 40
    mutation_rate = 0.05
    tournament_size = 3
    elite_fraction = 0.1
    num_elites = int(elite_fraction * population_size)

    # Initialize population
    population = []
    fitness = []
    population.append(h_init.copy())
    fitness.append(compute_c5(h_init))

    for _ in range(population_size - 1):
        h_rand = np.random.random(n_points)
        h_fixed = repair_solution(h_rand)
        population.append(h_fixed)
        fitness.append(compute_c5(h_fixed))

    best_c5 = min(fitness)
    best_h = population[np.argmin(fitness)].copy()

    while time.time() - start_time < budget_s - 5:
        new_population = []
        new_fitness = []

        # Elitism
        elite_indices = np.argsort(fitness)[:num_elites]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            new_fitness.append(fitness[idx])

        # Generate new population
        while len(new_population) < population_size:
            # Parent selection
            parent1, fit1 = tournament_selection(population, fitness, tournament_size)
            parent2, fit2 = tournament_selection(population, fitness, tournament_size)

            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child = repair_solution(child)

            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_rate, n_points)
                child += noise
                child = repair_solution(child)

            # Evaluate
            child_c5 = compute_c5(child)
            new_population.append(child)
            new_fitness.append(child_c5)

        population = new_population[:population_size]
        fitness = new_fitness[:population_size]

        current_best_c5 = min(fitness)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[np.argmin(fitness)].copy()

    return best_h, best_c5, n_points
