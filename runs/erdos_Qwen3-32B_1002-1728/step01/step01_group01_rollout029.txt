import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if np.isclose(current_sum, 0):
            return h_clipped
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)
    
    pop_size = 50
    population = []
    
    for _ in range(pop_size // 2):
        perturbed = initial_h_values.copy() + np.random.normal(0, 0.05, n_points)
        population.append(project(perturbed))
    
    for _ in range(pop_size // 2):
        rand_h = np.random.rand(n_points)
        population.append(project(rand_h))
    
    population = np.array(population)
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
