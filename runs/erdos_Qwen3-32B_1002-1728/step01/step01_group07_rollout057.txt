    max_iter = 500
    v_max = 0.1
    c1, c2 = 2.0, 2.0

    particles = np.array([np.random.rand(n_points) for _ in range(num_particles)])

        p *= target_sum / np.sum(p)  # Normalize to satisfy constraint
    velocities = np.random.uniform(-v_max, v_max, (num_particles, n_points))

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Initialize personal best and global best
    pbest_pos = particles.copy()
    pbest_obj = np.array([objective(p) for p in particles])
    gbest_idx = np.argmin(pbest_obj)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_val = pbest_obj[gbest_idx]

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        w = 0.9 - ((0.9 - 0.4) * iter / max_iter)  # Decreasing inertia

            soc = c2 * np.random.rand() * (gbest_pos - particles[i])

        # Update personal and global best
