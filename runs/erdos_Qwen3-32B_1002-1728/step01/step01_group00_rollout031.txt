import time

    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    p_list = [2, 4, 6, 8, 10]
    initial_guesses = [h_initial.copy()]
    for p in p_list:
        initial_guesses.append(generate_periodic_pattern(p, n_points))

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0.0, 1.0)
    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)

    best_c5 = None
    best_h = h_initial.copy()
    start_time = time.time()

    for init in initial_guesses:
        if time.time() - start_time > budget_s - 10:

        init = init.copy()
        init = np.clip(init, 0.0, 1.0)
        init_sum = np.sum(init)
        if abs(init_sum - target_sum) > 1e-8:
            init[0] += (target_sum - init_sum)
        init = np.clip(init, 0.0, 1.0)

        result = minimize(
            objective, x0=init, method='SLSQP', bounds=bounds, constraints=sum_constraint,
            options={'maxiter': 200, 'ftol': 1e-6})
        curr_h = result.x.copy() if result.success else init.copy()

        curr_h = np.clip(curr_h, 0, 1)
        curr_sum = np.sum(curr_h)
        curr_h[0] += (target_sum - curr_sum)

        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()

        if time.time() - start_time < budget_s - 20:

            new_init = np.clip(curr_h + perturbation, 0, 1)

            new_init = np.clip(new_init, 0, 1)

            result_perturb = minimize(
                objective, x0=new_init, method='SLSQP', bounds=bounds, constraints=sum_constraint,
                options={'maxiter': 100, 'ftol': 1e-6})
            perturbed_h = result_perturb.x.copy() if result_perturb.success else new_init.copy()
            perturbed_h = np.clip(perturbed_h, 0, 1)

            perturbed_h = np.clip(perturbed_h, 0, 1)

            corr_perturbed = np.correlate(perturbed_h, 1 - perturbed_h, mode='full') * dx
