                alpha *= 0.8
        
        # Decay alpha if stuck
        if iter_count % 500 == 0:
            alpha *= 0.95
    
    # Final constraint enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    if abs(total - n_points / 2) > 1e-8:
        best_h *= (n_points / 2) / total
    
    return best_h, best_c5, n_points
