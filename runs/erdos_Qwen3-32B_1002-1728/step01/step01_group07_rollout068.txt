import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    import time
    
    start_time = time.time()
    
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair(x, sum_target):
        x_clipped = np.clip(x, 0, 1)
        current_sum = np.sum(x_clipped)
        if np.isclose(current_sum, sum_target, atol=1e-6):
            return x_clipped
        delta = sum_target - current_sum
        if delta > 1e-6:
            mask = x_clipped < 1.0
            if np.sum(mask) == 0:
                return x_clipped
            possible_increase = np.sum(1.0 - x_clipped[mask])
            if possible_increase < delta:
                return x_clipped
            add_amount = delta / np.sum(mask)
            x_clipped[mask] += add_amount
            x_clipped = np.clip(x_clipped, 0, 1)
        elif delta < -1e-6:
            mask = x_clipped > 0.0
            if np.sum(mask) == 0:
                return x_clipped
            possible_decrease = np.sum(x_clipped[mask])
            if possible_decrease < abs(delta):
                return x_clipped
            subtract_amount = abs(delta) / np.sum(mask)
            x_clipped[mask] -= subtract_amount
            x_clipped = np.clip(x_clipped, 0, 1)
        return x_clipped
    
    def objective(x_vec):
        h = repair(x_vec, target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        popsize=20,
        strategy='best1bin',
        seed=seed,
        mutation=(0.5, 1),
        recombination=0.7,
        maxiter=50,
        polish=False,
        init='sobol',
        disp=False
    )
    
    optimized_h = repair(result.x, target_sum)
    corr = np.correlate(optimized_h, 1 - optimized_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    assert abs(np.sum(optimized_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((optimized_h >= -1e-6) & (optimized_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (optimized_h.tolist(), best_c5, n_points)
