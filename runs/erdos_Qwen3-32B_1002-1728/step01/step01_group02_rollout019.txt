    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - target_sum}]

    beta_sequence = [1.0, 4.0, 16.0, 64.0, 256.0]

    if hasattr(initial_h_values, '__len__'):
        seeds = [1234, 5678, 9101]
    else:
        seeds = [1234, 5678]

    perturbed_initializations = []
    for seed in seeds:
        np.random.seed(seed)
        perturbed = h_initial + np.random.normal(0.0, 0.05, n_points)
        perturbed = repair_h(perturbed, target_sum, n_points)
        perturbed_initializations.append(perturbed)

    for perturbed in perturbed_initializations:
        current_h = perturbed.copy()
        for beta in beta_sequence:
            if time.time() - start_time > budget_s - 30:
                break
            obj_func = partial(log_sum_exp_objective, beta=beta, n_points=n_points, dx=2.0 / n_points)
            result = minimize(obj_func, current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                current_h = result.x.copy()
                current_h = repair_h(current_h, target_sum, n_points)
                c5_val = evaluate_C5(current_h, n_points, 2.0 / n_points)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = current_h.copy()
            else:
                current_h = repair_h(current_h, target_sum, n_points)
        if time.time() - start_time < budget_s - 20:
            fine_obj = lambda x: evaluate_C5(x, n_points, 2.0 / n_points)
            fine_result = minimize(fine_obj, current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if fine_result.success:
                candidate = repair_h(fine_result.x, target_sum, n_points)
                c5_val = evaluate_C5(candidate, n_points, 2.0 / n_points)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = candidate.copy()
    return best_h, best_c5, n_points
