import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = 1e10  # Initialize with a high value

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def project(h):
        h_proj = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_proj)
        if h_sum < 1e-9:
            h_proj[:] = 0.0
        else:
            h_proj *= (target_sum / h_sum)
        h_proj = np.clip(h_proj, 0.0, 1.0)
        return h_proj

    start_time = time.time()
    best_c5 = compute_c5(best_h)

    # Global search using differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    def obj_de(x):
        x_proj = project(x)
        return compute_c5(x_proj)

    try:
        result_de = differential_evolution(
            obj_de, bounds,
            strategy='best1bin',
            maxiter=30,
            popsize=10,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.8,
            seed=seed,
            workers=1,
            polish=False,
            init='sobol'
        )
        h_de = project(result_de.x)
        c5_de = compute_c5(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    # Local refinement with SLSQP
    def obj_slp(x):
        return compute_c5(project(x))

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slp = Bounds(0.0, 1.0)

    result_slp = minimize(
        obj_slp,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_slp,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-6}
    )
    if result_slp.success:
        h_slp = project(result_slp.x)
        c5_slp = compute_c5(h_slp)
        if c5_slp < best_c5:
            best_h = h_slp
            best_c5 = c5_slp

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    # Perturbation restart
    h_perturbed = best_h + np.random.normal(0.0, 0.05, n_points)
    h_perturbed = project(np.clip(h_perturbed, 0.0, 1.0))
    new_h = h_perturbed

    result_restart = minimize(
        obj_slp,
        x0=new_h,
        method='SLSQP',
        bounds=bounds_slp,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-6}
    )
    if result_restart.success:
        h_restart = project(result_restart.x)
        c5_restart = compute_c5(h_restart)
        if c5_restart < best_c5:

            best_c5 = c5_restart
