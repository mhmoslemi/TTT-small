import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    h = np.array(h)
    current_sum = h.sum()
    if current_sum == 0:
        return h
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = Bounds(lb=0.0, ub=1.0)

    def de_objective(x_unprojected):
        x_projected = project_to_feasible(x_unprojected, target_sum)
        return compute_c5(x_projected, dx)

    start_time = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Global search with differential evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        mutation=(0.5, 1.0),
        recombination=0.8,
        popsize=8,
        maxiter=40,
        polish=False,
        tol=1e-4,
        updating='immediate',
        workers=1
    )
    de_sol = project_to_feasible(de_result.x, target_sum)
    de_val = compute_c5(de_sol, dx)
    if de_val < best_c5:
        best_h = de_sol
        best_c5 = de_val

    # Local refinement with SLSQP
    def objective(x):
        return compute_c5(x, dx)

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 300}
    )
    if result.success:
        sol = result.x
        val = compute_c5(sol, dx)
        if val < best_c5:
            best_h = sol
            best_c5 = val

    # Perturb and refine
    if time.time() - start_time < budget_s:
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h = np.clip(project_to_feasible(best_h + perturbation, target_sum), 0, 1)
        result = minimize(
            objective,
            x0=new_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        if result.success:
            sol = result.x
            val = compute_c5(sol, dx)
            if val < best_c5:
                best_h = sol
                best_c5 = val

    return (best_h, best_c5, n_points)
