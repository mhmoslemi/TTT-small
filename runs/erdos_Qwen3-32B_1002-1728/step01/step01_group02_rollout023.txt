                return max_int + (1/beta) * np.log(exp_terms.sum())

            # Constraints and bounds
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum})
            bounds = [(0, 1) for _ in range(n_points)]

            # Choose optimization method based on beta size
            method = 'SLSQP' if beta >= 16 else 'BFGS'

            # Run optimization
            result = minimize(objective_smoothed, current_h, method=method, 
                             bounds=bounds, constraints=cons)

            if result.success:
                optimized = result.x.reshape(n_points)
                # Enforce constraints
                optimized = np.clip(optimized, 0, 1)
                optimized_sum = np.sum(optimized)
                if abs(optimized_sum - required_sum) > 1e-8:
                    optimized = np.clip(optimized * (required_sum / optimized_sum), 0, 1)

                # Update best solution
                current_c5 = evaluate_C5(optimized)
                if current_c5 < best_c5:
                    best_c5, best_h = current_c5, optimized.copy()
                    current_h = optimized.copy()
            else:
                # Fallback to repaired current solution
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (required_sum / np.sum(current_h))

    # Fine optimization with exact max objective if time remains
    if time.time() - start_time < budget_s - 5:
        cons_fine = ({'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum})
        result_fine = minimize(evaluate_C5, best_h, method='SLSQP', bounds=bounds, constraints=cons_fine)
        if result_fine.success:
            optimized = result_fine.x.reshape(n_points)
            optimized = np.clip(optimized, 0, 1)
            optimized_sum = np.sum(optimized)
            if abs(optimized_sum - required_sum) > 1e-8:
                optimized = np.clip(optimized * (required_sum / optimized_sum), 0, 1)
            current_c5 = evaluate_C5(optimized)
            if current_c5 < best_c5:
                best_c5, best_h = current_c5, optimized

    # Final constraint cleaning
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - required_sum) > 1e-8:
        best_h = np.clip(best_h * (required_sum / best_h_sum), 0, 1)

    return best_h, best_c5, n_points
