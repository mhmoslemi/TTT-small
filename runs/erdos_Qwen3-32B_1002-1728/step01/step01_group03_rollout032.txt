import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, noise=0.2):
        h_pert = h.copy()
        n_perturb = max(1, int(0.2 * n_points))  # Perturb 20% of points
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        delta = np.random.uniform(-noise, noise, size=n_perturb)
        h_pert[indices] += delta
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert *= target_sum / current_sum
        h_pert = np.clip(h_pert, 0, 1)  # Final clip
        return h_pert
    
    max_tries = 30
    tries = 0
    noise_level = 0.2
    last_improvement_time = start_time
    improvement_window = 60  # seconds
    
    while time.time() - start_time < budget_s - 30 and tries < max_tries:
        tries += 1
        x0 = get_perturbed(best_h, noise=noise_level)
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-6}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val
                last_improvement_time = time.time()
                noise_level = max(0.1, noise_level * 0.9)  # Reset noise slightly
            else:
                if time.time() - last_improvement_time > improvement_window:
                    noise_level = max(0.1, noise_level * 0.8)
        else:
            # If optimization fails, perturb more strongly
            noise_level = min(0.5, noise_level * 1.2)
    
    return (best_h, best_c5, n_points)
