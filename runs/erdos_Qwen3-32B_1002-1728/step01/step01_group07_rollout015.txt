    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        # Local refinement with SLSQP
        def objective_slspq(h_vec):
            h_clipped = get_h(h_vec)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx

        bounds_slspq = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(get_h(h)) - target_sum})
        result_slspq = minimize(
            fun=objective_slspq,
            x0=h_repaired,
            method='SLSQP',
            bounds=bounds_slspq,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result_slspq.success:
            h_best = result_slspq.x
        else:
            h_best = h_repaired
    else:
        h_best = h_repaired

    h_final = get_h(h_best)
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    best_c5 = np.max(corr) * dx

    # Validate constraints
    assert np.allclose(np.sum(h_final), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_final.tolist(), best_c5, n_points)
