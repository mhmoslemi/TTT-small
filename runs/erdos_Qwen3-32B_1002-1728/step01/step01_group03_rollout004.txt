import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective_de(h_flat):
        h = h_flat.reshape(n_points)
        h_clamped = np.clip(h, 0, 1)
        corr = np.correlate(h_clamped, 1 - h_clamped, mode='full') * dx
        return np.max(corr)
    
    def objective_constrained(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def repair_to_feasible(h_flat):
        h = h_flat.reshape(n_points)
        h_clamped = np.clip(h, 0, 1)
        s = np.sum(h_clamped)
        if s == 0:
            return np.full(n_points, target_sum / n_points)
        scale = target_sum / s
        h_scaled = h_clamped * scale
        h_repaired = np.clip(h_scaled, 0, 1)
        return h_repaired
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    # Initial population: perturb initial_h_values
    pop_size = 20
    pop_init = [h_init + np.random.uniform(-0.1, 0.1, n_points) for _ in range(pop_size)]
    pop_init = [np.clip(p, 0, 1).flatten() for p in pop_init]
    
    # Run differential evolution to find good initial guess
    result_de = differential_evolution(
        objective_de, bounds, 
        strategy='best1exp',
        init=pop_init,
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-5,
        maxiter=100,
        seed=seed,
        polish=False,
        workers=1
    )
    
    best_h_de = result_de.x
    best_c5_de = result_de.fun
    
    # Refine with constrained optimization
    best_h = best_h_de.copy()
    best_c5 = np.inf
    
    if time.time() - start_time < budget_s - 60:
        x0_feasible = repair_to_feasible(best_h_de).copy()
        res = minimize(
            objective_constrained, 
            x0=x0_feasible, 
            method='SLSQP',
            bounds=bounds,

            if curr_c5 < best_c5:
                best_h = res.x
                best_c5 = curr_c5
    
    # Final repair to ensure feasibility
    h_opt = repair_to_feasible(best_h)
    final_c5 = objective_constrained(h_opt)
    
    return (h_opt, final_c5, n_points)
