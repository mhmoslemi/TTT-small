    # Random solutions (50%)
    for i in range(pop_size // 2, pop_size):
        vec = np.random.rand(n_points)
        sum_vec = np.sum(vec)
        vec = vec * (target_sum / sum_vec) if sum_vec != 0 else vec
        vec = np.clip(vec, 0, 1)
        pop[i] = vec

    def objective_func(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Initialize best solution
    objectives = np.array([objective_func(h) for h in pop])
    best_idx = np.argmin(objectives)
    best_vec = pop[best_idx].copy()
    best_c5 = objectives[best_idx]
    no_improve = 0
    MAX_NO_IMPROVE = 200

    while time.time() - start_time < budget_s - 10:
        for i in range(pop_size):
            # Select indices for mutation
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            # Mutate
            mutant = a + 0.8 * (b - c)
            mutant = np.clip(mutant, 0, 1)
            # Crossover
            cross_mask = np.random.rand(n_points) < 0.9
            cross_mask[np.random.randint(n_points)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            # Project to feasible set
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            if sum_trial != 0:
                trial = trial * (target_sum / sum_trial)
            trial = np.clip(trial, 0, 1)
            
            # Evaluate and replace
            trial_obj = objective_func(trial)
            if trial_obj < objective_func(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_vec = trial.copy()
                    best_c5 = trial_obj
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Perturb best solution to escape local optima
        if no_improve > MAX_NO_IMPROVE:
            perturbation = np.random.normal(0, 0.1, n_points)
            perturbed = best_vec + perturbation
            perturbed = np.clip(perturbed, 0, 1)
            sum_p = np.sum(perturbed)
            if sum_p != 0:
                perturbed = perturbed * (target_sum / sum_p)
            perturbed = np.clip(perturbed, 0, 1)
            pop[np.random.randint(0, pop_size)] = perturbed
            no_improve = 0

        if time.time() - start_time > budget_s - 10:
            break

    return best_vec, best_c5, n_points
