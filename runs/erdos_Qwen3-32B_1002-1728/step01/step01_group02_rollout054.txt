import time
from scipy.optimize import minimize

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points).astype(np.float64)

    corr = np.correlate(h, ones_minus_h, mode='full')
    max_corr = np.max(corr * dx)
    return max_corr

def repair_feasibility(h_vec, n_points):
    h = h_vec.reshape(n_points)

    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-8:
        scale = (n_points/2) / h_sum
        h = np.clip(h * scale, 0.0, 1.0)
    return h
