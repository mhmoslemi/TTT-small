            idx_end = int(np.ceil(x_end * original_n / 2))
            indices = np.arange(idx_start, min(idx_end, original_n))
            avg = np.mean(h_init[indices]) if len(indices) > 0 else 0.5
            new_h[i] = avg
        scaling = (new_n / 2.0) / np.sum(new_h)
        return np.clip(new_h * scaling, 0, 1)

    def optimize_grid(h_start, time_budget):
        n_points = len(h_start)
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds = [(0.0, 1.0)] * n_points
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        best_h_grid = h_start.copy()
        best_c5_grid = objective(best_h_grid)
        start_opt = time.time()
        tries = 0
        while tries < 3 and time.time() - start_opt < time_budget:
            perturbed = h_start + np.random.uniform(-0.1, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed *= target_sum / np.sum(perturbed)
            res = minimize(
                objective, x0=perturbed, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-4}
            )
            if res.success:
                val = objective(res.x)
                if val < best_c5_grid:
                    best_h_grid = res.x
                    best_c5_grid = val
            tries += 1
        return best_h_grid, best_c5_grid

    for res in res_list:
        if res < n_initial:
            h_start = downsample_h(h_init, res)
        elif res == n_initial:
            h_start = best_h.copy()
        else:
            h_start = upsample_h(best_h, res)

        time_left = budget_s - (time.time() - start_time)
        if time_left < 60:
            break
        grid_budget = min(200, time_left // (len(res_list) - len(h_start.shape)))
        optimized_h, optimized_c5 = optimize_grid(h_start, grid_budget)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_objective(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
