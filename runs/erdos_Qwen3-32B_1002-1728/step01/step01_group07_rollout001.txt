import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_best = h_init.copy()
    corr_best = np.correlate(h_best, 1 - h_best, mode='full') * dx
    best_c5 = np.max(corr_best)

    rng = np.random.RandomState(seed)

    # Precompute initial validation
    assert abs(np.sum(h_best) - target_sum) < 1e-6, "Initial state violates constraints"

    # Iterate while time remains
    while time.time() - start_time < budget_s - 30:  # 30s buffer
        # Perturb current best
        perturbed = h_best + rng.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)

        total = np.sum(perturbed)
        if total < 1e-8:
            perturbed.fill(target_sum / n_points)
        else:
            perturbed = (perturbed / total) * target_sum

        # Run optimization
        try:
            result = minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 120, 'ftol': 1e-8}
            )
        except Exception:
            continue

        if result.success:
            h_new = result.x.reshape(n_points)
            corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
            c5_new = np.max(corr)

            # Check constraints
            valid = True
            valid &= abs(np.sum(h_new) - target_sum) < 1e-6
            valid &= np.allclose(h_new, np.clip(h_new, 0, 1), atol=1e-6)

            if valid and c5_new < best_c5:
                h_best[:] = h_new  # in-place update to save memory
                best_c5 = c5_new

        # Respect time budget strictly (last iteration may take longer)
        if time.time() - start_time > budget_s - 10:
            break

    # Final constraint checks
    assert abs(np.sum(h_best) - target_sum) < 1e-6
    assert np.all((h_best >= -1e-6) & (h_best <= 1.0 + 1e-6))

    return (h_best.tolist(), best_c5, n_points)
