import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def get_perturbed(h, noise):
        h_pert = h.copy()
        # Add random perturbation
        delta = np.random.uniform(-noise, noise, n_points)
        h_pert += delta
        h_pert = np.clip(h_pert, 0, 1)
        # Normalize to maintain the sum
        current_sum = np.sum(h_pert)
        if current_sum != 0:  # prevent division by zero
            h_pert *= target_sum / current_sum
        return h_pert
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    max_tries = 100  # More tries
    tries = 0
    noise_level = 0.15
    max_no_improvement = 5  # threshold before increasing noise
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 60 and tries < max_tries:
        tries += 1
        # Generate perturbed starting point
        x0 = get_perturbed(best_h, noise_level)
        # Run optimization
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val
                no_improvement_count = 0  # reset counter on improvement
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Adapt noise level if no improvement for a while
        if no_improvement_count >= max_no_improvement:
            # Increase noise but cap it
            noise_level = min(noise_level * 1.5, 0.5)
            no_improvement_count = 0
    
    return (best_h, best_c5, n_points)
