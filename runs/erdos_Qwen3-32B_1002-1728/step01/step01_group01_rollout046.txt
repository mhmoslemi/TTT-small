import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def compute_c5(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        s = h_clipped.sum()
        if s == 0:
            return h_clipped
        return h_clipped * (target_sum / s)

    np.random.seed(seed)
    pop_size = 50
    pop = []
    for _ in range(pop_size // 2):
        pert = np.random.normal(0, 0.05, n_points)
        ind = np.clip(initial_h_values + pert, 0, 1)
        pop.append(project(ind))
    for _ in range(pop_size // 2):
        ind = np.random.uniform(0, 1, n_points)
        pop.append(project(ind))
    pop = np.array(pop)

    fit = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fit)
    best_h = pop[best_idx].copy()
    best_c5 = fit[best_idx]

    F = 0.8
    CR = 0.9

    while time.time() - start_time < budget_s - 10:
        new_pop = []
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            target = pop[i]
            cross_mask = np.random.rand(n_points) < CR
            trial = np.where(cross_mask, mutant, target)
            trial = project(trial)
            trial_c5 = compute_c5(trial)
            if trial_c5 < fit[i]:
                new_pop.append(trial)
                if trial_c5 < best_c5:
                    best_h[:] = trial
                    best_c5 = trial_c5
            else:
                new_pop.append(pop[i])
        pop = np.array(new_pop)
        fit = np.array([compute_c5(ind) for ind in pop])
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_h[:] = pop[best_idx]
            best_c5 = fit[best_idx]
        if time.time() - start_time > budget_s - 5:
            break

    return best_h, best_c5, n_points
