        for beta in betas:
            if time.time() - start_time > budget_s - 20:
                break
            def smooth_obj(x):
                h = x.reshape(n_points)
                ones_h = 1.0 - h
                cross = np.correlate(h, ones_h, mode='full')
                Ck = cross * dx
                return (1 / beta) * np.log(np.sum(np.exp(beta * Ck)))
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
            bounds = [(0, 1)] * n_points
            result = minimize(smooth_obj, current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                updated_h = result.x.copy()
                updated_h = np.clip(updated_h, 0.0, 1.0)
                sum_h = np.sum(updated_h)
                if not np.isclose(sum_h, n_points / 2):
                    factor = (n_points / 2) / sum_h
                    updated_h *= factor
                    updated_h = np.clip(updated_h, 0.0, 1.0)
                current_h = updated_h.copy()
                c5_val = np.max(np.correlate(current_h, 1 - current_h, mode='full') * dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = current_h.copy()

        if time.time() - start_time < budget_s - 20:
            def exact_obj(x):
                return np.max(np.correlate(x.reshape(n_points), (1 - x.reshape(n_points)), mode='full') * dx)
            result = minimize(exact_obj, current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                updated_h = result.x.copy()
                updated_h = np.clip(updated_h, 0.0, 1.0)
                sum_h = np.sum(updated_h)
                if not np.isclose(sum_h, n_points / 2):
                    factor = (n_points / 2) / sum_h
                    updated_h *= factor
                    updated_h = np.clip(updated_h, 0.0, 1.0)
                current_h = updated_h.copy()
                c5_val = np.max(np.correlate(current_h, 1 - current_h, mode='full') * dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = current_h.copy()

    return best_h, best_c5, n_points
