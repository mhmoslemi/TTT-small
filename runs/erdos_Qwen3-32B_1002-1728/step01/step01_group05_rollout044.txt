                best_c5 = c5_de
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Local refinement phase
    perturb_amp = 0.15
    perturb_decay = 0.92
    trials = 0
    max_trials = 60
    
    while remaining > 0 and trials < max_trials:
        if trials > 0 and trials % 8 == 0:
            perturbation = np.random.uniform(-perturb_amp, perturb_amp, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
            current_sum = np.sum(perturbed_h)
            if abs(current_sum) < 1e-9:
                perturbed_h = np.full_like(perturbed_h, target_sum / n_points)
            else:
                perturbed_h = perturbed_h * (target_sum / current_sum)
                perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            x0 = perturbed_h
        else:
            x0 = best_h
            
        def objective(x):
            h_proj = project_h(x, target_sum, n_points)
            return compute_c5(h_proj, dx)
        
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds_slsqp = Bounds(lb=0.0, ub=1.0)
        
        start_opt = time.time()
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        elapsed_opt = time.time() - start_opt
        remaining -= elapsed_opt
        
        if result.success:
            new_h = project_h(result.x, target_sum, n_points)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        trials += 1
        perturb_amp *= perturb_decay
    
    return (best_h, best_c5, n_points)
