import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        h *= (n / 2) / h.sum()
        h = np.clip(h, 0, 1)
    else:
        n = 280
        h = np.zeros(n)
        mid = n // 2
        h[mid-5:mid+5] = 0.5
        h *= (n / 2) / h.sum()

    def compute_c5(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, 'full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    sigma = 0.1
    lambda_ = 10
    adaptation_interval = 50
    patience = 0
    max_generations = 10000

    for generation in range(max_generations):
        if time.time() - start_time > max_time * 0.99:
            break

        candidates = []
        c5_values = []
        for _ in range(lambda_):
            new_h = best_h + sigma * np.random.randn(n)
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h)
            if current_sum == 0:
                new_h = best_h.copy()
            else:
                scaling = (n / 2) / current_sum
                new_h *= scaling
                new_h = np.clip(new_h, 0, 1)
            candidates.append(new_h)
            c5_val = compute_c5(new_h, n)
            c5_values.append(c5_val)

        current_best_index = np.argmin(c5_values)
        if c5_values[current_best_index] < best_c5:
            best_h = candidates[current_best_index].copy()
            best_c5 = c5_values[current_best_index]
            patience = 0
        else:
            patience += 1

        if patience >= adaptation_interval:
            sigma *= 0.5
            patience = 0

        if sigma < 1e-8:
            break

    return best_h, best_c5, n
