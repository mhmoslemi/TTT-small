import numpy as np
import time
from scipy.optimize import Bounds, differential_evolution, minimize
from functools import partial

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project_h(h, target_sum_val):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = h_clipped.sum()
        if h_sum < 1e-12:
            return h_clipped
        scaling = target_sum_val / h_sum
        return np.clip(h_clipped * scaling, 0.0, 1.0)

    def compute_c5(h, dx_val):
        corr = np.correlate(h, 1 - h, mode='full') * dx_val
        return np.max(corr)

    objective = lambda h: compute_c5(project_h(h, target_sum), dx)

    start_time = time.time()
    best_h = project_h(h_init, target_sum).copy()
    best_c5 = compute_c5(best_h, dx)

    # Global search with differential evolution
    if time.time() - start_time < budget_s:
        bounds = [(0.0, 1.0)] * n_points
        x0 = project_h(h_init + 0.05 * np.random.randn(n_points), target_sum)
        try:
            de_result = differential_evolution(
                func=objective,
                bounds=bounds,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 0.8),
                recombination=0.9,
                seed=seed + 1,
                maxiter=20,
                tol=1e-5,
                workers=1,
                disp=False,
                callback=None
            )
            if de_result.success and de_result.fun < best_c5:
                best_h = project_h(de_result.x, target_sum)
                best_c5 = compute_c5(best_h, dx)
        except:
            pass

    # Local refinement with SLSQP
    if time.time() - start_time < budget_s:
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(
            lambda x: compute_c5(x, dx),
            x0=best_h,
            method='SLSQP',
            bounds=Bounds(0.0, 1.0),
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-6}
        )
        if result.success:
            crsl = project_h(result.x, target_sum)
            curr_c5 = compute_c5(crsl, dx)
            if curr_c5 < best_c5:
                best_h, best_c5 = crsl, curr_c5

    # Perturbation restarts
    restarts = 0
    while time.time() - start_time < budget_s and restarts < 5:
        perturb = 0.2 + 0.3 * np.random.rand()
        new_h = project_h(best_h + perturb * np.random.randn(n_points), target_sum)
        result2 = minimize(
            lambda x: compute_c5(project_h(x, target_sum), dx),
            x0=new_h,
            method='SLSQP',
            bounds=Bounds(0.0, 1.0),
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        if result2.success:
            crsl_c5 = compute_c5(result2.x, dx)
            if crsl_c5 < best_c5:
                best_h, best_c5 = result2.x, crsl_c5
        restarts += 1

    best_h = project_h(best_h, target_sum)
    return (best_h, best_c5, n_points)
