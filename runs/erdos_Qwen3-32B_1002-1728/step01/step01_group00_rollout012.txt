        elites = [c[0] for c in candidates[:num_elites]]
        num_to_generate = population_size - num_elites
        parents = []
        for _ in range(num_to_generate * 2):
            parent = tournament_selection(candidates, tournament_size)
            parents.append(parent)

        new_pop = elites.copy()
        for i in range(0, len(parents) - 1, 2):
            p1 = parents[i]
            p2 = parents[i + 1]
            mask = np.random.rand(n_points) > 0.5
            c1 = np.where(mask, p1, p2)
            c2 = np.where(mask, p2, p1)
            c1 = fix_sum_constraint(c1, n_points)
            c2 = fix_sum_constraint(c2, n_points)
            c1 = c1 + np.random.normal(0, mutation_strength, n_points) * (np.random.rand(n_points) < 0.1)
            c1 = fix_sum_constraint(np.clip(c1, 0, 1), n_points)
            c2 = c2 + np.random.normal(0, mutation_strength, n_points) * (np.random.rand(n_points) < 0.1)
            c2 = fix_sum_constraint(np.clip(c2, 0, 1), n_points)
            new_pop.append(c1)
            new_pop.append(c2)
            if len(new_pop) >= population_size:
                break
        if len(new_pop) > population_size:
            new_pop = new_pop[:population_size]
        population = new_pop

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
