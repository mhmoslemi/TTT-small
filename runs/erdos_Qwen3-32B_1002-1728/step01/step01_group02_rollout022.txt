        initializations.append(perturbed)

    best_c5 = float('inf')
    best_h = None

    beta_sequence = [2**i for i in range(11)]

    for init_h in initializations:
        h_current = init_h.copy()
        for beta in beta_sequence:
            if time.time() - start_time >= budget_s - 10:
                break

            def objective(h_vec):
                h = h_vec.reshape(n_points)
                ones_h = 1.0 - h
                cross = np.correlate(h, ones_h, mode='full')
                overlaps = cross * dx
                exp_vals = np.exp(beta * (overlaps))
                sum_exp = exp_vals.sum()
                return (1.0 / beta) * np.log(sum_exp)

            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]

            result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                optimized_h = result.x.reshape(n_points)
                optimized_h = np.clip(optimized_h, 0.0, 1.0)
                total = optimized_h.sum()
                if not np.isclose(total, n_points / 2.0, atol=1e-8):
                    optimized_h = (optimized_h * (n_points / 2.0)) / total
                    optimized_h = np.clip(optimized_h, 0.0, 1.0)
                h_current = optimized_h

                ones_h = 1.0 - optimized_h
                cross = np.correlate(optimized_h, ones_h, mode='full')
                current_c5 = np.max(cross * dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h.copy()

            if time.time() - start_time >= budget_s - 10:
                break

        if time.time() - start_time >= budget_s - 10:
            break

    # Final refinement step
    if best_h is not None:
        def objective_exact(h_vec):
            h = h_vec.reshape(n_points)
            ones_h = 1.0 - h
            cross = np.correlate(h, ones_h, mode='full')
            return np.max(cross * dx)

        cons_exact = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}]
        bounds_exact = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(objective_exact, best_h, method='SLSQP', bounds=bounds_exact, constraints=cons_exact)
        if result.success:
            h_final = result.x.reshape(n_points)
            h_final = np.clip(h_final, 0.0, 1.0)
            total = h_final.sum()
            if not np.isclose(total, n_points / 2.0, atol=1e-8):
                h_final = (h_final * (n_points / 2.0)) / total
                h_final = np.clip(h_final, 0.0, 1.0)
            ones_h = 1.0 - h_final
            cross = np.correlate(h_final, ones_h, mode='full')
            final_c5 = np.max(cross * dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final

    # Ensure constraint compliance before return
    if best_h is None:
        best_h = h_initial.copy()
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = (best_h * (n_points / 2.0)) / best_h.sum()
        best_h = np.clip(best_h, 0.0, 1.0)
    
    ones_h = 1.0 - best_h
    cross = np.correlate(best_h, ones_h, mode='full')
    final_bound = np.max(cross * dx)
    
    return best_h, final_bound, n_points
