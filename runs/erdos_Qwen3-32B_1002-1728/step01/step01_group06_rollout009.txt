import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Configuration parameters
    n_points = 280
    pop_size = 50
    max_iter = 500
    pa = 0.25  # Probability of discovering alien egg

    # Initialize using initial_h_values or default
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
        old_n = len(h_init)
        if old_n != n_points:
            old_x = np.linspace(0, 2, old_n)
            new_x = np.linspace(0, 2, n_points)
            h_init = np.interp(new_x, old_x, h_init)
    else:
        # Simple default: a central pulse
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[mid-10:mid+10] = 0.5
    
    # Normalize to constraint ∑h = n_points / 2
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Feasibility repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / total
        return h

    # Objective function
    def evaluate(h):
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    # Create initial population with perturbations
    nests = []
    for _ in range(pop_size):
        perturbation = np.random.normal(0.0, 0.1, n_points)
        new_h = repair(h_init + perturbation)
        nests.append(new_h)
    
    # Evaluate initial population
    fitness = np.array([evaluate(n) for n in nests])
    best_idx = np.argmin(fitness)
    best_h = nests[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Cuckoo Search main loop
    for iter_num in range(max_iter):
        # Check time budget
        if time.time() - start_time > max_time:
            break
            
        # Generate new solutions via Levy flights
        beta = 1.5
        step_size = 0.1 * (1.0 - iter_num / max_iter)  # Linear decay
        
        for i in range(pop_size):
            # Choose a random nest
            chosen_nest = nests[i].copy()
            
            # Generate Levy flight step (simplified)
            sigma_u = (math.gamma(1+beta) * math.sin(math.pi*beta/2) / 
                      (math.gamma((1+beta)/2) * 2**((beta-1)/2) * 
                       math.sqrt(math.pi)))**(1/beta)
            u = np.random.normal(0, sigma_u**2, n_points)
            v = np.random.normal(0, 1, n_points)
            step = u / np.abs(v)**(1/beta)
            
            # Update solution using Levy flight step
            new_nest = chosen_nest + step_size * step
            new_nest = repair(new_nest)
            
            # Evaluate new solution
            new_c5 = evaluate(new_nest)
            
            # Replacement with better solution
            target_idx = np.random.randint(pop_size)
            if new_c5 < fitness[target_idx]:
                nests[target_idx] = new_nest
                fitness[target_idx] = new_c5
                
                # Update global best
                if new_c5 < best_c5:
                    best_h = new_nest.copy()
                    best_c5 = new_c5
        
        # Remove worst nests and add new random solutions

            if fitness[idx] < best_c5:
                best_h = new_h.copy()
                best_c5 = fitness[idx]
    
    return best_h, best_c5, n_points
