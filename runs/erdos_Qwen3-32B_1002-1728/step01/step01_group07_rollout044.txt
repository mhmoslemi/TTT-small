    target_sum = n_points / 2.0

    # Objective function
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: integral of h(x) must equal 1
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Bounds for h(x) in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to perturb the initial guess
    def perturb_h(h_current, seed):
        rs = np.random.RandomState(seed)
        perturbed = h_current + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Normalize to satisfy the integral constraint
        perturbed = (perturbed / np.sum(perturbed)) * target_sum
        return perturbed
    
    best_h = h_init.copy()
    best_c5 = objective(h_init)
    
    # Multi-start optimization with perturbations
    current_seed = seed
    while time.time() - start_time < budget_s * 0.95:
        current_seed += 1
        perturbed = perturb_h(best_h, current_seed)
        
        # Run optimization
        result = minimize(
            fun=objective,
            x0=perturbed.ravel(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            new_h = result.x.reshape(n_points)
            # Correct for numerical constraint violations
            new_sum = np.sum(new_h)
            new_h = np.clip(new_h * (target_sum / new_sum), 0, 1)
            # Evaluate
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = current_c5
    
    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
