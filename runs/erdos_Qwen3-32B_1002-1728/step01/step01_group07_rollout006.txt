import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Define optimization variables
    h = cp.Variable(n_points)
    C5_candidate = cp.Variable()

    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]

    # Add overlap constraints for all possible shifts
    for m in range(-(n_points - 1), n_points):
        valid_i = []
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:
                valid_i.append(i)
        if valid_i:
            overlap = 0
            for i in valid_i:
                j = i + m
                overlap += h[i] * (1 - h[j])
            constraints.append(overlap <= C5_candidate / dx)

    # Objective
    objective = cp.Minimize(C5_candidate)

    # Solve using Gurobi with non-convex handling
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(
            solver=cp.GUROBI,
            GurobiParams={'NonConvex': 2, 'TimeLimit': budget_s - 10},
            verbose=False
        )
    except Exception:
        # Fallback to SCS if Gurobi unavailable or fails
        problem.solve(solver=cp.SCS, verbose=False)

    # Post-process result
    optimized_h = np.clip(h.value, 0, 1) if h.value is not None else h_init.copy()
    optimized_h = (optimized_h / optimized_h.sum()) * target_sum

    # Compute final C5 bound
    cross_corr = np.correlate(optimized_h, 1 - optimized_h, mode='full') * dx
    c5_bound = np.max(cross_corr)

    return (optimized_h.tolist(), c5_bound, n_points)
