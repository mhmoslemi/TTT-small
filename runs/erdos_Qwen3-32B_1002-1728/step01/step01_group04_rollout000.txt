import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    # PSO parameters
    n_particles = 20
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_velocity = 0.1

    # Initialize swarm
    particles = [h.copy()]
    velocities = [np.zeros_like(h)]
    pbest = [h.copy()]
    pbest_score = [compute_c5(h)]
    for _ in range(n_particles - 1):
        # Initialize random particles maintaining constraints
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (target_sum / h_rand.sum())
        h_clipped = np.clip(h_rand, 0, 1)
        s = h_clipped.sum()
        if s != 0:
            h_clipped = h_clipped / s * target_sum
        h_final = np.clip(h_clipped, 0, 1)
        particles.append(h_final)
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(h_final.copy())
        pbest_score.append(compute_c5(h_final))

    gbest = pbest[np.argmin(pbest_score)].copy()
    gbest_score = min(pbest_score)

    while time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            # Velocity update
            r1, r2 = np.random.rand(2)
            vel_cog = cognitive * r1 * (pbest[i] - particles[i])
            vel_soc = social * r2 * (gbest - particles[i])
            velocities[i] = inertia * velocities[i] + vel_cog + vel_soc
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            # Position update with constraints
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = particles[i].sum()
            if s != 0:

                pbest_score[i] = score

                    gbest_score = score
