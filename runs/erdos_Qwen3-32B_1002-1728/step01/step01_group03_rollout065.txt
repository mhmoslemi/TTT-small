import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def constraint(h_flat):
        return np.sum(h_flat) - target_sum
    
    def penalty_objective(h_flat):
        h = h_flat.reshape(n_points)
        pen = 1e4 * abs(np.sum(h) - target_sum)
        return objective(h_flat) + pen
    
    bounds = [(0.0, 1.0)] * n_points
    current_sum = np.sum(h_init)
    h_init_adj = h_init * (target_sum / current_sum)
    h_init_adj = np.clip(h_init_adj, 0, 1)
    
    best_h = h_init_adj.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    try:
        # Differential evolution with penalty for constraint
        res_de = differential_evolution(penalty_objective, bounds,
                                        x0=best_h, 
                                        strategy='best1bin',
                                        maxiter=30, 
                                        popsize=10,
                                        mutation=(0.5, 1.0),
                                        recombination=0.7,
                                        tol=1e-5,
                                        seed=seed,
                                        workers=1)
        
        if res_de.success:
            h_de = res_de.x.reshape(n_points)
            # Enforce constraint exactly and bounds
            scale_factor = target_sum / np.sum(h_de)
            h_de = np.clip(h_de * scale_factor, 0, 1)
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except:
        pass
    
    remaining_time = max(0, budget_s - (time.time() - start_time) - 60)
    if remaining_time > 60:
        cons = [{'type': 'eq', 'fun': constraint}]
        # Perturb best solution
        for _ in range(3):
            h_pert = best_h * 0.9 + np.random.uniform(-0.05, 0.05, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            current_sum = np.sum(h_pert)

            res = minimize(objective, x0=h_pert,

            if res.success:
