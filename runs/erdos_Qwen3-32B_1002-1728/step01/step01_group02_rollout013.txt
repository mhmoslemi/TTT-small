import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h_initial[:] = 0.5

    best_c5 = evaluate_C5(h_initial, n_points, dx)
    best_h = h_initial.copy()

    def check_feasibility(h):

        current_sum = np.sum(h_clipped)

        if abs(current_sum - target_sum) > 1e-6:
            h_clipped = np.clip(h_clipped * target_sum / current_sum, 0.0, 1.0)
        return h_clipped
