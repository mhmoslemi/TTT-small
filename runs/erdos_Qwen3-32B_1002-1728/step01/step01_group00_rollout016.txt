    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Generate diverse initial guesses using structured patterns
    initial_patterns = [h_initial.copy()]
    for m in [3, 5]:
        if n_points % m != 0:
            continue
        groups = n_points // m
        if m == 3:
            # Pattern with two high values, one zero per group
            total_sum_per_group = 22.5 / groups
            a = total_sum_per_group / 2
            pattern = np.zeros(n_points)
            for i in range(groups):
                pattern[i*m:i*m+2] = a
        elif m == 5:
            # Pattern with three high values, two zeros per group
            total_sum_per_group = 22.5 / groups
            a = total_sum_per_group / 3
            pattern = np.zeros(n_points)
            for i in range(groups):
                pattern[i*m:i*m+3] = a
        initial_patterns.append(pattern.copy())

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    best_c5 = None
    best_h = h_initial.copy()

    start_time = time.time()
    np.random.seed(seed)

    for pattern in initial_patterns:
        # Generate multiple perturbed versions of each pattern
        n_perturbations = 3
        for _ in range(n_perturbations):
            # Perturb pattern with zero-mean noise
            x0 = pattern.copy()
            noise = np.random.uniform(-0.1, 0.1, size=n_points)
            noise -= np.mean(noise)
            x0 += noise
            x0 = np.clip(x0, 0.0, 1.0)
            # Ensure the perturbed guess satisfies sum constraint
            x0 -= (np.sum(x0) - n_points/2) / n_points
            x0 = np.clip(x0, 0.0, 1.0)

            if time.time() - start_time > budget_s - 1:
                break

            # Run local optimization
            result = minimize(
                objective,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=sum_constraint,
                options={'maxiter': 120, 'ftol': 1e-8}
            )

            curr_h = result.x.copy()
            # Post-processing: project to feasible region
            curr_h = np.clip(curr_h, 0.0, 1.0)
            offset = (n_points/2 - np.sum(curr_h)) / n_points
            curr_h += offset
            curr_h = np.clip(curr_h, 0.0, 1.0)

            corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            curr_c5 = np.max(corr_values)

            if (best_c5 is None) or (curr_c5 < best_c5):
                best_c5 = curr_c5
                best_h = curr_h.copy()

            # Perturb from best solution
            x0 = best_h.copy()
            x0 += np.random.normal(0, 0.02, n_points)
            x0 = np.clip(x0, 0.0, 1.0)
            x0 -= (np.sum(x0) - n_points/2) / n_points
            x0 = np.clip(x0, 0.0, 1.0)

    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)

    return best_h, c5_bound, n_points
