def project_h(h, target_sum):

    sum_clipped = h_clipped.sum()
    if np.allclose(sum_clipped, target_sum):
        return h_clipped
    # Compute the delta

    # Determine which indices are not at bounds
    not_at_upper = h_clipped < 1.
    not_at_lower = h_clipped > 0.

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project_h(h, target_sum, n_points):
        h = np.clip(h, 0, 1)
        sum_clipped = h.sum()
        if np.allclose(sum_clipped, target_sum):
            return h.copy()
        delta = target_sum - sum_clipped
        not_at_upper = h < 1.0 - 1e-8
        not_at_lower = h > 1e-8
        if delta > 0:  # Need to increase
            num_increase = np.sum(not_at_upper)
            if num_increase == 0:
                return h.copy()

            h += not_at_upper * increment
            h = np.clip(h, 0, 1)
        else:
            delta = -delta
            num_decrease = np.sum(not_at_lower)
            if num_decrease == 0:
                return h.copy()

            h -= not_at_lower * decrement
            h = np.clip(h, 0, 1)
