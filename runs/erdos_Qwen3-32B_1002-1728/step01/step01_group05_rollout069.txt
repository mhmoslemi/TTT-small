import numpy as np
import time
from scipy.optimize import minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)
    
    # Generate diverse initial candidates
    candidates = [h0]
    periods = [3, 4, 5, 6, 7, 8, 9, 10]
    for p in periods:

        x = np.arange(n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * p * x / n_points))

        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        current_sum = h.sum()

            scale = 40.0 / current_sum
            h = h * scale

        candidates.append(h)
