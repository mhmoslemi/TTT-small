    maxiter = int(budget_s * 0.6 / (0.01 * n_points ** 1.2))
    popsize = 20.0 / n_points if n_points else 1.0
    
    try:
        de_result = differential_evolution(
            func=objective_de,
            bounds=bounds_de,
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            seed=seed,
            polish=False,
            init="sobol",
            callback=lambda x, convergence: (time.time() - start_time) >= de_budget,
        )
    except:
        h_de = best_h.copy()
    else:
        h_de = feasibility_projection(de_result.x, target_sum)
    
    c5_de = compute_c5(h_de, dx)
    if c5_de < best_c5:
        best_h[:] = h_de
        best_c5 = c5_de
    
    # Local refinement
    def objective_slsqp(x):
        return compute_c5(x, dx)
    
    constraints_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = Bounds(0.0, 1.0)
    
    if not (np.isnan(best_c5) or np.isinf(best_c5)):
        slsq_result = minimize(
            objective_slsqp,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints_slsqp,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if slsq_result.success:
            h_new = slsq_result.x
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h[:] = h_new
                best_c5 = c5_new
    
    return (best_h, best_c5, n_points)
