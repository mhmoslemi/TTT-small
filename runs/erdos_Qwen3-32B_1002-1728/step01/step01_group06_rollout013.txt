import numpy as np
from scipy.optimize import minimize, Bounds
import time

def repair(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    if total == 0:
        return np.zeros_like(h_clipped)
    return h_clipped * (n_points / 2) / total

def constraint_sum(x, target):
    return np.sum(x) - target

def objective_C5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full') * (2 / n_points)
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280

    if 'initial_h_values' in globals():
        initial = initial_h_values.copy()
        old_n = len(initial)
        new_x = np.linspace(0, 2, n_points)
        old_x = np.linspace(0, 2, old_n)
        h = np.interp(new_x, old_x, initial)
        h = repair(h, n_points)
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid - 10:mid + 10] = 0.5
        h = repair(h, n_points)

    current_best = h.copy()
    best_c5 = objective_C5(current_best, n_points)

    T_initial = 1.0
    alpha = 0.995
    sigma = 0.1
    T = T_initial

    max_sa_iterations = 10000
    for iteration in range(max_sa_iterations):
        if time.time() - start_time > budget_s * 0.98:
            break
        noise = np.random.normal(0, sigma * T, n_points)
        proposal = current_best + noise
        proposal = repair(proposal, n_points)
        new_c5 = objective_C5(proposal, n_points)

        current_c5 = best_c5
        if new_c5 < current_c5:
            current_best = proposal.copy()
            best_c5 = new_c5
        else:
            if T < 1e-8:
                continue
            delta = new_c5 - current_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                current_best = proposal.copy()
                best_c5 = new_c5
        T *= alpha

    x0 = current_best.copy()
    target_sum = n_points / 2
    constraint = {'type': 'eq', 'fun': constraint_sum, 'args': (target_sum,)}
    bounds = Bounds(0, 1)

    def objective(x):
        x = x.flatten()
        x = repair(x, n_points)
        return objective_C5(x, n_points)

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[constraint], options={'maxiter': 200, 'ftol': 1e-6})

    if result.success:
        optimized_h = result.x.copy()
        optimized_h = repair(optimized_h, n_points)
        new_c5 = objective_C5(optimized_h, n_points)
        if new_c5 < best_c5:
            current_best = optimized_h.copy()
            best_c5 = new_c5

    final_h = repair(current_best, n_points)
    final_c5 = objective_C5(final_h, n_points)
    return final_h, final_c5, n_points
