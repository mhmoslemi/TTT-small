        return np.max(corr) * dx

        x0 -= offset / n_points

        # Run optimization
        result = minimize(

            x0=x0,
            method='COBYLA',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 150, 'ftol': 1e-6}
