import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        h_sum = np.sum(h)
        if np.abs(h_sum) > 1e-10:
            h *= (n / 2) / h_sum

    else:
        # Default initialization: pulse in center
        n = 280
        h = np.zeros(n)
        mid = n // 2
        h[mid-10:mid+10] = 0.25
        h *= (n / 2) / h.sum()
        h = np.clip(h, 0, 1)
    
    def compute_c5(h_vec):
        dx = 2.0 / len(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Parameters for evolutionary strategy
    sigma = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lambda_offspring = 10
    adaptation_interval = 50
    previous_check_c5 = best_c5
    last_progress_check = 0
    current_h = h.copy()
    generations = 0
    
    while time.time() - start_time < budget_s * 0.98:
        # Generate offspring
        offspring = []
        for _ in range(lambda_offspring):
            # Mutation
            mut = np.random.randn(n) * sigma
            child = current_h + mut
            # Repair to meet constraints
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum > 1e-10:

            child = np.clip(child, 0, 1)
            offspring.append(child)
        
        # Evaluate all offspring

        # Select best
        best_idx = np.argmin(offspring_c5)

        # Update best solution
        if next_c5 < best_c5:

            break
