import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = len(h0)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):

        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    # PSO parameters
    n_particles = 20
    w = 0.7  # inertia
    c1 = 2.0  # cognitive
    c2 = 2.0  # social
    velocity_clamp = 0.1
    max_iter = 9999999  # limit via time check

    # Initialize particles
    particles = np.zeros((n_particles, n_points))
    particles[0] = h0.copy()

        rand_h = np.random.rand(n_points)
        # normalize to sum
        s = rand_h.sum()
        if s == 0:
            particles[i] = np.zeros(n_points)
        else:
            particles[i] = rand_h * (target_sum / s)
            particles[i] = np.clip(particles[i], 0, 1)

    # Initialize velocities
