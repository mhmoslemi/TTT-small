from scipy.optimize import minimize

def smooth_cross_correlation_max(h_vec, n_points, dx, beta):
    h = h_vec.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')  # shape (2n_points -1)
    scaled = cross * dx
    exp_val = np.exp(beta * scaled)
    sum_exp = np.sum(exp_val)
    return (1 / beta) * np.log(sum_exp)

    cross = np.correlate(h, ones_minus_h, mode='full')  # shape (2n_points -1)
    return np.max(cross * dx)
