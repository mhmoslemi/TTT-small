def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = None

        dx = 2.0 / n_points
        target_sum = n_points / 2.0

        if resolution == 45:
            h = current_h.copy()
        else:
            upsamp_factor = resolution // current_n
            h = np.repeat(current_h, upsamp_factor)
            current_sum = np.sum(h)
            adjustment = target_sum / current_sum
            h *= adjustment

        corr = np.correlate(h, 1 - h, mode="full") * dx
        current_c = np.max(corr)
        if best_c5 is None or current_c < best_c5:
            best_c5 = current_c
            best_h = h.copy()

        low = 0.380
        high = max(best_c5, 0.381)
        for _ in range(8 if resolution == 180 else 12):
            if time.time() - start_time > budget_s - 60:
                break
            mid = (low + high) * 0.5
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    0 <= h_var,
                    h_var <= 1,
                    cp.sum(h_var) == target_sum
                ]
                for k in range(-n_points + 1, n_points):

                        j = i + k
                        sum_expr += h_var[i] * (1 - h_var[j])
                    constraints.append(sum_expr * dx <= mid)
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.CVXOPT, max_itrs=200)
                if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
