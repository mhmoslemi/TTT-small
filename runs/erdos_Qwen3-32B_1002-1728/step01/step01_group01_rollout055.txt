import numpy as np
import time

def project_h(h_vec):
    n_points = len(h_vec)
    target_sum = n_points / 2.0
    h_clipped = np.clip(h_vec, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return h_clipped.copy()
    delta = (target_sum - current_sum) / n_points
    h_adjusted = h_clipped + delta
    h_projected = np.clip(h_adjusted, 0, 1)
    return h_projected.copy()

    dx = 2.0 / n_points
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
