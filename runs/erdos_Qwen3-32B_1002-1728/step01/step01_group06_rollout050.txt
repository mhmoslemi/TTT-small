import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    target_n = 280
    max_elapsed = budget_s * 0.98

    # Load or create initial solution
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        if n != target_n:
            old_x = np.linspace(0, 2, n)
            new_x = np.linspace(0, 2, target_n)
            h = np.interp(new_x, old_x, h)
            h *= (target_n / 2) / h.sum()
    else:
        h = np.zeros(target_n)
        mid = target_n // 2
        h[mid-5:mid+5] = 0.5

    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(corr)
    best_h = h.copy()

    # Evolutionary parameters
    lambda_offspring = 10
    sigma = 0.1
    no_imp_count = 0

        # Select best offspring
        best_off_idx = np.argmin([c5 for h_vec, c5 in offspring])
