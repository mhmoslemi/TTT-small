import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    n_particles = 20
    particles = []
    velocities = []
    particles.append(h.copy())
    velocities.append(np.zeros_like(h))
    for _ in range(n_particles - 1):
        random_h = np.random.rand(n_points)
        sum_rand = random_h.sum()
        random_h = random_h * (target_sum / sum_rand)
        random_h = np.clip(random_h, 0, 1)
        particles.append(random_h)
        velocities.append(np.zeros_like(h))

    fitness = np.array([compute_c5(p, dx) for p in particles])
    personal_best_positions = [p.copy() for p in particles]
    personal_best_fitness = fitness.copy()
    global_best_idx = np.argmin(fitness)
    global_best = particles[global_best_idx].copy()
    global_best_fitness = fitness[global_best_idx]

    w, c1, c2 = 0.7, 2.0, 2.0
    max_vel = 0.1
    min_vel = -0.1

    while time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = (w * velocities[i] + c1 * r1 * (personal_best_positions[i] - particles[i]) +
                   c2 * r2 * (global_best - particles[i]))
            vel = np.clip(vel, min_vel, max_vel)
            velocities[i] = vel
            new_pos = particles[i] + vel
