import numpy as np
import time

def repair_position(h, target_sum, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        h_clipped = np.ones(n_points) * target_sum / n_points
    else:
        scaling = target_sum / current_sum
        h_clipped *= scaling
    h_clipped = np.clip(h_clipped, 0.0, 1.0)
    return h_clipped

def compute_objective(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
