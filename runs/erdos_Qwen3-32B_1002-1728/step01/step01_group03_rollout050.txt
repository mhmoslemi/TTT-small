import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, perturb_frac=0.2, noise=0.3):
        h_pert = h.copy()
        perturb_n = max(1, int(perturb_frac * n_points))
        indices = np.random.choice(n_points, size=perturb_n, replace=False)
        h_pert[indices] += np.random.uniform(-noise, noise, size=len(indices))
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = h_pert.sum()
        if current_sum != 0:
            h_pert *= target_sum / current_sum
        h_pert = np.clip(h_pert, 0, 1)
        return h_pert
    
    noise_level = 0.3
    max_initial_noise = noise_level
    perturb_frac = 0.2
    last_improvement_time = start_time
    
    while time.time() - start_time < budget_s - 60:
        x0 = get_perturbed(best_h, perturb_frac=perturb_frac, noise=noise_level)
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val
                last_improvement_time = time.time()
                noise_level = max_initial_noise  # reset noise after improvement
            else:
                if time.time() - last_improvement_time > 120:
                    noise_level = max(0.1, noise_level * 0.8)
        if noise_level < 1e-3:
            break
    
    return (best_h, best_c5, n_points)
