import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec, dx_val):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx_val
        return np.max(corr)

    x = (h - 0.5).copy()
    lambda_val = 1.0
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    learn_rate = 0.1

    while time.time() - start_time < budget_s - 5:
        h_logit = lambda_val * x
        h = 1.0 / (1 + np.exp(-h_logit))
        current_sum = h.sum()
        if np.isclose(current_sum, 0):
            h = np.zeros_like(h)
        else:
            h *= target_sum / current_sum
        h = np.clip(h, 0, 1)

        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()

        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        k = max_index - n_points + 1

        grad_h = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad_h[i] += (1 - h[j])
                grad_h[j] -= h[i]

        grad_x = grad_h * (lambda_val * h * (1 - h))
        x = np.clip(x - learn_rate * grad_x, -1, 1)

    return best_h, best_c5, n_points
