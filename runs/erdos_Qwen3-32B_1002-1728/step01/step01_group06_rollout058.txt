def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        h = repair(h, n)
    else:
        n = 280
        h = np.zeros(n)
        mid = n // 2
        h[mid-5:mid+5] = 0.5
        h *= (n / 2) / h.sum()
        h = repair(h, n)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, n)
    sigma = 0.1
    adaptation_interval = 50
    offspring_count = 10
    max_generations = 2000
    no_improvement_count = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > max_time * 0.98:
            break
        
        offspring = np.random.randn(offspring_count, n) * sigma + best_h
        offspring_repaired = np.array([repair(ind, n) for ind in offspring])
        
        c5_values = [evaluate_c5(ind, n) for ind in offspring_repaired]
        best_offspring_idx = np.argmin(c5_values)
        
        if c5_values[best_offspring_idx] < best_c5:
            best_h = offspring_repaired[best_offspring_idx].copy()
            best_c5 = c5_values[best_offspring_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= adaptation_interval:
            sigma *= 0.7
            no_improvement_count = 0
        else:
            sigma *= 0.95
        
        if gen % 200 == 0 and gen > 0:
            pass  # Progress tracking is implied but not verbose
    
    return best_h, best_c5, n

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair(x, n):
    x_clipped = np.clip(x, 0, 1)
    s = np.sum(x_clipped)
    if s == 0:
        x_scaled = np.zeros(n)
        x_scaled[0] = n / 2.0
        x_scaled = np.clip(x_scaled, 0, 1)
    else:
        x_scaled = x_clipped * (n / 2.0) / s
        x_scaled = np.clip(x_scaled, 0, 1)
    return x_scaled
