import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    def objective(h_flat):
        h = repair(h_flat)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return corr.max()

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    popsize = max(10, int(np.sqrt(n_points)))
    start = time.time()
    result = differential_evolution(
        objective, bounds=bounds, seed=seed,
        popsize=popsize, strategy='best1bin',
        mutation=(0.5, 1.5, 0.1), recombination=0.9,
        maxiter=100, polish=False, init='sobol',
        callback=lambda x, con, val: time.time()-start > budget_s-30
    )
    
    best_h = repair(result.x)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
