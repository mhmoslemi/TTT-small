def evaluate_C5_exact(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.full(n_points, 0.5)

    h_initial = np.clip(h_initial, 0.0, 1.0)
    sum_initial = np.sum(h_initial)
    if sum_initial != 0:
        scaling = (n_points / 2) / sum_initial
        h_initial *= scaling
    h_initial = np.clip(h_initial, 0.0, 1.0)

    best_c5 = evaluate_C5_exact(h_initial, n_points, dx)
    best_h = h_initial.copy()

    beta_values = [1, 2, 4, 8, 16, 32, 64]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    perturbations = []
    for _ in range(3):
        seed = int((time.time() % 1000) * 1000)
        np.random.seed(seed)
        perturbed = h_initial + np.random.normal(0.0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_p = np.sum(perturbed)
        if sum_p != 0:
            perturbed = perturbed * (n_points / 2) / sum_p
        perturbations.append(np.clip(perturbed, 0.0, 1.0))

    def objective_func(x, beta):
        h = x.reshape(n_points)
        ones_minus_h = 1.0 - h
        cross = np.correlate(h, ones_minus_h, mode='full')
        cross_dx = cross * dx
        max_val = cross_dx.max()
        exp_sum = np.sum(np.exp(beta * (cross_dx - max_val)))
        return (np.log(exp_sum) + beta * max_val) / beta

    for perturbed in perturbations:
        h_current = perturbed.copy()
        for beta in beta_values:
            if time.time() - start_time > budget_s - 20:
                break
            localscope = {'beta': beta}
            options = {'maxiter': 50, 'gtol': 1e-6, 'verbose': 0}
            result = minimize(
                lambda x: objective_func(x, localscope['beta']),
                x0=h_current,
                args=(),
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options=options
            )
            if result.success:
                h_current = np.clip(result.x, 0.0, 1.0)
                sum_h = np.sum(h_current)
                if sum_h != 0:
                    h_current = h_current * (n_points / 2) / sum_h
                h_current = np.clip(h_current, 0.0, 1.0)
                current_c5 = evaluate_C5_exact(h_current, n_points, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
    return best_h, best_c5, n_points
