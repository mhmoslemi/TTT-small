import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    S_desired = n_points / 2

    def compute_c5(h_vec):
        h = h_vec.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    np.random.seed(seed)
    h_best = h_initial.copy()
    best_c5 = compute_c5(h_best)
    current_h = h_best.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Clip to [0,1]
        current_h = np.clip(current_h, 0, 1)
        # Adjust sum constraint
        current_sum = current_h.sum()
        delta = current_sum - S_desired
        if abs(delta) > 1e-8:
            free = (current_h > 0) & (current_h < 1)
            if free.any():
                m = free.sum()
                correction = delta / m
                current_h[free] -= correction
                # Re-clip after adjustment
                current_h = np.clip(current_h, 0, 1)
        # Evaluate current solution
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = current_h.copy()
        # Perturb and proceed
        current_h += np.random.normal(0, 0.01, n_points)
    
    # Final validation projection
    h_best = np.clip(h_best, 0, 1)
    delta = h_best.sum() - S_desired
    free = (h_best > 0) & (h_best < 1)
    if np.any(free) and abs(delta) > 1e-8:
        h_best[free] -= delta / np.count_nonzero(free)
        h_best = np.clip(h_best, 0, 1)
    
    return h_best, best_c5, n_points
