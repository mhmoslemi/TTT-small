import numpy as np
import time

def compute_c5(h, dx):
    h = h.reshape(-1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, fitnesses, size=5):
    indices = np.random.choice(len(population), size, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in indices]
    best_index = indices[np.argmin(tournament_fitnesses)]
    return population[best_index]

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    np.random.seed(seed)

    # Genetic algorithm parameters
    population_size = 50
    elitism_size = 5
    mutation_rate = 0.1
    mutation_std = 0.1

    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_initial.copy().astype(float)
        h += np.random.normal(0, mutation_std, n_points)
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        h[0] -= current_sum - target_sum
        h = np.clip(h, 0, 1)
        population.append(h)

    best_c5 = float('inf')
    best_h = h_initial.copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitnesses = [compute_c5(h, dx) for h in population]

        # Update global best
        current_indices = np.argsort(fitnesses)
        if fitnesses[current_indices[0]] < best_c5:
            best_c5 = fitnesses[current_indices[0]]
            best_h = population[current_indices[0]].copy()

        # Select elites
        elites = [population[i] for i in current_indices[:elitism_size]]
        new_population = elites.copy()

        # Generate remaining individuals
        while len(new_population) < population_size:
            # Select parents
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)

            # Crossover
            child = 0.5 * p1 + 0.5 * p2
            child = np.clip(child, 0, 1)

            # Mutation
            mutation_mask = np.random.rand(n_points) < mutation_rate
            noise = np.random.normal(0, mutation_std, n_points)
            child += mutation_mask * noise

            # Repair constraints
            child = np.clip(child, 0, 1)
            current_sum = child.sum()
            delta = current_sum - target_sum
            child[0] -= delta
            child = np.clip(child, 0, 1)
            new_population.append(child)

        # Replace population
        population = new_population

    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
