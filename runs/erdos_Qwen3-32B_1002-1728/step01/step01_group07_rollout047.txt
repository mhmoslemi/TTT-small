import time
from numpy.random import uniform, normal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        scaling = target_sum / np.sum(h)
        return h * scaling

    def create_population():
        pop = []
        # Add initial solution and perturbed variants
        for _ in range(5):
            h = h_init + normal(0, 0.05, n_points)
            pop.append(repair(h))
        for _ in range(45):
            h = normal(0.5, 0.3, n_points)
            pop.append(repair(h))
        return np.array(pop)

    def crossover(parent1, parent2):
        alpha = 0.5 + uniform(-0.2, 0.2)
        child = (alpha * parent1 + (1 - alpha) * parent2)
        return repair(child)

    def mutate(h):
        mutation_rate = 0.1
        mask = uniform(0, 1, n_points) < mutation_rate
        h[mask] += normal(0, 0.02, size=np.sum(mask))
        return repair(h)

    population = create_population()
    best_h = h_init
    best_c5 = objective(best_h)
    
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        
        # Evaluate population
        fitnesses = [objective(p) for p in population]
        order = np.argsort(fitnesses)
        parents = population[order[:25]]  # Top 50%
        
        # Initialize next generation with elites
        next_gen = parents[:5]
        
        # Crossover
        for _ in range(18):
            p1, p2 = parents[np.random.randint(25, size=2)]
            child = crossover(p1, p2)
            next_gen = np.vstack([next_gen, child])
        
        # Mutation
        for _ in range(10):
            parent = parents[np.random.randint(25)]
            child = mutate(parent)
            next_gen = np.vstack([next_gen, child])
        
        population = next_gen
        
        # Update best solution if improved
        for p in population:
            c5 = objective(p)
            if c5 < best_c5:
                best_c5 = c5
                best_h = p.copy()
        
        # Check termination after 20 generations with no improvement

    # Final check for constraints
