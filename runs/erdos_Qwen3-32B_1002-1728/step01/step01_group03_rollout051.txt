```python
import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    max_delta = 0.3
    delta_decay_rate = 0.85
    no_improvement_counter = 0
    no_improvement_threshold = 3
    max_tries = 30
    tries = 0
    
    while tries < max_tries and time.time() - start_time < budget_s - 60:
        tries += 1
        h_pert = best_h.copy()
        delta = np.random.uniform(-max_delta, max_delta, size=n_points)
        delta -= np.mean(delta)
        h_pert = h_pert + delta
        h_pert = np.clip(h_pert, 0, 1)
        x0 = h_pert
        
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_c5 = curr_val
                best_h = res.x.copy()
                no_improvement_counter = 0
            else:
                no_improvement_counter += 1
        if no_improvement_counter >= no_improvement_threshold:
            max_delta *= delta_decay_rate
            no_improvement_counter = 0
    
    return (best_h, best_c5, n_points)
```
