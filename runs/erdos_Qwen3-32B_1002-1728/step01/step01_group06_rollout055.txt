import numpy as np
import time

def repair(h, n_points):
    S = n_points / 2.0
    h_clipped = np.clip(h, 0, 1)
    sum_h = h_clipped.sum()
    if sum_h == 0:
        return np.full_like(h_clipped, 0.25)
    scale = S / sum_h
    h_scaled = h_clipped * scale
    return np.clip(h_scaled, 0, 1)

def evaluate(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 280
    part_size = 50
    max_iterations = 500
    
    # Initialize base h
    if 'initial_h_values' in globals():
        base_h = initial_h_values.astype(np.float64).copy()

        base_h = np.interp(new_x, old_x, base_h)
    else:
        base_h = np.zeros(n_points)
        mid = n_points // 2
        base_h[mid-5:mid+5] = 0.5
    
    base_h = repair(base_h, n_points)
    
    # Initialize particles
    positions = np.zeros((part_size, n_points), dtype=np.float64)
    velocities = np.random.uniform(-0.1, 0.1, (part_size, n_points)).astype(np.float64)
    
    # Initialize particles with perturbations
    for i in range(part_size):
        positions[i] = base_h.copy() + np.random.uniform(-0.05, 0.05, n_points)
        positions[i] = repair(positions[i], n_points)
    
    # Evaluate personal bests
    pbest_positions = positions.copy()
    pbest_scores = np.array([evaluate(p, n_points) for p in pbest_positions])
    
    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_pos = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    inertia = 0.9
    inertia_decay = 0.95
    max_cog_soc = 1.5
    vel_clamp = 0.5
    no_imp_limit = 75

    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
