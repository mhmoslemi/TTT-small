    sum_h = np.sum(h_prev)
    if abs(sum_h - n_points / 2) > 1e-8:
        h_prev = np.clip(h_prev * (n_points / 2) / sum_h, 0.0, 1.0)

    best_c5 = evaluate_C5(h_prev, n_points, dx)
    best_h = h_prev.copy()

    while time.time() - start_time < budget_s:
