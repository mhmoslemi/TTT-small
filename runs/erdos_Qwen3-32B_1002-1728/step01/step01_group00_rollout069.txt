import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)

    start_time = time.time()
    best_c5 = None
    best_h = h_initial.copy()

    def create_periodic_initial(n_points, period, phase):

        for i in range(n_points):
            if (i + phase) % period == 0:
                positions.append(i)
        m = len(positions)
        if m == 0:
            return None
        total_val = n_points / 2.0
        v = 1.0
        try:
            u = (total_val - m * v) / (n_points - m)
        except:
            return None
        if u < 0:
            u = 0.0
            v = total_val / m
            if v > 1.0:
                return None
        elif u > 1.0:
            u = 1.0
            v = (total_val - (n_points - m) * u) / m
        h = np.full(n_points, u)
        h[positions] = v
        return h

    # Generate periodic initial guesses
    candidate_periods = [4, 5, 6, 7, 9]
    initial_guesses = []
    for p in candidate_periods:
        for phase in [0, 1]:
            h_guess = create_periodic_initial(n_points, p, phase)
            if h_guess is not None:
                initial_guesses.append(h_guess)
    initial_guesses.append(h_initial.copy())  # Include original

    # Process each initial guess
    for guess in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(
            objective,
            x0=guess.copy(),

            bounds=bounds,
            constraints=sum_constraint,

        )
