        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_coarse.copy()

    top_candidates = [best_h.copy()]
    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 10:
            break

        def fine_objective(h_vec):
            h = h_vec.reshape(n_points)
            cross = np.correlate(h, 1.0 - h, mode='full') * dx
            return np.max(cross)

        result = minimize(fine_objective, candidate, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            h_fine = result.x
            sum_h = np.sum(h_fine)
            if not np.isclose(sum_h, n_points / 2, atol=1e-6):
                scale = (n_points / 2) / sum_h
                h_fine = np.clip(h_fine * scale, 0.0, 1.0)
            cross = np.correlate(h_fine.reshape(n_points), 1.0 - h_fine.reshape(n_points), mode='full') * dx
            current_C5 = np.max(cross)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_fine.copy()

    return best_h, best_C5, n_points
