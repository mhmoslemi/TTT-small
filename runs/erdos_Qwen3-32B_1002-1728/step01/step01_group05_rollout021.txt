import numpy as np
import time
from scipy.optimize import differential_evolution, Bounds, minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        h_projected = np.zeros_like(h)
        h_projected[0] = target_sum
        return h_projected
    scale = target_sum / current_sum
    h_projected = h_clipped * scale
    h_projected = np.clip(h_projected, 0, 1)
    return h_projected

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    # Differential Evolution Global Search
    bounds = Bounds(0.0, 1.0)
    def objective_de(x):
        h_proj = project_to_feasible(x, target_sum)
        return compute_c5(h_proj, dx)

    de_result = differential_evolution(
        objective_de, 
        bounds=bounds, 
        popsize=15, 
        maxiter=50, 
        strategy='best1bin',
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-5,
        polish=False,
        seed=seed,
        workers=1
    )

    if de_result.success:
        h_de = project_to_feasible(de_result.x, target_sum)
        c5_de = compute_c5(h_de, dx)
        if c5_de < best_c5:
            best_h, best_c5 = h_de, c5_de

    # Local Refinement with SLSQP
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result_slspq = minimize(
        lambda x: compute_c5(x, dx),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result_slspq.success:
        h_new = result_slspq.x
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Perturbation-based Escapes
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        trials = 0
        amp = 0.3
        while trials < 15 and time.time() - start_time < budget_s:
            perturbation = np.random.uniform(-amp, amp, n_points)
            h_pert = project_to_feasible(best_h + perturbation, target_sum)
            result = minimize(

                method='L-BFGS-B',

                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
            amp *= 0.95
            trials += 1

    return (best_h, best_c5, n_points)
