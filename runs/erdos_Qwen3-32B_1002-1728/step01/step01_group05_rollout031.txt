            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if res.success:
            h_best = project(res.x, target_sum)
            best_c5 = compute_c5(h_best, dx)

    while time.time() - start_time < budget_s:
        pert = np.random.randn(n_points) * 0.1
        h_perturbed = project(h_best + pert, target_sum)
        res_nm = minimize(
            lambda x: compute_c5(project(x, target_sum), dx),
            x0=h_perturbed,
            method='Nelder-Mead',
            bounds=Bounds(0.0, 1.0),
            options={'maxiter': 20}
        )
        if res_nm.success:
            h_tmp = project(res_nm.x, target_sum)
            c5_tmp = compute_c5(h_tmp, dx)
            if c5_tmp < best_c5:
                h_best = h_tmp
                best_c5 = c5_tmp
        if time.time() - start_time >= budget_s:
            break

    return (h_best, best_c5, n_points)
