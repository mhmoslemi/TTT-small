import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()
    np.random.seed(seed)
    budget = budget_s
    n_points = 280
    target_sum = n_points / 2.0
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy().astype(float)
        # Resample to n_points if necessary
        if len(h) != n_points:
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
            h = repair(h, target_sum)
    else:
        # Default initialization: central pulse
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-20:mid+20] = 0.5
        h *= target_sum / h.sum()
        h = np.clip(h, 0, 1)
        h = repair(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    
    # ES parameters
    sigma = 0.1
    lambda_ = 10
    generations = 1000
    no_improve = 0
    sigma_decay = 0.5
    threshold = 50
    
    for gen in range(generations):
        if time() - start_time > budget * 0.98:
            break
        
        # Generate offspring
        offspring = []
        energies = []
        for _ in range(lambda_):
            noise = sigma * np.random.randn(n_points)
            child = best_h + noise
            child = repair(child, target_sum)
            c5 = compute_c5(child, n_points)
            offspring.append(child)
            energies.append(c5)
        
        # Select best
        candidates = np.array(offspring)
        best_idx = np.argmin(energies)
        current_best_c5 = energies[best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = candidates[best_idx].copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt sigma
        if no_improve >= threshold:
            sigma *= sigma_decay
            no_improve = 0
    
    return best_h, best_c5, n_points


def repair(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if abs(current_sum - target_sum) < 1e-8:
        return h
    
    delta = target_sum - current_sum
    n = len(h)
    for _ in range(5):
        if abs(delta) < 1e-8:
            break
        if delta > 0:
            idx = np.where(h < 1)[0]
            if len(idx) == 0:
                break
            delta_per = delta / len(idx)
            h[idx] += delta_per
            h = np.clip(h, 0, 1)
        else:
            idx = np.where(h > 0)[0]
            if len(idx) == 0:
                break
            delta_per = (-delta) / len(idx)
            h[idx] -= delta_per
            h = np.clip(h, 0, 1)
        current_sum = h.sum()
        delta = target_sum - current_sum
    
    if abs(current_sum - target_sum) > 1e-8 and current_sum != 0:
        h = h / current_sum * target_sum
        h = np.clip(h, 0, 1)
    return h


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
