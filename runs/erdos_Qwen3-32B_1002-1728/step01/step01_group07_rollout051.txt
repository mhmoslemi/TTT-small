import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    import time

    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_de(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e6 * (sum_h - target_sum) ** 2
        return c5 + penalty

    # Construct initial population with perturbations of h_init
    rng = np.random.default_rng(seed)
    popsize = 15
    init_pop = [h_init.tolist()]
    for _ in range(popsize - 1):
        perturbed = h_init + rng.normal(0, 0.1, h_init.shape)
        init_pop.append(np.clip(perturbed, 0, 1).tolist())

    result = differential_evolution(
        func=objective_de,
        bounds=bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=popsize,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        init=init_pop,
        disp=False,
        workers=1
    )

    optimized_h = result.x
    h = np.clip(optimized_h.reshape(n_points), 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, target_sum):
        scale = target_sum / h_sum
        h = h * scale
        h = np.clip(h, 0, 1)

    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx

    # Verify constraints
    assert abs(np.sum(h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((h >= -1e-6) & (h <= 1.0 + 1e-6)), "Value bounds violated"

    return (h.tolist(), best_c5, n_points)
