import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = np.float64(n_points / 2.0)

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    start_time = time.time()

    # Systematic exploration with COBYLA and improved constraint repair
    for _ in range(10):  # 10 restarts per strategy
        if time.time() - start_time > budget_s - 5:  # 5 sec buffer
            break
            
        # Build perturbed initial guess
        x0 = best_h.copy()
        x0 += np.random.normal(0, 0.02, n_points)  # Perturbation per strategy
        x0 = np.clip(x0, 0, 1)
        
        # Enforce sum constraint via distributed adjustment
        offset = np.sum(x0) - target_sum
        x0 -= offset / n_points
        x0 = np.clip(x0, 0, 1)

        # Optimization with COBYLA and controlled budget
        result = minimize(
            objective,
            x0=x0,
            method='COBYLA',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 150, 'rhobeg': 0.1}
        )

        # Feasibility repair with uniform redistribution
        curr_h = np.clip(result.x, 0, 1)
        offset = np.sum(curr_h) - target_sum
        curr_h -= offset / n_points
        curr_h = np.clip(curr_h, 0, 1)

        curr_c5 = np.max(corr_values)

            best_h[:] = curr_h[:]  # Direct assignment for numerical stability

    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
