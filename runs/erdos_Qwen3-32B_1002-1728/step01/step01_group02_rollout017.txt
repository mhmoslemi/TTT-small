import numpy as np
import time
import cvxpy as cp

def run():
    start_time = time.time()
    budget_s = 1100

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_current = initial_h_values.copy()
    else:
        n_points = 45
        h_current = np.ones(n_points) / 2  # Ensures sum is n_points / 2
    dx = 2.0 / n_points

    best_c5 = np.inf
    best_h = h_current.copy()

    while time.time() - start_time < budget_s - 1:
        # Compute current overlaps
        ones_minus_h = 1.0 - h_current
        cross = np.correlate(h_current, ones_minus_h, mode='full')
        cross_values = cross * dx
        max_overlap = np.max(cross_values)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_current.copy()
        k_max_index = np.argmax(cross_values)
        k = k_max_index - (n_points - 1)

        # Compute valid indices for the linearized constraint
        valid_i = []
        for i in range(n_points):
            if 0 <= i + k < n_points:
                valid_i.append(i)

        # Compute coefficients for the linear approximation
        coeffs = np.zeros(n_points)
        for i in valid_i:
            ik = i + k
            coeffs[i] += (1.0 - h_current[ik]) * dx
            coeffs[ik] -= h_current[i] * dx

        # Compute constant term for the linearized constraint
        lin_part_const = np.sum(coeffs * h_current)

        # Set up convex optimization problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [
            h_var >= 0.0,
            h_var <= 1.0,
            cp.sum(h_var) == n_points / 2,
        ]
        # Linearized constraint for worst-case overlap
        constraints.append(t >= max_overlap + cp.sum(coeffs * h_var) - lin_part_const)

        problem = cp.Problem(cp.Minimize(t), constraints)
        problem.solve(solver=cp.SCS, max_iters=1000, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = h_var.value
            # Ensure bounds
            new_h = np.clip(new_h, 0.0, 1.0)
            # Repair sum constraint with projection
