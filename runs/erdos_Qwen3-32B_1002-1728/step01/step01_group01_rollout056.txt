def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_best = initial_h_values.copy()
    n_points = len(h_best)
    dx = 2.0 / n_points
    best_c5 = None

    def compute_c5(h):
        O_values = []
        for k in range(-n_points + 1, n_points):
            val = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    val += h[i] * (1 - h[j])
            O_values.append(val)
        return max(O_values) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        if total == 0:
            return h.copy()
        scale = (n_points / 2) / total
        h *= scale
        h = np.clip(h, 0, 1)
        total = h.sum()
        if abs(total - n_points / 2) > 1e-8:
            scale = (n_points / 2) / total
            h *= scale
            h = np.clip(h, 0, 1)
        return h

    def compute_gradient(h, O_values, top_indices):
        n = len(h)
        grad = np.zeros(n)
        count = 0
        for index in top_indices:
            k = -n_points + 1 + index
            grad_k = np.zeros(n)
            for m in range(n):
                j1 = m + k
                if 0 <= j1 < n:
                    grad_k[m] += (1 - h[j1])
                j2 = m - k
                if 0 <= j2 < n:
                    grad_k[m] -= h[j2]
            grad += grad_k
            count += 1
        if count == 0:
            return np.zeros(n)
        return grad / count

    best_c5 = compute_c5(h_best)
    alpha = 0.1
    iteration = 0
    last_improvement = 0

    while time.time() - start_time < budget_s - 5:
        O_values = []
        for k in range(-n_points + 1, n_points):
            val = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    val += h_best[i] * (1 - h_best[j])
            O_values.append(val)
        current_c5 = max(O_values) * dx
        if best_c5 is None or current_c5 < best_c5:

        threshold = np.percentile(O_values, 95)
