import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 280
        h = np.zeros(n)
        mid = n // 2
        h[mid-5:mid+5] = 0.5
        h *= (n / 2) / h.sum()
        h = np.clip(h, 0, 1)

    def repair(h, n):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0:
            h = np.zeros_like(h)
        else:
            h = h * (n / 2) / s
        return h

    def evaluate(h, n):
        dx = 2 / n
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    parent = h.copy()
    best_parent = parent.copy()
    best_c5 = evaluate(parent, len(parent))
    n_points = len(parent)
    sigma = 0.1
    lambda_ = 10
    max_gen = 2000
    generation = 0
    adaptation_interval = 50
    last_best = best_c5

    while generation < max_gen and time.time() - start_time < max_time * 0.98:
        offspring = []
        for _ in range(lambda_):
            child = parent + sigma * np.random.randn(n_points)
            child = repair(child, n_points)
            offspring.append(child)

        # Evaluate and track best
        scores = []
        for child in offspring:
            c5 = evaluate(child, n_points)
            scores.append(c5)
            if c5 < best_c5:
                best_parent = child.copy()
                best_c5 = c5

        # Select best offspring
        scores = np.array(scores)
        best_idx = np.argmin(scores)
        parent = offspring[best_idx]

        # Adapt mutation rate
        if generation % adaptation_interval == 0:
            current_best = best_c5
            if current_best < last_best:
                sigma *= 0.95
            else:
                sigma *= 0.7
            last_best = current_best
            # Prevent sigma from being too small
            sigma = max(sigma, 1e-5)

        generation += 1

    return best_parent, best_c5, n_points
