    def objective(h):
        h = h.flatten()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)

    initial_guesses = []

    # Generate diverse periodic patterns as initial guesses
    period_candidates = [2, 3, 4, 5, 6, 7, 8, 9, 10]
    for T in period_candidates:
        m_ideal = T / 2
        m = int(m_ideal + 0.5)
        m = max(1, min(m, T - 1))
        base = np.zeros(T)
        base[:m] = 1.0
        n_periods = n_points // T
        remainder = n_points % T
        pattern = np.tile(base, n_periods)
        if remainder:
            pattern = np.concatenate([pattern, base[:remainder]])
        initial_guesses.append(pattern.copy())

    # Add original h_initial as base case
    initial_guesses.append(h_initial)

    best_c5 = None
    best_h = None
    start_time = time.time()

    # Optimize each periodic pattern and track best
    for x0 in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(
            fun=objective,
            x0=x0.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset
        curr_h = np.clip(curr_h, 0, 1)
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if (best_c5 is None or curr_c5 < best_c5):
            best_c5 = curr_c5
            best_h = curr_h.copy()
        # Break early if out of time
        if time.time() - start_time > budget_s - 10:
            break

    # Additional restarts with small perturbations
    if best_h is not None:
        num_perturb = 10
        for _ in range(num_perturb):
            if time.time() - start_time > budget_s - 10:
                break
            perturbed = best_h.copy()
            perturbed += np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            offset = np.sum(perturbed) - n_points/2
            perturbed[0] -= offset
            perturbed = np.clip(perturbed, 0, 1)
            result = minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=sum_constraint,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            curr_h = result.x.copy()
            curr_h = np.clip(curr_h, 0, 1)
            offset = np.sum(curr_h) - n_points/2
            curr_h[0] -= offset
            curr_h = np.clip(curr_h, 0, 1)
            corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            curr_c5 = np.max(corr_values)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()

    return best_h, best_c5, n_points
