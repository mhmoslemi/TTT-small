import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def evaluate_C5_smoothing(h_vector, n_points, dx, beta):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full') * dx
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * cross)))

def repair_h(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - n_points / 2) > 1e-8:
        h = np.clip(h * (n_points / 2) / s, 0.0, 1.0)
    return h
