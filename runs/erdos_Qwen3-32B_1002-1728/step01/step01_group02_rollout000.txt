import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def generate_perturbation(h, n_points, seed):
    np.random.seed(seed)
    perturbed = h + np.random.normal(0, 0.1, n_points)
    perturbed = np.clip(perturbed, 0.0, 1.0)
    current_sum = np.sum(perturbed)
    if abs(current_sum - n_points / 2) > 1e-8:
        scaling_factor = (n_points / 2) / current_sum
        perturbed = np.clip(perturbed * scaling_factor, 0.0, 1.0)
    return perturbed

        h_initial = initial_h_values.copy().astype(np.float64)

        n_points = 45

        h_initial[:] = 0.5

    dx = 2.0 / n_points
