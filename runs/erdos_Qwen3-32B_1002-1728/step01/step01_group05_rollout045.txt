def objective_de(x, target_sum, dx):
    h = project_to_feasible(x, target_sum)
    c5_value = compute_c5(h, dx)
    sum_diff = np.abs(h.sum() - target_sum)
    penalty = 1e6 * sum_diff
    return c5_value + penalty

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    # Global search with differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    max_de_time = 300
    de_maxiter = 50
    try:
        de_result = differential_evolution(
            objective_de,
            bounds,
            args=(target_sum, dx),
            popsize=20,
            strategy='best1bin',
            maxiter=de_maxiter,
            polish=False,
            disp=False,
            tol=1e-4,
            mutation=(0.5, 1.0),
            recombination=0.8,
            workers=1
        )
        if time.time() - start_time < budget_s - 200:  # Reserve time for refinement
            de_h = project_to_feasible(de_result.x, target_sum)
            # Local refinement with SLSQP
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            bounds_slsqp = Bounds(0.0, 1.0)
            result = minimize(
                lambda x: compute_c5(x, dx),
                x0=de_h,
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            if result.success:
                best_h = result.x
                best_c5 = compute_c5(best_h, dx)
    except Exception:
        pass

    # Final polishing from best solution so far
    perturb_amp = 0.1
    if time.time() - start_time < budget_s - 50:
        perturbation = np.random.uniform(-perturb_amp, perturb_amp, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = project_to_feasible(perturbed_h, target_sum)
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds_slsqp = Bounds(0.0, 1.0)
        result = minimize(
            lambda x: compute_c5(x, dx),
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success and compute_c5(result.x, dx) < best_c5:
            best_h = result.x
            best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n_points)
