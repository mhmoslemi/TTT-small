            subtract_amount = -delta / len(indices)
            candidate_clipped[indices] -= subtract_amount
            candidate_clipped = np.clip(candidate_clipped, 0, 1)
        return candidate_clipped

    pop_size = 50
    population = [fix_candidate(np.random.rand(n_points)) for _ in range(pop_size)]
    population[0] = fix_candidate(h)
    c5_values = [compute_c5(ind) for ind in population]
    best_idx = np.argmin(c5_values)
    best_h = population[best_idx].copy()
    best_c5 = c5_values[best_idx]

    start_time = time.time()
    F = 0.8
    CR = 0.9

    while time.time() - start_time < budget_s - 5:
        for idx in range(pop_size):
            while True:
                idxs = np.random.choice(pop_size, 3, replace=False)
                if idx not in idxs:
                    break
            a, b, c = population[idxs]
            mutant = a + F * (b - c)
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[idx])
            trial_fixed = fix_candidate(trial)
            trial_c5 = compute_c5(trial_fixed)
            if trial_c5 < c5_values[idx]:
                population[idx] = trial_fixed
                c5_values[idx] = trial_c5
                if trial_c5 < best_c5:
                    best_h[:] = trial_fixed
                    best_c5 = trial_c5

    return best_h, best_c5, n_points
