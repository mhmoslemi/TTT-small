import numpy as np
import time
import heapq

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    popul_size = 50
    n_elite = 5
    mutation_strength = 0.05
    pop = [h_initial.copy() for _ in range(popul_size)]

    # Helper for sum constraint
    def enforce_sum(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        s = h_vec.sum()
        if abs(s - target_sum) < 1e-8:
            return h_vec
        delta = target_sum - s
        h_vec[0] += delta
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    # Perturb initial population
    np.random.seed(seed)
    for i in range(popul_size):
        vec = pop[i].copy()
        vec += np.random.normal(0, 0.1, n_points)
        pop[i] = enforce_sum(vec)

    best_h = None
    best_val = float('inf')
    start_time = time.time()

    # Evaluate initial population
    def evaluate(individual):
        corr = np.correlate(individual, 1 - individual, mode='full')
        return np.max(corr) * dx

    while time.time() - start_time < budget_s - 2:

        # Evaluate
        scores = [evaluate(ind) for ind in pop]
        for i in range(popul_size):
            if scores[i] < best_val:
                best_val = scores[i]
                best_h = pop[i].copy()

        # Select parents via tournament
        ranks = np.argsort(scores)
        parents = []
        for _ in range(popul_size):
            tournament = np.random.choice(popul_size, 5, replace=False)
            winner = ranks[heapq.nsmallest(1, tournament, key=lambda j: scores[j])][0]
            parents.append(pop[winner])

        # Create new population
        new_pop = []
        for idx in range(n_elite):
            new_pop.append(pop[ranks[idx]].copy())

        for idx in range(n_elite, popul_size):
            p1, p2 = parents[idx % 10], parents[idx % 5 + 10]
            # Blend crossover
            child = 0.5 * p1 + 0.5 * p2 + np.random.normal(0, 0.02, n_points)
            # Mutation
            child += np.random.normal(0, mutation_strength, n_points)
            child = enforce_sum(child)
            new_pop.append(child)

        pop = new_pop

    # Final cleanup
    best_h = enforce_sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(corr) * dx
    return best_h, c5, n_points
