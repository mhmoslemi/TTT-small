import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    max_time = budget_s

    # Initialize with provided initial_h_values
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
        h_sum = h.sum()
        if h_sum > 0:
            h = np.clip(h * (n_points / 2) / h_sum, 0, 1)
    else:
        # Construct default starting point
        n_points = 280
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-15:mid+15] = 0.5
        h_sum = h.sum()
        h = np.clip(h * (n_points / 2) / h_sum, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    current_h = h.copy()
    current_c5 = best_c5

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    max_fast_iter = 1000000
    iteration = 0

    # SA loop
    while time.time() - start_time < max_time * 0.85:
        if iteration >= max_fast_iter:
            break
        # Generate proposal
        noise = np.random.normal(0, 0.1 * T, n_points)
        proposed_h = current_h + noise
        proposed_h = np.clip(proposed_h, 0, 1)
        proposed_sum = proposed_h.sum()
        if proposed_sum == 0:
            proposed_h = np.full(n_points, (n_points / 2) / n_points)
        else:
            proposed_h *= (n_points / 2) / proposed_sum

        new_c5 = compute_c5(proposed_h, n_points)
        # Acceptance probability
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0:
            accept = True
        else:
            try:
                metropolis = np.exp(-delta_c5 / T)
            except OverflowError:
                metropolis = np.inf
            accept = metropolis > np.random.rand()

        if accept:
            current_h = proposed_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                alpha = 0.9  # Faster cooling after improvement

        T *= alpha
        iteration += 1

    # SLSQP refinement
    def objective(x):
        x = x.ravel()
        h_opt = np.clip(x, 0, 1)
        return compute_c5(h_opt, n_points)

    res = minimize(
        objective,
        best_h.ravel(),
        method='SLSQP',
        bounds=Bounds(0, 1),
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n_points / 2}],
        options={'maxiter': 200, 'ftol': 1e-12}
    )

    if res.success:
        optimized_h = np.clip(res.x, 0, 1)
        optimized_h = np.clip(optimized_h * (n_points / 2) / optimized_h.sum(), 0, 1)
        candidate_c5 = compute_c5(optimized_h, n_points)
        if candidate_c5 < best_c5:
            best_h = optimized_h
            best_c5 = candidate_c5

    final_sum = best_h.sum()
    best_h = np.clip(best_h * (n_points / 2) / final_sum, 0, 1)
    final_c5 = compute_c5(best_h, n_points)

    return best_h, final_c5, n_points
