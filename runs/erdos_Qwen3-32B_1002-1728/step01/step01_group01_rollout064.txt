import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        return h_clipped * (n_points / 2) / s if s != 0 else h_clipped

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop = []
    for _ in range(25):
        perturbed = initial_h_values + np.random.normal(0, 0.1, n_points)
        pop.append(project(perturbed))
    for _ in range(25):
        random_vec = np.random.rand(n_points)
        pop.append(project(random_vec))
    pop = np.array(pop)

    pop_eval = np.array([evaluate(ind) for ind in pop])
    best_idx = np.argmin(pop_eval)
    best_h = pop[best_idx].copy()
    best_c5 = pop_eval[best_idx]

    F, CR = 0.8, 0.9
    while time.time() - start_time < budget_s - 10:
        for i in range(len(pop)):
            candidate_idx = np.array([idx for idx in range(len(pop)) if idx != i])
            a, b, c = np.random.choice(candidate_idx, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            target = pop[i]
            mask = np.random.rand(n_points) < CR
            rand_idx = np.random.randint(n_points)
            mask[rand_idx] = True
            trial = np.where(mask, mutant, target)
            trial = project(trial)
            trial_c5 = evaluate(trial)
            if trial_c5 < pop_eval[i]:
                pop[i] = trial
                pop_eval[i] = trial_c5
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5
            if time.time() - start_time > budget_s - 10:
                break
        if np.random.rand() < 0.2 and time.time() - start_time < budget_s - 20:
            perturbed = best_h + np.random.normal(0, 0.05, n_points)
            pop[0] = project(perturbed)
            pop_eval[0] = evaluate(pop[0])
            if pop_eval[0] < best_c5:
                best_h = pop[0].copy()
                best_c5 = pop_eval[0]
    return best_h, best_c5, n_points
