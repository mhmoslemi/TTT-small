import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import time
    
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # PSO Parameters
    n_particles = 50
    max_iter = 5000
    v_max = 0.1
    wp_start, wp_end = 0.9, 0.4
    wc, ws = 2.0, 2.0  # Cognitive and social coefficients
    
    # Initialize swarm
    rng = np.random.Generator(np.random.PCG64(seed))
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    for _ in range(n_particles):
        h = h_init.copy()
        h += rng.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        h = h * (target_sum / max(s, 1e-6))
        particles.append(h)
        velocities.append(rng.uniform(-0.1, 0.1, n_points))
        corr = np.correlate(h, 1 - h, mode='full')
        pbest_scores.append(np.max(corr) * dx)
        pbest_positions.append(h.copy())
    
    gbest = pbest_positions[np.argmin(pbest_scores)].copy()
    gbest_score = min(pbest_scores)
    
    # PSO Main Loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Adjust inertia weight
        inertia_weight = wp_start - (wp_start - wp_end) * iteration / max_iter
        
        for i in range(n_particles):
            h = particles[i].copy()
            v = velocities[i].copy()
            
            r1, r2 = rng.uniform(), rng.uniform()
            cognitive = wc * r1 * (pbest_positions[i] - h)
            social = ws * r2 * (gbest - h)
            v_new = inertia_weight * v + cognitive + social
            v_new = np.clip(v_new, -v_max, v_max)
            velocities[i] = v_new
            
            h_new = h + v_new
            h_new = np.clip(h_new, 0, 1)
            s = np.sum(h_new)
            h_new = h_new * (target_sum / max(s, 1e-6))
            particles[i] = h_new
            
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            score = np.max(corr) * dx
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = h_new.copy()
        
        # Update global best
        iter_best_idx = np.argmin(pbest_scores)
        if pbest_scores[iter_best_idx] < gbest_score:
            gbest[:] = pbest_positions[iter_best_idx]
            gbest_score = pbest_scores[iter_best_idx]
    
    # Final constraint checks and return
    assert np.allclose(np.sum(gbest), target_sum, atol=1e-6), "Sum constraint violated"
    assert np.all((0 <= gbest) & (gbest <= 1)), "Bounds violated"
    return (gbest.tolist(), gbest_score, n_points)
