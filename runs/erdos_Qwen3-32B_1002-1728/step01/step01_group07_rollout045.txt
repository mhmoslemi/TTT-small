import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 50
    population = []
    # Initialize population with 5 copies of h_init and 45 perturbed variations
    for _ in range(5):
        population.append(h_init.copy())
    for _ in range(45):
        perturbed = h_init.copy()
        perturbed += np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed *= target_sum / np.sum(perturbed)
        population.append(perturbed)
    population = np.array(population)

    best_h = h_init.copy()
    best_c5 = np.inf

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        c5_values = []
        for ind in population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(corr) * dx
            c5_values.append(c5)
        c5_values = np.array(c5_values)

        # Track the best solution found
        current_best_idx = np.argmin(c5_values)
        if c5_values[current_best_idx] < best_c5:
            best_c5 = c5_values[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Select parents (top 50% by fitness)
        sorted_indices = np.argsort(c5_values)
        parents = population[sorted_indices[:int(0.5 * population_size)]]

        # Create new population via crossover and mutation
        new_population = []
        # Elitism: carry over top 10%
        new_population.extend(population[sorted_indices[:int(0.1 * population_size)]])

        # Generate remaining individuals
        remaining = population_size - int(0.1 * population_size)
        n_crossover = int(0.8 * remaining)
        n_mutation = remaining - n_crossover

        # Crossover: blend of two parents
        for _ in range(n_crossover):
            if len(parents) < 2:
                new_population.append(parents[0].copy())
                continue
            p1, p2 = parents[np.random.choice(len(parents), 2, replace=False)]
            child = (p1 + p2) / 2.0
            child = np.clip(child, 0, 1)
            child = child * (target_sum / np.sum(child))
            new_population.append(child)

        # Mutation
        for _ in range(n_mutation):
            parent = parents[np.random.choice(len(parents))]
            child = parent + np.random.normal(0, 0.02, n_points)
            child = np.clip(child, 0, 1)
            child = child * (target_sum / np.sum(child))
            new_population.append(child)

        population = np.array(new_population)

    # Final evaluation of best solution
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx

    return (best_h.tolist(), final_c5, n_points)
