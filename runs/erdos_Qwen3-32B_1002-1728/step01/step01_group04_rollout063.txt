        h_scaled = np.clip(h_scaled, 0, 1)
        population.append(h_scaled)
    
    # Evaluate C5 for each individual in initial population
    c5_scores = [compute_c5(ind, dx) for ind in population]
    best_index = np.argmin(c5_scores)
    best_h = population[best_index].copy()
    best_c5 = c5_scores[best_index]
    
    # Differential Evolution parameters
    F = 0.8
    CR = 0.9
    pop_size = len(population)
    
    while time.time() - start_time < budget_s - 1:
        for i in range(pop_size):
            if time.time() - start_time >= budget_s - 1:
                break
            
            # Select three distinct indices different from i
            indices = list(range(pop_size))
            indices.remove(i)
            a, b, c = np.random.choice(indices, 3, replace=False)
            
            # Mutate
            v = population[a] + F * (population[b] - population[c])
            v = np.clip(v, 0, 1)
            
            # Crossover
            u = population[i].copy()
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if j == j_rand or np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0, 1)
            
            # Enforce constraint
            s = u.sum()
            if s != 0:
                u = u * (target_sum / s)
            else:
                u = np.full_like(u, target_sum / n_points)
            u = np.clip(u, 0, 1)
            
            # Evaluate trial
            trial_c5 = compute_c5(u, dx)
            
            # Selection
            if trial_c5 < c5_scores[i]:
                population[i] = u.copy()
                c5_scores[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h[:] = u.copy()
    
    # Final validation of best_h
    best_h = np.clip(best_h, 0, 1)
    s = best_h.sum()
    if not np.isclose(s, target_sum, rtol=1e-5, atol=1e-8):
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c5, n_points
