import numpy as np
import time

def _objective(h, n_points):
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    C5 = np.max(corr * dx)
    return C5

def run(seed=42, budget_s=1100, **kwargs):

    vel_clamp = 0.5

    h_initial = np.clip(h_initial, 0, 1)
