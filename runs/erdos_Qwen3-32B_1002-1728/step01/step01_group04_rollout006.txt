            p2 = population[np.argmin(np.random.choice(len(population), 3))]
            idx = np.random.randint(n_points)
            child = np.concatenate([p1[:idx], p2[idx:]])
            if np.random.rand() < 0.1:
                noise_scale = 0.05
                child += np.random.normal(0, noise_scale, child.shape)
            child = fix_candidate(child)
            new_population.append(child)
        population = new_population
        current_c5 = min([compute_c5(ind, dx) for ind in population])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[np.argmin([compute_c5(ind, dx) for ind in population])].copy()

    return best_h, best_c5, n_points
