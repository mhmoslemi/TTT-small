    tries = 0
    max_tries = 20
    while tries < max_tries and time.time() - start_time < budget_s - 60:
        tries += 1
        x0 = get_perturbed(best_h)
        res = minimize(
            objective, 
            x0=x0, 
            method='SLSQP',
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val

    # Final refinement with aggressive perturbation
    if time.time() - start_time < budget_s - 30:
        x0 = get_perturbed(best_h, noise=0.5)
        res = minimize(
            objective, 
            x0=x0, 
            method='SLSQP',
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val

    return (best_h, best_c5, n_points)
