import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_C5(h_vector, n_points, dx):
    """
    Return the worst-case overlap between h and (1-h) over all integer shifts
    """
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    # Cross-correlation of h and (1-h) in 'valid' mode to avoid padding
    cross_corr = correlate(h, ones_minus_h, mode='full')
    return np.max(cross_corr * dx)
