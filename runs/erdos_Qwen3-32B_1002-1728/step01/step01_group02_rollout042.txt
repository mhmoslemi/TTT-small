    h = h_vec.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    overlaps = cross * dx
    return np.max(overlaps)

def repair_feasibility(h_vec, n_points):
    h = h_vec.reshape(n_points)
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = h_clipped.sum()
    if np.isclose(current_sum, n_points / 2, atol=1e-8):
        return h_clipped.copy()
    else:
        scale = (n_points / 2) / current_sum
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0.0, 1.0).copy()

def run():
    start_time = time.time()
    budget_s = 1100

    n_points = 45
    dx = 2.0 / n_points

    # Initialize base h
    if 'initial_h_values' in globals():
        base_h = initial_h_values.copy().astype(np.float64)
    else:
        base_h = np.zeros(n_points).astype(np.float64)
        base_h[:] = 0.5  # Initial uniform guess

    # Generate perturbed initial candidates
    n_candidates = 5
    initial_candidates = []
    for _ in range(n_candidates):
        perturbed = base_h + np.random.normal(0, 0.1, base_h.shape)
        perturbed = np.clip(perturbed, 0, 1)
        s = perturbed.sum()
        if s <= 0:
            perturbed = base_h.copy()
        else:
            perturbed = perturbed * (n_points / 2) / s
            perturbed = np.clip(perturbed, 0, 1)
        initial_candidates.append(perturbed)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    # Betas for coarse phase
    betas = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    best_c5 = evaluate_C5(base_h, n_points, dx)
    best_h = base_h.copy()

    # Coarse optimization
    for candidate in initial_candidates:
        if time.time() - start_time > budget_s - 10:
            break
        current_h = candidate.copy()
        for beta in betas:
            if time.time() - start_time > budget_s - 10:
                break
            # Objective function for this beta
            result = minimize(lambda x: smoothed_objective(x, beta, n_points, dx), 
                              current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                optimized_h = repair_feasibility(result.x, n_points)
                current_c5 = evaluate_C5(optimized_h, n_points, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h.copy()
                current_h = optimized_h.copy()

    # Fine-tuning: exact optimization for best candidates
    top_candidates = [best_h.copy()]
    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 10:
            break
        current_h = candidate.copy()
        for _ in range(3):  # 3 iterations of exact optimization
            if time.time() - start_time > budget_s - 10:
                break
            result = minimize(lambda x: exact_objective(x, n_points, dx),
                              current_h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                optimized_h = repair_feasibility(result.x, n_points)
                current_c5 = evaluate_C5(optimized_h, n_points, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h.copy()
                current_h = optimized_h.copy()

    return best_h, best_c5, n_points
