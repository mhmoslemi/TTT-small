    if remaining_time > max_de_time:
        try:
            result_de = differential_evolution(
                de_objective,
                bounds=bounds,
                strategy='best1bin',
                popsize=20,
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.7,
                maxiter=100,
                polish=False,
                workers=1,
                disp=False
            )
            if result_de.success:
                h_de = result_de.x.copy()
                sum_h_de = np.sum(h_de)
                if sum_h_de != 0:
                    h_scaled = h_de * (target_sum / sum_h_de)
                else:
                    h_scaled = np.zeros_like(h_de)
                h_clipped = np.clip(h_scaled, 0, 1)
                h_initial = h_clipped
        except:
            pass
        remaining_time = max(budget_s - (time.time() - start_time), 0)

    # Step 2: SLSQP refinement
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = Bounds(lb=0.0, ub=1.0)
    max_slsqp_time = min(500, int(remaining_time * 0.6))
    if remaining_time > 0:
        result_slsqp = minimize(
            lambda x: compute_c5(x, dx),
            x0=h_initial,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-8}
        )
        if result_slsqp.success:
            h_refined = np.clip(result_slsqp.x, 0, 1)
            c5_refined = compute_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined

    # Perturbation and restart if time remains
    if remaining_time > 50:
        perturbation_counts = min(int(remaining_time / 10), 5)
        for _ in range(perturbation_counts):
            if time.time() - start_time >= budget_s:
                break
            idx = np.random.choice(n_points, size=int(n_points * 0.15), replace=False)
            perturbed_h = best_h.copy()
            perturbed_h[idx] += np.random.uniform(-0.2, 0.2, size=len(idx))
            perturbed_h = np.clip(perturbed_h, 0, 1)
            sum_perturbed = np.sum(perturbed_h)
            if sum_perturbed != 0:
                perturbed_h = perturbed_h * (target_sum / sum_perturbed)
            else:
                perturbed_h = best_h.copy()
            perturbed_h = np.clip(perturbed_h, 0, 1)
            result_restart = minimize(
                lambda x: compute_c5(x, dx),
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=constraints,
                options={'maxiter': 50}
            )
            if result_restart.success:
                c5_new = compute_c5(result_restart.x, dx)
                if c5_new < best_c5:
                    best_h = result_restart.x
                    best_c5 = c5_new

    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 1e-9:
        best_h = best_h * (target_sum / sum_best)
    best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, n_points)
