import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    max_time = budget_s

    # Initialize with initial_h_values or a default
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        h_sum = h.sum()
        scaling = (n / 2) / h_sum
        h *= scaling
        h = np.clip(h, 0, 1)
    else:
        # Default: central pulse
        n = 280
        h = np.zeros(n)
        mid = n // 2
        h[mid-5:mid+5] = 0.5
        h_sum = h.sum()
        scaling = (n / 2) / h_sum
        h *= scaling
        h = np.clip(h, 0, 1)

    best_h, best_c5 = h.copy(), compute_c5(h, n)

    # Evolutionary Strategy parameters
    population_lambda = 10
    mutation_sigma = 0.1
    max_generations = 1000
    adaptation_interval = 50
    current_h = h.copy()

    # Main ES loop
    for gen in range(max_generations):
        if time.time() - start_time > max_time * 0.98:
            break

        # Generate offspring
        offspring = []
        for _ in range(population_lambda):
            candidate = current_h + mutation_sigma * np.random.randn(n)
            candidate = np.clip(candidate, 0, 1)
            candidate_sum = candidate.sum()
            if candidate_sum == 0:
                offspring.append(current_h.copy())
            else:
                scale_factor = (n / 2) / candidate_sum
                candidate *= scale_factor
                candidate = np.clip(candidate, 0, 1)
                offspring.append(candidate)

        # Evaluate offspring
        offspring_c5 = np.array([compute_c5(offspr, n) for offspr in offspring])
        best_idx = np.argmin(offspring_c5)
        offspring_h = offspring[best_idx]
        offspring_c5_val = offspring_c5[best_idx]

        # Update best and current_h
        if offspring_c5_val < best_c5:
            best_h = offspring_h.copy()
            best_c5 = offspring_c5_val
            current_h = offspring_h.copy()
        else:
            current_h = current_h.copy()

        # Adapt mutation sigma
        if (gen + 1) % adaptation_interval == 0:
            if offspring_c5_val < best_c5:
                mutation_sigma *= 0.95
            else:
                mutation_sigma *= 0.7
            # Avoid vanishing mutation
            mutation_sigma = np.clip(mutation_sigma, 1e-5, 1.0)

    return best_h, best_c5, n
