import numpy as np
import time
from scipy.optimize import minimize, Bounds, differential_evolution

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def compute_feasible_c5(h_candidate, n_points, target_sum, dx):
    h = np.clip(h_candidate, 0.0, 1.0)
    current_sum = h.sum()
    if abs(current_sum) < 1e-9:
        h_projected = np.zeros_like(h)
    else:
        scaling = target_sum / current_sum
        h_projected = h * scaling
        h_projected = np.clip(h_projected, 0.0, 1.0)
    c5 = compute_c5(h_projected, dx)
    return c5
