import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    bounds = [(0.0, 1.0)] * n_points
    
    def obj_de(h_flat):
        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return float('inf')
        h_normalized = h_clipped * (target_sum / current_sum)
        return compute_c5(h_normalized, dx)
    
    # Global search with differential evolution
    de_start = time.time()
    result_de = differential_evolution(obj_de, bounds, strategy='best1bin',
        maxiter=20, popsize=15, mutation=(0.5, 1), recombination=0.8,
        polish=False, workers=1)
    de_duration = time.time() - de_start
    
    h_global = result_de.x.reshape(n_points)
    h_global_clipped = np.clip(h_global, 0, 1)
    current_sum = h_global_clipped.sum()
    h_global_clipped = h_global_clipped * (target_sum / current_sum) if current_sum != 0 else best_h
    c5_global = compute_c5(h_global_clipped, dx)
    if c5_global < best_c5:
        best_h, best_c5 = h_global_clipped, c5_global
    
    # Local refinement
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = Bounds(0.0, 1.0)
    
    max_slsqp = 40
    trials = 0
    perturb_amp = 0.1
    while time.time() - start_time < budget_s and trials < max_slsqp:

                h_perturbed = h_perturbed * (target_sum / current_sum)
        else:

        perturb_amp *= 0.95
    
    return (best_h, best_c5, n_points)
