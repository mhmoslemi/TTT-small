    sum_exp = np.sum(exp_values)
    log_sum_exp = np.log(sum_exp) / beta
    sum_h = np.sum(h)
    constraint = sum_h - target
    penalty = lambda_val * constraint + 0.5 * rho * (constraint**2)
    return log_sum_exp + penalty

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode='full')
    return np.max(cross * dx)

def run():
    start_time = time.time()
    budget_s = 1100
    n_points = 45
    dx = 2.0 / n_points
    target = n_points / 2.0

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy().astype(np.float64)
    else:
        h_initial = np.full(n_points, 0.5)
    # Normalize
    current_sum = np.sum(h_initial)
    if abs(current_sum - target) > 1e-8:
        h_initial = np.clip(h_initial * (target / current_sum), 0.0, 1.0)

    best_c5 = evaluate_C5(h_initial, n_points, dx)
    best_h = h_initial.copy()

    bounds_h = [(0.0, 1.0) for _ in range(n_points)]
    perturbation_strengths = [0.01, 0.02, 0.05, 0.1, 0.2]

    for _ in range(5):  # Perturbations
        np.random.seed(int((time.time() % 1000) * 1000))
        strength = perturbation_strengths[_ % 5]
        delta = np.random.normal(0.0, strength * np.max(h_initial), n_points)
        new_h = np.clip(h_initial + delta, 0.0, 1.0)
        new_sum = np.sum(new_h)
        new_h = np.clip(new_h * (target / new_sum), 0.0, 1.0) if abs(new_sum - target) > 1e-8 else new_h

        # ALM parameters
        lambda_val = 0.0
        rho = 1e-4
        beta_values = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

        for beta in beta_values:
            if time.time() - start_time + 10 >= budget_s:
                break
            for _ in range(3):  # ALM iterations
                # Optimization step
                result = minimize(
                    objective_aug,
                    x0=new_h,
                    args=(beta, lambda_val, rho, target, dx),
                    method='SLSQP',
                    jac='2-point',
                    bounds=bounds_h
                )
                if result.success:
                    new_h_opt = np.clip(result.x, 0.0, 1.0)
                    new_sum_opt = np.sum(new_h_opt)
                    if abs(new_sum_opt - target) > 1e-8:
                        new_h_opt = np.clip(new_h_opt * (target / new_sum_opt), 0.0, 1.0)
                    c5 = evaluate_C5(new_h_opt, n_points, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h[:] = new_h_opt

                    # ALM update
                    constraint = new_sum_opt - target
                    lambda_val += rho * constraint
                    if abs(constraint) > 1e-3:
                        rho = min(rho * 2.0, 1e6)
                else:
                    break
                if time.time() - start_time >= budget_s - 10:
                    break

    return best_h, best_c5, n_points
