            remaining_sum = target_sum - len(over_indices) * 1.0
            under_mask = ~over
            if remaining_sum <= 0:
                scaled_h[under_mask] = 0.0
            else:
                under_sum = h_clipped[under_mask].sum() * scaling_factor
                scaled_h[under_mask] = scaled_h[under_mask] * (remaining_sum / under_sum)
                scaled_h[under_mask] = np.clip(scaled_h[under_mask], 0, 1)
        return scaled_h

    pop_size = 50
    num_elites = int(pop_size * 0.1)
    mutation_rate = 0.1
    tournament_size = 5

    population = [normalize_h(np.random.rand(n_points), target_sum) for _ in range(pop_size)]
    population[0] = normalize_h(h_initial.copy(), target_sum)
    population = np.array(population)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    while time.time() - start_time < budget_s - 1:
        new_population = []
        elite_indices = np.argsort(fitness)[:num_elites]
        elites = population[elite_indices]
        new_population.extend(elites)

        while len(new_population) < pop_size:
            parent_indices = np.random.randint(0, len(population), tournament_size)
            parent_fitnesses = fitness[parent_indices]
            parent1_idx = parent_indices[np.argmin(parent_fitnesses)]
            parent_indices = np.random.randint(0, len(population), tournament_size)
            parent_fitnesses = fitness[parent_indices]
            parent2_idx = parent_indices[np.argmin(parent_fitnesses)]

            parent1 = population[parent1_idx].copy()
            parent2 = population[parent2_idx].copy()

            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, parent1, parent2)
            child = normalize_h(child, target_sum)

            if np.random.rand() < mutation_rate:
                child_perturbed = child + np.random.normal(0, 0.05, size=n_points)
                child = normalize_h(child_perturbed, target_sum)

            new_population.append(child)

        population = np.array(new_population)
        fitness = np.array([compute_c5(ind) for ind in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]

    return best_h, best_c5, n_points
