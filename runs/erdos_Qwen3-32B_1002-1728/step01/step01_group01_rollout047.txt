import numpy as np
import time
from scipy.optimize import OptimizeResult

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Differential Evolution parameters
    pop_size = 50
    F = 0.8  # Mutation scaling factor
    CR = 0.9  # Crossover rate

    # Initialize population
    pop = []
    for _ in range(pop_size // 2):
        pert = np.random.normal(0, 0.1, n_points)
        h = np.clip(initial_h_values + pert, 0, 1)
        h = h * (n_points / 2.0 / h.sum())
        pop.append(h)
    for _ in range(pop_size // 2):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1) * (n_points / 2.0 / h.sum())
        pop.append(h)
    pop = np.asarray(pop)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    # Track best solution
    best_idx = np.argmin([evaluate(p) for p in pop])
    best_h = pop[best_idx].copy()
    best_c5 = evaluate(best_h)

    # DE main loop
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            # Select three distinct vectors
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a + F * (b - c)
            # Crossover with target
            target = pop[i]
            cross_mask = np.random.rand(n_points) < CR
            trial = np.where(cross_mask, mutant, target)
            # Project onto feasible set
            trial_clipped = np.clip(trial, 0, 1)
            trial_sum = trial_clipped.sum()
            if trial_sum > 1e-9:
                trial_scaled = trial_clipped * (n_points / 2.0 / trial_sum)
            else:
                trial_scaled = np.zeros_like(trial_clipped)
                trial_scaled[0] = n_points / 2.0
            # Evaluate and replace
            trial_obj = evaluate(trial_scaled)
            target_obj = evaluate(pop[i])
            if trial_obj < target_obj:
                pop[i] = trial_scaled
                if trial_obj < best_c5:
                    best_h = trial_scaled.copy()
                    best_c5 = trial_obj
            # Early-exit if time is tight
            if time.time() - start_time > budget_s - 2:
                return best_h, best_c5, n_points

    return best_h, best_c5, n_points
