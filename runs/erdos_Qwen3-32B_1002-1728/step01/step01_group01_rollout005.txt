            h = np.clip(h + perturbation, 0.0, 1.0)
            h = h * (n_points / 2.0) / np.sum(h)
            last_improve_time = time.time()

        # Early exit if budget is nearly exhausted
        if time.time() - start_time > budget_s - 5:
            break

    # Final cleanup: project to constraints
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = h_best * (n_points / 2.0) / np.sum(h_best)
    return h_best, best_c5, n_points
