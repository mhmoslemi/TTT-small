import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_tries = max(100, int(budget_s // 15))  # Adapt max tries to budget
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    def get_perturbed(h, noise=0.1, noise_decay=0.5):
        h_pert = h.copy()
        noise_level = noise * (np.random.random() ** noise_decay)
        h_pert += np.random.uniform(-noise_level, noise_level, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        h_pert *= target_sum / current_sum
        return h_pert
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    tries = 0
    while time.time() - start_time < budget_s - 60:
        tries += 1
        perturbed = get_perturbed(best_h if tries % 5 == 0 else h_init)
        x0 = perturbed.copy()
        # Alternate perturbation scale
        noise = 0.3 if tries % 4 == 0 else 0.1
        x0 = get_perturbed(x0, noise=noise)
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 30, 'ftol': 1e-6}
        )

                )
