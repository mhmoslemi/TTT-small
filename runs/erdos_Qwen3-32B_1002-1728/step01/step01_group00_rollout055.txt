from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()
    x0 = x0.reshape(-1)

    # Initial repair to ensure sum constraint is satisfied
    sum_x0 = np.sum(x0)
    target_sum = n_points / 2.0
    adj = (sum_x0 - target_sum) / n_points
    x0 -= adj
    x0 = np.clip(x0, 0.0, 1.0)

    best_h = x0.copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    # Perform multiple restarts
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break

        # Perturb the current best_h
        current_x0 = best_h.copy()
        current_x0 += np.random.normal(0, 0.02, n_points)
        current_x0 = np.clip(current_x0, 0.0, 1.0)

        # Adjust sum constraint
        sum_current_x0 = np.sum(current_x0)
        adj = (sum_current_x0 - target_sum) / n_points
        current_x0 -= adj
        current_x0 = np.clip(current_x0, 0.0, 1.0)

        # Run optimization

            objective,
            x0=current_x0,
            method='COBYLA',
            bounds=bounds,

import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
    bounds = Bounds(0.0, 1.0)

    x0 = h_initial.copy().ravel()
    # Ensure initial guess satisfies sum constraint
    sum_x0 = np.sum(x0)
    target_sum = n_points / 2.0
    adj = (sum_x0 - target_sum) / n_points
    x0 -= adj
    x0 = np.clip(x0, 0, 1)

    best_h = x0.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Run 10 adaptive restarts with COBYLA
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break

        # Create perturbed starting point
        current_x0 = best_h.copy()
        perturbation = np.random.normal(0, 0.02, n_points)
        current_x0 += perturbation
        current_x0 = np.clip(current_x0, 0, 1)

        # Enforce sum constraint with uniform redistribution
        sum_current = np.sum(current_x0)
        adj = (sum_current - target_sum) / n_points
        current_x0 -= adj
        current_x0 = np.clip(current_x0, 0, 1)

        # Optimization with COBYLA
        result = minimize(
            objective,
            x0=current_x0,
            method='COBYLA',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 100, 'ftol': 1e-6, 'rhobeg': 0.02}
        )

        # Final repair - ensure valid constraints
        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        sum_curr = np.sum(curr_h)
        adj = (sum_curr - target_sum) / n_points
        curr_h = curr_h - adj
        curr_h = np.clip(curr_h, 0, 1)

        # Evaluate result
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)

        # Update best solution
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h[:] = curr_h[:]

    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    
    return best_h, c5_bound, n_points
