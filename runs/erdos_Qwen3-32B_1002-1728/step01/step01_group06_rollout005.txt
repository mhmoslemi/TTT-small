import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1-best_h, 'full') * (2/n))

    num_children = 10
    max_generations = 1000
    adaptation_interval = 50
