import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0  # Required sum

    def objective(h):
        h = h.reshape(-1)
        h_adj = h.copy()
        current_sum = h_adj.sum()
        delta = T - current_sum
        h_adj[0] += delta
        h_adj = np.clip(h_adj, 0, 1)
        corr = np.correlate(h_adj, 1 - h_adj, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 50
    elite_size = 5
    population = [h_initial.copy() + np.random.normal(0, 0.1, (n_points,)) for _ in range(pop_size)]
    population = np.array([np.clip(p, 0, 1) for p in population])

    best_c5 = float('inf')
    best_h = h_initial.copy()

    start_time = time()
    while time() - start_time < budget_s - 10:
        fitness = np.array([objective(ind) for ind in population])
        for idx in range(pop_size):
            if fitness[idx] < best_c5:
                best_c5 = fitness[idx]
                best_h = population[idx].copy()

        elite_indices = np.argsort(fitness)[:elite_size]
        elites = population[elite_indices]

        new_population = elites.copy()
        while len(new_population) < pop_size:
            t1 = np.random.choice(pop_size, 5, replace=False)
            p1_idx = t1[np.argmin([objective(population[i]) for i in t1])]
            t2 = np.random.choice(pop_size, 5, replace=False)
            p2_idx = t2[np.argmin([objective(population[i]) for i in t2])]

            p1 = population[p1_idx]
            p2 = population[p2_idx]
            child = 0.5 * p1 + 0.5 * p2 + np.random.normal(0, 0.05, n_points)
            new_population = np.vstack([new_population, np.clip(child, 0, 1)])

            if np.random.rand() < 0.2:
                child = new_population[-1].copy()
                mut = np.random.normal(0, 0.1, n_points)
                child += mut
                new_population[-1] = np.clip(child, 0, 1)

        population = new_population

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
