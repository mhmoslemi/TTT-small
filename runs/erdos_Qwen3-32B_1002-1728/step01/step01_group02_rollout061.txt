import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h_vector, n_points, dx):
    h = h_vector.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode="full")
    return np.max(cross * dx)

def smoothed_objective(h_vec, beta, n_points, dx):
    h = h_vec.reshape(n_points)
    ones_minus_h = 1.0 - h
    cross = np.correlate(h, ones_minus_h, mode="full") * dx
    max_c = np.max(cross)
    exp_vals = np.exp(beta * (cross - max_c))
    log_sum_exp = np.log(exp_vals.sum()) / beta + max_c
    return log_sum_exp

def repair_feasibility(h_vec, n_points):
    h = h_vec.reshape(n_points)
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-8:
        scale_factor = target_sum / current_sum
        h = h * scale_factor
        h = np.clip(h, 0.0, 1.0)
    return h.reshape(-1)

def constraint_eq_sum(h, n_points):
    return np.sum(h) - n_points / 2

def run():
    start_time = time.time()
    budget_s = 1100
    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_base = initial_h_values.copy().astype(np.float64)
    else:
        h_base = np.full(n_points, 0.5, dtype=np.float64)

    candidates = [h_base.copy()]
    for _ in range(4):
        perturbed = h_base.copy() + np.random.normal(0.0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = perturbed.sum()
        if not np.isclose(current_sum, n_points / 2.0):
            perturbed *= (n_points / 2.0) / current_sum
            perturbed = np.clip(perturbed, 0.0, 1.0)
        candidates.append(perturbed)

    cons = [{'type': 'eq', 'fun': constraint_eq_sum, 'args': (n_points,)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = np.inf
    best_h = None

    betas = [1., 2., 4., 8., 16., 32., 64., 128., 256., 512., 1024.]

    for candidate in candidates:
        current_h = candidate.copy()
        for beta in betas:
            if time.time() - start_time > budget_s - 10:
                break
            result = minimize(
                fun=smoothed_objective,
                x0=current_h,
                args=(beta, n_points, dx),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
            )
            if result.success:
                current_h = result.x.copy()
                current_h = repair_feasibility(current_h, n_points)
            current_c5 = evaluate_C5(current_h, n_points, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        if time.time() - start_time > budget_s - 10:
            break

    if best_h is not None:
        result_fine = minimize(
            fun=evaluate_C5,
            x0=best_h,
            args=(n_points, dx),
            method='SLSQP',
            bounds=bounds,
        )
        if result_fine.success:
            fine_h = result_fine.x.copy()
            fine_h = repair_feasibility(fine_h, n_points)
            fine_c5 = evaluate_C5(fine_h, n_points, dx)
            if fine_c5 < best_c5:
                best_c5 = fine_c5
                best_h = fine_h

    if best_h is None:
        best_h = h_base.copy()
        best_c5 = evaluate_C5(best_h, n_points, dx)

    return best_h, best_c5, n_points
