import numpy as np
import time
import cvxpy as cp

def run():
    start_time = time.time()
    budget_s = 1100
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = initial_h_values.copy().astype(np.float64)
    else:
        n_points = 45
        h = np.ones(n_points) / 2.0
    dx = 2.0 / n_points

    def evaluate_C5(h_vec, n_points, dx):
        h_np = h_vec.reshape(n_points)
        ones_minus_h = 1.0 - h_np
        cross = np.correlate(h_np, ones_minus_h, mode='full')
        relevant_cross = cross[n_points - 1: 2 * n_points - 1]
        return np.max(np.abs(relevant_cross) * dx)

    best_c5 = evaluate_C5(h, n_points, dx)
    best_h = h.copy()

    while time.time() - start_time < budget_s - 10:
        h_np = np.array(h)
        ones_minus_h = 1.0 - h_np
        cross = np.correlate(h_np, ones_minus_h, mode='full')
        relevant_cross = cross[n_points - 1: 2 * n_points - 1]
        worst_index = np.argmax(np.abs(relevant_cross))
        worst_k = worst_index

        # Compute convex part and gradient
        k = worst_k
        convex_part = 0.0
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                convex_part += h[i] * h[i + k]
                gradient[i] += h[i + k]
                gradient[i + k] += h[i]
        convex_part *= dx

        gradient *= dx  # Gradient of the convex part

        grad_dot_h = np.dot(gradient, h)
        constant_term = 1.0 - convex_part + grad_dot_h

        h_var = cp.Variable(n_points)
        t = cp.Variable()

        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            t >= constant_term - cp.sum(cp.multiply(gradient, h_var))
        ]

        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)

        prob.solve(solver=cp.ECOS, verbose=False)

        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2.0) / h.sum()
            continue

        new_h = h_var.value
        new_c5 = evaluate_C5(new_h, n_points, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            h = best_h
        else:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2.0) / h.sum()

    return best_h, best_c5, n_points
