import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h, noise_level=0.2, frac=0.3):
        h_pert = h.copy()
        n_change = max(1, int(frac * n_points))
        indices = np.random.choice(n_points, n_change, replace=False)
        h_pert[indices] += np.random.uniform(-noise_level, noise_level, n_change)
        h_pert = np.clip(h_pert, 0, 1)
        scale = target_sum / np.sum(h_pert)
        h_pert *= scale
        return h_pert
    
    noise_level = 0.2
    cooldown = 30.0
    while time.time() - start_time < budget_s - 30:
        x0 = get_perturbed(best_h, noise_level)
        try:
            res = minimize(
                objective, x0=x0, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 120, 'ftol': 1e-7}
            )
            if res.success:
                curr_val = objective(res.x)
                if curr_val < best_c5:
                    best_h = res.x.copy()
                    best_c5 = curr_val
                    noise_level = max(0.05, noise_level * 0.8)  # reduce noise after improvement
                else:
                    noise_level = min(0.4, noise_level * 1.2)  # increase noise after stale trials
        except Exception:
            pass
        
        # Dynamic cooldown to prevent divergence
        if (time.time() - start_time) % cooldown < 1:
            noise_level = min(0.4, noise_level * 1.1)
    
    return (best_h, best_c5, n_points)
