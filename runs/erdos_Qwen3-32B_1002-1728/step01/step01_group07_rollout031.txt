    best_c5 = np.max(corr) * dx
    
    # Final validation
    assert abs(np.sum(optimized_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((optimized_h >= -1e-6) & (optimized_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.8:
        # Perturb solution and reoptimize with different strategy
        perturbed = optimized_h + np.random.RandomState(seed + 1).normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = (perturbed / np.sum(perturbed)) * target_sum
        perturbed_result = differential_evolution(
            func=objective_with_penalty,
            bounds=bounds,
            init=perturbed.reshape((1, n_points)),
            strategy='rand1bin',
            maxiter=100,
            popsize=20,
            mutation=(0.8, 1.2),
            recombination=0.95,
            tol=1e-5,
            seed=seed + 2,
            polish=False
        )
        alt_h = perturbed_result.x.reshape(n_points)
        alt_h = np.clip(alt_h, 0, 1)
        alt_sum = np.sum(alt_h)
        if alt_sum > 1e-9:
            alt_h = alt_h * (target_sum / alt_sum)
        alt_h = np.clip(alt_h, 0, 1)
        alt_corr = np.correlate(alt_h, 1 - alt_h, mode='full')
        alt_c5 = np.max(alt_corr) * dx
        
        if alt_c5 < best_c5:
            optimized_h, best_c5 = alt_h, alt_c5
    
    return (optimized_h.tolist(), best_c5, n_points)
