        new_idx = np.argmin(pbest_scores)
        if pbest_scores[new_idx] < gbest_score:
            gbest_score = pbest_scores[new_idx]
            gbest_position = particles[new_idx].copy()

    corr_values = np.correlate(gbest_position, 1 - gbest_position, mode='full') * dx
    c5_bound = np.max(corr_values)
    return gbest_position, c5_bound, n_points
