import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def get_perturbed(h):
        h_pert = h.copy()
        n = len(h_pert)
        subset_size = max(3, int(n * 0.25 + np.random.rand()))
        subset = np.random.choice(n, subset_size, replace=False)
        noise = np.random.uniform(-0.3, 0.3, subset_size)
        h_pert[subset] += noise
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        scale_factor = target_sum / current_sum
        h_pert *= scale_factor
        h_pert = np.clip(h_pert, 0, 1)
        return h_pert
    
    max_tries = 50
    tries = 0
    decay_rate = 0.95
    
    while time.time() - start_time < budget_s - 10 and tries < max_tries:
        tries += 1
        x0 = get_perturbed(best_h)
        maxiter = min(100, int(150 * np.power(decay_rate, tries)))
        ftol = max(1e-8, 1e-5 * np.power(0.9, tries))
        res = minimize(
            objective, x0=x0, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if res.success:
            curr_val = objective(res.x)
            if curr_val < best_c5:
                best_h = res.x.copy()
                best_c5 = curr_val
                # Adaptive continuation after improvement
                x1 = get_perturbed(best_h)
                res1 = minimize(
                    objective, x0=x1, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': 60, 'ftol': 1e-7}
                )
                if res1.success:
                    v1 = objective(res1.x)
                    if v1 < best_c5:
                        best_h = res1.x.copy()
                        best_c5 = v1
            decay_rate *= 0.98
    
    return (best_h, best_c5, n_points)
