import numpy as np
import time
from scipy.optimize import minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def generate_sinusoidal_candidate(n_points, p):
    t = np.linspace(0, 2 * np.pi * p, n_points, endpoint=False)
    h = 0.5 + 0.5 * np.sin(t)
    return np.clip(h, 0.0, 1.0)

def generate_square_candidate(n_points, p):
    samples_per_period = n_points // p
    step = samples_per_period // 2
    h = np.zeros(n_points)
    for i in range(n_points):
        if (i % samples_per_period) < step:
            h[i] = 1.0
    return h

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-9:
        h[:] = 0.5
    else:
        h[:] *= target_sum / current_sum
    h[:] = np.clip(h, 0.0, 1.0)
    return h

    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # For n=80, this is 40.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
