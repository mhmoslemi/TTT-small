    def project(positions):
        positions_clipped = np.clip(positions, 0, 1)
        sum_positions = np.sum(positions_clipped, axis=1, keepdims=True)
        offset = sum_positions - target_sum

        positions_clipped = np.clip(positions_clipped, 0, 1)
        return positions_clipped

        return np.max(corr) * dx

    start_time = time.time()

            break

        elapsed_time = time.time() - start_time
        progress_ratio = min(elapsed_time / budget_s, 1.0)
        w = w_start + (w_end - w_start) * progress_ratio

        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        velocities = (w * velocities
                      + c1 * r1 * (pbest_positions - positions)
                      + c2 * r2 * (np.tile(gbest_position, (n_particles, 1)) - positions))
        positions += velocities
        positions = project(positions)

        current_values = np.array([compute_C5(p) for p in positions])
        is_better = current_values < pbest_values
        pbest_positions[is_better] = positions[is_better]
        pbest_values[is_better] = current_values[is_better]

        current_gbest_idx = np.argmin(pbest_values)
        current_gbest_value = pbest_values[current_gbest_idx]
        if current_gbest_value < best_c5:
            best_c5 = current_gbest_value
            gbest_position = pbest_positions[current_gbest_idx].copy()

        if iteration > max_iter:
            break

    return gbest_position, best_c5, n_points
