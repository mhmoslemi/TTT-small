import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_best = initial_h_values.copy()

    corr = np.correlate(h_best, 1 - h_best, mode='full')
    best_c5 = np.max(corr) * dx

    alpha = 0.05
    max_iter = 1000
    perturb_iter = 80
    perturb_strength = 0.05
    iter_no_improve = 0
    alpha_decay = 0.995
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlaps
        corr = np.correlate(h_best, 1 - h_best, mode='full')
        max_index = np.argmax(corr)
        k_star = max_index - (n_points - 1)
        
        # Compute gradient for worst shift
        grad = np.zeros(n_points)
        for m in range(n_points):

                grad[m] += (1 - h_best[m + k_star])
            if 0 <= m - k_star < n_points:
                grad[m] += -h_best[m - k_star]
        
        # Gradient step and projection
        h_proposal = h_best - alpha * grad
        h_proposal = np.clip(h_proposal, 0, 1)
        h_proposal = h_proposal * (n_points/2) / np.sum(h_proposal)
        
        # Evaluate proposal
        new_corr = np.correlate(h_proposal, 1 - h_proposal, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        # Update best solution
        if new_c5 < best_c5:
            h_best = h_proposal
            best_c5 = new_c5
            iter_no_improve = 0
            alpha *= alpha_decay  # Gradual step decay
        else:
            iter_no_improve += 1
        
        # Perturbation to escape local minima
        if iter_no_improve >= perturb_iter:
            h_best = np.clip(h_best + np.random.normal(0, perturb_strength, n_points), 0, 1)

            iter_no_improve = 0

            alpha = max(0.01, alpha * 1.5)

    h_best = np.clip(h_best, 0, 1)
