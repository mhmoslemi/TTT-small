import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    num_particles = 50
    c1, c2 = 2.0, 2.0
    v_max = 0.1

    # Initialize particles
    particles = np.repeat(h_init[None, ...], num_particles, axis=0)
    noise = np.random.randn(num_particles, n_points) * 0.05
    particles = np.clip(particles + noise, 0, 1)
    sums = particles.sum(axis=1, keepdims=True)
    particles = particles * (target_sum / sums)

    velocities = np.random.rand(num_particles, n_points) * 0.01 - 0.005

    # Objective function
    def objective(p):
        return np.max(np.correlate(p, 1 - p, mode='full')) * dx

    # Initialize personal best and global best
    pbest = particles.copy()
    pbest_obj = np.array([objective(p) for p in pbest])
    gbest_idx = np.argmin(pbest_obj)
    gbest = pbest[gbest_idx].copy()
    best_c5 = pbest_obj[gbest_idx]

    # PSO loop
    while time.time() - start_time < budget_s:
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:
            break

        elapsed_frac = (time.time() - start_time) / budget_s
        w = 0.9 * (1 - elapsed_frac) + 0.4 * elapsed_frac

        r1 = np.random.rand(num_particles, 1)
        r2 = np.random.rand(num_particles, 1)

        velocities = (w * velocities + 
                      c1 * r1 * (pbest - particles) + 
                      c2 * r2 * (gbest - particles))
        velocities = np.clip(velocities, -v_max, v_max)

        particles += velocities
        particles = np.clip(particles, 0, 1)
        sums = particles.sum(axis=1, keepdims=True)
        particles = particles * (target_sum / sums)

        # Update objectives and track best solutions
        for i in range(num_particles):
            current_obj = objective(particles[i])
            if current_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = current_obj

        current_min = np.min(pbest_obj)
        if current_min < best_c5:
            gbest = pbest[np.argmin(pbest_obj)].copy()
            best_c5 = current_min

    # Final validation
    assert abs(np.sum(gbest) - target_sum) < 1e-6
    assert np.all((gbest >= -1e-6) & (gbest <= 1.0 + 1e-6))

    return (gbest.tolist(), best_c5, n_points)
