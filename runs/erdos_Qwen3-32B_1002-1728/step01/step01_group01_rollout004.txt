    alpha = learning_rate
    
    while time.time() - start_time < budget_s - 10:
        # Find the index maximizing the correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Compute gradient contribution for maximum correlation index
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            if 0 <= i_plus_k < n_points:
                grad[i] += (1.0 - h[i_plus_k])
            i_minus_k = i - k
            if 0 <= i_minus_k < n_points:
                grad[i] -= h[i_minus_k]
        
        # Gradient descent step
        h -= alpha * grad
        
        # Project onto box constraints [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Project onto sum constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if current_sum != 0:
            scaling = target_sum / current_sum
            h *= scaling
            h = np.clip(h, 0.0, 1.0)  # rescale may push some back to bounds
        
        # Evaluate current C5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # More aggressive decay when finding improvement
            if alpha < 0.1:
                alpha *= decay_rate
        else:
            # Less aggressive decay if no improvement
            if alpha < 0.1:
                alpha *= decay_rate ** 1.5
        
        # Introduce occasional perturbation
        if np.random.rand() < 0.05:
            perturb = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0, 1)
    
    return best_h, best_c5, n_points
