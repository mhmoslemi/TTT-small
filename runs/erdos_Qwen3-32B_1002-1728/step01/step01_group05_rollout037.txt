    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    # Define constraints for sum(h) = target_sum
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(lb=0.0, ub=1.0)
    
    # Perturbation settings
    perturb_amp = 0.3
    perturb_decay = 0.99
    maxiter_per_run = 100
    tolerance = 1e-6
    max_trials = 500
    
    while time.time() - start_time < budget_s:
        # Perturb current best solution
        perturbation = np.random.uniform(-perturb_amp, perturb_amp, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        
        # Normalize sum constraint and clip to [0,1] again
        perturbed_sum = np.sum(perturbed_h)
        if abs(perturbed_sum) > 1e-9:
            adj_perturbed_h = perturbed_h * (target_sum / perturbed_sum)
        else:
            adj_perturbed_h = np.zeros_like(perturbed_h)
        adj_perturbed_h = np.clip(adj_perturbed_h, 0.0, 1.0)
        
        # Local optimization
        try:
            result = minimize(lambda x: compute_c5(x, dx), 
                              x0=adj_perturbed_h, 
                              method='SLSQP', 
                              bounds=bounds, 
                              constraints=constraints,
                              options={'maxiter': maxiter_per_run, 'ftol': tolerance})
            if result.success:
                new_h = result.x
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except:
            pass  # Skip failed optimizations
        
        # Adaptive perturbation
        perturb_amp *= perturb_decay
        
        # Time management
        if n_points < 50 or time.time() - start_time > budget_s * 0.95:
            break  # Avoid excessive late-stage oscillation
            
    return (best_h, best_c5, n_points)
