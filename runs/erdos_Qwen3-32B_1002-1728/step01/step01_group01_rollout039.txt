        new_h = current_h - alpha * grad

        # Projection to [0, 1]
        new_h = np.clip(new_h, 0, 1)
        # Rescale to meet sum constraint
        total = np.sum(new_h)
        scale = (n_points / 2.0) / total if total > 1e-8 else 1.0
        new_h = new_h * scale
        # Re-clip in case scaling exceeds bounds
        new_h = np.clip(new_h, 0, 1)
        total = np.sum(new_h)
        new_h = new_h * (n_points / 2.0) / total if total > 1e-8 else new_h

        current_h[:] = new_h

        # Periodic perturbation
        if iteration % perturb_interval == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(current_h + perturbation, 0, 1)
            perturbed *= (n_points / 2.0) / np.sum(perturbed)
            if np.correlate(perturbed, 1 - perturbed, mode='full').max() * dx < best_c5:
                current_h[:] = perturbed

    return best_h, best_c5, n_points
