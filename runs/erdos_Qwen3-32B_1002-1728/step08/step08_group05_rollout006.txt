import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0

        s = np.sum(h_half)
        if s > 1e-6:
            h_half = (h_half / s) * (target_sum_full / 2)
        return h_half

    def compute_c5(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(corr)

    # Initialize population
    pop_size = 30
    pop = []
    fitness = []
    random.seed(seed)
    np.random.seed(seed)
    for _ in range(pop_size):
        h = h_full_initial[:half_n].copy() + np.random.normal(0, 0.1, half_n)
        h = repair_half(h)

        fitness.append(fit)
