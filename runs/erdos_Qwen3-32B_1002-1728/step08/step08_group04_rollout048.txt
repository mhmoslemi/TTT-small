import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum <= 1e-10:
            return h_initial.copy()
        h_repaired = h_clipped * (target_sum / h_sum)
        h_repaired = np.clip(h_repaired, 0, 1)
        return h_repaired

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    population_size = 200
    population = np.zeros((population_size, n_points))
    population[0] = repair(h_initial)
    for i in range(1, population_size):
        h_rand = np.random.rand(n_points)
        population[i] = repair(h_rand)

    fitness = np.array([compute_c5(population[i]) for i in range(population_size)])
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_individual = population[best_idx].copy()

    while time.time() - start_time < budget_s - 10:
        sorted_indices = np.argsort(fitness)
        num_parents = population_size // 2
        parent_indices = sorted_indices[:num_parents]
        parents = population[parent_indices]

        offspring = []
        offspring_needed = population_size - int(population_size * 0.05)
        for _ in range((offspring_needed + 1) // 2):
            p1, p2 = random.sample(parents, 2)
            cp = random.randint(10, n_points - 1)
            c1 = np.concatenate([p1[:cp], p2[cp:]])
            c2 = np.concatenate([p2[:cp], p1[cp:]])
            offspring.append(repair(c1))
            offspring.append(repair(c2))
        offspring = offspring[:offspring_needed]

        for i in range(len(offspring)):
            if random.random() < 0.3:
                mut_indices = np.random.choice(n_points, int(n_points * 0.3), replace=False)
                noise = np.random.normal(0, 0.1, size=len(mut_indices))
                mut = offspring[i].copy()
                mut[mut_indices] += noise
                mut = repair(mut)
                offspring[i] = mut

        offspring_fitness = np.array([compute_c5(child) for child in offspring])
        elite_count = int(population_size * 0.05)
        elite_indices = sorted_indices[:elite_count]
        elites = population[elite_indices]
        elite_fitness = fitness[elite_indices]

        new_pop = np.concatenate((elites, offspring))
        new_pop = new_pop[:population_size]
        new_fitness = np.concatenate((elite_fitness, offspring_fitness))
        new_fitness = new_fitness[:population_size]

        population = new_pop
        fitness = new_fitness

        current_best_idx = np.argmin(new_fitness)
        if new_fitness[current_best_idx] < best_c5:
            best_c5 = new_fitness[current_best_idx]
            best_individual = new_pop[current_best_idx].copy()

    h_full = best_individual.tolist()
    return (h_full, best_c5, n_points)
