            x_new = np.linspace(0, 2, n_points)
            h_old = (prev_best if not symmetric else 
                     np.concatenate([prev_best[:half_n], prev_best[:half_n][::-1]]))
            h_init = np.interp(x_new, x_old, h_old)
            h_init = np.clip(h_init, 0, 1)
        else:
            h_init = np.full(half_n, target_sum / n_points)
            h_full = (np.concatenate([h_init, h_init[::-1]]) if symmetric else 
                      h_init)
            h_full = np.clip(h_full, 0, 1)
            total = h_full.sum()
            if total != 0:
                h_full = (h_full * (target_sum / total))
            h_init = (h_full[:half_n] if symmetric else 
                      h_full)
        
        # PSO setup
        num_particles = 25
        max_iter_pso = 20
        positions = np.random.uniform(0, 1, (num_particles, half_n))
        velocities = np.zeros_like(positions)
        personal_best = positions.copy()
        p_best_score = np.full(num_particles, np.inf)
        g_best = positions[0].copy()
        g_best_score = np.inf
        velocity_max = 0.2
        
        def pso_objective(x):
            if symmetric:
                full = np.concatenate([x, x[::-1]])
            else:
                full = x.copy()
            full_clipped = np.clip(full, 0, 1)
            sum_full = full_clipped.sum()
            if sum_full == 0:
                return np.inf
            scaled = full_clipped * (target_sum / sum_full)
            scaled_clipped = np.clip(scaled, 0, 1)
            corr = np.correlate(scaled_clipped, 1 - scaled_clipped, mode='full') * dx
            return np.max(corr)
        
        for _ in range(max_iter_pso):
            for i in range(num_particles):
                pos = positions[i]
                score = pso_objective(pos)
                if score < p_best_score[i]:
                    p_best_score[i] = score
                    personal_best[i] = pos.copy()
                    if score < g_best_score:
                        g_best_score = score
                        g_best = pos.copy()
            # Update velocities
            inertia = 0.9
            cognitive = 1.5
            social = 1.5
            for i in range(num_particles):
                r1, r2 = np.random.rand(2)
                cog = cognitive * r1 * (personal_best[i] - positions[i])
                soc = social * r2 * (g_best - positions[i])
                velocities[i] = (inertia * velocities[i] + cog + soc)
                velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
                positions[i] += velocities[i]
                positions[i] = np.clip(positions[i], 0, 1)
        
        # Trust-Region refinement
        def tr_objective(x, is_sym=symmetric):
            if is_sym:
                full = np.concatenate([x, x[::-1]])
            else:
                full = x.copy()
            full_clipped = np.clip(full, 0, 1)
            sum_full = full_clipped.sum()
            if sum_full == 0:
                return np.inf
            scaled = full_clipped * (target_sum / sum_full)
            scaled_clipped = np.clip(scaled, 0, 1)
            corr = np.correlate(scaled_clipped, 1 - scaled_clipped, mode='full') * dx
            return np.max(corr)
        
        x0 = g_best.copy()
        bounds = [(0,1) for _ in range(half_n)]
        res = minimize(tr_objective, x0=x0, method='trust-constr', bounds=bounds, 
                       options={'maxiter': 30})
        x_opt = res.x if res.success else g_best
        
        # Compute final h
        h_full = (np.concatenate([x_opt, x_opt[::-1]]) if symmetric else 
                  x_opt)
        h_full = np.clip(h_full, 0, 1)
        sum_full = h_full.sum()
        if sum_full != 0:
            h_full = h_full * (target_sum / sum_full)
        h_full = np.clip(h_full, 0, 1)
        
        current_c5 = pso_objective(x_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full.tolist()
            prev_best = h_full
            n_points_best = n_points
    
    return (best_h, best_c5, n_points_best)
