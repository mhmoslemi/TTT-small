                    p1, p2 = parents[i], parents[i+1]
                    alpha = np.random.uniform(0.3, 0.7)
                    c1 = alpha * p1 + (1 - alpha) * p2
                    c2 = (1 - alpha) * p1 + alpha * p2
                    c1 = np.clip(c1, 0, 1)
                    c2 = np.clip(c2, 0, 1)
                    offspring.extend([c1, c2])
            
            # Mutation
            for i in range(len(offspring)):
                if np.random.rand() < 0.2:
                    o = offspring[i].copy()
                    for _ in range(int(n_points * 0.1)):
                        i1, i2 = np.random.randint(n_points, size=2)
                        delta = np.random.uniform(-0.1, 0.1)
                        if o[i1] + delta <= 1 and o[i1] + delta >= 0:
                            o[i1] += delta
                            o[i2] -= delta
                    offspring[i] = np.clip(o, 0, 1)
            
            # Replace pop
            combined = np.concatenate([pop, offspring])
            combined_fit = []
            for ind in combined:
                corr = np.correlate(ind, 1 - ind, mode='full') * dx
                combined_fit.append(np.max(corr))
            idxs = np.argsort(combined_fit)[:pop_size]
            pop = combined[idxs]
    
    assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
    return (best_h.tolist(), best_c5, len(best_h))
