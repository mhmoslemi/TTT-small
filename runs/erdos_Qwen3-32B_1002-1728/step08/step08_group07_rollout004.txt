                        h_pert.fill(target_sum / n_points)
                    else:
                        h_pert = h_pert * (target_sum / h_sum)
                        h_pert = np.clip(h_pert, 0, 1)
                    val_plus = np.correlate(h_pert, 1 - h_pert, mode='full')[lag_idx] * dx
                    h_pert = individual.copy()
                    h_pert[i] -= epsilon
                    h_pert = np.clip(h_pert, 0, 1)
                    h_sum = h_pert.sum()
                    if h_sum < 1e-9:
                        h_pert.fill(target_sum / n_points)
                    else:
                        h_pert = h_pert * (target_sum / h_sum)
                        h_pert = np.clip(h_pert, 0, 1)
                    val_minus = np.correlate(h_pert, 1 - h_pert, mode='full')[lag_idx] * dx
                    grad[i] = (val_plus - val_minus) / (2.0 * epsilon)
                individual -= 0.02 * grad
                individual = np.clip(individual, 0, 1)
                h_sum = individual.sum()
                if h_sum < 1e-9:
                    individual.fill(target_sum / n_points)
                else:
                    individual = individual * (target_sum / h_sum)
                    individual = np.clip(individual, 0, 1)
                curr_c5 = evaluate(individual)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = individual.copy()

    best_h_sum = best_h.sum()
    if abs(best_h_sum - target_sum) > 1e-6:
        best_h = (best_h / best_h_sum) * target_sum
        best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
