    while time.time() - start_time < budget_s - 10:
        mid_c = (low + high) / 2
        feasible = False

        h_half_var = cp.Variable(half_n)
        constraints = [0 <= h_half_var, h_half_var <= 1,
                       cp.sum(h_half_var) == target_sum_half]
        problem = cp.Problem(cp.Minimize(0), constraints)

        # Evaluate worst-case overlap for current best_h_half
        full_h = np.zeros(n_points)
        full_h[:half_n] = best_h_half
        full_h[-half_n:] = best_h_half[::-1]
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        lag_idx = np.argmax(corr)
        worst_c = corr[lag_idx]

        # If current best is better, skip
        if worst_c <= mid_c:
            high = mid_c
            continue

        # Focus on the worst lag and create a linearized constraint
        if lag_idx >= n_points:
            k = lag_idx - n_points + 1  # Convert to positive shift
        else:
            k = n_points - 1 - lag_idx  # Adjust for full correlation indices

        # Construct constraint for this specific k
        if k != 0 and k != n_points:
            sum_hhalf = 0.0
            for i in range(n_points):
                if i + k < n_points:
                    sum_hhalf += full_h[i] * full_h[i + k] * dx
            target_sum = (1 - mid_c) / dx
            # Add linearized constraint: approximate nonlinear constraint
            # as linear term based on current best_h_half
            contributions = []
            for i in range(half_n):
                # Positions in h_full for i in h_half
                # h_full[i] comes from h_half[i]
                for j in range(half_n):
                    if i + k < n_points:
                        if i + k < half_n:
                            weight = best_h_half[j]
                        else:
                            weight = best_h_half[::-1][i + k - half_n]
                        contributions.append(h_half_var[i] * weight)  
            # Add constraint sum(contributions) * dx >= 1 - mid_c
            constraints.append(sum(contributions) * dx >= 1 - mid_c)
            problem = cp.Problem(cp.Minimize(0), constraints)
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                feasible = True
                new_h_half = h_half_var.value
                if new_h_half is not None:
                    new_h_half = np.clip(new_h_half, 0, 1)
                    best_h_half = new_h_half.copy()
                    # Update worst overlap using new h
                    full_h_new = np.zeros(n_points)
                    full_h_new[:half_n] = best_h_half
                    full_h_new[-half_n:] = best_h_half[::-1]
                    corr_new = np.correlate(full_h_new, 1 - full_h_new, mode='full') * dx
                    new_best_c = np.max(corr_new)
                    if new_best_c < best_c5:
                        best_c5 = new_best_c
                        high = mid_c
                    else:
                        low = mid_c
                else:
                    low = mid_c
            else:
                low = mid_c
        else:
            low = mid_c

    # Final validation
    full_h_best = np.zeros(n_points)
    full_h_best[:half_n] = best_h_half
    full_h_best[-half_n:] = best_h_half[::-1]
    corr = np.correlate(full_h_best, 1 - full_h_best, mode='full') * dx
    best_c5_value = np.max(corr)
    return (full_h_best.tolist(), best_c5_value, n_points)
