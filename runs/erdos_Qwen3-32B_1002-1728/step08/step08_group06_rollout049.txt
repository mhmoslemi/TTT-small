            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_list[idx_best]
            
            if time.time() - start_time > budget_s:
                break
    
    if best_h is not None:
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values.tolist(), best_c5, len(initial_h_values))
