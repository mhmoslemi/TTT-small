    h_clipped = np.clip(h_full, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial half
    current_half = h_full_initial[:half_n].copy()
    current_half = np.clip(current_half, 0, 1)
    current_sum = current_half.sum()
    if current_sum != 0:
        current_half = (current_half / current_sum) * target_sum_half
    current_half = np.clip(current_half, 0, 1)

    # Initialize population
    population_size = 30
    population = [current_half.copy() for _ in range(population_size)]
    for i in range(population_size):
        population[i] = perturb_and_repair(population[i], half_n, target_sum_half)

    best_half = current_half.copy()
    best_c5 = evaluate_C5(best_half, half_n, n_points, dx)

    while time.time() - start_time < budget_s - 0.1:
        # Parent selection via tournaments
        parents = []
        for _ in range(population_size // 2):
            idxs = random.sample(range(population_size), 3)
            candidates = [population[i] for i in idxs]
            best_c = min(candidates, key=lambda h: evaluate_C5(h, half_n, n_points, dx))
            parents.append(best_c)

        # Generate offspring
        offspring = []
        for _ in range(population_size // 2):
            p1, p2 = random.sample(parents, 2)
            alpha = random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            child = perturb_and_repair(child, half_n, target_sum_half)
            offspring.append(child)

        # Replace the worst half
        for i in range(population_size // 2, population_size):
            population[i] = offspring[i - population_size // 2]

        # Update best solution
        for h in population:
            c5 = evaluate_C5(h, half_n, n_points, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_half = h.copy()

    # Final repair and return
    h_final = np.zeros(n_points)
    h_final[:half_n] = best_half
    h_final[-half_n:] = best_half[::-1]
    h_final = np.clip(h_final, 0, 1)
    h_sum = h_final.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_final = (h_final / h_sum) * target_sum_full
    return (h_final.tolist(), best_c5, n_points)
