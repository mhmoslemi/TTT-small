    else:

    h_first = np.clip(h_first, 0, 1)
    best_h_first = h_first.copy()
    best_c5 = float('inf')

    # Main optimization loop
    while time.time() - start_time < budget_s - 20:
        # Generate candidate by permuting and scaling segments
        k_segments = 5
        seg_len = half_n // k_segments
        if seg_len * k_segments != half_n:
            seg_len = half_n // k_segments  # should handle it, assuming divisible
        segments = [best_h_first[i*seg_len : (i+1)*seg_len] for i in range(k_segments)]
        permuted_indices = np.random.permutation(k_segments)
        permuted_segments = [segments[i] for i in permuted_indices]
        scaled_segments = []
        for seg in permuted_segments:
            scale = np.random.uniform(0.5, 1.5)
            scaled_seg = seg * scale
            scaled_seg = np.clip(scaled_seg, 0, 1)
            scaled_segments.append(scaled_seg)
        new_h_candidate = np.concatenate(scaled_segments)
        new_h_candidate = np.clip(new_h_candidate, 0, 1)
        sum_new = np.sum(new_h_candidate)
        if sum_new < 1e-9:
            new_h_candidate = np.full_like(new_h_candidate, target_sum_half / half_n)
        else:
            scaling_factor = target_sum_half / sum_new
            new_h_candidate *= scaling_factor
            new_h_candidate = np.clip(new_h_candidate, 0, 1.0)

        full_h_candidate = np.zeros(n_points)
        full_h_candidate[:half_n] = new_h_candidate

    best_full[:half_n] = best_h_first
    best_full[-half_n:] = best_h_first[::-1]
    best_full = np.clip(best_full, 0, 1)

        best_full = np.clip(best_full, 0, 1)
    final_corr = np.correlate(best_full, 1 - best_full, mode='full') * dx
    best_c5 = np.max(final_corr)
    return (best_full.tolist(), best_c5, n_points)
