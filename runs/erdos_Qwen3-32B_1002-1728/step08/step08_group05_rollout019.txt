def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Particle Swarm Optimization Setup
    num_particles = 30
    np.random.seed(seed)
    initial_half = h_full_initial[:half_n].copy()
    particles_positions = np.zeros((num_particles, half_n))
    particles_velocities = np.zeros((num_particles, half_n))
    for i in range(num_particles):
        particles_positions[i] = initial_half + 0.1 * np.random.randn(half_n)
    for i in range(num_particles):
        particles_positions[i] = repair_particle(particles_positions[i], target_sum_half)

    pbest_positions = particles_positions.copy()
    pbest_scores = np.array([compute_c5(p, half_n, target_sum_half, n_points, dx) for p in particles_positions])
    gbest_index = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_index].copy()
    gbest_score = pbest_scores[gbest_index]
    best_h = gbest_position.copy()
    best_c5 = gbest_score

    phi = 0.7
    mu = 1.5
    nu = 1.5

    while time.time() - start_time < budget_s - 0.1:
        for i in range(num_particles):
            # Velocity Update
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = mu * r1 * (pbest_positions[i] - particles_positions[i])
            social = nu * r2 * (gbest_position - particles_positions[i])
            particles_velocities[i] = phi * particles_velocities[i] + cognitive + social
            particles_positions[i] += particles_velocities[i]
            particles_positions[i] = repair_particle(particles_positions[i], target_sum_half)
            current_score = compute_c5(particles_positions[i], half_n, target_sum_half, n_points, dx)
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles_positions[i].copy()
        # Update global best
        current_gbest_index = np.argmin(pbest_scores)
        if pbest_scores[current_gbest_index] < best_c5:
            best_c5 = pbest_scores[current_gbest_index]
            best_h = pbest_positions[current_gbest_index].copy()
            gbest_index = current_gbest_index

def repair_particle(half_part, target_sum_half):
    half_part = np.clip(half_part, 0, 1)
    current_sum = half_part.sum()
    if current_sum != 0:
        half_part = (half_part / current_sum) * target_sum_half
    half_part = np.clip(half_part, 0, 1)
    return half_part

def compute_c5(half_part, half_n, target_sum_half, n_points, dx):
    h_full = np.zeros(n_points)
    h_full[:half_n] = half_part
    h_full[-half_n:] = half_part[::-1]
    h_clipped = np.clip(h_full, 0, 1)
    correlate_result = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    c5 = np.max(correlate_result) * dx
    return c5
