def evaluate_objective(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    target_sum = n_points / 2.0

        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum != 0:
            h = (h / current_sum) * target_sum
        h = np.clip(h, 0, 1)
        if abs(h.sum() - target_sum) > 1e-6:
            scale = target_sum / h.sum()
            h = h * scale
            h = np.clip(h, 0, 1)
        return h
    
    h_full_initial = repair_constraints(h_full_initial, target_sum)
    best_h = h_full_initial.copy()
