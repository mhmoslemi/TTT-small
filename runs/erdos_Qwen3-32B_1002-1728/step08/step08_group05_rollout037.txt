            velocities[i] = phi * velocities[i] + cognitive + social
            new_pos = particles[i] + velocities[i]
            
            new_pos = np.clip(new_pos, 0, 1)
            new_sum = new_pos.sum()
            if new_sum != 0:
                new_pos = (new_pos / new_sum) * target_sum_half
            new_pos = np.clip(new_pos, 0, 1)
            particles[i] = new_pos
            c5_new = evaluate_particle(new_pos, half_n, target_sum_half, n_points, dx)
            
            if c5_new < pbest_c5[i]:
                pbest_c5[i] = c5_new
                pbest_pos[i] = new_pos.copy()
                if c5_new < gbest_c5:
                    gbest_c5 = c5_new
                    gbest_pos = new_pos.copy()

    h_best_half = gbest_pos
    h_full = np.concatenate([h_best_half, h_best_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), gbest_c5, n_points)
