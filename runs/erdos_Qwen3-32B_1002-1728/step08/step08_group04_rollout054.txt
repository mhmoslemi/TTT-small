def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Precompute Chebyshev basis (even degrees only)
    x_values = np.linspace(0.0, 2.0, n_points, endpoint=True)
    z_values = x_values - 1.0  # map [0,2] to [-1,1]
    max_degree = 20
    coeff_indices = [k for k in range(0, max_degree+1, 2)]
    num_coeffs = len(coeff_indices)
    basis_matrix = np.zeros((n_points, num_coeffs))
    for i, k in enumerate(coeff_indices):
        basis_matrix[:, i] = np.polynomial.chebyshev.chebval(z_values, np.array([0]*k + [1]))

    # Project initial solution onto basis
    h_initial_clipped = np.clip(h_initial, 0, 1)
    h_initial_sum = h_initial_clipped.sum()
    if h_initial_sum > 1e-9:
        h_initial_clipped = (h_initial_clipped / h_initial_sum) * target_sum
    h_initial_clipped = np.clip(h_initial_clipped, 0, 1)
    current_coeff, _, _, _ = np.linalg.lstsq(basis_matrix, h_initial_clipped, rcond=None)
    current_coeff = current_coeff.flatten()
    best_coeff = current_coeff.copy()
    best_c5 = float('inf')

    # Adam optimizer initialization
    learning_rate = 0.01
    beta1, beta2 = 0.9, 0.999
    m, v = np.zeros_like(current_coeff), np.zeros_like(current_coeff)
    t_adam = 0
    grad_eps = 1e-5

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        t_adam += 1

        # Compute current h and C5
        def compute_h(coeff):

            h_clipped = np.clip(h_unclipped, 0, 1)
            h_sum = h_clipped.sum()
            if h_sum < 1e-10:
                return np.zeros_like(h_clipped)
            h_scaled = h_clipped * (target_sum / h_sum)
            return np.clip(h_scaled, 0, 1)

        h = compute_h(current_coeff)
        corr = np.correlate(h, 1 - h, mode='full')

            corr_p = np.correlate(h_p, 1 - h_p, mode='full')
