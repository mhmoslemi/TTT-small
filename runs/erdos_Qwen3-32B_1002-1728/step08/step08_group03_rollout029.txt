import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0  # Ensure full sum is n/2
    dx = 2.0 / n_points
    
    h_half_initial = h_full_initial[:half_n]
    # Normalize initial half if needed
    current_sum = h_half_initial.sum()
    if abs(current_sum - target_sum_half) > 1e-6:
        h_half_initial = np.clip((h_half_initial / current_sum) * target_sum_half, 0, 1)
    
    np.random.seed(seed)
    num_starts = 20  # Number of random starts
    best_c5 = float('inf')
    best_h_half = h_half_initial.copy()
    
    for _ in range(num_starts):
        # Generate perturbed initial guess or use original
        if np.random.rand() < 0.5 or _ == 0:
            current_h_half = h_half_initial.copy()
        else:
            # 5% perturbation
            current_h_half = h_half_initial + 0.05 * np.random.randn(half_n)
            current_sum = current_h_half.sum()
            current_h_half = np.clip(current_h_half, 0, 1)
            if current_sum > 1e-9:
                current_h_half = (current_h_half / current_sum) * target_sum_half
                current_h_half = np.clip(current_h_half, 0, 1)
        
        # Define the objective function for this optimization
        def objective(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            h_full = np.clip(h_full, 0, 1)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        # Define constraints and bounds
        A = np.ones((1, half_n))
        constraint = LinearConstraint(A, lb=target_sum_half, ub=target_sum_half)
        bounds = Bounds(0, 1)
        
        # Run optimization with SLSQP
        result = minimize(
            objective, 
            x0=current_h_half,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h_half = result.x.copy()
        
        # Exit early if time is up
        if time.time() - start_time >= budget_s:
            break
    
    # Construct the full solution
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
