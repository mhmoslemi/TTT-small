        h_data = np.clip(h_data, 0, 1).copy()
        current_sum = h_data.sum()
        if current_sum == 0:
            h_data += 0.5
        else:
            h_data = (h_data / current_sum) * target_sum_val
        h_data = np.clip(h_data, 0, 1)
        return h_data

        valid_len = n_points - k

        else:
            overlaps_k = 0
        overlaps.append(overlaps_k)
    best_c5 = max(overlaps)

    random.seed(seed)

        grad = np.zeros(n_points)
        k = max_shift

            grad[i] -= h[i - k] * dx

        h_save = h.copy()
