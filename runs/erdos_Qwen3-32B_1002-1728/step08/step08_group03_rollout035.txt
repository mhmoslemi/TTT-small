        if c5_scores[curr_best_idx] < best_c5:
            best_h_half = population[curr_best_idx].copy()
            best_c5 = c5_scores[curr_best_idx]

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0.0, 1.0)
    total = np.sum(h_full)
    if abs(total - target_sum_full) > 1e-6:
        h_full = h_full * (target_sum_full / max(1e-9, total))
        h_full = np.clip(h_full, 0.0, 1.0)
    
    return (h_full.tolist(), best_c5, n_points)
