import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum_full / current_sum if current_sum > 1e-9 else 0
    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)
    h_sym_initial = h_full_initial[:half_n].copy()

    def create_full_h(sym_var):
        return np.concatenate((sym_var, sym_var[::-1]))

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped.astype(np.float64), 1 - h_clipped.astype(np.float64), mode='full')
        return np.max(corr) * dx

    def project_and_normalize(x):
        x_clipped = np.clip(x, 0, 1)
        h_sum = x_clipped.sum()
        if h_sum > 1e-9:
            return x_clipped * (target_sum_half / h_sum)
        return x_clipped

    def objective(x, beta):
        h_sym = project_and_normalize(x)
        h_full = create_full_h(h_sym)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlaps = corr * dx
        exp_vals = np.exp(beta * overlaps)
        return (1 / beta) * np.log(exp_vals.sum())

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    def constraint(x):
        return project_and_normalize(x).sum() - target_sum_half

    constraints = ({'type': 'eq', 'fun': constraint},)
    best_c5 = np.inf
    best_h = h_sym_initial.copy()

    beta_schedule = [5.0, 10.0, 20.0, 50.0]
    for beta in beta_schedule:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
        max_iter = min(int(max(10, remaining_time / 2)), 100)

        def f(x): return objective(x, beta)

        result = minimize(f, x0=best_h, method='L-BFGS-B', bounds=bounds,
                          constraints=constraints, options={'maxiter': max_iter})
        if result.success:
            h_current = project_and_normalize(result.x)
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()

    # Final projection and validation
    h_full = create_full_h(best_h)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if h_sum > 1e-9:
        h_full = h_full * (target_sum_full / h_sum)
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
