import numpy as np

from scipy.optimize import minimize

    h = np.array(initial_h_values)
    n_points = len(h)

        scale_factor = target_sum / current_sum
        h = h * scale_factor

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(corr)

    best_h = h.copy()

    tabu_capacity = 20
    tabu_list = deque([best_h.copy()], maxlen=tabu_capacity)

    # Tabu search parameters

    patience = 100
    no_improvement = 0

    num_candidates = 50
    min_perturb_indices, max_perturb_indices = 5, 8

    while time.time() - start_time < 0.7 * budget_s:
        candidates = []
        for _ in range(num_candidates):
            candidate = best_h.copy()
            k = np.random.randint(min_perturb_indices, max_perturb_indices + 1)
            indices = np.random.choice(n_points, k, replace=False)
            for idx in indices:
                if np.random.rand() < 0.5:

                else:
                    candidate[idx] = 1 - candidate[idx]
            candidate = np.clip(candidate, 0, 1)
            sum_candidate = candidate.sum()
            if sum_candidate != 0:
                candidate *= target_sum / sum_candidate
            candidate = np.clip(candidate, 0, 1)
            candidates.append(candidate)

        valid_candidates = []
        for c in candidates:
            unique = True
