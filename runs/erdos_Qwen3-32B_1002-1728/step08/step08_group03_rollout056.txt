        h_half -= learning_rate * g_half
        h_half = np.clip(h_half, 0, 1.0)
        total = max(1e-9, h_half.sum())
        h_half = (h_half / total) * target_sum_half
        
        learning_rate *= decay
    
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1.0)
    
    total_sum = h_full.sum()
    if abs(total_sum - n_points / 2.0) > 1e-6:
        h_full = (h_full / total_sum) * (n_points / 2.0)
        h_full = np.clip(h_full, 0, 1.0)
    
    return (h_full.tolist(), best_c5, n_points)

def evaluate_c5(h_half, target_sum_half, dx):
    half_n = len(h_half)
    h_full = np.zeros(2 * half_n)
    h_full[:half_n] = h_half
    h_full[half_n:] = h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
