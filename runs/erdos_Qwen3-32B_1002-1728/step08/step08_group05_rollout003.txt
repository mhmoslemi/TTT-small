    target_sum = n_points / 2.0

    def repair_h(h_vec, target_sum):

        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        else:
            h[:] = 0.5  # All elements set to 0.5
        h = np.clip(h, 0, 1)
        return h

    h = repair_h(h_initial, target_sum)

        return np.max(correlation) * dx
