    def optimize(basis, init_coeffs, max_iter=1000):
        best = init_coeffs.copy()
        best_h = None
        best_c5 = float('inf')
        t0 = time.time()
        eps = 1e-3
        eta = 0.1
        
        while time.time() - t0 < budget_s:
            h = 0.5 * np.ones(n_points)
            for i in range(len(basis)):
                h += coeffs[i] * basis[i]
            h = np.clip(h, 0, 1)
            tot = max(np.sum(h) * dx, 1e-8)
            h = h / tot * 1.0

            # Gradient estimation
            grad = np.zeros_like(coeffs)
            baseline = evaluate_erdos_solution(h)
            max_step = min(5, len(basis))
            sampled_indices = np.random.choice(len(basis), max_step)
            
            for i in sampled_indices:
                old = coeffs[i]
                coeffs[i] += eps
                h_plus = 0.5 * np.ones(n_points)
                for j in range(len(basis)):
                    h_plus += coeffs[j] * basis[j]
                h_plus = np.clip(h_plus, 0, 1)
                tot_plus = max(np.sum(h_plus) * dx, 1e-8)
                h_plus = h_plus / tot_plus * 1.0
                val_plus = evaluate_erdos_solution(h_plus)
                
                coeffs[i] = old - eps
                h_neg = 0.5 * np.ones(n_points)
                for j in range(len(basis)):
                    h_neg += coeffs[j] * basis[j]
                h_neg = np.clip(h_neg, 0, 1)
                tot_neg = max(np.sum(h_neg) * dx, 1e-8)
                h_neg = h_neg / tot_neg * 1.0
                val_neg = evaluate_erdos_solution(h_neg)
                
                coeffs[i] = old
                grad[i] = (val_plus - val_neg) / (2 * eps)
            
            coeffs -= eta * np.clip(grad, -0.1, 0.1)
            
            # Repair and track progress
            h_new = 0.5 * np.ones(n_points)
            for i in range(len(basis)):
                h_new += coeffs[i] * basis[i]
            h_new = np.clip(h_new, 0, 1)
            tot_new = max(np.sum(h_new) * dx, 1e-8)
            h_new = h_new / tot_new * 1.0
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < best_c5:
                best = coeffs.copy()
                best_h = h_new.copy()
                best_c5 = c5_new
            
            eta *= 0.999
            if time.time() - t0 > budget_s - 60:
                break
        
        return best_h, best_c5

    # Multi-scale refinement
    h_coarse, c5_coarse = optimize(coarse_bases, coeffs)
    budget_used = time.time() - time.time()
    
    # Final evaluation
    if h_coarse is not None:
        return (h_coarse.tolist(), c5_coarse, n_points)
    else:
        return (h_initial.tolist(), evaluate_erdos_solution(h_initial), n_points)
