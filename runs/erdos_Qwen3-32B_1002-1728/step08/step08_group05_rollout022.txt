def run(seed=42, budget_s=1100, **kwargs):

        h_full_initial = (h_full_initial / h_sum) * target_sum_full

    if h_half_sum != 0:
        h_half = (h_half / h_half_sum) * target_sum_half
    h_half = np.clip(h_half, 0, 1)

    best_c5 = np.inf
    best_h_half = h_half.copy()
    np.random.seed(seed)
    learning_rate = 0.1
    delta = 1e-4
    subset_size = 10  # Process only a subset of indices to reduce computation

        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = np.clip(h_full, 0, 1)
        h_sum = h_full.sum()

            h_full *= (target_sum_full / h_sum)

        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_c5 = np.max(corr)

            best_h_half = h_half.copy()

        # Compute gradient over a subset of indices
        grad = np.zeros_like(h_half)
        indices = np.random.choice(half_n, size=min(subset_size, half_n), replace=False)
        for j in indices:
            # Perturb +delta
            h_plus = h_half.copy()
            h_plus[j] += delta
            h_plus = np.clip(h_plus, 0, 1)
            sum_p = h_plus.sum()
            scale_p = (target_sum_half / sum_p) if sum_p != 0 else 0
            h_plus = np.clip(h_plus * scale_p, 0, 1)

            h_half *= (target_sum_half / sum_h)

    final_c5 = np.max(corr)
