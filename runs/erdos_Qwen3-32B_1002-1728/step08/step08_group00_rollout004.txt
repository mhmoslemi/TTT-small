import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize PSO particles
    np.random.seed(seed)
    n_particles = 30
    particles = []
    velocities = []
    for _ in range(n_particles):
        h = h_initial.copy()
        noise = 0.1 * np.random.randn(n_points)
        h += noise
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(0.01 * np.random.randn(n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    # Objective function
    def objective(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Initialize personal bests and global best
    personal_best_positions = particles.copy()
    personal_best_scores = [objective(h) for h in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = particles[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    current_best_c5 = global_best_score

    # PSO parameters
    max_iter_pso = 1000
    c1, c2 = 2.0, 2.0
    w_max, w_min = 0.7, 0.4
    v_max, v_min = 0.5, -0.5

    # Run PSO
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.6:
            break
        w = w_max - (w_max - w_min) * iter_pso / max_iter_pso
        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        cognitive = c1 * r1 * (personal_best_positions - particles)
        social = c2 * r2 * (global_best_position - particles)
        velocities = w * velocities + cognitive + social
        velocities = np.clip(velocities, v_min, v_max)
        new_positions = particles + velocities
        new_positions_clipped = np.clip(new_positions, 0, 1)
        for i in range(n_particles):
            sum_h = new_positions_clipped[i].sum()
            if sum_h != 0:
                new_h_scaled = new_positions_clipped[i] * (target_sum / sum_h)
                new_h_clipped = np.clip(new_h_scaled, 0, 1)
                particles[i] = new_h_clipped
            else:
                particles[i] = np.full(n_points, target_sum / n_points)
        new_scores = [objective(h) for h in particles]
        for i in range(n_particles):
            if new_scores[i] < personal_best_scores[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = new_scores[i]
                if new_scores[i] < global_best_score:
                    global_best_position[:] = particles[i].copy()
                    global_best_score = new_scores[i]
                    current_best_c5 = global_best_score

    # Gradient-based refinement
    x0 = global_best_position.flatten()
    def objective_slsqp(x):
        x = x.reshape(-1)
        x_clipped = np.clip(x, 0, 1)
        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    result = minimize(

        options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}

    if not np.allclose(sum_h, target_sum, atol=1e-6):
        if sum_h != 0:
            h_final = h_final / sum_h * target_sum
        h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), c5_bound, n_points)
