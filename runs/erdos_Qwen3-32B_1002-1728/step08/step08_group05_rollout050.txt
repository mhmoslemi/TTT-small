    obj_exprs = []

    for k in range(n_points):
        corr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                term = (h[i] + (1 - h[j])) ** 2
            else:
                term = (h[i] + 1) ** 2
            corr += term * dx / 4
        obj_exprs.append(corr <= C5)
    
    constraints += obj_exprs
    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)

    # Solve with ECOS or other available solver
    problem.solve(solver=cp.ECOS, max_iter=100, warm_start=True)

    # Extract solution
    h_opt = np.clip(h.value, 0, 1)
    h_opt = (h_opt + h_opt[::-1]) / 2  # Enforce symmetry if violated due to solver
    h_opt = h_opt * (target_sum / h_opt.sum())  # Re-normalize

    # Compute actual C5 for validation
    corr_values = np.correlate(h_opt, 1 - h_opt, mode="full") * dx
    c5_bound = np.max(corr_values)

    return (h_opt.tolist(), c5_bound, n_points)
