import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution to be symmetric and satisfy constraints
    h_repaired = np.clip(h_full_initial, 0, 1)
    current_sum = h_repaired.sum()
    if current_sum != 0:
        h_repaired = (h_repaired / current_sum) * target_sum
    h_repaired = np.clip(h_repaired, 0, 1)

    for i in range(n_points):
        j = n_points - 1 - i
        if h_repaired[i] != h_repaired[j] and i < j:
            h_repaired[j] = h_repaired[i]

    # Define CVXPY variables and constraints
    h_full = cp.Variable(n_points)
    C5 = cp.Variable()

    constraints = [
        0 <= h_full,
        h_full <= 1,
        cp.sum(h_full) == target_sum
    ]

    # Add symmetry constraints
    for i in range(n_points // 2):
        constraints.append(h_full[i] == h_full[n_points - 1 - i])

    # Compute all valid shifts and their constraints
    max_k = n_points - 1
    for k in range(-max_k, max_k + 1):
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append((i, j))
        if not pairs:
            continue
        terms = []
        for i, j in pairs:
            terms.append((h_full[i] + 1 - h_full[j])**2)
        expr = cp.sum(terms) * dx / 4.0
        constraints.append(expr <= C5)

    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 0.1:
