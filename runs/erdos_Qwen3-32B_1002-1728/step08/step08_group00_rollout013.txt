import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Projection to feasible set: [0,1]^n and sum(h) = target_sum
    def project_to_feasible(x):
        h = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h - x))
        constraints = [0 <= h, h <= 1, cp.sum(h) == target_sum]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='SCIP', verbose=False)
        except:
            return x
        if prob.status in ['optimal', 'optimal_inaccurate'] and h.value is not None:
            return np.clip(h.value, 0, 1).copy()
        return np.clip(x, 0, 1)

    # Objective evaluated with projected h
    def objective_full(x):
        h_proj = project_to_feasible(x)
        corr_vals = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr_vals) * dx

    # DE bounds and constraints
    bounds_full = [(0.0, 1.0)] * n_points

    # Restart strategy with improved DE parameters
    best_h = h_full_initial.copy()
    best_c5 = objective_full(best_h)

    # Heuristic to reallocate budget preference
    budget_phases = {
        'DE_phase': 0.5,       # More budget for DE with improved strategy
        'local_slsqp_phase': 0.5
    }

    # Phase 1: Differential evolution with adaptive strategy
    de_maxiter = 30
    de_popsize = 25
    de_budget = budget_s * budget_phases['DE_phase']
    de_start_time = time.time()

    try:
        def de_objective(x):
            return objective_full(x)

        de_result = differential_evolution(
            de_objective, bounds_full, strategy='randtobest1', 
            maxiter=de_maxiter, popsize=de_popsize, tol=1e-7,
            mutation=(0.5, 1.0), recombination=0.95, seed=seed, workers=1
        )

        if de_result.success and de_result.fun < best_c5:
            best_h = project_to_feasible(de_result.x)
            best_c5 = objective_full(best_h)

    except Exception:
        pass  # fallback behavior to continue with current

    remaining_budget_s = budget_s - (time.time() - start_time)
    if remaining_budget_s < 1:
        final_h = project_to_feasible(best_h)
        final_c5 = objective_full(final_h)
        return (final_h.tolist(), final_c5, n_points)

    # Phase 2: Local SLSQP with random restarts and reduced merit

        perturbed = project_to_feasible(perturbed)
