    best_c5 = float('inf')
    best_h = None
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate candidates
        c5_scores = np.array([compute_c5(ind, dx) for ind in population])
        # Track current best
        current_min = np.min(c5_scores)
        idx_min = np.argmin(c5_scores)
        if current_min < best_c5:
            best_c5 = current_min
            best_h = population[idx_min].copy()
        
        # Select parents (top 50%)
        idx_sorted = np.argsort(c5_scores)
        parents_idx = idx_sorted[:population_size // 2]
        parents = [population[i] for i in parents_idx]
        
        # Generate new children through crossover and mutation
        children = []
        for _ in range(population_size // 2):
            p1, p2 = random.sample(parents, 2)
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, p1, p2).copy()
            # Mutation
            mutation_mask = np.random.rand(n_points) < 0.1
            child[mutation_mask] += np.random.normal(0, 0.05, n_points)[mutation_mask]
            child = clip_and_project(child, target_sum)
            children.append(child)
        
        # Local search refinement on top 2 parents
        for i in range(2):
            if i < len(parents):
                refined = local_search(parents[i], dx, target_sum)
                children.append(refined)
        
        # Replace population with top candidates
        combined = population + children
        combined_c5 = [compute_c5(ind, dx) for ind in combined]
        idx_sorted = np.argsort(combined_c5)
        population = [combined[i] for i in idx_sorted[:population_size]]
    
    # Final validation
    final_h = clip_and_project(best_h if best_h is not None else population[0], target_sum)
    final_c5 = compute_c5(final_h, dx)
    return (final_h.tolist(), final_c5, n_points)
