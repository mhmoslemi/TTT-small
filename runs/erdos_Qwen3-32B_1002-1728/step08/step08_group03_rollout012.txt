import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points

    h_half_initial = h_initial[:half_n].copy()
    total = h_half_initial.sum()
    h_half_initial *= target_sum_half / max(1e-9, total)

    np.random.seed(seed)
    num_starts = 15
    starts = np.zeros((num_starts, half_n))
    starts[0] = h_half_initial.copy()
    for i in range(1, num_starts):
        perturbed = h_half_initial + 0.05 * np.random.randn(half_n)
        perturbed = np.clip(perturbed, 0, 1.0)
        total = max(1e-9, perturbed.sum())
        perturbed = (perturbed / total) * target_sum_half
        starts[i] = perturbed

    def compute_c5(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = np.clip(h_full, 0, 1.0)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: sum(x) - target_sum_half},)
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)

    for idx, init in enumerate(starts):
        if time.time() - start_time > budget_s - 0.1:
            break
        result = minimize(
            compute_c5, init, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h_half = result.x

    h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    h_full = np.clip(h_full, 0, 1.0)
    total_sum = max(1e-9, h_full.sum())
    h_full = (h_full / total_sum) * target_sum_full

    return (h_full.tolist(), best_c5, n_points)
