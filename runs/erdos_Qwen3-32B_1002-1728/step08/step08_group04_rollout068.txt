                best_coeffs = coeffs.copy()
                best_h = new_h.copy()
                best_c5 = new_c5
        temp *= alpha

        # Periodic restart
        if random.random() < 0.01 and n_points < 1000:
            coeffs = repair_coefficients(np.fft.rfft(h_from_coefficients(coeffs)))

    final_h = h_from_coefficients(best_coeffs)
    final_c5 = compute_c5(final_h)
    return (final_h.tolist(), final_c5, n_points)
