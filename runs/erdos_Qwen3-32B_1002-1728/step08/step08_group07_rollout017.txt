            h_up = h_up * (target_sum_half / h_up_sum)
            h_up = np.clip(h_up, 0, 1)
            full_up = create_full_h(h_up)
            corr_up = np.correlate(full_up, 1 - full_up, mode='full') * dx
            val_up = corr_up[lag_idx]
            # -epsilon
            h_down = best_h_first.copy()
            h_down[i] = max(0.0, h_down[i] - epsilon)
            h_down_sum = h_down.sum()
            h_down = h_down * (target_sum_half / h_down_sum)
            h_down = np.clip(h_down, 0, 1)
            full_down = create_full_h(h_down)
            corr_down = np.correlate(full_down, 1 - full_down, mode='full') * dx
            val_down = corr_down[lag_idx]
            grad[i] = (val_up - val_down) / (2 * epsilon)
        # Update with scaled gradient
        learning_rate = learning_rate_start * (lr_decay ** step)
        best_h_first -= learning_rate * grad
        # Project to feasible region
        best_h_first = np.clip(best_h_first, 0, 1)
        sum_h = best_h_first.sum()
        if sum_h < 1e-10:
            best_h_first = np.zeros_like(best_h_first) + target_sum_half / len(best_h_first)
        else:
            best_h_first = (best_h_first / sum_h) * target_sum_half
        best_h_first = np.clip(best_h_first, 0, 1)

    # Final projection and evaluation
    best_full = create_full_h(best_h_first)
    best_full = np.clip(best_full, 0, 1)
    sum_full = best_full.sum()
    if abs(sum_full - target_sum_full) > 1e-6:
        best_full = (best_full / sum_full) * target_sum_full
        best_full = np.clip(best_full, 0, 1)
    corr = np.correlate(best_full, 1 - best_full, mode='full') * dx
    c5_bound = np.max(corr)
    return (best_full.tolist(), c5_bound, n_points)
