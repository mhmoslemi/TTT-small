    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_full_sum = n_points / 2.0

    # Repair initial h_full to meet constraints
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_full_sum) > 1e-6:
        h_full_initial = (h_full_initial / current_sum) * target_full_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_full_initial = (h_full_initial / h_full_initial.sum()) * target_full_sum  # Final normalization
    h_half = h_full_initial[:half_n].copy()
    
    # Initialize best solution
    best_c5 = np.inf
    best_h_half = h_half.copy()
    
    # Optimization parameters
    learning_rate = 0.05
    epsilon = 1e-5
    decay = 0.999
    
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate current h
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = np.clip(h_full, 0, 1)
        h_sum = h_full.sum()
        if h_sum != 0:
            h_full = (h_full / h_sum) * target_full_sum
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()
        
        # Compute gradient via finite difference
        grad = np.zeros(half_n)
        for i in range(half_n):
            h_perturbed = h_half.copy()
            h_perturbed[i] += epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_tmp = np.concatenate([h_perturbed, h_perturbed[::-1]])
            h_tmp = np.clip(h_tmp, 0, 1)
            h_sum_tmp = h_tmp.sum()
            h_tmp = (h_tmp / h_sum_tmp) * target_full_sum if h_sum_tmp != 0 else h_tmp
            corr_tmp = np.correlate(h_tmp, 1 - h_tmp, mode='full') * dx
            c_tmp = np.max(corr_tmp)
            grad[i] = (c_tmp - current_c5) / epsilon
        
        # Update h_half with gradient descent
        h_half -= learning_rate * grad
        h_half = np.clip(h_half, 0, 1)
        
        # Re-normalize the full h and extract new h_half
        h_full_tmp = np.concatenate([h_half, h_half[::-1]])
        h_full_tmp = np.clip(h_full_tmp, 0, 1)
        h_sum_tmp = h_full_tmp.sum()
        h_full_tmp = (h_full_tmp / h_sum_tmp) * target_full_sum if h_sum_tmp != 0 else h_full_tmp
        h_half = h_full_tmp[:half_n].copy()
        
        # Decay learning rate
        learning_rate *= decay
    
    # Prepare final result
    h_full_best = np.concatenate([best_h_half, best_h_half[::-1]])
    h_full_best = np.clip(h_full_best, 0, 1)
    h_sum_best = h_full_best.sum()
    h_full_best = (h_full_best / h_sum_best) * target_full_sum if h_sum_best != 0 else h_full_best
    
    return (h_full_best.tolist(), best_c5, n_points)
