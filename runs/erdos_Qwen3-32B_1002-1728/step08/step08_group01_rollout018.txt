import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution to strict constraints
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_full_initial *= target_sum_full / h_full_initial.sum()
    h_full_initial = np.clip(h_full_initial, 0, 1)
    
    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n

    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    np.random.seed(seed)
    
    # Adaptive global search with diverse strategies
    de_budget = budget_s * 0.45
    de_max_time = int(de_budget)
    de_popsize = 25
    de_maxiter = max(10, int((de_max_time * 1e6 / (n_points**2)) // de_popsize))
    
    strategies = [
        {'strategy': 'currenttobest1bin', 'mutation': (0.8, 1.0), 'recombination': 0.9},
        {'strategy': 'best2bin', 'mutation': (0.5, 0.9), 'recombination': 0.85}
    ]
    
    for strat in strategies:
        try:
            de_result = differential_evolution(
                objective, bounds, strategy=strat['strategy'],
                maxiter=de_maxiter//2, popsize=de_popsize,
                mutation=strat['mutation'], recombination=strat['recombination'],
                seed=seed, workers=1
            )
            if de_result.success and de_result.fun < best_c5:
                new_h = np.clip(de_result.x, 0, 1)
                new_h *= target_sum_half / new_h.sum()
                new_h = np.clip(new_h, 0, 1)
                if objective(new_h) < best_c5:
                    final_h_half, best_c5 = new_h, objective(new_h)
        except:
            pass

    # Penalized objective for constraint-free solvers
    def penalized_obj(x, alpha=1e6):
        h_full = create_full_h(x)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5 = np.max(corr) * dx
        current_sum = x.sum()
        return c5 + alpha * (current_sum - target_sum_half)**2

    while time.time() - start_time < budget_s - 20:
        perturb_idx = np.random.choice(half_n, min(10, half_n), replace=False)
        noise_scale = 0.3 * (1 - (time.time() - start_time)/budget_s)**1.2
        perturbed = final_h_half.copy()
        perturbed[perturb_idx] += np.random.normal(0, noise_scale, len(perturb_idx))
        perturbed = np.clip(perturbed, 0, 1)

        for method in ['COBYLA', 'Nelder-Mead']:
            try:
                res = minimize(
                    penalized_obj, x0=perturbed, method=method,
                    bounds=bounds, options={'maxiter': 200}
                )
                if res.success:
                    cand = np.clip(res.x, 0, 1)
                    new_c5 = objective(cand)
                    if new_c5 < best_c5:
                        final_h_half, best_c5 = cand, new_c5
            except:
                continue

    # Final constraint enforcement
    h_full = create_full_h(final_h_half)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
