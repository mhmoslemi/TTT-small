    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_half})

    # Bounds for h_half
    bounds = [(0, 1) for _ in range(half_n)]

    # Use a single SLSQP optimization run
    result = minimize(
        objective_func,
        h_half_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': 500,
            'ftol': 1e-10,
            'disp': False
        }
    )

    if result.success:
        best_h = result.x
        best_c5 = objective_func(result.x)
    else:
        # fallback to initial if optimization fails
        best_h = h_half_initial.copy()
        best_c5 = objective_func(best_h)

    # Validate and reconstruct h_full
    h_full = np.concatenate([best_h, best_h[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = np.sum(h_full)
    if not np.isclose(h_sum, target_sum_full, atol=1e-6):
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    # Return the best solution found
    final_c5 = objective_func(best_h)
    return (h_full.tolist(), final_c5, n_points)
