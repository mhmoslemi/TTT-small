from scipy.optimize import minimize, _constraints

import warnings

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    def repair_h(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        x = (x / s) * target_sum
        x = np.clip(x, 0, 1)
        return x

    h = repair_h(h_init)

    # Objective function
    def objective(x):
        x = repair_h(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints for local opt
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0.0, 1.0)] * n_points

    # PSO parameters
    np.random.seed(seed)
    n_particles = 30
    particles = repair_h(np.random.uniform(0, 1, (n_particles, n_points)))
    velocities = 0.2 * np.random.randn(n_particles, n_points)
    pbest_pos = particles.copy()
    pbest_obj = np.array([objective(p) for p in particles])
    gbest_pos = pbest_pos[np.argmin(pbest_obj)].copy()
    gbest_obj = np.min(pbest_obj)
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter_pso = 30
    vel_limits = [0.5, -0.5]

    # Time budget tracking
    total_pso_time = 0.6 * budget_s
    time_checkpoint = start_time + budget_s - 10

    # PSO main loop
    for i in range(max_iter_pso):
        if time.time() > time_checkpoint:
            break
        for j in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = c1 * r1 * (pbest_pos[j] - particles[j])
            social = c2 * r2 * (gbest_pos - particles[j])
            velocities[j] = w * velocities[j] + cognitive + social
            velocities[j] = np.clip(velocities[j], vel_limits[1], vel_limits[0])
            particles[j] += velocities[j]
            particles[j] = repair_h(particles[j])
            obj = objective(particles[j])
            if obj < pbest_obj[j]:
                pbest_obj[j] = obj
                pbest_pos[j] = particles[j].copy()
                if obj < gbest_obj:
                    gbest_obj = obj
                    gbest_pos = particles[j].copy().copy()
        # Safety adjustment: reduce inertia if over time
        elapsed = time.time() - start_time
        if 0.5 * elapsed >= total_pso_time:
            w = max(0.4, w - 0.05)

    # Gradient-based refinement
    h_refine = gbest_pos.copy()
    final_h = h_refine
    best_c5 = gbest_obj

    try:

            result = minimize(

                bounds=bounds, constraints=cons,

                    best_c5 = c5_candidate

    final_h = repair_h(final_h)
    return (final_h.tolist(), best_c5, n_points)
