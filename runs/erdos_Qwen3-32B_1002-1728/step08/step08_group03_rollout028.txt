import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0

    np.random.seed(seed)
    
    num_starts = 20
    best_c5 = float('inf')
    best_h_half = h_half_initial.copy()

        h_full[:half_n] = x
        h_full[-half_n:] = x[::-1]

    for _ in range(num_starts):
