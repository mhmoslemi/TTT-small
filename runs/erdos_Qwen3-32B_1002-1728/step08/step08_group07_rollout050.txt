def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    np.random.seed(seed)

        h_best = h_best / current_sum * target_sum
        h_best = np.clip(h_best, 0, 1)
    corr = np.correlate(h_best, 1 - h_best, mode='full') * dx
    best_c5 = np.max(corr)
    
    n_segments = 10
    segment_len = n_points // n_segments
    while time.time() - start_time < budget_s - 20:
        # Generate candidate by permuting and scaling segments
        segments = np.split(h_initial, n_segments)
        permuted_indices = np.random.permutation(n_segments)
        reordered_segments = [segments[i] for i in permuted_indices]
        scale_factors = np.random.uniform(0.8, 1.2, size=n_segments)

        h_candidate = np.concatenate(scaled_segments)

        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            h_best = h_candidate.copy()
            best_c5 = current_c5
            # Gradient descent refinement
            learning_rate = 0.02
            for _ in range(25):
