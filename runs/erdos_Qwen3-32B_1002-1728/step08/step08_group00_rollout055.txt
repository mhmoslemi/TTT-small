import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = (h / h_sum) * target_sum
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-6:
            h = (h / h_sum) * target_sum
            h = np.clip(h, 0, 1)
        return h

    h_full_initial = repair_h(h_full_initial)
    
    # Objective function with clipping
    def objective_full(x):
        return np.max(np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')) * dx

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    cons_full = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Best solution tracker
    final_h = h_full_initial.copy()
    best_c5 = float('inf')
    for idx in range(n_points):
        lag = idx - n_points + 1
        val = np.sum(final_h * (1 - np.roll(final_h, -lag)))
        best_c5 = min(best_c5, val * dx)
    
    # Global search with enhanced differential evolution
    np.random.seed(seed)
    popsize = 40  # Increased population size
    maxiter = 25  # Increased generations
    de_budget = budget_s * 0.7  # 70% time for DE

    try:
        de_result = differential_evolution(
            objective_full, bounds_full, strategy='best1bin',
            maxiter=maxiter, popsize=popsize, tol=1e-8,

            workers=1, callback=lambda x, c: time.time()-start_time < de_budget
        )
        if de_result.success:
            candidate = repair_h(de_result.x)
            candidate_c5 = objective_full(candidate)
            if candidate_c5 < best_c5:
                final_h, best_c5 = candidate, candidate_c5
    except:
        pass

    # Local refinement with systematic exploration
    while time.time()-start_time < budget_s-1:

        perturbed = repair_h(final_h + np.random.normal(0, noise_scale, n_points))
        if not np.isnan(perturbed).any():
            try:
                result = minimize(
                    objective_full, x0=perturbed, method='SLSQP',
                    bounds=bounds_full, constraints=cons_full,
                    options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}

                if result.success:
                    candidate = repair_h(result.x)
                    candidate_c5 = objective_full(candidate)

            except:
                pass

    # Final reconstruction from rolling correlations

    h_full_sum = h_final.sum()
    if abs(h_full_sum - target_sum) > 1e-6:

    return (h_final.tolist(), best_c5, n_points)
