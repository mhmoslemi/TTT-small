        new_population = top_parents.copy()
        
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(top_parents, size=2, replace=False)
            mask = (np.random.rand(n_points) < 0.5)
            child = mask * p1 + (1 - mask) * p2
            child = project(child)
            n_mutate = int(n_points * 0.1)
            mutate_idx = np.random.choice(n_points, n_mutate, replace=False)
            child[mutate_idx] += np.random.normal(0, 0.05, size=n_mutate)
            child = project(child)
            new_population.append(child)
        population = new_population
        
        top_locals = [x[1].copy() for x in fits[:2]]
        for h_lo in top_locals:
            for _ in range(5):
                corr = np.correlate(h_lo, 1 - h_lo, mode='full') * dx
                lag_idx = np.argmax(corr)
                grad = np.zeros(n_points)
                epsilon = 1e-5
                for i in range(n_points):
                    h_pert = h_lo.copy()
                    h_pert[i] += epsilon
                    h_pert = project(h_pert)
                    corr_p = np.correlate(h_pert, 1 - h_pert, mode='full') * dx
                    val_p = corr_p[lag_idx]
                    h_pert[i] -= 2 * epsilon
                    h_pert = project(h_pert)
                    corr_m = np.correlate(h_pert, 1 - h_pert, mode='full') * dx
                    val_m = corr_m[lag_idx]
                    grad[i] = (val_p - val_m) / (2 * epsilon)
                h_lo -= 0.02 * grad
                h_lo = project(h_lo)
            population.append(h_lo)
        population = [project(h) for h in population[:population_size]]
    
    best_h = project(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    return (best_h.tolist(), final_c5, n_points)
