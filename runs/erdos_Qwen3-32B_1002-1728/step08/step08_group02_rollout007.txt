import numpy as np
import time
from collections import deque

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = np.clip(h_half_initial, 0, 1)
    sum_half = np.sum(h_half_initial)
    if sum_half != 0:
        h_half_initial = (h_half_initial / sum_half) * target_sum_half
    h_half_initial = np.clip(h_half_initial, 0, 1)

    np.random.seed(seed)
    simplex_size = 20
    simplex = [h_half_initial.copy()]
    for _ in range(simplex_size - 1):
        perturbed = simplex[0].copy() + np.random.normal(0, 0.1, size=half_n)
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        if sum_p != 0:
            perturbed = (perturbed / sum_p) * target_sum_half
        else:
            perturbed.fill(target_sum_half / half_n)
        perturbed = np.clip(perturbed, 0, 1)
        simplex.append(perturbed)

    best_h_half = h_half_initial.copy()
    best_c5 = float('inf')

    tabu_list = deque(maxlen=50)

    reflection, expansion, contraction, shrinkage = 1.0, 2.0, 0.5, 0.5

    while time.time() - start_time < budget_s - 10:
        evaluations = []
        for s in simplex:
            h_full = np.concatenate([s, s[::-1]])
            d = 1.0 - h_full
            corr = np.correlate(h_full, d, mode='full')
            max_corr = np.max(corr) * dx
            evaluations.append(max_corr)
        sorted_indices = np.argsort(evaluations)
        simplex = [simplex[i] for i in sorted_indices]
        evaluations = [evaluations[i] for i in sorted_indices]
        worst_index = -1
        best_index = 0
        second_best_index = -2
        n_vertices = len(simplex)

        if best_c5 > evaluations[0]:
            best_c5 = evaluations[0]
            best_h_half = simplex[best_index].copy()

        centroid = np.mean(simplex[:-1], axis=0)

        reflected = centroid + reflection * (centroid - simplex[worst_index])
        reflected = np.clip(reflected, 0, 1)
        sum_reflected = np.sum(reflected)
        if sum_reflected != 0:
            reflected = (reflected / sum_reflected) * target_sum_half
        else:
            reflected.fill(target_sum_half / half_n)
        reflected = np.clip(reflected, 0, 1)
        reflected_rounded = tuple(np.round(reflected, 5))
        in_tabu = any(rounded == reflected_rounded for rounded in tabu_list)

        if not in_tabu:
            h_full = np.concatenate([reflected, reflected[::-1]])
            d = 1.0 - h_full
            corr = np.correlate(h_full, d, mode='full')
            reflected_c5 = np.max(corr) * dx
            worst_c5 = evaluations[worst_index]
            if reflected_c5 < worst_c5:
                if reflected_c5 < evaluations[second_best_index]:
                    expanded = centroid + expansion * (reflected - centroid)
                    expanded = np.clip(expanded, 0, 1)
                    sum_expanded = np.sum(expanded)
                    if sum_expanded != 0:
                        expanded = (expanded / sum_expanded) * target_sum_half
                    else:
                        expanded.fill(target_sum_half / half_n)
                    expanded = np.clip(expanded, 0, 1)
                    h_expanded = np.concatenate([expanded, expanded[::-1]])
                    d = 1.0 - h_expanded
                    corr_expanded = np.correlate(h_expanded, d, mode='full')
                    expanded_c5 = np.max(corr_expanded) * dx
                    if expanded_c5 < reflected_c5:
                        simplex[worst_index] = expanded
                        tabu_list.append(tuple(np.round(expanded, 5)))
                    else:
                        simplex[worst_index] = reflected
                        tabu_list.append(reflected_rounded)
                else:
                    simplex[worst_index] = reflected
                    tabu_list.append(reflected_rounded)
            elif reflected_c5 >= evaluations[worst_index]:
                contracted = centroid + contraction * (simplex[worst_index] - centroid)
                contracted = np.clip(contracted, 0, 1)
                sum_contracted = np.sum(contracted)
                if sum_contracted != 0:
                    contracted = (contracted / sum_contracted) * target_sum_half
                else:
                    contracted.fill(target_sum_half / half_n)
                contracted = np.clip(contracted, 0, 1)
                h_contracted = np.concatenate([contracted, contracted[::-1]])
                d = 1.0 - h_contracted
                corr_contracted = np.correlate(h_contracted, d, mode='full')
                contracted_c5 = np.max(corr_contracted) * dx
                if contracted_c5 < evaluations[worst_index]:
                    simplex[worst_index] = contracted
                    tabu_list.append(tuple(np.round(contracted, 5)))
                else:
                    for i in range(1, n_vertices):
                        simplex[i] = simplex[best_index] + shrinkage * (simplex[i] - simplex[best_index])
                        sum_s = np.sum(simplex[i])
                        if sum_s != 0:
                            simplex[i] = (simplex[i] / sum_s) * target_sum_half
                        else:
                            simplex[i].fill(target_sum_half / half_n)
                        simplex[i] = np.clip(simplex[i], 0, 1)
        else:
            simplex[worst_index] = simplex[worst_index].copy() + np.random.normal(0, 0.1, size=half_n)
            sum_s = np.sum(simplex[worst_index])
            if sum_s != 0:
                simplex[worst_index] = (simplex[worst_index] / sum_s) * target_sum_half
            else:
                simplex[worst_index].fill(target_sum_half / half_n)
            simplex[worst_index] = np.clip(simplex[worst_index], 0, 1)
            tabu_list.append(tuple(np.round(simplex[worst_index], 5)))

        if time.time() - start_time >= budget_s - 10:
            break

    h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    current_sum = np.sum(h_full)
    if abs(current_sum - target_sum_full) > 1e-6:
        h_full = h_full / current_sum * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    h_full = h_full.tolist()
    return (h_full, best_c5, n_points)
