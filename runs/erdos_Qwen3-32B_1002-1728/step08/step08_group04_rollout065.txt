import numpy as np

import random

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)

        h = np.clip(h, 0, 1)
        total = max(h.sum(), 1e-9)
        h = (h / total) * target_sum
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Repair initial solution
    h_initial = repair_individual(h_initial)

    # Genetic algorithm parameters
    pop_size = 150
    generations = 120
    mutation_rate = 0.25
    mutation_sigma = 0.1
    elitism_fraction = 0.05

    # Initialize population with best initial solution and random samples
    population = [h_initial]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair_individual(h)
        population.append(h)

    start_time = time.time()

    best_individual = h_initial.copy()

    for gen in range(generations):
