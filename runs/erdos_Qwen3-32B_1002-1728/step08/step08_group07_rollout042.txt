import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Ensure initial guess is feasible
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = np.clip(h_half_initial, 0, 1)
    current_sum = h_half_initial.sum()
    if current_sum != 0:
        h_half_initial = h_half_initial * (target_sum_half / current_sum)
        h_half_initial = np.clip(h_half_initial, 0, 1)

    def create_full_h(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    def objective(h_half_vec):
        h_half = h_half_vec.reshape(half_n)
        full_h = create_full_h(h_half)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        return np.max(corr)

    def compute_gradient(h_half_vec, max_lag):
        h_half = h_half_vec.reshape(half_n)
        eps = 1e-5
        grad = np.zeros(half_n)
        full_h = create_full_h(h_half)
        for i in range(half_n):
            # Compute perturbation in h_half[i]
            delta = np.zeros(half_n)
            delta[i] = eps
            # Forward
            h_pert = h_half + delta
            full_pert = create_full_h(h_pert)
            corr_pert = np.correlate(full_pert, 1 - full_pert, mode='full') * dx
            val_plus = corr_pert[max_lag]
            # Backward
            h_pert = h_half - delta
            full_pert = create_full_h(h_pert)
            corr_pert = np.correlate(full_pert, 1 - full_pert, mode='full') * dx
            val_minus = corr_pert[max_lag]
            grad[i] = (val_plus - val_minus) / (2.0 * eps)
        return grad

    def gradient(h_half_vec):
        h_half = h_half_vec.reshape(half_n)
        full_h = create_full_h(h_half)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        max_lag = np.argmax(corr)
        return compute_gradient(h_half_vec, max_lag)

    bounds = [(0, 1) for _ in range(half_n)]
    def sum_constraint(h_half_vec):
        return np.sum(h_half_vec.reshape(half_n)) - target_sum_half
    con = {'type': 'eq', 'fun': sum_constraint}

    # Optimization with L-BFGS-B using gradient
    result = minimize(
        objective,
        h_half_initial,
        method='L-BFGS-B',
        bounds=bounds,
        jac=gradient,
        constraints=con,
        options={'maxiter': 150}
    )

    # Final result processing
    h_best_half = result.x.reshape(half_n)
    h_best_full = create_full_h(h_best_half)
    corr = np.correlate(h_best_full, 1 - h_best_full, mode='full') * dx
    c5_bound = np.max(corr)

    # Final clipping and normalization
    h_best_full = np.clip(h_best_full, 0, 1)
    total_mass = h_best_full.sum()
    if abs(total_mass - n_points / 2) > 1e-6:
        h_best_full = (h_best_full / total_mass) * (n_points / 2)
        h_best_full = np.clip(h_best_full, 0, 1)

    return (h_best_full.tolist(), c5_bound, n_points)
