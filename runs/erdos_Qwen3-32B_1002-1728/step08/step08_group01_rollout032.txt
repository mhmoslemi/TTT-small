import numpy as np
import scipy.interpolate
from scipy.optimize import differential_evolution, minimize
import time

def upscale_h_sym(old_vals, old_len, new_len):
    x_old = np.linspace(0, 1, old_len)
    f = scipy.interpolate.interp1d(x_old, old_vals, kind='linear', fill_value="extrapolate")
    x_new = np.linspace(0, 1, new_len)
    return f(x_new)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    resolutions = [10, 20, 40, 70]
    best_h_sym = None
    best_c5 = float('inf')
    prev_h_sym = None

    for n_points in resolutions:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        dx = 2.0 / n_points
        target_sum_half = n_points / 4.0
        half_n = n_points // 2

        if (initial_h_values is not None and 
            len(initial_h_values) == n_points and 
            n_points == 70):
            h_full = np.array(initial_h_values)
            h_sym = h_full[:half_n].copy()
            h_sum = h_sym.sum()
            if h_sum > 1e-9:
                h_sym = np.clip((target_sum_half / h_sum) * h_sym, 0, 1)
            else:
                h_sym = np.full(half_n, target_sum_half / half_n)
        elif prev_h_sym is not None:
            old_len = len(prev_h_sym)
            new_vals = upscale_h_sym(prev_h_sym, old_len, half_n)
            current_sum = new_vals.sum()
            if current_sum > 1e-9:
                h_sym = np.clip((target_sum_half / current_sum) * new_vals, 0, 1)
            else:
                h_sym = np.full(half_n, target_sum_half / half_n)
        else:
            h_sym = np.full(half_n, target_sum_half / half_n)

        def objective(sym_vars):
            s = np.sum(sym_vars)
            if abs(s) < 1e-9:
                return float('inf')
            h = sym_vars / s * target_sum_half
            h_clipped = np.clip(h, 0, 1)
            h_full = np.concatenate((h_clipped, h_clipped[::-1]))
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(half_n)]

        def repair_h(sym_vars):
            sym_vars = np.clip(sym_vars, 0, 1)
            s = np.sum(sym_vars)
            if s == 0:
                return np.full_like(sym_vars, target_sum_half / half_n)
            return np.clip((target_sum_half / s) * sym_vars, 0, 1)

        # Differential evolution
        try:
            de_result = differential_evolution(
                objective, bounds, strategy='best2bin',
                popsize=20, mutation=(0.5, 1.0),
                recombination=0.9, seed=seed, maxiter=50,
                polish=False, workers=1, disp=False
            )
            if de_result.success:
                h_candidate = repair_h(de_result.x)
                c5_candidate = objective(h_candidate)
            else:
                h_candidate = h_sym
                c5_candidate = objective(h_candidate)
        except:
            h_candidate = h_sym
            c5_candidate = objective(h_candidate)

        # Local optimization
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h_sym = h_candidate.copy()

        trial_h = repair_h(best_h_sym)
        for _ in range(3):
            perturbed = trial_h + np.random.normal(0, 0.1, size=half_n)
            perturbed = repair_h(perturbed)
            res = minimize(
                objective, perturbed, bounds=bounds,
                method='L-BFGS-B', options={'maxiter': 50, 'ftol': 1e-10}
            )
            if res.success:
                new_c5 = objective(repair_h(res.x))
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h_sym = res.x

        prev_h_sym = best_h_sym

    final_n = n_points if 'n_points' in locals() else 70
    h_full = np.concatenate((best_h_sym, best_h_sym[::-1]))
    h_full = np.clip(h_full, 0, 1)
    s = h_full.sum()
    if s > 1e-9:
        h_full = (h_full / s) * (final_n / 2.0)
    h_full = np.clip(h_full, 0, 1)

    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * (2.0 / final_n)

    return (h_full.tolist(), final_c5, final_n)
