from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Initial solution repair
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()

    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)
    initial_h_sym = h_full_initial[:half_n]

    np.random.seed(seed)
    def create_full_h(h_sym):
        return np.concatenate((h_sym, h_sym[::-1]))

    def evaluate(individual):
        h_full = create_full_h(individual)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

        h_clipped = np.clip(individual, 0, 1)
        h_sum = h_clipped.sum()

            return np.zeros_like(individual)

    while time.time() - start_time < budget_s - 20:
        # Tournament selection
        parent_indices = []
        for _ in range(pop_size // 2):

            fit_vals = fitness[t_indices]
