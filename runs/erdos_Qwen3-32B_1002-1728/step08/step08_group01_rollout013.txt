import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points

    x_half = np.linspace(0, 1, half_n, endpoint=False)
    K = 6  # Number of Fourier basis terms
    basis = np.zeros((half_n, K))
    for j in range(K):
        basis[:, j] = np.cos(j * np.pi * x_half)

    # Compute initial coefficients using least squares
    c_initial, _, _, _ = np.linalg.lstsq(basis, h_full_initial[:half_n], rcond=None)
    c_initial = np.clip(c_initial, -10, 10)  # Wrap within DE bounds

    def objective(c):
        c = np.array(c, dtype=float)
        h_sym = basis @ c
        h_clipped = np.clip(h_sym, 0, 1)
        current_sum = np.sum(h_clipped) * dx
        if current_sum <= 1e-9:
            return 1e18
        scale = target_half_sum / current_sum
        h_scaled = h_clipped * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(-10, 10)] * K

    # Global exploration
    de_maxiter = max(20, int((budget_s * 0.6) // (K * half_n)))
    de_result = differential_evolution(
        objective, bounds, strategy='best2bin', maxiter=de_maxiter, 
        popsize=15, mutation=(0.5, 1.0), recombination=0.9, 
        seed=seed, workers=1, disp=False
    )

    best_c = de_result.x
    best_c5 = de_result.fun

    # Local refinement
    try:
        local_result = minimize(
            objective, x0=best_c, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if local_result.success and local_result.fun < best_c5:
            best_c = local_result.x

    except:

    h_sym_best = basis @ best_c
