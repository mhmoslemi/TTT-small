        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr) * dx

    # Differential evolution
    de_bounds = [(-2.0, 2.0)] * k_basis
    de_max_time = int(budget_s * 0.5)
    de_result = differential_evolution(
        objective_c, de_bounds, strategy='best1bin', popsize=20,
        mutation=(0.5, 1.0), recombination=0.9, seed=seed,
        polish=False, workers=1, maxiter=50
    )

    # Refine with local search
    best_c = de_result.x
    best_c5 = de_result.fun
    while time.time() - start_time < budget_s - 5:
        perturbed = best_c + np.random.normal(0, 0.1, size=k_basis)
        perturbed = np.clip(perturbed, -2.0, 2.0)
        for method in ['SLSQP', 'L-BFGS-B']:
            try:
                res = minimize(
                    objective_c, perturbed, method=method, bounds=de_bounds,
                    options={'maxiter': 200}
                )
                if res.success and res.fun < best_c5:
                    best_c = res.x
                    best_c5 = res.fun
            except:
                continue

    # Final h reconstruction
    h_half = Phi @ best_c
    h_half_clipped = np.clip(h_half, 0.0, 1.0)
    h_sum = h_half_clipped.sum()
    if h_sum > 1e-6:
        h_half_clipped *= target_sum_half / h_sum
        h_half_clipped = np.clip(h_half_clipped, 0.0, 1.0)
    h_full = np.concatenate([h_half_clipped, h_half_clipped[::-1]])

    return h_full.tolist(), best_c5, n_points
