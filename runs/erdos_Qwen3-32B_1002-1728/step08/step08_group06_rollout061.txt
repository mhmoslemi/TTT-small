import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 70  # Single high-resolution grid
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    half_n = n_points // 2
    target_sum_sym = target_sum_full / 2.0
    
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        sym_vars = h_initial[:half_n]
    else:
        # fallback symmetric initialization
        sym_vars = np.full(half_n, target_sum_sym / half_n)
    
    # Ensure initial guess satisfies sum constraint
    sym_sum = np.sum(sym_vars)
    if not np.isclose(sym_sum, target_sum_sym, atol=1e-8):
        if sym_sum > 0:
            sym_vars = sym_vars * (target_sum_sym / sym_sum)
            sym_vars = np.clip(sym_vars, 0.0, 1.0)
        else:
            raise ValueError("Invalid initial guess")
    
    def create_full_h(sym_vars):
        return np.concatenate([sym_vars, sym_vars[::-1]])
    
    def objective(sym_vars):
        h = create_full_h(sym_vars)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    bounds = Bounds(0.0, 1.0)
    sum_constraint = LinearConstraint(np.ones(half_n), lb=target_sum_sym, ub=target_sum_sym)
    
    # Trust-region optimization
    optimized = False
    best_h = None
    best_c5 = np.inf
    
    # First attempt: trust-constr with gradient estimation
    result = minimize(
        objective,
        x0=sym_vars,
        method='trust-constr',
        bounds=bounds,
        constraints=sum_constraint,
        options={
            'maxiter': 200,
            'gtol': 1e-8,
            'xtol': 1e-8,
            'barrier_tol': 1e-8
        }
    )
    
    if result.success:
        optimized = True
        opt_sym = result.x
        h_opt = create_full_h(opt_sym)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
        best_h = h_opt
        best_c5 = c5
    
    if time.time() - start_time > budget_s - 30 or best_c5 == float('inf'):
        remaining = max(0, budget_s - (time.time() - start_time))
        return (initial_h_values, best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
