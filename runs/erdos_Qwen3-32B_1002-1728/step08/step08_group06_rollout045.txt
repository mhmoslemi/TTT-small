import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    levels = [10, 20, 40, 70]
    best_c5 = np.inf
    best_h = None
    
    h_previous = None
    initial_h_values_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None

            break

        half_n = n_points // 2
        target_sum = n_points / 2.0
        target_sum_half = target_sum / 2.0
        
        if n_points == 70 and initial_h_values_given is not None:
            h_init = initial_h_values_given.copy()

            x_new = np.linspace(0, 2, n_points)
            from_prev = np.interp(x_new, h_previous['x'], h_previous['h'])
            from_prev = np.clip(from_prev, 0, 1)
            from_sum = from_prev.sum()

            h[:half_n] = np.clip(sym_vars, 0, 1)
            h[-half_n:] = h[:half_n][::-1]
            return h

            h_full = create_full_h(sym_vars)
            h_sum = h_full.sum()
            if h_sum == 0:

                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-8}
            )
            sym_best = result.x if result.success else h_init[:half_n]
        except:
            sym_best = h_init[:half_n]
        
        h_full = create_full_h(sym_best)
        h_sum = h_full.sum()
