            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            current_sum = mutant.sum()
            if current_sum == 0:
                mutant[0] = 1e-9
                current_sum = 1e-9
            mutant *= target_sum / current_sum
            mutant = np.clip(mutant, 0, 1)

            cross_points = np.random.rand(n_points) < CR
            if not np.any(cross_points):
                j_rand = np.random.randint(n_points)
                cross_points[j_rand] = True
            trial = population[i].copy()
            trial[cross_points] = mutant[cross_points]
            trial = np.clip(trial, 0, 1)
            trial_sum = trial.sum()
            if trial_sum == 0:
                trial[0] = 1e-9
                trial_sum = 1e-9
            trial *= target_sum / trial_sum
            trial = np.clip(trial, 0, 1)

            trial_c5 = evaluate_c5(trial, target_sum, dx)
            if trial_c5 < evaluate_c5(population[i], target_sum, dx):
                population[i] = trial
                if trial_c5 < best_c5:
                    best_h[:] = trial
                    best_c5 = trial_c5

    # Final validation
    h_result = best_h
    h_result = np.clip(h_result, 0, 1)
    h_norm = h_result / h_result.sum() * target_sum
    h_result = np.clip(h_norm, 0, 1)
    return (h_result.tolist(), best_c5, n_points)

def evaluate_c5(h_vector, target_sum, dx):
    h = np.clip(h_vector, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
