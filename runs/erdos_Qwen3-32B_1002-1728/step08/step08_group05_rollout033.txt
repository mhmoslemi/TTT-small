def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points

    # Repair initial h to be symmetric and normalized
    for i in range(half_n):
        j = n_points - 1 - i
        h_initial[j] = h_initial[i]
    sum_h = h_initial.sum()
    if sum_h != 0:
        h_initial = h_initial * (n_points / 2) / sum_h
    h_initial = np.clip(h_initial, 0, 1)

    # Define CVXPy variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    constraints = []

    # Symmetry constraints
    for i in range(half_n):
        j = n_points - 1 - i
        constraints.append(h[i] == h[j])

    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)

    # Generate overlap constraints
    shifts = np.arange(- (n_points - 1), n_points, dtype=int)
    for shift in shifts:
        if time.time() - start_time > budget_s - 0.1:
            break  # Exit if time is up

        valid_indices = []
        for i in range(n_points):
            if 0 <= i + shift < n_points:
                valid_indices.append(i)
        if not valid_indices:
            continue

        # Construct sum expression for this shift
        sum_expr = 0
        for i in valid_indices:
            j = i + shift
            sum_expr += (h[i] + 1 - h[j])**2 / 4

        # Add constraint sum * dx <= C5
        constraints.append(sum_expr * dx <= C5)

    # Solve the problem
    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)
