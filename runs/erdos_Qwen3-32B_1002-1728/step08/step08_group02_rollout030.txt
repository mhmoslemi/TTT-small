            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False},
            callback=callback
        )
    except StopIteration:
        result = scipy.optimize.OptimizeResult()
        result.success = False
        result.x = h_half_initial.copy()

    h_full_opt = None
    c5_bound = 0.380952  # Fallback value
    if result.success:
        h_half_opt = result.x
        h_full_opt = np.concatenate([h_half_opt, h_half_opt[::-1]])
        h_full_opt = np.clip(h_full_opt, 0, 1)
        current_sum = h_full_opt.sum()
        if abs(current_sum - target_sum_full) > 1e-6:
            h_full_opt = (h_full_opt / current_sum) * target_sum_full
            h_full_opt = np.clip(h_full_opt, 0, 1)
        c5_bound = objective(h_full_opt[:n_points//2])
    else:
        h_full_opt = h_full_initial

    return (h_full_opt.tolist(), c5_bound, n_points)
