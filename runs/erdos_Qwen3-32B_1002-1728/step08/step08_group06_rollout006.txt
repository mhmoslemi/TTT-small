                    corr = np.correlate(scaled_h, 1 - scaled_h, mode='full') * dx
                    fit = corr.max()
                fitness.append(fit)
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                full_opt = create_full_h(population[best_idx]) * (target_sum / create_full_h(population[best_idx]).sum())
                full_opt = np.clip(full_opt, 0, 1)
                best_h = full_opt
                assert np.isclose(best_h.sum(), target_sum)
            elite_count = 10
            elite_indices = np.argsort(fitness)[:elite_count]
            elites = [population[i] for i in elite_indices]
            new_population = elites.copy()
            while len(new_population) < pop_size:
                parent = population[np.random.choice(elite_indices)]
                child = parent + np.random.normal(0, 0.1, len(parent))
                child = np.clip(child, 0, 1)
                new_population.append(child)
            population = new_population
        best_idx = np.argmin(fitness)
        best_params = population[best_idx]
        full_es = create_full_h(best_params) * (target_sum / create_full_h(best_params).sum())
        full_es = np.clip(full_es, 0, 1)
        def objective(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full') * dx
            return corr.max()
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        try:
            result = minimize(objective, full_es, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 100, 'gtol': 1e-6})
            if result.success:
                h_local = np.clip(result.x, 0, 1)
                corr = np.correlate(h_local, 1 - h_local, mode='full') * dx
                c5_local = corr.max()
                if c5_local < best_c5:
                    best_c5 = c5_local
                    best_h = h_local
        except:
            pass
    if best_h is not None:
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), 0.38085, 70)
