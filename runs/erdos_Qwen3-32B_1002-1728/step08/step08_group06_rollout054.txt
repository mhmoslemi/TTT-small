                pos = np.tile(h_previous['h'][np.newaxis, :], (num_particles, 1))
                pos += 0.025 * np.random.randn(num_particles, 70)
                pos = np.clip(pos, 0, 1)
            vel = np.zeros((num_particles, 70))
        
        pbest_pos = pos.copy()
        pbest_score = np.array([np.inf] * num_particles)
        gbest_pos = pos[0].copy()
        gbest_score = np.inf
        
        def evaluate_particle(particle):
            if n_points < 70:
                h_full = np.zeros(n_points)
                h_full[:half_n] = particle
                h_full[-half_n:] = particle[::-1]
            else:
                h_full = particle.copy()
            total = h_full.sum()
            if total < 1e-8:
                return np.inf
            h_scaled = h_full * (target_sum / total)
            h_scaled = np.clip(h_scaled, 0, 1)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
            return np.max(corr)
        
        # Initialize best scores
        for i in range(num_particles):
            score = evaluate_particle(pos[i])
            if score < pbest_score[i]:
                pbest_score[i] = score
                pbest_pos[i] = pos[i].copy()
            if score < gbest_score:
                gbest_score = score
                gbest_pos = pos[i].copy()
        
        # PSO iterations
        for iter in range(max_pso_iter):
            if time.time() - start_time > budget_s - 10:
                break
            w_current = max(w * (w_decay / w) ** (iter/max_pso_iter), 0.4)
            r1, r2 = np.random.rand(), np.random.rand()
            
            vel = (w_current * vel 
                   + c1 * r1 * (pbest_pos - pos) 
                   + c2 * r2 * (gbest_pos[np.newaxis,:] - pos))
            vel = np.clip(vel, -max_vel, max_vel)
            pos += vel
            pos = np.clip(pos, 0, 1)
            
            # Update bests
            for i in range(num_particles):
                score = evaluate_particle(pos[i])
                if score < pbest_score[i]:
                    pbest_score[i] = score
                    pbest_pos[i] = pos[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest_pos = pos[i].copy()
        
        # Local refinement
        remaining = budget_s - (time.time() - start_time)
        if remaining > 30:
            if n_points < 70:
                def obj(sym_vars):
                    h_full = np.zeros(n_points)
                    h_full[:half_n] = sym_vars
                    h_full[-half_n:] = sym_vars[::-1]
                    total = h_full.sum()
                    if total < 1e-8:
                        return np.inf
                    h_scaled = h_full * (target_sum / total)
                    h_scaled = np.clip(h_scaled, 0, 1)
                    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
                    return np.max(corr)
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - (target_sum / n_points) * half_n * 2})
                bounds = [(0, 1)] * half_n
                res = minimize(obj, gbest_pos, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-8}, tol=1e-8)
                if res.success:
                    gbest_pos = res.x
                    gbest_score = obj(res.x)
            else:
                def obj_flat(h_flat):
                    total = h_flat.sum()
                    if total < 1e-8:
                        return np.inf
                    h_scaled = h_flat * (target_sum / total)
                    h_scaled = np.clip(h_scaled, 0, 1)
                    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
                    return np.max(corr)
                res = minimize(obj_flat, gbest_pos, bounds=[(0, 1)]*70, method='SLSQP',
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    gbest_pos = res.x
                    gbest_score = res.fun
        
        # Final h construction
        if n_points < 70:
            h_full = np.zeros(n_points)
            h_full[:half_n] = gbest_pos
            h_full[-half_n:] = gbest_pos[::-1]
        else:
            h_full = gbest_pos.copy()
        total = h_full.sum()
        if total > 1e-8:
            h_scaled = h_full * (target_sum / total)
        else:
            h_scaled = np.zeros_like(h_full)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
        c5_current = np.max(corr)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_scaled.copy()
        h_previous = {'x': np.linspace(0, 2, n_points), 'h': h_scaled}
    
    return (best_h.tolist(), best_c5, len(best_h))
