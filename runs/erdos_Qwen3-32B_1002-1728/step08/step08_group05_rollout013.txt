import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial repair: clip and scale to target sum
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != 0:
        h = (h / h_sum) * target_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_max_overlap(h, dx, n_points)

    alpha = 0.01
    start_time = time.time()
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate current k*
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-6:
            if h_sum != 0:
                h = (h / h_sum) * target_sum
            h = np.clip(h, 0, 1)

        s_full = np.correlate(h, 1 - h, mode='full')
        relevant_s = s_full[n_points - 1 : 2 * n_points - 1]
        current_s = relevant_s * dx
        max_idx = np.argmax(current_s)
        k_star = max_idx

        # Compute gradient for k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if i + k_star < n_points:
                term1 = 1.0 - h[i + k_star]
            term2 = 0.0
            if i - k_star >= 0:
                term2 = h[i - k_star]
            grad[i] = (term1 - term2) * dx

        # Update step
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = (h / h_sum) * target_sum
        h = np.clip(h, 0, 1)

        # Check new C5
        new_c5 = compute_max_overlap(h, dx, n_points)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            alpha *= 1.1  # Increase if improved
        else:
            alpha *= 0.9  # Decrease if not
            h = best_h.copy()  # Revert and retry

    return (best_h.tolist(), best_c5, n_points)

def compute_max_overlap(h, dx, n_points):
    s_full = np.correlate(h, 1 - h, mode='full')
    relevant_s = s_full[n_points - 1 : 2 * n_points - 1]
    return np.max(relevant_s * dx)
