import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    x_grid = np.linspace(0, 2, n_points, endpoint=False)
    z_grid = x_grid - 1  # map to [-1,1]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def legendre_p(k, z):
        if k == 0:
            return np.ones_like(z)
        elif k == 1:
            return z
        else:
            p_prev2 = np.ones_like(z)
            p_prev1 = z
            for i in range(2, k+1):
                p = ((2*i - 1) * z * p_prev1 - (i - 1) * p_prev2) / i
                p_prev2, p_prev1 = p_prev1, p
            return p_prev2

    # Generate basis for even orders 0, 2, 4,... up to 20
    orders = np.arange(0, 21, 2)  # max_order=20
    bases = np.array([legendre_p(k, z_grid) for k in orders])

    # Initialize coefficients
    if 'initial_h_values' in globals():
        h_target = h_initial - 0.5 * legendre_p(0, z_grid)
        cols = bases[1:].T
        coeffs, _, _, _ = np.linalg.lstsq(cols, h_target, rcond=None)
        coeffs = np.concatenate([[0.5], coeffs[0]])  # [offset] + coefficients
    else:
        coeffs = 0.5 + np.random.uniform(-0.1, 0.1, size=(len(orders)))

    best_coeffs = coeffs.copy()
    best_c5 = float('inf')

    learning_rate = 0.1
    lr_decay = 0.999
    eps = 1e-5
    iter = 0

    while time.time() - start_time < budget_s - 10:
        iter += 1
        # Compute current h
        h = np.sum(coeffs[:, None] * bases, axis=0)
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        scale = target_sum / sum_h if sum_h > 1e-9 else 1.0
        h_scaled = np.clip(h_clipped * scale, 0, 1)

        # Compute C5
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
