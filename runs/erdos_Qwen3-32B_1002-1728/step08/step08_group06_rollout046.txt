import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    best_c5 = np.inf
    best_h = None
    
    levels = [10, 20, 40, 70]
    initial_h_values_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None

    for n_points in levels:
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        half_n = n_points // 2

        # Initialize h
        if n_points == 70 and initial_h_values_given is not None:
            h_init = initial_h_values_given.copy()
        else:
            h_init = np.full(n_points, target_sum / n_points)
            h_init[:half_n] = h_init[:half_n]
            h_init[-half_n:] = h_init[:half_n][::-1]

        # Normalize to meet the integral constraint
        h_init = h_init * (target_sum / h_init.sum())
        h_init = np.clip(h_init, 0, 1)

        # Convex relaxation using CVXPY
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        constraints = [
            0 <= h, h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Formulate max constraint over k
        for k in range(1, half_n + 1):
            corr = cp.sum((h - 2 * cp.multiply(h, np.roll(h, -k))))
            constraints += [C5 >= corr * dx]
        
        problem = cp.Problem(cp.Minimize(C5), constraints)
        problem.solve(solver=cp.GLPK, max_iters=1000, verbose=False)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            relaxed_sol = h.value
        else:
            relaxed_sol = h_init

        # Project to feasible region
        def project(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            scale = target_sum / h_clipped.sum()
            return np.clip(h_clipped * scale, 0, 1)

        def objective(h_flat):
            h_vec = project(h_flat)
            corr = correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(corr)

        def constraint_sum(h_flat):
            return np.sum(h_flat) - target_sum

        bounds = [(0, 1)] * n_points

        # Gradient-based refinement
        result = minimize(
            objective,
            x0=relaxed_sol,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=({'type': 'eq', 'fun': constraint_sum},),
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        h_opt = project(result.x if result.success else relaxed_sol)
        corr = correlate(h_opt, 1 - h_opt, mode='full') * dx
        c5_current = np.max(corr)

        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_opt

        if time.time() - start_time > budget_s - 30:
            break

    # Final validation
    if best_h is not None:

        assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
