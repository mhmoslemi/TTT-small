    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum_full / current_sum if current_sum > 1e-9 else 0
    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)
    
    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    repair_cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    np.random.seed(seed)
    
    x = np.linspace(0, 1, half_n)
    n_basis = 5
    fourier_basis = generate_fourier_basis(x, n_basis)
    b = h_full_initial[:half_n]
    A = fourier_basis
    try:
        initial_coeffs, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
    except:
        initial_coeffs = np.zeros(n_basis)
    initial_coeffs = initial_coeffs.flatten()
    
    def objective_coeffs(coeffs):
        h_basis = A @ coeffs
        h_clipped = np.clip(h_basis, 0, 1)
        s = h_clipped.sum()
        if s <= 1e-9:
            return np.inf
        h_scaled = h_clipped * (target_sum_half / s)
        h_repaired = np.clip(h_scaled, 0, 1)
        h_full = create_full_h(h_repaired)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_coeffs = [(-2, 2)] * n_basis
    de_budget = budget_s * 0.4
    max_de_time = int(de_budget)
    de_popsize = 20
    de_maxiter = max(5, int(max_de_time / (n_basis * 10)))
    
    try:
        de_result = differential_evolution(
            objective_coeffs, bounds_coeffs, strategy='best2bin', maxiter=de_maxiter,
            popsize=de_popsize, mutation=(0.5, 1.0), recombination=0.85,
            seed=seed, workers=1, disp=False
        )
        if de_result.success:
            best_coeffs = np.clip(de_result.x, -2, 2)
            h_basis = A @ best_coeffs
            h_clipped = np.clip(h_basis, 0, 1)
            s = h_clipped.sum()
            if s > 1e-9:
                h_scaled = h_clipped * (target_sum_half / s)
                h_repaired = np.clip(h_scaled, 0, 1)
                candidate_c5 = objective_coeffs(best_coeffs)
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    final_h_half = h_repaired
    except:
        pass

    while time.time() - start_time < budget_s - 5:
        perturb_size = max(3, int(half_n * 0.05))
        perturb_indices = np.random.choice(half_n, perturb_size, replace=False)
        noise_scale = 0.25 * (1 - (time.time() - start_time)/budget_s) ** 0.8
        perturbed = final_h_half.copy()
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, size=perturb_size)
        perturbed = np.clip(perturbed, 0, 1)
        pert_sum = perturbed.sum()
        if abs(pert_sum - target_sum_half) > 1e-6 and pert_sum != 0:
            perturbed *= target_sum_half / pert_sum
            perturbed = np.clip(perturbed, 0, 1)
        
        for method in ['SLSQP', 'Nelder-Mead']:
            try:
                result = minimize(
                    objective, x0=perturbed, method=method,
                    bounds=bounds, constraints=repair_cons,
                    options={'maxiter': 200, 'ftol': 1e-12}
                )
                if result.success:
                    cand = np.clip(result.x, 0, 1)
                    new_c5 = objective(cand)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        final_h_half = cand
                        break
            except:
                continue

    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
