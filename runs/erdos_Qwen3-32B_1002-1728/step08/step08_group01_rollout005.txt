            f_p = smooth_objective(x_p, n_points, dx, beta_softmax)
            x_m = x.copy()
            x_m[i] -= epsilon_fd
            f_m = smooth_objective(x_m, n_points, dx, beta_softmax)
            grad[i] = (f_p - f_m) / (2 * epsilon_fd)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x_update = x - alpha * m_hat / (np.sqrt(v_hat) + epsilon_adam)
        x_update = np.clip(x_update, 0, 1)
        x_sum = x_update.sum()
        if abs(x_sum - target_sum_half) > 1e-6 and x_sum != 0:
            x_update = x_update * (target_sum_half / x_sum)
            x_update = np.clip(x_update, 0, 1)
        x = x_update

        curr_c5 = compute_max_overlap(x, n_points, dx)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_x = x.copy()

        if t % decay_iter == 0:
            beta_softmax *= beta_increase

    h_full = create_full_h(best_x)
    h_full = np.clip(h_full, 0, 1)
    sum_h = h_full.sum()
    if abs(sum_h - target_sum_full) > 1e-6:
        h_full = (h_full / sum_h) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
