import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 2.0 / 2.0

    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum_half * 2 / current_sum if current_sum > 1e-9 else 0
    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)
    
    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def objective(sym_vars):
        h_full = create_full_h(np.clip(sym_vars, 0, 1))
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    repair_cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    np.random.seed(seed)
    
    # Adaptive global search
    de_budget = budget_s * 0.45
    max_de_time = int(de_budget)
    de_popsize = 40
    de_maxiter = max(10, int((max_de_time * 1e6 / (n_points**2 * 1e4)) // de_popsize))
    
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=de_maxiter, 
            popsize=de_popsize, mutation=(0.7, 1.0), recombination=0.85, 
            seed=seed, workers=1, disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
            best_h_half *= target_sum_half / best_h_half.sum()
            best_h_half = np.clip(best_h_half, 0, 1)
            best_c5 = objective(best_h_half)
            final_h_half = best_h_half
    except:
        pass

    # Dynamic local optimization with multiple methods and larger perturbations
    while time.time() - start_time < budget_s - 5:
        progress = (time.time() - start_time) / budget_s
        perturb_size = max(5, int(half_n * (0.15 - 0.05 * progress)))
        perturb_indices = np.random.choice(half_n, perturb_size, replace=False)
        
        noise_scale = 0.3 * (1 - progress) ** 0.8
        perturbed = final_h_half.copy()
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, size=perturb_size)
        perturbed = np.clip(perturbed, 0, 1)
        
        pert_sum = perturbed.sum()
        if abs(pert_sum - target_sum_half) > 1e-6 and pert_sum != 0:
            perturbed *= target_sum_half / pert_sum
            perturbed = np.clip(perturbed, 0, 1)
        
        for method in ['SLSQP', 'L-BFGS-B', 'TNC']:
            try:
                result = minimize(
                    objective, x0=perturbed, method=method,
                    bounds=bounds, constraints=repair_cons,
                    options={'maxiter': 300, 'ftol': 1e-12}
                )
                if result.success:
                    cand = np.clip(result.x, 0, 1)
                    cand_sum = cand.sum()
                    if abs(cand_sum - target_sum_half) < 1e-6:
                        new_c5 = objective(cand)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            final_h_half = cand
                            break
            except:
                continue

    # Final projection and validation
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_half * 2) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_half * 2
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
