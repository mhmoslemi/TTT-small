def run(seed=42, budget_s=1100, **kwargs):

        if n_points == levels[0]:
            h_sym_initial = np.full(half_n, target_sum_half / half_n)
        else:

            prev_full_h = best_full_h
            prev_half = prev_full_h[:len(prev_full_h)//2]
            x_old = np.linspace(0, 1, len(prev_half))
            x_new = np.linspace(0, 1, half_n)
            h_sym_initial = np.interp(x_new, x_old, prev_half)
            h_sym_initial_sum = np.sum(h_sym_initial)
            if h_sym_initial_sum > 1e-9:
                h_sym_initial *= target_sum_half / h_sym_initial_sum
            h_sym_initial = np.clip(h_sym_initial, 0, 1)

        if n_points == 70 and 'initial_h_values' in globals():
            initial_h_sym = np.array(initial_h_values)[:half_n].copy()
            initial_sum = np.sum(initial_h_sym)
            if initial_sum > 1e-9:
                initial_h_sym *= target_sum_half / initial_sum
            initial_h_sym = np.clip(initial_h_sym, 0, 1)
            
            def make_full_h(hs):
                return np.concatenate((hs, hs[::-1]))
            
            def evaluate(h_sym):

                h_clipped = np.clip(h_full, 0, 1)
                corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
                return np.max(corr) * dx
            
            obj_current = evaluate(h_sym_initial)
            obj_initial = evaluate(initial_h_sym)
            if obj_initial < obj_current:
                h_sym_initial = initial_h_sym
        
        bounds = [(0.0, 1.0)] * half_n
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

            h_full = np.concatenate((sym_vars, sym_vars[::-1]))
            h_clipped = np.clip(h_full, 0, 1)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx

            x_sum = np.sum(x)
            if x_sum < 1e-9:
                return x
            return x * (target_sum_half / x_sum)
        
        # Adaptive DE parameters
        remaining_time = max(0, budget_s - (time.time() - start_time))

                )

                    c5_de = objective(h_sym_de)

                    if c5_de < best_c5:

                else:
                    h_sym_current = h_sym_initial
            except:
                h_sym_current = h_sym_initial
        else:
            h_sym_current = h_sym_initial

        while time.time() - start_time < budget_s - 5:
            for method in ['L-BFGS-B', 'SLSQP']:
                try:
                    res = minimize(
                        objective, h_sym_current, method=method,
                        bounds=bounds, constraints=constraints,
                        options={'maxiter': 100, 'ftol': 1e-10}
                    )
                    if res.success:
                        h_sym_opt = repair(res.x)
                        c5_opt = objective(h_sym_opt)
                        if c5_opt < best_c5:
                            best_c5 = c5_opt
                            h_sym_current = h_sym_opt
                except:
                    pass
                if time.time() - start_time > budget_s - 5:
                    break
        
        # Create full h for this level
        h_full = np.concatenate((h_sym_current, h_sym_current[::-1]))
        h_full = np.clip(h_full, 0, 1)
        h_sum = h_full.sum()
        if abs(h_sum - target_sum_full) > 1e-6 and h_sum != 0:
            h_full = (h_full / h_sum) * target_sum_full
            h_full = np.clip(h_full, 0, 1)
        
        best_full_h = h_full
    
    return (best_full_h.tolist(), best_c5, n_points)
