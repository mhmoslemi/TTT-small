import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Because ∫₀² h(x) dx = 1

    n_particles = 30
    max_velocity = 0.1
    cog = 1.5  # Cognitive
    soc = 1.5  # Social
    inertia_initial = 0.9
    inertia_final = 0.4
    np.random.seed(seed)

    def repair_position(pos, target=target_sum):
        pos = np.clip(pos, 0, 1)
        pos_sum = np.sum(pos)
        if abs(pos_sum - target) > 1e-8:
            scale = target / pos_sum
            pos = pos * scale
            pos = np.clip(pos, 0, 1)
        return pos

    # Initialize particles around initial_h_values with small perturbations
    positions = np.array([repair_position(h_initial + 0.05 * np.random.randn(n_points)) for _ in range(n_particles)])
    velocities = 0.01 * np.random.randn(n_particles, n_points)
    pbest = positions.copy()
    pbest_c5 = np.array([max(np.correlate(repair_position(p), 1 - repair_position(p), mode='full')) * dx for p in positions])
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    best_c5 = pbest_c5[gbest_idx]
    best_position = gbest.copy()

    while time.time() - start_time < budget_s - 0.1:
        t_remaining = budget_s - (time.time() - start_time)
        t_fraction = t_remaining / budget_s
        inertia = inertia_final + (inertia_initial - inertia_final) * t_fraction

        for i in range(n_particles):
            current_pos = positions[i]
            current_vel = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * current_vel + cog * r1 * (pbest[i] - current_pos) + soc * r2 * (gbest - current_pos)
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            new_pos = current_pos + velocities[i]
            positions[i] = repair_position(new_pos)

        for i in range(n_particles):
            current_pos = positions[i]
            h_clipped = repair_position(current_pos)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < pbest_c5[i]:
                pbest_c5[i] = current_c5
                pbest[i] = current_pos.copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_position = current_pos.copy()
        gbest_idx = np.argmin(pbest_c5)
        gbest = pbest[gbest_idx].copy()

    # Final repair
    h_final = repair_position(best_position)
    return (h_final.tolist(), best_c5, n_points)
