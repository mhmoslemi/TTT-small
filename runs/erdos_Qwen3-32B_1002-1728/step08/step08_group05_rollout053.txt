    pbest_c5 = np.array([evaluate_c5(p) for p in positions])
    gbest_idx = np.argmin(pbest_c5)
    gbest_position = positions[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]

    while time.time() - start_time < budget_s - 0.1:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        inertia = inertia_initial - (inertia_initial - inertia_final) * (elapsed / budget_s)
        inertia = max(inertia, 0.0)

        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest_positions[i] - positions[i]) +
                social * r2 * (gbest_position - positions[i])
            )
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            s = np.sum(positions[i])
            if s != 0:
                positions[i] = (positions[i] / s) * target_sum_half
            positions[i] = np.clip(positions[i], 0, 1)

            current_c5 = evaluate_c5(positions[i])
            if current_c5 < pbest_c5[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_c5[i] = current_c5
                if current_c5 < gbest_c5:
                    gbest_position = positions[i].copy()
                    gbest_c5 = current_c5

    h_final = np.zeros(n_points)
    h_final[:half_n] = gbest_position
    h_final[-half_n:] = gbest_position[::-1]
    h_final = np.clip(h_final, 0, 1)
    h_sum = h_final.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_final = (h_final / h_sum) * target_sum_full

    return (h_final.tolist(), gbest_c5, n_points)
