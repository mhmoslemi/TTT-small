            if c5 < best_c5:
                best_c5 = c5
                best_half = half.copy()
        
        # Sort and select top half
        pop_sorted = sorted(pop_fitness, key=lambda x: x[0])
        top_half = [ind[1] for ind in pop_sorted[:pop_size//2]]
        
        # Generate new population: top half + offspring
        new_population = top_half.copy()
        children_needed = pop_size - len(new_population)
        
        for _ in range(children_needed):
            p1, p2 = random.sample(top_half, 2)
            alpha = 0.3 + 0.4 * np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            # Mutation
            if random.random() < 0.2:
                scale_choice = random.randint(0, 2)
                if scale_choice == 0:  # Single-point
                    idx = random.randint(0, half_n - 1)
                    noise = random.gauss(0, 0.15)
                    child[idx] += noise
                elif scale_choice == 1:  # Block mutation
                    size = min(random.randint(2, 5), half_n)
                    start = random.randint(0, half_n - size)
                    noise = np.random.normal(0, 0.1, size)
                    child[start:start+size] += noise
                else:  # Global mutation
                    noise = np.random.normal(0, 0.05, half_n)
                    child += noise
                child = np.clip(child, 0, 1)
                child_sum = child.sum()
                if child_sum != 0:
                    child = (child / child_sum) * target_sum_half
                child = np.clip(child, 0, 1)
            new_population.append(child.copy())
        
        population = new_population
    
    # Final repair
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_half
    h_full[-half_n:] = best_half[::-1]
    h_clipped = np.clip(h_full, 0, 1)
    h_sum = h_clipped.sum()
    if abs(h_sum - n_points / 2) > 1e-6:
        h_clipped = (h_clipped / h_sum) * (n_points / 2)
    return (h_clipped.tolist(), best_c5, n_points)
