                    if valid_indices:
                        sum_expr = 0
                        for i in valid_indices:
                            j = i + k
                            sum_expr += h_relax[i] - m_relax[i, j]
                        constraints_relax += [C5_relax >= dx * sum_expr]
                
                prob_relax = cp.Problem(cp.Minimize(C5_relax), constraints_relax)
                prob_relax.solve(solver=cp.GLPK, verbose=False)
                
                if prob_relax.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax_vals = np.clip(h_relax.value, 0, 1)
                    h_relax_vals /= h_relax_vals.sum() * 2.0 / n_points  # Adjust to integral constraint
                    h_relax_vals = np.clip(h_relax_vals, 0, 1)
                    
                    # Symmetrize the solution
                    h_relax_half = h_relax_vals[:half_n]
                    h_relax_full = create_full_h(h_relax_half)
                    h_relax_full = np.clip(h_relax_full, 0, 1)
                    h_relax_full *= target_sum / h_relax_full.sum()
                    
                    corr_relax = np.correlate(h_relax_full, 1 - h_relax_full, mode='full') * dx
                    c5_relax = np.max(corr_relax)
                    
                    if c5_relax < best_c5:
                        best_c5 = c5_relax
                        best_h = h_relax_full
            except Exception as e:
                pass
        
        if time.time() - start_time > budget_s - 30:
            break
    
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6), (best_h.sum(), target_sum)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
