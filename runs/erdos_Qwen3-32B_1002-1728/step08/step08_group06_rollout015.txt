                h = np.clip(h, 0, 1)
                population.append(h)
        else:
            for _ in range(population_size):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                s = h.sum()
                if s == 0:
                    h = np.full(n_points, target_sum / n_points)
                else:
                    h = h * (target_sum / s)
                h = np.clip(h, 0, 1)
                population.append(h)

        while time.time() - start_time < budget_s - 30:
            fitness = []
            for h in population:
                corr_vals = np.correlate(h, 1 - h, mode='full') * dx
                fitness.append(np.max(corr_vals))
            current_idx = np.argmin(fitness)
            current_c5 = fitness[current_idx]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[current_idx].copy()

            tournament_size = 5
            selected_parents = []
            for _ in range(population_size // 2):
                contestant_indices = np.random.randint(0, population_size, tournament_size)
                contestants = [population[i] for i in contestant_indices]
                if not contestants:
                    continue
                selected = min(contestants, key=lambda c: next(f for idx, f in zip(range(population_size), fitness) if population[idx] is c))
                selected_parents.append(selected)

            new_population = population[:2]  # elitism: keep 2 best
            for i in range(0, min(len(selected_parents), population_size // 2), 2):
                p1 = selected_parents[i]
                p2 = selected_parents[i + 1] if (i + 1) < len(selected_parents) else p1
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * p1 + (1 - alpha) * p2
                if np.random.rand() < 0.2:
                    idx = np.random.choice(n_points, int(0.2 * n_points))
                    noise = np.random.normal(0, 0.1, size=len(idx))
                    child[idx] += noise
                child = np.clip(child, 0, 1)
                s = child.sum()
                if s != 0:
                    child = child * (target_sum / s)
                child = np.clip(child, 0, 1)
                new_population.append(child)
            population = new_population[:population_size]

            if time.time() - start_time > budget_s - 5:
                break

        if best_h is None or n_points == 70:
            best_h = np.array(population[np.argmin(fitness)]).copy()
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target_sum / best_h.sum())

    if best_h is not None:
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_given.tolist(), best_c5, len(initial_h_given))
