                    simplex[i] = np.clip(simplex[i], 0, 1)
                    objectives[i] = evaluate(simplex[i])

        # Update best
        current_best_idx = np.argmin(objectives)
        current_val = objectives[current_best_idx]
        if current_val < best_val:
            best_val = current_val
            best_half = simplex[current_best_idx].copy()

    # Finalize solution
    h_full = np.concatenate([best_half, best_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    total_sum = h_full.sum()
    if abs(total_sum - target_sum_full) > 1e-6:
        h_full = (h_full / total_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_val, n_points)
