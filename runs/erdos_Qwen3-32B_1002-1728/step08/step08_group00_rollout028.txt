    # Final local optimization with SLSQP
    def objective(x):
        return np.max(np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': constraint_sum},)
    final_h = best_h.copy()

    remaining_time = max(budget_s - (time.time() - start_time), 0)
    if remaining_time > 5:
        result = minimize(
            objective, final_h, method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 100, 'disp': False}
        )
        if result.success:
            improved = np.clip(result.x, 0, 1)
            improved_sum = improved.sum()
            if abs(improved_sum - target_sum) < 1e-6:
                new_c5 = calc_c5(improved)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    final_h = improved.copy()

    # Final constraint repair
    final_h = np.clip(final_h, 0, 1)
    final_sum = final_h.sum()
    if abs(final_sum - target_sum) > 1e-6:
        final_h = (final_h / final_sum) * target_sum
        final_h = np.clip(final_h, 0, 1)

    return (final_h.tolist(), best_c5, n_points)
