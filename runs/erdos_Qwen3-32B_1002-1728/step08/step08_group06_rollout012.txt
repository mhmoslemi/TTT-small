        half_n = n_points // 2
        target_sum = n_points / 2.0
        
        # Initialize current h
        if n_points == 70 and initial_h_values_given is not None:
            # Multiple initializations for 70-point case
            init_list = [initial_h_values_given.copy()]
            # Perturb initial guess
            perturbed = np.clip(initial_h_values_given + 0.05 * np.random.randn(n_points), 0, 1)
            perturbed = perturbed * (target_sum / perturbed.sum())
            init_list.append(perturbed)
            # Symmetric initialization
            base = np.full(n_points, target_sum / n_points)
            base[:half_n] = np.linspace(0, 1, half_n)
            base[-half_n:] = base[:half_n][::-1]
            base = base * (target_sum / base.sum())
            init_list.append(base)
        else:
            # Default initialization
            h_init = np.full(n_points, target_sum / n_points)
            h_init[:half_n] = np.linspace(0.5, 0, half_n)
            h_init[-half_n:] = h_init[:half_n][::-1]
            base = h_init * (target_sum / h_init.sum())
            init_list = [base]
        
        # Optimization for each initialization
        for init in init_list:
            def create_full_h(sym_vars):
                h = np.zeros(n_points)
                h[:half_n] = sym_vars
                h[-half_n:] = sym_vars[::-1]
                return h

            h_full = create_full_h(init[:half_n] if n_points == 70 else init)
            h_full = np.clip(h_full, 0, 1)
            total = h_full.sum()
            if total != 0:
                h_scaled = h_full * (target_sum / total)
            else:
                h_scaled = np.zeros_like(h_full)
            h_scaled = np.clip(h_scaled, 0, 1)

            def objective(sym_vars):
                sym_vars = np.clip(sym_vars, 0, 1)
                h_full = create_full_h(sym_vars)
                total = h_full.sum()
                if total == 0:
                    return np.inf
                h_scaled = h_full * (target_sum / total)
                h_scaled = np.clip(h_scaled, 0, 1)
                corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
                return np.max(corr)

            bounds = [(0.0, 1.0)] * half_n
            method = 'L-BFGS-B' if n_points == 70 else 'SLSQP'
            cons = ({'type': 'eq', 'fun': lambda vars: np.sum(vars) - (target_sum / 2) / 2} if n_points == 70 else [])
            maxiter = 200 if n_points == 70 else 100

            # Differential evolution setup
            remaining = max(0, budget_s - (time.time() - start_time))
            de_time_budget = min(int(remaining * 0.4), 300)
            maxiter_de = max(1, int(de_time_budget / 5))

            try:
                de_result = differential_evolution(
                    objective, bounds, strategy='best1bin',
                    maxiter=maxiter_de, popsize=20, mutation=(0.5, 1.0),
                    recombination=0.95, seed=seed, workers=1, tol=1e-6
                )
                de_vars = de_result.x if de_result.success else init[:half_n]
            except:
                de_vars = init[:half_n]

            try:
                # Local optimization with L-BFGS-B
                bounds_loc = Bounds(np.zeros(half_n), np.ones(half_n))
                res = minimize(
                    objective, x0=de_vars, method=method, bounds=bounds_loc,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9}
                )
                sym_best = res.x if res.success else de_vars
            except:
                sym_best = de_vars

            h_full = create_full_h(sym_best)
            total = h_full.sum()
            if total != 0:
                h_scaled = h_full * (target_sum / total)
            else:
                h_scaled = np.zeros_like(h_full)
            h_scaled = np.clip(h_scaled, 0, 1)

            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
            c5_current = np.max(corr)

            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_scaled

            # Early stopping
            if time.time() - start_time > budget_s - 60 and n_points == 70:
                break
        
        # Additional refinement for 70
        if n_points == 70 and best_h is not None:
            h_refine = best_h
            rem_time = max(0, budget_s - (time.time() - start_time))
            if rem_time > 30:
                bounds_ref = Bounds(0, 1)
                h_vec = h_refine.copy()
                def refine_obj(v):
                    v = v.reshape(n_points)
                    total = v.sum()
                    if total == 0:
                        return np.inf
                    v = np.clip(v, 0, 1) * (target_sum / total)
                    corr = np.correlate(v, 1 - v, mode='full') * dx
                    return np.max(corr)
                res = minimize(refine_obj, h_refine, method='SLSQP', bounds=bounds_ref, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
                if res.success:
                    h_refine = np.clip(res.x, 0, 1) * (target_sum / res.x.sum())
                    corr = np.correlate(h_refine, 1 - h_refine, mode='full') * dx
                    best_c5 = min(best_c5, np.max(corr))
                    best_h = h_refine

    # Final validation
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
