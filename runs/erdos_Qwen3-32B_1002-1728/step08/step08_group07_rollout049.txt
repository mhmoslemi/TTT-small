def create_full_h(h_half, n_points):
    half_n = n_points // 2
    h = np.zeros(n_points)
    h[:half_n] = h_half
    h[-half_n:] = h_half[::-1]
    return h

def split_segments(h_half, n_segments):
    length = len(h_half)
    base = length // n_segments
    remainder = length % n_segments
    segment_lengths = []
    for i in range(n_segments):
        seg_len = base + (1 if i < remainder else 0)
        segment_lengths.append(seg_len)
    segments = []

    for l in segment_lengths:
        segments.append(h_half[start:start+l])
        start += l
    return segments, segment_lengths
