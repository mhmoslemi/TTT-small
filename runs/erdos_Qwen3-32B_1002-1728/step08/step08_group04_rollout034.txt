def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h_current = np.clip(h_full_initial, 0, 1)
    h_current_sum = h_current.sum()
    if h_current_sum > 1e-9:
        h_current = (h_current / h_current_sum) * target_sum
    h_current = np.clip(h_current, 0, 1)
    
    # Compute initial FFT
    h_fft = fft.rfft(h_current)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()
    
    temp = 0.1
    alpha = 0.99
    max_iter = 10000
    iter_count = 0
    
    while (time.time() - start_time < budget_s - 10) and iter_count < max_iter:
        m = len(h_fft)
        k = np.random.randint(0, m)  # Select random frequency index
        
        # Perturb real part of selected coefficient
        delta = np.random.normal(0, temp * 0.1)
        h_fft_perturbed = h_fft.copy()
        
        if k == 0 or (k == m - 1 and n_points % 2 == 0):  # DC or Nyquist
            real_part = h_fft[k].real + delta
            h_fft_perturbed[k] = real_part
        else:
            real_part = h_fft[k].real + delta
            h_fft_perturbed[k] = complex(real_part, h_fft[k].imag)
        
        # Inverse FFT to get real time domain signal
        h_perturbed = fft.irfft(h_fft_perturbed, n=n_points, norm='backward')
        
        # Repair constraints
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_sum = h_perturbed.sum()
        if h_sum > 1e-9:
            h_perturbed = (h_perturbed / h_sum) * target_sum
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Recompute FFT
        h_fft_new = fft.rfft(h_perturbed)
        new_c5 = compute_c5(h_perturbed)
        current_c5 = compute_c5(h_current)
        
        # Simulated Annealing acceptance
        delta_c5 = new_c5 - current_c5
        if np.exp(-delta_c5 / max(temp, 1e-9)) > np.random.rand():
            h_current = h_perturbed
            h_fft = h_fft_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_perturbed.copy()
        
        # Cool down
        temp *= alpha
        iter_count += 1
        
        # Reset from best every 100 iterations
        if iter_count % 100 == 0:
            h_current = best_h.copy()
            h_fft = fft.rfft(h_current)
    
    h_full = best_h.tolist()
    return (h_full, best_c5, n_points)
