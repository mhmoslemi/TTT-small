                    best_c5 = current_c5

        # Update velocities and positions
        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (w * velocities[i] +
                            phi1 * r1 * (pbest_positions[i] - positions[i]) +
                            phi2 * r2 * (gbest_position - positions[i]))
            positions[i] += velocities[i]
            # Clip and scale
            positions[i] = np.clip(positions[i], 0, 1)
            scale = target_sum_half / np.clip(positions[i].sum(), 1e-9, None)
            positions[i] *= scale
            positions[i] = np.clip(positions[i], 0, 1)

    # Final validation
    best_full = create_full_h(best_h)
    best_full = np.clip(best_full, 0, 1)
    best_full_sum = best_full.sum()
    if abs(best_full_sum - n_points / 2) > 1e-6:
        best_full = (best_full / best_full_sum) * (n_points / 2)
        best_full = np.clip(best_full, 0, 1)

    return (best_full.tolist(), best_c5, n_points)
