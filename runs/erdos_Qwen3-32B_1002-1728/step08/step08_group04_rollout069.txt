def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Compute initial FFT coefficients with DC fixed
    coeffs = np.fft.rfft(h_initial)
    coeffs[0] = target_sum  # Ensure the DC term is correct for sum constraint

    def compute_c5(coeffs_r):
        c = coeffs_r.copy()
        c[0] = target_sum  # Reset DC if modified
        h = np.fft.irfft(c, n=n_points)
        # Clip to [0, 1]
        h_clipped = np.clip(h, 0, 1)
        # Adjust sum to match target_sum
        S = h_clipped.sum()
        if S != 0:
            adjustment = (target_sum - S) / n_points
            h_clipped += adjustment
        # Final clip to ensure [0, 1]
        h_clipped = np.clip(h_clipped, 0, 1)
        # Compute overlap
        h_full = h_clipped
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    current_coeffs = coeffs.copy()
    best_coeffs = current_coeffs.copy()
    best_c5 = compute_c5(best_coeffs)

    start_time = time.time()
    temp = 1.0
    alpha = 0.995

    while time.time() - start_time < budget_s - 10:
        # Perturb non-DC coefficients with both real and imaginary components
        non_dc = current_coeffs[1:]
        real = non_dc.real.copy()
        imag = non_dc.imag.copy()
        
        perturb_real = np.random.normal(0, 0.1, len(real))
        perturb_imag = np.random.normal(0, 0.1, len(imag))
        new_real = real + perturb_real
        new_imag = imag + perturb_imag
        
        new_coeffs = current_coeffs.copy()
        new_coeffs[1:] = new_real + 1j * new_imag

        # Compute energies
        new_c5 = compute_c5(new_coeffs)
        current_c5 = compute_c5(current_coeffs)

        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-9)):
            current_coeffs = new_coeffs.copy()
            if new_c5 < best_c5:
                best_coeffs = current_coeffs.copy()
                best_c5 = new_c5

        temp *= alpha

            current_coeffs = best_coeffs.copy()

    h_final += adjustment
    h_final = np.clip(h_final, 0, 1)
