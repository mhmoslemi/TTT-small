        # Steepest descent direction scaled by trust radius
        direction = -grad.copy()
        norm = np.linalg.norm(direction)
        if norm > 1e-8:
            direction = direction / norm * trust_radius
        else:
            direction = np.zeros_like(grad)

        step_size = 1.0
        h_proposal = current_h + direction * step_size
        h_proposal = project_h_half(h_proposal, target_sum_half)

        h_full_proposal = construct_h_full(h_proposal, n_points)
        new_C5 = evaluate_C5(h_full_proposal, dx)

        # Update based on success
        if new_C5 < best_C5:
            best_h_half[:] = h_proposal[:]
            best_C5 = new_C5
            trust_radius *= gamma_grow
        else:
            trust_radius *= gamma_shrink

        iteration += 1

    # Final h_full construction
    h_full = construct_h_full(best_h_half, n_points)
    return (h_full.tolist(), best_C5, n_points)

def compute_gradient(h_half, dx, target_sum_half, n_points):
    n_half = len(h_half)
    grad = np.zeros(n_half)
    delta = 1e-5
    base_h_full = construct_h_full(h_half, n_points)
    base_C5 = evaluate_C5(base_h_full, dx)
    for i in range(n_half):
        perturbed = h_half.copy()
        perturbed[i] += delta
        perturbed = project_h_half(perturbed, target_sum_half)
        test_h_full = construct_h_full(perturbed, n_points)
        test_C5 = evaluate_C5(test_h_full, dx)
        grad[i] = (test_C5 - base_C5) / delta
    return grad

def evaluate_C5(h_full, dx):
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx

def project_h_half(h_proposal, target_sum_half):
    h_proposal = np.clip(h_proposal, 0, 1)
    h_sum = h_proposal.sum()
    if h_sum > 1e-8:
        h_proposal = (h_proposal / h_sum) * target_sum_half
    else:
        h_proposal.fill(target_sum_half / len(h_proposal))
    return h_proposal

def construct_h_full(h_half, n_points):
    h_full = np.concatenate([h_half, h_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    target_sum = n_points / 2.0
    if h_sum > 1e-8:
        h_full = (h_full / h_sum) * target_sum
    else:
        h_full.fill(target_sum / len(h_full))
    h_full = np.clip(h_full, 0, 1)
    return h_full
