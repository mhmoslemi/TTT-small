    target_sum_half = target_sum_full / 2.0  # since full h is twice the half part

    def create_full_h(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    # Initialize particles
    n_particles = 20
    particles = []
    for i in range(n_particles):
        if i == 0:
            h_half = h_full_initial[:half_n].copy()
        else:
            h_half = h_full_initial[:half_n].copy() + np.random.uniform(-0.05, 0.05, size=half_n)
        h_half = np.clip(h_half, 0, 1)
        sum_h = np.sum(h_half)
        if sum_h < 1e-9:
            h_half = np.full_like(h_half, target_sum_half / half_n)
        else:
            h_half = h_half * (target_sum_half / sum_h)
        h_half = np.clip(h_half, 0, 1)
        particles.append(h_half)
    particles = np.array(particles)
    velocities = np.zeros_like(particles)

    # Evaluate initial particle C5 values
    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    personal_best_positions = particles.copy()
    personal_best_values = np.array([evaluate_c5(p) for p in particles])
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]

    phi_p = 2.0
    phi_g = 2.0
    max_stagnation = 100
    stagnation_count = 0

    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        prev_best = global_best_value
        omega = max(0.4, 0.9 - 0.5 * (iteration / 1000))  # Linear decay from 0.9 to 0.4
        for i in range(n_particles):
            # Update velocity and position
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = omega * velocities[i] + phi_p * r1 * (personal_best_positions[i] - particles[i]) + phi_g * r2 * (global_best_position - particles[i])
            particles[i] += velocities[i]
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h < 1e-9:
                particles[i] = np.full_like(particles[i], target_sum_half / half_n)
            else:
                particles[i] = particles[i] * (target_sum_half / sum_h)
            particles[i] = np.clip(particles[i], 0, 1)
            # Evaluate and update best
            current_value = evaluate_c5(particles[i])
            if current_value < personal_best_values[i]:
                personal_best_values[i] = current_value
                personal_best_positions[i] = particles[i].copy()
            if current_value < global_best_value:
                global_best_value = current_value
                global_best_position = particles[i].copy()
        # Stagnation check
        if global_best_value < prev_best:
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                break

    # Final validation and return
    best_h_full = create_full_h(global_best_position)
    best_h_full = np.clip(best_h_full, 0, 1)
    sum_h = np.sum(best_h_full)
    if abs(sum_h - target_sum_full) > 1e-6:
        best_h_full = (best_h_full / sum_h) * target_sum_full
        best_h_full = np.clip(best_h_full, 0, 1)
    return (best_h_full.tolist(), global_best_value, n_points)
