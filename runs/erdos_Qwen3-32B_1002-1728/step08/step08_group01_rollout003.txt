import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Initial h_sym repair
    h_sym = h_full_initial[:half_n].copy()
    h_sym = np.clip(h_sym, 0, 1)
    current_sum = np.sum(h_sym)
    h_sym = h_sym * (target_sum_half / current_sum) if current_sum > 1e-9 else 0.5 * np.ones_like(h_sym)
    h_sym = np.clip(h_sym, 0, 1)

    # Dynamic smoothing parameter
    beta_schedule = lambda t: 10.0 * (1 + t * 0.5)
    beta = 10.0

    def repair_h_sym(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 1e-9:
            x = x * (target_sum_half / s)
        return np.clip(x, 0, 1)

    def objective(x_flattened):
        nonlocal beta
        x = x_flattened.copy()
        x = repair_h_sym(x)
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        smooth_max = (1 / beta) * np.log(np.sum(np.exp(beta * corr)))
        return float(smooth_max)

    # Optimization with dynamic beta updating
    result = None
    remaining = budget_s - (time.time() - start_time)
    best_c5 = np.inf
    best_h = None

    while remaining > 10:
        beta = beta_schedule(1100 - remaining)
        current_h = repair_h_sym(h_sym)
        bounds = [(0.0, 1.0)] * half_n
        res = minimize(
            objective, 
            x0=current_h, 
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'eps': 1e-4}
        )
        if res.success:
            h_sym = repair_h_sym(res.x)
