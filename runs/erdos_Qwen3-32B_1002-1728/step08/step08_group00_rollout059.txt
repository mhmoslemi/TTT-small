            w = 0.9 - (0.5 * t_frac)
            r1, r2 = np.random.rand(2)
            cognitive = 2.0 * r1 * (personal_best_pos[i] - positions[i])
            social = 2.0 * r2 * (global_solution - positions[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            positions[i] += velocities[i]
            positions[i] = normalize_h(positions[i])
            sc = objective(positions[i])
            if sc < personal_best[i]:
                personal_best[i] = sc
                personal_best_pos[i] = positions[i].copy()
                if sc < global_best:
                    global_best = sc
                    global_solution = positions[i].copy()

    x0 = global_solution
    bounds = [(0.0, 1.0)] * n_points
    def objective_local(x):
        x = np.clip(x, 0, 1)
        s = x.sum()
        if s > 1e-10:
            x = (x / s) * target_sum
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    if time.time() - start_time < budget_s - 1e-9:
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = minimize(objective_local, x0=x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-12, 'iprint':-1})
        if res.success:
            x = res.x
            x = np.clip(x, 0, 1)
            s = x.sum()
            if s > 1e-10:
                x = (x / s) * target_sum
            x = np.clip(x, 0, 1)
            sc = objective_local(x)
            if sc < global_best:
                global_solution = x
                global_best = sc

    final_h = normalize_h(global_solution)
    final_c5 = objective(final_h)
    return (final_h.tolist(), final_c5, n_points)
