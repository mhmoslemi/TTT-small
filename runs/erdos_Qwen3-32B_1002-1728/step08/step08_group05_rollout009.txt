import numpy as np
import time
import random
import math

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Process initial half
    initial_half = h_full_initial[:half_n].copy()
    initial_half = np.clip(initial_half, 0, 1)
    initial_sum = initial_half.sum()
    initial_half = (initial_half / initial_sum) * target_sum_half
    initial_half = np.clip(initial_half, 0, 1)

    # Initialize population
    POP_SIZE = 30
    TOURN_SIZE = 3
    population = []
    for _ in range(POP_SIZE):
        p = initial_half.copy()
        p += np.random.normal(0, 0.1, size=p.shape)
        p = np.clip(p, 0, 1)
        p_sum = p.sum()
        if p_sum != 0:
            p = (p / p_sum) * target_sum_half
        p = np.clip(p, 0, 1)
        population.append(p)

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Evaluate initial population
    population_c5 = [(compute_c5(p), p) for p in population]
    population_c5.sort()
    population = [p for (c, p) in population_c5]
    best_c5, best_half = population_c5[0][0], population_c5[0][1]

    random.seed(seed)
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 0.1:
        # Select parents using tournament
        parents = population[:POP_SIZE // 2]
        new_population = parents.copy()

        # Generate offspring via crossover and mutation
        for _ in range(POP_SIZE - len(new_population)):
            # Tournament selection

            # Crossover

            # Mutation

                    noise = np.random.normal(0, 0.1, size)
                    child[start:start+size] += noise

                    noise = np.random.normal(0, 0.05, half_n)
                    child += noise
