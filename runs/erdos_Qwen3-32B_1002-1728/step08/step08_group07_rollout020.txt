            h_perturbed[i] = orig - 1e-5
            full_perturbed = create_full_h(h_perturbed)
            corr_minus = np.correlate(full_perturbed, 1 - full_perturbed, mode='full') * dx
            val_minus = corr_minus[lag_idx]
            grad[i] = (val_plus - val_minus) / (2.0 * 1e-5)
        return grad

    while time.time() - start_time < budget_s - 10:
        full_h = create_full_h(h_first)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_first = h_first.copy()
        lag_idx = np.argmax(corr)
        update_even = (step % 2 == 0)
        indices = np.where(np.arange(h_first.size) % 2 == (0 if update_even else 1))[0]
        grad = compute_gradient(h_first, lag_idx, indices)
        velocity[indices] = beta * velocity[indices] + grad[indices]
        h_first[indices] -= learning_rate * velocity[indices]
        # Clip and project
        h_first = np.clip(h_first, 0, 1)
        sum_h = h_first.sum()
        if sum_h < 1e-9:
            h_first = np.ones_like(h_first) * target_sum_half / h_first.size
        else:
            h_first = h_first * (target_sum_half / sum_h)
        h_first = np.clip(h_first, 0, 1)
        # Adjust beta for momentum
        beta += (step / (step + 1)) * (0.9 - beta) * 0.01
        # Adjust learning rate
        learning_rate *= 0.995
        step += 1

    # Final projection
    best_h_first = np.clip(best_h_first, 0, 1)
    best_h = create_full_h(best_h_first)
    best_h_sum = best_h.sum()
    if abs(best_h_sum - target_sum_full) > 1e-6:
        best_h = (best_h / best_h_sum) * target_sum_full
        best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
