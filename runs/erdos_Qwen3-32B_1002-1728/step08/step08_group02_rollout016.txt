k = i - (n - 1)

So for example, when i = n - 1, displacement is 0.

Thus, for the chosen k (based on the i that produced the max corr), compute for each x:

if x +k is in [0, n-1], then the derivative of the corr at that lag for h_full[x] is (1 - h_full[x +k]) * dx.

Else, it's 1 * dx.

def compute_gradient_at_k(h_full, k, dx):
    n = len(h_full)
    grad = np.zeros_like(h_full)
    for i in range(n):
        # x is index i
        x_plus_k = i + k
        if 0 <= x_plus_k < n:
            grad[i] = (1.0 - h_full[x_plus_k]) * dx
        else:
            grad[i] = 1.0 * dx
    return grad

But wait, this would compute the derivative of the correlation at lag k with respect to the h_full array. For example, for a given x, the contribution to the correlation sum is h[x] * (1 - h[x +k]). The derivative with respect to h[x] is (1 - h[x +k]) * dx, and for h[x +k] is -h[x] * dx. However, in the gradient for the correlation sum with respect to h_full, we are only considering the derivative with respect to h[x], which is the (1 - h[x +k]) * dx part. The other part, -h[x] * dx for the derivative with respect to h[x +k], is accounted for when considering other positions. However, in the gradient for the sum over x, we are only taking the derivative of the current lag k's sum.

This function correctly computes the gradient for that specific lag.

def line_search(h_half, grad_half, target_sum_half, dx, h_full_len, **kwargs):
    max_steps = 20
    step_size = 1.0
    alpha = 0.5  # Backtracking step size reduction
    beta = 0.1   # Sufficient decrease condition parameter

    for _ in range(max_steps):
        # Compute proposed h_half
        proposed_h_half = h_half - step_size * grad_half
        # Clip to [0, 1]
        proposed_h_half = np.clip(proposed_h_half, 0, 1)
        # Normalize sum to target_sum_half
        proposed_sum = proposed_h_half.sum()
        if proposed_sum == 0:
            proposed_h_half = np.full_like(proposed_h_half, target_sum_half / h_half.shape[0])
        else:
            proposed_h_half = (proposed_h_half / proposed_sum) * target_sum_half
        # Construct full h and compute new c5
        h_full = np.concatenate([proposed_h_half, proposed_h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        new_c5 = np.max(corr) * dx

        # Check if the step decreases c5 enough
        # using the sufficient decrease condition
        # |f(x + step * d) - f(x)| <= beta * |step * grad|
        # but since it's a line search, perhaps compare new_c5 to current_c5
        if new_c5 < current_c5:  # Or use more sophisticated condition
            return proposed_h_half, new_c5
        else:
            step_size *= alpha
    # If no step found, return the original h_half
    return h_half, current_c5

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0  # Adjusted for h_half sum

    # Initialize h_half
    h_half = h_full_initial[:n_half].copy()
    h_half = np.clip(h_half, 0, 1)
    current_sum = h_half.sum()
    h_half = (h_half / current_sum) * target_sum_half
    h_half = np.clip(h_half, 0, 1)

    # Best solution tracking
    best_h_half = h_half.copy()
    best_c5 = float('inf')

    # Main loop
    max_iter = 10000
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = np.clip(h_full, 0, 1)

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()

        # Find k_max_index
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)

        # Compute gradient at lag k
        grad_full = np.zeros_like(h_full)
        for i in range(n_points):
            x_plus_k = i + k
            if 0 <= x_plus_k < n_points:
                grad_full[i] = (1.0 - h_full[x_plus_k]) * dx
            else:
                grad_full[i] = 1.0 * dx

        grad_half = 2.0 * grad_full[:n_half]

        # Line search
        step_size = 1.0
        alpha = 0.5  # reduction factor
        beta = 0.8  # acceptance threshold
        for _ in range(20):
            proposed_h = h_half - step_size * grad_half
            proposed_h = np.clip(proposed_h, 0, 1)
            proposed_sum = proposed_h.sum()
            if proposed_sum == 0:
                proposed_h = np.full_like(proposed_h, target_sum_half / n_half)
            else:
                proposed_h = (proposed_h / proposed_sum) * target_sum_half
            h_full_proposed = np.concatenate([proposed_h, proposed_h[::-1]])
            corr_proposed = np.correlate(h_full_proposed, 1 - h_full_proposed, mode='full')
            new_c5 = np.max(corr_proposed) * dx

            if new_c5 < current_c5:
                # Accept the step
                h_half[:] = proposed_h
                break
            else:
                step_size *= alpha
        else:
            # No step accepted, try another method
            continue

        iter_count += 1

    # Final best solution
    h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
