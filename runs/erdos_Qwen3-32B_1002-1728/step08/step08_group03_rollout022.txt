        corr = np.correlate(h_full, 1 - h_full, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()

        max_corr_idx = np.argmax(corr)
        k_max = max_corr_idx - (n_points - 1)

        grad_full = np.zeros(n_points)
        for x in range(n_points):
            x_plus_k = x + k_max
            if 0 <= x_plus_k < n_points:
                term1 = 1 - h_full[x_plus_k]
            else:
                term1 = 0.0
            y = x - k_max
            if 0 <= y < n_points:
                term2 = h_full[y]
            else:
                term2 = 0.0
            grad_full[x] = (term1 - term2) * dx

        grad_half = (grad_full[:half_n] + grad_full[-half_n:][::-1]) / 2.0

        h_half -= learning_rate * grad_half
        h_half = np.clip(h_half, 0, 1.0)
        total = max(1e-9, h_half.sum())
        h_half = (h_half / total) * target_sum_half

        learning_rate *= decay_rate

    h_final = np.concatenate([best_h_half, best_h_half[::-1]])
    h_final = np.clip(h_final, 0, 1.0)
    h_final_sum = h_final.sum()
    if abs(h_final_sum - (n_points / 2.0)) > 1e-9:
        h_final = (h_final / h_final_sum) * (n_points / 2.0)
        h_final = np.clip(h_final, 0, 1.0)

    return (h_final.tolist(), best_c5, n_points)
