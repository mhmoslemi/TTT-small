    h_full_initial = np.clip(h_full_initial, 0, 1)

        scale_factor = target_sum / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    def objective_full(x):

        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')

    final_h = h_full_initial.copy()

    def de_full_objective(x):
        h_clipped = np.clip(x, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    de_budget = budget_s * 0.6
    start_de = time.time()
    popsize = 40  # Larger population
    maxiter = 50  # More iterations
    strategy = 'rand1bin'  # Different strategy

        perturbed = np.clip(perturbed, 0, 1)
        perturbed[perturbed < 1e-8] = 0.0
