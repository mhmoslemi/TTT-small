    h = h_initial.astype(np.float64)
    h_sum = h.sum()
    if h_sum > 1e-8:
        h = h * (target_sum / h_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Tabu search parameters
    TABU_DURATION = 5
    tabu_dict = {}
    max_iterations = 5000
    cooldown_step = 0.1
    intensity = 0.2

    # Objective function
    def objective(x):
        x = np.clip(x, 0, 1)
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx
