    best_half = population[0].copy()
    best_c5 = objective(best_half)

    while time.time() - start_time < budget_s - 30:
        evaluations = np.array([objective(ind) for ind in population])
        current_best_idx = np.argmin(evaluations)
        if evaluations[current_best_idx] < best_c5:
            best_c5 = evaluations[current_best_idx]
            best_half = population[current_best_idx].copy()

        parents = [tournament_selection(population, evaluations) for _ in range(population_size)]
        offspring = []

        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                c1 = crossover(p1, p2)
                c2 = crossover(p2, p1)
                c1 = mutate(c1, 0.1, 0.2)
                c2 = mutate(c2, 0.1, 0.2)
                offspring.append(repair(c1, target_sum_half))
                offspring.append(repair(c2, target_sum_half))

        top_indices = np.argsort(evaluations)[:max(1, population_size // 10)]
        new_pop = [population[idx] for idx in top_indices]
        offspring.sort(key=lambda x: objective(x))
        new_pop += offspring[:population_size - len(new_pop)]
        population = np.array(new_pop)

    try:
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum_half}]
        result = minimize(
            lambda x: objective(x), x0=best_half,
            method='SLSQP', bounds=[(0,1)]*len(best_half),
            constraints=cons, options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            refined = np.clip(result.x, 0, 1)
            s = refined.sum()
            if abs(s - target_sum_half) > 1e-6 and s > 1e-9:
                refined *= target_sum_half / s
                refined = np.clip(refined, 0, 1)
            refined_c5 = objective(refined)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_half = refined
    except:
        pass

    h_full = create_full_h(best_half)
    h_full = np.clip(h_full, 0, 1)
    s = h_full.sum()
    if abs(s - target_sum_full) > 1e-6 and s > 1e-9:
        h_full = (h_full / s) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
