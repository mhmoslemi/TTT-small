import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != 0:
        h = h / h_sum * target_sum
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # Gradient descent parameters
    alpha = 0.01

    while time.time() - start_time < budget_s - 0.1:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_star_idx = np.argmax(overlaps)
        k_star = k_star_idx - (n_points - 1)

        # Compute gradient for k_star
        i = np.arange(n_points)
        i_plus = i + k_star
        i_minus = i - k_star

        contrib1 = np.zeros_like(h, dtype=np.float64)
        valid_plus = np.logical_and(i_plus >= 0, i_plus < n_points)
        contrib1[valid_plus] = (1 - h[i_plus[valid_plus]]) * dx

        contrib2 = np.zeros_like(h, dtype=np.float64)
        valid_minus = np.logical_and(i_minus >= 0, i_minus < n_points)
        contrib2[valid_minus] = -h[i_minus[valid_minus]] * dx

        grad = contrib1 + contrib2

        # Gradient step
        h -= alpha * grad

        # Repair constraints
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = h / h_sum * target_sum

        if current_c5 < best_c5:

            best_h = h.copy()

        if current_c5 < best_c5:
            alpha *= 1.1
        else:

        # Safeguard against slow progress
        if alpha < 1e-5:
