    h_half *= (target_sum_half / max(1e-9, current_sum))
    h_half = np.clip(h_half, 0, 1.0)
    best_h_half = h_half.copy()
    best_c5 = float('inf')
    
    learning_rate = 0.01
    decay_rate = 0.998
    max_no_progress = 1000
    no_progress = 0
    iteration = 0
    
    while time.time() - start_time < budget_s:
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1.0)
        
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        k_corr_idx = np.argmax(corr)
        k_shift = k_corr_idx - (n_points - 1)
        
        grad_full = np.zeros(n_points)
        for x in range(n_points):
            x_plus_k = x + k_shift
            term1 = (1 - h_full[x_plus_k]) * dx if x_plus_k < n_points else 0.0
            x_minus_k = x - k_shift
            term2 = -h_full[x_minus_k] * dx if x_minus_k >= 0 else 0.0
            grad_full[x] = term1 + term2
        
        grad_half = grad_full[:half_n] + grad_full[half_n:][::-1]
        h_half -= learning_rate * grad_half
        h_half = np.clip(h_half, 0, 1.0)
        current_sum = h_half.sum()
        if current_sum > 1e-9:
            h_half = h_half * (target_sum_half / current_sum)
        h_half = np.clip(h_half, 0, 1.0)
        
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1.0)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()
            no_progress = 0
        else:
            no_progress += 1
        
        learning_rate *= decay_rate
        iteration += 1
        if no_progress > max_no_progress:
            break
    
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1.0)
    total_sum = h_full.sum()
    if abs(total_sum - n_points / 2.0) > 1e-6:
        h_full = h_full * (n_points / 2.0) / total_sum
        h_full = np.clip(h_full, 0, 1.0)
    
    return (h_full.tolist(), best_c5, n_points)
