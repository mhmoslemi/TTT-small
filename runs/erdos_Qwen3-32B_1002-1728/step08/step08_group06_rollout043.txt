        return np.max(corr)
    
    def gradient(h_flat):
        h_clipped = np.clip(h_flat, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        idx_max = np.argmax(corr)
        k_samples = idx_max - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k_samples < n_points:
                grad[j] += (1 - h_clipped[j + k_samples]) * dx
            if j - k_samples >= 0:
                grad[j] -= h_clipped[j - k_samples] * dx
        return grad
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result = minimize(
        objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    h_opt = np.clip(result.x, 0, 1)
    h_full = h_opt.copy()
    c5 = objective(h_full)
    
    # Ensure sum constraint is satisfied
    h_full = h_full * (target_sum / np.sum(h_full))
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), c5, n_points)
