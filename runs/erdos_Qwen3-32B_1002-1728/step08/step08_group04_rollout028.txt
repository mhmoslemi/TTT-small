        basis = np.zeros((n_points, n_basis))

            deg = 2 * i
            poly = scipy.special.legendre(deg)
            basis[:, i] = poly(t_samples)
        return basis

            h_clipped *= scale
            h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_basis(basis, c0, basis_type, max_iter=1000, alpha=0.1, decay=0.999):
