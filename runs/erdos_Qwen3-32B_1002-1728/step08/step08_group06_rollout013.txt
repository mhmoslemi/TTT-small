import numpy as np
from scipy.optimize import differential_evolution, minimize, Bounds
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    levels = [10, 20, 40, 70]
    best_c5 = np.inf
    best_h = None
    initial_h_values_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None

    for n_points in levels:
        dx = 2.0 / n_points
        half_n = n_points // 2
        target_sum = n_points / 2.0

        if n_points == 70 and initial_h_values_given is not None:
            h_init = initial_h_values_given.copy()
        else:
            if n_points == 10:
                h_init = np.tile([0.5, 0.5, 0.5, 0.5, 0.5], 2)
            else:
                if HPrevious:
                    x = np.linspace(0, 2, n_points)
                    h_interpolated = np.interp(x, HPrevious['x'], HPrevious['h'])
                    h_interpolated = np.clip(h_interpolated, 0, 1)
                    h_interpolated *= target_sum / h_interpolated.sum()
                else:
                    h_init = np.full(n_points, target_sum / n_points)
                    h_init[:half_n] = h_init[:half_n] * 0.9 + 0.05
                    h_init[-half_n:] = h_init[:half_n][::-1]

        def create_full_h(sym_vars):
            h = np.zeros(n_points)
            h[:half_n] = sym_vars
            h[-half_n:] = sym_vars[::-1]
            return h

        h_full_init = create_full_h(h_init[:half_n])
        h_full_init = np.clip(h_full_init, 0, 1)
        total = h_full_init.sum()
        if total != 0:
            h_full_init = h_full_init * (target_sum / total)
        h_full_init = np.clip(h_full_init, 0, 1)

        def objective(sym_vars):
            sym_vars = np.clip(sym_vars, 0, 1)
            h_full = create_full_h(sym_vars)
            total = h_full.sum()
            if total == 0:
                return np.inf
            h_scaled = h_full * (target_sum / total)
            h_scaled = np.clip(h_scaled, 0, 1)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
            return np.max(corr)

        bounds = Bounds(np.zeros(half_n), np.ones(half_n))
        cons = [{'type': 'eq', 'fun': lambda vars: np.sum(vars) - (target_sum / 2)}]
        de_time_budget = max(10, int((budget_s - (time.time() - start_time)) * 0.3))
        max_iter_de = int(de_time_budget / 10) if de_time_budget > 0 else 1

        try:
            de_result = differential_evolution(
                objective, bounds, strategy='best1bin',
                maxiter=max_iter_de, popsize=10, mutation=(0.5, 1.0),
                recombination=0.9, seed=seed, disp=False, workers=1
            )
            if not de_result.success:
                de_result.x = h_init[:half_n]
        except:
            de_result.x = h_init[:half_n]

        result_vals = [de_result.x]
        perfs = [objective(de_result.x)]
        multi_start = [de_result.x]
        if 'initial_h_values' in globals():
            init_vals = np.array(initial_h_values_given[:half_n])
            multi_start.append(init_vals)

        best_sym = np.argmin(perfs)

        def proj(h_sym):
            h_full = create_full_h(h_sym)
            total = h_full.sum()
            if total == 0:
                return np.clip(h_sym, 0, 1)
            h_full_proj = h_full * (target_sum / total)
            h_full_proj = np.clip(h_full_proj, 0, 1)
            return h_full_proj[:half_n]

        result = minimize(
            objective, x0=multi_start[best_sym], method='L-BFGS-B',
            bounds=bounds, jac=None, constraints=cons,
            options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 200}
        )
        sym_best = proj(result.x) if result.success else proj(de_result.x)

        h_full = create_full_h(sym_best)
        total = h_full.sum()
        if total != 0:
            h_scaled = h_full * (target_sum / total)
        else:
            h_scaled = np.zeros_like(h_full)
        h_scaled = np.clip(h_scaled, 0, 1)

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
        c5_current = np.max(corr)

        if not np.isfinite(c5_current):
            h_scaled = h_init.copy()
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
            c5_current = np.max(corr)

        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_scaled

        HPrevious = {'x': np.linspace(0, 2, n_points), 'h': h_scaled}

        if time.time() - start_time > budget_s - 30:
            break

    if best_h is None:
        best_h = initial_h_values_given.tolist()
        best_c5 = 0.38085

    return (best_h.tolist(), best_c5, len(best_h))
