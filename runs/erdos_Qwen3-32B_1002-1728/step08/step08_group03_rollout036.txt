import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points

    h_half_initial = h_full_initial[:half_n].copy()
    np.random.seed(seed)
    n_pop = 20
    F = 0.8
    CR = 0.9

    population = np.zeros((n_pop, half_n))
    population[0] = h_half_initial.copy()
    for i in range(1, n_pop):
        perturbed = h_half_initial + 0.05 * np.random.randn(half_n)
        perturbed = np.clip(perturbed, 0, 1.0)
        total = max(1e-9, perturbed.sum())
        perturbed = (perturbed / total) * target_sum_half
        population[i] = perturbed

    def evaluate(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    fitness = np.array([evaluate(pop[idx]) for idx in range(n_pop)])
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    while time.time() - start_time < budget_s:
        for i in range(n_pop):
            indices = list(range(n_pop))
            np.random.shuffle(indices)
            a, b, c = population[indices[:3]]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1.0)
            mutant_total = max(1e-9, mutant.sum())
            mutant = (mutant / mutant_total) * target_sum_half

            target = population[i]
            trial = np.copy(target)
            cross_mask = np.random.rand(half_n) < CR
            if not np.any(cross_mask):

            trial[cross_mask] = mutant[cross_mask]
