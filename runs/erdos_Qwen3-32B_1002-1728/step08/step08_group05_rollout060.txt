import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial half
    current_half = h_full_initial[:half_n].copy()
    current_half = np.clip(current_half, 0, 1)
    current_sum = current_half.sum()

        current_half = (current_half / current_sum) * target_sum_half
    current_half = np.clip(current_half, 0, 1)

    # Initialize population
    population_size = 30
    population = [current_half.copy() for _ in range(population_size)]
    for i in range(1, population_size):  # perturb all but the first individual
        half_pert = population[i].copy()
        perturbation = np.random.normal(0, 0.1, half_n)
        half_pert += perturbation
        # Repair
        half_pert = np.clip(half_pert, 0, 1)
        sum_pert = half_pert.sum()
        if sum_pert > 1e-9:

    def evaluate(half_arr):
        h_full = np.zeros(n_points)
        h_full[:half_n] = half_arr

    population = np.array(population)
