def repair(h, target_sum):

    total = h.sum()
    if total < 1e-9:
        h[:] = 0
        h[0] = target_sum

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    pop_size = 200
    generations = 250
    mutation_rate = 0.3
    mutation_sigma = 0.1

    def evaluate(ind):
        corr = np.correlate(ind, 1.0 - ind, mode='full')
        return np.max(corr) * dx

    population = []
    h_initial = np.array(initial_h_values).copy()
    population.append(repair(h_initial, target_sum))
    for _ in range(pop_size - 1):
        rand_h = np.random.uniform(0, 1, n_points)
        population.append(repair(rand_h, target_sum))

    best_individual = None
    best_c5 = float('inf')

    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break

        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_individual = population[best_idx].copy()

        sorted_indices = np.argsort(fitness)
        parent_count = int(pop_size * 0.5)

        elitism_count = int(pop_size * 0.05)
        elites = [population[i] for i in sorted_indices[:elitism_count]]

        offspring = []
        remaining = pop_size - elitism_count
        while len(offspring) < remaining:
