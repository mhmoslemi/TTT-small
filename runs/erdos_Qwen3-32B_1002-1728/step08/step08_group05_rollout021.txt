import random

def compute_c5(h_half, n_points):
    half_n = len(h_half)
    dx = 2.0 / n_points
    target_full_sum = n_points / 2.0
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[-half_n:] = h_half[::-1]
    h_clipped = np.clip(h_full, 0, 1)
    h_sum = h_clipped.sum()
    if h_sum == 0:
        return 0.0
    h_normalized = h_clipped * (target_full_sum / h_sum)
    corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
    return np.max(corr) * dx
