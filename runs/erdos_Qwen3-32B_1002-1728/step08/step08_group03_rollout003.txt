def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)

    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    np.random.seed(seed)
    
    pop_size = 20
    population = np.zeros((pop_size, half_n))

        total = max(np.sum(population[i]), 1e-9)
        population[i] = (population[i] / total) * target_sum_half
        population[i] = np.clip(population[i], 0, 1.0)
    
    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1.0)

    best_half = population[best_idx].copy()

    F = 0.8
    CR = 0.9
    
    while time.time() - start_time < budget_s:
