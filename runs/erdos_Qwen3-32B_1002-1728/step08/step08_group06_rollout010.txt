            h_init = np.zeros(n_points)
            h_init[:half_n] = h_flat[:half_n]
            h_init[-half_n:] = h_flat[:half_n][::-1]
        
        # Project and scale the initial h if necessary
        def project_and_scale(h_half_input):
            h_full = np.zeros(n_points)
            h_full[:half_n] = h_half_input
            h_full[-half_n:] = h_half_input[::-1]
            h_clipped = np.clip(h_full, 0, 1)
            current_sum = h_clipped.sum()
            if current_sum == 0:
                return np.zeros_like(h_full)
            scale = target_sum / current_sum
            h_scaled = h_clipped * scale
            h_scaled = np.clip(h_scaled, 0, 1)
            return h_scaled
        
        h_half_initial = h_init[:half_n]
        
        # Objective function for optimization
        def objective(h_half):
            h_full = np.zeros(n_points)
            h_full[:half_n] = np.clip(h_half, 0, 1)
            h_full[-half_n:] = h_full[:half_n][::-1]
            h_clipped = np.clip(h_full, 0, 1)
            current_sum = h_clipped.sum()
            if current_sum == 0:
                return np.inf
            scale = target_sum / current_sum
            h_scaled = h_clipped * scale
            h_scaled = np.clip(h_scaled, 0, 1)
            # Compute correlation
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx
            return np.max(corr)
        
        # Optimization for this level
        bounds_half = [(0.0, 1.0) for _ in range(half_n)]
        
        # Time management for this optimization step
        remaining = max(0, budget_s - (time.time() - start_time))
        if remaining < 10:
            break
        time_budget = min(remaining * 0.7, 300)  # Assign 70% of remaining time, up to 300s
        maxiter = max(1, int(time_budget / 10))  # 10 seconds per iteration estimate
        
        # Early skip if time already too tight
        if time.time() - start_time > budget_s - 30:
            break
        
        # Perform optimization
        result = minimize(
            objective,
            h_half_initial,
            method='SLSQP',
            bounds=bounds_half,
            options={'maxiter': maxiter, 'ftol': 1e-8}
        )
        
        # After optimization, project and scale the solution
        optimized_h_half = result.x if result.success else h_half_initial
        h_full_proj = project_and_scale(optimized_h_half)
        corr = np.correlate(h_full_proj, 1 - h_full_proj, mode='full') * dx
        current_c5 = np.max(corr)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full_proj.copy()
        
        # Skip other levels if time is tight
        if time.time() - start_time > budget_s - 30:
            break
    
    # Final checks
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h_sum = best_h.sum()
        assert np.isclose(best_h_sum, target_sum, atol=1e-6), (best_h_sum, target_sum)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values.tolist(), best_c5, len(initial_h_values))
