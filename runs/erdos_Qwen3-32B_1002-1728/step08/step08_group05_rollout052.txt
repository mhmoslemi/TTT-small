            gbest_pos = p['pbest_pos'].copy()
    
    phi = 0.7
    mu = 1.5
    nu = 1.5
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 0.1:
        for p in particles:
            h_full = np.zeros(n_points)
            h_full[:half_n] = p['pos']
            h_full[-half_n:] = p['pos'][::-1]
            h_clipped = np.clip(h_full, 0, 1)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
            current_C5 = np.max(corr)
            if current_C5 < p['pbest_C5']:
                p['pbest_pos'] = p['pos'].copy()
                p['pbest_C5'] = current_C5
                if current_C5 < gbest_C5:
                    gbest_C5 = current_C5
                    gbest_pos = p['pbest_pos'].copy()
        
        for p in particles:
            r1 = np.random.rand()
            r2 = np.random.rand()
            new_vel = phi * p['vel'] + mu * r1 * (p['pbest_pos'] - p['pos']) + nu * r2 * (gbest_pos - p['pos'])
            new_vel = np.clip(new_vel, -0.5, 0.5)
            new_pos = p['pos'] + new_vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = new_pos.sum()
            if sum_new != 0:
                new_pos = (new_pos / sum_new) * target_sum_half
            new_pos = np.clip(new_pos, 0, 1)
            p['vel'] = new_vel
            p['pos'] = new_pos
        
        elapsed_time = time.time() - start_time
        phi = 0.7 - 0.3 * (elapsed_time / budget_s)
        phi = max(phi, 0.4)
    
    h_full_best = np.zeros(n_points)
    h_full_best[:half_n] = gbest_pos
    h_full_best[-half_n:] = gbest_pos[::-1]
    h_full_best = np.clip(h_full_best, 0, 1)
    sum_total = h_full_best.sum()
    if abs(sum_total - n_points/2) > 1e-6:
        h_full_best = (h_full_best / sum_total) * (n_points/2)
    h_full_best = np.clip(h_full_best, 0, 1)
    
    h_clipped = np.clip(h_full_best, 0, 1)
    corr_final = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
    final_c5 = np.max(corr_final)
    
    return (h_full_best.tolist(), final_c5, n_points)
