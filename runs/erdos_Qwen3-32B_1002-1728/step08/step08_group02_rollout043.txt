        idxs = np.argsort(objects)
        sorted_simplex = [simplex[i] for i in idxs]
        objects_sorted = objects[idxs]

        curr_best = sorted_simplex[0].copy()
        curr_best_obj = objects_sorted[0]
        if curr_best_obj < best_c5:
            best_h = curr_best.copy()
            best_c5 = curr_best_obj
            tabu_list.append(best_h.copy())

        # Centroid of all but worst
        worst = sorted_simplex[-1]
        bests = sorted_simplex[:-1]
        centroid = np.mean(bests, axis=0)

        # Reflection
        rho, gamma, chi = 1.0, 2.0, 0.5
        x_reflect = centroid + rho * (centroid - worst)
        x_reflect = np.clip(x_reflect, 0, 1)
        sum_r = x_reflect.sum()
        if sum_r != 0:
            x_reflect = x_reflect / sum_r * target_sum_half
        else:
            x_reflect.fill(target_sum_half / half_n)
        x_reflect = np.clip(x_reflect, 0, 1)

        # Tabu check
        in_tabu = False
        for t in tabu_list:
            if np.allclose(x_reflect, t, atol=1e-5):
                in_tabu = True
                break
        if in_tabu:
            noise = np.random.normal(0, 0.05, x_reflect.shape)
            x_reflect += noise
            x_reflect = np.clip(x_reflect, 0, 1)
            sum_r = x_reflect.sum()
            if sum_r != 0:
                x_reflect = x_reflect / sum_r * target_sum_half
            else:
                x_reflect.fill(target_sum_half / half_n)
            x_reflect = np.clip(x_reflect, 0, 1)
        tabu_list.append(x_reflect.copy())
        obj_reflect = evaluate(x_reflect)

        # Update simplex
        replace_index = -1
        if obj_reflect < objects_sorted[-2]:
            simplex[replace_index] = x_reflect
            continue

        # Expansion
        x_expand = centroid + gamma * (x_reflect - centroid)
        x_expand = np.clip(x_expand, 0, 1)
        sum_e = x_expand.sum()
        if sum_e != 0:
            x_expand = x_expand / sum_e * target_sum_half
        else:
            x_expand.fill(target_sum_half / half_n)
        x_expand = np.clip(x_expand, 0, 1)
        obj_expand = evaluate(x_expand)
        if obj_expand < obj_reflect:
            x_follow = x_expand
            obj_follow = obj_expand
        else:
            x_follow = x_reflect
            obj_follow = obj_reflect
        tabu_list.append(x_follow.copy())
        if obj_follow < objects_sorted[-1]:
            simplex[replace_index] = x_follow
            continue

        # Contraction
        x_contract = centroid + chi * (worst - centroid)
        x_contract = np.clip(x_contract, 0, 1)
        sum_c = x_contract.sum()
        if sum_c != 0:
            x_contract = x_contract / sum_c * target_sum_half
        else:
            x_contract.fill(target_sum_half / half_n)
        x_contract = np.clip(x_contract, 0, 1)
        obj_contract = evaluate(x_contract)
        if obj_contract < objects_sorted[-1]:
            simplex[replace_index] = x_contract
            tabu_list.append(x_contract.copy())
            continue

        # Shrink
        for i in range(1, num_simplex_vertices):
            simplex[i] = bests[0] + 0.5 * (simplex[i] - bests[0])
            simplex[i] = np.clip(simplex[i], 0, 1)
            sum_s = simplex[i].sum()
            if sum_s != 0:
                simplex[i] = simplex[i] / sum_s * target_sum_half
            else:
                simplex[i].fill(target_sum_half / half_n)
            simplex[i] = np.clip(simplex[i], 0, 1)
            tabu_list.append(simplex[i].copy())

    # Finalize
    h_full = np.concatenate([best_h, best_h[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = h_full / h_sum * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
