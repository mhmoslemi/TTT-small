import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h with constraint enforcement
    h_initial = np.clip(h_initial, 0, 1)
    h_sum = h_initial.sum()
    if h_sum != 0:
        h_initial = (h_initial / h_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Set up PSO parameters
    np.random.seed(seed)
    n_particles = 30
    particles = []
    velocities = []
    for i in range(n_particles):
        if i == 0:
            h = h_initial.copy()
        else:
            h = h_initial.copy() + np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if h_sum != 0:
                h = (h / h_sum) * target_sum
            h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))

    # Initialize pbest and global best
    pbest_positions = [p.copy() for p in particles]
    pbest_scores = []
    for p in particles:
        p_clipped = np.clip(p, 0, 1)
        corr_vals = np.correlate(p_clipped, 1 - p_clipped, mode='full')
        score = np.max(corr_vals) * dx
        pbest_scores.append(score)
    global_best_idx = np.argmin(pbest_scores)
    global_best = particles[global_best_idx].copy()
    global_best_score = pbest_scores[global_best_idx]

    # PSO loop with time tracking
    inertia_initial, inertia_final = 0.7, 0.4
    cognitive, social = 2.0, 2.0
    max_pso_time = budget_s * 0.6
    while time.time() - start_time < max_pso_time:
        elapsed = time.time() - start_time
        fraction = min(elapsed / max_pso_time, 1.0)
        inertia = inertia_initial + (inertia_final - inertia_initial) * fraction
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            pbest = pbest_positions[i]
            # Update velocity and position
            vel = (inertia * vel + cognitive * r1 * (pbest - pos) + social * r2 * (global_best - pos))
            vel = np.clip(vel, -0.5, 0.5)
            velocities[i] = vel
            pos += vel
            # Constraint enforcement
            pos_clipped = np.clip(pos, 0, 1)
            pos_sum = pos_clipped.sum()
            if pos_sum != 0:
                pos = (pos_clipped / pos_sum) * target_sum
            else:
                pos = np.zeros(n_points)
            particles[i] = pos
            # Compute score
            p_clipped = np.clip(pos, 0, 1)
            corr_vals = np.correlate(p_clipped, 1 - p_clipped, mode='full')
            score = np.max(corr_vals) * dx
            # Update pbest
            if score < pbest_scores[i]:
                pbest_positions[i] = p_clipped.copy()
                pbest_scores[i] = score
                if score < global_best_score:
                    global_best[:] = p_clipped.copy()
                    global_best_score = score
        # Break if time expired
        if time.time() - start_time >= budget_s:
            break

    # Local optimization with SLSQP
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        corr_vals = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr_vals) * dx

    # Enforce constraints on initial guess
    best_so_far = np.clip(global_best, 0, 1)
    best_sum = best_so_far.sum()
    if best_sum != 0:
        best_so_far = (best_so_far / best_sum) * target_sum
    best_so_far = np.clip(best_so_far, 0, 1)

    # Define bounds and constraints
    bounds = [(0.0, 1.0)] * n_points
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    # Run local optimization
    try:
        res = minimize(
            objective,
            x0=best_so_far,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            final_h_sum = final_h.sum()
            if final_h_sum != 0:
                final_h = (final_h / final_h_sum) * target_sum
            final_h = np.clip(final_h, 0, 1)
            final_c5 = objective(final_h)
        else:
            final_h = best_so_far.copy()
            final_c5 = objective(final_h)
    except:
        final_h = best_so_far.copy()
        final_c5 = objective(final_h)

    # Final constraint check
    final_h = np.clip(final_h, 0, 1)
    final_sum = final_h.sum()
    if abs(final_sum - target_sum) > 1e-6:
        final_h = (final_h / final_sum) * target_sum
        final_h = np.clip(final_h, 0, 1)
    return (final_h.tolist(), final_c5, n_points)
