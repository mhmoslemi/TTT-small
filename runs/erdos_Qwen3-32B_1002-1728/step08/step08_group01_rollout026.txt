    n_points = len(h_full_initial)

        if current_sum < 1e-9:
            sym_vars.fill(target_sum_half / half_n)
        else:
            scale_factor = target_sum_half / current_sum
            sym_vars *= scale_factor
            sym_vars = np.clip(sym_vars, 0, 1)
        return sym_vars

    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_full_initial = h_full_initial / h_full_initial.sum() * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_sym_initial = h_full_initial[:half_n].copy()

        h_full = create_full_h(repair_h_sym(sym_vars))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 3
    pop = [repair_h_sym(np.random.uniform(0, 1, half_n)) for _ in range(pop_size)]
    pop[0] = h_sym_initial  # prioritize initial solution
    best_c5 = np.inf
    best_h_sym = h_sym_initial

    while time.time() - start_time < budget_s - 60:  # reserve time for refinement
        fitness = np.array([objective(ind) for ind in pop])
        best_idx = np.argmin(fitness)
        best_c5 = min(best_c5, fitness[best_idx])
        best_h_sym = pop[best_idx].copy()

        new_pop = [pop[best_idx]]  # elite survival
        while len(new_pop) < pop_size:
