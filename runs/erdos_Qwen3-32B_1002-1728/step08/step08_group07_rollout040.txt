    def create_full_h(h_half):
        return np.concatenate([h_half, h_half[::-1]])

    def project_to_feasible(h_half):
        h_clipped = np.clip(h_half, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped < 1e-9:
            return np.full_like(h_clipped, target_sum_half / h_half.size)
        return np.clip(h_clipped * (target_sum_half / sum_clipped), 0, 1)

    def evaluate_c5(h_half):
        full_h = create_full_h(h_half)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        return np.max(corr), corr

    def mutate(h_half):
        h_mut = h_half.copy()
        num_mut = max(1, int(h_half.size * 0.1))
        indices = np.random.choice(h_half.size, num_mut, replace=False)
        noise = np.random.normal(0, 0.05, size=num_mut)
        h_mut[indices] += noise
        return project_to_feasible(h_mut)

    def crossover(parent1, parent2):
        mask = np.random.rand(*parent1.shape) < 0.5
        child = np.where(mask, parent1, parent2)
        return project_to_feasible(child)

    population_size = 20
    h_half_initial = project_to_feasible(h_full_initial[:half_n].copy())
    population = [h_half_initial] + [project_to_feasible(h_half_initial + np.random.normal(0, 0.01, half_n)) for _ in range(population_size - 1)]
    best_c5 = float('inf')
    best_h_half = h_half_initial.copy()

    while time.time() - start_time < budget_s - 10:
        scores = []
        corrs_list = []
        for h in population:
            c5, corr = evaluate_c5(h)
            scores.append(c5)
            corrs_list.append(corr)
        scores = np.array(scores)
        sort_idx = np.argsort(scores)
        parents = [population[i] for i in sort_idx[:max(1, population_size // 2)]]
        offspring = []
        while len(offspring) < population_size - len(parents):
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            child = crossover(p1, p2)
            if np.random.rand() < 0.5:
                child = mutate(child)
            offspring.append(child)
        for i in range(min(2, len(parents))):
            current_h = parents[i].copy()
            for _ in range(5):
                full_h = create_full_h(current_h)
                c5, corr = evaluate_c5(current_h)
                max_corr_idx = np.argmax(corr)
                grad = np.zeros(half_n)
                epsilon = 1e-4
                for j in range(half_n):
                    h_plus = current_h.copy()
                    h_plus[j] += epsilon
                    h_plus = project_to_feasible(h_plus)
                    _, corr_plus = evaluate_c5(h_plus)
                    val_plus = corr_plus[max_corr_idx]
                    h_minus = current_h.copy()
                    h_minus[j] -= epsilon
                    h_minus = project_to_feasible(h_minus)
                    _, corr_minus = evaluate_c5(h_minus)
                    val_minus = corr_minus[max_corr_idx]
                    grad[j] = (val_plus - val_minus) / (2 * epsilon)
                current_h -= 0.03 * grad
                current_h = project_to_feasible(current_h)
            parents[i] = current_h
        population = parents + offspring
        current_scores = [evaluate_c5(h)[0] for h in population]
        best_idx = np.argmin(current_scores)
        if current_scores[best_idx] < best_c5:
            best_c5 = current_scores[best_idx]
            best_h_half = population[best_idx].copy()

    full_h = create_full_h(best_h_half)
    sum_full = np.sum(full_h)
    desired = n_points / 2.0
    if abs(sum_full - desired) > 1e-6:
        full_h = (full_h / sum_full) * desired
        full_h = np.clip(full_h, 0, 1)
    return (full_h.tolist(), best_c5, n_points)
