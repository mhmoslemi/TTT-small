    def evaluate_population(pop):
        scores = []
        for ind in pop:
            full_h = create_full_h(ind)
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            score = np.max(corr) * dx
            scores.append(score)
        return np.array(scores)

    scores = evaluate_population(population)
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_half = population[best_idx].copy()

    np.random.seed(seed)

    current_mutation = mutation_rate
    for generation in range(generations):
        if time.time() - start_time > budget_s - 5:
            break

        parents_idx = tournament_selection(population, scores, pop_size, tournament_size)
        parents = population[parents_idx]

        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                child = uniform_crossover(p1, p2)
                mutated = mutate(child, current_mutation)
                repaired = repair_individual(mutated, target_sum_half)
                offspring.append(repaired)
            else:
                offspring.append(parents[-1].copy())
        offspring = np.array(offspring)
        offspring_scores = evaluate_population(offspring)

        elite_count = int(pop_size * elite_ratio)
        elite_indices = np.argsort(scores)[:elite_count]
        elites = population[elite_indices]
        elites_scores = scores[elite_indices]

        new_population = np.vstack([elites, offspring])
        new_scores = np.concatenate([elites_scores, offspring_scores])
        best_indices = np.argsort(new_scores)[:pop_size]
        population = new_population[best_indices]
        scores = new_scores[best_indices]

        current_best_idx = np.argmin(scores)
        if scores[current_best_idx] < best_score:
            best_score = scores[current_best_idx]
            best_half = population[current_best_idx].copy()

        current_mutation *= mutation_decay

    refinement_candidates = int(pop_size * 0.1)
    for idx in range(min(refinement_candidates, pop_size)):
        candidate = population[idx].copy()
        def objective(x):
            h_clipped = np.clip(x, 0, 1)
            h_full = create_full_h(h_clipped)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        bounds = [(0.0, 1.0)] * half_n
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)
        result = minimize(objective, candidate, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-10)
        if result.success:
            refined = np.clip(result.x, 0, 1)
            sum_refined = np.sum(refined)
            if not np.isclose(sum_refined, target_sum_half):
                refined = np.clip(refined, 0, 1)
                sum_refined = np.sum(refined)
                refined *= target_sum_half / sum_refined
                refined = np.clip(refined, 0, 1)
            full_h = create_full_h(refined)
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            score = np.max(corr) * dx
            if score < best_score:
                best_score = score
                best_half = refined

    h_full = create_full_h(best_half)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_score, n_points)
