    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    np.random.seed(seed)

    def repair_h_sym(h_sym):
        h_sym = np.clip(h_sym, 0, 1)
        current_sum = h_sym.sum()
        if current_sum < 1e-9:
            return h_sym * 0.0
        h_sym = h_sym * target_sum_half / current_sum
        h_sym = np.clip(h_sym, 0, 1)
        return h_sym

    # Initialize h_sym from initial_h_values
    h_sym = h_full_initial[:half_n].copy()
    h_sym = repair_h_sym(h_sym)
    best_c5 = np.inf
    best_h_sym = h_sym.copy()

    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    alpha = 0.05  # Learning rate
    m = np.zeros_like(h_sym)
    v = np.zeros_like(h_sym)
    t = 0
    beta_softmax = 20.0

    def compute_smoothing_objective(overlaps, beta):
        exp_vals = np.exp(beta * overlaps)
        smooth_max = (1 / beta) * np.log(np.sum(exp_vals))
        return smooth_max

    while time.time() - start_time < budget_s - 5:
        t += 1
        h_full = np.concatenate((h_sym, h_sym[::-1]))
        h_full = np.clip(h_full, 0, 1)

        grad = np.zeros_like(h_sym)
        eps = 1e-5

            h_pert[i] += eps

            grad[i] = (smooth_pert - smooth_pert_minus) / (2 * eps)
