import numpy as np
import time
from scipy.optimize import minimize

def create_full_h(h_half, n_points):
    half_n = len(h_half)
    full_h = np.zeros(n_points)
    full_h[:half_n] = h_half
    full_h[-half_n:] = h_half[::-1]
    return full_h

def compute_c5(full_h, dx):
    corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # target sum for the half part

    # Initialize particles
    n_particles = 20
    rng = np.random.default_rng(seed)
    particles = []
    velocities = []
    personal_bests = []

    # First particle from initial_h_values
    h_initial = h_full_initial[:half_n].copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (target_sum_half / h_initial.sum())
    h_initial = np.clip(h_initial, 0, 1)
    particles.append(h_initial)
    velocities.append(rng.uniform(-0.05, 0.05, half_n))
    personal_bests.append(h_initial.copy())

    # Generate remaining particles
    for _ in range(n_particles - 1):
        h = particles[0].copy()
        noise = 0.05 * rng.uniform(-1, 1, half_n)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum_half / h.sum())
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(rng.uniform(-0.05, 0.05, half_n))
        personal_bests.append(h.copy())

    particles = np.array(particles)
    velocities = np.array(velocities)
    personal_bests = np.array(personal_bests)
    personal_best_c5s = np.array([compute_c5(create_full_h(p, n_points), dx) for p in personal_bests])
    best_idx = np.argmin(personal_best_c5s)
    global_best = personal_bests[best_idx].copy()
    best_c5 = personal_best_c5s[best_idx]
    max_no_improve = 50
    no_improve_counter = 0
    omega_init = 0.9
    omega_decay = 0.995

    # Main PSO loop
    while time.time() - start_time < budget_s - 10:
        for i in range(n_particles):
            r1, r2 = rng.uniform(), rng.uniform()
            velocities[i] = (omega_init * velocities[i] +
                             2.0 * r1 * (personal_bests[i] - particles[i]) +
                             2.0 * r2 * (global_best - particles[i]))

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            # Project to feasible sum

            if h_sum > 1e-9:
                particles[i] = particles[i] / h_sum * target_sum_half
            particles[i] = np.clip(particles[i], 0, 1)

        # Evaluate and update bests
        current_c5s = np.array([compute_c5(create_full_h(p, n_points), dx) for p in particles])
        improved = np.where(current_c5s < personal_best_c5s)[0]
        if len(improved) > 0:
            personal_bests[improved] = particles[improved]
            personal_best_c5s[improved] = current_c5s[improved]
            if np.min(current_c5s) < best_c5:
                best_c5 = np.min(current_c5s)
                best_idx = np.argmin(current_c5s)
                global_best = particles[best_idx].copy()
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # Update inertia and early stop
        omega_init *= omega_decay
        if no_improve_counter >= max_no_improve:

    # Final validation and return

    full_sum = final_full.sum()
    if abs(full_sum - n_points / 2) > 1e-6:
        final_full = (final_full / full_sum) * (n_points / 2)
        final_full = np.clip(final_full, 0, 1)
    final_c5 = compute_c5(final_full, dx)
    return (final_full.tolist(), final_c5, n_points)
