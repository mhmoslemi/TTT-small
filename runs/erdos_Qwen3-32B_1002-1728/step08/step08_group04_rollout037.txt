import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    points_x = np.linspace(0, 2, n_points, endpoint=False)
    x_prime = points_x - 1.0  # Mapping [0, 2] to [-1, 1]

    # Precompute even Chebyshev polynomials T_0, T_2, ..., T_{2(M_even-1)}
    M_even = 10
    T_even = np.zeros((n_points, M_even))
    for k in range(M_even):
        k_poly = 2 * k
        for i in range(n_points):
            x = x_prime[i]
            if k_poly == 0:
                T_even[i, k] = 1.0
            else:
                T_even[i, k] = np.cos(k_poly * np.arccos(x))
    
    # Normalize initial h_initial to have integral 1 and project onto basis
    h_initial = np.clip(h_initial, 0, 1)
    integral_initial = np.sum(h_initial) * dx
    if integral_initial > 1e-9:
        h_initial /= integral_initial
    h_initial = np.clip(h_initial, 0, 1)
    A = T_even
    c_opt, _, _, _ = np.linalg.lstsq(A, h_initial, rcond=None)

    best_c5 = float('inf')
    best_c = c_opt.copy()
    max_iter = 10000
    learning_rate = 0.01
    decay_rate = 0.999
    epsilon = 1e-5
    current_iter = 0

    while time.time() - start_time < budget_s - 10:
        # Generate current h
        h_current = T_even @ c_opt
        h_current = np.clip(h_current, 0, 1)
        integral_current = np.sum(h_current) * dx
        if integral_current > 1e-9:
            h_current /= integral_current
        h_current = np.clip(h_current, 0, 1)

        # Compute C5
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        c5_val = np.max(corr) * dx

        if c5_val < best_c5:
            best_c5 = c5_val
            best_c = c_opt.copy()

        # Compute gradient using finite differences
        grad = np.zeros_like(c_opt)
        for i in range(M_even):
            c_plus = c_opt.copy()
            c_plus[i] += epsilon
            h_plus = T_even @ c_plus
            h_plus = np.clip(h_plus, 0, 1)
            integral_plus = np.sum(h_plus) * dx
            if integral_plus > 1e-9:
                h_plus /= integral_plus
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_val) / epsilon

        # Update coefficients
        c_opt -= learning_rate * grad

        # Project back to feasible region
        h_projected = T_even @ c_opt
        h_projected = np.clip(h_projected, 0, 1)
        integral = np.sum(h_projected) * dx
        if integral > 1e-9:
            h_projected /= integral
        h_projected = np.clip(h_projected, 0, 1)
        c_opt, _, _, _ = np.linalg.lstsq(A, h_projected, rcond=None)

        # Decay learning rate
        learning_rate *= decay_rate

        current_iter += 1
        if current_iter >= max_iter:
            break

    # Generate final h from best coefficients
    h_final = T_even @ best_c
    h_final = np.clip(h_final, 0, 1)
    integral = np.sum(h_final) * dx
    if integral > 1e-9:
        h_final /= integral
    h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), best_c5, n_points)
