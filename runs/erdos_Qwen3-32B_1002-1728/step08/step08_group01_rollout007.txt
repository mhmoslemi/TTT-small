import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    h_full_initial *= target_sum_full / current_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_sym_initial = h_full_initial[:half_n].copy()

    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def evaluate(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    def repair(sym_vars):
        sym_clipped = np.clip(sym_vars, 0, 1)
        sym_sum = sym_clipped.sum()
        if sym_sum < 1e-9:
            return np.zeros_like(sym_clipped)
        scale = target_sum_half / sym_sum
        return np.clip(sym_clipped * scale, 0, 1)

    np.random.seed(seed)
    pop_size = 50
    population = []
    for _ in range(pop_size):
        candidate = h_sym_initial.copy() + np.random.normal(0, 0.1, half_n)
        population.append(repair(candidate))
    population = np.array(population)
    fitness = np.array([evaluate(ind) for ind in population])
    best = population[fitness.argmin()].copy()
    best_val = np.min(fitness)

    mutation_rate = 0.1
    patience = 0

    while time.time() - start_time < budget_s - 5:
        new_pop = []
        for _ in range(pop_size // 2):
            tournament_size = 3
            parents = np.random.choice(population, tournament_size)
            p1, p2 = min(parents, key=evaluate), max(parents, key=evaluate)
            cross = (np.random.rand(half_n) < 0.5)
            c1 = np.where(cross, p1, p2) + np.random.normal(0, mutation_rate, half_n)
            c2 = np.where(cross, p2, p1) + np.random.normal(0, mutation_rate, half_n)
            new_pop.extend([repair(c1), repair(c2)])
        population = np.array(new_pop)
        fitness = np.array([evaluate(ind) for ind in population])
        candidates = np.vstack((population, best))
        candidates_fitness = np.concatenate((fitness, [best_val]))
        indices = np.argsort(candidates_fitness)[:pop_size]
        population = candidates[indices]
        current_best = population[candidates_fitness[indices].argmin()].copy()
        current_val = evaluate(current_best)
        if current_val < best_val:
            best, best_val = current_best.copy(), current_val
            patience = 0
            mutation_rate = max(0.05, mutation_rate * 0.95)
        else:
            patience += 1
            if patience > 50:
                mutation_rate = min(1.0, mutation_rate * 1.1)
                patience = 0

    for method in ['SLSQP', 'Nelder-Mead']:
        try:

                           options={'maxiter': 200, 'ftol': 1e-10})
