import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0  # sum(h) must equal this
    dx = 2.0 / n_points

    def clip_and_repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.clip(h.sum(), 1e-9, None)
        return (h / h_sum) * target_sum

    # Initialize population
    population_size = 200
    population = []
    # Add initial solution
    h = clip_and_repair(h_initial)
    population.append(h)
    # Generate random solutions
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = clip_and_repair(h)
        population.append(h)
    population = np.array(population)

    # Fitness function
    def fitness(individual):
        h_clipped = np.clip(individual, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s - 5:
        # Evaluate population
        fit_vals = np.array([fitness(ind) for ind in population])
        # Elites (top 5%)
        n_elites = int(0.05 * population_size)
        elite_indices = np.argsort(fit_vals)[:n_elites]
        elites = population[elite_indices]

        # Select parents (top 50%)
        parent_indices = np.argsort(fit_vals)[:population_size//2]
        parents = population[parent_indices]

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                cross_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:cross_point], p2[cross_point:]])
                child2 = np.concatenate([p2[:cross_point], p1[cross_point:]])
                for child in [child1, child2]:
                    child = clip_and_repair(child)
                    offspring.append(child)

        # Mutation
        mutation_rate = 0.3
        for idx in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
                noise = np.random.normal(0, 0.1, size=len(mut_indices))
                offspring[idx][mut_indices] += noise
                offspring[idx] = clip_and_repair(offspring[idx])

        # Create new population: elites + parents + offspring
        new_pop = np.concatenate([elites, parents, offspring])
        new_fit = np.array([fitness(ind) for ind in new_pop])
        # Truncate to size
        sorted_indices = np.argsort(new_fit)
        population = new_pop[sorted_indices[:population_size]]

        # Update best
        current_best = np.min(new_fit)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(new_fit)]

        # Check time
        if time.time() - start_time >= budget_s:
            break

    # Final verification
    best_h = clip_and_repair(best_h)
    final_c5 = fitness(best_h)
    return (best_h.tolist(), final_c5, n_points)
