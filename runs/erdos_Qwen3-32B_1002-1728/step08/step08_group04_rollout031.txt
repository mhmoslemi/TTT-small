    current_sum = np.sum(h)

        h[:] = 0.5
        current_sum = 0.5 * len(h)
    h = (h / current_sum) * target_sum
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h_vec, dx):
    h = np.clip(h_vec, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
