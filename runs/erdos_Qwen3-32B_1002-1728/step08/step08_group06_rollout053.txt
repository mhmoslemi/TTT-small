import numpy as np
from scipy.optimize import minimize
import time

def pso_step(obj_func, bounds, n_vars, pop_size=20, n_iter=100, w=0.7, c1=1.5, c2=1.5, tol=1e-6):
    npop, ndim = pop_size, n_vars
    low, high = np.asarray(bounds).T
    X = low + np.random.rand(npop, ndim) * (high - low)
    V = np.random.rand(npop, ndim) * 0.1
    p_best = X.copy()
    p_score = np.array([obj_func(x) for x in X])
    g_idx = p_score.argmin()
    g_best = X[g_idx].copy()
    g_score = p_score[g_idx]
    
    for _ in range(n_iter):
        r1, r2 = np.random.rand(ndim), np.random.rand(ndim)
        V = w * V + c1 * r1 * (p_best - X) + c2 * r2 * (g_best - X)
        V = np.clip(V, -0.2, 0.2)
        X = X + V
        X = np.clip(X, low, high)
        
        scores = np.array([obj_func(x) for x in X])
        improved = scores < p_score
        p_best[improved] = X[improved].copy()
        p_score[improved] = scores[improved]
        new_g_idx = p_score.argmin()
        if p_score[new_g_idx] < g_score:
            g_best, g_score = X[new_g_idx].copy(), p_score[new_g_idx]
            
        if np.abs(g_score) < tol:
            break
    return g_best, g_score
