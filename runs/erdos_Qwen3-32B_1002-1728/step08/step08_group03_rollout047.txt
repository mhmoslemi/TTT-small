    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points

    h_half_initial = h_full_initial[:half_n]
    current_sum = h_half_initial.sum()
    if abs(current_sum - target_sum_half) > 1e-6:
        h_half_initial = (h_half_initial / current_sum) * target_sum_half

    np.random.seed(seed)
    num_starts = 40
    best_c5 = np.inf
    best_h_half = h_half_initial.copy()

    def evaluate_c5(h_half):
        h_full = np.zeros(2 * half_n)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if evaluate_c5(h_half_initial) < best_c5:
        best_c5 = evaluate_c5(h_half_initial)
        best_h_half = h_half_initial.copy()

    while time.time() - start_time < budget_s:
        perturbation = 0.05 * np.random.randn(half_n)
        h_perturbed = h_half_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = h_perturbed.sum()
        if h_perturbed_sum < 1e-9:
            h_perturbed = h_half_initial.copy()
        else:
            h_perturbed = (h_perturbed / h_perturbed_sum) * target_sum_half

        cons = [{'type': 'eq', 'fun': constraint_func, 'args': (target_sum_half,)}]
        bounds = [(0, 1) for _ in range(half_n)]

        result = minimize(
            objective,
            h_perturbed,
            args=(target_sum_half, dx, half_n),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        if result.success:
            h_opt = result.x.copy()
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h_half = h_opt.copy()

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    h_full_final = np.zeros(n_points)
    h_full_final[:half_n] = best_h_half
    h_full_final[half_n:] = best_h_half[::-1]
    h_full_final = np.clip(h_full_final, 0, 1)

    total_sum = h_full_final.sum()
    if abs(total_sum - target_sum_full) > 1e-6:
        h_full_final = (h_full_final / total_sum) * target_sum_full
        h_full_final = np.clip(h_full_final, 0, 1)

    return (h_full_final.tolist(), best_c5, n_points)
