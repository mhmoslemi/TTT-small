def project_half(h_half, target_sum):
    h_clipped = np.clip(h_half, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return np.full_like(h_half, target_sum / len(h_half))
    scaled = h_clipped / current_sum * target_sum
    scaled_clipped = np.clip(scaled, 0, 1)
    return scaled_clipped
