import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    half_n = n_points // 2
    h_parent = np.array(initial_h_values) if 'initial_h_values' in globals() else None
    if h_parent is None:
        return ([], np.inf, n_points)
    
    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h
    
    def objective(sym_vars):
        sym_vars = np.clip(sym_vars, 0, 1)
        h_full = create_full_h(sym_vars)
        total = h_full.sum()
        if total == 0:
            return np.inf
        
        h_scaled = h_full * (target_sum / total)
        h_clipped = np.clip(h_scaled, 0, 1)
        total_clipped = h_clipped.sum()
        if total_clipped == 0:
            return np.inf
        
        scaling_factor = target_sum / total_clipped
        h_final = h_clipped * scaling_factor
        h_final = np.clip(h_final, 0, 1)
        
        corr = np.correlate(h_final, 1 - h_final, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    initial_guess = h_parent[:half_n].copy()
    
    result = minimize(
        objective,
        x0=initial_guess,
        bounds=bounds,
        method='trust-constr',
        options={
            'maxiter': 1000,
            'xatol': 1e-6,
            'gtol': 1e-6,
            'barrier_tol': 1e-6,
            'finite_diff_rel_step': 1e-6
        },
        tol=1e-8
    )
    
    optimized_sym = result.x if result.success else initial_guess

    total = h_full.sum()

        h_final = np.clip(h_scaled, 0, 1)

    h_final = np.clip(h_final, 0, 1)
    h_final_sum = h_final.sum()

    return (h_final.tolist(), c5_bound, len(h_final))
