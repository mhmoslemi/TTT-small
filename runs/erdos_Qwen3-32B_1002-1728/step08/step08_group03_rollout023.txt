        # Build full h
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1.0)

        # Compute cross-correlation
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_index = np.argmax(corr)
        current_c5 = corr[max_index] * dx

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()

        # Compute k_max
        m_max = max_index
        k_max = m_max - (n_points - 1)

        # Compute gradient for h_full
        x_vals = np.arange(n_points)
        grad_full = np.zeros(n_points)

        # Handle term1: 1 - h[x + k_max]
        x_plus_k = x_vals + k_max
        valid1 = (x_plus_k >= 0) & (x_plus_k < n_points)
        term1 = np.where(valid1, 1 - h_full[x_plus_k], 0)

        # Handle term2: -h[x - k_max]
        x_minus_k = x_vals - k_max
        valid2 = (x_minus_k >= 0) & (x_minus_k < n_points)
        term2 = np.where(valid2, -h_full[x_minus_k], 0)

        grad_full = term1 + term2

        # Symmetrize gradient
        grad_full = (grad_full + grad_full[::-1]) / 2.0

        # Extract gradient for h_half
        grad_half = grad_full[:half_n]

        # Gradient step
        h_half -= learning_rate * grad_half

        # Project back to feasible region
        h_half = np.clip(h_half, 0, 1.0)
        current_sum = h_half.sum()
        if current_sum != 0:
            scale = target_sum_half / current_sum
            h_half *= scale
            h_half = np.clip(h_half, 0, 1.0)

        # Decay learning rate
        learning_rate *= decay_rate

        iteration += 1

    # Final construction of h_full
    h_half = best_h_half
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[half_n:] = h_half[::-1]
    h_full = np.clip(h_full, 0, 1.0)

    return (h_full.tolist(), best_c5, n_points)
