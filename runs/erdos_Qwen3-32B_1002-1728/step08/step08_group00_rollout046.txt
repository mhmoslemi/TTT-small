            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h_sum = new_h.sum()
                if new_h_sum != 0:
                    new_h = (new_h / new_h_sum) * target_sum
                new_h = np.clip(new_h, 0, 1)
                # Evaluate new C5
                new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full') * dx)
                if new_c5 < best_c5:
                    final_h = new_h
                    best_c5 = new_c5
        except:
            pass

        # Perturb to avoid local optima
        perturbed = final_h.copy()
        perturb_indices = np.random.choice(n_points, size=min(n_points, 5), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, 0.1, size=len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        if perturbed.sum() != 0:
            perturbed = (perturbed / perturbed.sum()) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        perturbed[perturbed < 1e-8] = 0
        final_h = perturbed

    # Final cleanup
    h_full = final_h.copy()
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
