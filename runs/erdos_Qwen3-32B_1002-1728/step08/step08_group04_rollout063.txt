import numpy as np
import time
import random

def repair(individual):
    individual = np.clip(individual, 0, 1)
    total = np.sum(individual)
    if total < 1e-9:
        return np.full_like(individual, 0.5)
    scaling = 35.0 / total
    individual *= scaling
    individual = np.clip(individual, 0, 1)
    return individual

def evaluate(individual):
    h = np.array(individual)
    h_1_minus_h = 1.0 - h
    corr = np.correlate(h, h_1_minus_h, mode='full')
    dx = 2.0 / len(h)
    return np.max(corr) * dx
