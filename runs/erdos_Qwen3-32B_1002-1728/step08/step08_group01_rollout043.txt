                child = parent1.copy()

            # Mutation
            if np.random.rand() < mutation_rate:
                perturbation = np.random.normal(0, noise_std, size=half_n)
                child += perturbation

            # Repair
            child = repair(child)
            new_population.append(child)
        new_population = np.array(new_population)
        population_c5s = np.array([evaluate_individual(p) for p in new_population])
        population = new_population

        current_best_idx = np.argmin(population_c5s)
        current_best_c5 = population_c5s[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()

        # Dynamic parameter adjustment
        mutation_rate *= 0.999
        noise_std *= 0.995

    # Local refinement
    bounds = [(0.0, 1.0)] * half_n
    def objective(x):
        h_half = x.copy()
        return evaluate_individual(h_half)

    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
    for _ in range(20):
        if time.time() - start_time >= budget_s:
            break
        try:
            res = minimize(
                objective,
                x0=best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                tol=1e-12,
                options={'maxiter': 100, 'ftol': 1e-12}
            )
            if res.success:
                new_h = repair(res.x)
                new_c5 = evaluate_individual(new_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
        except:
            continue

    # Final projection
    h_full = create_full_h(best_h)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
