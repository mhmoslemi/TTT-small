def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    best_c5 = np.inf

        x_curr = np.linspace(0, 2, n_points)
        
        # Initialize h for this grid
        if initial_h_values_given is not None and n_points == 70:

        else:
            # Interpolate from previous best or start with uniform
            if best_h is not None and idx > 0:
                prev_x = np.linspace(0, 2, len(best_h))
                h_curr = np.interp(x_curr, prev_x, best_h)
            else:
                h_curr = np.full(n_points, target_sum / n_points)
            h_curr = np.clip(h_curr, 0, 1)
            h_curr_sum = h_curr.sum()
            if h_curr_sum != 0:
                h_curr *= target_sum / h_curr_sum
        
        # Extract symmetric part
        h_half = h_curr[:half_n].copy()
        
        # Parameters for Evolution Strategy
        pop_size = 50
        elite_size = 10
        max_gen = 50 if n_points < 70 else 30
        sigma = 0.1
        
        population = np.tile(h_half, (pop_size, 1))

            population[i] += np.random.randn(half_n) * sigma
