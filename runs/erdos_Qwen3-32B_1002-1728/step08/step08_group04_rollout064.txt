import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)
    
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair(individual):
        individual = np.clip(individual, 0, 1)
        total = max(np.sum(individual), 1e-9)
        return (individual / total) * target_sum

    def calculate_c5(individual):
        h = repair(individual)
        h_1_h = 1.0 - h
        corr = np.correlate(h, h_1_h, mode='full')
        return np.max(corr) * dx

    pop_size = 200
    elites_ratio = 0.05
    parents_ratio = 0.5
    mutate_rate = 0.3
    cx_min, cx_max = 10, 60

    population = np.zeros((pop_size, n_points))
    population[0] = repair(h_initial)
    for i in range(1, pop_size):
        candidate = np.random.rand(n_points)
        population[i] = repair(candidate)

    best_c5 = float('inf')
    best_individual = population[0].copy()

    while time.time() - start_time < budget_s - 10:
        fitness = np.array([calculate_c5(ind) for ind in population])
        ranked = np.argsort(fitness)
        elites_count = int(elites_ratio * pop_size)
        parents = population[ranked[:int(parents_ratio * pop_size)]]

        new_population = population[ranked[:elites_count]].copy()
        remaining = pop_size - elites_count

        for _ in range(remaining):
            p1 = parents[random.randint(0, len(parents)-1)].copy()
            p2 = parents[random.randint(0, len(parents)-1)].copy()
            cx_point = random.randint(cx_min, cx_max)
            child = np.concatenate([p1[:cx_point], p2[cx_point:]])
            child = repair(child)
            
            if random.random() < mutate_rate:
                n_mutate = min(random.randint(1, int(n_points * 0.3)), n_points)
                indices = np.random.choice(n_points, n_mutate, replace=False)
                noise = np.random.normal(0, 0.1, size=len(indices))
                child[indices] += noise
                child = repair(child)
            
            new_population = np.vstack([new_population, child]) if new_population.size else child.reshape(1, -1)

        population = new_population
        current_gen_best = np.min(fitness)
        if current_gen_best < best_c5:
            best_index = ranked[0]
            best_individual = population[best_index].copy()
            best_c5 = current_gen_best

    best_individual = repair(best_individual)
    final_c5 = calculate_c5(best_individual)
    return (best_individual.tolist(), final_c5, n_points)
