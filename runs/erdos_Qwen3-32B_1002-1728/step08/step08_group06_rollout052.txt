                # Mutation
                if np.random.rand() < 0.2:
                    mutation = np.random.normal(0, 0.05, n_points)
                    child += mutation
                child = repair_h(child, target_sum)
                offspring.append(child)
            offspring = np.array(offspring)
            
            # Replace part of population
            population = np.array([selected[i] if np.random.rand() < 0.5 else offspring[i] for i in range(population_size)])
            
            remaining = max(0, budget_s - (time.time() - start_time))
            if remaining < 2:
                break
            
            # Early stopping if no improvement
            if time.time() - start_time > budget_s - 30:
                break
    
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    return (initial_h_values_given.tolist(), 0.380952, len(initial_h_values_given))
