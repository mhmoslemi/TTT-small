    grid_sizes = [70, 50, 40, 30] if 'initial_h_values' in globals() else [50, 40, 30]
    pop_size = 100

    for n_points in grid_sizes:
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        target_sum = n_points / 2
        population = []

        # Initialize population
        if n_points == 70 and 'initial_h_values' in globals():
            base = np.array(initial_h_values)
            for _ in range(pop_size):
                h = base + np.random.normal(0, 0.1, n_points)
                population.append(repair(h, target_sum, n_points))
        else:
            for _ in range(pop_size):
                h = np.random.uniform(0, 1, n_points)
                population.append(repair(h, target_sum, n_points))

        # Evaluate initial population
        fitnesses = [compute_c5(h, dx) for h in population]
        generations = 0
        max_gen = 500

        while generations < max_gen and time.time() - start_time < budget_s:
            generations += 1
            children = []

            # Select parents and create children
            for _ in range(pop_size // 2):
                p1 = tournament_selection(population, fitnesses)
                p2 = tournament_selection(population, fitnesses)
                c1, c2 = crossover(p1, p2)
                c1 = mutate(c1, 0.1 if np.random.rand() < 0.8 else 0.3)
                c2 = mutate(c2, 0.1 if np.random.rand() < 0.8 else 0.3)
                children.extend([c1, c2])

            # Repair and evaluate children
            children = [repair(c, target_sum, n_points) for c in children]
            child_fitnesses = [compute_c5(c, dx) for c in children]

            # Combine and select best for next generation
            combined = population + children
            combined_fit = fitnesses + child_fitnesses
            idxs = np.argsort(combined_fit)
            population = [combined[i] for i in idxs[:pop_size]]
            fitnesses = [combined_fit[i] for i in idxs[:pop_size]]

            # Update best solution
            best_idx = np.argmin(fitnesses)
            curr_best = population[best_idx]
            curr_c5 = fitnesses[best_idx]
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_best

            if time.time() - start_time > budget_s - 10:
                break

    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), n_points / 2, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values.tolist(), best_c5, len(initial_h_values))
