    h_best = create_full_h(global_best_pos, half_n, n_points)
    h_best = np.clip(h_best, 0, 1)
    h_sum = np.sum(h_best)
    if not np.isclose(h_sum, target_sum_full):
        h_best = (h_best / h_sum) * target_sum_full
    return (h_best.tolist(), global_best_c5, n_points)
