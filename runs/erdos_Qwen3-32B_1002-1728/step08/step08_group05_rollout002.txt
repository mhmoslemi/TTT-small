    while time.time() - start_time < budget_s - 0.1:
        for i in range(num_particles):
            # Get current raw position
            current_pos = positions[i].copy()

            # Repair current position: clip and scale
            clipped = np.clip(current_pos, 0, 1)
            sum_clipped = clipped.sum()
            if sum_clipped != 0:
                scale = target_sum_first / sum_clipped
                repaired = np.clip(clipped * scale, 0, 1)
            else:
                repaired = clipped.copy()

            # Construct full h
            full_h = np.zeros(n_points)
            full_h[:half_n] = repaired
            full_h[half_n:] = repaired[::-1]
            full_h = np.clip(full_h, 0, 1)  # Ensure valid for correlation

            # Compute C5
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            current_c5 = np.max(corr) * dx

            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest_c5[i] = current_c5
                pbest_positions[i] = repaired.copy()
                # Check if this is new global best
                if current_c5 < gbest_c5:
                    gbest_c5 = current_c5
                    gbest_pos = repaired.copy()

            # Update velocity and position
            r1 = np.random.rand(half_n)
            r2 = np.random.rand(half_n)
            velocities[i] = (
                phi * velocities[i] + 
                mu * r1 * (pbest_positions[i] - current_pos) + 
                nu * r2 * (gbest_pos - current_pos)
            )
            # Update position
            new_pos = current_pos + velocities[i]
            # Repair new position
            clipped_new = np.clip(new_pos, 0, 1)
            sum_clipped = clipped_new.sum()
            if sum_clipped != 0:
                scale = target_sum_first / sum_clipped
                repaired_new = np.clip(clipped_new * scale, 0, 1)
            else:
                repaired_new = clipped_new.copy()
            positions[i] = repaired_new

    # After loop, construct final h
    full_h = np.zeros(n_points)
    full_h[:half_n] = gbest_pos
    full_h[half_n:] = gbest_pos[::-1]
    full_h = np.clip(full_h, 0, 1)
    sum_h = full_h.sum()
    if abs(sum_h - (n_points / 2)) > 1e-6:
        full_h = full_h / sum_h * (n_points / 2)
        full_h = np.clip(full_h, 0, 1)

    return (full_h.tolist(), gbest_c5, n_points)
