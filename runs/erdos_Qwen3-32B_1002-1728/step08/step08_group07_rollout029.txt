import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    h_first_initial = h_full_initial[:half_n].copy()
    h_first_initial = np.clip(h_first_initial, 0, 1)
    current_sum = h_first_initial.sum()
    if current_sum != 0:
        h_first_initial = (h_first_initial / current_sum) * target_sum_half
    h_first_initial = np.clip(h_first_initial, 0, 1)

    n_particles = 20
    np.random.seed(seed)
    particles = np.tile(h_first_initial[None], (n_particles, 1)).copy()
    pert = np.random.uniform(-0.05, 0.05, particles.shape)
    particles += pert
    for i in range(n_particles):
        p = particles[i]
        p = np.clip(p, 0, 1)
        s = p.sum()
        if s > 1e-9:
            p = p * target_sum_half / s
        else:
            p.fill(target_sum_half / half_n)
        particles[i] = np.clip(p, 0, 1)

    velocities = np.zeros_like(particles)
    pbest_positions = particles.copy()
    pbest_scores = np.full(n_particles, np.inf)

    def evaluate(current):
        full = create_full_h(current)
        corr = np.correlate(full, 1 - full, mode='full') * dx
        return np.max(corr)

    for i in range(n_particles):
        pbest_scores[i] = evaluate(pbest_positions[i])

    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    best_c5 = pbest_scores[gbest_idx]

    inertia = 0.9
    phi_p, phi_g = 2.0, 2.0
    max_iter = int(1e5)
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 2:
            break

        r_p = np.random.rand(n_particles, 1)
        r_g = np.random.rand(n_particles, 1)
        velocities = (inertia * velocities +

                      phi_g * r_g * (gbest_position - particles))
        particles += velocities

        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            p = particles[i]
            s = p.sum()

                p = p * (target_sum_half / s)
            else:
                p.fill(target_sum_half / half_n)
            particles[i] = np.clip(p, 0, 1)

            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                if score < best_c5:
