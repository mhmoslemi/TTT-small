import numpy as np
import time

def transform_coeffs(coeffs, n_points):
    return np.fft.irfft(coeffs, n_points).real

def compute_c5(h_full, dx):
    h_clipped = np.clip(h_full, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr) * dx

def perturb_coeffs(coeffs, temp):
    real_part = np.random.normal(0, temp, size=coeffs.shape)
    imag_part = np.random.normal(0, temp, size=coeffs.shape)
    new_coeffs = coeffs + real_part + 1j * imag_part
    new_coeffs[0] = new_coeffs[0].real  # Ensure DC is real
    new_coeffs[-1] = new_coeffs[-1].real  # Ensure Nyquist is real
    return new_coeffs

def clip_and_normalize(h_candidate, target_sum):
    h_candidate = np.clip(h_candidate, 0, 1)
    current_sum = np.sum(h_candidate)
    h_candidate = h_candidate * (target_sum / current_sum) if abs(current_sum) > 1e-10 else h_candidate
    h_candidate = np.clip(h_candidate, 0, 1)
    return h_candidate
