import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0  # 35 for n_points=70

    def create_full_h(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return np.clip(h, 0, 1)

    def project_h_half(h_half):
        h_half_clipped = np.clip(h_half, 0, 1)
        sum_h = h_half_clipped.sum()
        if sum_h < 1e-12:
            h_half_clipped = np.ones_like(h_half_clipped) * target_sum_half / h_half.size
        else:
            h_half_clipped = h_half_clipped * (target_sum_half / sum_h)
        return np.clip(h_half_clipped, 0, 1)

    num_particles = 20
    particles = []
    velocities = []
    personal_bests = []
    personal_best_values = []

    initial_h_half = project_h_half(h_full_initial[:half_n].copy())
    for _ in range(num_particles):
        pert = np.random.uniform(-0.05, 0.05, size=half_n)
        h_half = project_h_half(initial_h_half + pert)
        particles.append(h_half.copy())
        velocities.append(np.zeros_like(h_half))
        full_h = create_full_h(h_half)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        pbv = np.max(corr)
        personal_bests.append(h_half.copy())
        personal_best_values.append(pbv)

    global_best_value = min(personal_best_values)
    global_best_idx = np.argmin(personal_best_values)
    global_best_h_half = personal_bests[global_best_idx].copy()

    omega = 0.9
    phi_p = 2.0
    phi_g = 2.0
    stability = 0

    while time.time() - start_time < budget_s - 10:
        for i in range(num_particles):
            h_half = particles[i].copy()
            full_h = create_full_h(h_half)
            corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
            current_value = np.max(corr)
            if current_value < personal_best_values[i]:
                personal_bests[i] = h_half.copy()
                personal_best_values[i] = current_value

        current_gb_idx = np.argmin(personal_best_values)
        new_gb_value = personal_best_values[current_gb_idx]
        if new_gb_value < global_best_value:
            global_best_value = new_gb_value
            global_best_h_half = personal_bests[current_gb_idx].copy()
            stability = 0
        else:
            stability += 1

        if stability > 100:
            break

        elapsed = time.time() - start_time
        omega = max(0.4, omega - (0.5) * (elapsed / (budget_s - 10)))

        for i in range(num_particles):
            r1, r2 = np.random.rand(2, half_n)
            vel = (omega * velocities[i] +
                   phi_p * r1 * (personal_bests[i] - particles[i]) +
                   phi_g * r2 * (global_best_h_half - particles[i]))
            particles[i] += vel
            particles[i] = project_h_half(particles[i])
            velocities[i] = vel

    best_full = create_full_h(project_h_half(global_best_h_half))
    corr = np.correlate(best_full, 1 - best_full, mode='full') * dx
    final_c5 = np.max(corr)
    return (best_full.tolist(), final_c5, n_points)
