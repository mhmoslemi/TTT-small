        else:
            candidate_h = h0
            c5_tr = c5_es
        c5_current = min(c5_es, c5_tr)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = candidate_h
        
        h_previous = {'x': x, 'h': best_h}
    
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6), f"Sum constraint violated: {best_h.sum()} vs {target_sum}"
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
