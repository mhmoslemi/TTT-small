import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) * dx = 1 => sum(h) = n_points / 2

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum) > 1e-6:
        scale_factor = target_sum / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    # Objective function for DE and SLSQP
    def objective_full(x):
        h_clipped = np.clip(x, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    cons_full = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    final_h = h_full_initial.copy()
    best_c5 = objective_full(final_h)

    # Phase 1: Differential evolution with enhanced exploration
    def de_full_objective(x):
        h_clipped = np.clip(x, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    de_budget = budget_s * 0.7  # 70% time budget for DE
    start_de = time.time()
    popsize = 40
    maxiter = 30

    try:
        de_result = differential_evolution(
            de_full_objective, bounds_full, strategy='best1bin',
            maxiter=maxiter, popsize=popsize, tol=1e-6,
            mutation=(0.6, 1.0), recombination=0.9, seed=seed, workers=1
        )
        if de_result.success and de_result.fun < best_c5:
            best_h = np.clip(de_result.x, 0, 1)
            if best_h.sum() != 0:
                best_h = (best_h / best_h.sum()) * target_sum
            best_h = np.clip(best_h, 0, 1)
            final_h = best_h
            best_c5 = objective_full(final_h)
    except Exception:
        pass  # Skip on timeout/error

    # Phase 2: Systematic local optimization with SLSQP
    local_steps = 20
    for _ in range(local_steps):
        if time.time() - start_time >= budget_s - 5:
            break
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.2 * (1 - frac_remaining**1.3)

        # Create a perturbed guess
        perturbed = final_h.copy()
        perturb_indices = np.random.choice(n_points, size=min(n_points, 20), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, size=len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        if perturbed.sum() != 0:
            perturbed = (perturbed / perturbed.sum()) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        perturbed[perturbed < 1e-8] = 0

        # Local optimization with SLSQP
        try:
            result = minimize(
                objective_full, x0=perturbed, method='SLSQP',
                bounds=bounds_full, constraints=cons_full,
                options={'maxiter': 500, 'ftol': 1e-12}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                if abs(new_h.sum() - target_sum) < 1e-6:
                    new_c5 = objective_full(new_h)
                    if new_c5 < best_c5:
                        final_h = new_h
                        best_c5 = new_c5
        except:
            pass

    # Final cleanup and constraint repair
    h_full = final_h.copy()

    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
