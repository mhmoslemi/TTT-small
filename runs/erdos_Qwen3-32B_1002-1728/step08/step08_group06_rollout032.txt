                child2 = np.clip(child2, 0, 1)
                offspring[i] = child1
                offspring[i+1] = child2

            # Replace worst in pop with best offspring
            new_candidates = []
            for child in offspring:
                h = repair_full_h(child, n_points, target_sum)
                corr_vals = np.correlate(h, 1 - h, mode='full') * dx
                new_fitness = np.max(corr_vals)
                new_candidates.append((child, new_fitness))
            new_candidates.sort(key=lambda x: x[1])
            new_children = np.array([c[0] for c in new_candidates[:pop_size]])
            pop[:] = new_children
            fitness = np.zeros(pop_size)
            for i in range(pop_size):
                h = repair_full_h(pop[i], n_points, target_sum)
                corr_vals = np.correlate(h, 1 - h, mode='full') * dx
                fitness[i] = np.max(corr_vals)
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5_current:
                best_sym = pop[best_idx]
                best_h_full = repair_full_h(best_sym, n_points, target_sum)
                best_c5_current = fitness[best_idx]

        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_full

        h_previous = {'x': np.linspace(0, 2, n_points), 'h': best_h_full}

    return (best_h.tolist(), best_c5, len(best_h))
