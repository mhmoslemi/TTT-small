        full_h = create_full_h(h_first)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_first = h_first.copy()
        lag_idx = np.argmax(corr)
        active_subset = subsets[active_subset_index % 2]
        grad_subset = compute_gradient_subset(h_first, active_subset, lag_idx)
        velocity[active_subset] = momentum * velocity[active_subset] + grad_subset
        h_first[active_subset] -= learning_rate * velocity[active_subset]
        h_first = np.clip(h_first, 0, 1)
        sum_h = h_first.sum()
        if abs(sum_h - target_sum_half) > 1e-6:
            h_first = (h_first / sum_h) * target_sum_half
            h_first = np.clip(h_first, 0, 1)
        if current_c5 < best_c5:
            learning_rate *= 0.995
        else:
            learning_rate *= 0.5
        momentum = min(momentum + 0.01, 0.9)
        active_subset_index += 1

    best_full = create_full_h(best_h_first)
    best_full = np.clip(best_full, 0, 1)
    best_full_sum = best_full.sum()
    if abs(best_full_sum - target_sum_full) > 1e-6:
        best_full = (best_full / best_full_sum) * target_sum_full
        best_full = np.clip(best_full, 0, 1)
    return (best_full.tolist(), best_c5, n_points)
