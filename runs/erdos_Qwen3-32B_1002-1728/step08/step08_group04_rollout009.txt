                deg = 2 * (k + 1)
                A[i, k] = eval_legendre(deg, ti)
        return A

    # Phase 1: Coarse basis (low frequency)
    k_coarse = 10
    A_coarse = get_legendre_basis(k_coarse)
    target = h_initial - 0.5  # P_0 coefficient is 0.5, isolate the rest
    coeffs_coarse_init = np.linalg.pinv(A_coarse).dot(target)
    coeffs = coeffs_coarse_init.copy()
    best_c5 = float('inf')
    best_h = None
    best_coeffs = coeffs.copy()

    learning_rate = 0.01
    epsilon = 1e-5
    max_iter = 6000
    alpha = 0.999  # LR decay

    # Optimization loop (coarse basis)
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 60:
            break
        h = 0.5 + A_coarse @ coeffs
        h = np.clip(h, 0, 1)
        total = h.sum()
        h = h * (target_sum / total)
        current_c5 = max(np.correlate(h, 1 - h, mode='full') * dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_coeffs = coeffs.copy()
        
        # Compute gradient
        gradient = np.zeros(k_coarse)
        for j in range(k_coarse):
            coeffs_plus = coeffs.copy()
            coeffs_plus[j] += epsilon
            h_plus = 0.5 + A_coarse @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            total_plus = h_plus.sum()
            h_plus = h_plus * (target_sum / max(total_plus, 1e-8))
            c_plus = max(np.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[j] -= epsilon
            h_minus = 0.5 + A_coarse @ coeffs_minus
            h_minus = np.clip(h_minus, 0, 1)
            total_minus = h_minus.sum()
            h_minus = h_minus * (target_sum / max(total_minus, 1e-8))
            c_minus = max(np.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            
            gradient[j] = (c_plus - c_minus) / (2 * epsilon)
        
        coeffs -= learning_rate * gradient
        learning_rate *= alpha

    # Phase 2: Fine basis (switch to high frequency)
    k_fine = 30
    A_fine = get_legendre_basis(k_fine)
    # Project coarse solution to fine basis (zero-extend)
    coeffs_fine = np.zeros(k_fine)
    coeffs_fine[:k_coarse] = best_coeffs.copy()

    coeffs = coeffs_fine.copy()
    learning_rate_fine = 0.005
    epsilon_fine = 1e-5

    for iter in range(max_iter // 2):
        if time.time() - start_time > budget_s - 60:
            break
        h = 0.5 + A_fine @ coeffs
        h = np.clip(h, 0, 1)
        total = h.sum()
        h = h * (target_sum / max(total, 1e-8))
        current_c5 = max(np.correlate(h, 1 - h, mode='full') * dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_coeffs = coeffs.copy()
        
        gradient = np.zeros(k_fine)
        for j in range(k_fine):
            step = coeffs.copy()
            step[j] += epsilon_fine
            h_p = 0.5 + A_fine @ step
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (target_sum / h_p.sum())
            c_p = max(np.correlate(h_p, 1 - h_p, mode='full') * dx)
            
            step[j] -= 2 * epsilon_fine
            h_m = 0.5 + A_fine @ step
            h_m = np.clip(h_m, 0, 1)
            h_m = h_m * (target_sum / h_m.sum())
            c_m = max(np.correlate(h_m, 1 - h_m, mode='full') * dx)
            
            gradient[j] = (c_p - c_m) / (2 * epsilon_fine)
        
        coeffs -= learning_rate_fine * gradient
        learning_rate_fine *= 0.9995  # Faster decay

    # Final repair and validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    total = h.sum()
    h = h * (target_sum / max(total, 1e-8))
    final_c5 = max(np.correlate(h, 1 - h, mode='full') * dx)

    return (h.tolist(), final_c5, n_points)
