def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    levels = [10, 20, 40, 70]
    best_c5 = np.inf
    best_h = None
    
    initial_h_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None
    previous_best_h = None
    
    def repair(h, target_sum):
        h = np.clip(h, 0, 1).copy()
        total = np.sum(h)
        if total == 0:
            return np.ones_like(h) * (target_sum / h.size)
        return h * (target_sum / total)
    
    for idx, n_points in enumerate(levels):
