            bounds = [(0.0, 1.0)] * variable_count
            res = minimize(tr_objective, vars_tr, bounds=bounds, method='L-BFGS-B', options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                p_opt = res.x
                h_opt = create_full_h(p_opt, is_asymmetric, variable_count, x)
                total_opt = h_opt.sum()
                if total_opt != 0:
                    h_opt_scaled = h_opt * (target_sum / total_opt)
                else:
                    h_opt_scaled = np.zeros_like(h_opt)
                h_opt_scaled = np.clip(h_opt_scaled, 0, 1)
                score_opt = objective_full(h_opt_scaled, dx)
                if score_opt < global_best_score:
                    global_best_pos = p_opt.copy()
                    global_best_score = score_opt
        
        # Update best solution
        h_gbest = create_full_h(global_best_pos, is_asymmetric, variable_count, x)
        total = h_gbest.sum()
        if total != 0:
            h_scaled_gbest = h_gbest * (target_sum / total)
        else:
            h_scaled_gbest = np.zeros_like(h_gbest)
        h_scaled_gbest = np.clip(h_scaled_gbest, 0, 1)
        score_gbest = objective_full(h_scaled_gbest, dx)
        
        if score_gbest < best_c5:
            best_c5 = score_gbest
            best_h = h_scaled_gbest
            previous_solution = {'x': x, 'h': best_h}
        
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final validation
    if best_h is None:
        best_h = np.array(initial_h_values_given) if initial_h_values_given is not None else np.full(70, 0.5)
    assert np.isclose(best_h.sum(), len(best_h) / 2, atol=1e-6)
    return (best_h.tolist(), best_c5, len(best_h))
