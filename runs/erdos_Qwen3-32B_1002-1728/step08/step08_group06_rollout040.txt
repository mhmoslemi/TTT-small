import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        h_proj = np.full_like(h_clipped, 0.5)
        return h_proj
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    levels = [10, 20, 40, 70, 100]
    best_c5 = np.inf
    best_h = None
    initial_h_values_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None
    
    prev_solution = None  # To store solution from previous grid
    
    for n_points in levels:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        target_sum = n_points / 2.0
        
        # Initialize h_initial for current grid
        if n_points == 70 and initial_h_values_given is not None:
            h_initial = initial_h_values_given.copy()
        else:
            if prev_solution is None:
                h_initial = np.full(n_points, target_sum / n_points)
                h_initial = np.clip(h_initial, 0, 1)
            else:
                prev_h = prev_solution['h']
                prev_x = prev_solution['x']
                h_initial = np.interp(x, prev_x, prev_h)
                h_initial = np.clip(h_initial, 0, 1)
                h_initial = project_feasible(h_initial, target_sum)
        
        # Evolution Strategy
        pop_size = 50
        elite_size = 10
        num_generations = min(int(remaining_time * 0.5 // 0.5), 50)
        mutation_scale = 0.1
        
        # Initialize population
        population = []
        for _ in range(pop_size):
            candidate = np.random.uniform(0, 1, n_points)
            candidate = project_feasible(candidate, target_sum)
            corr = np.correlate(candidate, 1 - candidate, mode='full') * dx
            c5 = np.max(corr)
            population.append({'h': candidate, 'c5': c5})
        
        # Evolutionary generations
        for gen in range(num_generations):
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time <= 0:
                break
            
            # Sort and select elites
            population.sort(key=lambda x: x['c5'])
            elites = population[:elite_size]
            
            new_population = []
            for _ in range(pop_size):
                parent = elites[np.random.randint(elite_size)]
                candidate = parent['h'].copy()
                candidate += np.random.normal(0, mutation_scale, n_points)
                candidate = project_feasible(candidate, target_sum)
                corr = np.correlate(candidate, 1 - candidate, mode='full') * dx
                c5 = np.max(corr)
                new_population.append({'h': candidate, 'c5': c5})
            
            population = new_population
            
            # Adjust mutation scale
            if gen > 0 and population[0]['c5'] < population[:10][0]['c5']:
                mutation_scale *= 1.1 if mutation_scale < 1.0 else 1.0
            else:
                mutation_scale *= 0.9 if mutation_scale > 1e-5 else 1e-5
        
        # Select best from ES
        population.sort(key=lambda x: x['c5'])
        best_candidate = population[0]
        best_candidate_h = best_candidate['h']
        best_candidate_c5 = best_candidate['c5']
        
        # Trust-Region refinement
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(corr)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(
            objective,
            best_candidate_h,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50}
        )
        
        if result.success:
            optimized_h = result.x
            optimized_h = np.clip(optimized_h, 0, 1)
            corr = np.correlate(optimized_h, 1 - optimized_h, mode='full') * dx
            optimized_c5 = np.max(corr)
            if optimized_c5 < best_candidate_c5:
                best_candidate_h = optimized_h
                best_candidate_c5 = optimized_c5
        
        # Update best solution

            best_c5 = best_candidate_c5
            best_h = best_candidate_h.copy()
            prev_solution = {'x': x, 'h': best_h}
    
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
