        elapsed_time = time.time() - start_time
        t_ratio = elapsed_time / max_pso_time
        w = w_start - (w_start - w_end) * t_ratio
        w = max(w, w_end)

        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_cognitive = c1 * r1 * (personal_best_positions[i] - particles_positions[i])
            vel_social = c2 * r2 * (global_best_position - particles_positions[i])
            new_vel = w * velocities[i] + vel_cognitive + vel_social
            new_vel = np.clip(new_vel, -max_vel, max_vel)
            velocities[i] = new_vel

            new_pos = particles_positions[i] + new_vel
            new_pos = project(new_pos)
            particles_positions[i] = new_pos

            score = objective(new_pos)
            if score < personal_best_scores[i] - 1e-10:
                personal_best_positions[i] = new_pos.copy()
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best_position[:] = new_pos
                    global_best_score = score

    remaining_time = budget_s - (time.time() - start_time)
    final_h = project(global_best_position)
    final_c5 = objective(final_h)

    if remaining_time > 0:
        try:
            def objective_slsqp(x):
                return objective(project(x))
            bounds = [(0, 1)] * n_points
            cons = ({'type': 'eq', 'fun': lambda x: project(x).sum() - target_sum})
            result = minimize(
                objective_slsqp,
                x0=global_best_position,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12}
            )
            if result.success:
                final_h = project(result.x)
                final_c5 = objective(final_h)
        except:
            pass

    final_h = project(final_h)
    return (final_h.tolist(), final_c5, n_points)
