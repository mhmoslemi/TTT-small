            h_ref = project_to_feasible(create_full_h(x))
            return np.max(np.correlate(h_ref, 1 - h_ref, mode='full') * dx)
        
        res = minimize(
            refine_objective, x0=best_h[:half_n], bounds=bounds,
            method='SLSQP', options={'maxiter': 200, 'ftol': 1e-10}
        )
        if res.success:
            best_h = create_full_h(res.x)
            best_h = project_to_feasible(best_h)
            best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    assert best_h is not None
    best_h = np.clip(best_h, 0, 1)
    assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
    
    return (best_h.tolist(), best_c5, len(best_h))
