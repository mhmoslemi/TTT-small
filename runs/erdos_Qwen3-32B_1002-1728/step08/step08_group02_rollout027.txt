def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)

    target_sum = n_points / 2.0
    target_sum_half = target_sum / 2.0

    h_half = h_full_initial[:half_n].copy()
    h_half = np.clip(h_half, 0, 1)
    h_half = (h_half / np.clip(h_half.sum(), 1e-8, None) * target_sum_half)
    h_half = np.clip(h_half, 0, 1)

    λ = 0.0
    ρ = 1.0
    Δ = 0.1
    max_Δ = 1.0
    min_Δ = 0.01
    τ = 10.0

    best_h = h_half.copy()
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 10:
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = np.clip(h_full, 0, 1)
        if np.abs(h_full.sum() - target_sum) > 1e-6:
            h_full = (h_full / h_full.sum()) * target_sum
            h_full = np.clip(h_full, 0, 1)

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_index = np.argmax(corr)
        k_lag = max_index - (n_points - 1)

        grad_full = np.zeros(n_points)
        for x in range(n_points):
            x_plus_k = x + k_lag
            if 0 <= x_plus_k < n_points:
                grad_full[x] += 1 - h_full[x_plus_k]
                grad_full[x_plus_k] -= h_full[x]
        grad_full *= dx
        grad_half = 2 * grad_full[:half_n]

        sum_h_half = h_half.sum()
        constraint_violation = sum_h_half - target_sum_half
        grad_L = grad_half + λ + ρ * constraint_violation

        grad_norm = np.linalg.norm(grad_L)
        if grad_norm > 0:
            direction = -grad_L / grad_norm
            step = Δ * direction
            new_h_half = h_half + step
            new_h_half = np.clip(new_h_half, 0, 1)
            new_h_half = (new_h_half / np.clip(new_h_half.sum(), 1e-8, None) * target_sum_half)
