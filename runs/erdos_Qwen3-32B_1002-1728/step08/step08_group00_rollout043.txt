import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize h
    h = h_initial.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != 0:
        h = (h / h_sum) * target_sum
    h = np.clip(h, 0, 1)

    # Objective function
    def compute_c5(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    # Tabu search setup
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    tabu_list = []
    tabu_size = 20
    max_tabu_iter = 300
    perturb_strength = 0.2
    candidate_count = 50
    perturb_range = (5, 10)

    # Tabu search loop
    for tabu_iter in range(max_tabu_iter):
        if time.time() - start_time > budget_s * 0.7:
            break
        candidates = []
        for _ in range(candidate_count):
            candidate = best_h.copy()
            num = np.random.randint(*perturb_range)
            indices = np.random.choice(n_points, num, replace=False)
            for i in indices:
                delta = np.random.uniform(-perturb_strength, perturb_strength)
                candidate[i] += delta
                candidate[i] = np.clip(candidate[i], 0, 1)
            candidate_sum = candidate.sum()
            if candidate_sum != 0:
                candidate = (candidate / candidate_sum) * target_sum
            candidate = np.clip(candidate, 0, 1)
            # Check tabu
            rounded_c = tuple(np.round(candidate, 5))
            if rounded_c not in [tuple(np.round(t, 5)) for t in tabu_list]:
                candidates.append((candidate, rounded_c))
        if not candidates:
            continue
        scores = [compute_c5(c[0]) for c in candidates]
        best_idx = np.argmin(scores)
        new_candidate, new_rounded = candidates[best_idx]
        if scores[best_idx] < best_c5:
            best_h = new_candidate.copy()
            best_c5 = scores[best_idx]
        tabu_list.append(new_rounded)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)

    # Local refinement
    def objective(x):
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    while time.time() - start_time < budget_s - 5:
        # Perturb initial guess
        x0 = best_h + np.random.uniform(-0.05, 0.05, n_points)
        x0 = np.clip(x0, 0, 1)
        x0_sum = x0.sum()
        if x0_sum != 0:
            x0 = (x0 / x0_sum) * target_sum
        x0 = np.clip(x0, 0, 1)
        try:
            result = minimize(objective, x0=x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except:
            pass

    # Final constraint enforcement
    h_final = np.clip(best_h, 0, 1)
    h_sum = h_final.sum()
    if abs(h_sum - target_sum) > 1e-6:
        h_final = (h_final / h_sum) * target_sum
        h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), best_c5, n_points)
