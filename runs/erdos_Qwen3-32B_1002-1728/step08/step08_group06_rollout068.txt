def tournament_selection(population, fitnesses, tournament_size=5):

   # tournament_size is a proportion, e.g., 5% of population

   population_size = len(population)

   selected = []

   for _ in range(2):  # for selecting two parents per offspring

      tournament_indices = np.random.choice(population_size, tournament_size)

      candidates = [ (population[i], fitnesses[i]) for i in tournament_indices ]

      candidates.sort(key=lambda x: x[1])  # sort by fitness (lowest C₅)

      selected.append( candidates[0][0] )  # select best in tournament

   return selected[0], selected[1]

def crossover(parent1, parent2):

   alpha = np.random.uniform(0.3, 0.7)

   child1 = alpha * parent1 + (1 - alpha) * parent2

def mutate(child, mutation_rate=0.2, sigma=0.1):

   num_genes = len(child)

   mask = np.random.rand(num_genes) < mutation_rate

   child[mask] += np.random.normal(0, sigma, size=child[mask].shape)

   child = np.clip(child, 0, 1)
