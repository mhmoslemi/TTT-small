    start_time = time.time()
    np.random.seed(seed)
    
    levels = [10, 20, 40, 70]  # Multi-resolution grid sizes
    best_c5 = np.inf
    best_h = None
    prev_solution = None  # Store refined solution from previous grid level
    initial_h_values_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None
    
    for n_points in levels:
        if time.time() - start_time > budget_s - 30:
            break
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        half_n = n_points // 2
        target_half_sum = target_sum / 2.0
        
        # Initialize sym_vars for current grid
        sym_vars0 = None
        
        if prev_solution and prev_solution['n_points'] < n_points:
            # Interpolate from previous solution
            old_x = prev_solution['x']
            old_h_full = prev_solution['h']
            new_x = np.linspace(0, 2, n_points)
            new_h_full = np.interp(new_x, old_x, old_h_full)
            sym_init = new_h_full[:half_n]
            sym_sum = np.sum(sym_init)
            if sym_sum != 0:
                sym_scaled = sym_init * (target_half_sum / sym_sum)
            else:
                sym_scaled = np.zeros_like(sym_init)
            sym_vars0 = np.clip(sym_scaled, 0, 1)
        elif n_points == 70 and initial_h_values_given is not None:
            # Use provided initial construction for finest grid
            sym_init = initial_h_values_given[:half_n].copy()
            sym_sum = np.sum(sym_init)
            if sym_sum != 0:
                sym_scaled = sym_init * (target_half_sum / sym_sum)
            else:
                sym_scaled = np.zeros_like(sym_init)
            sym_vars0 = np.clip(sym_scaled, 0, 1)
        else:
            # Uniform initialization as fallback
            sym_vars0 = np.full(half_n, target_half_sum / half_n)
            sym_vars0 = np.clip(sym_vars0, 0, 1)
        
        # Objective function with symmetry
        def objective(sym_vars):
            sym_vars = np.clip(sym_vars, 0, 1)
            h_full = np.zeros(n_points)
            h_full[:half_n] = sym_vars
            h_full[-half_n:] = sym_vars[::-1]
            corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
            return np.max(corr)
        
        # Optimization constraints and bounds
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_half_sum})
        bounds = [(0.0, 1.0)] * half_n
        
        # Adaptive iteration limit based on remaining time
        remaining = budget_s - (time.time() - start_time)
        maxiter = max(1, min(100, int(remaining * 0.7)))
        
        # Perform constrained optimization
        result = None
        try:
            result = minimize(
                objective, 
                x0=sym_vars0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'maxiter': maxiter, 'ftol': 1e-8}
            )
        except Exception:
            pass
        
        # Extract optimized solution
        sym_best = sym_vars0 if (result is None or not result.success) else result.x
        
        # Construct full solution from optimized variables
        h_full = np.zeros(n_points)
        h_full[:half_n] = sym_best
        h_full[-half_n:] = sym_best[::-1]
        h_full = np.clip(h_full, 0, 1)
        
        # Final projection back to feasible set
        total = h_full.sum()
        if total != 0:
            h_full = h_full * (target_sum / total)
        h_full = np.clip(h_full, 0, 1)
        
        # Evaluate current solution
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        c5_current = np.max(corr)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_full
            
        # Store for next level
        x_current = np.linspace(0, 2, n_points)
        prev_solution = {'x': x_current, 'h': h_full, 'n_points': n_points}
    
    # Final validation and return
    if best_h is not None:
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
