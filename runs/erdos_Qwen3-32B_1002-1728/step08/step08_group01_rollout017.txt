from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution to strict constraints
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum_full / current_sum if current_sum > 1e-9 else 0
    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)
    
    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    repair_cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

    best_half = h_full_initial[:half_n].copy()
    best_h_full = create_full_h(best_half)
    best_c5 = objective(best_half)
    
    np.random.seed(seed)
    
    # Enhanced global search with currenttobest1bin strategy
    de_budget = 0.45 * budget_s
    max_de_time = int(de_budget)
    de_popsize = 40
    de_maxiter = max(12, int(max_de_time * 1e6 / (n_points**2.1 * de_popsize)))
    
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='currenttobest1bin',
            mutation=(0.8, 1.0), recombination=0.92, 
            maxiter=de_maxiter, popsize=de_popsize,
            seed=seed, workers=1, disp=False,
            polish=False, init='sobol'
        )
        if de_result.success and de_result.fun < best_c5:
            cand = np.clip(de_result.x, 0, 1)
            cand_sum = cand.sum()
            if cand_sum != 0:
                cand *= target_sum_half / cand_sum
                cand = np.clip(cand, 0, 1)
                cand_c5 = objective(cand)
                if cand_c5 < best_c5:
                    best_half = cand
                    best_c5 = cand_c5
                    best_h_full = create_full_h(best_half)
    except:
        pass

    # Dynamic local optimization with focused steps
    while time.time() - start_time < budget_s - 10:
        perturb_pct = max(0.1, 1.5 - (time.time()-start_time)/budget_s)
        perturb_size = max(3, int(half_n * perturb_pct))
        perturb_indices = np.random.choice(half_n, perturb_size, replace=False)
        
        noise_scale = 0.15 * (1 - (time.time()-start_time)/budget_s)**1.3
        perturbed = best_half.copy()
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        
        # Project to feasible region
        p_sum = perturbed.sum()
        if abs(p_sum - target_sum_half) > 1e-6:
            perturbed *= target_sum_half / p_sum
            perturbed = np.clip(perturbed, 0, 1)
        
        for method in ['L-BFGS-B', 'SLSQP']:
            try:
                result = minimize(
                    objective, perturbed, method=method,

        h_full = np.clip(h_full, 0, 1)
