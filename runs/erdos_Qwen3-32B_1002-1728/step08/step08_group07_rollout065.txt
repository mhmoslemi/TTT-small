        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        c5 = np.max(corr)
        particles_pbest_c5[i] = c5
    global_best_idx = np.argmin(particles_pbest_c5)
    global_best_pos = particles_pbest_pos[global_best_idx].copy()
    global_best_c5 = particles_pbest_c5[global_best_idx]
    global_best_full = create_full_h(global_best_pos)
    last_improvement = start_time
    
    # PSO parameters
    phi_p = 2.0
    phi_g = 2.0
    patience = 100
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i in range(n_particles):
            full_h = create_full_h(particles_positions[i])
            corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
            c5 = np.max(corr)
            if c5 < particles_pbest_c5[i]:
                particles_pbest_c5[i] = c5
                particles_pbest_pos[i] = particles_positions[i].copy()
        
        # Update global best
        current_global = particles_pbest_c5.min()
        current_global_idx = particles_pbest_c5.argmin()
        if current_global < global_best_c5:
            global_best_c5 = current_global
            global_best_pos = particles_pbest_pos[current_global_idx].copy()
            global_best_full = create_full_h(global_best_pos)
            last_improvement = time.time() - start_time
        
        # Check convergence
        if (time.time() - start_time - last_improvement) > (patience * budget_s / 1000):
            break
        
        # Update velocities and positions
        current_time = time.time() - start_time
        omega = max(0.4, 0.9 - (0.9 - 0.4) * current_time / budget_s)
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive = phi_p * r1 * (particles_pbest_pos[i] - particles_positions[i])
            social = phi_g * r2 * (global_best_pos - particles_positions[i])
            particles_velocities[i] = omega * particles_velocities[i] + cognitive + social
            
            # Update position
            particles_positions[i] += particles_velocities[i]
            particles_positions[i] = np.clip(particles_positions[i], 0, 1)
            sum_p = particles_positions[i].sum()
            if sum_p < 1e-9:
                particles_positions[i] = np.full(half_n, target_sum_half / half_n)
            else:
                particles_positions[i] = (particles_positions[i] / sum_p) * target_sum_half
            particles_positions[i] = np.clip(particles_positions[i], 0, 1)
    
    # Final processing
    best_full = create_full_h(global_best_pos)
    best_full_sum = best_full.sum()
    if abs(best_full_sum - n_points / 2) > 1e-6:
        best_full = (best_full / best_full_sum) * (n_points / 2)
        best_full = np.clip(best_full, 0, 1)
    corr = np.correlate(best_full, 1 - best_full, mode='full') * dx
    final_c5 = np.max(corr)
    return (best_full.tolist(), final_c5, n_points)
