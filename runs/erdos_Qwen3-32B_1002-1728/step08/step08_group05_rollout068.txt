import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)

    h_initial_full = np.clip(h_initial_full, 0, 1)
    current_sum = h_initial_full.sum()
    if abs(current_sum - target_sum) > 1e-6:
        h_initial_full *= target_sum / current_sum
        h_initial_half = h_initial_full[:n_half]

    # CVXPY setup
    h_half = cp.Variable(n_half, nonneg=True)
    h_full = cp.hstack([h_half, h_half[::-1]])
    constraints = [cp.sum(h_full) == target_sum]
    constraints += [h_full <= 1.0]

    C5 = cp.Variable(nonneg=True)
    max_k = min(20, n_half)  # consider top K shifts

    def add_overlap_constraints(k_values):
        for k in k_values:
            valid_i = []
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    valid_i.append(i)
            if not valid_i:
                continue
            terms = sum((h_full[i] + 1.0 - h_full[i + k])**2 for i in valid_i)
            constraints.append(terms <= 2.0 * C5 * n_points)

    # Add constraints for positive, negative, and zero shifts
    add_overlap_constraints(range(1, max_k + 1))  # positive shifts
    add_overlap_constraints(range(-max_k, 0))     # negative shifts

    # Objective and solve
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)

    remaining = budget_s - (time.time() - start_time)
