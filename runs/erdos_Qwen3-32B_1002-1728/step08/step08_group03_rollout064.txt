import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points
    
    h_half_initial = h_full_initial[:half_n].copy()
    current_sum = h_half_initial.sum()
    if abs(current_sum - target_sum_half) > 1e-9:
        h_half_initial = (h_half_initial / current_sum) * target_sum_half
    h_half_initial = np.clip(h_half_initial, 0, 1.0)
    
    np.random.seed(seed)
    
    num_starts = 20
    starts = [h_half_initial.copy()]
    for _ in range(num_starts - 1):
        perturbed = h_half_initial + 0.05 * np.random.randn(half_n)
        perturbed = np.clip(perturbed, 0, 1.0)
        perturbed_sum = perturbed.sum()
        if abs(perturbed_sum - target_sum_half) > 1e-9:
            perturbed = (perturbed / perturbed_sum) * target_sum_half
        starts.append(perturbed)
    
    best_c5 = np.inf
    best_h_half = h_half_initial.copy()
    
    for start in starts:
        if time.time() - start_time > budget_s - 1:
            break
        
        def objective(x):
            h_half = np.clip(x, 0, 1.0)
            h_full = np.zeros(n_points)
            h_full[:half_n] = h_half
            h_full[half_n:] = h_half[::-1]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
        bounds = [(0, 1)] * half_n
        
        result = scipy.optimize.minimize(
            objective, 
            x0=start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            tol=1e-6
        )
        
        if result.success and result.fun < best_c5:
            candidate = np.clip(result.x, 0, 1.0)
            h_full = np.zeros(n_points)
            h_full[:half_n] = candidate
            h_full[half_n:] = candidate[::-1]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            validated_c5 = np.max(corr) * dx
            if validated_c5 < best_c5:
                best_c5 = validated_c5
                best_h_half = candidate
    
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
