        if h_sum != 0:
            h_scaled = h_clipped / h_sum * target_sum
        else:
            h_scaled = np.zeros_like(h)
        return np.clip(h_scaled, 0, 1)

    def evaluate(h_half_candidate):
        h_full_candidate = np.concatenate([h_half_candidate, h_half_candidate[::-1]])
        h_full_candidate = normalize_h(h_full_candidate, target_sum_full)
        corr = np.correlate(h_full_candidate, 1 - h_full_candidate, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h_half_current):
        f_current = evaluate(h_half_current)
        eps = 1e-5
        grad = np.zeros_like(h_half_current)
        for i in range(len(h_half_current)):
            h_perturbed = h_half_current.copy()
            h_perturbed[i] += eps
            f_perturbed = evaluate(h_perturbed)
            grad[i] = (f_perturbed - f_current) / eps
        return grad

    h_half = h_half_initial.copy()
    best_c5 = evaluate(h_half)
    h_half_best = h_half.copy()
    trust_radius = 0.1
    max_iterations = 500
    iteration = 0

    while time.time() - start_time < budget_s - 10 and iteration < max_iterations:
        iteration += 1
        grad = compute_gradient(h_half)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-8:
            break
        direction = -grad / grad_norm
        step = direction * trust_radius
        h_half_candidate = h_half + step
        h_half_candidate = normalize_h(h_half_candidate, target_sum_half)
        c5_candidate = evaluate(h_half_candidate)
        c5_current = evaluate(h_half)
        if c5_candidate < c5_current:
            h_half = h_half_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                h_half_best = h_half.copy()
            trust_radius = min(trust_radius * 1.5, 0.5)
        else:
            trust_radius = max(trust_radius * 0.5, 1e-6)

    h_full = np.concatenate([h_half_best, h_half_best[::-1]])
    h_full = normalize_h(h_full, target_sum_full)
    return (h_full.tolist(), best_c5, n_points)
