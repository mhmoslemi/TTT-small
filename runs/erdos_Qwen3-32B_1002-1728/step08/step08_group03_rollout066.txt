    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points
    h_half_initial = h_full_initial[:half_n].copy()

    np.random.seed(seed)
    num_starts = 40
    initial_guesses = []

    for _ in range(num_starts):
        perturbation = 0.02 * np.random.randn(half_n)
        perturbed = h_half_initial + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        total = np.clip(perturbed.sum(), 1e-9, None)
        perturbed = (perturbed / total) * target_sum_half
        initial_guesses.append(perturbed)
    initial_guesses = [h_half_initial] + initial_guesses

    best_h_half = h_half_initial.copy()
    best_c5 = np.inf

    def objective(x):
        h_full = np.zeros(2 * half_n)
        h_full[:half_n] = x
        h_full[half_n:] = x[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    sum_constr = {
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum_half
    }

    for guess in initial_guesses:
        if time.time() - start_time >= budget_s:
            break

        result = opt.minimize(
            objective,
            x0=guess.copy(),
            bounds=bounds,
            constraints=sum_constr,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        if result.success:
            current_c5 = result.fun
            current_h_half = result.x
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_half = current_h_half.copy()

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    total = max(h_full.sum(), 1e-9)
    h_full = (h_full / total) * target_sum_full

    return (h_full.tolist(), best_c5, n_points)
