import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    def repair(h_half, target_sum):
        h_clipped = np.clip(h_half, 0, 1)
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0:
            return np.full_like(h_clipped, target_sum / len(h_clipped))
        scale = target_sum / sum_clipped
        h_scaled = h_clipped * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    population_size = 30
    population = []
    random.seed(seed)
    np.random.seed(seed)

    # Initialize population from initial_h_values
    h_half_initial = h_full_initial[:half_n].copy()
    for _ in range(population_size):
        individual = h_half_initial + np.random.normal(0, 0.1, half_n)
        population.append(repair(individual, target_sum_half))

    def evaluate_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * (2.0 / n_points)

    fitness = [evaluate_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h_half = population[best_idx].copy()

    while time.time() - start_time < budget_s - 0.1:
        # Select top 50%
        top_k = population_size // 2
        sorted_indices = np.argsort(fitness)
        parent_indices = sorted_indices[:top_k]
        parents = [population[i] for i in parent_indices]
        parent_fitness = [fitness[i] for i in parent_indices]

        # Generate offspring
        offspring = []
        offspring_fitness = []
        for _ in range(population_size - top_k):
            p1, p2 = random.sample(parents, 2)
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            mut_type = np.random.randint(3)
            if mut_type == 0:  # single
                idx = np.random.randint(half_n)
                child[idx] += np.random.normal(0, 0.1)
            elif mut_type == 1:  # block
                size = min(np.random.randint(2, 6), half_n)
                start = np.random.randint(half_n - size + 1)
                child[start:start+size] += np.random.normal(0, 0.1, size)
            else:  # global
                child += np.random.normal(0, 0.05, half_n)
            repaired = repair(child, target_sum_half)
            c5 = evaluate_c5(repaired)
            offspring.append(repaired)
            offspring_fitness.append(c5)

        # Combine and select best
        all_individuals = parents + offspring
        all_c5 = parent_fitness + offspring_fitness
        best_indices = np.argsort(all_c5)[:population_size]
        new_population = [all_individuals[i] for i in best_indices]
        new_fitness = [all_c5[i] for i in best_indices]

        population, fitness = new_population, new_fitness
        current_best_idx = np.argmin(fitness)

            best_c5 = fitness[current_best_idx]
            best_h_half = population[current_best_idx].copy()
