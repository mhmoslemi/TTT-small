import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    rng = np.random.RandomState(seed)
    
    # Perturb initial h to escape symmetry
    perturbation = (rng.rand(n_points) - 0.5) * 0.02
    h = h_initial.copy() + perturbation
    h = np.clip(h, 0, 1)
    sum_h = h.sum()
    h = h * (target_sum / sum_h) if sum_h != 0 else np.full(n_points, target_sum / n_points)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    even_indices = np.where(np.arange(n_points) % 2 == 0)[0]
    odd_indices = np.where(np.arange(n_points) % 2 == 1)[0]
    step = 0
    
    velocity = np.zeros_like(h)
    learning_rate = 0.01
    momentum = 0.1
    
    while time.time() - start_time < budget_s - 10:
        indices = even_indices if step % 2 == 0 else odd_indices
        
        # Compute current max overlap and lag
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        lag_idx = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Gradient computation for current indices
        grad = np.zeros(n_points)
        ε = 1e-5
        for i in indices:
            og = h[i]
            # Perturb upward
            h_up = h.copy()
            h_up[i] = og + ε
            corr_up = np.correlate(h_up, 1 - h_up, mode='full') * dx
            val_up = corr_up[lag_idx]
            
            # Perturb downward
            h_dn = h.copy()
            h_dn[i] = og - ε
            corr_dn = np.correlate(h_dn, 1 - h_dn, mode='full') * dx
            val_dn = corr_dn[lag_idx]
            
            grad[i] = (val_up - val_dn) / (2 * ε)
            
        # Update momentum and parameters
        velocity[indices] = momentum * velocity[indices] + grad[indices]
        h[indices] -= learning_rate * velocity[indices]
        
        # Projection to maintain constraints
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        h = h * (target_sum / sum_h) if sum_h != 0 else np.full(n_points, target_sum / n_points)
        h = np.clip(h, 0, 1)
        
        # Adaptive parameters
        momentum = min(0.9, momentum * 1.02)
        learning_rate *= 0.995
        step += 1
    
    return (best_h.tolist(), best_c5, n_points)
