import time

def create_full_h(h_half):
    n_half = len(h_half)
    full_h = np.zeros(2 * n_half)
    full_h[:n_half] = h_half
    full_h[-n_half:] = h_half[::-1]
    return full_h

def project_position(h_half, target_sum_half):
    h_clipped = np.clip(h_half, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum < 1e-9:
        return np.zeros_like(h_half) + target_sum_half / h_half.size
    h_scaled = h_clipped * (target_sum_half / current_sum)
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
