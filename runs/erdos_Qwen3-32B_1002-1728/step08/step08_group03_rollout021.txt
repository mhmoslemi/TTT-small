import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_half = h_full_initial[:half_n].copy()
    h_half *= target_sum_half / max(1e-9, h_half.sum())

    np.random.seed(seed)
    learning_rate = 0.01
    decay_rate = 0.995
    best_C5 = np.inf
    best_h_half = h_half.copy()

    while time.time() - start_time < budget_s:
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1.0)

        # Find k_max that maximizes the overlap integral
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        k_max = np.argmax(corr) - (n_points - 1)

        # Compute gradient for k_max
        grad_full = np.zeros(n_points)
        for y in range(n_points):
            yk = y + k_max
            if 0 <= yk < n_points:
                grad_full[y] += (1 - h_full[yk]) * dx
                grad_full[yk] -= h_full[y] * dx

        # Project gradient to h_half (symmetric positions)
        grad_half = np.zeros(half_n)
        for x in range(half_n):
            mirrored_x = n_points - x - 1
            grad_half[x] = grad_full[x] + grad_full[mirrored_x]

        # Gradient descent update
        h_half -= learning_rate * grad_half
        h_half = np.clip(h_half, 0, 1.0)
        h_half *= target_sum_half / max(1e-9, h_half.sum())

        # Evaluate current C5 and track best solution
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_C5 = np.max(corr)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h_half = h_half.copy()

        learning_rate *= decay_rate

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1.0)

    # Final normalization
    total = h_full.sum()
    if not np.isclose(total, target_sum_full, atol=1e-6):
        h_full = (h_full / total) * target_sum_full
        h_full = np.clip(h_full, 0, 1.0)

    return (h_full.tolist(), best_C5, n_points)
