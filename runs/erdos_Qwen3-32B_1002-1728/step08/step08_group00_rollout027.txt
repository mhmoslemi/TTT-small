from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    h_array = np.array(initial_h_values)
    n_points = len(h_array)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')

    h_sum = h.sum()
    if h_sum != 0:
        h = h / h_sum * target_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()

    tabu_size = 20
    tabu_list = []
    max_candidates = 50
    max_perturb = 0.2
    tabu_budget = budget_s * 0.7

    while time.time() - start_time < tabu_budget:
        candidates = []
        for _ in range(max_candidates):
            indices = np.random.choice(n_points, size=np.random.randint(5, 11), replace=False)

                delta = np.random.uniform(-max_perturb, max_perturb)
                new_h[idx] += delta
            new_h = np.clip(new_h, 0, 1)
            new_sum = new_h.sum()

            new_h = np.clip(new_h, 0, 1)
            c5 = compute_c5(new_h)
