        perturbed = np.clip(perturbed, 0, 1)
        perturbed = (perturbed / perturbed.sum()) * target_sum if perturbed.sum() != 0 else perturbed
        perturbed = np.clip(perturbed, 0, 1)

        try:
            result = minimize(
                objective_full, x0=perturbed, method='SLSQP',
                bounds=bounds_full, constraints=cons_full,
                options={'maxiter': 500, 'ftol': 1e-12}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h = (new_h / new_h.sum()) * target_sum if abs(new_h.sum() - target_sum) > 1e-6 else new_h
                new_c5 = objective_full(new_h)
                if new_c5 < best_c5:
                    final_h = new_h
                    best_c5 = new_c5
        except Exception as e:
            pass

    # Final constraint enforcement
    h_full = final_h.copy()
    h_full = np.clip(h_full, 0, 1)
    if abs(h_full.sum() - target_sum) > 1e-6:
        h_full = (h_full / h_full.sum()) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
