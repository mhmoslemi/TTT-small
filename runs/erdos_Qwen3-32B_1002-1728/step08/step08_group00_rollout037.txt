import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(x):
        x_clipped = np.clip(x, 0, 1)
        x_sum = x_clipped.sum()
        if x_sum == 0:
            scaled = np.zeros_like(x_clipped)
        else:
            scaled = (x_clipped / x_sum) * target_sum
        return np.clip(scaled, 0, 1)

    def objective(x):
        h_scaled = clip_and_scale(x)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    h_repaired = clip_and_scale(h_initial)
    best_h = h_repaired
    best_c5 = objective(h_repaired)

    # Phase 1: Differential Evolution with adaptive mutation
    pop_size = 40
    max_fevals_de = 30 * pop_size * n_points  # Rough estimate of cost
    de_time = min(0.6 * budget_s, budget_s - (time.time() - start_time))
    maxiter_de = int(de_time * pop_size / (n_points * n_points) * 100)

    try:
        de_result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            mutation=(0.6, 1.0),
            recombination=0.9,
            popsize=pop_size,
            tol=1e-6,
            maxiter=maxiter_de,
            seed=seed,
            workers=1,
            updating='deferred'
        )
        if de_result.convergence == 0:
            best_h = clip_and_scale(de_result.x)
            best_c5 = objective(best_h)
    except:
        pass

    # Phase 2: Systematic local refinement
    perturb_attempts = min(20, int(0.1 * n_points))
    while time.time() - start_time < budget_s - 5:
        frac = min(1.0, (budget_s - (time.time() - start_time)) / budget_s)
        perturbation_keys = [hash((seed + int(1000 * t)) % (1 << 64)) for t in np.linspace(0, frac, perturb_attempts)]

        for perturb_key in perturbation_keys[:int(frac * perturb_attempts)]:
            np.random.seed(int((perturb_key + 1e6) % (1 << 30)))
            perturbed = best_h.copy()
            perturb_indices = np.random.choice(n_points, size=min(n_points, 8), replace=False)
            noise = np.random.normal(0, 0.2 * frac**0.5, size=len(perturb_indices))
            perturbed[perturb_indices] = np.clip(perturbed[perturb_indices] + noise, 0, 1)

            try:
                result = minimize(
                    objective,
                    x0=clip_and_scale(perturbed),
                    method='SLSQP',
                    bounds=bounds,
                    options={
                        'maxiter': int(500 * frac**0.7),
                        'ftol': 1e-10,
                        'eps': 1e-6
                    }
                )
                if result.success:
                    candidate = clip_and_scale(result.x)
                    cand_c5 = objective(candidate)
                    if cand_c5 < best_c5:
                        best_h = candidate
                        best_c5 = cand_c5
            except:
                pass

    # Final repair
    best_h = clip_and_scale(best_h)
    return (best_h.tolist(), best_c5, n_points)
