def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initial repair
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    def compute_c5(h_eval):
        corr = np.correlate(h_eval, 1 - h_eval, mode='full') * dx
        return np.max(corr)
    best_c5 = compute_c5(best_h)

    alpha = 0.01  # Initial learning rate

    while time.time() - start_time < budget_s - 0.1:
        # Compute overlaps and find worst k
        corr = np.correlate(h, 1 - h, mode='full') * dx
        worst_k_index = np.argmax(corr)
        k = worst_k_index - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        if k >= 0:  # Right shift
            for m in range(n_points):
                if m + k < n_points:
                    grad[m] += (1 - h[m + k]) * dx
                if m >= k:
                    grad[m] += -h[m - k] * dx
        else:  # Left shift

                if m + l < n_points:
                    grad[m] += -h[m + l] * dx

        h -= alpha * grad

        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
