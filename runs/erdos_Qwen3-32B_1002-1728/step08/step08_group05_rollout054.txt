import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Initialize PSO parameters
    n_particles = 30
    np.random.seed(seed)

    # Initialize particles around initial solution
    particles = np.tile(h_initial[:half_n].copy(), (n_particles, 1)).astype(float)
    noise = np.random.normal(0, 0.05, particles.shape)
    particles += noise
    particles = np.clip(particles, 0.0, 1.0)
    sums = particles.sum(axis=1, keepdims=True)
    particles = (particles / np.where(sums == 0, 1, sums)) * target_sum_half
    particles = np.clip(particles, 0.0, 1.0)

    velocities = 0.1 * np.random.randn(n_particles, half_n)

    def evaluate(particle):
        h_full = np.zeros(n_points)
        h_full[:half_n] = particle
        h_full[-half_n:] = particle[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    personal_best_positions = particles.copy()
    personal_best_c5s = np.array([evaluate(p) for p in particles])
    global_idx = np.argmin(personal_best_c5s)
    global_pos = personal_best_positions[global_idx].copy()
    global_c5 = personal_best_c5s[global_idx]

    phi_initial = 0.9
    phi_final = 0.4
    mu = 1.5
    nu = 1.5

    while time.time() - start_time < budget_s - 0.1:
        r1 = np.random.rand(n_particles, half_n)
        r2 = np.random.rand(n_particles, half_n)
        global_tile = np.tile(global_pos[None, :], (n_particles, 1))

        velocities = phi_initial * velocities + mu * r1 * (personal_best_positions - particles) + nu * r2 * (global_tile - particles)

        particles += velocities
        particles = np.clip(particles, 0, 1)
        sums = particles.sum(axis=1, keepdims=True)
        particles = (particles / np.where(sums == 0, 1, sums)) * target_sum_half
        particles = np.clip(particles, 0, 1)

        current_c5s = np.array([evaluate(p) for p in particles])

        improved = current_c5s < personal_best_c5s

        personal_best_c5s[improved] = current_c5s[improved]

        global_c5_new = np.min(personal_best_c5s)
        if global_c5_new < global_c5:

            global_pos = personal_best_positions[np.argmin(personal_best_c5s)].copy()

    h_full[:half_n] = global_pos
