import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_random_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = max(np.sum(h), 1e-9)
        h = (h / h_sum) * target_sum
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = max(np.sum(h), 1e-9)
        h = (h / h_sum) * target_sum
        return h

    def crossover(parent1, parent2):
        cross_point = np.random.randint(10, n_points - 10)
        child1 = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
        child2 = np.concatenate([parent2[:cross_point], parent1[cross_point:]])
        return repair(child1), repair(child2)

    def mutate(individual):
        n_mutate = max(1, int(n_points * 0.3))
        indices = np.random.choice(n_points, n_mutate, replace=False)
        sigma = 0.1
        individual[indices] += np.random.normal(0, sigma, n_mutate)
        return repair(individual)

    pop_size = 200
    population = [h_full_initial.copy()]
    while len(population) < pop_size:
        population.append(generate_random_individual())
    population = np.array(population)
    fitness = np.array([compute_c5(ind) for ind in population])
    best_c5 = np.min(fitness)
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()

    max_generations = 250
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break

        sorted_indices = np.argsort(fitness)
        parent_indices = sorted_indices[:int(pop_size * 0.5)]
        parents = population[parent_indices]

        offspring = []
        while len(offspring) < pop_size - int(pop_size * 0.05):
            p1, p2 = parents[np.random.choice(len(parents), 2)]
            c1, c2 = crossover(p1, p2)
            offspring.extend([c1, c2])
        offspring = offspring[:pop_size - int(pop_size * 0.05)]

        for i in range(int(len(offspring) * 0.3)):
            idx = np.random.randint(0, len(offspring))
            offspring[idx] = mutate(offspring[idx])

        offspring = np.array(offspring)
        offspring_fitness = np.array([compute_c5(ind) for ind in offspring])

        elites = population[sorted_indices[:int(pop_size * 0.05)]]
        elites_fitness = fitness[sorted_indices[:int(pop_size * 0.05)]]

        new_population = np.concatenate([elites, offspring])
        new_fitness = np.concatenate([elites_fitness, offspring_fitness])

        if len(new_population) > pop_size:
            indices = np.argsort(new_fitness)
            new_population = new_population[indices][:pop_size]
            new_fitness = new_fitness[indices][:pop_size]

        population = new_population
        fitness = new_fitness

        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()

    h_values = best_individual.tolist()
    return (h_values, best_c5, n_points)
