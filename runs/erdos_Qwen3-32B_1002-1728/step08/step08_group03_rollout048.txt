import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Because full sum is n_points/2, and h is symmetric
    
    np.random.seed(seed)
    num_starts = 20
    
    h_half_initial = h_full_initial[:half_n].copy()
    current_sum = h_half_initial.sum()
    if not np.isclose(current_sum, target_sum_half, atol=1e-9):
        h_half_initial = h_half_initial * target_sum_half / current_sum
    h_half_initial = np.clip(h_half_initial, 0.0, 1.0)
    
    # Generate perturbed initial guesses
    initial_guesses = [h_half_initial.copy()]
    for _ in range(num_starts - 1):
        perturbation = 0.05 * np.random.randn(half_n)
        new_guess = initial_guesses[0] + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        current_sum = max(1e-9, new_guess.sum())
        new_guess = new_guess * target_sum_half / current_sum
        initial_guesses.append(new_guess)
    
    best_c5 = float('inf')
    best_h_half = h_half_initial.copy()
    
    for guess in initial_guesses:
        # Objective function
        def objective(h_half):
            h_full = np.zeros(n_points)
            h_full[:half_n] = np.clip(h_half, 0.0, 1.0)
            h_full[half_n:] = h_full[:half_n][::-1]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(half_n)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 100, 'ftol': 1e-6, 'disp': False})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_half = result.x
        if time.time() - start_time > budget_s - 5:

    h_full = np.zeros(n_points)
    h_full[:half_n] = np.clip(best_h_half, 0.0, 1.0)
