        h = np.clip(B @ c_in, 0, 1)
        total = np.sum(h) * dx
        if total < 1e-9:
            return np.zeros_like(h) + 0.5
        h = h * (1 / total)
        return np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = repair(c)
    best_c5 = compute_c5(best_h)
    best_c = c.copy()

    while time.time() - start_time < budget_s - 10:
        t += 1
        grad = np.zeros_like(c)
        eps_perturb = 1e-3
        for j in range(len(c)):
            # Perturb up
            cpup = c.copy()
            cpup[j] += eps_perturb
            h_perturbed = repair(cpup)
            c5_up = compute_c5(h_perturbed)
            # Perturb down
            cplo = c.copy()
            cplo[j] -= eps_perturb
            h_perturbed_lo = repair(cplo)
            c5_lo = compute_c5(h_perturbed_lo)
            grad[j] = (c5_up - c5_lo) / (2 * eps_perturb)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        c -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Repair and evaluate
        current_h = repair(c)
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            best_c = c.copy()

        # Learning rate decay
        if t % 500 == 0:
            lr *= 0.95

    return (best_h.tolist(), best_c5, n_points)
