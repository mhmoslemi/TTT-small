import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    h = h_initial.copy()
    h = np.clip(h, 0, 1)
    sum_h = h.sum()
    if sum_h != 0:
        h = (h / sum_h) * target_sum
    h = np.clip(h, 0, 1)

    # Initialize best solution
    best_h = h.copy()
    corr = np.correlate(np.clip(h, 0, 1), 1 - np.clip(h, 0, 1), mode='full')
    best_c5 = np.max(corr) * dx

    # Setup seed
    random.seed(seed)
    np.random.seed(seed)

    # Optimization parameters
    alpha = 0.01
    max_eval_time = budget_s - 0.1  # Leave 0.1s for final checks

    while time.time() - start_time < max_eval_time:
        # Compute current correlation
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        max_corr = np.max(corr)
        max_indices = np.where(corr == max_corr)[0]
        m_max = max_indices[0] if len(max_indices) else 0
        l = m_max - (n_points - 1)
        k = abs(l)
        k = min(k, n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            term1 = 0.0
            if j + k < n_points:
                term1 = 1.0 - h_clipped[j + k]
            term2 = 0.0
            if j - k >= 0:
                term2 = -h_clipped[j - k]
            grad[j] = (term1 + term2) * dx

        # Update with repair
        h_prev = h.copy()
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != 0:
            h = (h / sum_h) * target_sum
        h = np.clip(h, 0, 1)

        # Evaluate new C5
        h_clipped_new = np.clip(h, 0, 1)
        new_corr = np.correlate(h_clipped_new, 1 - h_clipped_new, mode='full')
        new_c5 = np.max(new_corr) * dx

        # Accept or reject update
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        elif new_c5 < np.max(np.correlate(np.clip(h_prev, 0, 1), 1 - np.clip(h_prev, 0, 1), mode='full')) * dx:

            best_c5 = new_c5

    sum_h = best_h.sum()
    if abs(sum_h - target_sum) > 1e-8:
        best_h = (best_h / sum_h) * target_sum

    return (best_h.tolist(), best_c5, n_points)
