import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Initialize and normalize h_half
    h_full_initial = np.clip(h_full_initial, 0, 1)
    if h_full_initial.sum() != 0:
        h_full_initial = (h_full_initial / h_full_initial.sum()) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_half = h_full_initial[:half_n]
    sum_h_half = h_half.sum()
    if sum_h_half != 0:
        h_half = (h_half / sum_h_half) * target_sum_half
    h_half = np.clip(h_half, 0, 1)

    def evaluate(hh):
        h_full = np.concatenate([hh, hh[::-1]])
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    best_h = h_half.copy()
    best_c5 = evaluate(h_half)

    np.random.seed(seed)

    max_iter = 5000
    step_size = 0.1
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s - 10:
            break
        hh = h_half.copy()
        h_full = np.concatenate([hh, hh[::-1]])
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        k_max_idx = np.argmax(corr)
        k_shift = k_max_idx - (n_points - 1)

        # Compute gradient of current max term
        grad = np.zeros(half_n)
        for i in range(n_points):
            j_plus = i + k_shift
            val1 = h_full[j_plus] if j_plus < n_points else 0.
            l_minus = i - k_shift
            val2 = h_full[l_minus] if l_minus >= 0 else 0.
            # Contribute to full gradient
            full_grad_i = (1 - val1 - val2) * dx
            # Project to h_half
            if i < half_n:
                grad[i] += full_grad_i
            elif i > n_points - 1 - half_n:
                idx = n_points - 1 - i
                grad[idx] += full_grad_i

        # Line search with backtracking
        alpha = 1.0
        beta = 0.5
        backtracks = 0
        while backtracks < 5:
            new_h = hh - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            sum_new = new_h.sum()
            if sum_new != 0:
                new_h = (new_h / sum_new) * target_sum_half
            new_h = np.clip(new_h, 0, 1)
            new_c5 = evaluate(new_h)
            if new_c5 < best_c5:
                h_half = new_h.copy()
                best_c5 = new_c5
                best_h = h_half.copy()
                break
            alpha *= beta
            backtracks += 1

    h_full = np.concatenate([best_h, best_h[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if h_sum != 0 and abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
