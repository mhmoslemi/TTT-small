    h_full = np.zeros(2 * half_n)
    h_full[:half_n] = h_half
    h_full[half_n:] = h_half[::-1]
    h_full = np.clip(h_full, 0, 1.0)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
