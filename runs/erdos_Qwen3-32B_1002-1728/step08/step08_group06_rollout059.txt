from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    levels = [10, 20, 40, 70]
    best_c5 = np.inf
    best_h = None
    initial_h_values_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None

    for n_points in levels:
        if time.time() - start_time > budget_s - 10:
            break
        
        dx = 2.0 / n_points
        half_n = n_points // 2
        target_sum_half = n_points / 4.0
        
        # Initialize symmetric variables
        if initial_h_values_given is not None and n_points == 70:
            h_init = np.array(initial_h_values_given)
            sym_init = h_init[:half_n]
            sum_sym = np.sum(sym_init)
            if sum_sym != 0:
                sym_init = sym_init * (target_sum_half / sum_sym)
            # Clip to ensure bounds
            sym_init = np.clip(sym_init, 0, 1)
        else:
            # Use uniform initialization
            sym_init = np.full(half_n, target_sum_half / half_n)
        
        # Define bounds and constraints
        bounds = [(0.0, 1.0)] * half_n
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}

        # Objective function
        def objective(sym_vars):
            sym_vars = np.clip(sym_vars, 0, 1)
            h = np.zeros(n_points)
            h[:half_n] = sym_vars
            h[half_n:] = sym_vars[::-1]
            corr = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(corr)

        # Allocate time for this level
        remaining = max(0, budget_s - (time.time() - start_time))
        time_alloc = min(remaining * 0.8 / len(levels), 300)
        options = {'maxiter': max(1, int(time_alloc / 3)), 'ftol': 1e-8}

        # Optimize using SLSQP with gradient
        try:
            result = minimize(
                objective,
                x0=sym_init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options=options
            )
            sym_best = result.x if result.success else sym_init
        except:
            sym_best = sym_init

        # Create full h and finalize variables
        h_best = np.zeros(n_points)
        h_best[:half_n] = sym_best
        h_best[half_n:] = sym_best[::-1]
        h_best = np.clip(h_best, 0, 1)

        # Evaluate C5 for this solution
        corr = np.correlate(h_best, 1 - h_best, mode='full') * dx
        c5_current = np.max(corr)

        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_best
            best_n = n_points

    if best_h is not None:
        assert np.isclose(np.sum(best_h), best_n / 2, atol=1e-6), "Constraint violated"
        return (best_h.tolist(), best_c5, best_n)
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
