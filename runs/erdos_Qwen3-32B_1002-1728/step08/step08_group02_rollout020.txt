def run(seed=42, budget_s=1100, **kwargs):

    num_particles = 40
    particles = np.zeros((num_particles, half_n))

        perturbed = particles[0].copy()
        perturbed += np.random.normal(0, 0.2, size=half_n)

        if sum_perturbed > 1e-10:

        return np.max(corr) * dx

    personal_best_positions = particles.copy()
