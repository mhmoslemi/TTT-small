def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * dx

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h <= 1e-9:
            sum_h = 1e-9
        h_scaled = h_clipped * (target_sum / sum_h)
        h_scaled = np.clip(h_scaled, 0, 1)
        sum_scaled = np.sum(h_scaled)
        if sum_scaled <= 1e-9:
            sum_scaled = 1e-9
        final_scale = target_sum / sum_scaled
        h_final = h_scaled * final_scale
        h_final = np.clip(h_final, 0, 1)
        return h_final

    def perturb_fft_coeffs(coeffs, scale_real, scale_imag):
        new_coeffs = coeffs.copy()
        num_coeffs = len(coeffs)
        max_idx = min(20, num_coeffs)
        indices = np.random.choice(max_idx, int(min(3, max_idx)), replace=False)
        for i in indices:
            delta_real = np.random.normal(0, scale_real)
            delta_imag = np.random.normal(0, scale_imag)
            new_coeffs[i] += complex(delta_real, delta_imag)
        return new_coeffs
