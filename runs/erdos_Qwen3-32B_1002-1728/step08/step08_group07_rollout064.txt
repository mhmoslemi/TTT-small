            current_c5 = compute_c5(particles[i])

                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest = particles[i].copy()

        # Update velocities and positions
        for i in range(N_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = omega * velocities[i] + phi_p * r1 * (pbest_positions[i] - particles[i]) + phi_g * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p < 1e-9:
                particles[i] = np.full(n_points, target_sum / n_points)
            else:
                particles[i] = np.clip(particles[i] * (target_sum / sum_p), 0, 1)

        omega = max(0.4, omega * 0.995)

    # Final validation
    best_sum = best_c5 = None
    for h in pbest_positions:
        c5 = compute_c5(h)
        if best_c5 is None or c5 < best_c5:
            best_c5 = c5
            best_sum = h.copy()
    if best_sum is None:
        best_sum = gbest

    best_sum_sum = np.sum(best_sum)
    if abs(best_sum_sum - target_sum) > 1e-6:
        best_sum = np.clip(best_sum * (target_sum / best_sum_sum), 0, 1)

    return (best_sum.tolist(), best_c5, n_points)
