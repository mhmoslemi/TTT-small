import numpy as np
import time
from scipy.optimize import minimize

def upscale_h_half(prev_h_half):
    half_m_current = len(prev_h_half) * 2
    new_h_half = np.zeros(half_m_current)
    for i in range(len(prev_h_half)):
        new_h_half[2 * i] = prev_h_half[i]
        new_h_half[2 * i + 1] = prev_h_half[i]
    return new_h_half

def optimize_resolution(h_half_initial, m_current, dx_current, max_iter=100):
    half_m = m_current // 2

    def objective_func(h_half_flat):
        h_half = h_half_flat.reshape((half_m,))
        h_full = np.zeros(m_current)
        h_full[:half_m] = h_half
        h_full[half_m:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        one_minus_h = 1 - h_full
        conv = np.correlate(h_full, one_minus_h, mode='full')
        return np.max(conv) * dx_current

    def projected_point(x):
        x = x.reshape((half_m,))
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total < 1e-9:
            x = np.full_like(x, 0.5)
            total = half_m * 0.5
        scale = (m_current / 4.0) / total
        x *= scale
        x = np.clip(x, 0, 1)
        return x

    # Initial guess after projection
    x0 = h_half_initial.reshape((half_m,))
    x0 = projected_point(x0)
    x0 = x0.reshape((half_m,))

    # Run optimization
    bounds = [(0, 1)] * half_m
    res = minimize(
        objective_func,
        x0=x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-6}
    )

    # After optimization, project again
    x_opt = res.x.reshape((half_m,))
    return projected_point(x_opt)
