    def repair_half(h_half):

        if np.isclose(h_half.sum(), 0):
            h_half[0] = 1.0
        h_half = (h_half / h_half.sum()) * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half

    # Initial solution repair
    current_half = h_full_initial[:half_n].copy()
    current_half = repair_half(current_half)
    best_half = current_half.copy()
    best_c5 = np.max(np.correlate(
        np.concatenate((best_half, best_half[::-1])),

    )) * dx

    # Setup seed and SA parameters
    random.seed(seed)
