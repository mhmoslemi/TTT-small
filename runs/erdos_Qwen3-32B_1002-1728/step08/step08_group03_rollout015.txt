from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    half_n = n_points // 2

    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points
    
    def objective_smooth(h_half, beta):
        h_full = np.zeros(n_points)
        h_full[:half_n] = np.clip(h_half, 0, 1)
        h_full[half_n:] = h_full[:half_n][::-1]
        h_full = np.clip(h_full, 0, 1.0)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        t = corr * dx
        exp_t = np.exp(beta * t)
        smooth_max = (1.0 / beta) * np.log(exp_t.sum())
        return smooth_max

    h_half_initial = h_full_initial[:half_n].copy()
    current_sum = h_half_initial.sum()
    if not np.isclose(current_sum, target_sum_half, atol=1e-6):
        h_half_initial = (h_half_initial / current_sum) * target_sum_half
        h_half_initial = np.clip(h_half_initial, 0, 1)

    bounds = [(0, 1)] * half_n
