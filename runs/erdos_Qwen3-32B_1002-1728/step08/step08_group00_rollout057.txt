from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial / h_initial.sum() * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Objective function
    def objective(x):
        h = np.clip(x, 0, 1)
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr_vals)

    # Constraint enforcement
    def project_x(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if abs(s - target_sum) > 1e-8:
            x = x * (target_sum / s)
            x = np.clip(x, 0, 1)
        return x

    # PSO parameters
    np.random.seed(seed)
    n_particles = 30
    w = 0.7
    w_decay = 0.3 / 100.0  # Linear decay per iteration
    c1 = 2.0
    c2 = 2.0
    v_max = 0.5

    # Initialize particles
    positions = np.zeros((n_particles, n_points))
    velocities = np.random.uniform(-v_max, v_max, (n_particles, n_points))
    for i in range(n_particles):
        perturbation = np.random.normal(0, 0.1, n_points)
        positions[i] = project_x(h_initial + perturbation)

    pbest_pos = positions.copy()
    pbest_obj = np.array([objective(p) for p in positions])
    gbest_idx = np.argmin(pbest_obj)
    gbest_pos = positions[gbest_idx].copy()
    gbest_obj = pbest_obj[gbest_idx]

    # PSO main loop
    pso_start = time.time()
    pso_budget = 0.6 * budget_s
    max_pso_iter = 100
    while time.time() - start_time < pso_budget:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_cognitive = c1 * r1 * (pbest_pos[i] - positions[i])
            vel_social = c2 * r2 * (gbest_pos - positions[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            # Update position
            positions[i] += velocities[i]
            positions[i] = project_x(positions[i])

            current_obj = objective(positions[i])

            gbest_obj = pbest_obj[gbest_idx]

        # Decay inertia weight
