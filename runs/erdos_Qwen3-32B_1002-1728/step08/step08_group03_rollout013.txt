from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)

    best_c5 = np.inf

    num_starts = 20

        if time.time() - start_time > budget_s - 5:

        trial_h = h_half.copy()

            h_full = np.concatenate([h_rec, h_rec[::-1]])
            h_full = np.clip(h_full, 0, 1)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
