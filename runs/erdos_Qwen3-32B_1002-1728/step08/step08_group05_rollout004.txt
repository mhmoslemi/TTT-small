import numpy as np
import time

def compute_gradient(h, k_lag, n_points, dx):
    grad = np.zeros(n_points)
    if k_lag >= 0:
        shift = k_lag
        valid_indices = np.arange(n_points - shift)
        i_plus_shift = valid_indices + shift
        grad[valid_indices] += 1 - h[i_plus_shift]
        grad[i_plus_shift] -= h[valid_indices]
    else:
        shift_abs = -k_lag
        valid_indices = np.arange(shift_abs, n_points)
        i_minus_shift = valid_indices - shift_abs
        grad[valid_indices] += 1 - h[i_minus_shift]
        grad[i_minus_shift] -= h[valid_indices]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0

    # Repair initial h to meet constraints
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum != 0:
        h_clipped = (h_clipped / current_sum) * target_sum
    h_clipped = np.clip(h_clipped, 0, 1)
    h = h_clipped.copy()

    best_c5 = np.inf
    best_h = h.copy()

    learning_rate = 0.02
    dx = 2.0 / n_points
    decay_rate = 0.995
    patience_counter = 0
    min_learning_rate = 1e-4
    random_seed = seed

    np.random.seed(random_seed)

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        best_index = np.argmax(corr)

        # Compute corresponding lag (shift)
        k_lag = best_index - (n_points - 1)

        # Compute gradient vector
        grad = compute_gradient(h, k_lag, n_points, dx)

        # Update h using gradient step
        h_new = h - learning_rate * grad

        # Project back into feasible set
        h_clipped = np.clip(h_new, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum != 0:
            h_norm = (target_sum / current_sum) * h_clipped
        else:
            h_norm = h_clipped
        h_norm = np.clip(h_norm, 0, 1)
        h = h_norm.copy()

        # Update best solution if improvement
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            patience_counter = 0
            # Adaptive learning rate increase if improvement
            learning_rate = min(0.1, learning_rate * 1.1)
        else:

            # Decrease learning rate if no improvement for 50 steps
            if patience_counter > 50:

        if learning_rate < min_learning_rate * 2 and patience_counter > 500:
