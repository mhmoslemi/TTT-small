import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    levels = [10, 20, 40, 70]
    best_h_full = None
    best_c5 = float('inf')
    best_n_points = 70
    initial_solution = np.array(initial_h_values)

    for n_points in levels:
        if time.time() - start_time > budget_s - 10:
            break
        n_half = n_points // 2
        dx = 2.0 / n_points
        target_sum_half = n_points / 4.0  # (integral=1 over [0,2] => sum(h) = dx * n_points / 2 => sum(h_half) = (dx * n_points / 2) / 2 = n_points / 4

        if n_points == 10:
            # Initial guess for level 10
            h_half = np.full(n_half, 0.5)
            h_half *= target_sum_half / h_half.sum()
        else:
            # Upample from previous level's h_half
            prev_n_half = len(previous_h_half)
            x_prev = np.arange(prev_n_half) / prev_n_half + 0.5 / prev_n_half
            x_new = np.arange(n_half) / n_half + 0.5 / n_half
            h_half = np.interp(x_new, x_prev, previous_h_half)
            h_half *= target_sum_half / h_half.sum()

        h_half = np.clip(h_half, 0, 1)
        previous_h_half = h_half.copy()

        def create_full(h_half):
            return np.concatenate([h_half, h_half[::-1]])

        def objective(x):
            h_full = create_full(np.clip(x, 0, 1))
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0)] * n_half
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)
        
        # Differential Evolution
        max_de_time = min(120, budget_s * 0.2 - (time.time() - start_time))
        de_result = differential_evolution(
            objective, bounds, strategy='best2bin', 
            mutation=(0.5, 1.0), recombination=0.9,
            popsize=30, tol=1e-6, maxiter=50, 
            seed=seed, workers=1, callback=None, disp=False
        )
        
        if de_result.success:
            h_candidate = np.clip(de_result.x, 0, 1)
            h_candidate *= target_sum_half / h_candidate.sum()
            h_candidate = np.clip(h_candidate, 0, 1)
            candidate_c5 = objective(h_candidate)
        else:
            h_candidate = h_half
            candidate_c5 = objective(h_half)

        # Local optimization using best from DE or initial guess
        for method in ['SLSQP', 'L-BFGS-B']:
            try:
                result = minimize(objective, h_candidate, method=method,
                                 bounds=bounds, constraints=cons,
                                 options={'maxiter': 100, 'ftol': 1e-10})
                if result.success:
                    refined = np.clip(result.x, 0, 1)
                    refined_sum = refined.sum()
                    if abs(refined_sum - target_sum_half) < 1e-6:
                        test_c5 = objective(refined)
                        if test_c5 < candidate_c5:
                            candidate_c5 = test_c5
                            h_candidate = refined
            except:
                continue

        # Final projection and verify integral
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate_sum = h_candidate.sum()
        if abs(h_candidate_sum - target_sum_half) > 1e-6:
            h_candidate *= target_sum_half / h_candidate_sum
            h_candidate = np.clip(h_candidate, 0, 1)
        
        h_full_candidate = create_full(h_candidate)
        current_c5 = np.max(np.correlate(h_full_candidate, 1 - h_full_candidate, 'full')) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_full = h_full_candidate.tolist()
            best_n_points = n_points

        previous_h_half = h_candidate  # Carry forward for next level

    # Final cleanup and projection
    if best_h_full is not None:
        best_h_full = np.clip(best_h_full, 0, 1)
        h_sum = np.sum(best_h_full)
        dx_final = 2.0 / best_n_points
        if abs(h_sum * dx_final - 1.0) > 1e-6:
            best_h_full = np.clip((best_h_full / h_sum) * (1.0 / dx_final), 0, 1)
        return (best_h_full.tolist(), best_c5, best_n_points)
    else:
        return (np.clip(initial_solution, 0, 1).tolist(), np.max(np.correlate(initial_solution, 1 - initial_solution, 'full')) * (2 / len(initial_solution)), len(initial_solution))
