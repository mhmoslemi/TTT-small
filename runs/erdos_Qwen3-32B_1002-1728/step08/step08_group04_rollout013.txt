            # Mutation
            if np.random.rand() < 0.3:  # 30% mutation chance
                mask = np.random.rand(n_points) < 0.3
                noise = np.random.normal(0, 0.1, n_points)
                child += mask * noise
            # Repair
            child = repair(child)
            new_population.append(child)
        
        # Update population
        population = np.array(new_population)
    
    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
