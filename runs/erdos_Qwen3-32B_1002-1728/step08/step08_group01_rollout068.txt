    import time
    from scipy.optimize import differential_evolution, minimize

    start_time = time.time()
    resolutions = [10, 20, 30, 60, 70]
    best_h_half = None

    for i, n_points in enumerate(resolutions):
        half_n = n_points // 2
        dx = 2.0 / n_points
        target_sum_half = n_points / 4.0  # sum(h_half) = total_sum / 2 = (n_points/2)/2 = n_points/4

        # Initial guess for current resolution
        if i == 0:
            # First level, create uniform h_half
            h_half = np.ones(half_n) / half_n * target_sum_half
            h_half = np.clip(h_half, 0, 1)
        else:
            # Upsample from previous best h_half
            prev_h_half = best_h_half
            prev_half_n = len(prev_h_half)
            # Define a function to upsample
            new_h_half = np.zeros(half_n)
            for i_new in range(half_n):
                pos = i_new / half_n * prev_half_n
                idx = int(np.floor(pos))
                new_h_half[i_new] = prev_h_half[idx]
            # Scale to meet sum constraint
            current_sum = new_h_half.sum()
            scaling = target_sum_half / current_sum if current_sum != 0 else 0
            new_h_half *= scaling
            new_h_half = np.clip(new_h_half, 0, 1)
            # Additional scaling if needed after clipping
            current_sum_after_clip = new_h_half.sum()
            if current_sum_after_clip != 0 and abs(current_sum_after_clip - target_sum_half) > 1e-6:
                scaling = target_sum_half / current_sum_after_clip
                new_h_half *= scaling
                new_h_half = np.clip(new_h_half, 0, 1)
            h_half = new_h_half

        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(half_n)]
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

        # Objective function
        def objective(sym_vars):
            h_full = np.concatenate((sym_vars, sym_vars[::-1]))
            h_clipped = np.clip(h_full, 0, 1)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx

        # Repair function to restore feasibility
        def repair_h_half(x):
            x = np.clip(x, 0, 1)
            current_sum = x.sum()

                scale = target_sum_half / current_sum
                x *= scale
                x = np.clip(x, 0, 1)
            return x

        popsize = 30
        maxiter = 50
        de_result = differential_evolution(
            objective, bounds, strategy='best2bin', mutation=(0.5, 1.0),
            recombination=0.9, popsize=popsize, maxiter=maxiter, seed=seed,
            tol=1e-8, polish=False, workers=1

        for method in ['SLSQP', 'L-BFGS-B']:

                    obj_val = objective(cand)
                    if obj_val < objective(de_h_half):

    c5_bound = np.max(corr) * (2.0 / final_n_points)

    return (final_h_full.tolist(), c5_bound, final_n_points)
