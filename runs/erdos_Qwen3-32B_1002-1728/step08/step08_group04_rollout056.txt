    best_c5 = compute_c5(compute_h(best_coeffs, basis_coarse))
    m, v = np.zeros_like(coeffs), np.zeros_like(coeffs)
    t, t_coarse = 0, 0

    while time.time() - start_time < budget_s * 0.55:
        h = compute_h(coeffs, basis_coarse)
        current_c5 = compute_c5(h)
        epsilon = 1e-4
        grad = np.zeros_like(coeffs)
        for i in range(len(coeffs)):
            coeffs_p = np.copy(coeffs)
            coeffs_p[i] += epsilon
            h_p = compute_h(coeffs_p, basis_coarse)
            c5_p = compute_c5(h_p)
            coeffs_n = np.copy(coeffs)
            coeffs_n[i] -= epsilon
            h_n = compute_h(coeffs_n, basis_coarse)
            c5_n = compute_c5(h_n)
            grad[i] = (c5_p - c5_n) / (2 * epsilon)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        coeffs -= lr_coarse * m_hat / (np.sqrt(v_hat) + 1e-8)
        coeffs = np.clip(coeffs, -1, 1)
        curr_c5 = compute_c5(compute_h(coeffs, basis_coarse))
        if curr_c5 < best_c5:
            best_coeffs = coeffs.copy()
            best_c5 = curr_c5
        t_coarse += 1

    # Fine optimization
    coeffs_fine = np.zeros(M_fine)
    coeffs_fine[:len(best_coeffs)] = best_coeffs
    best_fine_coeffs = coeffs_fine.copy()
    m_fine, v_fine = np.zeros_like(coeffs_fine), np.zeros_like(coeffs_fine)
    t_fine = 0

    while time.time() - start_time < budget_s - 10:
        h = compute_h(coeffs_fine, basis_fine)
        current_c5 = compute_c5(h)
        epsilon = 1e-4
        grad = np.zeros_like(coeffs_fine)
        for i in range(len(coeffs_fine)):
            coeffs_p = np.copy(coeffs_fine)
            coeffs_p[i] += epsilon
            h_p = compute_h(coeffs_p, basis_fine)
            c5_p = compute_c5(h_p)
            coeffs_n = np.copy(coeffs_fine)
            coeffs_n[i] -= epsilon
            h_n = compute_h(coeffs_n, basis_fine)
            c5_n = compute_c5(h_n)
            grad[i] = (c5_p - c5_n) / (2 * epsilon)
        t_fine += 1
        m_fine = beta1 * m_fine + (1 - beta1) * grad
        v_fine = beta2 * v_fine + (1 - beta2) * grad**2
        m_hat_fine = m_fine / (1 - beta1**t_fine)
        v_hat_fine = v_fine / (1 - beta2**t_fine)
        coeffs_fine -= lr_fine * m_hat_fine / (np.sqrt(v_hat_fine) + 1e-8)
        coeffs_fine = np.clip(coeffs_fine, -1, 1)
        h_fine = compute_h(coeffs_fine, basis_fine)
        c5_fine = compute_c5(h_fine)
        if c5_fine < best_c5:
            best_fine_coeffs = coeffs_fine.copy()
            best_c5 = c5_fine

    h_final = compute_h(best_fine_coeffs, basis_fine)
    h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), best_c5, n_points)
