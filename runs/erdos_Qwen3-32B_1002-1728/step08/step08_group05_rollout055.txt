import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair the initial solution for first half
    def repair_half(h_half, target_sum):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s != 0:
            h_half = (h_half / s) * target_sum
        return np.clip(h_half, 0, 1)

    initial_first_half = h_full_initial[:half_n].copy()
    initial_first_half = repair_half(initial_first_half, target_sum_half)

    # PSO Parameters
    n_particles = 30
    phi = 0.7
    mu = 1.5
    nu = 1.5
    max_velocity = 0.2

    # Initialize particles (positions & velocities)
    positions = np.zeros((n_particles, half_n))
    velocities = np.zeros((n_particles, half_n))
    for i in range(n_particles):
        pos = initial_first_half + np.random.uniform(-0.05, 0.05, half_n)
        positions[i] = pos
        velocities[i] = 0.1 * np.random.randn(half_n)

    # Personal best and global best
    personal_best_positions = positions.copy()
    personal_best_c5 = np.full(n_particles, np.inf)
    global_best_position = initial_first_half.copy()
    global_best_c5 = float('inf')

    def evaluate(particles):
        c5s = []
        for i in range(particles.shape[0]):
            first = particles[i].copy()
            h_full = np.concatenate([repair_half(first, target_sum_half), repair_half(first[::-1], target_sum_half)[::-1]])
            h_full = np.clip(h_full, 0, 1)
            total_sum = h_full.sum()
            if total_sum > 1e-9:
                h_full = (h_full / total_sum) * target_sum_full
                h_full = np.clip(h_full, 0, 1)
            corr = np.max(np.correlate(h_full, 1 - h_full, mode='full') * dx)
            c5s.append(corr)
            if corr < personal_best_c5[i]:
                personal_best_c5[i] = corr
                personal_best_positions[i] = particles[i].copy()
                if corr < global_best_c5:
                    global_best_c5 = corr
                    global_best_position[:] = particles[i].copy()
        return c5s

    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        # Evaluate particles
        c5s = evaluate(positions)

        # Update velocities and positions
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = phi * velocities[i] + mu * r1 * (personal_best_positions[i] - positions[i]) + nu * r2 * (global_best_position - positions[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            positions[i] += velocities[i]

    # Final repair and validation
    first_best = global_best_position.copy()
    h1 = repair_half(first_best, target_sum_half)
    h2 = repair_half(first_best[::-1], target_sum_half)
    h_full = np.concatenate([h1, h2[::-1]])
    h_full = np.clip(h_full, 0, 1)
    total_sum = h_full.sum()
    if total_sum > 1e-9:
        h_full = (h_full / total_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    
    best_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full') * dx)
    return (h_full.tolist(), best_c5, n_points)
