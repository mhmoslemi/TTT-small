import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum_full / current_sum if current_sum > 1e-9 else 1.0
    h_clipped = np.clip(h_full_initial * scale_factor, 0, 1)
    h_sym_initial = h_clipped[:half_n].copy()
    h_full_sym = np.concatenate((h_sym_initial, h_sym_initial[::-1]))
    current_sum_sym = h_full_sym.sum()
    if np.abs(current_sum_sym - target_sum_full) > 1e-9:
        scale_factor = target_sum_full / current_sum_sym
        h_full_sym = np.clip(h_full_sym * scale_factor, 0, 1)
    h_sym_initial = h_full_sym[:half_n]

    # ADAM parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(half_n)
    v = np.zeros(half_n)
    t = 0

    def compute_objective(h_sym_in, beta_in):
        h_full = np.concatenate((h_sym_in, h_sym_in[::-1]))
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        exp_term = np.exp(beta_in * corr)
        sum_exp = np.sum(exp_term)
        smooth_max = (1.0 / beta_in) * np.log(sum_exp)
        return smooth_max

    def compute_gradient(h_sym_in, beta_in, eps=1e-5):
        grad = np.zeros_like(h_sym_in)
        f_val = compute_objective(h_sym_in, beta_in)
        for i in range(len(h_sym_in)):
            h_perturbed = h_sym_in.copy()
            h_perturbed[i] += eps
            f_perturbed = compute_objective(h_perturbed, beta_in)
            grad[i] = (f_perturbed - f_val) / eps
        return grad

    h_sym = h_sym_initial.copy()
    beta_smooth = 10.0
    max_beta = 100.0
    beta_inc_epochs = 200
    best_c5 = np.inf
    best_h_sym = h_sym.copy()

    while time.time() - start_time < budget_s - 5:
        t += 1
        grad = compute_gradient(h_sym, beta_smooth)
        t_adam = t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        h_sym -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        h_sym = np.clip(h_sym, 0, 1)
        current_sum = h_sym.sum()
        if abs(current_sum - target_sum_half) > 1e-6:
            scale = target_sum_half / current_sum
            h_sym *= scale
            h_sym = np.clip(h_sym, 0, 1)

        if t % beta_inc_epochs == 0 and beta_smooth < max_beta:
            beta_smooth *= 2.0
            beta_smooth = min(beta_smooth, max_beta)

        if t % 10 == 0:
            h_full = np.concatenate((h_sym, h_sym[::-1]))

    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
    final_c5 = np.max(corr)

    return (h_clipped.tolist(), final_c5, n_points)
