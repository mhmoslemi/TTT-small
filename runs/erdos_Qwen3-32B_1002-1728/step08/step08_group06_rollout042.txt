                return compute_c5(create_full_h_and_scale(sym_vars, dx, target_sum), dx)
            
            optimized = minimize(
                objective, 
                x0=best_h[:half_n], 
                method='trust-constr', 
                bounds=[(0, 1)] * half_n, 
                options={'maxiter': 30}
            )
            if optimized.success:
                refined_h = create_full_h_and_scale(optimized.x, dx, target_sum)
                refined_c5 = compute_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h
    
    if best_h is not None:
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6), "Integral constraint violated"
        return (best_h.tolist(), best_c5, len(best_h))
    return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
