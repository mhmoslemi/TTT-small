import time
import random

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = (n_points / 2.0) / 2.0  # total half sum to satisfy full integral of 1

    # Particle Swarm Optimization setup
    num_particles = 30
    cognitive = 1.5
    social = 1.5
    inertia = 0.7
    max_vel = 0.1

    # Initialize particles and velocities around initial solution
    particles = []
    velocities = []
    for _ in range(num_particles):
        p_half = h_full_initial[:half_n].copy()
        p_half += np.random.normal(0, 0.05, half_n)
        # Repair to clamp and scale
        p_clipped = np.clip(p_half, 0, 1)
        p_sum = np.sum(p_clipped)
        if p_sum != 0:
            p_scaled = (p_clipped / p_sum) * target_sum_half
        else:
            p_scaled = np.zeros_like(p_clipped)
        p_scaled = np.clip(p_scaled, 0, 1)
        particles.append(p_scaled)
        velocities.append(np.random.normal(0, 0.1, half_n))

    def repair_p_half(p, target_sum_half):
        p_clipped = np.clip(p, 0, 1)
        p_sum = np.sum(p_clipped)
        if p_sum != 0:
            return np.clip((p_clipped / p_sum) * target_sum_half, 0, 1)
        return np.zeros_like(p_clipped)

    def evaluate_c5(p_half):
        p_clipped = np.clip(p_half, 0, 1)
        p_sum = np.sum(p_clipped)
        if p_sum != 0:
            p_scaled = p_clipped * target_sum_half / p_sum
        else:
            p_scaled = np.zeros_like(p_clipped)
        p_scaled = np.clip(p_scaled, 0, 1)
        full_h = np.concatenate([p_scaled, p_scaled[::-1]])
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        return np.max(corr)

    # Initialize personal bests and global best
    pbest_positions = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest_positions[gbest_idx].copy()
    best_c5 = pbest_scores[gbest_idx]

            current_p = particles[i]

            # Update velocity
