import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 20
    mutation_rate = 0.1
    mutation_std = 0.1
    local_search_steps = 5
    local_lr = 0.02

    population = []
    for _ in range(population_size):
        h_pert = h_initial.copy()
        h_pert += np.random.normal(0, 0.05, h_pert.shape)
        h_pert = project(h_pert, target_sum)
        population.append(h_pert)

    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 10:
        curr_c5_list = [evaluate_candidate(h, dx) for h in population]
        sorted_idx = np.argsort(curr_c5_list)
        parent_ids = sorted_idx[:population_size // 2]
        parents = [population[i] for i in parent_ids]

        new_population = []

        for _ in range(population_size - len(parents)):
            p1, p2 = np.random.choice(parents, size=2)
            child = crossover(p1, p2, target_sum)
            child = mutate(child, mutation_rate, mutation_std, target_sum)
            new_population.append(child)

        top_candidates = [population[i] for i in sorted_idx[:2]]
        for candidate in top_candidates:
            h_refined = candidate.copy()
            for _ in range(local_search_steps):
                corr = np.correlate(h_refined, 1 - h_refined, mode='full') * dx
                lag_idx = np.argmax(corr)
                grad = compute_gradient(h_refined, lag_idx, dx, target_sum)
                h_refined -= local_lr * grad

        new_population.extend(parents)

def evaluate_candidate(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def crossover(p1, p2, target_sum):
    mask = np.random.rand(len(p1)) < 0.5
    child = np.where(mask, p1, p2)
    return project(child, target_sum)

def mutate(child, mutation_rate, mutation_std, target_sum):
    n_mut = max(1, int(mutation_rate * len(child)))
    indices = np.random.choice(len(child), size=n_mut, replace=False)
    child = child.copy()
    child[indices] += np.random.normal(0, mutation_std, size=n_mut)
    return project(np.clip(child, 0, 1), target_sum)

def project(h, target_sum):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s < 1e-9:
        return np.ones_like(h) * target_sum / len(h)
    return h * target_sum / s

def compute_gradient(h, lag_idx, dx, target_sum):
    eps = 1e-5
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] = np.clip(h_plus[i] + eps, 0, 1)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
        vp = corr_plus[lag_idx]

        h_minus = h.copy()
        h_minus[i] = np.clip(h_minus[i] - eps, 0, 1)
        corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx
        vm = corr_minus[lag_idx]

        grad[i] = (vp - vm) / (2 * eps)
    return grad
