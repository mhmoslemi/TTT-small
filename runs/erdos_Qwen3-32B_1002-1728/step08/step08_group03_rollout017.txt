        if time.time() - start_time >= budget_s:
            break
            
        n_current = len(current_h)
        # Determine if we need to upsample or downsample
        if n_current < target_n:
            # Upsample
            x_old = np.linspace(0, 2, n_current)
            x_new = np.linspace(0, 2, target_n)
            # Interpolate the first half and mirror
            n_current_half = n_current // 2
            target_half = target_n // 2
            x_half_old = np.linspace(0, 1, n_current_half)
            x_half_new = np.linspace(0, 1, target_half)
            f = interp1d(x_half_old, current_h[:n_current_half], kind='linear', fill_value='extrapolate')
            h_half_new = f(x_half_new)
            h_half_new = np.clip(h_half_new, 0.0, 1.0)
            # Ensure the half sum corresponds to half of the target_sum
            target_sum_half = target_n / 4.0
            current_sum = h_half_new.sum() * (2 / target_n) * (target_n / 2)
            # Wait, the sum of h for full signal is sum(h_full) = sum(h_half * 2) - middle if odd. But for even target_n, it is 2 * sum(h_half_new)
            full_sum = 2 * np.sum(h_half_new) * (2 / target_n)  # because each point has dx = 2 / target_n
            full_target = target_n / 2.0
            if abs(full_sum - full_target) > 1e-6:
                scaling = full_target / full_sum
                h_half_new *= scaling
                h_half_new = np.clip(h_half_new, 0.0, 1.0)
        elif n_current > target_n:
            # Downsample
            x_old = np.linspace(0, 2, n_current)
            x_new = np.linspace(0, 2, target_n)
            n_current_half = n_current // 2
            target_half = target_n // 2
            x_half_old = np.linspace(0, 1, n_current_half)
            x_half_new = np.linspace(0, 1, target_half)
            f = interp1d(x_half_old, current_h[:n_current_half], kind='linear', fill_value='extrapolate')

            full_target = target_n / 2.0
            if abs(full_sum - full_target) > 1e-6:
                scaling = full_target / full_sum
                h_half_new *= scaling
                h_half_new = np.clip(h_half_new, 0.0, 1.0)
        else:
            # Same resolution
            n_current_half = n_current // 2
            h_half_new = current_h[:n_current_half]
        
        # Build full h from the half
        h_full_new = np.zeros(target_n)
        h_full_new[:target_half] = h_half_new
        h_full_new[-target_half:] = h_half_new[::-1]
        current_h = h_full_new
        
        # Now perform PSO optimization on this resolution
        n_points = target_n
        half_n = n_points // 2
        target_sum_full = n_points / 2.0

        h_half_initial = h_full_new[:half_n]
        target_sum_half = h_full_new.sum() * dx
        target_sum_half_half = target_sum_full / 2.0  # Because sum(h_half) = target_sum_full / 2.0

        particles = np.clip(particles, 0, 1.0)
        for i in range(num_particles):
            total = particles[i].sum()
            if abs(total) < 1e-10:
                particles[i] = 0.5 * np.ones(half_n)
            else:
                particles[i] = (particles[i] / total) * target_sum_half_half
        
        personal_best_positions = particles.copy()
        personal_best_fitness = np.array([evaluate_c5(p, target_sum_half_half, dx) for p in particles])
        global_best_idx = np.argmin(personal_best_fitness)
        global_best_position = personal_best_positions[global_best_idx].copy()
        global_best_c5 = personal_best_fitness[global_best_idx]
        
        velocities = np.zeros_like(particles)
        inertia = 0.7
        cognitive_weight = 1.5
        social_weight = 1.5
        velocity_clamp = 0.1
        
        # Run PSO for remaining time
        while time.time() - start_time < budget_s:
            for i in range(num_particles):
                current_fitness = evaluate_c5(particles[i], target_sum_half_half, dx)
                if current_fitness < personal_best_fitness[i]:
                    personal_best_positions[i] = particles[i].copy()
                    personal_best_fitness[i] = current_fitness
                    if current_fitness < global_best_c5:
                        global_best_position = particles[i].copy()
                        global_best_c5 = current_fitness
            
            # Update velocities and positions
            for i in range(num_particles):
                r1 = np.random.rand(half_n)
                r2 = np.random.rand(half_n)
                
                cognitive = r1 * (personal_best_positions[i] - particles[i])
                social = r2 * (global_best_position - particles[i])
                
                velocities[i] = inertia * velocities[i] + cognitive_weight * cognitive + social_weight * social
                velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
                
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1.0)
                total = particles[i].sum()
                if abs(total) < 1e-10:
                    particles[i] = 0.5 * np.ones(half_n)
                else:
                    particles[i] = (particles[i] / total) * target_sum_half_half
            
            # Break if no further time
            if time.time() - start_time >= budget_s:
                break
        
        # Update current_h to the best found
        h_half = global_best_position
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        
        # Final normalization
        total_sum = h_full.sum()
        if abs(total_sum - target_sum_full) > 1e-6:
            h_full = (h_full / total_sum) * target_sum_full
            h_full = np.clip(h_full, 0, 1)
        
        current_h = h_full

    corr = np.correlate(current_h, 1 - current_h, mode='full')
    c5_bound = np.max(corr) * (2.0 / len(current_h))
    return (current_h.tolist(), c5_bound, len(current_h))
