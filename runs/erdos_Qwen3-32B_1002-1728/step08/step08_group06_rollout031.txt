import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    levels = [10, 20, 40, 70]
    best_c5 = np.inf
    best_h = None
    best_n = 0
    
    initial_h_given = np.array(initial_h_values) if 'initial_h_values' in globals() else None
    
    for n_points in levels:
        half_n = n_points // 2
        dx = 2.0 / n_points
        target_sym_sum = n_points / 4.0  # Each half must sum to this
        
        population_size = 100
        population = []
        
        # Initialize with initial_h if available and grid size is 70
        if initial_h_given is not None and n_points == 70 and half_n * 2 == len(initial_h_given):
            h_sym = initial_h_given[:half_n].copy()
            h_sym = np.clip(h_sym, 0, 1)
            total = h_sym.sum()
            if total != 0:
                h_sym = h_sym * (target_sym_sum / total)
            h_sym = np.clip(h_sym, 0, 1)
            population.append(h_sym)
        
        # Fill remaining population
        while len(population) < population_size:
            h_sym = np.random.rand(half_n)
            h_sym = np.clip(h_sym, 0, 1)
            total = h_sym.sum()
            if total == 0:
                h_sym = np.zeros(half_n)
            else:
                h_sym = h_sym * (target_sym_sum / total)
            h_sym = np.clip(h_sym, 0, 1)
            population.append(h_sym)
        
        population = np.array(population)
        generation = 0
        max_generations_per_grid = max(1, int((budget_s - (time.time() - start_time)) // 10))
        
        while time.time() - start_time < budget_s - 10:
            # Evaluate fitness
            fitness = []
            for h_sym in population:
                h_full = np.concatenate((h_sym, h_sym[::-1]))
                corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
                fitness.append(np.max(corr))
            fitness = np.array(fitness)
            
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                h_full_best = np.concatenate((population[best_idx], population[best_idx][::-1]))
                best_h = h_full_best
                best_n = n_points
            
            new_population = []
            # Generate next generation
            for _ in range(population_size // 2):
                # Tournament selection
                parent1_indices = np.random.choice(population_size, size=5, replace=False)
                parent1_idx = parent1_indices[np.argmin(fitness[parent1_indices])]
                parent1 = population[parent1_idx]
                
                parent2_indices = np.random.choice(population_size, size=5, replace=False)
                parent2_idx = parent2_indices[np.argmin(fitness[parent2_indices])]
                parent2 = population[parent2_idx]
                
                # Crossover
                alpha = np.random.uniform(0.3, 0.7)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation and repair
                for child in [child1, child2]:
                    mut_indices = np.random.choice(half_n, size=int(0.15 * half_n), replace=False)
                    noise = np.random.normal(0, 0.1, len(mut_indices))
                    child[mut_indices] += noise
                    child = np.clip(child, 0, 1)
                    
                    total = child.sum()
                    if total != 0:
                        child = child * (target_sym_sum / total)
                    child = np.clip(child, 0, 1)
                    new_population.append(child)
            
            # Elitism
            top_indices = np.argsort(fitness)[:5]
            new_population[:5] = population[top_indices]
            population = np.array(new_population)
            
            generation += 1
            if generation >= max_generations_per_grid:
                break
        
        if time.time() - start_time > budget_s - 30:
            break
    
    return (best_h.tolist(), best_c5, best_n)
