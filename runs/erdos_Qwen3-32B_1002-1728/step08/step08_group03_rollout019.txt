import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points

    np.random.seed(seed)
    
    # Differential Evolution parameters
    pop_size = 30
    F = 0.8
    CR = 0.9

    # Initialize population with perturbed initial solution
    population = np.zeros((pop_size, half_n))
    population[0] = h_full_initial[:half_n].copy()
    for i in range(1, pop_size):
        population[i] = h_full_initial[:half_n] + 0.05 * np.random.randn(half_n)
    
    # Enforce constraints on initial population
    for i in range(pop_size):
        population[i] = np.clip(population[i], 0, 1.0)
        total = max(1e-9, population[i].sum())
        population[i] = (population[i] / total) * target_sum_half
    
    # Evaluate initial fitness
    fit = np.array([evaluate_c5(ind, target_sum_half, dx) for ind in population])
    best_idx = np.argmin(fit)
    best_h = population[best_idx].copy()
    best_c5 = fit[best_idx]
    
    # Main DE loop
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            # Mutation
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = idxs
            mutant = population[a] + F * (population[b] - population[c])
            
            # Clip and rescale mutant
            mutant = np.clip(mutant, 0, 1.0)
            m_total = max(1e-9, mutant.sum())
            mutant = (mutant / m_total) * target_sum_half
            
            # Crossover and feasibility
            trial = population[i].copy()
            crossover = np.random.rand(half_n) <= CR
            if not np.any(crossover):
                crossover[np.random.randint(half_n)] = True
            trial[crossover] = mutant[crossover]
            t_total = max(1e-9, trial.sum())
            trial = (trial / t_total) * target_sum_half
            
            # Selection
            trial_c5 = evaluate_c5(trial, target_sum_half, dx)
            if trial_c5 < fit[i]:
                population[i] = trial.copy()
                fit[i] = trial_c5
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5
    
    # Construct final solution
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h
    h_full[half_n:] = best_h[::-1]
    h_full = np.clip(h_full, 0, 1.0)
    
    # Final constraint check
    total_sum = h_full.sum()
    if abs(total_sum - target_sum_full) > 1e-6:
        h_full = (h_full / total_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1.0)
    
    return (h_full.tolist(), best_c5, n_points)

def evaluate_c5(h_half, target_sum_half, dx):
    half_n = len(h_half)
    h_full = np.zeros(2 * half_n)
    h_full[:half_n] = h_half
    h_full[half_n:] = h_half[::-1]
    h_full = np.clip(h_full, 0, 1.0)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
