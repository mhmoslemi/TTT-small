def run(seed=42, budget_s=1100, **kwargs):

    t = x - 1.0

    B = np.column_stack([eval_chebyt(2*k, t) for k in range(1, max_order+1)])
    n_bases = B.shape[1]

    h_initial_no_dc = h_initial - 0.5
    c_coeff, _, _, _ = np.linalg.lstsq(B, h_initial_no_dc, rcond=None)
    c_coeff = c_coeff.squeeze()

    def repair_h(c_coeff):
        h = 0.5 + B @ c_coeff
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = h_clipped.sum()
        target_sum = n_points / 2.0
        scaling = target_sum / sum_clipped if sum_clipped > 1e-9 else 1.0
        h_scaled = np.clip(h_clipped * scaling, 0, 1)
        return h_scaled

    def compute_c5(c_coeff):

        h_clipped = np.clip(h, 0, 1)
        sum_clipped = h_clipped.sum()
        target_sum = n_points / 2.0

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')

    beta2 = 0.999

    m = np.zeros_like(c_coeff)
    v = np.zeros_like(c_coeff)

    best_c5 = compute_c5(c_coeff)
