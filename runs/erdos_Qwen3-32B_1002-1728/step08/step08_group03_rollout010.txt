import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, copy=True)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')

    # Evaluate initial C5
    corr = np.correlate(h, 1 - h, mode='full') * dx
    current_max_c5 = np.max(corr)
    if current_max_c5 < best_c5:
        best_c5 = current_max_c5
        best_h = h.copy()

    np.random.seed(seed)

    while time.time() - start_time < budget_s:
        # Compute current max overlap and worst k
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max_c5 = np.max(corr)
        index_max = int(np.argmax(corr))
        k_worst = index_max - (n_points - 1)

        # Compute gradient for worst k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_worst
            term1 = 0.0
            if 0 <= j < n_points:
                term1 = 1.0 - h[j]
            m = i - k_worst
            term2 = 0.0
            if 0 <= m < n_points:
                term2 = -h[m]
            grad[i] = (term1 + term2) * dx

        # Step size with decay
