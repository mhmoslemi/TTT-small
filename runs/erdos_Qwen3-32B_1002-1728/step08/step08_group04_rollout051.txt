import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_full_initial = h_full_initial / h_full_initial.sum() * target_sum

    x = np.linspace(0, 2, n_points, endpoint=False)
    max_modes = n_points // 2

    k = np.arange(1, max_modes + 1)
    cos_term = np.cos(np.pi * k[:, np.newaxis] * x[np.newaxis, :])
    sin_term = np.sin(np.pi * k[:, np.newaxis] * x[np.newaxis, :])

    a_initial = np.zeros(max_modes)
    b_initial = np.zeros(max_modes)
    for i in range(max_modes):
        a_initial[i] = np.trapz(h_full_initial * cos_term[i], x) * 2 / 2
        b_initial[i] = np.trapz(h_full_initial * sin_term[i], x) * 2 / 2

    best_a = np.copy(a_initial)
    best_b = np.copy(b_initial)
    best_c5 = np.inf
    current_a = np.copy(a_initial)
    current_b = np.copy(b_initial)

        h = 0.5 + np.sum(a[:, np.newaxis] * cos_term, axis=0) + np.sum(b[:, np.newaxis] * sin_term, axis=0)
        h_clipped = np.clip(h, 0, 1)
        sum_h = h_clipped.sum()
        if sum_h < 1e-9:
            return np.inf
        scaling = target_sum / sum_h
        h_scaled = h_clipped * scaling

        corr = np.correlate(h_final, 1 - h_final, mode='full') * dx
        return np.max(corr)

    current_c5 = compute_c5(current_a, current_b)
    best_c5 = current_c5

    scale = 0.1
    alpha = 0.995
    temp = 1.0
    temp_decay = 0.995
