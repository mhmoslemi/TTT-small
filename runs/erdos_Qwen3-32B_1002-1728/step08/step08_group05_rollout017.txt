import random

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points

    n_particles = 30
    positions = np.zeros((n_particles, half_n))
    for i in range(n_particles):
        pos = h_initial[:half_n].copy()
        noise = np.random.normal(0, 0.05, half_n)
        pos += noise
        pos = np.clip(pos, 0, 1)
        s = pos.sum()
        if s != 0:
            pos = (pos / s) * target_sum_half
        positions[i] = np.clip(pos, 0, 1)

    velocities = np.random.uniform(-0.1, 0.1, (n_particles, half_n))
    pbest_positions = positions.copy()
    pbest_scores = np.zeros(n_particles)
    for i in range(n_particles):
        h_full = np.zeros(n_points)
        h_full[:half_n] = pbest_positions[i]
        h_full[-half_n:] = pbest_positions[i][::-1]
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        pbest_scores[i] = np.max(corr) * dx
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    phi = 0.7
    mu = 1.5
    nu = 1.5
    start_time = time.time()
    no_improvement = 0

    while time.time() - start_time < budget_s - 0.1:
        for i in range(n_particles):

            velocities[i] = phi * velocities[i] + mu * r1 * (pbest_positions[i] - positions[i]) + nu * r2 * (gbest_position - positions[i])

            current_score = np.max(corr) * dx
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = positions[i].copy()

        new_gbest_idx = np.argmin(pbest_scores)

        if new_gbest_score < gbest_score:

            gbest_position = pbest_positions[new_gbest_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement > 30 and time.time() - start_time > 10:

    h_best_full = np.zeros(n_points)
    h_best_full[:half_n] = gbest_position
