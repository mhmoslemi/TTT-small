                best_c5 = new_c5
                best_fft = new_fft.copy()
                best_h = new_h_repaired.copy()

        # Cool temperature
        temp *= alpha

        # Periodic re-evaluation
        if iteration % 200 == 0:
            test_h = repair_h(np.fft.irfft(best_fft))
            test_c5 = compute_c5(test_h)
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_h = test_h.copy()

    return (best_h.tolist(), best_c5, n_points)
