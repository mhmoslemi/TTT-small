    h_half_initial = h_full_initial[:half_n].copy()

    # Initialize simplex vertices
    np.random.seed(seed)
    num_vertices = 20
    simplex_vertices = []
    simplex_vertices.append(h_half_initial.copy())
    for i in range(1, num_vertices):
        new_point = h_half_initial.copy() + np.random.normal(0, 0.1, half_n)
        new_point = np.clip(new_point, 0, 1)
        new_sum = new_point.sum()
        if new_sum != 0:
            new_point = (new_point / new_sum) * target_sum_half
        else:
            new_point.fill(target_sum_half / half_n)
        new_point = np.clip(new_point, 0, 1)
        simplex_vertices.append(new_point)

    # Evaluate initial simplex to find best
    best_c5 = float('inf')
    best_h = simplex_vertices[0].copy()
    for v in simplex_vertices:
        h_full = np.concatenate([v, v[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = v.copy()

    # Tabu list setup
    tabu_list = []
    n_tabu = 50

    # Main simplex optimization loop
    while time.time() - start_time < budget_s - 10:
        # Evaluate all simplex vertices
        c5_values = []
        for v in simplex_vertices:
            h_full = np.concatenate([v, v[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx

        centroid = np.mean(non_worst, axis=0)

        reflected = centroid + alpha * (centroid - worst_vertex)

        # Repair reflected point
        reflected = np.clip(reflected, 0, 1)

                is_tabu = True
                break

        if is_tabu:
            reflected = worst_vertex + np.random.normal(0, 0.1, half_n)
            reflected = np.clip(reflected, 0, 1)
            refl_sum = reflected.sum()
            if refl_sum != 0:
                reflected = (reflected / refl_sum) * target_sum_half

                reflected.fill(target_sum_half / half_n)
            reflected = np.clip(reflected, 0, 1)

        h_full = np.concatenate([reflected, reflected[::-1]])
        corr_new = np.correlate(h_full, 1 - h_full, mode='full')
        new_c5 = np.max(corr_new) * dx

        # Replace worst if better
        if new_c5 < worst_c5:
            simplex_vertices[worst_idx] = reflected.copy()
            # Add to tabu list
            tabu_list.append(reflected.copy())
            if len(tabu_list) > n_tabu:
                tabu_list.pop(0)
            # Update best solution
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = reflected.copy()

    # Final repair and reconstruct
    h_full = np.concatenate([best_h, best_h[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if h_sum != 0:
        h_full = (h_full / h_sum) * target_sum_full
    else:
        h_full.fill(target_sum_full / n_points)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
