import numpy as np
import time
import scipy
import scipy.special


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    x_vals = np.linspace(0, 2.0 - dx, n_points)
    x_prime = x_vals - 1.0

    # Use M=10 for the expansion
    M = 10
    # Project initial h to Chebyshev coefficients
    basis = np.zeros((n_points, M))
    for j in range(M):
        degree = 2 * j
        for i in range(n_points):
            basis[i, j] = scipy.special.eval_chebyt(degree, x_prime[i])
    h_base = h_initial - 0.5
    c_initial, _, _, _ = np.linalg.lstsq(basis, h_base, rcond=None)
    current_c = c_initial.copy()

    # Adam optimizer parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(current_c)
    v = np.zeros_like(current_c)
    step = 0

    best_c5 = float('inf')
    best_coeffs = current_c.copy()

    # Optimization loop
    while time.time() - start_time < budget_s - 20:
        # Compute current h and C5
        h_vals = 0.5 + np.dot(basis, current_c)
        h_vals = np.clip(h_vals, 0, 1)
        total = np.sum(h_vals) * dx
        if total < 1e-9:
            h_vals[:] = 0.5  # fallback
        else:
            h_vals /= total
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_coeffs = current_c.copy()

        # Finite difference gradient estimation
        grad = np.zeros_like(current_c)
        delta = 1e-5
        obj_orig = current_c5

        for j in range(len(current_c)):
            c_perturbed = current_c.copy()
            c_perturbed[j] += delta
            h_perts = 0.5 + np.dot(basis, c_perturbed)
            h_perts = np.clip(h_perts, 0, 1)
            total_perts = np.sum(h_perts) * dx
            if total_perts < 1e-9:
                obj_pert = float('inf')
            else:
                h_perts /= total_perts
            corr_perts = np.correlate(h_perts, 1 - h_perts, mode='full') * dx
            obj_pert = np.max(corr_perts)
            grad[j] = (obj_pert - obj_orig) / delta

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        current_c -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        # Limit step size to prevent divergence
        current_c = np.clip(current_c, -10, 10)  # arbitrary clipping to prevent extreme values

    # Compute final h_vals
    h_vals = 0.5 + np.dot(basis, best_coeffs)
    h_vals = np.clip(h_vals, 0, 1)
    total = np.sum(h_vals) * dx
    if total < 1e-9:
        h_vals[:] = 0.5
    else:
        h_vals /= total

    # Final check for C5
    corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
