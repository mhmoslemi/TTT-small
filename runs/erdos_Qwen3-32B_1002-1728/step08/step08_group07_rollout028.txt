def run(seed=42, budget_s=1100, **kwargs):

        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    def process_h_half(h_half, target_sum):
        h = np.clip(h_half, 0, 1).copy()
        current_sum = h.sum()
        if current_sum < 1e-9:
            h[:] = target_sum / h.size
        else:
            h *= (target_sum / current_sum)
        if np.isnan(h).any():
            h[:] = target_sum / h.size
        return np.clip(h, 0, 1)

    n_particles = 20
    particles = []
    h_first_initial = h_full_initial[:half_n].copy()
    h_first_initial = process_h_half(h_first_initial, target_sum_half)
    particles.append(h_first_initial)
    for _ in range(1, n_particles):
        perturbation = np.random.uniform(-0.05, 0.05, half_n)
        new_h = particles[0].copy() + perturbation
        new_h = process_h_half(new_h, target_sum_half)
        particles.append(new_h)
    particles = np.array(particles)

    velocities = np.zeros((n_particles, half_n))
    personal_best_positions = particles.copy()
    personal_best_scores = []

    def compute_C5(h_half):
        full_h = create_full_h(h_half)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        return np.max(corr)

    for i in range(n_particles):
        personal_best_scores.append(compute_C5(particles[i]))

    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = particles[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
