import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum_full / current_sum if current_sum > 1e-9 else 0
    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)

    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def objective(sym_vars):
        h_full = create_full_h(np.clip(sym_vars, 0, 1))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    repair_cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    np.random.seed(seed + 1)
    
    # Enhanced DE with adaptive parameters
    de_budget = budget_s * 0.45
    de_popsize = 40
    de_maxiter = max(50, int((de_budget * 0.8) / (n_points * de_popsize * 0.01)))
    de_maxiter = min(de_maxiter, 1000)
    
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best2bin', maxiter=de_maxiter,

            seed=seed, workers=1, disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
            best_h_half *= target_sum_half / best_h_half.sum()

            best_c5 = objective(best_h_half)
            final_h_half = best_h_half
    except:

    # Structured local search with dynamic steps
    while time.time() - start_time < budget_s - 5:
        time_frac = (time.time() - start_time) / budget_s
        perturb_size = max(3, int(half_n * (0.15 - 0.05 * time_frac)))
        perturb_indices = np.random.choice(half_n, perturb_size, replace=False)
        
        # Adaptive noise scale
        noise_scale = 0.25 * (1 - time_frac) ** 1.2
        perturbed = final_h_half.copy()
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, size=perturb_size)
        perturbed = np.clip(perturbed, 0, 1)
        
        # Feasibility projection
        pert_sum = perturbed.sum()
        if abs(pert_sum - target_sum_half) > 1e-6 and pert_sum > 1e-9:
            perturbed *= target_sum_half / pert_sum
            perturbed = np.clip(perturbed, 0, 1)
        
        for method in ['SLSQP', 'L-BFGS-B', 'COBYLA']:
            try:
                result = minimize(
                    objective, x0=perturbed, method=method,
                    bounds=bounds, constraints=repair_cons,

                )
                if result.success:
                    cand = np.clip(result.x, 0, 1)
                    cand_sum = cand.sum()
                    if abs(cand_sum - target_sum_half) < 1e-6 or cand_sum > 1e-9 and time_frac > 0.8:
                        cand = (cand / cand_sum) * target_sum_half
                        cand = np.clip(cand, 0, 1)
                    new_c5 = objective(cand)
                    if new_c5 < best_c5:
                        best_c5 = new_c5

            except:
