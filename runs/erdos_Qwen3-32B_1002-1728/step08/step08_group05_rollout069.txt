            expr += term
        if expr > 0:
            expr *= dx
            constraints.append(expr <= C)

    # Objective and solve
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)

    # Warm start with initial solution
    x.value = x_init
    problem.solve(solver=cp.MOSEK, verbose=False)

    # Construct final h
    x_opt = x.value if x.value is not None else x_init
    h_opt = np.zeros(n_points)
    h_opt[:n_half] = x_opt
    h_opt[n_half:] = x_opt[::-1]
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = h_opt.sum()
    if abs(current_sum - target_sum_full) > 1e-8:
        h_opt = (h_opt / current_sum) * target_sum_full

    # Evaluate final C5
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(corr) * dx
    if time.time() - start_time >= budget_s:
        return (h_opt.tolist(), c5, n_points)
    return (h_opt.tolist(), c5, n_points)
