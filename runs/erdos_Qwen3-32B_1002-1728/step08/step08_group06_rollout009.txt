            grad_a = grad_subset_grad_full(grad_full, m)
            return grad_a
        
        bounds_a = Bounds(0.0, 1.0)
        cons_a = {'type': 'eq', 'fun': lambda a: np.sum(a) - target_sum_a}
        
        result = minimize(
            fun=objective,
            jac=compute_gradient_a,
            x0=a_initial,
            method='SLSQP',
            bounds=bounds_a,
            constraints=cons_a,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            a_opt = result.x
            full_h_opt = create_full_h(a_opt, m)
            full_h_opt = np.clip(full_h_opt, 0, 1)
            
            current_c5, _ = compute_c5_and_k_max(full_h_opt, dx, m)
            
            current_sum = full_h_opt.sum()
            if not np.isclose(current_sum, n_points/2, atol=1e-6):
                scale_factor = (n_points / 2) / current_sum
                full_h_opt = np.clip(full_h_opt * scale_factor, 0, 1)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = full_h_opt.copy()
        
        if time.time() - start_time > budget_s - 5:
            break
    
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), len(best_h) / 2, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), best_c5, len(initial_h_values_given))
