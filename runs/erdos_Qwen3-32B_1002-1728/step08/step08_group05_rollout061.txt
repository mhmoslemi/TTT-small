import time
import random

def run(seed=42, budget_s=1100, **kwargs):

    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum > 1e-10:
            h = (h / h_sum) * target_sum_half
        return np.clip(h, 0, 1)

    current_half = h_full_initial[:half_n].copy()
    current_half = repair(current_half)

    # Initialize population
    np.random.seed(seed)
    population_size = 30
    population = [current_half + 0.1 * np.random.randn(half_n) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = repair(population[i])

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    best_half = None
    best_c5 = float('inf')
    for h_half in population:
        c5 = compute_c5(h_half)
        if c5 < best_c5:
            best_c5 = c5
            best_half = h_half.copy()

    while time.time() - start_time < budget_s - 0.1:
        time_remaining = budget_s - (time.time() - start_time)

        # Select parents using tournament selection
        parents = []
        for _ in range(population_size // 2):
            indices = np.random.choice(population_size, 3)
            if compute_c5(population[indices[0]]) < compute_c5(population[indices[1]]):
                if compute_c5(population[indices[0]]) < compute_c5(population[indices[2]]):
                    best = population[indices[0]]
                else:
                    best = population[indices[2]]

                best = population[indices[1]]
            else:
                best = population[indices[2]]
            parents.append(best)

        offspring = []
        for i in range(population_size // 2):

                child = alpha * parent1 + (1 - alpha) * parent2

                child = parents[random.randint(0, len(parents) - 1)].copy()

                    size = min(random.randint(2, 5), half_n)
                    start = random.randint(0, half_n - size)
                    child[start:start + size] += np.random.normal(0, 0.1, size)
                else:  # Global
                    child += np.random.normal(0, 0.05, half_n)
                offspring.append(child)

        # Repair offspring
        for i in range(len(offspring)):
            offspring[i] = repair(offspring[i])

        # Combine population and offspring
        combined = population + offspring
        combined = np.array([repair(h) for h in combined])
        scores = np.array([compute_c5(h) for h in combined])
        
        # Sort and select best
        indices = np.argsort(scores)
        population = [combined[i] for i in indices[:population_size]]
        
        # Update best solution
        for h_half in population:

    return (h_final.tolist(), best_c5, n_points)
