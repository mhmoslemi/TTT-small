    # Final solution validation
    final_h = np.clip(result.x if result.success else h_initial, 0, 1)
    final_h = final_h / final_h.sum() * target_sum
    final_h = np.clip(final_h, 0, 1)
    
    best_c5 = objective(final_h)
    
    return (final_h.tolist(), best_c5, n_points)
