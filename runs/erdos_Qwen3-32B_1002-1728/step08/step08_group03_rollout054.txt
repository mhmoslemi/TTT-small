import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points
    
    h_half_initial = h_initial[:half_n]
    
    np.random.seed(seed)
    population_size = 20
    F = 0.8
    CR = 0.9
    
    population = np.zeros((population_size, half_n))
    population[0] = h_half_initial.copy()
    for i in range(1, population_size):

        total = max(1e-9, np.sum(population[i]))
        population[i] = (population[i] / total) * target_sum_half
    
    def evaluate(h_half):
        h_full = np.zeros(2 * half_n)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]

    while time.time() - start_time < budget_s:

    return (h_full.tolist(), best_c5, n_points)
