import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points
    h_half_initial = h_full_initial[:half_n].copy()
    
    np.random.seed(seed)
    num_pop = 30
    population = np.zeros((num_pop, half_n))
    population[0] = h_half_initial.copy()
    
    for i in range(1, num_pop):
        perturbed = h_half_initial + 0.05 * np.random.randn(half_n)
        perturbed = np.clip(perturbed, 0, 1.0)
        total = max(1e-9, perturbed.sum())
        perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1.0)
        population[i] = perturbed
    
    def evaluate_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1.0)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_c5 = np.inf
    best_h_half = None
    for i in range(num_pop):
        c5 = evaluate_c5(population[i])
        if c5 < best_c5:
            best_c5 = c5
            best_h_half = population[i].copy()
    
    F = 0.8
    CR = 0.9
    while time.time() - start_time < budget_s:
        for i in range(num_pop):
            while True:
                a, b, c = np.random.randint(0, num_pop, 3)
                if a != i and b != i and c != i and a != b and a != c and b != c:
                    break
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1.0)
            m_sum = max(1e-9, mutant.sum())
            mutant = (mutant / m_sum) * target_sum_half
            mutant = np.clip(mutant, 0, 1.0)
            trial = population[i].copy()
            cross_mask = np.random.rand(half_n) < CR
            if not cross_mask.any():
                cross_mask[0] = True
            trial[cross_mask] = mutant[cross_mask]
            t_sum = max(1e-9, trial.sum())
            trial = (trial / t_sum) * target_sum_half
            trial = np.clip(trial, 0, 1.0)
            trial_c5 = evaluate_c5(trial)
            current_c5 = evaluate_c5(population[i])
            if trial_c5 < current_c5:
                population[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h_half = trial.copy()
    
    h_best = np.zeros(n_points)
    h_best[:half_n] = best_h_half
    h_best[half_n:] = best_h_half[::-1]
    h_best = np.clip(h_best, 0, 1.0)
    total = h_best.sum()
    if abs(total - target_sum_full) > 1e-6:
        h_best = (h_best / total) * target_sum_full
        h_best = np.clip(h_best, 0, 1.0)
    return (h_best.tolist(), best_c5, n_points)
