def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Repair initial h_half to satisfy constraints
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_half_initial = h_full_initial[:half_n].copy()
    current_sum = h_half_initial.sum()
    if current_sum != 0:
        h_half_initial = (h_half_initial / current_sum) * target_sum_half
    else:
        h_half_initial.fill(target_sum_half / half_n)
    h_half_initial = np.clip(h_half_initial, 0, 1)

    # Initialize simplex with perturbed symmetric samples
    np.random.seed(seed)
    n_vertices = 20
    simplex = np.zeros((n_vertices, half_n))
    simplex[0] = h_half_initial.copy()

    for i in range(1, n_vertices):
        perturbed = h_half_initial.copy()
        perturbation = np.random.normal(0, 0.1, size=half_n)
        perturbed += perturbation
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = perturbed.sum()
        if sum_p != 0:
            perturbed = (perturbed / sum_p) * target_sum_half
        else:
            perturbed.fill(target_sum_half / half_n)
        perturbed = np.clip(perturbed, 0, 1)
        simplex[i] = perturbed

    # Nelder-Mead parameters
    alpha = 1.5  # reflection
    gamma = 2.0  # expansion
    rho = 0.5    # contraction
    sigma = 0.9  # shrink

    def evaluate(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    while time.time() - start_time < budget_s - 5:
        # Evaluate all points in simplex
        values = np.array([evaluate(simplex[i]) for i in range(n_vertices)])
        indices = np.argsort(values)
        simplex = simplex[indices]
        values = values[indices]

        if np.allclose(values[0], values[-1]):
            break

        # Compute centroid of best n_vertices-1 points
        centroid = np.mean(simplex[:-1], axis=0)

        # Reflection
        reflected = centroid + alpha * (centroid - simplex[-1])
        reflected = np.clip(reflected, 0, 1)
        sum_r = reflected.sum()
        if sum_r != 0:
            reflected = (reflected / sum_r) * target_sum_half
        else:
            reflected.fill(target_sum_half / half_n)
        reflected = np.clip(reflected, 0, 1)
        f_r = evaluate(reflected)

        if f_r < values[0]:
            # Expansion
            expanded = centroid + gamma * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            sum_e = expanded.sum()
            if sum_e != 0:
                expanded = (expanded / sum_e) * target_sum_half
            else:
                expanded.fill(target_sum_half / half_n)
            expanded = np.clip(expanded, 0, 1)
            f_e = evaluate(expanded)
            if f_e < f_r:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        elif f_r < values[-2]:
            simplex[-1] = reflected
        else:
            # Contraction
            if f_r < values[-1]:
                contraction = centroid + rho * (reflected - centroid)
            else:
                contraction = centroid + rho * (simplex[-1] - centroid)
            contraction = np.clip(contraction, 0, 1)
            sum_c = contraction.sum()
            if sum_c != 0:
                contraction = (contraction / sum_c) * target_sum_half
            else:
                contraction.fill(target_sum_half / half_n)
            contraction = np.clip(contraction, 0, 1)
            f_c = evaluate(contraction)
            if f_c < values[-1]:
                simplex[-1] = contraction
            else:
                # Shrink
                for k in range(1, n_vertices):
                    simplex[k] = simplex[0] + sigma * (simplex[k] - simplex[0])
                    simplex[k] = np.clip(simplex[k], 0, 1)
                    sum_k = simplex[k].sum()
                    if sum_k != 0:
                        simplex[k] = (simplex[k] / sum_k) * target_sum_half
                    else:
                        simplex[k].fill(target_sum_half / half_n)

        best_h_full = np.clip(best_h_full, 0, 1)
    final_value = evaluate(best_h_half)
    return (best_h_full.tolist(), final_value, n_points)
