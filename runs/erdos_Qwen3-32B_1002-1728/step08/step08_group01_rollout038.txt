    target_sum_half = target_sum_full / 2.0

    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum > 1e-9:
        scale_factor = target_sum_full / current_sum
        h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)
    
    # Symmetry, get initial h_sym
    h_sym_initial = h_full_initial[:half_n].copy()
    
    # Optimizer setup
    h_sym = h_sym_initial.copy()
    h_sym = np.clip(h_sym, 0, 1)
    # Project to target_sum_half
    current_sum = h_sym.sum()
    if current_sum > 1e-9:
        h_sym = h_sym * (target_sum_half / current_sum)
    h_sym = np.clip(h_sym, 0, 1)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    learning_rate = 0.01
    beta = 10.0
    decay_steps = 1000
    beta_inc_rate = 1.001  # Increase beta slowly over time

        iteration += 1

        h_full = np.concatenate((h_sym, h_sym[::-1]))
        h_full = np.clip(h_full, 0, 1)
        # Compute overlaps
        corr_full = np.correlate(h_full, 1 - h_full, mode='full')
        o = corr_full * dx
        # Compute softmax probabilities
        p = np.exp(beta * o)
        p_sum = np.sum(p)
        p = p / p_sum

        grad_h_full = np.zeros_like(h_full)

        for s in range(len(corr_full)):
