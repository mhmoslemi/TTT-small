        h_full_initial = (h_full_initial / current_sum) * target_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum) > 1e-6:
        scale_factor = target_sum / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    final_h = h_full_initial.copy()
    best_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx

    # SDP phase
    try:
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1
        ]

        # Sample a subset of shifts to check in the SDP relaxation
        selected_k = np.random.choice(n_points - 1, min(15, n_points // 2), replace=False)
        lambda_var = cp.Variable()
        for k in selected_k:
            if k == 0:
                term = cp.sum(h_vars ** 2)
                overlap = term * dx
            else:
                mask = np.zeros((n_points, n_points))
                for i in range(n_points - k):
                    mask[i, i + k] = 1
                overlap = cp.sum(cp.multiply(h_vars, cp.matmul(mask, h_vars))) * dx
            constraints.append(1 - overlap >= lambda_var)

        objective = cp.Minimize(1.0 - lambda_var)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, abstol=1e-7, reltol=1e-7, maxiter=1000, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sdp_h = h_vars.value.copy()
            sdp_h = np.clip(sdp_h, 0, 1)
            current_sum = sdp_h.sum()
            if abs(current_sum - target_sum) > 1e-6:
                sdp_h = (sdp_h / current_sum) * target_sum
                sdp_h = np.clip(sdp_h, 0, 1)
    except:
        sdp_h = h_full_initial.copy()

    # Local optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
    candidate = final_h.copy()

    # Refine with SLSQP
    time_limit = budget_s - 10
    while time.time() - start_time < time_limit:
        if np.random.rand() < 0.5:
            h0 = candidate
        else:
            h0 = sdp_h
        idx = np.argsort(np.random.rand(n_points))[:min(n_points, 15)]
        h0[idx] += np.random.randn(len(idx)) * 0.2
        h0 = np.clip(h0, 0, 1)
        current_sum = h0.sum()
        if abs(current_sum - target_sum) > 1e-6:
            h0 = (h0 / current_sum) * target_sum
        h0 = np.clip(h0, 0, 1)

        try:
            result = minimize(
                lambda x: np.max(np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')) * dx,
                x0=h0,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_sum = new_h.sum()
                if abs(new_sum - target_sum) < 1e-6:
                    new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        candidate = new_h
        except:
            pass

    # Final repair
    candidate = np.clip(candidate, 0, 1)
    current_sum = candidate.sum()
    if abs(current_sum - target_sum) > 1e-6:
        candidate = (candidate / current_sum) * target_sum
        candidate = np.clip(candidate, 0, 1)

    return (candidate.tolist(), best_c5, n_points)
