                            cand = np.clip(cand, 0, 1)
                            c5_cand = objective(cand)
                            if c5_cand < local_best_c5:
                                local_best_c5 = c5_cand
                                if np.isfinite(local_best_c5):
                                    current_h_sym = cand.copy()
                        break  # Break on first success to save time
                except:
                    pass

    # Final projection
    n_points = resolutions[-1]  # Last resolution
    half_n = n_points // 2
    h_full = create_full_h(np.clip(current_h_sym, 0, 1))
    h_sum = h_full.sum()
    # Repair constraints
    if abs(h_sum - target_sum_full) > 1e-6 and h_sum != 0:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    # Compute final C5
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)

    return (h_full.tolist(), c5_bound, n_points)

import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    resolutions = [10, 20, 40, 70]
    current_h_sym = None  # Will be updated per level

    for n_points in resolutions:
        if time.time() - start_time > budget_s - 10:
            break  # No time left

        dx = 2.0 / n_points
        half_n = n_points // 2
        target_sum_full = n_points / 2.0
        target_sum_half = target_sum_full / 2.0

        # Initialize h_sym for this level
        if current_h_sym is None:
            # First level: uniform initialization
            h_sym = np.full(half_n, 0.5 if half_n > 0 else 0.0)
            h_sum = h_sym.sum()
            h_sym = h_sym * (target_sum_half / h_sum)
        else:
            prev_half_n = len(current_h_sym)
            if n_points == 70:
                # Use provided initial_h_values if available
                initial_h_full = np.array(initial_h_values)
                current_h_sym = initial_h_full[:half_n].copy()
            elif prev_half_n * 2 == half_n:
                # Duplicate samples for scaling up
                current_h_sym = np.repeat(current_h_sym, 2)
            else:
                # Interpolate where resolution changes are not powers of 2
                x_old = np.linspace(0, 1, prev_half_n)
                x_new = np.linspace(0, 1, half_n)
                current_h_sym = np.interp(x_new, x_old, current_h_sym)
            current_h_sym = np.clip(current_h_sym, 0, 1)
            h_sum = current_h_sym.sum()
            if h_sum > 1e-9:
                current_h_sym = current_h_sym * (target_sum_half / h_sum)
            current_h_sym = np.clip(current_h_sym, 0, 1)

        def create_full_h(sym_vars):
            return np.concatenate((sym_vars, sym_vars[::-1]))

        def objective(sym_vars):
            h_full = create_full_h(np.clip(sym_vars, 0, 1))
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0)] * half_n
        repair_cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

        np.random.seed(seed)

        # Differential Evolution for global exploration
        de_budget = (budget_s - (time.time() - start_time)) * 0.3
        max_de_time = min(de_budget, 250)  # Cap per-level DE time
        de_popsize = max(15, int(half_n * 0.5))
        de_maxiter = max(5, int((max_de_time * 1e4) / (n_points**2 * 100)))

        try:
            de_result = differential_evolution(
                objective, bounds, strategy='best2bin',
                maxiter=de_maxiter, popsize=de_popsize,
                mutation=(0.5, 1.0), recombination=0.9,
                seed=seed, workers=1, disp=False, polish=False
            )
            if de_result.success:
                current_h_sym = np.clip(de_result.x, 0, 1)
                h_sum = current_h_sym.sum()
                if h_sum > 1e-9:
                    current_h_sym *= (target_sum_half / h_sum)
                current_h_sym = np.clip(current_h_sym, 0, 1)
        except Exception as e:
            pass

        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break

        # Adaptive Perturbation + Local Search
        perturb_fraction = max(0.1, min(0.5, 0.05 * n_points))
        perturb_size = int(half_n * perturb_fraction)
        perturb_indices = np.random.choice(half_n, perturb_size, replace=False)
        noise_scale = 0.3 * (1 - (time.time() - start_time) / budget_s)
        best_c5 = float('inf')

        while time.time() - start_time < budget_s - 10 and perturb_size > 0:
            perturbed = np.array(current_h_sym.copy())
            noise = np.random.normal(0, noise_scale, size=perturb_size)
            perturbed[perturb_indices] = np.clip(perturbed[perturb_indices] + noise, 0, 1)
            h_sum_perturbed = perturbed.sum()
            if h_sum_perturbed > 1e-9:
                perturbed = np.clip(perturbed * (target_sum_half / h_sum_perturbed), 0, 1)

            # Multi-method local optimization with early stopping
            for method in ('SLSQP', 'L-BFGS-B'):
                try:
                    result = minimize(
                        objective, perturbed, method=method,
                        bounds=bounds, constraints=repair_cons,
                        options={'maxiter': 30, 'ftol': 1e-10}
                    )
                    if result.success:
                        cand = np.clip(result.x, 0, 1)
                        h_sum_cand = cand.sum()
                        if h_sum_cand > 1e-9:
                            cand = cand * (target_sum_half / h_sum_cand)
                            cand = np.clip(cand, 0, 1)
                            c5_cand = objective(cand)
                            if c5_cand < best_c5:
                                best_c5 = c5_cand
                                current_h_sym = cand.copy()
                except:
                    continue

            # Reduce perturbation complexity over time
            perturb_fraction = max(0.05, perturb_fraction * 0.95)
            perturb_size = int(half_n * perturb_fraction)

    # Final projection and validation
    n_points = resolutions[-1]
    h_full = create_full_h(np.clip(current_h_sym, 0, 1))
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - n_points / 2.0) > 1e-6 and h_sum != 0:
        h_full = (h_full / h_sum) * (n_points / 2.0)
        h_full = np.clip(h_full, 0, 1)

    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx

    return (h_full.tolist(), c5_bound, n_points)
