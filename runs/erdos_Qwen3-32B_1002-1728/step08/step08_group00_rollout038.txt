    global_best_score = personal_best_scores[global_best_idx]
    
    c1, c2 = 2.0, 2.0
    inertia_min, inertia_max = 0.4, 0.7
    velocity_max, velocity_min = 0.5, -0.5
    
    pso_end_time = start_time + budget_s * 0.6
    while time.time() < pso_end_time:
        elapsed = time.time() - start_time
        inertia = inertia_max - (inertia_max - inertia_min) * (elapsed / (pso_end_time - start_time))
        inertia = max(inertia_min, inertia)
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - positions[i]) +
                             c2 * r2 * (global_best_position - positions[i]))
            velocities[i] = np.clip(velocities[i], velocity_min, velocity_max)
            new_pos = positions[i] + velocities[i]
            new_pos = np.clip(new_pos, 0, 1.0)
            current_sum = new_pos.sum()
            if current_sum != 0:
                scale_factor = target_sum / current_sum
                new_pos = (new_pos * scale_factor).clip(0, 1.0)
            positions[i] = new_pos
            score = objective_full(new_pos)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = new_pos
                if score < global_best_score:
                    global_best_position[:] = new_pos
                    global_best_score = score
    
    cons_full = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    best_h = global_best_position.copy()
    try:
        result = minimize(
            objective_full, x0=best_h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints=cons_full,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        if result.success and result.fun < global_best_score:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1.0)
            current_sum = best_h.sum()
            if current_sum != 0:
                best_h = (best_h / current_sum) * target_sum
    except:
        pass
    
    final_h = best_h.copy().clip(0, 1.0)
    current_sum = final_h.sum()
    if abs(current_sum - target_sum) > 1e-6:
        final_h = (final_h / current_sum) * target_sum
        final_h = final_h.clip(0, 1.0)
    
    return (final_h.tolist(), global_best_score, n_points)
