        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    # Initialize h_first
    h_first = h_full_initial[:half_n].copy()
    h_first = np.clip(h_first, 0, 1)
    current_sum = h_first.sum()
    if current_sum != 0:
        h_first = (h_first / current_sum) * target_sum_half
    h_first = np.clip(h_first, 0, 1)
    # Add perturbations
    np.random.seed(seed)
    h_first += np.random.normal(0, 0.01, size=h_first.shape)
    h_first = np.clip(h_first, 0, 1)
    current_sum = h_first.sum()
    if current_sum < 1e-9:
        h_first.fill(target_sum_half / half_n)
    else:
        h_first = (h_first / current_sum) * target_sum_half
        h_first = np.clip(h_first, 0, 1)

    best_h_first = h_first.copy()
    best_c5 = float('inf')
    velocity = np.zeros_like(h_first)
    learning_rate = 0.01
    momentum = 0.1
    even_step = True

    while time.time() - start_time < budget_s - 5:
        full_h = create_full_h(h_first)
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_first = h_first.copy()
        lag_idx = np.argmax(corr)
        
        if even_step:
            indices = np.arange(0, half_n, 2)
        else:
            indices = np.arange(1, half_n, 2)

        # Compute gradient subset
        grad = np.zeros_like(h_first)
        full_len = n_points
        eps = 1e-5
        for i in indices:
            orig = h_first[i]
            # Perturb up
            h_up = h_first.copy()
            h_up[i] = orig + eps
            full_up = create_full_h(h_up)
            corr_up = np.correlate(full_up, 1 - full_up, mode='full') * dx
            val_up = corr_up[lag_idx]
            # Perturb down
            h_down = h_first.copy()
            h_down[i] = orig - eps
            full_down = create_full_h(h_down)
            corr_down = np.correlate(full_down, 1 - full_down, mode='full') * dx
            val_down = corr_down[lag_idx]
            grad[i] = (val_up - val_down) / (2.0 * eps)

        # Update velocity and parameters
        velocity[indices] = momentum * velocity[indices] + learning_rate * grad[indices]
        h_first[indices] -= velocity[indices]
        h_first = np.clip(h_first, 0, 1)
        
        # Project to sum constraint
        sum_h = h_first.sum()
        if sum_h < 1e-9:
            h_first.fill(target_sum_half / half_n)
        else:
            h_first = (h_first / sum_h) * target_sum_half
        h_first = np.clip(h_first, 0, 1)

        even_step = not even_step
        momentum = min(momentum + 0.005, 0.9)
        learning_rate *= 0.995

    # Final projection
    best_h_full = create_full_h(best_h_first)
    best_h_full = np.clip(best_h_full, 0, 1)
    best_sum = best_h_full.sum()
    if abs(best_sum - target_sum_full) > 1e-6:
        best_h_full = (best_h_full / best_sum) * target_sum_full
        best_h_full = np.clip(best_h_full, 0, 1)

    return (best_h_full.tolist(), best_c5, n_points)
