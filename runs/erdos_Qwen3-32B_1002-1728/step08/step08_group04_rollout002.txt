import time
import scipy.optimize as opt
import scipy.special as sp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    # Load initial solution
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points)
    x_prime = x - 1.0  # Map [0,2] to [-1,1]

    def project_to_chebyshev(h, max_deg):
        T_basis = np.zeros((n_points, max_deg + 1))
        for m in range(max_deg + 1):
            deg = 2 * m  # Even degrees for symmetry
            T_basis[:, m] = sp.eval_chebyt(deg, x_prime)
        # Least squares solution: h = 0.5 + T_basis @ c => T_basis @ c = h - 0.5
        h_mod = h - 0.5
        c, _ = np.linalg.lstsq(T_basis, h_mod, rcond=None)
        return c.flatten()

    def compute_h(coeffs, T_basis):
        h = 0.5 + T_basis @ coeffs
        h_clipped = np.clip(h, 0, 1)
        sum_h = h_clipped.sum()
        if abs(sum_h - target_sum) > 1e-10:
            scale = target_sum / sum_h
            h_scaled = h_clipped * scale
            h_scaled = np.clip(h_scaled, 0, 1)
        else:
            h_scaled = h_clipped.copy()
        return h_scaled

    def objective(coeffs, T_basis, dx):
        h = compute_h(coeffs, T_basis)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_c = None
    best_c5 = float('inf')
    best_h = h_init.copy()

    stages = [
        {'M': 10, 'maxiter': 200},
        {'M': 30, 'maxiter': 700}
    ]
    current_h = h_init.copy()

    for stage in stages:
        if time.time() - start_time > budget_s - 10:
            break
        M = stage['M']
        maxiter = stage['maxiter']
        T_basis = np.zeros((n_points, M + 1))
        for m in range(M + 1):
            deg = 2 * m
            T_basis[:, m] = sp.eval_chebyt(deg, x_prime)
        # Project current_h into current basis
        c0 = project_to_chebyshev(current_h, M)
        # Optimization
        def callback(xk, d={}):

                    best_c5 = c5
                    best_c = xk.copy()
                    best_h = compute_h(xk, T_basis)

                    raise opt.OptimizeWarning("Time budget exceeded")
