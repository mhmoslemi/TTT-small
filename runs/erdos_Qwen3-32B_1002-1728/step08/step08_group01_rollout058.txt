    h_sym = np.clip(h_sym, 0, 1)

    best_c5 = float('inf')
    best_h_sym = h_sym.copy()

    beta = 10.0
    learning_rate = 0.01
    beta_increase = 1.05
    num_grad_eps = 1e-5
    max_iter = int(budget_s * 0.9 / 0.1)  # Estimation based on iteration timing
    decay_factor = 0.995

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        h_full = np.concatenate((h_sym, h_sym[::-1]))
        h_full = np.clip(h_full, 0, 1)
        h_sum = h_full.sum()
        if abs(h_sum - target_sum_full) > 1e-6:
            h_full = (h_full / h_sum) * target_sum_full
            h_full = np.clip(h_full, 0, 1)

        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        exp_beta_overlap = np.exp(beta * corr)
        smooth_max = (1.0 / beta) * np.log(exp_beta_overlap.sum())

        def obj(h_sym_vec):
            h_sym_vec = np.clip(h_sym_vec, 0, 1)
            h_full = np.concatenate((h_sym_vec, h_sym_vec[::-1]))
            h_full = np.clip(h_full, 0, 1)
            h_sum = h_full.sum()
            if abs(h_sum - target_sum_full) > 1e-6:
                h_full = (h_full / h_sum) * target_sum_full
                h_full = np.clip(h_full, 0, 1)
            corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
            return (1.0 / beta) * np.log(np.clip(np.exp(beta * corr), 1e-10, None).sum())

        grad = approx_fprime(h_sym, obj, num_grad_eps)
        h_sym -= learning_rate * grad
        h_sym = np.clip(h_sym, 0, 1)
        h_sum_half = h_sym.sum()
        if h_sum_half > 1e-9:
            h_sym *= target_sum_half / h_sum_half

        h_full = np.concatenate((h_sym, h_sym[::-1]))
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_sym = h_sym.copy()

        if iter % 50 == 0:
            beta *= beta_increase
        learning_rate *= decay_factor

    h_full = np.concatenate((best_h_sym, best_h_sym[::-1]))
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
