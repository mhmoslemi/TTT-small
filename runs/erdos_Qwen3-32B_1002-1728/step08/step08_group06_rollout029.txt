        total = best_h_current.sum()
        if total != 0:
            best_h_current *= (target_sum / total)
            best_h_current = np.clip(best_h_current, 0, 1)
        corr = np.correlate(best_h_current, 1 - best_h_current, mode='full') * dx
        c5_current = np.max(corr)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = best_h_current
    
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        assert np.isclose(best_h.sum(), target_sum, atol=1e-6)
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return (initial_h_values_given.tolist(), 0.380952, len(initial_h_values_given))
