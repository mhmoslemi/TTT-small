        h_perturbed = h_perturbed / h_perturbed.sum() * target_sum

        # Run local optimization
        new_h, new_c5 = refine_with_slsqp(h_perturbed)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Final SDP-inspired heuristic: create a Toeplitz matrix relaxation
    try:
        t = cp.Variable(n_points)
        T = toeplitz(t)
        lambda_var = cp.Variable()
        # Enforce that the matrix is close to being valid for autocorrelation
        sdp_prob = cp.Problem(
            cp.Maximize(lambda_var),
            [
                t[0] == (target_sum ** 2) / (n_points ** 2),
                t[1:] >= lambda_var / (dx * (n_points - np.arange(1, n_points)))
            ]
        )
        sdp_prob.solve(solver=cp.MOSEK, verbose=False)
        h_proposal = np.sqrt(t.value)
        h_proposal = h_proposal / h_proposal.sum() * target_sum
        h_proposal = np.clip(h_proposal, 0, 1)
        new_c5 = objective_full(h_proposal)
        if new_c5 < best_c5:
            best_h = h_proposal
            best_c5 = new_c5
    except Exception as e:
        pass

    # Final constraint repair
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * target_sum
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), best_c5, n_points)
