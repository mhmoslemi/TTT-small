import numpy as np
import time

def create_full_h(h_half):
    h_full = np.zeros(2*len(h_half))
    h_full[:len(h_half)] = h_half
    h_full[len(h_half):] = h_half[::-1]
    return h_full

def repair_half(h_half, target_sum):
    h_clipped = np.clip(h_half, 0, 1)
    sum_clipped = h_clipped.sum()
    if sum_clipped == 0:
        return np.zeros_like(h_clipped)
    scaling = target_sum / sum_clipped
    h_scaled = h_clipped * scaling
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
