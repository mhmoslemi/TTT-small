        for k in range(2, max_k + 1):
            T[:, k] = 2 * x_prime * T[:, k-1] - T[:, k-2]
        return T

    max_k = 10  # Use even indices up to k=10
    T = chebyshev_basis(x_prime, max_k)
    even_ks = np.array([k for k in range(0, max_k + 1, 2)])
    even_T = T[:, even_ks]  # Use only even-indexed Chebyshev polynomials

    def project_to_chebyshev(h):
        h_weights = h / h.sum() * target_sum
        h_weights = np.clip(h_weights, 0, 1)
        return np.linalg.lstsq(even_T.T @ even_T, even_T.T @ h_weights, rcond=None)[0]

    def reconstruct_h(coeffs):
        h = np.clip(even_T @ coeffs, 0, 1)
        h_sum = h.sum()
        if h_sum > 1e-9:
            h = (h / h_sum) * target_sum
        h = np.clip(h, 0, 1)
        return h

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial setup
    h_initial = np.clip(h_full_initial, 0, 1)
    h_initial_sum = h_initial.sum()
    if h_initial_sum > 1e-9:
        h_initial = h_initial / h_initial_sum * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Project to Chebyshev
    coeffs = project_to_chebyshev(h_initial)
    h_best = reconstruct_h(coeffs)
    best_c5 = evaluate_c5(h_best)
    current_coeffs = coeffs.copy()
    current_h = h_best.copy()

    # Annealing parameters
    temp = 0.25
    alpha = 0.995
    perturb_scale = 0.5

    while time.time() - start_time < budget_s - 10:
        # Perturb one random coefficient
        idx = np.random.randint(len(current_coeffs))
        delta = np.random.normal(0, perturb_scale)
        new_coeffs = current_coeffs.copy()
        new_coeffs[idx] += delta

        new_h = reconstruct_h(new_coeffs)
        new_c5 = evaluate_c5(new_h)
        old_c5 = evaluate_c5(current_h)

        delta_c5 = new_c5 - old_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            current_coeffs = new_coeffs
            current_h = new_h

        if new_c5 < best_c5:
            best_c5 = new_c5
            h_best = new_h

        temp *= alpha
        if np.random.rand() < 0.05 and time.time() - start_time < budget_s - 20:
            current_coeffs = project_to_chebyshev(h_best)
            current_h = reconstruct_h(current_coeffs)
            temp = max(0.1, temp * 0.5)

    return (h_best.tolist(), best_c5, n_points)
