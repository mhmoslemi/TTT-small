def run(seed=42, budget_s=1100, **kwargs):

    current_half = np.clip(current_half, 0, 1)

        s = np.sum(ind)
        if s != 0:
            ind = (ind / s) * target_sum_half
        ind = np.clip(ind, 0, 1)
        population.append(ind)

            tournament_size = 3

                    child[start:start + size] += noise
                elif mut_type == 2:  # Global
                    noise = np.random.normal(0, 0.05, half_n)
                    child += noise

            if s != 0:
                child = (child / s) * target_sum_half
            child = np.clip(child, 0, 1)

    return (h_final.tolist(), best_c5, n_points)
