def compute_gradient(h_full, corr, beta):
    n_points = h_full.shape[0]
    vals = corr * dx
    exp_vals = np.exp(beta * vals)
    sum_exp = np.sum(exp_vals)
    softmax_weights = exp_vals / sum_exp

    # Vectorize m and k
    m = np.arange(n_points)
    k = np.arange(vals.shape[0])

    # Compute term1 and term2
    m_col = m[:, None]  # (n_points, 1)
    k_row = k[None, :]  # (1, len(vals))

    j_plus = m_col + k_row  # (n_points, len(vals))
    mask_plus = j_plus < n_points
    term1 = np.where(mask_plus, 1 - h_full[j_plus], 1.0)

    j_minus = m_col - k_row
    mask_minus = j_minus >= 0
    term2 = np.where(mask_minus, -h_full[j_minus], 0.0)

    # Compute dCk_dh_m for all m and k
    dCk_dh_m = (term1 + term2) * dx  # (n_points, len(vals))

    # Multiply by softmax weights and sum over k
    gradient_full = np.sum(dCk_dh_m * softmax_weights, axis=1)  # (n_points, )

    # Compute gradient_sym using symmetry
    half_n = n_points // 2
    gradient_sym = np.zeros(half_n)
    for i in range(half_n):
        mirrored_i = n_points - 1 - i
        gradient_sym[i] = gradient_full[i] + gradient_full[mirrored_i]
    return gradient_sym
