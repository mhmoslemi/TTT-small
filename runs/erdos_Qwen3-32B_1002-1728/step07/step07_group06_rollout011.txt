def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    L = 2.0
    n_fourier = min(20, n_points // 2)

    # Compute initial Fourier coefficients
    a = scipy.fft.dct(h_initial, norm='ortho')[:n_fourier]
    b = scipy.fft.dst(h_initial, norm='ortho')[:n_fourier]
    variables = np.concatenate([a, b])

    best_c5 = float('inf')
    best_h = h_initial.copy()
    lambda_penalty = 1e3
    max_outer = 30
    outer_iter = 0

    def compute_h(a_coefs, b_coefs):
        x_vals = np.linspace(0, L, n_points, endpoint=False)
        h = np.zeros(n_points)
        for i in range(len(a_coefs)):
            k = i + 1
            h += a_coefs[i] * np.cos(k * np.pi * x_vals / L) + b_coefs[i] * np.sin(k * np.pi * x_vals / L)
        h = 0.5 + h / np.sqrt(len(a_coefs)) if len(a_coefs) > 0 else 0.5
        return np.clip(h, 0, 1)

    def objective(x):
        a_coefs = x[:n_fourier]
        b_coefs = x[n_fourier:]
        h = compute_h(a_coefs, b_coefs)
        corr = np.correlate(h, 1 - h, mode='full')
        C = np.max(corr) * dx

        sum_h = np.sum(h) * dx
        sum_contribution = (sum_h - 1) ** 2
        bound_contribution = np.sum((np.maximum(-h, 0) ** 2) + (np.maximum(h - 1, 0) ** 2))
        return C + lambda_penalty * (sum_contribution + bound_contribution)

    while time.time() - start_time < budget_s and outer_iter < max_outer:
        # Run inner optimization
        result_inner = opt.minimize(
            objective,
            x0=variables,
            method='L-BFGS-B',
            bounds=opt.Bounds(-np.inf, np.inf),
            options={'ftol': 1e-8, 'maxiter': 25, 'gtol': 1e-6}
        )
        if result_inner.success:
            a_new = result_inner.x[:n_fourier]
            b_new = result_inner.x[n_fourier:]
            h_opt = compute_h(a_new, b_new)
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5_opt = np.max(corr_opt) * dx
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

            sum_h_opt = np.sum(h_opt) * dx
            constraint_violation = np.any(h_opt < -1e-6) or np.any(h_opt > 1 + 1e-6) or np.abs(sum_h_opt - 1) > 1e-4

        return np.clip(h_clipped * (1.0 / s), 0, 1)

    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), final_c5, n_points)
