from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Set number of basis functions (low-frequency cosine modes)
    m = max(1, n_points // 4)
    x = np.linspace(0, 2, n_points)
    cos_basis = np.zeros((m, n_points))
    for j in range(1, m + 1):
        cos_basis[j - 1] = np.cos(np.pi * j * x)

    # Project initial h into basis to form starts
    h_proj = h_initial - 0.5
    a_initial = np.array([np.sum(h_proj * cb) * dx for cb in cos_basis])

    def build_h(a):
        h = 0.5 + a @ cos_basis
        h = np.clip(h, 0, 1)
        return h

    def objective(a):
        h = build_h(a)
        # Ensure valid solution for non-counting penalty
        if not (np.all(h >= 0) and np.all(h <= 1)):
            return np.inf
        # Compute correlation max criterion
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Differential Evolution with bounds
    bounds = [(-1.0, 1.0)] * m
    de_result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        strategy='best1bin',
        maxiter=150,
        popsize=12,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-6,
        polish=False,
        workers=1
    )

    # Gradient refinement
    refine_h = build_h(de_result.x)
    refine_c5 = objective(de_result.x)

    def refine_obj(a):
        h = build_h(a)
        if not (np.all(h >= 0) and np.all(h <= 1)):
            return 1e6  # Fallback val
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    final_bounds = [(-1.0, 1.0)] * m
    gl = differential_evolution(lambda a: objective(a), final_bounds, seed=seed + 1, 
                                strategy='currenttobest1bin', maxiter=30, polish=False)
