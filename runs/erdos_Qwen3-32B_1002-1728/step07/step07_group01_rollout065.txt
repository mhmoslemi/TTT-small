from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = (h_initial / h_initial.sum() if h_initial.sum() != 0 else np.zeros(n_points)) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Differential evolution phase
    np.random.seed(seed)
    de_budget = budget_s * 0.5
    start_de = time.time()

    try:
        de_result = differential_evolution(
            objective, bounds,
            strategy='best1bin',
            maxiter=30,
            popsize=30,
            mutation=(0.6, 1.2),
            recombination=0.95,
            tol=1e-6,
            seed=seed,
            workers=1
        )
        if de_result.success:
            best_h = np.clip(de_result.x, 0, 1)
            best_h = (best_h / best_h.sum()) * target_sum if best_h.sum() != 0 else np.zeros(n_points)
            best_h = np.clip(best_h, 0, 1)
        else:
            best_h = h_initial
    except:
        best_h = h_initial

    best_c5 = objective(best_h)

    # Local optimization with adaptive perturbations
    while time.time() - start_time < budget_s - 5:
        remaining = budget_s - (time.time() - start_time)
        frac_remain = remaining / budget_s
        noise_scale = 0.2 * (1 - frac_remain ** 1.2)

        # Full-vector perturbation
        perturbed = best_h.copy()
        perturb_indices = np.random.choice(n_points, size=min(n_points, 20), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, size=len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)

            perturbed = (perturbed / perturbed.sum()) * target_sum

                new_h = np.clip(result.x, 0, 1)
