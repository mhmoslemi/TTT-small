import numpy as np
import time
from scipy.optimize import differential_evolution, minimize, Bounds
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).astype(float)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure initial h is feasible
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = h_initial.sum()
    if current_sum > 1e-10:
        h_initial = (h_initial / current_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Objective function
    def objective(h_var):
        h = np.clip(h_var, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.abs(corr)) * dx

    # Bounds and constraints
    bounds = Bounds([0.0]*n, [1.0]*n)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Differential evolution with adaptive strategy
    pop_size = 30
    max_iter = 100
    de_result = None
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin', 
            mutation=(0.5, 1.2), recombination=0.9, 
            seed=seed, polish=False, workers=1, 
            maxiter=max_iter, popsize=pop_size
        )
    except:
        pass

    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    if de_result and de_result.success and de_result.fun < best_c5:
        best_h = de_result.x
        best_c5 = de_result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * target_sum
    best_h = np.clip(best_h, 0, 1)

    # Perturbation-based local search
    while time.time() - start_time < budget_s - 20:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = min(0.3, 0.1 + 0.2 * frac_remaining ** 1.5)

        # Create perturbed guess
        perturbed = best_h.copy()
        perturb_indices = np.random.choice(n, size=min(20, n//2), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)

        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-10:
            perturbed = (perturbed / perturbed_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        perturbed[perturbed < 1e-8] = 0

        # Local optimization
        try:
            result_slsqp = minimize(objective, perturbed, method='SLSQP', bounds=bounds, 
                                    constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
            if result_slsqp.success:
                new_c5 = result_slsqp.fun
                new_h = result_slsqp.x
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except:
            pass

    # Final scaling fix
    final_h = np.clip(best_h, 0, 1)
    final_sum = final_h.sum()
    if abs(final_sum - target_sum) > 1e-6:
        final_h = final_h * (target_sum / final_sum)
        final_h = np.clip(final_h, 0, 1)

    return (final_h.tolist(), best_c5, n)
