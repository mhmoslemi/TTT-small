                    h_de = result_de.x
                    h_de = h_de / h_de.sum() * target_sum_current if h_de.sum() != 0 else h_de * 0.5
                    h_de = np.clip(h_de, 0, 1)
                    current_c5 = objective(h_de)
                    if current_c5 < best_c5:
                        best_h = h_de
                        best_c5 = current_c5
            except Exception:
                pass

        try:
            cons_slsp = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_current})
            result_slsp = minimize(objective, current_best, method='SLSQP', bounds=bounds, constraints=cons_slsp,
                                   options={'maxiter': 80, 'ftol': 1e-8})
            if result_slsp.success:
                h_slsp = result_slsp.x
                h_slsp = h_slsp / h_slsp.sum() * target_sum_current if h_slsp.sum() != 0 else h_slsp * 0.5
                h_slsp = np.clip(h_slsp, 0, 1)
                c5 = objective(h_slsp)
                if c5 < best_c5:
                    best_h = h_slsp
                    best_c5 = c5
        except Exception:
            pass

    best_h = np.clip(best_h, 0, 1)
    if abs(best_h.sum() - target_sum_full) > 1e-6:
        best_h = best_h / best_h.sum() * target_sum_full
    best_h = np.clip(best_h, 0, 1)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), final_c5, n_points)
