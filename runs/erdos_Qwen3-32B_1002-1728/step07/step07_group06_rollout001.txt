import time
import scipy.optimize as opt
import cvxpy as cp
from scipy.ndimage import zoom

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        s = max(h_clipped.sum(), 1e-9)
        return h_clipped * (target_sum / s)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = [h_initial.copy()]
    for _ in range(5):
        h_rand = np.random.rand(n)
        h_rand = repair_h(h_rand)
        population.append(h_rand)

    best_h = min(population, key=evaluate_c5)
    best_c5 = evaluate_c5(best_h)

    # Genetic operations
    while time.time() - start_time < budget_s * 0.4:
        next_h = []
        candidates = sorted(population, key=evaluate_c5)[:3]

            alpha = np.random.choice([0.3, 0.5, 0.7])
            child = alpha * candidates[p1] + (1 - alpha) * candidates[p2]
            child = repair_h(child)
            # Mutation

            child[idx] += np.random.normal(0, 0.05, size=len(idx))
            child = repair_h(child)
            next_h.append(child)
        population = candidates + next_h
        current_best = min(population, key=evaluate_c5)
        if evaluate_c5(current_best) < best_c5:
            best_h, best_c5 = current_best, evaluate_c5(current_best)

    # Coarse-scale optimization
    coarse_n = max(n // 2, 35)
    def downsample(x):
        return zoom(x.reshape(1, -1), (1, coarse_n / n), order=1).ravel()
    def upsample(x):
        return zoom(x.reshape(1, -1), (1, n / coarse_n), order=0).ravel()

    while time.time() - start_time < budget_s * 0.7:
        h_coarse = downsample(best_h)
        h_coarse = repair_h(h_coarse * (target_sum / h_coarse.sum()))

            x = np.clip(x, 0, 1)
            x = repair_h(x * (target_sum / x.sum()))
            return evaluate_c5(upsample(x))
        res = opt.minimize(coarse_obj, h_coarse, method='L-BFGS-B', bounds=opt.Bounds(0, 1),
