            corr = np.correlate(h_final, 1 - h_final, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
        
        # Update penalty based on constraint satisfaction
        h_current = 0.5 * np.ones(n)
        for k in range(N_fourier):
            k_pi = (k + 1) * np.pi
            cos_term = np.cos(k_pi * x_grid)
            sin_term = np.sin(k_pi * x_grid)
            h_current += (current_a_b[k] * cos_term + current_a_b[N_fourier + k] * sin_term) / np.sqrt(N_fourier)
        violations = np.any(h_current < -1e-8) or np.any(h_current > 1 + 1e-8)
        lambda_ = min(max(lambda_ * (1.2 if violations else 0.9), 1e2), 1e4)
    
    return (best_h.tolist(), best_c5, n)
