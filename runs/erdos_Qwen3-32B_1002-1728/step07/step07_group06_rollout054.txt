        best_c5, best_h = candidates[0]

        # Optimization step
        current_time = time.time() - start_time
        if current_time < budget_s - 20:
            def objective(x):
                x = np.clip(x, 0, 1)
                x = repair_h(x)
                return evaluate_c5(x)
            res = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=opt.Bounds(0, 1),
                               options={'maxiter': 50, 'ftol': 1e-10})
            if res.success:
                opt_h = res.x.copy()
                opt_h = repair_h(opt_h)
                opt_c5 = evaluate_c5(opt_h)
                if opt_c5 < best_c5:
                    best_c5 = opt_c5
                    best_h = opt_h

        if current_time < budget_s - 15:
            z = cp.Variable(n)
            objective_fn = (cp.norm(z)**2) / 2 + 2.0 * cp.sum_squares(z - best_h)
            constraints = [cp.sum(z) == target_sum, 0 <= z, z <= 1]
            problem = cp.Problem(cp.Minimize(objective_fn), constraints)
            problem.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
            if problem.status == 'optimal':
                admm_h = repair_h(z.value)
                admm_c5 = evaluate_c5(admm_h)
                if admm_c5 < best_c5:
                    best_c5 = admm_c5
                    best_h = admm_h

    best_h = repair_h(best_h)
    return (best_h.tolist(), best_c5, n)
