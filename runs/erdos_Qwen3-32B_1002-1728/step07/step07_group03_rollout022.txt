    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Phase 1: Aggressive global search
    def de_objective(x):
        x = np.clip(x, 0, 1)
        sum_x = x.sum()
        if sum_x > 1e-9:
            x_scaled = (x / sum_x) * target_sum_half
        else:
            x_scaled = np.zeros_like(x)
        h_full = np.clip(create_full_h(x_scaled), 0, 1)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    popsize = 40
    maxiter = 10
    de_result = differential_evolution(
        de_objective, bounds, strategy='best1bin',
        maxiter=maxiter, popsize=popsize, tol=1e-4,
        mutation=(0.5, 1.5), recombination=0.95, seed=seed, workers=1, disp=False
    )
    best_h_half = np.clip(de_result.x, 0, 1)
    if best_h_half.sum() > 1e-9:
        best_h_half = (best_h_half / best_h_half.sum()) * target_sum_half

    best_c5 = objective(best_h_half)

    # Phase 2: Refine with custom local search
    def wrapped_objective(x):
        x_clipped = np.clip(x, 0, 1)
        x_sum = x_clipped.sum()
        if x_sum > 1e-9:
            x_scaled = (x_clipped / x_sum) * target_sum_half
        else:
            x_scaled = np.zeros_like(x_clipped)
        h_full = create_full_h(x_scaled)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr_vals) * dx
        return max(0.0, max_corr - best_c5 * 0.99)

    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, 0.05, half_n)
        perturbation = np.clip(perturbation, -0.2, 0.2)
        new_guess = best_h_half + perturbation
        new_guess = np.clip(new_guess, 0, 1)
        if new_guess.sum() > 1e-9:
            new_guess = (new_guess / new_guess.sum()) * target_sum_half
        
        result = minimize(
            wrapped_objective, x0=new_guess, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': 100, 'ftol': 1e-10}
        )
        candidate = result.x
        if result.success:
            candidate = np.clip(candidate, 0, 1)
            if abs(candidate.sum() - target_sum_half) < 1e-6:
                c5_val = objective(candidate)
                if c5_val < best_c5:
                    best_h_half = candidate
                    best_c5 = c5_val

    # Final constraint check
    h_full = np.clip(create_full_h(best_h_half), 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
