    gr = (1 + np.sqrt(5)) / 2  # golden ratio

def golden_section(f, args, a, b, tol=1e-5):

    d = a + (b - a) / gr
    while abs(c - d) > tol:
        fc = f(c, *args)
        fd = f(d, *args)

        else:
