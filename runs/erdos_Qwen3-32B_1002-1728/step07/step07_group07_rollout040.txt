    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break

        # Evaluate fitness
        fitness = [np.max(np.correlate(ind, 1 - ind, mode='full')) * dx for ind in population]
        fitness = np.array(fitness)

        # Select parents
        parent_indices = np.argsort(fitness)[:len(population) // 2]
        parents = population[parent_indices]

        # Generate new population
        new_population = []
        for _ in range(population_size):
            # Select two parents
            parent1 = parents[np.random.choice(len(parents))]
            parent2 = parents[np.random.choice(len(parents))]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Repair after crossover
            child = np.clip(child, 0, 1)
            total = child.sum()
            if abs(total) > 1e-10:
                child = (child / total) * target_sum
            child = np.clip(child, 0, 1)
            
            # Mutation
            if np.random.rand() < 0.1:
                mut_idx = np.random.choice(n_points, size=n_points // 5, replace=False)
                noise = np.random.normal(0, 0.2, size=len(mut_idx))
                child[mut_idx] += noise
                # Repair after mutation
                child = np.clip(child, 0, 1)
                total = np.sum(child)
                if abs(total) > 1e-10:
                    child = (child / total) * target_sum
                child = np.clip(child, 0, 1)
                
            new_population.append(child)
        population = np.array(new_population)

    # Post-GA refinement
    fitness = [np.max(np.correlate(ind, 1 - ind, mode='full')) * dx for ind in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]

    # Local optimization with SLSQP
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5:
        try:
            result = minimize(
                objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                best_h = np.clip(result.x, 0, 1)
        except:
            pass

    # Final repair
    best_h = np.clip(best_h, 0, 1)
    total = best_h.sum()
    if abs(total - target_sum) > 1e-6:
        best_h = (best_h / total) * target_sum
    best_h = np.clip(best_h, 0, 1)
    c5_final = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), c5_final, n_points)
