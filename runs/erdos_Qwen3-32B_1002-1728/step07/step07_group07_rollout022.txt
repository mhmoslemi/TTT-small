    values = np.random.uniform(0, 1, size=n_blocks)
    target_sum = n_blocks / 2.0
    total = np.sum(values)
    if total == 0:
        values.fill(target_sum / n_blocks)
    else:
        values = (values / total) * target_sum
    values = np.clip(values, 0, 1)
    total = np.sum(values)
    # After clipping, sum may be less than target_sum; adjust again
    if total != target_sum:
        excess = target_sum - total
        values += excess / n_blocks
        values = np.clip(values, 0, 1)
    return values
