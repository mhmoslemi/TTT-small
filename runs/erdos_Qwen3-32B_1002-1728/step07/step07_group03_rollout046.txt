    n_points = len(h_full_initial)

    # Initialize symmetric h from initial_h_values

    h_initial_half = np.clip(h_initial_half, 0, 1)
    
    # PSO parameters
    num_particles = 50
    max_iter = 200
    c1 = 2.0  # cognitive coefficient
    c2 = 2.0  # social coefficient
    w_initial = 0.8
    w_final = 0.4
    inertia_decay = (w_initial - w_final) / max_iter
    velocity_clippings = [-0.25, 0.25]

    np.random.seed(seed)
    particles = []
    for _ in range(num_particles):
        perturb = np.random.normal(0, 0.05, half_n)
        new_pos = h_initial_half + perturb
        new_pos = np.clip(new_pos, 0, 1)
        current_sum = new_pos.sum()
        if current_sum != 0:
            scaling = target_sum_half / current_sum
            new_pos = new_pos * scaling
        new_pos = np.clip(new_pos, 0, 1)
        particles.append(new_pos)
    particles = np.array(particles)
    
    velocities = np.random.uniform(velocity_clippings[0], velocity_clippings[1], (num_particles, half_n))
    pbest_pos = particles.copy()
    pbest_cost = np.full(num_particles, float('inf'))
    
    # Evaluate initial costs for particles
    for i in range(num_particles):
        h_full = np.zeros(n_points)
        h_full[:half_n] = particles[i]
        h_full[-half_n:] = particles[i][::-1]
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        cost = np.max(corr_vals) * dx
        pbest_cost[i] = cost
    
    gbest_idx = np.argmin(pbest_cost)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_cost = pbest_cost[gbest_idx]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 50:
            break  # Exit early if time is running out
        
        current_w = w_initial - inertia_decay * iteration
        # Update velocities and positions
        for i in range(num_particles):

            r2 = np.random.rand()
            
            cognitive = c1 * r1 * (pbest_pos[i] - particles[i])
            social = c2 * r2 * (gbest_pos - particles[i])
            new_v = current_w * velocities[i] + cognitive + social

            velocities[i] = new_v
            
            new_pos = particles[i] + new_v
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = new_pos.sum()
            if current_sum != 0:
                scaling = target_sum_half / current_sum
                new_pos = new_pos * scaling
            new_pos = np.clip(new_pos, 0, 1)
            particles[i] = new_pos
            
            # Compute cost
            h_full = np.zeros(n_points)
            h_full[:half_n] = new_pos
            h_full[-half_n:] = new_pos[::-1]
            corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
            cost = np.max(corr_vals) * dx
            
            # Update personal best

                pbest_pos[i] = new_pos.copy()
            
            # Update global best
            if cost < gbest_cost:
                gbest_cost = cost
                gbest_pos[:] = new_pos.copy()  # Ensure copy to prevent mutation
        
        # Check if we need to break early
        if time.time() - start_time > budget_s - 50:
            break
    
    # Local refinement with SLSQP
    def objective_sym(x):
        h_full = np.zeros(n_points)
        h_full[:half_n] = x
        h_full[-half_n:] = x[::-1]

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0.0, 1.0)] * half_n

        h_sum = final_h_sym.sum()

        final_h_sym = np.clip(final_h_sym, 0, 1)

    final_c5 = np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):

    target_sum_half = target_sum_full / 2.0

    # Symmetric initialization
    h_initial_half = h_full_initial[:half_n].copy()
    h_initial_half = np.clip(h_initial_half, 0, 1)
    h_initial_half = (h_initial_half / h_initial_half.sum()) * target_sum_half
    h_initial_half = np.clip(h_initial_half, 0, 1)

    # PSO parameters
    num_particles = 50
    max_iter = 200
    c1 = 2.0
    c2 = 2.0
    w_initial = 0.8
    w_final = 0.4
    velocity_clippings = (-0.25, 0.25)
    inertia_decay = (w_initial - w_final) / max_iter

    # Initialize particles
    np.random.seed(seed)
    particles = []
    for _ in range(num_particles):
        perturb = np.random.normal(0, 0.05, half_n)
        pos = h_initial_half + perturb
        pos = np.clip(pos, 0, 1)
        pos = (pos / pos.sum()) * target_sum_half if pos.sum() != 0 else 1e-6 * np.ones(half_n)
        pos = np.clip(pos, 0, 1)
        particles.append(pos)
    particles = np.array(particles)
    velocities = np.random.uniform(*velocity_clippings, (num_particles, half_n))

    pbest_pos = particles.copy()
    pbest_cost = np.full(num_particles, np.inf)
    for i in range(num_particles):
        h_full = np.zeros(n_points)
        h_full[:half_n] = particles[i]
        h_full[-half_n:] = particles[i][::-1]
        corr_vals = np.correlate(h_full, 1 - h_full, "full")
        pbest_cost[i] = np.max(corr_vals) * dx

    gbest_idx = np.argmin(pbest_cost)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_cost = pbest_cost[gbest_idx]

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 50:
            break
        current_w = w_initial - inertia_decay * iteration
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                current_w * velocities[i]
                + c1 * r1 * (pbest_pos[i] - particles[i])
                + c2 * r2 * (gbest_pos - particles[i])
            )

            new_pos = particles[i] + velocities[i]
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = new_pos.sum()

                new_pos = (new_pos / current_sum) * target_sum_half
