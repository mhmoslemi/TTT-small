import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    half_n = n_points // 2

    h_repaired = np.clip(h_full_initial, 0, 1)
    h_repaired = h_repaired / h_repaired.sum() * target_sum_full
    h_repaired = np.clip(h_repaired, 0, 1)

    def downscale_h(h_fine, n_coarse):
        n_fine = len(h_fine)
        k = n_fine // n_coarse
        h_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            start = i * k
            end = start + k
            h_coarse[i] = np.mean(h_fine[start:end])
        return h_coarse

    def upscale_h(h_coarse, k_scale):
        n_coarse = len(h_coarse)
        h_fine = np.zeros(n_coarse * k_scale)
        for i in range(n_coarse):
            h_fine[i * k_scale:(i + 1) * k_scale] = h_coarse[i]
        return h_fine

    def generate_variant(left_base, right_base, scale=0.1):
        perturbation = np.random.normal(0, scale, len(right_base))
        right_perturbed = np.clip(right_base + perturbation, 0, 1)
        left = left_base.copy()
        right = right_perturbed.copy()
        total = left.sum() + right.sum()
        if total > 0:
            right = right * (target_sum_full / total)
        right = np.clip(right, 0, 1)
        return np.concatenate([left, right])

    variants = []
    for _ in range(20):
        left = h_repaired[:half_n].copy()
        right = h_repaired[half_n:].copy()
        variant = generate_variant(left, right)
        variants.append(variant)

    variants = [h_repaired] + variants
    best_h = h_repaired
    best_c5 = float('inf')

    def coarse_objective(x_flat):
        x = np.clip(x_flat, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / len(x))

    n_coarse = half_n
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        h_coarse = downscale_h(variant, n_coarse)
        bounds = [(0.0, 1.0)] * len(h_coarse)
        try:
            result = differential_evolution(coarse_objective, bounds, strategy='best1bin',
                                            maxiter=10, popsize=15, mutation=(0.5, 1.5),
                                            recombination=0.9, seed=seed, workers=1, disp=False)
            if result.success:
                h_coarse_opt = np.clip(result.x, 0, 1)
                h_fine = upscale_h(h_coarse_opt, 2)
                h_fine = np.clip(h_fine, 0, 1)
                def objective_fine(x):
                    x_clipped = np.clip(x, 0, 1)
                    corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
                    return np.max(corr) * dx
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_full})
                bounds_fine = [(0.0, 1.0)] * len(h_fine)
                result_fine = minimize(objective_fine, h_fine, method='SLSQP',
                                       bounds=bounds_fine, constraints=cons,
                                       options={'maxiter': 200, 'ftol': 1e-8})
                if result_fine.success:
                    h_fine_opt = np.clip(result_fine.x, 0, 1)
                    c5 = np.max(np.correlate(h_fine_opt, 1 - h_fine_opt, mode='full')) * dx
