    def resample_coarse(points, target_length):
        return np.interp(np.linspace(0, len(points)-1, target_length), 
                         np.arange(len(points)), points)

    original_left = h_full_initial[:half_n]
    orig_half_len = len(original_left)
    coarsened_left = resample_coarse(original_left, coarse_half_n)
    coarsened_half = coarsened_left
    n_coarse_full = coarse_half_n * 2

    # Ensure symmetry and repair sum constraint for coarser grid
    def create_full_coarse(h_half):
        h = np.zeros(n_coarse_full)
        h[:coarse_half_n] = h_half
        h[-coarse_half_n:] = h_half[::-1]
        return h

    def objective_coarse(h_half):
        h_clipped = np.clip(h_half, 0, 1)
        h_summary = sum(h_clipped)
        if h_summary == 0:
            return 0.0
        h_normalized = (h_clipped / h_summary) * target_sum_half * (n_coarse_full / half_n)
        h_full = create_full_coarse(h_normalized)
        full_corr = np.correlate(h_full, 1 - h_full, mode='full')
        return max(np.max(full_corr) * (2.0 / n_coarse_full), 0.0)

    def repair_and_normalize(h_half):
        h_clipped = np.clip(h_half, 0, 1)
        h_sum = h_clipped.sum()

            return np.zeros_like(h_clipped)
        return (h_clipped / h_sum) * target_sum_half * (n_coarse_full / half_n)

    co_bounds = [(0.0, 1.0)] * coarse_half_n
