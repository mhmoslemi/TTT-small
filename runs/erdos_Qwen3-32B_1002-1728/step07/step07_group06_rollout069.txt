import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair(h):
        h_clipped = np.clip(h, 0, 1).copy()
        total = h_clipped.sum()
        if total != 0:
            h_clipped *= target_sum / total
        return np.clip(h_clipped, 0, 1)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [repair(h_initial + np.random.normal(0, 0.1, n)) for _ in range(100)]
    population = [repair(p) for p in population]
    best_h = repair(h_initial)
    best_c5 = evaluate(best_h)

    gen_budget = 0.4 * budget_s
    while time.time() - start_time < gen_budget:
        fits = np.array([evaluate(h) for h in population])
        idxs = np.argsort(fits)
        parents = [population[i] for i in idxs[:10]]
        offspring = parents.copy()
        
        while len(offspring) < 100:
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            alpha = np.random.uniform(0.5, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            child = repair(child)
            offspring.append(child)
            
            m_pos = np.random.choice(n, int(n * 0.05), replace=False)
            mutated = p1.copy()
            mutated[m_pos] += np.random.uniform(-0.2, 0.2, len(m_pos))
            mutated = repair(mutated)
            offspring.append(mutated)
        
        population = offspring[:100]
        fits = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fits)
        if fits[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fits[best_idx]

    # Convex refinement
    refine_budget = 0.4 * budget_s
    while time.time() - start_time < refine_budget:
        for h_start in [best_h] + [population[i] for i in np.argsort([evaluate(h) for h in population])[:5]]:
            z = cp.Variable(n)
            obj = cp.Minimize((cp.norm(z)**2 + 0.5 * cp.sum_squares(z - h_start)))
            cons = [cp.sum(z) == target_sum, 0 <= z, z <= 1]
            prob = cp.Problem(obj, cons)
            prob.solve(solver=cp.ECOS, max_iter=150, abstol=1e-4, reltol=1e-4)
            if prob.status == cp.OPTIMAL:
                trial = repair(np.clip(z.value, 0, 1))
                c = evaluate(trial)
                if c < best_c5:
                    best_h, best_c5 = trial, c
            if time.time() - start_time > refine_budget:
                break

    return (best_h.tolist(), best_c5, n)
