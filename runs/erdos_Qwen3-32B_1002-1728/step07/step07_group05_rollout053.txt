    while time.time() < time_end_ga:
        gen += 1
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_full = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        new_population = [population[best_idx].copy()]
        parents = np.argsort(fitness)[:5]

        for _ in range((population_size - 1) // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child1 = 0.5 * (population[p1] + population[p2])
            child2 = (population[p1] * population[p2]) ** 0.5
            child1 = repair_full(child1 + 0.05 * np.random.randn(n_points), target_sum)
            child2 = repair_full(child2 + 0.05 * np.random.randn(n_points), target_sum)
            new_population.extend([child1, child2])
        population = new_population[:population_size]

    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def objective(x):
        x = repair_full(x, target_sum)
        return compute_c5(x)

    try:
        result = minimize(objective, x0=best_h_full, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-6})
        if result.success:
            best_h_full = repair_full(result.x, target_sum)
            best_c5 = min(best_c5, compute_c5(best_h_full))
    except:
        pass

    h_full = np.clip(best_h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
