import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)

    target_sum = n / 2.0

    # Convex optimization using correct overlap constraints
    h = cp.Variable(n)
    C = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    overlaps = []

    # Handle k=0
    expr_k0 = cp.sum(h * (1 - h)) * dx - C
    overlaps.append(expr_k0)

    # Handle k=1 to n-1 and their negative counterparts
    for k in range(1, n):
        # Positive lag
        if n - k > 0:
            sum_pos = cp.sum(h[0:n - k] * (1 - h[k:n])) * dx - C
            overlaps.append(sum_pos)
        # Negative lag
        sum_neg = cp.sum(h[k:n] * (1 - h[0:n - k])) * dx - C
        overlaps.append(sum_neg)

    constraints += [o <= 0 for o in overlaps]
    objective = cp.Minimize(C)

    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.ECOS, max_iters=2000, abstol=1e-8, reltol=1e-8)
    except Exception:
        pass  # Fall back if solver fails

    elapsed = time.time() - start_time

    return (h_opt.tolist(), best_c5, n)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
