import numpy as np
import time
import cvxpy as cp
import numpy.random as nr

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n = len(initial_h)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return h.copy()
        scale = target_sum / current_sum
        return np.clip(h * scale, 0, 1)

    best_h = repair(initial_h.copy())
    best_c5 = evaluate_c5(best_h, dx)

    # Genetic Algorithm
    POPULATION_SIZE = 100
    population = [initial_h.copy()]
    for _ in range(POPULATION_SIZE - 1):
        h = population[0].copy()
        h += np.random.uniform(-0.1, 0.1, n)
        h = repair(h)
        population.append(h)

    start_time = time.time()
    budget_end = start_time + budget_s

    while time.time() < budget_end and len(population) == POPULATION_SIZE:
        evaluated = []
        for h in population:
            evaluated.append((evaluate_c5(h, dx), h))
        evaluated.sort(key=lambda x: x[0])

        top_count = max(1, int(0.1 * len(population)))
        top = [x[1] for x in evaluated[:top_count]]
        if evaluated[0][0] < best_c5:
            best_c5 = evaluated[0][0]
            best_h = evaluated[0][1].copy()

        new_population = top.copy()
        while len(new_population) < POPULATION_SIZE:
            p1 = top[nr.randint(len(top))]
            p2 = top[nr.randint(len(top))]
            alpha = nr.uniform(0.4, 0.6)
            child = alpha * p1 + (1 - alpha) * p2
            child = repair(child)
            if nr.rand() < 0.2:
                idx = nr.choice(n, size=int(n * 0.1), replace=False)
                child[idx] += nr.uniform(-0.2, 0.2, len(idx))
                child = repair(child)
            new_population.append(child)
        population = new_population

    # Convex refinement
    for h in population[:max(10, POPULATION_SIZE // 10)]:
        k_shift = find_worst_k(h, dx)
        if k_shift < 0:
            continue
        grads = np.zeros(n)
        for i in range(n - k_shift):
            grads[i] += h[i + k_shift]
            grads[i + k_shift] += h[i]
        h_var = cp.Variable(n)
        s_current = sum(h[i] * h[i + k_shift] for i in range(n - k_shift))
        obj = cp.Maximize(grads @ h_var - grads @ h + s_current)
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.GLPK, verbose=False)
        if problem.status == cp.OPTIMAL:
            trial = repair(h_var.value)
            c5 = evaluate_c5(trial, dx)
            if c5 < best_c5:

                best_c5 = c5

    # Simulated Annealing
