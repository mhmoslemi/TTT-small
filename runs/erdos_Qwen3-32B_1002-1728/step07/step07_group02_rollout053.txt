        idx = np.random.choice(half_n, perturb_count, replace=False)
        perturbed[idx] += np.random.normal(0, noise_scale, perturb_count)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-8:
            perturbed = (perturbed / perturbed_sum) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        # Local optimization with different methods
        for method in ['SLSQP', 'COBYLA']:
            try:
                result = minimize(
                    objective, x0=perturbed, method=method,
                    bounds=bounds, constraints=cons if method != 'COBYLA' else None,
                    options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
                )
                if result.success:
                    new_h = result.x.copy()
                    new_h = np.clip(new_h, 0, 1)
                    if np.allclose(new_h.sum(), target_sum_half, atol=1e-6):
                        obj = objective(new_h)
                        if obj < best_c5:
                            final_h_half = new_h.copy()
                            best_c5 = obj
            except:
                continue

    # Final cleanup and validation
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
