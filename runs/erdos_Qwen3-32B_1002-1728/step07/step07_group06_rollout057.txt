            return np.sum(h[lag_abs:lag_abs + valid] * (1 - h[:valid])) * dx

    # Perturb initial solutions
    perturbed_h_list = [h.copy() for _ in range(3)]
    for i in range(3):
        idx = np.random.randint(0, n, size=10)
        perturbed_h_list[i][idx] += np.random.uniform(-0.1, 0.1, size=10)
        perturbed_h_list[i] = repair_h(perturbed_h_list[i])
        current_c5 = evaluate_c5(perturbed_h_list[i], dx)
        if current_c5 < best_c5:
            best_h = perturbed_h_list[i].copy()
            best_c5 = current_c5

    # Constraints for optimization
    bounds = opt.Bounds(0, 1)
    sum_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    while time.time() - start_time < budget_s:
        # Evaluate current overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - n + 1
        if lag == 0:
            break  # Already optimal

        # Define objective for this lag
        def objective(x):
            return overlap_for_lag(x, lag)

        # Optimize
        res = opt.minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_h = repair_h(new_h)
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

        # Apply perturbed checks
        for i in range(2):
            idx = np.random.randint(0, n, size=5)
            perturbed = best_h.copy()
            perturbed[idx] += np.random.uniform(-0.1, 0.1, size=5)
            perturbed = repair_h(perturbed)
            perturbed_c5 = evaluate_c5(perturbed, dx)
            if perturbed_c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = perturbed_c5

    # Final repair
    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
