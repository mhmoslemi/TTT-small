from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Symmetrical reparation with exact constraint handling
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_half_repaired = h_full_initial[:half_n].copy()
    h_half_repaired = h_half_repaired / h_half_repaired.sum() * target_sum_half
    h_half_repaired = np.clip(h_half_repaired, 0, 1)

    # Create full solution from repaired half
    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    # Objective function with precise constraint verification
    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        if abs(h_clipped.sum() - target_sum_full) > 1e-6:
            h_clipped = (h_clipped / h_clipped.sum()) * target_sum_full
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Constraints and bounds
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Phase 1: Diversified differential evolution with adaptive parameters
    np.random.seed(seed)
    de_budget = 0.4 * budget_s

    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1exp', maxiter=maxiter, 
            popsize=popsize, mutation=0.8, recombination=0.9,
            seed=seed, tol=1e-6, workers=1, disp=False
        )
        best_h_half = np.clip(de_result.x, 0, 1)
        best_h_half = best_h_half / best_h_half.sum() * target_sum_half
        best_c5 = objective(best_h_half)
    except:
        de_result = None
        best_h_half = h_half_repaired.copy()
        best_c5 = objective(best_h_half)

    # Phase 2: Hybrid local search with multiple strategies
    while time.time() - start_time < budget_s - 20:
        time_left = budget_s - (time.time() - start_time)

        perturbed = np.clip(perturbed, 0, 1)
