def split_block(L, V):

    # Generate split lengths L1 and L2.

    split_ratio = np.random.beta(2, 2)  # Skewed towards 50-50 split.

    L1 = int(L * split_ratio)

    L2 = L - L1

    if L1 == 0 or L2 == 0:

        return None

    # Now compute possible V1 and V2.

    # total = L1 * V1 + L2 * V2 = L * V.

    # Let's set V1 to be a random value between V - delta and V + delta, but ensuring V2 is within [0,1].

    # Or generate V1 randomly such that V2 is in [0,1].

    # For example, solve for V2 = (L * V - L1 * V1) / L2

    # Ensure V2 in [0,1]

    # V1 must be such that:

    # 0 ≤ (L * V - L1 * V1)/L2 ≤ 1

    # Rearranged:

    # L * V - L2 * 1 ≤ L1 * V1 ≤ L * V

    # So:

    # V_min = (L*V - L2)/L1

    # V_max = L*V / L1

    V_min = max(0, (L * V - L2) / L1)

    V_max = min(1, (L * V) / L1)

    if V_min > V_max:

        return None  # Can't split without exceeding limits

    V1 = np.random.uniform(V_min, V_max)

    V2 = (L * V - L1 * V1) / L2

    return (L1, V1), (L2, V2)
