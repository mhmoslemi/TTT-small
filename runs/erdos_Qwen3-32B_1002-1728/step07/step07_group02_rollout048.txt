        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
            best_h_half = (best_h_half / best_h_half.sum()) * target_sum_half
            best_h_half = np.clip(best_h_half, 0, 1)
            final_h_half = best_h_half
            best_c5 = objective(final_h_half)
    except:
        pass  # Skip on timeout/error

    # Phase 2: Local optimization with multiple strategies and perturbations
    while time.time() - start_time < budget_s - 5:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.3 + 0.5 * (1 - frac_remaining ** 1.2)  # Higher initial noise

        # Create perturbed guess with more indices
        perturbed = final_h_half.copy()
        perturb_indices = np.random.choice(half_n, size=min(half_n, 15), replace=False)  # Perturb more
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        ifpert_sum = perturbed.sum()
        if ifpert_sum != 0:
            perturbed = (perturbed / ifpert_sum) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        perturbed[perturbed < 1e-8] = 0

        # Local optimization with different methods and lower maxiter
        try:
            result = minimize(
                objective, x0=perturbed, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                new_h_half = np.clip(result.x, 0, 1)
                new_h_half = (new_h_half / new_h_half.sum()) * target_sum_half
                new_h_half = np.clip(new_h_half, 0, 1)
                new_c5 = objective(new_h_half)
                if new_c5 < best_c5:
                    final_h_half = new_h_half
                    best_c5 = new_c5
        except:
            pass

        try:
            result_cobyla = minimize(
                objective, x0=perturbed, method='COBYLA',
                bounds=bounds,
                options={'maxiter': 100, 'eps': 1e-8, 'disp': False}
            )
            if result_cobyla.success:
                new_h_half = np.clip(result_cobyla.x, 0, 1)
                new_h_half = (new_h_half / new_h_half.sum()) * target_sum_half
                new_h_half = np.clip(new_h_half, 0, 1)
                new_c5 = objective(new_h_half)
                if new_c5 < best_c5:
                    final_h_half = new_h_half
                    best_c5 = new_c5
        except:
            pass

    # Final cleanup and validation
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
