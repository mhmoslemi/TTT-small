            best_h = pop[0].copy()  # Assume top is best

        if prev_best != 0:
            progress = (prev_best - best_c5) / prev_best
            if progress < 0.01:
                sigma *= 1.05
            else:
                sigma *= 0.95
        sigma = np.clip(sigma, 0.01, 0.5)

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    if time.time() - start_time < budget_s - 10:
        result = minimize(
            evaluate, x0=best_h, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12}
        )
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_c5 = evaluate(refined_h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h
    h_full[-half_n:] = best_h[::-1]
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
