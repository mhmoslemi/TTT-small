        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)

    # Create symmetric PSO particles
    np.random.seed(seed)
    num_particles = 20
    particles = np.zeros((num_particles, half_n))
    particles[0] = h_full_initial[:half_n].copy()

    for i in range(1, num_particles):
        perturbed = particles[0].copy()
        perturbed += np.random.normal(0, 0.1, size=half_n)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = perturbed.sum()
        if sum_perturbed != 0:
            perturbed = (perturbed / sum_perturbed) * target_sum_half
        else:
            perturbed.fill(target_sum_half / half_n)
        perturbed = np.clip(perturbed, 0, 1)
        particles[i] = perturbed

    velocities = np.zeros_like(particles)

    def evaluate_particle(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    personal_best_positions = particles.copy()
    personal_best_values = np.array([evaluate_particle(p) for p in particles])
    global_best_idx = np.argmin(personal_best_values)
    global_best_pos = personal_best_positions[global_best_idx]
    global_best_value = personal_best_values[global_best_idx]

    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    vel_clip = 0.1

    while time.time() - start_time < budget_s - 10:
        for i in range(num_particles):
            r1 = np.random.rand(half_n)
            r2 = np.random.rand(half_n)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best_positions[i] - particles[i]) +
                             social * r2 * (global_best_pos - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_clip, vel_clip)

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = particles[i].sum()
            if sum_p != 0:
                particles[i] = (particles[i] / sum_p) * target_sum_half
            else:
                particles[i].fill(target_sum_half / half_n)
            particles[i] = np.clip(particles[i], 0, 1)

        new_values = np.array([evaluate_particle(p) for p in particles])
        improved = new_values < personal_best_values
        personal_best_values[improved] = new_values[improved]
        personal_best_positions[improved] = particles[improved].copy()

        current_global_idx = np.argmin(new_values)
        current_val = new_values[current_global_idx]
        if current_val < global_best_value:
            global_best_value = current_val
            global_best_pos = particles[current_global_idx].copy()

    h_full = np.concatenate([global_best_pos, global_best_pos[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), global_best_value, n_points)
