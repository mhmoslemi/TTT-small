def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_full = len(h_initial)
    n_half = n_full // 2
    dx = 2.0 / n_full

    # Basis function parameters
    m = 5  # Number of basis functions
    x_half = np.arange(n_half) / n_half  # x in [0,1)
    Phi = np.zeros((n_half, m))
    for j in range(1, m+1):
        Phi[:, j-1] = np.cos(j * np.pi * x_half)

    # Initial guess for coefficients a from initial h_values
    h_half_initial = h_initial[:n_half]
    h_initial_avg = 0.5
    target = h_half_initial - h_initial_avg
    a_initial = np.linalg.pinv(Phi) @ target

    # Define constraints for the optimization
    lower = -0.5 * np.ones(n_half)
    upper = 0.5 * np.ones(n_half)
    constraint = LinearConstraint(Phi, lower, upper)

    # Objective function to minimize
    def objective(a_coeff):
        h_half = 0.5 + Phi @ a_coeff
        h_full = np.zeros(n_full)
        h_full[:n_half] = h_half
        h_full[n_half:2*n_half] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)  # Ensure bounds due to numerical precision
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Optimization
    result = minimize(
        objective,
        x0=a_initial,
        method='SLSQP',
        constraints=constraint,
        bounds=[(-1.0, 1.0)] * m,
