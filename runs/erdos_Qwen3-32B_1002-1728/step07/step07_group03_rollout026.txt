    h_full = create_full_h(h_half)
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if not np.isclose(current_sum, target_sum_full):
        h_full = h_full * target_sum_full / current_sum
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
