    # Convex objective: minimize sum of squares of overlaps
    objective = cp.Minimize(sum([o**2 for o in overlaps]))
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.SCS, verbose=False, max_iter=3000, eps=1e-8)
    except:
        pass

    # Extract solution
    ifnp = np.isfinite(h_half.value).all()
    h_half_val = h_half.value if ifnp else h_initial[:half_n].copy()
    h_half_val = np.clip(h_half_val, 0, 1)
    
    # Enforce integral constraint explicitly
    target_sum = n_points / 4.0
    current_sum = h_half_val.sum()
    h_half_val = h_half_val / current_sum * target_sum if abs(current_sum) > 1e-12 else target_sum/half_n * np.ones_like(h_half_val)

    # Construct full solution
    h_full_val = np.zeros(n_points)
    h_full_val[:half_n] = h_half_val
    h_full_val[-half_n:] = h_half_val[::-1]
    h_full_val = np.clip(h_full_val, 0, 1)
    
    # Quick constraint fix
    scaling = (n_points / 2.0) / h_full_val.sum()
    h_full_val = np.clip(h_full_val * scaling, 0, 1)

    # Refinement phase with local optimization
    if time.time() - start_time < budget_s * 0.8:
        def obj(x):
            h = np.zeros(n_points)
            h[:half_n] = x
            h[-half_n:] = x[::-1]
            corr = np.correlate(h, 1-h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0,1)]*half_n
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(obj, h_half_val, bounds=bounds, constraints=cons, method='SLSQP', 
                         options={'maxiter': 200, 'ftol': 1e-10})
        if result.success:
            h_half_val = result.x
            h_full_val = np.zeros(n_points)
            h_full_val[:half_n] = h_half_val
            h_full_val[-half_n:] = h_half_val[::-1]
            h_full_val = np.clip(h_full_val, 0, 1)

    # Final evaluation
    corr = np.correlate(h_full_val, 1-h_full_val, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_full_val.tolist(), c5_bound, n_points)
