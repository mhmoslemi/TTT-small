import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair symmetry and constraints
    def repair_symmetric_h(sym_vars, target_sum):
        sym_clipped = np.clip(sym_vars, 0, 1)
        total = sym_clipped.sum()
        if total == 0:
            return sym_clipped
        scale = target_sum / total
        scaled = sym_clipped * scale
        return np.clip(scaled, 0, 1)

    # Initialize symmetric half
    h_sym_current = h_full_initial[:half_n].copy()
    h_sym_current = repair_symmetric_h(h_sym_current, target_sum_half)
    best_half = h_sym_current.copy()
    best_c5 = None

    # Objective function with h_sym and full h reconstruction
    def objective(sym_vars):
        h_sym = np.clip(sym_vars, 0, 1)
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_sym
        h_full[-half_n:] = h_sym[::-1]
        h_clipped = np.clip(h_full, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum == 0:
            return float('inf')
        h_normalized = h_clipped / h_sum * target_sum_full
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    # Grid search with decreasing step sizes
    grid_levels = [0.25, 0.15, 0.08, 0.03]
    for step_size in grid_levels:
        if time.time() - start_time > budget_s - 5:
            break
        num_samples = 25 if step_size > 0.1 else 15
        for _ in range(num_samples):
            if time.time() - start_time > budget_s - 5:
                break
            # Perturb current best
            perturbation = np.random.normal(0, step_size, half_n)
            new_half = best_half + perturbation
            # Repair and evaluate
            repaired = repair_symmetric_h(new_half, target_sum_half)
            # Evaluate only if repaired is better
            current_score = objective(repaired)
            # Update best if needed
            if best_c5 is None or current_score < best_c5:
                best_c5 = current_score
                best_half = repaired.copy()

    # Local optimization with constraints
    bounds_half = [(0.0, 1.0) for _ in range(half_n)]
    cons_half = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    while time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 5:
            break
        # Create perturbation for local search
        current_pert = best_half.copy()
        current_pert += np.random.normal(0, 0.05, half_n)
        current_pert = repair_symmetric_h(current_pert, target_sum_half)

        # SLSQP optimization
        try:
            result_slsqp = minimize(
                objective, x0=current_pert, method='SLSQP', bounds=bounds_half,
                constraints=cons_half, options={'maxiter': 150, 'ftol': 1e-10, 'disp': False}
            )
            if result_slsqp.success:
                candidate = repair_symmetric_h(result_slsqp.x, target_sum_half)
                candidate_score = objective(candidate)
                if candidate_score < best_c5:
                    best_c5 = candidate_score
                    best_half = candidate.copy()
        except:
            pass

        # COBYLA as fallback
        try:
            result_cobyla = minimize(
                objective, x0=current_pert, method='COBYLA', bounds=bounds_half,
                options={'maxiter': 80, 'eps': 1e-7, 'disp': False}
            )
            if result_cobyla.success:
                candidate = repair_symmetric_h(result_cobyla.x, target_sum_half)
                candidate_score = objective(candidate)
                if candidate_score < best_c5:
                    best_c5 = candidate_score
                    best_half = candidate.copy()
        except:
            pass

    # Final validation and repair
    final_h_half = repair_symmetric_h(best_half, target_sum_half)
    h_full = np.zeros(n_points)
    h_full[:half_n] = final_h_half
    h_full[-half_n:] = final_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6 and h_sum != 0:
        h_full = h_full / h_sum * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5 if best_c5 is not None else float('inf'), n_points)
