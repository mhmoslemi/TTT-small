        if result.success:
            optimized = np.clip(result.x, 0, 1)
            optimized_sum = optimized.sum()
            if optimized_sum > 1e-8:
                optimized = (optimized / optimized_sum) * (current_n / 2.0)
                optimized = np.clip(optimized, 0, 1)
                if current_n == n_points:
                    current_c5 = compute_c5(optimized, current_n)
                    if current_c5 < best_c5:
                        h = optimized.copy()
                        best_c5 = current_c5
                else:
                    h = resample_h(optimized, n_points)
                    h_sum = h.sum()
                    h = (h / h_sum) * target_sum
                    h = np.clip(h, 0, 1)

    # Final cleanup and verification
    h_sum = h.sum()
    h = np.clip(h, 0, 1)
    if h_sum > 1e-8:
        h = (h / h_sum) * target_sum
        h = np.clip(h, 0, 1)
    return (h.tolist(), compute_c5(h, n_points), n_points)
