def compute_gradient_full(h, k_max, dx):
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        j = i + k_max
        if j < n:
            grad[i] += (1 - h[j])
            grad[j] -= h[i]
    grad *= dx
    return grad

    h = np.zeros(n)

    h[len(sym_vars):] = sym_vars[::-1]
