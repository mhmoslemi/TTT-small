import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):

        h_half = repair_half(h_half)
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half

        return np.max(corr) * dx
