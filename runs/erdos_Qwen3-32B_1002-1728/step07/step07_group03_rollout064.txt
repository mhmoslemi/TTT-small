from scipy.optimize import minimize, SLSQP
import time

def run(seed=42, budget_s=1100, **kwargs):

        h_sym_initial = (h_sym_initial / total) * target_sum_half
    h_sym_initial = np.clip(h_sym_initial, 0, 1)
    h_sym_initial = (h_sym_initial / h_sym_initial.sum()) * target_sum_half

    # PSO parameters

    velocity_clamp = 0.25
    c1, c2 = 2.0, 2.0
    max_iter = 200
    inertia_start, inertia_end = 0.9, 0.4

        total = positions[i].sum()
        if total != 0:

        positions[i] = np.clip(positions[i], 0, 1)
        total = positions[i].sum()
        positions[i] = (positions[i] / total) * target_sum_half
    velocities = np.random.uniform(-0.05, 0.05, size=(pop_size, half_n))

    personal_best_positions = positions.copy()
    personal_best_scores = np.full(pop_size, np.inf)
    for i in range(pop_size):
        h_full = np.concatenate([positions[i], positions[i][::-1]])
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        score = np.max(corr) * dx
        personal_best_scores[i] = score

    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    while time.time() - start_time < budget_s - 10:

            break

        inertia = inertia_start - iter_progress * (inertia_start - inertia_end)

        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(half_n), np.random.rand(half_n)
            cognitive = c1 * r1 * (personal_best_positions[i] - positions[i])

            velocities[i] = (inertia * velocities[i] + cognitive + social)
