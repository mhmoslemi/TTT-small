Now, considering the time budget, differential evolution with 70 variables may be slow, so the popsize and maxiter are adjusted. The phase 2 uses multiple restarts with both SLSQP and COBYLA.

import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_with_penalty(h_full):
        h_clipped = np.clip(h_full, 0, 1)
        current_sum = h_clipped.sum()
        penalty_scale = 1e4
        penalty = penalty_scale * (current_sum - target_sum)**2
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx + penalty

    def objective_clean(h_full):
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.)] * n_points

    # Phase 1: Global search with penalty
    np.random.seed(seed)
    popsize = 15
    maxiter = 60
    de_budget = budget_s * 0.7
    best_h = h_initial.copy()
    best_c5 = objective_clean(best_h)
    
    try:
        # DE with penalty-based constraint
        de_result = differential_evolution(
            objective_with_penalty, bounds, strategy='best1bin',
            maxiter=maxiter, popsize=popsize, mutation=(0.5, 1.5),
            recombination=0.85, seed=seed, workers=1, disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            best_h = np.clip(de_result.x, 0, 1).copy()
    except:
        pass

    # Postprocess solution to satisfy constraint exactly
    def repair_sum(h):
        h_clipped = np.clip(h, 0, 1)
        total = h_clipped.sum()
        if total != 0:
            h_clipped = (h_clipped / total) * target_sum
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    best_h = repair_sum(best_h)
    best_c5 = objective_clean(best_h)

    # Phase 2: Adaptive local refinement with perturbation
    while time.time() - start_time < budget_s - 15:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.3 * (1 - frac_remaining ** 1.2)
        perturb_size = min(n_points // 4, 8)
        
        # Create perturbed trial solution
        perturbed = best_h.copy()
        perturb_idx = np.random.choice(n_points, size=perturb_size, replace=False)
        perturbed[perturb_idx] += np.random.normal(0, noise_scale, perturb_size)
        perturbed = np.clip(perturbed, -1, 2)  # Allow temporary overflow
        
        # Feasible repair
        perturbed_sum = perturbed.sum()
        if abs(perturbed_sum - target_sum) > 0.2 * n_points:
            perturbed = repair_sum(perturbed)
        else:
            # Adjustment that preserves sum structure
            for _ in range(min(5, n_points // 10)):
                i = np.random.randint(n_points)
                perturbed[i] += np.random.uniform(-0.1, 0.1)
                if np.any(perturbed < -0.1) or np.any(perturbed > 2.1):
                    perturbed = repair_sum(perturbed)
                    break

        # Local optimization with multiple solvers
        try:
            result = minimize(
                objective_clean, x0=perturbed, method='L-BFGS-B',
                bounds=bounds,
                options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 120}
            )
            if result.success:
                new_h = repair_sum(result.x)
                new_c5 = objective_clean(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except:
            pass
        
        try:
            result = minimize(
                objective_clean, x0=perturbed, method='SLSQP',
                bounds=bounds,
                options={'maxiter': 150, 'ftol': 1e-10}
            )
            if result.success:
                new_h = repair_sum(result.x)
                new_c5 = objective_clean(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except:
            pass

    # Final strict validation
    final_h = repair_sum(best_h)
    final_c5 = objective_clean(final_h)

    return (final_h.tolist(), final_c5, n_points)
