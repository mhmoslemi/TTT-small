import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Symmetric initialization
    h_sym_initial = h_full_initial[:half_n].copy()
    h_sym_initial = np.clip(h_sym_initial, 0, 1)
    current_sum = h_sym_initial.sum()
    if current_sum != 0:
        h_sym_initial = (h_sym_initial / current_sum) * target_sum_half
    h_sym_initial = np.clip(h_sym_initial, 0, 1)

    def create_full_h(sym_vars):
        return np.concatenate([sym_vars, sym_vars[::-1]])

    def evaluate(h_sym):
        h_full = create_full_h(h_sym)
        h_scaled = np.clip(h_full, 0, 1)
        h_sum = h_scaled.sum()
        if h_sum != 0:
            h_scaled = (h_scaled / h_sum) * target_sum_full
        h_clipped = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    def enforce_constraints(h_sym):
        h_sym_clipped = np.clip(h_sym, 0, 1)
        h_sum = h_sym_clipped.sum()
        if h_sum != 0:
            h_sym_clipped = (h_sym_clipped / h_sum) * target_sum_half
        return np.clip(h_sym_clipped, 0, 1)

    # Evolution Strategy parameters
    np.random.seed(seed)
    popsize = 50
    sigma = 0.1
    num_gen = 150
    elite_ratio = 0.5
    best_h = h_sym_initial.copy()
    best_val = evaluate(best_h)
    best_h = enforce_constraints(best_h)

    # ES phase with adaptive sigma
    for gen in range(num_gen):
        if time.time() - start_time > budget_s - 30:
            break
        population = []
        for _ in range(popsize):
            h_mut = best_h + sigma * np.random.randn(half_n)
            h_valid = enforce_constraints(h_mut)
            population.append(h_valid)
        fitness = np.array([evaluate(ind) for ind in population])
        elite_idx = np.argsort(fitness)[:int(popsize * elite_ratio)]
        new_population = [population[i] for i in elite_idx]
        
        if len(new_population) > 0:
            best_idx = np.argmin([evaluate(ind) for ind in new_population])
            new_best = new_population[best_idx]
            if evaluate(new_best) < best_val:
                best_h = new_best
                best_val = evaluate(best_h)
        
        # Adapt sigma
        if len(new_population) > 1:
            fitness_change = fitness[elite_idx].mean() - best_val
            if fitness_change < 1e-6:
                sigma *= 1.1
            else:
                sigma *= 0.9
        sigma = np.clip(sigma, 0.01, 0.5)

    # Local optimization phase
    bounds = [(0.0, 1.0)] * half_n
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)
    
    while time.time() - start_time < budget_s - 10:
        init_guess = best_h + 0.05 * np.random.randn(half_n)
        init_guess = enforce_constraints(init_guess)
        
        result = minimize(
            evaluate, x0=init_guess, bounds=bounds, constraints=constraints,
            method='L-BFGS-B', options={'ftol': 1e-12, 'maxiter': 100}
        )
        
        if result.success:
            candidate = result.x
            candidate = enforce_constraints(candidate)
            val = evaluate(candidate)
            if val < best_val:
                best_h = candidate
                best_val = val

    # Final constraint enforcement
    h_final = create_full_h(best_h)
    h_final = np.clip(h_final, 0, 1)
    h_sum = h_final.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_final = (h_final / h_sum) * target_sum_full
    result = (
        h_final.tolist(),
        best_val if not np.isnan(best_val) else float('inf'),
        n_points
    )
    return result
