                x0=best_h_half, method='SLSQP',
                bounds=[(0, 1) for _ in range(half_n)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                if (c5_new := compute_c5(x_opt)) < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
