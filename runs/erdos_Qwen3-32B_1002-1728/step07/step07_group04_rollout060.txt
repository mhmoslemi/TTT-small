import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    m = 5  # number of basis functions to consider
    # Precompute the basis matrix: (m x n_points)
    basis = np.zeros((m, n_points))
    for j in range(1, m+1):
        basis[j-1] = np.cos(np.pi * j * x / 2)
    # Initial a's (all zero for initial flat h)
    a_initial = np.zeros(m)
    # Define the constraints: 0 <= 0.5 + basis @ a <=1 for all i
    # This can be represented as two sets of linear inequality constraints.
    # For each row i of basis, the constraint is:
    # lower: 0 - 0.5 <= basis_i @ a <= 0.5
    # upper: 0 <= basis_i @ a <= 1 - 0.5 = 0.5
    constraints = []
    lower = -0.5 * np.ones(n_points)
    upper = 0.5 * np.ones(n_points)
    for i in range(n_points):
        # For lower bound: basis_row[i] @ a >= lower[i]
        constraints.append({'type': 'ineq', 'fun': lambda a, i=i, b_row=basis[i]: lower[i] - np.dot(b_row, a)})
        # For upper bound: basis_row[i] @ a <= upper[i]
        constraints.append({'type': 'ineq', 'fun': lambda a, i=i, b_row=basis[i]: upper[i] - np.dot(b_row, a)})

    def objective(a):
        h_basis = np.dot(basis.T, a)
        h = 0.5 + h_basis
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Initial guess
    res = minimize(objective, x0=a_initial, constraints=constraints, method='SLSQP', options={'maxiter': 200, 'disp': False})
    a_opt = res.x
    # Compute the final h
    h_basis = np.dot(basis.T, a_opt)
    h = 0.5 + h_basis

    # Constraints: 0 <= 0.5 + h_basis(x) <= 1 for all x in sample grid
    def make_lower_bound_constraint(i):
        def constraint(a):
            return 0.5 + np.dot(basis[:, i], a)
        return {'type': 'ineq', 'fun': constraint}
    def make_upper_bound_constraint(i):
        def constraint(a):
            return 0.5 - np.dot(basis[:, i], a)
        return {'type': 'ineq', 'fun': constraint}

    constraints = []
    for i in range(n_points):
        constraints.append(make_lower_bound_constraint(i))
        constraints.append(make_upper_bound_constraint(i))

    # Objective function: minimize max_k [corr(h, (1-h))]
    def objective(a_coefs):
        h_basis = np.dot(basis, a_coefs)
        h = 0.5 + h_basis
        h = np.clip(h, 0, 1)  # Enforce bounds despite constraints
        # Compute correlation with (1 - h)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Perturb and refine solution
    best_c5 = np.inf
    best_h = h_initial.copy()
    for _ in range(3):  # Try different restarts
        # Run optimization from current best or random perturbation
        result = minimize(
            objective, 
            x0=a_initial if _ == 0 else np.random.normal(0, 0.05, m),
            constraints=constraints,
            method='SLSQP',
            bounds=[(-0.5, 0.5)] * m,  # Additional bounds to help convergence
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            a_opt = result.x
            h_basis = np.dot(basis, a_opt)
            h = 0.5 + h_basis
            h = np.clip(h, 0, 1)
            # Adjust to meet exact integral constraint
            h_sum = h.sum()
            if not np.isclose(h_sum, n_points / 2.):
                h = (h / h_sum) * (n_points / 2.)
                h = np.clip(h, 0, 1)
            # Compute final C5
            c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                a_initial = a_opt  # Use improved solution for next restart

    return (best_h.tolist(), best_c5, n_points)
