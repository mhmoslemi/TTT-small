        best_c5 = objective(h_half, n_points, target_sum_half, dx)

    while time.time() - start_time < time_left:
        perturbed = h_half.copy()
        idx = np.random.choice(half_n, size=min(half_n, 8), replace=False)
        perturbed[idx] += np.random.normal(0, 0.1, len(idx))
        perturbed = np.clip(perturbed, 0, 1)
        if perturbed.sum() != 0:
            perturbed = (perturbed / perturbed.sum()) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(lambda x: objective(x, n_points, target_sum_half, dx),
                          x0=perturbed,
                          method='SLSQP',
                          bounds=bounds,
                          constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
        if result.success:
            new_h_half = result.x
            new_h_half = constrain_h(new_h_half, half_n, target_sum_half)
            new_c5 = objective(new_h_half, n_points, target_sum_half, dx)
            if new_c5 < best_c5:
                h_half = new_h_half
                best_c5 = new_c5
    return h_half

def run(seed=42, budget_s=1100, **kwargs):
    start_time_total = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points_initial = len(h_full_initial)
    resolutions = [20, 40, 70]
    h_half_seq = []

    for i, res in enumerate(resolutions):
        if i == 0:
            h_prev = downsample_h(h_full_initial, n_points_initial, res)
        else:
            h_prev = upsample_h(h_half_seq[-1], n_points_prev, res)
            h_prev = create_symmetric_h(res, h_prev)
            h_prev = constrain_h(h_prev, res, res / 2.0)

        n_points_prev = res
        time_budget = budget_s * 0.3 if i == 2 else budget_s * 0.2
        time_left = max(60.0, budget_s * 0.3 - (time.time() - start_time_total))
        if time_left < 15:
            break

        half_n = res // 2
        h_half = optimize_resolution(res, h_prev[:half_n], time_left=time_left, seed=seed)
        h_half_seq.append(h_half)

    final_h_half = upsample_h(h_half_seq[-1], n_points_prev, 70)
    final_h_half = final_h_half[:35]
    h_final = create_symmetric_h(70, final_h_half)
    h_final = constrain_h(h_final, 70, 35.0)
    c5 = objective(final_h_half, 70, 35.0, 2.0 / 70)
    return (h_final.tolist(), c5, 70)
