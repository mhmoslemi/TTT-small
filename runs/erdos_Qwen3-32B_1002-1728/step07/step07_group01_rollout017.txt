    while time.time() - start_time < budget_s - 1e-4:
        iteration += 1

        # Compute current C5 and determine worst shift
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (n_points - 1)

        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        if k >= 0:  # Rightward shift
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    grad[i] += (1.0 - h[j]) * dx
                    grad[j] += -h[i] * dx
        else:  # Leftward shift
            abs_k = -k
            for i in range(abs_k, n_points):
                j = i + k
                grad[i] += (1.0 - h[j]) * dx
                grad[j] += -h[i] * dx

        # Update h with gradient step
        h_candidate = h - alpha * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        h_sum = h_candidate.sum()
        if h_sum != 0:
            h_candidate = h_candidate * (target_sum / h_sum)
            h_candidate = np.clip(h_candidate, 0, 1)

        # Evaluate new solution
        corr_vals = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr_vals) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h_candidate.copy()
        h[:] = h_candidate.copy()

        # Decay learning rate
        alpha *= alpha_decay

        # Apply random perturbation periodically
        if iteration % perturbation_interval == 0:
            pert = np.random.normal(0, perturb_scale, size=n_points)
            h_candidate = h.copy() + pert
            h_candidate = np.clip(h_candidate, 0, 1)
            h_sum = h_candidate.sum()
            if h_sum != 0:
                h_candidate = h_candidate * (target_sum / h_sum)
                h_candidate = np.clip(h_candidate, 0, 1)
            h[:] = h_candidate.copy()
            alpha = alpha_start  # Reset learning rate after perturbation

        # Early stopping based on time
        if time.time() - start_time >= budget_s - 1e-4:
            break

    # Final constraint enforcement
    best_h_sum = best_h.sum()
    if abs(best_h_sum - target_sum) > 1e-6:
        best_h = best_h * (target_sum / best_h_sum)
        best_h = np.clip(best_h, 0, 1)
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_vals) * dx
    return (best_h.tolist(), final_c5, n_points)
