import time
import copy
import bisect

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_area = 1.0
    
    # Parse initial_h_values into block structure
    breaks = [0.0]
    values = []
    current_value = h_initial[0]
    for i in range(1, n_points):
        if h_initial[i] != current_value:
            break_pos = (i) * dx
            breaks.append(break_pos)
            values.append(current_value)
            current_value = h_initial[i]
    breaks.append(2.0)
    values.append(current_value)
    
    def repair(breaks, values):
        dx_blocks = np.diff(breaks)
        area = np.sum(np.array(values) * dx_blocks)
        if area == 0:
            return breaks, values
        scaled_values = np.array(values) / area
        clipped_values = np.clip(scaled_values, 0, 1)
        new_area = np.sum(clipped_values * dx_blocks)
        if new_area == 0:
            return breaks, values
        final_values = clipped_values / new_area
        final_values = np.clip(final_values, 0, 1)
        return breaks, final_values.tolist()
    
    def get_h_array(breaks, values, n=n_points):
        h = np.zeros(n)
        dx_h = 2.0 / n
        for i in range(n):
            x = i * dx_h
            j = bisect.bisect_right(breaks, x) - 1
            if j >= 0 and j < len(values):
                h[i] = values[j]
        return h
    
    def evaluate_candidate(breaks, values):
        h_array = get_h_array(breaks, values, n_points)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx
    
    def perturb(candidate):
        breaks, values = copy.deepcopy(candidate)
        which = np.random.choice([0, 1, 2], p=[0.3, 0.5, 0.2])
        if which == 0:  # Perturb value
            if not values:
                return (breaks, values)
            idx = np.random.randint(len(values))
            values = values.copy()
            values[idx] += np.random.uniform(-0.2, 0.2)
        elif which == 1:  # Perturb break
            if len(breaks) < 3:
                return (breaks, values)
            idx = np.random.randint(1, len(breaks)-1)
            start, end = breaks[idx-1], breaks[idx+1]
            breaks = breaks.copy()
            breaks[idx] = np.random.uniform(start, end)
        elif which == 2:  # Split block
            if len(breaks) < 3:
                return (breaks, values)
            idx = np.random.randint(len(breaks)-1)
            b1, b2 = breaks[idx], breaks[idx+1]
            split_point = np.random.uniform(b1, b2)
            breaks = breaks[:idx+1] + [split_point] + breaks[idx+1:]
        return repair(breaks, values)
    
    beam_width = 10
    beam = [((breaks, values), np.inf)]
    best_c5 = np.inf
    best_h = h_initial
    
    while time.time() - start_time < budget_s:
        new_candidates = []
        for (candidate, old_score) in beam:
            for _ in range(3):
                new_candidate = perturb(candidate)
                if new_candidate[1] is None:
                    continue
                new_candidates.append((new_candidate, np.inf))
        all_candidates = beam + new_candidates
        # Evaluate and select top beam_width
        scored = []
        for (cand, _) in all_candidates:
            c5 = evaluate_candidate(*cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = get_h_array(*cand, n_points)
            scored.append((c5, cand))
        # Sort and keep top
        scored.sort()
        beam = []
        for s, cand in scored[:beam_width]:
            beam.append((cand, s))
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    # Final repair
    best_h = np.clip(best_h, 0, 1)
    total = best_h.sum() * dx
    if abs(total - target_area) > 1e-6:
        best_h = best_h / total * target_area / dx
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_candidate(get_h_from_blocks(breaks, values, n_points))
    return (best_h.tolist(), final_c5, n_points)
