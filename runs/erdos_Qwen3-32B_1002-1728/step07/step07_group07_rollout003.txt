            h_tmp = upscale_h_block(res.x, n_blocks_ga)
            c5_tmp = np.max(correlate(h_tmp, 1.0 - h_tmp, mode='full')) * dx
            if c5_tmp < best_c5:
                best_h = h_tmp
                best_c5 = c5_tmp
    
    # Phase 2: Block-based optimization with n_blocks=10
    n_blocks_ga2 = 10
    if n_points % n_blocks_ga2 == 0 and time.time() - start_time < budget_s * 0.9:
        samples_per_block2 = n_points // n_blocks_ga2
        cons_block2 = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum / samples_per_block2},)
        bounds_block2 = [(0.0, 1.0)] * n_blocks_ga2
        # Create initial guess from phase 1
        h_mid = upscale_h_block(create_blocks(best_h, n_blocks_ga2), n_blocks_ga2)
        x0 = create_blocks(h_mid, n_blocks_ga2)
        # Local optimization from phase 1 result
        def objective_block2(x):
            return objective_block(x, n_blocks_ga2)
        res2 = minimize(objective_block2, x0, bounds=bounds_block2, constraints=cons_block2, method='SLSQP')
        if res2.success:
            h_tmp = upscale_h_block(res2.x, n_blocks_ga2)
            c5_tmp = np.max(correlate(h_tmp, 1.0 - h_tmp, mode='full')) * dx
            if c5_tmp < best_c5:
                best_h = h_tmp
                best_c5 = c5_tmp
    
    # Final repair
    best_h = np.clip(best_h, 0, 1)
    if abs(best_h.sum() - target_sum) > 1e-6:
        best_h = np.clip(best_h / best_h.sum() * target_sum, 0, 1)
    return (best_h.tolist(), np.max(correlate(best_h, 1.0 - best_h, mode='full')) * dx, n_points)
