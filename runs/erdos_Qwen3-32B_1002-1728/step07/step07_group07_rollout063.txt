    n_blocks = len(breaks) - 1

import numpy as np

def sampled_to_blocks(samples):
    n = len(samples)
    breaks = [0.0]
    values = []
    i = 0
    while i < n:
        current_value = samples[i]
        j = i
        while j < n and np.isclose(samples[j], current_value, rtol=1e-8, atol=1e-8):
            j += 1
        length = (2.0 / n) * (j - i)
        breaks.append(breaks[-1] + length)
        values.append(current_value)
        i = j
    return breaks, values

def blocks_to_sampled(breaks, values, n_points):
    sampled = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(n_points):
        x = i * dx
        for b in range(len(breaks) - 1):
            if breaks[b] <= x < breaks[b + 1]:
                sampled[i] = values[b]
                break
    target_sum = n_points / 2.0
    current_sum = sampled.sum()
    if current_sum == 0:
        return np.zeros(n_points)
    scaled = sampled * (target_sum / current_sum)
    clipped = np.clip(scaled, 0, 1)
    return clipped

    return c5, sampled
