import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points

    # Enforce initial feasibility

        h = h * (target_sum / h_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf

    # Gradient descent parameters
    alpha = 0.5
    momentum = 0.9
    velocity = np.zeros(n_points)
    decay_rate = 0.995
    eval_budget = budget_s * 0.9
    start_time = time.time()
    max_iter = 5000  # Hard limit to prevent infinite loops

    while time.time() - start_time < eval_budget and max_iter > 0:
        max_iter -= 1

        # Compute current max overlap
        correlate_vals = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(correlate_vals)
        current_c5 = np.max(correlate_vals)

        if current_c5 < best_c5:

            best_h = h.copy()

        kd = max_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
