        result = minimize(objective, x0=h_start, method=method,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-10})
        
        if result.success:
            h_new = repair_sym_vars(result.x, target_sum_half)
            full_h_test = create_full_h_half(n_points, h_new)
            full_h_test = np.clip(full_h_test, 0, 1)
            c5_test = evaluate_objective(full_h_test, dx)
            if c5_test < best_c5:
                best_h = full_h_test
                best_c5 = c5_test

    # Final repairs and normalization
    best_h = np.clip(best_h, 0, 1)
    current_sum = best_h.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        best_h = (best_h / current_sum) * target_sum_full
        best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
