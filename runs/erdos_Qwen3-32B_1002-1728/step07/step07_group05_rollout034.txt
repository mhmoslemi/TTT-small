import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points

    full_h = cp.Variable(n_points)
    C5 = cp.Variable()

    constraints = []
    # Enforce symmetry in the second half
    for i in range(half_n):
        constraints.append(full_h[i + half_n] == full_h[half_n - 1 - i])
    # Total integral constraint
    constraints.append(cp.sum(full_h) == n_points / 2.0)
    # Bounds
    constraints.append(full_h <= 1)
    constraints.append(full_h >= 0)

    # Generate overlap constraints
    for k in range(-(n_points - 1), n_points):
        valid_a = [a for a in range(n_points) if 0 <= a + k < n_points]
        if not valid_a:
            continue
        terms = [full_h[a] * (1 - full_h[a + k]) for a in valid_a]
        sum_term = cp.sum(terms) * dx
        constraints.append(sum_term <= C5)

    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)

    # Set initial guess from symmetric initial solution
    full_h.value = h_full_initial
    C5.value = 0.380952  # Previous baseline

    # Calculate remaining time budget and set solver time limit
    time_budget = budget_s - (time.time() - start_time)
    solver_options = {
        "MOSEK": {
            "MSK_DPAR_OPTIMIZER_MAX_TIME": max(time_budget * 0.9, 1.0)

    problem.solve(solver=cp.MOSEK, **solver_options)
