    if abs(s - target_sum) > 1e-6:
        h_full_value = (h_full_value * target_sum) / s
        h_full_value = np.clip(h_full_value, 0, 1)

    # Final evaluation
    corr = np.correlate(h_full_value, 1 - h_full_value, mode='full') * dx
    c5_bound = np.max(corr)
    return (h_full_value.tolist(), c5_bound, n_points)
