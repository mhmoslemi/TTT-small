from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2

    def process_vars(vars_):
        vars_clipped = np.clip(vars_, 0, 1)
        if vars_clipped.sum() == 0:
            return vars_clipped.copy()
        scale = target_sum_half / vars_clipped.sum()
        return np.clip(vars_clipped * scale, 0, 1)

    def objective(vars_):
        h_full = create_full_h(process_vars(vars_))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_initial = h_full_initial[:half_n].copy()
    h_initial = process_vars(h_initial)
    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    popsize = 50
    sigma = 0.1
    population = [best_h + np.random.randn(half_n) * sigma for _ in range(popsize)]
    population = [process_vars(x) for x in population]
    evaluated = [(objective(x), x) for x in population]
    evaluated.sort()
    population = [x for c, x in evaluated]
    best_index = np.argmin([c for c, x in evaluated])
    candidate_h = population[best_index]
    candidate_c5 = evaluated[best_index][0]
    if candidate_c5 < best_c5:
        best_h = candidate_h.copy()

    while time.time() - start_time < budget_s * 0.7:

            new_offspring.append(process_vars(offspring))

        x = np.clip(x, 0, 1)
