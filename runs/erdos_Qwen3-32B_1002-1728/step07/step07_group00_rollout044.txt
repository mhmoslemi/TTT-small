import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def create_initial_h(n_points):
    half_n = n_points // 2
    h_half = np.full(half_n, 0.5)
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[-half_n:] = h_half[::-1]
    return h_full

def upsample_h(prev_h, new_n):
    assert new_n % 2 == 0, "new_n must be even"
    prev_half = prev_h[: (prev_h.size // 2)]
    x_prev = np.linspace(0, 1, len(prev_half))
    x_new = np.linspace(0, 1, new_n // 2)
    new_half = np.interp(x_new, x_prev, prev_half)
    h_full = np.zeros(new_n)
    h_full[: new_n//2] = new_half
    h_full[new_n//2 : ] = new_half[::-1]
    return h_full

def optimize_h(initial_guess, n_points, time_budget):
    start_opt = time.time()
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum / 2.0}

    # Initial guess processing
    h_half_init = initial_guess[:half_n]
    # Repair initial guess
    h_half_init = np.clip(h_half_init, 0, 1)
    h_full_init = create_full_h(h_half_init)
    total = h_full_init.sum()
    if total != 0:
        h_full_init = (h_full_init / total) * target_sum
    h_full_init = np.clip(h_full_init, 0, 1)

    best_c5 = objective(h_half_init)
    best_h_half = h_half_init.copy()

    # Differential evolution
    try:
        de_result = differential_evolution(
            lambda x: objective(x), bounds,
            strategy='best1bin', maxiter=15, popsize=20, tol=1e-6,
            mutation=(0.5, 1.0), recombination=0.9, seed=42
        )
        if de_result.success:
            de_h_half = np.clip(de_result.x, 0, 1)
            de_h_half = (de_h_half / de_h_half.sum() + 1e-9) * target_sum / 2.0
            de_h_half = np.clip(de_h_half, 0, 1)
            de_c5 = objective(de_h_half)
            if de_c5 < best_c5:
                best_h_half = de_h_half
                best_c5 = de_c5
    except Exception:
        pass

    # Local optimization with adaptive budget
    while time.time() - start_opt < time_budget - 5:
        frac_remaining = (time_budget - (time.time() - start_opt)) / time_budget
        perturb_indices = np.random.choice(half_n, size=int(0.15 * half_n), replace=False)
        perturbed = best_h_half.copy()
        perturbed[perturb_indices] += np.random.normal(0, 0.3 * frac_remaining**0.5, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        if perturbed.sum() != 0:
            perturbed = (perturbed / perturbed.sum()) * target_sum / 2.0
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result = minimize(
                objective, x0=perturbed, method='SLSQP', bounds=bounds,

                    if new_c5 < best_c5:

                        best_c5 = new_c5
        except Exception:
            pass

    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()

        h_full = (h_full / h_sum) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return h_full, best_c5
