    h_full = np.clip(h_full, 0, 1)
    h_full = h_full * (norms / np.clip(h_full, 0, 1).sum())
    
    grids = [14, 28, 56, 70]
    for g in grids:
        if g < n_points:
            h_full = interpolate_and_scale(h_full, g)
        if time.time() - start_time > 0.9 * budget_s:
            break
        dx = 2.0 / len(h_full)
        h_full = optimize_grid(h_full, dx)
        if time.time() - start_time > budget_s:
            break
    
    h_full = np.clip(h_full, 0, 1)
    final_norm = len(h_full) / 2.0
    h_full = h_full * (final_norm / np.clip(h_full, 0, 1).sum())
    h_full = np.clip(h_full, 0, 1)
    corr = np.correlate(h_full, 1-h_full, mode='full')
    c5 = np.max(corr) * (2.0 / len(h_full))
    return (h_full.tolist(), c5, len(h_full))
