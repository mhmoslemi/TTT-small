    start_idx = 0

        if abs(h_full_initial[i] - current_val) > 1e-6:
            break_pos = (i - 1) * dx + dx  # End of the previous block
            breaks.append(break_pos)
            values.append(current_val)

            start_idx = i
    breaks.append(2.0)
    values.append(current_val)
    breaks = [0.0] + breaks

    def blocks_to_h(breaks_b, values_b):
        h = np.zeros(n_points)
        for i in range(len(breaks_b) - 1):
            b1 = breaks_b[i]
            b2 = breaks_b[i + 1]
            val = values_b[i]
            for idx in range(n_points):
                pos = idx * dx
                if b1 <= pos < b2:
                    h[idx] = val
        return h

    def repair(breaks_b, values_b):
        values_b = np.clip(values_b, 0, 1)
        total_integral = sum((breaks_b[i+1] - breaks_b[i]) * v for i, v in enumerate(values_b))
        if total_integral == 0:
            return breaks_b, values_b
        scale = target_sum / total_integral
