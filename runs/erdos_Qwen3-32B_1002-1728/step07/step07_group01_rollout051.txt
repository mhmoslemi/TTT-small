import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial h
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum != 0:
        h = (h / current_sum) * target_sum
    h = np.clip(h, 0, 1)
    assert np.allclose(h.sum(), target_sum, atol=1e-6)

    best_h = h.copy()
    # Calculate C5 of current best
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Parameters for gradient descent
    alpha = 0.1
    alpha_decay = 0.99
    perturbation_interval = 100
    perturbation_scale = 0.05
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_max_shift = max_index - (n_points - 1)

        # Compute gradient for k_max_shift
        m_array = np.arange(n_points)

        valid_forward = (m_array + k_max_shift) >= 0
        valid_forward &= (m_array + k_max_shift) < n_points
        term1 = np.zeros(n_points)
        term1[valid_forward] = (1.0 - h[m_array[valid_forward] + k_max_shift]) * dx

        valid_backward = (m_array - k_max_shift) >= 0
        valid_backward &= (m_array - k_max_shift) < n_points
        term2 = np.zeros(n_points)
        term2[valid_backward] = h[m_array[valid_backward] - k_max_shift] * dx

        gradient = term1 - term2

        # Take a step in negative gradient direction
        h -= alpha * gradient

        # Clip and normalize
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = h_clipped.sum()
        if current_sum > 1e-10:
            h = (h_clipped / current_sum) * target_sum
        else:
            h.fill(0.0)
        h = np.clip(h, 0.0, 1.0)

        # Evaluate current C5
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_current) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Perturbation
        if iteration % perturbation_interval == 0:
            perturbed = h.copy()
            perturbed += np.random.normal(0, perturbation_scale, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = perturbed.sum()
            if perturbed_sum > 1e-10:
                perturbed = (perturbed / perturbed_sum) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
            # Do a few gradient steps on the perturbed h
            for _ in range(5):
                corr_p = np.correlate(perturbed, 1 - perturbed, mode='full')
                max_index_p = np.argmax(corr_p)
                k_p = max_index_p - (n_points - 1)

                m_arr = np.arange(n_points)
                valid_f_p = (m_arr + k_p) >=0
                valid_f_p &= (m_arr +k_p) < n_points
                term1_p = np.zeros(n_points)
                term1_p[valid_f_p] = (1.0 - perturbed[m_arr[valid_f_p] +k_p]) * dx
                valid_b_p = (m_arr -k_p) >=0
                valid_b_p &= (m_arr -k_p) < n_points
                term2_p = np.zeros(n_points)
                term2_p[valid_b_p] = perturbed[m_arr[valid_b_p] - k_p] * dx
                grad_p = term1_p - term2_p
                perturbed -= alpha * 0.1 * grad_p
                perturbed = np.clip(perturbed, 0, 1)
                ps = perturbed.sum()
                if ps > 1e-10:
                    perturbed = (perturbed / ps) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
            corr_p = np.correlate(perturbed, 1 - perturbed, mode='full')
            new_c5 = np.max(corr_p) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = perturbed.copy()
            h = perturbed.copy()

        # Decay alpha
        if iteration % 50 == 0:
            alpha *= alpha_decay

    return (best_h.tolist(), best_c5, n_points)
