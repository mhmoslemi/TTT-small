            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        c5_bound = compute_c5(h_solution)
        return h_solution.tolist(), c5_bound, n_points
    else:
        # Fallback: return best feasible guess or initial
        return h_full_sym_initial.tolist(), h_full_sym_initial @ h_full_sym_initial * dx, n_points
