            current_best = best_h_sym.copy()
            perturbation = np.random.normal(0, sigma, half_n)
            perturbed = current_best + perturbation
            perturbed = repair_h_sym(perturbed)
            h_full = create_full_h(perturbed)
            c5 = objective(h_full)
            if c5 < best_c5:
                best_h_sym = perturbed.copy()
                best_c5 = c5

    while budget_s - (time.time() - start_time) > 20:
        noise = np.random.normal(0, 0.05, half_n)
        perturbed = repair_h_sym(best_h_sym + noise)
        try:
            result = minimize(
                lambda x: objective(create_full_h(x)),
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if result.success:
                proposed = repair_h_sym(result.x)
                c5 = objective(create_full_h(proposed))
                if c5 < best_c5:
                    best_h_sym = proposed.copy()
                    best_c5 = c5
        except:
            pass

    h_full = create_full_h(best_h_sym)
    full_sum = h_full.sum()
    if abs(full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
