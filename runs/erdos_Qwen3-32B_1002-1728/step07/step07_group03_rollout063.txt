from scipy.optimize import differential_evolution, minimize, Bounds
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 2.0  # Because full sum is n_points/2, and we're using symmetry

    # Symmetric initialization
    h_sym_initial = h_full_initial[:half_n].copy()
    h_sym_initial = np.clip(h_sym_initial, 0, 1)
    h_sym_initial = (h_sym_initial / h_sym_initial.sum()) * target_sum_half if h_sym_initial.sum() != 0 else h_sym_initial
    h_sym_initial = np.clip(h_sym_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = np.flip(sym_vars, axis=0)
        return h

    def objective(sym_vars):
        h_full = create_full_h(np.clip(sym_vars, 0, 1))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(np.zeros(half_n), np.ones(half_n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

    best_c5 = objective(h_sym_initial)
    final_h_half = h_sym_initial.copy()

    # Phase 1: Global optimization with adjusted DE parameters
    np.random.seed(seed)
    maxiter_de = 20
    popsize_de = 40
    de_budget = min(600, budget_s * 0.5)  # Cap to prevent overuse
    start_de = time.time()
    try:
        de_result = differential_evolution(
            objective, bounds=[(0, 1)] * half_n, strategy='best1bin',
            maxiter=maxiter_de, popsize=popsize_de, tol=1e-4,
            mutation=(0.5, 1.0), recombination=0.85, seed=seed, polish=False,
            workers=1, disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            final_h_half = np.clip(de_result.x, 0, 1)
            best_c5 = de_result.fun
    except:
        pass

    # Phase 2: Local optimization with tighter tolerances and COBYLA
    while time.time() - start_time < budget_s - 30:
        perturbation = 0.1 * (1 - (time.time() - start_time) / budget_s)

        if np.sum(perturbed) == 0:
