        perturb_indices = np.random.choice(half_n, size=min(half_n, 20), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        ifpert_sum = perturbed.sum()
        if ifpert_sum != 0:
            perturbed = (perturbed / ifpert_sum) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        perturbed[perturbed < 1e-8] = 0

        # Local optimization with multiple methods
        try:
            result = minimize(
                objective, x0=perturbed, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
            if result.success:
                new_h_half = np.clip(result.x, 0, 1)
                if abs(new_h_half.sum() - target_sum_half) < 1e-6:
                    new_c5 = objective(new_h_half)
                    if new_c5 < best_c5:
                        final_h_half = new_h_half
                        best_c5 = new_c5
        except:
            pass

        try:
            result_cobyla = minimize(
                objective, x0=perturbed, method='COBYLA',
                bounds=bounds,
                options={'maxiter': 300, 'eps': 1e-8, 'disp': False})
            if result_cobyla.success:
                new_h_half = np.clip(result_cobyla.x, 0, 1)
                if abs(new_h_half.sum() - target_sum_half) < 1e-6:
                    new_c5 = objective(new_h_half)
                    if new_c5 < best_c5:
                        final_h_half = new_h_half
                        best_c5 = new_c5
        except:
            pass

        try:
            # Nelder-Mead with constraint-penalized objective
            def local_obj(x):
                x_clipped = np.clip(x, 0, 1)
                sum_x = x_clipped.sum()
                target = target_sum_half
                penalty = 1e6 * max(0, abs(sum_x - target))
                return objective(x_clipped) + penalty
            result_nm = minimize(
                local_obj, x0=perturbed, method='Nelder-Mead',
                bounds=bounds,
                options={'maxiter': 150, 'xatol': 1e-8, 'fatol': 1e-10, 'disp': False})
            if result_nm.success:
                new_h_half = np.clip(result_nm.x, 0, 1)
                if abs(new_h_half.sum() - target_sum_half) < 1e-6:
                    new_c5 = objective(new_h_half)
                    if new_c5 < best_c5:
                        final_h_half = new_h_half
                        best_c5 = new_c5
        except:
            pass
    # Final constraint projection
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
