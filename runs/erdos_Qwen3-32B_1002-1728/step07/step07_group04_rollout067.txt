    constraints = []

    for i in range(half_n):

    for k_index in range(2*n_points - 1):

import numpy as np
import time
import cvxpy as cp
import warnings

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) = n_points / 2

    h_level = np.array([0.0, 0.25, 0.5, 0.75, 1.0])

    h_symmetric = cp.Variable((half_n, 5), boolean=True)
    shift_overlap = cp.Variable(2 * n_points - 1, nonneg=True)

    h_front = h_level @ h_symmetric.T  # shape (half_n,)
    h_full = cp.hstack([h_front, cp.flip(h_front)])

    constraints = [
        cp.sum(h_symmetric[i, :]) == 1 for i in range(half_n)
    ] + [cp.sum(h_full) == target_sum]

    for k_index in range(2 * n_points - 1):
        actual_shift = k_index - (n_points - 1)
        valid_i = []
        for i in range(n_points):
            j = i + actual_shift
            if 0 <= j < n_points:
                valid_i.append(i)
        if valid_i:
            sum_expr = 0.0
            for i in valid_i:
                j = i + actual_shift
                sum_expr += h_full[i] * (1.0 - h_full[j])
            sum_expr *= dx
            constraints.append(shift_overlap[k_index] >= sum_expr)

    objective = cp.Minimize(cp.max(shift_overlap))
    prob = cp.Problem(objective, constraints)

    solver_options = {}
    if cp.GLPK_MI in cp.installed_solvers():
        solver = cp.GLPK_MI
        solver_options['max_seconds'] = int(budget_s * 0.8)
    else:
        solver = cp.ECOS_BB  # Fallback BB solver for MIP
    prob.solve(solver=solver, verbose=False, **solver_options)

    h_full_values = np.array(h_initial)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        best_h_symmetric = h_symmetric.value
        if best_h_symmetric is not None:
            h_front_values = np.sum(best_h_symmetric * h_level[None, :], axis=1)
            h_full_values = np.concatenate([h_front_values, np.flip(h_front_values)])
            h_full_values = np.clip(h_full_values, 0, 1)
        else:
            h_full_values = np.clip(h_full_values, 0, 1)

    if abs(h_full_values.sum() - target_sum) > 1e-6:

    return (h_full_values.tolist(), c5_bound, n_points)
