            frac_remaining = 1.0 - (time.time() - start_time) / budget_s
            noise_scale = 0.1 * (1.0 - frac_remaining ** 1.5)
            perturb_idx = np.random.choice(n_points, size=min(n_points, 5), replace=False)
            h[perturb_idx] = np.clip(h[perturb_idx] + np.random.normal(0, noise_scale, len(perturb_idx)), 0, 1)

    # Final cleanup
    h_sum = best_h.sum()
    if h_sum != 0:
        best_h = (best_h / h_sum) * target_sum
    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
