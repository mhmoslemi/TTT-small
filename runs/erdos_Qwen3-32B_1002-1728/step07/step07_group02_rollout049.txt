    sin_terms = np.zeros((M, n))
    for k in range(M):
        k1 = k + 1
        cos_terms[k] = np.cos(k1 * np.pi * x_grid / 2)
        sin_terms[k] = np.sin(k1 * np.pi * x_grid / 2)
    
    # Define CVXPY variables for Fourier coefficients
    a = cp.Variable(M + 1)  # Cosine coefficients
    b = cp.Variable(M)      # Sine coefficients

    # Construct h(x) using the Fourier series
    h = 0.5 * a[0] + a[1:].T @ cos_terms + b.T @ sin_terms
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum_full
    ]
    
    # Solve convex program with energy minimization objective
    objective = cp.Minimize(cp.sum_squares(a) + cp.sum_squares(b))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=200, verbose=False)
        status = prob.status
    except:
        status = "failed"
    
    # Get feasible solution or use initial
    if status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        a_vals = a.value
        b_vals = b.value
        h_feasible = (0.5 * a_vals[0] + 
                     a_vals[1:].dot(cos_terms) + 
                     b_vals.dot(sin_terms))
        h_feasible = np.clip(h_feasible, 0, 1)
    else:
        h_feasible = h_full.copy()
        h_feasible = np.clip(h_feasible, 0, 1)
    
    # Create symmetric initialization
    h_sym_init = np.empty(n)
    h_sym_init[:half_n] = h_feasible[:half_n]
    h_sym_init[-half_n:] = h_feasible[:half_n][::-1]  # Enforce symmetry
    h_sym_init = h_sym_init / h_sym_init.sum() * target_sum_full
    h_sym_init = np.clip(h_sym_init, 0, 1)
    
    def create_full_h(sym_vars):
        h = np.zeros(n)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h
    
    def objective_func(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx
    
    # Optimization bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    
    # Differential evolution phase
    final_h_half = h_sym_init[:half_n].copy()
    best_c5 = objective_func(final_h_half)
    
    if time.time() - start_time < budget_s:
        try:
            result = differential_evolution(
                objective_func, bounds, strategy='best1bin',
                maxiter=50, popsize=30, mutation=(0.5, 1.5),
                recombination=0.9, seed=seed, disp=False
            )
            if result.success and result.fun < best_c5:
                new_h_half = result.x.clip(0, 1)
                new_h_half = new_h_half / new_h_half.sum() * target_sum_half
                final_h_half = new_h_half
                best_c5 = result.fun
        except:
            pass
    
    # Local refinement phase
    while time.time() - start_time < budget_s - 10:
        # Create perturbed guess
        perturbed = final_h_half.copy()
        perturb_indices = np.random.choice(half_n, min(half_n, 10), replace=False)
        perturbed[perturb_indices] += np.random.randn(len(perturb_indices)) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) > 1e-8:
            perturbed = perturbed / np.sum(perturbed) * target_sum_half
        
        try:
            result = minimize(
                objective_func, perturbed, method='L-BFGS-B',
                bounds=bounds, options={'ftol': 1e-8, 'maxiter': 100}
            )
            if result.success:
                new_h_half = result.x.clip(0, 1)
                if abs(np.sum(new_h_half) - target_sum_half) < 1e-6:
                    c5_val = result.fun
                    if c5_val < best_c5:
                        final_h_half = new_h_half
                        best_c5 = c5_val
        except:
            pass
    
    # Final h construction
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full /= h_full.sum() * dx * 2  # Normalize to integral 1
    h_full = np.clip(h_full, 0, 1)
    
    # Final evaluation
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_full.tolist(), c5_bound, n)
