import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0

    sym_initial = np.clip(sym_initial, 0, 1)
    current_sum = np.sum(sym_initial)
    if current_sum != 0:
        sym_initial = (sym_initial / current_sum) * target_sum_half
    sym_initial = np.clip(sym_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(sym_vars):
        h_full = create_full_h(sym_vars)
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx

    current = sym_initial.copy()
    best = current.copy()
    best_c5 = compute_c5(best)

    np.random.seed(seed)
    temp = 0.1
    alpha = 0.99

    while time.time() - start_time < budget_s - 10:
        perturb_op = np.random.choice(['low', 'mid', 'high'])
        candidate = current.copy()

        if perturb_op == 'low':
            num_perturb = min(3, half_n)
            indices = np.random.choice(half_n, num_perturb, replace=False)
            noise = np.random.normal(0, 0.1, num_perturb)
            candidate[indices] += noise
        elif perturb_op == 'mid':
            start = np.random.randint(0, half_n - 5)
            end = min(start + 5, half_n)
            block_indices = np.arange(start, end)
            noise = np.random.normal(0, 0.2, len(block_indices))
            candidate[block_indices] += noise
        elif perturb_op == 'high':
            noise = np.random.normal(0, 0.05, half_n)
            candidate += noise

        candidate = np.clip(candidate, 0, 1)
