import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    full_to_half = np.zeros(n_points, dtype=int)
    for i in range(n_points):
        full_to_half[i] = i if i < half_n else n_points - 1 - i

    k_values = list(range(1, n_points // 2))
    n_k = len(k_values)

    # Precompute coefficient matrices for each k
    coeffs = []
    for k in k_values:
        coeff_k = np.zeros((half_n, half_n), dtype=int)
        for i in range(n_points):
            j = (i + k) % n_points
            a = full_to_half[i]
            b = full_to_half[j]
            coeff_k[a, b] += 1
        coeffs.append(coeff_k)

    # CVXPY optimization variables
    h_half_var = cp.Variable(half_n, nonneg=True)
    constr = [cp.sum(h_half_var) == target_sum_half, h_half_var <= 1.0]

    # Build objective: sum of squares of overlap terms for each k
    objective_terms = []
    for k_idx in range(n_k):
        coeff_k = coeffs[k_idx]
        expr = cp.sum(cp.multiply(coeff_k, cp.outer(h_half_var, h_half_var)))
        objective_terms.append(expr * expr)
    objective = cp.Minimize(cp.sum(objective_terms))

    problem = cp.Problem(objective, constr)
    problem.solve(solver=cp.SCS, verbose=False, feastol=1e-6, max_iter=1000)

    if h_half_var.value is None:
        # Fallback to initial solution if optimization fails
        h_half = h_initial[:half_n].copy()
        h_half = h_half / np.sum(h_half) * target_sum_half
        h_half = np.clip(h_half, 0, 1)
    else:
        h_half = h_half_var.value.copy()
        h_half = np.clip(h_half, 0, 1)
        s = np.sum(h_half)
        if abs(s - target_sum_half) > 1e-8:
            h_half = h_half / s * target_sum_half
            h_half = np.clip(h_half, 0, 1)

    # Construct full h and compute C5
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[half_n:] = h_half[::-1]
    h_full = np.clip(h_full, 0, 1)

    # Enforce integral constraint
    current_sum = np.sum(h_full)
    scale = (n_points / 2) / current_sum
    h_full *= scale
    h_full = np.clip(h_full, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h_full)

    return (h_full.tolist(), best_c5, n_points)
