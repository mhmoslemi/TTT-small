def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def find_worst_k_profile(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n - 1)
        return k, evaluate_c5(h)

    def compute_gradient(h, k):
        grad = np.zeros(n)
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        return grad

    h = repair_h(h_initial)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    alpha = 0.05
    alpha_decay = 0.997
    patience = 10
    wait = 0

    while time.time() - start_time < budget_s:
        k, current_max = find_worst_k_profile(h)
        grad = compute_gradient(h, k)
        new_h = h - alpha * grad
        new_h = repair_h(new_h)
        new_c5 = evaluate_c5(new_h)

        if new_c5 < best_c5:
            h = new_h.copy()
            best_h = h.copy()
            best_c5 = new_c5
            alpha *= 1.1
            wait = 0
        else:
            h = new_h.copy()
            wait += 1
            alpha *= 0.95 if wait % 3 == 0 else 1.0
        alpha = max(alpha, 0.0005)
        alpha *= alpha_decay

        if wait > patience:
            alpha *= 0.8
            wait = 0

    return (best_h.tolist(), best_c5, n)
