import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    h = repair_h(h_initial)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    # Generate structured candidates
    candidates = [best_h.copy()]
    if n % 10 == 0:
        block_size = n // 10
        for pattern in [[1]*5 + [0]*5, [1]*4 + [0]*6, [1]*3 + [0]*7]:
            h_candidate = np.tile(pattern, n // (block_size*len(pattern)))
            h_candidate = np.tile(h_candidate, block_size)
            h_candidate = repair_h(h_candidate)
            candidates.append(h_candidate)
    for cand in candidates:
        c5 = evaluate_c5(cand, dx)
        if c5 < best_c5:
            best_h = cand.copy()
            best_c5 = c5

    # Simulated Annealing
    current_h = best_h.copy()
    temp = 0.12
    cooling = 0.995
    best_c5_update = best_c5
    best_h_update = current_h.copy()

    while time.time() - start_time < budget_s * 0.8:
        for _ in range(5 if temp > 0.01 else 20):
            if time.time() - start_time >= budget_s:

            perturbation = np.zeros(n)
