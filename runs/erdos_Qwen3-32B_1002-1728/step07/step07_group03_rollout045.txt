import numpy as np
from scipy.optimize import minimize
import time

def create_full_h(sym_h, n_points):
    h = np.zeros(n_points)
    half_n = len(sym_h)
    h[:half_n] = sym_h
    h[-half_n:] = sym_h[::-1]
    return h

def objective_sym(x, target_sum_half, n_points, dx):
    x_clipped = np.clip(x, 0, 1)
    x_sum = np.sum(x_clipped)
    if x_sum == 0:
        return np.inf
    x_scaled = (x_clipped / x_sum) * target_sum_half
    h_full = create_full_h(x_scaled, n_points)
    corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr_vals) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    
    # Ensure initial solution is feasible
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_full_sum = h_full_initial.sum()
    if h_full_sum != 0:
        h_full_initial = (h_full_initial / h_full_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_sym_initial = h_full_initial[:half_n].copy()
    h_sym_sum = h_sym_initial.sum()
    if h_sym_sum != 0:
        h_sym_initial = (h_sym_initial / h_sym_sum) * target_sum_half
    h_sym_initial = np.clip(h_sym_initial, 0, 1)
    
    np.random.seed(seed)
    n_particles = 50
    positions = np.zeros((n_particles, half_n))
    for i in range(n_particles):
        perturbation = np.random.normal(0.0, 0.1, half_n)
        positions[i] = np.clip(h_sym_initial + perturbation, 0, 1)
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, half_n))
    pbest_positions = positions.copy()
    pbest_values = np.full(n_particles, np.inf)
    gbest_value = np.inf
    gbest_position = h_sym_initial.copy()
    
    max_iter = 200
    c1 = 2.0
    c2 = 2.0
    v_max = 0.25
    v_min = -0.25
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        inertia = 0.8 - (0.4 / max_iter) * iter
        
        for i in range(n_particles):
            current_pos = positions[i]
            sum_current = np.sum(current_pos)
            if sum_current == 0:
                current_c5 = np.inf
            else:
                current_clipped = np.clip(current_pos, 0, 1)
                current_scaled = (current_clipped / sum_current) * target_sum_half
                h_full = create_full_h(current_scaled, n_points)
                corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
                current_c5 = np.max(corr_vals) * dx
            
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = current_pos.copy()
            
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest_position[:] = current_pos.copy()
        
        for i in range(n_particles):
            r1 = np.random.rand(half_n)
            r2 = np.random.rand(half_n)

            final_h_sym = x0.copy()

    final_sum = final_h_sym.sum()
    if final_sum != 0:
        final_h_sym = (final_h_sym / final_sum) * target_sum_half
    final_h_sym = np.clip(final_h_sym, 0, 1)
    h_full = create_full_h(final_h_sym, n_points)

        h_full = np.clip(h_full, 0, 1)
    
    return (h_full.tolist(), best_c5, n_points)
