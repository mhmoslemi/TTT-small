    best_h_half = h_full_initial[:half_n].copy()

        prob.solve(solver="ECOS", max_itrs=150, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_half_opt = h_half.value
            if h_half_opt is not None:
                h_half_opt = np.clip(h_half_opt, 0, 1)
                new_sum = h_half_opt.sum()
                scale = target_sum_half / new_sum if abs(new_sum) > 1e-9 else 1.0
                h_half_opt *= scale
                h_half_opt = np.clip(h_half_opt, 0, 1)
                full_opt = create_full_h(h_half_opt)
                corr = np.correlate(full_opt, 1 - full_opt, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h_half = h_half_opt.copy()
                    best_c5 = c5
    except Exception as e:
        pass

    # Final validation and repair
    full_h_best = create_full_h(best_h_half)
    full_h_best = np.clip(full_h_best, 0, 1)
    assert abs(np.sum(full_h_best) - n_points/2) < 1e-8
    return (full_h_best.tolist(), best_c5, n_points)
