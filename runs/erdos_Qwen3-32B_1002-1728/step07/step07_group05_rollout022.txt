import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = 1/2 * integral

    def repair_half(x, target_sum):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = np.sum(x_clipped)
        if total < 1e-8:
            x_clipped[0] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 30
    pop = [repair_half(h_full_initial[:half_n].copy(), target_sum_half)]
    for _ in range(population_size - 1):
        x = h_full_initial[:half_n].copy()
        x += 0.2 * np.random.randn(half_n)
        x = repair_half(x, target_sum_half)
        pop.append(x)

    best_h_half = pop[0].copy()
    best_c5 = compute_c5(best_h_half)

    time_end = start_time + budget_s * 0.95
    iter_count = 0
    while time.time() < time_end:
        for i in range(population_size):
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a + 0.5 * (b - c)
            mutant = repair_half(mutant, target_sum_half)
            cross_mask = np.random.rand(half_n) < 0.9
            trial = np.where(cross_mask, mutant, pop[i])
            trial = repair_half(trial, target_sum_half)
            if compute_c5(trial) < compute_c5(pop[i]):
                pop[i] = trial.copy()
                if compute_c5(trial) < best_c5:
                    best_h_half = trial.copy()
                    best_c5 = compute_c5(trial)
        iter_count += 1

    def objective(x):
        h = create_full_h(repair_half(x, target_sum_half))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def bounds_fun(x):
        return np.all((0 <= x) & (x <= 1))

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(repair_half(x, target_sum_half)) - target_sum_half}

    try:
        result = minimize(
            objective, x0=best_h_half, method='L-BFGS-B',
            bounds=bounds, constraints=constraints,

        if result.success:

                best_h_half = refined
                best_c5 = refined_c5
    except:
        pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
