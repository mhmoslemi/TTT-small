            x_opt = repair_half(result.x, target_sum_half)
            c5_new = compute_c5(x_opt)
            if c5_new < best_c5:
                best_h_half = x_opt
                best_c5 = c5_new
    except:
        pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    final_corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (h_full.tolist(), final_c5, n_points)
