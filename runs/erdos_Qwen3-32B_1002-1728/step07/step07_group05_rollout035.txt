import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0

    def repair(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Repair initial
    h_best = repair(h_full_initial, target_sum)
    best_c5 = compute_c5(h_best)

    # Genetic algorithm configuration
    population_size = 100
    tournament_size = 3
    np.random.seed(seed)
    population = [repair(h_full_initial + 0.1 * np.random.randn(n_points), target_sum) for _ in range(population_size)]

    ga_time_budget = budget_s * 0.5
    time_end_ga = start_time + ga_time_budget
    while time.time() < time_end_ga:
        new_population = [h_best.copy()]

        # Cycle through population and generate offspring
        for i in range(population_size - 1):
            # Tournament selection
            candidates = np.random.choice(population_size, tournament_size, replace=False)
            parents = [population[j] for j in candidates]
            parent1 = parents[np.argmin([compute_c5(p) for p in parents])]

            candidates = np.random.choice(population_size, tournament_size, replace=False)
            parents = [population[j] for j in candidates]
            parent2 = parents[np.argmin([compute_c5(p) for p in parents])]

            # Blend crossover
            child = (parent1 + parent2) * 0.5
            # Adaptive mutation
            mutation = 0.2 * np.random.randn(n_points) * (np.random.rand(n_points) < 0.1)
            child += mutation
            child = repair(child, target_sum)
            new_population.append(child)

        population = new_population

        # Update best found so far
        current_fitness = [compute_c5(p) for p in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            h_best = population[best_idx].copy()
            best_c5 = current_fitness[best_idx]

    # Local refinement phase
    max_refinement = int(np.ceil(np.sqrt(budget_s)))
    for _ in range(max_refinement):
        try:
            result = minimize(
                compute_c5,
                x0=h_best,
                bounds=[(0., 1.)] * n_points,
                constraints=[{"type": "eq", "fun": lambda x: x.sum() - target_sum}],
                method='L-BFGS-B',
                options={'ftol': 1e-10, 'maxiter': 100}
            )
            if result.success:
                h_new = repair(result.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h_best = h_new.copy()
                    best_c5 = c5_new
        except:
            pass

    return (h_best.tolist(), best_c5, n_points)
