            best_h = np.clip(best_h, 0, 1)
            current_sum = best_h.sum()
            if current_sum != 0:
                best_h = (best_h / current_sum) * target_sum
            best_h = np.clip(best_h, 0, 1)

    # Final constraint enforcement
    h_full = best_h.copy()
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if abs(current_sum - target_sum) > 1e-6:
        h_full = (h_full / current_sum) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
