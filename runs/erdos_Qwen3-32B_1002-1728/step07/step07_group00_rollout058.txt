from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)

    target_sum_half = target_sum_full / 2.0
    
    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Initialize symmetric half
    h_half = h_full_initial[:half_n].copy()
    current_sum = h_half.sum()
    if current_sum != 0:
        h_half = (h_half / current_sum) * target_sum_half
    h_half = np.clip(h_half, 0, 1)
    current_sum = h_half.sum()
    if abs(current_sum - target_sum_half) > 1e-6:
        scale_factor = target_sum_half / current_sum
        h_half = h_half * scale_factor
        h_half = np.clip(h_half, 0, 1)

    # Multi-grid optimization setup
    best_c5 = evaluate_c5(create_full_h(h_half))

        N = len(arr)

        arr_fine[::2] = arr
        arr_fine[1::2] = arr
        return arr_fine

    def downscale(arr):
        N = len(arr)
        return (arr[::2] + arr[1::2]) / 2.0

    def scale_constrain(arr, target_sum, eps=1e-6):
        arr_clipped = np.clip(arr, 0, 1)
        current_sum = arr_clipped.sum()
        if abs(current_sum) < eps:
            return arr
        scale = target_sum / current_sum
        scaled = arr_clipped * scale
        scaled_clipped = np.clip(scaled, 0, 1)
        current_sum = scaled_clipped.sum()
        if abs(current_sum - target_sum) > 1e-6:
            scale = target_sum / current_sum

            final_clipped = np.clip(final, 0, 1)
            return final_clipped
        return scaled_clipped
