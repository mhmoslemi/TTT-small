        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_max = np.max(corr)
        k_idx = np.argmax(corr)
        
        # Convert k_idx to shift in points
        n = n_points
        shift_in_points = k_idx - (n_points - 1)
        
        # Compute valid indices for this shift
        valid_indices = []
        for i in range(n_points):
            j = i + shift_in_points
            if 0 <= j < n_points:
                valid_indices.append(i)
        
        if not valid_indices:
            continue
        
        # Define CVXPY problem for this specific shift
        h_var = cp.Variable(n_points)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # Build objective terms
        terms = []
        for i in valid_indices:
            j = i + shift_in_points
            terms.append(h_var[i] * (1 - h_var[j]) * dx)
        objective = cp.Minimize(cp.sum(terms))
        
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False, max_iter=150)
        
        if problem.status == cp.OPTIMAL:
            new_h = h_var.value.copy()
            new_h = np.clip(new_h, 0, 1)
            new_sum = new_h.sum()
            if new_sum != 0:
                new_h = new_h / new_sum * target_sum
            new_h = np.clip(new_h, 0, 1)
            
            # Check if this improves the solution
            new_corr = np.correlate(new_h, 1 - new_h, mode='full') * dx
            new_max = np.max(new_corr)
            if new_max < current_max:
                h_full = new_h.copy()
            # Try again with reinitialized optimization for possible meta-stability
            problem.solve(solver=cp.CVXOPT, verbose=False, max_iter=150)
        
    # Final validation and correction
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum
        h_full = np.clip(h_full, 0, 1)
    
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    c5_bound = np.max(np.abs(corr))
    
    return (h_full.tolist(), c5_bound, n_points)
