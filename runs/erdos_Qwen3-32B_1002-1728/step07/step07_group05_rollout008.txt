import time

def run(seed=42, budget_s=1100, **kwargs):
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h_half_repaired = repair_half(np.array(h_half), target_sum_half)
        h = create_full_h(h_half_repaired)
        corr = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr) * dx)

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)
    best_c5 = compute_c5(h_half_initial)

    start_time = time.time()

    # Differential Evolution Phase
    de_config = {
        'strategy': 'best1bin',
        'popsize': 20,
        'mutation': (0.5, 1.0),
        'recombination': 0.7,
        'seed': seed,

        'atol': 1e-6,
        'updating': 'deferred',
        'workers': 1,
        'init': 'sobol',
        'polish': False
    }

    de_config['maxiter'] = int(time_to_de / 10)  # Rough estimate for iterations
    de_result = differential_evolution(compute_c5, bounds, **de_config)
    if de_result.success:
        de_x_opt = repair_half(de_result.x, target_sum_half)
