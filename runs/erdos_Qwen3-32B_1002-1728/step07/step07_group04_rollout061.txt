        winner_idx = idx[np.argmin([fit[i] for i in idx])]
        return pop[winner_idx].copy()

    def create_population(initial_half, size=40):
        np.random.seed(seed)
        population = []
        for i in range(size):
            p = initial_half.copy() if i == 0 else np.random.uniform(0.0, 1.0, half_n)
            population.append(repair_half(p))
        return population

    initial_half = h_initial[:half_n].copy()
    population = create_population(initial_half, size=40)
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic Algorithm Phase with adaptive parameters
    pop_size = 40
    max_gen = 300
    mutation_rate = 0.2

    for gen in range(max_gen):
        time_elapsed = time.time() - start_time
        remaining = max(0, int((budget_s - time_elapsed) / 0.8))
        if time_elapsed > budget_s * 0.8 or gen * pop_size >= remaining:
            break

        new_pop = [best_h_half.copy()]
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            child = p1.copy()
            # Crossover
            if np.random.rand() < 0.3:
                cp = np.random.randint(1, half_n)
                child[:cp] = p2[:cp]
            # Mutation with adaptive strength
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    mutate_idx = np.random.choice(half_n, round(half_n * 0.2), replace=False)
                    child[mutate_idx] += np.random.normal(0, 0.1, mutate_idx.size)
                else:
                    mutate_idx = np.random.choice(half_n, 1)
                    child[mutate_idx] += np.random.uniform(-0.2, 0.2)
                child = np.clip(child, 0, 1)
            child = repair_half(child)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_h_half = population[new_best_idx].copy()
            best_c5 = fitness[new_best_idx]

    # Gradient Refinement Phase with tight tolerances
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0.0, 1.0)] * half_n
    result = minimize(compute_c5, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
    if result.success:
        refined = repair_half(result.x)
        c5 = compute_c5(refined)
        if c5 < best_c5:
            best_h_half = refined
            best_c5 = c5

    # Final H and validation
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    if abs(h_full.sum() - n_points / 2) > 1e-6:
        h_full = (h_full / h_full.sum()) * (n_points / 2)
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
