def get_blocks_from_array(h_array, n_points, threshold=1e-9):
    breaks = [0.0]
    values = []
    current_val = h_array[0]
    count = 1
    for i in range(1, n_points):
        if abs(h_array[i] - current_val) > threshold:
            dx_per_point = 2.0 / n_points
            break_pos = breaks[-1] + dx_per_point * count
            breaks.append(break_pos)
            values.append(current_val)
            current_val = h_array[i]
            count = 1

            count += 1
    values.append(current_val)
    breaks.append(2.0)
    return breaks, values
