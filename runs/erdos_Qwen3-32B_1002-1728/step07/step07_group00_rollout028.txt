    best_h_full[-half_n:] = best_h_half[::-1]
    best_h_full = np.clip(best_h_full, 0, 1)
    current_sum = best_h_full.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        best_h_full = (best_h_full / current_sum) * target_sum_full
        best_h_full = np.clip(best_h_full, 0, 1)
    return (best_h_full.tolist(), global_best_score, n_points)
