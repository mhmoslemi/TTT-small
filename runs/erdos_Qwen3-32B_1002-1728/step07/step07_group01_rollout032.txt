import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(h_initial, 0, 1)
    h = (h / h.sum()) * target_sum if h.sum() != 0 else np.zeros_like(h)
    h = np.clip(h, 0, 1)
    if abs(h.sum() - target_sum) > 1e-6:
        h = (h / h.sum()) * target_sum
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    np.random.seed(seed)
    alpha = 0.1
    alpha_decay = 0.999
    perturb_scale = 0.05
    perturb_freq = 50
    iteration = 0

    while time.time() - start_time < budget_s - 1e-3:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_lag = max_index - (n_points - 1)
        k_max = abs(k_lag)

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5 - 1e-8:
            best_h = h.copy()
            best_c5 = current_c5

        grad = np.zeros(n_points)
        for j in range(n_points):
            term1 = (1.0 - h[j + k_max]) * dx if j + k_max < n_points else 0.0
            term2 = -h[j - k_max] * dx if j - k_max >= 0 else 0.0
            grad[j] = term1 + term2

        h -= alpha * grad
        h = np.clip(h, 0, 1)
        if h.sum() != 0:
            h = (h / h.sum()) * target_sum
        h = np.clip(h, 0, 1)

        alpha *= alpha_decay

        if iteration % perturb_freq == 0 and iteration > 0 and np.random.rand() < 0.5:
            indices = np.random.choice(n_points, size=min(n_points, 10), replace=False)
            h[indices] += np.random.normal(0, perturb_scale, size=len(indices))
            h = np.clip(h, 0, 1)
            if h.sum() != 0:
                h = (h / h.sum()) * target_sum
            h = np.clip(h, 0, 1)

        iteration += 1

    h = best_h.copy()
    h = np.clip(h, 0, 1)
    if abs(h.sum() - target_sum) > 1e-6:
        h = (h / h.sum()) * target_sum
        h = np.clip(h, 0, 1)

    return (h.tolist(), best_c5, n_points)
