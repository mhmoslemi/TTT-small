def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import minimize

    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum_full = n_points / 2.0
    dx = 2.0 / n_points

    # Repair initial solution
    h_repaired = np.clip(h_full_initial, 0, 1)
    h_repaired = h_repaired / h_repaired.sum() * target_sum_full
    h_repaired = np.clip(h_repaired, 0, 1)
    best_h = h_repaired.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def downscale_h(h_fine, n_coarse):
        k = n_points // n_coarse
        h_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            start = i * k
            end = start + k
            h_coarse[i] = np.mean(h_fine[start:end])
        return h_coarse

    def upscale_h(h_coarse, k):
        h_fine = np.zeros(len(h_coarse) * k)
        for i in range(len(h_coarse)):
            h_fine[i * k:(i + 1) * k] = h_coarse[i]
        return h_fine

    # Coarse grid optimization
    n_coarse = 35  # Half of 70 points
    k_coarse = n_points // n_coarse
    h_coarse = downscale_h(best_h, n_coarse)

    # Coarse objective function
    def objective_coarse(x_flat):
        x = np.clip(x_flat, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        dx_coarse = 2.0 / n_coarse
        return np.max(corr) * dx_coarse

    # Coarse constraints
    bounds_coarse = [(0.0, 1.0)] * len(h_coarse)
    cons_coarse = ({'type': 'eq', 'fun': lambda x: x.sum() - (n_coarse / 2.0)})

    # Check time for this optimization
    while time.time() - start_time < budget_s * 0.5:  # Allocate 50% of budget here
        try:
            result_coarse = minimize(objective_coarse, h_coarse, bounds=bounds_coarse, constraints=cons_coarse, 
                                      method='SLSQP', options={'maxiter': 100, 'ftol': 1e-8})
            if result_coarse.success:
                h_coarse_opt = np.clip(result_coarse.x, 0, 1)
                h_fine = upscale_h(h_coarse_opt, k_coarse)
            else:
                h_fine = upscale_h(h_coarse, k_coarse)
            break  # Only one iteration for coarse
        except:
            h_fine = upscale_h(h_coarse, k_coarse)
            break

    # Fine-scale optimization
    def objective_fine(x_flat):
        x = np.clip(x_flat, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds_fine = [(0.0, 1.0)] * n_points
    cons_fine = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_full})

    while time.time() - start_time < budget_s * 0.8:  # Allocate 80% of budget here
        try:
            result_fine = minimize(objective_fine, h_fine, bounds=bounds_fine, constraints=cons_fine, 
                                    method='SLSQP', options={'maxiter': 200, 'ftol': 1e-8})
            if result_fine.success:
                best_h = np.clip(result_fine.x, 0, 1)
                best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
            break
        except:
            break

    # Final repair
    best_h = np.clip(best_h, 0, 1)
    if abs(best_h.sum() - target_sum_full) > 1e-6:
        best_h = (best_h / best_h.sum()) * target_sum_full
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h.tolist(), best_c5, n_points)
