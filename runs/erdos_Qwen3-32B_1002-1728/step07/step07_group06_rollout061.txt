            h_test = np.clip(h_test, 0, 1)

        current_c5 = evaluate_c5(best_h, dx)
        test_c5 = evaluate_c5(h_test, dx)
        delta = test_c5 - current_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            if test_c5 < best_c5:
                best_h = h_test.copy()
                best_c5 = test_c5
                h = h_test.copy()
        # Cool
        T *= cooling_rate
        elapsed = time.time() - start_time

    # Final repair
    best_h = np.clip(best_h, 0, 1)
    sum_h = best_h.sum()
    if abs(sum_h - target_sum) > 1e-8:
        best_h = best_h * (target_sum / sum_h)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
