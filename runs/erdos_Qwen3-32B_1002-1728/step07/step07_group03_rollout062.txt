    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Binary sparsity initialization with dynamic rounding
    h_upper = h_base[:half_n].copy()
    non_zero_idx = np.argsort(h_upper)[::-1][:18]
    h_bin = np.zeros_like(h_upper)
    h_bin[non_zero_idx] = 1.0

    # Build symmetric scaled initial guess
    h_init = np.zeros(n_points)
    h_init[:half_n] = h_bin
    h_init[-half_n:] = h_bin[::-1]
    h_init = (target_sum / h_init.sum()) * h_init
    h_init = np.clip(h_init, 0, 1)

    # Optimization with trust-region reflective method
    def obj_sym(x):
        x_clipped = np.clip(x, 0, 1)
        h = np.zeros(n_points)
        h[:half_n] = x_clipped
        h[-half_n:] = x_clipped[::-1]
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) + np.sum(x[::-1]) - target_sum

    bounds = [(0, 1)] * half_n
    x0 = h_init[:half_n].copy()

    result = minimize(
        obj_sym,
        x0=x0,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options={'maxiter': 400, 'ftol': 1e-12, 'disp': False, 'finite_diff_rel_step': 1e-6}
    )

    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)

    # Final h construction and constraint enforcement
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_opt
    h_full[-half_n:] = h_opt[::-1]
    h_full = np.clip(h_full, 0, 1)

    # Final scaling
    h_sum = h_full.sum()
    if abs(h_sum - target_sum) > 1e-6:
        h_full = (target_sum / h_sum) * h_full
        h_full = np.clip(h_full, 0, 1)

    c5_bound = obj_sym(h_full[:half_n])

    return (h_full.tolist(), c5_bound, n_points)
