    intervals = extract_intervals(h_full_initial, dx)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def best_c5_from_intervals(interval_list):
        h_samples = intervals_to_samples(interval_list, n_points)
        h_samples_clip = np.clip(h_samples, 0, 1)
        sum_h = np.sum(h_samples_clip)
        h_samples_clip = (h_samples_clip / sum_h) * n_points / 2.0
        h_samples = np.clip(h_samples_clip, 0, 1)
        c5 = objective(h_samples)
        return c5, h_samples

    c5_best, h_best = best_c5_from_intervals(intervals)
    best_intervals = intervals.copy()

    np.random.seed(seed)
    perturbation_scale = 0.1
    while time.time() - start_time < budget_s - 5:
        if not intervals:
            break
        idx = np.random.choice(len(intervals))
        c, w, a = intervals[idx]
        delta = perturbation_scale * (2.0 - w)
        direction = np.random.choice([-1, 1])
        c_new = c + direction * delta * np.random.rand()

        c_new = max(0.0, min(2.0 - w, c_new))
        if is_valid(c_new, w, intervals, idx):
            intervals[idx] = (c_new, w, a)
            current_c5, current_h = best_c5_from_intervals(intervals)
            if current_c5 < c5_best:
                c5_best = current_c5
                h_best = current_h
                best_intervals = intervals.copy()
            else:
                intervals[idx] = (c, w, a)

    final_h = h_best
    return (final_h.tolist(), c5_best, n_points)
