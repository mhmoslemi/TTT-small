        h_sum = h.sum()
        if not np.isclose(h_sum, 0):
            h = h * (target_sum / h_sum)
        h = np.clip(h, 0.0, 1.0)

        # Track best solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h  # in-place copy for efficiency

    # Final constraint enforcement
    h_sum = best_h.sum()
    if not np.isclose(h_sum, target_sum):
        best_h = best_h * (target_sum / h_sum)
        best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
