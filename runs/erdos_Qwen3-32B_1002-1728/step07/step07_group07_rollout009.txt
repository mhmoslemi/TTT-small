def find_blocks(h, tolerance=1e-8):
    # Tolerance to handle floating-point imprecision.
    blocks = []
    n = len(h)
    i = 0
    while i < n:
        j = i
        while j < n and abs(h[j] - h[i]) < tolerance:
            j += 1
        blocks.append({'start': i, 'end': j-1, 'value': h[i]})
        i = j
    return blocks
