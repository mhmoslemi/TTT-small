def run(seed=42, budget_s=1100, **kwargs):

            total = 1.0

        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(h_half):
        return np.concatenate([h_half, h_half[::-1]])

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_half_initial = h_full_initial[:half_n].copy()

        pop.append(x)
