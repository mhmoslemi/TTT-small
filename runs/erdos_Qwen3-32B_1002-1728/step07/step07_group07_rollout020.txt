        bounds = [(0.0, 1.0)] * len(h_coarse)

        try:
            result = differential_evolution(coarse_objective, bounds,
                                            strategy='best1bin',
                                            maxiter=10, popsize=15,
                                            mutation=(0.5, 1.5),
                                            recombination=0.9,
                                            seed=seed, workers=1)
            if result.success:
                h_coarse_opt = np.clip(result.x, 0, 1)
                h_fine = upscale_h(h_coarse_opt, 2)
                h_fine = np.clip(h_fine, 0, 1)

                def objective_fine(x):
                    x_clipped = np.clip(x, 0, 1)
                    corr = np.max(np.correlate(x_clipped, 1 - x_clipped, mode='full')) * dx
                    return corr

                # Add boundary constraint for sum
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_full},
                        {'type': 'ineq', 'fun': lambda x: np.min(x)},
                        {'type': 'ineq', 'fun': lambda x: 1 - np.max(x)})
                bounds_fine = [(0.0, 1.0)] * len(h_fine)
                try:
                    res = minimize(objective_fine, h_fine, method='SLSQP',
                                   bounds=bounds_fine, constraints=cons,
                                   options={'maxiter': 200, 'ftol': 1e-8})
                    if res.success:
                        h_fine_opt = np.clip(res.x, 0, 1)
                        c5 = np.max(np.correlate(h_fine_opt, 1 - h_fine_opt, mode='full')) * dx
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_fine_opt
                except Exception:
                    pass
        except Exception:
            pass

    # Final feasibility repair
    if abs(best_h.sum() - target_sum_full) > 1e-6:
        best_h = np.clip(best_h / best_h.sum() * target_sum_full, 0, 1)
    c5_final = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), c5_final, n_points)
