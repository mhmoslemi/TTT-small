import numpy as np
import time
import copy
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Access initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(70)
        h[35:] = 1.0
        h *= 35.0 / h.sum()
    n_points = len(h)
    dx = 2.0 / n_points
    # Convert initial h to block representation
    blocks = []
    current_val = h[0]
    start = 0.0
    for i in range(1, n_points):
        if h[i] != current_val:
            end = i * dx
            blocks.append((start, end, current_val))
            start = end
            current_val = h[i]
    blocks.append((start, 2.0, current_val))
    # Normalize to integral 1
    integral = sum((e - s) * v for s, e, v in blocks)
    blocks = [(s, e, v / integral) for s, e, v in blocks]
    best_blocks = blocks
    best_c5 = float('inf')
    beam = [blocks]
    beam_size = 5
    
    def perturb(candidate):
        new_candidate = list(candidate)
        idx = np.random.randint(len(new_candidate))
        if np.random.rand() < 0.3:  # Split block
            s, e, v = new_candidate[idx]
            split = s + np.random.rand() * (e - s)
            left = (s, split, v)
            right = (split, e, v)
            new_candidate[idx:idx+1] = [left, right]
        else:
            if np.random.rand() < 0.5:  # Adjust position
                s, e, v = new_candidate[idx]
                if np.random.rand() < 0.5:  # adjust start
                    delta = np.random.uniform(-0.1, 0.1) * (e - s)
                    new_s = max(0.0, s + delta)
                    new_e = e
                else:  # adjust end
                    delta = np.random.uniform(-0.1, 0.1) * (e - s)
                    new_s = s
                    new_e = min(2.0, e + delta)
                new_candidate[idx] = (new_s, new_e, v)
            else:  # Adjust value
                s, e, v = new_candidate[idx]
                delta = np.random.uniform(-0.2, 0.2)
                new_v = max(0.0, min(1.0, v + delta))
                new_candidate[idx] = (s, e, new_v)
        return new_candidate

    def rescale(candidate):
        integral = sum((e - s) * v for s, e, v in candidate)
        return [(s, e, v / integral) for s, e, v in candidate] if integral else candidate

    def sample(blocks, n_points):
        h = np.zeros(n_points)
        x = np.linspace(0, 2.0, n_points, endpoint=False)
        for i in range(n_points):
            xi = x[i]
            for s, e, v in blocks:
                if s <= xi < e:
                    h[i] = v
                    break
        return h

    while time.time() - start_time < budget_s * 0.9:
        new_candidates = []
        for c in beam:
            for _ in range(3):
                cand = perturb(c)
                if cand:
                    cand = rescale(cand)
                    new_candidates.append(cand)
        for cand in new_candidates:
            try:
                h_try = sample(cand, n_points)
                corr = np.correlate(h_try, 1 - h_try, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_blocks = cand
            except:
                pass
        # Update beam
        all_cands = beam + new_candidates
        all_cands = [c for c in all_cands if c]
        all_cands.sort(key=lambda x: np.max(np.correlate(sample(x, n_points), 1 - sample(x, n_points), 'full')) * dx)
        beam = all_cands[:10]
        beam = [b for b in beam if b]  # filter invalid

    h_final = sample(best_blocks, n_points)
    return (h_final.tolist(), best_c5, n_points)
