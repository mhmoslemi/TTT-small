        best_c5 = min(best_c5, evaluate_c5(perturbed_h_list[i], dx))

    # Coarse-scale optimization with SLSQP
    def coarse_optimize(h_coarse):
        def objective(x):
            h = np.clip(x, 0, 1)
            return evaluate_c5(h, dx)
        cons = opt.LinearConstraint(np.ones(n), target_sum, target_sum)
        bounds = opt.Bounds(0, 1)
        result = opt.minimize(objective, h_coarse, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
        return result if result.success else None

    # Grid search over two coarse resolutions
    for coarse_factor in [0.7, 0.5]:
        if time.time() - start_time >= budget_s * 0.6:
            break
        coarse_n = max(int(n * coarse_factor), 5)
        downsampled = np.mean(h.reshape(-1, n // coarse_n)[:, :coarse_n], axis=0) * target_sum / h.mean()
        for trial in range(2):
            result = coarse_optimize(downsampled)
            if result:
                refined = result.x
                refined = repair_h(refined)
                c5 = evaluate_c5(refined, dx)
                if c5 < best_c5:
                    best_h = refined.copy()
                    best_c5 = c5
            downsampled = best_h.copy()

    # Fine-scale optimization with gradient-based refinement
    elapsed = time.time() - start_time
    while elapsed < budget_s * 0.9:
        # Mix and match top candidates
        candidates = [best_h, perturbed_h_list[0], perturbed_h_list[1]]
        for alpha in [0.2, 0.4, 0.6, 0.8]:
            weights = [alpha, (1 - alpha) / 2, (1 - alpha) / 2]
            mix = np.average(np.stack(candidates), axis=0, weights=weights)
            mix = repair_h(mix)
            
            def objective(x):
                return evaluate_c5(np.clip(x, 0, 1), dx)
            
            cons = opt.LinearConstraint(np.ones(n), target_sum, target_sum)
            bounds = opt.Bounds(0, 1)
            result = opt.minimize(
                objective, mix,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            
            if result.success:
                trial_h = result.x
                trial_h = repair_h(trial_h)
                c5 = evaluate_c5(trial_h, dx)
                if c5 < best_c5:
                    best_h = trial_h.copy()
                    best_c5 = c5
        elapsed = time.time() - start_time

    # Final repair and validation
    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
