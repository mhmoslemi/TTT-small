import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_first_half) = n_points / 4 for full h

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s < 1e-8:
            h_half[:] = target_sum_half / half_n
        else:
            h_half = h_half / s * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half.copy()

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Enhanced Population Initialization
    def create_population(initial_half, size=20):
        np.random.seed(seed)
        population = []
        levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
        for i in range(size):
            if i == 0:
                p = initial_half.copy()
            else:
                p = levels[np.random.randint(len(levels), size=half_n)]
            population.append(repair_half(p))
        return population

    population = create_population(h_initial[:half_n].copy(), size=50)
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic Algorithm Parameters
    pop_size = 50
    max_gen = 500
    mutation_rate = 0.3
    levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])

    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.8:
            break
        new_pop = [best_h_half.copy()]
        while len(new_pop) < pop_size:
            p1 = population[np.random.choice(pop_size, p=np.exp(-0.1 * np.array(fitness)) / sum(np.exp(-0.1 * np.array(fitness))))]
            p2 = population[np.random.choice(pop_size, p=np.exp(-0.1 * np.array(fitness)) / sum(np.exp(-0.1 * np.array(fitness))))]
            child = p1.copy()
            # Crossover
            if np.random.rand() < 0.3:
                cp = np.random.randint(1, half_n)
                child[:cp] = p2[:cp]
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(half_n, size=int(half_n * 0.2), replace=False)
                new_levels = levels[np.random.randint(len(levels), size=len(mutate_indices))]
                child[mutate_indices] = new_levels
                child = repair_half(child)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

    # Gradient Refinement
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0.0, 1.0)] * half_n
    result = minimize(compute_c5, x0=best_h_half, method='COBYLA', bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'rhobeg': 0.1, 'tol': 1e-10})
    if result.success:
        refined = repair_half(result.x)
        c5 = compute_c5(refined)
        if c5 < best_c5:
            best_h_half = refined
            best_c5 = c5

        h_full = (h_full / h_full.sum()) * (n_points / 2)
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
