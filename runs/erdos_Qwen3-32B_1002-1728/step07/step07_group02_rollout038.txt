        noise_scale = 0.4 * (1 - (time.time() - start_time) / budget_s)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 1e-8:
            perturbed = (perturbed / perturbed_sum) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        for method in ['L-BFGS-B', 'SLSQP', 'COBYLA', 'Nelder-Mead']:
            try:
                if method in ['SLSQP', 'L-BFGS-B']:
                    result = minimize(
                        objective, x0=perturbed, method=method,
                        bounds=bounds,
                        options={'maxiter': 100, 'ftol': 1e-12, 'disp': False}
                    )
                elif method == 'COBYLA':
                    result = minimize(
                        objective, x0=perturbed, method=method,
                        bounds=bounds,
                        options={'maxiter': 100, 'eps': 1e-8, 'displ': False}
                    )
                elif method == 'Nelder-Mead':
                    result = minimize(
                        objective, x0=perturbed, method=method,
                        bounds=bounds,
                        options={'maxiter': 100, 'xtol': 1e-10, 'ftol': 1e-10}
                    )
                else:
                    continue

                if result.success:
                    new_h_half = np.clip(result.x, 0, 1)
                    new_sum = new_h_half.sum()
                    if abs(new_sum - target_sum_half) < 1e-4:
                        new_c5 = objective(new_h_half)
                        if new_c5 < best_c5:
                            best_h_half = new_h_half
                            best_c5 = new_c5
                            break  # Restart from improved solution
            except:
                pass

    # Final validation and output
    h_full = create_full_h(best_h_half)
    h_clipped = np.clip(h_full, 0, 1)
    final_sum = h_clipped.sum()
    if abs(final_sum - target_sum_full) > 1e-6:
        h_clipped = (h_clipped / final_sum) * target_sum_full
        h_clipped = np.clip(h_clipped, 0, 1)

    return (h_clipped.tolist(), best_c5, n_points)
