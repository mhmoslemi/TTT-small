        bounds = [(0.0, 1.0)] * n_points
        constraints = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
        
        np.random.seed(seed)
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=10, popsize=15, mutation=(0.5, 0.8), recombination=0.7,
            seed=seed, workers=1, polish=False, disp=False
        )
        improved = de_result.x if de_result.success else current_h
        improved = np.clip(improved, 0, 1)
        improved = improved / improved.sum() * target_sum
        
        while time.time() - start_time < budget_s * 0.9:
            perturbed = improved + np.random.normal(0, 0.2, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed / perturbed.sum() * target_sum
            
            result = minimize(
                objective, perturbed, method='L-BFGS-B',
                bounds=bounds, constraints=constraints,
                options={'ftol': 1e-11, 'maxiter': 20}
            )
            if result.success:
                candidate = np.clip(result.x, 0, 1)
                current_c5 = objective(candidate)
                if current_c5 < best_c5:
                    improved = candidate
                    best_c5 = current_c5
            if time.time() - start_time > budget_s:
                break
        
        current_h = improved
    
    h_full = np.clip(current_h, 0, 1)
    h_full_sum = h_full.sum()
    target_sum = len(h_full) / 2
    if abs(h_full_sum - target_sum) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points_initial)
