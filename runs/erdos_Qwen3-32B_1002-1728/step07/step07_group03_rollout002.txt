        if all_c5[best_idx] < best_c5:
            best_c5 = all_c5[best_idx]
            best_h_sym = all_candidates[best_idx]

        # Adaptive sigma
        success_count = sum(1 for c in c5_values if c < get_c5(best_h_sym))
        if success_count / n_offspring > 0.3:
            sigma *= 1.1
        else:
            sigma *= 0.9

    # Local Optimization
    def local_obj(x):
        h = np.clip(x, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = (h / h_sum) * target_sum_half
        h = np.clip(h, 0, 1)
        return get_c5(h)

    bounds = [(0.0, 1.0)] * half_n
    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_half})
    
    try:
        result = minimize(
            local_obj, x0=best_h_sym, 
            method='L-BFGS-B', bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_sum = h_opt.sum()
            if h_sum != 0:
                h_opt = (h_opt / h_sum) * target_sum_half
            h_opt = np.clip(h_opt, 0, 1)
            if get_c5(h_opt) < best_c5:
                best_h_sym = h_opt
                best_c5 = get_c5(best_h_sym)
    except:
        pass

    # Final full h construction
    h_full = create_full_h(best_h_sym)
    h_final = np.clip(h_full, 0, 1)
    h_sum = h_final.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_final = (h_final / h_sum) * target_sum_full
        h_final = np.clip(h_final, 0, 1)

    return (h_final.tolist(), best_c5, n_points)
