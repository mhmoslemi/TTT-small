def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective_sym(sym_vars):
        h = create_full_h(sym_vars)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum_sym(sym_vars):
        return np.sum(sym_vars) - target_sum_half

    bounds_sym = [(0.0, 1.0)] * half_n
    initial_sym = h_full_initial[:half_n].copy()
    best_c5 = objective_sym(initial_sym)
    best_h = create_full_h(initial_sym)

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 20:
        # Perturb initial guess
        perturb = np.random.normal(0, 0.1, half_n)
        x0 = initial_sym + perturb
        x0 = np.clip(x0, 0, 1)
        x0_sum = x0.sum()
        if x0_sum != 0:
            x0 = (x0 / x0_sum) * target_sum_half
        x0 = np.clip(x0, 0, 1)

        # Optimization
        result = minimize(
            objective_sym,
            x0=x0,
            bounds=bounds_sym,
            constraints={'type': 'eq', 'fun': constraint_sum_sym},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            sym_opt = result.x
            h_candidate = create_full_h(sym_opt)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate

        # Update initial for next iteration
        initial_sym = result.x if result.success else initial_sym

    # Final validation and repair
    best_h = np.clip(best_h, 0, 1)
    h_sum = best_h.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        best_h = (best_h / h_sum) * target_sum_full
        best_h = np.clip(best_h, 0, 1)

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(np.abs(corr)) * dx

    return (best_h.tolist(), final_c5, n_points)
