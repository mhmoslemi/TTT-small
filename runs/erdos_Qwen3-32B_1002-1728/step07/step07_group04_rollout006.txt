import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = n_points / 4.0

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s < 1e-8:
            h_half[:] = target_sum_half / half_n
        else:
            h_half = h_half / s * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, fit, k=3):
        idx = np.random.choice(len(pop), k, replace=False)
        winner_idx = idx[np.argmin([fit[i] for i in idx])]
        return pop[winner_idx].copy()

    def create_population(initial_half, size=30):
        np.random.seed(seed)
        population = []
        for i in range(size):
            if i == 0:
                p = initial_half.copy()
            else:
                p = np.random.uniform(0, 1, size=half_n)
                p = repair_half(p)
            population.append(p)
        return population

    initial_half = h_initial[:half_n].copy()
    population = create_population(initial_half, size=30)
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic Algorithm Phase
    pop_size = 30
    max_gen = 600
    mutation_rate = 0.2
    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.8:
            break
        new_pop = [best_h_half.copy()]
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fitness, k=5)
            p2 = tournament_selection(population, fitness, k=5)
            child = p1.copy()
            # Crossover (Uniform)
            if np.random.rand() < 0.3:
                mask = np.random.rand(half_n) < 0.5
                child[mask] = p2[mask]
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate = np.random.choice(half_n, int(half_n * 0.3), replace=False)
                child[mutate] += np.random.normal(0, 0.2, mutate.size)
                child[mutate] = np.clip(child[mutate], 0, 1)
            child = repair_half(child)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

    # Gradient Refinement Phase
    def objective_func(x):
        h_half = x.copy()
        return compute_c5(h_half)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0.0, 1.0)] * half_n
    result = minimize(objective_func, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=cons,

        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
