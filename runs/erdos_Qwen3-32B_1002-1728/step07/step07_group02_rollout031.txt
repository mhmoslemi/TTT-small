    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        scale_factor = target_sum_full / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_full / 2})
    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    # Stage 1: Global search with differential evolution
    np.random.seed(seed)
    popsize = 50  # Increased from 30
    maxiter = 60   # Increased from 50
    de_budget = budget_s * 0.5
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=maxiter, 
            popsize=popsize, mutation=(0.5, 1.5), recombination=0.9,
            seed=seed, workers=1, disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
            best_h_half = (best_h_half / best_h_half.sum()) * target_sum_full / 2
            best_h_half = np.clip(best_h_half, 0, 1)
            final_h_half = best_h_half
            best_c5 = objective(final_h_half)
    except Exception as e:
        pass  # Skip on error

    # Stage 2: Local optimization with various strategies and perturbations
    while time.time() - start_time < budget_s - 20:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.2 * (1 - frac_remaining ** 2)

        # Create perturbed starting guess
        perturbed = final_h_half.copy()
        perturb_indices = np.random.choice(half_n, size=min(half_n, 15), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)
        if np.abs(perturbed.sum()) > 1e-10:
            perturbed = (perturbed / perturbed.sum()) * target_sum_full / 2
        perturbed = np.clip(perturbed, 0, 1)

        # Try multiple local optimization methods
        try:
            result = minimize(
                objective, x0=perturbed, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
            )
            if result.success:
                new_h_half = np.clip(result.x, 0, 1)
                if np.all((new_h_half >= 0) & (new_h_half <= 1)):
                    new_h_half = (new_h_half / new_h_half.sum()) * target_sum_full / 2
                    new_h_half = np.clip(new_h_half, 0, 1)
                    new_c5 = objective(new_h_half)
                    if new_c5 < best_c5:
                        final_h_half = new_h_half
                        best_c5 = new_c5
        except:
            pass

        try:
            result_cobyla = minimize(
                objective, x0=perturbed, method='COBYLA',
                bounds=bounds,
                options={'maxiter': 300, 'rhobeg': 0.5, 'tol': 1e-10, 'disp': False}
            )
            if result_cobyla.success:
                new_h_half = np.clip(result_cobyla.x, 0, 1)
                new_h_half = (new_h_half / new_h_half.sum()) * target_sum_full / 2
                new_h_half = np.clip(new_h_half, 0, 1)
                new_c5 = objective(new_h_half)
                if new_c5 < best_c5:
                    final_h_half = new_h_half
                    best_c5 = new_c5
        except:
            pass

        # Early exit if time is tight
        if time.time() - start_time > budget_s * 0.95:
            break

    # Final validation and adjustment
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
