import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    target_half_sum = target_sum / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        scale = target_sum / current_sum
        h_full_initial *= scale
        h_full_initial = np.clip(h_full_initial, 0, 1)
    if abs(h_full_initial.sum() - target_sum) > 1e-6:
        scale = target_sum / h_full_initial.sum()
        h_full_initial *= scale
        h_full_initial = np.clip(h_full_initial, 0, 1)

    h_half_initial = h_full_initial[:half_n].copy()

    # Initialize particles
    np.random.seed(seed)
    num_particles = 20
    particles = np.zeros((num_particles, half_n))
    velocities = np.zeros((num_particles, half_n))
    for i in range(num_particles):
        if i == 0:
            particles[i] = h_half_initial.copy()
        else:
            perturbation = np.random.normal(0, 0.1, half_n)
            particle = np.clip(h_half_initial + perturbation, 0, 1)
            sum_p = particle.sum()
            if sum_p != 0:
                particle = (particle / sum_p) * target_half_sum
            else:
                particle = np.ones(half_n) * (target_half_sum / half_n)
            particles[i] = np.clip(particle, 0, 1)

    # Initialize personal and global bests
    personal_best = particles.copy()
    personal_best_c5 = np.full(num_particles, np.inf)
    for i in range(num_particles):
        c5 = 0.0
        h = np.zeros(n_points)
        h[:half_n] = particles[i]
        h[-half_n:] = particles[i][::-1]
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        personal_best_c5[i] = c5
    global_best_idx = np.argmin(personal_best_c5)
    global_best = particles[global_best_idx].copy()
    best_c5 = personal_best_c5[global_best_idx]

    # PSO parameters
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.1

    # Main PSO loop
    while time.time() - start_time < budget_s - 5:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best[i] - particles[i]) +
                             social * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            # Update position with constraint handling
            new_pos = particles[i] + velocities[i]
            new_pos = np.clip(new_pos, 0, 1)
            sum_clipped = new_pos.sum()
            if sum_clipped == 0:
                scaled = np.ones_like(new_pos) * (target_half_sum / half_n)
            else:
                scaled = (new_pos / sum_clipped) * target_half_sum
            particles[i] = np.clip(scaled, 0, 1)

            # Evaluate and update personal best
            h = np.zeros(n_points)
            h[:half_n] = particles[i]
            h[-half_n:] = particles[i][::-1]
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = c5
                if c5 < best_c5:
                    global_best = particles[i].copy()
                    best_c5 = c5

        # Early stopping if time is tight
        if time.time() - start_time > budget_s - 1:
            break

    # Final validation
    h_full = np.zeros(n_points)
    h_full[:half_n] = global_best
    h_full[-half_n:] = global_best[::-1]
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
