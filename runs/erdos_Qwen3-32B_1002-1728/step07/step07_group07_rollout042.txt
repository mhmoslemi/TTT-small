import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum_vals = n_points / 2.0

    # Convert array to block structure
    def get_blocks(h_flat):
        blocks = []
        current_val = h_flat[0]
        start_idx = 0
        for i in range(1, len(h_flat)):
            if h_flat[i] != current_val:
                blocks.append( (current_val, start_idx, i-1) )
                current_val = h_flat[i]
                start_idx = i
        blocks.append( (current_val, start_idx, len(h_flat)-1) )
        return blocks

    # Convert block structure to sample array with repair
    def blocks_to_sample(blocks):
        sample = np.zeros(n_points)
        for val, start, end in blocks:
            sample[start:end+1] = val
        sum_vals = np.sum(sample)
        if sum_vals != 0:
            scaling = target_sum_vals / sum_vals
            sample *= scaling
        # Clip to [0, 1]
        sample = np.clip(sample, 0, 1)
        return sample

    # Evaluate C5 score
    def evaluate(sample):
        corr = np.correlate(sample, 1 - sample, mode='full')
        return np.max(corr) * dx

    # Perturbation function
    def perturb_blocks(blocks):
        if not blocks:
            return blocks.copy()
        perturbed = blocks.copy()
        op = np.random.choice(['split', 'modify', 'merge', 'adjust'])
        if op == 'split':
            # Split a block
            idx = np.random.choice(len(blocks))
            val, start, end = perturbed[idx]
            if end - start == 0:
                return perturbed  # Can't split single sample
            split_pos = np.random.randint(start, end)
            new_left = (val, start, split_pos)
            new_right = (val, split_pos + 1, end)
            perturbed = (perturbed[:idx] + [new_left, new_right] + perturbed[idx+1:])
        elif op == 'modify':
            # Modify a block's value
            idx = np.random.choice(len(blocks))
            val, start, end = perturbed[idx]
            delta = np.random.uniform(-0.1, 0.1)
            new_val = val + delta
            perturbed[idx] = (max(0.0, min(1.0, new_val)), start, end)
        elif op == 'merge':
            # Merge adjacent blocks
            if len(blocks) < 2:
                return perturbed
            idx = np.random.choice(len(blocks) - 1)
            val1, start1, end1 = perturbed[idx]
            val2, start2, end2 = perturbed[idx+1]
            if end1 + 1 != start2:
                return perturbed  # Not adjacent
            len1 = end1 - start1 + 1
            len2 = end2 - start2 + 1
            new_val = (val1*len1 + val2*len2) / (len1 + len2)
            new_block = (new_val, start1, end2)
            perturbed = perturbed[:idx] + [new_block] + perturbed[idx+2:]
        elif op == 'adjust':
            # Adjust split between two blocks
            if len(blocks) < 2:
                return perturbed
            idx = np.random.choice(len(blocks) - 1)
            val1, start1, end1 = perturbed[idx]
            val2, start2, end2 = perturbed[idx+1]
            if end1 + 1 != start2:
                return perturbed
            # Move split by ±1
            new_end1 = end1 + np.random.choice([-1, 1])
            new_start2 = new_end1 + 1
            # Ensure new_end1 is within [start1, start2 - 1]
            new_end1 = min(max(start1, new_end1), start2 - 1)
            new_start2 = new_end1 + 1
            perturbed[idx] = (val1, start1, new_end1)
            perturbed[idx+1] = (val2, new_start2, end2)
        return perturbed

    # Initial blocks from h_initial
    initial_blocks = get_blocks(h_initial)
    best_sample = blocks_to_sample(initial_blocks)
    best_c5 = evaluate(best_sample)

    # Beam parameters
    beam_size = 10
    max_candidates = 50
    current_beam = [initial_blocks]
    iteration = 0

    # Main loop
    while time.time() - start_time < budget_s * 0.9:
        new_candidates = []
        for blocks in current_beam:
            for _ in range(3):  # Generate 3 variations per block
                new_blocks = perturb_blocks(blocks)
                new_candidates.append(new_blocks)
        # Evaluate all new candidates
        evaluated = []
        for new_blocks in new_candidates:
            sample = blocks_to_sample(new_blocks)
            c5 = evaluate(sample)
            evaluated.append( (c5, new_blocks, sample) )
            if c5 < best_c5:
                best_c5 = c5
                best_sample = sample
        # Select top candidates
        evaluated.sort()
        current_beam = [ blocks for c5, blocks, sample in evaluated[:beam_size] ]
        iteration +=1
        if iteration % 10 == 0:
            print(f"Iteration {iteration}, best_c5={best_c5}")

    # Final repair
    best_sample = blocks_to_sample(current_beam[0])
    final_c5 = evaluate(best_sample)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_sample.tolist(), best_c5, n_points)
