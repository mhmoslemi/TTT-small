        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_control = control_points.copy()
    
    # Final normalization
    final_control = best_control
    h_full = create_full_h(final_control)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if h_sum != 0:
        h_full = (h_full / h_sum) * target_sum_full
    h_full = np.clip(h_full, 0, 1)
    
    return (h_full.tolist(), best_c5, n_points)
