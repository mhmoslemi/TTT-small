    np.random.seed(seed)

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_best = repair_h(h_initial, target_sum)
    best_c5 = evaluate_c5(h_best, dx)
    
    population_size = 100
    population = [h_best.copy()]
    
    # Generate additional perturbed candidates
    for _ in range(99):
        if len(population) >= population_size:
            break
        perturbed = h_initial.copy()
        perturbed += np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed = repair_h(perturbed, target_sum)
        population.append(perturbed)
    
    # Evaluate initial population and find best
    pop_c5 = [evaluate_c5(h, dx) for h in population]
    best_idx = np.argmin(pop_c5)
    if pop_c5[best_idx] < best_c5:
        h_best = population[best_idx].copy()
        best_c5 = pop_c5[best_idx]
    
    # Genetic Algorithm loop
    ga_budget = budget_s * 0.6
    while time.time() - start_time < ga_budget:
        # Re-evaluate population each generation
        pop_c5 = [evaluate_c5(h, dx) for h in population]
        sorted_indices = np.argsort(pop_c5)
        population = [population[i] for i in sorted_indices]
        pop_c5 = [pop_c5[i] for i in sorted_indices]
        
        # Update best solution if improved
        if pop_c5[0] < best_c5:
            h_best = population[0].copy()
            best_c5 = pop_c5[0]
        
        top_k = max(1, int(0.1 * len(population)))
        parents = population[:top_k]
        new_candidates = parents.copy()
        
        # Generate offspring via crossover and mutation
        while len(new_candidates) < population_size:
            p1, p2 = np.random.choice(len(parents), 2, replace=False)
            parent1, parent2 = parents[p1], parents[p2]
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = repair_h(child, target_sum)
            
            # Mutation: 10% change with ±0.15
            mut_mask = np.random.rand(n_points) < 0.1
            child[mut_mask] += np.random.uniform(-0.15, 0.15, sum(mut_mask))
            child = repair_h(child, target_sum)
            
            new_candidates.append(child)
        
        population = new_candidates
        # Skip reevaluation during GA loop to save time (reevaluate at start of next gen)
    
    # Local optimization on top candidates
    top_candidates = [h for h in population[:5]]
    for h in top_candidates:
        def objective(x):
            x = x.reshape(-1)
            x = repair_h(x, target_sum)
            return evaluate_c5(x, dx)
        
        x0 = h.copy()
        bounds = opt.Bounds(np.zeros(n_points), np.ones(n_points))
        res = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            h_opt = repair_h(res.x, target_sum)
            c5 = evaluate_c5(h_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
                h_best = h_opt
    
    # Final repair and validation
    h_best = repair_h(h_best, target_sum)
    final_c5 = evaluate_c5(h_best, dx)
    return (h_best.tolist(), final_c5, n_points)
