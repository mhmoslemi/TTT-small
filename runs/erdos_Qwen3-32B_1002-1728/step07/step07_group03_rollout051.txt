import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Initialize symmetric control points
    control_initial = h_full_initial[:half_n].reshape(-1)
    control_initial = (control_initial / control_initial.sum()) * target_sum_half
    control_current = np.clip(control_initial, 0, 1)
    best_control = control_current.copy()
    best_c5 = float('inf')

    # Optimization parameters
    learning_rate = 0.02
    momentum = 0.9
    epsilon = 1e-5
    max_epochs = 1000
    prev_update = 0.0

    for epoch in range(max_epochs):
        if time.time() - start_time > budget_s - 5:
            break

        h_left = control_current
        h_right = h_left[::-1]
        h_full = np.concatenate([h_left, h_right])
        h_full = (h_full / h_full.sum()) * target_sum_full
        h_clipped = np.clip(h_full, 0, 1)
        
        # Compute overlaps
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        current_c5 = np.max(corr * dx)
        k_max_idx = np.argmax(corr)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_control = control_current.copy()

        # Compute gradient using finite differences
        n = len(control_current)
        gradient = np.zeros(n)
        
        for i in range(n):
            orig = control_current[i]
            control_current[i] = orig + epsilon
            h_perturbed = control_current
            h_full_p = np.concatenate([h_perturbed, h_perturbed[::-1]])
            h_full_p = (h_full_p / h_full_p.sum()) * target_sum_full
            h_clipped_p = np.clip(h_full_p, 0, 1)
            corr_p = np.correlate(h_clipped_p, 1 - h_clipped_p, mode='full')
            val_p = corr_p[k_max_idx] * dx
            
            control_current[i] = orig - epsilon
            h_full_n = np.concatenate([control_current, control_current[::-1]])
            h_full_n = (h_full_n / h_full_n.sum()) * target_sum_full
            h_clipped_n = np.clip(h_full_n, 0, 1)
            corr_n = np.correlate(h_clipped_n, 1 - h_clipped_n, mode='full')
            val_n = corr_n[k_max_idx] * dx
            
            gradient[i] = (val_p - val_n) / (2 * epsilon)
            control_current[i] = orig  # Restore original

        # Update with momentum
        update = momentum * prev_update - learning_rate * gradient
        control_current += update
        control_current = np.clip(control_current, 0, 1)
        
        # Normalize to maintain integral constraint
        control_current = (control_current / control_current.sum()) * target_sum_half
        control_current = np.clip(control_current, 0, 1)
        prev_update = update

    # Final validation
    h_final_left = best_control
    h_final_right = h_final_left[::-1]
    h_full_final = np.concatenate([h_final_left, h_final_right])

    corr_final = np.correlate(h_full_final, 1 - h_full_final, mode='full')
    final_c5 = np.max(corr_final * dx)
    
    return (h_full_final.tolist(), final_c5, n_points)
