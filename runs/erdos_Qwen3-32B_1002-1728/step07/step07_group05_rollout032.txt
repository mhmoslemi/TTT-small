        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial half repair
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)

    best_h_half = h_half_initial
    best_c5 = compute_c5(h_half_initial)

    # --- Genetic Algorithm Phase ---
    population_size = 30
    np.random.seed(seed)
    population = []
    for _ in range(population_size):
        x = h_half_initial.copy()
        x += 0.1 * np.random.randn(*x.shape)
        x = repair_half(x, target_sum_half)
        population.append(x)

    ga_time_budget = budget_s * 0.6
    time_end_ga = start_time + ga_time_budget
    gen = 0
    while time.time() < time_end_ga:
        gen += 1
        fitness = []
        for indiv in population:
            c5_val = compute_c5(indiv)
            fitness.append(c5_val)

        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        new_population = [population[best_idx]]  # elitism with best individual only

        for _ in range(population_size - 1):  # more offsprings
            # Tournament selection
            cand_ids = np.random.choice(population_size, 3, replace=False)
            parents = [population[i] for i in cand_ids]
            parent1 = parents[np.argmin([compute_c5(p) for p in parents])]
            cand_ids = np.random.choice(population_size, 3, replace=False)
            parents = [population[i] for i in cand_ids]
            parent2 = parents[np.argmin([compute_c5(p) for p in parents])]

            # Blend crossover
            child = (parent1 + parent2) / 2.0
            # Adaptive mutation (fixed scale now)
            mutation_mask = (np.random.rand(*child.shape) < 0.1)
            if mutation_mask.any():
                child[mutation_mask] += 0.1 * np.random.randn(np.sum(mutation_mask))
                child[mutation_mask] = np.clip(child[mutation_mask], 0, 1)
            child = repair_half(child, target_sum_half)
            new_population.append(child)

        population = new_population

        if gen % 20 == 0:
            elapsed = time.time() - start_time
            remaining = max(0, time_end_ga - start_time - elapsed)
            if remaining < 0.0001:
                break

    # --- Local Refinement with increased attempts ---
    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum_half
        else:
            x_scaled = x_clipped.copy()
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    # Try different methods for robustness
    for method in ['L-BFGS-B', 'Nelder-Mead', 'COBYLA']:
        try:
            result = minimize(
                bounded_objective, x0=best_h_half, method=method,
                bounds=bounds, constraints=constraints if method != 'Nelder-Mead' else [],
                options={'maxiter': 200 if method != 'Nelder-Mead' else 100, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
