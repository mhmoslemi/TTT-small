                h_half = np.clip(h_half, 0, 1)
        return h_half

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, fit, k=5):
        idx = np.random.choice(len(pop), k, replace=False)
        winner_idx = idx[np.argmin([fit[i] for i in idx])]
        return pop[winner_idx].copy()

    def create_population(initial_half, size=30):
        np.random.seed(seed)
        population = []
        for i in range(size):
            if i == 0:
                p = initial_half.copy()
            else:
                p = np.random.uniform(low=0.0, high=1.0, size=half_n)
                p = repair_half(p)
            population.append(p)
        return population

    initial_half = h_initial[:half_n].copy()
    population = create_population(initial_half, size=30)
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    pop_size = 30
    max_gen = 400
    mutation_rate = 0.3

    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.85:
            break
        new_pop = [best_h_half.copy()]
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            child = p1.copy()
            # Crossover
            if np.random.rand() < 0.5:
                cp = np.random.randint(1, half_n)
                child[:cp] = p2[:cp]
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate = np.random.choice(half_n, round(half_n * 0.3), replace=False)
                child[mutate] += np.random.normal(0, 0.1, mutate.size)
                child[mutate] = np.clip(child[mutate], 0, 1)
            child = repair_half(child)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

    def objective(x):
        h_half = x.reshape(-1)
        return compute_c5(h_half)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds = [(0.0, 1.0)] * half_n
    result = minimize(
        objective,
        x0=best_h_half,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
    )
    if result.success:
        refined = repair_half(result.x)
        c5 = compute_c5(refined)
        if c5 < best_c5:
            best_h_half = refined
            best_c5 = c5

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    if abs(h_full.sum() - n_points / 2) > 1e-6:
        h_full = (h_full / h_full.sum()) * (n_points / 2)
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
