import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        scale_factor = target_sum / current_sum
        h_scaled = h_clipped * scale_factor
        return np.clip(h_scaled, 0, 1)

    # Convex optimization setup
    h_var = cp.Variable(n)
    C = cp.Variable()

    # Define constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum
    ]

    overlaps = []
    # k=0
    overlap_k0 = cp.sum(h_var * (1 - h_var)) * dx - C
    overlaps.append(overlap_k0)

    # Positive and negative shifts
    for m in range(1, n):
        valid = n - m
        # Positive shift
        overlap_pos = cp.sum(h_var[:valid] * (1 - h_var[m:m + valid])) * dx - C
        overlaps.append(overlap_pos)
        # Negative shift
        overlap_neg = cp.sum(h_var[m:m + valid] * (1 - h_var[:valid])) * dx - C
        overlaps.append(overlap_neg)

    constraints += [o <= 0 for o in overlaps]

    # Objective: minimize C
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, max_iters=1000, verbose=False, abstol=1e-8, reltol=1e-8, feastol=1e-8)

    # Extract solution
    h_opt = repair_h(h_var.value)
    best_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
