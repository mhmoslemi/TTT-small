                    return np.max(corr) * dx

                def get_worst_k(h):
                    corr = np.correlate(h, 1 - h, mode='full')
                    idx = np.argmax(corr)
                    shifted_idx = idx % len(h)
                    return idx - len(h) + 1 if idx >= len(h) else 0  # Adjust index to k

                # Add constraints
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_full})
                bounds_fine = [(0.0, 1.0)] * len(h_fine)

                # Fine-tune iteratively
                h_fine_opt = np.clip(h_fine.copy(), 0, 1)
                current_c5 = evaluate_c5(h_fine_opt)
                sub_iter = 0
                while sub_iter < 3 and time.time() - start_time < budget_s:
                    # Get worst k
                    worst_k = get_worst_k(h_fine_opt)
                    if worst_k < 0 or worst_k >= n_points:
                        break

                    # Define objective for this k
                    def objective_sub(x):
                        x_clipped = np.clip(x, 0, 1)
                        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
                        return corr[np.argmax(corr)] * dx  # Max again

                    result_fine = minimize(
                        objective_sub, h_fine_opt, method='SLSQP',
                        bounds=bounds_fine, constraints=cons,
                        options={'maxiter': 150, 'ftol': 1e-8}
                    )
                    if result_fine.success:
                        new_c5 = evaluate_c5(result_fine.x)
                        if new_c5 < current_c5:
                            h_fine_opt = result_fine.x
                            current_c5 = new_c5
                        else:
                            break
                    else:
                        break
                    sub_iter += 1

                c5 = evaluate_c5(h_fine_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.clip(h_fine_opt, 0, 1)
        except Exception:
            pass

    # Final feasibility repair
    best_h = np.clip(best_h, 0, 1)
    if abs(best_h.sum() - target_sum_full) > 1e-6:
        best_h = (best_h / best_h.sum()) * target_sum_full
    best_h = np.clip(best_h, 0, 1)
    c5_final = evaluate_c5(best_h)
    return (best_h.tolist(), c5_final, n_points)
