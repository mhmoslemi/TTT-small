import numpy as np
import time
import cvxpy as cp
import numpy.random as npr

def run(seed=42, budget_s=1100, **kwargs):
    npr.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    def evaluate_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = repair_h(h_initial)
    best_c5 = evaluate_c5(best_h, dx)
    population = []
    # Initialize population with perturbed versions of h_initial
    for _ in range(20):
        perturbed = h_initial + npr.uniform(-0.1, 0.1, n)
        population.append(repair_h(perturbed))
    population.append(repair_h(h_initial))
    population = population[:25]  # Ensure 25 candidates

    # Genetic algorithm
    while time.time() - start_time < budget_s * 0.6:
        evaluated = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            evaluated.append((h, c5))
        evaluated.sort(key=lambda x: x[1])
        # Update global best
        for h, c5 in evaluated:
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        # Select parents (top 20%)
        parent_count = max(1, len(evaluated) // 5)
        parents = [p for p, _ in evaluated[:parent_count]]
        # Generate new population
        new_pop = parents.copy()
        while len(new_pop) < len(population):
            p1, p2 = npr.choice(parents, 2, replace=False)
            alpha = npr.uniform(0.4, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            child = repair_h(child)
            # Apply mutation

                child[idx] += npr.uniform(-0.1, 0.1, 5)
                child = repair_h(child)
            new_pop.append(child)
        population = new_pop[:len(population)]

    # Convex refinement
    candidates = [best_h] + [p for p, _ in evaluated[:3]]
    for h in candidates:

        constraints = [cp.sum(z) == target_sum, 0 <= z, z <= 1]
        problem = cp.Problem(cp.Minimize(objective), constraints)
        problem.solve(solver=cp.ECOS, abstol=1e-8)
        if problem.status == cp.OPTIMAL:
            new_h = repair_h(z.value.copy())
            c5 = evaluate_c5(new_h, dx)
            if c5 < best_c5:
                best_h = new_h
                best_c5 = c5
