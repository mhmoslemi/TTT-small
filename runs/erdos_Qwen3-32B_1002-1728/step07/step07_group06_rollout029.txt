def compute_h(coefs, x_values, n_fourier):
    h = np.zeros_like(x_values)
    h += 0.5 * np.ones_like(x_values)
    normalization = 1.0 / np.sqrt(n_fourier)
    for k in range(n_fourier):
        freq = (k + 1)
        a_k = coefs[k]
        b_k = coefs[n_fourier + k]
        h += normalization * (a_k * np.cos(freq * np.pi * x_values) + b_k * np.sin(freq * np.pi * x_values))
    h = np.clip(h, 0, 1)
    # Ensure sum is 1 by scaling
    dx = x_values[1] - x_values[0]
    h *= (1.0 / (np.sum(h) * dx))
    return h
