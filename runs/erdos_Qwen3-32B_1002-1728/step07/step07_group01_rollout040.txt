def cluster_intervals(h, dx):
    n = len(h)
    intervals = []
    i = 0
    while i < n:
        if h[i] < 1e-8:
            i += 1
            continue
        start = i
        while i < n and h[i] > 1e-8:
            i += 1
        end = i - 1
        c = start * dx
        sigma = (end - start + 1) * dx
        alpha = np.mean(h[start:end+1])
        intervals.append({'c': c, 'sigma': sigma, 'alpha': alpha})
    return intervals

def compute_overlap(max_k, intervals, dx, n_k=10000):
