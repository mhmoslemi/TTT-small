import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    h_repaired = np.clip(h_initial, 0, 1)
    h_repaired = h_repaired / h_repaired.sum() * target_sum
    h_repaired = np.clip(h_repaired, 0, 1)

    # Precompute DFT matrices
    F_real = np.zeros((n_points, n_points))
    F_imag = np.zeros((n_points, n_points))
    for m in range(n_points):
        for n in range(n_points):
            angle = 2 * np.pi * m * n / n_points
            F_real[m, n] = np.cos(angle)
            F_imag[m, n] = -np.sin(angle)

    constraints = [cp.sum(h_cvx) == target_sum, h_cvx >= 0, h_cvx <= 1]

    H_real = F_real @ h_cvx
    H_imag = F_imag @ h_cvx

    for m in range(n_points):
        constraints.append(H_real[m] ** 2 + H_imag[m] ** 2 <= gamma)

    objective = cp.Minimize(gamma)
    problem = cp.Problem(objective, constraints)

    problem.solve(solver=cp.MOSEK, verbose=False, max_iter=100)

    h_opt = np.clip(h_cvx.value, 0, 1) if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] else h_repaired
    if abs(h_opt.sum() - target_sum) > 1e-6:
