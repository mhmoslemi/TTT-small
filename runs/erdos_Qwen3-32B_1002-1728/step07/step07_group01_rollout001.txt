                return np.max(corr) * dx
            try:
                old_val = h[i]
                h[i] = golden_section_search(objective, 0.0, 1.0)
                # Prevent divergence
                if np.isnan(h[i]) or np.isinf(h[i]):
                    h[i] = old_val
            except:
                pass
        # Enforce sum constraint
        h_sum = h.sum()
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        h = np.clip(h, 0, 1)
        # Update best solution
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Perturbation phase
    if time.time() - start_time < budget_s - 5:
        perturbed = best_h.copy()
        # Add random noise to break local optima
        noise = np.random.normal(0, 0.15, size=n_points)
        perturbed = perturbed + noise
        perturbed = np.clip(perturbed, 0, 1)
        h_sum = perturbed.sum()
        if h_sum != 0:
            perturbed = perturbed * (target_sum / h_sum)
        perturbed = np.clip(perturbed, 0, 1)
        # Re-apply coordinate passes to perturbed solution
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            indices = np.random.permutation(n_points)
            for i in indices:
                def objective(x):
                    h_test = perturbed.copy()
                    h_test[i] = x
                    h_clipped = np.clip(h_test, 0, 1)
                    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
                    return np.max(corr) * dx
                try:
                    old_val = perturbed[i]
                    perturbed[i] = golden_section_search(objective, 0.0, 1.0)
                    if np.isnan(perturbed[i]) or np.isinf(perturbed[i]):
                        perturbed[i] = old_val
                except:
                    pass
            h_sum = perturbed.sum()
            if h_sum != 0:
                perturbed = perturbed * (target_sum / h_sum)
            perturbed = np.clip(perturbed, 0, 1)
            c5 = np.max(np.correlate(perturbed, 1 - perturbed, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = perturbed.copy()
    
    # Final sum check and clipping
    h_sum = best_h.sum()
    if abs(h_sum - target_sum) > 1e-6 and h_sum != 0:
        best_h = best_h * (target_sum / h_sum)
    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
