    h_full_initial = np.clip(h_full_initial, 0, 1)
    initial_h_half = h_full_initial[:half_n]
    
    # Phase 1: Enhanced global search with more population and iterations
    np.random.seed(seed)
    popsize = 40  # Increased population size
    maxiter = 20  # Increased iterations
    de_budget = budget_s * 0.6
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin',  # Alternative strategy
            maxiter=maxiter, popsize=popsize, mutation=(0.5, 1.0),
            recombination=0.9, seed=seed, workers=1, disp=False
        )
        best_h_half = np.clip(de_result.x, 0, 1)
        best_h_half = best_h_half / best_h_half.sum() * target_sum_half
    except:
        best_h_half = initial_h_half

    # Phase 2: Amplified local optimization with diverse restarts
    best_c5 = objective(best_h_half)
    num_restarts = 5  # More restarts for thorough search
    for _ in range(num_restarts): 
        if time.time() - start_time >= budget_s - 10:
            break
        perturbed = best_h_half + np.random.randn(half_n) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / perturbed.sum() * target_sum_half
        try:
            res = minimize(
                objective, x0=perturbed, method='L-BFGS-B',  # Alternative method
                bounds=bounds, constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 200}
            )
            if res.success and res.fun < best_c5:
                best_h_half = np.clip(res.x, 0, 1)
                best_c5 = res.fun
        except:
            pass

    # Final constraint enforcement
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = h_full / h_sum * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    
    return (h_full.tolist(), best_c5, n_points)
