import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum(h_half) = n_points / 4 for full sum(h)=n_points/2

    def repair(h, target):
        h = np.clip(h, 0, 1).copy()
        s = h.sum()
        if s < 1e-8:
            h[:] = target / len(h)
        else:
            h = h / s * target
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 40
    max_gen = 600
    mutation_strength = 0.1
    population = [repair(h_initial[:half_n], target_sum) for _ in range(pop_size)]
    fitness = np.array([compute_c5(np.concatenate([h, h[::-1]])) for h in population])
    best_idx = np.argmin(fitness)
    best_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Adaptive GA
    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.8:
            break
        new_pop = [best_half.copy()]  # Elitism
        while len(new_pop) < pop_size:
            # Tournament selection with 5 participants
            contestants = np.random.choice(population, 5, replace=False)
            parent1 = contestants[np.argmin([compute_c5(np.concatenate([c, c[::-1]])) for c in contestants])].copy()
            parent2 = contestants[np.argmin([compute_c5(np.concatenate([c, c[::-1]])) for c in contestants])].copy()
            
            # Crossover
            child = parent1.copy()
            if np.random.rand() < 0.6:
                cp = np.random.randint(1, half_n)
                child[:cp] = parent2[:cp]
            
            # Polynomial mutation
            for i in range(half_n):
                if np.random.rand() < 0.1:
                    eta = 20  # Distribution index
                    u = np.random.rand()
                    if u < 0.5:
                        delta = (2*u)**(1/(eta+1)) - 1
                    else:
                        delta = 1 - (2*(1-u))**(1/(eta+1))
                    child[i] = max(0, min(1, child[i] + delta))
            
            child = repair(child, target_sum)

        population = new_pop
