            if new_c5 < best_c5:
                best_h_half, best_c5 = new_h, new_c5
        temp *= alpha

    # Particle Swarm Optimization
    np.random.seed(seed)
    pop_size = 25
    population = [repair_half(h + 0.05 * np.random.randn(half_n), target_sum_half) for h in [best_h_half] * pop_size]
    velocities = [0.01 * np.random.randn(half_n) for _ in range(pop_size)]
    pbest_positions = [p.copy() for p in population]
    pbest_c5s = [compute_c5(p) for p in population]
    gbest_idx = np.argmin(pbest_c5s)
    gbest = pbest_positions[gbest_idx].copy()
    gbest_c5 = pbest_c5s[gbest_idx]

    pso_time = budget_s * 0.45
    w, c1, c2 = 0.9, 0.5, 1.5
    while time.time() < start_time + pso_time:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest_positions[i] - population[i]) + c2 * r2 * (gbest - population[i])
            population[i] += velocities[i]
            population[i] = repair_half(population[i], target_sum_half)
            current_c5 = compute_c5(population[i])
            if current_c5 < pbest_c5s[i]:
                pbest_positions[i], pbest_c5s[i] = population[i].copy(), current_c5
                if current_c5 < gbest_c5:
                    gbest, gbest_c5 = population[i].copy(), current_c5
            if current_c5 < best_c5:
                best_h_half, best_c5 = population[i].copy(), current_c5
        w *= 0.995

    # Local refinement
    def obj(x):
        x_repaired = repair_half(x, target_sum_half)
        h = create_full_h(x_repaired)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
    for _ in range(3):
        try:
            res = minimize(obj, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                x_opt = repair_half(res.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half, best_c5 = x_opt, c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
