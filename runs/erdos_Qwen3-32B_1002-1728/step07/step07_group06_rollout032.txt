import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        total = h_clipped.sum()
        if abs(total) > 1e-9:
            h_clipped = h_clipped * (target_sum / total)
        else:
            h_clipped = np.zeros_like(h)
        return np.clip(h_clipped, 0, 1)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [repair_h(h_initial)]
    while len(population) < 100:
        h = h_initial.copy()
        indices = np.random.choice(n, size=int(n * 0.2), replace=False)
        h[indices] += np.random.uniform(-0.1, 0.1, size=len(indices))
        population.append(repair_h(h))

    c5_values = [evaluate_c5(h) for h in population]
    best_h = min(population, key=evaluate_c5)
    best_c5 = evaluate_c5(best_h)

    while time.time() - start_time < budget_s * 0.8 and len(population) < 200:
        sorted_idx = np.argsort(c5_values)
        parents = [population[i] for i in sorted_idx[:10]]
        new_candidates = []
        for _ in range(10):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            alpha = np.random.uniform(0.4, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            child = repair_h(child)
            if np.random.rand() < 0.5:
                indices = np.random.choice(n, size=int(n * 0.1), replace=False)
                child[indices] += np.random.uniform(-0.2, 0.2, size=len(indices))
                child = repair_h(child)
            new_candidates.append(child)
        population.extend(new_candidates)
        c5_values.extend([evaluate_c5(h) for h in new_candidates])
        best_h = min(population, key=evaluate_c5)
        best_c5 = evaluate_c5(best_h)

    while time.time() - start_time < budget_s * 0.95:
        h_opt = opt.minimize(evaluate_c5, x0=best_h, method='SLSQP',
                             bounds=opt.Bounds(0, 1),
                             constraints=({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}),
                             options={'maxiter': 50})
        if h_opt.success:
            candidate = repair_h(h_opt.x)
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_h = candidate
                best_c5 = c5

    best_h = repair_h(best_h)
    return (best_h.tolist(), evaluate_c5(best_h), n)
