    # Each position selects exactly one level
    for i in range(half_n):
        constraints.append(cp.sum(y[i, :]) == 1)

    # Enforce sum(h) = n_points / 2.0
    sum_h_first_half = 0.0
    for i in range(half_n):
        for l in range(5):
            sum_h_first_half += y[i, l] * h_level[l]
    constraints.append(2.0 * sum_h_first_half == n_points / 2.0)

    # Maximize the minimum of overlaps
    max_shifts = 2 * n_points - 1
    for k in range(-n_points + 1, n_points):
        overlap = 0.0
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                # Mirror index for h[i]
                mirror_i = i if i < half_n else n_points - 1 - i
                h_i = sum(y[mirror_i, l] * h_level[l] for l in range(5))
                # Mirror index for h[j]
                mirror_j = j if j < half_n else n_points - 1 - j
                h_j = sum(y[mirror_j, l] * h_level[l] for l in range(5))
                overlap += h_i * (1 - h_j) * dx
        overlap_var = cp.Variable()
        constraints.append(overlap_var >= overlap)
        constraints.append(T >= overlap_var)

    # Objective: minimize T
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)

    # Solve with available time
    elapsed = time.time() - start_time
    timeout = max(0, budget_s - elapsed)
    result = prob.solve(solver="GLPK", verbose=False, time_limit=timeout)

    # Reconstruct h if solution found
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_full = np.zeros(n_points)
        for i in range(n_points):
            mirror_idx = i if i < half_n else n_points - 1 - i
            level_idx = np.argmax(y.value[mirror_idx] if mirror_idx < half_n else y.value[n_points - 1 - mirror_idx])
            h_full[i] = h_level[level_idx]
        # Ensure exact sum constraint
        h_full = (h_full / h_full.sum()) * (n_points / 2.0)
        h_full = np.clip(h_full, 0.0, 1.0)
        h_res = h_full.tolist()
        c5_bound = prob.value
    else:
        # Return initial solution if MILP fails
        h_res = h_initial.tolist()
        c5_bound = np.max(np.correlate(h_initial, 1-h_initial, mode='full')) * dx

    return (h_res, c5_bound, n_points)
