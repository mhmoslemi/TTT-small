    def objective(h_vars):
        h = repair_h(h_vars)
        return evaluate_c5(h, dx)

    bounds = [(0, 1) for _ in range(n)]
    initial_h = repair_h(h_initial)
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Differential evolution with time-limited run
    def de_eval(x):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.7:
            raise StopIteration
        return objective(x)

    try:
        result_de = opt.differential_evolution(de_eval, bounds,
                              strategy='best1bin', maxiter=50,
                              popsize=4, tol=1e-6, mutation=0.8,
                              recombination=0.9,
                              polish=False, workers=1,
                              init=['sobol' if n > 20 else 'latinhypercube'])
        if result_de.success:
            h_de = repair_h(result_de.x)
            c5_de = evaluate_c5(h_de, dx)
            if c5_de < best_c5:
                best_h, best_c5 = h_de, c5_de
    except StopIteration:
        pass

    # Local optimization refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.8:
        res = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                          options={'ftol': 1e-8, 'maxiter': 300})
        if res.success:
            h_refined = repair_h(res.x)
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_h, best_c5 = h_refined, c5_refined

    # Final perturbation-based search
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9:
        for _ in range(3):
            idx = np.random.choice(n, size=5)
            h_perturbed = best_h.copy()
            h_perturbed[idx] = np.clip(h_perturbed[idx] + np.random.uniform(-0.1, 0.1, size=5), 0, 1)
            h_perturbed = repair_h(h_perturbed)
            if evaluate_c5(h_perturbed, dx) < best_c5:
                best_h, best_c5 = h_perturbed, evaluate_c5(h_perturbed, dx)

    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n)
