import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0

    # Repair initial solution
    h_repaired = np.clip(h_full_initial, 0, 1)
    h_repaired = h_repaired / h_repaired.sum() * target_sum_full
    h_repaired = np.clip(h_repaired, 0, 1)

    n = n_points
    m_vals = np.arange(n // 2 + 1)
    k_vals = np.arange(n)
    theta = 2 * np.pi * np.outer(m_vals, k_vals) / n
    E_real = np.cos(theta)
    E_imag = -np.sin(theta)

    h_cpx = cp.Variable(n, nonneg=True)
    gamma = cp.Variable()

    constraints = [
        h_cpx <= 1,
        h_cpx >= 0,
        cp.sum(h_cpx) == target_sum_full
    ]

    for m in range(len(m_vals)):
        real_expr = cp.sum(cp.multiply(E_real[m], h_cpx))
        imag_expr = cp.sum(cp.multiply(E_imag[m], h_cpx))
        constraints.append( (real_expr**2 + imag_expr**2) <= gamma )

    problem = cp.Problem(cp.Minimize(gamma), constraints)

    # Estimate remaining time for solver
    remaining_time = budget_s - (time.time() - start_time)
    try:
        sol_result = problem.solve(
            solver=cp.MOSEK,
            verbose=False,
            time_limit=remaining_time
        )
    except Exception:
        sol_result = None

    h_opt = h_cpx.value if h_cpx.value is not None else h_repaired

    # Finalize and repair
    h_clipped = np.clip(h_opt, 0, 1)
    if abs(h_clipped.sum() - target_sum_full) > 1e-6:
        h_clipped = (h_clipped / h_clipped.sum()) * target_sum_full
    h_final = np.clip(h_clipped, 0, 1)
    c5 = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx

    return (h_final.tolist(), c5, n_points)
