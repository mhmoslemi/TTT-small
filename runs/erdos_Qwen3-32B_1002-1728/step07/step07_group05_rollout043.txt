import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)

    best_h_half = h_half_initial
    best_c5 = compute_c5(h_half_initial)

    # --- Genetic Algorithm Phase ---
    population_size = 30
    np.random.seed(seed)
    population = []
    for _ in range(population_size):
        x = h_half_initial.copy()
        x += 0.15 * np.random.randn(*x.shape)
        x = repair_half(x, target_sum_half)
        population.append(x)

    ga_time_budget = budget_s * 0.6
    time_end_ga = start_time + ga_time_budget
    gen = 0
    while time.time() < time_end_ga:
        gen += 1
        fitness = [compute_c5(indiv) for indiv in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        new_population = population[:3]  # elitism

        for _ in range(population_size - 3):
            # Tournament selection
            cand_ids = np.random.choice(population_size, 4, replace=False)
            parents = [population[i] for i in cand_ids]
            parent1 = parents[np.argmin([compute_c5(p) for p in parents])]
            cand_ids = np.random.choice(population_size, 4, replace=False)
            parents = [population[i] for i in cand_ids]
            parent2 = parents[np.argmin([compute_c5(p) for p in parents])]

            # Blend crossover
            child = (parent1 + parent2) / 2.0
            # Mutation with adaptive noise
            gen_factor = min(gen / 200, 1.0)
            noise_scale = 0.2 * (1.0 - gen_factor)
            mutation_mask = (np.random.rand(*child.shape) < 0.12)
            if mutation_mask.any():
                child[mutation_mask] += noise_scale * np.random.randn(np.sum(mutation_mask))
                child[mutation_mask] = np.clip(child[mutation_mask], 0, 1)
            child = repair_half(child, target_sum_half)
            new_population.append(child)

        population = new_population

    # --- Local Refinement ---
    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum_half
        else:
            x_scaled = x_clipped.copy()
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    for _ in range(5):  # More local optimization attempts
        try:
            result = minimize(

                bounds=bounds, constraints=constraints,

            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
