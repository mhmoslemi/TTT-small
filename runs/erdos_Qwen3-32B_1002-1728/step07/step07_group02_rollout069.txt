    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    
    # Initial symmetric half from smoothed guess
    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    # Adaptive budget allocation across phases
    np.random.seed(seed)
    def budget_timer():
        used = time.time() - start_time
        return used < budget_s

    # Phase 1: Global optimization with adaptive DE
    de_maxiter = min(100, int(budget_s * 0.3 // 10))
    de_popsize = 40
    try:
        de_result = differential_evolution(
            objective, bounds,
            strategy='best1bin', maxiter=de_maxiter,
            popsize=de_popsize, mutation=(0.5, 1.5), recombination=0.9,
            seed=seed, workers=1, tol=1e-5
        )
        if de_result.success and de_result.fun < best_c5:
            new_h = np.clip(de_result.x, 0, 1)
            new_h = new_h / new_h.sum() * target_sum_half
            final_h_half = new_h
            best_c5 = de_result.fun
    except Exception:
        pass  # Continue gracefully

    # Phase 2: Multi-periodic local search
    while budget_timer():
        use_slsqp = np.random.rand() > 0.5
        perturbation_width = 0.15 * (1 - (time.time()-start_time)/budget_s)**0.5
        perturbed = final_h_half.copy()
        
        # Select perturbation points using spectral pattern
        perturb_mask = np.random.rand(half_n) < 0.15
        if np.sum(perturb_mask) == 0:
            perturb_mask[0] = True  # Ensure at least one perturbation
        perturbed[perturb_mask] += np.random.normal(0, perturbation_width, np.sum(perturb_mask))
        perturbed = np.clip(perturbed, 0, 1)
        
        # Normalize
        if np.abs(np.sum(perturbed) - target_sum_half) > 1e-3:
            perturbed = perturbed / perturbed.sum() * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        try:
            if use_slsqp:
                result = minimize(objective, perturbed, bounds=bounds, method='SLSQP', 
                                 constraints=cons, options={'ftol': 1e-8, 'maxiter': 80})
            else:
                result = minimize(objective, perturbed, bounds=bounds, method='COBYLA', 
                                 options={'rhobeg': 0.05, 'tol': 1e-6, 'maxiter': 80})
                
            if result.success and result.fun < best_c5:
                final_h_half = np.clip(result.x, 0, 1)
                best_c5 = result.fun
                # Ensure constraint after rounding errors
                final_h_half = final_h_half / final_h_half.sum() * target_sum_half
        except Exception:
            pass

    # Final validation and constraint correction
    h_full = create_full_h(final_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = h_full.sum()
    if abs(h_full_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_full_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
