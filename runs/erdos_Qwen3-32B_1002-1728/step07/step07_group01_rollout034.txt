import time

def run():
    start_time = time.time()
    h_initial = initial_h_values  # provided by the environment
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial setup
    h = np.clip(h_initial, 0, 1).copy()
    current_sum = h.sum()
    if current_sum != 0:
        h = (h / current_sum) * target_sum
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if abs(current_sum - target_sum) > 1e-6:
        h_clipped = (h_clipped / current_sum) * target_sum
    h = np.clip(h_clipped, 0, 1)
    best_h = h.copy()
    best_c5 = None

    alpha = 0.1  # initial learning rate
    alpha_decay = 0.99
    perturb_scale = 0.05
    perturb_freq = 20  # add perturbation every 20 steps
    step = 0

    while time.time() - start_time < budget_s - 5:
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(corr)
        max_idx = np.argmax(corr)
        k = -(n_points - 1) + max_idx  # compute the corresponding shift in steps

        # Compute gradient for this k
        grad = np.zeros_like(h)
        start_i = max(0, -k)
        end_i = min(n_points, n_points -k)
        for i in range(start_i, end_i):
            j = i + k
            grad[i] += (1 - h[j]) * dx

        # Update h
        h -= alpha * grad
        h = np.clip(h, 0, 1)

        # Enforce sum constraint
        current_sum = h.sum()
        if current_sum != 0:
            scaling_factor = target_sum / current_sum
            h = np.clip(h * scaling_factor, 0, 1)
        else:
            h = np.zeros_like(h)
            h[0] = target_sum

        # Check if current is better
        current_c5 = max_val
        if best_c5 is None or current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Decay alpha
        alpha *= alpha_decay

        # Add perturbation
        if step % perturb_freq == 0:
            perturbation = np.random.normal(0, perturb_scale, size=n_points)

                scaling_factor = target_sum / current_sum
