import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Create symmetric h from first half
    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_sym = np.clip(sym_vars.copy(), 0, 1)
        h_full = create_full_h(h_sym)
        h_sum = h_full.sum()
        if abs(h_sum - target_sum_full) > 1e-6:
            h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    # Repair to maintain constraints
    def repair(sym_vars):
        h_clipped = np.clip(sym_vars, 0, 1)
        if h_clipped.sum() == 0:
            return np.zeros_like(h_clipped)
        return (h_clipped / h_clipped.sum()) * target_sum_half
