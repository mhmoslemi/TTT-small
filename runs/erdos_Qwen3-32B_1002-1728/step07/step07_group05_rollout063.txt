    population_size = 25
    np.random.seed(seed)
    particle_positions = []
    particle_velocities = []
    particle_pbest = []
    pbest_c5 = []
    for _ in range(population_size):
        pos = repair_half(h_half_initial + 0.1 * np.random.randn(half_n), target_sum_half)
        vel = 0.05 * np.random.randn(half_n)
        particle_positions.append(pos)
        particle_velocities.append(vel)
        c5_val = compute_c5(pos)
        particle_pbest.append(pos.copy())
        pbest_c5.append(c5_val)
    gbest_idx = np.argmin(pbest_c5)
    gbest = particle_pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]

    pso_time_budget = budget_s * 0.35
    time_end_pso = start_time + sa_time_budget + pso_time_budget
    while time.time() < time_end_pso:
        for i in range(population_size):
            pos = particle_positions[i]
            vel = particle_velocities[i]
            r1, r2 = np.random.rand(2)
            cognitive = r1 * (particle_pbest[i] - pos)
            social = r2 * (gbest - pos)
            inertia = 0.5
            vel = inertia * vel + 1.0 * cognitive + 2.0 * social
            pos += vel
            pos = repair_half(pos, target_sum_half)
            c5_val = compute_c5(pos)
            if c5_val < pbest_c5[i]:
                particle_pbest[i] = pos.copy()
                pbest_c5[i] = c5_val
                if c5_val < gbest_c5:
                    gbest = pos.copy()
                    gbest_c5 = c5_val
            particle_positions[i] = pos
            particle_velocities[i] = vel

    if gbest_c5 < best_c5:
        best_h_half = gbest.copy()
        best_c5 = gbest_c5

    # --- Local Refinement ---
    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if abs(total) < 1e-8:
            return 1e6  # Penalize invalid solutions
        x_scaled = (x_clipped / total) * target_sum_half
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    for _ in range(3):
        try:
            result = minimize(
                bounded_objective, x0=best_h_half, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
