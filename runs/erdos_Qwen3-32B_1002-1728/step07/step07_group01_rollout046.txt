import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # since sum(h) * dx = 1 => sum(h) = n_points / 2

    # Enforce initial constraints
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial / h_initial.sum() * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Objective function for DE and local optimization
    def objective(x):
        h = np.clip(x, 0, 1)
        # Compute max cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_solution(x):
        h = np.clip(x, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return np.zeros_like(x)
        h = h / h_sum * target_sum
        return np.clip(h, 0, 1)

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    final_h = repair_solution(h_initial)
    best_c5 = objective(final_h)

    # Phase 1: Differential Evolution with larger population and more iterations
    np.random.seed(seed)

    try:
        de_result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=30,
            popsize=30,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            workers=1,
            polish=False
        )
        final_h = repair_solution(de_result.x)
        best_c5 = objective(final_h)
    except Exception:
        pass

    remaining_time = max(0, budget_s - (time.time() - start_time))
    if remaining_time < 5:
        return (final_h.tolist(), best_c5, n_points)

    # Phase 2: Local optimization with multiple restarts and adaptive noise scaling
    while time.time() - start_time < budget_s - 5:
        # Generate perturbed initial guess
        h_pert = final_h.copy()
        perturbation_size = min(20, n_points)  # Perturb 20 elements
        idxs = np.random.choice(n_points, perturbation_size, replace=False)
        noise_scale = 0.15 * np.random.uniform(0.5, 1.0)
        h_pert[idxs] += np.random.normal(0, noise_scale, size=perturbation_size)
        h_pert = repair_solution(h_pert)

        # Local optimization with SLSQP
        try:
            result = minimize(
                objective,
                x0=h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-8}
            )
            if result.success:

                c5_new = objective(h_new)
                if c5_new < best_c5:
                    final_h = h_new
                    best_c5 = c5_new
        except:
            pass

    # Final validation and return
    final_h = repair_solution(final_h)
    return (final_h.tolist(), best_c5, n_points)
