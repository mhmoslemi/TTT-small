            corr = np.correlate(h_full, 1 - h_full, mode='full')
            max_index = np.argmax(corr)

                best_c5 = max_overlap
                best_control = control.copy()

            k_lag = max_index - (n_points - 1)

                        x_minus_k = x - k_lag

                        term1 = 0.0
                        if 0 <= x_plus_k < n_points:

                        term2 = 0.0
                        if 0 <= x_minus_k < n_points:
                            term2 = -h_full[x_minus_k]

                gradient[j] = total_deriv

            gradient_dir = -gradient

            velocity = momentum * velocity + learning_rate * gradient_dir

            control = control + velocity

        corr = np.correlate(h_final, 1 - h_final, mode='full')
        C5 = np.max(corr) * dx

        return (h_final.tolist(), C5, n_points)
