from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h) = n_points / 2, symmetric

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s < 1e-8:
            h_half[:] = target_sum_half / half_n
        else:
            h_half = h_half / s * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(h_half):
        h_half = repair_half(h_half)
        return compute_c5(h_half)

    bounds = [(0, 1)] * half_n
    initial_guess = h_initial[:half_n].copy()
    initial_guess = repair_half(initial_guess)
    best_half = initial_guess.copy()
    best_c5 = compute_c5(best_half)

    # Differential Evolution Phase
