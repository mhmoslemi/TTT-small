import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = 1/2 * integral

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial half repair
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)

    # Gradient descent phase
    learning_rate = 0.05
    time_end = start_time + (budget_s * 0.8)  # Main optimization loop

    while time.time() < time_end:
        h_current = create_full_h(best_h_half)
        corr_vals = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr_vals * dx)
        idx_k = max_idx - (n_points - 1)

        # Compute gradient for current worst k_lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + idx_k
            if 0 <= j < n_points:
                grad[i] += (1.0 - h_current[j]) * dx
                grad[j] -= h_current[i] * dx

        # Aggregate to h_half gradient
        grad_half = np.zeros(half_n)
        for j in range(half_n):
            grad_half[j] = grad[j] + grad[n_points - j - 1]

        # Step update with clipping and repair
        new_h_half = best_h_half - learning_rate * grad_half
        new_h_half = repair_half(new_h_half, target_sum_half)

        # Update if improved
        new_c5 = compute_c5(new_h_half)
        if new_c5 < best_c5:
            best_h_half.fill(0.0)
            best_h_half[:] = new_h_half
            best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
