    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = n_points/4.0

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        total = np.sum(h_half)
        if abs(total - target_sum_half) > 1e-8:
            if total < 1e-8:
                h_half[:] = target_sum_half / half_n
            else:
                h_half = h_half / total * target_sum_half
            h_half = np.clip(h_half, 0, 1)
        return h_half

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, fit, k=4):
        idx = np.random.choice(len(pop), k, replace=False)
        winner_idx = idx[np.argmin([fit[i] for i in idx])]
        return pop[winner_idx].copy()

    def create_population(initial_half, size=30):
        population = []
        for _ in range(size):
            p = np.random.uniform(0, 1, half_n)
            p = repair_half(p)
            population.append(p)
        population[0] = initial_half.copy()
        return population

    initial_half = h_initial[:half_n].copy()
    population = create_population(initial_half, size=30)
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic Algorithm Phase
    pop_size = 30
    max_gen = 250
    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.9:
            break
        new_pop = [best_h_half.copy()]
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            child = p1.copy()
            # Crossover: two-point
            if np.random.rand() < 0.4:
                cp1, cp2 = np.sort(np.random.choice(half_n, 2, replace=False))
                child[cp1:cp2] = p2[cp1:cp2]
            # Mutation: perturb a random subset
            if np.random.rand() < 0.15:
                mutate = np.random.choice(half_n, round(half_n * 0.3), replace=False)
                child[mutate] += np.random.normal(0, 0.2, mutate.size)
                child[mutate] = np.clip(child[mutate], 0, 1)
            child = repair_half(child)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h_half = population[best_idx].copy()
            best_c5 = fitness[best_idx]

    # CVXPY Relaxation Phase
    try:
        x = cp.Variable(half_n)
        h_full = cp.Parameter(n_points)
        h_full[:half_n] = x
        h_full[-half_n:] = cp.flip(x)
        constraints = [0 <= x, x <= 1, cp.sum(x) == target_sum_half]
        overlap_expr = []
        for k in range(-half_n, half_n+1):
            if k == 0:
                continue
            corr = cp.sum(cp.multiply(h_full, 1 - cp.roll(h_full, -k))) * dx
            overlap_expr.append(corr)
        if overlap_expr:
            t = cp.Variable()
            constraints += [overlap_expr[i] <= t for i in range(len(overlap_expr))]
            problem = cp.Problem(cp.Minimize(t), constraints)
            problem.solve(solver=cp.CVXOPT, maxiters=100)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_sol = x.value
                x_sol = repair_half(np.clip(x_sol, 0, 1))
                c5_candidate = compute_c5(x_sol)
                if c5_candidate < best_c5:
                    best_h_half = x_sol
                    best_c5 = c5_candidate
    except Exception as e:
        pass

    # Final H
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    if abs(np.sum(h_full) - n_points / 2) > 1e-6:
        h_full = h_full / np.sum(h_full) * (n_points / 2)
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
