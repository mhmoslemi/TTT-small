def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0  # sum(h) = 1/dx = n_points/2

    # Symmetrize and normalize initial guess
    h_sym_initial = h_full_initial[:half_n].copy()
    sum_sym = np.sum(h_sym_initial)
    if sum_sym != 0:
        h_sym_initial = (h_sym_initial / sum_sym) * (target_sum_full / 2.0)
    else:
        h_sym_initial[:] = 0.5
    h_full = np.concatenate([h_sym_initial, h_sym_initial[::-1]])
    h_full = np.clip(h_full, 0, 1)

    np.random.seed(seed)
    best_c5 = np.float64(0.38085)  # Start with best known upper bound
    best_h = h_full.copy()
    low = 0.3808
    high = 0.3809
    iterations = 0
    remaining_time = budget_s

    while remaining_time > 10 and iterations < 30:
        mid = (low + high) / 2.0
        h = cp.Variable(n_points)
        X = cp.Variable((n_points, n_points), symmetric=True)

        constraints = []
        # Symmetry constraint
        for i in range(half_n):
            constraints.append(h[i] == h[n_points - 1 - i])
        # Integral constraint and bounds
        constraints.append(cp.sum(h) == target_sum_full)
        constraints.append(0 <= h)
        constraints.append(h <= 1)

        # Linearization of product terms
        for i in range(n_points):
            for j in range(i, n_points):
                a, b = h[i], h[j]
                constraints.append(X[i, j] >= a + b - 1)
                constraints.append(X[i, j] <= a)
                constraints.append(X[i, j] <= b)
                constraints.append(X[i, j] >= 0)

        # PSD constraint
        constraints.append(X >> 0)

        # Overlap constraints for each lag
        for k in range(n_points):
            terms = [X[i, (i + k) % n_points] for i in range(n_points)]
            exp = cp.sum(terms) * dx
            constraints.append(exp >= 1 - mid)

        # Form and solve
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            t0 = time.time()
            prob.solve(solver=cp.CVXOPT, verbose=False, maxiters=100)
            solve_time = time.time() - t0
            remaining_time = budget_s - (time.time() - start_time)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = np.clip(h.value, 0, 1)
                corr = np.correlate(h_sol, 1 - h_sol, mode='full')
                actual_c5 = np.max(corr) * dx
                if actual_c5 < best_c5 and abs(actual_c5 - mid) < 1e-5:
                    best_c5 = actual_c5
                    best_h = h_sol
                    high = mid
                else:
                    low = mid
                iterations += 1
            else:
                low = mid
        except Exception:
            low = mid

    return (best_h.tolist(), best_c5, n_points)
