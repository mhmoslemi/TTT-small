            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h[:] = current_h[:]
                best_c5 = new_c5
        # Cool down
        T *= alpha
        iteration += 1

    # Final steps
    h_full = create_full_h(best_h)
    return (h_full.tolist(), best_c5, n_points)

def compute_c5(h_half, n_points):
    h_full = create_full_h(h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    dx = 2.0 / n_points
    return np.max(corr) * dx

def create_full_h(h_half):
    n_half = len(h_half)
    h_full = np.zeros(2 * n_half)
    h_full[:n_half] = h_half
    h_full[-n_half:] = h_half[::-1]
    return h_full

def repair(h, target_sum):
    h = np.clip(h, 0, 1)
    sum_h = h.sum()
    if sum_h == 0:
        return h
    factor = target_sum / sum_h
    h_repaired = h * factor
    return np.clip(h_repaired, 0, 1)

def perturb(h, scale, scales):
    h_new = h.copy()
    h_len = len(h)
    if scale == 'low':
        idx = np.random.randint(h_len)
        noise = np.random.normal(0, scales[scale])
        h_new[idx] += noise
    elif scale == 'mid':
        block_size = np.random.randint(2, min(5, h_len) + 1)
        start = np.random.randint(0, h_len - block_size + 1)
        end = start + block_size
        noise = np.random.normal(0, scales[scale], size=(end - start,))
        h_new[start:end] += noise
    else:  # high
        noise = np.random.normal(0, scales[scale], h_len)
        h_new += noise
    return h_new
