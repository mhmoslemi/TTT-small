                    c5 = np.max(np.correlate(h_fine_opt, 1 - h_fine_opt, mode='full')) * dx
                    if c5 < best_c5:
                        best_h = h_fine_opt
                        best_c5 = c5
                        success_count += 1
        except Exception as e:
            continue

    # Final repair pass
    best_h = np.clip(best_h, 0, 1)
    h_sum = best_h.sum()
    if not np.isclose(h_sum, target_sum, atol=1e-6):
        best_h = (best_h / h_sum) * target_sum
        best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
