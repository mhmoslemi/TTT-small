from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)

    def objective(h_full):
        h_clipped = np.array(h_full)
        current_sum = h_clipped.sum()
        if current_sum > 1e-8:
            h_clipped = (h_clipped / current_sum) * target_sum_full
        h_clipped = np.clip(h_clipped, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    best_c5 = objective(h_full_initial)
    final_h_full = h_full_initial.copy()

    # Phase 1: Global search with DE
    np.random.seed(seed)
    popsize = 15
    maxiter = 20
    de_budget = budget_s * 0.3  # 30% of total time
    de_start_time = time.time()
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=maxiter,
            popsize=popsize, mutation=(0.5, 1.0), recombination=0.9,
            seed=seed, workers=1, disp=False, tol=1e-6, updating='immediate'
        )
        if de_result.success and de_result.fun < best_c5:
            final_h_full = np.clip(de_result.x, 0, 1)
            best_c5 = objective(final_h_full)
    except Exception:
        pass

    # Phase 2: Local optimization with perturbations
    while time.time() - start_time < budget_s - 20:
        # Random perturbation
        perturbed = final_h_full.copy()
        perturb_indices = np.random.choice(n_points, size=min(n_points, 10), replace=False)
        perturbation = np.random.normal(0, 0.1, len(perturb_indices))
        perturbed[perturb_indices] += perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = perturbed.sum()
        perturbed = (perturbed / current_sum) * target_sum_full if current_sum > 1e-6 else np.zeros(n_points)
        perturbed = np.clip(perturbed, 0, 1)

        try:
            # Use SLSQP
            result = minimize(
                objective, x0=perturbed, method='SLSQP',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                new_h_full = np.clip(result.x, 0, 1)
                new_c5 = objective(new_h_full)
                if new_c5 < best_c5:
                    final_h_full = new_h_full
                    best_c5 = new_c5
        except:
            pass

        try:
            # Use COBYLA
            result = minimize(
                objective, x0=perturbed, method='COBYLA',
                bounds=bounds,
                options={'maxiter': 50, 'rhobeg': 0.1, 'disp': False}
            )
            if result.success:
                new_h_full = np.clip(result.x, 0, 1)
                new_c5 = objective(new_h_full)
                if new_c5 < best_c5:
                    final_h_full = new_h_full
                    best_c5 = new_c5
        except:
            pass

    # Final validation and normalization
