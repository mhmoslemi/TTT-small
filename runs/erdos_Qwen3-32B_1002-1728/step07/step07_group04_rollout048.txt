import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points / 4.0

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s < 1e-8:
            h_half[:] = target_sum_half / half_n
        else:
            h_half = h_half / s * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half

    initial_half = h_initial[:half_n].copy()
    initial_half = repair_half(initial_half)

    def sum_of_squares_objective(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.sum(corr ** 2) * (dx) ** 2  # dx is 2.0 / n_points

    cons_cohesion = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds_cohesion = [(0.0, 1.0)] * half_n

    h_cohesion = initial_half.copy()
    if time.time() - start_time < budget_s * 0.6:
        result_cohesion = minimize(sum_of_squares_objective, x0=h_cohesion, method='SLSQP', bounds=bounds_cohesion,
                                   constraints=cons_cohesion, options={'maxiter': 200, 'ftol': 1e-10})
        if result_cohesion.success:
            h_cohesion = repair_half(result_cohesion.x)

    def c5_objective(h_half):

        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
