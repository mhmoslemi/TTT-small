    best_h_half = res.x
    best_h_half = np.clip(best_h_half, 0, 1)

    # Construct final h
    h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    if abs(np.sum(h_full) - n_points / 2) > 1e-6:
        h_full = h_full / np.sum(h_full) * (n_points / 2)
        h_full = np.clip(h_full, 0, 1)

    best_c5 = compute_c5(best_h_half)
    return (h_full.tolist(), best_c5, n_points)
