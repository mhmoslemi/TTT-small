import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    scale_factor = target_sum / current_sum if current_sum > 1e-6 else 1.0
    h_full_initial = np.clip(h_full_initial * scale_factor, 0, 1)

    def objective_full(x):
        h_clipped = np.clip(x, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds_full = [(0.0, 1.0)] * n_points
    cons_full = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    final_h = h_full_initial.copy()
    best_c5 = objective_full(final_h)

    # Phase 1: Differential evolution with higher popsize and more generations
    def de_full_objective(x):
        h_clipped = np.clip(x, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    de_budget = budget_s * 0.7
    popsize = 50
    maxiter = 25
    try:
        de_result = differential_evolution(
            de_full_objective, bounds_full, strategy='currenttobest1bin',
            maxiter=maxiter, popsize=popsize, tol=1e-6,
            mutation=(0.5, 1.5), recombination=0.9, seed=seed, workers=1,
            updating='deferred'
        )
        if de_result.success:
            best_h = np.clip(de_result.x, 0, 1)
            if np.sum(best_h) > 1e-6:
                best_h = best_h * (target_sum / np.sum(best_h))
            best_h = np.clip(best_h, 0, 1)
            final_h = best_h
            best_c5 = objective_full(best_h)
    except Exception as e:
        pass

    # Phase 2: Local optimization with adaptive noise and larger stepsize
    while time.time() - start_time < budget_s - 5:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.22 * (1 - frac_remaining**1.2)

        perturbed = final_h.copy()
        perturb_indices = np.random.choice(n_points, size=int(n_points**0.5), replace=False)
        perturbed[perturb_indices] += np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed = np.clip(perturbed, 0, 1)

        if np.sum(perturbed) > 1e-6:
            perturbed = perturbed * (target_sum / np.sum(perturbed))
        perturbed = np.clip(perturbed, 0, 1)

        try:
            result = minimize(

            if result.success:

        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
