def repair(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    sum_clipped = h_clipped.sum()
    if sum_clipped == 0:
        return np.full_like(h_clipped, target_sum / len(h_clipped))
    h_scaled = (h_clipped / sum_clipped) * target_sum
    return np.clip(h_scaled, 0, 1)

def compute_c5(h_half, n_points, dx):
    half_n = len(h_half)
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[-half_n:] = h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
