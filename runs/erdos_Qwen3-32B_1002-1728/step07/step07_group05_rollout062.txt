import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def repair(x, target_sum):
        x_clipped = np.clip(x, 0., 1.)
        total = x_clipped.sum()
        if total == 0:
            x_clipped[np.random.randint(len(x))] = 1.0
        scale = target_sum / total
        x_scaled = np.clip(x_clipped * scale, 0, 1)
        return x_scaled

    def create_h(h_half):
        return np.concatenate([h_half, h_half[::-1]])

    def objective(x):
        h = create_h(repair(x, target_sum_half))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize from symmetry
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_best = repair(h_half_initial, target_sum_half)
    best_c5 = objective(h_half_best)

    # --- Simulated Annealing ---
    np.random.seed(seed)
    temp = 10.0
    cooling_rate = 0.995
    sa_time = budget_s * 0.4
    sa_end = start_time + sa_time
    while time.time() < sa_end:
        h_half_new = h_half_best + 0.1 * (np.random.rand(half_n) - 0.5)
        h_half_new = repair(h_half_new, target_sum_half)
        new_c5 = objective(h_half_new)
        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            h_half_best, best_c5 = h_half_new, new_c5
        temp *= cooling_rate

    # --- Particle Swarm ---
    n_swarm = 25
    particles = [h_half_best + 0.05 * (np.random.rand(half_n) - 0.5) for _ in range(n_swarm)]
    particles = [repair(p, target_sum_half) for p in particles]
    vel = 0.05 * np.random.randn(n_swarm, half_n)
    pbest = [p for p in particles]
    pbest_c5 = [objective(p) for p in particles]
    ibest_idx = np.argmin(pbest_c5)
    gbest = particles[ibest_idx]

    pso_time = budget_s * 0.4
    pso_end = start_time + pso_end
    while time.time() < pso_end and pso_end > time.time():
        for i in range(n_swarm):
            r1, r2 = np.random.rand(2)
            vel[i] = 0.7 * vel[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
            particles[i] = np.clip(particles[i] + vel[i], 0., 1.)
            particles[i] = repair(particles[i], target_sum_half)
            current_c5 = objective(particles[i])
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
        ibest_idx = np.argmin(pbest_c5)
        gbest = pbest[ibest_idx].copy()
        best_c5 = min(best_c5, pbest_c5[ibest_idx])

    # --- Local Refinement ---
    attemps = 3
    while attemps > 0 and time.time() < start_time + budget_s:
        res = minimize(
            objective, x0=gbest, method='L-BFGS-B',
            bounds=Bounds(0, 1),
            options={'maxiter': 50, 'ftol': 1e-10, 'disp': False})
        if res.success:

            c5_ref = objective(h_half_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                h_half_best = h_half_ref
        attemps -= 1

    h_full = create_h(h_half_best)
    return (h_full.tolist(), best_c5, n_points)
