import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Enforce constraints on initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    if abs(h_full_initial.sum() - target_sum_full) > 1e-6:
        h_full_initial = h_full_initial * (target_sum_full / h_full_initial.sum())
        h_full_initial = np.clip(h_full_initial, 0, 1)
    h_initial = h_full_initial[:half_n]

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_clipped = np.clip(sym_vars, 0, 1)
        h_scaled = h_clipped / h_clipped.sum() * target_sum_half
        h_full = create_full_h(h_scaled)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(sym_vars):
        h_clipped = np.clip(sym_vars, 0, 1)
        return h_clipped / h_clipped.sum() * target_sum_half

    best_h_half = h_initial.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)

    # Multi-resolution grid search
    for grid_scale in [0.3, 0.15, 0.05]:
        if time.time() - start_time > budget_s - 30:
            break
        num_candidates = 30 if grid_scale >= 0.3 else 20
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 30:
                break
            perturbed = best_h_half.copy()
            perturbed += np.random.uniform(-grid_scale, grid_scale, len(perturbed))
            perturbed = repair(perturbed)
            new_c5 = objective(perturbed)
            if new_c5 < best_c5:
                best_h_half = perturbed
                best_c5 = new_c5

    # Local optimization phase
    while time.time() - start_time < budget_s - 15:
        perturbed = best_h_half.copy()
        perturbed += np.random.normal(0, 0.03, len(perturbed))
        perturbed = repair(perturbed)

        try:
            result = minimize(
                objective, x0=perturbed, 
                method='L-BFGS-B',
                bounds=[(0,1) for _ in perturbed],
                options={'maxiter': 100, 'ftol': 1e-12}
            )
            if result.success:
                refined = repair(result.x)
                new_c5 = objective(refined)
                if new_c5 < best_c5:
                    best_h_half = refined
                    best_c5 = new_c5
        except:
            continue

    # Final validation and output
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        h_full = h_full * (target_sum_full / current_sum)
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
