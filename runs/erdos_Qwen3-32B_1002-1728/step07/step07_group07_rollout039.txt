import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    n_blocks = 10
    m_blocks = n_points // n_blocks  # 7

    def h_to_blocks(h):
        return h.reshape(n_blocks, m_blocks).mean(axis=1)

    block_values_initial = h_to_blocks(h_initial)

    def repair_block_values(block_vals):
        return np.clip(block_vals, 0.0, 1.0)

    def evaluate(block_vals):
        block_vals = repair_block_values(block_vals)
        h_expanded = np.repeat(block_vals, m_blocks)
        sum_h = np.sum(h_expanded)
        if sum_h == 0:
            return np.full(n_points, target_sum / n_points), float('inf')
        h_scaled = h_expanded * (target_sum / sum_h)
        h_expanded = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_expanded, 1 - h_expanded, mode='full')
        c5 = np.max(corr) * dx
        return h_expanded, c5

    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 1000

    def create_individual():
        base = block_values_initial + np.random.normal(0, 0.1, size=n_blocks)
        return np.clip(base, 0, 1)

    population = [create_individual() for _ in range(pop_size)]

    best_c5 = float('inf')
    best_h = None

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate
        fitness = []
        for ind in population:
            h_eval, c5 = evaluate(ind)
            fitness.append((c5, h_eval))
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        parents = [f[0] for f in fitness[:int(pop_size * 0.2)]]  # block values
        # Update best
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1]
        # Generate next population
        new_pop = parents.copy()
        while len(new_pop) < pop_size:
            # Select parents
            p1, p2 = np.random.choice(parents, size=2)
            # Crossover
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_blocks)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            # Mutation
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_blocks)
                noise = np.random.normal(0, 0.1)
                child[idx] += noise
                child = np.clip(child, 0, 1)
            new_pop.append(child)
        population = new_pop

    # Local optimization
    best_h = best_h if best_h is not None else evaluate(create_individual())[0]
    def objective_full(x):
        x_clipped = np.clip(x, 0, 1)
        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0.0, 1.0)] * n_points

    result_lo = minimize(objective_full, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
    h_refined = np.clip(result_lo.x, 0, 1)
    c5_refined = np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx
    if c5_refined < best_c5:

        best_h = h_refined

        best_h = (best_h / np.sum(best_h)) * target_sum
    best_h = np.clip(best_h, 0, 1)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), final_c5, n_points)
