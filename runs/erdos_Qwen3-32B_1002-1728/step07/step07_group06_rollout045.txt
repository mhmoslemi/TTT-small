import numpy as np
import time
import scipy.optimize as opt
import cvxpy as cp
from scipy.ndimage import zoom

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    def evaluate_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Construct periodic pattern with 2 ON, 2 OFF period
    period = 4
    base = [1.0, 1.0, 0.0, 0.0]
    h_periodic = np.tile(base, n // period + 1)[:, 0]
    h_periodic = h_periodic[:n]
    h_periodic = repair_h(h_periodic)
    best_h = h_periodic.copy()
    best_c5 = evaluate_c5(best_h, dx)

    # Simulated annealing with block flipping
    temp = 0.1
    iter_per_temp = 50
    while time.time() - start_time < budget_s and temp > 1e-5:
        for _ in range(iter_per_temp):
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            new_h = best_h.copy()
            # Flip values between i and j
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_h = repair_h(new_h)
            new_c5 = evaluate_c5(new_h, dx)
            delta_c5 = new_c5 - best_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
                best_h = new_h
                best_c5 = new_c5
        temp *= 0.95

    # Local gradient optimization
    bounds = [(0, 1)] * n
    res = opt.minimize(lambda h: evaluate_c5(h, dx), best_h, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 50})
    if res.success and evaluate_c5(res.x, dx) < best_c5:
        best_h = res.x
        best_c5 = evaluate_c5(best_h, dx)

    best_h = repair_h(best_h)
    return (best_h.tolist(), evaluate_c5(best_h, dx), n)
