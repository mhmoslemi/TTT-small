    while time.time() - start_time < sa_duration:
        noise = (np.random.rand(half_n) - 0.5) * 0.2 * T
        new_h = current_h_half + noise
        new_h = repair_half(new_h)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h_half = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h_half = current_h_half.copy()
                best_c5 = current_c5
        T *= cooling_rate
        if T < 1e-4:
            T = 1e-4

    pso_duration = 0.4 * budget_s
    population_size_pso = 20
    positions = [repair_half(best_h_half + 0.1 * np.random.randn(half_n)) for _ in range(population_size_pso)]
    velocities = [np.zeros(half_n) for _ in range(population_size_pso)]
    pbest_positions = [pos.copy() for pos in positions]
    pbest_values = [compute_c5(pos) for pos in positions]
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    pso_start = time.time()
    while time.time() - start_time < sa_duration + pso_duration:
        for i in range(population_size_pso):
            r1, r2 = np.random.rand(2)
            vel = omega * velocities[i] + c1 * r1 * (pbest_positions[i] - positions[i]) + c2 * r2 * (gbest_position - positions[i])
            new_pos = positions[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = repair_half(new_pos)
            new_c5 = compute_c5(new_pos)
            positions[i] = new_pos
            velocities[i] = vel
            if new_c5 < pbest_values[i]:
                pbest_positions[i] = new_pos.copy()
                pbest_values[i] = new_c5
                if new_c5 < gbest_value:
                    gbest_position = new_pos.copy()
                    gbest_value = new_c5
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h_half = gbest_position.copy()

    def bounded_objective(x):
        x = np.clip(x, 0, 1)
        total = x.sum()
        if total == 0:
            return np.inf
        x_scaled = (x / total) * target_sum_half
        x_scaled = np.clip(x_scaled, 0, 1)
        h = np.zeros(n_points)
        h[:half_n] = x_scaled
        h[-half_n:] = x_scaled[::-1]
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum_half}]
    for _ in range(3):
        try:
            result = minimize(
                bounded_objective, x0=best_h_half, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt.copy()
                    best_c5 = c5_new
        except:
            pass

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0.0, 1.0)
    return (h_full.tolist(), best_c5, n_points)
