                    if abs(new_sum - target_sum) < 1e-6:

                        if new_c5 < best_c5:
                            best_h = new_h
                            best_c5 = new_c5
            except:
                continue

    # Final validation
    h_opt = np.clip(best_h, 0, 1)
    h_opt_sum = h_opt.sum()
    if abs(h_opt_sum - target_sum) > 1e-6:
        h_opt = (h_opt / h_opt_sum) * target_sum
        h_opt = np.clip(h_opt, 0, 1)

    return (h_opt.tolist(), best_c5, n_points)
