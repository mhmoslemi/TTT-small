import numpy as np
import time
import scipy.optimize as opt
import numpy.random as random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = repair_h(h_initial)
    best_c5 = evaluate_c5(best_h)

    population_size = 100
    population = [best_h.copy()]
    for _ in range(population_size - 1):
        perturbed = h_initial.copy()
        indices = random.choice(n, size=20)
        perturbed[indices] += random.uniform(-0.1, 0.1, size=20)
        population.append(repair_h(perturbed))

    while time.time() - start_time < budget_s * 0.75:
        scores = np.array([evaluate_c5(h) for h in population])
        top_idx = np.argsort(scores)[:max(1, population_size // 10)]
        parents = [population[i] for i in top_idx]
        offspring = []
        for _ in range(population_size - len(parents)):
            p1, p2 = random.choice(parents, 2)
            alpha = random.uniform(0.4, 0.6)
            child = alpha * p1 + (1 - alpha) * p2
            if random.rand() < 0.2:
                indices = random.choice(n, size=5, replace=False)
                child[indices] += random.uniform(-0.1, 0.1, size=5)
            child = repair_h(child)
            offspring.append(child)
        new_pop = parents + offspring
        scores_new = [evaluate_c5(h) for h in new_pop]
        pool = sorted(zip(scores_new, new_pop), key=lambda x: x[0])[:population_size]
        population = [h for s, h in pool]
        current_best = min(scores_new)
        if current_best < best_c5:
            best_h = [h for s, h in pool if evaluate_c5(h) == current_best][0].copy()
            best_c5 = current_best

    # Local refinement
    candidates = [best_h]
    candidates.extend(random.sample(population, min(10, len(population))))
    for cand in candidates:
        if time.time() - start_time > budget_s:
            break
        def obj(x):
            x = repair_h(np.clip(x, 0, 1))
            return evaluate_c5(x)
        res = opt.minimize(obj, cand, method='L-BFGS-B', bounds=opt.Bounds(0, 1),
                          options={'maxiter': 50, 'ftol': 1e-9})
        if res.success:
            trial = res.x
            trial = repair_h(np.clip(trial, 0, 1))
            if evaluate_c5(trial) < best_c5:
                best_h = trial.copy()
                best_c5 = evaluate_c5(trial)

    best_h = repair_h(best_h)
    return (best_h.tolist(), best_c5, n)
