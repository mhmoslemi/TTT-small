import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2

    h_control = h_full_initial[:half_n].copy()
    h_control = np.clip(h_control, 0, 1)
    current_sum = np.sum(h_control)
    if current_sum > 1e-6:
        h_control[:] = (h_control / current_sum) * target_sum_half
    h_control = np.clip(h_control, 0, 1)

    # Set up parameters
    learning_rate = 0.01
    max_iterations = 2000
    best_c5 = np.inf
    best_h_control = h_control.copy()

    for iter_count in range(max_iterations):
        if time.time() - start_time > budget_s - 5:

        full_h = np.zeros(n_points)

        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_control = h_control.copy()

        # Find max overlap index

        for i in range(n_points):

import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    h_control = h_full_initial[:half_n].copy()
    h_control = np.clip(h_control, 0, 1)
    current_sum = np.sum(h_control)
    if abs(current_sum) > 1e-6:
        h_control[:] = (h_control / current_sum) * target_sum_half
    h_control = np.clip(h_control, 0, 1)

    best_c5 = np.inf
    best_h_control = h_control.copy()
    learning_rate = 0.01
    max_iterations = 2000

    for iter_count in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break

        full_h = np.zeros(n_points)
        full_h[:half_n] = h_control
        full_h[half_n:] = h_control[::-1]

        corr_vals = np.correlate(full_h, 1 - full_h, mode='full')
        valid_corr = corr_vals[n_points - 1 : 2 * n_points - 1]
        current_c5 = np.max(valid_corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_control = h_control.copy()

        max_indices = np.where(valid_corr * dx == best_c5)[0]
        if len(max_indices) == 0:
            continue
        k_star = max_indices[0]

        grad_full = np.zeros(n_points)
        for i in range(n_points):
            term1 = 1.0 - full_h[i + k_star] if (i + k_star) < n_points else 0.0
            term2 = full_h[i - k_star] if (i - k_star) >= 0 else 0.0
            grad_full[i] = dx * (term1 - term2)

        grad_control = np.zeros(half_n)
        for ic in range(half_n):
            mirrored_ic = n_points - 1 - ic
            grad_control[ic] = 0.5 * (grad_full[ic] + grad_full[mirrored_ic])

        h_control = np.clip(h_control, 0, 1)
