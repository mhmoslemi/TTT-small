import numpy as np
import time
from operator import itemgetter

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) * dx == 1 => sum(h) == target_sum

    def create_initial_candidate(h, n_blocks=5):
        block_breaks = np.linspace(0, 2, n_blocks + 1)
        block_values = np.zeros(n_blocks)
        for i in range(n_blocks):
            b_start = block_breaks[i]
            b_end = block_breaks[i + 1]
            j_start = int(np.floor(b_start * n_points / 2.0))
            j_end = int(np.ceil(b_end * n_points / 2.0))
            if j_end > n_points:
                j_end = n_points
            block_values[i] = np.mean(h[j_start:j_end])
        return {'block_breaks': block_breaks, 'block_values': block_values}

    def blocks_to_h(candidate):
        h = np.zeros(n_points)
        block_breaks = candidate['block_breaks']
        block_values = candidate['block_values']
        for i in range(len(block_values)):
            b_start = block_breaks[i]
            b_end = block_breaks[i + 1]
            j_start = int(np.floor(b_start * n_points / 2.0))
            j_end = int(np.ceil(b_end * n_points / 2.0))
            if j_end > n_points:
                j_end = n_points
            h[j_start:j_end] = block_values[i]
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        return h

    def generate_add_block_variant(candidate):
        n_blocks = len(candidate['block_values'])
        block_idx = np.random.randint(n_blocks)
        b_start = candidate['block_breaks'][block_idx]
        b_end = candidate['block_breaks'][block_idx + 1]
        new_break = np.random.uniform(b_start, b_end)
        new_breaks = np.insert(candidate['block_breaks'], block_idx + 1, new_break)
        new_values = np.insert(candidate['block_values'], block_idx + 1, candidate['block_values'][block_idx] / 2 + 0.25)
        new_values = np.insert(new_values, block_idx + 2, candidate['block_values'][block_idx] / 2 + 0.25)
        new_values = np.clip(new_values, 0, 1)
        return {'block_breaks': new_breaks, 'block_values': new_values}

    def generate_adjust_breaks_variant(candidate):
        n_breaks = len(candidate['block_breaks'])
        break_idx = np.random.randint(1, n_breaks - 1)
        b_prev = candidate['block_breaks'][break_idx - 1]
        b_current = candidate['block_breaks'][break_idx]
        b_next = candidate['block_breaks'][break_idx + 1]
        delta = np.random.uniform(-0.05, 0.05) * (b_next - b_prev)
        new_break = b_current + delta
        new_break = np.clip(new_break, b_prev + 1e-6, b_next - 1e-6)
        new_breaks = candidate['block_breaks'].copy()
        new_breaks[break_idx] = new_break
        return {'block_breaks': new_breaks, 'block_values': candidate['block_values']}

    def generate_modify_values_variant(candidate):
        n_blocks = len(candidate['block_values'])
        block_idx = np.random.randint(n_blocks)
        delta = np.random.uniform(-0.15, 0.15)
        new_value = candidate['block_values'][block_idx] + delta
        new_value = np.clip(new_value, 0, 1)
        new_values = candidate['block_values'].copy()
        new_values[block_idx] = new_value
        return {'block_breaks': candidate['block_breaks'], 'block_values': new_values}

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_n_blocks = 5
    initial_candidate = create_initial_candidate(h_full_initial, initial_n_blocks)
    candidates = []
    h0 = blocks_to_h(initial_candidate)
    c50 = evaluate(h0)
    candidates.append({'candidate': initial_candidate, 'h': h0, 'c5': c50})
    beam_size = 12

    while time.time() - start_time < budget_s:
        new_candidates = []
        for c in candidates:
            for _ in range(2):
                perturbation_type = np.random.choice(['add', 'adjust', 'modify'])
                if perturbation_type == 'add':
                    new_candidate = generate_add_block_variant(c['candidate'])
                elif perturbation_type == 'adjust':
                    new_candidate = generate_adjust_breaks_variant(c['candidate'])
                else:
                    new_candidate = generate_modify_values_variant(c['candidate'])
                h = blocks_to_h(new_candidate)
                c5 = evaluate(h)
                new_candidates.append({'candidate': new_candidate, 'h': h, 'c5': c5})
        all_candidates = candidates + new_candidates
        all_candidates.sort(key=itemgetter('c5'))
        candidates = all_candidates[:beam_size]
        if time.time() - start_time > budget_s:
            break
    best = min(candidates, key=lambda x: x['c5'])
    return (best['h'].tolist(), best['c5'], n_points)
