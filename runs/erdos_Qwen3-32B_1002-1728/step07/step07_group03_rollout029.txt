        if time.time() - start_time > budget_s:
            break
        current_inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)

        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            cog_part = cognitive * r1 * (personal_best_pos[i] - positions[i])
            soc_part = social * r2 * (global_best_pos - positions[i])
            velocities[i] = current_inertia * velocities[i] + cog_part + soc_part
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)

            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            s = positions[i].sum()
            if s != 0:
                positions[i] = (positions[i] / s) * target_sum_half
            positions[i] = np.clip(positions[i], 0, 1)

            score = objective_particle(positions[i], n_points, half_n, dx)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_pos[i] = positions[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best_pos = positions[i].copy()

    # Local Refinement
    best_h_half = global_best_pos.copy()
    while time.time() - start_time < budget_s - 5:
        perturbed = best_h_half + np.random.normal(0, 0.05, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        s = perturbed.sum()
        if s != 0:
            perturbed = (perturbed / s) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        def obj(x):
            h_full = np.zeros(n_points)
            h_full[:half_n] = x
            h_full[-half_n:] = x[::-1]
            h_clipped = np.clip(h_full, 0, 1)
            corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr_vals) * dx

        bounds = [(0, 1)] * half_n
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
        result = minimize(obj, x0=perturbed, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 30, 'ftol': 1e-12, 'disp': False})
        if result.success:
            candidate = result.x
            score = obj(candidate)
            if score < global_best_score:
                best_h_half = candidate.copy()
                global_best_score = score

    # Final Constraint Enforce
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), global_best_score, n_points)

def objective_particle(h_half, n_points, half_n, dx):
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[-half_n:] = h_half[::-1]
    h_clipped = np.clip(h_full, 0, 1)
    corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr_vals) * dx
