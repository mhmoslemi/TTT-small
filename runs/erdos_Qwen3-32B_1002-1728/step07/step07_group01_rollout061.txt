import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum_h = n_points / 2.0

    # Group into 10 groups to reduce variables
    n_groups = 10
    group_size = n_points // n_groups  # Assumes n_points divisible by n_groups
    groups_indices = [np.arange(i*group_size, (i+1)*group_size) for i in range(n_groups)]

    def expand_averages(averages):
        h = np.zeros_like(h_initial)
        for i in range(n_groups):
            h[groups_indices[i]] = averages[i]
        return h

    # Initial group averages
    avg_initial = np.array([h_initial[groups_indices[i]].mean() for i in range(n_groups)])
    avg_initial = np.clip(avg_initial, 0, 1)
    total = np.sum(avg_initial)
    if total != 0:
        scale = (target_sum_h / group_size) / total
        avg_initial *= scale
    avg_initial = np.clip(avg_initial, 0, 1)

    def objective_avg(avg):
        h = expand_averages(np.clip(avg, 0, 1))
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    cons_avg = [{'type': 'eq', 'fun': lambda x: np.sum(x) - (target_sum_h / group_size)}]
    bounds_avg = [(0.0, 1.0) for _ in range(n_groups)]

    best_avg = avg_initial.copy()
    best_c5 = objective_avg(best_avg)

    # Phase 1: Differential evolution
    de_result = differential_evolution(
        objective_avg, bounds_avg, strategy='currenttobest1bin',
        mutation=(0.5, 1.5), recombination=0.9,
        maxiter=30, popsize=20, tol=1e-5, seed=seed
    )
    if de_result.success:
        avg_candidate = np.clip(de_result.x, 0, 1)
        total = np.sum(avg_candidate)
        if abs(total - (target_sum_h / group_size)) > 1e-6:
            scale = (target_sum_h / group_size) / total
            avg_candidate *= scale
        avg_candidate = np.clip(avg_candidate, 0, 1)
        c5_candidate = objective_avg(avg_candidate)
        if c5_candidate < best_c5:
            best_avg = avg_candidate
            best_c5 = c5_candidate

    # Phase 2: Local optimization
    while time.time() - start_time < budget_s - 10:
        perturbed = best_avg + np.random.normal(0, 0.05, size=n_groups)
        perturbed = np.clip(perturbed, 0, 1)
        total_pert = np.sum(perturbed)
        if total_pert != 0:
            scale = (target_sum_h / group_size) / total_pert
            perturbed *= scale
        perturbed = np.clip(perturbed, 0, 1)
        try:
            result = minimize(
                objective_avg, x0=perturbed, method='SLSQP',
                bounds=bounds_avg, constraints=cons_avg,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if result.success:
                avg_candidate = np.clip(result.x, 0, 1)
                total_candidate = np.sum(avg_candidate)
                if abs(total_candidate - (target_sum_h / group_size)) > 1e-6:
                    scale = (target_sum_h / group_size) / total_candidate
                    avg_candidate *= scale
                avg_candidate = np.clip(avg_candidate, 0, 1)
                c5_candidate = objective_avg(avg_candidate)
                if c5_candidate < best_c5:
                    best_avg = avg_candidate
                    best_c5 = c5_candidate
        except:
            pass
        if time.time() - start_time >= budget_s:
            break

    # Final h reconstruction
    h_full = expand_averages(best_avg)
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if abs(current_sum - target_sum_h) > 1e-6:
        h_full = (h_full / current_sum) * target_sum_h
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
