    coarse_n = max(35, n // 2)
    def downsample(x):
        return zoom(x, (coarse_n / n), order=1)
    def upsample(x):
        return zoom(x, (n / coarse_n), order=1)

    h_coarse = downsample(best_h)
    h_coarse = repair_h(h_coarse)

    def objective_coarse(x):
        x = x.reshape(-1)
        x_repaired = repair_h(x)
        h_fine = upsample(x_repaired)
        return evaluate_c5(h_fine, dx)

    x0 = h_coarse.copy().ravel()
    res = opt.minimize(objective_coarse, x0, method='L-BFGS-B', bounds=opt.Bounds(0, 1),
                       options={'maxiter': 100, 'ftol': 1e-10})
    if res.success:
        h_coarse_opt = res.x.reshape(-1)
        h_fine_opt = repair_h(upsample(h_coarse_opt))
        c5 = evaluate_c5(h_fine_opt, dx)
        if c5 < best_c5:
            best_h[:] = h_fine_opt
            best_c5 = c5

    # ADMM refinement
    for _ in range(3):
        z = cp.Variable(n)
        rho = 2.0
        objective = (cp.norm(z)**2 / 2 + rho / 2 * cp.norm(z - best_h, 2)**2)
        constraints = [cp.sum(z) == target_sum, 0 <= z, z <= 1]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL:
            trial_h = repair_h(z.value)
            c5 = evaluate_c5(trial_h, dx)
            if c5 < best_c5:
                best_h[:] = trial_h
                best_c5 = c5

    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
