                    x_clipped = np.clip(x, 0, 1)
                    corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
                    return np.max(corr) * dx

                # Apply with L-BFGS-B (better handling of box bounds)
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_full})
                bounds_fine = [(0.0, 1.0)] * len(h_fine)
                try:
                    result_fine = minimize(
                        objective_fine, h_fine, method='L-BFGS-B',
                        bounds=bounds_fine,
                        jac=None,
                        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-4}
                    )
                    if result_fine.success:
                        h_fine_opt = np.clip(result_fine.x, 0, 1)
                        c5 = np.max(np.correlate(h_fine_opt, 1 - h_fine_opt, mode='full')) * dx
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_fine_opt
                except Exception:
                    pass
        except Exception:
            pass

    # Final feasibility repair
    best_h = np.clip(best_h, 0, 1)
    if abs(best_h.sum() - target_sum_full) > 1e-6:
        best_h = (best_h / best_h.sum()) * target_sum_full
    best_h = np.clip(best_h, 0, 1)
    c5_final = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), c5_final, n_points)
