import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = 1/2 * integral

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    h_half_initial = repair_half(h_full_initial[:half_n].copy(), target_sum_half)
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(h_half_initial)

    # Simulated Annealing Phase
    sa_time_budget = min(300.0, budget_s * 0.4)
    sa_iter = 0

    cooling_rate = 0.9995

        perturbation = 0.1 * np.random.randn(half_n)
        new_half = best_h_half + perturbation
        new_half = repair_half(new_half, target_sum_half)
        delta_c5 = compute_c5(new_half) - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            best_h_half[:] = new_half
            best_c5 = compute_c5(best_h_half)
        temp *= cooling_rate
        sa_iter += 1

    # Particle Swarm Optimization
    pso_time_budget = min(400.0, budget_s * 0.4)
    particles = repair_half(np.random.rand(25, half_n), target_sum_half, axis=1)
    velocities = 0.01 * np.random.randn(25, half_n)
    personal_best = particles.copy()
    pb_c5 = np.array([compute_c5(p) for p in particles])
    global_best_idx = np.argmin(pb_c5)

    pso_end = time.time() + pso_time_budget
