import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = half_n / 2.0
    dx = 2.0 / n_points

    def create_full_h(h_sym):
        h = np.zeros(n_points)
        h[:half_n] = h_sym
        h[-half_n:] = h_sym[::-1]
        return h

    def repair_h_sym(h_sym_candidate):
        h_clipped = np.clip(h_sym_candidate, 0, 1)
        current_sum = h_clipped.sum()
        scale = target_sum_half / current_sum if current_sum != 0 else 1.0
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    def objective(h_sym):
        h_full = create_full_h(repair_h_sym(h_sym))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective_for_optimizer(x):
        return objective(x)

    np.random.seed(seed)
    h_sym_initial = h_full_initial[:half_n]
    h_sym_initial = repair_h_sym(h_sym_initial)
    current_best_h = h_sym_initial.copy()
    best_c5 = objective(current_best_h)

    grid_levels = [
        {'num_samples': 20, 'std': 0.3},
        {'num_samples': 40, 'std': 0.15},
        {'num_samples': 60, 'std': 0.075},
    ]

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    remaining_time = budget_s - (time.time() - start_time)
    for level in grid_levels:
        if remaining_time < 10:
            break
        num_samples = level['num_samples']
        std = level['std']
        # Generate perturbed samples
        for _ in range(num_samples):
            perturbed = current_best_h + np.random.normal(0, std, size=half_n)
            perturbed = np.clip(perturbed, 0, 1)
            repaired = repair_h_sym(perturbed)
            c5 = objective(repaired)
            if c5 < best_c5:
                best_c5 = c5
                current_best_h = repaired.copy()
        remaining_time = budget_s - (time.time() - start_time)

    while remaining_time > 20:
        try:
            result = minimize(
                objective_for_optimizer,
                x0=current_best_h,
                method='SLSQP',

            if result.success:
