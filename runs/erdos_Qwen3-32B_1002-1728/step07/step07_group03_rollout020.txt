import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Initialize symmetric variables
    h_init = h_full_initial[:half_n].copy()
    h_sum = h_init.sum()
    if abs(h_sum - target_sum_half) > 1e-6:
        h_init = (h_init / h_sum) * target_sum_half
    h_init = np.clip(h_init, 0, 1)

    def create_full_h(sym_vars):
        return np.concatenate([sym_vars, sym_vars[::-1]])

    def objective(sym_vars):
        h = np.clip(sym_vars, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')
        h_scaled = (h / h_sum) * target_sum_half
        h_full = create_full_h(np.clip(h_scaled, 0, 1))
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * (2.0 / n_points)

    # Evolution Strategy (ES) with adaptive mutation
    np.random.seed(seed)
    pop_size = 50
    max_gen = 150
    sigma = 0.1
    current_best = h_init.copy()
    best_obj = objective(current_best)

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 5:
            break
        population = []
        for _ in range(pop_size):
            while True:
                perturbation = sigma * np.random.randn(half_n)
                candidate = current_best + perturbation
                candidate = np.clip(candidate, 0, 1)
                c_sum = candidate.sum()
                if c_sum > 1e-6:
                    candidate = (candidate / c_sum) * target_sum_half
                candidate = np.clip(candidate, 0, 1)
                population.append(candidate)
                break  # Always valid after scaling
        # Evaluate population
        scores = np.array([objective(p) for p in population])
        # Select top 50%
        elite_size = pop_size // 2
        elite_indices = np.argsort(scores)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Update current best
        new_best = np.mean(elites, axis=0)
        new_obj = objective(new_best)
        if new_obj < best_obj:
            current_best = new_best
            best_obj = new_obj
            # Success: expand sigma
            sigma = min(sigma * 1.1, 0.5)
        else:
            # Regress: shrink sigma
            sigma = max(sigma * 0.9, 1e-4)

    # Local optimization with L-BFGS-B
    bounds = [(0.0, 1.0)] * half_n
    cons = {
        'type': 'eq',
        'fun': lambda x: x.sum() - target_sum_half
    }
    result_local = minimize(
        objective,
        x0=current_best,
        method="L-BFGS-B",
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-12}
    )
    if result_local.success:
        current_best = result_local.x
        best_obj = result_local.fun

    # Final constraint adjustment and output
    final_h = create_full_h(np.clip(current_best, 0, 1))
    final_h_sum = final_h.sum()
    if abs(final_h_sum - target_sum_full) > 1e-6:
        final_h = (final_h / final_h_sum) * target_sum_full
        final_h = np.clip(final_h, 0, 1)
    
    return (final_h.tolist(), best_obj, n_points)
