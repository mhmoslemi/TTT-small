    result = minimize(objective_local, x0=final_h_half, method='SLSQP', bounds=bounds_local,
                      constraints=cons_local, options={'maxiter': 150, 'ftol': 1e-12})
    if result.success:
        final_h_half = np.clip(result.x, 0, 1)
    
    # Final constraint enforcement
    h_full = np.concatenate([final_h_half, final_h_half[::-1]])
    h_full = np.clip(h_full, 0, 1)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    
    # Final C5 evaluation
    corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr_vals) * dx
    return (h_full.tolist(), c5_bound, n_points)
