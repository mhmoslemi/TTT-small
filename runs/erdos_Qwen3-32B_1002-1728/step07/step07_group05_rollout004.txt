import numpy as np
import time
from scipy.optimize import minimize

def repair_half(x, target_sum, eps=1e-8):
    x_clipped = np.clip(x, 0.0, 1.0)
    total = x_clipped.sum()
    if abs(total) < eps:
        idx = np.random.choice(np.arange(len(x)))
        x_clipped[idx] = 1.0
        total = 1.0
    scale = target_sum / total
    x_scaled = x_clipped * scale
    x_scaled = np.clip(x_scaled, 0, 1)
    return x_scaled

def create_full_h(sym_vars, half_n):
    h = np.zeros(2*half_n)
    h[:half_n] = sym_vars
    h[-half_n:] = sym_vars[::-1]
    return h

def compute_c5(h_half, half_n, n_points):

    dx = 2.0 / n_points
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx

def repair_half(x, target_sum, eps=1e-8):
    x_clipped = np.clip(x, 0.0, 1.0)
    total = x_clipped.sum()
    if abs(total) < eps:
        idx = np.random.choice(np.arange(len(x)))
        x_clipped[idx] = 1.0
        total = 1.0
    scale = target_sum / total
    x_scaled = x_clipped * scale
    x_scaled = np.clip(x_scaled, 0, 1)
    return x_scaled
