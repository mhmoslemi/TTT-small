        dx = 2.0 / grid_size
        return np.max(corr) * dx

    # Define constraints and bounds
    target_sum_half = grid_size / 4.0
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    popsize = 20
    maxiter_de = 20

        seed=42, disp=False

    if de_result.success:
        h_half_best = np.clip(de_result.x, 0, 1)
        # Repair sum

    # Phase 2: Local optimization with various methods
    remaining_time = time_budget - (time.time() - start_time)

    # Apply perturbations and multi-method optimization
