import random
from scipy.ndimage import zoom

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)

        return np.max(corr) * dx

    h = repair(h_initial)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Simulated annealing
    temp = 0.5
    cooling_rate = 0.995
    max_steps = 30000
    for step in range(max_steps):
        if time.time() - start_time > budget_s - 10:
            break
        method = random.choice(['noise', 'uniform', 'block'], p=[0.6, 0.2, 0.2])

        # Generate candidate
        if method == 'noise':
            noise = np.random.normal(0, 0.05, n)
            candidate = h + noise

        else:  # block-driven

            pos = np.random.randint(0, n - block_size)
            candidate[pos:pos+block_size] = np.random.choice([0, 1])
        
        candidate = repair(candidate)
        candidate_c5 = evaluate_c5(candidate)

        # Acceptance check

    final_c5 = evaluate_c5(best_h)
