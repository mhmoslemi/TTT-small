        return evaluate_c5(x, dx)

    res = opt.minimize(objective_coarse, h_coarse, method='L-BFGS-B', bounds=opt.Bounds(0, 1),
                      options={'maxiter': 100, 'ftol': 1e-10})
    if res.success:
        refined = zoom(res.x, n / coarse_n, order=0)
        refined = repair_h(refined)
        best_c5 = min(best_c5, evaluate_c5(refined, dx))
        best_h = refined

    # Constrained optimization via ADMM
    z = cp.Variable(n)
    rho = 2.0
    objective_fn = cp.norm(z)**2 / 2 + rho / 2 * cp.norm(z - best_h)**2
    constraints = [cp.sum(z) == target_sum, 0 <= z, z <= 1]
    problem = cp.Problem(cp.Minimize(objective_fn), constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
    if problem.status == 'optimal':
        trial_h = repair_h(np.array(z.value))
        best_c5 = min(best_c5, evaluate_c5(trial_h, dx))
        best_h = trial_h

    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
