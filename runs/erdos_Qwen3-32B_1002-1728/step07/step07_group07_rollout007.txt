def split_block(blocks):
    if len(blocks) < 1:
        return None  # Cannot split
    # Randomly choose a block and a split position
    idx = random.randint(0, len(blocks) - 1)
    block_start, block_end, block_value = blocks[idx]

    new_block2 = (split_pos, block_end, block_value)

    # Replace the original block with the two new blocks

    return new_blocks

def adjust_break(blocks):

    # Select a break (between two blocks)

    current_start, current_end, current_value = blocks[break_idx]
    next_start, next_end, next_value = blocks[break_idx + 1]  # if break is between blocks[break_idx] and [break_idx+1]

    # Wait, the blocks are ordered by start position. The break between them is at current_end (which is next_start).
    # So the break to adjust is between blocks[break_idx] and blocks[break_idx + 1], if we are adjusting the split point at current_end.
    # To adjust this break, we can move it by a small amount.
    # The break is between current_end and next_start (they are the same). So adjust it by a small amount.

    # Determine how much to adjust the break
    current_width = current_end - current_start
    next_width = next_end - next_start
    # Adjust by ±5% of the width of the adjacent blocks
    delta = random.uniform(-0.05 * (current_width + next_width), 0.05 * (current_width + next_width))

    new_current_end = current_end + delta
    new_next_start = new_current_end
    # Ensure the new break doesn't cross with other breaks
    # Check that new_current_end is between current_start and next_end
    if new_current_end <= current_start or new_current_end >= next_end:
        return None  # Invalid adjustment, discard this perturbation

    # Update the blocks
    new_blocks = blocks[:break_idx]
    new_blocks.append( (current_start, new_current_end, current_value) )
    new_blocks.append( (new_current_end, next_end, next_value) )
    new_blocks.extend(blocks[break_idx + 2:])

def modify_value(blocks):
    if len(blocks) < 2:
        return None  # Need at least two blocks to perform modify value
    # Select two random blocks: one to change, another to compensate
    block_idx = random.randint(0, len(blocks) - 1)
    other_idx = random.choice([i for i in range(len(blocks)) if i != block_idx])
    block_start, block_end, block_value = blocks[block_idx]
    other_start, other_end, other_value = blocks[other_idx]

    # Compute their widths
    block_width = block_end - block_start
    other_width = other_end - other_start

    # Generate a new value for the selected block
    # Ensure that changing it won't push other_value out of [0,1]
    max_possible_change = min(block_value, 1 - block_value)

    # Perturb the block's value by a small amount
    delta_v = random.uniform(-0.1 * (1 - block_value), 0.1 * (1 - block_value)) if block_value < 1 else random.uniform(-0.1 * block_value, 0)
    new_block_v = block_value + delta_v
    new_block_v = np.clip(new_block_v, 0.0, 1.0)

    # Compute the delta in integral for the block
    delta_integral = (new_block_v - block_value) * block_width

    # Compute the delta needed in other block: other_value_new = other_value + delta_integral / other_width
    new_other_v = other_value + (delta_integral / other_width)
    if new_other_v < 0.0 or new_other_v > 1.0:

    # Update the blocks
    new_blocks = []
    for i, (s, e, v) in enumerate(blocks):

            new_blocks.append( (s, e, new_block_v) )
        elif i == other_idx:
            new_blocks.append( (s, e, new_other_v) )
        else:
            new_blocks.append( (s, e, v) )
    return new_blocks
