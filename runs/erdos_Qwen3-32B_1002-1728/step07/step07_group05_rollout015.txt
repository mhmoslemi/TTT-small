import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial_full = np.array(initial_h_values)
    n_points = len(h_initial_full)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Repair initial half
    h_half_initial = h_initial_full[:half_n].copy()
    h_half_initial = np.clip(h_half_initial, 0, 1)
    s = h_half_initial.sum()
    if s > 0:
        h_half_initial = (h_half_initial / s) * target_sum_half
    h_half_initial = np.clip(h_half_initial, 0, 1)

    # Optimization problem setup
    h_half = cp.Variable(half_n)
    C5 = cp.Variable()
    constraints = [
        h_half >= 0,
        h_half <= 1,
        cp.sum(h_half) == target_sum_half
    ]

    for k in range(n_points):
        total = 0.0
        for a in range(n_points - k):
            a_idx = a
            ak_idx = a + k
            if a_idx < half_n:
                h_a = h_half[a_idx]
            else:
                h_a = h_half[69 - a_idx]
            if ak_idx < half_n:
                h_ak = h_half[ak_idx]
            else:
                h_ak = h_half[69 - ak_idx]
            total += h_a * (1 - h_ak)
        constraints.append(C5 >= total * dx)

    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)

    # Solver budget
    remaining_time = max(int(budget_s - (time.time() - start_time)), 5)
    try:
        problem.solve(
            solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.CVXOPT,
            verbose=False,
            solver_args={"TimeLimit": remaining_time}
        )
    except:
        pass

    # Extract solution
    h_full_opt = create_full_h(np.clip(h_half_initial, 0, 1))
    if problem.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE} and h_half.value is not None:

        h_val = np.clip(h_val, 0, 1) * (target_sum_half / np.sum(h_val) if np.sum(h_val) > 1e-6 else 1.0)
        h_full_opt = create_full_h(h_val)

    c5_bound = compute_c5(h_full_opt)
    return (h_full_opt.tolist(), c5_bound, n_points)
