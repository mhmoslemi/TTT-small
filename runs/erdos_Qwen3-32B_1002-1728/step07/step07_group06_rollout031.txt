        h_clipped = np.clip(h, 0, 1)
        box_diff = (h - h_clipped) ** 2
        box_penalty = box_diff.sum() * lambda_box
        
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Define objective for inner optimization
        def objective_func(c):
            h = X @ c
            h_clipped = np.clip(h, 0, 1)
            
            sum_h = h.sum() * dx
            constraint_int_term = (sum_h - 1.0) ** 2 * lambda_int
            
            box_diff = (h - h_clipped) ** 2
            box_penalty_term = box_diff.sum() * lambda_box
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5_term = np.max(corr) * dx
            
            return c5_term + constraint_int_term + box_penalty_term
        
        # Optimize coefficients with L-BFGS-B
        result = opt.minimize(
            objective_func, coeffs,
            method='L-BFGS-B',
            options={'maxiter': 10, 'ftol': 1e-6}
        )
        
        if result.success:
            coeffs = result.x
        
        # Update penalty weights
        h = X @ coeffs
        h_clipped = np.clip(h, 0, 1)
        
        # Integral constraint
        sum_h = h.sum() * dx
        if abs(sum_h - 1.0) > 1e-6:
            lambda_int *= 1.2
        else:
            lambda_int *= 0.95
        
        # Box constraints
        viols = np.sum(np.abs(h - h_clipped) > 1e-6)
        if viols > 0:
            lambda_box *= 1.2
        else:
            lambda_box *= 0.95
    
    # Final cleanup: project to feasible set
    if best_h is None:
        best_h = X @ coeffs
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * (n_points / 2.0)  # Normalize density
    
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (best_h.tolist(), final_c5, n_points)
