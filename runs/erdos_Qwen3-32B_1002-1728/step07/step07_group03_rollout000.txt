        offspring = []
        for parent in population:
            if time.time() - start_time > budget_s - 5:
                break
            child = parent + np.random.randn(half_n) * sigma
            child = np.clip(child, 0, 1)
            current_sum = child.sum()
            if current_sum != 0:
                child = child / current_sum * target_sum_half
            child = np.clip(child, 0, 1)
            offspring.append(child)
        offspring = np.array(offspring)
        if len(offspring) == 0:
            continue
        offspring_objectives = [objective(c) for c in offspring]
        combined = list(zip(offspring_objectives, offspring))
        combined.sort()
        population = np.array([c for val, c in combined[:popsize]])
        current_gen_min = combined[0][0]
        if current_gen_min < best_c5:
            best_h_sym = combined[0][1]
            best_c5 = current_gen_min
        if gen > 0:
            progress = (prev_best - current_gen_min) / prev_best if prev_best != 0 else 0
            if progress < 0.01:
                sigma *= 1.1
            else:
                sigma *= 0.9
        prev_best = current_gen_min
    
    # Local optimization
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    
    def local_obj(x):
        h_full = np.concatenate([x, x[::-1]])
        h_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx
    
    try:
        result = minimize(
            local_obj,
            x0=best_h_sym,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12}
        )
        if result.success:
            new_h_sym = result.x
            new_h_sym = np.clip(new_h_sym, 0, 1)
            current_sum = new_h_sym.sum()
            if current_sum != 0:
                new_h_sym = new_h_sym / current_sum * target_sum_half
                new_h_sym = np.clip(new_h_sym, 0, 1)
            val = objective(new_h_sym)
            if val < best_c5:
                best_h_sym = new_h_sym
                best_c5 = val
    except:
        pass
    
    # Final constraint enforcement
    h_full = np.concatenate([best_h_sym, best_h_sym[::-1]])
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        h_full = h_full / current_sum * target_sum_full
        h_full = np.clip(h_full, 0, 1)
    
    return (h_full.tolist(), best_c5, n_points)
