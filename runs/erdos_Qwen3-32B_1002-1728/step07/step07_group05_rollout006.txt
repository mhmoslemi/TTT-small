import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)

    population_size = 30
    np.random.seed(seed)
    population = [h_half_initial.copy()]
    for _ in range(population_size - 1):
        x = h_half_initial.copy()
        x += 0.1 * np.random.randn(*x.shape)
        x = repair_half(x, target_sum_half)
        population.append(x)

    F = 0.5
    CR = 0.9
    de_time_budget = budget_s * 0.85
    time_end = start_time + de_time_budget

    while time.time() < time_end:
        for i in range(population_size):
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = idxs
            mutant = population[a] + F * (population[b] - population[c])
            mask = np.random.rand(half_n) < CR
            trial = np.where(mask, mutant, population[i])
            trial = repair_half(trial, target_sum_half)
            current_fitness = compute_c5(population[i])
            trial_fitness = compute_c5(trial)
            if trial_fitness < current_fitness:
                population[i] = trial
                if trial_fitness < best_c5:
                    best_h_half[:] = trial
                    best_c5 = trial_fitness
        if time.time() > time_end:
            break

    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum_half
        else:
            x_scaled = x_clipped.copy()
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
    for _ in range(3):
        try:
            result = minimize(
                bounded_objective, x0=best_h_half, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
