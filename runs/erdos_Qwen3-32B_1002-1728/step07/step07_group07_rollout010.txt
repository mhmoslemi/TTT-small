            problem.solve(solver='SCS', verbose=False, max_iter=int(0.8 * budget_s))
        except Exception:
            try:
                problem.solve(solver='ECOS', verbose=False, max_iter=int(0.7 * budget_s))
            except Exception:
                print("Fallback to repaired initial solution")
    
    h_opt = h_repaired if h.value is None else np.clip(h.value, 0, 1)
    if h.value is not None:
        sum_opt = h_opt.sum()
        if abs(sum_opt - target_sum) > 1e-6 and sum_opt != 0:
            h_opt = (h_opt / sum_opt) * target_sum
            h_opt = np.clip(h_opt, 0, 1)

    # Final C₅ evaluation
    corr = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr)

    return (h_opt.tolist(), c5_bound, n_points)
