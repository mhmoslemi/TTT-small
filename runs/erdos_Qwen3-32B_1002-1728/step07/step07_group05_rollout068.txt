        problem.solve(solver=cp.MOSEK, verbose=False, max_iter=100, warm_start=True)
        elapsed = time.time() - start_time
    except Exception as e:
        pass  # fall back to initial solution

    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        best_h_half = h_half_initial
    else:
        best_h_half = np.clip(h_half.value, 0, 1)

    # Final evaluation
    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx

    return (h_full.tolist(), final_c5, n_points)
