        return np.max(corr) * dx

    # Project and repair initial solution
    h_half_initial = (h_full_initial[:half_n] + h_full_initial[-half_n:][::-1]) / 2
    h_half_initial = repair_h_half(h_half_initial)

    # Generate diverse starting points
    np.random.seed(seed)
    def generate_random_half():
        samples = np.random.rand(half_n)
        return repair_h_half(samples)

    n_starts = 8
    starts = [generate_random_half() for _ in range(n_starts)]

    # Coarse-to-fine optimization
    best_c5 = float('inf')
    best_h_half = h_half_initial.copy()

    # Phase 1: Coarse optimization with multi-start COBYLA
    for start in starts:
        try:
            bounds = [(0.0, 1.0)] * half_n
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
            res = minimize(
                objective, x0=start, method='COBYLA', 
                bounds=bounds, constraints=constraints, 
                options={'maxiter': 100, 'tol': 1e-8}
            )
            if res.success:
                candidate = repair_h_half(res.x)
                c5 = objective(candidate)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h_half = candidate
        except:
            continue

    # Phase 2: Local refinement with L-BFGS-B
    while time.time() - start_time < budget_s * 0.9:
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        pert = best_h_half.copy() + (np.random.rand(half_n) - 0.5) * 0.3 * frac_remaining
        pert = repair_h_half(pert)
        
        try:
            res = minimize(
                objective, x0=pert, method='L-BFGS-B',
                bounds=[(0, 1)] * half_n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if res.success:
                new_candidate = repair_h_half(res.x)
                c5 = objective(new_candidate)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h_half = new_candidate
        except:
            continue

    # Final validation and return
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    
    # Ensure exact sum constraint
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
