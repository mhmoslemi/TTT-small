def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = np.array(initial_h_values)
    n = len(initial_h)
    dx = 2.0 / n
    L = 2.0
    N_fourier = 15

    x = np.arange(n) * dx
    a = np.zeros(N_fourier)
    b = np.zeros(N_fourier)
    for k in range(N_fourier):
        k1 = k + 1
        cos_t = np.cos(k1 * np.pi * x / L)
        a[k] = np.sum(initial_h * cos_t) * dx
        sin_t = np.sin(k1 * np.pi * x / L)
        b[k] = np.sum(initial_h * sin_t) * dx
    ab = np.concatenate([a, b])

    best_c5 = np.inf
    best_h = None
    lambda_ = 1e3

    while time.time() - start_time < budget_s:
        def objective(ab_flat):
            a = ab_flat[:N_fourier]
            b = ab_flat[N_fourier:]
            h = 0.5 * np.ones(n)
            for i in range(n):
                for k in range(N_fourier):
                    k1 = k + 1
                    h[i] += a[k] * np.cos(k1 * np.pi * x[i] / L) + b[k] * np.sin(k1 * np.pi * x[i] / L)
                h[i] = np.clip(h[i], 0, 1)
            sum_h = np.sum(h) * dx
            integral_penalty = (sum_h - 1.)**2
            box_penalty = np.sum(np.maximum(0, h - 1)**2 + np.maximum(0, -h)**2)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5 + lambda_ * (integral_penalty + 10 * box_penalty)

        res = opt.minimize(objective, ab, method='L-BFGS-B', options={'maxiter': 40})
        if res.success:
            ab = res.x
            a_new = ab[:N_fourier]
            b_new = ab[N_fourier:]
            h_new = 0.5 * np.ones(n)
            for i in range(n):
                for k in range(N_fourier):
                    k1 = k + 1
                    h_new[i] += a_new[k] * np.cos(k1 * np.pi * x[i] / L) + b_new[k] * np.sin(k1 * np.pi * x[i] / L)
                h_new[i] = np.clip(h_new[i], 0, 1)
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        sum_h = np.sum(h_new) * dx
        ival = max(1e-4, np.sum(np.maximum(0, h_new - 1)**2 + np.maximum(0, -h_new)**2))
        if abs(sum_h - 1) > 5e-2 or ival > 1e-6:
            lambda_ = min(lambda_ * 1.15, 1e4)
        else:
            lambda_ = max(lambda_ * 0.95, 1e2)

    return (best_h.tolist(), best_c5, n)
