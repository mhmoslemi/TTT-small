import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use 10 blocks since 70 points / 10 = 7 samples per block
    m = 10
    block_length = n_points // m
    
    # Generate initial block values from initial_h_values
    initial_blocks = np.array([np.mean(h_full_initial[i*block_length:(i+1)*block_length]) for i in range(m)])
    initial_blocks = np.clip(initial_blocks, 0, 1)
    # Repair sum constraint
    block_sum = m / 2.0
    scale = block_sum / max(1e-6, initial_blocks.sum())
    initial_blocks = np.clip(initial_blocks * scale, 0, 1)
    
    population_size = 20
    population = [initial_blocks.copy()]
    for _ in range(population_size - 1):
        candidate = initial_blocks + np.random.normal(0, 0.1, m)
        candidate = np.clip(candidate, 0, 1)
        scale = block_sum / max(1e-6, candidate.sum())
        candidate = np.clip(candidate * scale, 0, 1)
        population.append(candidate)
    
    def evaluate(blocks):
        h = np.repeat(blocks, block_length)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # GA parameters
    generations = 100
    mutation_rate = 0.1
    
    best_blocks = initial_blocks.copy()
    best_c5 = evaluate(best_blocks)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Selection: top 25%
        fitness = np.array([evaluate(p) for p in population])
        sorted_idx = np.argsort(fitness)
        elites = [population[i] for i in sorted_idx[:population_size//4]]
        
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Tournament selection
            p1, p2 = np.random.choice(np.arange(population_size), 2)
            parent1, parent2 = population[p1], population[p2]
            
            # Crossover
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                idxs = np.random.choice(m, size=int(0.3 * m), replace=False)
                child[idxs] += np.random.normal(0, 0.2, size=len(idxs))
                child = np.clip(child, 0, 1)
                
            # Repair sum constraint
            scale = block_sum / max(1e-6, child.sum())
            child = np.clip(child * scale, 0, 1)
            new_population.append(child)
            
        population = new_population
        
        # Update best
        current_fitness = [evaluate(p) for p in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_blocks = population[best_idx].copy()
            best_c5 = current_fitness[best_idx]
    
    # Final optimization on full scale
    h_ga = np.repeat(best_blocks, block_length)
    h_ga = np.clip(h_ga, 0, 1)
    
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr) * dx
    
    # SLSQP refinement
    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    bounds = [(0.0, 1.0)] * n_points
    
    result = minimize(

        bounds=bounds, constraints=constraints,
