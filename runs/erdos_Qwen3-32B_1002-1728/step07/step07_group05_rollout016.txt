        problem.solve(**solver_params)
    except Exception:
        pass

    if h_half.value is not None and C5.value is not None and np.isfinite(C5.value):
        h_full = np.zeros(n_points)
        h_full[:half_n] = np.clip(h_half.value, 0, 1)
        h_full[-half_n:] = h_full[:half_n][::-1]
        c5_bound = C5.value
    else:
        h_full = h_full_initial
        h_half_opt = h_full_initial[:half_n]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5_bound = np.max(corr) * dx

    return (h_full.tolist(), c5_bound, n_points)
