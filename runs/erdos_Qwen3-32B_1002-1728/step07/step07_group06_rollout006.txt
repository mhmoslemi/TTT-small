import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped.copy()
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    h = repair_h(h_initial)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    learning_rate = 0.05
    decay_rate = 0.999
    momentum = 0.9
    velocity = np.zeros(n)

    while time.time() - start_time < budget_s:
        # Compute convolution for max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        current_k = max_idx - (n - 1)

        # Compute gradient for current_k
        grad = np.zeros(n)
        for m in range(n):
            if 0 <= m + current_k < n:
                grad[m] += 1 - h[m + current_k]
            if 0 <= m - current_k < n:
                grad[m] -= h[m - current_k]
        
        # Apply momentum
        velocity = momentum * velocity - learning_rate * grad
        h_step = h + velocity
        h = repair_h(h_step)

        # Evaluate and update best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            learning_rate *= 1.02  # Increase if improving
        else:
            learning_rate *= decay_rate  # Decay otherwise

    # Final repair
    h = repair_h(h)
    final_c5 = evaluate_c5(h, dx)
    return (h.tolist(), min(best_c5, final_c5), n)
