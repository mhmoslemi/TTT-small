from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum_full
    h_full_initial = np.clip(h_full_initial, 0, 1)
    current_sum = h_full_initial.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        scale_factor = target_sum_full / current_sum
        h_full_initial = h_full_initial * scale_factor
        h_full_initial = np.clip(h_full_initial, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    final_h_half = h_full_initial[:half_n].copy()
    best_c5 = objective(final_h_half)

    # Phase 1: Global search with aggressive DE and adaptive parameters

    popsize = 20
    maxiter = 60
    de_budget = budget_s * 0.3  # Reduced to share time with local search
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=maxiter, 
            popsize=popsize, mutation=(0.3, 1.3), recombination=0.95, 
            seed=seed, workers=1, tol=1e-7, maxfun=int(1e5), 
            disp=False
        )
        if de_result.success and de_result.fun < best_c5:
            best_h_half = np.clip(de_result.x, 0, 1)
            best_h_half = (best_h_half / best_h_half.sum()) * target_sum_half
            best_h_half = np.clip(best_h_half, 0, 1)
            final_h_half = best_h_half
            best_c5 = objective(final_h_half)
    except:
        pass

    # Phase 2: Multiple local optimization strategies with adaptive perturbations
   _iters = 0
    while time.time() - start_time < budget_s - 20:

        perturbed = final_h_half.copy()

        noise = np.random.normal(0, noise_scale, len(perturb_indices))
        perturbed[perturb_indices] += noise
        perturbed = np.clip(perturbed, 0, 1)

        if pert_sum > 1e-8:

        for method in ['Nelder-Mead', 'COBYLA', 'SLSQP']:
            try:
