def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = 1/2 * integral

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial half repair
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)

    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)

    # ----------------- Simulated Annealing Phase ----------------
    current_h = best_h_half.copy()
    current_c5 = best_c5

    sa_end_time = start_time + 0.4 * budget_s
    temperature = 1.0
    while time.time() < sa_end_time:
        delta = 0.1 * np.random.randn(*current_h.shape)
        new_h = current_h + delta
        new_h = repair_half(new_h, target_sum_half)
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / temperature) > np.random.rand():
            current_h = new_h
            current_c5 = new_c5
        temperature *= 0.99
    if current_c5 < best_c5:
        best_h_half = current_h
        best_c5 = current_c5

    # ----------------- Particle Swarm Optimization Phase -------
    particle_count = 25
    np.random.seed(seed + 1)
    particles = [best_h_half.copy() + 0.1 * np.random.randn(*best_h_half.shape) for _ in range(particle_count)]
    particles = [repair_half(p, target_sum_half) for p in particles]
    particles = np.array(particles)
    velocities = np.zeros_like(particles)
    pbest_positions = particles.copy()
    pbest_c5 = [compute_c5(p) for p in particles]
    gbest_idx = np.argmin(pbest_c5)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]

    pso_end_time = sa_end_time + 0.35 * budget_s
    while time.time() < pso_end_time:
        for i in range(particle_count):
            r1, r2 = np.random.rand(2)
            vel = 0.9 * velocities[i] + r1 * 0.5 * (pbest_positions[i] - particles[i]) + r2 * 1.5 * (gbest_position - particles[i])
            vel = np.clip(vel, -0.1, 0.1)
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = repair_half(new_pos, target_sum_half)
            new_c5 = compute_c5(new_pos)
            if new_c5 < pbest_c5[i]:
                pbest_positions[i] = new_pos
                pbest_c5[i] = new_c5
            particles[i] = new_pos
        pbest_c5_array = np.array(pbest_c5)
        gbest_idx = np.argmin(pbest_c5_array)
        gbest_position = pbest_positions[gbest_idx]
        gbest_c5 = pbest_c5_array[gbest_idx]
    if gbest_c5 < best_c5:
        best_h_half = gbest_position
        best_c5 = gbest_c5

    # ----------------- Local Refinement with SLSQP ---------------
    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum_half
        else:
            x_scaled = x_clipped.copy()
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    for _ in range(3):
        try:
            result = minimize(
                bounded_objective, x0=best_h_half, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
