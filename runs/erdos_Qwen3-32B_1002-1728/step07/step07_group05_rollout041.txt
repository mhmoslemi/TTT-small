    h_half_initial = repair_half(h_half_initial, target_sum_half)
    population.append(h_half_initial)
    for _ in range(population_size - 1):
        x = h_half_initial.copy()
        x += 0.1 * np.random.randn(*x.shape)
        x = repair_half(x, target_sum_half)
        population.append(x)

    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover probability
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)

    # Differential Evolution Loop
    time_end_de = start_time + budget_s * 0.9
    while time.time() < time_end_de:
        for i in range(population_size):
            indices = list(range(population_size))
            a, b, c = np.random.choice(indices, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(population[i])
            j = np.random.randint(0, half_n)
            for k in range(half_n):
                if np.random.rand() < CR or k == j:
                    trial[k] = mutant[k]
            trial_repaired = repair_half(trial, target_sum_half)
            c5_trial = compute_c5(trial_repaired)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h_half[:] = trial_repaired.copy()
            if c5_trial < compute_c5(population[i]):
                population[i] = trial_repaired.copy()
            if time.time() > time_end_de:
                break
        if time.time() > time_end_de:
            break

    # Local Optimization
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total == 0:
            return float('inf')
        x_scaled = (x_clipped / total) * target_sum_half
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    for _ in range(3):
        try:
            result = minimize(
                bounded_objective, x0=best_h_half, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half[:] = x_opt.copy()
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
