import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points / 4.0
    
    def compute_sum_ss(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.sum(corr**2) * dx**2
    
    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return np.sum(x) - target_sum_half
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(half_n)]
    
    h_half_initial = h_initial[:half_n].copy()
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)
    
    # Phase 1: Sum of squares optimization
    t0 = time.time()
    result_ss = minimize(
        fun=compute_sum_ss,
        x0=best_h_half,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    if time.time() - start_time > budget_s * 0.8:
        result_ss.success = False

        h_half_ss = result_ss.x
        c5_ss = compute_c5(h_half_ss)
        if c5_ss < best_c5:
            best_h_half = h_half_ss
            best_c5 = c5_ss
    
    # Phase 2: Exact C₅ refinement
    result_c5 = minimize(
        fun=compute_c5,

        method='SLSQP',

        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-10}

    if time.time() - start_time > budget_s * 0.95:

        if c5_c5 < best_c5:
