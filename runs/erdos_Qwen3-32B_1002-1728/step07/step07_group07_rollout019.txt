import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair initial solution
    h_repaired = np.clip(h_initial, 0, 1)
    h_repaired = h_repaired / h_repaired.sum() * target_sum
    h_repaired = np.clip(h_repaired, 0, 1)

    def objective(h_flat):
        h = np.clip(h_flat, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_h(h):
        h = np.clip(h, 0, 1)
        if abs(h.sum() - target_sum) > 1e-6:
            h = (h / h.sum()) * target_sum
        return np.clip(h, 0, 1)

    # Asymmetric variant generator with controlled perturbation
    def generate_variant(h_base, scale=0.15):
        pert = np.random.normal(0, scale, len(h_base))
        h_pert = np.clip(h_base + pert, 0, 1)
        h_pert = repair_h(h_pert)
        return h_pert

    # Generate diverse variants
    variants = [h_repaired]
    for _ in range(12):
        variant = generate_variant(h_repaired)
        variants.append(variant)
        if time.time() - start_time > budget_s * 0.2:
            break

    # Global optimization with adaptive DE parameters
    for variant in variants:
        if time.time() - start_time > budget_s:
            break

        bounds = [(0.0, 1.0)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})

        def callback(x, convergence):
            return time.time() - start_time > budget_s * 0.9

        # Differential evolution with adaptive strategy
        de_result = differential_evolution(
            objective,
            bounds,
            strategy="best1bin",
            maxiter=15,
            popsize=15,
            recombination=0.7,
            mutation=(0.5, 1.5),
            tol=1e-5,
            updating="deferred",
            x0=variant,
            constraints=cons,
            seed=seed,
            callback=callback
        )

        if de_result.success:
            h_candidate = repair_h(de_result.x)
            c5_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if c5_candidate < np.max(np.correlate(variants[0], 1 - variants[0], mode='full')) * dx:
                variants[0] = h_candidate

    # Final convex relaxation for global constraint satisfaction
    h_cvx = cp.Variable(n_points)
    t = cp.Variable()
    objective_cvx = cp.Minimize(t)
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == target_sum
    ]
    for k in range(n_points):
        # Overlap constraints for adjacent shifts (simplification)
        overlap = h_cvx[:-k] * (1 - h_cvx[k:]) if k < n_points else 0
        constraints.append(cp.sum(overlap) * dx <= t)
    problem = cp.Problem(objective_cvx, constraints)
    if time.time() - start_time < budget_s * 0.9:

        h_cvx = repair_h(h_cvx)

    # Final selections
    best_h = min(variants + [h_cvx],
