from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Symmetric h handling
    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Repair h_sym to sum to target_sum_half and be within [0,1]
    def repair_h_sym(h_sym):
        h_clipped = np.clip(h_sym, 0, 1)
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0:
            return h_clipped.copy()
        h_scaled = (h_clipped / sum_clipped) * target_sum_half
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    # Initialize h_sym from initial_h_values
    h_sym = h_full_initial[:half_n].copy()
    h_sym = repair_h_sym(h_sym)

    current_best_h_sym = h_sym.copy()
