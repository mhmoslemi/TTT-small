import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def repair_half(x, target_sum):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < 1e-12:
            x_clipped[0] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        return np.clip(x_scaled, 0, 1)

    def create_full_h(sym_vars):
        return np.concatenate((sym_vars, sym_vars[::-1]))

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial repair
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)
    best_h_half = h_half_initial.copy()
    best_c5 = compute_c5(best_h_half)

    # Simulated Annealing
    sa_time_budget = budget_s * 0.35
    time_end_sa = start_time + sa_time_budget
    temp = 1.0
    cooling_rate = 0.99
    while time.time() < time_end_sa:
        current = best_h_half.copy()
        perturbation = 0.2 * np.random.randn(*current.shape)
        perturbation = np.clip(perturbation, -0.25, 0.25)
        candidate = current + perturbation
        candidate = repair_half(candidate, target_sum_half)
        delta = compute_c5(candidate) - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-6)):
            best_h_half = candidate.copy()
            best_c5 = min(best_c5, compute_c5(best_h_half))
        temp *= cooling_rate

    # Particle Swarm Optimization
    swarm_size = 20
    np.random.seed(int(seed) + 1)
    particles = [repair_half(best_h_half + 0.1 * np.random.randn(half_n), target_sum_half) for _ in range(swarm_size)]
    vels = [0.01 * np.random.randn(half_n) for _ in range(swarm_size)]
    personal_best = [p.copy() for p in particles]
    personal_c5 = [compute_c5(p) for p in particles]
    gbest_idx = np.argmin(personal_c5)
    gbest = particles[gbest_idx].copy()
    gbest_c5 = personal_c5[gbest_idx]
    omega = 0.7
    phi_p = 0.5
    phi_g = 1.5
    time_end_pso = start_time + budget_s * 0.8

    while time.time() < time_end_pso:
        for i in range(swarm_size):
            r_p, r_g = np.random.rand(2)
            vels[i] = omega * vels[i] + phi_p * r_p * (personal_best[i] - particles[i]) + phi_g * r_g * (gbest - particles[i])
            particles[i] += vels[i]
            particles[i] = repair_half(particles[i], target_sum_half)
            c5 = compute_c5(particles[i])
            if c5 < personal_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_c5[i] = c5
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5

        if gbest_c5 < best_c5:
            best_h_half = gbest.copy()
            best_c5 = gbest_c5

    # Local refinement

        return np.max(corr_vals) * dx

                best_c5 = c5_candidate

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
