    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair_full(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Initial repair
    h_initial = repair_full(h_full_initial, target_sum)

    best_vec = h_initial
    best_c5 = compute_c5(best_vec)

    # --- Genetic Algorithm Phase ---
    population_size = 40
    np.random.seed(seed)
    population = []
    for _ in range(population_size):
        x = h_initial.copy()
        x += 0.08 * np.random.randn(*x.shape)
        x = repair_full(x, target_sum)
        population.append(x)

    ga_time_budget = budget_s * 0.45
    time_end_ga = start_time + ga_time_budget
    gen = 0
    while time.time() < time_end_ga:
        gen += 1
        fitness = []
        for indiv in population:
            c5_val = compute_c5(indiv)
            fitness.append(c5_val)

        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_vec = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        new_population = population[:2]

        for _ in range(population_size - 2):
            # Tournament selection
            cand_ids = np.random.choice(population_size, 3, replace=False)
            parents = [population[i] for i in cand_ids]
            parent1 = parents[np.argmin([compute_c5(p) for p in parents])]
            cand_ids = np.random.choice(population_size, 3, replace=False)
            parents = [population[i] for i in cand_ids]
            parent2 = parents[np.argmin([compute_c5(p) for p in parents])]

            # Blend crossover
            child = (parent1 + parent2) / 2.0
            # Adaptive mutation
            noise_scale = 0.1 * (1.0 - min(max(gen / 500, 0.3), 1.0))
            mutation_mask = (np.random.rand(*child.shape) < 0.08)
            if mutation_mask.any():
                child[mutation_mask] += noise_scale * np.random.randn(np.sum(mutation_mask))
                child[mutation_mask] = np.clip(child[mutation_mask], 0, 1)
            child = repair_full(child, target_sum)
            new_population.append(child)

        population = new_population

    # --- Local Refinement ---
    def bounded_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum
        else:
            x_scaled = x_clipped.copy()
        x_scaled = np.clip(x_scaled, 0, 1)
        h = x_scaled
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for _ in range(5):
        try:
            result = minimize(
                bounded_objective, x0=best_vec, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
            if result.success:
                x_opt = repair_full(result.x, target_sum)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_vec = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = best_vec.copy()
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
