        x_clipped = np.clip(x, 0, 1)
        x_scaled = (x_clipped / x_clipped.sum()) * target_sum
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # Use multiple optimization methods
    methods = ['Nelder-Mead', 'Powell', 'SLSQP']  # Different local methods
    for method in methods:
        try:
            result = minimize(
                objective,
                x0=best_h_half,
                method=method,
                bounds=[(0., 1.) for _ in range(half_n)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
                options={'maxiter': 200, 'ftol': 1e-10, 'xatol': 1e-8, 'disp': False}
            )
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    # Final repair and validation
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
