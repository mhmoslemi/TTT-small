The repair function ensures the solution is feasible at each scale, and SLSQP is used for optimization. The time budget is checked at the start of each scale to avoid exceeding the allowed time.

import numpy as np
import time
from scipy.optimize import minimize

def downscale_h(h_fine, n_coarse):
    k = len(h_fine) // n_coarse
    h_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        start = i * k
        end = start + k
        h_coarse[i] = np.mean(h_fine[start:end])
    return h_coarse

def upscale_h(h_coarse, k):
    return np.repeat(h_coarse, k)

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    dx_initial = 2.0 / n_points_initial

    def repair_h(h, n_points):
        target_sum = n_points / 2.0
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        scale_factor = target_sum / current_sum
        h_scaled = h_clipped * scale_factor
        h_scaled_clipped = np.clip(h_scaled, 0, 1)
        return h_scaled_clipped

    h = h_initial.copy()
    h = repair_h(h, n_points_initial)

    # Define multi-scale resolutions
    scales = [10, 20, 40, 70]
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx_initial

    for scale in scales:
        if time.time() - start_time > budget_s:
            break

        current_scale_n = len(h)
        target_scale_n = scale
        if target_scale_n < current_scale_n:
            h = downscale_h(h, target_scale_n)
        else:

            corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        bounds = [(0.0, 1.0)] * current_n

            method='SLSQP',

                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()
