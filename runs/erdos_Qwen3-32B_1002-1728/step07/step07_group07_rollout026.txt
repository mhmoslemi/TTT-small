    for (start, end, value) in candidate_blocks:

def repair_h(h_array, target_sum):
    n_points = len(h_array)
    dx = 2.0 / n_points
    h_array = np.clip(h_array, 0, 1)
    current_sum = h_array.sum() * dx  # Because the sum of h_array * dx is 1
    if abs(current_sum - target_sum) > 1e-8:
        scaling_factor = target_sum / current_sum
        h_array = h_array * scaling_factor
        h_array = np.clip(h_array, 0, 1)
    return h_array

def evaluate_c5(h_array):
    n_points = len(h_array)
    dx = 2.0 / n_points
    corr = np.correlate(h_array, 1 - h_array, mode='full')
    return np.max(corr) * dx

import numpy as np
import time
import random

def h_to_blocks(h_array):
    blocks = []
    n = len(h_array)
    if n == 0:
        return blocks
    current_value = h_array[0]
    start = 0
    for i in range(1, n):
        if abs(h_array[i] - current_value) > 1e-8:
            blocks.append((start, i, current_value))
            current_value = h_array[i]
            start = i
    blocks.append((start, n, current_value))
    return blocks

def blocks_to_h_array(candidate_blocks, n_points):
    h_array = np.zeros(n_points)
    for (start, end, val) in candidate_blocks:
        h_array[start:end] = val
    return h_array

def repair_h(h_array, target_sum):
    n = len(h_array)
    dx = 2.0 / n
    h_array = np.clip(h_array, 0.0, 1.0)
    total = np.sum(h_array) * dx
    if abs(total - target_sum) > 1e-6:
        h_array *= (target_sum / total) if total != 0 else 1.0
        h_array = np.clip(h_array, 0.0, 1.0)
    return h_array

def evaluate_c5(h_array):
    n = len(h_array)
    dx = 2.0 / n
    corr = np.correlate(h_array, 1 - h_array, mode='valid')
    return np.max(corr) * dx

def generate_split(candidate_blocks):
    if len(candidate_blocks) < 2:
        return candidate_blocks[:]
    block_idx = random.randint(0, len(candidate_blocks) - 1)
    start, end, val = candidate_blocks[block_idx]
    if end - start < 2:
        return candidate_blocks[:]
    split = random.randint(start + 1, end - 1)
    new_blocks = candidate_blocks[:block_idx]
    new_blocks.append((start, split, val))
    new_blocks.append((split, end, val))
    new_blocks.extend(candidate_blocks[block_idx + 1:])
    return new_blocks

def generate_move(candidate_blocks):
    if len(candidate_blocks) < 2:
        return candidate_blocks[:]
    break_idx = random.randint(0, len(candidate_blocks) - 2)
    a_start, a_end, a_val = candidate_blocks[break_idx]
    b_start, b_end, b_val = candidate_blocks[break_idx + 1]
    move_by = random.randint(-2, 2)
    new_a_end = a_end + move_by
    new_b_start = new_a_end
    new_a_end = max(a_start + 1, min(new_a_end, b_start - 1))
    new_blocks = candidate_blocks[:break_idx]
    new_blocks.append((a_start, new_a_end, a_val))
    new_blocks.append((new_b_start, b_end, b_val))
    new_blocks.extend(candidate_blocks[break_idx + 2:])
    return new_blocks

def generate_value(candidate_blocks):
    if len(candidate_blocks) < 1:
        return candidate_blocks[:]
    block_idx = random.randint(0, len(candidate_blocks) - 1)
    start, end, val = candidate_blocks[block_idx]
    delta = random.uniform(-0.05, 0.05)
    new_val = max(0.0, min(1.0, val + delta))
    new_blocks = candidate_blocks[:block_idx]
    new_blocks.append((start, end, new_val))
    new_blocks.extend(candidate_blocks[block_idx + 1:])
    return new_blocks
