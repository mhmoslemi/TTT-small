    x_vals = np.linspace(0, 2, n, endpoint=False)

    m = n // 4
    k_vals = np.arange(1, m+1)
    cos_terms = np.cos(k_vals * np.pi * x_vals.reshape(-1, 1) / 2).T
    sin_terms = np.sin(k_vals * np.pi * x_vals.reshape(-1, 1) / 2).T

    def compute_fourier_coefficients(h):
        target = h - 0.5
        lhs = np.hstack([cos_terms, sin_terms])
        coeffs, _, _, _ = np.linalg.lstsq(lhs, target, rcond=None)
        return coeffs[:m], coeffs[m:]

    a0, b0 = compute_fourier_coefficients(h_samples)
    coeffs_initial = np.concatenate([a0, b0])

    def objective(coeffs):
        a = coeffs[:m]
        b = coeffs[m:]
        h = 0.5 + cos_terms @ a + sin_terms @ b
        h = np.clip(h, 0, 1)
        curr_sum = h.sum()
        if abs(curr_sum - target_sum) > 1e-6:
            h = (h / curr_sum) * target_sum
            h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def sum_constraint(coeffs):
        a = coeffs[:m]
        b = coeffs[m:]
        return (0.5 * n + cos_terms @ a.sum() + sin_terms @ b.sum()) - target_sum

    constraints = {'type': 'eq', 'fun': sum_constraint}

    result = minimize(
        objective,
        coeffs_initial,
        method='SLSQP',
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    a_opt = result.x[:m]
    b_opt = result.x[m:]
    h_opt = 0.5 + cos_terms @ a_opt + sin_terms @ b_opt
    h_opt = np.clip(h_opt, 0, 1)
    h_opt_sum = h_opt.sum()
    if abs(h_opt_sum - target_sum) > 1e-6:
        h_opt = (h_opt / h_opt_sum) * target_sum
        h_opt = np.clip(h_opt, 0, 1)

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_opt.tolist(), c5_bound, n)
