            x_val = np.clip(x_val, 0, 1)
            c5_val = objective_local(x_val)
            if c5_val < best_c5:
                best_h = x_val
                best_c5 = c5_val
                remained = budget_s - (time.time() - start_time)

    # Final polishing
    if remained > 100:
        result = minimize(
            objective_local, x0=best_h, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            x_pol = np.clip(result.x, 0, 1)
            c5_pol = objective_local(x_pol)
            if c5_pol < best_c5:
                best_h = x_pol
                best_c5 = c5_pol

    # Final constraint enforcement
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h
    h_full[-half_n:] = best_h[::-1]
    h_full = np.clip(h_full, 0, 1)
    total_sum = h_full.sum()
    if not np.isclose(total_sum, target_sum_full):
        h_full = h_full / total_sum * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
