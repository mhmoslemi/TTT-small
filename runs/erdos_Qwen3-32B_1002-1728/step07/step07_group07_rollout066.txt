    freq_mags = []
    for k in range(n_points):
        n = np.arange(n_points)

    ]
    
    # Precompute DFT real and imaginary parts
    k = np.arange(n_points)
    n = np.arange(n_points)
    K, N = np.meshgrid(k, n, indexing='ij')
    real_weights = np.cos(2 * np.pi * K * N / n_points)
    imag_weights = -np.sin(2 * np.pi * K * N / n_points)
    
    # Compute H_real and H_imag for all frequencies
    H_real = real_weights @ h
    H_imag = imag_weights @ h
    
    # Compute squared magnitudes
    H_squares = H_real**2 + H_imag**2
    
    # Introduce gamma and add constraints
    gamma = cp.Variable()
    constraints += [gamma >= H_squares[i] for i in range(n_points)]
    
    # Objective is to minimize the maximum squared magnitude
    objective = cp.Minimize(gamma)
    
    # Solve the problem
    problem = cp.Problem(objective, constraints)
    start_opt = time.time()
    try:
        problem.solve(solver=cp.MOSEK, verbose=False)
    except Exception:
        pass
    solve_time = time.time() - start_opt
    
    h_solution = h.value
    if h_solution is None or np.isnan(h_solution).any():
        h_solution = np.clip(h_full_initial, 0, 1)
        h_solution = h_solution / np.sum(h_solution) * target_sum
        h_solution = np.clip(h_solution, 0, 1)
    
    # Repair solution
    h_solution = np.clip(h_solution, 0, 1)
    current_sum = np.sum(h_solution)
    if abs(current_sum - target_sum) > 1e-6:
        h_solution = (h_solution / current_sum) * target_sum
        h_solution = np.clip(h_solution, 0, 1)
    
    # Compute C5
    corr = np.correlate(h_solution, 1 - h_solution, mode='full')
    c5 = np.max(corr) * dx
    
    return (h_solution.tolist(), c5, n_points)
