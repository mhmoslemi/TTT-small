        if result.success:

            if abs(x_new.sum() - target_sum_half) < 1e-6:
                c5_new = np.max(np.correlate(create_full_h(x_new), 1 - create_full_h(x_new), mode='full')) * dx
                if c5_new < best_c5:
                    final_h_half = x_new
                    best_c5 = c5_new

    # Final validation
    h_final = create_full_h(final_h_half)
    h_final = np.clip(h_final, 0, 1)
    h_final_sum = h_final.sum()
    if abs(h_final_sum - target_sum_full) > 1e-6:
        h_final = (h_final / h_final_sum) * target_sum_full
        h_final = np.clip(h_final, 0, 1)
    return (h_final.tolist(), best_c5, n_points)
