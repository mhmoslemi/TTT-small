        if h_total != 0:

        h_full = np.clip(h_full, 0, 1)
        current_h = h_full

    # Final validation
    h_full = current_h
    h_full = np.clip(h_full, 0, 1)
    h_total = h_full.sum()
    if abs(h_total - target_sum) > 1e-6:
        h_full = (h_full / h_total) * target_sum
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5 if best_c5 < np.inf else np.inf, n_points)
