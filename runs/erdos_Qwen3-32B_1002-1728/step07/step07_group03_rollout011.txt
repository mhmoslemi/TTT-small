    dx = 2.0 / n_points

    # Enforce constraints on initial solution
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = np.clip(h_half_initial, 0, 1)
    current_sum = h_half_initial.sum()
    if current_sum != 0:
        h_half_initial = (h_half_initial / current_sum) * target_sum_half
    h_half_initial = np.clip(h_half_initial, 0, 1)
    current_sum = h_half_initial.sum()
    if abs(current_sum - target_sum_half) > 1e-6:
        scale_factor = target_sum_half / current_sum
        h_half_initial = h_half_initial * scale_factor
        h_half_initial = np.clip(h_half_initial, 0, 1)

    def create_full_h(h_half):
        return np.concatenate([h_half, h_half[::-1]])

    def compute_c5(h_half):
        h = create_full_h(h_half)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # PSO parameters
    swarm_size = 50
    max_iter = 200
    cog = 2.0
    soc = 2.0

    w_min = 0.4

    while time.time() - start_time < budget_s * 0.9 and max_iter > 0:
