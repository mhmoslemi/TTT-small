        curr_pos = np.clip(curr_pos, 0, 1.0)
        curr_sum = curr_pos.sum()
        if curr_sum != 0:
            curr_pos = curr_pos * (target_sum_half / curr_sum)
        else:
            curr_pos.fill(target_sum_half / half_n)
        curr_pos = np.clip(curr_pos, 0, 1.0)
        particles_pos[i] = curr_pos
    particles_vel = np.zeros((num_particles, half_n))
    particles_pbest = particles_pos.copy()
    pbest_fitness = np.array([compute_c5(p) for p in particles_pbest])
    gbest_idx = np.argmin(pbest_fitness)
    gbest_pos = particles_pbest[gbest_idx].copy()
    best_c5 = pbest_fitness[gbest_idx]

    # PSO loop
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:  # Stop early to allow final cleanup
            break

        # Update all particles
        for i in range(num_particles):
            # Velocity update
            r1 = np.random.rand(half_n)
            r2 = np.random.rand(half_n)
            cognitive = cognitive_weight * r1 * (particles_pbest[i] - particles_pos[i])
            social = social_weight * r2 * (gbest_pos - particles_pos[i])
            new_vel = inertia * particles_vel[i] + cognitive + social
            new_vel = np.clip(new_vel, vel_min, vel_max)
            particles_vel[i] = new_vel

            # Position update
            new_pos = particles_pos[i] + new_vel
            new_pos_clipped = np.clip(new_pos, 0, 1.0)
            new_sum = new_pos_clipped.sum()
            if new_sum != 0:
                new_pos_scaled = new_pos_clipped * (target_sum_half / new_sum)
            else:
                new_pos_scaled = np.full_like(new_pos_clipped, target_sum_half / half_n)
            new_pos_final = np.clip(new_pos_scaled, 0, 1.0)
            particles_pos[i] = new_pos_final

        # Evaluate fitness and update pbest and gbest
        current_fitness = np.array([compute_c5(p) for p in particles_pos])
        improved_mask = current_fitness < pbest_fitness
        pbest_fitness[improved_mask] = current_fitness[improved_mask]
        particles_pbest[improved_mask] = particles_pos[improved_mask]
        new_gbest_idx = np.argmin(pbest_fitness)
        if pbest_fitness[new_gbest_idx] < best_c5:
            gbest_pos = particles_pbest[new_gbest_idx].copy()
            best_c5 = pbest_fitness[new_gbest_idx]

    # Final validation and cleanup
    h_full = create_full_h(gbest_pos)
    h_full = np.clip(h_full, 0, 1.0)
    h_sum = h_full.sum()
    if abs(h_sum - target_sum_full) > 1e-6:
        h_full = (h_full / h_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1.0)

    return (h_full.tolist(), best_c5, n_points)
