import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def downsample(h_fine, n_coarse):
    x_fine = np.linspace(0, 2, len(h_fine), endpoint=False)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    h_coarse = np.interp(x_coarse, x_fine, h_fine)
    dx_coarse = 2.0 / n_coarse
    current_sum = h_coarse.sum() * dx_coarse
    if current_sum != 0:
        h_coarse = h_coarse * (1.0 / current_sum)
    h_coarse = np.clip(h_coarse, 0, 1)
    return h_coarse

def upsample(h_coarse, n_fine):
    x_coarse = np.linspace(0, 2, len(h_coarse), endpoint=False)
    x_fine = np.linspace(0, 2, n_fine, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    dx_fine = 2.0 / n_fine
    current_sum = h_fine.sum() * dx_fine
    if current_sum != 0:
        h_fine = h_fine * (1.0 / current_sum)
    h_fine = np.clip(h_fine, 0, 1)
    return h_fine

def repair_h(h, n_points):
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h.copy()
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum != 0:
        h = (h / current_sum) * target_sum
    h = np.clip(h, 0, 1)
    return h
