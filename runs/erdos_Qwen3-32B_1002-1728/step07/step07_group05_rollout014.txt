import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum(h_half) = 1/2 * integral / dx

    # Define variables
    h_half = cp.Variable(half_n, lb=0.0, ub=1.0)
    C = cp.Variable()
    # Objective is to minimize C
    objective = cp.Minimize(C)

    constraints = [
        cp.sum(h_half) == target_sum
    ]

    # Build constraints for each shift k
    for k in range(n_points):
        current_sum = 0
        for a in range(n_points - k):
            # Get h_full[a] and h_full[a +k]
            # h_full[a] is h_half[ get_h_full_index(a) ]
            if a < half_n:
                idx_a = a
            else:
                idx_a = n_points - 1 - a
            # same for a +k
            pos_ak = a + k
            if pos_ak < half_n:
                idx_ak = pos_ak
            else:
                idx_ak = n_points - 1 - pos_ak
            term = h_half[idx_a] * (1 - h_half[idx_ak])
            current_sum += term
        current_sum *= dx
        constraints.append(current_sum <= C)

    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.GUROBI, verbose=False, warm_start=True)
    # or MOSEK if available?

    h_optimal_half = h_half.value
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_optimal_half[i]
        h_full[n_points - 1 - i] = h_optimal_half[i]
    # Ensure h_full is within [0,1]
    h_full = np.clip(h_full, 0, 1)
    # Compute C actual
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_full.tolist(), c5_bound, n_points)

import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum(h_half) must equal this

    # Repair initial half to sum to target_sum
    h_half_initial = h_full_initial[:half_n].copy()
    total = h_half_initial.sum()
    if total == 0:
        h_half_initial[0] = 1.0
        total = 1.0
    scale = target_sum / total
    h_half_initial = h_half_initial * scale
    h_half_initial = np.clip(h_half_initial, 0, 1)

    h_half = cp.Variable(half_n, lb=0.0, ub=1.0)
    h_half.value = h_half_initial

    C = cp.Variable()
    objective = cp.Minimize(C)

    constraints = [
        cp.sum(h_half) == target_sum
    ]

    # Build constraints for each shift k
    for k in range(n_points):
        current_sum = 0
        for a in range(n_points - k):
            # Get h_full[a] and h_full[a +k]
            if a < half_n:
                idx_a = a
            else:
                idx_a = n_points - 1 - a
            pos_ak = a + k
            if pos_ak < half_n:
                idx_ak = pos_ak
            else:
                idx_ak = n_points - 1 - pos_ak
            # Add term
            current_sum += h_half[idx_a] * (1 - h_half[idx_ak])
        current_sum *= dx
        constraints.append(current_sum <= C)

    problem = cp.Problem(objective, constraints)
    remaining_time = max(0, budget_s - (time.time() - start_time))
    try:
        problem.solve(solver=cp.GUROBI, verbose=False, TimeLimit=remaining_time)
    except:
        # Fallback if Gurobi is not available

import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum(h_half) == n_points / 4.0

    # Repair initial h_half to satisfy sum constraint
    h_half_initial = h_full_initial[:half_n].copy()
    total = h_half_initial.sum()
    if total == 0:
        h_half_initial[0] = 1.0
        total = 1.0
    scale = target_sum / total
    h_half_initial = h_half_initial * scale
    h_half_initial = np.clip(h_half_initial, 0, 1)

    # Define CVXPY variables and constraints
    h_half = cp.Variable(half_n, lb=0.0, ub=1.0)
    h_half.value = h_half_initial
    C = cp.Variable()
    objective = cp.Minimize(C)

    constraints = [
        cp.sum(h_half) == target_sum
    ]

    # Build overlap constraints
    for k in range(n_points):
        current_sum = 0
        for a in range(n_points - k):
            # Compute indices in h_half for h_full[a] and h_full[a+k]
            if a < half_n:
                idx_a = a
            else:
                idx_a = n_points - 1 - a
            pos_ak = a + k
            if pos_ak < half_n:
                idx_ak = pos_ak
            else:
                idx_ak = n_points - 1 - pos_ak
            current_sum += h_half[idx_a] * (1 - h_half[idx_ak])
        current_sum *= dx
        constraints.append(current_sum <= C)

    # Solve optimization problem
    problem = cp.Problem(objective, constraints)
    remaining_time = max(0, budget_s - (time.time() - start_time))
    try:
        problem.solve(solver=cp.GUROBI, verbose=False, TimeLimit=remaining_time)
    except:
        problem.solve(solver=cp.MOSEK, verbose=False)  # Fallback to MOSEK if Gurobi fails

    # Construct full h from optimized h_half
    h_optimal_half = np.clip(h_half.value, 0, 1)
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_optimal_half[i]
        h_full[n_points - 1 - i] = h_optimal_half[i]

    # Validate and compute final C5 bound
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_full.tolist(), c5_bound, n_points)
