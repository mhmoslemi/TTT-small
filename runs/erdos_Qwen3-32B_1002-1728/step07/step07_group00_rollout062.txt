                x1, x2 = x_current[j], x_current[j+1]
                h1, h2 = h_current[j], h_current[j+1]
                if x2 - x1 > 1e-8:
                    h_interp[i] = h1 + (x - x1) / (x2 - x1) * (h2 - h1)
        h_interp = np.clip(h_interp, 0, 1)
        current_integral = compute_integral(x_current, h_interp)
        if current_integral > 1e-8:
            h_interp = (h_interp / current_integral)
        x_current, h_current = optimize_level(x_current, h_interp, np.linspace(0.0, 2.0, n_points_uni))
        if time.time() - start_time > 0.9 * budget_s:
            break

    h_uni = np.zeros(n_points_uni)
    x_uni = np.linspace(0.0, 2.0, n_points_uni)
    for i in range(n_points_uni):
        x = x_uni[i]
        j = 0
        while j < len(x_current) - 1 and x_current[j] < x:
            j += 1
        j = min(j, len(x_current) - 2)
        x1, x2 = x_current[j], x_current[j+1]
        if x2 - x1 < 1e-8:
            h_uni[i] = h_current[j]
        else:
            frac = (x - x1) / (x2 - x1)
            h_uni[i] = h_current[j] * (1 - frac) + h_current[j+1] * frac
    h_uni = np.clip(h_uni, 0, 1)
    h_uni_sum = h_uni.sum() * dx_uni
    if abs(h_uni_sum - 1.0) > 1e-6:
        h_uni = h_uni / h_uni_sum

    corr = np.correlate(h_uni, 1 - h_uni, mode='full')
    best_c5 = np.max(corr) * dx_uni

    return (h_uni.tolist(), best_c5, n_points_uni)
