import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        return np.clip(h_clipped * (target_sum / current_sum), 0, 1)

    def evaluate_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.abs(corr).max() * dx

    population = [h_initial]
    for _ in range(10):
        h_rand = np.random.rand(n)
        h_repaired = repair_h(h_rand, target_sum)
        population.append(h_repaired)
    population = [repair_h(p, target_sum) for p in population]
    scores = [evaluate_c5(p, dx) for p in population]
    candidates = list(zip(population, scores))
    candidates.sort(key=lambda x: x[1])
    best_h, best_score = candidates[0]

    co_time = 0.5 * budget_s
    while time.time() - start_time < co_time:
        top = [c[0] for c in candidates[:3]]

            for i in range(2):
                for j in range(i+1, 3):
                    combo = alpha * top[i] + (1 - alpha) * top[j]
                    combo = repair_h(combo, target_sum)
                    score = evaluate_c5(combo, dx)
