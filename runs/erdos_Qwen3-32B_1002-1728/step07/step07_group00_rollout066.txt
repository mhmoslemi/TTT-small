import numpy as np
import time
import math
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0

    # Repair initial solution
    current_half = h_full_initial[:half_n].copy()
    current_half = np.clip(current_half, 0, 1)
    current_sum = current_half.sum()
    if current_sum != 0:
        current_half = (current_half / current_sum) * target_sum_half
    current_half = np.clip(current_half, 0, 1)

    # Initialize best
    best_half = current_half.copy()
    h_best_full = np.zeros(n_points)
    h_best_full[:half_n] = best_half
    h_best_full[-half_n:] = best_half[::-1]
    h_best_clipped = np.clip(h_best_full, 0, 1)
    best_corr = np.correlate(h_best_clipped, 1 - h_best_clipped, mode='full')
    best_c5 = np.max(best_corr) * dx
    current_c5 = best_c5

    # Setup seed and SA parameters
    random.seed(seed)
    np.random.seed(seed)
    T_initial = 1.0
    alpha = 0.999

    # Main Simulated Annealing loop
    while time.time() - start_time < budget_s - 0.1:
        time_remaining = budget_s - (time.time() - start_time)
        T = T_initial * alpha ** (budget_s - time_remaining)

        # Generate new candidate
        new_half = best_half.copy()
        scale_choice = random.randint(0, 2)
        if scale_choice == 0:  # Low-scale: Perturb single element
            idx = random.randint(0, half_n - 1)
            noise = random.gauss(0, 0.1)
            new_half[idx] += noise
        elif scale_choice == 1:  # Mid-scale: Perturb small block
            size = min(random.randint(2, 5), half_n)
            start = random.randint(0, half_n - size)
            noise = np.random.normal(0, 0.1, size)
            new_half[start:start + size] += noise
        else:  # High-scale: Perturb all elements
            noise = np.random.normal(0, 0.05, half_n)
            new_half += noise

        # Repair new_half
        new_half = np.clip(new_half, 0, 1)
        new_sum = new_half.sum()
        if new_sum != 0:
            new_half = (new_half / new_sum) * target_sum_half
        new_half = np.clip(new_half, 0, 1)

        # Evaluate new_c5
        h_new_full = np.zeros(n_points)
        h_new_full[:half_n] = new_half
        h_new_full[-half_n:] = new_half[::-1]
        h_clipped = np.clip(h_new_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        new_c5 = np.max(corr) * dx

        # Acceptance
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or random.random() < math.exp(-delta_c5 / T):
            current_c5 = new_c5
            current_half = new_half.copy()
