from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    x_vals = np.linspace(0.0, 2.0 - 1e-10, n_points)
    
    m = 3  # Number of Fourier basis terms beyond the constant term
    j_vals = np.arange(m + 1)  # [0, 1, 2, 3]
    
    # Precompute basis matrix (Phi[i][j] = cos(j * pi * x_i / 2))
    Phi = np.column_stack([np.cos(j * np.pi * x_vals / 2) for j in j_vals])
    
    # Objective function
    def objective(coeffs):
        a = np.zeros(m + 1)
        a[0] = 0.5  # Fixed coefficient for the constant term
        a[1:] = coeffs
        h = Phi @ a
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx
    
    # Linear constraints for 0 <= h_i <= 1
    Phi_upper = Phi[:, 1:]
    Phi_lower = -Phi[:, 1:]
    A_upper = np.vstack((Phi_lower, Phi_upper))
    b_upper = np.hstack((0.5 * Phi[:, 0], 1.0 - 0.5 * Phi[:, 0]))
    
    linear_constraints = LinearConstraint(A=A_upper, lb=-np.inf, ub=b_upper)
    
    # Initial guess: set coefficients to zero (h(x) = 0.5 for all x, feasible)
    coeffs_initial = np.zeros(m)
    
    # Bounds for variables (no explicit bounds, handled by constraints)
    bounds = [(-100.0, 100.0)] * m  # Wide bounds for optimization flexibility
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=coeffs_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=linear_constraints,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    
    if not result.success or time.time() - start_time > budget_s - 10:
        # Fallback to initial symmetric solution under time pressure or failure
        h_symmetric = np.concatenate([h_initial[:n_points//2], h_initial[n_points//2:][::-1]])
        return (h_symmetric.tolist(), objective(coeffs_initial), n_points)
