import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k):
        grad = np.zeros(n)
        indices = np.arange(n)
        shifted = indices + k
        valid = (shifted >= 0) & (shifted < n)
        grad[indices[valid]] += (1.0 - h[shifted[valid]]) * dx
        grad[shifted[valid]] += -h[indices[valid]] * dx
        return grad

    np.random.seed(seed)
    h = repair_h(h_initial)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    start_time = time.time()
    max_iter = 100000
    alpha = 0.02

    for t in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        m_max = np.argmax(corr)
        lag = m_max - (n - 1)

        grad = compute_gradient(h, lag)
        h = h - alpha * grad
        h = repair_h(h)

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Decay learning rate
        alpha *= 0.999

    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n)
