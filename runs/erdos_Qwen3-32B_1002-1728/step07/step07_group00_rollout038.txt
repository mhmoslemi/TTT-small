    while time.time() - start_time < budget_s:
        perturbed = h_fine_opt.copy()
        indices = np.random.choice(n_points, 10, replace=False)
        perturbed[indices] += np.random.normal(0, 0.05, size=10)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = (perturbed / perturbed.sum()) * target_sum
        result = minimize(objective_fine, perturbed, method='L-BFGS-B',
                          bounds=[(0, 1)]*n_points, tol=1e-8, options={'maxiter': 50})
        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_h = (new_h / new_h.sum()) * target_sum
            c5 = objective_fine(new_h)
            if c5 < best_c5_fine:
                h_fine_opt = new_h.copy()
                best_c5_fine = c5

    h_fine_opt = np.clip(h_fine_opt, 0, 1)
    h_fine_opt = (h_fine_opt / h_fine_opt.sum()) * target_sum
    return (h_fine_opt.tolist(), best_c5_fine, n_points)
