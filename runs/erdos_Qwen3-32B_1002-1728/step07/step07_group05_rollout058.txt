import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        return h

    def repair_half(x, target_sum):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < 1e-10:
            x_clipped[np.random.choice(half_n)] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 30
    population = [h_full_initial[:half_n].copy()]
    # Initialize with random variants of initial solution
    for _ in range(population_size - 1):
        x = h_full_initial[:half_n].copy()
        x += 0.1 * np.random.randn(half_n)
        x = repair_half(x, target_sum_half)
        population.append(x)
    population = np.array(population)

    F = 0.5
    CR = 0.9
    best_c5 = float('inf')
    best_h_half = None

    # Differential Evolution phase using 85% of budget
    de_end = start_time + budget_s * 0.85
    while time.time() < de_end:
        for i in range(population_size):
            # Select 3 distinct indices
            candidates = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[candidates]
            # Mutation and crossover
            trial = a + F * (b - c)
            mask = np.random.rand(half_n) < CR
            trial_vector = np.where(mask, trial, population[i])
            trial_repaired = repair_half(trial_vector, target_sum_half)
            # Evaluate
            new_c5 = compute_c5(trial_repaired)
            # Compare with current
            current_c5 = compute_c5(population[i])
            if new_c5 < current_c5:
                population[i] = trial_repaired
            # Update best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_half = trial_repaired.copy()
        # Early exit if not improving
        if time.time() - start_time > 0.99 * (de_end - start_time):
            break

    # Local Optimization phase
    def objective(x):
        x_repaired = repair_half(x, target_sum_half)
        h = create_full_h(x_repaired)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        try:
            result = minimize(
                objective, 
                x0=best_h_half, 
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_h_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
