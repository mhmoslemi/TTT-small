            best_c5 = max_c5_current

        if best_c5 <= 0.38085:
            break

        max_idx = np.argmax(corr_vals)
        k = max_idx - (n_points - 1)
        if k in added_lags:
            continue
        added_lags.add(k)

        a = np.zeros(n_points)
        S_k_ref = 0.0
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                S_k_ref += h_val[i] * (1 - h_val[j])
                a[i] += (1 - h_val[j])
                a[j] -= h_val[i]

        constraints.append(a @ h_opt + S_k_ref <= t_opt)

        try:
            problem = cp.Problem(cp.Minimize(t_opt), constraints)
            problem.solve(solver=cp.CVXOPT, verbose=False)
            if problem.status == 'optimal':
                h = h_opt.value.copy()
                h = np.clip(h, 0, 1)
        except:
            pass

    h = h_opt.value.copy() if h_opt.value is not None else h
    h = np.clip(h, 0, 1)
    return (h.tolist(), best_c5, n_points)
