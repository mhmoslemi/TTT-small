        lambda_box = lambda_box * 1.2 if violation else lambda_box * 0.9

        if np.abs(best_c5 - objective(best_ab)) < 1e-5:
            break
    
    # Final repair to constraints
    h_best = 0.5 + A @ best_ab
    h_best = np.clip(h_best, 0, 1)
    h_best_sum = h_best.sum()
    h_best = h_best * (n / 2.0) / h_best_sum  # Enforce integral constraint
    h_best = np.clip(h_best, 0, 1)
    
    c5_bound = np.max(np.correlate(h_best, 1 - h_best, mode='full') * dx)
    return (h_best.tolist(), c5_bound, n)
