import numpy as np
import time
import math
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = 1/2 * integral

    def repair_half(x, target_sum, eps=1e-8):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if abs(total) < eps:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial half repair
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)
    best_half = h_half_initial.copy()
    best_c5 = compute_c5(best_half)

    # --- Simulated Annealing Phase ---
    sa_time_budget = budget_s * 0.45
    sa_end_time = start_time + sa_time_budget
    np.random.seed(seed)
    current_half = best_half.copy()
    current_c5 = best_c5
    temp = 1.0e5
    temp_decay = 0.9995
    perturbation_scale = 0.2

    while time.time() < sa_end_time:
        # Perturb current solution
        new_half = current_half.copy()
        noise = perturbation_scale * np.random.randn(half_n)
        new_half += noise
        new_half = repair_half(new_half, target_sum_half)
        new_c5 = compute_c5(new_half)

        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            current_half = new_half.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_half = new_half.copy()
                best_c5 = new_c5

        temp *= temp_decay

    # --- Particle Swarm Optimization Phase ---
    pso_time_budget = budget_s * 0.35
    pso_end_time = start_time + budget_s * 0.8
    num_particles = 30
    particles = np.zeros((num_particles, half_n))
    velocities = np.zeros_like(particles)
    personal_best_positions = np.zeros_like(particles)
    personal_best_values = np.full(num_particles, np.inf)

    for i in range(num_particles):
        p = best_half.copy()
        p += 0.1 * np.random.randn(half_n)
        p = repair_half(p, target_sum_half)
        particles[i] = p
        pb = p.copy()
        pb_c5 = compute_c5(pb)
        personal_best_positions[i] = pb
        personal_best_values[i] = pb_c5
        if pb_c5 < best_c5:
            best_half = pb.copy()
            best_c5 = pb_c5

    global_best_index = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_index].copy()
    global_best_value = personal_best_values[global_best_index]
    w_inertia = 0.9
    c1, c2 = 1.5, 1.5

    while time.time() < pso_end_time:
        for i in range(num_particles):
            # Update velocity
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w_inertia * velocities[i] + cognitive + social
            # Limit velocity to reasonable scale
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            # Update position
            particles[i] += velocities[i]
            # Repair position
            particles[i] = repair_half(particles[i], target_sum_half)
            # Evaluate and update personal best
            current_c5 = compute_c5(particles[i])
            if current_c5 < personal_best_values[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_values[i] = current_c5
                if current_c5 < best_c5:
                    best_half = particles[i].copy()
                    best_c5 = current_c5

        # Update global best
        global_best_index = np.argmin(personal_best_values)
        global_best_position = personal_best_positions[global_best_index].copy()
        global_best_value = personal_best_values[global_best_index]

        if time.time() - pso_end_time > 0:  # Check for time overflow
            break

    # --- Local Refinement Phase ---
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total != 0:
            x_scaled = (x_clipped / total) * target_sum_half
        else:
            x_scaled = x_clipped.copy()
        x_scaled = np.clip(x_scaled, 0, 1)
        h = create_full_h(x_scaled)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    for _ in range(3):
        if time.time() > start_time + budget_s - 0.1:
            break
        try:
            result = minimize(
                objective, x0=best_half, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-12, 'disp': False})
            if result.success:
                x_opt = repair_half(result.x, target_sum_half)
                c5_new = compute_c5(x_opt)
                if c5_new < best_c5:
                    best_half = x_opt
                    best_c5 = c5_new
        except:
            pass

    h_full = create_full_h(best_half)
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
