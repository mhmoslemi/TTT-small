def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import minimize

    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) should equal n_points/4.0

    def repair_half(x, target_sum):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if total == 0:
            idx = np.random.choice(np.arange(len(x)))
            x_clipped[idx] = 1.0
            total = 1.0
        scale = target_sum / total
        x_scaled = x_clipped * scale
        x_scaled = np.clip(x_scaled, 0, 1)
        return x_scaled

    def compute_c5(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[-half_n:] = h_half[::-1]
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize solution
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial, target_sum_half)
    best_h = h_half_initial.copy()
    best_c5 = compute_c5(best_h)
    
    np.random.seed(seed)
    
    # Simulated Annealing
    current_h = best_h.copy()
    current_c5 = best_c5
    T_initial = 1.5
    T = T_initial
    cooling_rate = 0.995
    sa_time_end = start_time + 0.4 * budget_s
    
    while time.time() < sa_time_end:
        # Perturbation
        perturbation = 0.2 * np.random.randn(half_n)
        new_h = current_h + perturbation
        new_h = repair_half(new_h, target_sum_half)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h[:] = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h[:] = current_h.copy()
                best_c5 = new_c5
        
        T *= cooling_rate
    
    # Particle Swarm Optimization
    population_size = 25
    particles_positions = [repair_half(current_h.copy() + 0.1 * np.random.randn(half_n), target_sum_half) for _ in range(population_size)]
    velocities = [0.05 * np.random.randn(half_n) for _ in range(population_size)]
    personal_best_positions = [p.copy() for p in particles_positions]
    personal_best_c5 = [compute_c5(p) for p in particles_positions]
    global_best_idx = np.argmin(personal_best_c5)
    global_best = particles_positions[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    pso_time_end = start_time + 0.75 * budget_s  # 40% SA + 35% PSO = 75%
    
    while time.time() < pso_time_end:
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = 0.7 * velocities[i] + 0.5 * r1 * (personal_best_positions[i] - particles_positions[i]) + 1.5 * r2 * (global_best - particles_positions[i])
            new_pos = particles_positions[i] + vel
            new_pos = repair_half(new_pos, target_sum_half)
            new_c5 = compute_c5(new_pos)
            
            if new_c5 < personal_best_c5[i]:
                personal_best_positions[i] = new_pos.copy()
                personal_best_c5[i] = new_c5
                if new_c5 < global_best_c5:
                    global_best = new_pos.copy()
                    global_best_c5 = new_c5
            
            particles_positions[i] = new_pos
            velocities[i] = vel
    
    # Update best values from PSO
    if global_best_c5 < best_c5:
        best_h = global_best.copy()
        best_c5 = global_best_c5
    
    # Local Optimization (SLSQP)
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = x_clipped.sum()
        if total == 0:
            return np.inf
        x_scaled = repair_half(x_clipped, target_sum_half)
        h = np.zeros(n_points)
        h[:half_n] = x_scaled
        h[-half_n:] = x_scaled[::-1]
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    
    try:
        result = minimize(objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
        if result.success:
            x_opt = repair_half(result.x, target_sum_half)
            h_candidate = x_opt.copy()
            h_full = np.zeros(n_points)
            h_full[:half_n] = h_candidate
            h_full[-half_n:] = h_candidate[::-1]
            h_candidate = h_candidate if np.all((0 <= h_candidate) & (h_candidate <= 1)) else repair_half(h_candidate, target_sum_half)
            new_c5 = compute_c5(h_candidate)
            if new_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = new_c5
    except:
        pass
    
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h
    h_full[-half_n:] = best_h[::-1]
    h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
