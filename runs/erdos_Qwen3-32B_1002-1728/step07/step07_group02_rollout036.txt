        if h_sum > 1e-8:
            h_half = (h_half / h_sum) * target_sum_half
        else:
            h_half[:] = target_sum_half / len(h_half)
        h_half = np.clip(h_half, 0, 1)
        return h_half

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def objective(sym_vars):
        h_full = create_full_h(sym_vars)
        h_clipped = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    h_half = repair_h(h_full_initial[:half_n])
    best_c5 = objective(h_half)

    # Step 1: Enhanced Differential Evolution
    np.random.seed(seed)
    de_maxiter = 100
    de_popsize = 50
    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=de_maxiter, popsize=de_popsize,
            mutation=(0.1, 2.0), recombination=0.95,
            seed=seed, polish=False, workers=1
        )
        if de_result.success and de_result.fun < best_c5:
            h_half = repair_h(de_result.x)
            best_c5 = objective(h_half)
    except:
        pass

    # Step 2: Guided Perturbation Local Search with Multi-Strategy
    def perturb_half(h_half, frac_remaining):
        noise_scale = 0.4 * (1 - frac_remaining ** 1.7)
        perturbed = h_half.copy()
        n_perturb = max(3, int(half_n * 0.3 * (1 - frac_remaining)))
        indices = np.random.choice(half_n, size=min(half_n, n_perturb), replace=False)
        # Structured pattern perturbation: alternating zones
        pattern = np.cos(2 * np.pi * np.linspace(0, 1, half_n))
        pattern = (pattern - pattern.min()) / (pattern.ptp() + 1e-8)
        pattern = pattern - 0.5  # Symmetric around zero
        perturbed += noise_scale * (0.5 + 0.25 * np.random.rand()) * pattern
        return np.clip(perturbed, 0, 1)

    for _ in range(3):  # Staged reinitialization
        while time.time() - start_time < budget_s - 10:
            frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
            perturbed = perturb_half(h_half, frac_remaining)
            # Projection to constraint
            perturbed_sum = perturbed.sum()
            if perturbed_sum > 1e-8:
                perturbed = (perturbed / perturbed_sum) * target_sum_half

            # Dual strategy local optimization: trust-step + constraint
            def opt_cons(x):
                return objective(x)

            try:
                result = minimize(
                    opt_cons, x0=perturbed, method='L-BFGS-B',
                    bounds=bounds,
                    options={'ftol': 1e-12, 'gtol': 1e-10, 'maxiter': 100}
                )
                if result.success:
                    failfast = np.abs(np.sum(result.x) - target_sum_half) < 1e-6
                    if failfast:
                        new_c5 = result.fun
                        if new_c5 < best_c5:
                            h_half = result.x.clip(0, 1)
                            best_c5 = new_c5
            except:
                pass

            # Surrogate guided constraint repair
            def surrogate(x):
                corr = np.correlate(create_full_h(x).clip(0, 1), (1 - create_full_h(x).clip(0, 1)), mode='full')
                return np.max(corr) * dx

            try:
                result = minimize(
                    surrogate, x0=perturbed, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'ftol': 1e-12, 'maxiter': 50}
                )
                if result.success and np.isfinite(result.fun):
                    if result.fun < best_c5:
                        h_half = repair_h(result.x)
                        best_c5 = result.fun
            except:
                pass

    # Final validation and normalization
    h_full = create_full_h(h_half)
    h_full = np.clip(h_full, 0, 1)
    current_sum = h_full.sum()
    if abs(current_sum - target_sum_full) > 1e-6:
        h_full = (h_full / current_sum) * target_sum_full
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
