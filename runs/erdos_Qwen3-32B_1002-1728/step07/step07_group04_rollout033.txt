    objective = cp.Minimize(cp.max(z))
    
    constraints = []
    
    # Each position has exactly one level active
    for i in range(half_n):
        constraints.append(cp.sum(h_b[i, :]) == 1)

    h_half = cp.sum(h_b * levels, axis=1)
    h_full = cp.Variable(n_points)
    
    for i in range(half_n):
        constraints.extend([
            h_full[i] == h_half[i],
            h_full[n_points - 1 - i] == h_half[i]
        ])

    constraints.append(cp.sum(h_full) == target_sum)
