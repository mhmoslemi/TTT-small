import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h) = n_points / 2, symmetry implies this

    def repair_half(h_half):
        h_half = np.clip(h_half, 0, 1)
        s = h_half.sum()
        if s < 1e-8:
            h_half[:] = target_sum_half / half_n
        else:
            h_half = h_half / s * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        return h_half

    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(half_n)]
    initial_half = h_initial[:half_n].copy()
    initial_half = repair_half(initial_half)
    best_c5 = compute_c5(initial_half)

    # Differential Evolution Phase
    de_iter = 50
    de_time_limit = budget_s * 0.3  # 30% of time

    def objective(x):
        x = np.array(x)
        return compute_c5(repair_half(x))

    result_de = differential_evolution(
        objective,
        bounds,
        popsize=15,
        strategy='best1bin',
        maxiter=de_iter,
        mutation=(0.5, 1.0),
        recombination=0.7,
        tol=1e-4,
        seed=seed,
        callback=lambda x, _: time.time() - start_time > de_time_limit
    )

    best_h_de = repair_half(result_de.x)
    best_c5 = compute_c5(best_h_de)

    # Gradient Descent Refinement
    remaining_time = budget_s - (time.time() - start_time)
    max_iter_gd = int(remaining_time * 0.8 / 0.1)  # Conservative estimate

    res = minimize(
        compute_c5,
        x0=best_h_de,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
        options={'maxiter': max_iter_gd, 'ftol': 1e-10}
    )

    best_h = repair_half(res.x if res.success else best_h_de)
    best_c5 = min(best_c5, compute_c5(best_h))

    # Construct final h
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h
    h_full[-half_n:] = best_h[::-1]
    h_full = np.clip(h_full, 0, 1)
    
    # Final repair
    if abs(h_full.sum() - n_points/2) > 1e-6:
        h_full = (h_full / h_full.sum()) * (n_points/2)
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
