import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h[:half_n]) = 1/2 * integral

    def repair_half(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = x_clipped.sum()
        if total < 1e-10:
            return x_clipped * 0 + 1.0 / len(x)
        x_scaled = x_clipped * (target_sum_half / total)
        return np.clip(x_scaled, 0, 1)

    def create_full_h(sym_vars):
        h = np.zeros(n_points)
        h[:half_n] = sym_vars
        h[-half_n:] = sym_vars[::-1]
        return h

    def compute_c5(h_half):
        h = create_full_h(h_half)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial feasible population generation
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = repair_half(h_half_initial)
    bounds_half = [(0.0, 1.0) for _ in range(half_n)]

    # Differential Evolution with repair
    def objective(x):
        x_repaired = repair_half(x.copy())
        return compute_c5(x_repaired)

    de_time_budget = budget_s * 0.55  # Allocate more time to global search
    time_end_de = start_time + de_time_budget

    np.random.seed(seed)
    result_de = differential_evolution(
        objective, bounds_half,
        popsize=20,
        strategy='best1bin',
        tol=1e-4,
        mutation=(0.5, 1.0),
        recombination=0.8,
        updating='replace',
        seed=seed,
        workers=1,
        polish=False,  # Custom polish later
        integrality=False,
        callback=None,
        maxiter=int(budget_s * 0.6 / (half_n * 4))
    )

    de_best = repair_half(result_de.x)
    best_c5 = compute_c5(de_best)

    # Local refinement after DE
    def bounded_objective(x, *args):
        x_repaired = repair_half(x)
        return compute_c5(x_repaired)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
    bounds = bounds_half

    local_budget = budget_s * 0.35
    time_end_local = start_time + local_budget

    for _ in range(4):  # Try multiple local search rounds
        if time.time() >= time_end_local:
            break
        result = minimize(

            bounds=bounds,
