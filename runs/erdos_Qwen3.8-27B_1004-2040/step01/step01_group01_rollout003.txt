        alpha = 0.1 / (1 + k * 0.001)
        
        # Compute numerical subgradient
        g = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            # Need to adjust another element to maintain sum
            # Actually, let's just compute C5 of the perturbed version

            # Simpler: just clip and use fix_sum

            c5_plus = compute_c5(h_plus)
            g[i] = (c5_plus - c5) / eps
        
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, temp):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = np.max(corr)
    return (mx + np.log(np.sum(np.exp(corr - mx)))) / temp

def _eq_fun(x):
    return np.sum(x) - x.size / 2.0

def _eq_jac(x):
    return np.ones(x.size)

def sa_optimize(h0, n, rng, deadline, max_iters):
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = 0.04
    T_min = 1e-9
    stalls = 0
    stall_thresh = max(200, max_iters // 15)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it > 0 and it % (max_iters // 40) == 0:
            T = max(T * 0.88, T_min)
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi <= lo + 1e-12:
            continue
        delta = rng.uniform(lo, hi)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c5 = compute_c5(new_h)
        if new_c5 <= c5 or (T > T_min and rng.random() < np.exp(-(new_c5 - c5) / T)):
            h = new_h
            c5 = new_c5
            stalls = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stalls += 1
        if stalls > stall_thresh:
            T = min(T * 4.0, 0.08)
            stalls = 0
    return best_h, best_c5

def multi_sa(h0, n, rng, deadline, max_iters):
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = 0.015
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 600 == 599:
            T = max(T * 0.92, 1e-9)
        n_move = 2 if rng.random() < 0.55 else rng.randint(3, 5)
        idx = rng.choice(n, size=n_move, replace=False)
        scale = rng.uniform(0.02, 0.1)
        pert = rng.randn(n_move)
        pert -= pert.mean()
        new_h = h.copy()
        new_h[idx] += scale * pert
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = fix_sum(new_h, n)
        new_c5 = compute_c5(new_h)
        if new_c5 <= c5 or (T > 1e-9 and rng.random() < np.exp(-(new_c5 - c5) / T)):
            h = new_h
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def slsqp_run(h0, temp, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_slsqp_obj, h0, args=(temp,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return None

def subgradient_method(h0, n, rng, deadline, n_iter):
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    eps = 0.008
    for k in range(n_iter):
        if time.time() > deadline:
            break
        alpha = 0.05 / (1.0 + 0.002 * k)
        g = np.zeros(n)
        for i in range(n):
            hp = h.copy()
            hp[i] = min(1.0, h[i] + eps)
            hp = fix_sum(hp, n)
            g[i] = (compute_c5(hp) - c5) / eps
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def local_polish(h0, n, rng, deadline):
    h = h0.copy()
    c5 = compute_c5(h)
    steps = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for ss in steps:
        if time.time() > deadline:
            break
        progress = True
        while progress:
            if time.time() > deadline:
                break
            progress = False
            order = rng.permutation(n)
            for i in order:
                if time.time() > deadline:
                    break
                for j in order:
                    if i == j:
                        continue
                    d = ss
                    ni = h[i] + d
                    nj = h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        new_h = h.copy()
                        new_h[i] = ni
                        new_h[j] = nj
                        nc = compute_c5(new_h)
                        if nc < c5 - 1e-14:
                            h = new_h
                            c5 = nc
                            progress = True
                    d = -ss
                    ni = h[i] + d
                    nj = h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        new_h = h.copy()
                        new_h[i] = ni
                        new_h[j] = nj
                        nc = compute_c5(new_h)
                        if nc < c5 - 1e-14:
                            h = new_h
                            c5 = nc
                            progress = True
    return h, c5

def make_start(n, rng, mode):
    target = n / 2.0
    fill = int(target)
    h = np.zeros(n)
    if mode == 0:
        if fill > 0:

                h[pos[0]] = min(1.0, 1.0 + rem)
    elif mode == 1:
        if fill > 0:
            pos = np.round(np.linspace(0, n - 1, fill)).astype(int)
            pos = np.unique(pos)
            h[pos[:min(fill, len(pos))]] = 1.0
            got = min(fill, len(pos))
            rem = target - got
            if rem > 0.01 and got > 0:
                h[pos[0]] = min(1.0, 1.0 + rem)
    elif mode == 2:
        p = 0
        rem = fill
        while rem > 0 and p < n:
            b = min(rem, max(1, rng.randint(2, n // 4 + 1)))
            h[p:p + b] = 1.0
            rem -= b
            p += b + max(1, rng.randint(1, n // 7 + 1))
    else:
        h = 0.25 + 0.5 * rng.rand(n)
    return fix_sum(np.clip(h, 0.0, 1.0), n)
