            h_opt, c5_opt = anneal_slsqp(h_start, n, t_limit)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Phase 3: Final intensive refinement on best solution
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)

        # Multiple perturbation rounds
        for round_i in range(6):
            if time.time() > deadline - 8:
                break
            strength = 0.05 * (1.0 + round_i * 0.3)
            h_p = perturb(best_h, n, rng, strength)
            h_opt, c5_opt = anneal_slsqp(h_p, n, min(deadline - 8, time.time() + 80))
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

        # Final SLSQP with very low temperature
        result = try_slsqp(best_h, n, 5, 3000, 1e-15)
        if result is not None:
            result = fix_sum(result, n)
            c5 = compute_c5(result)
            if c5 < best_c5:
                best_h = result
                best_c5 = c5

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
