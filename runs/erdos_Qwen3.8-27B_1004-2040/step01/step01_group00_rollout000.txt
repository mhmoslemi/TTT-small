import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(200):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        a = -k
        g[:n - a] += 1.0 - h[a:]
        g[a:] -= h[:n - a]
    return g

def lsa_optimize(h, n, T, betas, iters_per_beta, lr_scale, elapsed_fn, reserve):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in betas:
        if elapsed_fn() < reserve:
            break
        h = best_h.copy()
        lr = lr_scale / (1.0 + np.log1p(beta) * 0.3)
        for it in range(iters_per_beta):
            if elapsed_fn() < reserve:
                break
            dx = 2.0 / n
            c = np.correlate(h, 1.0 - h, mode="full") * dx
            cm = c.max()
            z = np.maximum(beta * (c - cm), -60.0)
            exps = np.exp(z)
            w = exps / exps.sum()
            g = np.zeros(n)
            for m in range(len(c)):
                if w[m] > 1e-8:
                    g += w[m] * grad_k(h, m - (n - 1))
            h = project_sum(h - lr * g, T)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if it % 400 == 399:
                lr *= 0.65
    return best_h, best_c5

def binary_push(h, T, n_push):
    h = h.copy()
    for _ in range(n_push):
        dists = np.minimum(h, 1.0 - h)
        dists[h <= 1e-8] = 1.0
        dists[h >= 1.0 - 1e-8] = 1.0
        idx = int(np.argmin(dists))
        direction = 1.0 if h[idx] > 0.5 else -1.0
        h[idx] += direction * 0.08
        h = project_sum(h, T)
    return h

def eval_coord(h, j, val, T):
    trial = h.copy()
    trial[j] = val
    trial = project_sum(trial, T)
    return compute_c5(trial)

def coord_refine(h, n, T, rounds, elapsed_fn, reserve):
    h = h.copy()
    c5 = compute_c5(h)
    for r in range(rounds):
        if elapsed_fn() < reserve:
            break
        improved = False
        for j in range(n):
            if elapsed_fn() < reserve:
                break
            if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
                continue
            best_val = h[j]
            best_c5 = c5
            coarse = np.linspace(0.0, 1.0, 12)
            for v in coarse:
                cv = eval_coord(h, j, v, T)
                if cv < best_c5:
                    best_c5 = cv
                    best_val = v
            lo = max(0.0, best_val - 0.06)
            hi = min(1.0, best_val + 0.06)
            fine = np.linspace(lo, hi, 25)
            for v in fine:
                cv = eval_coord(h, j, v, T)
                if cv < best_c5:
                    best_c5 = cv
                    best_val = v
            if best_c5 < c5 - 1e-10:
                h[j] = best_val
                h = project_sum(h, T)
                c5 = best_c5
                improved = True
        if not improved:
            break
    return h, c5

def de_optimize(h0, n, T, pop_size, F, CR, max_gen, seed, elapsed_fn, reserve):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        if i % 4 == 0:
            pop[i] = h0 + rng.normal(0, 0.12, n)
        elif i % 4 == 1:
            pop[i] = (rng.random(n) < 0.5).astype(float)
            frac = rng.random(n) < 0.25
            pop[i][frac] = 0.5
        elif i % 4 == 2:
            pop[i] = rng.uniform(0.05, 0.95, n)
            n_bin = n // 5
            pop[i][rng.choice(n, size=n_bin, replace=False)] = rng.choice([0.0, 1.0], n_bin)
        else:
            pop[i] = h0 + rng.normal(0, 0.25, n)
        pop[i] = project_sum(pop[i], T)
    fitness = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]
    for gen in range(max_gen):
        if elapsed_fn() < reserve:
            break
        for i in range(pop_size):
            if elapsed_fn() < reserve:
                break
            cands = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(cands, size=3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            v = np.clip(v, 0, 1)
            v = project_sum(v, T)
            trial = v.copy()
            j_keep = rng.randint(n)
            for j in range(n):
                if rng.random() < CR or j == j_keep:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]
            trial = project_sum(trial, T)
            f_trial = compute_c5(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = trial.copy()
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h0 = np.full(47, 0.5)

    n = len(h0)
    T = n * 0.5

    best_h = h.copy()

    h_bp = binary_push(best_h, T, 15)

    if c5 < best_c5:
        best_c5, best_h = c5, h

    best_h, best_c5 = coord_refine(best_h, n, T, 12, elapsed, 40)

    # Phase 6: Restarts with simulated annealing acceptance
    temp = 0.0005
    for i in range(100):
        if elapsed() > budget_s - 30:

        s = i % 6
        if s == 0:
