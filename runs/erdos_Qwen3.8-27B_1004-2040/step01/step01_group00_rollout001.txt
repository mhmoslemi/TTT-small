import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

def to_binary(h, T):
    T = int(round(T))
    idx = np.argsort(-h)
    b = np.zeros(len(h))
    b[idx[:T]] = 1.0
    return b

def grad_k(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        a = -k
        g[:n-a] += 1.0 - h[a:]
        g[a:] -= h[:n-a]
    return g

def lsa_optimize(h, n, T, time_fn, reserve, n_betas=5, max_it=2500):
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    h = h.copy()
    betas = [5.0, 12.0, 25.0, 50.0, 100.0, 200.0][:n_betas]
    for beta in betas:
        if time_fn() < reserve:
            break
        lr = 0.4 / (1.0 + 0.02*beta)
        for it in range(max_it):
            if time_fn() < reserve:
                break
            c5, c = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            cm = c.max()
            z = np.clip(beta * (c - cm), None, -60.0)
            exps = np.exp(z)
            s = exps.sum()
            if s < 1e-30:
                break
            w = exps / s
            g = np.zeros(n)
            for m in range(2*n-1):
                if w[m] > 1e-14:
                    g += w[m] * grad_k(h, m-(n-1), n)
            h = project_sum(h - lr*g, T)
            if it % 300 == 299:
                lr *= 0.6
    return best_h, best_c5

def binary_local_search(h_bin, n, max_iter, time_fn, reserve):
    h = h_bin.copy()
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = c_raw.max() * dx
    k_arr = np.arange(2*n-1) - (n-1)
    for it in range(max_iter):
        if time_fn() < reserve:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        if not ones or not zeros:
            break
        rem = {}
        for j in ones:
            r = np.zeros(2*n-1)
            idx = j + k_arr
            mask = (idx >= 0) & (idx < n)
            r[mask] = -(1.0 - h[idx[mask]])
            idx2 = j - k_arr
            mask2 = (idx2 >= 0) & (idx2 < n)
            r[mask2] += h[idx2[mask2]]
            rem[j] = r
        add = {}
        for l in zeros:
            a = np.zeros(2*n-1)
            idx = l + k_arr
            mask = (idx >= 0) & (idx < n)
            a[mask] = (1.0 - h[idx[mask]])
            idx2 = l - k_arr
            mask2 = (idx2 >= 0) & (idx2 < n)
            a[mask2] += -h[idx2[mask2]]
            add[l] = a
        best_j = None
        best_l = None
        best_new_max = c_raw.max()
        for j in ones:
            for l in zeros:
                nm = (c_raw + rem[j] + add[l]).max()
                if nm < best_new_max:
                    best_new_max = nm
                    best_j = j
                    best_l = l
        if best_j is not None:
            h[best_j] = 0.0
            h[best_l] = 1.0
            c_raw = c_raw + rem[best_j] + add[best_l]
            new_c5 = c_raw.max() * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break
    return h, best_c5

def binary_sa(h_bin, n, max_iter, time_fn, reserve, rng):
    h = h_bin.copy()
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    cur_c5 = c_raw.max() * dx
    best_c5 = cur_c5
    best_h = h.copy()
    k_arr = np.arange(2*n-1) - (n-1)
    for it in range(max_iter):
        if time_fn() < reserve:
            break
        T = 0.5 * (0.01 / 0.5) ** (it / max_iter)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = ones[rng.randint(len(ones))]
        l = zeros[rng.randint(len(zeros))]
        r = np.zeros(2*n-1)
        idx = j + k_arr
        mask = (idx >= 0) & (idx < n)
        r[mask] = -(1.0 - h[idx[mask]])
        idx2 = j - k_arr
        mask2 = (idx2 >= 0) & (idx2 < n)
        r[mask2] += h[idx2[mask2]]
        a = np.zeros(2*n-1)
        idx = l + k_arr
        mask = (idx >= 0) & (idx < n)
        a[mask] = (1.0 - h[idx[mask]])
        idx2 = l - k_arr
        mask2 = (idx2 >= 0) & (idx2 < n)
        a[mask2] += -h[idx2[mask2]]
        delta = (c_raw + r + a).max() - c_raw.max()
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h[j] = 0.0
            h[l] = 1.0
            c_raw = c_raw + r + a
            cur_c5 = c_raw.max() * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def coord_refine(h, n, T, rounds, time_fn, reserve):
    h = h.copy()
    cur = compute_c5(h)[0]
    for r in range(rounds):
        if time_fn() < reserve:
            break
        improved = False
        for j in range(n):
            if time_fn() < reserve:
                break
            if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
                continue
            lo = -h[j] + 1e-9
            hi = 1.0 - h[j] - 1e-9
            if hi - lo < 1e-7:
                continue
            def obj(d):
                t = h.copy()
                t[j] = h[j] + d
                t = project_sum(t, T)
                return compute_c5(t)[0]
            try:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-7, 'maxiter': 50})
                if res.fun < cur - 1e-10:
                    h[j] = h[j] + res.x
                    h = project_sum(h, T)
                    cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, cur

def subgrad_optimize(h, n, T, max_iter, lr0, time_fn, reserve):
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    lr = lr0
    h = h.copy()
    for it in range(max_iter):
        if time_fn() < reserve or lr < 1e-9:
            break
        c5, c = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        m = int(c.argmax())
        k = m - (n - 1)
        g = grad_k(h, k, n)
        h -= lr * g
        h = project_sum(h, T)
        lr *= 0.9995
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    try:
        h_init = np.asarray(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.full(47, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)[0]
    best_n = len(h_init)
    rs = np.random.RandomState(seed)
    n_init = len(h_init)

    def make_init(n, kind):
        if n == n_init and kind == 0:
            return h_init.copy()
        if kind == 0:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n)
            return np.interp(x_new, x_old, h_init)
        elif kind == 1:
            x = np.arange(n)
            return 0.5 + 0.3 * np.sin(2.0*np.pi*x/n + 0.7)
        else:
            return np.clip(0.5 + rs.normal(0, 0.25, n), 0.01, 0.99)

    n_list = [48, 50, 47, 52, 46, 54]
    n_kinds = {47: [0, 1], 48: [0, 1, 2], 50: [0, 1], 52: [0, 1], 46: [0, 1], 54: [0, 1]}

    for n in n_list:
        if remaining() < 55:
            break
        T = n * 0.5
        kinds = n_kinds.get(n, [0])
        for kind in kinds:
            if remaining() < 45:
                break
            h0 = make_init(n, kind)
            h0 = project_sum(h0, T)
            c5_0 = compute_c5(h0)[0]
            if c5_0 >= best_c5 * 1.03 and kind != 0:
                continue

            # Phase 1: LSA
            h, c5 = lsa_optimize(h0, n, T, remaining, 40, n_betas=4, max_it=2000)

            # Phase 2: Subgradient
            h2, c52 = subgrad_optimize(h, n, T, 8000, 0.06, remaining, 35)
            if c52 < c5:
                h, c5 = h2, c52

            # Phase 3: Binary for even n
            if n % 2 == 0:
                h_bin = to_binary(h, T)
                hb, cb = binary_local_search(h_bin, n, 15000, remaining, 35)
                if cb < c5:
                    h, c5 = hb, c5
                # SA from current binary
                hsa, csa = binary_sa(h, n, 8000, remaining, 30, rs)
                if csa < c5:
                    h, c5 = hsa, csa
                    # Re-run local search from SA result
                    hb2, cb2 = binary_local_search(h, n, 10000, remaining, 30)
                    if cb2 < c5:
                        h, c5 = hb2, c5

            # Phase 4: Coordinate refinement
            h, c5 = coord_refine(h, n, T, 4, remaining, 25)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

    # Final: try a few more aggressive coordinate passes on best
    if remaining() > 30:
        best_h, best_c5 = coord_refine(best_h, best_n, best_n * 0.5, 3, remaining, 15)

    best_h = project_sum(best_h, best_n * 0.5)
    best_c5 = compute_c5(best_h)[0]
    return (best_h.tolist(), float(best_c5), best_n)
