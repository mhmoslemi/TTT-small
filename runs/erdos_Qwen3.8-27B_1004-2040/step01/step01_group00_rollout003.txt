def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        a = -k
        g[:n-a] += 1.0 - h[a:]
        g[a:] -= h[:n-a]
    return g

def lsa_gradient(h, c, beta):
    n = len(h)
    cm = c.max()
    z = np.maximum(beta * (c - cm), -50.0)
    exps = np.exp(z)
    w = exps / exps.sum()
    g = np.zeros(n)
    for m in range(len(c)):
        if w[m] > 1e-12:
            g += w[m] * grad_k(h, m - (n-1))
    return g
