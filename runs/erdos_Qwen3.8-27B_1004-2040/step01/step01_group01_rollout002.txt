import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    """Project h onto {0 <= h[i] <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on the offset
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def subgradient_step(h, n, alpha):
    """One subgradient step for the min-max problem"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    j_star = int(np.argmax(corr))
    l_star = j_star - n + 1  # lag
    
    # Compute subgradient
    sg = np.zeros(n)
    for m in range(n):
        # Contribution from h[i+l] term: (1-h[m-l]) if 0 <= m-l < n
        if 0 <= m - l_star < n:
            sg[m] += (1.0 - h[m - l_star])
        # Contribution from (1-h[i]) term: -h[m+l] if 0 <= m+l < n
        if 0 <= m + l_star < n:
            sg[m] -= h[m + l_star]
    
    sg *= dx
    h_new = h - alpha * sg
    h_new = project_feasible(h_new, n)
    return h_new, corr[j_star]

def ls_exp_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = temp * corr
    mx = np.max(scaled)
    val = (mx + np.log(np.sum(np.exp(scaled - mx)))) / temp
    return val * dx

def projected_grad_descent(h0, n, temp, step, max_iters, time_limit):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        
        # Compute objective and gradient
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = temp * corr
        mx = np.max(scaled)
        weights = np.exp(scaled - mx) / np.sum(np.exp(scaled - mx))

        grad = np.zeros(n)

            for m in range(n):
                if 0 <= m - l < n:
                    grad[m] += weights[k] * (1.0 - h[m - l])
                if 0 <= m + l < n:
                    grad[m] -= weights[k] * h[m + l]

        alpha = step
        obj_curr = ls_exp_obj(h, n, temp)

        grad += weights[k] * (shift_right(one_minus_h, l) - shift_left(h, l))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def shift_r(x, l):
    n = len(x)
    out = np.zeros(n)
    if 0 <= l < n: out[l:] = x[:n-l]
    elif -n < l < 0: out[:n+l] = x[-l:]
    return out

def shift_l(x, l):
    n = len(x)
    out = np.zeros(n)
    if 0 <= l < n: out[:n-l] = x[l:]
    elif -n < l < 0: out[-l:] = x[:n+l]
    return out

def ls_exp_value(h, n, temp):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    s = temp * c
    m = s.max()
    return float((m + np.log(np.exp(s - m).sum())) / temp * dx)

def ls_exp_grad(h, n, temp):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    s = temp * c
    m = s.max()
    w = np.exp(s - m)
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        if w[k] < 1e-18: continue
        l = k - n + 1
        grad += w[k] * (shift_r(omh, l) - shift_l(h, l))
    return grad * dx

def constrained_slsqp(h0, n, temp, maxiter, deadline):
    try:
        def obj(x): return ls_exp_value(x, n, temp)
        def grad(x): return ls_exp_grad(x, n, temp)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0,1)]*n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and time.time() < deadline:
            return np.clip(res.x, 0, 1)
    except:
        pass
    return None

def subgrad_polish(h, n, n_iter, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline: break
        alpha = 0.1 / (1 + 0.01 * it)
        corr = np.correlate(h, 1-h, mode="full") * dx
        l_star = int(np.argmax(corr)) - n + 1
        grad = shift_r(1-h, l_star) - shift_l(h, l_star)
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coord_descent_maxlag(h, n, deadline, max_rounds=50):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n

        max_val = corr.max()

                if new_val < 0 or new_val > 1: continue
                # Check sum constraint: need to adjust another element

                        improved = True
                    h[i] -= delta  # undo
                    break
                h[i] -= delta  # undo
        if not improved: break
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def shift_r(x, l):
    n = len(x)
    out = np.zeros(n)
    if 0 <= l < n:
        out[l:] = x[:n-l]
    elif -n < l < 0:
        out[:n+l] = x[-l:]
    return out

def shift_l(x, l):
    n = len(x)
    out = np.zeros(n)
    if 0 <= l < n:
        out[:n-l] = x[l:]
    elif -n < l < 0:
        out[-l:] = x[:n+l]
    return out

def ls_exp_value(h, n, temp):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    s = temp * c
    m = s.max()
    return float((m + np.log(np.exp(s - m).sum())) / temp * dx)

def ls_exp_grad(h, n, temp):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    s = temp * c
    m = s.max()
    w = np.exp(s - m)
    w /= w.sum()
    grad = np.zeros(len(h))
    omh = 1.0 - h
    for k in range(2 * len(h) - 1):
        if w[k] < 1e-18:
            continue
        l = k - len(h) + 1
        grad += w[k] * (shift_r(omh, l) - shift_l(h, l))
    return grad * dx

                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return np.clip(res.x, 0.0, 1.0)

        pass
    return None
