def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_exact(h, n, max_sweeps=200, time_limit=None, start_time=None):
    """Exact coordinate descent: for each coordinate, find optimal value."""
    dx = 2.0 / n
    c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Find optimal x for coordinate i
            # Build the evaluation function
            best_x = old
            best_val = c5
            # Sample candidate points
            # The function is max of one concave quad and linear functions
            # Minimum is at 0, 1, or intersection points

        if accept:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        t *= cooling
    h = fix_sum(best_h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def structured_start(n, rng):
    h = np.zeros(n)
    count = int(n / 2)
    rem = (n / 2.0) - count
    positions = np.round(np.arange(count) * n / count).astype(int) % n
    for p in positions:
        h[p] = 1.0
    if rem > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[rng.choice(zeros)] = rem
    return h

def random_binary_start(n, rng):
    h = np.zeros(n)
    count = int(n / 2)
    rem = (n / 2.0) - count
    idx = rng.permutation(n)
    h[idx[:count]] = 1.0
    if rem > 1e-10:
        h[idx[count]] = rem
    return h

def try_slsqp(h0, n, temp, maxiter, ftol):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        scaled = temp * corr
        mx = np.max(scaled)
        return (mx + np.log(np.sum(np.exp(scaled - mx)))) / temp
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
         'jac': lambda x: np.ones(n)}
    ]
    try:
        res = minimize(obj, h0, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    # Phase 1: initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = fix_sum(h0, n0)
        c5 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c5

        # Strong coordinate descent
        t_lim = deadline - 400
        h_cd, c5_cd = coord_descent(h0, n0, max_sweeps=1000, time_limit=t_lim, n_cand=100)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

        # SLSQP
        for temp in [200, 500, 1000]:
            if time.time() > deadline - 300:
                break
            result = try_slsqp(best_h, n0, temp, 500, 1e-12)
            if result is not None:
                result = fix_sum(result, n0)
                c5 = compute_c5(result)
                if c5 < best_c5:
                    best_h = result.copy()
                    best_c5 = c5
                h_cd2, c5_cd2 = coord_descent(best_h, n0, 200, deadline - 200, 80)
                if c5_cd2 < best_c5:
                    best_h = h_cd2.copy()
                    best_c5 = c5_cd2

        # SA
        if time.time() < deadline - 100:
            h_sa, c5_sa = sa_optimize(best_h, n0, deadline - 80, rng,
                                      t_init=0.02, t_final=0.0005, cooling=0.9997)
            if c5_sa < best_c5:
                best_h = h_sa.copy()
                best_c5 = c5_sa
            h_cd3, c5_cd3 = coord_descent(best_h, n0, 300, deadline - 40, 100)
            if c5_cd3 < best_c5:
                best_h = h_cd3.copy()
                best_c5 = c5_cd3
    except Exception:
        pass

    # Phase 2: other n values
    for n in [46, 50, 68, 94]:
        if time.time() > deadline - 80:
            break
        for mode in range(2):
            if time.time() > deadline - 40:
                break
            if mode == 0:
                h_start = structured_start(n, rng)
            else:
                h_start = random_binary_start(n, rng)
            h_start = fix_sum(h_start, n)
            h_cd, c5_cd = coord_descent(h_start, n, 500, deadline - 40, 100)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd
            if time.time() < deadline - 30:
                result = try_slsqp(h_cd, n, 300, 300, 1e-11)
                if result is not None:
                    result = fix_sum(result, n)
                    c5 = compute_c5(result)
                    if c5 < best_c5:
                        best_h = result.copy()
                        best_c5 = c5
                    h_cd2, c5_cd2 = coord_descent(best_h, n, 200, deadline - 30, 80)
                    if c5_cd2 < best_c5:
                        best_h = h_cd2.copy()
                        best_c5 = c5_cd2

    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_cd, c5_cd = coord_descent(best_h, n, 300, deadline - 5, 120)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
