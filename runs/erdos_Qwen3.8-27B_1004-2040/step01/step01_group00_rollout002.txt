    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

    lo, hi = -1.5, 1.5

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = c.max()
    if beta * cm > 400:
        return cm
    return cm + np.log(np.exp(beta * (c - cm)).sum() / len(c)) / beta

def local_optimize(h, n, T, time_budget, start_beta=20, end_beta=200):
    """Run soft-max optimization with increasing beta"""
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    
    betas = np.geomspace(start_beta, end_beta, 8)
    for beta in betas:
        if time_budget() < 30:
            break

        def obj(x):
            return soft_c5(x, beta)
        
        def obj_grad(x):
            n = len(x)
            dx = 2.0 / n
            c = np.correlate(x, 1.0 - x, mode="full") * dx
            cm = c.max()
            if beta * cm > 400:
                # Fallback: gradient of hard max
                m = c.argmax()
                k = m - (n - 1)
                g = np.zeros(n)
                if k >= 0:
                    g[k:] += (1.0 - x[:n-k]) * dx
                    g[:n-k] -= x[k:] * dx
                else:
                    a = -k
                    g[:n-a] += (1.0 - x[a:]) * dx
                    g[a:] -= x[:n-a] * dx
                return g

            z = np.clip(z, -400, 400)
            w = np.exp(z) / np.exp(z).sum()
            g = np.zeros(n)

                        a = -k
                        g[:n-a] += w[m] * (1.0 - x[a:]) * dx
                        g[a:] -= w[m] * x[:n-a] * dx
            return g
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - T, 
                 'jac': lambda x: np.ones(n)}]

                if v < best_c5:

**Idea: Use the dual problem / KKT conditions**

At the optimal point, the KKT conditions say:
- For the active constraint (max of c_k), the gradient of c_k_active should be in the span of {1, active bounds}
- This means: at the optimum, the gradient of the active c_k, projected onto the null space of the constraint (sum=0) and the inactive bounds, should be zero.

This gives us a system of equations we can solve. But it's complex to implement.

**Idea: Use the "minimax" structure**

The problem is: minimize max_k f_k(h) subject to constraints.

This is equivalent to: find the smallest t such that f_k(h) <= t for all k, subject to constraints.

We can use a bisection on t:
- For a given t, check if there exists h with f_k(h) <= t for all k
- This is a feasibility problem with nonlinear constraints

But checking feasibility is itself hard.

**Idea: Use a penalty method**

This is a smooth approximation that can be optimized with standard methods.

from scipy.optimize import minimize, minimize_scalar, differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _grad_k(h, k, n, dx):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k]) * dx
        g[:n-k] -= h[k:] * dx
    else:
        a = -k
        g[:n-a] += (1.0 - h[a:]) * dx
        g[a:] -= h[:n-a] * dx
    return g

def smooth_obj(x, beta, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = c.max()
    if beta * cm > 300:
        return cm
    z = np.clip(beta * (c - cm), -300, 300)
    return cm + np.log(np.exp(z).sum()) / beta

def smooth_grad(x, beta, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = c.max()
    if beta * cm > 300:
        m = int(c.argmax())
        return _grad_k(x, m - (n-1), n, dx)
    z = np.clip(beta * (c - cm), -300, 300)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-300:
        m = int(c.argmax())
        return _grad_k(x, m - (n-1), n, dx)
    w /= ws
    g = np.zeros(n)
    for m in range(len(c)):
        if w[m] > 1e-12:
            g += w[m] * _grad_k(x, m - (n-1), n, dx)
    return g

def slsqp_optimize(h, n, T, beta, maxiter=400):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - T,
             'jac': lambda x: np.ones(n)}]

                      jac=lambda x: smooth_grad(x, beta, n),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_sum(res.x, T)
    except Exception:
        pass
    return h
