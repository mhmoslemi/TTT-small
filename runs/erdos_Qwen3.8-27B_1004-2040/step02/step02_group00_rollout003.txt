import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_val(corr, n):
    return float(corr.max() * 2.0 / n)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def to_binary(h, T):
    T = int(round(T))
    idx = np.argsort(-h)
    b = np.zeros(len(h))
    b[idx[:T]] = 1.0
    return b

def make_delta_rm(h, j, lags, n):
    d = np.zeros(2 * n - 1)
    m1 = (lags >= -j) & (lags < n - j)
    d[m1] = -(1.0 - h[j + lags[m1]])
    m2 = (lags >= j - n + 1) & (lags <= j)
    d[m2] += h[j - lags[m2]]
    d[n - 1] = 0.0
    return d

def make_delta_ad(h, l, lags, n):
    d = np.zeros(2 * n - 1)
    m1 = (lags >= -l) & (lags < n - l)
    d[m1] = (1.0 - h[l + lags[m1]])
    m2 = (lags >= l - n + 1) & (lags <= l)
    d[m2] -= h[l - lags[m2]]
    d[n - 1] = 0.0
    return d

def fast_bitap_ls(h, n, max_iter, t0, budget, reserve, rng):
    h = h.astype(float).copy()
    corr = compute_corr(h)
    best_c5 = c5_val(corr, n)
    best_h = h.copy()
    lags = np.arange(2 * n - 1) - (n - 1)
    L = 2 * n - 1

    for it in range(max_iter):
        if time.time() - t0 > budget - reserve:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        nr, nz = len(ones), len(zeros)
        if nr == 0 or nz == 0:
            break

        delta_rm = np.array([make_delta_rm(h, j, lags, n) for j in ones])
        delta_ad = np.array([make_delta_ad(h, l, lags, n) for l in zeros])

        best_val = corr.max() + 1
        best_bi, best_bj = -1, -1

        chunk = max(1, 500000 // (nr * L))
        for s in range(0, nz, chunk):
            e = min(s + chunk, nz)
            batch = e - s
            base = corr[None, :] + delta_rm[:, None, :]  # (nr, L)
            # new_corr for each (b, a): base[b] + delta_ad[s+a]
            # = (nr, batch, L)
            new_vals = base[:, None, :] + delta_ad[s:e, None, :]
            new_maxes = new_vals.max(axis=2)  # (nr, batch)
            # Apply interaction correction: +1 at two positions
            # pos1 = ones[b] - zeros[s+a] + n-1, pos2 = zeros[s+a] - ones[b] + n-1
            # We need: max(new_vals + correction)
            # Simplify: check if correction positions are the max
            # For speed, just compute corrected max:
            pos1_arr = (ones[:, None] - zeros[s:e, None] + n - 1)  # (nr, batch)

            v1 = np.take_along_axis(new_vals, pos1_arr[:, :, None], axis=2).squeeze(-1)
            v2 = np.take_along_axis(new_vals, pos2_arr[:, :, None], axis=2).squeeze(-1)
            corrected = np.maximum(new_maxes, np.maximum(v1 + 1, v2 + 1))

    T_init, T_fin = 0.15, 0.0005
    decay = (T_fin / T_init) ** (1.0 / max(max_iter, 1))

    for it in range(max_iter):
        if time.time() - t0 > budget - reserve:
            break
        T = T_init * (decay ** it)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

            break
        j = ones[rng.randint(len(ones))]

        d = make_delta_rm(h, j, lags, n) + make_delta_ad(h, l, lags, n)

        d[p2] += 1

                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5
