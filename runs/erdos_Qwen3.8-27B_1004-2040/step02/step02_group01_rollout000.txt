    z = np.maximum(beta * (c - cm), -50.0)
    exps = np.exp(z)
    w = exps / exps.sum()
    g = np.zeros(n)
    sig = w > threshold * exps.max()
    for m in np.where(sig)[0]:
        g += w[m] * grad_k(h, m - (n - 1))
    return g

def nesterov_lsa(h, n, T, beta, lr, mu, max_iter, deadline):
    vel = np.zeros(n)
    h_nest = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    for it in range(max_iter):
        if time.time() > deadline:
            break
        vel_new = mu * vel
        h_trial = h_nest - lr * vel_new
        c5, c = compute_c5(h_trial)
        g = lsa_gradient_fast(h_trial, c, beta)
        h = project_sum(h_trial - lr * g, T)
        vel = vel_new + (h - h_nest)
        h_nest = h
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 400 == 399:
            lr *= 0.6
    return best_h, best_c5

def subgrad_descent(h, n, T, top_k, lr, max_iter, deadline):
    best_h = h.copy().astype(float)
    best_c5 = compute_c5(h)[0]
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c5, c = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for m in idx:
            g += grad_k(h, m - (n - 1))
        g /= len(idx)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        h = project_sum(h - lr * g / gnorm, T)
        lr *= 0.998
    return best_h, best_c5

def upsample(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    x_c = np.linspace(0, 2, n_coarse, endpoint=False)
    x_f = np.linspace(0, 2, n_fine, endpoint=False)
    return np.interp(x_f, x_c, h_coarse)

def downsample(h_fine, n_coarse):
    n_fine_len = len(h_fine)
    x_f = np.linspace(0, 2, n_fine_len, endpoint=False)
    x_c = np.linspace(0, 2, n_coarse, endpoint=False)
    return np.interp(x_c, x_f, h_fine)

def push_boundary(h, n, T, max_steps, deadline):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)[0]
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        interior = (h > 0.01) & (h < 0.99)
        if not interior.any():
            break
        idxs = np.where(interior)[0]
        dists = np.minimum(h[idxs], 1.0 - h[idxs])
        j = idxs[np.argmin(dists)]
        h0 = h.copy(); h0[j] = 0.0
        h0 = project_sum(h0, T)
        c5_0 = compute_c5(h0)[0]
        h1 = h.copy(); h1[j] = 1.0
        h1 = project_sum(h1, T)
        c5_1 = compute_c5(h1)[0]
        if c5_0 < best_c5:
            h, best_c5 = h0, c5_0
        elif c5_1 < best_c5:
            h, best_c5 = h1, c5_1
        else:
            break
    return h, best_c5

def coord_refine(h, n, T, max_rounds, deadline):
    h = h.copy().astype(float)
    cur_c5 = compute_c5(h)[0]
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            lo = max(0.0, h[j] - 0.5)
            hi = min(1.0, h[j] + 0.5)
            best_val = cur_c5
            best_x = h[j]
            for frac in np.linspace(0, 1, 13):
                nv = lo + frac * (hi - lo)
                trial = h.copy()
                trial[j] = nv
                trial = project_sum(trial, T)
                v = compute_c5(trial)[0]
                if v < best_val:
                    best_val = v
                    best_x = nv
            if best_val < cur_c5 - 1e-11:
                h[j] = best_x
                h = project_sum(h, T)
                cur_c5 = compute_c5(h)[0]
                improved = True
        if not improved:
            break
    return h, cur_c5

def optimize_at_n(h_init, n, T, deadline, seed):
    """Full optimization pipeline at a given n."""
    rs = np.random.RandomState(seed)
    h = project_sum(h_init.copy(), T)
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    
    # LSA with increasing beta
    for beta, lr0 in [(5, 0.4), (15, 0.3), (40, 0.2), (100, 0.12), (250, 0.06)]:
        if time.time() > deadline:
            break
        h2, c52 = nesterov_lsa(best_h, n, T, beta, lr0, 0.7, 5000, deadline)
        if c52 < best_c5:
            best_c5, best_h = c52, h2
        h2, c52 = subgrad_descent(best_h, n, T, 5, 0.04, 500, deadline)
        if c52 < best_c5:
            best_c5, best_h = c52, h2
        # Perturbation escape
        if time.time() < deadline and rs.rand() < 0.5:
            h_pert = best_h + rs.randn(n) * 0.03
            h_pert = project_sum(np.clip(h_pert, 0, 1), T)
            h3, c53 = nesterov_lsa(h_pert, n, T, 40, 0.15, 0.7, 2000, deadline)
            if c53 < best_c5:
                best_c5, best_h = c53, h3
    
    # Boundary and coordinate
    if time.time() < deadline:
        best_h, best_c5 = push_boundary(best_h, n, T, 20, deadline)
    if time.time() < deadline:
        best_h, best_c5 = coord_refine(best_h, n, T, 2, deadline)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rs = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init_global = np.asarray(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init_global = np.full(47, 0.5)
    
    best_h = h_init_global.copy()
    best_c5 = compute_c5(best_h)[0]
    best_n = len(best_h)
    
    # Phase 1: Try multiple n values (quick pass)
    dl1 = t0 + budget_s * 0.30
    n_values = [47, 49, 51, 53, 55]
    
    for n_try in n_values:
        if time.time() > dl1:
            break
        T_try = float(n_try * 0.5)
        if n_try == len(h_init_global):
            h_try = h_init_global.copy()
        else:
            h_try = upsample(h_init_global, n_try)
        h_try = project_sum(h_try, T_try)
        
        # Quick LSA
        h2, c52 = nesterov_lsa(h_try, n_try, T_try, 40, 0.2, 0.7, 2000, dl1)
        if c52 < best_c5:
            best_c5, best_h, best_n = c52, h2, n_try
        h2, c52 = subgrad_descent(h_try, n_try, T_try, 5, 0.05, 500, dl1)
        if c52 < best_c5:
            best_c5, best_h, best_n = c52, h2, n_try
    
    # Phase 2: Thorough optimization at best n
    dl2 = t0 + budget_s * 0.65
    n_best = best_n
    T_best = float(n_best * 0.5)
    h_cur = project_sum(best_h.copy(), T_best)
    
    # Extended LSA with fine beta schedule
    for beta, lr0 in [(10, 0.35), (25, 0.25), (60, 0.18), (150, 0.1), (300, 0.05), (500, 0.03)]:
        if time.time() > dl2:
            break
        h2, c52 = nesterov_lsa(h_cur, n_best, T_best, beta, lr0, 0.72, 6000, dl2)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
            h_cur = h2
        # Subgradient refinement
        h2, c52 = subgrad_descent(h_cur, n_best, T_best, 3, 0.03, 800, dl2)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
            h_cur = h2
    
    # Phase 3: Restart-based escape
    dl3 = t0 + budget_s * 0.80
    for restart in range(5):
        if time.time() > dl3:
            break
        h_pert = best_h + rs.randn(n_best) * 0.02 * (restart + 1)
        h_pert = project_sum(np.clip(h_pert, 0, 1), T_best)
        h2, c52 = nesterov_lsa(h_pert, n_best, T_best, 100, 0.1, 0.7, 3000, dl3)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
        h2, c52 = subgrad_descent(h2, n_best, T_best, 5, 0.03, 500, dl3)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
    
    # Phase 4: Multi-n retry with best solution as seed
    dl4 = t0 + budget_s * 0.90
    for n_try in [47, 49, 51, 53]:
        if time.time() > dl4:
            break
        if n_try == best_n:
            continue
        T_try = float(n_try * 0.5)
        h_try = upsample(best_h, n_try)
        h_try = project_sum(h_try, T_try)
        h2, c52 = nesterov_lsa(h_try, n_try, T_try, 60, 0.15, 0.7, 3000, dl4)
        h2, c52 = subgrad_descent(h2, n_try, T_try, 5, 0.03, 500, dl4)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
            best_n = n_try
    
    # Phase 5: Final polish at best_n
    best_n = len(best_h)
    T_final = float(best_n * 0.5)
    best_h = project_sum(best_h, T_final)
    
    dl5 = t0 + budget_s * 0.96
    best_h, best_c5 = push_boundary(best_h, best_n, T_final, 30, dl5)
    best_h, best_c5 = coord_refine(best_h, best_n, T_final, 3, dl5)
    
    # Final LSA micro-refinement
    if time.time() < t0 + budget_s * 0.99:
        h2, c52 = nesterov_lsa(best_h, best_n, T_final, 200, 0.04, 0.7, 2000,
                                t0 + budget_s * 0.99)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
    
    # Validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, T_final)
    best_c5 = compute_c5(best_h)[0]
    best_n = len(best_h)
    
    return (best_h.tolist(), float(best_c5), best_n)
