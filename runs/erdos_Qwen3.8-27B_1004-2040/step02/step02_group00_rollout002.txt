    h = np.zeros(n)
    h[idx[:T_int]] = 1.0
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = c_raw.max() * dx
    for it in range(80000):
        if time_fn() < reserve:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rem = np.zeros((len(ones), 2 * n - 1))
        for ii, j in enumerate(ones):
            idx1 = j + k_arr
            m1 = (idx1 >= 0) & (idx1 < n)
            rem[ii, m1] = -(1.0 - h[idx1[m1]])
            idx2 = j - k_arr
            m2 = (idx2 >= 0) & (idx2 < n)
            rem[ii, m2] += h[idx2[m2]]
        add = np.zeros((len(zeros), 2 * n - 1))
        for ii, l in enumerate(zeros):
            idx3 = l + k_arr
            m3 = (idx3 >= 0) & (idx3 < n)
            add[ii, m3] = (1.0 - h[idx3[m3]])
            idx4 = l - k_arr
            m4 = (idx4 >= 0) & (idx4 < n)
            add[ii, m4] += -h[idx4[m4]]
        new_max = (c_raw[None, None, :] + rem[:, None, :] + add[None, :, :]).max(axis=2)
        bi = np.unravel_index(new_max.argmin(), new_max.shape)
        if new_max[bi] >= c_raw.max() - 1e-15:
            break
        j, l = ones[bi[0]], zeros[bi[1]]
        h[j] = 0.0
        h[l] = 1.0
        c_raw = c_raw + rem[bi[0]] + add[bi[1]]
        noc = c_raw.max() * dx
        if noc < best_c5:
            best_c5 = noc
    return h, best_c5

def coord_refine(h, n, T, time_fn, reserve, rounds):
    h = h.copy()
    cur = compute_c5(h)[0]
    grid_coarse = np.linspace(0, 1, 15)
    for r in range(rounds):
        if time_fn() < reserve:
            break
        improved = False
        for j in range(n):
            if time_fn() < reserve:
                break
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            best_obj = cur
            best_val = h[j]
            for trial in grid_coarse:
                t = h.copy()
                t[j] = trial
                t = project_sum(t, T)
                obj = compute_c5(t)[0]
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_val = trial
            if best_val != h[j]:
                t = h.copy()
                t[j] = best_val
                t = project_sum(t, T)
                lo_v = max(0.0, best_val - 0.1)
                hi_v = min(1.0, best_val + 0.1)
                grid_fine = np.linspace(lo_v, hi_v, 20)
                for trial in grid_fine:
                    t2 = t.copy()
                    t2[j] = trial
                    t2 = project_sum(t2, T)
                    obj2 = compute_c5(t2)[0]
                    if obj2 < best_obj - 1e-13:
                        best_obj = obj2
                        best_val = trial
                t[j] = best_val
                t = project_sum(t, T)
                h = t
                cur = best_obj
                improved = True
        if not improved:
            break
    return h, cur

def near_binary_init(h, n, T):
    T_int = int(T)
    T_frac = T - T_int
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:T_int]] = 1.0
    if T_frac > 1e-10:
        n_mid = max(1, int(np.ceil(T_frac * 2)))
        frac_val = T_frac / n_mid
        b[idx[T_int:T_int + n_mid]] = frac_val
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    try:
        h_init = np.asarray(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.full(47, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)[0]
    best_n = len(h_init)
    rs = np.random.RandomState(seed)
    n_init = len(h_init)

    def make_init(n, kind):
        if n == n_init and kind == 0:
            return h_init.copy()
        if kind == 0:
            xo = np.linspace(0, 2, n_init, endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            return np.interp(xn, xo, h_init)
        elif kind == 1:
            x = np.arange(n)
            return np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x / n + 0.7), 0.01, 0.99)
        elif kind == 2:
            x = np.arange(n)
            return np.clip(0.5 + 0.25 * np.cos(4 * np.pi * x / n) + 0.15 * np.sin(6 * np.pi * x / n + 0.3), 0.01, 0.99)
        else:
            return np.clip(0.5 + rs.normal(0, 0.3, n), 0.01, 0.99)

    n_configs = [
        (n_init, [0, 1]),
        (n_init + 2, [0, 1, 2]),
        (n_init + 1, [0, 1]),
        (n_init - 1, [0, 1]),
        (n_init + 4, [1, 2, 3]),
        (n_init - 2, [1, 2]),
    ]

    for n, kinds in n_configs:
        if remaining() < 65:
            break
        T = n * 0.5
        for kind in kinds:
            if remaining() < 55:
                break
            h0 = project_sum(make_init(n, kind), T)
            c5_0 = compute_c5(h0)[0]
            if c5_0 > best_c5 * 1.06 and kind != 0:
                continue
            h1, c5_1 = lsa_momentum(h0, n, T, remaining, 55)
            if n % 2 == 0:
                h2, c5_2 = vec_binary_ls(h1, n, T, remaining, 45)
                if c5_2 < c5_1:
                    h1, c5_1 = h2, c5_2
            else:
                h_nb = near_binary_init(h1, n, T)
                h_nb, c5_nb = coord_refine(h_nb, n, T, remaining, 40, 4)
                if c5_nb < c5_1:
                    h1, c5_1 = h_nb, c5_nb
            h3, c5_3 = coord_refine(h1, n, T, remaining, 35, 6)
            if c5_3 < c5_1:
                h1, c5_1 = h3, c5_3
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
                best_n = n

    if remaining() > 25:
        best_h, best_c5 = coord_refine(best_h, best_n, best_n * 0.5, remaining, 12, 4)

    best_h = project_sum(best_h, best_n * 0.5)
    best_c5 = compute_c5(best_h)[0]
    return (best_h.tolist(), float(best_c5), best_n)
