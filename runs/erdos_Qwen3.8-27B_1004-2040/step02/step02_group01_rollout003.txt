    best_h, best_c5 = boundary_push(best_h, n, T, 15, dl)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, T)
    best_c5 = compute_c5(best_h)[0]
    return (best_h.tolist(), float(best_c5), n)
