from scipy.optimize import minimize_scalar

    c = np.correlate(h, 1.0 - h, mode="full") * dx

    g = np.zeros(n)
    if k >= 0:

        g[:n - k] -= h[k:]
    else:
        a = -k
        g[:n - a] += 1.0 - h[a:]
        g[a:] -= h[:n - a]
    return g

    ga = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        ga[:, m] = _grad_k(h, m - (n - 1), n)
    return ga

def lsa_optimize(h, n, T, time_fn, reserve, max_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    h = h.copy()
    betas = [10, 40, 150, 500, 2000]
    per = max_iter // len(betas)

                best_h = h.copy()

            q = max(per // 4, 1)

    t[j] = h[j] + d

def coord_refine(h, n, T, rounds, time_fn, reserve):
    h = h.copy()
    cur = compute_c5(h)[0]

    n = len(h)

    b[sorted_idx[:n_full]] = 1.0
    rem = T - n_full
    if rem > 0.01:
