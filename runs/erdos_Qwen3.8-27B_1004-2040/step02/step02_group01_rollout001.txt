def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        a = -k
        g[:n-a] += 1.0 - h[a:]
        g[a:] -= h[:n-a]
    return g

def lsa_gradient(h, c, beta):
    n = len(h)
    cm = c.max()
    z = np.maximum(beta * (c - cm), -60.0)
    exps = np.exp(z)
    s = exps.sum()
    if s < 1e-30:
        # All mass on the max; just use the argmax
        idx = np.argmax(c)
        return grad_k(h, idx - (n-1))
    w = exps / s
    g = np.zeros(n)
    for m in range(len(c)):
        if w[m] > 1e-10:
            g += w[m] * grad_k(h, m - (n-1))
    return g

import numpy as np
import time
from scipy.optimize import differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        a = -k
        g[:n-a] += 1.0 - h[a:]
        g[a:] -= h[:n-a]
    return g

def lsa_gradient(h, c, beta):
    n = len(h)
    cm = c.max()
    z = np.maximum(beta * (c - cm), -50.0)
    exps = np.exp(z)
    s = exps.sum()
    if s < 1e-30:
        idx = np.argmax(c)
        return grad_k(h, idx - (n-1))
    w = exps / s
    g = np.zeros(n)
    for m in range(len(c)):
        if w[m] > 1e-10:
            g += w[m] * grad_k(h, m - (n-1))
    return g

def momentum_lsa(h, n, T, betas, lr_base, mu, deadline, iters_per_beta):
    vel = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    h = h.copy()
    for beta in betas:
        lr = lr_base / (1.0 + beta * 0.015)
        for it in range(iters_per_beta):
            if time.time() > deadline:
                return best_h, best_c5
            c5, c = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            if it % 400 == 399:

    return best_h, best_c5

def subgrad_step(h, n, T, lr, deadline, max_iter):

    best_h = h.copy()
    for it in range(max_iter):

            break
        c5, c = compute_c5(h)
        if c5 < best_c5:

            best_h = h.copy()

        g = grad_k(h, k)
        h = project_sum(h - lr * g, T)
        if it % 300 == 299:
            lr *= 0.92
    return best_h, best_c5

def push_boundary(h, n, T, max_steps, deadline, aggressive=False):
    h = h.copy()
    best_c5 = compute_c5(h)[0]
    for step in range(max_steps):
        if time.time() > deadline:
            break

                    trial[j] = tv
                    trial = project_sum(trial, T)

                        best_new_h = trial
            if best_new_h is not None:

            else:

        else:

            j = idxs[np.argmin(dists)]

def coord_descent(h, n, T, rounds, deadline, n_evals=7):
    h = h.copy()
    cur_c5 = compute_c5(h)[0]
    for r in range(rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):

                continue

            best_h = None
            cands = np.unique(np.concatenate([
                np.linspace(0, 1, n_evals),

                    best_val = cv
                    best_h = trial
            if best_h is not None:
                h, cur_c5 = best_h, best_val
                improved = True
        if not improved:
            break
    return h, cur_c5

class DEObj:
    def __init__(self, T):
        self.T = T
    def __call__(self, x):
        trial = project_sum(np.asarray(x), self.T)
        return compute_c5(trial)[0]

def de_search(h_init, n, T, seed, deadline, maxiter=100, popsize=15):
    if time.time() > deadline:
        return h_init, compute_c5(h_init)[0]
    bounds = [(0.0, 1.0)] * n
    obj = DEObj(T)
    try:
        res = differential_evolution(obj, bounds, seed=seed, maxiter=maxiter,
                                      popsize=popsize, polish=False, tol=1e-8)
        if res.x is not None:
            trial = project_sum(np.clip(np.asarray(res.x), 0, 1), T)
            v = compute_c5(trial)[0]
            return trial, v
    except Exception:
        pass
    return h_init, compute_c5(h_init)[0]

def perturb_h(h, n, T, strength, rs):
    h2 = h.copy()
    mask = (h2 > 0.05) & (h2 < 0.95)
    if mask.any():
        h2[mask] += strength * (2.0 * rs.rand(mask.sum()) - 1.0)
    h2 = np.clip(h2, 0, 1)
    return project_sum(h2, T)

def anneal_local(h, n, T, n_iter, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    temp = 0.02
    for it in range(n_iter):
        if time.time() > deadline:
            break
        j = np.random.randint(n)
        delta = np.random.uniform(-0.4, 0.4)
        h2 = h.copy()
        h2[j] = np.clip(h2[j] + delta, 0, 1)
        h2 = project_sum(h2, T)
        c2 = compute_c5(h2)[0]
        c_cur = compute_c5(h)[0]
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        if c2 <= c_cur:
            h = h2
        elif np.random.rand() < np.exp(-(c2 - c_cur) / temp):
            h = h2
        temp *= 0.9995
    return best_h, best_c5
