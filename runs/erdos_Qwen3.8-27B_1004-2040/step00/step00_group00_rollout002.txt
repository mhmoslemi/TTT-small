from scipy.optimize import minimize, linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix any remaining discrepancy
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def objective_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Smooth max
    m = np.max(corr)
    if m < 1e-15:
        return m
    return m + np.log(np.sum(np.exp(corr - m))) / beta

def local_slsqp(h, n, beta=200, maxiter=300):
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return objective_smooth(x, beta)
    
    def grad(x):
        # Numerical gradient
        g = np.zeros(n)
        eps = 1e-7
        f0 = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - f0) / eps
            x[i] -= eps
        return g
    
    result = minimize(obj, h, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    return result.x, result.fun
