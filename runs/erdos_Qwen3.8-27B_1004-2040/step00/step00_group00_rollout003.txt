    """Get pairs (i, j) such that h[i] * (1-h[j]) contributes to correlation at index c_idx."""
    k = c_idx - (n - 1)  # actual correlation lag
    pairs = []
    if k >= 0:

            pairs.append((i + k, i))
    else:

        # c_idx = n-1+k, contribution: h[i] * (1-h[i-k]) for i = -k..n-1
        for i in range(-k, n):

from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h *= (n / 2.0) / s
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = np.max(c)
    # Avoid overflow
    shifted = alpha * (c - c_max)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def scipy_optimize(h0, n, time_limit, seed=42):
    t0 = time.time()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    alpha = 20.0
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    bounds = [(0, 1)] * n
    
    while time.time() - t0 < time_limit:
        result = minimize(
            obj, best_h, method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 5000, 'xatol': 1e-9, 'fatol': 1e-12, 'adaptive': True}

        h_new = np.clip(result.x, 0, 1)
        # Repair sum constraint
        s = np.sum(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * (n / 2.0) / s
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def scipy_refine(h0, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = 30.0
    t0 = time.time()

    def obj(x):
        return smooth_max_obj(x, n, alpha)

    bounds = [(0.0, 1.0)] * n
    while time.time() - t0 < time_limit:
        res = minimize(obj, h, method='Nelder-Mead', bounds=bounds,
                       options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = repair(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        alpha = min(alpha * 1.25, 400.0)
        if rng.random() < 0.35:
            h = h + rng.normal(0, 0.03, n)
            h = repair(h)
    return best_h, best_c5

def local_search(h0, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    t0 = time.time()

    stagnant = 0
    eval_count = 0

    while time.time() - t0 < time_limit:
        if eval_count % 500 == 0:
            batch = 100
        else:
            batch = 1

        for _ in range(batch):
            if time.time() - t0 >= time_limit:
                break
            g = 1.0 - h
            c = np.correlate(h, g, mode="full") * dx
            c_max_idx = int(np.argmax(c))
            k = c_max_idx - (n - 1)

            if k >= 0 and k < n:
                idx1 = np.arange(k, n)
                idx2 = np.arange(0, n - k)
            elif -n < k < 0:
                idx1 = np.arange(0, n + k)
                idx2 = np.arange(-k, n)
            else:
                idx1 = np.array([rng.randint(n)])
                idx2 = np.array([rng.randint(n)])

            if len(idx1) > 1:
                p = rng.randint(len(idx1))
                i, j = int(idx1[p]), int(idx2[p])
                if h[i] > h[j] + 1e-8:
                    delta = min(step, (h[i] - h[j]) * 0.4)
                elif h[j] > h[i] + 1e-8:
                    delta = -min(step, (h[j] - h[i]) * 0.4)
                else:

            h_try[i] += delta
            h_try[j] -= delta

            cnt = int(mask.sum())
            if cnt > 0:
                h[mask] += rng.uniform(-0.12, 0.12, cnt)
                h = repair(h)
            step = 0.07

    return best_h, best_c5
