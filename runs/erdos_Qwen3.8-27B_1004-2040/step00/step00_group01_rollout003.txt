    return float(np.max(c)), c

def project_sum(h, target):

    for _ in range(100):

        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    """Gradient of c_k with respect to h, where c_k = sum_i h[i+k]*(1-h[i])."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # (1 - h[j-k]) for j >= k
        g[k:] += 1.0 - h[:n-k]
        # -h[j+k] for j < n-k
        g[:n-k] -= h[k:]
    else:
        a = -k
        # (1 - h[j+k]) for j < n-k (i.e., j < n-a)
        g[:n-a] += 1.0 - h[a:]
        # -h[j-k] = -h[j+a] for j >= a
        g[a:] -= h[:n-a]
    return g

def coordinate_refine(h, n, T, max_rounds=5):
    from scipy.optimize import minimize_scalar
    for _ in range(max_rounds):
        for j in range(n):
            if h[j] <= 0 or h[j] >= 1:
                continue
            best_val = compute_c5(h)[0]
            def sub_obj(delta):
                trial = h.copy()
                trial[j] = h[j] + delta
                trial = project_sum(trial, T)
                return compute_c5(trial)[0]
            res = minimize_scalar(sub_obj, bounds=(-h[j], 1-h[j]), method='bounded', options={'xatol': 1e-4})
            trial = h.copy()
            trial[j] = h[j] + res.x
            trial = project_sum(trial, T)
            if compute_c5(trial)[0] < best_val:
                h = trial
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        a = -k
        g[:n - a] += 1.0 - h[a:]
        g[a:] -= h[:n - a]
    return g

def lsa_step(h, beta, lr, n, T):
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    cm = c.max()
    z = np.maximum(beta * (c - cm), -60.0)
    exps = np.exp(z)
    w = exps / exps.sum()
    g = np.zeros(n)
    for m in range(len(c)):
        if w[m] > 1e-10:
            g += w[m] * grad_k(h, m - (n - 1))
    h_new = h - lr * g
    h_new = project_sum(h_new, T)
    return h_new

def lsa_optimize(h, n, T, betas, max_iter, lr_scale, time_fn, reserve):
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    for beta in betas:
        if time_fn() < reserve:
            break
        h = best_h.copy()
        lr = lr_scale / (1.0 + beta * 0.05)
        for it in range(max_iter):
            if time_fn() < reserve:
                break
            h = lsa_step(h, beta, lr, n, T)
            c5 = compute_c5(h)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if it % 800 == 799:
                lr *= 0.6
    return best_h, best_c5

def subgrad_optimize(h, n, T, max_iter, lr0, time_fn, reserve):
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    lr = lr0
    h = best_h.copy()
    for it in range(max_iter):
        if time_fn() < reserve or lr < 1e-8:
            break
        c5, c = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        m = int(c.argmax())
        k = m - (n - 1)
        g = grad_k(h, k)
        h -= lr * g
        h = project_sum(h, T)
        lr *= 0.9997
    return best_h, best_c5

def scipy_polish(h, n, T, time_fn, reserve, maxiter=500):
    if time_fn() < reserve:
        return h, compute_c5(h)[0]
    from scipy.optimize import minimize
    def f(x):
        return compute_c5(np.asarray(x))[0]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - T, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            hx = project_sum(np.asarray(res.x), T)
            v = compute_c5(hx)[0]
            if v < compute_c5(h)[0]:
                return hx, v
    except Exception:
        pass
    return h, compute_c5(h)[0]

def coord_refine(h, n, T, rounds, time_fn, reserve):
    from scipy.optimize import minimize_scalar
    h = h.copy()
    cur_c5 = compute_c5(h)[0]
    for r in range(rounds):
        if time_fn() < reserve:
            break

                    cur_c5 = v
                    improved = True
            except Exception:
                pass
