            h = new_h
            patience = 0
            beta = min(beta * 1.04, 400.0)
        else:
            h = new_h
            patience += 1
            if patience % 15 == 0:
                lr *= 0.88
            if patience > 250:
                h = best_h.copy()
                beta = max(beta_init, beta * 0.4)
                lr = 0.01
                patience = 0
    return best_h, best_c5

def coordinate_polish(h, n, max_rounds, time_limit, start_time, budget_frac=0.92):
    target_sum = n / 2.0
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    deadline = start_time + budget_frac * time_limit

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            h[i] = 0.0
            h = project_feasible(h, target_sum)
            c5 = evaluate_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
                improved = True
            else:
                h = best_h.copy()
            old_val = h[i]
            h[i] = 1.0
            h = project_feasible(h, target_sum)
            c5 = evaluate_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
                improved = True
            else:
                h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        n_init = 50
        h_init = np.zeros(n_init)
        h_init[n_init // 4: 3 * n_init // 4] = 1.0
        h_init = project_feasible(h_init, n_init / 2.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    for n in [60, 80, 100, 120]:
        if time.time() - start_time > budget_s * 0.55:
            break
        h = upsample_to(h_init, n)
        h, c5 = optimize_gradient(h, n, 40.0, 2000, budget_s, start_time, 0.58)
        h, c5 = coordinate_polish(h, n, 30, budget_s, start_time, 0.62)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    if best_h is None:
        best_n = 80
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n / 2.0)
        best_c5 = evaluate_c5(best_h)

    n = best_n
    target_sum = n / 2.0

    for attempt in range(12):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.88:
            break
        remaining = budget_s - elapsed
        if remaining < 30:
            break
        frac = min(0.90 - 0.01 * attempt, 0.88)

        if attempt % 4 == 0:
            h = best_h + 0.12 * np.random.randn(n)
        elif attempt % 4 == 1:
            h = best_h + 0.04 * np.random.randn(n)
        elif attempt % 4 == 2:
            h = best_h.copy()
            idx = np.random.choice(n, size=max(1, n // 5), replace=False)
            h[idx] = 1.0 - h[idx]
        else:
            h = np.random.uniform(0, 1, n)

        h = project_feasible(h, target_sum)
        h, c5 = optimize_gradient(h, n, 60.0, 1000, budget_s, start_time, frac)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h.copy()

    if time.time() - start_time < budget_s * 0.94:
        best_h, best_c5 = coordinate_polish(best_h, best_n, 50, budget_s, start_time, 0.97)

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, best_n)
