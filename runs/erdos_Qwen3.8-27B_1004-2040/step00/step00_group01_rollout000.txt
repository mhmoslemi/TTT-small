import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def sum_constraint_eq(x, n):
    return np.sum(x) - n / 2.0


def sum_constraint_jac(x, n):
    return np.ones(n)


def fix_sum(h, n):
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def make_obj(n, temp):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        scaled = temp * corr
        mx = np.max(scaled)
        return (mx + np.log(np.sum(np.exp(scaled - mx)))) / temp
    return obj


def make_constraint(n):
    return {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
            'jac': lambda x: np.ones(n)}


def try_slsqp(h0, n, temp, maxiter, ftol):
    try:
        res = minimize(
            make_obj(n, temp),
            h0,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[make_constraint(n)],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return None


def random_start(n, rng, mode=0):

        step = n / float(fill) if fill > 0 else n
        for i in range(fill):
            pos = int(round(i * step)) % n
            h[pos] = 1.0
    elif mode == 1:
        h = rng.rand(n) * 0.5 + 0.25
    elif mode == 2:
        h = (rng.rand(n) > 0.5).astype(float)

            h[start:start + blk] = 1.0
            remaining -= blk
            start += blk + rng.randint(1, max(2, n // 6))
    h = fix_sum(h, n)
    return h
