from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def soft_max_obj(h, n, eps):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(c)
    return m + eps * np.log(np.sum(np.exp((c - m) / eps)))

def project_feasible(h, n):
    """Project h onto the feasible set: 0≤h≤1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # First try to distribute among non-saturated variables
        mask = (h > 1e-6) & (h < 1 - 1e-6)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        # If still not feasible, adjust more aggressively
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                # Need to increase; use variables closest to 0
                order = np.argsort(h)
                for idx in order:
                    room = 1.0 - h[idx]
                    add = min(room, diff)
                    h[idx] += add
                    diff -= add
                    if diff < 1e-12:
                        break
            else:
                # Need to decrease; use variables closest to 1
                order = np.argsort(-h)
                for idx in order:
                    room = h[idx]
                    rem = min(room, -diff)
                    h[idx] -= rem
                    diff += rem
                    if diff > -1e-12:
                        break
    return h

    h = project_feasible(h, n)
    target = n / 2.0
    
    best_local_c5 = compute_c5(h)
    best_local_h = h.copy()
    
    for eps in eps_schedule:
        if time.time() - start_time > total_budget * budget_frac:
            break
        
        def obj(x):
            return soft_max_obj(x, n, eps)
        
        def obj_grad(x):
            # Numerical gradient
            dx = 2.0 / n
            c = np.correlate(x, 1-x, mode="full") * dx
            m = np.max(c)
            # Soft max gradient
            weights = np.exp((c - m) / eps) / np.sum(np.exp((c - m) / eps))
            # Gradient of c[m_idx] w.r.t. x[j]
            grad = np.zeros(n)
            n_half = n - 1
            for mi, w in enumerate(weights):
                if w < 1e-10:
                    continue
                s = mi - n_half
                for j in range(n):
                    js_minus = j - s
                    if 0 <= js_minus < n:
                        grad[j] += w * (1 - x[js_minus])
                    js_plus = j + s
                    if 0 <= js_plus < n:
                        grad[j] -= w * x[js_plus]
            grad *= dx
            return grad
        
        # Use L-BFGS-B (handles bounds) with penalty for sum constraint
        # Or use SLSQP with the equality constraint
        
        bounds = [(0, 1)] * n
        
        try:
            res = minimize(
                obj, h,
                jac=obj_grad,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            h = project_feasible(res.x, n)
        except Exception:
            pass
        
        c5_now = compute_c5(h)
        if c5_now < best_local_c5:
            best_local_c5 = c5_now
            best_local_h = h.copy()
    
    return best_local_h, best_local_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def soft_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(c)
    return m + eps * np.log(np.sum(np.exp((c - m) / eps)))

def soft_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(c)
    exp_c = np.exp((c - m) / eps)
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    n_half = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = abs(k - n_half)
        if s == 0:
            contrib = (1 - h) - h
        else:
            contrib = np.zeros(n)
            contrib[s:] += (1 - h[:n - s])
            contrib[:n - s] -= h[s:]
        grad += w[k] * dx * contrib
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-10:
            if diff2 > 0:
                order = np.argsort(h)
            else:
                order = np.argsort(-h)
            for idx in order:
                if diff2 > 0:
                    room = 1.0 - h[idx]
                    add = min(room, diff2)
                    h[idx] += add
                    diff2 -= add
                else:
                    room = h[idx]
                    rem = min(room, -diff2)
                    h[idx] -= rem
                    diff2 += rem
                if abs(diff2) < 1e-12:
                    break
    return h

def optimize_soft(h0, n, start_time, budget_s, frac):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eps_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.003]
    bounds = [(0.0, 1.0)] * n
    for eps in eps_list:
        if time.time() - start_time > budget_s * frac:
            break
        res = minimize(
            soft_max_obj, h,
            args=(eps,),
            jac=lambda x, e: soft_max_grad(x, e),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-7}
        )
        h = project_feasible(res.x, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def local_search(h, n, start_time, budget_s, frac):
    h = h.copy()
    c5 = compute_c5(h)
    target = n / 2.0
    deltas = [0.01, 0.03, 0.05, 0.1, 0.2]
    improved = True
    while improved and (time.time() - start_time) < budget_s * frac:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - start_time > budget_s * frac:
                    return h, c5
                for d in deltas:
                    for sign in [1, -1]:
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if hi < -1e-10 or hi > 1 + 1e-10 or hj < -1e-10 or hj > 1 + 1e-10:
                            continue
                        h_test = h.copy()
                        h_test[i] = hi
                        h_test[j] = hj
                        c5_test = compute_c5(h_test)
                        if c5_test < c5 - 1e-9:
                            h = h_test
                            c5 = c5_test
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
    return h, c5

def make_initial(n, h_ref=None):
    if h_ref is not None and len(h_ref) > 1:
        x_old = np.linspace(0, 2, len(h_ref), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, h_ref)
    elif n >= 4:
        h = np.full(n, 0.5)
        # Create a structured initial: blocks of high and low
        h[:n // 4] = 0.8
        h[n // 4:n // 2] = 0.3
        h[n // 2:3 * n // 4] = 0.7
        h[3 * n // 4:] = 0.4
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

            best_h = h2.copy()

    # Phase 3: Try different n values
    n_candidates = sorted(set([n_init + 1, n_init + 2, n_init + 4, 55, 65, 80, 100]))
    for n_try in n_candidates:
        if time.time() - start_time > budget_s * 0.85:

        if n_try == n_init or n_try < 10:

        h_try = make_initial(n_try, h_ref=best_h)

        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            # Quick local search on the new best

                h3b, c5_3b = local_search(h3, n_try, start_time, budget_s, 0.88)

            h_pert = project_feasible(h_pert, n_fin)

            if c5_pert < best_c5:
