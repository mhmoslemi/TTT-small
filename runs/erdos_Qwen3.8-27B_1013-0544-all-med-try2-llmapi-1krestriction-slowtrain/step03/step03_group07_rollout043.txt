            valid_js = (j - s >= 0) & (j - s < n)
            valid_kp = (k + s >= 0) & (k + s < n)

            h[k] -= delta

                best_c5 = c5

def smoothmax_slsqp(h, beta, maxiter=300):

    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = corr - m
        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(beta * shifted)
        w = exps / np.sum(exps)
        grad = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            if s >= 0:
                idx = np.arange(s, n)
                grad[idx] += wk * (1.0 - x[idx - s])
                idx2 = np.arange(0, n - s)
                grad[idx2] += wk * (-x[idx2 + s])
            elif s < 0:
                t = -s
                idx = np.arange(0, n - t)
                grad[idx] += wk * (1.0 - x[idx + t])
                idx2 = np.arange(t, n)
                grad[idx2] += wk * (-x[idx2 - t])
            else:
                grad[:] += wk * (1.0 - 2.0 * x)
        return grad * dx
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)
