    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for li in top_idx:
        lag = int(li) - (n - 1)
        for j in range(n):
            i1 = j - lag
            i2 = j + lag
            if 0 <= i1 < n:
                grad[j] += (1.0 - h[i1])
            if 0 <= i2 < n:
                grad[j] -= h[i2]
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return None
    return -grad / gn

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(280, 0.5)

    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = [{'type': 'eq', 'fun': make_constraint(target), 'jac': lambda x: make_con_jac(n)}]

    # Phase 1: Top-k lag targeting with SLSQP
    h_cur = best_h.copy()
    cur_c5 = best_c5

    for rnd in range(20):
        if budget_s - elapsed() < 40:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_k = min(7, 2 * n - 1)
        lags = np.argsort(corr)[::-1][:top_k]

        def obj_topk(x, lags=lags):
            c = np.correlate(x, 1.0 - x, mode="full")
            vals = np.array([c[li] for li in lags])
            return smooth_max(vals, 300) * dx

        try:
            res = minimize(obj_topk, h_cur, method='SLSQP', bounds=bounds,
                          constraints=constr, options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h_cur = h_new
                cur_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: Binary simulated annealing
    h_bin = (h_cur >= 0.5).astype(float)
    deficit = target - h_bin.sum()
    if deficit > 0.5:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(h_cur[zeros])[::-1]]
        h_bin[order[:int(np.ceil(deficit))]] = 1.0
    elif deficit < -0.5:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h_cur[ones])]
        h_bin[order[:int(np.ceil(-deficit))]] = 0.0

    bin_c5 = compute_c5(h_bin)
    if bin_c5 < best_c5:
        best_h = h_bin.copy()
        best_c5 = bin_c5

    h_sa = h_bin.copy()
    sa_c5 = compute_c5(h_sa)
    T = 0.05
    cooling = 0.99995
    sa_iter = 0
    max_sa = 60000

    while sa_iter < max_sa and budget_s - elapsed() > 30:
        T = max(0.0005, T * cooling)
        if T <= 0.0006 and rng.random() > 0.1:
            pass

        move_type = rng.random()
        if move_type < 0.6:
            h_trial = sa_binary_move(h_sa, n, target, rng)
        else:
            bs = int(rng.choice([2, 3, 4, 5, 7]))
            h_trial = sa_block_move(h_sa, n, target, rng, bs)

        if h_trial is None:
            sa_iter += 1
            continue

        trial_c5 = compute_c5(h_trial)
        delta = trial_c5 - sa_c5
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / T):
            h_sa = h_trial
            sa_c5 = trial_c5
            if sa_c5 < best_c5:
                best_h = h_sa.copy()
                best_c5 = sa_c5
        sa_iter += 1

    # Phase 3: Continuous refinement with progressive smooth_max
    if budget_s - elapsed() > 40:
        h_cur = best_h.copy()
        cur_c5 = best_c5

        for beta in [80, 200, 500, 1000, 2000, 5000]:
            if budget_s - elapsed() < 15:
                break

            def obj_full(x, b=beta):
                corr = np.correlate(x, 1.0 - x, mode="full")
                return smooth_max(corr, b) * dx

            try:
                res = minimize(obj_full, h_cur, method='SLSQP', bounds=bounds,
                              constraints=constr, options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h_cur = h_new
                    cur_c5 = c5_new
                    if c5_new < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5_new
            except Exception:
                pass

    # Phase 4: Subgradient polishing with backtracking
    if budget_s - elapsed() > 8:
        cur = best_h.copy()
        cur_c5 = best_c5
        momentum = np.zeros(n)
        base_step = 0.08

        for it in range(30000):
            if budget_s - elapsed() < 3:
                break
            step = base_step / (1.0 + 0.001 * it)

            g = subgradient_step(cur, 3, n)
            if g is None:
                g = subgradient_step(cur, 5, n)
            if g is None:
                break

            momentum = 0.75 * momentum + 0.25 * g
            h_trial = project_h(cur + step * momentum, target)
            c5_trial = compute_c5(h_trial)

            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                base_step = min(base_step * 1.1, 0.15)
                if cur_c5 < best_c5:
                    best_h = cur.copy()
                    best_c5 = cur_c5
            else:
                base_step *= 0.7
                momentum = 0.5 * momentum
                if base_step < 0.001:
                    break

    return best_h, best_c5, int(len(best_h))
