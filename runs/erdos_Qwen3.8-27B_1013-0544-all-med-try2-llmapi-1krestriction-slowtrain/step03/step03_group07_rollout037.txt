    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    # For each lag, add weighted gradient contribution
    # Use the formula: for lag s (s = k_idx - (n-1)):
    # if s >= 0: grad[s:] += w * (1-h[:n-s]), grad[:n-s] += w * (-h[s:])
    # if s < 0 (t=-s): grad[:n-t] += w * (1-h[t:]), grad[t:] += w * (-h[:n-t])

```python
def lp_cutting_plane(h, target, n_top=20):
    n = len(h)
    corr = compute_corr(h)
    # Get top lags
    top_idx = np.argsort(corr)[-n_top:]
    
    # Build constraint matrix
    # Variables: h'[0],...,h'[n-1], t (total n+1)
    # Constraints: for each top lag s_l: grad_c[s_l] @ h' - t <= c_s_l - grad_c[s_l] @ h
    #              sum(h') = n/2
    #              0 <= h'[i] <= 1
    
    A_ub = np.zeros((n_top, n + 1))
    b_ub = np.zeros(n_top)
    
    for ii, k_idx in enumerate(top_idx):
        s = k_idx - (n - 1)
        gc = np.zeros(n)
        if s >= 0:
            gc[s:] += (1.0 - h[:n-s])
            gc[:n-s] += (-h[s:])
        elif s < 0:
            t = -s
            gc[:n-t] += (1.0 - h[t:])
            gc[t:] += (-h[:n-t])
        A_ub[ii, :n] = gc
        A_ub[ii, n] = -1.0
        b_ub[ii] = corr[k_idx] - gc @ h
    
    # Equality constraint: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h'[i] <= 1, t unbounded below (but will be >= 0)
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = project_feasible(res.x[:n], target)
        return h_new
    return h
```
