import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -100, 100)
    return m + np.logaddexp.reduce(x_shift) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

            break

        new_c5 = compute_c5(proposal)

            h = proposal
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        
        # Cool

    return best_h, best_c5

    return np.interp(x_new, x_old, h)

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sml_grad(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    weights = np.exp(np.clip(beta * (c - m), -200, 200))
    weights /= weights.sum()
    n = len(h_in)
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += w * (1.0 - h_in[lag:])
            grad[lag:] -= w * h_in[:n-lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h_in[:n-la])
            grad[:n-la] -= w * h_in[la:]
    return grad * dx

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    def obj(x):
        return sml_obj(x, beta, 2.0/n)
    def cons_eq(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(x.shape)
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_subgrad(h, target, n_iter, step0, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    vel = np.zeros(n)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    momentum = 0.9
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 1:
            break
        # Compute smooth gradient
        c = np.correlate(cur, 1.0 - cur, mode="full")
        m = np.max(c)
        beta = 200.0
        weights = np.exp(np.clip(beta * (c - m), -200, 200))
        weights /= weights.sum()
        
        # Compute gradient of smooth max
        grad = np.zeros(n)
        for k in range(2*n-1):
            if weights[k] < 1e-12:
                continue
            lag = k - (n-1)
            if lag >= 0:
                grad[:n-lag] += weights[k] * (1.0 - cur[lag:])
                grad[lag:] -= weights[k] * cur[:n-lag]
            else:
                la = -lag
                grad[la:] += weights[k] * (1.0 - cur[:n-la])
                grad[:n-la] -= weights[k] * cur[la:]
        grad *= dx
        
        # Nesterov step
        step = step0 / (1.0 + 0.01 * it)
        vel = momentum * vel - step * grad
        trial = project_h(cur + vel, target)
        cur = trial
        
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cur.copy()
    
    return best_h, best_c5

def sa_block(h, target, n_iter, t0, budget_s, block_sizes=None, T_initial=0.03):
    if block_sizes is None:
        block_sizes = [3, 5, 7, 10, 15, 20]
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    T = T_initial
    T_min = 0.002
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 0.5:
            break
        bs = block_sizes[np.random.randint(len(block_sizes))]
        if bs >= n:
            bs = max(1, n // 4)
        start = np.random.randint(0, n - bs + 1)
        
        # Propose: perturbation
        proposal = cur.copy()
        noise = np.random.randn(bs) * T
        proposal[start:start+bs] += noise
        proposal = project_h(proposal, target)
        
        new_c5 = compute_c5(proposal)
        delta = new_c5 - cur_c5
        
        if delta < 0 or (T > T_min and np.random.rand() < np.exp(-delta / T)):
            cur = proposal
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
        
        # Cooling
        cool_rate = 1.0 - 1.0 / max(n_iter, 10)
        T = max(T_min, T * cool_rate)
    
    return best_h, best_c5

def coord_descent_fine(cur, target, t0, budget_s):
    best_c5 = compute_c5(cur)
    n = len(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    sw = 0
    while budget_s - (time.time() - t0) > 2:
        improved = False
        for j in range(n):
            if budget_s - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        sw += 1
        if not improved:
            break
    return cur, best_c5
