    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_beta_g = 400.0

def fft_corr(h, f):
    n = len(h)
    N = 2 * n - 1
    Nf = int(2 ** np.ceil(np.log2(N + 1)))
    return np.fft.irfft(np.fft.rfft(h, Nf) * np.fft.rfft(f[::-1], Nf))[:N]

def c5_val(h):
    return float(np.max(fft_corr(h, 1.0 - h)) * (2.0 / len(h)))

def proj(h, tgt):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
        d2 = tgt - h.sum()
        if abs(d2) > 1e-14:
            fr = (h > 1e-10) & (h < 1.0 - 1e-10)
            if fr.sum():
                h[fr] += d2 / fr.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def grad_topk(h, k):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h, 1.0 - h)
    idx = np.argsort(corr)[-k:]
    g = np.zeros(n)
    for ii in idx:
        s = ii - (n - 1)
        if s == 0:
            g[:] += 1.0 - 2.0 * h
        elif s > 0:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] += -h[s:]
        else:
            t = -s
            g[:n - t] += 1.0 - h[t:]
            g[t:] += -h[:n - t]
    return g / k * dx

def smx_val(h, beta):
    n = len(h); dx = 2.0 / n
    corr = fft_corr(h, 1.0 - h)
    m = corr.max()
    e = np.exp(beta * (corr - m))
    return (m + np.log(e.sum()) / beta) * dx

def smx_grad(h, beta):
    n = len(h); dx = 2.0 / n
    corr = fft_corr(h, 1.0 - h)
    m = corr.max()
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    g = np.zeros(n)
    for ii in range(len(corr)):
        if w[ii] < 1e-11:
            continue
        s = ii - (n - 1)
        if s == 0:
            g[:] += w[ii] * (1.0 - 2.0 * h)
        elif s > 0:
            g[s:] += w[ii] * (1.0 - h[:n - s])
            g[:n - s] += w[ii] * (-h[s:])
        else:
            t = -s
            g[:n - t] += w[ii] * (1.0 - h[t:])
            g[t:] += w[ii] * (-h[:n - t])
    return g * dx

def _sf(x):
    return smx_val(x, _beta_g)

def _sj(x):
    return smx_grad(x, _beta_g)

def _cf(x):
    return x.sum() - len(x) / 2.0

def _cj(x):
    return np.ones(len(x))

def slsqp(h, mi, beta):
    global _beta_g
    _beta_g = beta
    n = len(h); tgt = n / 2.0
    try:
        r = minimize(_sf, h, jac=_sj, method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        return proj(r.x, tgt)
    except Exception:
        return proj(h.copy(), tgt)

def nesterov(h, mi, lr0, decay, tk):
    n = len(h); tgt = n / 2.0
    x = h.copy().astype(float); xp = x.copy()
    bx = x.copy(); bv = c5_val(x)
    for t in range(mi):
        lrt = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            y = proj(x + ((t - 1) / (t + 2)) * (x - xp), tgt)
        g = grad_topk(y, tk)
        xn = proj(y - lrt * g, tgt)
        v = c5_val(xn)
        if v < bv:
            bv = v; bx = xn.copy()
        xp = x; x = xn
    return bx, bv

def interp_h(h, nn):
    xo = (np.arange(len(h)) + 0.5) * (2.0 / len(h))
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    return proj(np.clip(f(xn), 0, 1), nn / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:

    hb = proj(hb, len(hb) / 2.0)
    bc = c5_val(hb)
    nb = len(hb)

    # Phase 1: SLSQP warmup to establish good basin
    for b in [100, 300, 800]:

    # Phase 2: Long Nesterov subgradient with FFT (fast iterations)
    for tk, lr0, dec in [(5, 0.02, 0.99995), (10, 0.015, 0.99996), (15, 0.012, 0.99997)]:

    # Phase 3: SLSQP high-beta polish
