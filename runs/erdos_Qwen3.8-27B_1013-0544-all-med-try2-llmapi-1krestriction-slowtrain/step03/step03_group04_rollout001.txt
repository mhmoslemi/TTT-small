def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_weighted(x, beta, top_n=None):
    m = np.max(x)
    if top_n is not None and top_n < len(x):
        idx = np.argsort(x)[::-1][:top_n]
        weights = np.ones(len(x))
        weights[idx] = 10.0
    else:
        weights = np.ones(len(x))
    x_shift = np.clip(beta * (x - m), -500, 500)
    logsumexp = np.sum(weights * np.exp(x_shift))
    return m + np.log(logsumexp / np.sum(weights)) / beta

    sm = smooth_max_weighted(corr, _g_beta, top_n=10)

    return sm * _g_dx + penalty
