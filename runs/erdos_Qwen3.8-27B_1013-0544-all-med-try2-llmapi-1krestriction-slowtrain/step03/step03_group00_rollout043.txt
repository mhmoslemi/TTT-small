import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx, k in enumerate(top_idx):
        wk = w_sel[idx]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def slsqp_optimize(h, beta, top_k, target, maxiter):
    n = len(h)
    def fun(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    def jac(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < fun(h):
            return project_feasible(res.x, target)
    except Exception:
        pass
    return project_feasible(h, target)

def coord_desc(h, target, n_sweeps=100, grid_size=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    grid = np.linspace(0, 1, grid_size)
    
    for sweep in range(n_sweeps):
        changed = False
        for i in range(n):
            old_val = h[i]
            # Determine affected correlation indices
            # result[k] = sum_j h[j]*(1-h[j+k-(n-1)])
            # h[i] affects k where: i+k-(n-1) in [0,n) [as j=i]
            #   => k in [n-1-i, 2n-2-i]
            #   or: j+k-(n-1)=i [as j+s=i], j in [0,n)
            #   => k = i + n - 1 - j, j in [0,n) => k in [n-1-i, n-1+i]
            
            k_lo = n - 1 - i
            k_hi1 = 2 * n - 2 - i  # from j=i case
            k_hi2 = n - 1 + i       # from j+s=i case
            k_hi = max(k_hi1, k_hi2)
            k_hi = min(k_hi, 2 * n - 2)
            
            if k_lo > k_hi:
                continue
            
            # Save affected correlations
            affected = corr[k_lo:k_hi+1].copy()
            
            best_t = old_val
            best_obj = np.max(corr) * dx
            
            for t in grid:
                if abs(t - old_val) < 1e-10:
                    continue
                h[i] = t
                # Update affected correlations
                for k in range(k_lo, k_hi + 1):
                    s = k - (n - 1)
                    delta = 0.0
                    if 0 <= i + s < n:
                        delta += (t - old_val) * (1.0 - h[i + s])
                    if 0 <= i - s < n:
                        delta += h[i - s] * (old_val - t)
                    corr[k] += delta
                obj = np.max(corr) * dx
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_t = t
                # Restore
                corr[k_lo:k_hi+1] = affected.copy()
            
            if best_t != old_val:
                h[i] = best_t
                for k in range(k_lo, k_hi + 1):
                    s = k - (n - 1)

                    if 0 <= i - s < n:

                    corr[k] += delta
                changed = True
