    except Exception:
        return project_feasible(h_init.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: L-BFGS-B progressive beta (fast, many iterations)
    for beta in [30, 80, 200, 500]:
        if tl() < 30:
            break
        h_new = lbfgsb_opt(h_best, beta, maxiter=300, mu=100.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Nesterov top-k for peak reduction
    if tl() > 80:
        h_ng, c_ng = nesterov_topk(h_best, 4000, 0.03, 0.9999, top_k=10)
        if c_ng < best_c5:
            h_best, best_c5 = h_ng, c_ng

    # Phase 3: High beta L-BFGS-B
    for beta in [500, 1000, 2000]:
        if tl() < 30:
            break
        h_new = lbfgsb_opt(h_best, beta, maxiter=250, mu=200.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: Alternating low/high beta with perturbations
    for trial in range(12):
        if tl() < 50:
            break
        scale = 0.008 + 0.003 * trial
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, n / 2.0)
        b = 200 if trial % 2 == 0 else 1500
        h_new = lbfgsb_opt(h_pert, b, maxiter=200, mu=150.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
        if tl() > 20 and c < best_c5 + 1e-6:
            h_new2 = lbfgsb_opt(h_new, 3000, maxiter=150, mu=300.0)
            c2 = compute_c5(h_new2)
            if c2 < best_c5:
                h_best, best_c5 = h_new2, c2

    # Phase 5: Projected gradient with many steps
    if tl() > 80:
        h_pg, c_pg = proj_grad(h_best, 800, 5000, 0.008, 0.99995)
        if c_pg < best_c5:
            h_best, best_c5 = h_pg, c_pg
        if tl() > 30:
            h_new = lbfgsb_opt(h_best, 2000, maxiter=200, mu=200.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 6: Higher resolution
    for n_try in [500, 600]:
        if tl() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [100, 500, 2000]:
            if tl() < 40:
                break
            h_new = lbfgsb_opt(h_interp, beta, maxiter=250, mu=200.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        n = len(h_best)

    # Phase 7: SLSQP final polish
    for beta in [1000, 3000]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 8: Final L-BFGS-B very high beta
    if tl() > 25:
        h_final = lbfgsb_opt(h_best, 5000, maxiter=400, mu=500.0)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
