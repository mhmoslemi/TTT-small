import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:

        else:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    """Alternating LP: fix g=1-h, solve LP for h, update g, repeat."""
    t0 = time.time()
    h_cur = h.copy()

            break

                idx = np.arange(n - lag)
                coeff = g[lag:]  # g[i+lag] for i in 0..n-lag-1
            else:

                coeff = g[:n+lag]  # g[i+lag] for i in -lag..n-1
            constraints.append(cp.sum(h_var[idx] * coeff) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
        except:

    return smooth_max(corr, beta) * dx

    n = len(h)
    # d/dh_i of corr[j] for lag k = j-(n-1)
    # corr[j] = sum_i h[i]*(1-h[i+k]) for lag k>=0
    #         = sum_i h[i+k]*(1-h[i]) for lag k<0
    # grad_j[i] = (1-h[i+k]) for i where i+k valid, when k>=0
    #            = -h[i] for i where i+k valid, when k>=0

    # Weight by softmax
    m = np.max(corr)
    w = np.exp(beta * (corr - m))

            grad[idx] += weight * (1.0 - h[idx + lag])

        else:
            la = -lag
            idx = np.arange(la, n)
            grad[idx] += weight * (1.0 - h[idx - la])
            grad[idx - la] -= weight * h[idx]
    
    return grad * dx, grad * dx * beta  # value, gradient

def augmented_lagrangian(h0, dx, beta, n, max_iter=200, time_limit=120):

    t0 = time.time()
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    target = n / 2.0
    lam = 0.0
    mu = 1.0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        def aug_obj(x, lam=lam, mu=mu):
            val = smooth_max(np.correlate(x, 1.0-x, mode='full'), beta) * dx
            constraint = x.sum() - target
            val += lam * constraint + (mu/2) * constraint**2
            return val
        
        def aug_grad(x, lam=lam, mu=mu):
            corr = np.correlate(x, 1.0-x, mode='full')
            m = np.max(corr)
            w = np.exp(beta * (corr - m))
            w /= w.sum()
            grad = np.zeros(n)
            for j in range(len(corr)):
                if w[j] < 1e-15:
                    continue
                lag = j - (n-1)
                if lag >= 0:
                    idx = np.arange(n-lag)
                    grad[idx] += w[j]*(1.0-x[idx+lag])
                    grad[idx+lag] -= w[j]*x[idx]
                else:
                    la = -lag
                    idx = np.arange(la, n)
                    grad[idx] += w[j]*(1.0-x[idx-la])
                    grad[idx-la] -= w[j]*x[idx]
            grad *= dx
            constraint = x.sum() - target
            grad += lam + mu * constraint
            return grad
        
        # Use L-BFGS-B with bound constraints
        from scipy.optimize import minimize as sci_min
        res = sci_min(aug_obj, h_cur, jac=aug_grad, method='L-BFGS-B',
                     bounds=[(0,1)]*n, options={'maxiter': 500, 'ftol': 1e-14})
        
        h_new = project_feasible(res.x, target)
        c5 = compute_c5(h_new)
        
        # Update multiplier
        constraint = h_new.sum() - target
        lam += mu * constraint
        if abs(constraint) > 1e-6:
            mu *= 2.0
            mu = min(mu, 1000.0)
        
        h_cur = h_new
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    
    return best_h, best_c5

```python
from scipy.sparse import csr_matrix, coo_matrix

def build_lp_matrix(n, g):
    """Build the A matrix for the LP.
    Variables: h[0..n-1], t
    Constraints: for each lag k in -(n-1)..(n-1): sum h[i]*g[i+k] - t <= 0
    """
    rows = []
    cols = []
    vals = []
    n_lags = 2*n - 1
    for lag_idx in range(n_lags):
        lag = lag_idx - (n-1)
        if lag >= 0:
            # sum_{i=0}^{n-1-lag} h[i]*g[i+lag] - t <= 0
            for i in range(n - lag):
                rows.append(lag_idx)
                cols.append(i)
                vals.append(g[i+lag])
        else:
            # sum_{i=-lag}^{n-1} h[i]*g[i+lag] - t <= 0
            for i in range(-lag, n):
                rows.append(lag_idx)
                cols.append(i)
                vals.append(g[i+lag])
        # -t coefficient
        rows.append(lag_idx)
        cols.append(n)
        vals.append(-1.0)
    
    A = coo_matrix((vals, (rows, cols)), shape=(n_lags, n+1)).tocsr()
    b = np.zeros(n_lags)
    return A, b
```

Then use scipy.optimize.linprog with method='highs':

```python
from scipy.optimize import linprog

def solve_lp(h, n, g):
    target = n / 2.0
    # Build A matrix
    A, b = build_lp_matrix(n, g)
    # c vector: minimize t (last variable)
    c = np.zeros(n+1)
    c[-1] = 1.0
    # A_eq: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
```

    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            return h
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)

    return m + np.log(np.sum(np.exp(shifted))) / beta

    """Compute gradient of smooth_max(corr, beta) * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for li in range(n_lags):
        if w[li] < 1e-15:
            continue

        if lag >= 0:
            idx = np.arange(n - lag)

            grad[idx + lag] -= w[li] * h[idx]

            idx = np.arange(la, n)
            grad[idx] += w[li] * (1.0 - h[idx - la])
            grad[idx - la] -= w[li] * h[idx]

    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            return h
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10

            return h

                const = 0.0
                lin = 0.0
                if 0 <= i + k < n:
                    lin += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    const += h[i - k]
                    lin += (-h[i - k])
                a_arr[li] = const
                b_arr[li] = lin
            x_opt = min_1d_convex(a_arr, b_arr)
            h[i] = x_opt
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def smooth_grad(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for li in range(n_lags):
        if w[li] < 1e-15:
            continue
        lag = li - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += w[li] * (1.0 - h[idx + lag])
            grad[idx + lag] -= w[li] * h[idx]
        else:
            la = -lag
            idx = np.arange(la, n)
            grad[idx] += w[li] * (1.0 - h[idx - la])
            grad[idx - la] -= w[li] * h[idx]
    return grad * dx

def polish_smooth(h, n, time_limit=60, beta_start=50, beta_end=10000, n_betas=8):
    t_start = time.time()
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    dx = 2.0 / n
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for beta in betas:
        if time.time() - t_start > time_limit:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, beta) * dx + 5.0 * (x.sum() - target) ** 2
        def jac(x):
            g = smooth_grad(x, n, dx, beta)
            g += 10.0 * (x.sum() - target)
            return g
        try:
            res = minimize(obj, h_cur, jac=jac, method='L-BFGS-B',
                         bounds=[(0, 1)] * n, options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h_init = np.full(100, 0.5)

    n = len(h_init)
    target_sum = n / 2.0
    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Alternating LP at initial resolution
    if tl() > 100:
        h1, c1 = solve_alt_lp(best_h, n, time_limit=50, rng=rng, perturb_every=20, perturb_scale=0.02)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Coordinate descent
    if tl() > 100:
        h2, c2 = coordinate_descent(best_h, n, time_limit=30, max_sweeps=300)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Smooth polish
    if tl() > 80:
        h3, c3 = polish_smooth(best_h, n, time_limit=40, beta_start=50, beta_end=20000, n_betas=10)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Multi-resolution exploration
    n_cur = n
    resolutions = sorted(set([n_cur * 2, n_cur * 3, n_cur + 30, n_cur + 50,
                               n_cur + 80, n_cur + 120, n_cur + 200,
                               n_cur - 20, n_cur - 50] and
                              [r for r in [n_cur * 2, n_cur * 3, n_cur + 30, n_cur + 50,
                                           n_cur + 80, n_cur + 120, n_cur + 200,
                                           n_cur - 20, n_cur - 50]
                               if 40 <= r <= 1000]))
    for n_try in resolutions:
        if n_try == n_cur or tl() < 80:
            continue
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        # Alt LP
        h4, c4 = solve_alt_lp(h_interp, n_try, time_limit=min(35, tl() * 0.3),
                              rng=rng, perturb_every=30, perturb_scale=0.015)
        improved = c4 < best_c5
        if improved:
            best_h, best_c5 = h4, c4
            n_cur = n_try
        # Coordinate descent
        if tl() > 40:
            h5, c5 = coordinate_descent(h4, n_try, time_limit=min(20, tl() * 0.2), max_sweeps=150)
            if c5 < best_c5:
                best_h, best_c5 = h5, c5
                n_cur = n_try
        # Smooth polish
        if tl() > 30:
            h6, c6 = polish_smooth(h5 if 'h5' in dir() else h4, n_cur, time_limit=min(20, tl()*0.15),
                                   beta_start=100, beta_end=10000, n_betas=6)
            if c6 < best_c5:
                best_h, best_c5 = h6, c6
                n_cur = n_try

    # Phase 5: Final combined polish
    n_final = len(best_h)
    if tl() > 40:
        h7, c7 = solve_alt_lp(best_h, n_final, time_limit=min(20, tl()*0.4),
                              rng=rng, perturb_every=40, perturb_scale=0.008)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7
    if tl() > 25:
        h8, c8 = coordinate_descent(best_h, n_final, time_limit=min(15, tl()*0.5), max_sweeps=200)
        if c8 < best_c5:
            best_h, best_c5 = h8, c8
    if tl() > 15:
        h9, c9 = polish_smooth(best_h, n_final, time_limit=min(12, tl()*0.7),
                               beta_start=500, beta_end=100000, n_betas=5)
        if c9 < best_c5:
            best_h, best_c5 = h9, c9

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
