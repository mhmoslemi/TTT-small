import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, beta):
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(x)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n, dx):
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0*h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:n-lag]) * dx
        g[:n-lag] += -h[lag:] * dx
    else:
        l = -lag
        g[:n-l] += (1.0 - h[l:]) * dx
        g[l:] += -h[:n-l] * dx
    return g

def adam_optimize(h, n, target, dx, max_iters, time_fn, min_time, best_state):
    """Adam optimizer on the smooth max objective with projection."""
    cur = h.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    cur_c5 = compute_c5(cur)
    
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        t += 1
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr * dx
        # Use top-k with softmax weighting
        top_k = min(4, 2*n//100 + 2)
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 200.0  # high temperature for peaked weighting
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        
        # Compute gradient as weighted average
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_lag(cur, lag, n, dx)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.05 / (1 + 0.01 * it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        h_trial = project_h(cur - lr * update / (np.linalg.norm(update) + 1e-10), target)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < cur_c5 - 1e-16:
            cur = h_trial
            cur_c5 = c5_trial
            if cur_c5 < best_state[0]:
                best_state[0] = c5_trial
                best_state[1] = cur.copy()
                best_state[2] = n
        else:

                    if c < cur_c5 - 1e-15:
                        cur = trial
                        cur_c5 = c
                        break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur.copy()
    
    # Try alternate resolutions
    if budget_s - (time.time() - t0) > 80:
        for n_alt in (200, 300, 350):
            if budget_s - (time.time() - t0) < 50:
                break
            cur_a = project_h(resample_h(best_h, n_alt), n_alt / 2.0)
            c5_a = compute_c5(cur_a)
            dx_a = 2.0 / n_alt
            c_a = n_alt - 1
            for it in range(15000):
                if budget_s - (time.time() - t0) < 25:
                    break
                corr = np.correlate(cur_a, 1.0 - cur_a, mode="full")
                cv = corr * dx_a
                ti = np.argsort(cv)[::-1][:3]
                v = cv[ti]
                w = np.exp((v - v[0]) * 500.0)
                w /= w.sum()
                grad = np.zeros(n_alt)
                for t_i, wi in zip(ti, w):
                    grad += wi * grad_lag(cur_a, int(t_i) - c_a, n_alt, dx_a)
                gn = np.linalg.norm(grad)
                if gn < 1e-15:
                    break
                grad /= gn
                step = 0.005 / (1.0 + 0.003 * it)
                ht = project_h(cur_a - step * grad, n_alt / 2.0)
                ct = compute_c5(ht)
                if ct < c5_a:
                    cur_a = ht
                    c5_a = ct
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = cur_a.copy()
    
    # Final ultra-fine
    if budget_s - (time.time() - t0) > 10:
        n_f = len(best_h)
        t_f = n_f / 2.0
        dx_f = 2.0 / n_f
        c_f = n_f - 1
        cur = best_h.copy()
        cur_c5 = best_c5
        for it in range(50000):
            if budget_s - (time.time() - t0) < 3:
                break
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            cv = corr * dx_f
            ti = np.argsort(cv)[::-1][:2]
            v = cv[ti]
            w = np.exp((v - v[0]) * 2000.0)
            w /= w.sum()

            for t_i, wi in zip(ti, w):
                grad += wi * grad_lag(cur, int(t_i) - c_f, n_f, dx_f)
            gn = np.linalg.norm(grad)

            step = 5e-5
            for _ in range(15):

                ct = compute_c5(ht)
                if ct < cur_c5 - 1e-17:

                    cur_c5 = ct
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                    break
                step *= 0.5
    
    return best_h, best_c5, len(best_h)

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, lag, n, dx):
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag]) * dx
        g[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        g[:n - l] += (1.0 - h[l:]) * dx
        g[l:] += -h[:n - l] * dx
    return g


def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2.0, n_new), np.linspace(0, 2.0, len(h)), h)
