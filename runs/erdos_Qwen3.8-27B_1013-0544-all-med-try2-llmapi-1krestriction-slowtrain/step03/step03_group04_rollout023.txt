                    best_t = tv
            
            if abs(best_t - h[j]) > 1e-12:
                h[j] = best_t
                h[l] = s - best_t
                corr = C + best_t * D  # update corr
                improved = True
        
        if not improved:
            break
        
        corr = compute_corr(h)  # recompute for accuracy
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5

import numpy as np
from scipy.optimize import differential_evolution, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_B(h, j, n):
    B = np.zeros(2 * n - 1)
    B[n - 1 - j:2 * n - 1 - j] += (1.0 - h)
    B[j:j + n] -= h[::-1]
    return B

def find_optimal_t(C, D, t_lo, t_hi):
    best_t = (t_lo + t_hi) / 2
    best_val = np.max(C + best_t * D)
    for tv in [t_lo, t_hi]:
        v = np.max(C + tv * D)
        if v < best_val - 1e-14:
            best_val = v
            best_t = tv
    pos_m = D > 1e-10
    neg_m = D < -1e-10
    if np.any(pos_m) and np.any(neg_m):
        lo, hi = t_lo, t_hi
        for _ in range(50):
            mid = (lo + hi) / 2.0
            vals = C + mid * D
            mv = np.max(vals)
            tol = 1e-10 + 1e-14 * abs(mv)
            amax = np.where(np.abs(vals - mv) < tol)[0]
            if len(amax) == 0:
                break
            if D[amax].min() > 1e-10:
                hi = mid
            elif D[amax].max() < -1e-10:
                lo = mid
            else:
                break
        tv = (lo + hi) / 2.0
        v = np.max(C + tv * D)
        if v < best_val - 1e-14:
            best_val = v
            best_t = tv
    return best_t, best_val

def paired_coord_descent(h, target, max_passes=200, rng_seed=42, time_limit=None, t0=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    rng = np.random.RandomState(rng_seed)
    rev_h = h[::-1]
    for p in range(max_passes):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            l = rng.randint(n)
            if l == j:
                l = (j + 1) % n
            s = h[j] + h[l]
            if s < 1e-12 or s > 2.0 - 1e-12:
                continue
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-12:
                continue
            B_j = compute_B(h, j, n)
            B_l = compute_B(h, l, n)
            D = B_j - B_l
            C = corr - h[j] * B_j - h[l] * B_l + s * B_l
            best_t, best_val = find_optimal_t(C, D, t_lo, t_hi)
            if abs(best_t - h[j]) > 1e-12:
                h[j] = best_t
                h[l] = s - best_t
                corr = C + best_t * D
                rev_h = h[::-1]
                improved = True
        if not improved:
            break
        corr = compute_corr(h)
        rev_h = h[::-1]
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def smooth_gradient_descent(h, target, dx, beta_schedule, lr_scale, max_iter, time_limit, t0):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        beta = beta_schedule(it)
        corr = compute_corr(h)
        vals = corr * dx
        m = np.max(vals)
        shift = np.clip(beta * (vals - m), -500, 500)
        e = np.exp(shift)
        p = e / e.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = p[k]
            if w < 1e-15:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                idx1 = np.arange(n - lag)
                grad[idx1] += w * (1.0 - h[idx1 + lag])
                idx2 = np.arange(lag, n)
                grad[idx2] -= w * h[idx2 - lag]
            else:
                il = -lag
                idx1 = np.arange(il, n)
                grad[idx1] += w * (1.0 - h[idx1 - il])
                idx2 = np.arange(n - il)
                grad[idx2] -= w * h[idx2 + il]
        grad *= dx
        lr = lr_scale / (beta * dx * n + 1.0)
        h_new = project_h(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

def try_binary_round(h, target, time_limit=None, t0=None):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in np.linspace(0.2, 0.8, 25):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        hb = (h >= t).astype(float)
        deficit = target - hb.sum()
        if abs(deficit) > n * 0.3:
            continue
        if deficit > 0:
            zeros = np.where(hb == 0)[0]
            order = zeros[np.argsort(h[zeros])[::-1]]
            k = int(np.ceil(deficit))
            if k <= len(order):
                hb[order[:k]] = 1.0
        elif deficit < 0:
            ones = np.where(hb == 1)[0]
            order = ones[np.argsort(h[ones])]
            k = int(np.ceil(-deficit))
            if k <= len(order):
                hb[order[:k]] = 0.0
        if abs(hb.sum() - target) < 1e-6:
            c5 = compute_c5(hb)
            if c5 < best_c5:
                best_h = hb.copy()
                best_c5 = c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)

    target = n_base / 2.0
    h_parent = project_h(h_parent, target)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Paired coordinate descent at base resolution
    tl = min(400, budget_s * 0.35)
    h_cd, c5_cd = paired_coord_descent(best_h, target, max_passes=500, rng_seed=seed,
                                        time_limit=tl, t0=t0)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd.copy(), c5_cd

    # Phase 2: Smooth gradient descent with continuation
    if budget_s - elapsed() > 100:
        dl = min(300, budget_s * 0.25)
        h_gd, c5_gd = smooth_gradient_descent(
            best_h, target, 2.0 / n_base,
            lambda it: min(50.0 * (1.5 ** (it // 50)), 2000.0),
            lr_scale=1.0, max_iter=2000,
            time_limit=dl, t0=t0
        )
        if c5_gd < best_c5:
            best_h, best_c5 = h_gd.copy(), c5_gd

    # Phase 3: More coordinate descent (different random order)
    if budget_s - elapsed() > 80:
        tl2 = min(250, budget_s * 0.2)
        h_cd2, c5_cd2 = paired_coord_descent(best_h, target, max_passes=500,
                                             rng_seed=seed + 12345,
                                             time_limit=tl2, t0=t0)
        if c5_cd2 < best_c5:
            best_h, best_c5 = h_cd2.copy(), c5_cd2

    # Phase 4: Try upscaling to higher resolution
    for mult in [2.0, 2.5]:
        if budget_s - elapsed() < 120:
            break
        n_new = int(n_base * mult)
        if n_new > 1200:
            continue
        target_new = n_new / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_up = project_h(np.interp(x_new, x_old, best_h), target_new)
        c5_up = compute_c5(h_up)
        # Coordinate descent at higher res
        tl3 = min(150, budget_s - elapsed() - 60)
        if tl3 > 30:
            h_up2, c5_up2 = paired_coord_descent(h_up, target_new, max_passes=200,
                                                  rng_seed=seed + 999,
                                                  time_limit=tl3, t0=t0)
            if c5_up2 < c5_up:
                h_up, c5_up = h_up2, c5_up2
        if c5_up < best_c5:
            best_h, best_c5 = h_up.copy(), c5_up

    # Phase 5: Final coordinate descent polish on best
    if budget_s - elapsed() > 30:
        n_f = len(best_h)
        tgt_f = n_f / 2.0
        tl4 = min(100, budget_s - elapsed() - 5)
        h_f, c5_f = paired_coord_descent(best_h, tgt_f, max_passes=300,
                                         rng_seed=seed + 777,
                                         time_limit=tl4, t0=t0)
        if c5_f < best_c5:
            best_h, best_c5 = h_f.copy(), c5_f

    # Phase 6: Binary rounding attempt
    if budget_s - elapsed() > 10:

        tgt_b = n_b / 2.0
        h_bin, c5_bin = try_binary_round(best_h, tgt_b,
                                         time_limit=min(60, budget_s - elapsed() - 2), t0=t0)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin.copy(), c5_bin

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_h(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(n_final)
