def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

    t_vals = np.arange(-(n-1), n)
    for l in range(n):

        mask1 = (a >= 0) & (a < n)
        C[l, mask1] += -(1.0 - h[a[mask1]])
        b = l - t_vals
        mask2 = (b >= 0) & (b < n)
        C[l, mask2] += h[b[mask2]]
    return C

def gradient_refine(h, n, max_iters=5000, time_fn=None, min_time=5):
    target = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time_fn is not None and time_fn() < min_time:
            break
        
        c5_vec = compute_c5_vec(h)
        top_k = min(5, 2*n//100 + 3)
        top_idx = np.argsort(c5_vec)[::-1][:top_k]
        vals = c5_vec[top_idx]
        temp = 100.0 + 100.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(h, lag, n, dx)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        step = 0.05 / (1.0 + 0.002 * it)
        improved = False
        for _ in range(30):
            h_trial = project_h(h - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5 - 1e-15:
                h = h_trial
                c5 = c5_trial
                improved = True
                break
            step *= 0.5
        
        if not improved and it % 100 == 0:
            # Small perturbation
            T = 0.002 * np.exp(-it / 2000.0)
            noise = np.random.normal(0, T, n)
            h_trial = project_h(h + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5 + T * 5:
                h = h_trial
                c5 = c5_trial
    
    return h, c5
