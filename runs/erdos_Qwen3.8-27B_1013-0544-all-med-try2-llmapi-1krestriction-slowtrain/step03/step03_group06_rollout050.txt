∂r[τ]/∂h[i]:
- For i in [0, n-1-τ]: h[i] is the first factor, derivative is (1-h[i+τ])
- For i in [τ, n-1]: h[i] appears in the (1-h[i-τ]) term (where i-τ ranges from 0 to n-1-τ), derivative is -h[i-τ]

- h[i] as first factor: i must be in [t, n-1] (so i-t is in [0, n-1-t]), derivative is (1-h[i-t])
- h[i] in (1-h) factor: i must be in [0, n-1-t], derivative is -h[i+t]

So:

grad[0:n-t] -= wk * h[t:n]  ✓

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def _sm_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr * dx
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    ws = np.exp(xs) / np.sum(np.exp(xs))
    grad = np.zeros(n)
    for k in range(len(ws)):
        tau = k - (n - 1)
        wk = ws[k]
        if wk < 1e-12:
            continue
        if tau == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif tau > 0:
            grad[:n - tau] += wk * (1.0 - h[tau:n])
            grad[tau:] -= wk * h[:n - tau]
        else:
            t = -tau
            grad[t:] += wk * (1.0 - h[:n - t])
            grad[:n - t] -= wk * h[t:]
    return grad * dx

def _subgrad_hard(h, n_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[::-1][:n_top]
    grad = np.zeros(n)
    for k in top:
        tau = int(k) - (n - 1)
        if tau == 0:
            grad += (1.0 - 2.0 * h)
        elif tau > 0:
            grad[:n - tau] += (1.0 - h[tau:n])
            grad[tau:] -= h[:n - tau]
        else:
            t = -tau
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] -= h[t:]
    return grad

def _lbfgs_step(h, n, beta, dx, target_sum, mu, maxiter=2000):
    def obj(z):
        hc = np.clip(z, 0, 1)
        return _sm(np.correlate(hc, 1.0 - hc, mode="full") * dx, beta) + mu * (hc.sum() - target_sum) ** 2
    def jac(z):
        hc = np.clip(z, 0, 1)
        g = _sm_grad(hc, beta, dx)
        g += 2.0 * mu * (hc.sum() - target_sum)
        mask = (z > 0) & (z < 1)
        g[~mask] = 0.0
        return g
    res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return np.clip(res.x, 0, 1)

def _polytope_descent(h, target_sum, max_iter=3000):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[::-1][:min(5, n // 20 + 2)]
        grad = np.zeros(n)
        for k in top:
            tau = int(k) - (n - 1)
            if tau == 0:
                grad += (1.0 - 2.0 * h)
            elif tau > 0:
                grad[:n - tau] += (1.0 - h[tau:n])
                grad[tau:] -= h[:n - tau]
            else:
                t = -tau
                grad[t:] += (1.0 - h[:n - t])
                grad[:n - t] -= h[t:]
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        found = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            ht = project_h(h - step * grad, target_sum)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                h = ht
                best_c5 = c5t
                found = True
                break
        if not found and it % 200 == 199:
            pert = np.random.randn(n) * 0.01
            ht = project_h(h + pert, target_sum)
            if compute_c5(ht) < best_c5:
                h = ht
                best_c5 = compute_c5(h)
    return h, best_c5

def _upsample(h_old, n_new, target_sum):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, target_sum)

def _make_structured(n):
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    h[x < 0.5] = 1.0
    h[(x >= 1.0) & (x < 1.5)] = 1.0
    return project_h(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = _make_structured(200)
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    resolutions = [100, 160, 200, 280]
    beta_sched = [15, 30, 60, 120, 250, 500, 1000, 2000]
    mu_sched = [5, 20, 100, 500]

    for n_pts in resolutions:
        if tl() < 60:
            break
        target = n_pts / 2.0
        dx = 2.0 / n_pts

        if n_pts == best_n:
            h_cur = best_h.copy()
        elif n_pts > best_n:
            h_cur = _upsample(best_h, n_pts, target)
        else:
            h_cur = _upsample(best_h, n_pts, target)
        cur_c5 = compute_c5(h_cur)

        for mu in mu_sched:
            if tl() < 30:
                break
            for beta in beta_sched:
                if tl() < 15:
                    break
                try:
                    h_new = _lbfgs_step(h_cur, n_pts, beta, dx, target, mu, maxiter=1500)
                    h_new = project_h(h_new, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < cur_c5:
                        h_cur = h_new
                        cur_c5 = c5_new
                except Exception:
                    pass
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5
            best_n = n_pts

    # Also try structured starts at n=200
    if tl() > 120:
        for n_try in [200, 280]:
            if tl() < 60:
                break
            target = n_try / 2.0
            dx = 2.0 / n_try
            h_struct = _make_structured(n_try)
            c5s = compute_c5(h_struct)
            for beta in [50, 200, 800]:
                if tl() < 20:
                    break
                for mu in [50, 200]:
                    try:
                        h_new = _lbfgs_step(h_struct, n_try, beta, dx, target, mu, maxiter=1000)
                        h_new = project_h(h_new, target)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5s:
                            h_struct, c5s = h_new, c5_new
                    except Exception:
                        pass
            if c5s < best_c5:
                best_h = h_struct.copy()
                best_c5 = c5s
                best_n = n_try

    # Final polytope descent polish
    tgt = best_n / 2.0
    h_pol, c5_pol = _polytope_descent(best_h, tgt, max_iter=5000)
    if c5_pol < best_c5:
        best_h, best_c5 = h_pol, c5_pol

    # One more L-BFGS-B round at final resolution with very high beta
    if tl() > 10:
        tgt = best_n / 2.0
        dx = 2.0 / best_n
        try:
            h_new = _lbfgs_step(best_h, best_n, 5000, dx, tgt, 500, maxiter=3000)
            h_new = project_h(h_new, tgt)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Final short subgradient

    h_pol2, c5_pol2 = _polytope_descent(best_h, tgt, max_iter=2000)
    if c5_pol2 < best_c5:
        best_h, best_c5 = h_pol2, c5_pol2

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
```
