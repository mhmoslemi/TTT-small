   def slsqp_optimize(h, beta, maxiter=300):

       target = n / 2.0

           return smoothmax_obj_grad(x, beta)[0]

           return smoothmax_obj_grad(x, beta)[1]

           return np.ones(n)

   def nesterov_topm(h, max_iters, lr0, decay, top_m, momentum_base=0.9):
       n = len(h)
       dx = 2.0 / n
       target = n / 2.0
       x = h.copy().astype(float)
       x_prev = x.copy()
       best_x = x.copy()

           x_prev = x
           x = x_new
       
       return best_x, best_v
