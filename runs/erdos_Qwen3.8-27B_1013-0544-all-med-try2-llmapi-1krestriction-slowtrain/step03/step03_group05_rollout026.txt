    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def lse_objective(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta

def lse_gradient(h, dx, beta, top_k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Get top-k lags
    top_idx = np.argsort(corr)[::-1][:top_k]
    vals = corr[top_idx]
    # Softmax weights
    m = vals[0]
    w = np.exp(beta * (vals - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for t_idx, wi in zip(top_idx, w):
        lag = int(t_idx) - center
        # ∂c_lag/∂h[i]
        if lag == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] += -wi * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wi * (1.0 - h[l:])
            grad[l:] += -wi * h[:n - l]
    return grad * dx

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)

    x_old = np.linspace(0, 2, n_old)

    val = lse_objective(h, dx, beta)
    val += penalty * (np.sum(h) - target) ** 2

import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _lse_obj(x, dx, beta, target, penalty):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    val = m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta
    val += penalty * (np.sum(h) - target) ** 2
    return val


def _lse_grad(x, dx, beta, target, penalty, top_k):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(corr)[::-1][:top_k]
    vals = corr[top_idx]
    m = vals[0]
    w = np.exp(beta * (vals - m))
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for t_idx, wi in zip(top_idx, w):
        lag = int(t_idx) - center
        if lag == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] += -wi * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wi * (1.0 - h[l:])
            grad[l:] += -wi * h[:n - l]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(h) - target)
    return grad


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)


def subgrad_step(h, n, dx, top_k, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(corr)[::-1][:top_k]
    vals = corr[top_idx]
    w = np.exp((vals - vals[0]) * temp)
    w /= w.sum()
    grad = np.zeros(n)
    center = n - 1
    for t_idx, wi in zip(top_idx, w):
        lag = int(t_idx) - center
        if lag == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] += -wi * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wi * (1.0 - h[l:])
            grad[l:] += -wi * h[:n - l]
    return grad * dx


def polish_phase(h, target, dx, max_iters, time_fn, min_time, best_state):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        grad = subgrad_step(cur, n, dx, top_k=3, temp=300.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        step = 0.015 / (1.0 + 0.002 * it)
        improved = False
        for _ in range(20):
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-16:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved and it % 100 == 0:
            T = 0.003 * np.exp(-it / 600.0)
            if T > 1e-7:
                noise = np.random.normal(0, T, n)
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 2:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
    return cur, cur_c5

        return budget_s - (time.time() - start_time)

    best_state = [best_c5, best_h.copy(), best_n]
