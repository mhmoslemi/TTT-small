import numpy as np
from scipy.optimize import minimize, Bounds
import time

_params = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _obj(x):
    beta = _params['beta']
    dx = _params['dx']
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max_val(corr, beta) * dx

def _con_eq(x):
    return np.sum(x) - _params['target']

def _con_eq_jac(x):
    return np.ones(x.shape)

def subgrad_topk(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] += (-h[lag:])
        else:
            p = -lag
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] += (-h[:n - p])
    return grad / top_k

def slsqp_round(h_init, n_pts, maxiter=3000):
    _params['target'] = n_pts / 2.0
    _params['dx'] = 2.0 / n_pts
    bounds_s = Bounds(np.zeros(n_pts), np.ones(n_pts))
    constr = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    h_work = h_init.copy()
    work_c5 = compute_c5(h_work)
    for beta in [40, 100, 250, 600, 1500, 4000, 8000]:
        _params['beta'] = float(beta)
        try:
            res = minimize(_obj, h_work, method='SLSQP', bounds=bounds_s,
                           constraints=constr,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_h(res.x, n_pts / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < work_c5 - 1e-14:
                h_work = h_new
                work_c5 = c5_new
        except Exception:
            pass
    return h_work, work_c5

def subgrad_polish(h_init, budget_remaining, top_k_schedule=None):
    n = len(h_init)
    tgt = n / 2.0
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    best_c5 = cur_c5
    best_cur = cur.copy()
    momentum = np.zeros(n)
    t_start = time.time()
    no_improve = 0
    for it in range(50000):
        if time.time() - t_start > budget_remaining:
            break
        if top_k_schedule is not None:

        else:
            tk = 3
        ss = 0.06 / (1.0 + 0.0015 * it)
        g = subgrad_topk(cur, tk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            g = subgrad_topk(cur, 8)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
        g = g / gn
        momentum = 0.75 * momentum + 0.25 * g
