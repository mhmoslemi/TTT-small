import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    v = np.exp(x)
    v_sum = v.sum()
    weights = v / v_sum
    val = (m + np.log(v_sum / len(corr)) / beta) * dx
    threshold = 1e-8
    active = weights > threshold
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        w = weights[idx] * dx
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    return val, grad


def subgrad_descend(h, target_sum, n_steps, time_limit):
    n = len(h)
    t_start = time.time()
    best_c5 = compute_c5(h)
    no_imp = 0
    for it in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = min(5, max(3, n // 40))
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_lags = [int(i) - (n - 1) for i in top_idx]
        grad = np.zeros(n)
        for lag in top_lags:
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            elif -n < lag < 0:
                l = -lag
                grad[:n - l] += (1.0 - h[l:])
                grad[l:] -= h[:n - l]
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        found = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_t = project_h(h - step * grad, target_sum)
            c_t = compute_c5(h_t)
            if c_t < best_c5 - 1e-15:
                h = h_t
                best_c5 = c_t
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp > 300 and it % 200 == 199:
                pert = np.random.randn(n) * 0.005
                h_t = project_h(h + pert, target_sum)
                c_t = compute_c5(h_t)
                if c_t < best_c5:
                    h = h_t
                    best_c5 = c_t
                    no_imp = 0
            if no_imp > 5000:
                break
    return h, best_c5


def lbfgs_optimize(h0, target_sum, betas, max_iter_each, time_limit):
    n = len(h0)
    t_start = time.time()
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time.time() - t_start > time_limit:
            break
        best_in_beta = h_cur.copy()
        best_c5_beta = c5_cur
        for restart in range(3):
            if time.time() - t_start > time_limit:
                break
            def obj(x):
                return smooth_max_and_grad(x, beta)[0]
            def jac(x):
                return smooth_max_and_grad(x, beta)[1]
            x0 = best_in_beta.copy()
            if restart > 0:
                x0 = project_h(x0 + np.random.randn(n) * 0.003, target_sum)
            try:
                res = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': max_iter_each, 'ftol': 1e-16,
                                        'gtol': 1e-14, 'iprint': -1})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5_beta:
                    best_in_beta = h_new.copy()
                    best_c5_beta = c5_new
            except Exception:
                pass
        h_cur = best_in_beta
        c5_cur = best_c5_beta
    return h_cur, c5_cur

    start_time = time.time()
