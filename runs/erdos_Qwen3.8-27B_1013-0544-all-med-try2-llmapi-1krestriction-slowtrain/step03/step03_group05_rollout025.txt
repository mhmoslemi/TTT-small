    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(250, 0.5)

    n = len(h_cur)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = project_h(h_cur, target)

    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    best_state = [best_c5, best_h.copy(), n]

    # Phase 1: SLSQP with increasing beta
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in [80, 200, 500, 1000, 2000, 4000]:
        if time_left() < 40:
            break
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
        try:
            res = minimize(lse_obj, best_state[1], args=(beta, dx), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_state[0]:
                best_state[0] = c5_new
                best_state[1] = h_new.copy()
                best_state[2] = n
        except Exception:
            pass

    # Phase 2: Nesterov subgradient (with momentum)
    if time_left() > 100:
        h_cur = subgrad_descent(best_state[1], n, target, dx,
                                max_iters=6000, time_fn=time_left, min_time=80,
                                best_state=best_state, use_momentum=True)

    # Phase 3: Another SLSQP round with high beta
    if time_left() > 40:
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [1500, 3000, 6000]:
            if time_left() < 20:
                break
            cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
            try:
                res = minimize(lse_obj, best_state[1], args=(beta, dx), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-16})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_state[0]:
                    best_state[0] = c5_new
                    best_state[1] = h_new.copy()
                    best_state[2] = n
            except Exception:
                pass

    # Phase 4: Plain subgradient (no momentum, very fine)
    if time_left() > 60:
        h_cur = subgrad_descent(best_state[1], n, target, dx,
                                max_iters=8000, time_fn=time_left, min_time=30,
                                best_state=best_state, use_momentum=False)

    # Phase 5: Local pair search
    if time_left() > 30:
        h_cur = local_search(best_state[1], n, target, dx,
                             max_iters=80000, time_fn=time_left, min_time=10,
                             best_state=best_state)

    # Phase 6: Final SLSQP polish
    if time_left() > 10:
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [3000, 8000, 15000]:
            if time_left() < 5:
                break
            cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
            try:
                res = minimize(lse_obj, best_state[1], args=(beta, dx), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-17})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_state[0]:
                    best_state[0] = c5_new
                    best_state[1] = h_new.copy()
                    best_state[2] = n
            except Exception:
                pass

    # Phase 7: Ultra-fine subgradient
    if time_left() > 5:
        h_best = best_state[1].copy()
        c5_best = best_state[0]
        for it in range(5000):
            if time_left() < 3:
                break
            corr_val = np.correlate(h_best, 1.0 - h_best, mode="full") * dx
            top_idx = np.argsort(corr_val)[::-1][:3]
            vals = corr_val[top_idx]
            weights = np.exp((vals - vals[0]) * 200.0)
            weights /= weights.sum()
            grad = np.zeros(n)
            center = n - 1
            for t_idx, w in zip(top_idx, weights):
                lag = int(t_idx) - center
                grad += w * grad_for_lag(h_best, lag, n, dx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad /= gnorm
            step = 0.002
            for _ in range(20):
                h_trial = project_h(h_best - step * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_best - 1e-16:
                    h_best = h_trial
                    c5_best = c5_trial
                    best_state[0] = c5_best
                    best_state[1] = h_best.copy()
                    break
                step *= 0.5

    final_h = best_state[1]
    final_c5 = float(compute_c5(final_h))
    return final_h, final_c5, n
