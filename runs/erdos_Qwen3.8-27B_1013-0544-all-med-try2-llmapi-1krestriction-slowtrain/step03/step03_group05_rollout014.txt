def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

def grad_lag(h, lag, n, dx):
    """Gradient of overlap at a specific lag."""
    grad = np.zeros(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()

            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:

        h = np.clip(h, 0.0, 1.0)

def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:]) * dx
        grad[lag:] += -h[:n - lag] * dx
    else:
