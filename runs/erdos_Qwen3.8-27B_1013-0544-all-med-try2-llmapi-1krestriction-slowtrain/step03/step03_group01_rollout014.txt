        h_new = np.clip(h + v, 0.0, 1.0)
        # Project to satisfy sum constraint

        c = compute_c5(h, n)
        if c < best_c:

        if it % 50 == 49:

            else:
                lr *= 1.1
    return best_h, best_c

                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= s * d
                            h[j] += s * d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                return np.clip(h, 0, 1)
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                return np.clip(h, 0, 1)
            h -= (-diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full')
    c = c_full * dx
    m = c.max()
    exp_c = np.exp(c - m)
    s = exp_c.sum()
    f = (m + np.log(s)) / alpha
    w = exp_c / s
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        L = m_idx - (n - 1)
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        jp = np.arange(n) + L
        mask_p = (jp >= 0) & (jp < n)
        if mask_p.any():
            grad[mask_p] += wm * (1.0 - h[jp[mask_p]])
        jm = np.arange(n) - L
        mask_m = (jm >= 0) & (jm < n)
        if mask_m.any():
            grad[mask_m] -= wm * h[jm[mask_m]]
    grad *= dx
    return f, grad
