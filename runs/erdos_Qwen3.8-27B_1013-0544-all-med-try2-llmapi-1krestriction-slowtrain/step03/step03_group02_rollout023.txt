def smooth_max(x, beta):

        if c_t < c_best - 1e-14:
            cur = trial
            c_best = c_t
            step = min(step * 1.01, 0.1)
        else:
            step *= 0.96
            if step < 0.0003:
                break
    return cur, c_best

def coord_desc(cur, target, t0, budget):
    n = len(cur)
    c_best = compute_c5(cur)
    deltas = (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(20):
        if budget - (time.time() - t0) < 5:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 3:
                return cur, c_best
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_t = compute_c5(trial)
                if c_t < c_best - 1e-14:
                    cur = trial
                    c_best = c_t
                    improved = True
                    break
        if not improved:
            break
    return cur, c_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(320, 0.5)

    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: Momentum subgradient
    cur = best_h.copy()
    cur, c1 = momentum_subgrad(cur, target, n, 4000, 0.04, t0, budget_s, top_k=12)
    if c1 < best_c5:
        best_h, best_c5 = cur.copy(), c1

    # Phase 2: Binary + pair flip
    if budget_s - (time.time() - t0) > 120:
        h_bin = binary_round(cur, target)
        h_bin, c_bin = pair_flip(h_bin, target, 25000, t0, budget_s)
        if c_bin < best_c5:
            best_h, best_c5 = h_bin.copy(), c_bin
            cur = h_bin.astype(float).copy()
        # Also try from best_h
        h_bin2 = binary_round(best_h, target)
        h_bin2, c_bin2 = pair_flip(h_bin2, target, 10000, t0, budget_s)
        if c_bin2 < best_c5:
            best_h, best_c5 = h_bin2.copy(), c_bin2
            cur = h_bin2.astype(float).copy()

    # Phase 3: SLSQP smooth max
    if budget_s - (time.time() - t0) > 100:
        for beta in [300, 800, 2000]:
            if budget_s - (time.time() - t0) < 25:
                break
            cur = do_slsqp(best_h, n, target, beta, dx, bounds, 2000, 1e-14)
            c_s = compute_c5(cur)
            if c_s < best_c5:
                best_h, best_c5 = cur.copy(), c_s

    # Phase 4: Block coordinate
    if budget_s - (time.time() - t0) > 60:
        cur, c_bc = block_opt(best_h, target, 5, 4, t0, budget_s)
        if c_bc < best_c5:
            best_h, best_c5 = cur.copy(), c_bc

    # Phase 5: Fine subgradient
    if budget_s - (time.time() - t0) > 40:
        cur, c_f = momentum_subgrad(best_h, target, n, 6000, 0.012, t0, budget_s, top_k=20)
        if c_f < best_c5:
            best_h, best_c5 = cur.copy(), c_f

    # Phase 6: Perturbation + re-optimize
    if budget_s - (time.time() - t0) > 50:
        for att in range(5):
            if budget_s - (time.time() - t0) < 10:
                break
            scale = 0.015 / (att + 1)
            hp = project_h(best_h + np.random.randn(n) * scale, target)
            hp = do_slsqp(hp, n, target, 500, dx, bounds, 800, 1e-13)
            c_p = compute_c5(hp)
            if c_p < best_c5:
                best_h, best_c5 = hp.copy(), c_p

    # Phase 7: Coordinate descent polish
    if budget_s - (time.time() - t0) > 15:
        best_h, best_c5 = coord_desc(best_h, target, t0, budget_s)

    # Phase 8: Final ultra-fine subgradient
    if budget_s - (time.time() - t0) > 10:
        cur, c_fin = momentum_subgrad(best_h, target, n, 5000, 0.005, t0, budget_s, top_k=30)
        if c_fin < best_c5:
            best_h, best_c5 = cur.copy(), c_fin

    return best_h, float(best_c5), n
