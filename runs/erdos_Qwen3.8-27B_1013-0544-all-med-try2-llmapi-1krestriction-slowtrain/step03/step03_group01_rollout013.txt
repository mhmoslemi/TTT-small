    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def subsgrad_refine(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    v = np.zeros(n)  # momentum
    
    eta = 0.01  # initial step size
    top_m = 3  # number of top shifts to average
    
    t = 0
    no_improve = 0
    
    while time.time() < deadline:
        t += 1
        c_vals = get_correlations(h, n)
        current_max = c_vals.max()
        
        # Find top-m shifts
        top_indices = np.argsort(c_vals)[-top_m:]
        
        # Average subgradient
        g = np.zeros(n)
        for k_idx in top_indices:
            k = k_idx - (n - 1)  # convert to lag
            g += subgradient_at_shift(h, n, k)
        g /= top_m
        
        # Nesterov momentum
        momentum = 0.9 if t < 100 else 0.95
        v = momentum * v - eta * g
        h_new = project_feasible(h + v, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            # Reject, but keep momentum
            no_improve += 1
            if no_improve > 50:
                # Adaptive step reduction
                eta *= 0.8
                if eta < 1e-6:
                    # Try a restart
                    h = project_feasible(best_h + 0.05 * rng.randn(n), n)
                    v *= 0.1
                    eta = 0.01
                    no_improve = 0
                    top_m = min(top_m + 2, 10)
        
        # Periodically increase top_m for smoother gradients
        if t % 200 == 0:
            top_m = min(top_m + 1, 15)
        
        # Periodically decrease top_m for sharper gradients near convergence
        if best_c < 0.382 and t % 500 == 0:
            top_m = max(top_m - 1, 2)
    
    return best_h, best_c

For the current max shift k*, I want to find a pair (a, b) and a step δ such that:
- h_a increases by δ, h_b decreases by δ (maintaining sum constraint)
- The max correlation decreases

The change in c_k for all k is:
Δc_k = δ * [g_k(a) - g_k(b)] where g_k is the subgradient at shift k.

We want Δc_k ≤ 0 for all k that are currently at or near the max.

1. Find k* = argmax c_k
2. Find (a, b) that maximizes g_k*(a) - g_k*(b) (to decrease c_k*)

5. If yes, accept; if no, try smaller δ or different pair

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def subgrad_shift(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    lo = max(0, k)
    hi = min(n, n + k)
    if hi > lo:
        g[lo:hi] += (1.0 - h[lo - k:hi - k])
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 + k:hi2 + k]
    return g * dx
