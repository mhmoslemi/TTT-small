import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def _obj_fn(x, beta, top_k):
    return smoothmax_val_grad(x, beta, top_k)[0]


def _grad_fn(x, beta, top_k):
    return smoothmax_val_grad(x, beta, top_k)[1]


def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0


def _cons_jac(x):
    return np.ones(len(x))


def slsqp_refine(h0, beta, maxiter, top_k):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac},)
    try:
        res = minimize(_obj_fn, h0, jac=_grad_fn, args=(beta, top_k),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h0, n / 2.0)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def fourier_noise(n, rng, noise_scale, max_freq=5):
    f = np.zeros(n, dtype=complex)
    for k in range(1, min(max_freq, n // 2) + 1):
        amp = rng.randn() * noise_scale / k
        phase = rng.rand() * 2 * np.pi
        f[k] = amp * np.exp(1j * phase)
        f[n - k] = np.conj(f[k])
    noise = np.real(np.fft.ifft(f))
    return noise


def basin_hop(h_best, best_c5, n_hops, beta_list, maxiter, noise_base,
              noise_decay, seed, t0, budget_s, max_freq=4):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 40:
            break
        noise = fourier_noise(n, rng, noise_base * (noise_decay ** hop), max_freq)
        h_pert = project_feasible(h_best + noise, target)
        h_local = h_pert
        for beta in beta_list:
            if budget_s - (time.time() - t0) < 25:
                break
            h_local = slsqp_refine(h_local, beta, maxiter, top_k=20)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5


def generate_cosine_init(n, freq1, freq2, amp, rng):
    x = np.linspace(0.5 / n * 2.0, 2.0 - 0.5 / n * 2.0, n)
    h = 0.5 + amp * np.cos(np.pi * freq1 * x + rng.rand() * 2 * np.pi) \
        + amp * 0.5 * np.cos(np.pi * freq2 * x + rng.rand() * 2 * np.pi)
    return project_feasible(np.clip(h, 0.0, 1.0), n / 2.0)
