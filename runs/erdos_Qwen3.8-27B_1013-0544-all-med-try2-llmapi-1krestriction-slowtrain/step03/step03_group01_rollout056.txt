def project_feasible_fast(h, n):
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # clip(h - lambda, 0, 1) = max(0, min(1, h - lambda))
    # As lambda increases, each component decreases from h[i] (when h[i]-lambda > 0 and < 1)
    # to 0 (when h[i]-lambda <= 0)
    # The sum is a piecewise linear decreasing function of lambda
    
    h_clipped = np.clip(h, 0, 1)
    s = h_clipped.sum()
    if abs(s - target) < 1e-13 * n:
        return h_clipped
    
    if s > target:
        # Need to decrease sum by (s - target)
        # Reducing lambda by delta reduces sum by delta * (number of components in (0,1))
        # until some hit 0
        # Binary search on lambda
        lo, hi = 0, 1
        for _ in range(60):
            mid = (lo + hi) / 2
            proj = np.clip(h_clipped - mid, 0, 1)
            if proj.sum() > target:
                lo = mid
            else:
                hi = mid
        result = np.clip(h_clipped - (lo+hi)/2, 0, 1)
    else:
        # Need to increase sum by (target - s)
        lo, hi = -1, 0
        for _ in range(60):
            mid = (lo + hi) / 2
            proj = np.clip(h_clipped + (-mid), 0, 1)  # h_clipped - mid where mid < 0
            if proj.sum() < target:
                lo = mid
            else:
                hi = mid
        result = np.clip(h_clipped - (lo+hi)/2, 0, 1)
    
    # Final adjustment (should be very small)
    diff = target - result.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - result
            mask = room > 1e-15
            if mask.sum() > 0:
                result[mask] += diff / mask.sum()
        else:
            mask = result > 1e-15
            if mask.sum() > 0:
                result[mask] += diff / mask.sum()
    return np.clip(result, 0, 1)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    sum_exp = exp_shifted.sum()
    value = m + (shifted_max + np.log(sum_exp)) / alpha
    weights = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += weights[k] * (1 - h[s:]) * dx
            if s < n:
                grad[s:] -= weights[k] * h[:n-s] * dx
        else:
            p = -s
            if n - p > 0:
                grad[p:] += weights[k] * (1 - h[:n-p]) * dx
            if p < n:
                grad[:n-p] -= weights[k] * h[p:] * dx
    return value, grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        v, _ = smooth_max_and_grad(np.clip(x, 0, 1), n, alpha)
        return v
    def obj_grad(x):
        v, g = smooth_max_and_grad(np.clip(x, 0, 1), n, alpha)
        return v, g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj_grad, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h
