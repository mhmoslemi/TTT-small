    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max_obj_grad(h, dx, beta, penalty):
    """Compute smooth max objective and gradient with penalty for sum constraint."""
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    val = dx * (m + np.log(exps.sum() / len(corr)) / beta)
    
    # Penalty for sum constraint
    sum_diff = h.sum() - target
    val += penalty * sum_diff ** 2
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[n-1+k]
        if w < 1e-14:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:-k])
            grad[:n-k] -= w * h[k:]
        else:
            m2 = -k
            grad[:n-m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n-m2]
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * penalty * sum_diff
    
    return val, grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def sm_val(x, dx, beta, penalty):
    n = len(x)
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    val = dx * (m + np.log(exps.sum() / len(corr)) / beta)
    sd = x.sum() - target
    val += penalty * sd * sd
    return val


def sm_grad(x, dx, beta, penalty):
    n = len(x)
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    wts = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = wts[n - 1 + k]
        if w < 1e-14:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            grad[k:] += w * (1.0 - x[:-k])
            grad[:n - k] -= w * x[k:]
        else:
            m2 = -k
            grad[:n - m2] += w * (1.0 - x[m2:])
            grad[m2:] -= w * x[:n - m2]
    grad *= dx
    sd = x.sum() - target
    grad += 2.0 * penalty * sd
    return grad


def slsqp_obj(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    return dx * (m + np.log(exps.sum() / len(corr)) / beta)


def slsqp_eq(x, target):
    return x.sum() - target


def slsqp_eq_jac(x):
    return np.ones(len(x))


def interp_h(h_old, n_new, target):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_h(np.interp(x_new, x_old, h_old), target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with smooth max at current resolution
    n_cur = len(best_h)
    h_cur = best_h.copy()
    target = n_cur / 2.0
    dx = 2.0 / n_cur
    bounds = [(0.0, 1.0)] * n_cur

    for beta in [30, 60, 120, 250, 500, 1000, 2500, 5000]:
        if time_left() < 40:
            break
        for pen in [500, 2000]:
            if time_left() < 15:
                break
            try:
                res = minimize(sm_val, h_cur, jac=sm_grad, args=(dx, beta, pen),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-10,
                                        'maxls': 50})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h_cur = h_new
            except Exception:
                pass

    # Phase 2: Resolution cycling with L-BFGS-B
    resolutions = [128, 160, 192, 200, 224]
    for n_t in resolutions:
        if time_left() < 80:
            break
        target_t = n_t / 2.0
        dx_t = 2.0 / n_t
        bounds_t = [(0.0, 1.0)] * n_t
        h_t = interp_h(best_h, n_t, target_t)

        for beta in [50, 100, 300, 800, 2000, 5000]:
            if time_left() < 20:
                break
            for pen in [500, 2000]:
                if time_left() < 10:
                    break
                try:
                    res = minimize(sm_val, h_t, jac=sm_grad, args=(dx_t, beta, pen),
                                   method='L-BFGS-B', bounds=bounds_t,
                                   options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-10,
                                            'maxls': 50})
                    h_new = project_h(res.x, target_t)
                    c5_new = compute_c5(h_new)

                        best_h = h_new.copy()
                        best_c5 = c5_new
                    h_t = h_new
                except Exception:
                    pass

    # Phase 3: Subgradient refinement with periodic L-BFGS restarts
    n_final = len(best_h)
    target_final = n_final / 2.0
    dx_final = 2.0 / n_final
    bounds_final = [(0.0, 1.0)] * n_final
    no_improve = 0

    for iteration in range(30000):
        if time_left() < 5:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_active = min(5, max(3, n_final // 40))
        top_idx = np.argsort(corr)[::-1][:n_active]
        top_lags = [int(i) - (n_final - 1) for i in top_idx]
        grad = compute_subgrad(best_h, top_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm

        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_h(best_h - step * grad, target_final)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = c5_trial
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve % 500 == 499 and time_left() > 30:
                # Basin hopping: perturb + reoptimize
                scale = 0.02 + 0.03 * (no_improve % 5) / 5.0
                perturb = np.random.randn(n_final) * scale
                h_pert = project_h(best_h + perturb, target_final)
                c5_pert = compute_c5(h_pert)
                try:
                    res = minimize(sm_val, h_pert, jac=sm_grad,
                                   args=(dx_final, 500, 1000),
                                   method='L-BFGS-B', bounds=bounds_final,
                                   options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-9})
                    h_new = project_h(res.x, target_final)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        no_improve = 0
                    elif c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                        no_improve = 0
                except Exception:
                    if c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                        no_improve = 0
            if no_improve > 5000:
                break

    # Phase 4: SLSQP polish with equality constraint
    if time_left() > 15:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        dx_pol = 2.0 / n_pol
        bounds_pol = Bounds(np.zeros(n_pol), np.ones(n_pol))
        for beta in [2000, 5000, 10000]:
            if time_left() < 8:
                break
            try:
                res = minimize(slsqp_obj, best_h, args=(dx_pol, beta),
                               method='SLSQP', bounds=bounds_pol,
                               constraints=[{'type': 'eq', 'fun': slsqp_eq,
                                             'jac': slsqp_eq_jac, 'args': (target_pol,)}],
                               options={'maxiter': 5000, 'ftol': 1e-18})
                h_new = project_h(res.x, target_pol)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 5: Final subgradient sweep (very fine steps)
    n_fin = len(best_h)
    target_fin = n_fin / 2.0
    for iteration in range(10000):
        if time_left() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        top_lags = [int(i) - (n_fin - 1) for i in top_idx]

        if gnorm < 1e-14:

        grad /= gnorm
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = project_h(best_h - step * grad, target_fin)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = c5_trial
                break

    return best_h, best_c5, len(best_h)
