        klo, khi = max(-m, -(n-1)), min(n-1-m, n-1)
        if klo <= khi:
            a[klo+n-1:khi+n] += s * (1.0 - h[m+klo:m+khi+1])
        klo2, khi2 = max(m-n+1, -(n-1)), min(m, n-1)
        if klo2 <= khi2:
            a[klo2+n-1:khi2+n] -= s * h[m-klo2:m-khi2+1][::-1]
    return a

def find_optimal_d(corr, a, d_min, d_max):
    """Find d minimizing max_k(corr + d*a) over [d_min, d_max]."""
    if d_min >= d_max - 1e-12:
        return 0.0
    d_grid = np.linspace(d_min, d_max, 30)
    t_grid = np.max(corr[None, :] + d_grid[:, None] * a[None, :], axis=1)
    bi = np.argmin(t_grid)
    d_lo = d_grid[max(0, bi - 1)]
    d_hi = d_grid[min(len(d_grid) - 1, bi + 1)]
    for _ in range(30):
        d_mid = (d_lo + d_hi) / 2.0
        t_mid = np.max(corr + d_mid * a)
        act = np.abs(corr + d_mid * a - t_mid) < 1e-9
        if not act.any():
            act = np.zeros(len(corr), dtype=bool)
            act[np.argmax(corr + d_mid * a)] = True
        s_lo, s_hi = np.min(a[act]), np.max(a[act])
        if s_lo > 1e-12:
            d_hi = d_mid
        elif s_hi < -1e-12:
            d_lo = d_mid
        else:
            break
    return (d_lo + d_hi) / 2.0

def full_sweep_2coord(h, dx, target, n_iter=1, rng=None):
    """Do full sweeps of all 2-coordinate pairs."""
    n = len(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    total_improvements = 0
    for _ in range(n_iter):
        any_improved = False
        for i in range(n):
            for j in range(i + 1, n):
                d_min = max(-best_h[i], best_h[j] - 1.0)
                d_max = min(1.0 - best_h[i], best_h[j])
                if d_min >= d_max - 1e-12:
                    continue
                a = compute_a_vec(best_h, i, j)
                d_opt = find_optimal_d(corr, a, d_min, d_max)
                c5_new = np.max(corr + d_opt * a) * dx
                if c5_new < best_c5 - 1e-13:
                    best_h[i] += d_opt
                    best_h[j] -= d_opt
                    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
                    best_c5 = c5_new
                    any_improved = True
                    total_improvements += 1
        if not any_improved:
            break
    return best_h, best_c5, total_improvements

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    dx = 2.0 / n

    # Phase 1: Full 2-coordinate sweeps (exact moves)
    for sweep_round in range(8):
        if time_left() < 80:
            break
        n_iter = 3 if sweep_round < 3 else 2
        h_cur, c5_cur, n_imp = full_sweep_2coord(best_h, dx, target, n_iter=n_iter, rng=rng)
        if c5_cur < best_c5 - 1e-14:
            best_h = h_cur.copy()
            best_c5 = c5_cur
        if n_imp == 0:
            break

    # Phase 2: SLSQP with smooth max at current resolution
    if time_left() > 40:
        h_cur = best_h.copy()
        bounds = Bounds(np.zeros(n), np.ones(n))
        con = make_eq_con(target)
        jac = make_eq_jac((n,))
        cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
        for beta in [50, 200, 800, 3000, 10000]:
            if time_left() < 15:
                break
            obj = make_slsqp_obj(dx, beta)
            try:
                res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_cur) - 1e-14:
                    h_cur = h_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 3: More 2-coordinate sweeps after SLSQP
    if time_left() > 30:
        for _ in range(5):
            if time_left() < 15:
                break
            h_cur, c5_cur, n_imp = full_sweep_2coord(best_h, dx, target, n_iter=2, rng=rng)
            if c5_cur < best_c5 - 1e-14:
                best_h = h_cur.copy()
                best_c5 = c5_cur
            if n_imp == 0:
                break

    # Phase 4: Try at different resolutions
    if time_left() > 50:
        for n_alt in [150, 100, 300, 400]:
            if time_left() < 25:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_alt)
            h_alt = project_h(np.interp(x_new, x_old, best_h), n_alt / 2.0)
            dx_alt = 2.0 / n_alt
            target_alt = n_alt / 2.0
            bounds_alt = Bounds(np.zeros(n_alt), np.ones(n_alt))
            con_alt = make_eq_con(target_alt)
            jac_alt = make_eq_jac((n_alt,))
            cons_alt = [{'type': 'eq', 'fun': con_alt, 'jac': jac_alt}]
            for beta in [200, 1000, 5000]:
                if time_left() < 10:
                    break
                obj_alt = make_slsqp_obj(dx_alt, beta)
                try:
                    res = minimize(obj_alt, h_alt, method='SLSQP', bounds=bounds_alt,
                                   constraints=cons_alt, options={'maxiter': 2000, 'ftol': 1e-15})
                    h_new = project_h(res.x, target_alt)
                    c5_new = compute_c5(h_new)
                    if c5_new < compute_c5(h_alt) - 1e-14:
                        h_alt = h_new
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                except Exception:
                    pass

    # Phase 5: Final 2-coordinate polish
    if time_left() > 10:
        n_final = len(best_h)
        dx_final = 2.0 / n_final
        target_final = n_final / 2.0
        for _ in range(4):
            if time_left() < 5:
                break
            h_cur, c5_cur, _ = full_sweep_2coord(best_h, dx_final, target_final, n_iter=2, rng=rng)
            if c5_cur < best_c5 - 1e-14:
                best_h = h_cur.copy()
                best_c5 = c5_cur
            else:
                break

    # Phase 6: Subgradient with active lags for final micro-improvement
    if time_left() > 8:
        n_s = len(best_h)
        target_s = n_s / 2.0
        h_s = best_h.copy()
        for it in range(20000):
            if time_left() < 2:
                break
            corr = np.correlate(h_s, 1.0 - h_s, mode="full")
            top_k = min(5, max(3, n_s // 30))
            top_idx = np.argsort(corr)[::-1][:top_k]
            top_lags = [int(idx) - (n_s - 1) for idx in top_idx]
            grad = compute_subgrad(h_s, top_lags) / top_k
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            grad /= gnorm
            improved = False
            for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
                h_trial = project_h(h_s - step * grad, target_s)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    h_s = h_trial
                    best_h = h_s.copy()
                    best_c5 = c5_trial
                    improved = True
                    break
            if not improved and it % 500 == 499:
                perturb = np.random.randn(n_s) * 0.003
                h_trial = project_h(h_s + perturb, target_s)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    h_s = h_trial
                    best_h = h_s.copy()
                    best_c5 = c5_trial

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
