def balanced_subgradient(h, S):
    """Compute the balanced subgradient for the active set S."""
    n = len(h)
    grad = np.zeros(n)
    for k in S:
        lag = k - (n - 1)  # Convert from correlation index to actual lag
        if lag >= 0:
            grad[:n-lag] += (1 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            la = -lag
            grad[la:] += (1 - h[:n-la])
            grad[:n-la] -= h[la:]
    return grad / len(S)

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_lags(h, top_k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    return corr, top_idx

def grad_for_lag(h, lag):
    """Gradient of corr at given lag with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[:n-lag] += (1 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        la = -lag
        g[la:] += (1 - h[:n-la])
        g[:n-la] -= h[la:]
    return g

def balanced_grad(h, top_idx, top_k):
    """Balanced subgradient over top_k lags."""
    n = len(h)
    g = np.zeros(n)
    for idx in top_idx[:top_k]:
        lag = int(idx) - (n - 1)
        g += grad_for_lag(h, lag)
    return g / top_k

    """Projected subgradient with Nesterov acceleration."""
    mom = np.zeros(n)
    best = compute_c5(cur)
    y = cur.copy()
    prev_t = 1.0
    stall = 0
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        # Nesterov extrapolation
        t = (1 + np.sqrt(1 + 4 * prev_t * prev_t)) / 2

        if gn < 1e-14:
            break
        g /= gn
        step = base_step / (1 + 0.1 * it)

        trial = project_h(cur - step * g, target)
        c_try = compute_c5(trial)
        
        if c_try < best - 1e-15:
            cur = trial
            y = y_new  # (this isn't quite right for Nesterov, but close enough)
            best = c_try
            stall = 0

                    best = c_try
                    stall = 0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        la = -lag
        if la < n:
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g

def subgrad_optimize(cur, target, n, max_iter, t0, budget, step0=0.025):
    best = compute_c5(cur)
    mom = np.zeros(n)
    stall = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += grad_for_lag(cur, lag)
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = step0 * (0.997 ** it)
        mom = 0.85 * mom - step * g
        trial = project_h(cur + mom, target)
        c = compute_c5(trial)
        if c < best - 1e-15:
            cur = trial
            best = c
            stall = 0
        else:
            stall += 1
            mom *= 0.4
            trial = project_h(cur + mom, target)
            c = compute_c5(trial)
            if c < best - 1e-15:
                cur = trial
                best = c
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 200:
            break
    return cur, best

def coord_descent(h, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(h)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 1:
            break
        improved = False
        base_step = 0.03 * (0.7 ** (sw // 3))  # Anneal step size
        for j in range(n):
            if budget - (time.time() - t0) < 0.5:
                break
            for mult in [1.0, -1.0, 0.5, -0.5, 0.25, -0.25, 0.1, -0.1]:
                d = base_step * mult
                nv = h[j] + d
                if nv < 0.0 or nv > 1.0:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return h, best_c5

def pair_refine(h, target, n, t0, budget, max_iters=5000):
    best_c5 = compute_c5(h)
    for _ in range(max_iters):
        if budget - (time.time() - t0) < 1:
            break

        m = np.random.randint(n)
        if j == m:
            continue

            nm = h[m] - d
            if nj < 0 or nj > 1 or nm < 0 or nm > 1:

            trial = h.copy()
            trial[j] = nj
            trial[m] = nm
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                h = trial
                best_c5 = c
                break
    return h, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

    return h, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
