    n = len(h0)
    h = h0.copy()
    best = compute_c5(h)
    for it in range(max_iters):
        corr = overlaps_of(h)
        top_idx = int(np.argmax(corr))
        k = top_idx - (n - 1)
        improved = False
        if 0 < abs(k) < n:
            abs_k = abs(k)
            n_valid = n - abs_k
            for _ in range(8):
                i = np.random.randint(n_valid)
                j = i + abs_k if k > 0 else i - k
                for d in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                    h_t = h.copy()
                    h_t[i] += d
                    h_t[j] -= d
                    if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                        h_t = project_h(h_t, target_sum)
                        c = compute_c5(h_t)
                        if c < best - 1e-16:
                            h = h_t
                            best = c
                            improved = True
                            break
                if improved:
                    break
        if not improved and it % 300 == 299:
            p = np.random.randn(n) * 0.002
            h_t = project_h(h + p, target_sum)
            c = compute_c5(h_t)
            if c < best:
                h = h_t
                best = c
        if not improved and it > 2000:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 200
        best_h = np.full(n, 0.5)

    target_sum = n / 2.0
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing beta
    betas = [15, 30, 60, 120, 250, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas:
        if tl() < 40:
            break
        h_new = lbfgsb_round(best_h, target_sum, beta, maxiter=200, inner=3)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    # Phase 2: Subgradient descent
    if tl() > 60:
        h_new, c_new = subgrad_descent(best_h, target_sum, max_iters=15000,
                                       n_active_min=2, n_active_max=5,
                                       base_step=0.025, perturb_every=60,
                                       perturb_scale=0.005)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    # Phase 3: More L-BFGS-B polish
    for beta in [5000, 15000, 40000]:
        if tl() < 20:
            break
        h_new = lbfgsb_round(best_h, target_sum, beta, maxiter=400, inner=3)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    # Phase 4: Fine subgradient with smaller steps
    if tl() > 40:
        h_new, c_new = subgrad_descent(best_h, target_sum, max_iters=20000,
                                       n_active_min=2, n_active_max=4,
                                       base_step=0.01, perturb_every=100,
                                       perturb_scale=0.003)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    # Phase 5: Pairwise moves
    if tl() > 30:
        h_new, c_new = pairwise_moves(best_h, target_sum, max_iters=50000)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    # Phase 6: Final L-BFGS-B at very high beta
    if tl() > 15:
        h_new = lbfgsb_round(best_h, target_sum, 100000, maxiter=500, inner=2)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    # Phase 7: Ultra-fine subgradient
    if tl() > 10:
        h_new, c_new = subgrad_descent(best_h, target_sum, max_iters=10000,
                                       n_active_min=1, n_active_max=3,
                                       base_step=0.005, perturb_every=150,
                                       perturb_scale=0.001)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
