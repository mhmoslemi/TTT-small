import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smax_val(c, beta):
    m = np.max(c)
    s = np.clip(beta * (c - m), -200, 200)
    return m + (np.log(np.sum(np.exp(s))) - np.log(len(c))) / beta

def smax_w(c, beta):
    m = np.max(c)
    s = np.clip(beta * (c - m), -200, 200)
    e = np.exp(s)
    return e / e.sum()

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_gradient(h, beta, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    obj = smax_val(c, beta) * dx
    w = smax_w(c, beta)
    g = np.zeros(n)
    thr = 1e-12
    idxs = np.where(w > thr)[0]
    for idx in idxs:
        wi = w[idx]
        lag = int(idx) - (n - 1)
        if lag >= 0:
            g[:n - lag] += wi * (1.0 - h[lag:])
            g[lag:] -= wi * h[:n - lag]
        else:
            la = -lag
            g[la:] += wi * (1.0 - h[:n - la])
            g[:n - la] -= wi * h[la:]
    g *= dx
    return obj, g

def sum_eq_fun(x, tgt):
    return np.sum(x) - tgt

def sum_eq_jac(x, tgt):
    return np.ones(x.shape)

def make_slsqp_fn(h, beta, dx):
    o, g = obj_gradient(h, beta, dx)
    return o, g

def do_slsqp(h0, n, tgt, beta, bounds, maxiter, ftol):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (tgt,)}]
    def f(x):
        return obj_gradient(x, beta, dx)
    try:
        r = minimize(f, h0, method='SLSQP', bounds=bounds, constraints=cons, jac=True,
                     options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_h(r.x, tgt)
    except Exception:
        return h0

def subgrad_opt(cur, tgt, n, max_iter, t0, budget):

    best_c = compute_c5(cur)
    step = 0.015
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        c = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = int(np.argmax(c))
        lag = top_idx - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[:n - lag] += (1.0 - cur[lag:])
            g[lag:] -= cur[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - cur[:n - la])
            g[:n - la] -= cur[la:]
        g *= dx
        trial = project_h(cur - step * g, tgt)
        c_try = compute_c5(trial)
        if c_try < best_c - 1e-15:
            cur = trial
            best_c = c_try

                step = min(0.03, step * 1.1)
        else:
            step *= 0.6
            trial = project_h(cur - step * g, tgt)
            c_try = compute_c5(trial)
            if c_try < best_c - 1e-15:

def coord_desc(cur, tgt, n, sweeps, t0, budget, base_step=0.008):
