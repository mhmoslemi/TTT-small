    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target_sum):
    n = len(h)
    if n == 0:
        return h

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
