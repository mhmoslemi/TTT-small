def binary_local_search_efficient(h, max_iters=5000, rng=None):
    n = len(h)
    target_ones = n // 2
    h = h.astype(float).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    
    # Ensure correct count
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    
    while len(ones) > target_ones:
        i = ones.pop(rng.randint(len(ones)))
        h[i] = 0.0
        zeros.append(i)
    while len(ones) < target_ones:
        j = zeros.pop(rng.randint(len(zeros)))
        h[j] = 1.0
        ones.append(j)
    
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = np.max(c) * dx
    
    for it in range(max_iters):
        # Compute g for all positions
        # g[p] is a vector of length 2n-1
        g = np.zeros((n, 2*n-1))
        for p in range(n):
            for lag in range(-(n-1), n):
                k = lag + (n-1)
                coeff = 0.0
                pp = p + lag
                pm = p - lag
                if 0 <= pp < n:
                    coeff += (1.0 - h[pp])
                if 0 <= pm < n:
                    coeff -= h[pm]
                g[p, k] = coeff
        
        # Find best swap
        best_improve = 1e-12
        best_swap = None
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # New correlation: c + g[j] - g[i]
                new_c = c + g[j] - g[i]
                new_c5 = np.max(new_c) * dx
                improve = best_c5 - new_c5
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        c = c + g[j] - g[i]
        best_c5 = np.max(c) * dx
    
    return h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_binary(h, n, rng):
    """Convert h to binary with exactly n//2 ones."""
    target = n // 2
    # Sort by value, keep top n//2 as ones
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)

def continuous_refine(h, mu=100.0, max_rounds=50):
    """Coordinate descent with penalty for sum constraint."""
    n = len(h)
    target = n / 2.0
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = np.sum(h)
    
    for rnd in range(max_rounds):
        max_change = 0.0
        for j in range(n):
            old_v = h[j]
            # Compute g vector
            g = np.zeros(2*n-1)
            for lag in range(-(n-1), n):
                k = lag + (n-1)
                coeff = 0.0
                jp = j + lag
                jm = j - lag
                if 0 <= jp < n:
                    coeff += (1.0 - h[jp])
                if 0 <= jm < n:
                    coeff -= h[jm]
                g[k] = coeff
            
            def obj(v):
                c_new = corr + (v - old_v) * g
                return np.max(c_new) + mu * (s + (v - old_v) - target) ** 2
            
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            new_v = res.x
            delta = new_v - old_v
            if abs(delta) > 1e-10:
                h[j] = new_v
                s += delta
                corr = corr + delta * g
                max_change = max(max_change, abs(delta))
        
        if max_change < 1e-8:
            break
    
    # Project to exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h = np.clip(h + diff / n, 0, 1)
        # Iterate a few times
        for _ in range(10):
            diff2 = target - np.sum(h)
            if abs(diff2) < 1e-12:
                break
            h = np.clip(h + diff2 / n, 0, 1)
    
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n, rng=None):
    target = n // 2
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    h_bin[order[:target]] = 1.0

def binary_local_search(h, max_iters=10000, time_limit=30.0, allow_worse=False):
    n = len(h)
    target = n // 2
    h = make_binary(h, n)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])

    best_c5 = float(np.max(c)) * dx
    n_corr = 2 * n - 1
    new_c = np.empty(n_corr)

                    if 0 <= im < n:

                    if 0 <= jp < n:
                        delta += (1.0 - h[jp])
                    if 0 <= jm < n:
                        delta -= h[jm]
                    if L == j - i:
                        delta += 1.0

                        delta += 1.0
