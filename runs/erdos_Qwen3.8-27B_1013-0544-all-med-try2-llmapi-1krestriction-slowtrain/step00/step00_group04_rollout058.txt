```python
def compute_c5(h, dx):

def compute_overlaps(h):
    """Compute all overlap values (before multiplying by dx)"""
    return np.correlate(h, 1-h, mode='full')

def smooth_objective(h, dx, beta):
    overlaps = compute_overlaps(h)
    # log-sum-exp for numerical stability
    max_ov = np.max(overlaps)
    lse = max_ov + np.log(np.sum(np.exp(beta * (overlaps - max_ov))))
    return lse / beta

def gradient(h, beta):

    n = len(h)
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)

    weights /= weights.sum()

            if 0 <= hp < n:

            if 0 <= hm < n:

    return grad
```

```python
    max_ov = np.max(overlaps)
    logits = beta * (overlaps - max_ov)
    weights = np.exp(logits)
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_constraints(v, target):
    n = len(v)
    lo = -v.max() - 0.1
    hi = 1.0 - v.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    n = len(h)
    ov = compute_overlaps(h)
    mx = ov.max()
    w = np.exp(beta * (ov - mx))
    sw = w.sum()
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    grad = np.zeros(n)
    half = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - half
        if s >= 0:
            # valid i: 0 <= i < n-s
            grad[:n-s] += wi * (1.0 - h[s:])
            if s > 0:
                grad[s:] -= wi * h[:n-s]
        else:
            ss = -s
            # valid i: ss <= i < n
            grad[ss:] += wi * (1.0 - h[:n-ss])
            grad[:n-ss] -= wi * h[ss:]
    return grad

def opt_continuous(h0, target, beta, t_limit, lr_init=0.1):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    lr = lr_init
    for _ in range(100000):
        if time.time() - t0 > t_limit:
            break
        g = smooth_grad(h, beta)
        h_new = h - lr * g
        h_new = project_constraints(h_new, target)
        c5 = compute_c5(h_new, dx)
```

```python
def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_constraints(v, target):
    n = len(v)
    lo = -v.max() - 0.5
    hi = 1.0 - v.min() + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    n = len(h)
    ov = compute_overlaps(h)
    mx = ov.max()
    w = np.exp(beta * (ov - mx))
    sw = w.sum()
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    grad = np.zeros(n)
    half = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - half
        if s >= 0:
            if s < n:
                grad[:n - s] += wi * (1.0 - h[s:])
            if s > 0:
                grad[s:] -= wi * h[:n - s]
        else:
            ss = -s
            if ss < n:
                grad[ss:] += wi * (1.0 - h[:n - ss])
                grad[:n - ss] -= wi * h[ss:]
    return grad
```
