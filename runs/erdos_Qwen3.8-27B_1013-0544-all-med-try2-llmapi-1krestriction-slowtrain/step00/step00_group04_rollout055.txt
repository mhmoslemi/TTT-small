import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def simulate(h, time_limit, rng, T0=0.03, cooling=0.999):
    n = len(h)
    current = h.astype(np.float64).copy()
    current_score = compute_c5(current)
    best = current.copy()
    best_score = current_score
    T = T0
    start = time.time()
    
    while time.time() - start < time_limit:
        # Get ones and zeros
        h_int = current.astype(int)
        ones = np.where(h_int == 1)[0]
        zeros = np.where(h_int == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        current[i] = 1.0
        current[j] = 0.0
        new_score = compute_c5(current)
        delta = new_score - current_score

            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            current[i] = 0.0
            current[j] = 1.0
        
        T *= cooling
    
    return best, best_score
