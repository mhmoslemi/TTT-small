import numpy as np
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) * 0.5
    h = np.clip(h + offset, 0.0, 1.0)
    # Correct small numerical error
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            h[interior] += diff / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def _compute_grad(h):
    """Compute (value, gradient) of max overlap at h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    max_val = float(np.max(overlaps))
    
    tol = max(1e-3, 0.01 * max_val)
    active = np.where(overlaps >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        k_lo = max(0, m - n + 1)
        k_hi = min(m, n - 1)
        xs = np.arange(k_lo, k_hi + 1)
        grad[xs] += 1.0 - h[m - xs]
        x_lo2 = m - k_hi
        x_hi2 = m - k_lo
        xs2 = np.arange(x_lo2, x_hi2 + 1)
        grad[xs2] -= h[m - xs2]
    
    grad /= len(active)
    grad *= dx
    return max_val, grad

def _hill_climb_binary(h, rng, time_limit=30.0):
    """Best-improvement hill climbing on binary h with swap moves"""
    n = len(h)
    h = h.copy()
    cur_c5 = _compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    
    start = time.time()
    stall = 0
    
    while stall < 30 and time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        
        best_swap_c5 = cur_c5
        best_a, best_b = -1, -1
        
        # For large n, sample; for small n, check all
        total_swaps = len(ones) * len(zeros)
        if total_swaps > 3000:
            indices = rng.choice(total_swaps, size=3000, replace=False)
            for idx in indices:
                a = ones[idx // len(ones)]
                b = zeros[idx % len(ones)]
                h[a], h[b] = h[b], h[a]
                c5 = _compute_c5(h)
                h[a], h[b] = h[b], h[a]
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_a, best_b = a, b
        else:
            for a in ones:
                for b in zeros:
                    h[a], h[b] = h[b], h[a]
                    c5 = _compute_c5(h)
                    h[a], h[b] = h[b], h[a]
                    if c5 < best_swap_c5 - 1e-12:
                        best_swap_c5 = c5
                        best_a, best_b = a, b
        
        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]
            cur_c5 = best_swap_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _simulated_anneal_binary(h, rng, time_limit=30.0):
    """Simulated annealing on binary h"""
    n = len(h)
    h = h.copy()
    cur_c5 = _compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    
    start = time.time()
    temp = 0.05
    cool = 0.9995
    
    while time.time() - start < time_limit:
        a = int(rng.choice(np.where(h > 0.5)[0]))
        b = int(rng.choice(np.where(h <= 0.5)[0]))
        
        h[a], h[b] = h[b], h[a]
        new_c5 = _compute_c5(h)
        
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a], h[b] = h[b], h[a]
        
        temp *= cool
    
    return best_h, best_c5

def _gradient_descent(h, time_limit=30.0):

            if patience > 100:
                lr *= 0.5
                patience = 0
                if lr < 1e-6:

    return best_h, best_c5

import numpy as np
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            h[interior] += diff / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def _compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    max_val = float(np.max(overlaps))
    tol = max(1e-3, 0.05 * max_val)
    active = np.where(overlaps >= max_val - tol)[0]
    grad = np.zeros(n)
    for m in active:
        k_lo = max(0, m - n + 1)
        k_hi = min(m, n - 1)
        xs = np.arange(k_lo, k_hi + 1)
        grad[xs] += 1.0 - h[m - xs]
        x_lo2 = m - k_hi
        x_hi2 = m - k_lo
        xs2 = np.arange(x_lo2, x_hi2 + 1)
        grad[xs2] -= h[m - xs2]
    grad /= len(active)
    grad *= dx
    return max_val, grad

def _gradient_descent(h, time_limit=30.0, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = _project(h.copy(), target_sum)
    best_c5 = _compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.3
    patience = 0
    while time.time() - start < time_limit:
        val, grad = _compute_grad(h)
        h_new = _project(h - lr * grad, target_sum)
        new_c5 = _compute_c5(h_new)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 50:
                lr *= 0.5
                patience = 0
                if lr < 1e-7:
                    break
        h = h_new
    return best_h, best_c5

def _hill_climb_binary(h, rng, time_limit=30.0):
    n = len(h)
    h = h.copy()
    cur_c5 = _compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    start = time.time()
    stall = 0
    while stall < 40 and time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        best_swap_c5 = cur_c5
        best_a, best_b = -1, -1
        total = len(ones) * len(zeros)
        if total > 4000:
            sample_size = 4000
            a_s = rng.choice(len(ones), size=sample_size, replace=False)
            b_s = rng.choice(len(zeros), size=sample_size, replace=False)
            for i in range(sample_size):
                a = int(ones[a_s[i]])
                b = int(zeros[b_s[i]])
                h[a], h[b] = h[b], h[a]
                c5 = _compute_c5(h)
                h[a], h[b] = h[b], h[a]
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_a, best_b = a, b
        else:
            for a in ones:
                for b in zeros:
                    h[a], h[b] = h[b], h[a]
                    c5 = _compute_c5(h)
                    h[a], h[b] = h[b], h[a]
                    if c5 < best_swap_c5 - 1e-12:
                        best_swap_c5 = c5
                        best_a, best_b = int(a), int(b)
        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]

                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sim_anneal_binary(h, rng, time_limit=15.0):
    n = len(h)
    h = h.copy()
    cur_c5 = _compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    start = time.time()
    temp = 0.03
    cool = 0.9997
    while time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        h[a], h[b] = h[b], h[a]
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a], h[b] = h[b], h[a]
        temp *= cool
    return best_h, best_c5

def _random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _make_binary_from_frac(h, n):
    if len(h) == n:
        order = np.argsort(h)[::-1]
        hb = np.zeros(n)
        hb[order[:n // 2]] = 1.0
        return hb
    return None
