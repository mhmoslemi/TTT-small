                best_h, best_c5 = do_active_gradient(best_h, n, tgt, dx, t0, sub_deadline, max_iters=4000)

    # Final: one more polishing pass
    if time.time() < deadline:
        best_h, best_c5 = do_active_gradient(best_h, n, tgt, dx, t0, deadline, max_iters=10000)

    return best_h, compute_c5(best_h), n
