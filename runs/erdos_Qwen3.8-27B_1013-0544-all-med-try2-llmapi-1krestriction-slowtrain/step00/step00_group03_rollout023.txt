    c = np.correlate(h, 1.0 - h, mode="full")
    best = np.max(c)
    t0 = time.time()
    
    while time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(500):
            i = zeros[rng.integers(len(zeros))]
            j = ones[rng.integers(len(ones))]
            d = compute_delta(h, i, j, n)
            nm = np.max(c + d)
            if nm < best - 1e-12:
                h[i], h[j] = 1.0, 0.0
                c += d
                best = nm
                improved = True
                break
        if not improved:
            break
    return h, best * dx

def binary_sa(h, rng, time_limit=10.0):
    """Simulated annealing on binary sequences"""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best_h = h.copy()
    best = np.max(c)
    t0 = time.time()
    T = 1.0
    
    while time.time() - t0 < time_limit:
        T = max(0.001, T * 0.9995)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        d = compute_delta(h, i, j, n)
        nm = np.max(c + d)
        cur = np.max(c)
        if nm < cur or rng.random() < np.exp(-(nm - cur) / T):
            h[i], h[j] = 1.0, 0.0
            c += d
            if np.max(c) < best:
                best = np.max(c)
                best_h = h.copy()
    return best_h, best * dx

def continuous_optimize(h, rng, time_limit=5.0):
    """Projected gradient descent with smoothing"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = project_onto_box_sum(h.copy(), target)
    best_x = x.copy()
    best_f = eval_c5(x)
    t0 = time.time()
    mu = 5.0
    step = 0.02
    stall = 0
    
    while time.time() - t0 < time_limit:
        # Gradient
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(c)
        e = np.exp(mu * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            off = k - (n - 1)
            wq = w[k] * dx
            lo1, hi1 = max(0, -off), min(n, n - off)
            if lo1 < hi1:
                g[lo1:hi1] += wq * (1.0 - x[lo1 + off:hi1 + off])
            lo2, hi2 = max(0, off), min(n, n + off)
            if lo2 < hi2:
                g[lo2:hi2] -= wq * x[lo2 - off:hi2 - off]
        
        # Line search
        improved = False
        s = step
        for _ in range(30):
            x_try = project_onto_box_sum(x - s * g, target)
            f_try = eval_c5(x_try)
            if f_try < best_f:
                best_f = f_try
                best_x = x_try.copy()
                x = x_try
                improved = True
                stall = 0
                break
            s *= 0.5
        
        if improved:
            step = min(step * 1.05, 0.1)
        else:
            stall += 1
            if stall > 5:
                mu = min(mu * 2.0, 100.0)
                step = 0.05
                stall = 0
                if mu >= 100.0:
                    break
    
    return best_x, best_f

def make_binary_initial(n, rng, pattern="random"):
    """Create initial binary sequence"""
    h = np.zeros(n)
    if pattern == "random":
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    elif pattern == "block":
        h[:n//2] = 1.0
    elif pattern == "alternating":
        h[::2] = 1.0
        # Fix count
        n_ones = np.sum(h == 1.0)
        if n_ones > n // 2:
            excess = n_ones - n // 2
            ones_idx = np.where(h == 1.0)[0]
            h[ones_idx[-excess:]] = 0.0
        elif n_ones < n // 2:
            deficit = n // 2 - n_ones
            zeros_idx = np.where(h == 0.0)[0]
            h[zeros_idx[:deficit]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):

Wait, I have a concern. The `compute_delta` function has a loop over 2n-1 values, and for each, it does 4 condition checks. For n=70, that's 139 iterations with 4 if-checks each = 556 conditionals per swap. In a loop of 500 tries per pass, that's 278,000 conditionals per pass. In Python, this might be slow.
