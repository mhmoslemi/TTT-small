def topk_gradient(h, K=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    topk = np.argsort(corr)[-K:]
    lags = topk - (n - 1)
    
    grad = np.zeros(n)
    for k in lags:
        # For each m, the contribution is:
        # (1-h[m+k]) if 0<=m+k<n, minus h[m-k] if 0<=m-k<n
        if k >= 0:
            # m ranges from 0 to n-1-k for the first term
            # m ranges from k to n-1 for the second term
            grad[:n-k] += (1.0 - h[k:n])
            grad[k:] -= h[:n-k]
        else:
            # k < 0, let k' = -k
            kp = -k
            # first term: m+k >= 0 means m >= kp, m < n
            # (1-h[m+k]) = (1-h[m-kp]) for m in [kp, n-1]
            grad[kp:] += (1.0 - h[:n-kp])
            # second term: m-k = m+kp >= 0 (always for m>=0), m+kp < n means m < n-kp
            # h[m-k] = h[m+kp] for m in [0, n-1-kp]
            grad[:n-kp] -= h[kp:]
    
    return grad * dx / K

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def topk_gradient(h, K=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    topk_idx = np.argsort(corr)[-K:]
    lags = topk_idx - (n - 1)
    grad = np.zeros(n)
    for k in lags:
        if k >= 0:
            if k < n:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] -= h[:n-k]
        else:
            kp = -k
            if kp < n:
                grad[kp:] += (1.0 - h[:n-kp])
                grad[:n-kp] -= h[kp:]
    return grad * dx / K

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def adam_step(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, target_sum=40.0):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = h - step
    h_new = project_h(h_new, target_sum)
    return h_new, m, v, t + 1

def optimize_adam(h_init, n, target_sum, max_steps, t0, deadline, lr=0.01, K=10):
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_lr = lr
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        grad = topk_gradient(h, K)
        h, m, v, t = adam_step(h, grad, m, v, t, lr=best_lr, target_sum=target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
        else:
            best_lr *= 0.9
    
    return best_h, best_c5
