import numpy as np
import time
import random
import math

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def greedy_swap(h, dx, time_limit=5.0):
    t0 = time.time()
    h = h.copy().astype(float)
    best_c5 = compute_c5(h, dx)
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_move = None
        best_val = best_c5
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                c5v = compute_c5(h, dx)
                if c5v < best_val:
                    best_val = c5v
                    best_move = (i, j)
                h[i] = 1; h[j] = 0
        if best_move is None:
            break
        h[best_move[0]] = 0
        h[best_move[1]] = 1
        best_c5 = best_val
    return h, best_c5

def sa_search(h, dx, time_limit=20.0, T0=0.2):

            j = int(zeros[random.randrange(len(zeros))])
