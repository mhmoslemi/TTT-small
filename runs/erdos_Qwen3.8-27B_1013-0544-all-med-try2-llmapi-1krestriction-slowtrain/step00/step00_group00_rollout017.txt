import numpy as np
from scipy.optimize import minimize
import time


def _obj_grad(x, n, dx, penalty, alpha):
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode="full")

    if alpha > 0:
        scaled = corr / alpha
        max_s = np.max(scaled)
        exp_s = np.exp(scaled - max_s)
        sum_exp = np.sum(exp_s)
        c5 = alpha * (max_s + np.log(sum_exp))
        weights = exp_s / sum_exp
    else:
        c5 = np.max(corr)
        k_max = np.argmax(corr)
        weights = np.zeros(len(corr))
        weights[k_max] = 1.0

    int_viol = np.sum(x) * dx - 1.0
    obj = c5 * dx + penalty * int_viol * int_viol

    grad = np.zeros(n)
    n_corr = len(corr)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k_shift = k_idx - (n - 1)
        j_min = max(0, -k_shift)
        j_max = min(n, n - k_shift)
        if j_max <= j_min:
            continue
        js = np.arange(j_min, j_max)
        m1 = np.arange(j_min + k_shift, j_max + k_shift)
        grad[m1] += w * (1.0 - x[m1 - k_shift])
        grad[js] += w * (-x[js + k_shift])

    grad += 2.0 * penalty * int_viol * dx
    grad *= dx
    return obj, grad


def _run_lbfgs(h0, n, penalty, alpha, max_iter):
    dx = 2.0 / n
    h0 = np.clip(h0, 0.0, 1.0)
    try:
        res = minimize(
            _obj_grad, h0.copy(),
            args=(n, dx, penalty, alpha),
            method='L-BFGS-B', jac=True,
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter, 'maxfun': max_iter * 10,
                     'ftol': 1e-14, 'gtol': 1e-10,
                     'maxls': 50, 'maxcor': 30}
        )
    except Exception:
        return h0, _c5_of(h0)
    if res.x is not None:
        h_opt = np.clip(res.x, 0.0, 1.0)
        return h_opt, _c5_of(h_opt)
    return h0, _c5_of(h0)


def _c5_of(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h
