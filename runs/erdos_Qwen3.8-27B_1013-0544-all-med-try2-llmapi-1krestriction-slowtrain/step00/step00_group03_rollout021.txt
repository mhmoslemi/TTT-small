import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single_k(h, k, n):
    L = k - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        m = np.arange(n - L)
        grad[m + L] += (1.0 - h[m])
        grad[m] -= h[m + L]
    else:
        p = -L
        m = np.arange(n - p)
        grad[m] += (1.0 - h[m + p])
        grad[m + p] -= h[m]
    return grad

def smooth_obj_and_grad(h, tau=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Numerical stability
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    weights /= wsum
    obj = max_c + tau * np.log(wsum)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-15:
            grad += weights[k] * grad_single_k(h, k, n) * dx

import numpy as np
from scipy.optimize import minimize
import time

_n_g = 70
_tau_g = 0.005

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                h[:] = 0.5
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                h[:] = 0.0
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single_k(h, k, n):
    L = k - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        m = np.arange(n - L)
        grad[m + L] += (1.0 - h[m])
        grad[m] -= h[m + L]
    else:
        p = -L
        m = np.arange(n - p)
        grad[m] += (1.0 - h[m + p])
        grad[m + p] -= h[m]
    return grad

def smax_val(h):
    n = _n_g
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    sh = (corr - mc) / _tau_g
    return mc + _tau_g * np.log(np.sum(np.exp(sh)))

def smax_grad(h):
    n = _n_g
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    sh = (corr - mc) / _tau_g
    w = np.exp(sh)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_single_k(h, k, n) * dx
    return grad

def _obj(h):
    return smax_val(h)

def _grad(h):
    return smax_grad(h)

def _con_fun(h):
    return np.sum(h) - _n_g / 2.0

def _con_jac(h):
    return np.ones(_n_g)

def subgrad_search(h, n, n_iter, lr0):
    h = h.copy()
    for it in range(n_iter):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            grad += grad_single_k(h, k, n)
        grad /= len(active)
        grad *= dx
        lr = lr0 / (1.0 + 0.01 * it)
        h_new = h - lr * grad
        h_new = repair_h(h_new, n)
        h = h_new
    return h

    # Phase 1: Coarse-to-fine local optimization
    tau_sequence = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

        elapsed = time.time() - t0
        if elapsed > budget_s * 0.35:
            break
        try:
            res = minimize(_obj, best_h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = repair_h(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 1b: Subgradient refinement
    elapsed = time.time() - t0
