        objective, x0, method='SLSQP', jac=objective_grad,
        constraints=constraints, bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    
    if result.success:
        h_opt = result.x[:n]
        c5 = evaluate_c5(h_opt)
        return h_opt, c5
    return h0, evaluate_c5(h0)

import numpy as np
import time
import scipy.optimize

def evaluate_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def coord_update(h, i):
    n = len(h)
    old_val = h[i]
    h[i] = 0.0
    corr_0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr_1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old_val
    a = corr_1 - corr_0
    B = corr_0
    g0 = B.max()
    g1 = (B + a).max()
    best_val = min(g0, g1)
    best_x = 0.0 if g0 <= g1 else 1.0
    m = len(a)
    if m > 2:
        da = a[:, None] - a[None, :]
        dB = B[None, :] - B[:, None]
        valid = np.abs(da) > 1e-14
        if valid.any():
            safe_da = np.where(valid, da, 1.0)
            x_int = dB / safe_da
            mask = valid & (x_int > 1e-10) & (x_int < 1.0 - 1e-10)
            if mask.any():
                xc = x_int[mask]
                vals = B[None, :] + a[None, :] * xc[:, None]
                g_vals = vals.max(axis=1)
                idx = np.argmin(g_vals)
                if g_vals[idx] < best_val - 1e-14:
                    best_val = g_vals[idx]
                    best_x = xc[idx]
    return best_x

def project_sum(h, target):
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def coord_descent(h, max_sweeps=300, time_limit=30.0):
    t0 = time.time()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        for i in range(n):
            h[i] = coord_update(h, i)
        h = project_sum(h, target)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_polish(h0):
    n = len(h0)
    dx = 2.0 / n
    m = 2 * n - 1
    offset = n - 1

    def objective(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = []
    for k in range(m):
        d = k - offset
        def make_c(k, d):
            def fun(x):
                h = x[:n]
                c = float(np.correlate(h, 1.0 - h, mode='full')[k]) * dx
                return x[n] - c
            def jac(x):
                h = x[:n]
                g = np.zeros(n + 1)
                g[n] = 1.0
                for i in range(n):
                    av = 0.0
                    if 0 <= i - d < n:
                        av += (1.0 - h[i - d])
                    if 0 <= i + d < n:
                        av -= h[i + d]
                    g[i] = -av * dx
                return g
            return {'type': 'ineq', 'fun': fun, 'jac': jac}
        cons.append(make_c(k, d))

    def sum_fun(x):
        return np.sum(x[:n]) - n / 2.0
    def sum_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': sum_fun, 'jac': sum_jac})

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.append(h0.copy(), evaluate_c5(h0) + 0.05)
    try:
        res = scipy.optimize.minimize(objective, x0, method='SLSQP', jac=obj_grad,
                                       constraints=cons, bounds=bounds,
                                       options={'maxiter': 300, 'ftol': 1e-14})
        if res.x[:n].min() >= -1e-9:
            h_opt = np.clip(res.x[:n], 0, 1)
            h_opt = project_sum(h_opt, n / 2.0)
            return h_opt, evaluate_c5(h_opt)
    except Exception:
        pass
    return h0, evaluate_c5(h0)

def binarize_and_optimize(h, n, time_limit=10.0):
    h_bin = (h >= 0.5).astype(float)
    diff = n // 2 - int(h_bin.sum())
    if diff != 0:
        if diff > 0:
            below = np.where(h_bin == 0)[0]
            below = below[np.argsort(-h[below])][:diff]
            h_bin[below] = 1.0
        else:
            above = np.where(h_bin == 1)[0]
            above = above[np.argsort(h[above])[:diff]]
            h_bin[above] = 0.0
    c5 = evaluate_c5(h_bin)
    h_opt, c5_opt = coord_descent(h_bin.copy(), max_sweeps=200, time_limit=time_limit)
    if c5_opt < c5:
        return h_opt, c5_opt
    return h_bin, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.asarray(initial_h_values, dtype=float)
        n_parent = len(h_parent)

        h_parent = np.full(45, 0.5)
        n_parent = 45

    best_h = h_parent.copy()
    best_c5 = evaluate_c5(h_parent)
    best_n = n_parent

    n_list = sorted(set([n_parent, 45, 60, 90, 120, 180, 250, 350, 500]))

    for ni, n in enumerate(n_list):
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.85:
            break
        remaining = budget_s * 0.85 - elapsed
        n_remaining = len(n_list) - ni
        time_per_n = remaining / max(1, n_remaining)

        if n == n_parent:
            h = h_parent.copy()
        else:
            x_old = np.linspace(0, 2, n_parent, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, h_parent)
            h = np.clip(h, 0, 1)
        h = project_sum(h, n / 2.0)

        # Phase 1: coordinate descent
        t_cd = min(time_per_n * 0.4, 50.0)
        h_cd, c5_cd = coord_descent(h, max_sweeps=400, time_limit=t_cd)
        if c5_cd < best_c5:

            h_s, c5_s = slsqp_polish(h_cd)

            # Random binary initialization

            h_noise = h_cd + rng.normal(0, 0.15, n)

            h_noise = project_sum(h_noise, n / 2.0)

            if c5_r2 < best_c5:

    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
