import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def steepest_descent(h, max_rounds=500, time_limit=10.0):
    n = len(h)
    best_c5 = evaluate_c5(h)
    start = time.time()
    for _ in range(max_rounds):
        if time.time() - start > time_limit:
            break
        ones = np.where(h > 0.5)[0].tolist()
        zeros = np.where(h < 0.5)[0].tolist()
        best_val = best_c5
        bi = bj = -1
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                v = evaluate_c5(h)
                h[i], h[j] = h[j], h[i]
                if v < best_val - 1e-12:
                    best_val = v
                    bi, bj = i, j
        if bi < 0:
            break
        h[bi], h[bj] = h[bj], h[bi]
        best_c5 = best_val
    return h, best_c5

def perturb_random(h, rng, nsw):

def interpolate_to_size(h_old, n_new, rng):
    n_old = len(h_old)
    h_new = np.zeros(n_new)
