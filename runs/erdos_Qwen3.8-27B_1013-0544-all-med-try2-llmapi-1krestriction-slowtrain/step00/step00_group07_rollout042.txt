def grad_c_at_k(h, k, n):
    """Compute d(c[k])/d(h) where c[k] = sum_i h[i+k](1-h[i]) for valid i."""
    dx = 2.0 / n
    grad = np.zeros(n)
    
    if k >= 0:
        # i from 0 to n-1-k
        # h[j] as h[i+k]: j = i+k, i = j-k, valid if 0 ≤ j-k ≤ n-1-k → k ≤ j ≤ n-1
        #   contribution: (1-h[i]) = (1-h[j-k])
        for j in range(k, n):
            i = j - k
            if 0 <= i <= n-1-k:
                grad[j] += (1 - h[i])
        # h[j] as h[i]: j = i, valid if 0 ≤ j ≤ n-1-k
        #   contribution: -h[i+k] = -h[j+k]
        for j in range(0, n-k):
            if 0 <= j+k < n:
                grad[j] -= h[j+k]

        k_neg = -k
        # i from k_neg to n-1

    # The term m+k=j (m=j-k=i): same term!
    # So c[k=j-i] includes h[j] - h[i]*h[j] = h[j](1-h[i])

    # The max of these functions is convex in h[i] (max of convex functions is convex).
