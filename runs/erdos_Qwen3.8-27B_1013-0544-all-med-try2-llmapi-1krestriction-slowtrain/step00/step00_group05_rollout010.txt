def grad_corr(h, n):
    """Compute gradient of all correlation values w.r.t. h.
    Returns array of shape (2*n-1, n) where grad_corr[k, j] = dc_k/dh_j"""
    m = n - 1
    g = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - m  # shift
        for j in range(n):
            if 0 <= j+s < n:
                g[k, j] += (1 - h[j+s])
            if 0 <= j-s < n:
                g[k, j] -= h[j-s]
    return g

        if 0 <= j-s < n:
            val -= h[j-s]

    # For pairs (i, l): t = (b[l] - b[i]) / (a[i] - a[l])

    B = b[None, :] - b[:, None]

    valid = np.abs(A) > 1e-12
    t_vals = np.where(valid, B / np.where(valid, A, 1), -1)
    t_vals = t_vals[(t_vals >= 0) & (t_vals <= 1)]
    t_vals = np.unique(t_vals)
    
    # Add endpoints
    t_vals = np.concatenate([[0.0, 1.0], t_vals])
    
    # Evaluate f(t) = max_k(a_k * t + b_k) for all t
    # f_vals[t_idx, k] = a[k] * t_vals[t_idx] + b[k]
    vals = a[None, :] * t_vals[:, None] + b[None, :]
    f_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(f_vals)

np.correlate(h, 1-h, mode="full") for h of length n gives an array of length 2n-1.

The k-th element (0-indexed) is: sum_i h[i] * (1-h)[i + k - (n-1)]

where the sum is over all valid i (both indices in [0, n-1]).

So for k=0: sum over i where i-(n-1) >= 0, i.e., i >= n-1. Only i=n-1: h[n-1]*(1-h[0]).
For k=n-1: sum over i where both i and i >= 0 and i <= n-1: sum_i h[i]*(1-h[i]). This is the "zero lag" correlation.
For k=2(n-1): sum over i where i+n-1 <= n-1, i.e., i <= 0. Only i=0: h[0]*(1-h[n-1]).

So the maximum will typically be at k near n-1 (zero lag) or at small/large k.

The zero-lag value is sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2. Since sum h[i] = n/2, this is n/2 - sum h[i]^2. To minimize this, we want sum h[i]^2 to be as large as possible, which means h should be as "binary" as possible (all 0s and 1s). If h is exactly 0/1 with n/2 ones, then sum h[i]^2 = n/2, and the zero-lag value is 0.

But other lags will be non-zero. The problem is a trade-off.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_grad(h, n, beta):
    dx = 2.0 / n
    m = n - 1
    corr = np.correlate(h, 1 - h, 'full')
    x = beta * corr
    x -= x.max()
    w = np.exp(x)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-9:
            continue
        s = k - m
        wk = w[k] * beta * dx
        for j in range(n):
            if 0 <= j + s < n:
                g[j] += wk * (1.0 - h[j + s])
            if 0 <= j - s < n:
                g[j] -= wk * h[j - s]
    return g

def coord_opt(h, j, n):
    m = n - 1
    corr = np.correlate(h, 1 - h, 'full')
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - m
        if 0 <= j + s < n:
            a[k] += (1.0 - h[j + s])
        if 0 <= j - s < n:
            a[k] -= h[j - s]
    b = corr - a * h[j]
    A = a[:, None] - a[None, :]
    B = b[None, :] - b[:, None]
    valid = np.abs(A) > 1e-12
    tri = np.triu(np.ones((2 * n - 1, 2 * n - 1), dtype=bool), k=1)
    valid &= tri
    t_vals = np.zeros((2 * n - 1, 2 * n - 1))
    t_vals[valid] = B[valid] / A[valid]
    t_vals = t_vals[valid]
    t_vals = t_vals[(t_vals >= 0.0) & (t_vals <= 1.0)]
    cands = np.concatenate([[0.0, 1.0], t_vals])
    vals = np.max(a[None, :] * cands[:, None] + b[None, :], axis=1)
    return cands[np.argmin(vals)]

def projected_gradient_phase(h, n, beta_start, lr_start, max_iters, time_limit):
    start = time.time()
    beta = beta_start
    lr = lr_start
    best = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        g = lse_grad(h, n, beta)
        h_new = project_h(h - lr * g, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
        else:
            lr *= 0.92
        beta *= 1.015
        if lr < 1e-9:
            break
    return best, best_c5

def coord_descent_phase(h, n, max_sweeps, time_limit):
    start = time.time()
    best = h.copy()
    best_c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            h[j] = coord_opt(h, j, n)
        h = project_h(h, n)
        c5_new = compute_c5(h, n)
        if c5_new <= best_c5:
            best_c5 = c5_new
            best = h.copy()
    return best, best_c5
