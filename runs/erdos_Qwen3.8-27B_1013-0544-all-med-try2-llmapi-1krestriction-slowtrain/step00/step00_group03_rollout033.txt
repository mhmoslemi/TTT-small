    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)

    for k in max_ks:
        p = k - (n - 1)  # lag
        g = np.zeros(n)
        for i in range(n):

            if 0 <= i + p < n:
                g[i] += (1 - h[i + p])
            # Contribution from h[i] in (1 - h[j+p]) where j+p = i
            if 0 <= i - p < n:
                g[i] -= h[i - p]
        grad += g
    grad *= dx / len(max_ks)
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h, dx):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h_proj = np.clip(h - alpha, 0.0, 1.0)
    # Fix numerical drift
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def subgradient_c5(h, dx):
    """Subgradient of C5 at h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # All indices achieving max (within tolerance)
    tol = 1e-10

    for k in max_ks:

    grad *= dx / len(max_ks)
    return grad

def smooth_max_objective(h, dx, temperature=100.0):
    """Smooth approximation of max using log-sum-exp"""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Shift for numerical stability
    c = np.max(corr)
    return c + np.log(np.sum(np.exp(corr - c)) / len(corr)) / temperature

    """SLSQP optimization with smooth objective"""

        return smooth_max_objective(x, dx, temperature)

        return np.sum(x) - target_sum

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h_proj = np.clip(h - alpha, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def subgradient_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    max_ks = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for k in max_ks:
        p = int(k) - (n - 1)
        for i in range(n):
            if 0 <= i + p < n:
                grad[i] += (1.0 - h[i + p])
            if 0 <= i - p < n:
                grad[i] -= h[i - p]
    grad *= dx / len(max_ks)
    return grad

def projected_subgradient(h, dx, target_sum, max_iter=3000, c=0.01):
    n = len(h)
    h_curr = project_feasible(h.copy(), target_sum)
    best_h = h_curr.copy()
    best_val = compute_c5(best_h, dx)
    
    for it in range(1, max_iter + 1):
        g = subgradient_c5(h_curr, dx)
        lr = c / np.sqrt(it)
        h_new = h_curr - lr * g
        h_new = project_feasible(h_new, target_sum)
        val = compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h_curr = h_new

            h_kick = project_feasible(h_curr + kick, target_sum)
            if compute_c5(h_kick, dx) < val:
                h_curr = h_kick
    
    return best_h, best_val

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.max(corr)
    return c + np.log(np.sum(np.exp(corr - c)) / len(corr)) / temp

        return smoothed_obj(x, dx, temp)

        return np.sum(x) - target_sum

        return h_out

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h_p.sum()
    if abs(diff) > 1e-12:
        m = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if m.sum() > 0:
            h_p[m] += diff / m.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def subgradient_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = corr.max()
    mks = np.where(corr >= mv - 1e-12)[0]
    grad = np.zeros(n)
    for k in mks:
        p = int(k) - n + 1
        g = np.zeros(n)
        if p == 0:
            g[:] = 1.0 - 2.0 * h
        elif p > 0:
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
        else:
            q = -p
            g[q:] += 1.0 - h[:n - q]
            g[:n - q] -= h[q:]
        grad += g
    return grad * dx / len(mks)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h_p.sum()
    if abs(diff) > 1e-12:
        m = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if m.sum() > 0:
            h_p[m] += diff / m.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def subgradient_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = corr.max()
    mks = np.where(corr >= mv - 1e-12)[0]
    grad = np.zeros(n)
    for k in mks:
        p = int(k) - n + 1
        g = np.zeros(n)
        if p == 0:
            g[:] = 1.0 - 2.0 * h
        elif p > 0:
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
        else:
            q = -p
            g[q:] += 1.0 - h[:n - q]
            g[:n - q] -= h[q:]
        grad += g
    return grad * dx / len(mks)

def subgrad_optimize(h, dx, tsum, maxit=3000, c=0.01, momentum=0.0):
