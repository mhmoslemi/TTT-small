            h_cont, c5_cont = continuous_optimize(h0, max_iter=30000)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
                best_n = n0
            # Multi-start: perturb and re-optimize
            for trial in range(5):
                if remaining() < 60:
                    break
                h_pert = h0 + 0.1 * np.random.randn(n0)
                h_pert = make_feasible(h_pert, n0)
                h_c2, c5_c2 = continuous_optimize(h_pert, max_iter=10000)
                if c5_c2 < best_c5:
                    best_c5 = c5_c2
                    best_h = h_c2.copy()
                    best_n = n0
    except Exception:
        pass

    # Phase 2: Binary local search for even n
    n_list = [46, 48, 50, 52, 54, 56, 58, 60, 64, 68, 72, 80]
    for n in n_list:
        if remaining() < 90:
            break
        if n <= 54:
            n_restarts = 12
            max_rounds = 80
        elif n <= 68:
            n_restarts = 8
            max_rounds = 50
        else:
            n_restarts = 5
            max_rounds = 30
        for r in range(n_restarts):
            if remaining() < 90:
                break
            h = np.zeros(n)
            h[:n // 2] = 1.0
            np.random.shuffle(h)
            h_opt, c5_opt = binary_local_search(h, max_rounds=max_rounds)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            # Kick: perturb and re-optimize
            if remaining() > 60:
                h_kick = h_opt.copy()
                for _ in range(4):
                    ones_k = np.where(h_kick == 1)[0]
                    zeros_k = np.where(h_kick == 0)[0]
                    if len(ones_k) > 0 and len(zeros_k) > 0:
                        i = int(np.random.choice(ones_k))
                        j = int(np.random.choice(zeros_k))
                        h_kick[i], h_kick[j] = 1.0, 0.0
                h_kick, c5_kick = binary_local_search(h_kick, max_rounds=max_rounds)
                if c5_kick < best_c5:
                    best_c5 = c5_kick
                    best_h = h_kick.copy()
                    best_n = n

    # Phase 3: Final refinement on best solution
    if best_h is not None and remaining() > 30:
        n = len(best_h)
        if n % 2 == 0:
            h_ref, c5_ref = binary_local_search(best_h, max_rounds=300)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
                best_n = n
        if remaining() > 20:
            h_cont, c5_cont = continuous_optimize(best_h, max_iter=15000)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont
                best_n = n

    # Fallback
    if best_h is None:
        n = 50
        best_h = np.zeros(n)
        best_h[:n // 2] = 1
        np.random.shuffle(best_h)
        best_h, best_c5 = binary_local_search(best_h, max_rounds=50)
        best_n = n

    best_h = make_feasible(best_h, len(best_h))
    return best_h, best_c5, best_n
