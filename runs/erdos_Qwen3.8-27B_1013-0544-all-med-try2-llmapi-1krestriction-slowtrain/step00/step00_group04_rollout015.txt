def compute_c5(h, dx):
    h_f = h.astype(float)
    corr = np.correlate(h_f, 1.0 - h_f, mode="full")
    return float(np.max(corr) * dx)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    h = np.zeros(n, dtype=float)
    for i in range(1, n):
        h[i] = h[i >> 1] ^ (i & 1)
    return h

        h[i] = 0

        h[i] = 1

def compute_c5(h, dx):
    h_f = h.astype(float)
    corr = np.correlate(h_f, 1.0 - h_f, mode="full")
    return float(np.max(corr) * dx)

def thue_morse(n):
    h = np.zeros(n, dtype=float)
    # t(0) = 0
    for i in range(1, n):
        # Count trailing zeros in binary representation of i
        # t(i) = popcount(i) mod 2
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        h[i] = c % 2
    return h

import time

def compute_c5(h, dx):
    h_f = h.astype(np.float64)
    corr = np.correlate(h_f, 1.0 - h_f, mode="full")
    return float(np.max(corr) * dx)

def thue_morse(n):
    h = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        c = np.binary_repr(i).count('1')
        h[i] = c % 2
    return h

def fix_sum(h, n, rng):
    """Ensure sum(h) = n/2 for binary h."""
    h = h.copy().astype(np.float64)
    target = n // 2
    s = int(np.round(h.sum()))
    if s > target:
        ones = np.where(h >= 0.5)[0]
        for _ in range(s - target):
            h[ones[rng.integers(len(ones))]] = 0
    elif s < target:
        zeros = np.where(h < 0.5)[0]
        for _ in range(target - s):
            h[zeros[rng.integers(len(zeros))]] = 1
    h[h >= 0.5] = 1
    h[h < 0.5] = 0
    return h

        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h, dx)
        delta = c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        # Cool down
        T *= 0.999
        if T < 0.001:
            T = 0.001
            h = best_h.copy()
            cur_c5 = best_c5
    
    return best_h, best_c5

def compute_c5(h, dx):
    h_f = h.astype(np.float64)
    corr = np.correlate(h_f, 1.0 - h_f, mode="full")
    return float(np.max(corr) * dx)

def thue_morse(n):
    h = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        h[i] = c % 2
    return h

    h = h.copy().astype(np.float64)
    h[h >= 0.5] = 1.0

        ones = np.where(h == 1)[0]
        for _ in range(s - target):
            h[ones[rng.integers(len(ones))]] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        for _ in range(target - s):
            h[zeros[rng.integers(len(zeros))]] = 1
    return h
