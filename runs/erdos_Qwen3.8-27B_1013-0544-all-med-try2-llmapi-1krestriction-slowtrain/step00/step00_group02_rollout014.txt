    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr_raw)[::-1][:topk]

            grad[nl:] += (1.0 - h[:n - nl])

    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    # Use top-k for efficiency, or all if n is small
    if topk is not None and topk < len(corr_raw):
        top_idx = np.argsort(corr_raw)[::-1][:topk]
        c = corr_raw[top_idx]
    else:
        top_idx = np.arange(len(corr_raw))
        c = corr_raw
    
    # Log-sum-exp
    c_max = c.max()
    weights = np.exp((c - c_max) / eps)
    weights /= weights.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):

            grad[nl:] += w * (1.0 - h[:n-nl])
            grad[:nl] -= w * h[nl:]
    
    obj = eps * (c_max + np.log(weights.sum()))

    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]

        g[nl:] += (1.0 - h[:n - nl])

    n = len(h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    if topk is not None and topk < len(corr_raw):
        top_idx = np.argsort(corr_raw)[::-1][:topk]

    else:
        top_idx = np.arange(len(corr_raw))

    c_max = c.max()
    scaled = (c - c_max) / eps
    exp_vals = np.exp(scaled)

    weights = exp_vals / total
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w < 1e-8:
            continue
        lag = int(k) - (n - 1)
        grad += w * grad_single_lag(h, lag)
    obj = float(eps * (c_max + np.log(max(total, 1e-300))))

        v = momentum * v + grad
        h = h - lr * v
        h = project_box_sum(h, target_sum)
        if it % 15 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                lr = min(0.05, lr * 1.02)
            else:
                lr = max(0.001, lr * 0.92)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 46
        h0 = np.ones(n0) * 0.5

    n0 = len(h0)
    ts0 = n0 / 2.0
    h0 = project_box_sum(h0, ts0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: optimize at native resolution
    rem = budget_s - (time.time() - start_time) - 180
    if rem > 20:
        h1, c5_1 = optimize_smoothed(h0, ts0, 0.1, 0.01, 2000, 0.02, 0.85, rem)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: upsample and optimize at higher resolutions
    for n_new in [max(2 * n0, 90), 180]:
        elapsed = time.time() - start_time
        rem = budget_s - elapsed - 80
        if rem < 15:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_box_sum(h_new, n_new / 2.0)
        rem2 = budget_s - (time.time() - start_time) - 40
        if rem2 < 10:
            continue
        h_new, c5_new = optimize_smoothed(h_new, n_new / 2.0, 0.08, 0.005, 3000, 0.01, 0.85, rem2)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: perturbation restarts
    n_best = len(best_h)
    ts_best = n_best / 2.0
    for r in range(8):
        elapsed = time.time() - start_time
        rem = budget_s - elapsed - 15
        if rem < 15:
            break
        sigma = 0.03 * (1.0 - r / 10.0)
        noise = np.random.randn(n_best) * sigma
        h_pert = project_box_sum(best_h + noise, ts_best)
        h_pert, c5_pert = optimize_smoothed(h_pert, ts_best, 0.06, 0.003, 2000, 0.008, 0.85, rem - 5)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert

    # Phase 4: final fine-tune with very small eps
    elapsed = time.time() - start_time
    rem = budget_s - elapsed - 5
    if rem > 10:
        h_fin, c5_fin = optimize_smoothed(best_h, ts_best, 0.02, 0.001, 3000, 0.003, 0.8, rem)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final validation
    best_h = project_box_sum(best_h, n_best / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
