```python
    lo, hi = -1.0, 2.0
```

```python
    grad = np.zeros(n)
    # First term: (1-h[j+shift]) for j where 0 <= j+shift < n
    if shift >= 0:
        for j in range(n - shift):
            grad[j] = 1.0 - h[j + shift]
    else:
        for j in range(-shift, n):
            grad[j] = 1.0 - h[j + shift]
    # Second term: -h[j-shift] for j where 0 <= j-shift < n
    if shift >= 0:
        for j in range(shift, n):
            grad[j] -= h[j - shift]
    else:
        for j in range(n + shift):
            grad[j] -= h[j - shift]
    return grad
```

```python
def smoothed_gradient(h, n_points, beta=20.0, top_k=5):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    
    # Find top k values
    top_indices = np.argsort(vals)[-top_k:]
    top_vals = vals[top_indices]
    
    # Softmax weights
    max_val = np.max(top_vals)
    weights = np.exp(beta * (top_vals - max_val))
    weights /= np.sum(weights)
    
    # Compute weighted gradient
    grad = np.zeros(n_points)
    for w, k in zip(weights, top_indices):
        shift = k - (n_points - 1)
        grad += w * compute_grad(h, shift)
    
    return grad, float(np.max(vals))
```

```python
    n = len(h)
    grad = np.zeros(n)
    if shift == 0:
        return grad
    if shift > 0:
```
```python
        if n > shift:
            grad[:n - shift] = 1.0 - h[shift:]
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if n > s:
            grad[s:] = 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    return grad

def optimize_continuous(h, n, rng, max_time, beta=15.0):
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    target_sum = n / 2.0
    lr = 0.1
    no_improve = 0

    while time.time() - start < max_time:
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr * dx
        current_c5 = float(np.max(vals))

        if current_c5 < best_c5 - 1e-10:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(5, len(vals))
        top_idx = np.argsort(vals)[-top_k:]
        top_vals = vals[top_idx]
        w = np.exp(beta * (top_vals - top_vals[-1]))
        w /= w.sum()

        grad = np.zeros(n)
        for weight, k in zip(w, top_idx):
            shift = int(k) - (n - 1)
            grad += weight * compute_grad(h, shift)

        h = project(h - lr * grad, target_sum)
        lr *= 0.9998

        if no_improve > 1000:
            n_pert = max(3, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            h[idx] += rng.normal(0, 0.05, size=n_pert)
            h = project(h, target_sum)
            no_improve = 0
            lr = 0.1

    return best_h, best_c5

    h_work = (h >= 0.5).astype(float)
    target = n // 2

    best_h = h_work.copy()
    ones_list = list(np.where(h_work == 1)[0])

        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue
        h_work[i] = 0
        h_work[j] = 1
        new_c5 = compute_c5(h_work)
        if new_c5 <= best_c5:
            best_c5 = new_c5
            best_h = h_work.copy()
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
        else:
            h_work[i] = 1
            h_work[j] = 0

    return best_h, best_c5
```
