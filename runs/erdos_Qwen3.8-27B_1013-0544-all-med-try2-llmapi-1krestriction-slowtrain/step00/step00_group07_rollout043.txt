import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full") * dx
    return np.max(corr)

def project_onto_feasible(x, target_sum):
    """Project x onto {0 <= h <= 1, sum(h) = target_sum}"""
    # Binary search on tau
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        clamped = np.clip(x - mid, 0, 1)
        s = np.sum(clamped)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2, 0, 1)

    n = len(h)
    corr = np.correlate(h, 1-h, mode="full") * dx
    max_val = np.max(corr)

    temperature = 0.01
    weights = np.exp((corr - max_val) / temperature)
    weights /= np.sum(weights)
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    # For each shift s, compute contribution
    for k in range(2*n-1):
        s = k - (n-1)
        w = weights[k]
        if w < 1e-10:
            continue

            # grad[j] -= h[j+s] for j in [0, n-1-s]

        else:

    return best_h, best_c5

    return np.max(np.correlate(h, 1-h, mode="full") * dx)

def project_feasible(x, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(x - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

    n = len(h)
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    w = np.exp((corr - m) / beta)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += w[k] * (1.0 - h[:n-s])
            grad[:n-s] -= w[k] * h[s:]
        else:
            p = -s
            grad[:n-p] += w[k] * (1.0 - h[p:])
            grad[p:] -= w[k] * h[:n-p]
    return grad * dx

    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) / beta)) / len(corr)) / beta
