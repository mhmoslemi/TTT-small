        if proj.sum() > target:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    """Negative subgradient of max correlation, projected to sum=0 plane."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.argmax(corr)
    offset = m - n + 1

    idx = np.arange(n)
    i1 = idx - offset
    v1 = (i1 >= 0) & (i1 < n)
    g[v1] += 1.0 - h[i1[v1]]
    i2 = idx + offset
    v2 = (i2 >= 0) & (i2 < n)
    g[v2] -= h[i2[v2]]
    g -= g.mean()  # project to tangent space
    norm = np.linalg.norm(g)
    if norm > 1e-12:
        g /= norm
    return g

                c5 = new_c5
                stagnation = 0
                last_improve_t = time.time()
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
            elif rng.rand() < np.exp(-(new_c5 - c5) / (T + 1e-15)):
                c5 = new_c5
            else:
                h[i], h[j] = oi, oj
        else:
            # Multi-pair kick
            k = min(4, n // 2)
            idx = rng.choice(n, size=k * 2, replace=False)
            for p in range(k):
                a, b = idx[2 * p], idx[2 * p + 1]
                delta = rng.randn() * T * 3.0
                oa, ob = h[a], h[b]
                h[a], h[b] = h[a] - delta, h[b] + delta
                if h[a] < 0 or h[a] > 1 or h[b] < 0 or h[b] > 1:
                    h[a], h[b] = oa, ob
            new_c5 = compute_c5(h, n)
            if new_c5 < c5 or rng.rand() < np.exp(-(new_c5 - c5) / (T + 1e-15)):
                c5 = min(new_c5, c5)
                if new_c5 < c5:
                    pass
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
                stagnation = 0
                last_improve_t = time.time()
        
        T = max(T * 0.999995, 1e-8)
        if stagnation > 8000 and time.time() - last_improve_t > 2.0:
            T = max(T, 0.008)
            stagnation = 0
        if T <= 1.1e-8 and c5 == best_c5:
            T = 0.003
            h = project_feasible(h + rng.randn(n) * 0.01, n)
            c5 = compute_c5(h, n)
            best_h, best_c5 = h.copy(), c5
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h, best_n, best_c5 = h0.copy(), n0, compute_c5(h0, n0)
    
    # Phase 1: optimize at n0 (50% budget)
    t1 = t0 + budget_s * 0.50
    bh, bc = optimize_n(h0.copy(), n0, t1, rng)
    if bc < best_c5:
        best_h, best_n, best_c5 = bh, n0, bc
    
    # Phase 2: try 2*n0 (25% budget)
    t2 = t0 + budget_s * 0.75
    n1 = n0 * 2
    xo = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n1, endpoint=False)
    h_up = project_feasible(np.interp(xn, xo, best_h), n1)
    bh, bc = optimize_n(h_up, n1, t2, rng)
    if bc < best_c5:
        best_h, best_n, best_c5 = bh, n1, bc
    
    # Phase 3: final refinement at best_n (20% budget)
    t3 = t0 + budget_s * 0.95
    bh, bc = optimize_n(best_h.copy(), best_n, t3, rng)
    if bc < best_c5:
        best_h, best_n, best_c5 = bh, best_n, bc
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
