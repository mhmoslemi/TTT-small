import numpy as np
from scipy.optimize import minimize
import time

_G = {'ts': 0.0, 'n': 0, 'beta': 500.0}


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m))) + 1e-300) / beta


def project_box_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            if d > 0:
                idx = int(np.argmin(h))
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + d)
        else:
            h[mask] += d / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h


def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    eb = np.exp(beta * (corr - mc))
    se = np.sum(eb)
    p = eb / se
    obj = dx * (mc + np.log(se) / beta)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx] * dx
        if w < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nk = n - k
            grad[:nk] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:nk]
        else:
            k2 = -k
            nk = n - k2
            grad[k2:] += w * (1.0 - h[:nk])
            grad[:nk] -= w * h[k2:]
    return obj, grad


def _pen_obj(x):
    obj, _ = obj_grad(x, _G['beta'])
    cv = x.sum() - _G['ts']
    return obj + 200.0 * cv * cv


def _pen_grad(x):
    _, g = obj_grad(x, _G['beta'])
    cv = x.sum() - _G['ts']
    return g + 400.0 * cv * np.ones(_G['n'])


def _slsqp_obj(x):
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, _G['beta']) * (2.0 / n)


def _slsqp_grad(x):
    _, g = obj_grad(x, _G['beta'])
    return g


def _cons_eq(x):
    return x.sum() - _G['ts']


def _cons_eq_jac(x):
    return np.ones(_G['n'])


def _bfgs(h, n, ts, beta, max_iter):
    _G['n'] = n
    _G['ts'] = ts
    _G['beta'] = beta
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': max_iter, 'ftol': 1e-16})
        return project_box_sum(r.x, ts)
    except Exception:
        return h


def _slsqp(h, n, ts, beta, max_iter):
    _G['n'] = n
    _G['ts'] = ts
    _G['beta'] = beta
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        r = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': 1e-15})
        return project_box_sum(r.x, ts)
    except Exception:
        return h


def _pgd(h, n, ts, beta, max_iter, step0):
    step = step0
    h = h.copy()
    for _ in range(max_iter):
        _, grad = obj_grad(h, beta)
        cv = h.sum() - ts
        grad = grad + 80.0 * cv * np.ones(n)
        h_new = project_box_sum(h - step * grad, ts)
        obj_new, _ = obj_grad(h_new, beta)
        obj_old, _ = obj_grad(h, beta)
        cn = compute_c5(h_new)
        if cn <= compute_c5(h) + 1e-14:
            h = h_new
        else:
            step *= 0.75
            if step < 1e-9:
                break
    return h


def _binary_search(h, n, ts, max_rounds):
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(max_rounds):
        improved = False
        ones = list(np.where(best > 0.5)[0])
        zeros = list(np.where(best < 0.5)[0])
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:15]:
            for j in zeros[:15]:
                trial = best.copy()
                trial[i], trial[j] = trial[j], trial[i]
                trial = project_box_sum(trial, ts)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best, best_c = trial, c
                    improved = True
        if not improved:
            break
    return best, best_c

    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = project_box_sum(h_init, ts)

    # Phase 1: SLSQP initial refinement

        if strat == 0:

            k_star = int(np.argmax(corr)) - (n - 1)

            idxs = np.random.choice(span, min(8, span), replace=False)

        else:
            nf = np.random.randint(1, 4)
            idxs = np.random.choice(n, nf, replace=False)
            for i in idxs:

        h = project_box_sum(h, ts)
        h2 = _pgd(h, n, ts, 200, 800, 0.03)

        if c < best_c5:
