        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_smooth_max_vec(h, beta, top_k=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    total = np.sum(exps)
    weights = exps / total
    
    if top_k is not None and top_k < len(weights):
        idx = np.argsort(weights)[::-1][:top_k]
        w = np.zeros_like(weights)
        w[idx] = weights[idx]
        w /= np.sum(w)
        weights = w
    
    grad = np.zeros(n)
    for m_idx in range(len(weights)):
        w = weights[m_idx]
        if w < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:n] += w * (1.0 - h[0:n-k])
                grad[0:n-k] += w * (-h[k:n])
        else:
            p = -k
            if n - p > 0:
                grad[0:n-p] += w * (1.0 - h[p:n])
                grad[p:n] += w * (-h[0:n-p])
    return grad

def pgd_step(h, target_sum, beta, top_k=50, max_iter=500, lr_init=0.1):
    """Projected gradient descent with backtracking line search."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    h = h.copy()
    
    for it in range(max_iter):
        g = grad_smooth_max_vec(h, beta, top_k)
        # Project gradient onto feasible direction (zero mean component)
        g = g - np.mean(g)
        
        lr = lr_init
        for attempt in range(30):
            h_new = project_h(h - lr * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new <= best_val + 1e-14:
                h = h_new
                if c5_new < best_val:
                    best_val = c5_new
                    best_h = h_new.copy()
                break
            lr *= 0.5
        else:
            break  # No improvement possible
    
    return best_h, best_val

            corr = corr_arr(best_h)
            active = np.argmax(corr)
            k = active - (n_points - 1)
            h0 = best_h.copy()
            if k > 0:
                h0[0:n_points-k] -= 0.04
                h0[n_points-k:] += 0.04
            elif k < 0:
                h0[-k:] -= 0.04
                h0[:-k] += 0.04
            h0 += rng.randn(n_points) * 0.008
        elif s == 5:
            h0 = best_h.copy()
            for _ in range(rng.randint(2, 5)):
                bs = rng.randint(0, n_points - 3)
                bl = rng.randint(2, 5)
                h0[bs:bs+bl] += rng.randn() * 0.04
        elif s == 6:
            h0 = best_h.copy()
            for _ in range(rng.randint(2, 6)):
                i, j = rng.choice(n_points, 2, replace=False)
                h0[i], h0[j] = h0[j] + 0.02*rng.randn(), h0[i] + 0.02*rng.randn()
        else:
            h0 = 0.6 * best_h + 0.4 * rng.rand(n_points)

        h0 = project_h(h0, target_sum)
        h_new, c5_new = pgd_optimize(h0, target_sum, 500, 40, 1500, 0.04)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: SLSQP with high-precision smooth max
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    def sabc(h, b=2000.0):
        corr = corr_arr(h)
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(b * (corr - m)))) / b) * dx

    def sacg(h, b=2000.0):
        return grad_corr(h, b, 30) * dx

    def cfun(h):
        return h.sum() - target_sum

    def cjac(h):
        return np.ones(n_points)

    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    for bf in [1500, 3000, 5000]:
        if time_left() < 40:
            break
        try:
            def obj_wrap(hv):
                corr = corr_arr(hv)
                m = np.max(corr)
                return (m + np.log(np.sum(np.exp(bf * (corr - m)))) / bf) * dx

            def obj_grad_wrap(hv):
                return grad_corr(hv, bf, 25) * dx

            res = minimize(obj_wrap, best_h, jac=obj_grad_wrap, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 8000, 'ftol': 1e-17})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 5: Ultra-fine local search
    if time_left() > 40:
        for it in range(200):
            if time_left() < 40:
                break
            h0 = best_h + rng.randn(n_points) * 0.002
            h0 = project_h(h0, target_sum)
            h_new, c5_new = pgd_optimize(h0, target_sum, 5000, 15, 800, 0.004)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 6: Coordinate-level fine tuning
    if time_left() > 20:
        for _ in range(2000):
            if time_left() < 20:
                break
            j = rng.randint(n_points)
            h0 = best_h.copy()
            h0[j] += rng.choice([-0.01, 0.01])
            h0 = project_h(h0, target_sum)
            c5_new = compute_c5(h0)
            if c5_new < best_c5:
                best_h = h0
                best_c5 = c5_new

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
