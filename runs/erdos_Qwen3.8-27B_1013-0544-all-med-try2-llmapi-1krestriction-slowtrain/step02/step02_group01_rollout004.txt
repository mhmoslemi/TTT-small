import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_n_global = 200
_beta_global = 100.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wk * (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += wk * (-h[mp + s])
        elif s < 0:

            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

    v, _ = smoothmax_val_grad(x, _beta_global)
    return v

def obj_grad(x):
    _, g = smoothmax_val_grad(x, _beta_global)
    return g

def slsqp_refine(h, maxiter, beta):

    _n_global = len(h)
    _beta_global = beta
    target = _n_global / 2.0
    bounds = Bounds(np.zeros(_n_global), np.ones(_n_global))
    one_vec = np.ones(_n_global)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

def subgradient_exp(h, max_iters, lr, decay, beta_w):

    target = n / 2.0

        m = np.max(corr)

        w = w / sw
        grad = np.zeros(n)
        for k in range(len(corr)):

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_topk(h, max_iters, lr0, decay, top_k):

    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)

    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)

        y = project_feasible(y, target)
        corr = compute_corr(y)

                mp = np.arange(0, n - s)
                grad[mp] += (-y[mp + s])
            elif s < 0:

                ms = np.arange(0, n - tt)
