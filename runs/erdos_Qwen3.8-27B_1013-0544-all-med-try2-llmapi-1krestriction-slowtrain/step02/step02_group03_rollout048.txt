import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def multi_lag_grad(h, top_k=10, beta_w=200.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta_w * (corr - m), -40.0, 40.0)
    w = np.exp(shifted)
    total = w.sum()
    if total < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= total
    
    grad = np.zeros(n)
    # Only iterate over lags with non-negligible weight
    thresh = 1e-5
    for k in np.where(w > thresh)[0]:
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad, corr


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def slsqp_run(h, n, target_sum, beta, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def multi_lag_grad(h, beta_w):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta_w * (corr - m), -40.0, 40.0)
    w = np.exp(shifted)
    total = w.sum()
    if total < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= total
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-8:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            end = n - lag
            grad[:end] += wk * (1.0 - h[lag:lag + end])
            grad[lag:lag + end] -= wk * h[:end]
        else:
            la = -lag
            end = n - la
            grad[la:la + end] += wk * (1.0 - h[:end])
            grad[:end] -= wk * h[la:la + end]
    return grad


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def slsqp_run(h, n, target_sum, beta, maxiter, ftol):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h.copy()


def subgrad_momentum(h, target_sum, n, max_iter, t0, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.85
    lr = 0.007
    beta_w = 150.0
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 25:
            break
        grad = multi_lag_grad(h, beta_w)
        v = mu * v + (1.0 - mu) * grad
        current_f = compute_c5(h)
        alpha = lr
        found = False
        for _ in range(10):
            h_try = project_feasible(h - alpha * v, target_sum)
            f_try = compute_c5(h_try)
            if f_try < current_f - 1e-12:
                h = h_try
                if f_try < best_c5:
                    best_c5 = f_try
                    best_h = h.copy()
                found = True
                break
            alpha *= 0.5
        if not found:
            lr = max(lr * 0.92, 1e-7)
            if lr <= 1.001e-7:
                v = np.zeros(n)
                lr = 0.005
            h = project_feasible(h - lr * v, target_sum)
            f_now = compute_c5(h)
            if f_now < best_c5:
                best_c5 = f_now
                best_h = h.copy()
        if it % 6000 == 5999:
            beta_w = min(beta_w * 2.0, 4000.0)
            mu = max(mu * 0.96, 0.5)
    return best_h, best_c5
