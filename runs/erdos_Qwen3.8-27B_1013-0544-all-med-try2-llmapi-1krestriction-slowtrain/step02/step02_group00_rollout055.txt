    """
    Find optimal delta for h[i]+=delta, h[j]-=delta.
    lag_indices: array of correlation indices (0 to 2n-2)
    corr_vals: corresponding correlation values
    Returns the optimal delta.
    """
    a = np.zeros(len(lag_indices))
    for p in range(len(lag_indices)):
        m = lag_indices[p]
        L = m - (n - 1)
        di = 0.0
        if 0 <= i + L < n:
            di += 1.0 - h[i + L]
        if 0 <= i - L < n:
            di -= h[i - L]
        dj = 0.0
        if 0 <= j + L < n:
            dj += 1.0 - h[j + L]
        if 0 <= j - L < n:
            dj -= h[j - L]

    best = np.max(c)

    # Check boundaries

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_optimal_delta(h, i, j, lag_idx, corr_vals, n):
    a = np.zeros(len(lag_idx))
    for p in range(len(lag_idx)):
        L = int(lag_idx[p]) - (n - 1)
        di = 0.0
        if 0 <= i + L < n:
            di += 1.0 - h[i + L]
        if 0 <= i - L < n:
            di -= h[i - L]
        dj = 0.0
        if 0 <= j + L < n:
            dj += 1.0 - h[j + L]
        if 0 <= j - L < n:
            dj -= h[j - L]
        a[p] = di - dj
    c = corr_vals
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-13:
        return 0.0
    best = float(np.max(c))
    best_d = 0.0
    for d in (d_lo, d_hi):
        val = float(np.max(c + a * d))
        if val < best:
            best = val
            best_d = d
    na = len(a)
    for p in range(na):
        for q in range(p + 1, na):
            da = a[p] - a[q]
            if abs(da) > 1e-13:
                d = (c[q] - c[p]) / da
                if d_lo <= d <= d_hi:
                    val = float(np.max(c + a * d))
                    if val < best:
                        best = val
                        best_d = d
    return best_d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)

    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Random pairwise coordinate descent with exact line search
    top_k = 10
    recompute_interval = 30
    pairs_since_recompute = 0
    total_pairs = 0

    while elapsed() < budget_s - 200:
        if pairs_since_recompute >= recompute_interval:
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argsort(corr)[::-1][:top_k]
            lag_idx = idx
            corr_vals = corr[idx]
            pairs_since_recompute = 0

        i, j = np.random.choice(n, 2, replace=False)
        delta = get_optimal_delta(h, i, j, lag_idx, corr_vals, n)
        if abs(delta) > 1e-12:
            h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h[j] = np.clip(h[j] - delta, 0.0, 1.0)
        pairs_since_recompute += 1
        total_pairs += 1

        if total_pairs % 3000 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-14:
                best_h = h.copy()
                best_c5 = c5_cur
            # Periodic perturbation to escape local optima
            if total_pairs > 30000 and total_pairs % 30000 == 0:
                sigma = 0.01 + 0.02 * (1.0 - elapsed() / budget_s)
                noise = np.random.normal(0, sigma, n)
                h_trial = project_h(h + noise, target)
                if compute_c5(h_trial) < compute_c5(h):
                    h = h_trial

    c5_cur = compute_c5(h)
    if c5_cur < best_c5:
        best_h, best_c5 = h.copy(), c5_cur

    # Phase 2: Structured full sweep
    h = best_h.copy()
    while elapsed() < budget_s - 100:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[::-1][:top_k]
        lag_idx = idx
        corr_vals = corr[idx]
        n_changed = 0
        for i in range(n):
            for j in range(i + 1, n):
                delta = get_optimal_delta(h, i, j, lag_idx, corr_vals, n)
                if abs(delta) > 1e-12:
                    h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                    h[j] = np.clip(h[j] - delta, 0.0, 1.0)
                    n_changed += 1
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_h, best_c5 = h.copy(), c5_cur
        if n_changed < 5:
            break

    # Phase 3: SLSQP polish
    if elapsed() < budget_s - 30:
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        beta = 400.0

        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            m = np.max(c)
            return float(m + np.log(np.sum(np.exp(np.clip(beta * (c - m), -500, 500)) / len(c))) / beta) * dx

        def con_fn(h_in):
            return np.sum(h_in) - target

        def con_jac(h_in):
            return np.ones(n)

        constraints = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
        try:
            res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

    # Phase 4: Final pairwise refinement
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        for total in range(300000):
            if elapsed() > budget_s - 3:
                break
            if total % 50 == 0:
                corr = np.correlate(h, 1.0 - h, mode="full")
                idx = np.argsort(corr)[::-1][:top_k]
                lag_idx = idx
                corr_vals = corr[idx]
            i, j = np.random.choice(n, 2, replace=False)
            delta = get_optimal_delta(h, i, j, lag_idx, corr_vals, n)
            if abs(delta) > 1e-12:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h[j] = np.clip(h[j] - delta, 0.0, 1.0)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_h, best_c5 = h.copy(), c5_cur

    return best_h, best_c5, n
