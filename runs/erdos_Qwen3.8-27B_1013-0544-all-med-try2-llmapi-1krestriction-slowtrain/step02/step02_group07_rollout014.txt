    e = np.exp(beta * (x - m))

    if tleft() > 30:
        h = best_h.copy()
        c5 = best_c5
        step = 0.0005
        stall = 0
        for it in range(50000):
            if tleft() < 3:
                break
            corr = compute_corrs(h)
            mx = corr.max()
            active = corr >= mx - 1e-10
            w = np.float64(active) / max(1, active.sum())
            g = active_grad(h, w) * dx
            g = g - g.mean()
            ok = False
            for ls in range(20):
                h2 = project_h(h - step * g, T)
                c52 = compute_c5(h2)
                if c52 < c5 - 1e-15:
                    h, c5 = h2, c52
                    ok = True
                    if c5 < best_c5:
                        best_h, best_c5 = h.copy(), c5
                        stall = 0
                    else:
                        stall += 1
                    break
                step *= 0.5
            if not ok:
                stall += 1
            if stall > 100:
                break

    best_h = project_h(best_h, T)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
