def subgradient_polish(h, n, t0, budget_s, n_steps=5000, base_lr=0.01, decay=0.999):
    """Subgradient method with decaying step size."""
    lr = base_lr
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for step in range(n_steps):
        if budget_s - (time.time() - t0) < 10:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(c)
        m = k_star - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + m <= n - 1:
                grad[i] += (1.0 - h[i + m])
            if 0 <= i - m <= n - 1:
                grad[i] -= h[i - m]
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-10:
            break
        h_new = project_feasible(h - lr * grad / norm_g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
            h = h_new
        lr *= decay
    return best_h, best_c5

def sa_optimize(h, n, t0, budget_s, rng, n_iter=200000, e_start=0.05, e_end=0.0001,
                d_start=0.2, d_end=0.01):

    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    E = e_start
    D = d_start
    e_ratio = (e_end / e_start) ** (1.0 / n_iter)
    d_ratio = (d_end / d_start) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 30:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-D, D)
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta < 0 or (E > 1e-10 and rng.random() < np.exp(-delta / E)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = hi - d
            h[j] = hj + d
        E *= e_ratio
        D *= d_ratio
    return best_h, best_c5

def smooth_max_obj(h, n, alpha):
    """Smooth approximation of max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    out = shifted.max()
    if out > 500:
        s = np.sum(np.exp(shifted - out))
    else:
        s = np.sum(np.exp(shifted))
    return float(m + np.log(s) / alpha)

def slsqp_smooth(h, n, alpha, maxiter=200, ftol=1e-12):
    """SLSQP with smooth max objective."""
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_max_obj(x, n, alpha), h,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

3. Solving the 1D QP exactly

For the 1D QP: given that k* is the active constraint, minimize r_{k*}(d) subject to 0≤h[i]+d≤1 and 0≤h[j]-d≤1.

This is the **active set method in 1D**, which is exact for the pair optimization step.

def project_feasible(h, n):

def slsqp_smooth(h, n, alpha, maxiter=200, ftol=1e-14):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        out = shifted.max()
        if out > 500:
            s = np.sum(np.exp(shifted - out))
        else:
            s = np.sum(np.exp(shifted))
        return m + np.log(s) / alpha
    def obj_grad(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        out = shifted.max()
        if out > 500:
            weights = np.exp(shifted - out)
            s = weights.sum()
        else:
            weights = np.exp(shifted)
            s = weights.sum()
        w = weights / s  # softmax weights
        # Gradient of sum w_k * c_k where c_k = r_k * dx
        # dc_k/dx[i] = dx * [(1-x[i+m]) * 1_{valid} - x[i-m] * 1_{valid}]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            m = k - (n - 1)
            for i in range(n):
                if 0 <= i + m <= n - 1:
                    grad[i] += w[k] * dx * (1.0 - x[i + m])
                if 0 <= i - m <= n - 1:
                    grad[i] -= w[k] * dx * x[i - m]
        return grad

             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
