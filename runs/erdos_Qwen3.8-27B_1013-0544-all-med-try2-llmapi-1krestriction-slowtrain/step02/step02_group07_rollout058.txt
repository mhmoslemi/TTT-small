def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, target_ones):
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target_ones - 1]
    binary = (h >= threshold).astype(float)
    # Fix if we have too many or too few ones
    while binary.sum() > target_ones:
        # Flip the smallest value that's 1
        ones = np.where(binary == 1)[0]
        flip_ind = ones[np.argmin(h[ones])]
        binary[flip_ind] = 0
    while binary.sum() < target_ones:
        # Flip the largest value that's 0
        zeros = np.where(binary == 0)[0]
        flip_ind = zeros[np.argmax(h[zeros])]
        binary[flip_ind] = 1
    return binary

def binary_local_search(h, c, max_iter=10000):
    n = len(h)
    dx = 2.0 / n
    current_max = c.max()
    
    for _ in range(max_iter):
        best_gain = 0
        best_swap = None
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                # Compute delta_c(k) for all k
                # delta_c(k) = h[i+k] + h[i-k] - h[j+k] - h[j-k]
                # where out-of-range indices give 0
                delta = np.zeros(2*n-1)
                # For each lag k (indexed as k - (n-1) in the full correlation)
                # The actual shift is from -(n-1) to (n-1)
                for shift in range(-(n-1), n):
                    # delta at this shift = h[i+shift] + h[i-shift] - h[j+shift] - h[j-shift]
                    val = 0.0
                    if 0 <= i + shift < n:
                        val += h[i + shift]
                    if 0 <= i - shift < n:
                        val += h[i - shift]
                    if 0 <= j + shift < n:
                        val -= h[j + shift]
                    if 0 <= j - shift < n:
                        val -= h[j - shift]
                    delta[shift + (n-1)] = val
                
                new_max = (c + delta).max()
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_swap = (i, j)
                    best_delta = delta
        
        if best_swap is not None and best_gain > 0:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            c = c + best_delta
            current_max = c.max()
        else:
            break
    
    return h, c, current_max

def binary_sa(h, c, start_time, time_limit, seed=42):
    np.random.seed(seed)
    n = len(h)
    current_max = c.max()
    best_max = current_max
    best_h = h.copy()
    best_c = c.copy()
    
    T = 5.0  # Initial temperature (in terms of c values, which are ~10-20)
    T_min = 0.1
    cooling = 0.9995
    
    while time.time() - start_time < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute delta
        delta = np.zeros(2*n-1)
        for shift in range(-(n-1), n):
            val = 0.0
            if 0 <= i + shift < n:
                val += h[i + shift]
            if 0 <= i - shift < n:
                val += h[i - shift]
            if 0 <= j + shift < n:
                val -= h[j + shift]
            if 0 <= j - shift < n:
                val -= h[j - shift]
            delta[shift + (n-1)] = val
        
        new_c = c + delta
        new_max = new_c.max()
        
        delta_obj = new_max - current_max
        if delta_obj < 0 or np.random.random() < np.exp(-delta_obj / T):
            h[i], h[j] = h[j], h[i]
            c = new_c
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                best_c = c.copy()
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c, best_max

    h_binary = h_binary.copy()
