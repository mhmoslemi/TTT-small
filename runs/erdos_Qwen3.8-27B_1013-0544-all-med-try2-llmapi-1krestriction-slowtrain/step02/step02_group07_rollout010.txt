   Even simpler: since the function is convex and piecewise linear on [0,1], we can find the minimum by:
   - Computing the subgradient at 0 and at 1
   - If subgradient at 0 > 0, minimum is at 0
   - If subgradient at 1 < 0, minimum is at 1
   - Otherwise, use bisection to find where subgradient crosses 0

   The subgradient at t is {max of slopes of active lines at t}, where active lines are those achieving the max at t.

   Actually, for a convex piecewise linear function, the subgradient at t is [min_active_slope, max_active_slope]. The minimum is where 0 ∈ subgradient, i.e., min_active_slope ≤ 0 ≤ max_active_slope.

           m_lo = max(0, n-1-j)
           m_hi = min(n-1, n-1+j)

               if 0 <= k <= n-1:

           m = i - j + (n-1)

               B[j] = 1.0

           total = 0.0
           for m in range(m_lo, m_hi+1):
               if m == i:
                   continue
               k = m + j - (n-1)
               if 0 <= k <= n-1:
                   total += h[m] * (1 - h[k])
           base[j] = total

       c = base + B  # constant term
       s = A - B  # slope
       
       # Minimize max_j (c[j] + s[j]*t) over t in [0,1]
       # This is a 1D convex problem
       # Use bisection on the subgradient
       lo, hi = 0.0, 1.0
       for _ in range(30):
           mid = (lo + hi) / 2
           vals = c + s * mid
           max_idx = np.argmax(vals)
           # Subgradient: slope of the active constraint
           # If the max is uniquely at max_idx, subgradient = s[max_idx]
           # If multiple achieve the max, subgradient = [min_slope, max_slope]
           active = np.where(np.abs(vals - vals[max_idx]) < 1e-10)[0]
           min_slope = s[active].min()
           max_slope = s[active].max()
           if min_slope > 0:
               hi = mid
           elif max_slope < 0:
               lo = mid
           else:
               break
       return mid

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-10:
