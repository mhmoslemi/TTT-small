import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_periodic(period, m):
    p = len(period)
    n = p * m
    h = np.tile(period, m)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

class PeriodicOBJ:
    def __init__(self, m, dx, beta):
        self.m = m
        self.dx = dx
        self.beta = beta
    def __call__(self, x):
        full = np.tile(x, self.m)
        c = np.correlate(full, 1.0 - full, mode="full")
        return smooth_max(c, self.beta) * self.dx

class PeriodicCon:
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

class DirectOBJ:
    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta
    def __call__(self, x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, self.beta) * self.dx

class DirectCon:
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

def coordinate_descent_period(cur, p, m, target, max_rounds=100):
    cur_c5 = compute_c5_periodic(cur, m)
    for r in range(max_rounds):
        improved = False
        for i in range(p):
            lo = max(0.0, cur[i] - 0.2)
            hi = min(1.0, cur[i] + 0.2)
            trials = np.linspace(lo, hi, 21)
            for tv in trials:
                trial = cur.copy()
                trial[i] = tv
                trial = project_h(trial, target)
                c5t = compute_c5_periodic(trial, m)
                if c5t < cur_c5 - 1e-13:
                    cur = trial.copy()
                    cur_c5 = c5t
                    improved = True
        if not improved:
            break
    return cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
        best_h = parent.copy()
        best_c5 = compute_c5(parent)
        best_n = len(parent)

    periods_to_try = [
        (7, 20), (7, 10), (7, 15), (7, 28),
        (10, 14), (10, 10), (10, 7), (10, 20),
        (14, 10), (14, 5), (14, 15),
        (5, 28), (5, 14), (5, 20),
        (20, 7), (20, 5),
        (12, 10), (12, 7),
        (8, 14), (8, 10),
        (4, 17), (4, 25),
        (6, 15), (6, 12),
        (28, 5),
        (35, 4),
    ]

    for (p, m) in periods_to_try:
        if time_left() < 25:
            break
        n = p * m
        if n > 500:
            continue
        target = p / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(p), np.ones(p))

        inits = []
        inits.append(np.full(p, 0.5))
        for freq in range(1, p // 2 + 1):
            s = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(p) / p)
            inits.append(project_h(s, target))
        for amp in [0.25, 0.4, 0.5]:
            alt = 0.5 + amp * np.where(np.arange(p) % 2 == 0, 1.0, -1.0)
            inits.append(project_h(alt, target))
        for _ in range(4):
            r = np.random.uniform(0.05, 0.95, p)
            inits.append(project_h(r, target))

        if best_h is not None and len(best_h) >= p:
            n_b = len(best_h)
            if n_b % p == 0:
                npers = n_b // p
                avg = np.zeros(p)
                for k in range(npers):
                    avg += best_h[k*p:(k+1)*p]
                avg /= npers
                inits.append(project_h(avg, target))
            n_seg = max(2, min(p, n_b))
            x_s = np.linspace(0, 1, n_seg)
            x_p = np.linspace(0, 1, p)
            seg = best_h[:n_seg]
            interp_p = np.interp(x_p, x_s, seg)
            inits.append(project_h(interp_p, target))

        local_best_period = None
        local_best_c5 = float('inf')

        for h_init in inits:
            if time_left() < 12:
                break
            cur = project_h(h_init.copy(), target)
            cur_c5 = compute_c5_periodic(cur, m)
            if cur_c5 < local_best_c5:
                local_best_c5 = cur_c5
                local_best_period = cur.copy()

            for beta in [40, 80, 160, 320, 640, 1200]:
                if time_left() < 8:
                    break
                obj = PeriodicOBJ(m, dx, beta)
                con = PeriodicCon(target)
                constraints = [{'type': 'eq', 'fun': con, 'jac': con.jac}]
                try:
                    res = minimize(obj, cur, method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 3000, 'ftol': 1e-14, 'maxls': 80})
                    h_new = project_h(res.x, target)
                    c5_new = compute_c5_periodic(h_new, m)
                    if c5_new < cur_c5 - 1e-14:
                        cur = h_new.copy()
                        cur_c5 = c5_new
                    if cur_c5 < local_best_c5:
                        local_best_c5 = cur_c5
                        local_best_period = cur.copy()
                except Exception:
                    pass

            if time_left() > 8:
                cur, cur_c5 = coordinate_descent_period(cur, p, m, target, max_rounds=60)
                if cur_c5 < local_best_c5:
                    local_best_c5 = cur_c5
                    local_best_period = cur.copy()

        if local_best_period is not None and local_best_c5 < best_c5:
            full_h = np.tile(local_best_period, m)
            best_h = full_h.copy()
            best_c5 = local_best_c5
            best_n = n

        if local_best_period is not None and time_left() > 20:
            cur = local_best_period.copy()
            for nm_beta in [200, 500]:
                if time_left() < 8:
                    break
                obj = PeriodicOBJ(m, dx, nm_beta)
                con = PeriodicCon(target)
                constraints = [{'type': 'eq', 'fun': con, 'jac': con.jac}]
                try:
                    res = minimize(obj, cur, method='Nelder-Mead',
                                   options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12})
                    h_new = project_h(res.x, target)
                    c5_new = compute_c5_periodic(h_new, m)
                    if c5_new < best_c5:
                        full_h = np.tile(h_new, m)
                        best_h, best_c5, best_n = full_h.copy(), c5_new, n
                        cur = h_new.copy()
                except Exception:
                    pass

    # Direct optimization on best solution to break periodicity
    if time_left() > 30 and best_h is not None:
        n_d = len(best_h)
        target_d = n_d / 2.0
        dx_d = 2.0 / n_d
        bounds_d = Bounds(np.zeros(n_d), np.ones(n_d))
        cur_d = best_h.copy()
        cur_d_c5 = best_c5

        for beta in [200, 400, 800, 1500]:
            if time_left() < 15:
                break
            obj = DirectOBJ(dx_d, beta)
            con = DirectCon(target_d)
            constraints = [{'type': 'eq', 'fun': con, 'jac': con.jac}]
            try:
                res = minimize(obj, cur_d, method='SLSQP', bounds=bounds_d,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-14, 'maxls': 100})
                h_new = project_h(res.x, target_d)
                c5_new = compute_c5(h_new)
                if c5_new < cur_d_c5 - 1e-14:
                    cur_d = h_new.copy()
                    cur_d_c5 = c5_new

            except Exception:
                pass

        # Subgradient refinement
        for iteration in range(4000):
            if time_left() < 5:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top_k = np.argsort(corr)[::-1][:4]
            grad = np.zeros(n_d)
            for lag_idx in top_k:

                    i2 = j + lag
                    if 0 <= i1 < n_d:

                    if 0 <= i2 < n_d:
                        grad[j] -= best_h[i2]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad /= gnorm
            for step_mult in [1.0, 0.5, 0.2]:

                h_trial = project_h(best_h - step * grad, target_d)

                if c5_trial < best_c5 - 1e-14:
                    best_h = h_trial
                    best_c5 = c5_trial
                    break

    return best_h, best_c5, best_n
