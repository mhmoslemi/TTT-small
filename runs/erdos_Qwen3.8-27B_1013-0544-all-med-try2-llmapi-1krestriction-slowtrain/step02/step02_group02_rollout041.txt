def project_onto_constraint(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # This is monotone decreasing in lambda
    
    lo = h.min() - target  # lambda such that all become 0
    hi = h.max()           # lambda such that all become 1... not quite
    
    # Better bounds:
    # If lambda <= min(h): result is h (sum = sum(h))
    # If lambda >= max(h) - target... 
    
    # Let's just use bisection
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

Let me also think about the 2-point optimal step approach:

For a move from (h[i], h[j]) to (h[i]+d, h[j]-d), the change in each correlation c_k is:
Δc_k = d * dx * [(1-h[i+lag_k]_valid - h[i-lag_k]_valid) - (1-h[j+lag_k]_valid - h[j-lag_k]_valid)]

Let a_k = dx * [(1-h[i+lag_k]_valid - h[i-lag_k]_valid) - (1-h[j+lag_k]_valid - h[j-lag_k]_valid)]

Then new c_k = c_k + d * a_k

We want to minimize max_k (c_k + d * a_k) over d in [d_min, d_max] (where d_min and d_max ensure feasibility).

This is a 1D Chebyshev approximation problem. The optimal d can be found by:
1. The function f(d) = max_k (c_k + d*a_k) is convex in d (max of linear functions)
2. We can use ternary search or find the point where the two "active" constraints are equal

Actually, since it's convex, we can use bisection on the derivative (subgradient). The subgradient of f at d is a_k* where k* is an argmax.

Or more simply: since f is convex and piecewise linear, the minimum is where the slope changes sign, i.e., where there exist k1, k2 with k1 ≠ k2 such that c_{k1} + d*a_{k1} = c_{k2} + d*a_{k2} and this value is ≥ all other c_k + d*a_k.

For a practical approach, ternary search on the 1D convex function works well.

def optimal_2pt_step(h, n, i, j, d_bounds):
    """Find optimal d for 2-point move (h[i]+=d, h[j]-=d)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Compute a_k for each k
    lags = np.arange(n - 1, -(n - 1), -1)
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = lags[k]
        vi = 1.0
        if 0 <= i + lag < n:
            vi -= h[i + lag]
        if 0 <= i - lag < n:
            vi -= h[i - lag]
        vj = 1.0
        if 0 <= j + lag < n:
            vj -= h[j + lag]
        if 0 <= j - lag < n:
            vj -= h[j - lag]
        a[k] = dx * (vi - vj)
    
    # f(d) = max_k (c[k] + d * a[k])
    # Find d that minimizes f(d) in [d_bounds[0], d_bounds[1]]

        f2 = np.max(c + m2 * a)

            hi = m2
        else:
            lo = m1

            # So g[:n-lag, k] -= h[lag:n]

            g[-lag:, k] -= h[:n+lag]
            # h[i-lag] valid for i=0,...,n-1+lag
            g[:n+lag, k] -= h[-lag:]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        w = np.exp(shifted - mx)
        w /= w.sum()
    else:
        w = np.exp(shifted)
        w /= w.sum()
    grad = np.zeros(n)
    lags = np.arange(n - 1, -(n - 1), -1)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = lags[k]
        val = np.ones(n)
        if lag >= 0:
            val[:n - lag] -= h[lag:]
            val[lag:] -= h[:n - lag]
        else:
            val[-lag:] -= h[:n + lag]
            val[:n + lag] -= h[-lag:]
        grad += w[k] * val
    return grad * dx

def compute_g(h, n):
    dx = 2.0 / n
    g = np.ones((n, 2 * n - 1)) * dx
    lags = np.arange(n - 1, -(n - 1), -1)
    for k in range(2 * n - 1):
        lag = lags[k]
        if lag >= 0:
            g[:n - lag, k] -= h[lag:]
            g[lag:, k] -= h[:n - lag]
        else:
            g[-lag:, k] -= h[:n + lag]
            g[:n + lag] -= h[-lag:]
    return g

def two_point_sweep(h, n, t0, budget_s, max_rounds=15, ts_iters=45):

    h = h.copy()

        g = compute_g(h, n)

                break

                d_lo = max(-h[i], h[j] - 1.0)

def pg_optimize(h, n, t0, budget_s):

    best_c5 = compute_c5(h, n)
    for phase in range(6):

def slsqp_polish(h, n, alpha, maxiter=2000):
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha)
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_optimize(h, n, t0, budget_s, rng, n_iter, e_start, e_min, d_scale):
    E = e_start
    e_ratio = (e_min / e_start) ** (1.0 / max(n_iter, 1))
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    h = h.copy()
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 15:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0, d_scale) * (1.0 if rng.random() < 0.5 else -1.0)
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta < 0 or (delta < 1.0 and rng.random() < np.exp(-delta / max(E, 1e-15))):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = hi - d
            h[j] = hj + d
        E *= e_ratio
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Projected gradient
    if remaining() > 60:
        best_h, best_c5 = pg_optimize(best_h, n, t0, budget_s)

    # Phase 2: 2-point sweep
    if remaining() > 60:
        best_h, best_c5 = two_point_sweep(best_h, n, t0, budget_s, max_rounds=10)

    # Phase 3: SLSQP polish
    if remaining() > 40:
        for a in [5000, 50000]:
            if remaining() < 20:
                break
            ht = slsqp_polish(best_h, n, a)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = ht, c

    # Phase 4: SA exploration
    if remaining() > 80:
        best_h, best_c5 = sa_optimize(best_h, n, t0, budget_s, rng,
                                      300000, 0.04, 1e-6, 0.12)

    # Phase 5: Polish after SA
    if remaining() > 60:
        best_h, best_c5 = pg_optimize(best_h, n, t0, budget_s)
    if remaining() > 40:
        best_h, best_c5 = two_point_sweep(best_h, n, t0, budget_s, max_rounds=8)

    # Phase 6: Random restarts
    while remaining() > 100:
        scale = rng.choice([0.02, 0.04, 0.08, 0.12])
        h_p = project_feasible(best_h + rng.normal(0, scale, size=n), n)
        h_p = slsqp_polish(h_p, n, 5000, maxiter=800)
        c = compute_c5(h_p, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_p, c
            if remaining() > 60:
                best_h, best_c5 = pg_optimize(best_h, n, t0, budget_s)
            if remaining() > 40:
                best_h, best_c5 = two_point_sweep(best_h, n, t0, budget_s, max_rounds=8)

    # Phase 7: Final intensive 2-point sweep
    if remaining() > 20:
        best_h, best_c5 = two_point_sweep(best_h, n, t0, budget_s, max_rounds=20, ts_iters=50)

    # Phase 8: Final SLSQP
    if remaining() > 15:
        ht = slsqp_polish(best_h, n, 200000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
