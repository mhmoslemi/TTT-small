    h = h.copy()
    lo, hi = -1.0, 2.0  # wider range for safety
    for _ in range(80):

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def obj_and_grad(h, n, beta, lam):
    """Objective and gradient for L-BFGS-B."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Smooth max
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()  # shape (2n-1,)
    sm = m + np.log(exps.sum()) / beta
    
    # Gradient of smooth max * dx
    grad_sm = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for j in range(n):
            k1 = i - j  # lag for (1-h[j]) term
            k2 = j - i  # lag for h[j] term
            idx1 = k1 + (n - 1)
            idx2 = k2 + (n - 1)
            s1 += w[idx1] * (1.0 - h[j])
            s2 += w[idx2] * h[j]

    penalty = lam * (h.sum() - n / 2.0) ** 2
    grad_pen = 2.0 * lam * (h.sum() - n / 2.0)
    
    obj = sm * dx + penalty
    grad = grad_sm + grad_pen
    return obj, grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0

        tau = (lo + hi) / 2.0
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    penalty = lam * (h.sum() - n / 2.0) ** 2
    return sm * dx + penalty

    idx = np.arange(n)

    W2 = w[(idx[None, :] - idx[:, None]) + (n - 1)]  # (n,n), W2[i,j] = w[(j-i)+(n-1)]
    
    grad_sm = dx * (W1 @ (1.0 - h) - W2 @ h)

        def fun(x):
            return smooth_max_obj(x, n, beta, lam)
        def jac(x):
            return smooth_max_grad(x, n, beta, lam)
        try:

                          bounds=bnds, options={'maxiter': max_iters_per_beta, 'ftol': 1e-16, 'gtol': 1e-10})
            current = project_polytope(res.x, target)
        except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def _obj(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    return sm * dx + lam * (h.sum() - n / 2.0) ** 2

def _grad(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()
    idx = np.arange(n)
    W1 = w[(idx[:, None] - idx[None, :]) + (n - 1)]
    W2 = w[(idx[None, :] - idx[:, None]) + (n - 1)]
    g = dx * (W1 @ (1.0 - h) - W2 @ h)
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

def lbfgs_refine(h, n, bnds, beta, lam, maxiter):
    def fun(x):
        return _obj(x, n, beta, lam)
    def jac(x):
        return _grad(x, n, beta, lam)
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_polytope(res.x, n / 2.0)
    except Exception:
        return h

def subgrad_worst(h, n, target, step, n_worst=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[-n_worst:]
    g = np.zeros(n)
    for idx_k in top:
        k = idx_k - (n - 1)
        for i in range(n):
            v = 0.0
            j1 = i - k
            j2 = i + k
            if 0 <= j1 < n:
                v += 1.0 - h[j1]
            if 0 <= j2 < n:
                v -= h[j2]
            g[i] += v
    g /= n_worst
    return project_polytope(h - step * g, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def _obj(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    return sm * dx + lam * (h.sum() - n / 2.0) ** 2

def _grad(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()
    idx = np.arange(n)
    W1 = w[(idx[:, None] - idx[None, :]) + (n - 1)]
    W2 = w[(idx[None, :] - idx[:, None]) + (n - 1)]
    g = dx * (W1 @ (1.0 - h) - W2 @ h)
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

def lbfgs_refine(h, n, bnds, beta, lam, maxiter):
    def fun(x):
        return _obj(x, n, beta, lam)
    def jac(x):
        return _grad(x, n, beta, lam)
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_polytope(res.x, n / 2.0)
    except Exception:
        return h

def subgrad_worst(h, n, target, step, n_worst=3):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[-n_worst:]
    g = np.zeros(n)
    for idx_k in top:
        k = idx_k - (n - 1)
        for i in range(n):
            v = 0.0
            j1 = i - k
            j2 = i + k
            if 0 <= j1 < n:
                v += 1.0 - h[j1]
            if 0 <= j2 < n:
                v -= h[j2]
            g[i] += v
    g /= float(n_worst)
    return project_polytope(h - step * g, target)

def coordinate_descent(h, n, target, n_sweeps=2):
    for _ in range(n_sweeps):
        for i in range(n):
            if rem() < 30:
                return project_polytope(h, target)
            lo_i, hi_i = 0.0, 1.0
            old = h[i]
            for _ in range(30):
                mid = (lo_i + hi_i) / 2.0
                h[i] = mid
                c_new = compute_c5(h)
                h[i] = (lo_i + hi_i) / 4.0
                c_q1 = compute_c5(h)
                h[i] = (lo_i + hi_i) * 3 / 4.0
                c_q3 = compute_c5(h)
                h[i] = mid
                if c_q1 < c_q3:
                    hi_i = mid
                else:
                    lo_i = mid
            h[i] = (lo_i + hi_i) / 2.0
            h = project_polytope(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    target = n / 2.0
    bnds = [(0.0, 1.0)] * n

    def rem():
        return budget_s - (time.time() - t0)

    best_h = project_polytope(h_init, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-scale refinement from initial
    if rem() > 50:
        for beta, lam, mi in [(20, 5, 200), (50, 15, 200), (100, 40, 250), (200, 80, 250)]:
            if rem() < 20:
                break
            h_t = lbfgs_refine(best_h, n, bnds, beta, lam, mi)
            c = compute_c5(h_t)
            if c < best_c5:
                best_h, best_c5 = h_t, c

    # Phase 2: Diverse starting points
    starts = [best_h.copy()]
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    for t in range(8):
        a1 = np.random.uniform(0.1, 0.5)
        f1 = np.random.uniform(1.0, 5.0) * np.pi
        p1 = np.random.uniform(0, 2 * np.pi)
        h_s = 0.5 + a1 * np.sin(f1 * x + p1)
        if np.random.rand() > 0.4:
            a2 = np.random.uniform(0, 0.2)
            f2 = np.random.uniform(2.0, 8.0) * np.pi
            p2 = np.random.uniform(0, 2 * np.pi)
            h_s += a2 * np.sin(f2 * x + p2)
        starts.append(project_polytope(h_s, target))
    for t in range(3):
        nb = np.random.randint(3, 7)
        h_b = np.zeros(n)
        for b in range(nb):
            s = b * n // nb
            e = (b + 1) * n // nb if b < nb - 1 else n
            h_b[s:e] = np.random.uniform(0, 1)
        starts.append(project_polytope(h_b, target))
    starts.append(project_polytope(np.random.uniform(0, 1, n), target))

    for h_s in starts:
        if rem() < 40:
            break
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_h, best_c5 = h_s, c_s
        for beta, lam, mi in [(50, 20, 150), (150, 60, 200), (400, 150, 200)]:
            if rem() < 15:
                break
            h_t = lbfgs_refine(h_s, n, bnds, beta, lam, mi)
            c = compute_c5(h_t)
            if c < best_c5:
                best_h, best_c5 = h_t, c

    # Phase 3: Iterative subgradient + noise + L-BFGS-B
    iters = 0
    while rem() > 100:
        iters += 1
        h = best_h.copy()
        for st in [0.02, 0.01, 0.005]:
            h = subgrad_worst(h, n, target, st, n_worst=3)
        noise_scale = max(0.003, 0.025 * np.exp(-iters / 12.0))
        h += np.random.randn(n) * noise_scale
        h = project_polytope(h, target)
        for beta, lam, mi in [(200, 100, 200), (500, 300, 200)]:
            if rem() < 20:
                break
            h_t = lbfgs_refine(h, n, bnds, beta, lam, mi)
            c = compute_c5(h_t)
            if c < best_c5:
                best_h, best_c5 = h_t, c

    # Phase 4: Coordinate descent polish on best
    if rem() > 40:
        h_cd = coordinate_descent(best_h.copy(), n, target, n_sweeps=1)
        c_cd = compute_c5(h_cd)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd
        h_t = lbfgs_refine(best_h, n, bnds, 500, 300, 300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_h, best_c5 = h_t, c

    # Phase 5: Final high-precision polish
    if rem() > 25:
        for beta, lam, mi in [(400, 200, 400), (800, 500, 400), (1500, 1000, 500)]:
            if rem() < 12:
                break
            h_t = lbfgs_refine(best_h, n, bnds, beta, lam, mi)
            c = compute_c5(h_t)
            if c < best_c5:
                best_h, best_c5 = h_t, c

    # Phase 6: Final subgradient nudge + quick refine
    if rem() > 12:
        h = best_h.copy()
        for st in [0.003, 0.001, 0.0005]:
            h = subgrad_worst(h, n, target, st, n_worst=5)
        h += np.random.randn(n) * 0.003
        h = project_polytope(h, target)
        h_t = lbfgs_refine(h, n, bnds, 800, 500, 300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_h, best_c5 = h_t, c

    best_h = project_polytope(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
