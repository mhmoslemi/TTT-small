                for idx in near:
                    if idx != worst_idx:
                        lag_other = idx - (n - 1)
                        grad += 0.2 * lag_gradient(cur, lag_other, n)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm

            if g_prev is not None:
                s = cur - h_prev
                y = grad - g_prev
                ss = s @ s
                sy = s @ y
                yy = y @ y
                if ss > 1e-20 and yy > 1e-20:
                    if sy > 1e-10:
                        step_bb = ss / sy
                    else:
                        step_bb = max(-sy / yy, 1e-5)
                    step = 0.5 * step_prev + 0.5 * min(step_bb, 0.08)
                else:
                    step = step_prev * 0.85
            else:
                step = 0.03

            accepted = False
            for factor in [1.0, 0.5, 0.25, 0.1]:
                h_trial = project_h(cur - step * factor * grad, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    accepted = True
                    break

            if accepted:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 80:
                    scale = 0.003 + 0.008 * min(no_improve / 200.0, 1.0)
                    delta = np.random.randn(n) * scale
                    h_trial = project_h(cur + delta, target_sum)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < cur_c5:
                        cur = h_trial
                        cur_c5 = c5_trial
                        no_improve = 0

            if cur_c5 < best_c5:
                best_h = cur.copy()
                best_c5 = cur_c5
                best_n = n

            g_prev = grad.copy()
            h_prev = cur.copy()
            step_prev = step if accepted else step * 0.6

        # SLSQP polish
        if time_left() > 70:
            bounds = Bounds(np.zeros(n), np.ones(n))

            def obj_fn(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, 3000.0) * dx

            def con_fn(x_in):
                return np.sum(x_in) - target_sum

            def con_jac_fn(x_in):
                return np.ones(x_in.shape)

            constraints = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac_fn}]
            try:
                res = minimize(obj_fn, cur.copy(), method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 30000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    cur = h_new
                    cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h = cur.copy()
                    best_c5 = cur_c5
                    best_n = n
            except Exception:
                pass

    # Final high-precision polish
    if time_left() > 30:
        n_f = best_n
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))

        def final_obj(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, 8000.0) * dx_f

        def final_con(x_in):
            return np.sum(x_in) - target_f

        def final_jac(x_in):
            return np.ones(x_in.shape)

        constraints_f = [{'type': 'eq', 'fun': final_con, 'jac': final_jac}]
        try:
            res = minimize(final_obj, best_h.copy(), method='SLSQP', bounds=bounds_f,
                           constraints=constraints_f,
                           options={'maxiter': 50000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

        for it in range(15000):
            if time_left() < 5:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - (n_f - 1)
            grad = lag_gradient(best_h, worst_lag, n_f)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.002 / (1.0 + 0.001 * it)
            h_trial = project_h(best_h - step * grad, target_f)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = c5_trial

    return best_h, best_c5, best_n
