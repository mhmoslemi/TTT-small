    weights = exp_shifted / S

        w = weights[k]
        if w < 1e-18:

        lag = k - (n - 1)
        # First term: (1-h[j-lag]) for valid j
        if lag >= 0:

            j_start, j_end = lag, n
            idx = np.arange(j_start, j_end)
            grad[idx] += w * (1.0 - h[idx - lag])
            # Second term: -h[j+lag] for j in [0, n-1-lag]
            j2_end = n - lag
            if j2_end > 0:
                idx2 = np.arange(0, j2_end)
                grad[idx2] += w * (-h[idx2 + lag])
        else:
            # lag < 0, let l = -lag > 0
            l = -lag
            # j-lag = j+l in [0, n-1], so j in [0, n-1-l]
            j_end = n - l
            if j_end > 0:
                idx = np.arange(0, j_end)
                grad[idx] += w * (1.0 - h[idx + l])
            # Second term: -h[j+lag] = -h[j-l] for j in [l, n-1]
            idx2 = np.arange(l, n)
            grad[idx2] += w * (-h[idx2 - l])
    
    return grad

                res = minimize(_obj, best_h, jac=_grad, method='SLSQP',
                              bounds=bounds_f, constraints=cons_f,
                              options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_h_sum(res.x, ts_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Diversification: try perturbed restarts if time remains
    if best_h is not None and time.time() < t_start + budget_s - 10:
        n_f = best_n
        ts_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        cons_f = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        _MS['ts'] = ts_f
        for scale in [0.05, 0.02, 0.01]:
            if time.time() > t_start + budget_s - 5:
                break
            h_pert = best_h + np.random.randn(n_f) * scale
            h_pert = project_h_sum(h_pert, ts_f)
            _MS['beta'] = 500
            _MS['dx'] = dx_f
            try:
                res = minimize(_obj, h_pert, jac=_grad, method='SLSQP',
                              bounds=bounds_f, constraints=cons_f,
                              options={'maxiter': 1000, 'ftol': 1e-14})
                h_new = project_h_sum(res.x, ts_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
