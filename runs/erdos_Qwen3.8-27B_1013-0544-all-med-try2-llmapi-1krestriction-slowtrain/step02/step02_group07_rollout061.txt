        return budget_s - (time.time() - start_time)
    bounds = Bounds(np.zeros(n), np.ones(n))
    lam = 500.0

    # Phase 1: L-BFGS-B with increasing beta
    for beta in [30, 60, 120, 250, 500, 1000, 2000]:
        if tleft() < 25:
            break
        try:
            res = minimize(smax_obj_grad, best_h, args=(beta, lam, n, dx, target),
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-10, 'maxfun': 20000})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Iterative reweighting on top lags
    if tleft() > 80:
        for it in range(15):
            if tleft() < 25:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top_m = max(3, min(8, (2*n-1)//10))
            top_idx = np.argsort(corr)[-top_m:]
            weights = np.full(2*n-1, 0.05)
            weights[top_idx] = 1.0
            weights /= np.sum(weights)
            try:
                res = minimize(weighted_obj_grad, best_h, args=(weights, lam, n, dx, target),
                               method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass
            # Also try with smooth max at high beta from the new point
            if tleft() < 25:
                break
            try:
                res = minimize(smax_obj_grad, best_h, args=(500, lam, n, dx, target),
                               method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 3: Simulated annealing with targeted moves
    if tleft() > 60:
        h_cur = best_h.copy()
        c5_cur = best_c5
        T = 0.01
        T_min = 0.00005
        cooling = 0.9993
        no_improve = 0
        max_iters = 200000
        for it in range(max_iters):
            if tleft() < 25:
                break
            h_new = h_cur.copy()
            mt = np.random.random()
            if mt < 0.20:
                # Lag-targeted: gradient step on worst lag
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                k_m = int(np.argmax(corr))
                d = k_m - (n - 1)
                n_adj = np.random.randint(3, min(10, n//5 + 1))
                if d > 0:
                    idx = np.random.choice(n - d, min(n_adj, n - d), replace=False)
                    for i in idx:
                        h_new[i + d] -= 0.03 + 0.05 * np.random.random()
                        h_new[i] += 0.02 + 0.04 * np.random.random()
                elif d < 0:
                    da = -d
                    idx = np.random.choice(n - da, min(n_adj, n - da), replace=False)
                    for i in idx:
                        h_new[i] -= 0.03 + 0.05 * np.random.random()
                        h_new[i + da] += 0.02 + 0.04 * np.random.random()
                else:
                    idx = np.random.choice(n, n_adj, replace=False)
                    for i in idx:
                        h_new[i] += 0.06 if h_cur[i] < 0.5 else -0.06
            elif mt < 0.38:
                # Top-k lag targeted (multiple lags)
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                top3 = np.argsort(corr)[-3:]
                for k_m in top3:
                    d = k_m - (n - 1)
                    n_adj = np.random.randint(1, 4)
                    if d > 0:
                        idx = np.random.choice(n - d, min(n_adj, n - d), replace=False)
                        for i in idx:
                            h_new[i + d] -= 0.02 + 0.03 * np.random.random()
                    elif d < 0:
                        da = -d
                        idx = np.random.choice(n - da, min(n_adj, n - da), replace=False)
                        for i in idx:
                            h_new[i] -= 0.02 + 0.03 * np.random.random()
            elif mt < 0.52:
                # Block swap
                blen = np.random.randint(2, min(6, n//12 + 1))
                i = np.random.randint(0, n - blen + 1)
                j = np.random.randint(0, n - blen + 1)
                tr = 0
                while abs(j - i) < blen and tr < 20:
                    j = np.random.randint(0, n - blen + 1)
                    tr += 1
                if abs(j - i) >= blen:
                    tmp = h_new[i:i+blen].copy()
                    h_new[i:i+blen] = h_new[j:j+blen]
                    h_new[j:j+blen] = tmp
            elif mt < 0.68:
                # Gaussian perturbation on random subset
                frac = 0.08 + 0.25 * (T / 0.01)
                mask = np.random.random(n) < frac
                nm = int(mask.sum())
                if nm > 0:
                    h_new[mask] += np.random.randn(nm) * (0.02 + 0.06 * T / 0.01)
            elif mt < 0.80:
                # Pair swaps
                nsw = np.random.randint(1, 4)
                for _ in range(nsw):
                    i, j = np.random.choice(n, 2, replace=False)
                    h_new[i], h_new[j] = h_new[j], h_new[i]
            elif mt < 0.90:
                # Gradient-guided step (approximate)
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                m_c = np.max(corr)
                w = np.exp(200.0 * (corr - m_c))
                w /= np.sum(w)
                gstep = np.zeros(n)
                for k in range(2*n-1):
                    wk = w[k]
                    if wk < 1e-10:
                        continue
                    d = k - (n-1)
                    if d >= 0:
                        gstep[:n-d] += wk * (1.0 - h_cur[d:])
                        gstep[d:]   -= wk * h_cur[:n-d]
                    else:
                        dp = -d
                        gstep[dp:]   += wk * (1.0 - h_cur[:n-dp])
                        gstep[:n-dp] -= wk * h_cur[dp:]
                step = 0.02 + 0.03 * np.random.random()
                h_new += step * gstep
            else:
                # Mirror a small region
                blen = np.random.randint(3, min(7, n//10 + 1))
                i = np.random.randint(0, n - blen + 1)
                old_sum = h_new[i:i+blen].sum()
                h_new[i:i+blen] = 1.0 - h_new[i:i+blen]
                new_sum = h_new[i:i+blen].sum()
                others = np.setdiff1d(np.arange(n), np.arange(i, i+blen))
                if len(others) > 0:
                    h_new[others] += (old_sum - new_sum) / len(others)
            h_new = project_h(h_new, target)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(150.0, delta / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            T = max(T * cooling, T_min)
            if no_improve > 400:
                T = max(T, 0.006)
                no_improve = 0

    # Phase 4: Multi-start L-BFGS-B from perturbed best
    if tleft() > 30:
        for attempt in range(5):
            if tleft() < 15:
                break
            h0 = best_h.copy()
            n_pert = np.random.randint(3, 10)
            idx = np.random.choice(n, n_pert, replace=False)
            h0[idx] += np.random.randn(n_pert) * 0.04
            h0 = project_h(h0, target)
            beta = np.random.choice([300, 600, 1000])
            try:
                res = minimize(smax_obj_grad, h0, args=(beta, lam, n, dx, target),
                               method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-11})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 5: Final high-precision polish
    if tleft() > 10:
        for beta in [1500, 3000, 5000]:
            if tleft() < 8:
                break
            try:
                res = minimize(smax_obj_grad, best_h, args=(beta, lam, n, dx, target),
                               method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-12})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
