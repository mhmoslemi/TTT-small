from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def periodic_corr_matrix(h):
    """Compute S_r = sum_j h[j] * (1 - h[(j+r) % p]) for all r."""
    p = len(h)
    S = np.zeros(p)
    for r in range(p):
        shifted = np.roll(h, -r)
        S[r] = np.sum(h * (1.0 - shifted))
    return S

def optimize_periodic(p, target_c5=None, max_iter=50, seed=42):
    """Optimize a periodic pattern of length p to minimize max_r S_r."""
    rng = np.random.RandomState(seed)
    best_pattern = None

    for trial in range(20):

        h0 = project_h(h0.copy(), target)

        def obj(params):

            # Renormalize
            params = params / params.sum() * target
            S = periodic_corr_matrix(params)

    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

                step = 0.06 * mult / (1.0 + 0.005 * it)
                h_trial = project_h(cur - step * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    break
            if cur_c5 < best_c5:
                best_h = cur.copy()
                best_c5 = cur_c5

    return best_h, best_c5, len(best_h)
