import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    e = np.exp(beta * (x - m))
    return m + np.log(np.sum(e)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_periodic(src, n_src, n_dst):
    xo = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xn, xo, src, period=2.0)

def optimize_lifecycle(h0, n_pts, t_deadline, start_time, heavy=False):
    """Multi-phase optimization at fixed resolution."""
    ts = n_pts / 2.0
    dx = 2.0 / n_pts
    bnd = Bounds(np.zeros(n_pts), np.ones(n_pts))

    def cfun(x):
        return np.sum(x) - ts
    def cjac(x):
        return np.ones(n_pts)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    best = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase A: L-BFGS-B with penalty (fast, handles bounds)
    h = best.copy()
    betas_a = [25, 50, 100, 200, 350]
    pen_lambda = 5000.0
    for beta in betas_a:
        if time.time() >= t_deadline:
            break
        def obj_a(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(corr, float(beta)) * dx + pen_lambda * (np.sum(x) - ts) ** 2
        try:
            res = minimize(obj_a, h, method='L-BFGS-B', bounds=bnd,
                           options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
            h_new = project_h(res.x, ts)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass

    # Phase B: SLSQP with smooth max (handles constraint exactly)
    if time.time() < t_deadline:
        betas_b = [80, 150, 300, 600]
        for beta in betas_b:
            if time.time() >= t_deadline:
                break
            def obj_b(x):
                corr = np.correlate(x, 1.0 - x, mode="full")
                return smooth_max(corr, float(beta)) * dx
            try:
                res = minimize(obj_b, best, method='SLSQP', bounds=bnd, constraints=cons,
                               options={'maxiter': 1500, 'ftol': 1e-16, 'maxls': 80})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best = h_new.copy()
            except Exception:
                pass

    # Phase C: Top-k active lag focus
    if time.time() < t_deadline:
        corr = np.correlate(best, 1.0 - best, mode="full")
        max_corr = np.max(corr)
        for margin in [0.003, 0.001, 0.0005]:
            if time.time() >= t_deadline:
                break
            idx = np.where(corr >= max_corr - margin)[0]
            if len(idx) == 0:
                continue
            for pb in [150, 300]:
                if time.time() >= t_deadline:
                    break
                def obj_c(x, ii=idx):
                    c = np.correlate(x, 1.0 - x, mode="full")
                    return smooth_max(c[ii], float(pb)) * dx
                try:
                    res = minimize(obj_c, best, method='SLSQP', bounds=bnd, constraints=cons,
                                   options={'maxiter': 1000, 'ftol': 1e-16})
                    h_new = project_h(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best = h_new.copy()
                except Exception:
                    pass

    # Phase D: Coordinate perturbation local search (escape local optima)
    if time.time() < t_deadline and heavy:
        rng = np.random.RandomState(42)
        no_improve = 0
        for it in range(40):
            if time.time() >= t_deadline or no_improve > 20:
                break
            h_try = best.copy()
            # Perturb a random subset of coordinates
            n_perturb = max(3, n_pts // 10)
            idx_perturb = rng.choice(n_pts, size=n_perturb, replace=False)
            h_try[idx_perturb] += rng.randn(n_perturb) * 0.05
            h_try = project_h(h_try, ts)

            def obj_d(x):
                corr = np.correlate(x, 1.0 - x, mode="full")
                return smooth_max(corr, 200.0) * dx
            try:
                res = minimize(obj_d, h_try, method='L-BFGS-B', bounds=bnd,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best = h_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            except Exception:
                no_improve += 1

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(70, 0.5)

    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Resolution list: parent + nearby + a few higher
    res_list = [pn]
    for d in [-2, -1, 1, 2, 3, 5, 8, 14]:
        n = pn + d
        if 40 <= n <= 200 and n not in res_list:
            res_list.append(n)
    res_list.sort()

    # Time allocation
    total_avail = budget_s - 25
    # Parent resolution gets 35%, others share the rest
    parent_time = total_avail * 0.35
    other_time = total_avail * 0.65 / max(1, len(res_list) - 1)

    for n_pts in res_list:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 20:
            break
        t_alloc = parent_time if n_pts == pn else other_time

        ts = n_pts / 2.0

        # Starting point: interpolate parent (or use directly)
        if n_pts == pn:
            h0 = parent.copy()
        else:
            h0 = interp_periodic(parent, pn, n_pts)
            h0 = project_h(h0, ts)

        # Starting point 2: interpolate current best
        if best_n != n_pts:
            h0b = interp_periodic(best_h, best_n, n_pts)
            h0b = project_h(h0b, ts)
        else:
            h0b = best_h.copy()

        # Optimize from each start
        t_dl = time.time() + t_alloc
        h_opt, c5_opt = optimize_lifecycle(h0, n_pts, t_dl, start_time, heavy=(n_pts == pn))
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_pts

        # Second start point (if time remains)
        if time.time() < t_dl - 5 and n_pts == pn:
            t_dl2 = time.time() + t_alloc * 0.5
            h_opt2, c5_opt2 = optimize_lifecycle(h0b, n_pts, t_dl2, start_time, heavy=False)
            if c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2.copy(), c5_opt2, n_pts

    # Final intensive polish at best resolution
    if time.time() < budget_s - 15:
        t_final = budget_s - 8
        h_fin, c5_fin = optimize_lifecycle(best_h, best_n, t_final, start_time, heavy=True)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    return best_h, best_c5, best_n
