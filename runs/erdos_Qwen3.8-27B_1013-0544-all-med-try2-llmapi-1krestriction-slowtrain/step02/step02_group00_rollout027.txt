def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx), corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_max(h, n_points, active_lags):
    """Compute subgradient of max overlap w.r.t. h, using only active lags."""
    grad = np.zeros(n_points)
    for lag_idx in active_lags:
        lag = int(lag_idx) - (n_points - 1)
        if lag == 0:
            grad += (1.0 - h) - h  # = 1 - 2h
        elif lag > 0:
            if lag < n_points:
                grad[:n_points - lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n_points - lag]
        else:
            lag_abs = -lag
            if lag_abs < n_points:
                grad[lag_abs:] += (1.0 - h[:n_points - lag_abs])
                grad[:n_points - lag_abs] -= h[lag_abs:]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_max(h, n_points, active_lags):
    grad = np.zeros(n_points)
    for lag_idx in active_lags:
        lag = int(lag_idx) - (n_points - 1)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif 0 < lag < n_points:
            grad[:n_points - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n_points - lag]
        elif -n_points < lag < 0:
            m = -lag
            grad[m:] += (1.0 - h[:n_points - m])
            grad[:n_points - m] -= h[m:]
    return grad

            inits.append(project_h(1.0 - h_parent.copy(), target_sum))
        else:

            x_new = np.arange(n_points) * (2.0 / n_points)

            inits.append(project_h(np.interp(x_new, x_old, h_ext), target_sum))
