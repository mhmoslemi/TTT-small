            k_shift = int(lag_idx) - (n - 1)
            h = _subgrad_step(h, k_shift, 0.004)
        h += np.random.randn(n) * 0.004
        h = project_box_sum(h, ts)
        try:
            r = minimize(_obj_smooth, h, args=(n, ts, 1500, 3000.0),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 300, 'ftol': 1e-16})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 4: Powell (derivative-free, handles non-smooth well)
    if tl() > 120:
        h_start = best_h.copy()
        try:
            def _c5_direct(h):
                h_c = np.clip(h, 0.0, 1.0)
                c = np.correlate(h_c, 1.0 - h_c, mode="full")
                return np.max(c) * (2.0 / n) + 500.0 * (h_c.sum() - ts) ** 2
            r = minimize(_c5_direct, h_start, method='Powell',
                         options={'maxiter': 3000, 'xtol': 1e-10, 'ftol': 1e-14})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 5: High-precision polish
    for beta in [3000, 8000, 15000]:
        if tl() < 15:
            break
        try:
            r = minimize(_obj_smooth, best_h, args=(n, ts, beta, 8000.0),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 2000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 6: Tiny random perturbation search
    for it in range(200):
        if tl() < 5:
            break
        h = best_h + np.random.randn(n) * 0.0008
        h = project_box_sum(h, ts)
        c = compute_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c
            # Small local refinement
            try:
                r = minimize(_obj_smooth, h, args=(n, ts, 5000, 10000.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 200, 'ftol': 1e-17})
                h_new = project_box_sum(r.x, ts)
                c2 = compute_c5(h_new)
                if c2 < best_c5:
                    best_h, best_c5 = h_new, c2
            except Exception:
                pass

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
