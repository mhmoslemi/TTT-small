def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n, top_k=3):
    """Compute subgradient using top_k active lags, weighted by proximity to max."""
    C = np.correlate(h, 1.0 - h, mode="full")
    max_val = C.max()
    # Get top_k lags
    idx = np.argsort(C)[::-1][:top_k]
    # Weights: exponential of (C[i] - max_val) * scale
    weights = np.exp(50 * (C[idx] - max_val))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, idx):
        lag = k - (n - 1)
        if lag >= 0:
            grad[0:n-lag] += w * (1.0 - h[lag:n])
            grad[lag:n] -= w * h[0:n-lag]
        else:
            lag2 = -lag
            grad[lag2:n] += w * (1.0 - h[0:n-lag2])
            grad[0:n-lag2] -= w * h[lag2:n]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def corr_grad_single(h, n, k):
    """Gradient of C[k] = correlate(h,1-h,full)[k] w.r.t. h."""
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[0:n - lag] += (1.0 - h[lag:n])
        g[lag:n] -= h[0:n - lag]
    else:
        lag2 = -lag
        g[lag2:n] += (1.0 - h[0:n - lag2])
        g[0:n - lag2] -= h[lag2:n]
    return g


def weighted_subgrad(h, n, top_k=5, temp=50.0):
    """Subgradient using top_k active lags with softmax weighting."""
    C = np.correlate(h, 1.0 - h, mode="full")
    m = C.max()
    weights = np.exp(temp * (C - m))
    idx = np.argsort(weights)[::-1][:top_k]
    w = weights[idx]
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        g += wi * corr_grad_single(h, n, ki)
    return g

        elif s == 2:

        elif s == 4:

def sum_constraint_fun(x, target_sum):
    return np.sum(x) - target_sum


def sum_constraint_jac(x, n_points):
    return np.ones(n_points)


def slsqp_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx


def slsqp_grad(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    sw = np.sum(weights)
    g = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] / sw
        if w < 1e-14:
            continue
        g += w * corr_grad_single(h, n, k)
    return g * dx


def make_slsqp_obj(dx, beta):
    def obj(h):
        return slsqp_obj(h, dx, beta)
    return obj


def make_slsqp_grad(n, dx, beta):
    def grad(h):
        return slsqp_grad(h, n, dx, beta)
    return grad


def make_cons(target_sum, n_points):
    def cfun(x):
        return sum_constraint_fun(x, target_sum)
    def cjac(x):
        return sum_constraint_jac(x, n_points)
    return [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
