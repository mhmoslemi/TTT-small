    if len(active) == 0:
        active = np.argsort(corr)[::-1][:top_k]

    for j in range(n):
        slopes = []
        intercepts = []
        for li in active:
            d = int(li) - (n - 1)
            a = 0.0
            b = 0.0
            if 0 <= j + d < n:
                a += 1.0 - h_cur[j + d]
            if 0 <= j - d < n:
                a -= h_cur[j - d]
                b += h_cur[j - d]
            slopes.append(a)
            intercepts.append(b)
        h_cur[j] = opt_x1d(slopes, intercepts)

    return project_h(h_cur, target_sum)

def subgrad_dir(h, top_k):
    n = len(h)
    corr = get_corr(h)
    max_c = corr.max()
    near = np.where(corr >= max_c - 0.01)[0]
    if len(near) > top_k:
        order = np.argsort(corr[near])[::-1][:top_k]
        active = near[order]
    else:
        active = near

    grad = np.zeros(n)
    j = np.arange(n)
    for li in active:
        d = int(li) - (n - 1)
        jd = j + d
        jdm = j - d
        m1 = (jd >= 0) & (jd < n)
        m2 = (jdm >= 0) & (jdm < n)
        grad[m1] += 1.0 - h[jd[m1]]
        grad[m2] -= h[jdm[m2]]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    else:
        n0 = 140
        hp = 0.5 * np.ones(n0)

    ts = n0 / 2.0
    best_h = project_h(hp.copy(), ts)
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Phase 1: Coordinate descent (increasing top_k)
    h_cur = best_h.copy()
    prev_c5 = compute_c5(h_cur)
    for p in range(150):
        if tl() < 80:
            break
        tk = min(5 + p // 15, 20)
        h_cur = coord_pass(h_cur, ts, tk)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_h, best_c5 = h_cur.copy(), c5
        if abs(c5 - prev_c5) < 1e-12 and p > 20:
            break
        prev_c5 = c5

    # Phase 2: Subgradient with momentum and line search
    h_cur = best_h.copy()
    c5_cur = compute_c5(h_cur)
    step = 0.04
    grad_ema = np.zeros(n0)
    mom = 0.7
    stag = 0
    for it in range(3000):
        if tl() < 80:
            break
        g = subgrad_dir(h_cur, 5)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g /= gn
        grad_ema = mom * grad_ema + (1 - mom) * g
        ema_norm = np.linalg.norm(grad_ema)
        if ema_norm < 1e-10:
            break
        grad_ema /= ema_norm

        h_try = project_h(h_cur - step * grad_ema, ts)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-15:
            h_cur = h_try
            c5_cur = c5_try
            step = min(step * 1.03, 0.12)
            stag = 0
            if c5_cur < best_c5:
                best_h, best_c5 = h_cur.copy(), c5_cur
        else:
            step *= 0.6
            stag += 1
            if stag > 80:
                # Random perturbation to escape
                h_try = project_h(h_cur + 0.015 * np.random.randn(n0), ts)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur:
                    h_cur = h_try
                    c5_cur = c5_try
                    if c5_cur < best_c5:
                        best_h, best_c5 = h_cur.copy(), c5_cur
                else:
                    h_try = project_h(h_cur + 0.005 * np.random.randn(n0), ts)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur:
                        h_cur = h_try
                        c5_cur = c5_try
                step = 0.03
                stag = 0

    # Phase 3: More coordinate descent on best
    h_cur = best_h.copy()
    prev_c5 = compute_c5(h_cur)
    for p in range(100):
        if tl() < 60:
            break
        h_cur = coord_pass(h_cur, ts, 15)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_h, best_c5 = h_cur.copy(), c5
        if abs(c5 - prev_c5) < 1e-13:
            break
        prev_c5 = c5

    # Phase 4: Try nearby resolutions
    for n_try in [100, 120, 160, 200, 280]:
        if tl() < 60:
            break
        tsum = n_try / 2.0
        xb = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, n_try)
        h_try = project_h(np.interp(xn, xb, best_h), tsum)
        c5_try = compute_c5(h_try)
        if c5_try > best_c5 + 0.005:
            continue
        for p in range(50):
            if tl() < 40:
                break
            tk = min(8 + p // 10, 15)
            h_try = coord_pass(h_try, tsum, tk)
            c5_try = compute_c5(h_try)
        # Subgradient on this resolution
        step2 = 0.03
        stag2 = 0
        for it2 in range(500):
            if tl() < 40:
                break
            g2 = subgrad_dir(h_try, 5)
            gn2 = np.linalg.norm(g2)
            if gn2 < 1e-10:
                break
            g2 /= gn2
            h_t2 = project_h(h_try - step2 * g2, tsum)
            c5_t2 = compute_c5(h_t2)
            if c5_t2 < c5_try - 1e-14:
                h_try = h_t2
                c5_try = c5_t2
                step2 = min(step2 * 1.05, 0.1)
                stag2 = 0
            else:
                step2 *= 0.6
                stag2 += 1
                if stag2 > 60:
                    step2 = 0.02
                    stag2 = 0
        if c5_try < best_c5:
            best_h, best_c5, best_n = h_try.copy(), c5_try, n_try
            # Re-run coord descent on new best
            h_cur = best_h.copy()
            tf = best_n / 2.0
            for p in range(50):
                if tl() < 40:
                    break
                h_cur = coord_pass(h_cur, tf, 15)
                c5 = compute_c5(h_cur)
                if c5 < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5

    # Phase 5: Final fine polish
    if tl() > 10:
        n_f = best_n
        tf = n_f / 2.0
        h_cur = best_h.copy()
        prev_c5 = compute_c5(h_cur)
        for p in range(200):
            if tl() < 5:
                break
            h_cur = coord_pass(h_cur, tf, 25)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_h, best_c5 = h_cur.copy(), c5
            if abs(c5 - prev_c5) < 1e-14:
                break
            prev_c5 = c5

        # Tiny subgradient
        step = 0.003
        for it in range(1000):
            if tl() < 3:
                break
            g = subgrad_dir(best_h, 3)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            g /= gn
            h_try = project_h(best_h - step * g, tf)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_h = h_try
                best_c5 = c5_try
                step = min(step * 1.05, 0.01)
            else:
                step *= 0.7

    return best_h, best_c5, best_n
