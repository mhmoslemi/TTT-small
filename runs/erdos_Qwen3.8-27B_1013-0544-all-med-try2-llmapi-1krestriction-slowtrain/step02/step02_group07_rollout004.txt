def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        if abs(s - target_sum) < 1e-12:
            break

        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, lr):
    """One subgradient step on the max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    k = k_max - (n - 1)  # actual lag
    
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            # h[i+k] * (1-h[i])
            # d/dh[i+k] = (1 - h[i])
            grad[i+k] += (1 - h[i])
            # d/dh[i] = -h[i+k]
            grad[i] -= h[i+k]
    else:
        ak = -k
        for i in range(ak, n):
            # h[i] * (1-h[i-ak])
            # d/dh[i] = (1 - h[i-ak])
            grad[i] += (1 - h[i - ak])
            # d/dh[i-ak] = -h[i]
            grad[i - ak] -= h[i]
    
    # Projected gradient descent (minimize, so go against gradient)
    h_new = h - lr * grad
    return h_new

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)


def con_fun(x):
    return np.sum(x) - len(x) / 2.0


def con_jac(x):
    return np.ones(len(x))


def make_ramp(n, freq=1.0, phase=0.0):
    t = np.linspace(0, 1, n)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t + phase)
    return h


def make_blocks(n, n_blocks=5, width=0.6):
    h = np.zeros(n)
    seg = n // n_blocks
    for i in range(n_blocks):
        if i % 2 == 0:
            center = i * seg + seg // 2
            lo = max(0, center - int(seg * width // 2))
            hi = min(n, center + int(seg * width // 2) + 1)
            h[lo:hi] = 1.0
    return h


def make_staircase(n, n_levels=4):
    h = np.zeros(n)
    seg = n // n_levels
    for i in range(n_levels):
        h[i*seg:(i+1)*seg] = (i + 1) / n_levels
    return h


def top_k_objective(h, k=5, beta=50):
    corr = np.correlate(h, 1.0 - h, mode="full")
    sorted_c = np.sort(corr)[::-1]
    top = sorted_c[:k]
    return smooth_max(top, beta)


def sa_optimize(h, n, target_sum, t_start, n_iters, start_time, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_cur = h.copy()
    c5_cur = best_c5
    T = t_start
    cool_rate = (0.00005 / t_start) ** (1.0 / max(1, n_iters))
    no_improve = 0

    for it in range(n_iters):
        if time.time() > deadline:
            break

        h_new = h_cur.copy()
        r = np.random.random()

        if r < 0.35:
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            k_max = int(np.argmax(corr))
            d = k_max - (n - 1)
            n_adj = np.random.randint(3, max(4, n // 4))
            if d > 0 and d < n:
                idx = np.random.choice(n - d, min(n_adj, n - d), replace=False)
                for i in idx:
                    if np.random.random() < 0.5:
                        h_new[i + d] -= 0.04 + 0.08 * np.random.random()
                    else:
                        h_new[i] += 0.04 + 0.08 * np.random.random()
            elif d < 0 and -d < n:
                da = -d
                idx = np.random.choice(da, min(n_adj, da), replace=False)
                for i in idx:
                    if np.random.random() < 0.5:
                        h_new[i] -= 0.04 + 0.08 * np.random.random()
                    else:
                        h_new[i + da] += 0.04 + 0.08 * np.random.random()
            else:
                idx = np.random.choice(n, min(n_adj, n), replace=False)
                for i in idx:
                    h_new[i] += (0.5 - h_cur[i]) * 0.15
        elif r < 0.55:
            frac = 0.1 + 0.4 * (T / t_start)
            mask = np.random.random(n) < frac
            cnt = int(mask.sum())
            if cnt > 0:
                h_new[mask] += np.random.randn(cnt) * (0.02 + 0.08 * T / t_start)
        elif r < 0.70:
            blen = np.random.randint(2, max(3, n // 10))
            i = np.random.randint(0, n - blen + 1)
            j = np.random.randint(0, n - blen + 1)
            tries = 0
            while abs(j - i) < blen and tries < 30:
                j = np.random.randint(0, n - blen + 1)
                tries += 1
            if abs(j - i) >= blen:
                tmp = h_new[i:i+blen].copy()
                h_new[i:i+blen] = h_new[j:j+blen]
                h_new[j:j+blen] = tmp
        elif r < 0.85:
            n_swaps = np.random.randint(3, max(4, n // 8))
            idx = np.random.choice(n, n_swaps, replace=False)
            h_new[idx] += np.random.randn(n_swaps) * 0.06
        else:
            # Mirror symmetry exploitation
            half = n // 2
            src = np.random.randint(0, half)
            dst = n - 1 - src
            h_new[dst] = 1.0 - h_new[src] + np.random.randn() * 0.03

        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new)

        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-min(500.0, delta / max(T, 1e-12))):
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_h = h_cur.copy()
                best_c5 = c5_cur
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cool_rate
        if no_improve > 500:
            T = max(T, t_start * 0.3)
            no_improve = 0

    return best_h, best_c5


def slsqp_refine(h, n, target_sum, betas, start_time, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    for beta in betas:
        if time.time() > deadline:
            break

        def obj_func(x, b=beta):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(corr, b)

        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 1500, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5


def optimize_at_n(n, seed, start_time, deadline):
    np.random.seed(seed)
    target_sum = n / 2.0

    inits = []
    inits.append(np.full(n, 0.5))
    inits.append(make_ramp(n, 1.0, 0.0))
    inits.append(make_ramp(n, 1.0, np.pi / 2))
    inits.append(make_ramp(n, 2.0, 0.0))
    inits.append(make_ramp(n, 2.0, np.pi / 3))
    inits.append(make_ramp(n, 3.0, 0.0))
    inits.append(make_blocks(n, 4))
    inits.append(make_blocks(n, 6))
    inits.append(make_blocks(n, 8))
    inits.append(make_staircase(n, 4))
    inits.append(make_staircase(n, 6))

    best_h = None
    best_c5 = float('inf')

    for h0 in inits:
        h0 = project_h(h0, target_sum)
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_h = h0.copy()
            best_c5 = c5

    if best_h is None:
        best_h = project_h(np.full(n, 0.5), target_sum)
        best_c5 = compute_c5(best_h)

    # Phase 1: Coarse SLSQP
    sa_deadline = start_time + (deadline - start_time) * 0.55
    phase1_deadline = start_time + (sa_deadline - start_time) * 0.3
    best_h, best_c5 = slsqp_refine(best_h, n, target_sum, [30, 60, 100], start_time, phase1_deadline)

    # Phase 2: SA
    n_sa = 60000
    best_h, best_c5 = sa_optimize(best_h, n, target_sum, 0.015, n_sa, start_time, sa_deadline)

    # Phase 3: Fine SLSQP
    best_h, best_c5 = slsqp_refine(best_h, n, target_sum, [200, 400, 800, 1500], start_time, deadline)

    return best_h, best_c5, n


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
