    return grad

def interp_h(h_src, n_tgt):
    if len(h_src) == n_tgt:
        return h_src.copy()
    xo = np.linspace(0, 2, len(h_src), endpoint=False)
    xn = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_sum(np.interp(xn, xo, h_src, period=2.0), n_tgt / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = None

    def rem():
        return budget_s - 10 - (time.time() - t0)

    def build_starts(n, ts):
        starts = []
        if h_parent is not None:
            starts.append(interp_h(h_parent, n))
        if best_h is not None:
            starts.append(interp_h(best_h, n))
        h_alt = np.zeros(n); h_alt[::2] = 1.0
        starts.append(project_sum(h_alt, ts))
        h_blk = np.zeros(n)
        h_blk[:n//4] = 1.0; h_blk[3*n//4:] = 1.0
        starts.append(project_sum(h_blk, ts))
        h_rand = np.clip(0.5 + 0.4*np.random.randn(n), 0, 1)
        starts.append(project_sum(h_rand, ts))
        return starts

    def slsqp_optimize(h_c, n, ts, bounds, cons, beta, maxiter=500, ftol=1e-15):
        def obj(h, b=beta):
            return smooth_max(lags_array(h), float(b))
        try:
            res = minimize(obj, h_c, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            return project_sum(res.x, ts)
        except Exception:
            return None

    def subgradient_phase(h_init, n, ts, max_iter=300):
        h_sg = h_init.copy()
        v = np.zeros(n)
        c5_cur = compute_c5(h_sg)
        lr0 = 0.004
        for it in range(max_iter):
            if rem() < 12:
                break
            lr = lr0 / (1.0 + 0.05 * it)
            l = lags_array(h_sg)
            mx = np.max(l)
            active = np.where(l > mx - 0.003)[0]
            grad = np.zeros(n)
            for k in active:
                grad += grad_correlation(h_sg, k)
            grad /= len(active)
            v = 0.8 * v - lr * grad
            h_new = project_box_sum(h_sg + v, ts)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_cur + 1e-7:
                h_sg = h_new
                c5_cur = c5_new
        return h_sg, c5_cur

    n_list = [70, 72, 75, 80, 68]

    for n in n_list:
        if rem() < 50:
            break
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

        def cf(x):
            return np.sum(x) - ts
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]

        starts = build_starts(n, ts)
        curr_h, curr_c5 = None, float('inf')

        for h0 in starts:
            if rem() < 20:
                break
            h0 = project_sum(h0, ts)
            h_c = h0.copy()

            for beta in [20, 50, 100, 200, 400]:
                if rem() < 12:
                    break
                hn = slsqp_optimize(h_c, n, ts, bounds, cons, beta)
                if hn is not None:
                    c5 = compute_c5(hn)
                    if c5 < curr_c5:
                        curr_h, curr_c5 = hn.copy(), c5
                        h_c = hn.copy()

            if curr_h is not None and rem() > 45:
                h_sg, c5_sg = subgradient_phase(curr_h, n, ts, max_iter=250)
                if c5_sg < curr_c5:
                    curr_h, curr_c5 = h_sg.copy(), c5_sg
                    h_c = h_sg.copy()

            if curr_h is not None and rem() > 25:
                for pert in [0.015, 0.04, 0.008]:
                    if rem() < 10:
                        break
                    h_p = project_sum(curr_h + np.random.randn(n) * pert, ts)
                    hn = slsqp_optimize(h_p, n, ts, bounds, cons, 250, maxiter=300)
                    if hn is not None:
                        c5 = compute_c5(hn)
                        if c5 < curr_c5:
                            curr_h, curr_c5 = hn.copy(), c5
                            h_c = hn.copy()

        if curr_h is not None and curr_c5 < best_c5:
            best_h, best_c5, best_n = curr_h.copy(), curr_c5, n

    # Final polish
    if best_h is not None and rem() > 8:
        n = best_n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        def cf(x):
            return np.sum(x) - ts
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        h_c = best_h.copy()
        for beta in [500, 800, 1200]:
            if rem() < 4:
                break
            hn = slsqp_optimize(h_c, n, ts, bounds, cons, beta, maxiter=3000, ftol=1e-16)
            if hn is not None:
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_h, best_c5 = hn.copy(), c5
                    h_c = hn.copy()

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
