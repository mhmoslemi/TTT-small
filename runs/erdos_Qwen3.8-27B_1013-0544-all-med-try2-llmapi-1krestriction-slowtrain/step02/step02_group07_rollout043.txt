Let me try yet another approach: **交替优化 (alternating optimization)**.

If we fix all h[i] except h[j], the objective as a function of h[j] is:
C₅(t) = max_k [sum_{i≠j, i+d≠j} h[i](1-h[i+d]) + terms involving h[j]]

The terms involving h[j] are:
- For lag d where j is in [0, n-d-1]: h[j]*(1-h[j+d]) → linear in h[j]
- For lag d where j is in [d, n-1]: h[j-d]*(1-h[j]) → linear in h[j] (as -h[j-d]*h[j] + h[j-d])

So for each j, the objective is the max of affine functions of h[j]. This is a 1D convex problem! We can solve it exactly.

This gives us a **coordinate descent** algorithm:
1. For each j (or a random subset of j's):
   a. Compute the objective as a function of h[j] (with all other h fixed)
   b. Find the optimal h[j] that minimizes this 1D convex function
   c. Update h[j]

This is guaranteed to converge (though possibly slowly) and each step is O(n) to compute the 1D function and O(n) to find the minimum.

where a_k is the contribution to corr_k from all terms not involving h[j], and b_k is the coefficient of h[j] in corr_k.

The optimal h[j] is found by:

- This is a piecewise-linear convex function
- The minimum is at one of the breakpoints (where two affine functions are equal) or at 0 or 1

This is solvable in O(n) time per coordinate!

- B_k = coefficient of h[j] in corr_k

- If d <= j <= n-1: h[j] appears as h[j-d]*(1-h[j]), contributing -h[j-d] to B

- If d' <= j <= n-1: h[j] appears as h[j]*(1-h[j-d']), contributing (1-h[j-d']) to B
- If 0 <= j <= n-d'-1: h[j] appears as h[j+d']*(1-h[j]), contributing -h[j+d'] to B

So B_k = (1-h[j+d])*I(0<=j<=n-d-1) - h[j-d]*I(d<=j<=n-1) for d>0

And A_k = corr_k(h) - B_k * h[j] (where corr_k(h) is the current value)

So A_k is easily computed from the current correlations.

where x = h[j].

This is a 1D convex piecewise-linear minimization. The minimum occurs at:
- x = 0
- x = 1
- A breakpoint where A_k + B_k * x = A_{k'} + B_{k'} * x for some k, k'

The number of breakpoints is O(n²) in the worst case, but in practice much less.

def obj_slsqp(h, beta):

    return smooth_max(corr, beta) * (2.0 / len(h))

def con_eq(x):
    return np.sum(x) - len(x) / 2.0

def compute_B(h, j, n):
    B = np.zeros(2 * n - 1)
    mid = n - 1
    B[mid] = 1.0 - 2.0 * h[j]
    d_pos = np.arange(1, n)
    v1 = d_pos < (n - j)
    if np.any(v1):
        B[mid + d_pos[v1]] += (1.0 - h[j + d_pos[v1]])
    v2 = d_pos <= j
    if np.any(v2):
        B[mid + d_pos[v2]] -= h[j - d_pos[v2]]
    v3 = d_pos <= j
    if np.any(v3):
        B[mid - d_pos[v3]] += (1.0 - h[j - d_pos[v3]])
    v4 = d_pos < (n - j)
    if np.any(v4):
        B[mid - d_pos[v4]] -= h[j + d_pos[v4]]
    return B

def coord_descent(h, target_sum, n_sweeps=200, rng=None):

    h = h.copy()
    if rng is None:
        rng = np.random.default_rng()
    
    for sweep in range(n_sweeps):
        corr_all = np.correlate(h, 1.0 - h, mode="full")
        max_before = np.max(corr_all)
        
        # Random permutation for coordinate order
        order = rng.permutation(n)
        
        for j in order:
            j2 = (j + 1) % n
            if j2 == j:
                continue
            s = h[j] + h[j2]
            if s <= 0 or s >= 2:
                continue
            
            B_j = compute_B(h, j, n)
            B_j2 = compute_B(h, j2, n)
            
            A = corr_all - B_j * h[j] - B_j2 * h[j2]
            C = A + B_j2 * s
            D = B_j - B_j2
            
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            
            if hi - lo < 1e-15:
                continue
            
            lo_save, hi_save = lo, hi
            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = C + D * mid
                max_v = np.max(vals)
                active = np.flatnonzero(vals >= max_v - 1e-13)
                if len(active) == 0:
                    break
                d_min = np.min(D[active])
                d_max = np.max(D[active])
                if -1e-14 <= d_min and d_max <= 1e-14:
                    break
                elif d_min > 1e-14:
                    hi = mid
                elif d_max < -1e-14:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) * 0.5
            h[j] = x_new
            h[j2] = s - x_new
        
        # Project to maintain feasibility
        h = project_h(h, target_sum)
    
    return h

                if d_min <= 1e-14 and d_max >= -1e-14:
                    break
                elif d_min > 1e-14:
                    hi = mid
                elif d_max < -1e-14:
                    lo = mid
                else:
                    break
            x_new = (lo + hi) * 0.5
            delta = x_new - h[j]
            if abs(delta) > 1e-15:
                h[j] = x_new
                h[j2] = s - x_new
                corr_all += D * delta
    h = project_h(h, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n_points = len(h_init)
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - (time.time() - start_time)

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    constraints = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]

    # Phase 1: SLSQP with increasing smoothness
    for beta in [50, 100, 200, 400, 800]:
        if tleft() < 70:
            break
        try:
            res = minimize(obj_slsqp, best_h, args=(beta,),
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Coordinate descent polish
    if tleft() > 80:
        h_cd = coord_descent(best_h, target_sum, n_sweeps=400, rng=rng)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h, best_c5 = h_cd.copy(), c5

    # Phase 3: Active-set SLSQP (focus on critical lags)
    if tleft() > 60:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_val = np.max(corr)
        active_lags = np.flatnonzero(corr > max_val - 1e-5)
        if len(active_lags) > 0 and len(active_lags) < 15:
            try:
                res = minimize(lambda h: obj_active(h, active_lags), best_h,
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass
        # SLSQP with very high beta
        for beta in [1500, 3000]:
            if tleft() < 40:
                break
            try:
                res = minimize(obj_slsqp, best_h, args=(beta,),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 4000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 4: More coordinate descent
    if tleft() > 60:
        h_cd = coord_descent(best_h, target_sum, n_sweeps=400, rng=rng)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h, best_c5 = h_cd.copy(), c5

    # Phase 5: Targeted simulated annealing
    if tleft() > 120:
        h_cur = best_h.copy()
        c5_cur = best_c5
        T = 0.004
        T_min = 0.00005
        cooling = 0.9992
        while tleft() > 40:
            T = max(T, T_min)
            h_new = h_cur.copy()
            move = rng.random()
            if move < 0.25:
                n_pts = int(rng.integers(2, 12))
                idx = rng.choice(n_points, n_pts, replace=False)
                h_new[idx] += rng.standard_normal(n_pts) * (0.01 + 0.06 * T / 0.004)
            elif move < 0.55:
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                k_max = int(np.argmax(corr))
                d = k_max - (n_points - 1)
                n_adj = int(rng.integers(2, 7))
                if d > 0:
                    idx = rng.choice(n_points - d, min(n_adj, n_points - d), replace=False)
                    for i in idx:
                        if h_cur[i] > 0.5:
                            h_new[i] -= 0.02 + 0.04 * rng.random()
                        else:
                            h_new[i + d] -= 0.02 + 0.04 * rng.random()
                elif d < 0:
                    da = -d
                    idx = rng.choice(da, min(n_adj, da), replace=False)
                    for i in idx:
                        if h_cur[i + da] > 0.5:
                            h_new[i + da] -= 0.02 + 0.04 * rng.random()
                        else:
                            h_new[i] -= 0.02 + 0.04 * rng.random()
                else:
                    idx = rng.choice(n_points, n_adj, replace=False)
                    for i in idx:
                        h_new[i] += (0.5 - h_cur[i]) * 0.08
            elif move < 0.8:
                i, j = rng.choice(n_points, 2, replace=False)
                delta = 0.01 + 0.04 * rng.random()
                direction = 1.0 if h_cur[i] > h_cur[j] else -1.0
                h_new[i] += delta * direction
                h_new[j] -= delta * direction
            else:
                blen = int(rng.integers(2, 6))
                i = int(rng.integers(0, n_points - blen + 1))
                j = int(rng.integers(0, n_points - blen + 1))
                tries = 0
                while abs(j - i) < blen and tries < 30:
                    j = int(rng.integers(0, n_points - blen + 1))
                    tries += 1
                if abs(j - i) >= blen:
                    tmp = h_new[i:i+blen].copy()
                    h_new[i:i+blen] = h_new[j:j+blen]
                    h_new[j:j+blen] = tmp
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta_c = c5_new - c5_cur
            if delta_c < 0 or rng.random() < np.exp(-min(400.0, delta_c / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5_cur
            T *= cooling
        # CD after SA
        if tleft() > 30:
            h_cd = coord_descent(best_h, target_sum, n_sweeps=300, rng=rng)
            c5 = compute_c5(h_cd)
            if c5 < best_c5:
                best_h, best_c5 = h_cd.copy(), c5

    # Phase 6: Final active-set + CD polish
    if tleft() > 40:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_val = np.max(corr)
        active_lags = np.flatnonzero(corr > max_val - 2e-5)
        if len(active_lags) > 0 and len(active_lags) < 20:
            try:
                res = minimize(lambda h: obj_active(h, active_lags), best_h,
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-17})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass
    if tleft() > 20:
        h_cd = coord_descent(best_h, target_sum, n_sweeps=300, rng=rng)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h, best_c5 = h_cd.copy(), c5

    # Final verification
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
