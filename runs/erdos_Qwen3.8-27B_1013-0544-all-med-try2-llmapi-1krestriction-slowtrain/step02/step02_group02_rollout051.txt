def _fourier_eval(coeffs, n, K, basis):
    h = 0.5 + basis.dot(coeffs)
    h = project_feasible(h, n)
    return compute_c5(h, n), h

def de_search(n, K, pop_size, gens, F, CR, t0, budget, rng):
    basis = _precompute_basis(n, K)
    np_ = 2 * K
    best_c5 = 1.0
    best_h = None
    population = rng.uniform(-0.35, 0.35, size=(pop_size, np_))
    fitness = np.zeros(pop_size)
    for i in range(pop_size):
        fitness[i], h_i = _fourier_eval(population[i], n, K, basis)
        if fitness[i] < best_c5:
            best_c5 = fitness[i]
            best_h = h_i.copy()
    for gen in range(gens):
        if budget - (time.time() - t0) < 60:
            break
        for i in range(pop_size):
            if budget - (time.time() - t0) < 60:
                break
            others = [x for x in range(pop_size) if x != i]
            a, b, c = rng.choice(others, 3, replace=False)
            v = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            j0 = rng.randint(np_)
            for j in range(np_):
                if rng.random() < CR or j == j0:
                    trial[j] = v[j]
            fit_t, h_t = _fourier_eval(trial, n, K, basis)
            if fit_t <= fitness[i]:
                population[i] = trial
                fitness[i] = fit_t
                if fit_t < best_c5:
                    best_c5 = fit_t
                    best_h = h_t.copy()
    return best_h, best_c5

def sa_optimize(h, n, t0, budget, rng, n_iter, e_start, e_min, d_start, d_min):
    E = e_start
    D = d_start
    e_ratio = (e_min / e_start) ** (1.0 / n_iter)
    d_ratio = (d_min / d_start) ** (1.0 / n_iter)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    curr_c5 = best_c5
    last_polish = time.time()
    for it in range(n_iter):
        if budget - (time.time() - t0) < 30:
            break
        if time.time() - last_polish > 12.0:
            last_polish = time.time()
            h = best_h.copy()
            curr_c5 = best_c5
            ht = slsqp(h, n, 20000, 400, 1e-14)
            c = compute_c5(ht, n)
            if c < best_c5:
                best_h = ht.copy()
                best_c5 = c
                h = ht.copy()
                curr_c5 = c
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.01, D) * rng.choice([1.0, -1.0])
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - curr_c5
        if delta < 0 or (delta < 3.0 * E and rng.random() < np.exp(-delta / max(E, 1e-10))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = hi - d
            h[j] = hj + d
        E *= e_ratio
        D *= d_ratio
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    except Exception:
        n = 80
        best_h = np.full(n, 0.5)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation
    for alpha in [300, 1000, 5000, 20000]:
        if remaining() < 10:
            break
        ht = slsqp(best_h, n, alpha, 3000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 2: Golden-section coordinate descent (key improvement)
    if remaining() > 60:
        best_h, best_c5 = coord_descent(best_h, n, t0, budget_s, n_passes=5, iters=22)

    # Phase 3: SLSQP after coordinate descent
    for alpha in [50000, 200000]:
        if remaining() < 10:
            break
        ht = slsqp(best_h, n, alpha, 4000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 4: DE global search on Fourier coefficients
    if remaining() > 150:
        for K in [8, 10]:
            if remaining() < 60:
                break
            de_h, de_c5 = de_search(n, K, 35, 150, 0.8, 0.8, t0, budget_s, rng)
            if de_h is not None and de_c5 < best_c5:
                best_h, best_c5 = de_h, de_c5
                # Polish after DE
                for a in [5000, 50000]:
                    if remaining() < 10:
                        break
                    ht = slsqp(best_h, n, a, 2000, 1e-16)
                    c = compute_c5(ht, n)
                    if c < best_c5 - 1e-15:
                        best_h, best_c5 = ht, c
                best_h, best_c5 = coord_descent(best_h, n, t0, budget_s, n_passes=3, iters=20)

    # Phase 5: Simulated annealing
    if remaining() > 100:
        best_h, best_c5 = sa_optimize(best_h, n, t0, budget_s, rng,
                                      200000, 0.03, 1e-6, 0.2, 0.005)

    # Phase 6: Coordinate descent again
    if remaining() > 50:
        best_h, best_c5 = coord_descent(best_h, n, t0, budget_s, n_passes=4, iters=25)

    # Phase 7: Random restarts
    while remaining() > 80:
        scale = rng.choice([0.02, 0.05, 0.08, 0.12, 0.18])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        improved = False
        for a in [5000, 50000]:
            if remaining() < 15:
                break
            ht = slsqp(h_pert, n, a, 1500, 1e-14)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = ht, c
                improved = True
                break
        if improved:
            best_h, best_c5 = coord_descent(best_h, n, t0, budget_s, n_passes=2, iters=18)

    # Phase 8: Final polish
    for alpha in [200000, 1000000, 5000000]:
        if remaining() < 10:
            break
        ht = slsqp(best_h, n, alpha, 6000, 1e-20)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    if remaining() > 20:
        best_h, best_c5 = coord_descent(best_h, n, t0, budget_s, n_passes=2, iters=30)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
