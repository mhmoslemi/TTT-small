def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:

- sum h[i] = n/2
- f_k(h) ≤ t for all k

The constraints f_k(h) ≤ t are quadratic (non-convex). But if we linearize around the current point, we get a convex QP (or even LP if we drop the quadratic terms in the linearization).

**Linearized approach**:
At the current point h*, linearize f_k(h) ≈ f_k(h*) + ∇f_k(h*)·(h - h*)
Then the problem becomes:
min t
s.t. f_k(h*) + ∇f_k(h*)·(h - h*) ≤ t for all k
     0 ≤ h[i] ≤ 1
     sum h[i] = n/2

This is a linear program! We can solve it with any LP solver (CBC, GLPK, etc.).

The gradient of f_k with respect to h:
f_k(h) = sum_i h[i+k](1-h[i]) (for lag k > 0)
∂f_k/∂h[j] = (1-h[j-k]) if j is in the range where it appears as h[i+k]
             - h[j+k] if j is in the range where it appears as h[i]

This is exactly what the subgradient computation gives!

So the algorithm is:
1. Start with h*
2. Compute all f_k(h*) and their gradients
3. Solve the LP: min t s.t. f_k(h*) + ∇f_k·(h-h*) ≤ t, 0≤h≤1, sum h = n/2
4. Get new h from the LP solution
5. Repeat

This is the "cutting plane" or "outer approximation" method for minimax problems!

- It handles all constraints simultaneously
- It uses the LP solver to find the best direction
- It converges (in principle) to the optimum

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def subgrad_topk(h, top_k_indices, n):
    grad = np.zeros(n)
    for k_idx in top_k_indices:
        lag = k_idx - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx + la])
            grad[idx] -= h[idx + la]
    return grad

def con_sum_fun(h, ts):
    return h.sum() - ts

def con_sum_jac(h, n):
    return np.ones(n)

def make_slsqp_cons(ts, n):
    return [{'type': 'eq', 'fun': con_sum_fun, 'jac': con_sum_jac, 'args': (ts, n)}]

def optimize_slsqp(h, n, dx, ts, betas, t0, budget_s, max_iter_each=2000):
    h = project_feasible(h.copy(), ts)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = make_slsqp_cons(ts, n)
    for beta in betas:
        remaining = budget_s - (time.time() - t0)
        if remaining < 20:
            break
        for trial in range(2):
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            x0 = h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.008
                x0 = project_feasible(x0, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter_each, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                if compute_c5(h_new) < compute_c5(h):
                    h = h_new
            except Exception:
                pass
    return h

def fista_subgrad(h, n, ts, t0, budget_s, max_iter=50000):
    h = h.copy()
    h_y = h.copy()
    momentum = 0.9
    lr = 0.005
    for it in range(max_iter):
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode='full')
        top_k = np.argsort(corr)[-3:]
        grad = subgrad_topk(h_y, top_k, n)
        h_new = h_y - lr * grad
        h_new = project_feasible(h_new, ts)
        h_y = h_new + momentum * (h_new - h_y)
        h_y = project_feasible(h_y, ts)
        h = h_y.copy()
        if it % 10000 == 9999:
            lr = lr * 0.7
    return h

def make_starts(n, parent_h, rng):
    starts = []
    starts.append(resample_h(parent_h, n))
    starts.append(np.full(n, 0.5))
    x = np.arange(n)
    for freq in [1, 2, 3]:
        starts.append(0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n))
    starts.append(0.5 + 0.35 * np.cos(2 * np.pi * 2 * x / n))
    r = rng.random(n)
    starts.append(r)
    return starts

    candidates.sort(key=lambda x: x[0])

    # Phase 2: Full optimization on top candidates
    max_cands = min(5, len(candidates))
    for ci in range(max_cands):

        c5_cand, n_cand, h_cand = candidates[ci]
