import numpy as np

import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break

        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    B = base - slopes * h[i]
    val_0 = np.max(B)
    val_1 = np.max(B + slopes)
    best_val = min(val_0, val_1)
    best_t = 0.0 if val_0 <= val_1 else 1.0
    for a in range(num_lags):
        for b in range(a + 1, num_lags):
            ds = slopes[a] - slopes[b]
            if abs(ds) < 1e-15:
                continue
            t = (B[b] - B[a]) / ds
            if 0.0 <= t <= 1.0:
                val = B[a] + slopes[a] * t
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
    return best_t


def coord_descent(h, n, target_sum, max_sweeps=100):
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        h_work = h.copy()
        max_change = 0.0
        for i in range(n):
            t_opt = coord_optimal(h_work, i, n)
            max_change = max(max_change, abs(t_opt - h_work[i]))
            h_work[i] = t_opt
        h_work = project_feasible(h_work, target_sum)
        c5 = compute_c5(h_work)
        if c5 <= best_c5 + 1e-12:
            h = h_work.copy()
            if c5 < best_c5:
                best_c5 = c5
        if max_change < 1e-12:
            break
    return h, best_c5


def lp_step(h, n, target_sum):
    dx = 2.0 / n
    num_lags = 2 * n - 1
    corrs = np.correlate(h, 1.0 - h, mode='full')
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((num_lags, n + 1))
    b_ub = np.zeros(num_lags)
    for m in range(num_lags):
        lag = m - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] = g
        A_ub[m, :n] = dx * grad
        A_ub[m, n] = -1.0
        b_ub[m] = dx * (corrs[m] - np.dot(grad, h))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h.copy()


def lp_iterate(h, n, target_sum, max_iters=500):
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        h_new = lp_step(h, n, target_sum)
        h_new = project_feasible(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            h = h_new.copy()
            best_c5 = c5
        elif abs(c5 - best_c5) < 1e-12 and np.max(np.abs(h_new - h)) < 1e-10:
            break
        else:
            h = h_new.copy()
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target_sum = n / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    best_n = n

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent
    if tl() > 10:
        best_h, best_c5 = coord_descent(best_h, n, target_sum, max_sweeps=200)

    # Phase 2: LP linearization
    if tl() > 60:
        best_h, best_c5 = lp_iterate(best_h, n, target_sum, max_iters=300)
        best_h, best_c5 = coord_descent(best_h, n, target_sum, max_sweeps=50)

    # Phase 3: SLSQP with high beta
    if tl() > 80:
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [2000, 10000, 50000]:
            if tl() < 30:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n
            except Exception:
                pass
        best_h, best_c5 = coord_descent(best_h, best_n, best_n / 2.0, max_sweeps=50)
        best_h, best_c5 = lp_iterate(best_h, best_n, best_n / 2.0, max_iters=200)

    # Phase 4: Try different resolutions
    if tl() > 150:
        for n_try in [n, n + 10, n - 10, n + 20, 50, 90, 110, 140]:
            if n_try < 20 or tl() < 50:
                continue
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            h_t, c5_t = coord_descent(h_t, n_try, ts_t, max_sweeps=50)
            h_t, c5_t = lp_iterate(h_t, n_try, ts_t, max_iters=100)
            if tl() > 30:
                dx_t = 2.0 / n_try
                bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
                cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, 5000), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_t:
                        h_t = h_new
                        c5_t = c5_new
                    h_t, c5_t = coord_descent(h_t, n_try, ts_t, max_sweeps=30)
                    h_t, c5_t = lp_iterate(h_t, n_try, ts_t, max_iters=100)
                except Exception:
                    pass
            if c5_t < best_c5:
                best_h = h_t.copy()
                best_c5 = c5_t
                best_n = n_try
                target_sum = ts_t

    # Phase 5: Random perturbations + local optimization
    if tl() > 60:
        n = best_n
        target_sum = n / 2.0
        for attempt in range(30):
            if tl() < 15:
                break
            scale = 0.04 / (1.0 + attempt * 0.3)
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = project_feasible(h_pert, target_sum)
            h_pert, c5_pert = coord_descent(h_pert, n, target_sum, max_sweeps=40)
            h_pert, c5_pert = lp_iterate(h_pert, n, target_sum, max_iters=50)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert

    # Phase 6: Final polish
    if tl() > 15:
        n = best_n
        target_sum = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [100000, 200000]:
            if tl() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
        best_h, best_c5 = coord_descent(best_h, best_n, best_n / 2.0, max_sweeps=100)
        best_h, best_c5 = lp_iterate(best_h, best_n, best_n / 2.0, max_iters=200)

    return best_h, best_c5, len(best_h)
