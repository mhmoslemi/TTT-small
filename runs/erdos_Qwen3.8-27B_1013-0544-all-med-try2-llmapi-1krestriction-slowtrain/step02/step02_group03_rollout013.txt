def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def con_fun(x, target_sum):
    return x.sum() - target_sum


def con_jac(x, target_sum):
    return np.ones(len(x))


def obj_smax(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, beta) * dx


def subgrad_at(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation with increasing smoothness
    for beta in [5, 20, 100, 500, 2000, 10000]:
        if budget_s - (time.time() - t0) < 20:
            break
        try:
            res = minimize(obj_smax, best_h, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Phase 2: Subgradient with Polyak adaptive step size
    if budget_s - (time.time() - t0) > 80:
        h_curr = best_h.copy()
        for it in range(150000):
            if budget_s - (time.time() - t0) < 30:
                break
            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
            k_w = int(np.argmax(corr))
            grad = subgrad_at(h_curr, k_w, n)
            c5_curr = np.max(corr) * dx
            gap = max(c5_curr - 0.28, 1e-6)
            gnorm2 = np.dot(grad, grad)
            lr = min(gap * gap / (gnorm2 + 1e-10), 0.3)
            h_curr = h_curr - lr * grad
            h_curr = project_feasible(h_curr, target_sum)
            if it % 3000 == 2999:
                c5 = compute_c5(h_curr)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()

    # Phase 3: Pair coordinate descent (transfer moves preserving sum exactly)
    if budget_s - (time.time() - t0) > 60:
        h_local = best_h.copy()
        stagnant = 0
        while stagnant < 600 and budget_s - (time.time() - t0) > 30:
            i, j = np.random.choice(n, 2, replace=False)
            found = False
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                vi = h_local[i] + delta
                vj = h_local[j] - delta
                if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
                    h_try = h_local.copy()
                    h_try[i] = vi
                    h_try[j] = vj
                    c5 = compute_c5(h_try)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        best_h = h_try.copy()
                        h_local = h_try.copy()
                        stagnant = 0
                        found = True
                        break
            if not found:
                stagnant += 1

    # Phase 4: SLSQP high-beta polish
    for beta in [10000, 50000, 200000]:
        if budget_s - (time.time() - t0) < 15:
            break
        try:
            res = minimize(obj_smax, best_h, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Phase 5: Multi-resolution exploration
    n_cands = []
    for delta in [10, 20, 30, 40, 50, 60]:
        for sign in [1, -1]:
            nc = n + sign * delta
            if nc >= 40 and nc % 2 == 0:
                n_cands.append(nc)
    n_cands = sorted(set(n_cands))

    for nc in n_cands:
        if budget_s - (time.time() - t0) < 25:
            break
        h_t = resample_h(best_h, nc)
        h_t = project_feasible(h_t, nc / 2.0)
        dx_t = 2.0 / nc
        ts_t = nc / 2.0
        bn_t = Bounds(np.zeros(nc), np.ones(nc))
        cn_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
        for beta in [2000, 10000]:
            if budget_s - (time.time() - t0) < 12:
                break
            try:
                res = minimize(obj_smax, h_t, args=(dx_t, beta), method='SLSQP',
                               bounds=bn_t, constraints=cn_t,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts_t)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                    h_t = h_new.copy()
            except Exception:
                pass

    # Phase 6: Final subgradient + SLSQP at best resolution
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]

    if budget_s - (time.time() - t0) > 30:
        h_curr = best_h.copy()
        for it in range(50000):
            if budget_s - (time.time() - t0) < 15:
                break
            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
            k_w = int(np.argmax(corr))
            grad = subgrad_at(h_curr, k_w, n)
            c5_curr = np.max(corr) * dx
            gap = max(c5_curr - 0.28, 1e-6)
            gnorm2 = np.dot(grad, grad)
            lr = min(gap * gap / (gnorm2 + 1e-10), 0.2)
            h_curr = h_curr - lr * grad
            h_curr = project_feasible(h_curr, target_sum)
            if it % 5000 == 4999:
                c5 = compute_c5(h_curr)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()

    # Phase 7: Final SLSQP squeeze
    for beta in [100000, 500000]:
        if budget_s - (time.time() - t0) < 8:
            break
        try:
            res = minimize(obj_smax, best_h, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
