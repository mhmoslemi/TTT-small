    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # For each possible lag
    for lag in range(-(n-1), n):
        # Get valid indices
        if lag >= 0:
            idx = np.arange(0, n - lag)
        else:
            idx = np.arange(-lag, n)
        
        # h[idx] and h[idx + lag]
        h_a = h[idx]
        h_b = h[idx + lag]
        
        # Linearized: sum (h_a - h0_a * h_b - h_a * h0_b + h0_a * h0_b)
        h0_a = h0[idx]
        h0_b = h0[idx + lag]
        
        lin_sum = cp.sum(h_a) - cp.sum(h0_a * h_b) - cp.sum(h_a * h0_b) + np.sum(h0_a * h0_b)
        constraints.append(lin_sum * dx <= t)
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    problem.solve(solver=solver)
    
    if h.value is not None:

import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve(h0, n, target_sum, solver=None):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    for lag in range(-(n-1), n):
        if lag >= 0:
            idx = np.arange(0, n - lag)
        else:
            idx = np.arange(-lag, n)
        if len(idx) == 0:
            continue
        h0_a = h0[idx]
        h0_b = h0[idx + lag]
        lin_sum = cp.sum(h[idx]) - cp.sum(h0_a * h[idx + lag]) - cp.sum(h[idx] * h0_b) + np.sum(h0_a * h0_b)
        constraints.append(lin_sum * dx <= t)
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    if solver is None:
        for s in ['CLARABEL', 'ECOS', 'SCS', 'GLPK', 'CBC']:
            try:
                problem.solve(solver=s)
                if h.value is not None:
                    return np.array(h.value).flatten()
            except Exception:
                continue
    else:
        problem.solve(solver=solver)
    
    if h.value is not None:
        return np.array(h.value).flatten()
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    def tl():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Successive Linear Programming
    h = best_h.copy()
    for iteration in range(200):
        if tl() < 60:
            break
        h_new = lp_solve(h, n, target_sum)
        h_new = project_feasible(h_new, target_sum)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h = h_new.copy()
            best_c5 = c
        # Damping
        alpha = 1.0 if c < best_c5 else 0.5
        h = alpha * h_new + (1 - alpha) * best_h
        h = project_feasible(h, target_sum)
        # Check convergence
        if np.max(np.abs(h - h_new)) < 1e-8:
            break
    
    # Phase 2: SLSQP polish
    def obj_smooth(h, dx, beta):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, beta) * dx

from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lp_solve(h0, n, target_sum):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    for lag in range(-(n - 1), n):
        if lag >= 0:
            idx = np.arange(0, n - lag)
        else:
            idx = np.arange(-lag, n)
        if len(idx) == 0:
            continue
        h0_a = h0[idx]
        h0_b = h0[idx + lag]
        lin = cp.sum(h[idx]) - cp.sum(h0_a * h[idx + lag]) - cp.sum(h[idx] * h0_b) + np.sum(h0_a * h0_b)
        constraints.append(lin * dx <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in ['CLARABEL', 'ECOS', 'SCS', 'GLPK']:
        try:
            prob.solve(solver=solver)
            if h.value is not None:
                return np.array(h.value).flatten()
        except Exception:
            continue
    return h0.copy()

    corr = np.correlate(h, 1.0 - h, mode='full')

def con_fun(h, ts, n):
    return h.sum() - ts


def con_jac(h, ts, n):
    return np.ones(n)

            break

                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target_sum)
                c = compute_c5(h_new)
                if c < best_c:
                    best_h = h_new.copy()
                    best_c = c
            except Exception:
                pass
    return best_h, best_c
