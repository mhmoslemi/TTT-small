```python
def subgrad_multi(h, corr, n, top_m=5):
    """Compute a weighted subgradient from the top m lags."""
    dx = 2.0 / n
    # Get the top m lags
    top_idx = np.argsort(corr)[::-1][:top_m]
    # Weight by correlation value (higher = more weight)
    weights = np.exp((corr[top_idx] - corr[top_idx[0]]) * 10)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += w * (1.0 - h[idx + lag])
            grad[idx + lag] -= w * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += w * (1.0 - h[idx])
            grad[idx] -= w * h[idx + la]
    return grad
```

```python
    # O_0(1) = dx * (1*0 + sum_{i≠p} h[i](1-h[i])) = dx * sum_{i≠p} h[i](1-h[i])
    # So O_0(0) = O_0(1)! The parabola has its vertex at x=0.5.
    # O_0(0.5) = dx * (0.25 + sum_{i≠p} h[i](1-h[i]))
    
    # So for d=0: a_0 = corr1[d0] - corr0[d0] = 0 (they're equal)
    # The quadratic term: O_0(x) = O_0(0) + dx*(x - x²)
    
    d0_idx = n - 1  # index of lag 0 in the full correlation
    
    # Candidate points:
    candidates = set()
    candidates.add(0.0)
    candidates.add(1.0)
    candidates.add(0.5)  # vertex of O_0
    
    # Intersections of O_0 with each O_d (d≠0):
    # O_0(x) = O_0(0) + dx*(x-x²) = c0 + dx*x - dx*x²
    # O_d(x) = corr0[d] + (corr1[d] - corr0[d])*x
    # Setting equal: c0 + dx*x - dx*x² = corr0[d] + (corr1[d]-corr0[d])*x
    # -dx*x² + (dx - (corr1[d]-corr0[d]))*x + (c0 - corr0[d]) = 0
    # dx*x² - (dx - (corr1[d]-corr0[d]))*x + (corr0[d] - c0) = 0
    
    c0 = corr0[d0_idx]
    for d_idx in range(len(corr0)):
        if d_idx == d0_idx:
            continue
        a = dx
        b = -(dx - (corr1[d_idx] - corr0[d_idx]))
        c = corr0[d_idx] - c0
        disc = b*b - 4*a*c
        if disc >= 0:
            x1 = (-b + np.sqrt(disc)) / (2*a)
            x2 = (-b - np.sqrt(disc)) / (2*a)
            if 0 <= x1 <= 1:
                candidates.add(x1)
            if 0 <= x2 <= 1:
                candidates.add(x2)
    
    # Intersections of O_d with O_d' (d≠d', both ≠0):
    # O_d(x) = O_d'(x)
    # corr0[d] + (corr1[d]-corr0[d])*x = corr0[d'] + (corr1[d']-corr0[d'])*x
    # (corr1[d]-corr0[d] - corr1[d']+corr0[d'])*x = corr0[d'] - corr0[d]
    
    # This gives O(n²) pairs, which is ~19600 for n=70 (138 lags squared).
    # That's a lot. Let me skip this for now and just use the O_0 intersections + boundaries.
    # Actually, the minimum of the max of linear functions is at a vertex of the upper envelope,
    # which is at an intersection of two lines or at a boundary. But with 138 lines, there
    # could be many such intersections. Let me just sample.
    
    # Add some random samples
    for _ in range(20):
        candidates.add(np.random.uniform(0, 1))

    best_x = h[p]
    best_val = np.inf
    for x in candidates:

        val = eval_max_overlap_at(h, p, x)
```
