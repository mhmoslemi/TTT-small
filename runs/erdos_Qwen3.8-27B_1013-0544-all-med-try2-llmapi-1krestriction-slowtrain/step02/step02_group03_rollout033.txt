    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_eq(h, target):
    return h.sum() - target

def con_eq_jac(h, target):
    return np.ones(len(h))

def binary_local_search(b, n, max_passes=500, time_limit=None, t_start=None):
    """Local search on binary sequence: swap 1s and 0s to reduce max correlation."""
    best_c5 = compute_c5(b.astype(float))
    ones = np.where(b == 1)[0].tolist()
    zeros = np.where(b == 0)[0].tolist()
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passages = 0
        for i in range(len(ones)):
            for j in range(len(zeros)):
                if time_limit is not None and (time.time() - t_start) > time_limit:
                    return b.copy(), best_c5
                a, z = ones[i], zeros[j]
                if a == z:
                    continue
                b[a], b[z] = 0, 1
                c5 = compute_c5(b.astype(float))
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    ones[i] = z
                    zeros[j] = a
                    improved = True
                else:
                    b[a], b[z] = 1, 0
                passages += 1
        passes += 1
    return b.copy(), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    best_h = project_feasible(h_init.copy(), n0 / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at native resolution
    n = n0
    target = n / 2.0
    dx = 2.0 / n
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target,)}]
    for beta in [10, 50, 200, 1000, 5000]:
        if budget_s - (time.time() - t0) < 25:
            break
        try:
            res = minimize(obj_fn, best_h, args=(dx, beta), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Binary projection + local search
    if budget_s - (time.time() - t0) > 200:
        b = (best_h > 0.5).astype(int)
        # Ensure exactly n/2 ones
        n_ones = b.sum()
        if n_ones > n / 2:
            # Remove excess ones (those closest to 0.5)
            one_idx = np.where(b == 1)[0]
            order = np.argsort(best_h[one_idx])
            for k in range(n_ones - n // 2):
                b[one_idx[order[k]]] = 0
        elif n_ones < n / 2:
            zero_idx = np.where(b == 0)[0]
            order = np.argsort(-best_h[zero_idx])
            for k in range(n // 2 - n_ones):
                b[zero_idx[order[k]]] = 1
        b_ls, c5_bin = binary_local_search(b.copy(), n, max_passes=200,
                                           time_limit=60, t_start=t0)
        h_bin = b_ls.astype(float)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin.copy(), c5_bin
        # SLSQP from binary starting point
        if budget_s - (time.time() - t0) > 60:
            for beta in [100, 1000, 10000]:
                if budget_s - (time.time() - t0) < 15:
                    break
                try:
                    res = minimize(obj_fn, h_bin, args=(dx, beta), method='SLSQP',
                                   bounds=bnds, constraints=cons,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    # Phase 3: Multi-resolution sweep
    if budget_s - (time.time() - t0) > 200:
        n_best = len(best_h)
        tried = set([n_best])
        # Generate candidate resolutions sorted by distance
        candidates = []
        for d in range(1, 80):
            for sign in [1, -1]:
                nt = n_best + sign * d
                if 20 <= nt <= 160 and nt not in tried:
                    candidates.append((abs(d), nt))
        candidates.sort()
        for _, n_try in candidates[:25]:
            if budget_s - (time.time() - t0) < 30:
                break
            dx_t = 2.0 / n_try
            target_t = n_try / 2.0
            h_t = project_feasible(resample_h(best_h, n_try), target_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_t,)}]
            for beta in [50, 500, 5000]:
                if budget_s - (time.time() - t0) < 12:
                    break
                try:
                    res = minimize(obj_fn, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 2000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    # Phase 4: Simulated annealing on continuous h
    if budget_s - (time.time() - t0) > 150:
        h = best_h.copy()
        n_sa = len(h)
        target_sa = n_sa / 2.0
        current_c5 = compute_c5(h)
        T_init = 0.003
        T_ratio = 0.005
        n_iter = 40000
        for it in range(n_iter):
            if budget_s - (time.time() - t0) < 40:
                break
            T = T_init * T_ratio ** (it / n_iter)
            h_trial = h.copy()
            n_pert = np.random.randint(2, 5)
            idx = np.random.choice(n_sa, n_pert, replace=False)
            delta = np.random.randn(n_pert) * 0.08
            delta -= delta.mean()
            for i, j in enumerate(idx):
                h_trial[j] += delta[i]
            h_trial = project_feasible(h_trial, target_sa)
            c5_t = compute_c5(h_trial)
            if c5_t < current_c5 or (np.random.rand() < np.exp(-(c5_t - current_c5) / max(T, 1e-10))):
                h = h_trial
                current_c5 = c5_t
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5

    # Phase 5: Final high-beta SLSQP polish
    if budget_s - (time.time() - t0) > 50:
        n_fin = len(best_h)
        target_fin = n_fin / 2.0
        dx_fin = 2.0 / n_fin
        bnds_fin = Bounds(np.zeros(n_fin), np.ones(n_fin))
        cons_fin = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_fin,)}]
        for beta in [5000, 20000, 100000]:
            if budget_s - (time.time() - t0) < 15:
                break
            try:
                res = minimize(obj_fn, best_h, args=(dx_fin, beta), method='SLSQP',
                               bounds=bnds_fin, constraints=cons_fin,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_fin)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 6: Fine resolution tuning
    if budget_s - (time.time() - t0) > 60:
        n_fb = len(best_h)
        for dn in [-3, -2, -1, 1, 2, 3, 5, -5, 7, -7]:
            n_try = n_fb + dn
            if n_try < 20 or budget_s - (time.time() - t0) < 18:
                continue
            dx_t = 2.0 / n_try
            target_t = n_try / 2.0
            h_t = project_feasible(resample_h(best_h, n_try), target_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_t,)}]
            for beta in [1000, 10000]:
                if budget_s - (time.time() - t0) < 10:
                    break
                try:
                    res = minimize(obj_fn, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    return best_h, best_c5, len(best_h)
