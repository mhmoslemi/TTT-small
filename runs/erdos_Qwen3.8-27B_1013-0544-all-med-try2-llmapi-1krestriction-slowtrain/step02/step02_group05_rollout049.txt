    n = len(h)
    dx = 2.0 / n

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    n = len(h)

        else:

    n = len(h)

        #           - h[j-s] for j in [s, n-1]

    for t in range(n_iter):
        corr = np.correlate(y, 1.0 - y, mode="full")
        s_star = int(np.argmax(corr)) - (n - 1)
        grad = corr_gradient(y, s_star)
        alpha = base_alpha / (1.0 + t * 0.02)
        h_new = project_box_sum(y - alpha * grad, target_sum)
        mu = (1.0 + t * 0.1) / (2.0 + t * 0.1 + t * 0.05)
        y = project_box_sum(h_new + mu * (h_new - h), target_sum)
        h = h_new
        v = compute_c5(h)
        if v < best_val:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def slsqp_optimize(h0, n, target_sum, dx, beta, maxiter, ftol):
    """SLSQP with smooth_max objective and sum constraint."""
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    def con_val(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(x.shape[0])
    cons = [{'type': 'eq', 'fun': con_val, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_box_sum(res.x, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()

    n_list = [70, 98, 126]
    time_alloc = {70: 0.55, 98: 0.25, 126: 0.20}

    for n in n_list:
        if remaining() < 30:
            break
        t_budget = budget_s * time_alloc.get(n, 0.15)
        t_n_start = time.time()

        ts = n / 2.0
        dx = 2.0 / n

        starts = []
        if parent is not None:
            pn = len(parent)
            if pn == n:
                starts.append(parent.copy())
            else:
                xo = np.linspace(0, 2, pn, endpoint=False)
                xn = np.linspace(0, 2, n, endpoint=False)
                starts.append(project_box_sum(np.interp(xn, xo, parent), ts))
            p_c = 1.0 - (parent if pn == n else np.interp(xn, xo, parent))
            starts.append(project_box_sum(p_c, ts))

        # Step-function inspired starts
        for frac in [0.4, 0.45, 0.5]:
            h_step = np.zeros(n)
            boundary = int(n * frac)
            h_step[:boundary] = 1.0
            starts.append(project_box_sum(h_step + np.random.randn(n) * 0.05, ts))

        starts.append(project_box_sum(0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) / n * 3), ts))
        starts.append(project_box_sum(np.random.random(n) * 0.6 + 0.2, ts))

        cur_h = None
        cur_c5 = float('inf')

        for si, h0 in enumerate(starts):
            if time.time() - t_n_start > t_budget * 0.85:
                break
            h0 = project_box_sum(h0, ts)
            c5_0 = compute_c5(h0)

            # Stage 1: Nesterov subgradient (fast, good for escaping)
            if si == 0 and c5_0 > 0.381:
                h_sg, c5_sg = nesterov_subgrad(h0, ts, n_iter=300, base_alpha=0.008)
                if c5_sg < c5_0:
                    h0 = h_sg
                    c5_0 = c5_sg

            # Stage 2: Progressive SLSQP
            betas = [30, 60, 120, 200]
            for beta in betas:
                if time.time() - t_n_start > t_budget * 0.9 or remaining() < 15:
                    break
                h0, c5_new = slsqp_optimize(h0, n, ts, dx, beta, 600, 1e-13)
                if c5_new < cur_c5:
                    cur_h, cur_c5 = h0.copy(), c5_new

            # Stage 3: High-precision SLSQP
            for beta in [350, 500]:
                if time.time() - t_n_start > t_budget * 0.95 or remaining() < 10:
                    break
                h0, c5_new = slsqp_optimize(h0, n, ts, dx, beta, 1500, 1e-14)
                if c5_new < cur_c5:
                    cur_h, cur_c5 = h0.copy(), c5_new

        if cur_h is not None and cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n

    # Final ultra-precise polish on best
    if best_h is not None and remaining() > 20:
        n_f = best_n
        ts_f = n_f / 2.0
        dx_f = 2.0 / n_f
        for beta in [500, 700, 1000, 1500]:
            if remaining() < 8:
                break
            h_new, c5_new = slsqp_optimize(best_h, n_f, ts_f, dx_f, beta, 5000, 1e-15)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new

        # Additional: try small perturbations + re-optimize
        if remaining() > 15:
            for _ in range(3):
                if remaining() < 10:
                    break
                h_pert = project_box_sum(best_h + np.random.randn(n_f) * 0.005, ts_f)
                h_new, c5_new = slsqp_optimize(h_pert, n_f, ts_f, dx_f, 600, 3000, 1e-14)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
