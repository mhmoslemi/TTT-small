        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def coord_descent(h, dx, max_sweeps=100, grid_size=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    t_grid = np.linspace(0.0, 1.0, grid_size)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            t_old = h[i]
            # Compute the effect of changing h[i] to t
            # Save and set to zero
            h[i] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode='full')
            
            # Change for lag k when h[i] goes from 0 to t:
            # For k≠0: delta = t * (1-h[i+k]) - t*h[i-k] = t*(1-h[i+k]-h[i-k])
            # For k=0: delta = t - t²
            # 
            # So new_corr[k] = corr_base[k] + t*(1-h[i+k]-h[i-k]) for k≠0
            #                  = corr_base[k] + t - t² for k=0
            #
            # Let me compute the coefficient vector d_k:
            k_vals = np.arange(-(n-1), n)  # lags
            # i+k valid where 0 <= i+k < n
            ipk = i + k_vals
            vm_ipk = (ipk >= 0) & (ipk < n)
            # i-k valid where 0 <= i-k < n
            imk = i - k_vals
            vm_imk = (imk >= 0) & (imk < n)
            
            d = np.zeros(2*n-1)
            d[vm_ipk] += (1.0 - h[ipk[vm_ipk]])
            d[vm_imk] -= h[imk[vm_imk]]
            
            # For k=0 (index n-1 in the corr array):
            # The formula gives d[0] contribution = (1-h[i]) - h[i] = 1-2h[i] but h[i]=0 so d=1
            # But the actual change for k=0 is t - t², not t*1 = t
            # So for k=0, the "linear" formula overestimates by t²
            # We'll handle this by using the exact formula in the evaluation
            
            idx_k0 = n - 1  # index of lag 0 in the full correlation
            
            # Evaluate f(t) = max_k new_corr[k] * dx
            # For k≠0: new_corr[k] = corr_base[k] + d[k]*t
            # For k=0: new_corr[k0] = corr_base[k0] + t - t²
            
            # Build the function
            f_vals = np.empty(grid_size)
            for gi, t in enumerate(t_grid):
                new_corr = corr_base + d * t
                new_corr[idx_k0] = corr_base[idx_k0] + t - t*t
                f_vals[gi] = np.max(new_corr)
            
            best_gi = np.argmin(f_vals)
            t_new = t_grid[best_gi]
            
            # Refine around the best point
            if 0 < best_gi < grid_size - 1:
                t_lo = t_grid[best_gi - 1]
                t_hi = t_grid[best_gi + 1]
                # Bisection-like refinement
                for _ in range(10):
                    t_mid = (t_lo + t_hi) / 2
                    # Evaluate at 3 points
                    for t_test in [t_lo, t_mid, t_hi]:
                        new_corr = corr_base + d * t_test
                        new_corr[idx_k0] = corr_base[idx_k0] + t_test - t_test*t_test
                        f_test = np.max(new_corr)
                        if f_test < f_vals[best_gi]:
                            f_vals[best_gi] = f_test
                            t_new = t_test
                    if t_mid <= t_lo or t_mid >= t_hi:
                        break
                    # Compare f at t_lo, t_mid, t_hi to narrow
                    new_corr_lo = corr_base + d * t_lo
                    new_corr_lo[idx_k0] = corr_base[idx_k0] + t_lo - t_lo*t_lo
                    new_corr_hi = corr_base + d * t_hi
                    new_corr_hi[idx_k0] = corr_base[idx_k0] + t_hi - t_hi*t_hi
                    f_lo = np.max(new_corr_lo)
                    f_hi = np.max(new_corr_hi)
                    if f_lo < f_hi:
                        t_hi = t_mid
                    else:
                        t_lo = t_mid
                
                # Final evaluation
                for t_test in [t_lo, t_mid, t_hi]:
                    new_corr = corr_base + d * t_test
                    new_corr[idx_k0] = corr_base[idx_k0] + t_test - t_test*t_test
                    f_test = np.max(new_corr)
                    if f_test < f_vals[best_gi]:
                        f_vals[best_gi] = f_test
                        t_new = t_test
            
            h[i] = t_new
            max_change = max(max_change, abs(t_new - t_old))
        
        # Project to maintain sum constraint
        h = project_feasible(h, target)
        
        if max_change < 1e-14:
            break
    
    return h

def subgrad_momentum(h, dx, n_iters=50000, lr_base=0.003, momentum=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        grad = subgrad_step(h, k_worst, n)
        
        lr = lr_base / (1.0 + it * 0.001)
        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, target)
    
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def project_feasible(h, target_sum):

    # Phase 3: More coordinate descent (finer grid)
    if elapsed_remaining() > 80:
        h_cd2 = coord_descent(best_h, dx, max_sweeps=200, grid_size=60, t0=t0, budget=budget_s - 60)
        c5 = compute_c5(h_cd2)
        if c5 < best_c5:
            best_h = h_cd2
            best_c5 = c5

    # Phase 4: SLSQP with very high beta
    if elapsed_remaining() > 40:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (target_sum,)}]
        for beta in [5000, 20000, 100000]:
            if elapsed_remaining() < 15:
                break
            try:
                res = minimize(smooth_max_obj, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 5: Binary approach
    if elapsed_remaining() > 60:
        h_bin = binary_search(best_h, n, target_sum)
        h_bin_ref, c5_bin = binary_local_search(h_bin, t0, budget_s - 40)
        if c5_bin < best_c5:
            best_h = h_bin_ref
            best_c5 = c5_bin

    # Phase 6: Multi-resolution
    if elapsed_remaining() > 80:
        for n_try in [n - 10, n + 10, n - 20, n + 20, 50, 80, 90, 100]:
            if n_try < 20 or elapsed_remaining() < 30:
                continue
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, n_try / 2.0)
            h_t = coord_descent(h_t, 2.0 / n_try, max_sweeps=80, grid_size=30, t0=t0, budget=budget_s - 20)
            c5 = compute_c5(h_t)
            if c5 < best_c5:
                best_h = h_t
                best_c5 = c5
                n = n_try
                dx = 2.0 / n
                target_sum = n / 2.0

    # Phase 7: Final coordinate descent polish on best
    if elapsed_remaining() > 20:
        h_final = coord_descent(best_h, dx, max_sweeps=150, grid_size=50, t0=t0, budget=budget_s - 5)
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_h = h_final
            best_c5 = c5

    # Phase 8: Final SLSQP
    if elapsed_remaining() > 12:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (target_sum,)}]
        try:
            res = minimize(smooth_max_obj, best_h, args=(dx, 50000), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
