def project_feasible(x, target_sum):
    lo = -0.5
    hi = 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_lag(h, L):
    n = len(h)
    o = 1.0 - h
    g = np.zeros(n)
    if L == 0:
        g[:] = o - h
    elif L > 0:
        g[L:] += o[:n - L]
        g[:n - L] -= h[L:]
    else:
        g[:n + L] += o[-L:]
        g[-L:] -= h[:n + L]
    return g

def smooth_obj(x):

    vals = corr * _dx
    m = vals.max()
    shifted = vals - m
    e = np.exp(_g_alpha * shifted)
    return m + np.log(e.sum()) / _g_alpha

def smooth_grad(x):

    vals = corr * _dx

    for k in range(len(w)):

            continue
