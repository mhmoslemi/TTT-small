from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_cvector(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-10
            if mask.sum() == 0:
                break
            h[mask] *= (target_sum / h.sum()) if h.sum() > 0 else 1.0
            # Better: uniform reduction
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_subgradient(h, n, active_lags, weights=None):
    """Compute subgradient of max_k c[k]*dx at h using active lags."""
    # c[m] where m = 0..2n-2, corresponding to shift s = m - (n-1)
    grad = np.zeros(n)
    for m in active_lags:
        s = m - (n - 1)
        w = weights[m - active_lags[0]] if weights is not None else 1.0
        # grad[i] += w * [(1-h[i+s]) * 1_{valid} - h[i-s] * 1_{valid}]
        if s >= 0:
            # i+s valid means i in [0, n-1-s]
            for i in range(n - s):
                grad[i] += w * (1 - h[i + s])
            # i-s valid means i in [s, n-1]
            for i in range(s, n):
                grad[i] -= w * h[i - s]
        else:
            ss = -s  # positive shift in the other direction
            for i in range(ss, n):
                grad[i] += w * (1 - h[i - ss])
            for i in range(n - ss):
                grad[i] -= w * h[i + ss]
    return grad

    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wi = w[m_idx]
        if wi < 1e-10:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] -= wi * h[s:]
        else:
            ss = -s
            grad[:n - ss] += wi * (1.0 - h[ss:])
            grad[ss:] -= wi * h[:n - ss]
    grad *= dx
    return grad

def subgradient_phase(h, n, alpha, max_iter, budget_fn):
    x = h.copy()
    v = np.zeros(n)
    beta = 0.9
    obj = smooth_obj(x, n, alpha)
    for it in range(max_iter):
        if budget_fn() < 15:
            break
        g = smooth_grad(x, n, alpha)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        v = beta * v + (1.0 - beta) * g
        base_eta = 0.05 / (np.linalg.norm(v) + 1e-10)
        best_x = x.copy()
        best_obj = obj
        for factor in [0.1, 0.5, 1.0, 2.0, 5.0]:
            eta = base_eta * factor
            x_try = project_feasible(x - eta * v, n)
            obj_try = smooth_obj(x_try, n, alpha)
            if obj_try < best_obj:
                best_obj = obj_try
                best_x = x_try
        if np.allclose(best_x, x):
            break
        x = best_x
        obj = best_obj
    return x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    bounds = [(0.0, 1.0)] * n

    def sum_con(x):
        return np.sum(x) - n / 2.0

    def sum_con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]

    # Phase 1: Continuation subgradient with momentum
    alphas = [3, 10, 30, 100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if remaining() < 40:
            break
        iters = int(min(30000, max(5000, remaining() * 20 / len(alphas))))
        x = subgradient_phase(best_h, n, alpha, iters, remaining)
        c5 = compute_c5(x, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()

    # Phase 2: SLSQP polish
    for alpha in [500, 2000, 10000]:
        if remaining() < 20:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, n, alpha), best_h,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-17})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_c5 = c
                best_h = h_try
        except Exception:
            pass

    # Phase 3: Multi-restart exploration
    n_restarts = 80
    for r in range(n_restarts):
        if remaining() < 40:
            break
        # Vary perturbation scale
        scale = 0.01 + 0.06 * (r / max(1, n_restarts - 1)) + 0.01 * rng.random()
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)

        # Quick subgradient from perturbed start
        x = subgradient_phase(h_pert, n, 500, 3000, remaining)
        if remaining() < 20:
            break
        c5 = compute_c5(x, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()

        # SLSQP from this point
        if remaining() > 25:
            try:
                res = minimize(lambda x: smooth_obj(x, n, 3000), x,
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 3000, 'ftol': 1e-17})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_try
            except Exception:
                pass

    # Phase 4: Targeted coordinate descent
    if remaining() > 15:
        c_vec = compute_cvec(best_h, n)
        max_val = c_vec.max()
        active = np.where(c_vec > max_val - 0.02)[0]
        for sweep in range(50):
            if remaining() < 10:
                break
            improved = False
            for i in range(n):
                if remaining() < 10:
                    break
                for d in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                    h_try = best_h.copy()
                    h_try[i] += d
                    h_try = project_feasible(h_try, n)
                    c = compute_c5(h_try, n)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                        best_h = h_try
                        improved = True
                        break
            if not improved:
                break

    # Phase 5: Final SLSQP with very high alpha
    if remaining() > 8:
        try:
            res = minimize(lambda x: smooth_obj(x, n, 50000), best_h,
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 10000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_c5 = c
                best_h = h_try
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
