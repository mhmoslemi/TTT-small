import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] -= diff / cnt
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, dx, eps):
    """Log-sum-exp smooth approximation of C5"""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if eps < 1e-10:
        return m
    shifted = eps * (corr - m)
    # For numerical stability
    log_sum = m + np.log(np.sum(np.exp(shifted))) / eps
    return log_sum

def smooth_grad(h, dx, eps):
    """Gradient of the smooth objective"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # these are g_k values
    m = np.max(corr)
    shifted = eps * (corr - m)

    # m_k = k - (n-1) for k = 0, ..., 2n-2

    grad = np.zeros(n)
    # Only compute for significant weights
    threshold = 1e-10
    active = p > threshold
    for k in np.where(active)[0]:
        mk = k - (n - 1)  # lag
        # Term 1: (1-h[j-mk]) for j where 0 <= j-mk < n
        # j-mk in [0, n-1] => j in [mk, mk+n-1] intersect [0, n-1]
        lo1 = max(0, mk)
        hi1 = min(n-1, mk + n - 1)
        if lo1 <= hi1:

        # Term 2: -h[j+mk] for j where 0 <= j+mk < n
        # j+mk in [0, n-1] => j in [-mk, n-1-mk] intersect [0, n-1]
        lo2 = max(0, -mk)
        hi2 = min(n-1, n-1-mk)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += p[k] * dx * (-h[lo2+mk:hi2+mk+1])
    
    return grad

    def obj(x):
        return smooth_obj(x, dx, eps)
    
    def obj_grad(x):
        return smooth_grad(x, dx, eps)
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    def constraint_grad(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': constraint_fun, 'jac': constraint_grad}
    
    res = minimize(
        obj, h_start, jac=obj_grad,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, n)
