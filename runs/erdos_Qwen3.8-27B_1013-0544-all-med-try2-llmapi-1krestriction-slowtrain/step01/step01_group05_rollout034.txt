import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 1.0
_g_ts = 0.0

def project_feasible(x, target_sum):
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_grad(h, dx, tau):
    n = len(h)
    nh = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    s = tau * f
    m = s.max()
    e = np.exp(s - m)
    se = float(e.sum())
    p = e / se
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        L = k - nh
        if L >= 0:
            g[L:] += pk * (1.0 - h[:n - L])
            g[:n - L] -= pk * h[L:]
        else:
            L2 = -L
            g[:n - L2] += pk * (1.0 - h[L2:])
            g[L2:] -= pk * h[:n - L2]
    g *= dx
    return g

def active_set_grad(h, dx, tol_frac=0.01):
    n = len(h)
    nh = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    tol = max(c_max * tol_frac, 1e-6)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > 12:
        active = np.argsort(corr)[-12:]
    g = np.zeros(n)
    for idx in active:
        L = int(idx) - nh
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            L2 = -L
            g[:n - L2] += (1.0 - h[L2:])
            g[L2:] -= h[:n - L2]
    g /= float(len(active))
    g *= dx
    return g

def _obj(x):
    return compute_c5(x, _g_dx)

def _con(x):
    return np.sum(x) - _g_ts

def _slsqp_run(h_init, dx, ts, maxiter=3000):
    global _g_dx, _g_ts
    _g_dx = dx
    _g_ts = ts
    res = minimize(_obj, h_init, method='SLSQP',
                   bounds=[(0.0, 1.0)] * len(h_init),
                   constraints=[{'type': 'eq', 'fun': _con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None:
        hr = project_feasible(res.x, ts)
        return hr, compute_c5(hr, dx)
    return None, np.inf

    global _g_dx, _g_ts
    start = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = np.inf
    best_h = None
    best_n = 70

    for ni, np_ in enumerate(n_list):
        remaining = (start + budget_s - 5) - time.time()
        if remaining < 60:
            break
        n_rem = len(n_list) - ni
        t_alloc = remaining / n_rem * 0.92

        dx = 2.0 / np_
        ts = np_ / 2.0
        _g_dx = dx
        _g_ts = ts

        try:
            if np_ == 70:

        lc5 = compute_c5(h, dx)
        lb = h.copy()

        # Phase A: Smooth max annealing with Nesterov momentum
        t_sm = time.time()
        sm_budget = t_alloc * 0.55
        tau_list = [5.0, 12.0, 25.0, 60.0, 150.0, 400.0]

        for tau in tau_list:

            if (start + budget_s - 5) - time.time() < 25:

            mu = 0.35
