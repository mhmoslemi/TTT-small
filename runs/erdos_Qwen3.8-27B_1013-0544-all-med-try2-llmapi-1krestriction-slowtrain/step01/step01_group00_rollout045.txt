def binary_local_search(h_binary, n_swaps, T_init, T_final, rng):
    """Simulated annealing on binary sequence."""
    n = len(h_binary)
    h = h_binary.copy()
    corr = np.correlate(h, 1-h, mode='full') * (2.0/n)
    c5 = np.max(corr)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for step in range(n_swaps):
        T = T_init * (T_final / T_init) ** (step / n_swaps)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1-h, mode='full') * (2.0/n)
        new_c5 = np.max(new_corr)
        delta = new_c5 - c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = new_c5
            corr = new_corr
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    
    return best_h, best_c5

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

    return v[-1]

    n = len(v) - 1

    t = v[-1]

    dx = 2.0 / len(h)

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)
