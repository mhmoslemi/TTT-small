    c = compute_overlaps(h, n) * (2.0 / n)
    m = c.max()
    if m * alpha > 500:
        s = np.exp(c - m)
        return m + np.log(s.sum()) / alpha

def slsqp_polish(h0, n, alpha=5000, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        return smooth_obj(x, n, alpha)
    
    def obj_grad(x):
        return grad_smooth(x, n, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
    h_try = project_to_feasible(res.x, n)
    return h_try, compute_c5(h_try, n)
