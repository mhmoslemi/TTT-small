    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    if diff > 0:
        cap = (1.0 - h[mask])
    else:
        cap = h[mask]
    total_cap = cap.sum()
    if total_cap < 1e-14:
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    else:
        h[mask] += cap * (diff / total_cap)
    return np.clip(h, 0.0, 1.0)

        return smooth_c5(x, tau)

        # Numerical gradient

        n = len(x)
        g = np.zeros(n)

            x_old = x[i]
            x[i] = x_old + eps
            g[i] = (smooth_c5(x, tau) - f0) / eps
            x[i] = x_old
        return g

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    if diff > 0:
        cap = (1.0 - h[mask])
    else:
        cap = h[mask]
    total_cap = cap.sum()
    if total_cap < 1e-14:
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    else:
        h[mask] += cap * (diff / total_cap)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau) * dx

def slsqp(h_start, tau, maxiter=300):
    n = len(h_start)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
            'jac': lambda x: np.ones(n)}
    res = minimize(lambda x: smooth_c5(x, tau), h_start, method='SLSQP',
                   bounds=bnds, constraints=[cons],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    x = fix_sum(res.x, target)
    return x, evaluate_c5(x)

def binary_push(h, rounds=15, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = evaluate_c5(h)
    if rng is None:
        rng = np.random.default_rng(7)
    for _ in range(rounds):
        imp = False
        for i in rng.permutation(n):
            old = h[i]
            for nv in (0.0, 1.0):
                h[i] = nv
                ht = fix_sum(h, target)
                v = evaluate_c5(ht)
                if v < best - 1e-13:
                    h, best = ht, v
                    imp = True
                    break
            else:
                h[i] = old
        if not imp:
            break
    return h, best

def pairwise_push(h, rng=None, max_rounds=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = evaluate_c5(h)
    if rng is None:
        rng = np.random.default_rng(13)
    for _ in range(max_rounds):
        imp = False
        for _ in range(500):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            for vi in (0.0, 1.0):
                for vj in (0.0, 1.0):
                    h[i] = vi
                    h[j] = vj
                    ht = fix_sum(h, target)
                    v = evaluate_c5(ht)
                    if v < best - 1e-13:
                        h, best = ht, v
                        imp = True
                        break
                if imp:
                    break
        if not imp:
            break
    return h, best

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return fix_sum(h_new, n_new / 2.0)

def sa_explore(h, n_steps, t_init, t_final, t0, budget_s, rng):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for step in range(n_steps):
        if (time.time() - t0) > budget_s - 5:
            break
        temp = t_init * (t_final / t_init) ** (step / max(n_steps - 1, 1))
        i = rng.integers(n)
        h[i] = rng.random()
        ht = fix_sum(h, target)
        v = evaluate_c5(ht)
        delta = v - evaluate_c5(best_h)
        if v < best_c5 or rng.random() < np.exp(-(v - best_c5) / max(temp, 1e-12)):
            h = ht
            if v < best_c5:
                best_c5 = v
                best_h = ht.copy()
        else:
            h = fix_sum(best_h, target)
    return best_h, best_c5
