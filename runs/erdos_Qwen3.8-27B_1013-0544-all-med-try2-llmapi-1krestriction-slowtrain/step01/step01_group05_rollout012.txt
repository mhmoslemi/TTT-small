            h_sl, c5_sl = slsqp_optimize(h_loc, dx_t, target_t, n_try, maxiter=1000)
            if c5_sl < c5_loc:
                h_loc, c5_loc = h_sl, c5_sl
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc
            cur_n = n_try

    # Phase 5: Final polish with fine active-set gradient
    remaining = deadline - time.time()
    if remaining > 3:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        h_pol, c5_pol = projected_gradient(best_h, dx_f, target_f, n_f,
                                           int(min(3000, remaining * 400)),
                                           ss=0.0005, mu=0.4, tol_frac=0.005)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final SLSQP squeeze if time allows
    remaining = deadline - time.time()
    if remaining > 20:
        n_f = len(best_h)
        h_sl, c5_sl = slsqp_optimize(best_h, 2.0 / n_f, n_f / 2.0, n_f, maxiter=2000)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n_final)
    return (best_h, best_c5, n_final)
