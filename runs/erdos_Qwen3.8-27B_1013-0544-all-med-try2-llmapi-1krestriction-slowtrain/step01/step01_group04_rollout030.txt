import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_s = np.exp(T * shifted)
    s = exp_s.sum()
    w = exp_s / s
    val = max_c + np.log(s) / T
    lags = np.arange(2*n-1) - (n-1)
    moh = 1.0 - h
    grad = np.zeros(n)
    for m in range(2*n-1):
        wm = w[m]
        if wm < 1e-13:
            continue
        k = lags[m]
        if k >= 0:
            t1 = np.zeros(n)
            t2 = np.zeros(n)
            t1[k:] = moh[:n-k]
            t2[:n-k] = h[k:]
        else:
            kp = -k
            t1 = np.zeros(n)
            t2 = np.zeros(n)
            t1[kp:] = moh[:n-kp]
            t2[:n-kp] = h[kp:]
        grad += wm * (t1 - t2)
    grad *= dx
    return val, grad

def nesterov_descent(h0, dx, n, max_time, T_init=5.0, T_final=500.0, seed=42):
    start_t = time.time()
    rng = np.random.RandomState(seed)
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    iter_count = 0
    last_improve = 0
    
    while time.time() - start_t < max_time:
        iter_count += 1
        prog = min(1.0, (time.time() - start_t) / max_time)
        T = T_init * (T_final / T_init) ** prog
        lr = 1.0 / (T + T_init)
        
        # Periodic kick to escape local minima
        if iter_count % 500 == 0 and (iter_count - last_improve) > 300:
            h = project_feasible(h + rng.randn(n) * 0.05, n)
            v = np.zeros(n)
        
        _, grad = smooth_max_and_grad(h, dx, T)
        
        # Nesterov lookahead
        h_lk = project_feasible(h - beta * v, n)
        _, grad_lk = smooth_max_and_grad(h_lk, dx, T)
        v = beta * v - lr * grad_lk
        h_new = project_feasible(h + v, n)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improve = iter_count
        
        h = h_new
    
    return best_h, best_c5

def slsqp_smoothed(h_start, n, dx, target, T, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: smooth_max_and_grad(x, dx, T)[0],
        h_start.copy(),
        jac=lambda x: smooth_max_and_grad(x, dx, T)[1],
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

def slsqp_true(h_start, n, dx, target, max_iter=100):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: compute_c5(x, dx),
        h_start.copy(),
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, n)

One more optimization: I can vectorize the gradient computation using the full correlation structure.

Instead of looping over lags, I can compute:
grad[j] = dx * (sum_k w[k] * shift_k(1-h)[j] - sum_k w[k] * shift_{-k}(h)[j])

The first term is: for each j, sum over k of w[k] * (1-h[j-k]) where j-k is in [0,n)
= sum over j-k = i (where i is in [0,n)) of w[k] * (1-h[i]) where k = j-i
= sum over i in [0,n) of w[j-i] * (1-h[i]) (where w is zero outside its range)

This is a convolution! Specifically:
first_term[j] = sum_i w[j-i] * (1-h[i]) = (w * (1-h))[j] (convolution)

Similarly:
second_term[j] = sum_k w[k] * h[j+k] where j+k is in [0,n)
= sum over j+k = i (where i in [0,n)) of w[k] * h[i] where k = i-j
= sum over i in [0,n) of w[i-j]
