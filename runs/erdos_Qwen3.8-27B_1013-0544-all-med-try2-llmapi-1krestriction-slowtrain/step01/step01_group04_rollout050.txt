from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n, max_iter=50):
    target = n / 2.0
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(alpha * (corr - m)))
    return (m + np.log(s) / alpha) * dx

def local_polish(h, n, dx, target_sum, max_iter=200, alpha=None):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    if alpha is not None:
        obj = lambda x: soft_max_c5(x, dx, alpha)
    else:
        obj = lambda x: compute_c5(x, dx)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraint,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def de_generation(pop, pop_fit, n, dx, rng, f_mu, cr):
    nt = len(pop)
    for i in range(nt):
        idx = [j for j in range(nt) if j != i]
        a, b, c = rng.choice(idx, 3, replace=False)
        v = pop[a] + f_mu * (pop[b] - pop[c])
        v = project_feasible(v, n)
        r = rng.random()
        mask = rng.random(n) < cr
        mask[int(r * n) % n] = True
        u = np.where(mask, v, pop[i])
        u = project_feasible(u, n)
        fu = compute_c5(u, dx)
        if fu < pop_fit[i]:
            pop[i] = u
            pop_fit[i] = fu
    return pop, pop_fit
