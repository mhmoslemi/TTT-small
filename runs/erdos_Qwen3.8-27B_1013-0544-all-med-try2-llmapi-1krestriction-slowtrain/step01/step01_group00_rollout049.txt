    h_best = h.copy()

    # Phase 1: SLSQP t-var initial refinement
    if time_left() > 250:
        h_new, c5_new = optimize_t_var(h, max_iter=200, ftol=1e-10)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: L-BFGS-B multi-temperature with penalty
    temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for temp in temps:
        if time_left() < 50:
            break
        h_new, c5_new = optimize_lbfgs(h, temp, 50.0, max_iter=300)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()
        if time_left() > 50:
            h_new, c5_new = optimize_lbfgs(h, temp, 500.0, max_iter=200)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 3: Active lag descent
    if time_left() > 100:
        h_new, c5_new = active_lag_descent(h_best, n_steps=200, step_scale=0.001)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 4: SLSQP polish
    if time_left() > 150:
        h_new, c5_new = optimize_t_var(h, max_iter=300, ftol=1e-11)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 5: Structured perturbations + L-BFGS-B
    n_pert = 12
    for i in range(n_pert):
        if time_left() < 80:
            break
        if i < 4:
            scale = 0.01 + 0.005 * i
            h_p = h_best + rng.randn(n) * scale
        elif i < 8:
            # Circular shift perturbation
            shift = rng.randint(1, 5)
            h_p = np.roll(h_best, shift)
            h_p = project_h(h_p, n * 0.5)
        else:
            # Section flip
            start = rng.randint(0, n - 10)
            end = min(start + rng.randint(5, 20), n)
            h_p = h_best.copy()
            h_p[start:end] = 1.0 - h_p[start:end]
            h_p = project_h(h_p, n * 0.5)
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_lbfgs(h_p, 0.003, 100.0, max_iter=200)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()
        if time_left() > 120:
            h_new, c5_new = optimize_t_var(h_best, max_iter=150, ftol=1e-11)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 6: Higher resolution n=270
    n_new = 270
    if time_left() > 250:
        h_interp = interp_h(h_best, n_new)
        h_cur = h_interp.copy()
        c5_cur = compute_c5(h_cur)
        if time_left() > 150:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=150, ftol=1e-10)
            if c5_new < c5:
                h_cur, c5 = h_new, c5_new
                h_best = h_cur.copy()
        for temp in [0.01, 0.005, 0.002, 0.001]:
            if time_left() < 60:
                break
            h_new, c5_new = optimize_lbfgs(h_cur, temp, 100.0, max_iter=200)
            if c5_new < c5:
                h_cur, c5 = h_new, c5_new
                h_best = h_cur.copy()
        if time_left() > 100:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=200, ftol=1e-11)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new

    # Final: try n=360 if time permits
    n_new2 = 360
    if time_left() > 200:
        h_interp = interp_h(h_best, n_new2)
        h_cur = h_interp.copy()
        if time_left() > 100:
            h_new, c5_new = optimize_lbfgs(h_cur, 0.005, 100.0, max_iter=200)
            if c5_new < c5:
                h_cur, c5 = h_new, c5_new
                h_best = h_cur.copy()
        if time_left() > 80:
            h_new, c5_new = optimize_lbfgs(h_cur, 0.002, 500.0, max_iter=200)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
