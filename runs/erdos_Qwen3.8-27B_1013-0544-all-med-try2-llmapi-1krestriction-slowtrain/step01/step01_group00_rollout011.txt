        h = 0.5 + 0.3 * np.cos(2 * np.pi * rng.randint(2, 8) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    return project_h(h, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    # Initialize
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = project_h(h_init, n_init * 0.5)
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)

    # Phase 1: Optimize at initial resolution
    if tl() > 150:
        h, c5 = opt_t_topk(h_init, topk=10, max_iter=500)
        if c5 < c5_best:
            h_best, c5_best = h.copy(), c5
    if tl() > 100:
        for temp in [0.03, 0.01, 0.005, 0.002]:
            if tl() < 60:
                break
            h, c5 = opt_smooth(h_best, temp, max_iter=150)
            if c5 < c5_best:
                h_best, c5_best = h.copy(), c5

    # Phase 2: Try different resolutions
    for n_try in [60, 90, 120, 240, 360]:
        if tl() < 120:
            break
        h_try = interp_h(h_best, n_try)
        h_try, c5_try = opt_t_topk(h_try, topk=8, max_iter=400)
        if c5_try < c5_best:
            h_best, c5_best = h_try.copy(), c5_try
        if tl() > 60:
            h_try2, c5_try2 = opt_smooth(h_try, 0.01, max_iter=150)
            if c5_try2 < c5_best:
                h_best, c5_best = h_try2.copy(), c5_try2
                h_try = h_try2
            if tl() > 50:
                h_try3, c5_try3 = opt_t_topk(h_try, topk=6, max_iter=300)
                if c5_try3 < c5_best:
                    h_best, c5_best = h_try3.copy(), c5_try3

    # Phase 3: Structured multi-start at best resolution
    n_cur = len(h_best)
    for trial in range(25):
        if tl() < 50:
            break
        pt = trial % 4
        if trial < 5:
            h_try = h_best + rng.randn(n_cur) * (0.01 + 0.005 * trial)
        else:
            h_try = make_structured_start(n_cur, rng, pt)
        h_try = project_h(h_try, n_cur * 0.5)
        h_try, c5_try = opt_t_topk(h_try, topk=8, max_iter=300)
        if c5_try < c5_best:
            h_best, c5_best = h_try.copy(), c5_try

    # Phase 4: Gaussian multi-start at best resolution
    for trial in range(30):
        if tl() < 40:
            break
        scale = 0.008 + 0.003 * trial
        h_try = h_best + rng.randn(n_cur) * scale
        h_try = project_h(h_try, n_cur * 0.5)
        h_try, c5_try = opt_t_topk(h_try, topk=8, max_iter=300)
        if c5_try < c5_best:
            h_best, c5_best = h_try.copy(), c5_try
            n_cur = len(h_best)

    # Phase 5: Full constraint polish
    if tl() > 80:
        h, c5 = opt_t_full(h_best, max_iter=500, ftol=1e-12)
        if c5 < c5_best:
            h_best, c5_best = h.copy(), c5

    # Phase 6: Coordinate descent polish
    if tl() > 20:
        h_best = fast_coord_desc(h_best, n_passes=1, n_try=4)
        c5_check = compute_c5(h_best)
        if c5_check < c5_best:
            c5_best = c5_check

    # Final verification
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
