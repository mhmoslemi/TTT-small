def sa_binary(n, budget_s, t0):
    """Simulated annealing on binary sequences with weight n/2."""
    # Initialize randomly
    h = np.zeros(n)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1.0
    
    current_energy = np.max(n/2.0 - np.correlate(h, h, mode='full'))
    best_h = h.copy()
    best_energy = current_energy
    
    T = 10.0  # Initial temperature
    cooling = 0.999
    
    while time.time() - t0 < budget_s:
        # Pick a 1 to flip to 0 and a 0 to flip to 1
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        
        # Apply move
        h[i] = 0
        h[j] = 1
        
        new_energy = np.max(n/2.0 - np.correlate(h, h, mode='full'))
        delta = new_energy - current_energy
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_energy = new_energy
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1
            h[j] = 0
        
        T *= cooling
        if T < 0.01:
            break
    
    return best_h, best_energy * 2.0 / n

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = h > 0 and h < 1
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def sa_binary(n, max_time, t0):
    """Simulated annealing on binary sequences with weight n/2."""
    h = np.zeros(n)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1.0

    current_energy = np.max(n_half - np.correlate(h, h, mode='full'))
    best_h = h.copy()
    best_energy = current_energy

    T_min = 0.01
    cooling = 0.9995

        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]

        h[j] = 1.0

        delta = new_energy - current_energy

            h[j] = 0.0

    n = len(h_start)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))

        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except:
        return h_start, evaluate_c5(h_start)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 0.0) & (h < 1.0)
        if mask.sum() == 0:
            # All at bounds, try adjusting the one closest to the needed direction
            if diff > 0:
                idx = np.argmax(h)
            else:
                idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            continue
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def sa_binary(n, time_limit, t0):
    """SA on binary sequences with weight n/2, optimizing C5."""
    h = np.zeros(n)
    h[np.random.choice(n, n//2, replace=False)] = 1.0
    
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 0.3  # Temperature (in C5 units)
    T_min = 0.001
    cooling = 0.9999
    
    while time.time() - t0 < time_limit and T > T_min:
        # Pick random swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        delta = new_c5 - current_c5

                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

def refine(h_start, tau=100, maxiter=300):

        return smooth_max(corr, tau)

        c5 = evaluate_c5(x)
        if c5 < 1.0:

    except:
        pass
    return h_start, evaluate_c5(h_start)

import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * np.asarray(obj, dtype=float)
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def smooth_max_grad(obj, tau):
    """Return (value, gradient) of smooth max."""
    x = tau * np.asarray(obj, dtype=float)
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    softmax = exp_x / np.sum(exp_x)
    val = x_max / tau + np.log(np.sum(exp_x)) / tau
    grad = softmax * tau  # d/dobj of smooth_max = tau * softmax
    return val, grad / tau  # gradient w.r.t. obj

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            break
    return h

def slsqp_refine(h_start, tau, maxiter=500):
