import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, target_sum):
    n = len(h)
    k = int(target_sum)
    order = np.argsort(-h)
    result = np.zeros(n)
    result[order[:k]] = 1.0
    return result

def get_influence_fast(pos, h):
    n = len(h)
    infl = np.zeros(2 * n - 1)
    # idx1 = pos + p - n + 1, valid for p in [n-1-pos, n-1+pos]
    p1s = max(0, n - 1 - pos)
    p1e = min(2 * n - 2, n - 1 + pos)
    if p1s <= p1e:
        idx = pos + np.arange(p1s, p1e + 1) - n + 1
        infl[p1s:p1e+1] += h[idx]
    # idx2 = n - 1 + pos - p, valid for p in [pos, n-1+pos]
    p2s = max(0, pos)
    p2e = min(2 * n - 2, n - 1 + pos)
    if p2s <= p2e:
        idx = n - 1 + pos - np.arange(p2s, p2e + 1)
        infl[p2s:p2e+1] += h[idx]
    return infl

        best_max = np.max(overlap)
        best_swap = None
        
        for ii, i in enumerate(ones):
            for jj, j in enumerate(zeros):
                change = infl_ones[ii] - infl_zeros[jj]
                new_max = np.max(overlap + change)

                    best_swap = (i, j, change)

            h[i] = 0.0
            h[j] = 1.0

            no_improve = 0
        else:

    final_c5 = compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, target_sum):
    n = len(h)
    k = int(target_sum)
    order = np.argsort(-h)
    result = np.zeros(n)
    result[order[:k]] = 1.0
    return result

def get_influence(pos, h):
    n = len(h)
    infl = np.zeros(2 * n - 1)
    p1s = max(0, n - 1 - pos)
    p1e = min(2 * n - 2, n - 1 + pos)
