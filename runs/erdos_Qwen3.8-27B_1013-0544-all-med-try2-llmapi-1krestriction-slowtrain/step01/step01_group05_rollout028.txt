- Successive QP: linearize the quadratic constraints at the current point
- Or use a general NLP solver

The successive QP approach:
At current point h_current, linearize:
h^T Q_k h ≈ 2*h_current^T Q_k * (h - h_current) + h_current^T Q_k h_current
= 2*(Q_k * h_current)^T * h - h_current^T Q_k h_current

So the linearized constraint becomes:
sum_i h[i] - 2*(Q_k * h_current)^T * h + h_current^T Q_k h_current <= t/dx

This is linear in h and t! So at each iteration, we solve a linear program.

Actually, I want to also try a different approach that I think could be powerful: **successive linearization (cutting plane)**.

The idea:
1. At current point h, identify the active lags (those with correlation near the max)
2. Linearize the problem: minimize t subject to:
   - For each active lag k: linearized constraint corr_k(h) ≤ t/dx
   - 0 ≤ h ≤ 1
   - sum(h) = n/2
3. Solve this LP
4. Update h, go to step 1

The linearized constraint for lag k at point h_cur:
corr_k(h) ≈ corr_k(h_cur) + ∇_h corr_k(h_cur) · (h - h_cur)

So: corr_k(h_cur) + ∇_k · (h - h_cur) ≤ t/dx
i.e., ∇_k · h - t/dx ≤ corr_k(h_cur) - ∇_k · h_cur

This is linear in (h, t)! So we can solve it as an LP.

Using scipy.optimize.linprog:
- Variables: h[0..n-1], t (total n+1 variables)
- Objective: minimize t → c = [0, 0, ..., 0, 1]
- Equality: sum(h) = n/2 → A_eq = [1,1,...,1, 0], b_eq = n/2
- Inequality constraints:
  - ∇_k · h - t/dx ≤ corr_k(h_cur) - ∇_k · h_cur for each active k
  - -h[i] ≤ 0 (i.e., h[i] ≥ 0)
  - h[i] ≤ 1
- Bounds: 0 ≤ h[i] ≤ 1, t ≥ 0

This is a standard LP! We can use linprog with the 'highs' method.

This approach is essentially the Bundle Method for non-smooth optimization, specialized to this problem. It should converge to a local minimum.

def solve_active_lp(h_cur, active_lags, dx, n, target_sum):
    """Solve the LP relaxation at the current point with given active lags."""
    from scipy.optimize import linprog
    n_half = n - 1
    omh = 1.0 - h_cur
    
    # Compute gradients for active lags
    grads = []
    corr_vals = []
    for idx in active_lags:
        L = int(idx) - n_half
        g = np.zeros(n)
        if L >= 0:
            g[L:] += omh[:n - L]
            if n - 1 - L >= 0:
                g[0:n - L] -= h_cur[L:]
        else:
            J = -L
            if n - 1 - J >= 0:
                g[0:n - J] += omh[J:]
            if J <= n - 1:
                g[J:] -= h_cur[:n - J]
        grads.append(g)
    
    # Current correlation values
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    for idx in active_lags:
        corr_vals.append(float(corr[idx]))
    
    # Build LP:
    # Variables: h[0..n-1], t (index n is t)
    # Minimize: t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k:
    #   grad_k · h - t/dx <= corr_k - grad_k · h_cur
    #   grad_k · h + (-1/dx) * t <= corr_k - grad_k · h_cur
    A_rows = []
    b_rows = []
    for i, (g, cv) in enumerate(zip(grads, corr_vals)):
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0 / dx
        A_rows.append(row)
        b_rows.append(cv - np.dot(g, h_cur))
    
    # h[i] >= 0 → -h[i] <= 0
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # h[i] <= 1 → h[i] <= 1
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0)
    
    if A_rows:
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
    else:
        A_ub = None
        b_ub = None
    
    # Equality: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], float(res.t)
    return None, None

    return np.correlate(h, 1.0 - h, mode='full')

    """Gradient of correlation at lag L w.r.t. h."""
    n = len(h)
    omh = 1.0 - h
    g = np.zeros(n)
    if L >= 0:
        g[L:] += omh[:n - L]
        if n - 1 - L >= 0:
            g[0:n - L] -= h[L:]
    else:
        J = -L

        if J <= n - 1:
            g[J:] -= h[:n - J]

    corr = get_corr(x)
    vals = corr * dx
    m = vals.max()
    if m < 1e-15:
        return 0.0
    shifted = np.clip(beta * (vals - m), -50, 50)
    return m + np.log(np.exp(shifted).sum()) / beta

    n_half = n - 1

    vals = corr * dx

def project_feasible(x, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_at_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    g = np.zeros(n)
    if L >= 0:
        g[L:] += omh[:n - L]
        if n - 1 - L >= 0:
            g[0:n - L] -= h[L:]
    else:
        J = -L
        if n - 1 - J >= 0:
            g[0:n - J] += omh[J:]
        if J <= n - 1:
            g[J:] -= h[:n - J]
    return g

    corr = get_corr(x)
    vals = corr * dx
    m = vals.max()
    if m < 1e-15:
        return 0.0
    shifted = np.clip(beta * (vals - m), -50.0, 50.0)
    return m + np.log(np.exp(shifted).sum()) / beta

def smooth_grad_val(x, dx, beta):
    n = len(x)
    n_half = n - 1
    corr = get_corr(x)
    vals = corr * dx
    m = vals.max()
    shifted = np.clip(beta * (vals - m), -50.0, 50.0)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-12:
            continue
        L = k - n_half
        g = grad_corr_at_lag(x, L)
        grad += p[k] * g
    return dx * grad
