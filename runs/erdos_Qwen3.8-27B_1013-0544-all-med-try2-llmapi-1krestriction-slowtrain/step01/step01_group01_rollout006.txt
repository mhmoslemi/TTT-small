        h += diff / len(h)

    """Compute overlap at specific lag"""
    n = len(h)
    abs_lag = abs(lag)
    if abs_lag == 0:
        return np.sum(h * (1 - h))
    if abs_lag >= n:
        return 0.0
    return np.sum(h[:n-abs_lag] * (1 - h[abs_lag:]))
