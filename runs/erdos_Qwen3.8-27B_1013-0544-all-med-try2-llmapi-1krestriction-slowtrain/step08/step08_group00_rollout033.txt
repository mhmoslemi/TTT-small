def build_active_gradients(h, active, n):
    r=int(active.size); G=np.zeros((r,n), dtype=float)

    if r==0: return None
    cobj=np.zeros(v+1); cobj[-1]=1.0

import time
import numpy as np
from scipy.optimize import minimize, linprog

def eval_c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def eval_c_max(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))))

def fix_sum_exact(h, target):
    h = np.array(h, dtype=float, copy=True)
    if h.size == 0:
        return h
    idx = int(np.argmax(np.minimum(h, 1.0 - h)))
    val = target - (float(np.sum(h, dtype=np.float64)) - float(h[idx]))
    if 0.0 <= val <= 1.0:
        h[idx] = val
        return np.clip(h, 0.0, 1.0, out=h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(12):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0 - 1e-15) if diff > 0.0 else (h > 1e-15)
        idxs = np.nonzero(mask)[0]
        if idxs.size == 0:
            break
        h[idxs] = np.clip(h[idxs] + diff / idxs.size, 0.0, 1.0)
    idx = int(np.argmax(np.minimum(h, 1.0 - h)))
    val = target - (float(np.sum(h)) - float(h[idx]))
    h[idx] = min(1.0, max(0.0, val))
    return h

def project_feasible(h, target=None):
    h = np.asarray(h, dtype=float)
    if target is None:
        target = 0.5 * float(len(h))
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-12:
        return fix_sum_exact(h, target)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return fix_sum_exact(np.clip(h - 0.5 * (lo + hi), 0.0, 1.0), target)

def adjoint_gradient(h, n, w):
    if n <= 1:
        return np.zeros_like(h)
    a = np.correlate((1.0 - h)[::-1], w, mode='full')
    b = np.correlate(h, w, mode='full')
    s = n - 1
    return a[s:s + n] - b[s:s + n][::-1]

def smooth_objective(x, n, alpha, mu):
    c = eval_c_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    sm = float(sh.max())
    e = np.exp(sh - sm)
    s = float(x.sum()) - 0.5 * n
    return float(m + (sm + float(np.log(e.sum()))) / alpha + mu * s * s)

def smooth_gradient(x, n, alpha, mu):
    c = eval_c_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    sm = float(sh.max())
    p = np.exp(sh - sm)
    p /= p.sum()
    g = (2.0 / float(n)) * adjoint_gradient(x, n, p)
    return g + 2.0 * mu * (float(x.sum()) - 0.5 * n)

def lbfgs_phase(h, n, deadline, alphas=None, mu=1.0, maxit=250):
    if alphas is None:
        alphas = (20, 200, 2000, 20000, 200000, 2000000)
    h = np.array(h, dtype=float, copy=True)
    best_h = h.copy()
    best_c = eval_c_max(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() >= deadline:
            break
        rem = max(0.0, deadline - time.time())
        it = int(min(maxit, max(40, int(rem * 20))))
        try:
            res = minimize(smooth_objective, h, jac=smooth_gradient, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': it, 'ftol': 1e-13, 'gtol': 1e-10, 'maxls': 50})
            h2 = project_feasible(res.x)
            c2 = eval_c_max(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                h = h2.copy()
        except Exception:
            pass
    return best_h, best_c

def build_active_gradients(h, active, n):
    G = np.zeros((active.size, n))
    dx = 2.0 / float(n)
    g = np.zeros(n)
    for i, m in enumerate(active):
        g.fill(0.0)
        k = int(m - (n - 1))
        if k >= 0:
            L = n - k
            g[k:k + L] += 1.0 - h[:L]
            g[:L] -= h[k:k + L]
        else:
            L = n + k
            g[:L] += 1.0 - h[-L:]
            g[-L:] -= h[:L]
        G[i] = g * dx
    return G

def lp_direction(h, n, G, step):
    r, v = G.shape
    if r == 0:
        return None
    cobj = np.zeros(v + 1)
    cobj[-1] = 1.0
    A_ub = np.empty((r, v + 1))
    A_ub[:, :v] = G
    A_ub[:, v] = -1.0
    A_eq = np.zeros((1, v + 1))
    A_eq[0, :v] = 1.0
    b_eq = np.array([0.0])
    bds = []
    for i in range(v):
        bds.append((-h[i] * step, (1.0 - h[i]) * step))
    bds.append((-1e6, 1e6))
    try:
        res = linprog(cobj, A_ub=A_ub, b_ub=np.zeros(r), A_eq=A_eq, b_eq=b_eq, bounds=bds, method='highs')
    except Exception:
        try:
            res = linprog(cobj, A_ub=A_ub, b_ub=np.zeros(r), A_eq=A_eq, b_eq=b_eq, bounds=bds)
        except Exception:
            return None
    if res.status != 0:
        return None
    t = float(res.x[-1])
    if t > -1e-11:
        return None
    d = res.x[:v].copy()
    return d if float(np.max(np.abs(d))) > 1e-15 else None

def line_search_step(h, n, d, c0):
    eps_max = 1.0
    pos = d > 0.0
    if pos.any():
        eps_max = min(eps_max, float(np.min((1.0 - h[pos]) / d[pos])))
    neg = d < 0.0
    if neg.any():
        eps_max = min(eps_max, float(np.min(-h[neg] / (-d[neg]))))
    if eps_max <= 1e-12:
        return h, c0, 0.0
    eps_max = min(eps_max, 1.0)
    best_v, best_e = c0, 0.0
    for f in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.0009765625):
        e = f * eps_max
        v = eval_c_max(h + e * d, n)
        if v < best_v:
            best_v, best_e = v, e
    if best_e > 0.0:
        lo = max(0.0, best_e - 0.05 * eps_max)
        hi = min(eps_max, best_e + 0.1 * eps_max)
        for e in np.linspace(lo, hi, 9):
            v = eval_c_max(h + e * d, n)
            if v < best_v:
                best_v, best_e = v, e
    return h + best_e * d, best_v, best_e

def gradient_direction_step(h, n, step, alpha=100000.0):
    c = eval_c_vec(h, n)
    C = float(c.max())
    sh = alpha * (c - C)
    sm = float(sh.max())
    p = np.exp(sh - sm)
    p /= p.sum()
    d = -(2.0 / float(n)) * adjoint_gradient(h, n, p)
    d -= float(np.mean(d))
    nm = float(np.max(np.abs(d)))
    if nm <= 1e-15:
        return h, C, 0.0
    return line_search_step(h, n, step * d / nm, C)

def active_lp_polish(h, n, deadline, steps=(0.05, 0.02, 0.01, 0.005, 0.002, 0.001), max_iter=20):
    h = np.array(h, dtype=float, copy=True)
    best_h = h.copy()
    best_c = eval_c_max(h, n)
    for step in steps:
        if time.time() >= deadline:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            c = eval_c_vec(h, n)
            C = float(c.max())
            if C < best_c:
                best_c = C
                best_h = h.copy()
            tol = max(2e-5, 2e-5 * max(1.0, best_c))
            near = np.where(c >= C - tol)[0]
            if near.size > 80:
                near = near[np.argsort(c[near])[::-1][:80]]
            else:
                top = np.argsort(c)[::-1][:40]
                near = np.unique(np.concatenate((near, top)))
            if near.size == 0:
                near = np.argsort(c)[::-1][:20]
            G = build_active_gradients(h, near, n)
            d = lp_direction(h, n, G, step)
            if d is None:
                h2, v2, eps = gradient_direction_step(h, n, step)
            else:
                h2, v2, eps = line_search_step(h, n, d, C)
            if eps > 0.0 and v2 < C - 1e-13:
                h = project_feasible(h2, n)
                improved = True
    c = eval_c_max(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return project_feasible(best_h, n), best_c

def interpolate_h(h, no, nn):
    if no == nn:
        return project_feasible(h.copy())
    xo = np.linspace(0, 2.0, no, endpoint=False)
    xn = np.linspace(0, 2.0, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, h))

def smooth_wave_start(n, rng, nf=8, amp=0.4):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for f in range(1, int(nf) + 1):
        a = 2.0 * np.pi * f * x / 2.0
        h += (amp / (f * 1.5)) * rng.randn() * np.cos(a + rng.rand() * 2.0 * np.pi)
        h += (amp / (f * 1.8)) * rng.randn() * np.sin(a + rng.rand() * 2.0 * np.pi)
    return project_feasible(h)

def block_start(n, rng, nb=16):
    b = rng.rand(nb)
    h = np.tile(b, (n + nb - 1) // nb)[:n]
    return project_feasible(h)

def binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    if n % 2 and idx.size:
        h[idx[0]] = 0.5
    return project_feasible(h)

def periodic_start(n, rng):
    m = max(1, n // 2)
    a = 2.0 * np.pi * np.arange(m) / float(m)
    h = 0.5 + 0.35 * np.sin(a + rng.rand() * 6.283185307179586) + 0.08 * rng.randn(m)
    h = np.tile(h, (n + m - 1) // m)[:n]
    return project_feasible(h)

def perturb_start(h, n, rng, s):
    return project_feasible(h + s * rng.randn(n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(30.0, float(budget_s)) - 4.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        h0 = smooth_wave_start(650, rng, 10, 0.4)
        n0 = 650

    h0 = project_feasible(h0, n0)
    best_h, best_n = h0.copy(), n0
    best_c = eval_c_max(h0, n0)

    t = min(deadline - 650.0, time.time() + 180.0)
    if time.time() < t:
        best_h, best_c = lbfgs_phase(best_h, best_n, t, maxit=300)

    t = min(deadline - 420.0, time.time() + 100.0)
    if time.time() < t:
        best_h, best_c = active_lp_polish(best_h, best_n, t,
                                          steps=(0.05, 0.02, 0.01, 0.005, 0.002, 0.001),
                                          max_iter=25)

    for cand in (800, 1000, 1200):
        rem = deadline - time.time()
        if rem < 220 or cand == best_n:
            continue
        hc = interpolate_h(best_h, best_n, cand)
        te = min(time.time() + 80.0, deadline - 130.0)
        if te - time.time() < 20:
            continue
        ht, ct = lbfgs_phase(hc, cand, te, alphas=(10, 100, 1000, 10000, 100000), maxit=200)
        if ct < best_c:
            best_c, best_h, best_n = ct, ht, cand
        if ct < best_c + 5e-4:
            te = min(time.time() + 45.0, deadline - 100.0)
            if te - time.time() > 15:
                hp, cp = active_lp_polish(ht, cand, te,
                                          steps=(0.02, 0.01, 0.005, 0.002),
                                          max_iter=15)
                if cp < best_c:
                    best_c, best_h, best_n = cp, hp, cand

    ns = best_n
    starts = [
        smooth_wave_start(ns, rng, 8, 0.35),
        smooth_wave_start(ns, rng, 14, 0.5),
        block_start(ns, rng, 18),
        binary_start(ns, rng),
        periodic_start(ns, rng),
        perturb_start(best_h, ns, rng, 0.02),
        perturb_start(best_h, ns, rng, 0.05),
        perturb_start(best_h, ns, rng, 0.10),
    ]
    if n0 != ns:
        starts.append(interpolate_h(h0, n0, ns))

    for s in starts:
        rem = deadline - time.time()
        if rem < 120:
            break
        s = project_feasible(s, ns)
        te = min(time.time() + 25.0, deadline - 80.0)
        if te - time.time() < 10:
            break
        hs, cs = lbfgs_phase(s, ns, te, alphas=(50, 500, 5000, 50000, 200000), maxit=140)
        if cs < best_c + 0.003:
            te = min(time.time() + 12.0, deadline - 50.0)
            if te - time.time() > 5:
                hs, cs = active_lp_polish(hs, ns, te,
                                          steps=(0.01, 0.005, 0.002),
                                          max_iter=10)
        if cs < best_c:
            best_c, best_h, best_n = cs, hs, ns

    t = deadline - 2.0
    if time.time() < t:
        best_h, best_c = active_lp_polish(best_h, best_n, t,
                                          steps=(0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001),
                                          max_iter=40)

    if time.time() < deadline - 8:
        best_h, best_c = lbfgs_phase(best_h, best_n, deadline - 1,
                                     alphas=(1000000, 10000000), mu=10.0, maxit=80)

    best_h = project_feasible(best_h, best_n)
    return best_h, eval_c_max(best_h, best_n), best_n
