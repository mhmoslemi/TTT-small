    # Phase 2: Coordinate descent to find binary local opt
    if _can_go(10):
        hb, _ = coord_descent_quick(best_h, n, max_sweeps=200, deadline=deadline - 10)
        cb = compute_c5(hb, n)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
        # SLSQP from binary
        for alpha in [10000, 100000, 1000000]:
            if not _can_go(5):
                break
            h2 = slsqp_round(hb, n, alpha, maxiter=2000, ftol=1e-15)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()

    # Phase 3: Diverse restarts
    restart_inits = []
    # Binary random
    hr = np.zeros(n)
    hr[rng.choice(n, n // 2, replace=False)] = 1.0
    restart_inits.append(hr)
    # Block pattern (alternating)
    hr = np.zeros(n)
    blk = max(1, n // 12)
    for s in range(0, n, 2 * blk):
        hr[s:s + blk] = 1.0
    restart_inits.append(hr)
    # Sinusoidal freq 2
    restart_inits.append(np.clip(0.5 + 0.45 * np.sin(4 * np.pi * np.arange(n) / n), 0, 1))
    # Sinusoidal freq 5
    restart_inits.append(np.clip(0.5 + 0.45 * np.sin(10 * np.pi * np.arange(n) / n + 0.7), 0, 1))
    # Golden ratio
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    fr = np.mod(np.arange(n, dtype=float) * phi, 1.0)
    restart_inits.append((fr < 0.5).astype(float))
    # Perturbed parent (moderate)
    restart_inits.append(best_h + 0.15 * rng.randn(n))

    for hr in restart_inits:
        if not _can_go(8):
            break
        hr = project_feasible(hr, n)
        for alpha in [500, 5000, 50000, 500000, 5000000]:
            if not _can_go(5):
                break
            hr = slsqp_round(hr, n, alpha, maxiter=2500, ftol=1e-15)
            c = compute_c5(hr, n)
            if c < best_c:
                best_c, best_h = c, hr.copy()

    # Phase 4: Multi-resolution
    for nh in [300, 500, 600]:
        if not _can_go(20):
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hn = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [5000, 50000, 500000, 5000000]:
            if not _can_go(8):
                break
            hn = slsqp_round(hn, nh, alpha, maxiter=2500, ftol=1e-15)
            cn = compute_c5(hn, nh)
            if cn < best_c:
                best_c, best_h, n = cn, hn.copy(), nh

    # Phase 5: Small perturbation polish
    for trial in range(4):
        if not _can_go(5):
            break
        hq = best_h + 0.005 * rng.randn(n)
        hq = project_feasible(hq, n)
        for alpha in [5000000, 50000000]:
            if not _can_go(3):
                break
            hq = slsqp_round(hq, n, alpha, maxiter=3000, ftol=1e-16)
            cq = compute_c5(hq, n)
            if cq < best_c:
                best_c, best_h = cq, hq.copy()

    # Phase 6: Final coordinate descent + SLSQP
    if _can_go(10):
        hc, _ = coord_descent_quick(best_h, n, max_sweeps=150, deadline=deadline - 8)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()
        if _can_go(5):
            hc = slsqp_round(hc, n, 10000000, maxiter=2000, ftol=1e-16)
            cc = compute_c5(hc, n)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
