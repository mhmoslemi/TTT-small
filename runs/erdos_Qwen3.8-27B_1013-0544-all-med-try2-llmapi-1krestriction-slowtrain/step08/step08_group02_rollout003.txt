    target=0.5*n

        if abs(diff)<1e-13: break
        if diff>0:
            room=1.0-h
            mask=room>1e-14
            if not np.any(mask): break

            if tot<=0: break
            h[mask]+=diff/tot*room[mask]

            if tot<=0: break

def smooth_max_grad_C(x,n,alpha):

    c=np.correlate(x,1.0-x,mode='full')*dx
    m=c.max()

import numpy as np
import time
import math
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def current_c(c, n):
    return float(np.max(c)) * (2.0 / n)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            mask = room > 1e-14
            if not np.any(mask):
                break
            tot = room[mask].sum()
            if tot <= 0.0:
                break
            h[mask] += (diff / tot) * room[mask]
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            tot = h[mask].sum()
            if tot <= 0.0:
                break
            h[mask] += (diff / tot) * h[mask]
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = int(np.argmax(h))
            dec = min(-diff, h[idx])
            h[idx] -= dec
    return np.clip(h, 0.0, 1.0)


def smooth_max_val_C(x, n, alpha):
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(cv.max())
    z = alpha * (cv - m)
    sm = z.max()
    s = np.sum(np.exp(z - sm))
    return float(m + math.log(s) / alpha)


def smooth_max_grad_C(x, n, alpha):
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(cv.max())
    z = alpha * (cv - m)
    sm = z.max()
    w = np.exp(z - sm)
    tot = w.sum()
    if tot <= 0.0:
        return np.zeros(n)
    w /= tot
    g = np.zeros(n)
    for idx in np.nonzero(w > 1e-12)[0]:
        e = int(idx - (n - 1))
        wm = float(w[idx])
        if e > 0:
            nv = n - e
            g[e:] += wm * (1.0 - x[:nv])
            g[:nv] += -wm * x[e:]
        elif e < 0:
            s = -e
            nv = n - s
            g[:nv] += wm * (1.0 - x[s:])
            g[s:] += -wm * x[:nv]
        else:
            g[:] += wm * (1.0 - 2.0 * x)
    return g * dx


def pen_obj(x, n, alpha, mu):
    return smooth_max_val_C(x, n, alpha) + mu * (float(x.sum()) - 0.5 * n) ** 2


def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad_C(x, n, alpha)
    return g + 2.0 * mu * (float(x.sum()) - 0.5 * n)


def lbfgs_run(h, n, deadline, alphas=(50000.0,), max_iters=50):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if max_iters <= 0 or time.time() > deadline - 3:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(
                pen_obj, h, jac=pen_grad, args=(n, float(alpha), 100.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': int(max_iters), 'ftol': 1e-17,
                         'gtol': 1e-13, 'maxfun': int(max(100, 20 * max_iters))}
            )
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                best_h = hn.copy()
                h = hn
        except Exception:
            pass
    return best_h, best_c


def slsqp_obj(x, n, alpha):
    return smooth_max_val_C(x, n, alpha)


def slsqp_eq(x, n):
    return float(x.sum()) - 0.5 * n


def slsqp_jac(x, n):
    return np.ones(n)


def slsqp_run(h, n, deadline, alpha=50000.0, maxiter=25):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if maxiter <= 0 or time.time() > deadline - 6:
        return best_h, best_c
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]
        res = minimize(
            slsqp_obj, h, args=(n, float(alpha)), method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-15}
        )
        hn = project_feasible(res.x, n)
        cn = compute_c5(hn, n)
        if cn < best_c:
            return hn, cn
    except Exception:
        pass
    return best_h, best_c


def update_c_single(c, h, p, delta, n):
    if delta == 0.0:
        return
    c[n - 1] += delta * (1.0 - 2.0 * h[p] - delta)

    if p > 0:
        c[n:n + p] += delta * (1.0 - h[p - 1:0:-1])
    L = n - 1 - p
    if L > 0:
        c[n:n + L] -= delta * h[p + 1:p + L + 1]

    L = n - 1 - p
    if L > 0:
        c[n - 1 - L:n - 1] += delta * (1.0 - h[n - 1:p:-1])
    L = p
    if L > 0:
        c[n - 1 - L:n - 1] -= delta * h[:L]


def trial_pair_c(c, h, p, q, s, n, t):
    newp = float(t)
    newq = s - newp
    if newp < 0.0 or newp > 1.0 or newq < 0.0 or newq > 1.0:
        return float('inf')
    dp = newp - h[p]
    dq = newq - h[q]
    if abs(dp) < 1e-14 and abs(dq) < 1e-14:
        return current_c(c, n)
    c_t = c.copy()
    oldp = h[p]
    if abs(dp) > 1e-14:
        update_c_single(c_t, h, p, dp, n)
    if abs(dq) > 1e-14:
        h[p] = newp
        try:
            update_c_single(c_t, h, q, dq, n)
        finally:
            h[p] = oldp
    return current_c(c_t, n)


def refine_line(c, h, p, q, s, n, a, b, bestx, bestf, iters):
    if iters <= 0 or b - a <= 1e-12:
        return bestx, bestf
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = a + gr * (b - a)
    x2 = b - gr * (b - a)
    f1 = trial_pair_c(c, h, p, q, s, n, x1)
    f2 = trial_pair_c(c, h, p, q, s, n, x2)
    if f1 < bestf:
        bestx, bestf = x1, f1
    if f2 < bestf:
        bestx, bestf = x2, f2
    for _ in range(iters):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = a + gr * (b - a)
            f1 = trial_pair_c(c, h, p, q, s, n, x1)
            if f1 < bestf:
                bestx, bestf = x1, f1
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = b - gr * (b - a)
            f2 = trial_pair_c(c, h, p, q, s, n, x2)
            if f2 < bestf:
                bestx, bestf = x2, f2
    mid = 0.5 * (a + b)
    fm = trial_pair_c(c, h, p, q, s, n, mid)
    if fm < bestf:
        bestx, bestf = mid, fm
    return bestx, bestf


def line_search_pair(c, h, p, q, s, n, baseC, refine_iters=6):
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:
        return None
    oldp = float(h[p])
    oldp = min(max(oldp, lo), hi)
    bestx = oldp
    bestf = baseC
    pts = [lo, hi, oldp, lo + 0.25 * (hi - lo), lo + 0.5 * (hi - lo), lo + 0.75 * (hi - lo)]
    seen = []
    for t in pts:
        t = float(t)
        if t < lo - 1e-12 or t > hi + 1e-12:
            continue
        t = min(max(t, lo), hi)
        if any(abs(t - u) <= 1e-10 for u in seen):
            continue
        seen.append(t)
        f = trial_pair_c(c, h, p, q, s, n, t)
        if f < bestf:
            bestf = f
            bestx = t
    if bestf < baseC - 1e-13:
        rad = 0.25 * (hi - lo)
        a = max(lo, bestx - rad)
        b = min(hi, bestx + rad)
        if b - a > 1e-10:
            bestx, bestf = refine_line(c, h, p, q, s, n, a, b, bestx, bestf, refine_iters)
    if bestf < baseC - 1e-12:
        return bestx, bestf
    return None


def apply_pair(h, c, p, q, s, n, t):
    newp = float(t)
    newq = s - newp
    dp = newp - h[p]
    if abs(dp) > 1e-14:
        update_c_single(c, h, p, dp, n)
        h[p] = newp
    dq = newq - h[q]
    if abs(dq) > 1e-14:
        update_c_single(c, h, q, dq, n)
        h[q] = newq


def top_lags(c, r):
    els = len(c)
    if r <= 0:
        return np.array([], dtype=int)
    if r >= els:
        return np.arange(els)
    idx = np.argpartition(c, -r)[-r:]
    return idx[np.argsort(c[idx])[::-1]]


def guided_pairs(c, h, n, total, rng, max_per_lag=24):
    pairs = []
    seen = set()

    maxc = float(c.max())
    if maxc > 0.0 and c[n - 1] >= maxc - max(1e-9, 1e-4 * maxc):
        highs = np.flatnonzero(h > 0.5 + 1e-5)
        lows = np.flatnonzero(h < 0.5 - 1e-5)
        if len(highs) and len(lows):
            m = min(12, len(highs), len(lows))
            hi = highs[rng.choice(len(highs), m, replace=False)]
            lo = lows[rng.choice(len(lows), m, replace=False)]
            for a, b in zip(hi, lo):
                key = (int(a), int(b))
                if key not in seen:
                    seen.add(key)
                    pairs.append(key)
                    if len(pairs) >= total:
                        return pairs

    for m in top_lags(c, 6):
        e = int(m - (n - 1))
        if e == 0:
            continue
        if e > 0:
            nv = n - e
            score = h[e:] - h[:nv]
            inds = np.flatnonzero(score > 1e-5)
            if len(inds):
                if len(inds) > max_per_lag:
                    inds = inds[np.argpartition(score[inds], -max_per_lag)[-max_per_lag:]]
                for i in inds:
                    p = int(i + e)
                    q = int(i)
                    key = (p, q)
                    if p != q and key not in seen:
                        seen.add(key)
                        pairs.append(key)
                        if len(pairs) >= total:
                            return pairs
        else:
            s = -e
            nv = n - s
            score = h[:nv] - h[s:]
            inds = np.flatnonzero(score > 1e-5)
            if len(inds):
                if len(inds) > max_per_lag:
                    inds = inds[np.argpartition(score[inds], -max_per_lag)[-max_per_lag:]]
                for i in inds:
                    p = int(i)
                    q = int(i + s)
                    key = (p, q)
                    if p != q and key not in seen:
                        seen.add(key)
                        pairs.append(key)
                        if len(pairs) >= total:
                            return pairs

    for _ in range(20 * total):
        p = int(rng.randint(n))
        q = int(rng.randint(n))
        if p == q:
            continue
        s = h[p] + h[q]
        width = min(1.0, s) - max(0.0, s - 1.0)
        if width < 1e-6:
            continue
        key = (p, q)
        if key in seen:
            continue
        seen.add(key)
        pairs.append(key)
        if len(pairs) >= total:
            break
    return pairs


def pair_polish(h, n, block_deadline, rng, total_pairs=160, refine_iters=6):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    C = current_c(c, n)
    best_h = h.copy()
    best_c = C
    no_improve = 0

    while time.time() < block_deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        C = current_c(c, n)
        if C < best_c:
            best_c = C
            best_h = h.copy()
        pairs = guided_pairs(c, h, n, total_pairs, rng)
        if not pairs:
            break
        improved = False
        for p, q in pairs:
            if time.time() > block_deadline:
                break
            s = h[p] + h[q]
            step = line_search_pair(c, h, p, q, s, n, C, refine_iters)
            if step is not None:
                xt, _ = step
                apply_pair(h, c, p, q, s, n, xt)
                C = current_c(c, n)
                if C < best_c:
                    best_c = C
                    best_h = h.copy()
                improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    return best_h, best_c


def to_binary(h, n, rng):
    hb = (h >= 0.5).astype(float)
    target = n // 2
    cnt = int(round(hb.sum()))
    if cnt > target:
        ones = np.flatnonzero(hb == 1)
        if len(ones):
            rem = cnt - target
            rem = min(rem, len(ones))
            idx = ones[np.argsort(h[ones])[:rem]]
            hb[idx] = 0.0
    elif cnt < target:
        zeros = np.flatnonzero(hb == 0)
        if len(zeros):
            add = target - cnt
            add = min(add, len(zeros))
            idx = zeros[np.argsort(h[zeros])[::-1][:add]]
            hb[idx] = 1.0
    return hb


def binary_directed(h, n, block_deadline, rng, max_tries=8000):

    best_c = C
    temp = 0.0045
    stagnant = 0
    it = 0

    while time.time() < block_deadline and it < max_tries:

        found_improve = False

        for m in top_lags(c, 4):

                s2 = h[p] + h[q]

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def fourier_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for f in range(1, 7):
        amp = rng.uniform(0.15, 0.55)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(2.0 * np.pi * f * x / 2.0 + phase)
    h = 0.5 + h
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def perturb_h(h, n, rng, scale=0.08):
    h2 = h.copy()
    k = max(1, int(0.2 * n))
    idx = rng.choice(n, k, replace=False)
    h2[idx] += rng.uniform(-scale, scale, size=len(idx))
    return project_feasible(h2, n)

                best_c, best_h = c2, h2

    if time.time() < deadline - 180:
        t = min(time.time() + min(25.0, 0.06 * (deadline - time.time())), deadline - 160.0)
        if t > time.time() + 4.0:
            hb, cb = binary_directed(best_h, n, t, rng, max_tries=5000)
            if cb < best_c:
                best_c, best_h = cb, hb

    if time.time() < deadline - 60:
        alloc = max(30.0, 0.36 * (deadline - time.time()))
        t = min(time.time() + alloc, deadline - 80.0)
        if t > time.time() + 5.0:
            h2, c2 = pair_polish(best_h, n, t, rng, total_pairs=170, refine_iters=6)
            if c2 < best_c:
                best_c, best_h = c2, h2

    mode = 0
    while time.time() < deadline - 85:
        rem = deadline - time.time()
        if rem < 45.0:
            break
        if mode == 0:
            cand = perturb_h(best_h, n, rng, 0.09)
        elif mode == 1:
            cand = to_binary(best_h, n, rng)
        elif mode == 2:
            cand = fourier_start(n, rng)
        elif mode == 3:
            cand = random_binary(n, rng)
        else:
            cand = perturb_h(best_h, n, rng, 0.18)

        h2 = project_feasible(cand, n)
        if mode in (1, 3):
            tb = min(time.time() + max(8.0, 0.18 * rem), deadline - 55.0)
            if tb > time.time() + 3.0:
                h2, _ = binary_directed(h2, n, tb, rng, max_tries=2000)

        tp = min(time.time() + max(14.0, 0.38 * rem), deadline - 45.0)
        if tp > time.time() + 4.0:
            h2, c2 = pair_polish(h2, n, tp, rng, total_pairs=130, refine_iters=5)
            if c2 < best_c:
                best_c, best_h = c2, h2
        mode += 1

    if time.time() < deadline - 70:
        t = min(time.time() + 25.0, deadline - 45.0)
        if t > time.time() + 4.0:
            h2, c2 = lbfgs_run(best_h, n, t, alphas=(100000.0,), max_iters=35)
            if c2 < best_c:
                best_c, best_h = c2, h2

    if time.time() < deadline - 45:
        t = min(time.time() + 25.0, deadline - 15.0)
        if t > time.time() + 5.0:
            h2, c2 = slsqp_run(best_h, n, t, alpha=80000.0, maxiter=18)
            if c2 < best_c:
                best_c, best_h = c2, h2

    if time.time() < deadline - 5:
        h2, c2 = pair_polish(best_h, n, deadline - 4.0, rng, total_pairs=120, refine_iters=6)
        if c2 < best_c:
            best_c, best_h = c2, h2

    if abs(float(best_h.sum()) - 0.5 * n) > 1e-9:
        best_h = project_feasible(best_h, n)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, int(n)
