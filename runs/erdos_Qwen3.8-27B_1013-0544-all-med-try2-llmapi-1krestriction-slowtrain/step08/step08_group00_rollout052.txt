        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Active set L-BFGS-B on parent grid
    t1 = min(t0 + 150, deadline - 500)
    best_h, best_c = active_lbfgs(best_h, best_n, t1, max_active=30)

    # Phase 2: MINMAX refinement
    if time.time() < deadline - 350:
        t2 = min(time.time() + 60, deadline - 350)
        h, c = minmax_refine(best_h, best_n, t2, max_iters=8000, rng=rng)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Active set L-BFGS-B with larger active set
    if time.time() < deadline - 280:
        t3 = min(time.time() + 70, deadline - 280)
        h, c = active_lbfgs(best_h, best_n, t3, max_active=60)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 4: Try n=800 (finer grid)
    if time.time() < deadline - 200:
        n_fine = 800
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_fine, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_fine)
        alloc = min(100, (deadline - time.time() - 120) * 0.5)
        if alloc > 40:
            h_t, c_t = active_lbfgs(h_interp, n_fine, time.time() + alloc, max_active=50)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_fine
            if time.time() < deadline - 80:
                h_t, c_t = minmax_refine(best_h, best_n, min(time.time() + 30, deadline - 80), max_iters=4000, rng=rng)
                if c_t < best_c:
                    best_c, best_h = c_t, h_t

    # Phase 5: Try n=1000
    if time.time() < deadline - 130:
        n_1k = 1000
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_1k, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_1k)
        alloc = min(70, (deadline - time.time() - 70) * 0.45)
        if alloc > 35:
            h_t, c_t = active_lbfgs(h_interp, n_1k, time.time() + alloc, max_active=50)
            if time.time() < deadline - 50:
                h_t, c_t = minmax_refine(h_t, n_1k, min(time.time() + 20, deadline - 50), max_iters=3000, rng=rng)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_1k

    # Phase 6: Gradient-informed local search
    if time.time() < deadline - 25:
        best_h, best_c = grad_local_search(best_h, best_n, deadline - 12, k_top=50)

    # Phase 7: Final active set + full smooth
    if time.time() < deadline - 3:
        h_f, c_f = active_lbfgs(best_h, best_n, deadline - 2, max_active=80)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
