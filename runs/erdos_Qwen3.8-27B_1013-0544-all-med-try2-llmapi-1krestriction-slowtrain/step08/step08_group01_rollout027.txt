            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_phase(h, n, deadline, alpha_list, max_iters=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time() - 2
        if remaining < 5:
            break
        mu = 10.0 * alpha
        try:
            res = minimize(penalty_obj, best_h, jac=penalty_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-18,
                                    'gtol': 1e-16, 'maxfun': max_iters * 40,
                                    'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=100000):
    bounds = [(0.0, 1.0)] * n
    c0 = n / 2.0
    def eq_fun(x):
        return float(x.sum() - c0)
    def eq_jac(x):
        return np.ones_like(x)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_guided_pair(h, n, deadline, top_k=20):
    best_c = compute_c5(h, n)
    c_all = compute_all_c(h, n)
    active_set = np.argsort(c_all)[-top_k:]
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in steps:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            c_all = compute_all_c(h, n)
            cur_max = c_all.max()
            active_set = np.argsort(c_all)[-top_k:]
            # Compute gradient
            g = smooth_max_grad(h, n, 100000)
            # Sort by gradient: move mass from high g to low g
            order = np.argsort(g)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                src = order[i]  # low gradient, receive mass
                for j in range(i + 1, min(n, i + 80)):
                    dst = order[j]  # high gradient, give mass
                    if h[dst] < d or h[src] > 1.0 - d:
                        continue
                    h[dst] -= d
                    h[src] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        any_imp = True
                    else:
                        h[dst] += d
                        h[src] -= d
    return h, best_c

def exact_coord_pair(h, n, deadline, top_k=10):
    """For each pair, find the optimal transfer amount via 1D search."""
    best_c = compute_c5(h, n)
    c_all = compute_all_c(h, n)
    active = np.argsort(c_all)[-top_k:]
    max_val = c_all[active].max()
    n_tries = 0
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        c_all = compute_all_c(h, n)
        active = np.argsort(c_all)[-top_k:]
        max_val = c_all[active].max()
        g = smooth_max_grad(h, n, 50000)
        order = np.argsort(g)
        for i in range(n):
            if time.time() > deadline or n_tries > 50000:
                return h, best_c
            src = order[i]
            for j in range(i + 1, min(n, i + 60)):
                dst = order[j]
                if g[dst] - g[src] < 1e-8:
                    continue
                if h[dst] < 1e-6 or h[src] > 1.0 - 1e-6:
                    continue
                # Binary search for optimal transfer
                lo, hi = 0.0, min(h[dst], 1.0 - h[src])
                if hi < 1e-7:
                    continue
                best_t = 0.0
                best_obj = best_c
                for _ in range(8):
                    mid = (lo + hi) / 2
                    h[dst] -= mid
                    h[src] += mid
                    c_test = compute_c5(h, n)
                    n_tries += 1
                    if c_test < best_obj:
                        best_obj = c_test
                        best_t = mid
                        lo = mid
                    else:
                        hi = mid
                    if hi - lo < 1e-6:
                        break
                h[dst] -= best_t
                h[src] += best_t
                if best_obj < best_c - 1e-15:
                    best_c = best_obj
                    improved_any = True
    return h, best_c

def multi_start_h(n, rng, variant):
    if variant == 0:
        h = np.zeros(n)
        half = n // 2
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        pos = np.floor(np.arange(half) * phi).astype(int) % n
        h[pos] = 1.0
    elif variant == 1:
        h = np.zeros(n)
        half = n // 2
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
    elif variant == 2:
        h = np.full(n, 0.5)
        x = np.arange(n)
        for freq in range(1, 5):
            amp = rng.uniform(0.1, 0.3)
            ph = rng.uniform(0, 2 * np.pi)
            h += amp * np.cos(2 * np.pi * freq * x / n + ph)
    elif variant == 3:
        h = np.full(n, 0.5)
        x = np.arange(n)
        h += 0.2 * np.sin(np.pi * x / n)
        h += 0.1 * np.cos(2 * np.pi * x / n)
    else:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.15
    return project_feasible(h, n)

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS with increasing alpha
    alphas_low = [50, 200, 500, 2000, 10000]
    best_h, best_c = lbfgs_phase(best_h, n, min(t0 + 120, deadline - 500), alphas_low, max_iters=200)

    # Phase 2: Higher alpha L-BFGS
    alphas_high = [50000, 200000, 1000000, 5000000]
    best_h, best_c = lbfgs_phase(best_h, n, min(time.time() + 150, deadline - 350), alphas_high, max_iters=150)

    # Phase 3: SLSQP polish
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        h3, c3 = slsqp_phase(best_h, n, t3, alpha=500000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 100:
        t4 = min(time.time() + 120, deadline - 100)
        best_h, best_c = grad_guided_pair(best_h, n, t4, top_k=15)

    # Phase 5: Exact coordinate pair optimization
    if time.time() < deadline - 60:
        t5 = min(time.time() + 100, deadline - 60)
        best_h, best_c = exact_coord_pair(best_h, n, t5, top_k=10)

    # Phase 6: Final L-BFGS with very high alpha
    if time.time() < deadline - 40:
        t6 = min(time.time() + 50, deadline - 40)
        best_h, best_c = lbfgs_phase(best_h, n, t6, [5000000, 20000000], max_iters=100)

    # Phase 7: Final SLSQP
    if time.time() < deadline - 20:
        t7 = min(time.time() + 30, deadline - 20)
        h7, c7 = slsqp_phase(best_h, n, t7, alpha=10000000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Very fine pair polish
    if time.time() < deadline - 5:
        t8 = deadline - 3
        best_h, best_c = grad_guided_pair(best_h, n, t8, top_k=10)

    # Phase 9: Multi-start exploration (if time remains)
    if time.time() < deadline - 30:
        remaining = deadline - time.time() - 5
        for variant in range(4):
            if time.time() > deadline - 30:
                break
            h_rs = multi_start_h(n, rng, variant)
            alloc = min(remaining * 0.25, 60)
            if alloc < 15:
                break
            h_rs, c_rs = lbfgs_phase(h_rs, n, time.time() + alloc * 0.7,
                                     [1000, 10000, 100000], max_iters=100)
            if c_rs < best_c:
                best_c = c_rs
                best_h = h_rs

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
