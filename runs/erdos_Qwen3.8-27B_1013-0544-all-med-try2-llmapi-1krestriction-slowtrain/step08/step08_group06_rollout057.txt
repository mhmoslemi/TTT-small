    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        lag = m_idx - n + 1
        if lag >= 0:
            idx = n - lag
            if idx > 0:
                grad[:idx] += w * (1.0 - h[lag:])
            if lag > 0:
                grad[lag:] -= w * h[:idx]
        else:
            al = -lag
            idx = n - al
            if idx > 0:
                grad[al:] += w * (1.0 - h[:idx])
            grad[:idx] -= w * h[al:]
    return grad

    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    return m + np.log(np.exp(shifted).sum()) / alpha

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        cnt = n - lag
        g[:cnt] += (1.0 - h[lag:])
        if lag > 0:
            g[lag:] -= h[:cnt]
    else:
        a = -lag
        cnt = n - a
        g[a:] += (1.0 - h[:cnt])
        g[:cnt] -= h[a:]
    return g, corr.max()

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    w = np.exp(sh)
    w /= w.sum()
    thresh = w.max() * 1e-5
    active = np.where(w > thresh)[0]
    grad = np.zeros(n)
    for mi in active:
        wi = w[mi]
        lag = mi - (n - 1)
        if lag >= 0:
            cnt = n - lag
            grad[:cnt] += wi * (1.0 - h[lag:])
            if lag > 0:
                grad[lag:] -= wi * h[:cnt]
        else:
            a = -lag
            cnt = n - a
            grad[a:] += wi * (1.0 - h[:cnt])
            grad[:cnt] -= wi * h[a:]
    return grad

def gradient_descent(h, n, lr, deadline, mode='subgrad', alpha=1000, noise=0.0, momentum=0.85):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_lr = lr
    rng = np.random.RandomState(int(time.time()) % 10000)
    step = 0
    stall = 0
    while time.time() < deadline:
        if mode == 'subgrad':
            g, _ = subgrad_step(h, n)
        else:
            g = lse_grad(h, n, alpha)
        v = momentum * v + (1.0 - momentum) * g
        cur_lr = base_lr / (1.0 + step * 0.005)
        h_new = h - cur_lr * v
        if noise > 0:
            h_new += noise * rng.randn(n)
        h = project_feasible(h_new, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        step += 1
    return best_h, best_c

def binary_anneal(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = 0.003
    cool = 0.99997
    scale = 10000.0
    while time.time() < deadline:
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = int(zeros[ai])
        j = int(ones[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
