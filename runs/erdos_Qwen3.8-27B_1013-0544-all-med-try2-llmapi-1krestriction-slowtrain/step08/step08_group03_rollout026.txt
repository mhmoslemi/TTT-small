Actually, let me reconsider the pair polish. The issue is that it's very slow. Let me use a smarter approach:
- Instead of trying all pairs in a neighborhood, use the gradient to identify the best pairs
- The gradient tells us where to add and remove mass
- So: find the position with the most negative gradient (add mass) and the most positive gradient (remove mass)
- This is O(n) per step instead of O(n * radius)

This is essentially what the PGD does, but with a "greedy" selection of the best pair at each step.

Let me call this "greedy pair descent":
1. Compute gradient g
2. Find i* = argmax(g[i] * (1-h[i])) (best place to add)
   Wait, we want to decrease the objective. The objective change from moving mass from j to i is:
   Δ ≈ d * (g[i] - g[j]) where d is the amount moved.
   We want Δ < 0, so we want g[i] < g[j].
   Best pair: i = argmin(g), j = argmax(g), with the constraint that h[i]+d ≤ 1 and h[j]-d ≥ 0.

This is O(n) per step and much faster than the neighborhood search.

Let me implement this as "greedy descent":

```python
def greedy_descent(h, n, deadline, d=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        g = smooth_max_grad(best_h, n, 500.0)
        # Find best pair: minimize g[i] - g[j] subject to feasibility
        # We want to move mass from j to i: h[i] += d, h[j] -= d
        # Objective change ≈ d * (g[i] - g[j])
        # We want g[i] - g[j] < 0, i.e., g[i] < g[j]
        # Best: i = argmin g (among those with h[i] + d ≤ 1)
        #       j = argmax g (among those with h[j] - d ≥ 0)
        
        candidates_i = np.where(best_h + d <= 1.0)[0]
        candidates_j = np.where(best_h - d >= 0.0)[0]
        
        if len(candidates_i) == 0 or len(candidates_j) == 0:
            break
        
        gi = g[candidates_i]
        gj = g[candidates_j]
        
        i_best = candidates_i[np.argmin(gi)]
        j_best = candidates_j[np.argmax(gj)]
        
        if g[i_best] >= g[j_best]:
            # No beneficial move
            break
        
        h_new = best_h.copy()
        h_new[i_best] += d
        h_new[j_best] -= d
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
        else:
            break
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad * (2.0 / n)

def all_grads(h, n):
    """Compute gradients for all lags. Returns array of shape (2n-1, n)."""
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        G[idx] = grad_single(h, n, k)
    return G

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    top_k = min(100, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(len(x))
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (len(x) - 1)
        grad += w[i] * grad_single(x, len(x), k)
    return grad

def trust_region_lp(h, n, delta, max_iter):
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    best_h = h_cur.copy()
    for it in range(max_iter):
        c = compute_c5_vec(h_cur, n)
        G = all_grads(h_cur, n)
        n_lags = 2 * n - 1
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = c - G @ h_cur
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(max(0.0, h_cur[i] - delta), min(1.0, h_cur[i] + delta)) for i in range(n)]
        bounds.append((0.0, None))
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status == 0:
                h_new = project_feasible(res.x[:n], n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_cur = h_new
                else:
                    delta *= 0.5
                    if delta < 0.003:
                        break
            else:
                break
        except Exception:
            break
    return best_h, best_c

def nesterov_pgd(h, n, deadline, alpha=200.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    vel = np.zeros(n)
    step = 0.02
    mu = 0.9
    stall = 0
    while time.time() < deadline:
        g = smooth_max_grad(h, n, alpha)
        vel = mu * vel + g
        h_new = project_feasible(h - step * vel, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        elif c_new < best_c + 1e-12:
            h = h_new
            stall += 1
        else:
            stall += 1
            step *= 0.7
        if stall > 80:
            break
    return best_h, best_c

def pair_polish(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
```
