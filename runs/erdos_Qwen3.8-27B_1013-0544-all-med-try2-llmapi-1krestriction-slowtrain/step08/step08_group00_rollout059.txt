        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-14: break
            h += (d/t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d/t) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()

    sm = s.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-14: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(-n + 1, n):
        wi = w[n - 1 + k] * dx
        if wi < 1e-15: continue
        if k >= 0:
            nv = n - k
            g[:nv] += wi * (1.0 - x[k:])
            g[k:] -= wi * x[:nv]
        else:
            kk = -k; nv = n - kk
            g[kk:] += wi * (1.0 - x[:nv])
            g[:nv] -= wi * x[kk:]
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)

def _slsqp_grad(x, n, alpha):
    return smooth_grad(x, n, alpha, 0.0)

def lbfgs_opt(h, n, deadline, alphas):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3: break
        mu = max(0.01, alpha * 0.005)
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-18, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=20000000, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), jac=_slsqp_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def proj_grad(h, n, alpha, deadline, max_iters=20000):
    best_c = compute_c5(h, n)
    for _ in range(max_iters):
        if time.time() > deadline - 1: break
        g = smooth_grad(h, n, alpha, 0.0)
        gc = g - g.mean()
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h + step * gc, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h, best_c = h_new, c_new
                improved = True
                break
        if not improved:
            # Try individual coordinate steps
            gi = np.argsort(gc)[::-1]
            gj = np.argsort(gc)
            for a in range(min(30, n)):
                for b in range(min(30, n)):
                    i, j = gi[a], gj[b]
                    if i == j: continue
                    for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                        if step <= 1.0 - h[i] and step <= h[j]:
                            h[i] += step; h[j] -= step
                            c2 = compute_c5(h, n)
                            if c2 < best_c - 1e-16:
                                best_c = c2; improved = True
                                break
                            else:
                                h[i] -= step; h[j] += step
                        if step <= 1.0 - h[j] and step <= h[i]:
                            h[j] += step; h[i] -= step
                            c2 = compute_c5(h, n)
                            if c2 < best_c - 1e-16:
                                best_c = c2; improved = True
                                break
                            else:
                                h[j] -= step; h[i] += step
                    if improved: break
                if improved: break
            if not improved: break
    return h, best_c

def pair_search(h, n, deadline, steps, radius):
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline - 2:
            any_imp = False
            for i in range(n):
                if time.time() > deadline - 2: return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j: continue
                    if d <= 1.0 - h[i] and d <= h[j]:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c: best_c = c; any_imp = True
                        else: h[i] -= d; h[j] += d
                    if d <= 1.0 - h[j] and d <= h[i]:
                        h[j] += d; h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c: best_c = c; any_imp = True
                        else: h[j] -= d; h[i] += d
            if not any_imp: break
    return h, best_c
