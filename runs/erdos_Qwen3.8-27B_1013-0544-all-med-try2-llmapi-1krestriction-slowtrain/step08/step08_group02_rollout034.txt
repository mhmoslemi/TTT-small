import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(len(w)):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_run(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    # Try increasing alpha for tighter approximation
    for alpha in [5000, 50000, 500000, 2000000]:
        if time.time() > deadline - 5:
            break
        mu = 100.0
        for _ in range(3):
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': max_iters, 'ftol': 1e-20,
                                      'gtol': 1e-16, 'maxfun': max_iters * 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
                mu *= 10
            except Exception:
                pass
    return h, best_c

                no_improve = 0
            else:
                temp *= 0.93
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h[i] = 0
        h[j] = 1
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 120000:
                nkick = rng.randint(5, 12)
                for _ in range(nkick):
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    a = oi[rng.randint(len(oi))]
                    b = zi[rng.randint(len(zi))]
                    h[a] = 0
                    h[b] = 1
                best_h = h.copy()
                best_c = compute_c5(h, n)
                no_improve = 0
                temp = 0.02
            elif rng.random() >= np.exp(-(c_new - best_c) / max(temp, 1e-10)):
                h[i] = 1
                h[j] = 0
        flips += 1
    return best_h, best_c

def continuous_sa(h_cont, n, deadline, rng, max_iters=200000):
    h = project_feasible(h_cont, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    temp = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            temp *= 0.95
        idx = rng.randint(n)
        delta = rng.uniform(-0.3, 0.3)
        h_new = h.copy()
        h_new[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-10)):
            h = h_new
        else:
            pass
    return best_h, best_c

def lbfgs_polish(h, n, deadline):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in [5000, 50000, 500000, 2000000]:
        if time.time() > deadline - 4:
            break
        mu = 50.0
        for _ in range(3):
            if time.time() > deadline - 4:
                break
            try:
                res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 400, 'ftol': 1e-22,
                                      'gtol': 1e-18, 'maxfun': 12000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
                mu *= 5.0
            except Exception:
                pass
    return h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def quad_binary(n, rng):
    h = np.zeros(n)
    for x in range(1, n):
        h[(x * x) % n] = 1.0
    return to_binary(h, n, rng)

def lfsr_binary(n, rng):
    h = np.zeros(n)
    state = rng.randint(1, 128)
    taps = [0, 3, 5]
    for i in range(n):
        h[i] = state & 1
        fb = 0
        for t in taps:
            fb ^= (state >> t) & 1
        state = (state >> 1) | (fb << 6)
    return to_binary(h, n, rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = random_binary(500, rng)
        n_init = 500
    n = n_init
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Multi-start quick SA (80s)
    t1 = min(t0 + 80, deadline - 600)
    starts = [to_binary(best_h, n, rng), random_binary(n, rng),
              quad_binary(n, rng), lfsr_binary(n, rng), random_binary(n, rng)]
    cands = []
    for s in starts:
        if time.time() > t1 - 4:
            break
        h_s, c_s = binary_sa(s, n, min(time.time() + 12, t1 - 4), rng, max_flips=300000)
        cands.append((c_s, h_s))
    cands.sort(key=lambda x: x[0])
    if cands and cands[0][0] < best_c:
        best_c = cands[0][0]
        best_h = cands[0][1]
    # Phase 2: Long SA on best (300s)
    t2 = min(time.time() + 300, deadline - 300)
    h_sa, c_sa = binary_sa(to_binary(best_h, n, rng), n, t2, rng, max_flips=15000000)
    if c_sa < best_c:
        best_c = c_sa
        best_h = h_sa
    # Phase 3: SA on 2nd best (120s)
    if len(cands) > 1 and time.time() < deadline - 200:
        t3 = min(time.time() + 120, deadline - 180)
        h_s2, c_s2 = binary_sa(cands[1][1], n, t3, rng, max_flips=8000000)
        if c_s2 < best_c:
            best_c = c_s2
            best_h = h_s2
    # Phase 4: L-BFGS-B polish (120s)
    if time.time() < deadline - 80:
        t4 = min(time.time() + 120, deadline - 60)
        h_p, c_p = lbfgs_polish(best_h, n, t4)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    # Phase 5: Try other n values (150s)
    n_options = sorted(set([n - 50, n + 50, n + 100, n + 200, n + 100]))
    for n_try in n_options:
        if n_try < 100 or n_try > 2000 or n_try % 2 != 0:
            continue
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        h_int_b = to_binary(h_int, n_try, rng)
        alloc = min(35, (deadline - time.time() - 35) * 0.4)
        if alloc < 10:
            continue
        h_t, c_t = binary_sa(h_int_b, n_try, time.time() + alloc, rng, max_flips=3000000)
        if time.time() < deadline - 20:
            h_t, c_t = lbfgs_polish(h_t, n_try, time.time() + 25)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            n = n_try
    # Phase 6: Final SA + polish
    if time.time() < deadline - 5:
        t6 = deadline - 3
        h_f, c_f = binary_sa(to_binary(best_h, n, rng), n, t6, rng, max_flips=8000000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
