    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def slsqp_polish(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def corr_delta(h, n, i, j, m_range):
    """Compute correlation change when h[i]: 0->1, h[j]: 1->0."""
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full')
    cur_e = corr.max() * dx
    best_h = h.copy()
    best_e = cur_e
    m_range = np.arange(-(n - 1), n)
    
    n_restarts = 30
    iters_per_restart = 150000
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        T = 0.04 * (0.75 ** r)
        if r > 0:
            h = best_h.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            cur_e = corr.max() * dx
        for it in range(iters_per_restart):
            if time.time() > deadline:
                break
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            i = zi[rng.randint(len(zi))]
            j = oi[rng.randint(len(oi))]
            delta = corr_delta(h, n, i, j, m_range)
            new_e = (corr + delta).max() * dx
            de = new_e - cur_e
            if de < 0 or rng.rand() < np.exp(-de / max(T, 1e-15)):
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_e = new_e
                if cur_e < best_e:
                    best_e = cur_e
                    best_h = h.copy()
    return best_h, best_e

def greedy_ls(h, n, deadline):

    corr = np.correlate(h, 1 - h, mode='full')
    cur_e = corr.max() * dx
    m_range = np.arange(-(n - 1), n)
    while time.time() < deadline:
        best_de = 0.0
        best_i, best_j = -1, -1

        zi = list(np.where(h == 0)[0])
        for i in zi:

        corr += best_delta
        cur_e += best_de
    return h, cur_e
