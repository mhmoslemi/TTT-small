def compute_c(h, n):
    # returns length (2n-1) array of c_k = sum h[n+k](1-h[n]) * dx, k=-(n-1)..n-1
    dx = 2.0/n
    P = np.empty(n+1); P[0]=0.0; P[1:]=np.cumsum(h)  # P[i]=sum h[:i]
    A = np.empty(2*n-1)
    # neg shifts idx 0..n-2 -> k=-(n-1)..-1 -> A=P[n+k], n+k in 1..n-1
    A[:n-1] = P[1:n]
    # nonneg idx n-1..2n-2 -> k=0..n-1 -> A = P[n]-P[k]
    A[n-1:] = P[n] - P[:n]
    # autocorr
    N = 1
    while N < 2*n: N <<= 1
    Fh = np.fft.rfft(h, N)
    acf = np.fft.irfft(Fh*np.conj(Fh), N)  # circular autocorr of zero-padded h to len N
    # R(k)=acf[|k|] for |k|<=n-1
    ks = np.arange(-(n-1), n)
    B = acf[np.abs(ks)]
    return (A - B)*dx

def grad_direct(h, n, w):
    # w length 2n-1, idx=k+(n-1)
    g = np.zeros(n)
    for idx in np.where(w>1e-14)[0]:
        k = idx-(n-1); wk=w[idx]
        if wk<=1e-14: continue
        if k>=0:
            # dc_k/dh_i = (1-h[i-k])1_{i-k in [0,n-1]} - h[i+k]1_{i+k in [0,n-1]}
            # i-k in range: i in k..n-1
            ii = np.arange(k, n)
            g[ii] += wk*(1-h[ii-k])
            # i+k in range: i in 0..n-1-k
            jj = np.arange(0, n-k)
            g[jj] -= wk*h[jj+k]
        else:
            kk=-k
            # i-k = i+kk in range: i in 0..n-1-kk
            ii = np.arange(0, n-kk)
            g[ii] += wk*(1-h[ii+kk])
            # i+k = i-kk in range: i in kk..n-1
            jj = np.arange(kk, n)
            g[jj] -= wk*h[jj-kk]
    return g

def proj_exact(z, n):
    target=0.5*n
    # find t in [tmin,tmax] such that sum clip(z+t)=target
    zt_max = 1.0 - z  # for clip zero at +inf: h=1 when z+t≥1
    # lower bound t: when sum=0 -> z+t≤0 all -> t ≤ -z.max(); but we can bisect between t_lo and t_hi.
    t_lo = 0.0 - (z.max()+1.0)   # ensures h=0
    t_hi = 1.0 - (z.min()-1.0)  # ensures h=1
    # Actually t_lo: to make all h=0 need t ≤ -z_i for all i -> t ≤ -max z. use t_lo = -z.max() - 1e-9? but then one might be exactly 0. Use t_lo=-z.max()-2, t_hi=1-z.min()+2.
    t_lo = -z.max() - 2.0
    t_hi = 1.0 - z.min() + 2.0
    for _ in range(60):
        t=0.5*(t_lo+t_hi)
        h=np.clip(z+t,0.0,1.0)
        s=h.sum()
        if s>target: t_hi=t
        else: t_lo=t
    t=0.5*(t_lo+t_hi)
    h=np.clip(z+t,0.0,1.0)
    # fix residual to exact 0 in double: adjust by tiny distrib
    d=target-h.sum()
    if abs(d)>1e-14:
        # spread d among interior coords
        mask=(h>0)&(h<1)
        if mask.sum()>0:
            h[mask]+=d/mask.sum()
            h=np.clip(h,0,1)
    return h

import numpy as np

USE_FFT = True


def compute_c(h, n):
    dx = 2.0 / n
    P = np.empty(n + 1); P[0] = 0.0; P[1:] = np.cumsum(h)
    A = np.empty(2 * n - 1)
    A[:n - 1] = P[1:n]
    A[n - 1:] = P[n] - P[:n]
    N = 1
    while N < 2 * n:
        N <<= 1
    Fh = np.fft.rfft(h, N)
    acf = np.fft.irfft(Fh * np.conj(Fh), N)
    ks = np.abs(np.arange(-(n - 1), n))
    B = acf[ks]
    return (A - B) * dx


def compute_c5_fast(h, n):
    return float(np.max(compute_c(h, n)))


def smoothmax_val(c, alpha):
    m = c.max()
    x = alpha * (c - m)
    return float(m + np.log(np.exp(x).sum()) / alpha)


def softmax_w(c, alpha):
    m = c.max()
    x = alpha * (c - m)
    e = np.exp(x - x.max())
    return e / e.sum()


def grad_fft(h, n, w):
    L = 1 << max(1, (3 * n - 2).bit_length())
    Fh = np.fft.rfft(h, L)
    Wp = np.zeros(L); Wp[:2 * n - 1] = w
    c2 = np.fft.irfft(np.fft.rfft(Wp, L) * Fh, L)
    J2 = c2[n - 1:2 * n - 1]
    W3p = np.zeros(L); W3p[:2 * n - 1] = w[::-1]
    c3 = np.fft.irfft(np.fft.rfft(W3p, L) * Fh, L)
    J3 = c3[n - 1:2 * n - 1]
    Wpre = np.zeros(2 * n); Wpre[1:] = np.cumsum(w)
    ii = np.arange(n)
    J1 = Wpre[ii + n] - Wpre[ii]
    return J1 - J2 - J3


def grad_direct(h, n, w):
    g = np.zeros(n)
    for idx in np.where(w > 1e-13)[0]:
        wk = w[idx]
        if wk <= 1e-13:
            continue
        k = int(idx) - (n - 1)
        if k >= 0:
            ii = np.arange(k, n)
            g[ii] += wk * (1.0 - h[ii - k])
            jj = np.arange(0, n - k)
            g[jj] -= wk * h[jj + k]
        else:
            kk = -k
            ii = np.arange(0, n - kk)
            g[ii] += wk * (1.0 - h[ii + kk])
            jj = np.arange(kk, n)
            g[jj] -= wk * h[jj - kk]
    return g


def grad_use(h, n, w):
    if USE_FFT:
        return grad_fft(h, n, w)
    return grad_direct(h, n, w)

def proj_exact(z, n):
    target = 0.5 * n
    z = np.clip(z, 0.0, 1.0)
    t_lo = -z.max() - 2.0
    t_hi = 1.0 - z.min() + 2.0
    for _ in range(48):
        t = 0.5 * (t_lo + t_hi)
        h = np.clip(z + t, 0.0, 1.0)

            t_hi = t
        else:
            t_lo = t

def _F(x, n, alpha, mu):
    c = compute_c(x, n)
    return float(smoothmax_val(c, alpha) + mu * (x.sum() - 0.5 * n) ** 2)


def _G(x, n, alpha, mu):
    c = compute_c(x, n)
    w = softmax_w(c, alpha)
    g = grad_use(x, n, w)
    return g + 2.0 * mu * (x.sum() - 0.5 * n)


def lbfgs_stage(h, n, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_F, h, jac=_G, args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16,
                                'maxfun': maxiter * 60})
        if hasattr(res, 'x') and res.x is not None:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h.copy()
