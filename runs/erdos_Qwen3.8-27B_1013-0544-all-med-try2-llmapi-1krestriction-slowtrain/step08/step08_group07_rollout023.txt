    c = np.correlate(h, 1 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-15:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0

def softobj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def softobj_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    obj = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    p = np.exp(shifted - sm)
    p /= p.sum()
    
    # grad[i] = dx * sum_m p[m] * [(1-x[i+m-(n-1)]) - x[i-m+(n-1)]]

        j2 = i - m_vals + (n - 1)

def coord_descent_fast(h, n, max_sweeps=300, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            
            # Find x in [0,1] that minimizes max(corr + B*x)
            # Bisection on subgradient
            vals0 = corr + B * 0.0
            vals1 = corr + B * 1.0
            m0 = vals0.max()
            m1 = vals1.max()
            
            # Check if x=0 is optimal: need all B[k] >= 0 for active k at x=0
            active0 = vals0 >= m0 - 1e-14
            sg_hi_at_0 = B[active0].max()
            active1 = vals1 >= m1 - 1e-14
            sg_lo_at_1 = B[active1].min()

                x_opt = 0.0

                x_opt = 1.0
            else:

                        hi = mid
                        break

                        lo = mid
                    else:
                        hi = mid

                corr += B * (x_opt - h[i])
                h[i] = x_opt

    return h

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-18):

    except Exception:

    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                i<j  # skip if i >= j to avoid duplicates? No, let's check all
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                # Correct for double counting at the same positions
                # When i and j are both changed, the delta at the cross terms
                # Actually, setting h[i]=1 and h[j]=0:
                # delta from h[i]: 0->1, delta from h[j]: 1->0
                # The cross term at m where both affect: need to subtract the interaction
                # For position m in delta:
                # from i: B_i[m] = (1-h[i+m-(n-1)]) - h[i-m+(n-1)]
                # from j: B_j[m] = -(1-h[j+m-(n-1)]) + h[j-m+(n-1)]
                # But when we set h[i]=1, it affects B_j (if i is in j's neighborhood)
                # And when we set h[j]=0, it affects B_i
                
                # The current delta already accounts for h[i]=0 and h[j]=1 (current state)
                # After the swap: h[i]=1, h[j]=0
                # The change in corr[m] = sum_i h[i](1-h[i+m-(n-1)])
                # = old_corr[m] + (1-h[i+m-(n-1)]) - h[i-m+(n-1)]  [from setting h[i]=1]
                #   - (1-h[j+m-(n-1)]) + h[j-m+(n-1)]  [from setting h[j]=0]
                # But setting h[i]=1 changes the terms where i appears as the second index:
                # term with h[i](1-h[i+k]): when i is in the (1-h[j]) part for j=i-m or j=i+m
                # This is the interaction term!
                
                # The interaction: if we set h[i]=1, it affects corr[m] through:
                # - The term h[i](1-h[i+m-(n-1)]): adds (1-h[i+m-(n-1)])
                # - The term where i appears in (1-h[j+m-(n-1)]): if i = j+m-(n-1) for some j
                #   i.e., m = i-j+(n-1), then (1-h[i]) changes from 1 to 0, contributing -h[j]
                #   But h[j] was 1 (before swap), so contribution is -1
                #   Wait, this is getting complicated.
                
                # Actually, the B-vector approach computes the change assuming only ONE variable
                # changes. When two variables change, there's an interaction term.
                
                # Let me recompute properly:
                # corr_new[m] - corr_old[m] = sum_i (h_new[i] - h_old[i]) * (1 - h_new[i+m-(n-1)])
                #   + sum_i h_old[i] * (h_old[i+m-(n-1)] - h_new[i+m-(n-1)])
                # = (h_new[i]-h_old[i]) * (1-h_new[i+m-(n-1)]) for i (the one that goes 0->1)
                #   + h_old[i] * (h_old[i+m-(n-1)] - h_new[i+m-(n-1)]) for i (the one that goes 0->1)
                #   + (h_new[j]-h_old[j]) * (1-h_new[j+m-(n-1)]) for j (the one that goes 1->0)
                #   + h_old[j] * (h_old[j+m-(n-1)] - h_new[j+m-(n-1)]) for j (the one that goes 1->0)
                
                # For i (0->1): delta_h = 1
                #   Term 1: 1 * (1 - h_new[i+m-(n-1)])
                #   Term 2: 0 * (...) = 0
                # For j (1->0): delta_h = -1
                #   Term 1: -1 * (1 - h_new[j+m-(n-1)])
                #   Term 2: 1 * (h_old[j+m-(n-1)] - h_new[j+m-(n-1)])
                
                # But h_new differs from h_old only at positions i and j.
                # So h_new[i+m-(n-1)] = h_old[i+m-(n-1)] unless i+m-(n-1) = i or j
                # If i+m-(n-1) = i: m = n-1, h_new[i] = 1 (was 0), so 1-h_new[i] = 0, 1-h_old[i] = 1
                # If i+m-(n-1) = j: m = j-i+n-1, h_new[j] = 0 (was 1), so 1-h_new[j] = 1, 1-h_old[j] = 0
                
                # This is getting very complex. The previous code's approach was:
                # delta = B_i + B_j (computed with the OLD h)
                # Then adjust for the interaction at m = n-1 (self) and m = n-1+(j-i), n-1+(i-j)
                
                # Let me use the same approach as the previous code:
                # delta[n-1] -= 2.0  (because at m=n-1, the "self" terms interact)
                # delta[n-1+(j-i)] += 1.0
                # delta[n-1+(i-j)] += 1.0
                
                # This accounts for the fact that when h[i] changes, it affects the
                # (1-h[.]) terms for positions that include i, and vice versa.
                
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
