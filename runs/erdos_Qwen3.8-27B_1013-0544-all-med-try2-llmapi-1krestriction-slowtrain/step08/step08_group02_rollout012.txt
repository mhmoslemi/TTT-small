```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            t = (1.0 - h)[m].sum()
            if t < 1e-15: break
            h[m] += d / t * (1.0 - h)[m]
        else:
            m = h > 0.0
            t = h[m].sum()
            if t < 1e-15: break
            h[m] *= 1.0 + d / t
    return np.clip(h, 0.0, 1.0)

def get_c(h, n):
    return np.correlate(h, 1 - h, mode='full')

def smooth_max_val(x, alpha):
    c = get_c(x, len(x))
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_max_grad(x, alpha):
    n = len(x)
    c = get_c(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15: continue
        s = i - (n - 1)
        if s >= 0:
            nv = n - s
            g[:nv] += wi * (1.0 - x[s:s+nv])
            g[s:s+nv] -= wi * x[:nv]
        else:
            t = -s
            nv = n - t
            g[t:t+nv] += wi * (1.0 - x[:nv])
            g[:nv] -= wi * x[t:t+nv]
    return g

def pen_obj(x, alpha, mu):
    n = len(x)
    return smooth_max_val(x, alpha) + mu * (x.sum() - n/2.0)**2

def pen_grad(x, alpha, mu):
    n = len(x)
    g = smooth_max_grad(x, alpha)
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

def lbfgs(h, n, deadline, cfgs):
    bh, bc = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu, maxit in cfgs:
        rem = deadline - time.time()
        if rem < 10: break
        mi = min(maxit, max(50, int(rem * 3)))
        try:
            r = minimize(pen_obj, bh, jac=pen_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': mi, 'ftol': 1e-20,
                                'gtol': 1e-16, 'maxfun': mi*50,
                                'maxls': 50, 'maxcor': 20})
            hn = project_feasible(r.x, n)
            cn = compute_c5(hn, n)
            if cn < bc:
                bc, bh = cn, hn
        except Exception:
            pass
    return bh, bc
```
