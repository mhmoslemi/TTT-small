    return best_h, best_c

def pair_search_fine(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] += d
    return h, best_c

def multi_n_search(h_best, n_best, c_best, deadline):
    """Try different resolutions."""
    for n_try in [200, 250, 350, 400]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(45, deadline - time.time() - 30)
        if alloc < 20:
            break
        h_t, c_t = adam_optimize(h_interp, n_try, time.time() + alloc * 0.5,
                                 alpha_start=50, alpha_end=50000, n_steps=100, lr=0.01)
        if time.time() < deadline - 35:
            h_t, c_t = lbfgs_phase(h_t, n_try, min(time.time() + 20, deadline - 35), alpha=100000)
        c_t = compute_c5(h_t, n_try)
        if c_t < c_best:
            c_best = c_t
            h_best = h_t
            n_best = n_try
    return h_best, c_best, n_best

def perturbation_escape(h, n, deadline, n_restarts=3):
    best_h, best_c = h.copy(), compute_c5(h, n)
    rng = np.random.RandomState(777)
    per_restart = max(20, (deadline - time.time()) / n_restarts)
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        t_end = time.time() + per_restart
        h_pert = h.copy()
        n_pert = max(10, n // 8)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx] = np.clip(h_pert[idx] + rng.uniform(-0.25, 0.25, n_pert), 0, 1)
        h_pert = project_feasible(h_pert, n)
        # Quick optimize
        h_pert, c_pert = adam_optimize(h_pert, n, min(time.time() + per_restart * 0.5, t_end - 5),
                                       alpha_start=50, alpha_end=10000, n_steps=80, lr=0.01)
        if time.time() < t_end - 3:
            h_pert, c_pert = lbfgs_phase(h_pert, n, min(time.time() + 15, t_end - 2), alpha=50000)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Adam with wide alpha range
    t1 = min(t0 + 120, deadline - 500)
    h_a, c_a = adam_optimize(h, n, t1, alpha_start=10, alpha_end=100000, n_steps=350, lr=0.02)
    if c_a < best_c:
        best_c = c_a; best_h = h_a

    # Phase 2: L-BFGS sweep
    t2 = min(t0 + 350, deadline - 350)
    for alpha in [2000, 10000, 50000, 200000, 1000000]:
        if time.time() > t2 or time.time() > deadline - 250:
            break
        h_l, c_l = lbfgs_phase(best_h, n, min(time.time() + 45, t2, deadline - 250), alpha)
        if c_l < best_c:
            best_c = c_l; best_h = h_l

    # Phase 3: SLSQP
    if time.time() < deadline - 250:
        for a in [50000, 200000, 1000000]:
            if time.time() > deadline - 250:
                break
            h_s, c_s = slsqp_phase(best_h, n, min(time.time() + 40, deadline - 250), alpha=a, maxiter=800)
            if c_s < best_c:
                best_c = c_s; best_h = h_s

    # Phase 4: Active-set focused optimization
    if time.time() < deadline - 200:
        h_as, c_as = active_set_optimize(best_h, n, min(time.time() + 80, deadline - 200))
        if c_as < best_c:
            best_c = c_as; best_h = h_as

    # Phase 5: Multi-n search
    if time.time() < deadline - 150:
        best_h, best_c, n = multi_n_search(best_h, n, best_c, deadline)

    # Phase 6: Perturbation escape
    if time.time() < deadline - 100:
        h_pe, c_pe = perturbation_escape(best_h, n, min(time.time() + 100, deadline - 60), n_restarts=3)
        if c_pe < best_c:
            best_c = c_pe; best_h = h_pe; n = len(best_h)

    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 40:
        for a in [500000, 2000000, 5000000]:
            if time.time() > deadline - 40:
                break
            h_p, c_p = lbfgs_phase(best_h, n, min(time.time() + 15, deadline - 40), alpha=a, maxiter=300)
            if c_p < best_c:
                best_c = c_p; best_h = h_p

    # Phase 8: Active set final
    if time.time() < deadline - 20:
        h_af, c_af = active_set_optimize(best_h, n, min(time.time() + 15, deadline - 15))
        if c_af < best_c:
            best_c = c_af; best_h = h_af

    # Phase 9: Ultra-fine pair search
    if time.time() < deadline - 2:
        best_h, best_c = pair_search_fine(best_h, n, deadline - 1, radius=20)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
