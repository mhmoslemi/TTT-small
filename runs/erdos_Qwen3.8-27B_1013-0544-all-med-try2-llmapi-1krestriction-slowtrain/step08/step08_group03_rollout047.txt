def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = (1.0 - h[k:])
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = (1.0 - h[:nv])
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

    dx = 2.0 / n

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = (1.0 - h[k:])
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = (1.0 - h[:nv])
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 200.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_shifts = len(c)
    top_k = min(120, n_shifts)
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(len(x))
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 400.0 * s
    return grad

def grad_directed_polish(h, n, deadline, top_k_shifts=50, top_k_grad=25):

    alpha = 400.0
    stall = 0
    while time.time() < deadline and stall < 8:
        c = compute_c5_vec(h, n)
        m = c.max()
        sh = alpha * (c - m)
        w = np.exp(sh - sh.max())
        w /= w.sum()
        top_idx = np.argsort(c)[-top_k_shifts:]
        grad = np.zeros(n)
        for i in range(top_k_shifts):
            wi = w[top_idx[i]]
            if wi < 1e-12:
                continue
            k = top_idx[i] - (n - 1)
            grad += wi * grad_single(h, n, k)
        inc = np.argsort(grad)[:top_k_grad]
        dec = np.argsort(grad)[-top_k_grad:]
        improved = False
        for i in inc:
            if h[i] >= 1.0 - 1e-12:
                continue
            for j in dec:
                if i == j or h[j] <= 1e-12:
                    continue

                for frac in [0.05, 0.15, 0.3, 0.5, 0.8, 1.0]:
                    d = d_max * frac
                    if d < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c:

                        improved = True

                    h[i] -= d
                    h[j] += d
                if improved:

        else:
            stall += 1
            rng = np.random.RandomState(int(best_c * 1e10) + stall * 7919)

                j = rng.randint(n)

def fine_polish(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.0003, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
