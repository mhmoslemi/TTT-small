f_L = n/2 - sum_i h[i]*h[i+L]

def chirp_start(n, rng):
    """Generate a chirp-like starting point."""
    p = n  # use n as the "prime" (not necessarily prime, but works OK)
    h = np.zeros(n)
    # Quadratic chirp: h[i] = 1 if (i*i) mod p < p/2
    for i in range(n):
        if (i * i) % p < p // 2:
            h[i] = 1.0
    # Adjust to get exactly n/2 ones
    h = project_feasible(h, n)
    # Add some randomness
    h = h + rng.randn(n) * 0.05
    h = project_feasible(h, n)
    return h

def random_binary_start(n, rng):
    """Random binary starting point."""
    h = np.zeros(n)
    ones_positions = rng.choice(n, size=n//2, replace=False)
    h[ones_positions] = 1.0
    h = project_feasible(h, n)
    return h
