def grad_descent_refine(h, n, deadline, alpha=500000.0):
    """High-alpha gradient descent with careful line search."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.005
    stall = 0
    
    while time.time() < deadline and lr > 1e-12:
        g = grad_smooth_max(h, n, alpha)
        # Project gradient to maintain feasibility (sum constraint)
        g = g - np.mean(g)  # This maintains the sum
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        improved = False
        step = lr
        for _ in range(50):
            h_try = h - step * g
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try <= best_c - 1e-15:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.4
        if improved:
            lr = min(lr * 1.05, 0.05)
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if stall > 30:
                alpha *= 2
                if alpha > 10000000:
                    break
                stall = 0
    
    return best_h, best_c

def pair_search_smart(h, n, deadline, fractions=(0.1, 0.1)):
    """Gradient-guided pair search: only try promising pairs."""
    best_c = compute_c5(h, n)
    alpha = 5000.0
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in step_sizes:
        passes = 0
        while time.time() < deadline and passes < 20:
            g = grad_smooth_max(h, n, alpha)
            # Top fraction: should decrease (high gradient)
            # Bottom fraction: should increase (low gradient)
            top_n = max(5, int(n * fractions[0]))
            bot_n = max(5, int(n * fractions[1]))
            top_idx = np.argsort(g)[-top_n:]  # highest gradient
            bot_idx = np.argsort(g)[:bot_n]    # lowest gradient
            
            improved = False
            for i in bot_idx:
                if time.time() > deadline:
                    return h, best_c
                for j in top_idx:
                    if i == j:
                        continue
                    # Try moving d from j (high grad) to i (low grad)
                    if h[j] - d >= 0.0 and h[i] + d <= 1.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            passes += 1
            if not improved:
                break
    
    return h, best_c

def subgradient_averaging(h, n, deadline, alpha=10000):
    """Subgradient method with running average."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h_curr = h.copy()
    avg_h = h.copy()
    lr = 0.02
    t = 0
    
    while time.time() < deadline and lr > 1e-10:
        t += 1
        c_vec = compute_c5_vec(h_curr, n)
        k_star = np.argmax(c_vec)
        k = k_star - (n - 1)
        dx = 2.0 / n
        
        # Subgradient at active constraint
        sg = np.zeros(n)
        if k >= 0:
            nv = n - k
            sg[0:nv] += (1.0 - h_curr[k:n])
            sg[k:n] += (-h_curr[0:nv])
        else:
            kk = -k
            nv = n - kk
            sg[kk:n] += (1.0 - h_curr[0:nv])
            sg[0:nv] += (-h_curr[kk:n])
        sg *= dx
        
        # Update
        h_curr = h_curr - lr * sg
        h_curr = project_feasible(h_curr, n)
        avg_h = avg_h + (h_curr - avg_h) / t
        
        c_curr = compute_c5(h_curr, n)
        c_avg = compute_c5(avg_h, n)
        
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
        if c_avg < best_c:
            best_c = c_avg
            best_h = avg_h.copy()
        
        lr = 0.02 / np.sqrt(t)
    
    return best_h, best_c
