from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

            ub = 1.0
            if pos_B.any():
                ub = min(ub, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            lb = 0.0
            if neg_B.any():
                lb = max(lb, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            x_opt = np.clip((lb + ub) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                nm = corr.max()
                if nm < cur_max - 1e-14:
                    improved = True
                cur_max = nm
                h[i] = x_opt
        if not improved and sweep > 5:
            break
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)

    def obj_and_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        ev = np.exp(shifted - sm)
        sv = ev.sum()
        val = m + (sm + np.log(sv)) / alpha
        weights = ev / sv
        grad = np.zeros(n)
        thresh = 1e-12
        for mi in range(2 * n - 1):
            w = weights[mi] * dx
            if abs(w) < thresh:
                continue
            lag = lags[mi]
            if lag == 0:
                grad[:] += w * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[:n - lag] += w * (1.0 - x[lag:])
                grad[lag:] -= w * x[:n - lag]
            else:
                llag = -lag
                grad[llag:] += w * (1.0 - x[:-llag])
                grad[:n - llag] -= w * x[llag:]
        return val, grad

    bounds = [(0.0, 1.0)] * n
    one_vec = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: one_vec}]
    try:
        res = minimize(obj_and_grad, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Coord descent warm-up
    p1 = t0 + 0.10 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=300, deadline=p1)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 2: SLSQP exploration (low to mid alpha)
    p2 = t0 + 0.35 * total
    h = best_h.copy()
    for alpha in [30, 100, 500, 2000, 10000]:
        if time.time() > p2 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 2000, 1e-14)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Heavy coordinate descent polish
    p3 = t0 + 0.55 * total
    hc2, _ = coord_descent(best_h, n, max_sweeps=500, deadline=p3)
    hc2 = project_feasible(hc2, n)
    cc2 = compute_c5(hc2, n)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 4: SLSQP high alpha polish
    p4 = t0 + 0.78 * total
    h = best_h.copy()
    for alpha in [50000, 200000, 1000000, 10000000, 100000000]:
        if time.time() > p4 - 4:
            break
        h = slsqp_smooth(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # Alternate with coord descent
        if time.time() < p4 - 4:
            hc3, _ = coord_descent(h, n, max_sweeps=100, deadline=p4 - 4)
            hc3 = project_feasible(hc3, n)
            cc3 = compute_c5(hc3, n)
            if cc3 < best_c:
                best_c, best_h, h = cc3, hc3.copy(), hc3.copy()

    # Phase 5: Perturbation restarts with quick polish
    scales = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.02]
    for sc in scales:
        if time.time() > dl - 4:
            break
        h_try = best_h + sc * rng.randn(n)
        h_try = project_feasible(h_try, n)
        # Quick coord descent
        hc4, _ = coord_descent(h_try, n, max_sweeps=150, deadline=dl - 4)
        hc4 = project_feasible(hc4, n)
        cc4 = compute_c5(hc4, n)
        if cc4 < best_c:
            best_c, best_h = cc4, hc4.copy()
        # Quick SLSQP high alpha
        if time.time() < dl - 6:
            h_try = slsqp_smooth(hc4, n, 100000000, 1000, 1e-18)
            c5 = compute_c5(h_try, n)
            if c5 < best_c:
                best_c, best_h = c5, h_try.copy()

    # Phase 6: Try nearby n values
    if time.time() < dl - 20:
        for nh in [390, 410, 380, 420]:
            if time.time() > dl - 5:
                break
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh, _ = coord_descent(hh, nh, max_sweeps=80, deadline=dl - 5)
            hh = project_feasible(hh, nh)
            ch = compute_c5(hh, nh)
            if ch < best_c - 1e-6:
                best_c, best_h, n = ch, hh.copy(), nh
                hh = slsqp_smooth(hh, n, 50000000, 500, 1e-18)
                ch2 = compute_c5(hh, n)
                if ch2 < best_c:
                    best_c, best_h = ch2, hh.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
