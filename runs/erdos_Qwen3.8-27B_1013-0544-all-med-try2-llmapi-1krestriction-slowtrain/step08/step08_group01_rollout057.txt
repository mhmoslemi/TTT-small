                        s1_lo = max(-(n-1), -j)
                        s1_hi = min(n-1, n-1-j)

                            dc[idx] += d * (1.0 - h[j + s_vals])
                        # Second factor: s in [j-(n-1), j]

                        if s2_lo <= s2_hi:
                            s_vals = np.arange(s2_lo, s2_hi+1)
                            idx = s_vals + (n-1)
                            dc[idx] += -d * h[j - s_vals]
                        
                        c += dc
                        h[i] -= d
                        h[j] += d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_jac(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def fast_lbfgs(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 1.0), (5000, 5.0), (50000, 20.0), (500000, 200.0), (5000000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        tr = deadline - time.time() - 3
        if tr < 5:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_jac,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-17,
                                  'gtol': 1e-15, 'maxfun': max_iters * 20, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fast_local_search(h, n, deadline, radius=100):

    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = c.max()
    nc = 2 * n - 1
    shifts_all = np.arange(-(n - 1), n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] < 0.005:
                    continue

                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - d < -1e-12 or h[j] + d > 1.0 + 1e-12:
                        continue
                    feasible = True
                    nw_max = best_c
                    for idx in crit_k:
                        s = int(crit_s[idx - crit_k[0]] ) if len(crit_k) > 1 else int(crit_s[0])
                        # Recompute s properly
                    # Redo: iterate over critical shifts properly
                    feasible = True

                    for ci in range(len(crit_k)):
                        k_idx = crit_k[ci]
                        s = shifts_all[k_idx]
                        dc = 0.0

                        if 0 <= ip_s < n:
                            dc += -d * (1.0 - h[ip_s])

                        if 0 <= im_s < n:

                            break
                    if feasible and nw_max < best_c - 1e-14:
                        # Update c for all shifts (vectorized)
                        dc_full = np.zeros(nc)
                        s1_lo = max((-(n-1)), -i)

                        s1_lo = max((-(n-1)), -j)

                        if s1_lo <= s1_hi:

                        s2_lo = max((-(n-1)), j-(n-1))

                        if s2_lo <= s2_hi:

                        c += dc_full
                        h[i] -= d
                        h[j] += d
                        best_c = nw_max
                        improved_any = True
    return h, best_c

def golden_ratio_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 9):
        amp = rng.uniform(0.15, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)
