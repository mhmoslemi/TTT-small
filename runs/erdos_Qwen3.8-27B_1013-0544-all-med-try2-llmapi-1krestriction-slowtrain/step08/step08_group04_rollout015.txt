import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        return m + 1.0 / alpha
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        return m + 1.0 / alpha
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 1e4 * (x.sum() - n / 2.0) ** 2
    return val

def _grad_con(x, n, alpha):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    wsum = w.sum()
    w /= wsum
    grad = np.zeros(n)
    h = x
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx

            nv = n - k

            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    grad += 2e4 * (x.sum() - n / 2.0) * np.ones(n)
    return grad

    return x.sum() - n / 2.0

    return np.ones(n)

    best_c = compute_c5(best_h, n)

            break

                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-14, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        if time.time() > deadline - 5:

        # SLSQP

                best_h = h_new

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _con(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def _bare_obj(x, n, alpha):
    return _smax(compute_c5_vec(x, n), alpha)

def lbfgs_phase(h, n, deadline, max_iters=200, alphas=(200, 1000, 5000)):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = alpha * 0.01
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=20000, maxiter=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_bare_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def gradient_direction_search(h, n, deadline, n_moves=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for alpha in [200, 1000, 5000]:
        moves = 0
        while time.time() < deadline and moves < n_moves:
            g = _smax_grad(h, n, alpha)
            idx = np.argsort(np.abs(g))[::-1]
            improved = False
            for i in idx[:30]:
                gi = g[i]
                if abs(gi) < 1e-14:
                    continue
                if gi > 0:
                    target_dir = -1.0
                else:
                    target_dir = 1.0
                for j in idx[:30]:
                    if j == i:
                        continue
                    gj = g[j]
                    if target_dir > 0:
                        if gj >= gi:
                            continue
                    else:
                        if gj <= gi:
                            continue
                    for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
                        di = target_dir * step
                        dj = -target_dir * step
                        if 0 <= h[i] + di <= 1 and 0 <= h[j] + dj <= 1:
                            h[i] += di
                            h[j] += dj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                                moves += 1
                                break
                            else:
                                h[i] -= di
                                h[j] -= dj
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
            moves += 1
    return h, best_c

def grad_gd(h, n, deadline, lr_init=0.02, alpha=1000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = _smax_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            if lr < 1e-14:
                break
            alpha *= 2.0
            lr = lr_init * 0.5
    return best_h, best_c

def local_pair_search(h, n, deadline, steps=(0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002), radius=80):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for step in steps:
        if time.time() > deadline:
            break
        sweep = True
        while sweep and time.time() < deadline:
            sweep = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            sweep = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            sweep = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def interp_to(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(parent)
        parent = project_feasible(parent, n_parent)
    except Exception:
        parent = np.full(300, 0.5)
        n_parent = 300

    best_h = parent.copy()
    best_n = n_parent
    best_c = compute_c5(best_h, best_n)

    # Phase 1: Multi-resolution optimization
    grids = [16, 32, 64, 128, 256]
    allocs = [(t0 + 20), (t0 + 40), (t0 + 70), (t0 + 110), (t0 + 160)]
    for gi, (ng, dl) in enumerate(zip(grids, allocs)):
        if time.time() > dl - 3:
            break
        # Try multiple inits
        inits = []
        # Constant
        inits.append(np.full(ng, 0.5))
        # Step
        h_step = np.zeros(ng)
        h_step[:ng // 2] = 1.0
        inits.append(h_step)
        # Sine
        xs = np.linspace(0, 2, ng)
        inits.append(0.5 + 0.4 * np.cos(np.pi * xs))
        # Interpolated parent
        inits.append(interp_to(best_h, best_n, ng))
        # Random
        inits.append(np.clip(rng.rand(ng) * 0.8 + 0.1, 0, 1))

        best_coarse = None
        best_coarse_c = 1.0
        for init in inits:
            if time.time() > dl - 2:
                break
            hc = project_feasible(init, ng)
            hc, cc = lbfgs_phase(hc, ng, time.time() + 3, max_iters=100,
                                  alphas=(100, 500, 2000))
            if cc < best_coarse_c:
                best_coarse_c = cc
                best_coarse = hc

        if best_coarse is not None:
            # Interpolate to parent size and compare
            interp = interp_to(best_coarse, ng, best_n)
            interp = project_feasible(interp, best_n)
            ci = compute_c5(interp, best_n)
            if ci < best_c:
                best_c = ci
                best_h = interp

    # Phase 2: Optimize parent grid
    if time.time() < deadline - 200:
        t2 = min(time.time() + 60, deadline - 200)
        best_h, best_c = lbfgs_phase(best_h, best_n, t2, max_iters=200,
                                      alphas=(200, 1000, 5000, 20000))

    # Phase 3: SLSQP
    if time.time() < deadline - 150:
        t3 = min(time.time() + 50, deadline - 150)
        for a in [10000, 50000, 200000]:
            if time.time() > t3:
                break
            h3, c3 = slsqp_phase(best_h, best_n, time.time() + 20, alpha=a, maxiter=400)
            if c3 < best_c:
                best_c = c3
                best_h = h3

    # Phase 4: Gradient descent
    if time.time() < deadline - 100:
        t4 = min(time.time() + 50, deadline - 100)
        h4, c4 = grad_gd(best_h, best_n, t4, lr_init=0.01, alpha=2000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient direction search
    if time.time() < deadline - 80:
        t5 = min(time.time() + 60, deadline - 80)
        h5, c5 = gradient_direction_search(best_h, best_n, t5, n_moves=3000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine pair search
    if time.time() < deadline - 30:
        t6 = min(time.time() + 120, deadline - 30)
        h6, c6 = local_pair_search(best_h, best_n, t6,
                                    steps=(0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001),
                                    radius=60)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try other grid sizes
    if time.time() < deadline - 60:
        for n_try in [200, 400, 500]:
            if time.time() > deadline - 40:
                break
            ht = interp_to(best_h, best_n, n_try)
            ht = project_feasible(ht, n_try)
            ht, ct = lbfgs_phase(ht, n_try, time.time() + 25, max_iters=150,
                                  alphas=(500, 2000, 10000))
            if time.time() < deadline - 20:
                ht, ct = slsqp_phase(ht, n_try, time.time() + 20, alpha=50000, maxiter=300)
            if ct < best_c:
                best_c = ct
                best_h = ht
                best_n = n_try

    # Phase 8: Final local search
    if time.time() < deadline - 5:
        best_h, best_c = local_pair_search(best_h, best_n, deadline - 2,
                                            steps=(0.001, 0.0005, 0.0002, 0.0001),
                                            radius=40)

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
