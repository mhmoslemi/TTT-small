import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha, k):
    n = len(c)
    kk = min(k, n)
    top = np.partition(c, n-kk)[n-kk:]
    cx = alpha * top
    mx = cx.max()
    return float((mx + np.log(np.sum(np.exp(cx - mx)))) / alpha)

def grad_smooth_max(h, n, alpha, k=10):
    c = compute_c5_vec(h, n)
    ntot = len(c)
    kk = min(k, ntot)
    top_idx = np.argpartition(c, ntot-kk)[ntot-kk:]
    cx = alpha * c[top_idx]
    mx = cx.max()
    w = np.exp(cx - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for ii in range(kk):
        m = top_idx[ii]
        wm = w[ii]
        shift = m - (n - 1)
        if shift >= 0:
            nv = n - shift
            grad[0:nv] += wm * (1.0 - h[shift:n])
            grad[shift:n] -= wm * h[0:nv]
        else:
            kk2 = -shift
            nv = n - kk2
            grad[kk2:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] -= wm * h[kk2:n]
    return grad

def nesterov_optimize(h, n, deadline, alpha_init=100.0, k=10, 
                       lr_init=0.005, momentum=0.9):
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = alpha_init
    lr = lr_init
    stuck_count = 0
    for it in range(50000):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        g = grad_smooth_max(y, n, alpha, k)
        # Gradient step
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 100:
                # Reduce learning rate, increase alpha
                lr *= 0.5
                alpha *= 2.0
                v *= 0.5
                stuck_count = 0
                if lr < 1e-10:
                    break
        h = h_new
    return best_h, best_c

def lbfgs_smooth(h, n, deadline, alpha, k, maxiter=300):
    def obj(x):
        c = compute_c5_vec(x, n)
        return smooth_max_val(c, alpha, k)
    def grad(x):
        return grad_smooth_max(x, n, alpha, k)
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    try:

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

        c_new = compute_c5(h_new, n)

def grad_pair_search(h, n, deadline, step_size=0.003, k=10):

            order = np.argsort(g)

                i = order[it]  # high gradient (want to decrease h[i])

               (2000, 8, 500.0), (10000, 8, 2000.0), (50000, 6, 10000.0)]
    for alpha, k, mu in configs:
        if time.time() > deadline - 100:
            break
        h_new, c_new = lbfgs_pen(best_h, n, time.time() + 70, alpha, k, mu, maxiter=250)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: SLSQP with high alpha for precise constraint handling
    if time.time() < deadline - 60:
        h_new, c_new = slsqp_refine(best_h, n, time.time() + 50, 50000, 6, maxiter=400)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 3: Nesterov momentum
    if time.time() < deadline - 50:
        t2 = min(time.time() + 70, deadline - 50)
        h_new, c_new = nesterov_opt(best_h, n, t2, alpha=2000, k=6, lr=0.001)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 40:
        t3 = min(time.time() + 80, deadline - 40)
        best_h, best_c = grad_pair(best_h, n, t3, step=0.002, k=6, max_rounds=200)

    # Phase 5: Another SLSQP pass
    if time.time() < deadline - 30:
        h_new, c_new = slsqp_refine(best_h, n, time.time() + 30, 200000, 5, maxiter=600)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 6: Perturbation restarts
    mags = [0.02, 0.04, 0.01, 0.03, 0.05]
    for mag in mags:
        if time.time() > deadline - 20:
            break
        h_pert = project_feasible(best_h + rng.normal(0, mag, n), n)
        h_start = h_pert if compute_c5(h_pert, n) < best_c else best_h.copy()
        if time.time() > deadline - 15:
            break
        t_rem = deadline - time.time() - 8
        if t_rem < 15:
            break
        h_new, c_new = lbfgs_pen(h_start, n, time.time() + min(25, t_rem * 0.6), 20000, 5, 5000.0, maxiter=100)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 6:
            h_new2, c_new2 = grad_pair(h_new, n, time.time() + min(10, deadline - time.time() - 4),
                                       step=0.001, k=5, max_rounds=50)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2

    # Phase 7: Try other resolutions
    for n_try in [200, 250, 400]:
        if time.time() > deadline - 12:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_t, c_t = lbfgs_pen(h_interp, n_try, time.time() + 12, 5000, 8, 2000.0, maxiter=100)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Final tiny refinement
    if time.time() < deadline - 2:
        nf = len(best_h)
        best_h, best_c = grad_pair(best_h, nf, deadline - 1, step=0.0005, k=4, max_rounds=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
