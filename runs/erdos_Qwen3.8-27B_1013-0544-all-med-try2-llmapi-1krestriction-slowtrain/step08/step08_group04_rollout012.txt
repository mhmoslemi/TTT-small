            if tot < 1e-15:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    return grad_smooth_max(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def slsqp_obj(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 5.0), (1000, 10.0), (10000, 50.0), (100000, 200.0), (500000, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=200000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def momentum_gd(h, n, deadline, alpha=500, lr=0.005, max_iter=5000):
    h = project_feasible(h, n)
    velocity = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        velocity = 0.85 * velocity - lr * g
        h_new = project_feasible(h + velocity, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        else:
            velocity *= 0.5
        h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        found = True
        while found and time.time() < deadline:
            found = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; found = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; found = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def grad_pair_search(h, n, deadline, step_size=0.003, radius=70):
    best_c = compute_c5(h, n)
    for alpha in [50, 200, 1000, 5000, 20000]:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if 0.0 <= h[i] + move <= 1.0 and 0.0 <= h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= move; h[j] += move
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS warmup with progressive alpha
    t1 = min(t0 + 120, deadline - 400)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=150)

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 300:
        t2 = min(time.time() + 40, deadline - 300)
        for a in [100000, 500000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_phase(best_h, n_init, min(time.time() + 25, t2), alpha=a, maxiter=800)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Momentum gradient descent
    if time.time() < deadline - 250:
        t3 = min(time.time() + 60, deadline - 250)
        for alpha in [100, 500, 2000]:
            if time.time() > t3:
                break
            h3, c3 = momentum_gd(best_h, n_init, min(time.time() + 20, t3), alpha=alpha, lr=0.003)
            if c3 < best_c:
                best_c = c3
                best_h = h3

    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 200:
        t4 = min(time.time() + 120, deadline - 200)
        best_h, best_c = grad_pair_search(best_h, n_init, t4, step_size=0.002, radius=50)

    # Phase 5: Extended fine pair search (main refinement)
    if time.time() < deadline - 80:
        t5 = deadline - 80
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        best_h, best_c = pair_search(best_h, n_init, t5, deltas, radius=50)

    # Phase 6: Try different n values
    if time.time() < deadline - 60:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(40, deadline - time.time() - 30)
            if alloc < 15:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=100)
            if time.time() < deadline - 20:
                h_t, c_t = pair_search(h_t, n_try, deadline - 20, [0.002, 0.001, 0.0005], radius=30)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    # Phase 7: Final ultra-fine pair search
    nf = len(best_h)
    if time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                     [0.0002, 0.0001, 0.00005, 0.00002], radius=30)

    # Final perturbation + re-localize
    if time.time() < deadline - 2:
        h_pert = best_h + rng.randn(nf) * 0.001
        h_pert = project_feasible(h_pert, nf)
        c_pert = compute_c5(h_pert, nf)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
        else:
            h_refine = project_feasible(best_h + rng.randn(nf) * 0.0005, nf)
            c_ref = compute_c5(h_refine, nf)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_refine

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
