import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    mx = c.max()
    sh = alpha * (c - mx)
    smx = sh.max()
    if smx > 500:
        e = np.exp(sh - smx)
        logsum = smx + np.log(e.sum())
    else:
        e = np.exp(sh)
        logsum = np.log(e.sum())
    val = mx + logsum / alpha
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:k + nv])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return val, grad

def _obj(x, n, alpha, mu):
    v, _ = smooth_max_val_grad(x, n, alpha)
    return v + mu * (x.sum() - n / 2.0) ** 2

def _grad(x, n, alpha, mu):
    _, g = smooth_max_val_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_obj, best_h, jac=_grad,
                           args=(n, alpha, 5.0), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': 4000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-17:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def pd_refine(h, n, deadline, alpha_init=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    alpha = alpha_init
    for _ in range(20000):
        if time.time() > deadline:
            break
        _, g = smooth_max_val_grad(best_h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for s in [lr, lr * 0.5, lr * 0.25]:
            h_new = project_feasible(best_h - s * g / gn, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-17:
                best_h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-13:
                break
        alpha = min(alpha * 1.05, 5000000)
    return best_h, best_c

def restricted_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def grad_guided_pairs(h, n, deadline, alpha=100000, step=0.003):
    best_c = compute_c5(h, n)
    for _ in range(5000):
        if time.time() > deadline:
            break
        _, g = smooth_max_val_grad(h, n, alpha)
        g_proj = g - g.mean()
        i_c = int(np.argmin(g_proj))
        j_c = int(np.argmax(g_proj))
        if g_proj[j_c] - g_proj[i_c] < 1e-13:
            break
        improved = False
        for frac in [1.0, 0.5, 0.2, 0.1]:
            s = step * frac
            if h[i_c] + s <= 1.0 and h[j_c] - s >= 0.0:
                h[i_c] += s; h[j_c] -= s
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    improved = True
                    break
                h[i_c] -= s; h[j_c] += s
        if not improved:
            alpha = min(alpha * 1.2, 5000000)
            step *= 0.7
            if step < 1e-5:
                break
    return h, best_c

def perturb_and_optimize(h, n, deadline, rng, scale=0.05):
    h_p = np.clip(h + rng.randn(n) * scale, 0.0, 1.0)
    h_p = project_feasible(h_p, n)
    h_new, c_new = lbfgs_opt(h_p, n, deadline, alphas=(100, 1000, 10000, 100000))
    return h_new, c_new
