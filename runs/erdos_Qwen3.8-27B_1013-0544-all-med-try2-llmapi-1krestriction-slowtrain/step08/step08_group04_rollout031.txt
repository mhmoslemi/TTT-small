def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def correlate_c5(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    c = correlate_c5(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_grad(h, n, alpha):
    c = correlate_c5(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        idx = k + n - 1
        wm = w[idx] * dx
        if wm < 1e-14:
            continue
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, alphas, max_iter_each=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        rem = deadline - time.time() - 3
        if rem < 5:
            break
        mi = min(max_iter_each, max(10, int(rem / 0.3)))
        def _o(x, a=alpha):
            return smooth_obj(x, n, a)
        def _g(x, a=alpha):
            return smooth_grad(x, n, a)
        try:
            res = minimize(_o, h, jac=_g, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mi, 'ftol': 1e-17, 'gtol': 1e-14,
                                   'maxfun': mi * 8})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                h = hn
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, steps, radius):
    best_c = compute_c5(h, n)
    for s in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + s <= 1.0 and h[j] - s >= 0.0:
                        h[i] += s; h[j] -= s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= s; h[j] += s
                    if h[i] - s >= 0.0 and h[j] + s <= 1.0:
                        h[i] -= s; h[j] += s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += s; h[j] -= s
            if not improved:
                break
    return h, best_c

def binary_swap(h, n, deadline):
    best_c = compute_c5(h, n)
    h_bin = (h >= 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin >= best_c:
        return h, best_c
    h_cur = h_bin
    best_c = c_bin
    zeros = np.where(h_cur < 0.5)[0]
    ones = np.where(h_cur > 0.5)[0]
    for radius in [8, 20, 50, n]:
        while time.time() < deadline:
            improved = False
            zlist = zeros.tolist()
            olist = ones.tolist()
            for i in zlist:
                if time.time() > deadline:
                    return h_cur, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in olist:
                    if j < lo or j >= hi:
                        continue
                    h_cur[i] = 1.0; h_cur[j] = 0.0
                    c = compute_c5(h_cur, n)
                    if c < best_c - 1e-15:
                        best_c = c; improved = True
                        zlist.remove(i)
                        olist.append(i)
                        olist.remove(j)
                        zlist.append(j)
                        zeros = np.array(zlist)
                        ones = np.array(olist)
                    else:
                        h_cur[i] = 0.0; h_cur[j] = 1.0
            if not improved:
                break
    return h_cur, best_c

def interp_resp(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

    coarse_levels = [24, 40, 60, 90, 130]
    for nc in coarse_levels:
        if time.time() > deadline - 350:
            break
        h_c = project_feasible(interp_resp(best_h, best_n, nc), nc)
        alloc = min(70, deadline - time.time() - 280)
        if alloc < 15:
            break
        h_c, c_c = lbfgs_opt(h_c, nc, time.time() + alloc,
                             [50, 200, 1000, 5000, 20000], max_iter_each=80)
        h_f = project_feasible(interp_resp(h_c, nc, best_n), best_n)
        c_f = compute_c5(h_f, best_n)
        if c_f < best_c - 1e-7:
            best_c = c_f
            best_h = h_f

    # Phase 2: Intensive L-BFGS on best_n
    if time.time() < deadline - 250:
        alloc = min(180, deadline - time.time() - 200)
        best_h, best_c = lbfgs_opt(best_h, best_n, time.time() + alloc,
                                   [200, 1000, 5000, 20000, 100000, 500000, 2000000],
                                   max_iter_each=150)

    # Phase 3: Try different grid sizes
    alt_ns = [n0, n0 // 2, n0 * 3 // 2, n0 * 2 // 3]
    alt_ns = [x for x in alt_ns if x >= 30 and x <= 600 and x != n0]
    for nt in alt_ns:
        if time.time() > deadline - 150:
            break
        alloc = min(80, deadline - time.time() - 100)
        if alloc < 25:
            break
        h_t = project_feasible(interp_resp(best_h, best_n, nt), nt)
        h_t, c_t = lbfgs_opt(h_t, nt, time.time() + alloc,
                             [500, 5000, 50000, 500000], max_iter_each=100)
        if c_t < best_c - 1e-7:
            best_c = c_t
            best_h = h_t
            best_n = nt

    # Phase 4: Pair search on best
    if time.time() < deadline - 100:
        alloc = min(120, deadline - time.time() - 80)
        best_h, best_c = pair_search(best_h, best_n, time.time() + alloc,
                                      [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
                                      radius=40)

    # Phase 5: Binary swap search
    if time.time() < deadline - 80:
        alloc = min(80, deadline - time.time() - 50)
        h_b, c_b = binary_swap(best_h, best_n, time.time() + alloc)
        if c_b < best_c - 1e-15:
            best_h = h_b
            best_c = c_b

    # Phase 6: Restart with perturbation
    if time.time() < deadline - 60:
        for _ in range(5):
            if time.time() > deadline - 40:
                break
            noise = rng.uniform(-0.05, 0.05, best_n)
            h_p = project_feasible(best_h + noise, best_n)
            alloc = min(30, deadline - time.time() - 25)
            if alloc < 10:
                break
            h_p, c_p = lbfgs_opt(h_p, best_n, time.time() + alloc,
                                  [10000, 100000, 500000], max_iter_each=80)
            if c_p < best_c - 1e-9:
                best_c = c_p
                best_h = h_p

    # Phase 7: Final fine pair search
    if time.time() < deadline - 5:
        best_h, best_c = pair_search(best_h, best_n, deadline,
                                      [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                      radius=25)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
