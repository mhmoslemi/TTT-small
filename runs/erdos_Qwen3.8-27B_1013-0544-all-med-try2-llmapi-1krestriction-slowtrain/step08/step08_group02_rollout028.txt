def flip_delta(h, i, j, n):
    """Compute delta_c for flipping h[i] 1->0 and h[j] 0->1. Returns array of length 2n-1."""
    delta = np.zeros(2*n-1)
    # i: 1->0
    # Term 1: h[i+k] for k where 0<=i+k<n, i.e., -i <= k < n-i
    lo1 = max(-i, -(n-1))
    hi1 = min(n-i, n)  # exclusive
    if lo1 < hi1:
        idx = np.arange(lo1, hi1) + (n-1)  # convert to c-index
        vals = i + np.arange(lo1, hi1)  # indices into h
        delta[idx] += h[vals]
    # Term 2: -(1-h[i-k]) for k where 0<=i-k<n, i.e., i-n+1 < k <= i, i.e., i-n+1 <= k-1... 
    # i-k in [0,n) means k in [i-n+1, i]
    lo2 = max(i-n+1, -(n-1))
    hi2 = min(i, n-1) + 1  # exclusive
    if lo2 < hi2:
        idx = np.arange(lo2, hi2) + (n-1)
        vals = i - np.arange(lo2, hi2)
        delta[idx] -= (1 - h[vals])
    
    # j: 0->1
    # Term 3: -h[j+k] for k where 0<=j+k<n
    lo3 = max(-j, -(n-1))
    hi3 = min(n-j, n)
    if lo3 < hi3:
        idx = np.arange(lo3, hi3) + (n-1)
        vals = j + np.arange(lo3, hi3)
        delta[idx] -= h[vals]
    # Term 4: +(1-h[j-k]) for k where 0<=j-k<n
    lo4 = max(j-n+1, -(n-1))
    hi4 = min(j, n-1) + 1
    if lo4 < hi4:
        idx = np.arange(lo4, hi4) + (n-1)
        vals = j - np.arange(lo4, hi4)
        delta[idx] += (1 - h[vals])
    
    return delta

def peak_focused_optimize(h, n, deadline, top_m=10, step_size=0.1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    while time.time() < deadline - 5:
        c = np.correlate(h, 1-h, mode='full')
        # Find top M peaks
        m = min(top_m, len(c))
        top_idx = np.argsort(c)[-m:]
        top_weights = c[top_idx] / c.max()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, w in zip(top_idx, top_weights):
            k = idx - (n - 1)
            # grad_c[k][j] = (1-h[j-k]) - h[j+k] with bounds
            if k >= 0:
                # j-k valid when j >= k, j+k valid when j+k < n
                grad[k:] += w * (1 - h[:n-k])  # (1-h[j-k]) for j >= k
                grad[:n-k] += w * (-h[k:])  # -h[j+k] for j+k < n
            else:
                kk = -k
                grad[kk:] += w * (1 - h[:n-kk])
                grad[:n-kk] += w * (-h[kk:])
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

        else:

            c_try = compute_c5(h, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
            step_size *= 0.7
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot < 1e-14: break
            h[mask] += (diff / tot) * h[mask]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14: break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14: continue
        k = mi - (n - 1)
        wm = w[mi]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n/2.0)**2

    g = smooth_max_grad(x, n, alpha)

                          method='L-BFGS-B', bounds=bounds,
