        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = project_polytope(np.full(n, 0.5), n)
    best_h, best_c, best_n = h0.copy(), compute_c5(h0, n), n
    # Phase 1: SLSQP warmup
    p1 = t0 + 0.12 * total_time
    h = best_h.copy()
    for alpha in [200, 2000, 20000, 200000]:
        if time.time() > p1 - 2:
            break
        h = slsqp_pass(h, n, alpha, 400)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 2: Projected gradient (main)
    p2 = t0 + 0.50 * total_time
    h = best_h.copy()
    h, c = proj_grad_opt(h, n, p2, n_iters=300000, noise_period=8000, noise_amp=0.008)
    if c < best_c:
        best_c, best_h = c, h
    # Phase 3: Binary greedy
    p3 = t0 + 0.65 * total_time
    hb, cb = greedy_bin(best_h, n, p3)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 4: Continuous from binary
    p4 = t0 + 0.80 * total_time
    h = best_h.copy()
    h, c = proj_grad_opt(h, n, p4, n_iters=200000, noise_period=0, noise_amp=0.0)
    if c < best_c:
        best_c, best_h = c, h
    # Phase 5: Multi-resolution
    p5 = dl - 5
    for nh in [250, 350, 500]:
        if time.time() > p5 - 25:
            break
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_polytope(np.interp(xn, xo, best_h), nh)
        hh, ch = proj_grad_opt(hh, nh, p5 - 18, n_iters=40000, noise_period=5000, noise_amp=0.005)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh, nh
    # Phase 6: Final polish
    if time.time() < dl - 3:
        h = best_h.copy()
        h, c = proj_grad_opt(h, best_n, dl - 3, n_iters=50000, noise_period=0, noise_amp=0.0)
        if c < best_c:
            best_c, best_h = c, h
    best_h = project_polytope(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
