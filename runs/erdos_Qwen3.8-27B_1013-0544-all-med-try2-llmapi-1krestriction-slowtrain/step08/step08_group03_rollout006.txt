        s = float(h.sum())
        diff = float(target) - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = s
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    diff = float(target) - float(h.sum())
    if diff > 1e-15:
        idx = np.argsort(1.0 - h)[::-1]
        rem = diff
        for ii in idx:
            if rem <= 1e-15:
                break
            add = min(1.0 - h[ii], rem)
            if add > 0.0:
                h[ii] += add
                rem -= add
    elif diff < -1e-15:
        idx = np.argsort(h)
        rem = -diff
        for ii in idx:
            if rem <= 1e-15:
                break
            sub = min(h[ii], rem)
            if sub > 0.0:
                h[ii] -= sub
                rem -= sub
    return np.clip(h, 0.0, 1.0)


def symmetrize_even(h, n):
    h = np.asarray(h, dtype=float)
    if n % 2 != 0 or n < 2:
        return project_to_sum(0.5 * (h + h[::-1]), 0.5 * n)
    L = n // 2
    u = 0.5 * (h[:L] + h[n - 1:n - L - 1:-1])
    u = project_to_sum(u, 0.5 * L)
    return np.concatenate((u, u[::-1]))


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw) * (2.0 / n))


def init_c_arrays(h, n):
    h = np.asarray(h, dtype=float)
    raw = np.correlate(h, 1.0 - h, mode="full")
    cpos = raw[n - 1:].astype(float).copy()
    cneg = raw[:n].astype(float)[::-1].copy()
    return cpos, cneg


def apply_delta_full(cpos, cneg, h, idxs, deltas, n):
    cnt = len(idxs)
    for a in range(cnt):
        r = idxs[a]
        d = deltas[a]
        if abs(d) < 1e-18:
            continue
        old = h[r]
        c0 = d - 2.0 * old * d - d * d
        cpos[0] += c0
        cneg[0] += c0
        m = n - r
        if m > 1:
            seg = d * h[r + 1:r + m]
            cpos[1:m] += d
            cpos[1:m] -= seg
            cneg[1:m] -= seg
        if r > 0:
            seg = d * h[r - 1::-1]
            cpos[1:r + 1] -= seg
            cneg[1:r + 1] += d
            cneg[1:r + 1] -= seg
    for a in range(cnt):
        da = deltas[a]
        if da == 0.0:
            continue
        ia = idxs[a]
        for b in range(a + 1, cnt):
            db = deltas[b]
            if db == 0.0:
                continue
            j = ia - idxs[b]
            if j < 0:
                j = -j
            if j > 0:
                val = da * db
                cpos[j] -= val
                cneg[j] -= val


def descent_direction(h, cpos, cneg, n, mode_sym):
    if mode_sym:
        raw = cpos
        top = min(80, n)
        idx = np.argsort(raw)[-top:]
        grad = np.zeros(n, dtype=float)
        for ii in idx:
            j = int(ii)
            L = n - j
            if L <= 0:
                continue
            grad[:L] += 1.0 - h[j:]
            if L > 0:
                grad[j:] += -h[:L]
        Lh = n // 2
        gu = grad[:Lh] + grad[n - 1:n - Lh - 1:-1]
        return -gu
    raw = np.empty(2 * n - 1, dtype=float)
    raw[n - 1:] = cpos
    raw[:n - 1] = cneg[1:][::-1]
    top = min(80, 2 * n - 1)
    idx = np.argsort(raw)[-top:]
    grad = np.zeros(n, dtype=float)
    for ii in idx:
        k = int(ii) - (n - 1)
        if k == 0:
            grad += 1.0 - 2.0 * h
        elif k > 0:
            j = k
            L = n - j
            grad[:L] += 1.0 - h[j:]
            if L > 0:
                grad[j:] += -h[:L]
        else:
            j = -k
            L = n - j
            grad[j:] += 1.0 - h[:L]
            if L > 0:
                grad[:L] += -h[j:]
    return -grad


def pair_search(h, n, mode_sym, deadline, rng):
    h = np.asarray(h, dtype=float).copy()
    cpos, cneg = init_c_arrays(h, n)
    cur_rs = float(max(cpos.max(), cneg.max()))
    best_rs = cur_rs
    best_h = h.copy()
    best_cpos = cpos.copy()
    best_cneg = cneg.copy()
    step = 0.05
    temperature = 1e-6
    no_improve = 0
    since_best = 0
    guide = None
    guide_age = 99999
    attempts = 0
    dx = 2.0 / n

    while time.time() < deadline:
        attempts += 1
        guide_age += 1
        if guide_age >= 800:
            guide = descent_direction(h, cpos, cneg, n, mode_sym)
            guide_age = 0
        if attempts % 20000 == 0:
            step = max(1e-5, step * 0.75)
        if attempts % 1000 == 0:
            if since_best > 35000:
                h = best_h.copy()
                cpos = best_cpos.copy()
                cneg = best_cneg.copy()
                cur_rs = best_rs
                step = 0.04
                temperature = 1e-6
                no_improve = 0
                since_best = 0
                guide = None
            if no_improve > 7000:
                step = max(1e-5, step * 0.65)
                no_improve = 0
        temperature = max(1e-10, temperature * 0.99995)

        if mode_sym:
            L = n // 2
            if L < 2:
                break
            if guide is not None and rng.random() < 0.70:
                order = np.argsort(guide)
                K = min(32, L)
                i = int(order[L - K + int(rng.randint(K))])
                j = int(order[int(rng.randint(K))])
                if i == j:
                    i = (i + 1) % L
                    if i == j:
                        i = (j + 1) % L
                if guide[i] > guide[j] + 1e-10:
                    sgn = 1
                elif guide[j] > guide[i] + 1e-10:
                    sgn = -1
                else:
                    sgn = 1 if rng.random() < 0.5 else -1
            else:
                i = int(rng.randint(L))
                j = int(rng.randint(L))
                if i == j:
                    j = (j + 1) % L
                sgn = 1 if rng.random() < 0.5 else -1
            hi = h[i]
            hj = h[j]
            max_pos = min(1.0 - hi, hj)
            max_neg = min(hi, 1.0 - hj)
            if sgn == 1 and max_pos <= 1e-13:
                sgn = -1
            elif sgn != 1 and max_neg <= 1e-13:
                sgn = 1
            ma = max_pos if sgn == 1 else max_neg
            if ma <= 1e-13:
                continue
        else:
            if guide is not None and rng.random() < 0.70:
                order = np.argsort(guide)
                K = min(48, n)
                a = int(order[n - K + int(rng.randint(K))])
                b = int(order[int(rng.randint(K))])
                if a == b:
                    b = (b + 1) % n
                if guide[a] > guide[b] + 1e-10:
                    sgn = 1
                elif guide[b] > guide[a] + 1e-10:
                    sgn = -1
                else:
                    sgn = 1 if rng.random() < 0.5 else -1
            else:
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    b = (b + 1) % n
                sgn = 1 if rng.random() < 0.5 else -1
            ha = h[a]
            hb = h[b]
            if sgn == 1:
                ma = min(1.0 - ha, hb)
            else:
                ma = min(ha, 1.0 - hb)
            if ma <= 1e-13:
                sgn = -sgn
                ma = min(ha, 1.0 - hb) if sgn == 1 else min(1.0 - ha, hb)
            if ma <= 1e-13:
                continue

        if rng.random() < 0.035:
            eps = ma * (0.15 + 0.85 * rng.random())
        else:
            eps = step
            if eps > 0.85 * ma:
                eps = max(1e-6, 0.45 * ma)
        eps = min(eps, ma * (1.0 - 1e-12))
        if eps <= 1e-12:
            continue

        if mode_sym:
            idxs = (i, n - 1 - i, j, n - 1 - j)
            deltas = (sgn * eps, sgn * eps, -sgn * eps, -sgn * eps)
        else:
            idxs = (a, b)
            deltas = (sgn * eps, -sgn * eps)

        ncpos = cpos.copy()
        ncneg = cneg.copy()
        apply_delta_full(ncpos, ncneg, h, idxs, deltas, n)
        new_rs = float(max(ncpos.max(), ncneg.max()))
        tol = 1e-12

        if new_rs < cur_rs - tol:
            for rr, dd in zip(idxs, deltas):
                h[rr] += dd
            cpos = ncpos
            cneg = ncneg
            cur_rs = new_rs
            no_improve = 0
            if cur_rs < best_rs - tol:
                best_rs = cur_rs
                best_h = h.copy()
                best_cpos = cpos.copy()
                best_cneg = cneg.copy()
                since_best = 0
        else:
            since_best += 1
            no_improve += 1
            delta_c = (new_rs - cur_rs) * dx
            if delta_c > 0.0 and temperature > 1e-11:
                lp = -delta_c / temperature
                if lp > -50.0 and rng.random() < math.exp(lp):
                    for rr, dd in zip(idxs, deltas):
                        h[rr] += dd
                    cpos = ncpos
                    cneg = ncneg
                    cur_rs = new_rs

        if attempts % 25000 == 0:
            cpos, cneg = init_c_arrays(h, n)
            cur_rs = float(max(cpos.max(), cneg.max()))
            if cur_rs < best_rs - 1e-12:
                best_rs = cur_rs
                best_h = h.copy()
                best_cpos = cpos.copy()
                best_cneg = cneg.copy()
                since_best = 0

    return best_h, best_cpos, best_cneg, float(best_rs)


def det_polish_sym(h, cpos, cneg, n, deadline):
    if n % 2 != 0 or n < 4:
        return h, cpos, cneg, float(max(cpos.max(), cneg.max()))
    h = np.asarray(h, dtype=float).copy()
    cpos = np.asarray(cpos, dtype=float).copy()
    cneg = np.asarray(cneg, dtype=float).copy()
    cur_rs = float(max(cpos.max(), cneg.max()))
    best_h = h.copy()
    best_cpos = cpos.copy()
    best_cneg = cneg.copy()
    best_rs = cur_rs
    L = n // 2
    steps = (3e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6)

    for eps in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(L):
                for j in range(i + 1, L):
                    if time.time() > deadline:
                        break
                    hi = h[i]
                    hj = h[j]
                    signs = (-1, 1) if hi > hj else (1, -1)
                    for sgn in signs:
                        if sgn == 1:
                            ma = min(1.0 - hi, hj)
                        else:
                            ma = min(hi, 1.0 - hj)
                        if ma <= 1e-14:
                            continue
                        ep = min(eps, ma * (1.0 - 1e-12))
                        if ep <= 1e-12:
                            continue
                        idxs = (i, n - 1 - i, j, n - 1 - j)
                        deltas = (sgn * ep, sgn * ep, -sgn * ep, -sgn * ep)
                        ncp = cpos.copy()
                        ncn = cneg.copy()
                        apply_delta_full(ncp, ncn, h, idxs, deltas, n)
                        nr = float(max(ncp.max(), ncn.max()))
                        if nr < cur_rs - 1e-12:
                            for rr, dd in zip(idxs, deltas):
                                h[rr] += dd
                            cpos = ncp
                            cneg = ncn
                            cur_rs = nr
                            improved = True
                            if cur_rs < best_rs - 1e-12:
                                best_rs = cur_rs
                                best_h = h.copy()
                                best_cpos = cpos.copy()
                                best_cneg = cneg.copy()
                            break
                if time.time() > deadline:
                    break
    return best_h, best_cpos, best_cneg, float(best_rs)


def interpolate_to_n(h_old, n_old, n_new):
    if n_old == n_new:
        return project_to_sum(np.asarray(h_old, dtype=float).copy(), 0.5 * n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.asarray(h_old, dtype=float))
    return project_to_sum(h_new, 0.5 * n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_start = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h_start = None

    if h_start is None or len(h_start) < 4:
        n0 = 500
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        h0 = 1.0 - np.abs(x - 1.0)
        h0 = project_to_sum(h0, 0.5 * n0)
    else:
        h0 = project_to_sum(h_start, 0.5 * len(h_start))

    n0 = len(h0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    working_sym = False
    if n0 % 2 == 0 and n0 >= 4:
        hs = symmetrize_even(h0, n0)
        cs = compute_c5(hs, n0)
        asym = float(np.max(np.abs(h0 - h0[::-1])))
        if cs < best_c + 2e-4 or asym < 0.05:
            best_h = hs.copy()
            best_c = cs
            working_sym = True

    if best_n > 800:
        best_h = interpolate_to_n(best_h, best_n, 800)
        if best_n % 2 == 0 and working_sym:
            best_h = symmetrize_even(best_h, 800)
        best_n = 800
        best_c = compute_c5(best_h, best_n)
        working_sym = bool(best_n % 2 == 0)

    # Main direct exact pair-annealing on the current resolution.
    if time.time() < deadline - 60:
        d1 = min(t0 + 660.0, deadline - 260.0)
        if d1 < t0 + 50.0:
            d1 = min(deadline - 20.0, t0 + 120.0)
        bh, bcp, bcn, brs = pair_search(best_h, best_n, working_sym, d1, rng)
        cand_c = brs * (2.0 / best_n)
        if cand_c < best_c:
            best_h = bh
            best_n = len(bh)
            best_c = cand_c
            working_sym = bool(best_n % 2 == 0)

    # Coordinate-descent polish on the best symmetric candidate.
    if best_n % 2 == 0 and best_n <= 750 and time.time() < deadline - 80:
        cp, cn = init_c_arrays(best_h, best_n)
        d2 = min(time.time() + 170.0, deadline - 70.0)
        bh, bcp, bcn, brs = det_polish_sym(best_h, cp, cn, best_n, d2)
        if brs * (2.0 / best_n) < best_c:
            best_h = bh
            best_c = brs * (2.0 / best_n)
            best_n = len(bh)
            working_sym = True

    # Shorter second pass at current resolution if time remains.
    if time.time() < deadline - 50:
        d3 = min(time.time() + 95.0, deadline - 30.0)
        bh, bcp, bcn, brs = pair_search(best_h, best_n, working_sym, d3, rng)
        cand_c = brs * (2.0 / best_n)
        if cand_c < best_c:
            best_h = bh
            best_n = len(bh)
            best_c = cand_c
            working_sym = bool(best_n % 2 == 0)

    # Try a higher-resolution refinement if enough time remains.
    if time.time() < deadline - 140:
        target = 800 if best_n < 800 else 1000
        if target is not None:
            h_try = interpolate_to_n(best_h, best_n, target)
            if target % 2 == 0:
                h_try = symmetrize_even(h_try, target)
            c_try = compute_c5(h_try, target)
            if c_try < best_c:
                best_h = h_try
                best_n = target
                best_c = c_try
                working_sym = True
            rem = deadline - time.time()
            if rem > 90.0:
                d4 = min(time.time() + 130.0, deadline - 25.0)
                bh, bcp, bcn, brs = pair_search(h_try, target, True, d4, rng)
                cand_c = brs * (2.0 / target)
                if cand_c < best_c:
                    best_h = bh
                    best_n = target
                    best_c = cand_c
                    working_sym = True

    # Final feasibility/quality check.
    if best_n % 2 == 0 and best_n >= 4:
        hs = symmetrize_even(best_h, best_n)
        cs = compute_c5(hs, best_n)
        co = compute_c5(best_h, best_n)
        if cs <= co:
            best_h = hs
        else:
            best_h = project_to_sum(best_h, 0.5 * best_n)
    else:
        best_h = project_to_sum(best_h, 0.5 * best_n)

    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
