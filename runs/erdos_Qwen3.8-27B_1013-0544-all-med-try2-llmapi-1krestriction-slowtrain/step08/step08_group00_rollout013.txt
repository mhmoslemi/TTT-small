from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_all_gradients(h, n):
    """Compute gradients of all c_k with respect to h.
    Returns array of shape (2n-1, n) where row m corresponds to shift k = m-(n-1)."""
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            # h[i] * (1-h[i+k]) for i in [0, nv)
            # d/dh[j]: (1-h[j+k]) for j in [0, nv)  [j appears as i]
            #         -h[j-k] for j in [k, n)       [j appears as i+k, i=j-k in [0,nv))
            grads[m, :nv] += (1.0 - h[k:])       # j in [0,nv): h[j+k] = h[k+j]
            grads[m, k:]  += (-h[:nv])           # j in [k,n): h[j-k] = h[j-k]
        else:
            kk = -k
            nv = n - kk
            # h[i] * (1-h[i-kk]) for i in [kk, n)
            # d/dh[j]: (1-h[j-kk]) for j in [kk, n)  [j appears as i]
            #          -h[j+kk] for j in [0, nv)      [j appears as i-kk, i=j+kk in [kk,n))
            grads[m, kk:] += (1.0 - h[:nv])      # j in [kk,n): h[j-kk]
            grads[m, :nv] += (-h[kk:])           # j in [0,nv): h[j+kk]
    return grads  # shape (2n-1, n)

    """Successive Convex Approximation using LP subproblems."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute current correlations and gradients
        c_vals = compute_c5_vec(h, n)  # shape (2n-1), already includes dx
        grads = compute_all_gradients(h, n)  # shape (2n-1, n), no dx factor
        
        # SCA: min t
        # s.t. c_k_linearized <= t for each k
        #      0 <= h[j] <= 1
        #      sum h[j] * dx = 1
        #
        # c_k_linearized = c_k(h) + grad_c_k^T (h_new - h)
        #                 = (c_k(h) - grad_c_k^T h) + grad_c_k^T h_new
        #
        # Note: c_vals already includes dx, but grads don't have dx.
        # So: c_k_linearized = c_vals[k] + dx * grads[k]^T (h_new - h)
        #     = (c_vals[k] - dx * grads[k]^T h) + dx * grads[k]^T h_new
        
        A_ineq = np.zeros((2*n-1, n+1))

            A_ineq[k, :n] = -dx * grads[k]
            A_ineq[k, n] = 1.0
            b_ineq[k] = -const

        A_eq[0, :n] = 1.0

def lbfgs_smooth_max(h, n, deadline, max_iters=300):
    """L-BFGS-B with smooth max (log-sum-exp)."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        alpha = 2000.0
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        alpha = 2000.0
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        g = np.zeros(n)
        for mm in range(2*n-1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n-1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                g[:nv] += wm * (1.0 - x[k:])
                g[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[kk:])
        return g

        pass

        for m in active[:40]:
            ag += dx * grads[m]
        desc = np.argsort(ag)
        asc = np.argsort(-ag)
        any_imp = False
        for _ in range(300):
            if time.time() > deadline:
                return h, best_c
            for ii in range(min(n_top, len(desc))):
                for jj in range(min(n_top, len(asc))):
                    i = desc[ii]
                    j = asc[jj]
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                            any_imp = True
                        else:
                            h[j] -= d
                            h[i] += d
        if not any_imp:
            break
    return h, best_c

def random_pair_search(h, n, deadline, n_trials=3000):
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    steps = [0.003, 0.001, 0.0004, 0.0001, 4e-5]
    for d in steps:
        trials = 0
        any_imp = False
        while trials < n_trials and time.time() < deadline:
            trials += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    any_imp = True
                else:
                    h[i] -= d
                    h[j] += d
            if trials > 800 and not any_imp:
                break
        if not any_imp:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: SCA (primary new approach)
    t1 = min(t0 + budget_s * 0.30, deadline - 350)
    h1, c1 = sca_optimize(best_h, best_n, t1, max_iter=60, damp_start=1.0, damp_end=0.6)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: L-BFGS polish
    t2 = min(time.time() + 70, deadline - 250)
    h2, c2 = lbfgs_optimize(best_h, best_n, t2, alphas=(2000, 10000, 50000))
    if c2 < best_c:
        best_c, best_h = c2, h2

    # Phase 3: SCA again (continued)
    t3 = min(time.time() + 70, deadline - 180)
    h3, c3 = sca_optimize(best_h, best_n, t3, max_iter=40, damp_start=0.8, damp_end=0.4)
    if c3 < best_c:
        best_c, best_h = c3, h3

    # Phase 4: Targeted local search
    t4 = min(time.time() + 80, deadline - 100)
    h4, c4 = targeted_search(best_h, best_n, t4, n_top=20)
    if c4 < best_c:
        best_c, best_h = c4, h4

    # Phase 5: Random pair search
    t5 = min(time.time() + 60, deadline - 40)
    h5, c5 = random_pair_search(best_h, best_n, t5, n_trials=5000)
    if c5 < best_c:
        best_c, best_h = c5, h5

    # Phase 6: Final SCA squeeze
    if time.time() < deadline - 15:
        t6 = min(time.time() + 30, deadline - 5)
        h6, c6 = sca_optimize(best_h, best_n, t6, max_iter=20, damp_start=0.5, damp_end=0.3)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Try nearby grid sizes
    if time.time() < deadline - 10:
        for n_try in [best_n + 50, best_n - 50]:
            if time.time() > deadline - 8:
                break
            if n_try < 200:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            hi = np.interp(x_new, x_old, best_h)
            hi = project_feasible(hi, n_try)
            ht, ct = sca_optimize(hi, n_try, deadline - 3, max_iter=10, damp_start=0.7, damp_end=0.4)
            if ct < best_c:
                best_c, best_h, best_n = ct, ht, n_try

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
