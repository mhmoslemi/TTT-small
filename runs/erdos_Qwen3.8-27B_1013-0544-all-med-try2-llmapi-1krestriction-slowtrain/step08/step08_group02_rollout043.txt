def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):

    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g
