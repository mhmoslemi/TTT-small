```python
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 50.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_pts = len(x)
    top_k = min(100, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n_pts)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 100.0 * s
    return grad

def coord_desc(h, n, deadline, max_sweeps=500, init_range=0.08, n_grid=21):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    search_range = init_range
    stall_count = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c5_vec = compute_c5_vec(h, n).copy()
        sweep_improved = False
        
        for i in range(n):
            k_vals = np.arange(-(n - 1), n)
            i_pk = i + k_vals
            i_mk = i - k_vals
            mask_p = (i_pk >= 0) & (i_pk < n)
            mask_m = (i_mk >= 0) & (i_mk < n)
            slopes = np.zeros(2 * n - 1)
            if mask_p.any():
                slopes[mask_p] += (1.0 - h[i_pk[mask_p]]) * dx
            if mask_m.any():
                slopes[mask_m] -= h[i_mk[mask_m]] * dx
            
            lo = max(-h[i], -search_range)
            hi = min(1.0 - h[i], search_range)
            if lo >= hi:
                continue
            
            top_lags = np.argsort(c5_vec)[-4:]
            candidates = [0.0, lo, hi]
            for a_idx in range(len(top_lags)):
                for b_idx in range(a_idx + 1, len(top_lags)):
                    ka, kb = top_lags[a_idx], top_lags[b_idx]
                    sa, sb = slopes[ka], slopes[kb]
                    if abs(sa - sb) > 1e-15:
                        d_cross = (c5_vec[kb] - c5_vec[ka]) / (sa - sb)
                        if lo <= d_cross <= hi:
                            candidates.append(d_cross)
            candidates.extend(np.linspace(lo, hi, n_grid).tolist())
            candidates = np.array(sorted(set(np.round(candidates, 10))))
            
            new_c5_mat = c5_vec[None, :] + candidates[:, None] * slopes[None, :]
            pos0 = n - 1
            new_c5_mat[:, pos0] -= candidates**2 * dx
            new_max = new_c5_mat.max(axis=1)
            
            best_idx = new_max.argmin()
            if new_max[best_idx] < c5_vec.max() - 1e-15:
                d = candidates[best_idx]
                h[i] = np.clip(h[i] + d, 0.0, 1.0)
                c5_vec += d * slopes
                c5_vec[pos0] -= d * d * dx
                sweep_improved = True
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 30:
                search_range = max(0.002, search_range * 0.6)
                if stall_count > 80:
                    break
    
    return best_h, best_c

def pair_polish(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return best_h, best_c

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = random_binary_start(500, rng)
        n = 500
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    # Phase 1: L-BFGS-B
    if time.time() < deadline - 300:
        t1 = min(time.time() + 120, deadline - 180)
        for alpha in [3000, 8000, 15000]:
            if time.time() > t1:
                break
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 150, 'ftol': 1e-17, 'gtol': 1e-15})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    
    # Phase 2: Coordinate descent (main workhorse)
    if time.time() < deadline - 60:
        h, best_c = coord_desc(h, n, min(time.time() + 400, deadline - 30),
                              max_sweeps=500, init_range=0.06, n_grid=25)
    
    # Phase 3: Second coord desc pass with tighter range
    if time.time() < deadline - 30:
        h, best_c = coord_desc(h, n, min(time.time() + 200, deadline - 10),
                              max_sweeps=300, init_range=0.015, n_grid=31)
    
    # Phase 4: Try n=600
    if time.time() < deadline - 80:
        n2 = 600
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h)
        h2 = project_feasible(h2, n2)
        t2 = min(time.time() + 80, deadline - 20)
        h2, c2 = coord_desc(h2, n2, t2, max_sweeps=200, init_range=0.04, n_grid=21)
        if c2 < best_c:
            best_c = c2
            h = h2
            n = n2
    
    # Phase 5: Final pair polish
    if time.time() < deadline - 2:
        h, best_c = pair_polish(h, n, deadline - 1, radius=30)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c, n
```
