import time
import math
import numpy as np
from scipy.optimize import minimize


def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of(h):
    S = corr_raw(h)
    return float(S.max()) * 2.0 / h.size


def project_obj(h):
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = 0.5 * n
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
            if not np.any(m):
                break
            room = (1.0 - h) * m
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff * room / s
        else:
            m = h > 1e-12
            if not np.any(m):
                break
            mass = h * m
            s = mass.sum()
            if s < 1e-15:
                break
            h = h + diff * mass / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(m):
            h = h.copy()
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def keep_sum(h):
    n = h.size
    diff = 0.5 * n - h.sum()
    if abs(diff) > 1e-15:
        h = h + diff / n
    return h


def build_L(h):
    n = h.size
    m = 2 * n - 1
    L = np.zeros((n, m), dtype=float)
    rev_oh = 1.0 - h[::-1]
    neg = -h
    for r in range(n):
        L[r, r:r + n] += rev_oh
        L[r, n - 1 - r:2 * n - r - 1] += neg
    return L


def max_box_step(h, d):
    eps = 1e-12
    st = np.inf
    pos = (d > 1e-14) & (h < 1.0 - eps)
    if np.any(pos):
        st = min(st, float(np.min((1.0 - h[pos] - eps) / d[pos])))
    neg = (d < -1e-14) & (h > eps)
    if np.any(neg):
        st = min(st, float(np.min((h[neg] - eps) / (-d[neg]))))
    if not np.isfinite(st):
        return 0.0
    return max(0.0, st)


def softmax_weights(S, alpha):
    mx = float(S.max())
    if alpha <= 1e-12:
        return np.full(S.shape, 1.0 / S.size)
    z = alpha * (S - mx)
    ez = np.exp(z)
    s = ez.sum()
    if (not np.isfinite(s)) or s <= 1e-300:
        w = np.zeros(S.shape)
        mask = S >= mx - 1e-12
        c = mask.sum()
        if c == 0:
            w[np.argmax(S)] = 1.0
        else:
            w[mask] = 1.0 / c
        return w
    return ez / s


def smooth_obj(x, alpha):
    S = corr_raw(x)
    mx = float(S.max())
    if alpha <= 1e-12:
        return mx
    z = alpha * (S - mx)
    ez = np.exp(z)
    s = ez.sum()
    if (not np.isfinite(s)) or s <= 1e-300:
        return mx
    return float(mx + math.log(float(s)) / alpha)


def smooth_jac(x, alpha):
    S = corr_raw(x)
    w = softmax_weights(S, alpha)
    return build_L(x) @ w


def sum_con(x, n):
    return float(x.sum() - 0.5 * n)


def run_slsqp(h, c, deadline, alphas):
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                smooth_obj, h,
                jac=smooth_jac,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 60, 'ftol': 1e-12, 'disp': False},
                args=(alpha,)
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                xx = project_obj(res.x)
                cc = c5_of(xx)
                if cc < c - 1e-15:
                    h, c = xx, cc
        except Exception:
            pass
    h = project_obj(h)
    c = c5_of(h)
    return h, c


def smooth_descent(h, c, deadline, alphas):
    h = h.copy()
    c = c5_of(h)
    best_h = h.copy()
    best_c = c
    fracs = (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 0.6, 1.0)

    for alpha in alphas:
        if time.time() > deadline:
            break
        no = 0
        while time.time() < deadline:
            S = corr_raw(h)
            c = float(S.max()) * 2.0 / h.size
            if c < best_c:
                best_c = c
                best_h = h.copy()

            w = softmax_weights(S, alpha)
            G = build_L(h) @ w
            d = -(G - G.mean())
            if not np.all(np.isfinite(d)):
                break
            if np.linalg.norm(d) < 1e-8:
                break

            ms = max_box_step(h, d)
            if ms < 1e-11:
                break

            base = min(ms, 0.25)
            best_try = None
            best_try_c = np.inf

            for fr in fracs:
                t = fr * base
                if t <= 1e-16:
                    continue
                ht = keep_sum(h + t * d)
                if ht.min() < 0.0 or ht.max() > 1.0:
                    ht = np.clip(ht, 0.0, 1.0)
                    ht = keep_sum(ht)
                ct = float(corr_raw(ht).max()) * 2.0 / ht.size
                if ct < best_try_c:
                    best_try_c = ct
                    best_try = ht

            if best_try is not None and best_try_c < c - 1e-13:
                h, c = best_try, best_try_c
                no = 0
            else:
                no += 1
                if no >= 4:
                    break

        if time.time() > deadline:
            break

    h = project_obj(h)
    c = c5_of(h)
    if c < best_c:
        best_h, best_c = h, c
    return project_obj(best_h), c5_of(project_obj(best_h))


def pair_raw_max(S, Ld, t, d, n):
    vals = S + Ld * t
    t2 = t * t
    idx = n - 1
    if d > 0:
        vals[idx - d] += t2
        vals[idx + d] += t2
    vals[idx] -= 2.0 * t2
    return float(vals.max())


def pair_search(S, L, h, p, q):
    n = h.size
    Tmax = min(h[p], 1.0 - h[q])
    if Tmax <= 1e-12:
        return 0.0
    Ld = L[q] - L[p]
    d = abs(p - q)
    base_raw = float(S.max())
    fracs = (1e-7, 1e-6, 3e-6, 1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3,
             1e-2, 3e-2, 0.1, 0.3, 0.6, 1.0)
    best_t = 0.0
    best_raw = base_raw
    for fr in fracs:
        t = fr * Tmax
        if t <= 1e-16:
            continue
        raw = pair_raw_max(S, Ld, t, d, n)
        if raw < best_raw - 1e-14:
            best_raw = raw
            best_t = t
    return best_t


def pair_descent(h, c, deadline, alphas, seed=7):
    n = h.size
    best_h = h.copy()
    best_c = c
    rng = np.random.RandomState(seed)

    for alpha in alphas:
        if time.time() > deadline:
            break
        no = 0
        while time.time() < deadline:
            S = corr_raw(h)
            c = float(S.max()) * 2.0 / n
            if c < best_c:
                best_c = c
                best_h = h.copy()

            w = softmax_weights(S, alpha)
            L = build_L(h)
            G = L @ w
            order = np.argsort(G)
            K = 5
            cand = []
            seen = set()

            for ii in range(n - K, n):
                p = int(order[ii])
                for jj in range(0, K):
                    q = int(order[jj])
                    if p == q:
                        continue
                    if h[p] <= 1e-12 or h[q] >= 1.0 - 1e-12:
                        continue
                    if G[p] <= G[q] + 1e-7:
                        continue
                    key = (p, q)
                    if key not in seen:
                        seen.add(key)
                        cand.append(key)

            mm = float(S.max())
            act = np.argsort(S)[-4:]
            for k_idx in act:
                k = int(k_idx)
                if S[k] < mm - 0.25:
                    continue
                col = L[:, k]
                p = int(np.argmax(col))
                q = int(np.argmin(col))
                if p != q and h[p] > 1e-12 and h[q] < 1.0 - 1e-12:
                    key = (p, q)
                    if key not in seen:
                        seen.add(key)
                        cand.append(key)

            if len(cand) < 6:
                for _ in range(8):
                    p = int(rng.randint(n))
                    q = int(rng.randint(n))
                    if p != q and h[p] > 1e-12 and h[q] < 1.0 - 1e-12 and G[p] > G[q] + 1e-9:
                        key = (p, q)
                        if key not in seen:
                            seen.add(key)
                            cand.append(key)

            improved = False
            for p, q in cand:
                t = pair_search(S, L, h, p, q)
                if t <= 1e-14:
                    continue
                ht = h.copy()
                ht[p] -= t
                ht[q] += t
                ht = np.clip(ht, 0.0, 1.0)
                ht = keep_sum(ht)
                ct = float(corr_raw(ht).max()) * 2.0 / n
                if ct < c - 1e-13:
                    h, c = ht, ct
                    improved = True
                    break

            if not improved:
                no += 1
                if no >= 2:
                    break

        if time.time() > deadline:
            break

    h = project_obj(h)
    c = c5_of(h)
    if c < best_c:
        best_h, best_c = h, c
    return project_obj(best_h), c5_of(project_obj(best_h))


def resample_h(h, nnew):
    n = h.size
    xold = (np.arange(n) + 0.5) * (2.0 / n)
    xnew = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    hn = np.interp(xnew, xold, h)
    return project_obj(hn)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 6.0

    try:
        init = initial_h_values
        h = np.asarray(init, dtype=float).copy()
        if h.ndim != 1:
            h = h.reshape(-1)
        if h.size == 0 or not np.all(np.isfinite(h)):
            raise ValueError
    except Exception:
        h = np.linspace(0.0, 1.0, 250)

    h = project_obj(h)
    c = c5_of(h)
    best_h = h.copy()
    best_c = c
    best_n = h.size

    p1 = t0 + min(0.14 * budget, 200.0)
    if time.time() < p1:
        h, c = run_slsqp(h, c, p1, (5.0, 20.0, 100.0))
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = h.size

    p2 = p1 + min(0.28 * budget, 450.0)
    if time.time() < p2:
        h, c = smooth_descent(best_h, best_c, p2, (0.1, 0.5, 2.0, 10.0, 50.0))
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = h.size

    p3 = p2 + min(0.30 * budget, 500.0)
    if time.time() < p3:
        h, c = pair_descent(best_h, best_c, p3, (1.0, 5.0, 20.0, 100.0), seed=seed)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    p4 = t0 + 0.87 * budget

        for nh in (200, 230, 270, 300, 350):
            if time.time() > p4:
                break
            if nh == best_n:

            hn = resample_h(best_h, nh)

    if time.time() < deadline:
        h, c = pair_descent(best_h, best_c, deadline, (50.0, 200.0), seed=seed + 123)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = h.size

    best_h = project_obj(best_h)
    best_c = c5_of(best_h)
    return best_h, best_c, best_h.size
