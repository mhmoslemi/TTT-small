    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    zeros_arr = np.where(h == 0)[0].copy()
    ones_arr = np.where(h == 1)[0].copy()
    T = 0.02
    cool = 0.99997
    scale = 2000.0
    staggard = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if staggard > 50000 and rng.random() < 0.01:
            staggard = 0
            T = max(T, 0.005)
        else:
            staggard += 1
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i, j = int(zeros_arr[ai]), int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range; msk = (im >= 0) & (im < n); delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range; msk = (imn >= 0) & (imn < n); delta[msk] -= h[imn[msk]]
        jm = j + m_range; msk = (jm >= 0) & (jm < n); delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range; msk = (jmn >= 0) & (jmn < n); delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i); k2 = n - 1 + (i - j)
        if 0 <= k1 < 2 * n - 1: delta[k1] += 1.0
        if 0 <= k2 < 2 * n - 1: delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j; ones_arr[bi] = i
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                staggard = 0
        T *= cool
    return best_h, best_c

def multi_res(h_in, n_in, h_best, best_c, deadline):
    n = n_in
    best_h = h_best.copy()
    for nh in [200, 300, 500]:
        if time.time() > deadline:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hb, cb = greedy_binary(hh, nh, deadline - 20)
        if cb < compute_c5(hh, nh):
            hh = hb
        hh, ch = cont_opt(hh, nh, deadline - 5, [1e4, 1e5, 1e6])
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
    return best_h, best_c, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: binary greedy
    p1 = t0 + 0.06 * (dl - t0)
    hb, cb = greedy_binary(h0, n, p1)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: continuous L-BFGS-B
    p2 = t0 + 0.35 * (dl - t0)
    alphas2 = [200, 1000, 5000, 20000, 100000, 1e6]
    best_h, best_c = cont_opt(best_h, n, p2, alphas2)

    # Phase 3: annealing (multiple restarts)
    p3 = t0 + 0.70 * (dl - t0)
    per_restart = max(30000, int(0.30 * (dl - t0) / 0.02))
    for rs in range(4):
        if time.time() > p3:
            break
        seed_r = seed + rs * 1000
        has, cas = anneal_binary(best_h, n, p3 - rs * 2, seed_r, 5000000)
        if cas < best_c:
            best_c, best_h = cas, has.copy()
        # Quick continuous polish from annealed
        if time.time() < p3 - 3:
            best_h, best_c = cont_opt(best_h, n, p3 - 2, [5e4, 5e5, 5e6])

    # Phase 4: multi-resolution
    p4 = t0 + 0.88 * (dl - t0)
    best_h, best_c, n = multi_res(h0, len(h0), best_h, best_c, p4)

    # Phase 5: final polish
    if time.time() < dl - 3:
        best_h, _ = cont_opt(best_h, n, dl - 2, [1e6, 1e7, 1e8])
        # Try one more anneal + polish cycle
        if time.time() < dl - 5:
            has, cas = anneal_binary(best_h, n, dl - 4, seed + 9999, 2000000)
            if cas < best_c:
                best_c = cas
                best_h = has
                if time.time() < dl - 2:
                    best_h, _ = cont_opt(best_h, n, dl - 1, [1e7])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
