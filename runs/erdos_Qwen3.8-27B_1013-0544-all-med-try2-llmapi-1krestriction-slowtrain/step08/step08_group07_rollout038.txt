        sm = shifted.max()

        m = g.max()

        sm = shifted.max()

        weights /= weights.sum()

        # where a = j + m - (n-1), b = j - m + (n-1) (with bounds)
        # First term: convolve weights with (1-x) appropriately
        # Let's compute directly for clarity (n=400 is fine)
        grad_out = np.zeros(n)
        w = weights * dx
        for j in range(n):
            # a = j + m - (n-1), valid when 0 <= a < n, i.e., (n-1)-j <= m <= (n-1)-j+n-1
            a_lo = 0
            a_hi = n - 1
            m_lo = max(0, (n-1) - j)
            m_hi = min(2*n-2, (n-1) - j + n - 1)
            if m_lo < m_hi + 1:
                # a ranges from j+m_lo-(n-1) to j+m_hi-(n-1)
                a_start = j + m_lo - (n-1)
                a_end = j + m_hi - (n-1)
                grad_out[j] += np.sum(w[m_lo:m_hi+1] * (1 - x[a_start:a_end+1]))
            # b = j - m + (n-1), valid when 0 <= b < n, i.e., j-(n-1) <= m <= j+(n-1)
            m_lo2 = max(0, j - (n-1))
            m_hi2 = min(2*n-2, j + (n-1))
            if m_lo2 < m_hi2 + 1:
                b_start = j - m_hi2 + (n-1)
                b_end = j - m_lo2 + (n-1)
                # w[m_lo2:m_hi2+1] * x[b_end:b_start-1:-1] -- reversed
                # b = j - m + (n-1), so when m = m_lo2, b = b_end; when m = m_hi2, b = b_start
                grad_out[j] -= np.sum(w[m_lo2:m_hi2+1] * x[b_end:b_start-1:-1])
        return grad_out
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfun': 200000})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_to_continuous(h_b, n, fraction):

                A_ub[m, 1:] = B[m, :]

            for bi in range(block_size):
                A_ub[2*n-1+bi, bi+1] = -1.0
                b_ub[2*n-1+bi] = 0.0

                A_ub[2*n-1+block_size+bi, bi+1] = 1.0
                b_ub[2*n-1+block_size+bi] = 1.0

                for bi, i in enumerate(idx):
                    corr += B[:, bi] * (x_opt[bi] - h[i])
                    h[i] = x_opt[bi]
    
    return h, corr

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=100, deadline=None, bisect_iters=30):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = float(np.max(corr))
            val1 = float(np.max(corr + B))
            lo_v = min(val0, val1) - 1e-10
            hi_v = max(val0, val1) + 1e-10
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = float(np.max(corr[zero_B])) if zero_B.any() else -np.inf
            for _ in range(bisect_iters):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, float(np.max((mid - corr[neg_B]) / B[neg_B])))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, float(np.min((mid - corr[pos_B]) / B[pos_B])))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, float(np.max((hi_v - corr[neg_B]) / B[neg_B])))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, float(np.min((hi_v - corr[pos_B]) / B[pos_B])))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def smooth_lbfgs(h, n, alpha, penalty, maxiter=3000, deadline=None):
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        lse = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        s = x.sum() - n / 2.0
        return lse + penalty * s * s
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxfun': 100000})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    hc, _ = coord_descent(best_h, n, max_sweeps=150, deadline=p4, bisect_iters=30)

    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    
    # Phase 5: Higher alpha smooth
    p5 = t0 + 0.80 * total
    h = best_h.copy()
    for alpha in [300000, 1000000, 3000000]:
        if time.time() > p5 - 5:
            break
        h = smooth_lbfgs(h, n, alpha, 200.0, maxiter=3000, deadline=p5 - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 6: Multi-resolution
    p6 = t0 + 0.90 * total
    for nh in [100, 200, 300]:
        if time.time() > p6 - 15:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=40, deadline=p6 - 12, bisect_iters=25)
        hh = project_feasible(hh, nh)
        hh = smooth_lbfgs(hh, nh, 10000, 50.0, maxiter=1000, deadline=p6 - 10)
        hh_full = np.interp(xo, xn, hh)
        hh_full = project_feasible(hh_full, n)
        hh_full, _ = coord_descent(hh_full, n, max_sweeps=60, deadline=p6 - 12, bisect_iters=28)
        hh_full = project_feasible(hh_full, n)
        ch = compute_c5(hh_full, n)
        if ch < best_c:
            best_c, best_h = ch, hh_full.copy()
    
    # Phase 7: Noise restart + refine
    if time.time() < dl - 15:
        for trial in range(5):
            if time.time() > dl - 5:
                break
            h = best_h + 0.004 * rng.randn(n)
            h = project_feasible(h, n)
            h, _ = coord_descent(h, n, max_sweeps=40, deadline=dl - 10, bisect_iters=25)
            h = project_feasible(h, n)
            h = smooth_lbfgs(h, n, 500000, 200.0, maxiter=2000, deadline=dl - 6)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
