import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx

            grad[:nv] += wm * (1.0 - h[k:])

            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])

    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def exact_pair_step(h, n, i, j):
    """Find optimal h[i] given h[j]=c_ij-h[i]. Returns (new_hi, new_c)."""
    c_ij = h[i] + h[j]
    lo = max(0.0, c_ij - 1.0)
    hi = min(1.0, c_ij)
    if lo >= hi - 1e-12:
        # Can't move
        t = max(lo, min(hi, h[i]))
        h_new = h.copy()
        h_new[i] = t
        h_new[j] = c_ij - t
        return h_new, compute_c5_vec(h_new, n)
    
    nc = 2 * n - 1
    r_arr = np.arange(nc) - (n - 1)

    partA_i = np.zeros(nc)
    v = (i + r_arr >= 0) & (i + r_arr < n)
    partA_i[v] = 1.0 - h[i + r_arr[v]]
    partB_i = np.zeros(nc)
    v2 = (i - r_arr >= 0) & (i - r_arr < n)
    partB_i[v2] = -h[i - r_arr[v2]]

    partA_j = np.zeros(nc)
    v = (j + r_arr >= 0) & (j + r_arr < n)
    partA_j[v] = 1.0 - h[j + r_arr[v]]
    partB_j = np.zeros(nc)
    v2 = (j - r_arr >= 0) & (j - r_arr < n)
    partB_j[v2] = -h[j - r_arr[v2]]
    g_j = partA_j + partB_j
    
    a = g_i - g_j
    c_current = np.correlate(h, 1 - h, mode='full')
    b = c_current - a * h[i]

    for _ in range(40):

            lo = mid

    h_new[i] = t
    h_new[j] = c_ij - t
    return h_new, compute_c5_vec(h_new, n)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            continue

            grad[k:] += wm * (-h[:nv])
        else:
