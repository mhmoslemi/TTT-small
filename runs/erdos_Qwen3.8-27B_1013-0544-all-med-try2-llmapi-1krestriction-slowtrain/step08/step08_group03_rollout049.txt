import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _grad_single(h, n, k):
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:])
        grad[k:] -= h[:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] -= h[kk:]
    return grad

def grad_smooth(h, n, alpha, top_k=50):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    sh = alpha * (c - c.max())
    sh -= sh.max()
    w = np.exp(sh)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(w))
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * _grad_single(h, n, k)
    return dx * grad

def nesterov_loop(h, n, deadline, alpha=200, top_k=40, init_step=0.04, momentum=0.85):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    v = np.zeros(n)
    t = 1.0
    step = init_step
    stall = 0
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = np.clip(h + beta * (h - y), 0.0, 1.0)
        g = grad_smooth(y, n, alpha, top_k)
        v = momentum * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 15 == 0:
                step *= 0.6
                momentum *= 0.97
                if step < 1e-7:
                    step = init_step
                    momentum = 0.85
                    t = 1.0
        if c_new <= best_c + 1e-10:
            h = h_new
        else:
            h = best_h.copy()
        t = t_new
        if stall > 300:
            break
    return best_h, best_c

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 200.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha):
    g = grad_smooth(x, n, alpha, top_k=40)
    s = x.sum() - n / 2.0
    g += 400.0 * s
    return g

def thorough_pair_polish(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        passes = 0
        while improved and time.time() < deadline:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def neighborhood_pair_polish(h, n, deadline, radius=60):

                            improved = True
                        else:

def binary_round_and_optimize(h, n, deadline):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = project_feasible(h_bin, n)
    if abs(h_bin.sum() - n / 2.0) > 0.5:
        return h.copy(), compute_c5(h, n)
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    c_orig = compute_c5(h, n)
    if best_c >= c_orig:
        return h.copy(), c_orig
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                best_h[i] = 0.0
                best_h[j] = 1.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
    return best_h, best_c
