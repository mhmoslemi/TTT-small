import time
import math
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize, linprog


def compute_c5_vec(h, n):
    dx = 2.0 / float(n)
    return correlate(h, 1.0 - h, mode='full', method='fft') * dx


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def project_box_target(z, target, lower, upper):
    z = np.asarray(z, dtype=float).ravel()
    n = z.size
    lower = np.full(n, float(lower), dtype=float) if np.isscalar(lower) else np.asarray(lower, dtype=float).ravel()
    upper = np.full(n, float(upper), dtype=float) if np.isscalar(upper) else np.asarray(upper, dtype=float).ravel()
    if lower.size != n:
        lower = np.zeros(n, dtype=float)
    if upper.size != n:
        upper = np.ones(n, dtype=float)

    if not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=float)

    if np.all(z >= lower - 1e-13) and np.all(z <= upper + 1e-13) and abs(float(np.sum(z)) - float(target)) < 1e-13:
        return z.copy()

    lo_target = float(np.sum(lower))
    hi_target = float(np.sum(upper))
    target = float(np.clip(target, lo_target, hi_target))

    lo = float(np.min(lower - z)) - 1.0
    hi = float(np.max(upper - z)) + 1.0
    if not (lo < hi):
        lo, hi = -1.0, 2.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z + mid, lower, upper)))
        if s <= target:
            lo = mid
        else:
            hi = mid

    out = np.clip(z + 0.5 * (lo + hi), lower, upper)
    diff = target - float(np.sum(out))

    if abs(diff) > 1e-13:
        mask = (out > lower + 1e-13) & (out < upper - 1e-13)
        if np.any(mask):
            out[mask] += diff / float(np.count_nonzero(mask))
            out = np.clip(out, lower, upper)

    if abs(target - float(np.sum(out))) > 1e-11:
        diff = target - float(np.sum(out))
        if diff > 0.0:
            idx = np.where(out < upper - 1e-12)[0]
        else:
            idx = np.where(out > lower + 1e-12)[0]
        if idx.size:
            out[idx] += diff / float(idx.size)
            out = np.clip(out, lower, upper)

    return out


def project_feasible(h, n):
    return project_box_target(h, 0.5 * float(n), np.zeros(n, dtype=float), np.ones(n, dtype=float))


def grad_single(h, n, k):
    nv = n - abs(k)
    g = np.zeros(n, dtype=float)
    if k >= 0:
        g[:nv] = 1.0 - h[k:]
        g[k:] -= h[:nv]
    else:
        kk = -k
        g[kk:] = 1.0 - h[:nv]
        g[:nv] -= h[kk:]
    return g


def grad_active(h, n, A):
    A = np.asarray(A, dtype=int)
    G = np.zeros((len(A), n), dtype=float)
    off = int(n) - 1
    for i, idx in enumerate(A):
        G[i, :] = grad_single(h, n, int(idx) - off)
    return G


def top_active_indices(c, max_count, margin=0.0):
    c = np.asarray(c, dtype=float)
    max_count = int(max_count)
    if max_count <= 0:
        return np.array([], dtype=int)
    max_count = min(max_count, len(c))
    C = float(np.max(c))
    order = np.argsort(c, kind='stable')[::-1]
    idx = order[:max_count]
    if margin > 0.0:
        near = np.where(c >= C - margin)[0]
        if near.size:
            combined = np.unique(np.concatenate((idx, near)))
            if combined.size > max_count:
                combined = combined[np.argsort(c[combined], kind='stable')[::-1]][:max_count]
            idx = combined
    return idx[np.argsort(c[idx], kind='stable')[::-1]]


def active_obj(z, n):
    return float(z[-1])


def active_obj_jac(z, n):
    out = np.zeros(int(n) + 1, dtype=float)
    out[-1] = 1.0
    return out


def active_ineq(z, n, A):
    h = z[:-1]
    t = float(z[-1])
    c = compute_c5_vec(h, n)
    return np.asarray(t - c[np.asarray(A, dtype=int)], dtype=float)


def active_ineq_jac(z, n, A):
    h = z[:-1]
    A = np.asarray(A, dtype=int)
    G = grad_active(h, n, A)
    J = np.zeros((len(A), int(n) + 1), dtype=float)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def sum_constraint(z, n):
    return float(np.sum(z[:-1]) - 0.5 * float(n))


def sum_constraint_jac(z, n):
    out = np.zeros(int(n) + 1, dtype=float)
    out[:n] = 1.0
    return out


def solve_active_nlp(h, n, A, t_init, maxiter):
    A = np.asarray(A, dtype=int)
    if len(A) == 0:
        return None
    z = np.concatenate((np.asarray(h, dtype=float), np.array([float(t_init)], dtype=float)))
    bounds = [(0.0, 1.0)] * int(n)
    bounds.append((0.0, max(float(t_init) + 0.05, 1.0)))
    ineq = {'type': 'ineq', 'fun': active_ineq, 'jac': active_ineq_jac, 'args': (int(n), A)}
    eq = {'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (int(n),)}
    try:
        return minimize(
            active_obj, z, jac=active_obj_jac, args=(int(n),),
            method='SLSQP', bounds=bounds, constraints=[ineq, eq],
            options={'maxiter': int(maxiter), 'ftol': 1e-13, 'iprint': 0}
        )
    except Exception:
        return None


def random_perturb(h, n, rng, scale):
    noise = rng.normal(0.0, float(scale), int(n))
    noise -= float(np.mean(noise))
    return project_feasible(h + noise, n)

def compute_path_terms(h, n, d):
    dx = 2.0 / float(n)
    L = (correlate(d, 1.0 - h, mode='full', method='fft') -
         correlate(h, d, mode='full', method='fft')) * dx
    Q = correlate(d, d, mode='full', method='fft') * dx
    return L, Q


def path_value(a, c0, L, Q):
    return float(np.max(c0 + a * L - (a * a) * Q))


def golden_path_values(c0, L, Q, a_left, a_right):
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = a_right - gr * (a_right - a_left)
    x2 = a_left + gr * (a_right - a_left)
    f1 = path_value(x1, c0, L, Q)
    f2 = path_value(x2, c0, L, Q)
    for _ in range(35):
        if f1 < f2:
            a_right = x2
            x2 = x1
            f2 = f1
            x1 = a_right - gr * (a_right - a_left)
            f1 = path_value(x1, c0, L, Q)
        else:
            a_left = x1
            x1 = x2
            f1 = f2
            x2 = a_left + gr * (a_right - a_left)
            f2 = path_value(x2, c0, L, Q)
    a = 0.5 * (a_left + a_right)
    return a, path_value(a, c0, L, Q)


def line_search_c(c0, L, Q):
    alphas = np.linspace(0.0, 1.0, 25)
    aa = alphas[:, None]
    vals = np.max(c0[None, :] + aa * L[None, :] - (aa * aa) * Q[None, :], axis=1)
    if not np.all(np.isfinite(vals)):
        return 0.0, float(np.max(c0))
    i = int(np.argmin(vals))
    a = float(alphas[i])
    v = float(vals[i])
    if 0 < i < len(alphas) - 1:
        a, v = golden_path_values(c0, L, Q, float(alphas[i - 1]), float(alphas[i + 1]))
    return a, v


def lp_direction(h, n, c0, A, mu):
    if mu <= 1e-15:
        return None
    A = np.asarray(A, dtype=int)
    if len(A) == 0:
        return None

    lower = np.maximum(-h, -float(mu))
    upper = np.minimum(1.0 - h, float(mu))
    bad = lower > upper
    if np.any(bad):
        lower[bad] = 0.5 * (lower[bad] + upper[bad])
        upper[bad] = lower[bad]

    J = grad_active(h, n, A)
    r, nvars = J.shape
    A_ub = np.empty((r, nvars + 1), dtype=float)
    A_ub[:, :nvars] = J
    A_ub[:, nvars] = -1.0
    b_ub = np.asarray(c0[A], dtype=float)
    A_eq = np.ones((1, nvars), dtype=float)
    b_eq = np.zeros(1, dtype=float)

    bounds = list(zip(lower, upper))
    bounds.append((0.0, float(np.max(c0)) + 1.0))
    cvec = np.zeros(nvars + 1, dtype=float)
    cvec[-1] = 1.0

    res = None
    for meth in ('HiGHS', 'highs', None):
        try:
            if meth is None:
                res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds)
            else:
                res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method=meth)
            if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                break
        except Exception:
            pass

    if res is None or res.x is None:
        return None
    d = np.asarray(res.x[:nvars], dtype=float)
    if not np.all(np.isfinite(d)):
        return None
    d = project_box_target(d, 0.0, -h, 1.0 - h)
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    return d

def retarget(h, n, m):
    m = int(m)
    if m <= 0 or m == int(n):
        return project_feasible(h, n)
    old = np.linspace(0.0, 2.0, int(n), endpoint=False)
    new = np.linspace(0.0, 2.0, m, endpoint=False)
    y = np.interp(new, old, np.asarray(h, dtype=float))
    return project_feasible(y, m)

    if time.time() < t3 - 5:
        c0 = compute_c5_vec(best_h, best_n)
        best_h, best_c = active_lp_polish(best_h, best_n, c0, min(t3, deadline), rng, 0.03)
        best_n = len(best_h)

    if time.time() < deadline - 220:
        m = 600 if best_n < 600 else 700
        h2 = retarget(best_h, best_n, m)
        c2 = compute_c5_vec(h2, m)
        b2 = float(np.max(c2))
        if time.time() < deadline - 80:
            h2, b2 = active_lp_polish(h2, m, c2, min(deadline - 50, time.time() + 90.0), rng, 0.03)
        if b2 < best_c - 1e-12:
            best_h = h2
            best_c = float(b2)
            best_n = m

    if time.time() < deadline - 70 and best_n % 2 == 0:
        hb = to_binary(best_h, best_n)
        if hb is not None:
            cb = compute_c5(hb, best_n)
            if cb < best_c:
                best_h = hb
                best_c = cb
            alloc = min(70.0, deadline - time.time() - 20.0)
            if alloc > 5.0:
                hb2, cb2 = binary_polish(hb, best_n, rng, time.time() + alloc,
                                         int(min(6000, alloc * 110)))
                if cb2 < best_c - 1e-13:
                    best_h = hb2
                    best_c = cb2

    if time.time() < deadline - 25:
        c0 = compute_c5_vec(best_h, best_n)
        best_h, best_c = active_lp_polish(best_h, best_n, c0, deadline - 3.0, rng, 0.01)
        best_n = len(best_h)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
