import numpy as np
from scipy.optimize import minimize
import time

_CUR_ALPHA = 1.0
_CUR_WEIGHTS = None
_CUR_PEN = 0.0


def project_feasible(h):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mu = 0.5 * (lo + hi)
        s = np.clip(h + mu, 0.0, 1.0).sum()
        if s < target:
            lo = mu
        else:
            hi = mu
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if diff != 0.0:
        if diff > 0.0:
            idx = np.flatnonzero(1.0 - h > 1e-15)
            if idx.size:
                h[idx] += diff / idx.size
        else:
            idx = np.flatnonzero(h > 1e-15)
            if idx.size:
                h[idx] += diff / idx.size
    return np.clip(h, 0.0, 1.0)


def exact_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def full_c_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def top_weights(c, width):
    c = np.asarray(c, dtype=float)
    m = float(c.max())
    w = np.exp((c - m) / max(1e-8, width))
    s = w.sum()
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    return w / s


def causal_conv(ker, x):
    n = x.size
    if n < 2 or ker.size == 0:
        return np.zeros(n)
    y = np.zeros(n)
    y[1:] = np.convolve(ker, x, mode="full")[:n - 1]
    return y


def anti_conv(ker, x):
    return causal_conv(ker, x[::-1])[::-1]


def jac_from_weights(x, w):
    n = x.size
    if w.size != 2 * n - 1:
        return np.zeros(n)
    dx = 2.0 / n
    g = w[n - 1] * (1.0 - 2.0 * x)
    if n > 1:
        p = w[n:]
        q = w[:n - 1][::-1]
        xm = 1.0 - x
        g = g + causal_conv(p, xm) - causal_conv(q, x) + anti_conv(q, xm) - anti_conv(p, x)
    return dx * g


def logsumexp_fun(x):
    c = full_c_values(x)
    alpha = _CUR_ALPHA
    mx = float(c.max())
    z = alpha * (c - mx)
    s = np.exp(z).sum()
    return float(mx + np.log(s) / alpha)


def logsumexp_jac(x):
    c = full_c_values(x)
    alpha = _CUR_ALPHA
    mx = float(c.max())
    z = alpha * (c - mx)
    w = np.exp(z)
    w /= w.sum()
    return jac_from_weights(x, w)


def weighted_sum_fun(x):
    w = _CUR_WEIGHTS
    if w is None or w.size != 2 * len(x) - 1:
        return 0.0
    return float(np.dot(w, full_c_values(x)))


def weighted_sum_jac(x):
    w = _CUR_WEIGHTS
    if w is None or w.size != 2 * len(x) - 1:
        return np.zeros_like(x)
    return jac_from_weights(x, w)


def penalty_logsum_fun(x):
    val = logsumexp_fun(x)
    pen = _CUR_PEN
    if pen == 0.0:
        return val
    s = x.sum() - 0.5 * len(x)
    return float(val + 0.5 * pen * s * s)


def penalty_logsum_jac(x):
    g = logsumexp_jac(x)
    pen = _CUR_PEN
    if pen == 0.0:
        return g
    s = x.sum() - 0.5 * len(x)
    return g + pen * s * np.ones_like(g)


def eq_sum(x):
    return float(x.sum() - 0.5 * len(x))


def eq_sum_jac(x):
    return np.ones_like(x)


def slsqp_with(h, fun, jac, maxiter, ftol):
    hb = project_feasible(h)
    bounds = [(0.0, 1.0)] * len(hb)
    cons = [{"type": "eq", "fun": eq_sum, "jac": eq_sum_jac}]
    opts = {"maxiter": int(maxiter), "ftol": float(ftol), "disp": False}
    try:
        res = minimize(fun, hb, method="SLSQP", jac=jac, bounds=bounds,
                       constraints=cons, options=opts)
    except Exception:
        return hb
    if hasattr(res, "x") and np.all(np.isfinite(res.x)):
        return project_feasible(res.x)
    return hb


def lbfgs_with(h, fun, jac, maxiter, ftol):
    hb = project_feasible(h)
    bounds = [(0.0, 1.0)] * len(hb)
    opts = {"maxiter": int(maxiter), "ftol": float(ftol), "disp": False}
    try:
        res = minimize(fun, hb, method="L-BFGS-B", jac=jac, bounds=bounds, options=opts)
    except Exception:
        return hb
    if hasattr(res, "x") and np.all(np.isfinite(res.x)):
        return project_feasible(res.x)
    return hb


def binary_top(h):
    n = len(h)
    k = int(round(0.5 * n))
    if k <= 0 or k >= n:
        k = n // 2
    hb = np.zeros(n)
    idx = np.argsort(h)[::-1][:k]
    hb[idx] = 1.0
    return project_feasible(hb)


def binary_fourier_start(n, rng, k=12):
    x = np.linspace(0.0, 2.0, n)
    z = np.zeros(n)
    for m in range(1, k + 1):
        z += rng.normal() * np.cos(np.pi * m * x) + rng.normal() * np.sin(np.pi * m * x)
    if np.linalg.norm(z) < 1e-8:
        z = rng.normal(size=n)
    hb = (z > np.quantile(z, 0.5)).astype(float)
    return project_feasible(hb)


def smooth_fourier_start(n, rng, k=8, amp=0.2):
    x = np.linspace(0.0, 2.0, n)
    z = np.zeros(n)
    for m in range(1, k + 1):
        z += rng.normal() * np.cos(np.pi * m * x) + rng.normal() * np.sin(np.pi * m * x)
    nv = np.linalg.norm(z)
    if nv > 1e-8:
        z = amp * z / nv
    return project_feasible(0.5 + z)


def resample_values(h, nh):
    n = len(h)
    if nh <= 1 or nh == n:

    return project_feasible(hh)
