def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (target / tot) if tot > 0 else 1.0
    return np.clip(h, 0.0, 1.0)

def smax_val(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha

def smax_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    grad = np.ones(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        ok = abs(k)
        nv = n - ok
        wm = w[mi]
        if k >= 0:
            grad[:nv] -= wm * h[k:]
            grad[k:] -= wm * h[:nv]
        else:
            grad[ok:] -= wm * h[:nv]
            grad[:nv] -= wm * h[ok:]
    return grad

def pen_obj(x, n, alpha):
    return smax_val(x, n, alpha) + 0.0  # constraint handled by projection

            return smax_val(x, n, alpha) + 1e6 * (x.sum() - n/2)**2

            g = smax_grad(x, n, alpha)
            g += 2e6 * (x.sum() - n/2) * np.ones(n)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-14})

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

**Key algorithmic insight**: Since we want to maximize min_k R_f(k), and R_f(k) is related to the periodogram of f, we can use **spectral methods**:
1. Design f in the frequency domain to have a flat magnitude spectrum (which gives flat autocorrelation)
2. Project back to the time domain with the constraint f ∈ [-1,1]

This is like designing a **minimum-peak sidelobe (MPSL)** sequence, but in continuous time.

For the continuous case, one approach is:
1. Start with a candidate f
2. Compute its spectrum
3. Boost frequencies where the "sidelobes" are too low
4. Project back to [-1,1]

This is essentially the **alternating projections** approach I mentioned earlier.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_val(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha

def smax_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    grad = np.ones(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        ok = abs(k)
        nv = n - ok
        wm = w[mi]
        if k >= 0:
            grad[:nv] -= wm * h[k:]
            grad[k:] -= wm * h[:nv]
        else:
            grad[ok:] -= wm * h[:nv]
            grad[:nv] -= wm * h[ok:]
    return grad

def lbfgs_optimize(h, n, deadline, alpha_list=None, max_iters=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alpha_list:
        if time.time() > deadline - 4:
            break
        if deadline - time.time() < 8:
            break
        mu = 1e5
        def obj(x):
            return smax_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def grad(x):
            g = smax_grad(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-17,
                                    'gtol': 1e-15, 'maxfun': max_iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_ls(h, n, deadline, rng, max_rounds=1000):

    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()

    max_val = c_corr.max()

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        improved = False
        n_try = min(3000, max(500, int((deadline - time.time()) * 1500)))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0
            h_bin[j] = 1
            c_corr = np.correlate(h_bin, 1 - h_bin, mode='full')

def spectral_start(n, rng):
    f = np.zeros(n)
    F = np.fft.rfft(f)
    nfreq = len(F)
    phase = rng.uniform(0, 2 * np.pi, nfreq)
    F[1:] = np.exp(1j * phase[1:])
    F[1] = 0
    f = np.fft.irfft(F, n)
    f = np.sort(f)
    half = n // 2
    thresh = f[half]
    h = (f > thresh).astype(float)
    return project_feasible(h, n)

    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)

        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)

        if w_sum < 1:
