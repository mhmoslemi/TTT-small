def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, lam):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += lam * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, lam):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2.0 * lam * s
    return grad

        w = w / w.sum()
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if w[m_idx] < 1e-13:
                continue
            k = m_idx - (n - 1)
            wm = w[m_idx] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - best_h[k:])
                grad[k:] += wm * (-best_h[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - best_h[:nv])
                grad[:nv] += wm * (-best_h[kk:])
        # Sort by gradient: most negative first (want to increase), most positive last (want to decrease)
        order = np.argsort(grad)
        improved = False
        for p in range(min(k_pairs, n // 3)):
            if time.time() > deadline:
                break
            i = order[p]
            j = order[n - 1 - p]
            if i == j:
                continue
            if grad[i] - grad[j] < -1e-10:
                # Try increasing h[i], decreasing h[j]
                lo, hi = 0.0, min(1.0 - best_h[i], best_h[j])
                if hi <= 0:
                    continue
                for frac in [0.5, 0.3, 0.7, 0.15, 0.85, 0.1, 0.2, 0.4, 0.6]:
                    d = hi * frac
                    if d < 1e-8:
                        continue
                    best_h[i] += d
                    best_h[j] -= d
                    c_val = compute_c5(best_h, n)
                    if c_val < best_c - 1e-14:
                        best_c = c_val
                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                    if improved:
                        break
            elif grad[j] - grad[i] < -1e-10:
                lo, hi = 0.0, min(best_h[i], 1.0 - best_h[j])
                if hi <= 0:
                    continue
                for frac in [0.5, 0.3, 0.7, 0.15, 0.85, 0.1, 0.2, 0.4, 0.6]:
                    d = hi * frac
                    if d < 1e-8:
                        continue
                    best_h[i] -= d
                    best_h[j] += d
                    c_val = compute_c5(best_h, n)
                    if c_val < best_c - 1e-14:
                        best_c = c_val
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                    if improved:
                        break
        if not improved:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline):
    """Exhaustive fine pair search with very small steps"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    for d in steps:
        if time.time() > deadline:
            break
        any_imp = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best_h, best_c
                if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        any_imp = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                if best_h[j] + d <= 1.0 and best_h[i] - d >= 0.0:
                    best_h[j] += d
                    best_h[i] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        any_imp = True
                    else:
                        best_h[j] -= d
                        best_h[i] += d
        if not any_imp:
            break
    return best_h, best_c

def binary_push(h, n, deadline, max_rounds=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        idx = np.argsort(best_h)
        for t in range(min(30, n // 10)):
            if time.time() > deadline:
                break
            i = idx[t]
            j = idx[n - 1 - t]
            if i == j:
                continue
            for step in [0.05, 0.02, 0.01, 0.005]:
                if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                    best_h[i] += step
                    best_h[j] -= step
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] -= step
                        best_h[j] += step
                    if improved:
                        break
        if not improved:
            break
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS on parent
    t1 = min(t0 + 150, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, best_n, t1)

    # Phase 2: Gradient-informed search
    if time.time() < deadline - 400:
        t2 = min(time.time() + 80, deadline - 400)
        h2, c2 = gradient_informed_search(best_h, best_n, t2, k_pairs=300)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Binary push
    if time.time() < deadline - 300:
        t3 = min(time.time() + 60, deadline - 300)
        h3, c3 = binary_push(best_h, best_n, t3)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Try different n values
    n_candidates = [488, 520, 700]
    for n_new in n_candidates:
        if time.time() > deadline - 200:
            break
        alloc = min(80, (deadline - time.time() - 100) * 0.3)
        if alloc < 30:
            break
        h_int = project_feasible(interpolate_h(best_h, best_n, n_new), n_new)
        h_t, c_t = lbfgs_phase(h_int, n_new, time.time() + alloc,
                               alphas=[5000, 50000, 500000, 5000000], max_iters=300)
        if c_t < best_c:
            best_c, best_h, best_n = c_t, h_t, n_new

    # Phase 5: SLSQP on best
    if time.time() < deadline - 150:
        t5 = min(time.time() + 60, deadline - 150)
        h5, c5v = slsqp_phase(best_h, best_n, t5, alpha=10000000, maxiter=4000)
        if c5v < best_c:
            best_c, best_h = c5v, h5

    # Phase 6: Gradient-informed again
    if time.time() < deadline - 100:
        t6 = min(time.time() + 50, deadline - 100)
        h6, c6 = gradient_informed_search(best_h, best_n, t6, k_pairs=500)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Fine pair search (most important for final improvement)
    if time.time() < deadline - 10:
        h7, c7 = fine_pair_search(best_h, best_n, deadline - 5)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Final SLSQP squeeze
    if time.time() < deadline - 2:
        h8, c8 = slsqp_phase(best_h, best_n, deadline - 1, alpha=20000000, maxiter=2000)
        if c8 < best_c:
            best_c, best_h = c8, h8

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
