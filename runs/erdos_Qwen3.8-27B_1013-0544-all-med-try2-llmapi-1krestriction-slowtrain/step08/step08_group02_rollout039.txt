import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for ki in range(2 * n - 1):
        if w[ki] < 1e-13:
            continue
        k = ki - (n - 1)
        wm = w[ki]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * omh[k:]
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * omh[:nv]
            grad[:nv] -= wm * h[kk:]
    return grad

def obj_sm(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha

def run_adam(h, n, alpha, lr, max_iters, deadline):
    h = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        g = smooth_max_grad(h, n, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        mhat = m / (1 - 0.9**t)
        vhat = v / (1 - 0.999**t)
        h = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h = np.clip(h, 0.0, 1.0)

def lbfgs_polish(h, n, deadline, alpha=50000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu = 100.0
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        smax = sh.max()
        sv = m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha
        s = x.sum() - n / 2.0
        return sv + mu * s * s
    def grads(x):
        g = smooth_max_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        g += 2 * mu * s
        return g
    try:
        res = minimize(obj, h, jac=grads, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except:
        pass
    return h, best_c

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14: break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for ki in range(2 * n - 1):
        if w[ki] < 1e-13: continue
        k = ki - (n - 1)
        wm = w[ki]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * omh[k:]
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * omh[:nv]
            grad[:nv] -= wm * h[kk:]
    return grad

def smooth_max_val(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha

def run_adam(h, n, alpha, lr, max_iters, deadline):

    m = np.zeros(n); v = np.zeros(n); t = 0
    for _ in range(max_iters):

        t += 1
        g = smooth_max_grad(h, n, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        mhat = m / (1 - 0.9**t)
        vhat = v / (1 - 0.999**t)
        h = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
    return h

    return best_h, best_c

def lbfgs_polish(h, n, deadline, alpha=50000, mu=50.0):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val(x, n, alpha) + mu * (x.sum() - n/2)**2
    def grads(x):
        g = smooth_max_grad(x, n, alpha)
        g += 2 * mu * (x.sum() - n/2)
        return g
    try:
        res = minimize(obj, h, jac=grads, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c: return h_new, c_new

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for ki in range(2 * n - 1):
        if w[ki] < 1e-13:
            continue
        k = ki - (n - 1)
        wm = w[ki]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * omh[k:]
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * omh[:nv]
            grad[:nv] -= wm * h[kk:]
    return grad

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha

def run_adam(h, n, alpha, lr, max_iters, deadline):

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        g = smooth_max_grad(h, n, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        mhat = m / (1.0 - 0.9**t)
        vhat = v / (1.0 - 0.999**t)
        h = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h = np.clip(h, 0.0, 1.0)

    schedule = [(50, 0.05), (200, 0.03), (500, 0.02),
