import numpy as np
from scipy.optimize import minimize, line_search
from scipy.fft import fft, ifft
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(c, h, n, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_fn(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    val = smooth_max_value(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    g = smooth_max_grad(c, x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_total_time=180):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 50000, 500000]
    mus = [1.0, 10.0, 100.0, 1000.0, 10000.0, 100000.0]
    for i, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 10:
            break
        t_phase = min(time.time() + remaining * 0.35, time.time() + 40)
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': 10000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgradient_polish(h, n, deadline):

    dx = 2.0 / n
    step = 0.005
    for _ in range(50000):
        if time.time() > deadline:
            break
        c = compute_c5_all(h, n)
        cmax = c.max()

        eps = 1e-6
        active = np.where(c >= cmax - eps)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Average subgradient over active lags
        g = np.zeros(n)
        for k in active:

            if shift >= 0:
                nv = n - shift
                g[:nv] += (1.0 - h[shift:])
                g[shift:] += (-h[:nv])
            else:
                s = -shift
                nv = n - s
                g[s:] += (1.0 - h[:nv])
                g[:nv] += (-h[s:])
        g /= len(active)
        g *= dx
        # Take step

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(c, h, n, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        s = idx - (n - 1)
        wm = w[idx] * dx
        if s >= 0:
            nv = n - s
            grad[:nv] += wm * (1.0 - h[s:])
            grad[s:] += wm * (-h[:nv])
        else:
            t = -s
            nv = n - t
            grad[t:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[t:])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    v = smooth_max_val(c, alpha)
    if mu > 0:
        v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def _grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    g = smooth_max_grad(c, x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline):

    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

            if c_new < best_c:

                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgrad_cycle(h, n, deadline):

    dx = 2.0 / n
    step = 0.002
    momentum = np.zeros(n)

        h3, c3 = slsqp_run(best_h, best_n, t3, alpha=500000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try n=750
    if time.time() < deadline - 250:
        n2 = 750
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n2)
        alloc = min(150, (deadline - time.time() - 150) * 0.55)
        if alloc > 40:
            h2, c2 = lbfgs_opt(h_int, n2, time.time() + alloc)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n2
            if time.time() < deadline - 100:
                h2, c2 = subgrad_cycle(h2, n2, min(time.time() + 40, deadline - 100))
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
                    best_n = n2

    # Phase 5: Alternative starts
    if time.time() < deadline - 120:
        starts = [golden_start(best_n), random_start(best_n, rng)]
        h_pert = best_h + rng.randn(best_n) * 0.04
        starts.append(project_feasible(h_pert, best_n))
        for s in starts:
            if time.time() > deadline - 60:
                break
            alloc = min(40, (deadline - time.time() - 30) * 0.5)
            if alloc < 15:
                break
            h_s, c_s = lbfgs_opt(s, best_n, time.time() + alloc)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
                best_n = best_n

    # Phase 6: Final subgradient
    if time.time() < deadline - 20:
        t6 = min(time.time() + 60, deadline - 15)
        best_h, best_c = subgrad_cycle(best_h, best_n, t6)

    # Phase 7: Pair polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 2, radius=25)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
