def smooth_grad(x,n,alpha):
  c=cvec(x,n)
  m=c.max()
  z=alpha*(c-m)
  w=np.exp(z-z.max()); w/=w.sum()
  thresh=1e-13
  idx=np.nonzero(w>thresh)[0]
  dx=2.0/n
  grad=np.zeros(n)
  off=n-1
  for ii in idx:
     w_i=w[ii]*dx
     k=int(ii-off)
     aa=abs(k); nv=n-aa
     if k>=0:
        grad[:nv]+=w_i*(1-x[aa:])
        grad[aa:]-=w_i*x[:nv]
     else:
        kk=-k
        grad[kk:]+=w_i*(1-x[:nv])
        grad[:nv]-=w_i*x[kk:]
  return grad
```

def project_feasible(y,S):
    y=np.asarray(y,dtype=float).copy(); n=y.size
    lo=-2.0; hi=2.0
    for _ in range(70):
        m=(lo+hi)*0.5
        s=np.sum(np.clip(y-m,0.0,1.0))
        if s>S: lo=m
        else: hi=m
    p=np.clip(y-(lo+hi)*0.5,0.0,1.0)
    d=S-p.sum()
    if d>1e-12:
        idx=np.argsort(1.0-p)
        for ii in idx:
            add=min(1.0-p[ii],d)
            if add>0: p[ii]+=add; d-=add
            if d<=1e-14: break
    elif d<-1e-12:
        d=-d; idx=np.argsort(p)
        for ii in idx:
            sub=min(p[ii],d)
            if sub>0: p[ii]-=sub; d-=sub
            if d<=1e-14: break

```python
import time
import numpy as np
from scipy.signal import correlate as spc
from scipy.optimize import minimize, linprog

def cvec(h, n):
    return spc(h, 1.0 - h, mode='full', method='fft') * (2.0 / float(n))

def cval(h, n):
    return float(np.max(cvec(h, n)))

def project_feasible(y, S):
    y = np.asarray(y, dtype=float).copy()
    n = y.size
    lo, hi = -2.0, 2.0
    for _ in range(70):
        m = (lo + hi) * 0.5
        s = np.sum(np.clip(y - m, 0.0, 1.0))
        if s > S:
            lo = m
        else:
            hi = m
    p = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    d = S - p.sum()
    if d > 1e-12:
        for ii in np.argsort(1.0 - p):
            add = min(1.0 - p[ii], d)
            if add > 0.0:
                p[ii] += add
                d -= add
            if d <= 1e-14:
                break
    elif d < -1e-12:
        d = -d
        for ii in np.argsort(p):
            sub = min(p[ii], d)
            if sub > 0.0:
                p[ii] -= sub
                d -= sub
            if d <= 1e-14:
                break
    return p

def grad_single(h, n, k):
    aa = abs(k)
    nv = n - aa
    g = np.zeros(n)
    if k >= 0:
        g[aa:] += 1.0 - h[:nv]
        g[:nv] -= h[aa:]
    else:
        kk = -k
        g[:nv] += 1.0 - h[kk:]
        g[kk:] -= h[:nv]
    return g

def smooth_value(x, n, alpha):
    c = cvec(x, n)
    m = float(np.max(c))
    z = alpha * (c - m)
    z = z - np.max(z)
    return float(m + (np.max(z) + np.log(np.sum(np.exp(z)))) / alpha)

def smooth_jac(x, n, alpha):
    c = cvec(x, n)
    z = alpha * (c - np.max(c))
    z = z - np.max(z)
    w = np.exp(z)
    w /= w.sum()
    idx = np.nonzero(w > 1e-14)[0]
    g = np.zeros(n)
    dx = 2.0 / float(n)
    off = n - 1
    for ii in idx:
        bw = float(w[ii]) * dx
        k = int(ii - off)
        aa = abs(k)
        nv = n - aa
        if k >= 0:
            g[aa:] += bw * (1.0 - x[:nv])
            g[:nv] -= bw * x[aa:]
        else:
            kk = -k
            g[:nv] += bw * (1.0 - x[kk:])
            g[kk:] -= bw * x[:nv]
    return g

def smooth_obj(x, n, alpha):
    return smooth_value(x, n, alpha)

def smooth_jacobj(x, n, alpha):
    return smooth_jac(x, n, alpha)

def eq_fun(x, n):
    return float(np.sum(x) - 0.5 * n)

def eq_jac(x, n):
    return np.ones((1, x.size))

def slsqp_smooth(x, n, alpha, maxit):
    try:
        res = minimize(smooth_obj, x, args=(n, alpha), jac=smooth_jacobj, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': int(maxit), 'ftol': 1e-14})
        return project_feasible(res.x, 0.5 * n)
    except Exception:
        return project_feasible(x, 0.5 * n)

def top_obj(x, n, idx, weights):
    c = cvec(x, n)
    return float(np.dot(weights, c[idx]))

def top_jac(x, n, idx, weights):
    g = np.zeros(n)
    dx = 2.0 / float(n)
    off = n - 1
    for r in range(idx.size):
        bw = float(weights[r]) * dx
        if bw <= 0.0:
            continue
        k = int(idx[r] - off)
        aa = abs(k)
        nv = n - aa
        if k >= 0:
            g[aa:] += bw * (1.0 - x[:nv])
            g[:nv] -= bw * x[aa:]
        else:
            kk = -k
            g[:nv] += bw * (1.0 - x[kk:])
            g[kk:] -= bw * x[:nv]
    return g

def slsqp_top(x, n, idx, weights, maxit):
    try:
        res = minimize(top_obj, x, args=(n, idx, weights), jac=top_jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': int(maxit), 'ftol': 1e-14})
        return project_feasible(res.x, 0.5 * n)
    except Exception:
        return project_feasible(x, 0.5 * n)

def lp_direction(h, n, L, tmax):
    c = cvec(h, n)
    idx = np.argsort(c)[-L:]
    A = np.zeros((len(idx), n + 1))
    dx = 2.0 / float(n)
    off = n - 1
    for r, ii in enumerate(idx):
        A[r, :n] = grad_single(h, n, int(ii - off)) * dx
        A[r, n] = 1.0
    co = np.zeros(n + 1)
    co[-1] = -1.0
    Ae = np.zeros((1, n + 1))
    Ae[0, :n] = 1.0
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(0.0, tmax)]
    try:
        res = linprog(co, A, np.zeros(len(idx)), Ae, np.zeros(1), bounds, method='HiGHS')
    except Exception:
        return None
    if res is None or not res.success or res.x[-1] <= 1e-7:
        return None
    d = res.x[:n]
    lo = np.maximum(-h + 1e-12, -1.0)
    hi = np.minimum(1.0 - h - 1e-12, 1.0)
    d = np.clip(d, lo, hi)
    d -= d.mean()
    if np.linalg.norm(d) < 1e-9:
        return None
    return d

def lp_polish(h, n, deadline):
    h = project_feasible(h, 0.5 * n)
    best = float(cval(h, n))
    Ls = (12, 22)
    tms = (1e-4, 5e-4, 2e-3)
    scales = (1.0, 0.7, 0.4, 0.2, 0.08)
    while time.time() < deadline - 3:
        imp = False
        for L in Ls:
            for tm in tms:
                d = lp_direction(h, n, L, tm)
                if d is None:
                    continue
                for s in scales:
                    h2 = project_feasible(h + s * d, 0.5 * n)
                    v = float(cval(h2, n))
                    if v < best - 1e-14:
                        h = h2
                        best = v
                        imp = True
                        break
                if imp:
                    break
            if imp:
                break
        if not imp:
            break
    return h, best

def add_flip(arr, bits, pos, delta):
    n = bits.size
    off = n - 1
    p = int(pos)
    a = p
    b = n - 1 - p
    if a > 0:
        rev = bits[p - 1:p - 1 - a:-1]
        arr[off + 1:off + 1 + a] += delta * (1.0 - rev)
        arr[off - 1:off - 1 - a:-1] += delta * (-rev)
    if b > 0:
        fwd = bits[p + 1:p + 1 + b]
        arr[off + 1:off + 1 + b] += delta * (-fwd)
        arr[off - 1:off - 1 - b:-1] += delta * (1.0 - fwd)

def binary_hill(bits, n, deadline, rng=None):
    if rng is None:
        rng = np.random.RandomState(999)
    bits = project_feasible(bits, 0.5 * n).astype(float)
    bits[bits >= 0.5] = 1.0
    bits[bits < 0.5] = 0.0
    counts = np.rint(spc(bits, 1.0 - bits, mode='full', method='fft')).astype(float)
    dx = 2.0 / float(n)
    best = float(counts.max()) * dx
    off = n - 1
    while time.time() < deadline:
        ones = np.nonzero(bits > 0.5)[0]
        zeros = np.nonzero(bits < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        g = smooth_jac(bits, n, 5000.0)
        src = ones[np.argsort(-g[ones])[:10]]
        tgt = zeros[np.argsort(g[zeros])[:10]]
        base_mx = float(counts.max())
        best_counts = None
        best_pair = None
        best_mx = base_mx
        for p in src:
            p = int(p)
            for q in tgt:
                q = int(q)
                new = counts.copy()
                add_flip(new, bits, p, -1.0)
                add_flip(new, bits, q, 1.0)
                dist = abs(p - q)
                if dist > 0:
                    new[off + dist] += 1.0
                    new[off - dist] += 1.0
                new[off] = 0.0
                mx = float(new.max())
                if mx < best_mx - 1e-9:
                    best_mx = mx
                    best_counts = new
                    best_pair = (p, q)
        if best_pair is None:
            for _ in range(15):
                p = int(rng.choice(ones))
                q = int(rng.choice(zeros))
                if p == q:
                    continue
                new = counts.copy()
                add_flip(new, bits, p, -1.0)
                add_flip(new, bits, q, 1.0)
                dist = abs(p - q)
                if dist > 0:
                    new[off + dist] += 1.0
                    new[off - dist] += 1.0
                new[off] = 0.0
                mx = float(new.max())
                if mx < best_mx - 1e-9:
                    best_mx = mx
                    best_counts = new
                    best_pair = (p, q)
                    break
        if best_pair is None:
            break
        bits[best_pair[0]] = 0.0
        bits[best_pair[1]] = 1.0
        counts = best_counts
        best = best_mx * dx
        if best <= 0.3808:
            break
    return bits, best

def threshold_bits(h):
    n = len(h)
    S = n // 2
    order = np.argsort(h)
    b = np.zeros(n)
    if S > 0:
        b[order[n - S:]] = 1.0
    return b

def score_to_h(n, score):
    s = score - np.min(score)
    mx = np.max(s)
    if mx > 0.0:
        s = s / mx
    return project_feasible(s, 0.5 * n)

def score_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    f = float(rng.uniform(0.4, 2.2))
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    score = np.sin(np.pi * f * x + ph) + 0.55 * np.cos(2.0 * np.pi * f * x + 2.0 * ph)
    return score_to_h(n, score)

def random_initial(n, rng):
    return score_start(n, rng)

def resample_h(h, n_dst):

    return project_feasible(np.interp(x1, x0, h), 0.5 * n_dst)

            hB, cB = cont_optimize(hB, best_n, endB2)
            if cB < best_c:
                best_h, best_c = hB, cB

    tried = set()
    for nt in (best_n, 300, 400, 600, 750):
        nt = int(nt)
        if nt % 2:
            nt += 1
        if nt < 300 or nt > 900 or nt in tried:
            continue
        tried.add(nt)
        if time.time() > deadline - 130:
            break
        hS = resample_h(best_h, nt)
        endS = min(time.time() + (60 if nt <= 600 else 45), deadline - 90)
        hR, cR = cont_optimize(hS, nt, endS)
        if cR < best_c:
            best_h, best_c, best_n = hR, cR, nt

    if time.time() < deadline - 100 and best_c > 0.3807:
        nS = best_n if 300 <= best_n <= 800 else 600
        for _ in range(2):
            if time.time() > deadline - 70:
                break
            hS = score_start(nS, rng)
            endS = min(time.time() + 45, deadline - 50)
            hR, cR = cont_optimize(hS, nS, endS)
            if cR < best_c:
                best_h, best_c, best_n = hR, cR, nS

    if time.time() < deadline - 3:
        best_h, best_c = lp_polish(best_h, best_n, deadline - 1)

    best_h = project_feasible(best_h, 0.5 * best_n)
    best_c = float(cval(best_h, best_n))
    return best_h, best_c, best_n
