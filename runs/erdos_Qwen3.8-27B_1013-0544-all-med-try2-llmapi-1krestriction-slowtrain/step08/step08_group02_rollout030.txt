    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:

            tot = h[mask].sum()
            if tot < 1e-14:

            h[mask] *= (1.0 + diff / tot)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    wsum = w.sum()
    return float(m + np.log(wsum) / alpha)

def smooth_max_grad(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)

    grad = np.zeros(n)

This can be precomputed! For a given h, I can precompute for each position a the vector Δc_a[m] (the change in c if h[a] goes from 1 to 0). Then for a swap (a,b), the total change is Δc_a + Δc_b (where Δc_b is the change for h[b] going from 0 to 1).

- For each b with h[b]=0: delta_0[b, m] = change in c[m] if h[b] goes 0→1

Then for swap (a,b): Δc[m] = delta_1[a, m] + delta_0[b, m]
And the new max is max_m (c[m] + delta_1[a,m] + delta_0[b,m])

def compute_deltas(h, n):
    """Precompute delta arrays for binary h."""
    c = np.correlate(h, 1 - h, mode='full')
    delta_1 = np.zeros((n, 2*n-1))
    delta_0 = np.zeros((n, 2*n-1))
    
    for a in range(n):
        for m in range(2*n-1):
            idx1 = a + m - (n-1)
            idx2 = a - m + (n-1)
            
            if h[a] == 1:
                # h[a] 1->0
                if 0 <= idx1 < n:
                    delta_1[a, m] = h[idx1] - 1.0
                if 0 <= idx2 < n:
                    delta_1[a, m] += h[idx2]
            else:
                # h[a] 0->1
                if 0 <= idx1 < n:
                    delta_0[a, m] = 1.0 - h[idx1]
                if 0 <= idx2 < n:
                    delta_0[a, m] -= h[idx2]
    
    return c, delta_1, delta_0
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] *= max(0.0, 1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:n])
    return grad

def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, alpha_list=None):
    if alpha_list is None:
        alpha_list = [(1000, 1.0), (10000, 10.0), (100000, 100.0),
                      (1000000, 1000.0), (5000000, 10000.0)]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alpha_list:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        if remaining < 4:
            break
        max_iter = min(300, int(remaining * 15))
        if max_iter < 10:
            break
        try:
            res = minimize(pen_obj, best_h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iter * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

    return best_h, best_c

def slsqp_obj(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def slsqp_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, deadline, alpha=2000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]

        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,

        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

    # Phase 5: SLSQP (50s)
    if time.time() < dl - 25:
        hf, cf = slsqp(bh, bn, min(time.time() + 45, dl - 20), a=5e6, mi=8000)
        if cf < bc:
            bh, bc = hf, cf

    # Phase 6: Final high-alpha L-BFGS-B (30s)
    if time.time() < dl - 5:
        bh, bc = lbfgs(bh, bn, dl - 4, alphas=[(1e7, 1e4), (5e7, 5e4)])

    # Phase 7: Final binary search + polish
    if time.time() < dl - 4:
        hb, cb = bsearch(bh, bn, min(time.time() + 15, dl - 8), rng)
        if cb < bc:
            bh, bc = hb, cb
        if time.time() < dl - 3:
            hr, cr = lbfgs(bh, bn, dl - 2, alphas=[(1e7, 1e4)])
            if cr < bc:
                bh, bc = hr, cr

    bh = project_feasible(bh, bn)
    bc = compute_c5(bh, bn)
    return bh, bc, bn
```
