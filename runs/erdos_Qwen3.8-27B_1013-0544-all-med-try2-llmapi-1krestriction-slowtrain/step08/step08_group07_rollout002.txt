def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = 0.5 * n
    if h.min() >= -1e-13 and h.max() <= 1.0 + 1e-13 and abs(h.sum() - target) < 1e-12:
        np.clip(h, 0.0, 1.0, out=h)
        return h
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        v = h - mid
        np.clip(v, 0.0, 1.0, out=v)
        if v.sum() > target:
            lo = mid
        else:
            hi = mid
    v = h - 0.5 * (lo + hi)
    np.clip(v, 0.0, 1.0, out=v)
    for _ in range(25):
        res = target - v.sum()
        if abs(res) < 1e-14:
            break
        if res > 0:
            room = 1.0 - v
            sr = room.sum()
            if sr < 1e-15:
                break
            v += (res / sr) * room
        else:
            sv = v.sum()
            if sv < 1e-15:
                break
            v += (res / sv) * v
        np.clip(v, 0.0, 1.0, out=v)
    return v


def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    target = 0.5 * n
    res = target - h.sum()
    if abs(res) > 1e-14:
        score = np.minimum(h, 1.0 - h)
        idx = int(np.argmax(score))
        h[idx] = float(np.clip(h[idx] + res, 0.0, 1.0))
        res = target - h.sum()
        if abs(res) > 1e-14:
            idxs = np.where(score > 1e-12)[0]
            if len(idxs):
                h[idxs] += res / float(len(idxs))
                np.clip(h, 0.0, 1.0, out=h)
    return h


def to_binary(h, n):
    v = np.zeros(n)
    k = n // 2
    if k <= 0:
        return v
    v[np.argsort(h)[::-1][:k]] = 1.0
    return v


def make_lag_indices(n):

                    c += B * dy
                    c[diag] = diag_other + best_y * (1.0 - best_y)
        start = (start + 17) % n
        if (sweep + 1) % project_interval == 0 or sweep == max_sweeps - 1:
            hp = project_feasible(h, n)
            h = hp
            c = corr_full(h)
            cv = float(c.max()) * (2.0 / n)
            if cv < best_c:
                best_c = cv
                best_h = h.copy()
    hp = project_feasible(h, n)
    cv = compute_c5(hp, n)
    if cv < best_c:
        return hp, cv
    return best_h, best_c


def binary_sa(h0, n, deadline, rng, firsts, seconds, max_moves=400000):
    dx = 2.0 / n
    m = 2 * n - 1
    diag = n - 1
    h = to_binary(h0, n)
    corr = corr_full(h)
    cur = float(corr.max())
    best = cur
    best_h = h.copy()
    ones = []
    zeros = []
    pos1 = np.full(n, -1, dtype=int)
    pos0 = np.full(n, -1, dtype=int)
    for i, v in enumerate(h):
        if v > 0.5:
            pos1[i] = len(ones)
            ones.append(i)
        else:
            pos0[i] = len(zeros)
            zeros.append(i)
    if not ones or not zeros:
        return best * dx, best_h
    E = 4.0
    Emin = 0.02
    cool = (Emin / E) ** (1.0 / max(1, max_moves))
    noimp = 0
    active = int(np.argmax(corr)) - diag
    delta = np.zeros(m)
    for it in range(max_moves):
        if time.time() > deadline:
            break
        z = -1
        o = -1
        if active != 0 and rng.random() < 0.4:
            l = active
            lo = 0 if l > 0 else -l
            hi = n if l < 0 else n - l
            if hi > lo:
                for _ in range(4):
                    i = int(rng.randint(lo, hi))
                    if h[i] > 0.5 and h[i + l] < 0.5:
                        z = i + l
                        o = i
                        break
        if z < 0:
            z = int(zeros[int(rng.randint(len(zeros)))])
            o = int(ones[int(rng.randint(len(ones)))])
        if z == o:
            continue
        delta.fill(0.0)
        delta[firsts[z]] += 1.0 - h
        delta[seconds[z]] -= h
        delta[firsts[o]] += h - 1.0
        delta[seconds[o]] += h
        delta[diag] = 0.0
        delta[diag + (o - z)] += 1.0
        delta[diag + (z - o)] += 1.0
        new = float(np.max(corr + delta))
        d = new - cur
        acc = False
        if d <= 1e-12:
            acc = True
        elif d < 700.0 * E and rng.random() < math.exp(-d / E):
            acc = True
        if acc:
            h[z] = 1.0
            h[o] = 0.0
            corr += delta
            cur = new
            pz = pos0[z]
            last0 = zeros[-1]
            zeros[pz] = last0
            pos0[last0] = pz
            zeros.pop()
            pos1[z] = len(ones)
            ones.append(z)
            po = pos1[o]
            last1 = ones[-1]
            ones[po] = last1
            pos1[last1] = po
            ones.pop()
            pos0[o] = len(zeros)
            zeros.append(o)
            if cur < best - 1e-12:
                best = cur
                best_h = h.copy()
                noimp = 0
            else:
                noimp += 1
            if noimp > 8000:
                E = min(25.0, E * 1.7)
                noimp = 0
            E = max(Emin, E * cool)
            if it % 10000 == 9999:
                c2 = corr_full(h)
                cm = float(c2.max())
                if abs(cm - cur) > 1e-6:
                    corr = c2
                    cur = cm
                    if cur < best:
                        best = cur
                        best_h = h.copy()
        else:
            E = max(Emin, E * cool)
        if it % 1000 == 999:
            active = int(np.argmax(corr)) - diag
    return best * dx, best_h


def smooth_objective(h, n, alpha):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(c.max())
    s = alpha * (c - cm)
    s = np.maximum(s, -745.0)
    sm = float(s.max())
    e = np.exp(s - sm)
    val = cm + (sm + float(np.log(e.sum()))) / alpha
    return float(val * (2.0 / n))


def smooth_gradient(h, n, alpha):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(c.max())
    s = alpha * (c - cm)
    s = np.maximum(s, -745.0)
    sm = float(s.max())
    e = np.exp(s - sm)
    p = e / e.sum()
    grad = np.zeros(n)
    diag = n - 1
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-16:
            continue
        l = idx - n + 1
        if l == 0:
            grad += w * (1.0 - 2.0 * h)
        elif l > 0:
            grad[:-l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:-l]
        else:
            d = -l
            grad[d:] += w * (1.0 - h[:-d])
            grad[:-d] -= w * h[d:]
    return grad * (2.0 / n)


def smooth_fun(x):
    return smooth_objective(x, _OPT_N, _OPT_ALPHA)


def smooth_jac(x):
    return smooth_gradient(x, _OPT_N, _OPT_ALPHA)


def cons_fun(x):
    return float(x.sum() - 0.5 * _OPT_N)


def cons_jac(x):
    return np.ones_like(x)


def slsqp_smooth(h, n, alpha, maxiter, deadline=None):
    global _OPT_N, _OPT_ALPHA
    if deadline and time.time() > deadline:
        return h
    remain = (deadline - time.time()) if deadline else 60.0
    if remain < 8.0:
        return h
    maxiter = int(min(maxiter, max(10, remain * 1.1)))
    _OPT_N = n
    _OPT_ALPHA = float(alpha)
    x0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(smooth_fun, x0, method='SLSQP', jac=smooth_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if not np.all(np.isfinite(res.x)):
            return h
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s) - 60.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(400, 0.5)

    n = len(h0)
    if n % 2 == 1:
        n = n - 1 if n > 2 else 2
        xo = np.linspace(0.0, 2.0, len(h0), endpoint=False)
        xn = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(xn, xo, h0)
    if n < 2:
        n = 2
        h0 = np.full(2, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    firsts, seconds = make_lag_indices(n)

    d1 = t0 + 0.13 * total
    h1, c1 = coord_descent_exact(best_h, n, 12, d1, firsts, seconds, 4)
    if c1 < best_c:
        best_c, best_h = c1, h1

    d2 = t0 + 0.37 * total
    hb, cb = binary_sa(best_h, n, d2, rng, firsts, seconds, 500000)
    if cb < best_c:
        best_c, best_h = cb, hb

    d3 = t0 + 0.47 * total
    if time.time() < d3 - 10.0:
        hr = rng.rand(n)
        hb2, cb2 = binary_sa(hr, n, d3, rng, firsts, seconds, 250000)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2

    d4 = t0 + 0.60 * total
    h2, c2 = coord_descent_exact(best_h, n, 30, d4, firsts, seconds, 4)
    if c2 < best_c:
        best_c, best_h = c2, h2

    ds = min(t0 + 0.82 * total, deadline - 30.0)
    h = best_h.copy()
    for alpha in [200.0, 5000.0, 50000.0, 500000.0, 5000000.0]:
        if time.time() > ds - 12.0:
            break
        hp = slsqp_smooth(h, n, alpha, 450, ds)
        hp = project_feasible(hp, n)
        cv = compute_c5(hp, n)
        if cv < best_c:
            best_c, best_h = cv, hp
            h = hp
        elif cv <= best_c + 0.02:
            h = hp
        else:
            h = best_h

    d5 = deadline - 8.0
    h3, c3 = coord_descent_exact(best_h, n, 25, d5, firsts, seconds, 3)
    if c3 < best_c:
        best_c, best_h = c3, h3

    best_h = fix_sum(project_feasible(best_h, n), n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
