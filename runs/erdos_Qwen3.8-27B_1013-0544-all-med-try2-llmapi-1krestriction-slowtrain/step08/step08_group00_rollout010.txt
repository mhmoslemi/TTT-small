    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    return float(val)

    s = x.sum() - n / 2.0

def fine_pair_search(h, n, deadline, max_step=0.001, min_step=1e-7):
    """Very fine pair search with decreasing step sizes."""
    best_c = compute_c5(h, n)
    # Geometric sequence of step sizes
    log_max = np.log(max_step)
    log_min = np.log(min_step)
    n_steps = 12
    steps = np.exp(np.linspace(log_max, log_min, n_steps))
    
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    # Move from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Move from i to j
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[j] -= d
                            h[i] += d
    return h, best_c

def active_set_polish(h, n, deadline):
    """Polish by focusing on the active (max) correlation."""
    best_c = compute_c5(h, n)
    lr = 0.001
    patience = 0
    
    while time.time() < deadline and patience < 200:
        c = compute_c5_vec(h, n)
        k_star = np.argmax(c)
        c_max = c[k_star]
        
        if c_max < best_c - 1e-15:
            best_c = c_max
            patience = 0
        else:
            patience += 1
        
        k = k_star - (n - 1)
        dx = 2.0 / n
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[:nv] = dx * (1.0 - h[k:])
            grad[k:] = dx * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] = dx * (1.0 - h[:nv])
            grad[:nv] = dx * (-h[kk:])
        
        # Negative gradient step with projection
        h_try = h - lr * grad
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        
        if c_try < best_c:
            h = h_try
            best_c = c_try
            patience = 0
            lr = min(lr * 1.1, 0.01)
        else:
            lr *= 0.8
            if lr < 1e-12:
                break
    
    return h, best_c

For very small steps (d < 1e-5), the change in C5 is approximately linear in d. So we can use a first-order approximation to determine which direction to move, and then just verify with the exact computation.

The first-order change in C5 when we move d from j to i is:
ΔC5 ≈ d * (∂C5/∂h[i] - ∂C5/∂h[j])

If this is negative, the move improves C5. We don't need to try all pairs - we just need to find the best pair (i*, j*) that minimizes this, and make that move.

This is essentially the steepest descent direction for the pair search. We can compute this in O(n) time (after computing the gradient) instead of O(n^2).

def steepest_pair_descent(h, n, deadline):
    """At each step, find the best (i, j) pair to move and do it."""
    best_c = compute_c5(h, n)
    alpha = 10000000  # Very high for near-exact max
    
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        c_max = c.max()
        
        # Compute smooth max gradient
        c_shift = alpha * (c - c_max)
        mx = c_shift.max()
        w = np.exp(c_shift - mx)
        w /= w.sum()
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-12:
                continue
            k = m - (n - 1)
            wm = w[m] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] += wm * (-h[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
        
        # Best pair: minimize grad[i] - grad[j] subject to h[i] + d <= 1, h[j] - d >= 0
        # The objective change is d * (grad[i] - grad[j]) (negative = improvement)
        # So we want the most negative (grad[i] - grad[j])
        # Which is: minimize grad[i] and maximize grad[j]
        # But we also need h[i] < 1 and h[j] > 0
        
        # Find best i (minimize grad[i] where h[i] < 1)
        # Find best j (maximize grad[j] where h[j] > 0)
        
        valid_i = np.where(h < 1.0 - 1e-10)[0]
        valid_j = np.where(h > 1e-10)[0]
        
        if len(valid_i) == 0 or len(valid_j) == 0:
            break
        
        best_i = valid_i[np.argmin(grad[valid_i])]
        best_j = valid_j[np.argmax(grad[valid_j])]
        
        if best_i == best_j:
            # Try second best
            sorted_i = valid_i[np.argsort(grad[valid_i])]
            sorted_j = valid_j[np.argsort(-grad[valid_j])]
            found = False
            for i_cand in sorted_i[:5]:
                for j_cand in sorted_j[:5]:
                    if i_cand != j_cand:
                        best_i, best_j = i_cand, j_cand
                        found = True
                        break
                if found:
                    break
            if not found:
                break
        
        # Determine step size (limited by bounds)
        max_step_i = 1.0 - h[best_i]
        max_step_j = h[best_j]
        d = min(0.01, max_step_i, max_step_j)
        
        # Try a few step sizes
        found = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            d_try = d * frac
            h[best_i] += d_try
            h[best_j] -= d_try
            c_try = compute_c5(h, n)
            if c_try < best_c - 1e-16:
                best_c = c_try
                found = True
                break
            else:
                h[best_i] -= d_try
                h[best_j] += d_try
        
        if not found:
            break
    
    return h, best_c

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def steepest_pair_descent(h, n, deadline, max_step=0.01):
    """Use gradient to find the best pair move at each iteration."""
    best_c = compute_c5(h, n)
    alpha = 100000000
    step_decay = 0.9
    d = max_step
    stall = 0
    
    while time.time() < deadline and stall < 500:
        if d < 1e-8:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        c_shift = alpha * (c - c_max)
        mx = c_shift.max()
        w = np.exp(c_shift - mx)
        w /= w.sum()
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-12:
                continue
            k = m - (n - 1)
            wm = w[m] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] += wm * (-h[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
        
        # Find best (i, j) pair: minimize grad[i] - grad[j]
        # i should have low grad (adding mass helps)
        # j should have high grad (removing mass helps)
        valid_i = np.where(h < 1.0 - 1e-12)[0]
        valid_j = np.where(h > 1e-12)[0]
        
        if len(valid_i) == 0 or len(valid_j) == 0:
            break
        
        # Sort by gradient
        i_order = valid_i[np.argsort(grad[valid_i])]
        j_order = valid_j[np.argsort(-grad[valid_j])]
        
        found = False
        # Try top candidates
        for i_cand in i_order[:20]:
            if found:
                break
            for j_cand in j_order[:20]:
                if i_cand == j_cand:
                    continue
                d_try = min(d, 1.0 - h[i_cand], h[j_cand])
                if d_try < 1e-10:
                    continue
                h[i_cand] += d_try
                h[j_cand] -= d_try
                c_try = compute_c5(h, n)
                if c_try < best_c - 1e-16:
                    best_c = c_try
                    stall = 0
                    found = True
                    break
                else:
                    h[i_cand] -= d_try
                    h[j_cand] += d_try
        
        if not found:
            d *= step_decay
            stall += 1
    
    return h, best_c
