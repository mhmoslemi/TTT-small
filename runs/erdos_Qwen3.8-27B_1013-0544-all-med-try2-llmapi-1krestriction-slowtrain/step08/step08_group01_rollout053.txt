    m1 = (a >= 0) & (a < n)
    v[m1] = 1.0 - h[a[m1]]
    m2 = (b >= 0) & (b < n)
    v[m2] -= h[b[m2]]
    return v

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        wm = w[mi] * dx
        k = mi - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj_lbfgs(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_lbfgs(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _obj_slsqp(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n))

def _eq_c(x, n):
    return float(x.sum() - n / 2.0)

def _eq_j(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0, 300), (1000, 5.0, 200), (5000, 20.0, 200),
               (20000, 100.0, 200), (100000, 500.0, 150),
               (500000, 2000.0, 100), (2000000, 5000.0, 80)]
    for alpha, mu, mx in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_obj_lbfgs, h, jac=_grad_lbfgs, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': mx, 'ftol': 1e-16, 'gtol': 1e-13})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_c, 'jac': _eq_j, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fast_local_search(h, n, deadline, radius=40):
    dx = 2.0 / n
    corr = compute_corr(h, n)
    best_c = float(np.max(corr)) * dx
    p = np.arange(2 * n - 1) - (n - 1)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]

    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                vi = infl_vec(h, i, p, n)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        vj = infl_vec(h, j, p, n)
                        delta = d * (vi - vj)
                        new_max = float(np.max(corr + delta)) * dx
                        if new_max < best_c - 1e-14:
                            corr += delta
                            h[i] += d
                            h[j] -= d
                            best_c = new_max
                            improved = True
                            vi = vj + 2.0 * d * (infl_vec(h, i, p, n) - vi)
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        vj = infl_vec(h, j, p, n)
                        delta = -d * (vi - vj)
                        new_max = float(np.max(corr + delta)) * dx
                        if new_max < best_c - 1e-14:
                            corr += delta
                            h[i] -= d
                            h[j] += d
                            best_c = new_max
                            improved = True
    return h, best_c

def binary_sa(h, n, deadline, n_iter=50000):
    dx = 2.0 / n
    b = binary_project(h, n)
    corr = compute_corr(b, n)
    best_c = float(np.max(corr)) * dx
    best_b = b.copy()
    cur_c = best_c
    p = np.arange(2 * n - 1) - (n - 1)
    ones_list = np.where(b == 1)[0]
    zeros_list = np.where(b == 0)[0]
    rng = np.random.RandomState(123)
    T0 = 0.05
    T_end = 0.001

    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max(n_iter, 1))
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        vi = infl_vec(b, i, p, n)
        vj = infl_vec(b, j, p, n)
        delta = -vi + vj
        new_c = float(np.max(corr + delta)) * dx
        dc = new_c - cur_c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-10)):
            corr += delta
            b[i], b[j] = 0.0, 1.0
            cur_c = new_c
            np.delete(ones_list, np.where(ones_list == i))
            ones_list = np.append(ones_list, j)
            np.delete(zeros_list, np.where(zeros_list == j))
            zeros_list = np.append(zeros_list, i)
            if cur_c < best_c:
                best_c = cur_c
                best_b = b.copy()
        else:
            pass
    return best_b, best_c

def golden_start(n, rng=None):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = golden_start(500, rng)
        n_init = len(h_init)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Gradient optimization
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, best_n, t1)

    # Phase 2: SLSQP
    if time.time() < deadline - 400:
        t2 = min(time.time() + 50, deadline - 350)
        h_s, c_s = slsqp_phase(best_h, best_n, t2, alpha=1000000, maxiter=3000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 3: Binary SA
    if time.time() < deadline - 300:
        t3 = min(time.time() + 120, deadline - 250)
        b_sa, c_sa = binary_sa(best_h, best_n, t3, n_iter=80000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = b_sa
            best_n = best_n

    # Phase 4: Continuous local search from best
    if time.time() < deadline - 200:
        t4 = min(time.time() + 180, deadline - 150)
        best_h, best_c = fast_local_search(best_h, best_n, t4, radius=50)

    # Phase 5: Try different n values
    for n_try in [300, 400, 600]:
        if time.time() > deadline - 120:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time() - 60) * 0.5)
        if alloc < 30:
            continue
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6)
        if c_t < best_c:
            h_t, c_t = fast_local_search(h_t, n_try, min(time.time() + 60, deadline - 40), radius=40)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 6: More local search
    if time.time() < deadline - 30:
        t6 = min(time.time() + 120, deadline - 10)
        best_h, best_c = fast_local_search(best_h, best_n, t6, radius=30)

    # Phase 7: Final SLSQP
    if time.time() < deadline - 10:
        t7 = min(time.time() + 40, deadline - 5)
        h_f, c_f = slsqp_phase(best_h, best_n, t7, alpha=5000000, maxiter=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 8: Final micro-search
    if time.time() < deadline - 2:
        best_h, best_c = fast_local_search(best_h, best_n, deadline - 1, radius=20)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
