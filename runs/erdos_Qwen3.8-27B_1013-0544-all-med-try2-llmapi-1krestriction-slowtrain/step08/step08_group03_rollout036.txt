def compute_c5_vec(h, n):

def compute_c5(h, n):

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(beta, alpha):
    """Minimize max_m (beta[m] + alpha[m]*x) over x in [0,1]."""
    f0 = beta.max()
    f1 = (beta + alpha).max()
    if f0 <= f1:
        # Check if derivative at 0 is >= 0
        active0 = beta >= f0 - 1e-14
        if alpha[active0].min() >= 0:
            return 0.0
    if f1 <= f0:
        active1 = (beta + alpha) >= f1 - 1e-14
        if alpha[active1].max() <= 0:
            return 1.0
    # Binary search
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = beta + alpha * mid
        f_mid = vals.max()
        active = vals >= f_mid - 1e-14
        if not active.any():
            break
        sg = alpha[active].mean()
        if sg > 1e-14:
            hi = mid
        elif sg < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, max_passes=200, deadline=None):
    """Coordinate descent for the min-max problems."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    s_arr = np.arange(-(n-1), n)  # length 2n-1
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    improves_streak = 0
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        pass_improvement = 0.0
        for i in range(n):
            old_val = h[i]
            # Compute alpha (linear coefficient of h[i] in each c_k)
            idx1 = i + s_arr
            valid1 = (idx1 >= 0) & (idx1 < n)
            term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
            idx2 = i - s_arr
            valid2 = (idx2 >= 0) & (idx2 < n) & (s_arr != 0)
            term2 = np.where(valid2, -h[np.clip(idx2, 0, n-1)], 0.0)
            # Quadratic correction for s=0
            term_sq = np.where(s_arr == 0, -h[i], 0.0)
            alpha = dx * (term1 + term2 + term_sq)
            
            beta = c_vec - alpha * old_val
            
            new_val = solve_1d_minimax(beta, alpha)
            new_val = max(0.0, min(1.0, new_val))
            
            if abs(new_val - old_val) < 1e-12:
                continue
            
            # Update c_vec incrementally
            c_vec_trial = c_vec + alpha * (new_val - old_val)
            # Correct the quadratic term for lag 0
            c_vec_trial[n-1] += dx * ((new_val - new_val**2) - (old_val - old_val**2)) - dx * (1 - 2*old_val) * (new_val - old_val)
            
            old_max = c_vec.max()
            new_max = c_vec_trial.max()
            
            if new_max <= old_max + 1e-13:
                h[i] = new_val
                c_vec = c_vec_trial
                pass_improvement += max(0, old_max - new_max)
        
        # Project to maintain sum constraint
        h = project_feasible(h, n)
        # Recompute c_vec to avoid accumulated error
        c_vec = compute_c5_vec(h, n)
        cur_max = c_vec.max()
        
        if cur_max < best_c - 1e-15:
            best_c = cur_max
            improves_streak = 0
        else:
            improves_streak += 1
        
        if improves_streak > 20:
            break
    
    return h, c_vec.max()

def pgd(h0, n, deadline, alpha=300.0, top_k=40):
    """Projected gradient descent with smooth max."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    stall = 0
    cur_alpha = alpha
    dx = 2.0 / n
    
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        m = c.max()
        top_idx = np.argsort(c)[-top_k:]
        vals = c[top_idx]
        sh = cur_alpha * (vals - m)
        mx = sh.max()
        exps = np.exp(sh - mx)
        w = exps / exps.sum()
        
        grad = np.zeros(n)
        for ii in range(top_k):
            k = top_idx[ii] - (n - 1)
            nv = n - abs(k)
            g = np.zeros(n)
            if k >= 0:
                g[:nv] = 1.0 - h[k:]
                g[k:] -= h[:nv]
            else:
                kk = -k
                g[kk:] = 1.0 - h[:nv]
                g[:nv] -= h[kk:]
            grad += w[ii] * dx * g
        
        accepted = False
        for _ in range(15):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 1e-13:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.45
        if not accepted:
            step = max(step * 0.3, 1e-7)
            if step < 1e-7:
                break
        if stall > 30:
            cur_alpha = min(cur_alpha * 1.5, 5000.0)
            step = max(0.01, step)
            stall = 0
        if stall > 150:
            break
    return best_h, best_c

def pair_polish(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(77)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_rounds):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            # Try random pairs
            for _ in range(200):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:

                if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:

                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
            if not improved:
                break
    return best_h, best_c
