def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j: continue

def corr_partial_update(h, cor, pos, delta):
    """Update cor when h[pos] changes by delta. Must be called BEFORE modifying h."""
    n = len(h)
    # First factor: m where pos is valid as x
    # k = m - (n-1), need max(0,-k) <= pos < min(n, n-k)
    # -k <= pos → k >= -pos → m >= n-1-pos
    # pos < n-k → k < n-pos → m < n-1+n-pos = 2n-1-pos
    # Also need pos >= 0 and pos+k >= 0 (i.e., k >= -pos) and pos+k < n (i.e., k < n-pos)
    # So m ranges from n-1-pos to 2n-2-pos (inclusive), clamped to [0, 2n-2]
    m1_lo = max(0, n-1-pos)
    m1_hi = min(2*n-2, 2*n-2-pos)
    if m1_lo <= m1_hi:
        ks = np.arange(m1_lo, m1_hi+1) - (n-1)
        idx = pos + ks  # indices into h for the second factor
        cor[m1_lo:m1_hi+1] += delta * (1 - h[idx])
    
    # Second factor: m where pos = x+k, i.e., x = pos-k
    # Need max(0,-k) <= pos-k < min(n, n-k)
    # pos-k >= 0 → k <= pos → m <= n-1+pos
    # pos-k >= -k → pos >= 0 (always)
    # pos-k < n → k > pos-n → m > n-1+pos-n = pos-1, so m >= pos
    # pos-k < n-k → pos < n (always)
    # So m ranges from pos to n-1+pos (inclusive), clamped to [0, 2n-2]
    m2_lo = max(0, pos)
    m2_hi = min(2*n-2, n-1+pos)
    if m2_lo <= m2_hi:
        ks = np.arange(m2_lo, m2_hi+1) - (n-1)
        idx = pos - ks  # indices into h for the first factor
        cor[m2_lo:m2_hi+1] += (-delta) * h[idx]

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h

    cor = compute_corr(h, n)

    c = cor * dx
    m = c.max()

    w /= w.sum()

            break
        try:
            res = minimize(_sm_obj, h, jac=_sm_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': maxiter * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new; h = h_new
        except Exception:
            pass
    return h, best_c

def gradient_pair_search(h, n, deadline, step=0.003):
    dx = 2.0 / n
    cor = compute_corr(h, n).copy()
    best_c = cor.max() * dx
    for alpha in [50, 200, 1000, 5000, 20000]:
        while time.time() < deadline:
            improved = False
            g = _sm_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = step * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    di = move; dj = -move
                    if h[i] + di >= 0.0 and h[i] + di <= 1.0 and h[j] + dj >= 0.0 and h[j] + dj <= 1.0:
                        _corr_apply(h, cor, i, j, di, dj, n)
                        h[i] += di; h[j] += dj
                        c = cor.max() * dx
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= di; h[j] -= dj
                            _corr_apply(h, cor, i, j, -di, -dj, n)
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_use = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5); n_use = 300
    best_h = project_feasible(h_init, n_use)
    best_c = compute_c5(best_h, n_use)
    # Phase 1: L-BFGS
    t1 = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, n_use, t1, maxiter=300)
    # Phase 2: Gradient-guided pair search
    if time.time() < deadline - 200:
        t2 = min(time.time() + 250, deadline - 200)
        best_h, best_c = gradient_pair_search(best_h, n_use, t2, step=0.002)
    # Phase 3: Fast pair search
    if time.time() < deadline - 150:
        t3 = min(time.time() + 350, deadline - 150)
        best_h, best_c = fast_pair_search(best_h, n_use, t3,
            deltas=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            radius=80)
    # Phase 4: Multi-resolution
    if time.time() < deadline - 60:
        for n_try in [150, 200, 400, 500]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(80, deadline - time.time() - 30)
            if alloc < 30:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5, maxiter=200)
            if time.time() < deadline - 20:
                h_t, c_t = fast_pair_search(h_t, n_try, time.time() + min(40, deadline - time.time() - 10),
                    deltas=[0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)
            if c_t < best_c:
                best_c = c_t; best_h = h_t
    # Phase 5: Final refinement
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = lbfgs_optimize(best_h, nf, deadline - 3, maxiter=100)
        if time.time() < deadline - 1:
            best_h, best_c = fast_pair_search(best_h, nf, deadline - 0.5,
                deltas=[0.001, 0.0005, 0.0002, 0.0001], radius=40)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
