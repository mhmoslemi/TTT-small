    # Build h from Fourier modes
    x = np.arange(n)
    h = np.zeros(n)
    h += 0.5  # DC component
    for m in range(1, d+1):
        h += (a[2*(m-1)] + 1j*a[2*(m-1)+1]) * np.exp(2j*np.pi*m*x/n)
    h = np.real(h)

    smax = sh.max()

= [h_new[i] * (1 - h_new[i+k])] - [h_old[i] * (1 - h_old[i+k])] + [h_new[j] * (1 - h_new[j+k])] - [h_old[j] * (1 - h_old[j+k])]

For most k, h_new[i+k] = h_old[i+k] and h_new[j+k] = h_old[j+k], so:
delta[k] = -(1-h[k+i]) + (1-h[k+j])  (when i+k≠j and j+k≠i)

When i+k = j (i.e., k = j-i):
delta[k] = [0*(1-h_new[j])] - [1*(1-h_old[j])] + [h_new[j]*(1-h_new[j+k])] - [h_old[j]*(1-h_old[j+k])]
= 0 - 1 + [1*(1-h_new[j+k])] - [0*(1-h_old[j+k])]
= -1 + (1 - h_new[j+k])

But j+k = j+(j-i) = 2j-i. If 2j-i ≠ i (i.e., j ≠ i, which is true), then h_new[2j-i] = h_old[2j-i].
So delta[j-i] = -1 + (1 - h[2j-i]) = -h[2j-i]

When j+k = i (i.e., k = i-j):
delta[k] = [h_new[i]*(1-h_new[i+k])] - [1*(1-h_old[i+k])] + [1*(1-h_new[i])] - [0*(1-h_old[i])]
= [0*(1-h_new[i+k])] - [1*(1-h_old[i+k])] + (1-0) - 0
= -(1-h_old[i+k]) + 1

i+k = i+(i-j) = 2i-j. If 2i-j ≠ j (i.e., i≠j, true), then h_old[2i-j] = h_new[2i-j].
So delta[i-j] = -(1-h[2i-j]) + 1 = h[2i-j]

OK so the special cases are:
- At k = j-i: delta = -h[2j-i] (if 0 ≤ 2j-i < n)
- At k = i-j: delta = h[2i-j] (if 0 ≤ 2i-j < n)

And for all other valid k:
- From source i: delta -= (1-h[i+k])
- From source j: delta += (1-h[j+k])

This is implementable! The total work per swap evaluation is O(n) (to compute the max after updating ~2n entries).

But actually, I realize that for the "best improvement" approach, I need to evaluate ALL 62,500 swaps and find the best one. Each evaluation is O(n) = O(500). So total per pass is 62,500 * 500 = 31,250,000 operations. At ~1 ns per operation in Python (with numpy vectorization, maybe faster), this is about 30 milliseconds if I can vectorize, or about 30 seconds if I do it in a Python loop.

The key is to vectorize! I can precompute the "base" change for each possible (i, j) pair and then just find the one that minimizes the resulting max.

Actually, let me think of a simpler vectorized approach:

For a given h, the correlation corr[k] = sum_i h[i]*(1-h[i+k]).

If I flip h[i] from 1 to 0, the change in corr is:
delta_i[k] = -(1-h[i+k]) for all valid k

If I then flip h[j] from 0 to 1, the change is:
delta_j[k] = +(1-h_new[j+k]) for all valid k
where h_new differs from h only at position i (which is now 0).

So for j+k ≠ i: delta_j[k] = (1-h[j+k])
For j+k = i (k = i-j): delta_j[i-j] = (1-0) = 1

Total change: delta[k] = delta_i[k] + delta_j[k]

For the "best improvement" search, I need:
min_{i,j} max_k (corr[k] + delta_{ij}[k])

This is a min-max problem. It's not trivially vectorizable because of the max over k.

However, I can approximate: the max will likely be at the same k as the current max (or one of the top few). So I can just check the top T lags (say T=10) and find the swap that minimizes the max over those T lags. This is a good heuristic that's much faster.

Let me implement this:
1. Compute corr = np.correlate(h, 1-h, mode='full')
2. Find top T lags by value
3. For each (i, j) swap, compute the new values at those T lags and find the min-max
4. Apply the best swap

For step 3, I can vectorize over (i, j) pairs:
- For a fixed set of T lags, the change at each lag k for swap (i,j) is:
  delta[k, i, j] = -(1-h[i+k]) + (1-h[j+k])  (with special cases)
  
This is a 3D array of size T × n × n which might be too large. For T=10, n=500: 10*500*500 = 2,500,000 floats = 20MB. That's borderline.

Alternative: for each lag k in the top T, compute the matrix M_k[i,j] = -(1-h[i+k]) + (1-h[j+k]) (with special cases). Then for each (i,j), the new value at lag k is corr[k] + M_k[i,j]. The max over k is what we want to minimize.

I can compute this as: for each (i,j), compute max_k (corr[k] + M_k[i,j]). This is a T × n × n problem.

Actually, let me simplify further. Instead of the full min-max, let me just use a **greedy approach**:
1. Find the current max lag k*
2. Find the swap (i,j) that maximally reduces corr[k*]
3. Apply it
4. Repeat

This is "hill climbing on the max" and is much simpler. It's essentially the same as what the smooth max L-BFGS-B does, but in the binary space.

For step 2: I want to maximize delta[k*] (make it as negative as possible).
delta[k*] for swap (i,j) = -(1-h[i+k*]) + (1-h[j+k*]) (ignoring special cases for now)

To minimize delta[k*]:
- Choose i such that (1-h[i+k*]) is maximized, i.e., h[i+k*] = 0
- Choose j such that (1-h[j+k*]) is minimized, i.e., h[j+k*] = 1

So: pick i from {ones} where h[i+k*] = 0, and j from {zeros} where h[j+k*] = 1.

This gives a very fast greedy! In O(n) time per step.

But this only optimizes one lag. Other lags might increase. So I should check the full max after each swap.

Let me combine:
1. Greedily find a good swap that reduces the current max
2. Verify by computing the full correlation
3. If the overall max decreased, accept; else, reject
4. Repeat

This is a "first improvement" approach that's very fast.

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        nv = n - abs(k)
        if k >= 0:
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            grad[-k:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[-k:])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def softmax_overlap(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    am = a.max()
    return float(m + (am + np.log(np.sum(np.exp(a - am)))) / alpha)

def softmax_overlap_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    am = a.max()
    w = np.exp(a - am)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        ni = n - abs(k)
        if k >= 0:
            grad[:ni] += wi * (1.0 - x[k:])
            grad[k:] += wi * (-x[:ni])
        else:
            grad[-k:] += wi * (1.0 - x[:ni])
            grad[:ni] += wi * (-x[-k:])
    return grad

def obj_fn(x, n, alpha, mu):
    return softmax_overlap(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def grad_fn(x, n, alpha, mu):
    g = softmax_overlap_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def optimize_stage(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_alpha_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stages = [(10, 5.0, 200), (50, 20.0, 200), (200, 100.0, 300),
              (1000, 500.0, 300), (5000, 2000.0, 400),
              (20000, 10000.0, 500), (100000, 50000.0, 600)]
    for alpha, mu, iters in stages:
        if time.time() > deadline - 4:
            break
        h = optimize_stage(best_h, n, alpha, mu, iters, deadline)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h
    return best_h, best_c

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

def binary_smart_search(h, n, deadline, rng):
    h_bin = to_binary(h, n, rng)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    stall = 0
    while time.time() < deadline - 3:

        c_cur = best_c
        improved_this_round = False

            c_val = corr.max()
            k_star = int(np.argmax(corr)) - (n - 1)
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            # Targeted: i where h[i+k_star]==0, j where h[j+k_star]==1
            if k_star >= 0:
                iv = ones[ones + k_star < n]
                ic = iv[h_cur[iv + k_star] == 0]
                jv = zeros[zeros + k_star < n]
                jc = jv[h_cur[jv + k_star] == 1]
            else:
                iv = ones[ones + k_star >= 0]
                ic = iv[h_cur[iv + k_star] == 0]
                jv = zeros[zeros + k_star >= 0]
                jc = jv[h_cur[jv + k_star] == 1]
            if len(ic) > 0 and len(jc) > 0:
                i = ic[rng.randint(len(ic))]
                j = jc[rng.randint(len(jc))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            h_cur[i] = 0
            h_cur[j] = 1
            c_new = compute_c5(h_cur, n)
            if c_new < c_val - 1e-15:
                c_cur = c_new
                improved_this_round = True
            else:
                h_cur[i] = 1
                h_cur[j] = 0
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                stall = 0
        # Perturbation
        n_pert = rng.randint(2, 8 + stall * 2)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        for _ in range(min(n_pert, n // 4)):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            best_h[i] = 0
            best_h[j] = 1
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
    return best_h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 12):
        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)
