```python
def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_smooth(h, n, alpha, maxiter=1000, ftol=1e-14):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        # Numerically stable log-sum-exp
        if shifted.max() > 700:
            return m
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - target

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

        return h
```

∂corr[k]/∂h[j] = 1[0≤j≤n-1-d] - h[j+d]*1[0≤j≤n-1-d] - h[j-d]*1[d≤j≤n-1]

For d < 0 (let L = -d):

Wait, for d < 0: j+d = j-L (since d=-L). And j-d = j+L.

Let me redo for d < 0, L = -d:
- 1[j ∈ S_d] = 1[L ≤ j ≤ n-1]
- h[j+d] = h[j-L], valid when 0 ≤ j-L < n, i.e., L ≤ j < n+L (but j < n, so L ≤ j ≤ n-1)
  Combined: h[j-L] * 1[L ≤ j ≤ n-1]
- h[j-d] = h[j+L], with j-d ∈ S_d meaning L ≤ j+L ≤ n-1, i.e., 0 ≤ j ≤ n-1-L
  So: h[j+L] * 1[0 ≤ j ≤ n-1-L]

Therefore for d < 0:
∂corr[k]/∂h[j] = 1[L≤j≤n-1] - h[j-L]*1[L≤j≤n-1] - h[j+L]*1[0≤j≤n-1-L]

OK so in both cases, the gradient has a simple structure. Let me code this up:

```python
def compute_grad_at_k(h, n, k):
    """Compute gradient of corr[k] w.r.t. h (before dx scaling)."""
    d = k - (n - 1)  # actual lag
    grad = np.zeros(n)
    if d >= 0:
        grad[:n-d] = 1.0 - h[d:n]  # 1[0<=j<=n-1-d] - h[j+d] = 1 - h[j+d] for j in [0, n-1-d]
        if d > 0:
            grad[d:] -= h[:n-d]  # -h[j-d] for j in [d, n-1]
    else:
        L = -d
        grad[L:] = 1.0 - h[:n-L]  # 1[L<=j<=n-1] - h[j-L] for j in [L, n-1]
        grad[:n-L] -= h[L:]  # -h[j+L] for j in [0, n-1-L]
    return grad
```

Let me verify: for d=0:
grad[:n] = 1 - h[0:n] = 1 - h
That's correct: corr at center = sum h[i](1-h[i]) = 0 for binary, and the gradient is 1-2h[i].

For d=1:
grad[:n-1] = 1 - h[1:n]  (for j=0 to n-2)
grad[1:] -= h[:n-1]  (for j=1 to n-1)
So grad[0] = 1-h[1]
grad[j] = 1-h[j+1] - h[j-1] for j=1 to n-2
grad[n-1] = -h[n-2]

∂/∂h[1] = (1-h[2]) - h[0] ✓
∂/∂h[n-2] = (1-h[n-1]) - h[n-3] ✓
∂/∂h[n-1] = -h[n-2] ✓

So SLSQP with numerical Hessian is not viable for n=400 with many iterations.

**Alternative**: Use L-BFGS-B which uses a limited-memory BFGS approximation of the Hessian. It only needs O(n) storage and O(n) per iteration (for the Hessian-vector product). But it doesn't handle equality constraints.

**Solution for L-BFGS-B**: Use a penalty for the equality constraint.

f_penalized(h) = smooth_max(h) + mu * (sum(h) - n/2)^2

As mu → ∞, this approaches the constrained problem. Start with small mu and increase.

- Each gradient eval: O(n^2) for correlation + O(n) for penalty grad
- Per L-BFGS-B iteration: O(n) line search evaluations × O(n^2) = O(n^3)
- But L-BFGS-B is usually very efficient, maybe 20-50 line search evals per iteration
- So per iteration: ~50 * O(n^2) = O(8M) for n=400
- With 500 iterations: 4B operations ≈ 40 seconds

That's much better than SLSQP!

```python
def lbbf_optimize(h, n, alpha, mu, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        base = m + logsum / alpha
        pen = mu * (x.sum() - target) ** 2
        return base + pen
    
    def grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()

        return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_grad_smooth(h, n, alpha):
    """Compute gradient of the smooth max objective."""
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # All weights underflow, just use the argmax
        k_star = np.argmax(g)
        d = k_star - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            grad[:n-d] = 1.0 - h[d:n]
            if d > 0:
                grad[d:] -= h[:n-d]
        else:
            L = -d
            grad[L:] = 1.0 - h[:n-L]
            grad[:n-L] -= h[L:]
        return grad * dx
    
    w /= w_sum
    
    # Compute weighted sum of gradients efficiently
    total_grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-8:
            continue
        d = k - (n - 1)
        wg = w[k]
        if d >= 0:
            total_grad[:n-d] += wg * (1.0 - h[d:n])
            if d > 0:
                total_grad[d:] -= wg * h[:n-d]
        else:
            L = -d
            total_grad[L:] += wg * (1.0 - h[:n-L])
            total_grad[:n-L] -= wg * h[L:]
    
    return total_grad * dx

def lbbf_optimize(h, n, alpha, mu, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        base = m + logsum / alpha
        pen = mu * (x.sum() - target) ** 2
        return base + pen
    
    def grad(x):
        base_grad = compute_grad_smooth(x, n, alpha)
        pen_grad = 2.0 * mu * (x.sum() - target)
        return base_grad + pen_grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_optimize(h, n, alpha, maxiter=100):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    
    def grad(x):
        return compute_grad_smooth(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_init, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_list = list(np.where(h == 0)[0])
    
    rng = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try random moves
        for _ in range(200):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == 0 and h[j] == 1:
                h[i], h[j] = 1.0, 0.0
                nc = np.max(np.correlate(h, 1-h, mode='full')) * dx
                if nc >= cur_c + 1e-15:
                    h[i], h[j] = h[j], h[i]  # revert
                else:
                    corr = np.correlate(h, 1-h, mode='full').astype(float)
                    cur_c = nc
                    ones_set.add(i)
                    ones_set.discard(j)
                    if i in zeros_list:
                        zeros_list.remove(i)
                    zeros_list.append(j)
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    try:

    except:

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    p1 = t0 + 0.4 * budget_s
    h = best_h.copy()

        if time.time() > p1 - 30:

        for mu in [10, 100, 1000]:
            if time.time() > p1 - 20:

            h = lbbf_optimize(h, n, alpha, mu, maxiter=300)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    
    # Phase 2: SLSQP with analytical gradient (slower but handles constraint exactly)
    p2 = t0 + 0.65 * budget_s
    for alpha in [5000, 50000, 500000]:
        if time.time() > p2 - 20:
            break
        h = slsqp_optimize(best_h, n, alpha, maxiter=100)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 3: Binary local search

    hb, cb = binary_local_search(best_h, n, p3)

    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
