    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] *= max(0.0, 1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def compute_a(h, idx, n):
    a = np.zeros(2 * n - 1)
    m_lo1 = max(idx, 0)
    m_hi1 = min(idx + n - 1, 2 * n - 2)
    if m_lo1 <= m_hi1:
        hs = idx + n - 1 - m_hi1
        he = idx + n - 1 - m_lo1
        a[m_lo1:m_hi1 + 1] = 1.0 - h[hs:he + 1][::-1]
    m_lo2 = max(n - 1 - idx, 0)
    m_hi2 = min(2 * n - 2 - idx, 2 * n - 2)
    if m_lo2 <= m_hi2:
        hs2 = idx + m_lo2 - (n - 1)
        he2 = idx + m_hi2 - (n - 1)
        a[m_lo2:m_hi2 + 1] += -h[hs2:he2 + 1]
    return a

def two_coord_step(h, C, n, i, j):
    hi = h[i]
    hj = h[j]
    if hi < 1e-12 and hj > 1.0 - 1e-12:
        return False
    ai = compute_a(h, i, n)
    aj = compute_a(h, j, n)
    d = aj - ai
    d_lo = max(hi - 1.0, -hj)
    d_hi = min(hi, 1.0 - hj)
    if d_lo >= d_hi - 1e-14:
        return False
    lo, hi2 = d_lo, d_hi
    for _ in range(50):
        mid = (lo + hi2) / 2.0
        vals = C + d * mid
        if d[np.argmax(vals)] < 0:
            lo = mid
        else:
            hi2 = mid
    delta = (lo + hi2) / 2.0
    old_max = C.max()
    new_max = (C + d * delta).max()
    if new_max < old_max - 1e-14:
        C += d * delta
        h[i] -= delta
        h[j] += delta
        return True
    return False

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha + mu * (x.sum() - n / 2.0) ** 2

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_polish(h, n, deadline, max_iters=600):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(1000, 50.0), (10000, 200.0), (100000, 1000.0),
               (1000000, 5000.0), (10000000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        if deadline - time.time() < 8:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': max_iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

        h = best_h.copy()
        t_final = min(time.time() + 25, deadline - 4)
        h, c_fin = lbfgs_polish(h, n, t_final, max_iters=300)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
