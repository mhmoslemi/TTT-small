    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n-kk])
        grad[:n-kk] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    val += 200.0 * (x.sum() - n/2.0)**2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    ntot = len(x)
    # Only use significant weights
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    grad = np.zeros(ntot)
    for i in active:
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    grad += 400.0 * (x.sum() - n/2.0)
    return grad

def optimize_phase(h, n, t_end, alpha=10000, method='L-BFGS-B'):
    if method == 'L-BFGS-B':
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
        except Exception:
            return h, compute_c5(h, n)
    return h, compute_c5(h, n)

def pgd_optimize(h, n, t_end, top_k=40, alpha_smooth=50.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Use Barzilai-Borwein step size
    step = 0.01
    prev_h = None
    prev_g = None
    stall = 0
    while time.time() < t_end:
        c = compute_c5_vec(h, n)
        idx = np.argsort(c)[-top_k:]
        vals = c[idx]
        mx = vals.max()
        sh = alpha_smooth * (vals - mx)
        w = np.exp(sh - sh.max())
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for i in range(top_k):
            if w[i] < 1e-10:
                continue
            k = idx[i] - (n - 1)
            grad += w[i] * dx * grad_single(h, n, k)
        
        # BB step
        if prev_h is not None:
            s_vec = h - prev_h
            y_vec = grad - prev_g
            sy = s_vec @ y_vec
            ss = s_vec @ s_vec
            if sy > 1e-14:
                step_bb = ss / sy
                step = 0.7 * step + 0.3 * max(0.001, min(0.05, step_bb))
        prev_h = h.copy()
        prev_g = grad.copy()
        
        # Line search
        accepted = False
        for _ in range(20):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-14:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.4
        if not accepted:
            break
        if stall > 100:
            # Restart with perturbation
            h = best_h + 0.001 * np.random.randn(n)
            h = project_feasible(h, n)
            stall = 0
    return best_h, best_c

def fine_local(h, n, t_end):
    """Very fine local search with small steps"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(42)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in steps:
        improved = True
        while improved and time.time() < t_end:
            improved = False
            # Compute gradient for guidance
            c = compute_c5_vec(best_h, n)
            top_k = np.argsort(c)[-30:]
            mx = c.max()
            sh = 20.0 * (c[top_k] - mx)
            w = np.exp(sh - sh.max())
            w /= w.sum()
            dx = 2.0 / n
            grad = np.zeros(n)
            for i in range(len(top_k)):
                k = top_k[i] - (n-1)
                grad += w[i] * dx * grad_single(best_h, n, k)
            
            # Try gradient-guided moves
            # Move mass from high-grad to low-grad
            for attempt in range(200):
                if time.time() > t_end or improved:
                    break
                # Pick i (to increase) preferentially from low-grad region
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                if grad[i] < grad[j]:
                    # Move from j to i
                    if best_h[i] + d > 1 or best_h[j] - d < 0:
                        continue
                    h_new = best_h.copy()
                    h_new[i] += d
                    h_new[j] -= d
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_new
                        improved = True
                elif grad[j] < grad[i]:
                    if best_h[j] + d > 1 or best_h[i] - d < 0:
                        continue
                    h_new = best_h.copy()
                    h_new[j] += d
                    h_new[i] -= d
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_new
                        improved = True
    return best_h, best_c
