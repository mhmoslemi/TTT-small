                break
            x=h[i]
            B=np.zeros(C, dtype=float)
            B[off-i:off-i+n] += 1.0 - h
            B[off+i-n+1:off+i+1] -= h[::-1]

            zero_max = float(np.max(corr[zero])) if zero.any() else -1e300
            v0 = float(np.max(corr - B*x))
            v1 = float(np.max(corr + B*(1.0-x)))
            hi = max(v0, v1) + 1e-11

import time
import numpy as np
from scipy.optimize import minimize

_OPT_N = 0
_OPT_TG = 0.0
_OPT_AL = 1.0

def conv_full(x, y):
    m = x.size + y.size - 1
    L = 1 << (m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(x, L) * np.fft.rfft(y, L), L)[:m]

def c_raw(x):
    n = x.size
    if n <= 600:
        return np.correlate(x, 1.0 - x, mode='full')
    return conv_full(x, (1.0 - x)[::-1])

def grad_weights(x, w):
    n = x.size
    off = n - 1
    wr = w[::-1]
    if n <= 600:
        F = np.convolve(x, wr, mode='full')
        P = np.convolve(x[::-1], wr, mode='full')
    else:
        F = conv_full(x, wr)
        P = conv_full(x[::-1], wr)
    Fk = F[off:off+n]
    Pk = P[2*n-2::-1][:n]
    pref = np.empty(2*n, dtype=float)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    L0 = pref[n:2*n] - pref[0:n]
    return L0 - Fk - Pk

def lse_loss(x):
    c = c_raw(x)
    m = float(np.max(c))
    z = _OPT_AL * (c - m)
    zm = float(np.max(z))
    ez = np.exp(z - zm)
    return m + (zm + float(np.log(ez.sum()))) / _OPT_AL

def lse_jac(x):
    c = c_raw(x)
    al = _OPT_AL
    m = float(np.max(c))
    z = al * (c - m)
    zm = float(np.max(z))
    ez = np.exp(z - zm)
    w = ez / float(ez.sum())
    return grad_weights(x, w)

def sum_con(x):
    return float(np.sum(x) - _OPT_TG)

    return np.ones(_OPT_N, dtype=float)

import time
import numpy as np
from scipy.optimize import minimize

_OPT_N = 0
_OPT_TG = 0.0
_OPT_AL = 1.0


def conv_full(x, y):
    m = x.size + y.size - 1
    L = 1 << (m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(x, L) * np.fft.rfft(y, L), L)[:m]


def c_raw(x):
    n = x.size
    if n <= 600:
        return np.correlate(x, 1.0 - x, mode='full')
    return conv_full(x, (1.0 - x)[::-1])


def grad_weights(x, w):
    n = x.size
    off = n - 1
    wr = w[::-1]
    if n <= 600:
        F = np.convolve(x, wr, mode='full')
        P = np.convolve(x[::-1], wr, mode='full')
    else:
        F = conv_full(x, wr)
        P = conv_full(x[::-1], wr)

    Fk = F[off:off + n]
    Pk = P[2 * n - 2::-1][:n]

    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    L0 = pref[n:2 * n] - pref[0:n]
    return L0 - Fk - Pk


def lse_loss(x):
    c = c_raw(x)
    m = float(np.max(c))
    z = _OPT_AL * (c - m)
    zm = float(np.max(z))
    ez = np.exp(z - zm)
    return m + (zm + float(np.log(ez.sum()))) / _OPT_AL


def lse_jac(x):
    c = c_raw(x)
    al = _OPT_AL
    m = float(np.max(c))
    z = al * (c - m)
    zm = float(np.max(z))
    ez = np.exp(z - zm)
    w = ez / float(ez.sum())
    return grad_weights(x, w)


def sum_con(x):
    return float(np.sum(x) - _OPT_TG)


def sum_con_jac(x):
    return np.ones(_OPT_N, dtype=float)


def project_feasible(x, n):
    x = np.asarray(x, dtype=float)
    target = 0.5 * n
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0

        t = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - t, 0.0, 1.0)))
        if s > target:
            lo = t
        else:
            hi = t

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    d = float(target - np.sum(y))
    if abs(d) > 1e-14:
        margin = np.minimum(y, 1.0 - y)
        idx = int(np.argmax(margin))
        if margin[idx] + 1e-15 >= abs(d):
            y[idx] += d
        elif d > 0:
            room = 1.0 - y
            sr = float(room.sum())
            if sr > 1e-15:
                y += (d / sr) * room
        else:
            sy = float(y.sum())
            if sy > 1e-15:
                y += (d / sy) * y
        y = np.clip(y, 0.0, 1.0)
    return y


def exact_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))


def fallback_h(n, seed=1):
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    h = (np.sin(4.0 * np.pi * t + 0.37) > 0.0).astype(float)
    return project_feasible(h, n)


def to_binary(x, n):
    b = np.zeros(n, dtype=float)
    k = int(0.5 * n)
    idx = np.argsort(x)[::-1]
    if k > 0:
        b[idx[:k]] = 1.0
    if n % 2 != 0 and k < n:
        b[idx[k]] = 0.5
    return project_feasible(b, n)


def quantize_candidate(h, n, k):
    k = int(k)
    if k < 0:
        k = 0
    if k > n // 2:
        k = n // 2
    q = h.copy()
    if k > 0:
        idx = np.argsort(q)
        q[idx[:k]] = 0.0
        q[idx[n - k:]] = 1.0
    return project_feasible(q, n)


def smooth_candidate(h, n, w):
    if w <= 1:
        return project_feasible(h, n)
    pads = (w - 1) // 2
    ph = np.pad(h, (pads, pads), mode='edge')
    ker = np.ones(w, dtype=float) / float(w)
    return project_feasible(np.convolve(ph, ker, mode='valid'), n)


def interp_to(h, n, nh):
    if nh == n:
        return h.copy()
    xo = np.linspace(0.0, 2.0, n)
    xn = np.linspace(0.0, 2.0, nh)
    return project_feasible(np.interp(xn, xo, h), nh)

def run_slsqp_once(h, n, alpha, maxiter):
    global _OPT_N, _OPT_TG, _OPT_AL
    h = project_feasible(h, n)
    _OPT_N = int(n)
    _OPT_TG = 0.5 * _OPT_N
    _OPT_AL = float(alpha)

    cons = {'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}
    try:
        res = minimize(
            lse_loss, h, method='SLSQP', jac=lse_jac,
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-11, 'disp': False}
        )
        out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        out = h.copy()
    return project_feasible(out, n)


def coord_descent(h, n, deadline=None, max_sweeps=20):
    h = project_feasible(h, n).astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    C = 2 * n - 1
    off = n - 1

    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            x = h[i]
            B = np.zeros(C, dtype=float)
            B[off + i - n + 1:off + i + 1] += 1.0 - h[::-1]
            B[off - i:off - i + n] -= h

            pos = B > 1e-13
            neg = B < -1e-13
            zero = ~(pos | neg)
            zero_max = float(np.max(corr[zero])) if zero.any() else -1e300

            v0 = float(np.max(corr - B * x))
            v1 = float(np.max(corr + B * (1.0 - x)))
            hi = max(v0, v1) + 1e-11
            lo = min(v0, v1) - 1e-11

            for __ in range(30):
                mid = 0.5 * (lo + hi)
                if zero_max > mid + 1e-13:
                    lo = mid
                    continue
                lb = 0.0
                if neg.any():
                    lb = max(lb, x + float(np.max((mid - corr[neg]) / B[neg])))
                ub = 1.0
                if pos.any():
                    ub = min(ub, x + float(np.min((mid - corr[pos]) / B[pos])))
                if lb <= ub + 1e-12 and zero_max <= mid + 1e-13:
                    hi = mid
                else:
                    lo = mid

            lb = 0.0
            if neg.any():
                lb = max(lb, x + float(np.max((hi - corr[neg]) / B[neg])))
            ub = 1.0
            if pos.any():
                ub = min(ub, x + float(np.min((hi - corr[pos]) / B[pos])))

            if lb <= ub + 1e-11 and zero_max <= hi + 1e-11:
                t = float(np.clip(0.5 * (lb + ub), 0.0, 1.0))
            else:
                t = x

            if abs(t - x) > 1e-14:
                corr += B * (t - x)
                h[i] = t

        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)

    return h
