import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_for_shift(h, n, s):
    j = np.arange(n)
    grad = np.zeros(n)
    mask1 = (j + s >= 0) & (j + s < n)
    grad[mask1] += (1.0 - h[j[mask1] + s])
    mask2 = (j - s >= 0) & (j - s < n)
    grad[mask2] -= h[j[mask2] - s]
    return grad

def all_grads(h, n):
    """Return matrix G where G[m, j] = d(c[m])/d(h[j]), m=0..2n-2"""
    ns = 2 * n - 1
    G = np.zeros((ns, n))
    for m in range(ns):
        s = m - (n - 1)
        G[m] = grad_for_shift(h, n, s)
    return G

def lp_linearize(h0, n, max_step=1.0):
    """Solve LP: min t s.t. linearized c[m] <= t, sum(h)=n/2, 0<=h<=1"""
    ns = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h0, 1 - h0, mode='full')
    
    # Build constraint matrix for linprog
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize: t => c = [0,...,0,1]
    # Constraints: for each m: sum_j G[m,j]*h_j - t <= corr[m] - sum_j G[m,j]*h0_j
    #              sum_j h_j = n/2
    #              -h_j <= 0 (lower bound handled by bounds)
    #              h_j <= 1 (upper bound handled by bounds)
    
    G = all_grads(h0, n)
    rhs = corr - G @ h0  # c[m]_0 - sum_j G[m,j]*h0[j]
    
    # A_ub: each row m: [G[m,0], ..., G[m,n-1], -1]
    A_ub = np.zeros((ns, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = rhs
    
    # A_eq: [1, 1, ..., 1, 0] = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return h0
    return res.x[:n]

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_for_shift(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    m1 = (j + s >= 0) & (j + s < n)
    g[m1] += (1.0 - h[j[m1] + s])
    m2 = (j - s >= 0) & (j - s < n)
    g[m2] -= h[j[m2] - s]
    return g

def build_G(h, n):
    ns = 2 * n - 1
    G = np.zeros((ns, n))
    for m in range(ns):
        G[m] = grad_for_shift(h, n, m - (n - 1))
    return G

def lp_solve(h0, n, delta=0.08):
    ns = 2 * n - 1
    corr = np.correlate(h0, 1 - h0, mode='full')
    G = build_G(h0, n)
    rhs = corr - G @ h0
    A_ub = np.column_stack([G, -np.ones(ns)])
    b_ub = rhs.copy()
    if delta is not None:
        A_d = np.zeros((2 * n, n + 1))
        b_d = np.zeros(2 * n)
        for j in range(n):
            A_d[j, j] = 1.0
            b_d[j] = h0[j] + delta
            A_d[n + j, j] = -1.0
            b_d[n + j] = h0[j] - delta
        A_ub = np.vstack([A_ub, A_d])
        b_ub = np.concatenate([b_ub, b_d])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return np.clip(res.x[:n], 0, 1)
    except Exception:
        pass
    return h0.copy()

def lp_search(h, n, deadline, max_iter=10000):
    h = h.copy()
    cur_c = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_lp = lp_solve(h, n, delta=0.08)
        if np.max(np.abs(h_lp - h)) < 1e-12:
            stall += 1
            if stall > 3:
                corr = np.correlate(h, 1 - h, mode='full')
                ms = int(np.argmax(corr))
                s = ms - (n - 1)
                g = grad_for_shift(h, n, s)
                gn = np.linalg.norm(g)
                if gn < 1e-12:
                    break
                found = False
                for eta in [0.05 / gn, 0.02 / gn, 0.01 / gn, 0.005 / gn]:
                    h_try = project_feasible(h - eta * g, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < cur_c - 1e-14:
                        h = h_try
                        cur_c = c_try
                        stall = 0
                        found = True
                        break
                if not found:
                    break
            continue
        improved = False
        for alpha in [1.0, 0.6, 0.4, 0.2, 0.1]:
            h_try = project_feasible(h + alpha * (h_lp - h), n)
            c_try = compute_c5(h_try, n)
            if c_try < cur_c - 1e-14:
                h = h_try
                cur_c = c_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 8:
                break
    return h, cur_c

def coord_descent(h, n, deadline, max_sweeps=300):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            ms1 = (k1 >= 0) & (k1 < n)
            B[ms1] += (1.0 - h[k1[ms1]])
            k2 = i + m_range
            ms2 = (k2 >= 0) & (k2 < n)
            B[ms2] -= h[k2[ms2]]
            v0 = (corr - B * h[i]).max()
            v1 = (corr + B * (1.0 - h[i])).max()
            hi = max(v0, v1) + 1e-9
            lo = min(v0, v1) - 1e-9
            pos = B > 1e-14
            neg = B < -1e-14
            zm = ~(pos | neg)
            zm_max = corr[zm].max() if zm.any() else -np.inf
            for _ in range(25):
                mid = (lo + hi) / 2.0
                if zm_max > mid + 1e-14:
                    lo = mid
                    continue
                lb = 0.0
                if neg.any():
                    lb = max(lb, h[i] + np.max((mid - corr[neg]) / B[neg]))
                ub = 1.0
                if pos.any():
                    ub = min(ub, h[i] + np.min((mid - corr[pos]) / B[pos]))
                if lb <= ub + 1e-12:
                    hi = mid
                else:
                    lo = mid
            lb = 0.0
            if neg.any():
                lb = max(lb, h[i] + np.max((hi - corr[neg]) / B[neg]))
            ub = 1.0
            if pos.any():
                ub = min(ub, h[i] + np.min((hi - corr[pos]) / B[pos]))
            xo = np.clip((lb + ub) / 2.0, 0.0, 1.0)
            if abs(xo - h[i]) > 1e-15:
                corr += B * (xo - h[i])
                h[i] = xo
    return h, compute_c5(h, n)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def slsqp_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_cons(x):
    return x.sum() - len(x) / 2.0

def slsqp_polish(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': slsqp_cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def make_noisy(h, n, scale, rng):
    return project_feasible(h + scale * rng.randn(n), n)

def make_random_binary(n, rng):
    return to_binary(rng.rand(n), n)

    p1 = t0 + 0.30 * total
    h1, c1 = lp_search(best_h, n, p1, max_iter=10000)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Coordinate descent polish (12%)
    p2 = t0 + 0.42 * total
    h2, c2 = coord_descent(best_h, n, p2, max_sweeps=500)
    h2 = project_feasible(h2, n)
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    # Phase 3: SLSQP smooth polish (13%)
    p3 = t0 + 0.55 * total
    h3 = best_h.copy()
    for alpha in [5000, 50000, 500000, 5000000]:
        if time.time() > p3 - 5:
            break
        h3 = slsqp_polish(h3, n, alpha, 2000)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: LP search again from polished result (10%)
    p4 = t0 + 0.65 * total
    h4, c4 = lp_search(best_h, n, p4, max_iter=5000)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Diverse restarts (25%)
    p5 = t0 + 0.90 * total
    n_trials = 10
    per_trial = max(4.0, (p5 - time.time()) / n_trials)
    for t in range(n_trials):
        if time.time() > p5 - 3:
            break
        r = t % 5
        if r == 0:
            h_r = make_noisy(best_h, n, 0.04, rng)
        elif r == 1:
            h_r = to_binary(make_noisy(best_h, n, 0.08, rng), n)
        elif r == 2:
            h_r = make_random_binary(n, rng)
        elif r == 3:
            h_r = make_noisy(best_h, n, 0.12, rng)
        else:
            perm = rng.permutation(n)
            h_shuffled = best_h[perm]
            h_r = project_feasible(h_shuffled, n)
        trial_dl = min(p5 - 2, time.time() + per_trial * 0.55)
        h_t, c_t = lp_search(h_r, n, trial_dl, max_iter=3000)
        if c_t < best_c:
            best_c, best_h = c_t, h_t.copy()
        trial_dl2 = min(p5 - 1, time.time() + per_trial * 0.35)
        h_cd, c_cd = coord_descent(h_t, n, trial_dl2, max_sweeps=150)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd.copy()

    # Phase 6: Final polish (10%)
    h6, c6 = lp_search(best_h, n, dl, max_iter=10000)
    if c6 < best_c:
        best_c, best_h = c6, h6.copy()
    h6b, c6b = coord_descent(best_h, n, dl, max_sweeps=300)
    h6b = project_feasible(h6b, n)
    c6b = compute_c5(h6b, n)
    if c6b < best_c:
        best_c, best_h = c6b, h6b.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
