def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

    v_hi = max(c_at_0, c_at_max) + 1e-12
    v_lo = min(c_at_0, c_at_max) - 1e-12
    for _ in range(55):
        v_mid = (v_lo + v_hi) / 2.0
        if feasible(v_mid):
            v_hi = v_mid
        else:
            v_lo = v_mid
    v_opt = v_hi
    pos = a_k > 1e-18
    neg = a_k < -1e-18
    lower = 0.0
    upper = delta_max
    if np.any(pos):
        upper = min(upper, np.min((v_opt - c[pos]) / a_k[pos]))
    if np.any(neg):
        lower = max(lower, np.max((v_opt - c[neg]) / a_k[neg]))
    delta = max(0.0, min(delta_max, (lower + upper) / 2.0))
    return delta

def exact_pair_search(h, n, deadline, alpha=100000, max_rounds=5000):
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        V = precompute_derivs(h, n)
        c = compute_c5_vec(h, n)
        # Get gradient for pair selection
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = V.T @ w
        # Select best pair
        g = grad.copy()
        g[h >= 1.0 - 1e-9] = -np.inf
        g[h <= 1e-9] = np.inf
        i_rec = int(np.argmax(g))
        i_don = int(np.argmin(g))
        if not (np.isfinite(g[i_rec]) and np.isfinite(g[i_don])) or i_rec == i_don:
            # Try random pair
            i_don = int(rng.randint(n))
            i_rec = int(rng.randint(n))
            if i_rec == i_don:
                break
        # Solve 1D
        delta = pair_1d_solve(h, n, V, c, i_rec, i_don)
        if delta > 1e-14:
            h[i_rec] += delta
            h[i_don] -= delta
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
            else:
                h[i_rec] -= delta
                h[i_don] += delta
        else:
            # Try a few random pairs
            improved = False
            for _ in range(30):
                j = int(rng.randint(n))
                k = int(rng.randint(n))
                if j == k:
                    continue
                d = pair_1d_solve(h, n, V, c, j, k)
                if d > 1e-14:
                    h[j] += d
                    h[k] -= d
                    c_try = compute_c5(h, n)
                    if c_try < best_c - 1e-15:
                        best_c = c_try
                        improved = True
                    else:
                        h[j] -= d
                        h[k] += d
                if improved:
                    break
            if not improved:
                # Try reverse direction on best pair
                d2 = pair_1d_solve(h, n, V, c, i_don, i_rec)
                if d2 > 1e-14:
                    h[i_don] += d2
                    h[i_rec] -= d2
                    c_try = compute_c5(h, n)
                    if c_try < best_c - 1e-15:
                        best_c = c_try
                    else:
                        h[i_don] -= d2
                        h[i_rec] += d2
                else:
                    break
    return h, best_c

def fine_local_pair(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in steps:
        stagnant = 0
        while time.time() < deadline and stagnant < 500:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 80)
                hi = min(n, i + 81)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[j] -= d
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[j] += d
                            h[i] -= d
            if not any_imp:
                stagnant += 1
            else:
                stagnant = 0
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(650, 0.5)
        n = 650

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing alpha
    alpha_schedule = [200, 2000, 20000, 200000, 2000000]
    for alpha in alpha_schedule:
        rem = deadline - time.time()
        if rem < 40:
            break
        alloc = min(90, rem * 0.2)
        h_new, c_new = lbfgs_round(best_h, n, time.time() + alloc, alpha, maxiter=300)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Exact pair search (main workhorse)
    rem = deadline - time.time()
    if rem > 100:
        h_new, c_new = exact_pair_search(best_h, n, time.time() + rem * 0.5, alpha=100000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 3: Another round of exact pair search with different alpha
    rem = deadline - time.time()
    if rem > 80:
        h_new, c_new = exact_pair_search(best_h, n, time.time() + rem * 0.4, alpha=5000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Fine local pair search
    rem = deadline - time.time()
    if rem > 30:
        best_h, best_c = fine_local_pair(best_h, n, time.time() + rem * 0.6)

    # Phase 5: Final L-BFGS-B squeeze
    rem = deadline - time.time()
    if rem > 15:
        h_new, c_new = lbfgs_round(best_h, n, time.time() + rem * 0.7, 5000000, maxiter=200)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 6: Final exact pair polish
    rem = deadline - time.time()
    if rem > 10:
        best_h, best_c = exact_pair_search(best_h, n, time.time() + rem * 0.8, alpha=500000)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
