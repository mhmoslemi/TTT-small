import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _smooth_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12: continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

    return _smooth_obj(h, n, alpha) + mu * (h.sum() - n/2.0)**2
