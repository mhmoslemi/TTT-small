```python
def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def minimize_1d_smooth(A, B, S, alpha, mu, target):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)

        m = vals.max()

        return m + np.log(e.sum()) / alpha + mu * (S + t - target) ** 2

            d_pt = a + gr * (b - a); fd = f(d_pt)
        if b - a < 1e-12:

    return (a + b) / 2.0

def coord_descent(h, n, alpha, mu, max_sweeps, rng, tol=1e-11):

    target = n / 2.0
    h = h.copy()
    c = compute_c5_vec(h, n)
    best_c = c.max()
    m_all = np.arange(2 * n - 1)

    for sweep in range(max_sweeps):
        total_change = 0.0
        order = rng.permutation(n)
        for i in order:
            old_h = h[i]
            B = np.zeros(2 * n - 1)
            m1 = (i + k_all >= 0) & (i + k_all < n)
            B[m1] += 1.0 - h[i + k_all[m1]]
            m2 = (i - k_all >= 0) & (i - k_all < n)
            B[m2] -= h[i - k_all[m2]]

            delta = t_opt - old_h
            if abs(delta) > 1e-13:
                h[i] = t_opt
                c += B * delta
                total_change += abs(delta)
        new_c = c.max()
        if new_c < best_c:
            best_c = new_c
        if total_change < tol:
            break
    return h, best_c

def two_coord_polish(h, n, deadline, max_rounds=50):

    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 200.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n

    grad = np.zeros(len(x))

            grad[:nv] -= w[i] * dx * x[kk:]

    return grad
```

```python
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def minimize_1d_smooth(A, B, S, alpha, mu, target):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)

    def f(t):
        vals = A + B * t
        m = vals.max()
        e = np.exp(alpha * (vals - m))
        return m + np.log(e.sum()) / alpha + mu * (S + t - target) ** 2

    fc = f(c_pt)
    fd = f(d_pt)
    for _ in range(45):
        if fc < fd:
            b = d_pt; d_pt = c_pt; fd = fc
            c_pt = b - gr * (b - a); fc = f(c_pt)
        else:
            a = c_pt; c_pt = d_pt; fc = fd
            d_pt = a + gr * (b - a); fd = f(d_pt)
        if b - a < 1e-12:
            break
    return (a + b) / 2.0

def coord_descent(h, n, alpha, mu, max_sweeps, rng, tol=1e-11):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    c = compute_c5_vec(h, n)
    best_c = c.max()
    m_all = np.arange(2 * n - 1)
    k_all = m_all - (n - 1)

    for sweep in range(max_sweeps):
        total_change = 0.0
        order = rng.permutation(n)
        for i in order:
            old_h = h[i]
            B = np.zeros(2 * n - 1)
            m1 = (i + k_all >= 0) & (i + k_all < n)
            B[m1] += 1.0 - h[i + k_all[m1]]
            m2 = (i - k_all >= 0) & (i - k_all < n)
            B[m2] -= h[i - k_all[m2]]
            B *= dx
            A = c - B * old_h
            S = h.sum() - old_h
            t_opt = minimize_1d_smooth(A, B, S, alpha, mu, target)
            delta = t_opt - old_h
            if abs(delta) > 1e-13:
                h[i] = t_opt
                c += B * delta
                total_change += abs(delta)
        new_c = c.max()
        if new_c < best_c:
            best_c = new_c
        if total_change < tol:
            break
    return h, best_c

def two_coord_polish(h, n, deadline, n_pairs=300):
    dx = 2.0 / n
    h = h.copy()
    c = compute_c5_vec(h, n)
    best_c = c.max()
    m_all = np.arange(2 * n - 1)
    k_all = m_all - (n - 1)

    def get_B(i, h_curr):
        B = np.zeros(2 * n - 1)
        m1 = (i + k_all >= 0) & (i + k_all < n)
        B[m1] += 1.0 - h_curr[i + k_all[m1]]
        m2 = (i - k_all >= 0) & (i - k_all < n)
        B[m2] -= h_curr[i - k_all[m2]]
        return B * dx

    rng = np.random.RandomState(999)
    improved_total = True
    while improved_total and time.time() < deadline:
        improved_total = False
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i, j = rng.choice(n, 2, replace=False)
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            hi_b = min(1.0 - hi, hj)
            if lo >= hi_b - 1e-12:
                continue
            Bi = get_B(i, h)
            Bj = get_B(j, h)
            dB = Bi - Bj
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi_b
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)

            def f2(d):
                return float((c + d * dB).max())

            fc = f2(c_pt)
            fd = f2(d_pt)
            for _ in range(35):
                if fc < fd:
                    b = d_pt; d_pt = c_pt; fd = fc
                    c_pt = b - gr * (b - a); fc = f2(c_pt)
                else:
                    a = c_pt; c_pt = d_pt; fc = fd
                    d_pt = a + gr * (b - a); fd = f2(d_pt)
                if b - a < 1e-10:
                    break
            d_opt = (a + b) / 2.0
            new_max = float((c + d_opt * dB).max())
            if new_max < best_c - 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                c += d_opt * dB
                best_c = new_max
                improved_total = True
    return h, best_c

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 300.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(200, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(len(x))
    for idx in top_idx:
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        nv = n - abs(k)
        if k >= 0:
            grad[:nv] += w[idx] * dx * (1.0 - x[k:])
            grad[k:] -= w[idx] * dx * x[:nv]
        else:
            kk = -k
            grad[kk:] += w[idx] * dx * (1.0 - x[:nv])
            grad[:nv] -= w[idx] * dx * x[kk:]
    grad += 600.0 * (x.sum() - n / 2.0)
    return grad

def optimize_at_n(h_start, n, deadline, rng, alphas=(100, 300, 1000, 3000, 8000), mu=500.0):
    best_h = project_feasible(h_start, n)
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        sweeps = 12 if alpha < 1000 else 8
        best_h, c_cd = coord_descent(best_h, n, alpha, mu, sweeps, rng)
        best_h = project_feasible(best_h, n)
        c_cd = compute_c5(best_h, n)
        if c_cd < best_c:
            best_c = c_cd
    return best_h, best_c
```
