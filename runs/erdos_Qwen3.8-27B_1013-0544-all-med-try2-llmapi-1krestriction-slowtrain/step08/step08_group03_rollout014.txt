def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_k(h, n, k):
    """Gradient of the k-th correlation w.r.t. h"""
    nv = n - abs(k)
    g = np.zeros(n)
    if k >= 0:
        g[:nv] += (1.0 - h[k:k+nv])
        g[k:k+nv] -= h[:nv]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:nv])
        g[:nv] -= h[kk:]
    return g

def smooth_max_grad(h, n, alpha, top_k=50):

    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n

        k = idx[i] - (n - 1)

The issue is that if multiple lags are at the maximum, the gradient of the max is not well-defined (it's the subdifferential, which is the convex hull of the individual gradients). For minimization, we want to move in a direction that decreases ALL active lags simultaneously. The average gradient is a reasonable choice, but it might not be the best.

A better approach: Find the direction d that minimizes the maximum of (g_k · d) over active k, where g_k is the gradient of c_k. This is a linear program:
min_t t
s.t. g_k · d ≤ t for all active k
     ||d||² ≤ 1

This gives the "minimum maximum" descent direction, which is optimal for the objective.

    """Find direction d that minimizes max_k (g_k · d) subject to ||d||_2 ≤ 1"""

    dx = 2.0 / n

    G = np.zeros((m, n))

        G[i] = dx * grad_c_k(h, n, k)

    # This is a QP. Use a simple approach:
    # The optimal d is in the direction of -avg(G rows weighted by dual variables)
    # For a quick approximation, just use the negative average:
    d = -G.mean(axis=0)
    norm = np.linalg.norm(d)
    if norm > 0:
        d /= norm
    return d

                continue
            found = False
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    no_improve = 0
                    found = True
                else:
                    h[i] -= d
                    h[j] += d
            if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                h[i] -= d
                h[j] += d
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    no_improve = 0
                    found = True
                else:
                    h[i] += d
                    h[j] -= d
            if not found:
                no_improve += 1
    return h, best_c

def directed_local_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    alpha = 5000
    top_k = 40
    d = 0.0005
    for _ in range(20000):
        if time.time() > deadline:
            break
        g = active_grad(h, n, alpha, top_k)
        neg_idx = np.where(g < -0.005)[0]
        pos_idx = np.where(g > 0.005)[0]
        if len(neg_idx) == 0 or len(pos_idx) == 0:
            break
        i = neg_idx[rng.randint(len(neg_idx))]
        j = pos_idx[rng.randint(len(pos_idx))]
        found = False
        for dd in [d, d * 0.5, d * 0.25, d * 0.1]:
            if h[i] + dd <= 1.0 and h[j] - dd >= 0.0:
                h[i] += dd
                h[j] -= dd
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    found = True
                    break
                h[i] -= dd
                h[j] += dd
        if not found:
            d *= 0.7
            if d < 1e-6:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.zeros(n)
        h_init[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with penalty
    for alpha, lam in [(2000, 80), (5000, 80), (10000, 100)]:
        if time.time() > deadline - 350:
            break
        try:
            res = minimize(obj_pen, best_h, jac=grad_pen, args=(n, alpha, lam),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-13})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Gradient descent with high alpha
    if time.time() < deadline - 250:
        t2 = min(time.time() + 120, deadline - 150)
        best_h, best_c = grad_desc(best_h, n, t2, alpha=3000, top_k=50, max_iter=3000)

    # Phase 3: Multi-restart
    for restart in range(5):
        if time.time() > deadline - 180:
            break
        scale = 0.015 * (1 + restart * 0.3)
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.005:
            t_r = min(time.time() + 50, deadline - 80)
            h_r, c_r = grad_desc(h_pert, n, t_r, alpha=5000, top_k=40, max_iter=1500)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 4: Try other n values
    for n_try in [480, 520, 540, 560]:
        if time.time() > deadline - 130:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        alloc = deadline - time.time() - 60
        if alloc < 30:
            continue
        try:
            res = minimize(obj_pen2, h_interp, jac=grad_pen2, args=(n_try, 5000, 80),
                        method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_try,
                        options={'maxiter': 250, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n_try)
            c_new = compute_c5(h_new, n_try)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                n = n_try
        except Exception:
            pass

    # Phase 5: Final gradient descent
    if time.time() < deadline - 40:
        best_h, best_c = grad_desc(best_h, n, deadline - 25, alpha=8000, top_k=60, max_iter=2000)

    # Phase 6: Directed local search
    if time.time() < deadline - 10:
        best_h, best_c = directed_local_search(best_h, n, deadline - 5, rng)

    # Phase 7: Random local search
    if time.time() < deadline - 1:
        best_h, best_c = local_search(best_h, n, deadline - 0.5, rng)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
