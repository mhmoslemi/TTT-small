import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 20.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, top_k=80):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    idx = np.argsort(w)[-top_k:] if top_k < len(w) else np.arange(len(w))
    w_sp = w[idx] / w[idx].sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i, ii in enumerate(idx):
        if w_sp[i] < 1e-12:
            continue
        k = ii - (n - 1)
        nv = n - abs(k)
        if nv <= 0:
            continue
        if k >= 0:
            grad[k:k+nv] += w_sp[i] * (1.0 - x[:nv])
            grad[:nv] -= w_sp[i] * x[k:k+nv]
        else:
            mg = -k
            grad[:nv] += w_sp[i] * (1.0 - x[mg:mg+nv])
            grad[mg:mg+nv] -= w_sp[i] * x[:nv]
    grad *= dx
    grad += 40.0 * (x.sum() - n / 2.0)
    return grad

def fast_coord_descent(h, n, deadline, top_lags=40, max_passes=5000):
    """Fast coordinate descent with O(n) per-coordinate evaluation."""
    dx = 2.0 / n
    c_vec = compute_c5_vec(h, n)
    best_c = float(c_vec.max())
    h_best = h.copy()
    stall = 0
    candidates = np.array([-0.2, -0.1, -0.05, -0.03, -0.01, -0.005,
                           0.005, 0.01, 0.03, 0.05, 0.1, 0.2, 0.5, -0.5])
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        c_vec = compute_c5_vec(h, n)
        cur_max = float(c_vec.max())
        top_idx = np.argsort(c_vec)[-top_lags:]
        
        for m in range(n):
            old_v = h[m]
            if old_v <= 0 or old_v >= 1:
                # Boundary: try only inward
                if old_v <= 0:
                    cands = [0.05, 0.1, 0.2]
                else:
                    cands = [old_v - 0.05, old_v - 0.1, old_v - 0.2]
            else:
                cands = old_v + candidates
                
            best_val = cur_max
            best_v = old_v
            
            for v in cands:
                if v < 0 or v > 1:
                    continue
                dv = v - old_v
                if abs(dv) < 1e-10:
                    continue
                # Compute new values for top lags only
                max_new = 0.0
                for ti in top_idx:
                    d = ti - (n - 1)
                    if d == 0:
                        nv = c_vec[ti] + dv * (1.0 - 2.0 * old_v - dv)
                    else:
                        coeff = 0.0
                        if 0 <= m + d < n:
                            coeff += (1.0 - h[m + d])
                        if 0 <= m - d < n:
                            coeff -= h[m - d]
                        nv = c_vec[ti] + dv * coeff
                    if nv > max_new:
                        max_new = nv
                if max_new < best_val - 1e-13:
                    best_val = max_new
                    best_v = v
            
            if best_v != old_v:
                h[m] = best_v
                improved = True
        
        if improved:
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                h_best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 3:
                break
    
    return h_best, best_c

def pair_polish(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def random_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

            h_new = project_feasible(res.x, best_n)
