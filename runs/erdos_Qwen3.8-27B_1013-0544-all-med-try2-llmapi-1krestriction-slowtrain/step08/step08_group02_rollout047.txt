def binary_local_search(h_bin, n, deadline, rng):
    # h_bin is binary array with sum = n/2
    # Compute all correlations
    c = np.correlate(h_bin, 1 - h_bin, mode='full')
    dx = 2.0 / n
    best_c = c.max() * dx
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    no_improve = 0
    while time.time() < deadline and no_improve < 200:
        found_improvement = False
        # Try random swaps
        n_tries = min(len(ones) * len(zeros), 50000)
        best_swap = None
        best_swap_c = best_c
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            # Swap
            h_bin[i] = 0
            h_bin[j] = 1
            c_new = np.correlate(h_bin, 1 - h_bin, mode='full')
            c_max = c_new.max()
            # Undo
            h_bin[i] = 1
            h_bin[j] = 0
            if c_max < best_swap_c:
                best_swap_c = c_max
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0
            h_bin[j] = 1
            c = np.correlate(h_bin, 1 - h_bin, mode='full')
            best_c = c.max() * dx
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            no_improve = 0
            found_improvement = True
        else:
            no_improve += 1
    
    return h_bin, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(x, n, alpha):
    """Compute smooth max of correlation and its gradient w.r.t. x."""
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    z = alpha * (c - m)
    z_max = z.max()
    log_sum = z_max + np.log(np.sum(np.exp(z - z_max)))
    val = (alpha * m + log_sum) / alpha
    w = np.exp(z - z_max)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        shift = k - (n - 1)
        wm = w[k]
        if shift >= 0:
            nv = n - shift
            grad[:nv] += wm * (1.0 - x[shift:shift + nv])
            grad[shift:shift + nv] += wm * (-x[:nv])
        else:
            ss = -shift
            nv = n - ss
            grad[ss:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[ss:])
    return val, grad

def nesterov_search(h, n, deadline, alpha_init=100, alpha_max=500000, 
                     lr_init=0.01, momentum=0.9):
    """Nesterov momentum projected gradient descent with adaptive alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_init
    v = np.zeros(n)
    x = h.copy()
    lr = lr_init
    
    while time.time() < deadline:
        # Compute gradient
        val, grad = smooth_max_val_grad(x, n, alpha)
        
        # Nesterov step
        x_nest = x + momentum * v
        _, grad_nest = smooth_max_val_grad(x_nest, n, alpha)
        
        # Update
        v = momentum * v - lr * grad_nest
        x_new = x + v
        
        # Project
        x_new = project_feasible(x_new, n)
        
        # Evaluate
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        
        x = x_new
        
        # Adaptive learning rate
        if c_new < best_c + 1e-10:
            lr *= 1.1
        else:
            lr *= 0.9
        
        # Increase alpha periodically
        alpha = min(alpha * 1.05, alpha_max)
        
        # Check convergence
        if np.max(np.abs(v)) < 1e-10:
            break
    
    return best_h, best_c
