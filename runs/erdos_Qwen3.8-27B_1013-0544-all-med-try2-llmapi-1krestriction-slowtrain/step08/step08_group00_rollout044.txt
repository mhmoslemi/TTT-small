    def _eq(x, n):
        return float(x.sum() - n / 2.0)
    def _eq_jac(x, n):
        return np.ones(n)
    def _obj_s(x, n, alpha):
        c = compute_correlation(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_s, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-19})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = (np.floor(np.arange(half) * phi) % n).astype(int)
    h[positions] = 1.0
    return project_feasible(h, n)

def ramp_start(n, rng):
    t = np.linspace(0, 1, n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * t + rng.uniform(0, 2*np.pi))
    h += rng.randn(n) * 0.1
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 650
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B (main optimization)
    t1 = min(t0 + 200, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=800)

    # Phase 2: SLSQP polish
    if time.time() < deadline - 350:
        t2 = min(time.time() + 60, deadline - 350)
        h_s, c_s = slsqp_optimize(best_h, best_n, t2, alpha=5000000, maxiter=5000)
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 3: Bottleneck coordinate descent
    if time.time() < deadline - 200:
        t3 = min(time.time() + 120, deadline - 200)
        h_d, c_d = bottleneck_coord_descent(best_h, best_n, t3)
        if c_d < best_c:
            best_c, best_h = c_d, h_d

    # Phase 4: Local pair search (targeted)
    if time.time() < deadline - 80:
        t4 = min(time.time() + 60, deadline - 80)
        best_h, best_c = pair_local_search(best_h, best_n, t4, radius=60)

    # Phase 5: Try golden ratio start
    if time.time() < deadline - 100:
        h_g = golden_start(best_n)
        alloc = min(50, deadline - time.time() - 60)
        if alloc > 20:
            h_t, c_t = lbfgs_optimize(h_g, best_n, time.time() + alloc, max_iters=300)
            if c_t < best_c:
                best_c, best_h = c_t, h_t

    # Phase 6: Try ramp start
    if time.time() < deadline - 60:
        h_r = ramp_start(best_n, rng)
        alloc = min(30, deadline - time.time() - 30)
        if alloc > 10:
            h_t, c_t = lbfgs_optimize(h_r, best_n, time.time() + alloc, max_iters=200)
            if c_t < best_c:
                best_c, best_h = c_t, h_t

    # Phase 7: Final SLSQP squeeze
    if time.time() < deadline - 5:
        h_f, c_f = slsqp_optimize(best_h, best_n, deadline - 3, alpha=10000000, maxiter=3000)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    # Phase 8: Final fine pair search
    if time.time() < deadline - 1:
        best_h, best_c = pair_local_search(best_h, best_n, deadline - 1, radius=40)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
