def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

def smooth_obj_and_grad(x, n, alpha):
    """Objective and gradient for log-sum-exp approximation"""
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()

def compute_delta_swap(h, i, j, n):
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0
    k1 = n-1 + (j-i)
    if 0 <= k1 < 2*n-1:
        delta[k1] += 1.0
    k2 = n-1 + (i-j)
    if 0 <= k2 < 2*n-1:
        delta[k2] += 1.0
    return delta

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0].astype(int))
    zeros_list = list(np.where(h == 0)[0].astype(int))
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try to find any improving swap
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                if time.time() > deadline:
                    break
                delta = compute_delta_swap(h, i, j, n)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=10000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    
    T = 0.02
    cool = 0.99999
    scale = 5000.0
    reheat_interval = 20000
    reheat_T = 0.01
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it > 0 and it % reheat_interval == 0 and T < reheat_T * 0.5:
            T = reheat_T
            cool = 0.999995
        
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        
        delta = compute_delta_swap(h, i, j, n)
        nc = (corr + delta).max() * dx
        
        accept = False
        if nc <= cur_c:
            accept = True
        elif T > 1e-14:
            ratio = scale * (cur_c - nc) / T
            if ratio < 50 and rng.random() < np.exp(-ratio):
                accept = True
        
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

    m_range = np.arange(-(n-1), n)

    imn = i - m_range
