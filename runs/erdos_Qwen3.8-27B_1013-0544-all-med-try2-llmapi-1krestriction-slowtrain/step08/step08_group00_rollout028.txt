        diff=target-h.sum()

            if step<=0: break

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = h < 1.0
            if not mask.any():
                break
            num = int(np.count_nonzero(mask))
            step = min(float((1.0 - h[mask]).max()), diff / num)
            if step <= 0.0:
                break
            h[mask] += step
        else:
            mask = h > 0.0
            if not mask.any():
                break
            num = int(np.count_nonzero(mask))
            step = min(float(h[mask].min()), -diff / num)
            if step <= 0.0:
                break
            h[mask] -= step
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            i = int(np.argmax(1.0 - h))
            add = min(1.0 - h[i], diff)
            if add <= 0.0:
                break
            h[i] += add
        else:
            i = int(np.argmax(h))
            add = -min(h[i], -diff)
            if add >= 0.0:
                break
            h[i] += add
    return h


def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h):
    return float(np.max(compute_c5_vec(h)))


def fft_convolve(a, b):
    need = len(a) + len(b) - 1
    f = 1
    while f < need:
        f <<= 1
    fa = np.fft.rfft(a, f)
    fb = np.fft.rfft(b, f)
    return np.fft.irfft(fa * fb, f)[:need]


def gradient_for_weights(h, n, w, mu):
    w = np.asarray(w, dtype=float)
    ws = float(w.sum())
    if ws == 0.0:
        return np.zeros(n)
    w = w / ws
    center = n - 1
    dx = 2.0 / float(n)
    conv1 = fft_convolve(1.0 - h, w)
    conv2 = fft_convolve(h, w[::-1])
    g = dx * (conv1[center:center + n] - conv2[center:center + n])
    if mu:
        g += 2.0 * mu * (h.sum() - 0.5 * n)
    return g


def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x)
    m = float(c.max())
    sh = alpha * (c - m)
    sh = sh - sh.max()
    val = m + (float(sh.max()) + np.log(np.sum(np.exp(sh)))) / alpha
    val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)


def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x)
    m = float(c.max())
    sh = alpha * (c - m)
    sh = sh - sh.max()
    p = np.exp(sh)
    p /= p.sum()
    return gradient_for_weights(x, n, p, mu)


def weighted_obj(x, n, active, mu):
    c = compute_c5_vec(x)
    return float(np.mean(c[active]) + mu * (x.sum() - 0.5 * n) ** 2)


def weighted_grad(x, n, active, mu):
    m = 2 * n - 1
    w = np.zeros(m)
    w[active] = 1.0 / float(len(active))
    return gradient_for_weights(x, n, w, mu)


def sum_eq(x):
    return float(x.sum() - 0.5 * len(x))


def sum_eq_jac(x):
    return np.ones(len(x))


def lbfgs_smooth(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    alphas = (100.0, 1000.0, 10000.0, 100000.0)
    mu = 1.0
    for alpha in alphas:
        if time.time() > deadline - 4.0:
            break
        try:
            res = minimize(
                lse_obj, h, jac=lse_grad, args=(n, alpha, mu),
                method="L-BFGS-B", bounds=bounds,
                options={"maxiter": 70, "maxfun": 220, "ftol": 1e-12, "gtol": 1e-8}
            )
            if res is not None and np.all(np.isfinite(res.x)):
                hn = project_feasible(res.x, n)
                cn = compute_c5(hn)
                if cn < best_c - 1e-15:
                    best_c = cn
                    best_h = hn
        except Exception:
            pass
    hc = best_h.copy()
    cc = compute_c5_vec(hc)
    return hc, cc, float(cc.max() if best_c <= cc.max() + 1e-12 else best_c)


def active_slsqp(h, c, deadline, active_tol=5e-5, max_active=220):

def apply_move(c, h, src, dst, d, dx, center, n):
    x = d * dx
    s = center + src
    c[s:s - n:-1] -= x * (1.0 - h)
    c[center - src:center - src + n] += x * h
    t = center + dst
    c[t:t - n:-1] += x * (1.0 - h)
    c[center - dst:center - dst + n] -= x * h
    c[center] -= 2.0 * x * d
    c[center + src - dst] += x * d
    c[center + dst - src] += x * d


def step_list_for(dmax):
    raw = []
    for f in (1.0, 0.4, 0.15, 0.05, 0.012, 0.003, 0.0007):
        d = f * dmax
        if d > 1e-8:
            raw.append(d)
    for av in (0.02, 0.005, 0.001, 0.0002):
        if av < dmax:
            raw.append(av)
    raw.sort()
    out = []
    for d in raw:
        if not out or d > out[-1] * 1.25:
            out.append(d)
    if len(out) > 9:
        out = out[:9]
    return out

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
