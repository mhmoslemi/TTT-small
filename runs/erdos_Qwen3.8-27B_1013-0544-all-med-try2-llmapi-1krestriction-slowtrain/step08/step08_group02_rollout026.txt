            if tot < 1e-14: break
            h += (diff / tot) * h

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    """Continuous optimization using L-BFGS-B with smooth max."""
    best_c = compute_c5(h, n)

            if tot < 1e-14: break
