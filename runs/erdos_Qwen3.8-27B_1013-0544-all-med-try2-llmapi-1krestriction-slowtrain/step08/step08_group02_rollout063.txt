def eval_swap(c, h, i, j, n):
    """Compute change in correlation c when swapping h[i]=1 to 0 and h[j]=0 to 1."""
    # m = 0, ..., 2n-2
    # The index into h that corresponds to position k in the sum for c[m] is: i+m-n+1
    # We need: 2*h[i+m-n+1] - 2*h[j+m-n+1] for the normal case
    # Special cases when i+m-n+1 == j or j+m-n+1 == i
    
    m = np.arange(2*n - 1)  # m = 0, ..., 2n-2
    
    # Index into h for the i-part: idx_i = i + m - n + 1
    idx_i = i + m - n + 1
    # Index into h for the j-part: idx_j = j + m - n + 1
    idx_j = j + m - n + 1
    
    # Normal formula: 2*h[idx_i] - 2*h[idx_j] (with out-of-range being 0)
    delta = np.zeros(2*n - 1)
    
    # i-part: valid when 0 <= idx_i < n
    valid_i = (idx_i >= 0) & (idx_i < n)
    delta[valid_i] += 2 * h[idx_i[valid_i]]
    
    # j-part: valid when 0 <= idx_j < n
    valid_j = (idx_j >= 0) & (idx_j < n)
    delta[valid_j] -= 2 * h[idx_j[valid_j]]
    
    # Special case 1: i + m - n + 1 == j → m = n - 1 + j - i
    m_special1 = n - 1 + j - i
    if 0 <= m_special1 < 2*n - 1:
        # In this case, the formula should be 1 - 2*h[2j-i] instead of 2*h[j] - 2*h[2j-i]
        # The normal formula gives: 2*h[j] - 2*h[2j-i] = 2*0 - 2*h[2j-i] = -2*h[2j-i]
        # The correct formula gives: 1 - 2*h[2j-i]
        # Difference: 1 - 2*h[2j-i] - (-2*h[2j-i]) = 1
        # So we add 1 to delta[m_special1]
        delta[m_special1] += 1
    
    # Special case 2: j + m - n + 1 == i → m = n - 1 + i - j
    m_special2 = n - 1 + i - j
    if 0 <= m_special2 < 2*n - 1:
        # The normal formula gives: 2*h[2i-j] - 2*h[i] = 2*h[2i-j] - 2
        # The correct formula gives: 2*h[2i-j] - 1
        # Difference: 2*h[2i-j] - 1 - (2*h[2i-j] - 2) = 1
        # So we add 1 to delta[m_special2]
        delta[m_special2] += 1
