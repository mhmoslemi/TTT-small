from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h.sum()
            if tot < 1e-14:
                break
            h[mask] = h[mask] * (1 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        # Remove the ones with lowest h values
        order = idx[np.argsort(h[idx])]
        h_bin[order[:cnt-target]] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        # Add the zeros with highest h values
        order = idx[np.argsort(-h[idx])]
        h_bin[order[:target-cnt]] = 1
    return h_bin

def full_swap_search(h_bin, n, deadline):
    """Try all single swaps and pick the best improving one"""
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        best_move_c = best_c
        best_i, best_j = -1, -1
        for i in ones:
            h_bin[i] = 0
            for j in zeros:
                if j == i:
                    continue
                h_bin[j] = 1
                c = compute_c5(h_bin, n)
                h_bin[j] = 0
                if c < best_move_c - 1e-15:
                    best_move_c = c
                    best_i, best_j = i, j
            h_bin[i] = 1
            if time.time() > deadline:
                return best_h, best_c
        if best_i >= 0:
            ones = np.where(h_bin == 1)[0]
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            zeros = np.where(h_bin == 0)[0]
            best_c = best_move_c
            best_h = h_bin.copy()
            improved = True
    return best_h, best_c

For binary h with set S = {i : h[i]=1}, |S| = n/2:

This is related to the concept of a "Bose-Mesner algebra" or "association scheme" in combinatorics. Sets that maximize minimum intersection with their shifts are related to difference sets and other combinatorial designs.

Wait, is the objective convex? max_k f_k(h) where f_k(h) = ∫h(x)(1-h(x+k))dx. 
f_k(h) = ∫h(x)dx - ∫h(x)h(x+k)dx
The first term is linear. The second term ∫h(x)h(x+k)dx is a quadratic form. Is it convex?

The quadratic form is ∫h(x)h(x+k)dx, which can be written as h^T A_k h where A_k is the correlation matrix. Is A_k positive semidefinite? 

A_k[i,j] = 1 if j-i=k (or i-j=k depending on convention), 0 otherwise. This is not symmetric in general, so it's not a standard quadratic form.

Actually, ∫h(x)h(x+k)dx = sum_i h[i]*h[i+k] for valid i. This is not a standard quadratic form h^T Q h (which would be sum_{i,j} h[i] Q[i,j] h[j]). Instead it's sum_i h[i] h[i+k], which is like a "shifted" inner product.

We can write it as: h^T S_k h where S_k has 1s on the k-th superdiagonal. But S_k is not symmetric, so h^T S_k h = h^T ((S_k + S_k^T)/2) h (the antisymmetric part cancels).

(S_k + S_k^T)/2 has 1/2 on the k-th superdiagonal and subdiagonal. This is like a banded matrix with 1/2 on the k-th off-diagonals.

Is this PSD? For k=1, it's (1/2) * (adjacency matrix of a path graph). The eigenvalues of the path adjacency matrix are 2cos(jπ/(n+1)) for j=1,...,n. The smallest eigenvalue is -2cos(π/(n+1)) ≈ -2 for large n. So (1/2)*(-2) = -1. Not PSD!

So the quadratic form is indefinite, meaning the objective is not convex. This means the optimum is NOT necessarily at a vertex of the constraint polytope.

def continuous_refine(h, n, deadline):
    """Use L-BFGS-B to refine continuous solution"""
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        return float(c.max())
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(c)
        k = k_max - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] += (1.0 - x[k:])
            g[k:] += (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += (1.0 - x[:nv])
            g[:nv] += (-x[kk:])
        return g

def swap_block(h_bin, n, rng):
    """Move a block of consecutive 1s to a random location"""
    h_new = h_bin.copy()
    # Find a block of 1s of length 2-5
    ones = np.where(h_bin == 1)[0]
    if len(ones) < 5:
        return h_new
    # Pick a starting position in the 1s
    start = ones[rng.randint(len(ones))]
    # Extend the block
    length = rng.randint(2, 6)
    end = start
    while end + 1 < n and h_bin[end + 1] == 1 and end - start + 1 < length:
        end += 1
    block_len = end - start + 1
    if block_len < 2:
        return h_new
    # Clear the block
    h_new[start:end+1] = 0
    # Find a random empty location
    zeros = np.where(h_new == 0)[0]
    zeros = zeros[(zeros >= end + block_len) | (zeros + block_len <= start)]  # avoid overlap
    # Actually just pick any zero position far enough from the block
    candidates = np.where(h_new == 0)[0]
    if len(candidates) < block_len:
        return h_new
    # Pick a random starting position for the new block
    valid_starts = candidates[candidates + block_len - 1 < n]
    if len(valid_starts) == 0:
        return h_new
    new_start = valid_starts[rng.randint(len(valid_starts))]
    # Check that the destination is all zeros
    if h_new[new_start:new_start+block_len].any():
        return h_new
    h_new[new_start:new_start+block_len] = 1
    return h_new
