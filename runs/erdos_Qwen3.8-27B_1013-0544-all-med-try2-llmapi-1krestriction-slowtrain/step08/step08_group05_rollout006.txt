def project_box_sum(y,target):
    y=np.asarray(y,dtype=float)
    lo = -float(np.max(y)) - 1.0
    hi = 1.0 - float(np.min(y))
    if hi < lo: hi=lo
    for _ in range(70):
       mid=0.5*(lo+hi)
       s=float(np.sum(np.clip(y+mid,0.0,1.0)))
       if s > target: hi=mid

    x=np.clip(y+0.5*(lo+hi),0.0,1.0)

       else:

          x[j] += diff
       x=np.clip(x,0.0,1.0)

def soft_lse_grad(x,n,beta):
    raw=raw_corr(x,n)
    sc=raw*(2.0/n)
    m=float(sc.max())
    z=np.exp(beta*(sc-m))
    zsum=float(z.sum())
    w=z/zsum
    dx=2.0/n
    g=np.zeros(n,float)
    tol=1e-13
    c=n-1
    w0=w[c]
    if abs(w0)>tol:
       g += (1.0-2.0*x)*w0
    for d in range(1,n):
       wp=w[c+d]
       if abs(wp)>tol:
          m=n-d
          g[:m] -= wp*x[d:]
          g[d:] += wp*(1.0-x[:m])
       wn=w[c-d]
       if abs(wn)>tol:
          m=n-d
          g[d:] -= wn*x[:m]
          g[:m] += wn*(1.0-x[d:])
    return g * (beta*dx)

            if h[i] > 0.5: continue

import numpy as np
import time
from scipy.optimize import minimize


def raw_corr(h, n):
    h = np.asarray(h, dtype=float)
    return np.ascontiguousarray(np.correlate(h, 1.0 - h, mode='full'), dtype=float)


def score(h, n):
    return float(np.max(raw_corr(h, n) * (2.0 / n)))


def project_box_sum(y, target):
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-15:
        fr = np.where((x > 1e-14) & (x < 1.0 - 1e-14))[0]
        if fr.size:
            x[fr] += diff / float(fr.size)
        else:
            j = int(np.argmax(x)) if diff > 0 else int(np.argmin(x))
            x[j] += diff
        x = np.clip(x, 0.0, 1.0)
    return x


def make_bounds(n):
    return [(0.0, 1.0)] * n


def resample(h, n):
    h = np.asarray(h, dtype=float)
    m = len(h)
    if m == n:
        return h.copy()
    if m == 1:
        return np.full(n, float(h[0]))
    x0 = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return np.interp(xn, x0, h)


def binarize(x, n):
    x = np.asarray(x, dtype=float)
    t = n // 2
    out = np.zeros(n, dtype=float)
    if t > 0:
        idx = np.argsort(x, kind='stable')[::-1]
        out[idx[:t]] = 1.0
    return out


def random_start(n, seed):
    return np.random.RandomState(seed).rand(n)


def perturb_start(h0, n, seed, amount):
    rng = np.random.RandomState(seed)
    if h0 is None:
        x = rng.rand(n)
    else:
        x = resample(h0, n)
    return np.clip(x + amount * rng.randn(n), 0.0, 1.0)


def quad_start(n, seed):
    i = np.arange(n)
    p = max(8, n // 4)
    x = (((i * i + seed) % p) * 2.0 < p).astype(float)
    return x + 0.01 * np.random.RandomState(seed).rand(n)


def build_D(h, n):
    h = np.asarray(h, dtype=float)
    c = n - 1
    D = np.zeros((n, 2 * n - 1), dtype=float)
    for d in range(1, n):
        pos = c + d
        neg = c - d
        m = n - d
        D[d:, pos] += 1.0 - h[:m]
        D[:m, pos] -= h[d:]
        D[:m, neg] += 1.0 - h[d:]
        D[d:, neg] -= h[:m]
    return D


def update_D_flip(D, p, delta, n):
    c = n - 1
    val = -delta
    for d in range(1, p + 1):
        D[p - d, c + d] += val
        D[p - d, c - d] += val
    for d in range(1, n - p):
        D[p + d, c + d] += val
        D[p + d, c - d] += val


def binary_greedy(start, n, deadline, seed=1, chunk=200):
    start = resample(start, n)
    h = binarize(start, n)
    corr = raw_corr(h, n).astype(float)
    D = build_D(h, n)
    cur = float(corr.max())
    best_h = h.copy()
    best_cur = cur
    zeros = np.where(h < 0.5)[0].tolist()
    ones = np.where(h >= 0.5)[0].tolist()
    c = n - 1
    accept_count = 0

    while time.time() < deadline:
        zlist = zeros[:]
        improved = False
        for i in zlist:
            if time.time() > deadline:
                break
            if h[i] > 0.5:
                continue
            base = corr + D[i]
            best_c = cur
            best_j = -1
            for st in range(0, len(ones), chunk):
                J = np.asarray(ones[st:st + chunk], dtype=int)
                if J.size == 0:
                    continue
                J = J[(h[J] == 1.0) & (J != i)]
                if J.size == 0:
                    continue
                mat = base[None, :] - D[J]
                m0 = mat.max(axis=1)
                idx1 = (J - i) + c
                idx2 = (i - J) + c
                rows = np.arange(J.size)
                act = np.maximum(np.maximum(m0, mat[rows, idx1] + 1.0),
                                 mat[rows, idx2] + 1.0)
                p = int(np.argmin(act))
                if act[p] < best_c - 1e-10:
                    best_c = float(act[p])
                    best_j = int(J[p])

            if best_j >= 0:
                idx1 = (best_j - i) + c
                idx2 = (i - best_j) + c
                delta = D[i].copy() - D[best_j].copy()
                corr += delta
                corr[idx1] += 1.0
                corr[idx2] += 1.0
                update_D_flip(D, i, 1.0, n)
                update_D_flip(D, best_j, -1.0, n)

                h[i] = 1.0
                h[best_j] = 0.0
                try:
                    zeros.remove(i)
                except ValueError:
                    pass
                try:
                    ones.remove(best_j)
                except ValueError:
                    pass
                ones.append(i)
                zeros.append(best_j)

                cur = best_c
                improved = True
                accept_count += 1
                if cur < best_cur:
                    best_cur = cur
                    best_h = h.copy()

                if accept_count % 2000 == 0:
                    corr = raw_corr(h, n).astype(float)
                    D = build_D(h, n)
                    cur = float(corr.max())
                    if cur < best_cur:
                        best_cur = cur
                        best_h = h.copy()

        if not improved:
            break

    return best_h, best_cur * (2.0 / n)


def soft_lse_value(x, n, beta):
    raw = raw_corr(x, n)
    sc = raw * (2.0 / n)
    m = float(sc.max())
    if beta <= 1e-8:
        return m
    z = np.exp(beta * (sc - m))
    return m + float(np.log(z.sum()) / beta)


def soft_lse_grad(x, n, beta):
    raw = raw_corr(x, n)
    sc = raw * (2.0 / n)
    m = float(sc.max())
    z = np.exp(beta * (sc - m))
    w = z / float(z.sum())
    g = np.zeros(n, dtype=float)
    tol = 1e-13
    c = n - 1
    w0 = float(w[c])
    if abs(w0) > tol:
        g += w0 * (1.0 - 2.0 * x)
    for d in range(1, n):
        m = n - d
        wp = float(w[c + d])
        if abs(wp) > tol:
            g[:m] -= wp * x[d:]
            g[d:] += wp * (1.0 - x[:m])
        wn = float(w[c - d])
        if abs(wn) > tol:
            g[d:] -= wn * x[:m]
            g[:m] += wn * (1.0 - x[d:])
    return g * (beta * (2.0 / n))


def penal_value(x, n, beta, lam):
    v = soft_lse_value(x, n, beta)
    s = float(np.sum(x) - 0.5 * n)
    return v + lam * s * s


def penal_grad(x, n, beta, lam):
    g = soft_lse_grad(x, n, beta)
    s = float(np.sum(x) - 0.5 * n)
    return g + 2.0 * lam * s


def sum_eq(x, target):
    return float(np.sum(x) - target)


def solve_binary_candidate(start, n, deadline, seed=1):
    if n % 2 != 0:
        n -= 1
    if n < 20:
        n = 20
    h = binarize(resample(start, n), n)
    if time.time() < deadline - 10:
        h, c = binary_greedy(start, n, deadline, seed)
    else:
        c = score(h, n)
    return h, float(c)


def continuous_opt(h, n, deadline, seed=1):
    target = 0.5 * n
    best = project_box_sum(h, target)
    best_c = score(best, n)
    bounds = make_bounds(n)
    for beta in (50.0, 200.0, 800.0, 3000.0):
        rem = deadline - time.time()
        if rem < 25:
            break
        iters = 160 if rem > 150 else 70
        lam = 50.0 if beta < 1000.0 else 200.0
        x = best
        try:
            res = minimize(penal_value, best, method='L-BFGS-B',
                           jac=penal_grad, args=(n, beta, lam),
                           bounds=bounds,
                           options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-12})
            if hasattr(res, 'x'):
                x = project_box_sum(res.x, target)
                if not np.all(np.isfinite(x)):
                    x = best
        except Exception:
            pass
        c = score(x, n)
        if c < best_c:
            best = x.copy()
            best_c = c
    return best, float(best_c)


def slsqp_polish(h, n, deadline, seed=1):
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 70 or n > 350:
        x = project_box_sum(h, target)
        return x, float(score(x, n))
    best = project_box_sum(h, target)
    best_c = score(best, n)
    bounds = make_bounds(n)
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (target,)}]
    betas = (500.0, 2000.0) if rem > 300 else (1000.0,)
    for beta in betas:
        rem = deadline - time.time()
        if rem < 50:
            break
        iters = 100 if rem > 250 else 50
        try:
            res = minimize(soft_lse_value, best, method='SLSQP',
                           jac=soft_lse_grad, args=(n, beta),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': iters, 'ftol': 1e-15})
            x = project_box_sum(res.x, target)
            if not np.all(np.isfinite(x)):
                continue
            c = score(x, n)
            if c < best_c:
                best = x.copy()
                best_c = c
        except Exception:
            break
    return best, float(best_c)


def solve_continuous_candidate(h, n, deadline, seed=1):
    if time.time() > deadline - 25:
        x = project_box_sum(h, 0.5 * n)
        return x, float(score(x, n))
    x, c = continuous_opt(h, n, deadline, seed)
    x = project_box_sum(x, 0.5 * n)
    c = score(x, n)
    return x, float(c)


def update_state(state, h, c, n):
    if h is None or n is None:
        return state
    try:
        c = float(c)
        if not np.isfinite(c):
            return state
    except Exception:
        return state
    if c < state['c'] - 1e-15:
        state['h'] = np.asarray(h, dtype=float).copy()
        state['c'] = c
        state['n'] = int(n)
    return state

        if quick_improved and time.time() < m1a - 5:
            h, c = solve_binary_candidate(state['h'], state['n'], m1a, seed)
            state = update_state(state, h, c, state['n'])
            if time.time() < m1b - 15:
                h, c = solve_binary_candidate(start0, n0, m1b, seed + 1)
                state = update_state(state, h, c, n0)
        else:
            h, c = solve_binary_candidate(start0, n0, m1b, seed)
            state = update_state(state, h, c, n0)

    if state['h'] is not None and time.time() < m2 - 20:
        h, c = solve_continuous_candidate(state['h'], state['n'], m2, seed + 2)
        state = update_state(state, h, c, state['n'])

    nwork = state['n'] if state['h'] is not None else n0
    nwork -= nwork % 2
    if nwork < 20:
        nwork = n0

    if state['h'] is not None and time.time() < m3 - 20:
        rem = m3 - time.time()
        starts = []
        starts.append(state['h'])
        if base is not None:
            starts.append(perturb_start(base, nwork, seed + 10, 0.35))
        starts.append(perturb_start(state['h'], nwork, seed + 11, 0.50))
        starts.append(random_start(nwork, seed + 12))
        starts.append(quad_start(nwork, seed + 13))
        for idx, st in enumerate(starts):
            dl_i = min(m3, time.time() + max(15.0, rem / float(len(starts) - idx)))
            if time.time() < dl_i - 5:
                h, c = solve_binary_candidate(st, nwork, dl_i, seed + 20 + idx)
                state = update_state(state, h, c, hlen if False else (len(h) if len(h) % 2 == 0 else nwork))

    if state['h'] is not None and time.time() < m4 - 20:
        rem = m4 - time.time()
        sizes = []
        for k in (-48, -32, -16, 16, 32, 48):
            nn = state['n'] + k
            if nn % 2 == 0 and 120 <= nn <= 400:
                sizes.append(nn)
        sizes = list(dict.fromkeys(sizes))
        for idx, nn in enumerate(sizes):
            dl_i = min(m4, time.time() + max(15.0, rem / float(max(1, len(sizes) - idx))))
            if time.time() < dl_i - 5:
                st = resample(state['h'], nn)
                h, c = solve_binary_candidate(st, nn, dl_i, seed + 40 + idx)
                state = update_state(state, h, c, nn)

    end_cont = deadline - 30.0
    if state['h'] is not None and time.time() < end_cont - 80.0:
        h, c = solve_continuous_candidate(state['h'], state['n'], end_cont, seed + 60)
        state = update_state(state, h, c, state['n'])

    end_sls = deadline - 10.0
    if state['h'] is not None and time.time() < end_sls - 40.0:
        h, c = slsqp_polish(state['h'], state['n'], end_sls, seed + 70)
        state = update_state(state, h, c, state['n'])

    end_bin = deadline - 4.0
    if state['h'] is not None and time.time() < end_bin - 5.0:
        h0 = binarize(resample(state['h'], state['n']), state['n'])
        h, c = solve_binary_candidate(h0, state['n'], end_bin, seed + 80)
        state = update_state(state, h, c, state['n'])

    if state['h'] is None:
        h = np.full(n0, 0.5)
        n = n0
    else:
        h = state['h']
        n = state['n']

    h = project_box_sum(h, 0.5 * n)
    c = score(h, n)
    state = update_state(state, project_box_sum(h, 0.5 * n), c, n)

    return state['h'], float(state['c']), int(state['n'])
