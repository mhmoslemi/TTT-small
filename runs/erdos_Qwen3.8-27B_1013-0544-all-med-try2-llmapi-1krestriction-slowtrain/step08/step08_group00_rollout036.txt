            grad[:nv] += wmi * (1.0 - x[k:])

            nv = n - kk
            grad[kk:] += wmi * (1.0 - x[:nv])
            grad[:nv] -= wmi * x[kk:]

def lbfgs_optimize(h, n, alpha, deadline, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, args=(n, alpha), jac=grad_func,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def coord_search(h, n, deadline, max_sweeps=20):

        improved = False
        for i in order:
            for d in deltas:

                    h_p = project_feasible(h, n)
                    if not np.allclose(h_p, h):

                        improved = True

                    else:

                        improved = True
                        break
                    else:

def active_lag_optimization(h, n, k_active=5, max_iters=100):
    best_c = compute_c5(h, n)
    for _ in range(max_iters):
        c = compute_c5_vec(h, n)
        # Find top k_active lags
        top_idx = np.argsort(c)[-k_active:]
        top_c = c[top_idx]
        target = top_c.min()  # We want all active lags to be at most their min
        
        # Build sensitivity matrix for active lags
        # S[j, m] = dc[m] / dh[j]
        dx = 2.0 / n
        S = np.zeros((n, len(top_idx)))
        for col, m_idx in enumerate(top_idx):
            k = m_idx - (n - 1)
            if k >= 0:
                nv = n - k
                S[:nv, col] += (1.0 - h[k:]) * dx
                S[k:, col] -= h[:nv] * dx
            else:
                kk = -k
                nv = n - kk
                S[kk:, col] += (1.0 - h[:nv]) * dx
                S[:nv, col] -= h[kk:] * dx
        
        # We want: S @ delta = target - top_c (make all active lags equal to target)
        # But we also want sum(delta) = 0 (to maintain the constraint)
        # Add the constraint: sum(delta) = 0
        # Augmented system: [S; ones^T] @ delta = [target - top_c; 0]
        A = np.vstack([S, np.ones((1, n))])
        b = np.concatenate([target - top_c, [0.0]])
        
        # Solve least squares
        delta, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
        
        # Try different step sizes
        best_step = 0
        best_new_c = best_c
        for step in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:

                best_step = step

        h = project_feasible(h + best_step * delta, n)
