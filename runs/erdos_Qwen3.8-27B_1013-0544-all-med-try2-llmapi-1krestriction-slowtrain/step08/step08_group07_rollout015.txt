    delta[m1] -= (1.0 - h[i + k_range[m1]])
    
    m2 = (i - k_range >= 0) & (i - k_range < n)
    # p+k=i term (p=i-k): old = h[i-k]*(1-1) = 0, new = h_new[i-k]*(1-0) = h_new[i-k]
    vals = h[i - k_range[m2]].copy()
    special = m2 & (i - k_range == j)
    vals[special] = 1.0  # h_new[j] = 1
    delta[m2] += vals
    
    m3 = (j + k_range >= 0) & (j + k_range < n)
    # p=j term: old = 0*(1-h[j+k]) = 0, new = 1*(1-h_new[j+k])
    vals3 = h[j + k_range[m3]].copy()
    special3 = m3 & (j + k_range == i)
