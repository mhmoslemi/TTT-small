    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    return grad_smooth_max(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def slsqp_obj(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def slsqp_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, maxiter=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    schedule = [(5, 0.5), (20, 1.0), (100, 5.0), (500, 20.0), (2000, 100.0),
                (10000, 500.0), (50000, 2000.0), (200000, 10000.0),
                (1000000, 50000.0), (5000000, 200000.0), (20000000, 1000000.0)]
    for i, (alpha, mu) in enumerate(schedule):
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        frac = remaining / (len(schedule) - i)
        it = max(15, min(maxiter, int(frac * 0.6)))
        if it < 12:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': it, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def grad_pair_search(h, n, deadline, top_k=100, step_init=0.005):
    best_c = compute_c5(h, n)
    step = step_init
    alpha = 200
    for _ in range(300):
        if time.time() > deadline - 2:
            break
        g = grad_smooth_max(h, n, alpha)
        idx = np.argsort(g)
        improved = False
        steps = [step, step * 0.5, step * 0.25, step * 0.125]
        for d in steps:
            if time.time() > deadline - 2:
                break
            for a in range(min(top_k, n // 2)):
                if time.time() > deadline - 2:
                    break
                for b in range(max(0, n - top_k), n):
                    i, j = int(idx[a]), int(idx[b])
                    if i == j:
                        continue
                    if h[j] - d >= -1e-15 and h[i] + d <= 1.0 + 1e-15:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
        if not improved:
            step *= 0.5
            alpha = min(int(alpha * 2.5), 5000000)
            if step < 1e-6:
                break
    return h, best_c

def perturbation_restart(h, n, deadline, rng, max_restarts=4):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(max_restarts):
        if time.time() > deadline - 15:
            break
        scale = 0.03 * (1.0 - r / max_restarts)
        h_pert = h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)
        alloc = min(50, deadline - time.time() - 10)
        if alloc < 20:
            break
        h_new, c_new = lbfgs_phase(h_pert, n, time.time() + alloc, maxiter=80)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new
            h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS with adaptive alpha
    t1 = min(t0 + 280, deadline - 450)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, maxiter=200)

    # Phase 2: SLSQP with hard constraint
    if time.time() < deadline - 300:
        t2 = min(time.time() + 100, deadline - 300)
        h_s, c_s = slsqp_phase(best_h, n_init, t2, alpha=200000, maxiter=1000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 3: Gradient-guided pair search
    if time.time() < deadline - 250:
        t3 = min(time.time() + 130, deadline - 250)
        best_h, best_c = grad_pair_search(best_h, n_init, t3, top_k=80, step_init=0.004)

    # Phase 4: Second L-BFGS round
    if time.time() < deadline - 200:
        t4 = min(time.time() + 100, deadline - 200)
        best_h, best_c = lbfgs_phase(best_h, n_init, t4, maxiter=150)

    # Phase 5: SLSQP again
    if time.time() < deadline - 150:
        t5 = min(time.time() + 60, deadline - 150)
        h_s, c_s = slsqp_phase(best_h, n_init, t5, alpha=2000000, maxiter=800)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 6: Perturbation restarts
    if time.time() < deadline - 120:
        t6 = min(time.time() + 100, deadline - 120)
        best_h, best_c = perturbation_restart(best_h, n_init, t6, rng)

    # Phase 7: Fine pair search
    if time.time() < deadline - 40:
        t7 = min(time.time() + 80, deadline - 40)
        best_h, best_c = grad_pair_search(best_h, n_init, t7, top_k=60, step_init=0.001)

    # Phase 8: Try alternative n values
    for n_try in [250, 350, 400, 200]:
        if time.time() > deadline - 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(70, deadline - time.time() - 25)
        if alloc < 25:
            break
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6, maxiter=120)
        if time.time() < deadline - 20:
            h_t, c_t = grad_pair_search(h_t, n_try, min(time.time() + 25, deadline - 20),
                                         top_k=60, step_init=0.002)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 9: Final polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = lbfgs_phase(best_h, nf, deadline - 5, maxiter=50)
        if time.time() < deadline - 2:
            best_h, best_c = grad_pair_search(best_h, nf, deadline - 1, top_k=50, step_init=0.001)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
