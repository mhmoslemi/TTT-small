def solve_1d_subproblem(A, B, t_lo=0.0, t_hi=1.0):
    """Minimize max_m (A[m] + B[m]*t) for t in [t_lo, t_hi]."""
    def f(t):
        return np.max(A + B * t)
    
    def subgrad_range(t):
        vals = A + B * t
        mx = vals.max()
        active = np.where(np.abs(vals - mx) < 1e-14)[0]
        return B[active].min(), B[active].max()
    
    # Check endpoints
    f_lo = f(t_lo)
    f_hi = f(t_hi)
    
    # If decreasing at left endpoint, minimum is to the right
    sg_lo_min, sg_lo_max = subgrad_range(t_lo)
    if sg_lo_min > 0:
        return t_lo  # f is increasing everywhere near t_lo
    
    # If increasing at right endpoint, minimum is to the left  
    sg_hi_min, sg_hi_max = subgrad_range(t_hi)
    if sg_hi_max < 0:
        return t_hi  # f is decreasing everywhere near t_hi
    
    # Binary search for where 0 crosses the subgradient
    lo, hi = t_lo, t_hi
    for _ in range(50):
        mid = (lo + hi) / 2
        sg_min, sg_max = subgrad_range(mid)
        if sg_min > 0:
            lo = mid  # f increasing at mid, minimum is to the left
        elif sg_max < 0:
            hi = mid  # f decreasing at mid, minimum is to the right
        else:
            return mid  # 0 is in subgradient at mid
    
    return (lo + hi) / 2

def project_feasible(h, n):

        if abs(diff) < 1e-14:

            if tot < 1e-14:

def solve_1d_convex_pl(A, B):
    """Solve min_{t in [0,1]} max_m (A[m] + B[m]*t)."""
    # Evaluate at endpoints
    f_lo = np.max(A)
    f_hi = np.max(A + B)
    
    # Check subgradient at endpoints
    active_lo = np.where(A == f_lo)[0]
    if len(active_lo) > 0 and B[active_lo].min() > 1e-14:
        return 0.0  # increasing at t=0
    
    active_hi = np.where(A + B == f_hi)[0]
    if len(active_hi) > 0 and B[active_hi].max() < -1e-14:
        return 1.0  # decreasing at t=1
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = A + B * mid
        f_mid = vals.max()
        active = np.where(np.abs(vals - f_mid) < 1e-12 * max(1.0, f_mid))[0]
        sg_min = B[active].min()
        sg_max = B[active].max()
        if sg_min > 1e-14:
            hi = mid
        elif sg_max < -1e-14:
            lo = mid
        else:

    f_1 = np.max(A + B)
    if f_t >= min(f_0, f_1) - 1e-15:
        if f_0 <= f_1:
            return 0.0
        else:
            return 1.0
    return t

            break

            sh2_lo = max(0, j)
            sh2_hi = min(n_corr - 1, n - 1 + j)
            if sh2_lo <= sh2_hi:

                B[sh2_lo:sh2_hi+1] += (-h[j - shifts2]) * dx
            
            # A_m = c[m] - B_m * h[j]
            A = c - B * h[j]
            
            # Solve 1D subproblem
            t_star = solve_1d_convex_pl(A, B)
            
            if abs(t_star - h[j]) > 1e-14:
                h[j] = t_star
                improved = True

            break

            if c_current < best_c:
                best_c = c_current

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlate(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_correlate(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_correlate(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def slsqp_con_fun(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp_obj(h, n, alpha):
    c = compute_correlate(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def lbfgs_phase(h, n, deadline, alphas=None, max_iters=200):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000, 500000, 2000000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        mu = max(5.0, alpha * 0.005)
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=5000000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def coordinate_descent(h, n, deadline, max_sweeps=2000, n_grid=200):
    """Coordinate descent for the Erdős overlap problem."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h_rev = h[::-1].copy()
    
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Randomize coordinate order every 10 sweeps
        if sweep % 10 == 0:
            order = rng.permutation(n)
        
        c = np.correlate(h, 1 - h, mode='full') * dx
        improved = False
        
        for j in order:
            # Compute B (coefficient of h[j] in each c[m])
            B = np.zeros(n_corr)
            B[n-1-j : 2*n-1-j] += (1.0 - h) * dx
            B[j : n+j] += -h_rev * dx
            # Note: for m = n-1 (shift 0), the actual dependence is quadratic
            # B[n-1] is the "linear" coefficient but we'll handle it separately
            
            # Solve 1D subproblem: min_t max(c + B*(t-h[j]) for all m, 
            #   except m=n-1 where it's quadratic)
            h_j = h[j]

            # g_lin(t) = max_{m≠n-1} (c[m] + B[m]*(t - h_j))
            # = max_{m≠n-1} ((c[m] - B[m]*h_j) + B[m]*t)
            A = c - B * h_j  # A[m] = c[m] - B[m]*h_j
            A_lin = A.copy()
            A_lin[n-1] = -np.inf  # exclude k=0 shift
            
            # Bisection on the linear part
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = A_lin + B * mid  # B[n-1] doesn't matter since A_lin[n-1]=-inf
                vals[n-1] = -np.inf
                f_mid = vals.max()

                active = active[active != n-1]
                if len(active) == 0:
                    break
                sg = B[active]
                if sg.min() > 1e-13:
                    hi = mid
                elif sg.max() < -1e-13:
                    lo = mid
                else:
                    break
            t_lin
