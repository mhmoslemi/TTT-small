def weighted_grad_c5(w,h,n):
  dx=2.0/n
  b=1.0-h
  M=next_pow2(3*n-2)
  Fh=np.fft.rfft(h,M)

def project_box_sum(x,n):
  x=np.asarray(x,dtype=float)
  S=0.5*float(n)
  if x.size != n:
      x=x.reshape(n) # not safe; assume
  lo=float(np.min(x))-1.0
  hi=float(np.max(x))+1.0
  for _ in range(70):
     tau=0.5*(lo+hi)
     z=np.clip(x-tau,0.0,1.0)
     if float(z.sum()) > S: lo=tau
     else: hi=tau
  tau=0.5*(lo+hi)
  z=np.clip(x-tau,0.0,1.0)
  diff=S-float(z.sum())
  if abs(diff)>1e-13:
     if diff>0:
        idxs=np.flatnonzero(z < 1.0 - 1e-12)
     else:
        idxs=np.flatnonzero(z > 1e-12)
     for idx in idxs[:200]:
        if diff>0:
           room=1.0-z[idx]
           adj=min(diff,room)
           if adj<=0: continue
           z[idx]+=adj; diff-=adj
        else:
           room=z[idx]
           adj=min(-diff,room)
           if adj<=0: continue
           z[idx]-=adj; diff+=adj
        if abs(diff)<=1e-14: break
  return z

import numpy as np
import time
import math
from scipy.optimize import minimize

_MIN_DEADLINE = -1.0

def _timeout_cb(x):
    if _MIN_DEADLINE > 0.0 and time.time() > _MIN_DEADLINE:
        raise RuntimeError("timeout")

def next_pow2(m):
    m = int(m)
    return 1 << max(1, (m - 1)).bit_length()

def project_box_sum(x, n):
    x = np.array(x, dtype=float)
    S = 0.5 * n
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        t = 0.5 * (lo + hi)
        z = np.clip(x - t, 0.0, 1.0)
        if float(z.sum()) > S:
            lo = t
        else:
            hi = t
    z = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    d = S - float(z.sum())
    if abs(d) > 1e-13:
        idx = np.flatnonzero(z < 1.0 - 1e-12) if d > 0 else np.flatnonzero(z > 1e-12)
        for k in idx[:200]:
            if d > 0:
                a = min(d, 1.0 - z[k])
                if a > 0:

            else:

                if a > 0:
                    z[k] -= a
                    d += a
            if abs(d) < 1e-14:

    return z

def repair_feasible(x, n):
    x = np.asarray(x, dtype=float)
    np.clip(x, 0.0, 1.0, out=x)
    d = 0.5 * n - float(x.sum())
    if abs(d) > 1e-12:
        if d > 0:
            k = int(np.argmax(1.0 - x))
            a = min(d, 1.0 - x[k])
            if a > 0:
                x[k] += a
        else:
            k = int(np.argmax(x))
            a = min(-d, x[k])
            if a > 0:
                x[k] -= a
    if abs(0.5 * n - float(x.sum())) > 1e-10:
        return project_box_sum(x, n)
    return x

def corr_c5_fft(h, n):
    b = 1.0 - h
    M = next_pow2(2 * n - 1)
    c = np.fft.irfft(np.fft.rfft(h, M) * np.fft.rfft(b[::-1], M), M)[:2 * n - 1]
    return c * (2.0 / n)

def eval_c5(h, n):
    return float(np.max(corr_c5_fft(h, n)))

def eval_c5_direct(h, n):
    if n > 4000:
        return eval_c5(h, n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))

def soft_value(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    mx = float(np.max(s))
    return m + (mx + float(np.log(np.sum(np.exp(s - mx))))) / alpha

def softmax_weights(c, alpha):
    L = len(c)
    if alpha <= 1e-8:
        return np.full(L, 1.0 / L)
    s = alpha * (c - float(np.max(c)))
    mx = float(np.max(s))
    e = np.exp(s - mx)
    t = float(e.sum())
    if t == 0.0 or not np.isfinite(t):
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / t

def weighted_grad_c5(w, h, n):
    dx = 2.0 / n
    b = 1.0 - h
    M = next_pow2(3 * n - 2)
    Fh = np.fft.rfft(h, M)
    Fb = np.fft.rfft(b, M)
    Fw = np.fft.rfft(w, M)
    Fwr = np.fft.rfft(w[::-1], M)
    upto = 3 * n - 2
    first = np.fft.irfft(Fw * Fb, M)[:upto]
    second = np.fft.irfft(Fwr * Fh, M)[:upto]
    s = n - 1
    return dx * (first[s:s + n] - second[s:s + n])

def _obj(x, n, alpha, mu):
    c = corr_c5_fft(x, n)
    s = float(np.sum(x)) - 0.5 * n
    return soft_value(c, alpha) + mu * s * s

def _jac(x, n, alpha, mu):
    c = corr_c5_fft(x, n)
    w = softmax_weights(c, alpha)
    g = weighted_grad_c5(w, x, n)
    s = float(np.sum(x)) - 0.5 * n
    if mu != 0.0:
        g += 2.0 * mu * s
    return g

def lbfgs_polish(h0, n, deadline):
    global _MIN_DEADLINE
    h = project_box_sum(h0, n)
    best = eval_c5(h, n)
    cwork = best
    _MIN_DEADLINE = max(0.0, deadline - 4.0)
    if deadline - time.time() < 15.0:
        return h, best
    if n < 800:
        alphas = [20, 100, 500, 2000, 10000, 50000, 200000]
    elif n < 1600:

    else:

    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 15.0:
            break
        mu = max(1.0, 0.02 * alpha)
        if n > 500:
            mu *= float(n) / 500.0
        mu = min(50000.0, mu)

        scale = min(1.0, max(0.2, rem / 100.0))

def pair_polish(h0, n, deadline):

                bt = (i, j, dd)
        if bt is None:

        i, j, dd = bt
        h = h.copy()
        h[i] += dd
        h[j] -= dd
        h = repair_feasible(h, n)
        best = bnd
        c = corr_c5_fft(h, n)

        if ct < best_c - 1e-12 or (abs(ct - best_c) <= 1e-12 and nt < best_n):
            best_h, best_c, best_n = ht, ct, nt
        if deadline - time.time() < 300.0:
            break
        alloc2 = min(120.0, deadline - time.time() - 320.0)
        if alloc2 > 50.0:
            hr = random_binary(nt, rng)
            ht2, ct2 = optimize_start(hr, nt, time.time() + alloc2)
            if ct2 < best_c - 1e-12:
                best_h, best_c, best_n = ht2, ct2, nt

    if deadline - time.time() > 25.0:
        hf, cf = optimize_start(best_h, best_n, deadline - 4.0)
        if cf < best_c - 1e-12:
            best_h, best_c = hf, cf

    best_h = project_box_sum(best_h, best_n)
    try:
        best_c = eval_c5_direct(best_h, best_n)
    except Exception:
        best_c = eval_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
