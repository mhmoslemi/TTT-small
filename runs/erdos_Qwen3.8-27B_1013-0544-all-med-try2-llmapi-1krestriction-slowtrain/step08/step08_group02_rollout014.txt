    s = alpha * (c - m)

        else:

            if tot < 1e-15:

        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, -diff_t, replace=False)] = 0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    tabu = {}
    tabu_len = 500
    for rnd in range(200):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        best_move_c = best_c
        best_i = best_j = -1
        n_try = min(40000, len(ones) * len(zeros))
        for _ in range(n_try):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            key = (i, j)
            if key in tabu and tabu[key] > rnd - tabu_len:
                continue
            h_bin[i] = 0
            h_bin[j] = 1
            c_new = compute_c5(h_bin, n)
            h_bin[i] = 1
            h_bin[j] = 0
            if c_new < best_move_c - 1e-15:
                best_move_c = c_new
                best_i, best_j = i, j
        if best_i >= 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            tabu[(best_i, best_j)] = rnd
            best_c = best_move_c
            best_h = h_bin.copy()
        else:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Multi-alpha L-BFGS-B
    t1 = min(t0 + 350, deadline - 350)
    alphas = [50, 200, 1000, 10000, 100000]
    for a in alphas:
        if time.time() > t1:
            break
        h_try = best_h + rng.normal(0, 0.01, n)
        h_try = project_feasible(h_try, n)
        h_new, c_new = lbfgs(h_try, n, t1, a, maxiter=30000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 2: High-alpha refinement
    if time.time() < deadline - 250:
        t2 = min(time.time() + 120, deadline - 200)
        h_new, c_new = lbfgs(best_h, n, t2, 2000000, maxiter=80000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 3: Subgradient polish
    if time.time() < deadline - 150:
        t3 = min(time.time() + 60, deadline - 120)
        h_new, c_new = subgrad_polish(best_h, n, t3, base_step=0.03)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 4: Binary local search + polish
    if time.time() < deadline - 80:
        t4 = min(time.time() + 50, deadline - 60)
        h_bin, c_bin = binary_local(best_h, n, t4, rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        if time.time() < deadline - 40:
            t4b = min(time.time() + 30, deadline - 30)
            h_new, c_new = lbfgs(best_h.copy(), n, t4b, 1000000, maxiter=50000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 5: Restart from perturbation
    for r in range(4):
        if time.time() > deadline - 30:
            break
        alloc = max(20, (deadline - time.time() - 20) / (4 - r))
        h_try = best_h + rng.normal(0, 0.015, n)
        h_try = project_feasible(h_try, n)
        h_new, c_new = lbfgs(h_try, n, time.time() + alloc * 0.6, 500000, maxiter=40000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 15:
            h_new2, c_new2 = subgrad_polish(best_h, n, time.time() + 10, base_step=0.02)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2
    # Final
    if time.time() < deadline - 3:
        h_new, c_new = lbfgs(best_h, n, deadline - 1, 5000000, maxiter=30000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
