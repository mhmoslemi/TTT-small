import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def nesterov_projected_grad(h, n, deadline, alphas=None, max_iters=2000):
    """Nesterov-accelerated projected gradient with path Following alpha."""
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        # Estimate Lipschitz constant
        h_test = h + 0.01 * np.random.randn(n)
        h_test = project_feasible(h_test, n)
        g1 = grad_smooth_max(h, n, alpha)
        g2 = grad_smooth_max(h_test, n, alpha)
        diff = np.linalg.norm(h - h_test)
        if diff > 1e-10:
            L = np.linalg.norm(g1 - g2) / diff + 1.0
        else:
            L = 1.0
        step = 1.0 / (alpha * L)
        
        y = h.copy()
        t = 1.0
        for it in range(max_iters):
            if time.time() > deadline:
                break
            g = grad_smooth_max(y, n, alpha)
            # Projected gradient step
            x_new = y - step * g
            x_new = project_feasible(x_new, n)
            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            beta = (t - 1.0) / t_new
            y = x_new + beta * (x_new - h)
            h = x_new.copy()
            t = t_new
            
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
        
        # After each alpha, do a few backtracking steps
        step = 1.0 / (alpha * L)
        for _ in range(10):
            if time.time() > deadline:
                break
            g = grad_smooth_max(best_h, n, alpha)
            x_new = best_h - step * g
            x_new = project_feasible(x_new, n)
            c = compute_c5(x_new, n)
            if c < best_c:
                best_c = c
                best_h = x_new
    
    return best_h, best_c

def lbfgs_smooth(h, n, deadline, alpha, max_iters=300):
    """Single L-BFGS-B run with given alpha."""
    dx = 2.0 / n
    
    def obj(x):
        return smooth_max(compute_c5_vec(x, n), alpha)
    
    def obj_grad(x):
        return grad_smooth_max(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, args=(), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': max_iters, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxfun': max_iters * 20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coordinate_descent(h, n, deadline, n_rounds=10):

            max_idx = np.argmax(c)
            k = max_idx - (n - 1)
            # The shift k is the bottleneck

            if k >= 0:
                # overlap = sum h[i+k]*(1-h[i]) for i=0..n-1-k

def golden_ratio_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def van_der_corput_start(n):
    h = np.zeros(n)
    half = n // 2
    # Van der Corput sequence (base 2)
    positions = set()
    for i in range(half):
        x = 0.0
        den = 1.0
        j = i
        while j > 0:
            x += (j % 2) / den
            j //= 2
            den *= 2.0
        pos = int(x * n) % n
        positions.add(pos)
    h[list(positions)] = 1.0
    return project_feasible(h, n)

def low_disc_start(n):
    """Low discrepancy using golden ratio in [0,1) scaled to [0,n)."""
    h = np.zeros(n)
    half = n // 2
    phi_inv = (np.sqrt(5.0) - 1.0) / 2.0
    positions = np.floor((np.arange(half) * phi_inv) % 1.0 * n).astype(int)
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 10):
        amp = rng.uniform(0.05, 0.3) / np.sqrt(freq)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def binary_random_start(n, rng):

def pair_polish_fast(h, n, deadline, radius=30):
    """Faster pair polish using bubble sort idea."""
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(0, n, 2):  # stride 2 for speed
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try i += d, j -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try i -= d, j += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def subgrad_max(h, n):
    """Subgradient of the true max (not smooth)."""
    c = compute_c5_vec(h, n)
    mx = c.max()
    active = c >= mx - 1e-12
    dx = 2.0 / n
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if not active[mi]:
            continue
        k = mi - (n - 1)
        wm = dx / active.sum()
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def lbfgs_run(h, n, deadline, alpha, maxit=300):
    def f(x):
        return smooth_max(compute_c5_vec(x, n), alpha)
    def df(x):
        return grad_smmax(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=df, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-19, 'gtol': 1e-17})
        hn = project_feasible(res.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def nesterov_run(h, n, deadline, alpha, maxit=400):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 1.0 / (alpha * 50.0)
    y = h.copy()
    t = 1.0
    for it in range(maxit):
        if time.time() > deadline:
            break
        g = grad_smmax(y, n, alpha)
        x = project_feasible(y - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = x + beta * (x - best_h)
        best_h = x.copy()
        t = t_new
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
    return best_h, best_c

def subgrad_descent(h, n, deadline, steps=500, decay=0.999):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.5 / (2.0 / n)
    h_c = h.copy()
    for it in range(steps):
        if time.time() > deadline:
            break
        g = subgrad_max(h_c, n)
        h_c = project_feasible(h_c - step * g, n)
        c = compute_c5(h_c, n)
        if c < best_c:
            best_c = c
            best_h = h_c.copy()
        step *= decay
    return best_h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    pos = np.floor(np.arange(half) * phi).astype(int) % n
    h[pos] = 1.0
    return project_feasible(h, n)

def pair_polish(h, n, deadline, radius=20):
    best_c = compute_c5(h, n)
    for d in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo, hi = max(0, i - radius), min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def multi_alpha_lbfgs(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha, nit in [(5000, 200), (20000, 250), (100000, 300),
                       (500000, 350), (2000000, 400), (10000000, 500)]:
        if time.time() > deadline - 8:
            break
        rem = deadline - time.time() - 8
        if rem < 10:
            break
        hn, cn = lbfgs_run(best_h, n, time.time() + rem, alpha, maxit=nit)
        if cn < best_c - 1e-17:
            best_c = cn
            best_h = hn
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _rng
    _rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = golden_start(500)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Multi-alpha L-BFGS (main optimization)
    t1 = min(t0 + 350, deadline - 300)
    best_h, best_c = multi_alpha_lbfgs(best_h, best_n, t1)

    # Phase 2: Nesterov accelerated gradient
    if time.time() < deadline - 250:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 200:
                break
            hn, cn = nesterov_run(best_h, best_n,
                                   min(time.time() + 60, deadline - 190), alpha, maxit=300)
            if cn < best_c:
                best_c = cn
                best_h = hn

    # Phase 3: Try alternative n
    for n_try in [400, 600]:
        if time.time() > deadline - 180:
            break
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, n_try, endpoint=False)
        hi = project_feasible(np.interp(xn, xo, best_h), n_try)
        alloc = min(80, (deadline - time.time() - 100) * 0.4)
        if alloc < 25:
            continue
        hn, cn = multi_alpha_lbfgs(hi, n_try, time.time() + alloc)
        if cn < best_c:
            best_c = cn
            best_h = hn
            best_n = n_try

    # Phase 4: Diverse restarts
    if time.time() < deadline - 100:
        starts = [golden_start(best_n),
                  project_feasible(golden_start(best_n) + _rng.randn(best_n) * 0.08, best_n)]
        hf = np.full(best_n, 0.5)
        xf = np.linspace(0, 2, best_n, endpoint=False)
        for k in range(1, 10):
            hf += _rng.uniform(0.05, 0.25) * np.cos(np.pi * k * xf + _rng.uniform(0, 6.28))
        starts.append(project_feasible(hf, best_n))
        for s in starts:
            if time.time() > deadline - 50:
                break
            alloc = min(30, (deadline - time.time() - 40) * 0.3)
            if alloc < 12:
                break
            hn, cn = multi_alpha_lbfgs(s, best_n, time.time() + alloc)
            if cn < best_c:
                best_c = cn
                best_h = hn

    # Phase 5: Subgradient refinement
    if time.time() < deadline - 40:
        hn, cn = subgrad_descent(best_h, best_n,
                                  min(time.time() + 35, deadline - 30), steps=800)
        if cn < best_c:
            best_c = cn
            best_h = hn

    # Phase 6: Final high-alpha L-BFGS
    if time.time() < deadline - 20:
        hn, cn = lbfgs_run(best_h, best_n, min(time.time() + 30, deadline - 10),
                           50000000, maxit=500)
        if cn < best_c:
            best_c = cn
            best_h = hn

    # Phase 7: Final Nesterov with very high alpha
    if time.time() < deadline - 10:
        hn, cn = nesterov_run(best_h, best_n, min(time.time() + 20, deadline - 5),
                              20000000, maxit=300)
        if cn < best_c:
            best_c = cn
            best_h = hn

    # Phase 8: Pair polish
    if time.time() < deadline - 1:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1, radius=15)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
