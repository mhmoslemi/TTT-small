    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 500.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha):

    mx = sh.max()

    top_k = 100
    top_idx = np.argsort(w)[-top_k:]

            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    grad += 1000.0 * (x.sum() - n / 2.0)
    return grad

                h = h_new2

    return best_h, best_c
