def delta_c_at_k(h, i, j, k, n):
    """Compute delta in c_fwd[k] when swapping h[i]:1→0, h[j]:0→1.
    k is the forward lag (0 <= k <= n-1)."""

        delta += h[i-k] - 1
    if k <= n-1-i:
        delta += h[i+k]
    # j: 0→1
    if k <= j <= n-1-k:
        delta += 1 - h[j-k]
    if k <= n-1-j:
        delta -= h[j+k]
    return delta

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:

            mask = h > 0.0

            h[mask] += (diff / tot) * h[mask]

def smooth_max(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)
