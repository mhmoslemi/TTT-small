```python
    """Gradient of smooth max of c w.r.t. c itself, and the smooth max value."""
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    val = c.max() + (mx + np.log(weights.sum())) / alpha

    val, w = smooth_max_grad(c, alpha)
```

```python
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = compute_c5_all(h, n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-16:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] -= wm * h[kk:]
    val = c.max() + (mx + np.log(w.sum())) / alpha
    return val, grad

def _obj(x, n, alpha):
    c = compute_c5_all(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    return float(c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _grad(x, n, alpha):
    _, g = grad_c5(x, n, alpha)
    return g

def run_lbfgs(h, n, deadline, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-17,
                                               'gtol': 1e-15, 'maxfun': maxiter*40})
        h_new = project_box_sum(res.x, n/2.0)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)
```

```python
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_all = compute_c5_all(h, n)
            top_idx = np.argsort(c_all)[-top_m:]
            ks = top_idx - (n - 1)
            
            # Precompute B matrix: B[k_idx, j] = (1-h[j+ks[k_idx]]) - h[j-ks[k_idx]]
            # with 0 where indices are out of bounds
            B = np.zeros((top_m, n))
            for mi, k in enumerate(ks):
                for j in range(n):
                    b = 0.0
                    if 0 <= j + k < n:
                        b += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        b -= h[j - k]
                    B[mi, j] = b
            
            for i in range(n):
                if time.time() > deadline:
                    break
                # A_k for fixed i: A_k = (1-h[i+k]) - h[i-k]
                A = np.zeros(top_m)
                for mi, k in enumerate(ks):
                    a = 0.0
                    if 0 <= i + k < n:
                        a += (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        a -= h[i - k]
                    A[mi] = a
                
                # For each j: need B[:,j] > A (all components)
                # i.e., (B - A)[:,j] > 0 for all rows
                margin = (B - A[None, :]).min(axis=0)
                candidates = np.where(margin > 1e-12)[0]

                    h[j] -= step

                        # Update B and break to recompute
                        break
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_all(h, n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] -= wm * h[kk:]
    return w, grad

def _smoob(x, n, alpha):
    c = compute_c5_all(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    return float(c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _smog(x, n, alpha):
    _, g = grad_smooth_max(x, n, alpha)
    return g

def lbfgs_phase(h, n, deadline, alpha_list):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(_smoob, h, jac=_smog, args=(n, alpha),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 500, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': 20000})
            h_new = project_box_sum(res.x, n / 2.0)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

            else:
                kk = -k
                nv = n - kk
                grad[kk:] += dx * (1.0 - h[:nv])
                grad[:nv] -= dx * h[kk:]

def fine_search(h, n, deadline):

                margin = (B - A[None, :]).min(axis=0)

                j_candidates = j_candidates[j_candidates != i]

                        improved = True
                        break
                    else:
                        h[i] -= step
                        h[j] += step
                if improved:
                    break
```

```python
                margins_valid = margins[valid_j]
                best_j_local = valid_j[np.argmax(margins_valid)]
                best_gain_local = margins_valid.max()
                if best_gain_local > best_gain:
                    best_gain = best_gain_local
                    best_i, best_j = i, best_j_local
            if best_i < 0:
                break
            # Apply
            h[best_i] += step
            h[best_j] -= step
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
            else:
                h[best_i] -= step
                h[best_j] += step
                break
    return h, best_c
```
