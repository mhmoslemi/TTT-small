import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * (c - c.max())
    w = np.exp(cs - cs.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += 0.05 * s * s
    return float(val)

    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 0.1 * s
    return g

                    jkn = j - ks
                    m1 = (ik >= 0) & (ik < n)
                    m2 = (ikn >= 0) & (ikn < n)
                    m3 = (jk >= 0) & (jk < n)
                    m4 = (jkn >= 0) & (jkn < n)
                    dC = np.zeros(2 * n - 1)
                    dC += d * (1.0 - h[icrop(ik, n)]) * m1
                    dC -= d * h[icrop(ikn, n)] * m2
                    dC -= d * (1.0 - h[icrop(jk, n)]) * m3
                    dC += d * h[icrop(jkn, n)] * m4
                    if j > i:
                        dC[(j - i) + (n - 1)] += d2
                    else:
                        dC[(i - j) + (n - 1)] += d2
                    new_max = (C + dC).max()
                    if new_max < best_c - 1e-15:
                        h[i] += d
                        h[j] -= d
                        C += dC
                        best_c = float(new_max)
                        improved = True
    return h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 9):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS warmup on initial
    t1 = min(t0 + 80, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=120)

    # Phase 2: Fast local search (main improvement phase)
    t2 = min(t0 + 400, deadline - 200)
    best_h, best_c = fast_local_search(best_h, best_n, t2, radius=30)

    # Phase 3: Gradient-directed refinement
    if time.time() < deadline - 100:
        t3 = min(time.time() + 80, deadline - 80)
        best_h, best_c = gradient_directed(best_h, best_n, t3)

    # Phase 4: Try n = 250 (faster local search, can do more iterations)
    if time.time() < deadline - 200:
        n_try = 250
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_t = np.interp(x_new, x_old, best_h)
        h_t = project_feasible(h_t, n_try)
        alloc = min(100, (deadline - time.time() - 100) * 0.6)
        if alloc > 40:
            h_t, c_t = lbfgs_optimize(h_t, n_try, time.time() + 30, max_iters=80)
            h_t, c_t = fast_local_search(h_t, n_try, time.time() + alloc, radius=40)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 5: Diverse restarts
    if time.time() < deadline - 80:
        starts = [random_binary(best_n, rng), fourier_start(best_n, rng)]
        for s in starts:
            if time.time() > deadline - 50:
                break
            alloc = min(40, (deadline - time.time() - 30) * 0.5)
            if alloc < 15:
                break
            h_s, c_s = lbfgs_optimize(s, best_n, time.time() + 20, max_iters=60)
            h_s, c_s = fast_local_search(h_s, best_n, time.time() + 20, radius=25)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 6: Perturbation + reoptimize
    if time.time() < deadline - 60:
        h_p = best_h + rng.randn(best_n) * 0.03
        h_p = project_feasible(h_p, best_n)
        alloc = min(35, (deadline - time.time() - 20) * 0.6)
        if alloc > 10:
            h_p, c_p = lbfgs_optimize(h_p, best_n, time.time() + 12, max_iters=50)
            h_p, c_p = fast_local_search(h_p, best_n, time.time() + 12, radius=20)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 7: Final fine local search
    if time.time() < deadline - 5:
        t7 = deadline - 2
        best_h, best_c = fast_local_search(best_h, best_n, t7, radius=15)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
