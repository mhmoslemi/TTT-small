from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def fw_gradient(h, n, m_index):
    """Compute gradient of correlation at index m_index with respect to h."""
    shift = m_index - (n - 1)
    grad = np.zeros(n)
    # dc/dh[j] = (1-h[j-shift]) - h[j+shift] (with bounds)
    if shift >= 0:
        # j-shift valid when j >= shift
        grad[shift:] += (1 - h[:n-shift])
        # j+shift valid when j+shift < n, i.e., j < n-shift
        if shift > 0:
            grad[:n-shift] -= h[shift:]
    else:
        s = -shift
        # j-shift = j+s valid when j+s < n, i.e., j < n-s
        grad[:n-s] += (1 - h[s:])
        # j+shift = j-s valid when j >= s
        grad[s:] -= h[:n-s]
    return grad

def frank_wolfe(h, n, max_iter=2000, deadline=None, verbose=False):
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h, n)
    h = h.copy()
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        c_vals = corr * dx
        c_max = c_vals.max()
        
        if c_max < best_val:
            best_val = c_max
            best_h = h.copy()
        
        # Use top-k active constraints for stability
        top_k = min(5, len(c_vals))
        thresholds = np.sort(c_vals)[-top_k]
        thresh = thresholds[0]  # threshold for top_k
        active = np.where(c_vals >= thresh)[0]
        
        # Compute averaged gradient over active constraints
        grad = np.zeros(n)
        for m in active:
            grad += fw_gradient(h, n, m)
        grad /= len(active)
        
        # Frank-Wolfe linear step: minimize grad · h subj to 0<=h<=1, sum(h)=n/2
        order = np.argsort(grad)
        w = np.zeros(n)
        w[order[:n//2]] = 1.0
        
        # Step size: use 2/(it+2) (standard FW) or line search
        # Let's use a mini line search
        t_candidates = [2.0 / (it + 2), 1.0 / (it + 1), 0.5, 0.25, 0.1]
        best_t = 2.0 / (it + 2)
        best_f = c_max
        for t in t_candidates:

            ht = project_feasible(ht, n)
            ft = compute_c5(ht, n)
            if ft < best_f:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def fw_gradient(h, n, m_index):
    shift = m_index - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            grad[:n - shift] -= h[shift:]
    else:
        s = -shift
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    return grad

def frank_wolfe_step(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c_vals = corr * dx
    c_max = c_vals.max()
    
    idx_sorted = np.argsort(c_vals)[::-1]
    active = idx_sorted[:top_k]
    
    grad = np.zeros(n)
    for m in active:
        grad += fw_gradient(h, n, m)
    grad /= len(active)
    
    # FW vertex: set n/2 smallest grad components to 1
    order = np.argsort(grad)
    w = np.zeros(n)
    w[order[:n // 2]] = 1.0
    return w, c_max

    # For the full correlation array indexed by m (0 to 2n-2):
    # corr[m] = sum_i h_i * (1-h_{i-(m-n+1)})
    # d(corr[m])/d(h_j) = (1-h_{j-(m-n+1)}) - h_{j+(m-n+1)}

        # (1-h_{j-s}) for j-s in [0,n)
        if s >= 0:
            grad[s:] += wval * (1.0 - h[:n-s])
        else:
            grad[:n+s] += wval * (1.0 - h[-s:])
        # -h_{j+s} for j+s in [0,n)
        if s > 0:
            grad[:n-s] -= wval * h[s:]
        elif s < 0:
            grad[-s:] -= wval * h[:n+s]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def fw_gradient(h, n, m_idx):
    shift = m_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            grad[:n - shift] -= h[shift:]
    else:
        s = -shift
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    return grad

def weighted_gradient(h, w, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wv = w[idx]
        if abs(wv) < 1e-16:
            continue
        grad += wv * fw_gradient(h, n, idx)
    return grad * dx

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    obj = m + (sm + np.log(w.sum())) / alpha
    grad = weighted_gradient(h, w, n)
    return obj, grad

def slsqp_with_grad(h, n, alpha, maxiter=3000, ftol=1e-17):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return lse_obj_grad(x, n, alpha)[0]
    def grad(x):
        return lse_obj_grad(x, n, alpha)[1]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def alternating_max(h_init, n, max_outer=30, deadline=None):
    h = h_init.copy()
    dx = 2.0 / n
    w = np.ones(2 * n - 1) / (2 * n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for outer in range(max_outer):
        if deadline and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        wcopy = w.copy()
        ncopy = n
        dxcopy = dx
        def obj2(x):
            corr = np.correlate(x, 1 - x, mode='full') * dxcopy
            return float(np.dot(wcopy, corr))
        def grad2(x):
            return weighted_gradient(x, wcopy, ncopy)
        try:
            res = minimize(obj2, h, jac=grad2, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            h_new = h
        corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
        c_max = corr.max()
        beta = 20.0 + 10.0 * outer
        w = np.exp(beta * (corr - c_max))
        w /= w.sum()
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def fw_optimize(h, n, max_iter=300, deadline=None, top_k=5):
    best_h = h.copy()
    best_val = compute_c5(h, n)
    h = h.copy()
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c_vals = corr * (2.0 / n)
        c_max = c_vals.max()
        if c_max < best_val:
            best_val = c_max
            best_h = h.copy()
        idx_sorted = np.argsort(c_vals)[::-1]
        active = idx_sorted[:top_k]
        grad = np.zeros(n)
        for m in active:
            grad += fw_gradient(h, n, m)
        grad /= len(active)
        order = np.argsort(grad)
        w = np.zeros(n)
        w[order[:n // 2]] = 1.0
        best_t = 2.0 / (it + 2)
        best_f = c_max
        for t in [2.0 / (it + 2), 1.0, 0.5, 0.25, 0.1]:
            ht = project_feasible(h + t * (w - h), n)
            ft = compute_c5(ht, n)
            if ft < best_f:
                best_f = ft
                best_t = t
        h = project_feasible(h + best_t * (w - h), n)
    return best_h, best_val

def coord_descent_binary(h, n, max_sweeps=100, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    h[np.where(h > 0.5)[0]] = 1.0
    h[np.where(h <= 0.5)[0]] = 0.0
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            if h[i] > 0.5 and val0 < val1 - 1e-12:
                corr += B * (0.0 - h[i])
                h[i] = 0.0
                improved = True
            elif h[i] <= 0.5 and val1 < val0 - 1e-12:
                corr += B * (1.0 - h[i])
                h[i] = 1.0
                improved = True
        if not improved:
            break
    return h
