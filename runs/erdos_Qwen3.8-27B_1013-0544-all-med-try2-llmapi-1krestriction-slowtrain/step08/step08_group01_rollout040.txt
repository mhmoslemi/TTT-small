            if compute_c5(h_new, n) < compute_c5(h, n) + 1e-12:
                h = h_new
        except Exception:
            pass
    return h

def slsqp_phase(h, n, deadline, alpha=1000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_eq, 'jac': slq_jac, 'args': (n,)}]
    try:
        res = minimize(slq_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgrad_polish(h, n, deadline, max_iters=3000):
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    step = 0.02
    dx = 2.0 / n
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h_curr, n)
        cmax = c.max()
        active = c >= cmax - 1e-7
        if active.sum() == 0:
            active[np.argmax(c)] = True
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if not active[m]:
                continue
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += dx * (1.0 - h_curr[k:n])
                grad[k:n] += dx * (-h_curr[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h_curr[:nv])
                grad[:nv] += dx * (-h_curr[kk:n])
        grad /= float(active.sum())
        h_new = project_feasible(h_curr - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h_curr = h_new
            step = min(step * 1.08, 0.15)
            stuck = 0
        else:
            step *= 0.75
            stuck += 1
            if stuck > 200 or step < 1e-7:
                break
    return h_curr, best_c

def pair_polish(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0004, 0.0001]
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def fourier_interp_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: L-BFGS-B on parent (n=500) with well-tuned alpha
    alphas_500 = [1000, 5000, 20000, 100000, 500000, 2000000]
    t1_dl = min(t0 + 250, deadline - 500)
    h = lbfgs_phase(h, n, t1_dl, alphas_500, max_iter=250)
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: SLSQP polish at n=500
    if time.time() < deadline - 450:
        t2 = min(time.time() + 80, deadline - 420)
        h2, c2 = slsqp_phase(h, n, t2, alpha=2000000, maxiter=3000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Subgradient polish at n=500
    if time.time() < deadline - 400:
        t3 = min(time.time() + 60, deadline - 380)
        h3, c3 = subgrad_polish(h, n, t3, max_iters=2000)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Try n=1000
    n1000 = 1000
    if time.time() < deadline - 300:
        h1000 = fourier_interp_to(best_h, best_n, n1000)
        alphas_1000 = [2000, 10000, 50000, 200000, 1000000]
        t4_dl = min(time.time() + 280, deadline - 200)
        h1000 = lbfgs_phase(h1000, n1000, t4_dl, alphas_1000, max_iter=200)
        c1000 = compute_c5(h1000, n1000)
        if c1000 < best_c:
            best_c, best_h, best_n = c1000, h1000, n1000

        # Phase 5: SLSQP at n=1000
        if time.time() < deadline - 120 and best_n == n1000:
            t5 = min(time.time() + 70, deadline - 100)
            h5, c5 = slsqp_phase(h1000, n1000, t5, alpha=3000000, maxiter=2500)
            if c5 < best_c:
                best_c, best_h = c5, h5

    # Phase 6: Subgradient polish on best
    if time.time() < deadline - 60:
        t6 = min(time.time() + 50, deadline - 40)
        h6, c6 = subgrad_polish(best_h, best_n, t6, max_iters=3000)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Pair polish
    if time.time() < deadline - 5:
        t7 = min(time.time() + 45, deadline - 3)
        best_h, best_c = pair_polish(best_h, best_n, t7, radius=20)

    # Final: try a perturbation restart if time remains
    if time.time() < deadline - 30:
        h_pert = best_h + rng.randn(best_n) * 0.03
        h_pert = project_feasible(h_pert, best_n)
        h_pert = lbfgs_phase(h_pert, best_n, min(time.time() + 25, deadline - 5),
                            [50000, 200000], max_iter=150)
        c_pert = compute_c5(h_pert, best_n)
        if c_pert < best_c:
            best_c, best_h = c_pert, h_pert

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
