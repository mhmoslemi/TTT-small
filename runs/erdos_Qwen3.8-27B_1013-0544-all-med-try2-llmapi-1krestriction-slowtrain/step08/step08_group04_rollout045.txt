                    j_range = [int(time.time() * 31) % n]
                for j in j_range:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return best_h, best_c

def random_pair_search(h, n, deadline, deltas=None, n_per_round=2000):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        improved = False
        stall_count = 0
        while time.time() < deadline:
            improved = False
            rng = np.random.RandomState(int(time.time() * 104729) % 999983)
            pairs = list(zip(rng.randint(0, n, n_per_round), rng.randint(0, n, n_per_round)))
            for i, j in pairs:
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; best_h = h.copy(); improved = True
                    else:
                        h[i] -= d; h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d; h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; best_h = h.copy(); improved = True
                    else:
                        h[i] += d; h[j] -= d
            stall_count += 0 if improved else 1
            if stall_count >= 3:
                break
    return best_h, best_c

def block_moves(h, n, deadline, block_sizes=(2, 3, 5, 8)):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(777)
    for bl in block_sizes:
        while time.time() < deadline:
            improved = False
            for _ in range(800):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.randint(0, n - bl)
                j = rng.randint(0, n - bl)
                if abs(i - j) < bl:
                    continue
                for delta in [0.02, 0.01, 0.005, 0.002]:
                    b1 = h[i:i+bl].copy()
                    b2 = h[j:j+bl].copy()
                    if b1.min() + delta > 1.0 or b2.min() - delta < -0.5:
                        continue
                    h[i:i+bl] = np.clip(b1 + delta, 0, 1)
                    h[j:j+bl] = np.clip(b2 - delta, 0, 1)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; best_h = h.copy(); improved = True
                        break
                    h[i:i+bl] = b1
                    h[j:j+bl] = b2
            if not improved:
                break
    return best_h, best_c

def slsqp_refine(h, n, deadline, alpha=200000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x, n):
        return x.sum() - n / 2.0
    def cons_jac(x, n):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    mu = 0.01
    try:
        res = minimize(obj_fn, h, jac=obj_grad, args=(n, alpha, mu),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS at original resolution
    t1 = min(t0 + 120, deadline - 400)
    best_h, best_c = run_lbfgs(best_h, n_init, t1, max_iters=120)

    # Phase 2: SLSQP at original resolution
    if time.time() < deadline - 300:
        t2 = min(time.time() + 80, deadline - 300)
        h2, c2 = slsqp_refine(best_h, n_init, t2, alpha=200000, maxiter=2000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Interpolate to higher resolution and optimize
    for n_hr in [400, 500, 600]:
        if time.time() > deadline - 180:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_hr)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_hr)
        alloc = min(100, deadline - time.time() - 120)
        if alloc < 40:
            break
        h_hr, c_hr = run_lbfgs(h_interp, n_hr, time.time() + alloc * 0.5, max_iters=100)
        if time.time() < deadline - 80:
            h_hr, c_hr = random_pair_search(h_hr, n_hr, time.time() + min(30, deadline - time.time() - 50),
                                            deltas=[0.01, 0.005, 0.002, 0.001])
        if c_hr < best_c:
            best_c = c_hr
            best_h = h_hr

    # Phase 4: Aggressive local search at best resolution
    nf = len(best_h)
    if time.time() < deadline - 80:
        t4 = min(time.time() + 120, deadline - 80)
        best_h, best_c = random_pair_search(best_h, nf, t4,
                                            deltas=[0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5])

    # Phase 5: Block moves
    if time.time() < deadline - 50:
        t5 = min(time.time() + 40, deadline - 50)
        best_h, best_c = block_moves(best_h, nf, t5)

    # Phase 6: Final fine-tuning
    if time.time() < deadline - 15:
        t6 = min(time.time() + 30, deadline - 15)
        best_h, best_c = random_pair_search(best_h, nf, t6,
                                            deltas=[0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5])

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 10:
        nf = len(best_h)
        h7, c7 = slsqp_refine(best_h, nf, min(time.time() + 20, deadline - 5), alpha=500000, maxiter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final micro-tuning
    if time.time() < deadline - 3 and len(best_h) > 0:
        nf = len(best_h)
        best_h, best_c = random_pair_search(best_h, nf, deadline - 1,
                                            deltas=[5e-5, 2e-5, 1e-5, 5e-6])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
