import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alpha_seq=None, max_iters=500):
    best_c = compute_c5(h, n)
    if alpha_seq is None:
        alpha_seq = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alpha_seq:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': max_iters * 50,
                                  'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def sim_anneal(h, n, deadline, t_start=0.005, t_end=1e-7, n_sweeps=200):
    """Simulated annealing with pair moves"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    c_current = best_c
    
    total_iters = n_sweeps * n
    for it in range(total_iters):
        if time.time() > deadline:
            break
        # Temperature schedule (geometric)
        frac = it / total_iters
        T = t_start * (t_end / t_start) ** frac
        
        # Choose a random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Determine transfer direction and amount
        delta = rng.uniform(0.001, 0.01)
        
        # Try: h[i] += delta, h[j] -= delta
        if h[i] + delta <= 1.0 + 1e-12 and h[j] - delta >= -1e-12:
            h[i] += delta
            h[j] -= delta
            h[i] = min(h[i], 1.0)
            h[j] = max(h[j], 0.0)
            c_new = compute_c5(h, n)
            delta_c = c_new - c_current
            if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-15)):
                c_current = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
        # Try reverse
        elif h[i] - delta >= -1e-12 and h[j] + delta <= 1.0 + 1e-12:
            h[i] -= delta
            h[j] += delta
            h[i] = max(h[i], 0.0)
            h[j] = min(h[j], 1.0)
            c_new = compute_c5(h, n)
            delta_c = c_new - c_current
            if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-15)):
                c_current = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i] += delta
                h[j] -= delta
    
    # Project back to feasible
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0

    # Ensure unique
    unique_pos = np.unique(positions)
    h[unique_pos[:half]] = 1.0
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wt = w[m_idx]
        if wt < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = wt * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, deadline, alpha_list, max_it=400):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alpha_list:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smax_obj, h, jac=smax_grad, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': max_it, 'ftol': 1e-18, 'gtol': 1e-16,
                                  'maxfun': max_it * 40, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
