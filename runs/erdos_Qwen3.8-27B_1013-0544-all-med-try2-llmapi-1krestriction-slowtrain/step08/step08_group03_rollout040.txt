from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(100, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        wi = w[i]
        if wi < 1e-14:
            continue
        k = i - (n - 1)
        nv = n - abs(k)
        if k >= 0:
            grad[:nv] += wi * dx * (1.0 - x[k:])
            grad[k:] -= wi * dx * x[:nv]
        else:
            kk = -k
            grad[kk:] += wi * dx * (1.0 - x[:nv])
            grad[:nv] -= wi * dx * x[kk:]
    s = x.sum() - n / 2.0
    grad += 200.0 * s
    return grad

def compute_sensitivity(h, n):
    """Compute S[i, lag_index] = d(corr[lag_index])/d(h[i])"""
    # corr has length 2n-1, index n-1 is lag 0
    S = np.zeros((n, 2*n-1))
    for i in range(n):
        for k in range(-(n-1), n):
            lag_idx = n - 1 + k
            # h[i] appears in corr[n-1+k] as:
            # first factor: h[i]*(1-h[i+k]) if 0<=i<n and 0<=i+k<n

            S[i, lag_idx] = val
    return S

    # For each i, S[i, n-1+k] gets contributions from:

from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]

def nesterov_pgd(h, n, deadline, alpha=200.0, top_k=80):
    """Nesterov-accelerated projected gradient descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    t = 1.0
    step = 2.0 / (alpha * 2.0 / n)  # Lipschitz estimate
    
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - y)
        
        c = compute_c5_vec(y, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        top_idx = np.argsort(w)[-top_k:]
        dx_val = 2.0 / n
        g = np.zeros(n)
        for idx in top_idx:
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = idx - (n - 1)
            nv = n - abs(k)
            if k >= 0:
                g[:nv] += wi * dx_val * (1.0 - y[k:])
                g[k:] -= wi * dx_val * y[:nv]
            else:
                kk = -k
                g[kk:] += wi * dx_val * (1.0 - y[:nv])
                g[:nv] -= wi * dx_val * y[kk:]
        s = y.sum() - n / 2.0

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        t = t_new

def compute_S(h, n):
    S = np.zeros((n, 2 * n - 1))
    center = n - 1
    for i in range(n):
        # k >= 0 part: p from center to 2*center
        # (1-h[i-k]) for k in [0, min(i, n-1-i)]
        k_max1 = min(i, n - 1 - i)
        if k_max1 >= 0:
            ks = np.arange(0, k_max1 + 1)
            S[i, center + ks] += (1.0 - h[i - ks])
        # -h[i+k] for k in [0, n-1-i]
        k_max2 = n - 1 - i
        if k_max2 >= 0:
            ks = np.arange(0, k_max2 + 1)
            S[i, center + ks] -= h[i + ks]
        
        # k < 0 part: m = -k, p from 0 to center-1
        # (1-h[i+m]) for m in [1, n-1-i]
        m_max1 = n - 1 - i
        if m_max1 >= 1:
            ms = np.arange(1, m_max1 + 1)
            S[i, center - ms] += (1.0 - h[i + ms])
        # -h[i-m] for m in [1, min(i, n-1)]
        m_max2 = min(i, n - 1)
        if m_max2 >= 1:
            ms = np.arange(1, m_max2 + 1)
            S[i, center - ms] -= h[i - ms]
    return S

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(100, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        nv = n - abs(k)
        if k >= 0:
            grad[:nv] += wi * dx * (1.0 - x[k:])
            grad[k:] -= wi * dx * x[:nv]
        else:
            kk = -k
            grad[kk:] += wi * dx * (1.0 - x[:nv])
            grad[:nv] -= wi * dx * x[kk:]
    s = x.sum() - n / 2.0
    grad += 200.0 * s
    return grad

def compute_S(h, n):
    S = np.zeros((n, 2 * n - 1))
    center = n - 1
    for i in range(n):
        k_max1 = min(i, n - 1 - i)
        if k_max1 >= 0:
            ks = np.arange(0, k_max1 + 1)
            S[i, center + ks] += (1.0 - h[i - ks])
        k_max2 = n - 1 - i
        if k_max2 >= 0:
            ks = np.arange(0, k_max2 + 1)
            S[i, center + ks] -= h[i + ks]
        m_max1 = n - 1 - i
        if m_max1 >= 1:
            ms = np.arange(1, m_max1 + 1)
            S[i, center - ms] += (1.0 - h[i + ms])
        m_max2 = min(i, n - 1)
        if m_max2 >= 1:
            ms = np.arange(1, m_max2 + 1)
            S[i, center - ms] -= h[i - ms]
    return S

def local_search(h, n, deadline, min_radius=10, max_radius=100):
    O = compute_c5_vec(h, n)
    best_c = float(np.max(O))
    S = compute_S(h, n)
    nlcen = 2 * n - 1
    rounds = [(max_radius, [0.01, 0.005, 0.002]),
              (max_radius // 2, [0.001, 0.0005, 0.0002]),
              (max_radius // 4, [0.0001, 5e-5, 2e-5]),
              (max_radius // 8, [1e-5, 5e-6])]
    for radius, deltas in rounds:
        if time.time() > deadline:
            break
        radius = max(radius, min_radius)
        for d in deltas:
            if time.time() > deadline:
                break
            changed = True
            while changed and time.time() < deadline:
                changed = False
                for i in range(n):
                    if time.time() > deadline:
                        break
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    for j in range(lo, hi):
                        if i == j:
                            continue
                        Si_minus_Sj = S[i] - S[j]
                        if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                            nm = np.max(O + d * Si_minus_Sj)
                            if nm < best_c - 1e-15:
                                h[i] += d
                                h[j] -= d
                                O += d * Si_minus_Sj
                                best_c = nm
                                changed = True
                        if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                            nm = np.max(O - d * Si_minus_Sj)
                            if nm < best_c - 1e-15:
                                h[i] -= d
                                h[j] += d
                                O -= d * Si_minus_Sj
                                best_c = nm
                                changed = True
    return h, best_c

def nesterov_pgd(h, n, deadline, alpha=500.0, top_k=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    t = 1.0
    step = 0.02
    stall = 0
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - y)
        c = compute_c5_vec(y, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        top_idx = np.argsort(w)[-top_k:]
        dxv = 2.0 / n
        g = np.zeros(n)
        for idx in top_idx:
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = idx - (n - 1)
            nv = n - abs(k)
            if k >= 0:
                g[:nv] += wi * dxv * (1.0 - y[k:])
                g[k:] -= wi * dxv * y[:nv]
            else:
                kk = -k
                g[kk:] += wi * dxv * (1.0 - y[:nv])
                g[:nv] -= wi * dxv * y[kk:]
        s = y.sum() - n / 2.0
        g += 200.0 * s
        h_new = project_feasible(y - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            step *= 0.8
        h = h_new
        t = t_new
        if stall > 80:
            break
    return best_h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

            h_new = project_feasible(res.x, n)

            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:

    # Phase 4: Final L-BFGS polish

                          args=(n, 10000), method='L-BFGS-B',

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
