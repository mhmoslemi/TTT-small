def perturb_h(h,n,rng,strength,mode):
    y=h.copy()
    if mode == 0:
        y += strength * rng.normal(0.0, 1.0, n)
    elif mode == 1:
        for _ in range(8):
            i=int(rng.randint(n)); j=int(rng.randint(n))
            if i == j: continue
            d = strength * float(rng.uniform(0.5, 1.5))
            if y[i] + d <= 1.0 and y[j] - d >= 0.0:
                y[i] += d; y[j] -= d
            elif y[i] - d >= 0.0 and y[j] + d <= 1.0:

        ph = float(rng.uniform(0.0, 2.0*np.pi))

import numpy as np
import time
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize, linprog


def overlap_counts(h):
    n = len(h)
    if n < 256:
        out = sp_correlate(h, 1.0 - h, mode='full', method='direct')
    else:
        out = sp_correlate(h, 1.0 - h, mode='full', method='fft')
    if np.iscomplexobj(out):
        out = np.real(out)
    return out


def c5_counts(c, n):
    return float(np.max(c) * (2.0 / n))


def c5_fft(h):
    return c5_counts(overlap_counts(h), len(h))


def c5_exact(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))


def force_sum(h, target):
    for _ in range(8):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        mask = (h > 0.0) & (h < 1.0)
        if not mask.any():
            break
        idxs = np.nonzero(mask)[0]
        vals = h[idxs]
        k = int(np.argmin(np.minimum(vals, 1.0 - vals)))
        idx = int(idxs[k])
        if diff > 0.0:
            add = min(diff, 1.0 - h[idx])
        else:
            add = max(diff, -h[idx])
        h[idx] += add
    return np.clip(h, 0.0, 1.0)


def project_feasible(y, target):
    y = np.asarray(y, dtype=float).copy()
    lo = -float(np.max(y)) - 1.0
    hi = 1.0 - float(np.min(y)) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    out = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    return force_sum(out, target)


def add_grad_single(g, h, n, k, w):
    if w == 0.0:
        return
    nk = abs(int(k))
    nv = n - nk
    if nv <= 0:
        return
    if k >= 0:
        g[:nv] += w * (1.0 - h[nk:])
        g[nk:] -= w * h[:nv]
    else:
        g[nk:] += w * (1.0 - h[:nv])
        g[:nv] -= w * h[nk:]


def top_indices(c, top):
    top = int(min(top, len(c)))
    if top <= 0:
        return np.empty(0, dtype=int)
    if top >= len(c):
        return np.arange(len(c))
    return np.argpartition(c, -top)[-top:]


def softmax_weights(vals, alpha):
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    w = np.exp(sh)
    s = float(w.sum())
    if s <= 0.0:
        w = (vals >= m - 1e-12).astype(float)
        s = float(w.sum())
    if s <= 0.0:
        w = np.ones_like(vals)
        s = float(vals.size)
    return w / s


def weighted_grad(h, n, c, top, alpha):
    idx = top_indices(c, top)
    w = softmax_weights(c[idx], alpha)
    g = np.zeros(n)
    off = n - 1
    for ii in range(len(idx)):
        add_grad_single(g, h, n, off - int(idx[ii]), float(w[ii]))
    return g


def smooth_obj_top(x, n, alpha, pen, top):
    c = overlap_counts(x)
    idx = top_indices(c, top)
    vals = c[idx]
    m = float(vals.max())
    sh = alpha * (vals - m)
    shm = float(sh.max())
    z = sh - shm
    val = m + (shm + float(np.log(np.sum(np.exp(z))))) / alpha
    s = float(x.sum()) - 0.5 * n
    val += pen * s * s
    return float(val)


def smooth_grad_top(x, n, alpha, pen, top):
    c = overlap_counts(x)
    idx = top_indices(c, top)
    w = softmax_weights(c[idx], alpha)
    g = np.zeros(n)
    off = n - 1
    for ii in range(len(idx)):
        add_grad_single(g, x, n, off - int(idx[ii]), float(w[ii]))
    s = float(x.sum()) - 0.5 * n
    g += 2.0 * pen * s
    return g


def lbfgs_step(h, n, alpha, pen, top, maxiter):
    n = int(n)
    c0 = c5_fft(h)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            smooth_obj_top, h,
            jac=smooth_grad_top,
            args=(n, alpha, pen, top),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': int(maxiter), 'ftol': 1e-15, 'gtol': 1e-10, 'maxcor': 12}
        )
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h2 = project_feasible(np.asarray(res.x, dtype=float), 0.5 * n)
            c2 = c5_fft(h2)
            if c2 < c0 - 1e-14:
                return h2, c2
    except Exception:
        pass
    return h, c0


def pgd_refine(h, n, deadline, max_iters=1000, top=50, alpha=300.0):
    n = int(n)
    target = 0.5 * n
    best_h = h.copy()
    best_c = c5_fft(h)
    step = 0.03
    it = 0
    while it < int(max_iters) and time.time() < float(deadline):
        it += 1
        c = overlap_counts(h)
        C = c5_counts(c, n)
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        g = weighted_grad(h, n, c, top, alpha)
        oldC = C
        accepted = False
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625):
            lr = step * f
            if lr <= 0.0:
                continue
            h2 = project_feasible(h - lr * g, target)
            C2 = c5_counts(overlap_counts(h2), n)
            if C2 < C - 1e-14:
                h = h2
                C = C2
                best_c = C2
                best_h = h2.copy()
                accepted = True
                if oldC - C2 > 1e-9:
                    step = min(0.25, step * 1.1)
                else:
                    step *= 0.95
                break
        if not accepted:
            step *= 0.7
            if step < 1e-5:
                break
    return best_h, best_c


def pair_search(h, n, deadline, max_sweeps=20, low_high=25):
    n = int(n)
    target = 0.5 * n
    best_h = h.copy()
    best_c = c5_fft(h)
    fracs = (0.001, 0.0025, 0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0)
    for _ in range(int(max_sweeps)):
        if time.time() > float(deadline):
            break
        c = overlap_counts(h)
        C = c5_counts(c, n)
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        g = weighted_grad(h, n, c, 50, 100.0)
        order = np.argsort(g)
        low = []
        high = []
        for p in order:
            if h[p] < 1.0 - 1e-12 and len(low) < int(low_high):
                low.append(int(p))
        for p in order[::-1]:
            if h[p] > 1e-12 and len(high) < int(low_high):
                high.append(int(p))
        improved = False
        for i in low:
            hiav = 1.0 - h[i]
            if hiav <= 1e-12:
                continue
            for j in high:
                if i == j:
                    continue
                maxd = min(hiav, float(h[j]))
                if maxd <= 1e-12:
                    continue
                for fr in fracs:
                    d = fr * maxd
                    if d > maxd:
                        d = maxd
                    if d <= 1e-14:
                        continue
                    oi = float(h[i])
                    oj = float(h[j])
                    ni = oi + d
                    nj = oj - d
                    if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    C2 = c5_counts(overlap_counts(h), n)
                    if C2 < C - 1e-14:
                        C = C2
                        improved = True
                        if C < best_c - 1e-14:
                            best_c = C
                            best_h = h.copy()
                        hiav = 1.0 - h[i]
                        break
                    h[i] = oi
                    h[j] = oj
                if hiav <= 1e-12:
                    break
        if not improved:
            break
        h = project_feasible(h, target)
        C = c5_counts(overlap_counts(h), n)
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
    return best_h, best_c


def lp_direction(h, n, c, tau=0.03, K=220):
    n = int(n)
    cmax = float(c.max())
    idx = top_indices(c, K)
    if len(idx) == 0:
        return None, 0.0
    A = np.zeros((len(idx), n + 1))
    b = np.zeros(len(idx))
    off = n - 1
    for r, p in enumerate(idx):
        row = A[r, :n]
        add_grad_single(row, h, n, off - int(p), 1.0)
        A[r, n] = 1.0
        b[r] = cmax - float(c[p])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    c_obj[:n] = 1e-8
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    lo = np.maximum(-tau, -h)
    hi = np.minimum(tau, 1.0 - h)
    bounds = list(zip(lo.tolist(), hi.tolist())) + [(0.0, 1.0)]
    try:
        res = linprog(c_obj, A, b, Aeq, 0.0, bounds=bounds, method='highs')
    except Exception:
        return None, 0.0
    if res is not None and res.success:
        delta = np.asarray(res.x, dtype=float)[:n].copy()
        r = float(res.x[-1])
        if r > 1e-8:
            return delta, r
    return None, 0.0


def lp_refine(h, n, deadline, max_iters=20, tau=0.03, K=220):
    n = int(n)
    target = 0.5 * n
    best_h = h.copy()
    best_c = c5_fft(h)
    for it in range(int(max_iters)):
        if time.time() > float(deadline):
            break
        c = overlap_counts(h)
        C = c5_counts(c, n)
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        delta, r = lp_direction(h, n, c, tau, K)
        if delta is None:
            break
        accepted = False
        for step in (1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04):
            h2 = project_feasible(h + step * delta, target)
            C2 = c5_counts(overlap_counts(h2), n)
            if C2 < C - 1e-14:
                h = h2
                C = C2
                best_c = C2
                best_h = h2.copy()
                accepted = True
                break
        if not accepted:
            if tau > 4e-4:
                tau *= 0.5
                continue
            break
    return best_h, best_c


def random_binary_start(n, rng):
    n = int(n)
    h = np.zeros(n)
    cnt = int(np.round(0.5 * n))
    idx = rng.choice(n, size=cnt, replace=False)
    h[idx] = 1.0
    return force_sum(h, 0.5 * n)


def perturb_h(h, n, rng, strength, mode):
    n = int(n)
    y = h.copy()
    if mode == 0:
        y += strength * rng.normal(0.0, 1.0, n)
    elif mode == 1:
        for _ in range(8):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = strength * float(rng.uniform(0.5, 1.5))
            if y[i] + d <= 1.0 and y[j] - d >= 0.0:
                y[i] += d
                y[j] -= d
            elif y[i] - d >= 0.0 and y[j] + d <= 1.0:
                y[i] -= d
                y[j] += d
    else:
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        m = 1 + int(rng.randint(0, 5))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        y += strength * np.sin(np.pi * m * x + ph)
    return project_feasible(y, 0.5 * n)


def basin_hop(h, n, deadline, rng, max_trials=15):
    n = int(n)
    best_h = h.copy()
    best_c = c5_fft(h)
    cur_h = h.copy()
    cur_c = best_c
    strengths = (0.003, 0.006, 0.0015, 0.01, 0.0045, 0.012, 0.0025, 0.018, 0.008, 0.02)
    for t in range(int(max_trials)):
        if time.time() > float(deadline) - 20.0:
            break
        s = strengths[t % len(strengths)]
        mode = t % 3
        hp = perturb_h(cur_h, n, rng, s, mode)
        hp, chp = lbfgs_step(hp, n, 500.0, 20.0, 120, 30)
        hp, chp = pgd_refine(hp, n, min(float(deadline) - 10.0, time.time() + 25.0), max_iters=60)
        if chp < best_c - 1e-13:
            best_h = hp.copy()
            best_c = chp
            cur_h = hp.copy()
            cur_c = chp
        elif chp < cur_c + 2e-5:
            cur_h = hp.copy()
            cur_c = chp
    return best_h, best_c


def resample_to_n(h_old, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return project_feasible(h_old, 0.5 * n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, 0.5 * n_new)


def absorb_candidate(best_h, best_c, work_h, work_n, work_c, h2, c2, n2, allow_n_change=False):
    ce = c5_exact(h2)
    improved_best = ce < best_c
    if improved_best:
        best_h = h2.copy()
        best_c = ce
        work_h = h2.copy()
        work_n = int(n2)
        work_c = c2
    elif c2 < work_c - 1e-14:
        if int(n2) == int(work_n) or allow_n_change:
            work_h = h2.copy()
            work_n = int(n2)
            work_c = c2
    return best_h, best_c, work_h, work_n, work_c
