```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for Euclidean projection
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    val = m + (a_max + np.log(np.sum(np.exp(a - a_max)))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for shift_m in range(2 * n - 1):
        if w[shift_m] < 1e-12:
            continue
        k = shift_m - (n - 1)
        wm = w[shift_m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_with_penalty(h, n, alpha, mu):
    return smooth_max_obj(h, n, alpha) + mu * (h.sum() - n / 2.0) ** 2

def grad_with_penalty(h, n, alpha, mu):
    return smooth_max_grad(h, n, alpha) + 2.0 * mu * (h.sum() - n / 2.0) * np.ones(n)

def run_lbfgs(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    mus = [10, 50, 100, 500, 1000, 5000, 10000, 50000]
    for i, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(obj_with_penalty, best_h, jac=grad_with_penalty,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 80, 'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_slsqp(h, n, deadline, alpha=500000, maxiter=2000):
    def _obj(x, n, a):
        return smooth_max_obj(x, n, a)
    def _cons_eq(x, n):
        return x.sum() - n / 2.0
    def _jac_eq(x, n):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _jac_eq, 'args': (n,)}]
    try:
        res = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    return float(m + (a_max + np.log(np.sum(np.exp(a - a_max)))) / alpha)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for shift_m in range(2 * n - 1):
        if w[shift_m] < 1e-12:
            continue
        k = shift_m - (n - 1)
        wm = w[shift_m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(h, n, alpha, mu):
    return smooth_max_obj(h, n, alpha) + mu * (h.sum() - n / 2.0) ** 2

def penalty_grad(h, n, alpha, mu):
    return smooth_max_grad(h, n, alpha) + 2.0 * mu * (h.sum() - n / 2.0)

def slsqp_obj(h, n, alpha):
    return smooth_max_obj(h, n, alpha)

def slsqp_eq(h, n):
    return h.sum() - n / 2.0

def slsqp_eq_jac(h, n):
    return np.ones(n)

def lwfh_step(h, n, alpha):
    """Linearized feasible half-step: project gradient onto feasible set"""
    g = smooth_max_grad(h, n, alpha)
    # Solve min g·z s.t. 0<=z<=1, sum(z)=n/2
    idx = np.argsort(g)
    z = np.zeros(n)
    z[idx[:n // 2]] = 1.0
    return z

def frank_wolfe(h, n, deadline, alpha=500, max_iters=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for t in range(max_iters):
        if time.time() > deadline:
            break
        gamma = 2.0 / (t + 2.0)
        z = lwfh_step(h, n, alpha)
        h_new = (1.0 - gamma) * h + gamma * z
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def run_lbfgs(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 10.0), (500, 50.0), (2000, 100.0),
               (10000, 500.0), (50000, 2000.0), (200000, 10000.0),
               (1000000, 50000.0), (5000000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        remaining = deadline - time.time() - 2
        if remaining < 1:
            break
        try:
            res = minimize(penalty_obj, best_h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 80, 'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_slsqp(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_polish(h, n, deadline, alpha_init=5000, lr_init=0.005):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = alpha_init
    lr = lr_init
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g /= g_norm
        improved = False
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    return float(m + (a_max + np.log(np.sum(np.exp(a - a_max)))) / alpha)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for shift_m in range(2 * n - 1):
        if w[shift_m] < 1e-12:
            continue
        k = shift_m - (n - 1)
        wm = w[shift_m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

                                     deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
