def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_box_sum(y, target):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(y - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def compute_smooth_grad(h, alpha):

    s -= s.max()
