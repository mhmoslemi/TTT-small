    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w_total = w.sum()
    if w_total < 1e-30: return np.zeros(n)
    w = w / w_total
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2*n-1):
        if w[m] < 1e-14: continue
        k = m - (n-1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _slsqp_objective(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _slsqp_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def _slsqp_eq(x, n):
    return float(x.sum() - n/2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 5: break
        alloc = deadline - time.time() - 3
        maxiter = int(max(100, alloc * 80))
        try:
            res = minimize(_slsqp_objective, best_h,
                         jac=_slsqp_grad, args=(n, alpha),
                         method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def projected_gradient(h, n, deadline, alpha=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    s_prev = None
    y_prev = None
    for it in range(100000):
        if time.time() > deadline: break
        g = grad_smooth_max(h, n, alpha)
        g_proj = g - g.mean()
        if s_prev is not None and y_prev is not None:
            sy = float(s_prev @ y_prev)
            if sy > 1e-12:
                step = float(s_prev @ s_prev) / sy
                step = float(np.clip(step, 1e-8, 0.05))
            else:
                step = 1.0 / (1.0 + float(np.abs(g_proj).max()) + 1e-30)
        else:
            step = 1.0 / (1.0 + float(np.abs(g_proj).max()) + 1e-30)
        h_new = project_feasible(h - step * g_proj, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            if s_prev is not None:
                y_curr = grad_smooth_max(h_new, n, alpha)
                s_prev = h_new - h
                y_prev = y_curr - g
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.5
            if step < 1e-12:
                h = h_new
                s_prev = None
                y_prev = None
    return best_h, best_c

def golden_ratio_start(n):

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def random_start(n, rng):

    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0

def fourier_start(n, rng):

        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.cos(2*np.pi*freq*x/2.0 + phase)
    return project_feasible(h, n)

def coordinate_polish(h, n, deadline):
