        if time.time() > deadline:
            break
        # Step 1: Compute current c values
        c = compute_c5_all(h, n)  # shape (2n-1,)
        
        # Step 2: Update weights (exponentially boost lags with high c)
        # Actually, we want to focus on lags with HIGH c (since we're minimizing max c)
        # So boost weights for lags where c is high
        alpha_w = 5.0
        w_new = w * np.exp(alpha_w * c / c.max())
        w_new /= w_new.sum()
        w = w_new
        
        # Step 3: Optimize h to minimize sum_k w_k * c_k
        # This is minimizing a weighted sum of overlaps
        # = 1 - sum_k w_k * autocorr_k
        # = 1 - sum_k w_k * sum_i h[i]*h[i+k] * dx
        # So we want to maximize sum_k w_k * sum_i h[i]*h[i+k]
        # This is a quadratic objective: h^T A h where A_{i,j} = w_{|i-j|} * dx (approximately)
        # Actually it's: sum_{k} w_k * sum_{i=0}^{n-k-1} h[i]*h[i+k] * dx
        # Let me just use L-BFGS with this as the objective
        
        # For efficiency, just do a few gradient steps
        g = np.zeros(n)
        for k_idx in range(2*n-1):
            if w[k_idx] < 1e-8:
                continue
            k = k_idx - (n-1)
            wk = w[k_idx] * dx  # dx = 2.0/n
            if k >= 0:
                nv = n - k
                g[0:nv] += wk * (1.0 - h[k:n])  # gradient of h[i]*(1-h[i+k])
                g[k:n] += wk * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wk * (1.0 - h[0:nv])
                g[0:nv] += wk * (-h[kk:n])
        
        # Gradient descent step on h (negative gradient since we want to minimize)
        # But we need to project...
        
        # Actually, let me just use the smooth max approach with the weights
        # The weighted smooth max is: log(sum exp(alpha * w_k * c_k)) / alpha
        # This converges to max_k w_k * c_k as alpha → ∞
        # But that's not the same as max_k c_k...
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(h, n, alpha, mu):
    c = compute_c5_all(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    val = c.max() + np.log(np.sum(np.exp(sh - mx))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(h, n, alpha, mu):
    c = compute_c5_all(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu != 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, deadline, alphas=None, mu=10.0, max_iters=500):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000, 200000, 1000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smax_obj, h, jac=smax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 30,
                                   'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def grad_local_search(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for base_step in steps:
        improved_any = False
        alpha_ls = 5000.0
        while time.time() < deadline - 5:
            improved = False
            g = smax_grad(h, n, alpha_ls, mu=0)
            top_src = np.argsort(g)[::-1][:20]
            top_dst = np.argsort(g)[:20]
            for d in [base_step, base_step * 0.5, base_step * 0.2]:
                for i in top_src:
                    for j in top_dst:
                        if i == j:
                            continue
                        if h[i] - d >= 0 and h[j] + d <= 1:
                            h[i] -= d
                            h[j] += d
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                                improved_any = True
                                break
                            else:
                                h[i] += d
                                h[j] -= d
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
        if not improved_any:
            break
    return h, best_c

def multi_alpha_lbfgs(h, n, deadline):
    h2, c2 = lbfgs_run(h, n, deadline, alphas=[100, 500, 2000], mu=5.0, max_iters=400)
    return h2, c2

    deadline = t0 + budget_s - 5

    t1 = min(t0 + 150, deadline - 500)

    t2 = min(t0 + 350, deadline - 350)

        best_h = h2

            best_h = h3

        h4, c4 = lbfgs_run(h, n, t4, alphas=[50000, 200000, 1000000], mu=5.0, max_iters=300)

        x_new = np.linspace(0, 2, n_lo)

        h_lo = project_feasible(h_lo, n_lo)

            best_h = h_lo

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
