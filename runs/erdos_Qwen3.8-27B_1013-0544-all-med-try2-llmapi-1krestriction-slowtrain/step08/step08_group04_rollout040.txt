                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Progressive L-BFGS with increasing alpha
    configs = [(100, 1.0, 200), (500, 10.0, 200), (2000, 50.0, 200),
               (10000, 100.0, 200), (50000, 500.0, 200),
               (200000, 1000.0, 300), (1000000, 5000.0, 300)]
    for alpha, mu, iters in configs:
        rem = deadline - time.time() - 350
        if rem < 10:
            break
        t_alloc = min(70, rem)
        if t_alloc < 8:
            break
        h = lbfgs_optimize(h, n, bounds, alpha, mu, iters, t0 + t_alloc)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()

    # Phase 2: LP-based exploration
    if time.time() < deadline - 200:
        for it in range(200):
            if time.time() > deadline - 150:
                break
            h_lp = lp_step(h, n, top_m=30)
            if h_lp is None:
                if it > 5:
                    break
                continue
            c_lp = compute_c5(h_lp, n)
            if c_lp < best_c - 1e-10:
                best_c = c_lp
                best_h = h_lp.copy()
                h = h_lp
            elif c_lp < best_c + 0.003:
                h_trial = h_lp.copy()
                h_ref = lbfgs_optimize(h_trial, n, bounds, 20000, 100.0, 80,
                                       min(time.time() + 10, deadline - 140))
                h_ref = project_feasible(h_ref, n)
                c_ref = compute_c5(h_ref, n)
                if c_ref < best_c:
                    best_c = c_ref
                    best_h = h_ref.copy()
                    h = h_ref
                else:
                    h = best_h.copy()
            else:
                h = best_h.copy()

    # Phase 3: Random perturbation + restart
    if time.time() < deadline - 120:
        for _ in range(5):
            if time.time() > deadline - 100:
                break
            eps = 0.05
            h_pert = best_h + eps * rng.randn(n)
            h_pert = project_feasible(h_pert, n)
            h_ref = lbfgs_optimize(h_pert, n, bounds, 5000, 100.0, 150,
                                   min(time.time() + 15, deadline - 100))
            h_ref = project_feasible(h_ref, n)
            c_ref = compute_c5(h_ref, n)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
                h = h_ref

    # Phase 4: Final L-BFGS refinement
    if time.time() < deadline - 60:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > deadline - 50:
                break
            h = lbfgs_optimize(h, n, bounds, alpha, 5000.0, 200,
                               min(time.time() + 20, deadline - 50))
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 5: Pair search
    if time.time() < deadline - 15:
        h = best_h.copy()
        h, best_c = pair_search(h, n, deadline - 3,
                                [0.005, 0.002, 0.001, 0.0005], radius=35, max_sweeps=2)
        best_h = h.copy()

    # Phase 6: Multi-n exploration
    if time.time() < deadline - 30:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 25:
                break
            if n_try == n:
                continue
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            bounds_try = [(0.0, 1.0)] * n_try
            alloc = min(20, deadline - time.time() - 20)
            if alloc < 10:
                break
            h_t = lbfgs_optimize(h_interp, n_try, bounds_try, 50000, 500.0, 200,
                                 time.time() + alloc * 0.7)
            h_t = project_feasible(h_t, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()
                n = n_try

    best_h = project_feasible(best_h, len(best_h))
    final_c = compute_c5(best_h, len(best_h))
    return best_h, final_c, len(best_h)
