    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def run_lbfgs(h, n, deadline, topk=15):

    bounds = [(0.0, 1.0)] * n
    # (alpha, mu) pairs - more aggressive than before
    configs = [(200, 0.5), (800, 2.0), (3000, 10.0), (10000, 50.0),
               (40000, 200.0), (150000, 800.0), (600000, 3000.0),
               (2000000, 10000.0), (8000000, 40000.0), (30000000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(obj_topk, h.copy(), jac=grad_obj_topk,
                         args=(n, alpha, topk, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 250, 'ftol': 1e-17,
                                  'gtol': 1e-15, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new

def run_slsqp(h, n, deadline, alpha=2000000, topk=10, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(x):
        return float(x.sum() - n / 2.0)
    
    def eq_jac(x):
        return np.ones(n)
    
    def obj(x):
        c = compute_c5_vec(x, n)
        k_idx = np.argsort(c)[::-1][:topk]
        c_t = c[k_idx]
        m = c_t.max()
        sh = alpha * (c_t - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    
    def obj_grad(x):
        return grad_smooth_topk(x, n, alpha, topk)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

                    i = rng.randint(nv)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_indices(c, topk_max, epsilon):
    m = c.max()
    active = np.where(c >= m - epsilon)[0]
    if len(active) > topk_max:
        return np.argsort(c)[::-1][:topk_max]
    return active

def grad_smooth_topk(h, n, alpha, active_idx):
    c = compute_c5_vec(h, n)
    c_t = c[active_idx]
    sh = alpha * (c_t - c_t.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(active_idx)):
        if w[idx] < 1e-14:
            continue
        m = active_idx[idx]
        k = m - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:nv+k])
            grad[k:nv+k] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:nv+kk] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:nv+kk])
    return grad

    active = get_active_indices(c, topk, eps)
    c_t = c[active]

    sh = alpha * (c_t - m)
    mx_v = sh.max()
    val = m + (mx_v + np.log(np.sum(np.exp(sh - mx_v)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_vec(x, n)
    active = get_active_indices(c, topk, eps)
    g = grad_smooth_topk(x, n, alpha, active)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

            if alloc < 35:

            h_t, c_t = run_lbfgs(h_interp, n_try, time.time() + alloc, topk=15, eps=0.003)
            if time.time() < deadline - 40:
                h_t, c_t = run_slsqp(h_t, n_try,
                                     min(time.time() + 30, deadline - 35),
                                     alpha=10000000, topk=8, maxiter=2000)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 4: Diverse restarts
    for rtype in range(3):
        if time.time() > deadline - 50:
            break
        if rtype == 0:
            h_s = np.full(best_n, 0.5)
            x = np.linspace(0, 2, best_n, endpoint=False)
            for freq in range(1, 8):
                h_s += rng.uniform(0.12, 0.35) * np.cos(
                    2 * np.pi * freq * x / 2.0 + rng.uniform(0, 6.2832))
        elif rtype == 1:
            h_s = np.zeros(best_n)
            h_s[rng.choice(best_n, size=best_n // 2, replace=False)] = 1.0
        else:
            h_s = best_h + rng.randn(best_n) * 0.05
        h_s = project_feasible(h_s, best_n)
        alloc = min(30, (deadline - time.time() - 20) * 0.4)
        if alloc < 10:
            break
        h_r, c_r = run_lbfgs(h_s, best_n, time.time() + alloc, topk=10, eps=0.002)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r

    # Phase 5: Final SLSQP squeeze
    if time.time() < deadline - 18:
        h_f, c_f = run_slsqp(best_h, best_n, min(time.time() + 30, deadline - 10),
                            alpha=50000000, topk=5, maxiter=3000, eps=0.0005)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 6: Targeted polish
    if time.time() < deadline - 3:
        best_h, best_c = targeted_polish(best_h, best_n, deadline - 1, rng)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
