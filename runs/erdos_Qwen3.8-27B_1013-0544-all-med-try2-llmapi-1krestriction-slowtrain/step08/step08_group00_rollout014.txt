                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c; any_imp = True
                        else:
                            best_h[j] -= d; best_h[i] += d
            if not any_imp:
                break
    return best_h, best_c

def slsqp_finish(h, n, deadline, alpha=5000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(sm_val, h, jac=sm_grad, args=(n, alpha, 0.0),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def try_resolution(h_best, c_best, n_best, n_new, deadline, alloc):
    t_end = time.time() + alloc
    x_old = np.linspace(0, 2, n_best, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_new)
    t1 = min(time.time() + alloc * 0.5, t_end - alloc * 0.4)
    h_t, c_t = lbfgs_phase(h_interp, n_new, t1)
    if time.time() < t_end - 5:
        h_t, c_t = grad_pair_search(h_t, n_new, min(t_end - 3, time.time() + alloc * 0.4), k_cand=25)
    if c_t < c_best:
        return h_t, c_t, n_new
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(650, 0.5)
        n_init = 650
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    # Phase 1: L-BFGS polish
    t1 = min(t0 + 60, deadline - 600)
    best_h, best_c = lbfgs_phase(best_h, best_n, t1)
    # Phase 2: Main gradient-guided pair search
    t2 = min(time.time() + 350, deadline - 250)
    best_h, best_c = grad_pair_search(best_h, best_n, t2, k_cand=40)
    # Phase 3: Try other resolutions
    for n_try in [800, 500, 620]:
        remain = deadline - time.time() - 60
        if remain < 40:
            break
        alloc = min(120, remain * 0.4)
        if alloc < 25:
            break
        h_t, c_t, n_t = try_resolution(best_h, best_c, best_n, n_try, deadline, alloc)
        if c_t < best_c:
            best_h, best_c, best_n = h_t, c_t, n_t
    # Phase 4: More pair search on best
    if time.time() < deadline - 80:
        t4 = min(time.time() + 150, deadline - 80)
        best_h, best_c = grad_pair_search(best_h, best_n, t4, k_cand=30)
    # Phase 5: Micro search
    if time.time() < deadline - 20:
        best_h, best_c = micro_search(best_h, best_n, deadline - 15)
    # Phase 6: Final SLSQP
    if time.time() < deadline - 3:
        h_f, c_f = slsqp_finish(best_h, best_n, deadline - 2, alpha=10000000, maxiter=2000)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
