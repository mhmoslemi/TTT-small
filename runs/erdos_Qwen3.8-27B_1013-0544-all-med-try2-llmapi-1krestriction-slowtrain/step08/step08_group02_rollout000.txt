```python
import numpy as np
import time
from scipy.optimize import minimize

class TE(Exception):
    pass

_CH = {'x': None, 'w': None, 'a': None, 'm': None}

def c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, n):
    h = np.nan_to_num(np.array(h, dtype=float, copy=True))
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo = -h.max() - 1e-12
    hi = 1.0 - h.min() + 1e-12
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        ms = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(ms):
            h[ms[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def svmw(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = float(c.max())
    r = a * (c - cm)
    rm = r.max()
    e = np.exp(r - rm)
    ss = e.sum()
    if ss <= 1e-300:
        ss = 1.0
        e = np.ones_like(e)
    w = e / ss
    val = float(cm + (rm + np.log(ss)) / a)
    return val, w

def fwgrad(x, n, w):
    if n <= 1:
        return np.zeros(n)
    om = 1.0 - x
    P = np.ascontiguousarray(w[n - 1:2 * n - 1])
    Q = np.zeros(n)
    if n > 1:
        Q[1:] = w[n - 2::-1]
    t1 = np.convolve(P, om, mode='full')[:n]
    t2 = np.correlate(x, P, mode='full')[n - 1:2 * n - 1]
    t3 = np.correlate(om, Q, mode='full')[n - 1:2 * n - 1]
    t4 = np.convolve(Q, x, mode='full')[:n]
    return t1 - t2 + t3 - t4

def lbf(x, n, a, mu, dl):
    if time.time() > dl:
        raise TE
    v, w = svmw(x, n, a)
    _CH['x'] = x.copy()
    _CH['w'] = w
    _CH['a'] = a
    _CH['m'] = mu
    s = x.sum() - n / 2.0
    return float(v + mu * s * s)

def lbj(x, n, a, mu, dl):
    if time.time() > dl:
        raise TE
    cx = _CH['x']
    if (cx is not None and _CH['a'] == a and _CH['m'] == mu and
            cx.size == x.size and np.array_equal(x, cx)):
        w = _CH['w']
    else:
        _, w = svmw(x, n, a)
        _CH['x'] = x.copy()
        _CH['w'] = w
        _CH['a'] = a
        _CH['m'] = mu
    g = fwgrad(x, n, w)
    s = x.sum() - n / 2.0
    return g + 2.0 * mu * s

def lbfgs(h, n, dl, cfg=None):
    h = proj(h, n)
    best = c5(h, n)
    bh = h.copy()
    if cfg is None:
        cfg = [(10, 1e2, 80), (50, 1e3, 120), (200, 1e4, 160),
               (800, 1e5, 220), (3000, 1e6, 300), (12000, 1e6, 400),
               (50000, 1e6, 600)]
    bounds = [(0.0, 1.0)] * n
    scale = 1.0 if n <= 500 else max(0.35, 500.0 / float(n))
    for a, mu, it in cfg:
        if time.time() > dl - 5:
            break
        it2 = max(20, int(it * scale))
        try:
            res = minimize(lbf, h, jac=lbj, args=(n, a, mu, dl),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': it2, 'ftol': 1e-18,
                                    'gtol': 1e-16, 'maxcor': 50})
            hn = proj(res.x, n)
            cn = c5(hn, n)
            if cn < best:
                best = cn
                bh = hn.copy()
                h = hn
            elif cn < best + 0.02 and np.all(np.isfinite(hn)):
                h = hn
        except TE:
            pass
        except Exception:
            pass
    return proj(bh, n), best

def glag(x, n, k):
    g = np.zeros(n)
    if k >= 0:
        l = n - k
        if 0 < l <= n and k < n:
            g[k:] += 1.0 - x[:l]
            g[:l] -= x[k:]
    else:
        l = -k
        m = n - l
        if 0 < m <= n and l < n:
            g[:m] += 1.0 - x[l:]
            g[l:] -= x[:m]
    return g

def subgrad(h, n, dl, iters=200):
    h = proj(h, n)
    best = c5(h, n)
    besth = h.copy()
    step = 0.04
    fail = 0
    for _ in range(iters):
        if time.time() > dl:
            break
        cv = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(cv))
        k = idx - (n - 1)
        g = glag(h, n, k)
        g -= g.mean()
        nn = np.linalg.norm(g)
        if nn < 1e-12:
            break
        d = g / nn
        im = False
        s = step
        for __ in range(11):
            cand = proj(h - s * d, n)
            val = c5(cand, n)
            if val < best - 1e-13:
                h = cand
                best = val
                besth = h.copy()
                im = True
                break
            s *= 0.5
        if im:
            step = min(0.15, step * 1.05)
            fail = 0
        else:
            step *= 0.6
            fail += 1
            if fail > 12:
                break
    return proj(besth, n), best

def smf(x, n, a, dl):
    if time.time() > dl:
        raise TE
    v, w = svmw(x, n, a)
    _CH['x'] = x.copy()
    _CH['w'] = w
    _CH['a'] = a
    _CH['m'] = 0.0
    return float(v)

def smj(x, n, a, dl):
    if time.time() > dl:
        raise TE
    cx = _CH['x']
    if (cx is not None and _CH['a'] == a and _CH['m'] == 0.0 and
            cx.size == x.size and np.array_equal(x, cx)):
        w = _CH['w']
    else:
        _, w = svmw(x, n, a)
        _CH['x'] = x.copy()
        _CH['w'] = w
        _CH['a'] = a
        _CH['m'] = 0.0
    return fwgrad(x, n, w)

def eqf(x, n, *args):
    return float(x.sum() - n / 2.0)

def eqj(x, n, *args):
    return np.ones(n)

def slsqp_smooth(h, n, dl, a=50000, it=80):
    h = proj(h, n)
    best = c5(h, n)
    besth = h.copy()
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj, 'args': (n,)}]
    try:
        res = minimize(smf, h, args=(n, a, dl), method='SLSQP',
                       jac=smj, bounds=[(0.0, 1.0)] * n,
                       constraints=cons,
                       options={'maxiter': it, 'ftol': 1e-12})
        hn = proj(res.x, n)
        cn = c5(hn, n)
        if cn < best:
            best = cn
            besth = hn.copy()
    except TE:
        pass
    except Exception:
        pass
    return proj(besth, n), best

def fstart(n, rng, m=8):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for q in range(1, m + 1):
        amp = rng.uniform(0.0, 0.35 / max(1.0, np.sqrt(q)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * q * x + ph)
    return proj(h, n)

def rstart(n, rng):
    h = np.zeros(n)
    half = n // 2
    if half > 0:
        idx = rng.choice(n, half, replace=False)
        h[idx] = 1.0
    if n % 2:
        z = np.where(h == 0.0)[0]
        if len(z):
            h[z[0]] = 0.5
    return proj(h, n)

def diverse(h, n, cb, dl, rng):
    cfg = [(100, 1e4, 120), (1000, 1e5, 180), (5000, 1e6, 250)]
    starts = (fstart(n, rng, 12), fstart(n, rng, 5), rstart(n, rng))
    for s in starts:
        if time.time() > dl - 15:
            break
        hn, cn = lbfgs(s, n, min(dl - 15, time.time() + 65), cfg)
        if cn < cb:
            h = hn
            cb = cn
    return h, cb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = fstart(n0, rng, 12)

    if n0 < 20:
        n0 = 500
        h0 = fstart(n0, rng, 12)

    h1 = proj(h0, n0)
    c1 = c5(h1, n0)

    main_dl = min(dl - 380.0, t0 + 700.0)
    h1, c1 = lbfgs(h1, n0, main_dl)

    if time.time() < dl - 300.0:
        h1, c1 = subgrad(h1, n0, min(dl - 250.0, time.time() + 55.0), 140)

    if time.time() < dl - 180.0:
        h1, c1 = diverse(h1, n0, c1, min(dl - 160.0, time.time() + 190.0), rng)

    if time.time() < dl - 110.0:
        h1, c1 = slsqp_smooth(h1, n0, min(dl - 90.0, time.time() + 45.0), 60000, 90)

    if time.time() < dl - 50.0:
        h1, c1 = subgrad(h1, n0, min(dl - 30.0, time.time() + 25.0), 90)

    rem = dl - time.time()
    if rem > 25.0:
        if rem > 220.0:
            cfg2 = [(5000, 1e6, 300), (30000, 1e6, 650), (120000, 1e6, 1100)]
        elif rem > 90.0:
            cfg2 = [(10000, 1e6, 350), (80000, 1e6, 700)]
        else:
            cfg2 = [(30000, 1e6, 250), (150000, 1e6, 400)]
        h1, c1 = lbfgs(h1, n0, dl - 6.0, cfg2)

    h1 = proj(h1, n0)
    c1 = c5(h1, n0)
    return h1, float(c1), int(n0)
```
