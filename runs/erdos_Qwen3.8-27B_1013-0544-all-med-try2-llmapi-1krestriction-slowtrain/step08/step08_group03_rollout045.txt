            if tot < 1e-14:

def grad_guided_swap(h, n, deadline, m=30, max_rounds=200):
    """Use gradient to guide pair swaps between high-grad and low-grad coordinates"""
    best_c = compute_c5(h, n)
    h = h.copy()
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        for _ in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            # Compute gradient
            alpha = 200.0
            c = compute_c5_vec(h, n)
            mval = c.max()
            sh = alpha * (c - mval)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            dx = 2.0 / n
            # Compute gradient using top weights
            top_k = min(50, n)
            top_idx = np.argsort(w)[-top_k:]
            grad = np.zeros(n)
            for i in top_idx:
                if w[i] < 1e-14:
                    continue
                k = i - (n - 1)
                grad += w[i] * dx * grad_single(h, n, k)
            
            # Find best pair: high grad (to increase) and low grad (to decrease)
            grad_sorted = np.argsort(grad)
            # Coordinates to increase: those with highest gradient
            # Coordinates to decrease: those with lowest gradient
            inc_candidates = grad_sorted[-m:]  # highest gradient
            dec_candidates = grad_sorted[:m]   # lowest gradient
            
            best_improve = 1e-15
            best_ii = -1
            best_jj = -1
            best_dir = 1  # 1 means increase ii, decrease jj
            
            for ii in inc_candidates:
                if h[ii] + step > 1.0:
                    continue
                for jj in dec_candidates:
                    if h[jj] - step < 0.0:
                        continue
                    if ii == jj:
                        continue
                    # Try increasing h[ii] by step, decreasing h[jj] by step
                    h[ii] += step
                    h[jj] -= step
                    c_new = compute_c5(h, n)
                    h[ii] -= step
                    h[jj] += step
                    if c_new < best_c - best_improve:
                        best_improve = best_c - c_new
                        best_ii = ii
                        best_jj = jj
                        best_dir = 1
                    # Also try the reverse
            if best_ii >= 0:
                h[best_ii] += step
                h[best_jj] -= step
                best_c = compute_c5(h, n)
            else:
                break
    return h, best_c

            improved = False
            for k_idx in top_idx:
                k = k_idx - (n - 1)
                nv = n - abs(k)
                if nv < 2:
                    continue
                if k >= 0:
                    prod = h[:nv] * (1.0 - h[k:k+nv])
                else:
                    kk = -k
                    prod = h[kk:kk+nv] * (1.0 - h[:nv])
                total = prod.sum()
                if total <= 1e-10:
                    continue
                # Find position where prod is largest - that's the biggest contributor
                # Try to fix it by adjusting h
                biggest = np.argmax(prod)
                if k >= 0:
                    i_src = k + biggest  # position where h is low (1-h is high)
                    i_dst = biggest      # position where h is high
                else:
                    kk = -k
                    i_src = biggest
                    i_dst = kk + biggest
                # Move mass: decrease h[i_dst], increase h[i_src]
                if h[i_dst] - step >= 0.0 and h[i_src] + step <= 1.0:
                    h[i_dst] -= step
                    h[i_src] += step
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i_dst] += step
                        h[i_src] -= step
                if not improved:
                    break
            if not improved:
                break
    return h, best_c

def pair_polish_fine(h, n, deadline, radius=80):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for d in deltas:
        improved = True
        max_passes = 10
        for _ in range(max_passes):
            if not improved or time.time() > deadline:
                break
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if time.time() > deadline:
                return h, best_c
    return h, best_c

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = random_binary_start(n, rng)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: L-BFGS-B with multiple alphas
    t1 = min(t0 + 180, deadline - 300)
    for alpha in [200, 500, 2000, 8000]:
        if time.time() > t1:
            break
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(best_n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Gradient-guided pair swap
    if time.time() < deadline - 250:
        t2 = min(time.time() + 200, deadline - 180)
        best_h, best_c = grad_guided_swap(best_h, best_n, t2, m=30)

    # Phase 3: Shift attack
    if time.time() < deadline - 150:
        t3 = min(time.time() + 80, deadline - 100)
        best_h, best_c = shift_attack(best_h, best_n, t3, n_top=5)

    # Phase 4: Try different n values
    for n_try in [400, 600, 700, 800]:
        if time.time() > deadline - 120:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc_check = deadline - time.time() - 80
        if alloc_check < 15:
            continue
        bounds = [(0.0, 1.0)] * n_try
        try:
            res = minimize(lbfgs_obj, h_interp, jac=lbfgs_grad,
                          args=(n_try, 3000), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-15})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass

    # Phase 5: Gradient-guided swap again on best
    if time.time() < deadline - 60:
        t5 = min(time.time() + 50, deadline - 40)
        best_h, best_c = grad_guided_swap(best_h, best_n, t5, m=20)

    # Phase 6: Shift attack
    if time.time() < deadline - 30:
        t6 = min(time.time() + 25, deadline - 15)
        best_h, best_c = shift_attack(best_h, best_n, t6, n_top=4)

    # Phase 7: Fine pair polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_polish_fine(best_h, best_n, deadline - 1, radius=60)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
