def project_feasible(y, n):
    target = 0.5*n
    y = np.asarray(y, dtype=float)
    if y.shape != (n,): y = np.resize(y,n)
    if np.all(y >= -1e-12) and np.all(y <= 1+1e-12) and abs(y.sum()-target) < 1e-10:
        return np.clip(y,0,1)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):
       mid = 0.5*(lo+hi)
       s = np.clip(y-mid,0,1).sum()
       if s > target: lo = mid
       else: hi = mid
    z = np.clip(y - 0.5*(lo+hi), 0,1)
    diff = target - z.sum()
    if diff > 0:
       idx = np.where(1.0 - z > 1e-12)[0]
       if idx.size:
          amt = min(diff, (1-z[idx]).sum())
          room = 1-z[idx]
          z[idx] += amt * room / room.sum()
    elif diff < 0:
       idx = np.where(z > 1e-12)[0]
       if idx.size:
          amt = min(-diff, z[idx].sum())
          z[idx] -= amt * z[idx] / z[idx].sum()
    return np.clip(z,0,1)

def project_feasible(y, n):
    t = 0.5 * n
    y = np.asarray(y, dtype=float)
    if y.size != n:
        y = np.resize(y, n)
    if abs(y.sum() - t) < 1e-12 and float(y.min()) >= -1e-12 and float(y.max()) <= 1.0 + 1e-12:
        return np.clip(y, 0.0, 1.0)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - z.sum()
    if d > 0:
        idx = np.where(z < 1.0 - 1e-12)[0]
        if idx.size:
            r = 1.0 - z[idx]
            z[idx] += min(d, r.sum()) * r / r.sum()
    elif d < 0:
        idx = np.where(z > 1e-12)[0]
        if idx.size:
            z[idx] -= min(-d, z[idx].sum()) * z[idx] / z[idx].sum()

    if nn == n:
        return h.copy()

    x2 = np.linspace(0.0, 2.0, nn, endpoint=False) + 2.0 / nn
    return project_feasible(np.interp(x2, x1, h, left=float(h[0]), right=float(h[-1])), nn)

    z -= z.max()

def lp_line_search(hdir, h0, n, deadline):

def lp_polish(h,n,deadline,tau=0.05,active=40,max_fail=20):

            break
        val, grad = smooth_value_grad(h, n, alpha)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-12:
            break
        st = min(1.0, step / max(gn, 1e-4))
        ok = False
        for _ in range(24):
            cand = project_feasible(h - st * grad, n)
            v2 = smooth_value(cand, n, alpha)
            if v2 < val:
                h = cand
                val = v2
                ok = True
                break
            st *= 0.5
            if st < 1e-9:
                break
        if not ok:
            for _ in range(3):
                cand = project_feasible(h + 0.02 * rng.randn(n), n)
                v2 = smooth_value(cand, n, alpha)
                if v2 < val - 1e-12:
                    h = cand
                    val = v2
                    ok = True
                    break
            if not ok:
                bad += 1
                if bad > 20:
                    break
        else:
            bad = 0
            last = time.time()
            c = compute_c5(h, n)
            if c < bc - 1e-15:
                bc = c
                bh = h.copy()
    c = compute_c5(h, n)
    if c < bc:
        bh = h.copy()
        bc = c
    return bh, bc

def lp_propose(h, n, tau, n_active):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, 'full') * dx
    L = 2 * n - 1
    k = min(L, max(1, int(n_active)))
    top = np.argsort(g)[::-1][:k]
    idx = np.unique(np.concatenate((top, L - 1 - top)))
    m = len(idx)
    G = np.zeros((m, n), dtype=float)
    v = 1.0 - h
    for r, ip in enumerate(idx):
        s_off = int(ip - (n - 1))
        a = max(0, -s_off)
        b = min(n, n - s_off)
        if b > a:
            G[r, a:b] = v[a + s_off:b + s_off]
        a = max(0, s_off)
        b = min(n, s_off + n)
        if b > a:
            G[r, a:b] -= h[a - s_off:b - s_off]
    G *= dx
    f = g[idx]
    rhs = -f + G.dot(h)
    A = np.empty((m, n + 1), dtype=float)
    A[:, :n] = G
    A[:, n] = -1.0
    cobj = np.zeros(n + 1)
    cobj[-1] = 1.0
    target = 0.5 * n
    lb = np.maximum(0.0, h - tau)
    ub = np.minimum(1.0, h + tau)
    if lb.sum() > target + 1e-8 or ub.sum() < target - 1e-8:
        return None
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = list(zip(lb, ub))
    bounds.append((0.0, 1.0))
    try:
        res = linprog(cobj, A_ub=A, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status == 0 or res.success:
        return np.clip(res.x[:n], 0.0, 1.0)
    return None

def lp_line_search(hdir, h0, n, deadline):
    bh = h0.copy()
    bc = compute_c5(h0, n)
    d = hdir - h0
    for lam in (1.0, 0.66, 0.44, 0.31, 0.22, 0.16, 0.11, 0.08, 0.056, 0.04):
        if time.time() > deadline:
            break
        cand = project_feasible(h0 + float(lam) * d, n)
        c = compute_c5(cand, n)
        if c < bc - 1e-14:
            bc = c
            bh = cand.copy()
    return bh, bc

def lp_polish(h, n, deadline, tau=0.05, active=40, max_fail=20):
    h = project_feasible(h, n)
    bh = h.copy()
    bc = compute_c5(h, n)
    fail = 0
    last = time.time()
    it = 0
    cur_tau = float(tau)
    while time.time() < deadline:
        it += 1
        if it > 3000 or time.time() - last > 90.0:
            break
        hd = lp_propose(h, n, cur_tau, active)
        if hd is None:
            fail += 1
            cur_tau *= 0.7
            if fail > max_fail or cur_tau < 0.005:
                break
            continue
        h2, c2 = lp_line_search(hd, h, n, deadline)
        if c2 < bc - 1e-14:
            h = h2.copy()
            bh = h2.copy()
            bc = c2
            fail = 0
            last = time.time()
            cur_tau = float(tau)
        else:
            fail += 1
            cur_tau *= 0.7
            if fail > max_fail or cur_tau < 0.005:
                break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    best_h = None
    best_n = -1
    best_c = 1e9
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = hp.size
        if n0 > 0:
            hp = project_feasible(hp, n0)
            c = compute_c5(hp, n0)
            best_h = hp
            best_n = n0
            best_c = c
    except Exception:
        pass
    if best_h is None:
        n0 = 250
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_n = n0
        best_c = compute_c5(best_h, n0)

    p1 = t0 + 0.16 * budget_s
    h2, c2 = lp_polish(best_h, best_n, min(deadline, p1), tau=0.05, active=30, max_fail=25)
    if c2 < best_c:
        best_h, best_n, best_c = h2, best_n, c2

    p2 = t0 + 0.32 * budget_s
    h2, c2 = lp_polish(best_h, best_n, min(deadline, p2), tau=0.1, active=50, max_fail=20)
    if c2 < best_c:
        best_h, best_n, best_c = h2, best_n, c2

    p3 = t0 + 0.50 * budget_s
    h2, c2 = lp_polish(best_h, best_n, min(deadline, p3), tau=0.2, active=80, max_fail=18)
    if c2 < best_c:
        best_h, best_n, best_c = h2, best_n, c2

    p4 = t0 + 0.62 * budget_s
    h2, c2 = smooth_gradient_polish(best_h, best_n, 5000.0, min(deadline, p4), 2000, 0.08, seed=seed)
    if c2 < best_c:
        best_h, best_n, best_c = h2, best_n, c2

    if best_n % 2 == 0:
        hb = binarize(best_h, best_n)
        cb = compute_c5(hb, best_n)
        if cb < best_c:
            best_h, best_n, best_c = hb, best_n, cb

    p5 = t0 + 0.82 * budget_s
    if best_n % 2 == 0 and p5 - time.time() > 60.0:
        hr = random_binary(best_n, seed + 7)
        h2, c2 = lp_polish(hr, best_n, min(p5, time.time() + 50.0), tau=0.05, active=30, max_fail=18)
        if c2 < best_c:
            best_h, best_n, best_c = h2, best_n, c2

    for nn in (300, 256, 400, 320, 200):
        rem = p5 - time.time()
        if rem < 45.0:
            break
        if nn == best_n:
            continue
        alloc = min(80.0, rem * 0.9)
        hc = interpolate_h(best_h, best_n, nn)
        h2, c2 = lp_polish(hc, nn, min(p5, time.time() + 0.55 * alloc), tau=0.06, active=25, max_fail=18)
        if c2 < best_c:
            best_h, best_n, best_c = h2, nn, c2
        rem = p5 - time.time()
        if rem > 25.0:
            h3, c3 = smooth_gradient_polish(h2, nn, 3000.0, min(p5, time.time() + 0.3 * alloc), 1000, 0.05, seed=seed + nn)
            if c3 < best_c:
                best_h, best_n, best_c = h3, nn, c3

    rem = deadline - time.time()
    if rem > 15.0:
        h2, c2 = lp_polish(best_h, best_n, time.time() + 0.5 * rem, tau=0.025, active=60, max_fail=20)
        if c2 < best_c:
            best_h, best_n, best_c = h2, best_n, c2
        rem = deadline - time.time()
        if rem > 15.0:
            h2, c2 = smooth_gradient_polish(best_h, best_n, 20000.0, time.time() + 0.7 * rem, 1500, 0.03, seed=seed + 123)
            if c2 < best_c:
                best_h, best_n, best_c = h2, best_n, c2

    if best_n % 2 == 0 and time.time() < deadline - 2.0:
        hb = binarize(best_h, best_n)
        for w in (0.25, 0.5, 0.75):
            hc = project_feasible((1.0 - w) * best_h + w * hb, best_n)
            cc = compute_c5(hc, best_n)
            if cc < best_c:
                best_h, best_c = hc, cc
                break

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
