import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def grad_smooth_max(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(top_k):
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single(h, n, k)
    return grad

def nesterov_pgd(h, n, deadline, top_k=50, alpha=100.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    momentum = 0.9
    y = h.copy()
    prev_h = h.copy()
    stall = 0
    while time.time() < deadline:
        g = grad_smooth_max(y, n, top_k, alpha)
        y_new = y + momentum * (y - prev_h) - step * g
        y_new = project_feasible(y_new, n)
        c_new = compute_c5(y_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = y_new.copy()
            stall = 0
        else:
            stall += 1
            # try smaller step
            for s_factor in [0.5, 0.25, 0.1]:
                y_try = project_feasible(y - s_factor * step * g, n)
                c_try = compute_c5(y_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = y_try.copy()
                    y_new = y_try
                    stall = 0
                    break
        prev_h = y
        y = y_new
        if stall > 100:
            break
    return best_h, best_c

    c = compute_c5_vec(h, n)

    ks = np.arange(-(n-1), n)  # shape (2n-1,)

    imk = i - ks
    jpk = j + ks
    jmk = j - ks

        return np.where((idx >= 0) & (idx < n), idx, 0)

        vals = np.zeros(len(idx))

        return vals

    g = (1.0 - h_ipk - h_imk) - (1.0 - h_jpk - h_jmk)

    g = h_jpk + h_jmk - h_ipk - h_imk
    
    # Minimize max_k(c + dx*d*g) over d in [d_min, d_max]
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    
    if d_max - d_min < 1e-15:
        return 0.0
    
    # Ternary search (vectorized inner)
    lo, hi = d_min, d_max
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c + dx * m1 * g)
        f2 = np.max(c + dx * m2 * g)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

Actually, a simpler and effective approach:
1. Compute the smooth-max gradient
2. Do a few steps of Nesterov PGD
3. Then do coordinate descent: for each i (in random order), find the best j to transfer to/from

For coordinate descent on i: fix all h[j] for j≠i, find the best h[i] value.
The objective as a function of h[i] = v is:
f(v) = max_k (c(k) + (v - h[i]) * dx * g_i(k))
where g_i(k) is the partial derivative of c(k) with respect to h[i].

This is the same 1D convex problem! And we can solve it for each i independently.

g_i(k) = (1 - h[i+k]) - h[i-k] (same as a_i(k) above)

So for each i:
- Compute a_i(k) for all k (this is O(n))
- Find v that minimizes max_k (c(k) + (v - h[i]) * dx * a_i(k))
- Subject to constraints: h[i] = v, and sum(h) = n/2 (so we'd need to adjust another coordinate)

The constraint coupling makes this harder. If I only change h[i], I violate the sum constraint. So I need to change two coordinates.

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_vec(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def grad_smooth_max(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(top_k):
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single_vec(h, n, k)
    return grad

# Precompute the lag indices for vectorized g computation
def compute_g_for_pair(h, n, i, j):
    """Compute g(k) for all k, where g is the direction of transfer i<-j"""
    ks = np.arange(-(n-1), n)
    ipk = i + ks
    imk = i - ks
    jpk = j + ks
    jmk = j - ks
    
    mask_ipk = (ipk >= 0) & (ipk < n)
    mask_imk = (imk >= 0) & (imk < n)
    mask_jpk = (jpk >= 0) & (jpk < n)
    mask_jmk = (jmk >= 0) & (jmk < n)
    
    h_ipk = np.zeros(2*n-1)
    h_ipk[mask_ipk] = h[ipk[mask_ipk]]
    h_imk = np.zeros(2*n-1)
    h_imk[mask_imk] = h[imk[mask_imk]]
    h_jpk = np.zeros(2*n-1)
    h_jpk[mask_jpk] = h[jpk[mask_jpk]]
    h_jmk = np.zeros(2*n-1)
    h_jmk[mask_jmk] = h[jmk[mask_jmk]]
    
    # g(k) = [(1-h[i+k]) - h[i-k]] - [(1-h[j+k]) - h[j-k]]
    #       = h[j+k] - h[j-k] - h[i+k] + h[i-k]
    return h_jpk - h_jmk - h_ipk + h_imk

def exact_pair_d(h, n, i, j, c, g):
    """Find optimal d for h[i]+=d, h[j]-=d given precomputed c and g"""
    dx = 2.0 / n
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    if d_max - d_min < 1e-12:
        return 0.0
    # Ternary search on f(d) = max(c + dx*d*g)
    lo, hi = d_min, d_max
    for _ in range(25):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c + dx * m1 * g) < np.max(c + dx * m2 * g):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def smart_pair_optimize(h, n, deadline, top_k=60, alpha=100.0):
    """Pair optimization using gradient-guided selection"""
    best_c = compute_c5(h, n)
    improved = True
    sweep = 0
    while improved and time.time() < deadline:
        improved = False
        sweep += 1
        # Compute gradient to guide pair selection
        g = grad_smooth_max(h, n, top_k, alpha)
        c = compute_c5_vec(h, n)
        # Sort by gradient: high g wants to decrease, low g wants to increase
        order = np.argsort(g)
        # Try pairs: (low g index, high g index) -> transfer from high to low
        n_try = min(5000, n * n // 2)
        rng = np.random.RandomState(sweep)
        for _ in range(n_try):
            if time.time() > deadline:
                return h, best_c
            # Pick i from low-gradient side, j from high-gradient side
            i = order[rng.randint(0, n // 2)]
            j = order[rng.randint(n // 2, n)]
            if i == j:
                continue
            gv = compute_g_for_pair(h, n, i, j)
            d = exact_pair_d(h, n, i, j, c, gv)
            if abs(d) < 1e-10:
                continue
            h[i] += d
            h[j] -= d
            # Check feasibility
            if h[i] < -1e-10 or h[i] > 1.0 + 1e-10 or h[j] < -1e-10 or h[j] > 1.0 + 1e-10:
                h[i] -= d
                h[j] += d
                continue
            h[i] = np.clip(h[i], 0, 1)
            h[j] = np.clip(h[j], 0, 1)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                c = compute_c5_vec(h, n)
                improved = True
            else:
                h[i] -= d
                h[j] += d
    return h, best_c

def nesterov_pgd(h, n, deadline, top_k=50, alpha=100.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.03
    momentum = 0.7
    y = h.copy()
    prev_y = h.copy()
    stall = 0

        y_new = y + momentum * (y - prev_y) - step * g
        y_new = project_feasible(y_new, n)
        c_new = compute_c5(y_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = y_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                step *= 0.8
                if step < 0.001:
                    break
        prev_y = y
        y = y_new
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def precompute_v(h, n):
    """Precompute v_p for all p: v_p(k) = h[p+k] - h[p-k]"""
    # v has shape (n, 2n-1)
    # v[p, k_idx] = h[p+k] - h[p-k] where k = k_idx - (n-1)

    ks = np.arange(-(n-1), n)
    for p in range(n):

        pmk = p - ks
        v[p, (pk >= 0) & (pk < n)] = h[pk[(pk >= 0) & (pk < n)]]
        v[p, (pmk >= 0) & (pmk < n)] -= h[pmk[(pmk >= 0) & (pmk < n)]]
    return v

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_vec(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def grad_smooth_max(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(top_k):
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single_vec(h, n, k)
    return grad

def compute_g_for_pair(h, n, i, j):
    ks = np.arange(-(n - 1), n)
    ipk = i + ks
    imk = i - ks
    jpk = j + ks
    jmk = j - ks
    m1 = (ipk >= 0) & (ipk < n)
    m2 = (imk >= 0) & (imk < n)
    m3 = (jpk >= 0) & (jpk < n)

    cl = np.clip(jmk, 0, n - 1)
    return np.where(m3, h[ck], 0.0) - np.where(m4, h[cl], 0.0) - np.where(m1, h[ci], 0.0) + np.where(m2, h[cj], 0.0)

def exact_pair_d(h, n, i, j, c, g):
