```python
def grad_c5_k(h, n, k):

    return grad
```

```python
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5_k(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:nv + k] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 500.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(120, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(len(x))
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_c5_k(x, n, k)
    grad += 1000.0 * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alphas=None, max_iter=500):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def pgd_optimize(h, n, deadline, alpha=2000.0, top_k=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.02
    prev_h = None
    prev_g = None
    stall = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        top_idx = np.argsort(w)[-top_k:]
        g = np.zeros(n)
        for i in top_idx:
            if w[i] < 1e-12:
                continue
            k = i - (n - 1)
            g += w[i] * dx * grad_c5_k(h, n, k)
        g += 200.0 * (h.sum() - n / 2.0)
        if prev_h is not None:
            s = h - prev_h
            y = g - prev_g
            sy = s @ y
            ss = s @ s
            if sy > 1e-18:
                step = ss / sy
                step = float(np.clip(step, 0.001, 0.08))
        prev_h = h.copy()
        prev_g = g.copy()
        accepted = False
        for _ in range(15):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-12:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.5
        if not accepted:
            stall += 5
        if stall > 100:
            break
    return best_h, best_c

def targeted_local_search(h, n, deadline):
    """Focus on top lags: find best pair move that reduces the max lag"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(best_h, n)
            top_idx = np.argsort(c)[-10:]
            top_lags = top_idx - (n - 1)
            for _ in range(3000):
                if time.time() > deadline:
                    return best_h, best_c
                lag = top_lags[rng.randint(0, len(top_lags))]
                nv = n - abs(lag)
                i = rng.randint(0, nv)
                if d > 0:
                    j = i + lag
                else:
                    j = i - lag
                j = int(j)
                if j < 0 or j >= n:
                    continue
                if best_h[i] >= d and best_h[j] <= 1.0 - d:
                    best_h[i] -= d
                    best_h[j] += d
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                if best_h[j] >= d and best_h[i] <= 1.0 - d:
                    best_h[j] -= d
                    best_h[i] += d
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        best_h[j] += d
                        best_h[i] -= d
    return best_h, best_c

def golden_start(n, rng=None):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    offset = rng.random() if rng is not None else 0.3
    positions = np.floor((np.arange(half) + offset) * phi).astype(int) % n
    used = set()
    for p in positions:
        if p not in used:
            h[p] = 1.0
            used.add(p)
    if len(used) < half:
        for i in range(n):
            if i not in used and len(used) < half:
                h[i] = 1.0
                used.add(i)
    return project_feasible(h, n)

def sine_start(n, freq, phase):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 * (1.0 + np.sin(freq * x * np.pi + phase))
    return project_feasible(h, n)

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    t1 = min(t0 + 250, deadline - 350)

    # Phase 3: L-BFGS again with very high alpha
    if time.time() < deadline - 200:

                                         alphas=[100000, 500000], max_iter=400)

    # Phase 4: Targeted local search on top lags
    if time.time() < deadline - 80:
        t4 = min(time.time() + 80, deadline - 40)

        x_old = np.linspace(0, 2, best_n, endpoint=False)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
