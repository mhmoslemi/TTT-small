def smooth_gradient(h,c,alpha,n):

    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    sw = w.sum()
    if sw <=0: w = np.ones_like(c)/len(c)
    else: w /= sw
    g = np.empty(n)

def linear_min_point(g,target):
    x = np.zeros(g.shape, dtype=float)
    order = np.argsort(g)
    k = int(np.floor(target + 1e-12))
    if k > 0:
        x[order[:k]] = 1.0
    frac = target - k
    if frac > 1e-12 and k < len(g):

    return x

import numpy as np

def compute_c_vec(h,n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def ensure_feasible(h,target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(20):
        r = target - h.sum()
        if abs(r) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += r / mask.sum()
        else:
            idx = int(np.argmin(h)) if r > 0.0 else int(np.argmax(h))
            h[idx] += r
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_gradient(h,c,alpha,n):
    m = float(c.max())
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    sw = w.sum()
    if sw > 0.0:
        w = w / sw
    else:
        w = np.full(c.shape, 1.0 / c.size)
    g = w[n-1] * (1.0 - 2.0 * h)
    omh = 1.0 - h
    for q in range(1, n):
        a = w[n-1+q]
        b = w[n-1-q]
        if a == 0.0 and b == 0.0:
            continue
        L = n - q
        if a != 0.0:
            g[:L] += a * omh[q:q+L]
            g[q:] += -a * h[0:L]
        if b != 0.0:
            g[q:] += b * omh[0:L]
            g[:L] += -b * h[q:q+L]
    return g

def compute_c_vec(h,n):
    return np.correlate(h, 1.0-h, mode="full")*(2.0/n)

def ensure_feasible(h,target):
    h=np.asarray(h,dtype=float).copy(); np.clip(h,0,1,out=h)
    for _ in range(20):
        r=target-h.sum()
        if abs(r)<1e-13: break
        m=(h>1e-12)&(h<1-1e-12)
        if m.any(): h[m]+=r/m.sum()
        else:
            i=int(np.argmin(h)) if r>0 else int(np.argmax(h)); h[i]+=r
        np.clip(h,0,1,out=h)
    return h

def smooth_gradient(h,c,alpha,n):
    m=float(c.max()); s=alpha*(c-m); s-=s.max(); w=np.exp(s)
    sw=w.sum(); w=w/sw if sw>0 else np.full(c.shape,1.0/c.size)
    g=w[n-1]*(1.0-2.0*h); omh=1.0-h
    for q in range(1,n):
        a=w[n-1+q]; b=w[n-1-q]
        if a==0.0 and b==0.0: continue
        L=n-q
        if a:
            g[:L]+=a*omh[q:q+L]; g[q:]+=-a*h[0:L]
        if b:
            g[q:]+=b*omh[0:L]; g[:L]+=-b*h[q:q+L]
    return g

def linear_min_point(g,target):
    x=np.zeros(g.shape); o=np.argsort(g)
    k=int(np.floor(target+1e-12))
    if k>0: x[o[:k]]=1.0
    f=target-k
    if f>1e-12 and k<g.size: x[o[k]]=f
    return x

def add_single_delta(h,p,dg,n,dt):

import numpy as np
import time

def compute_c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def ensure_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(20):
        r = target - h.sum()
        if abs(r) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += r / m.sum()
        else:
            i = int(np.argmin(h)) if r > 0.0 else int(np.argmax(h))
            h[i] += r
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_gradient(h, c, alpha, n):
    m = float(c.max())
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    sw = w.sum()
    if sw > 0.0:
        w = w / sw
    else:
        w = np.full(c.shape, 1.0 / c.size)
    g = w[n - 1] * (1.0 - 2.0 * h)
    omh = 1.0 - h
    for q in range(1, n):
        a = w[n - 1 + q]
        b = w[n - 1 - q]
        if a == 0.0 and b == 0.0:
            continue
        L = n - q
        if a != 0.0:
            g[:L] += a * omh[q:q + L]
            g[q:] += -a * h[0:L]
        if b != 0.0:
            g[q:] += b * omh[0:L]
            g[:L] += -b * h[q:q + L]
    return g

def linear_min_point(g, target):
    x = np.zeros(g.shape, dtype=float)
    o = np.argsort(g)
    k = int(np.floor(target + 1e-12))
    if k > 0:
        x[o[:k]] = 1.0
    f = target - k
    if f > 1e-12 and k < g.size:
        x[o[k]] = f
    return x

def add_single_delta(h, p, dg, n, dt):
    if dg == 0.0:
        return
    dt[n - 1] += dg * (1.0 - 2.0 * h[p]) - dg * dg
    fu = n - 1 - p
    if fu > 0:
        y = h[p + 1:p + fu + 1]
        dt[n:n + fu] += dg * (1.0 - y)
        dt[n - fu - 1:n - 1] += -dg * y[::-1]
    ps = p
    if ps > 0:
        dt[n:n + ps] += -dg * h[0:ps][::-1]
        dt[n - ps - 1:n - 1] += dg * (1.0 - h[0:ps])

def pair_value(h, c, i, j, d, n, dt, tm):
    dt.fill(0.0)
    add_single_delta(h, i, d, n, dt)
    add_single_delta(h, j, -d, n, dt)
    diff = i - j
    if diff < 0:
        diff = -diff
    if diff:
        d2 = d * d
        dt[n - 1 + diff] += d2
        dt[n - 1 - diff] += d2
    np.add(c, dt, out=tm)
    return float(tm.max())

                if time.time() > deadline - 2.0:
                    break
                if h[i] + d > 1.0 + 1e-12:
                    continue
                gi = g[i]
                js = np.flatnonzero((h >= d) & (g > gi + 1e-13) & (idx != i))
                for j in js:
                    if h[i] + d > 1.0 + 1e-12:
                        break
                    if h[j] - d < -1e-12:
                        continue
                    val = pair_value(h, c, i, int(j), d, n, dt, tm)
                    if val + 1e-12 < lb:
                        c += dt
                        h[i] += d
                        h[j] -= d
                        lb = val
                        acc = True
            if not acc and time.time() < deadline - 3.0:
                for _ in range(500):
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        val = pair_value(h, c, i, j, d, n, dt, tm)
                        if val + 1e-12 < lb:
                            c += dt
                            h[i] += d
                            h[j] -= d
                            lb = val
                            acc = True
            if acc:
                ct = compute_c_vec(h, n)
                tb = float(ct.max())
                if tb <= sb + 1e-10:
                    c = ct
                    if tb < best_c:
                        best_c = tb
                        best_h = h.copy()
                    if time.time() > deadline - 5.0:
                        break
                else:
                    h[:] = sh
                    c[:] = sc
            else:
                break
    return best_h, best_c, compute_c_vec(best_h, len(best_h))

def refine_pairs(h, c, deadline, best_h, best_c, iters=3000, seed=55,
                 fracs=(1.0, 0.5, 0.25, 0.125),
                 smalls=(0.002, 0.001, 0.0005, 0.0002)):
    n = len(h)
    h = h.copy()
    c = c.copy()
    dt = np.zeros(2 * n - 1)
    tm = np.empty_like(c)
    rng = np.random.RandomState(seed)
    g = None
    lo = hi = None
    for it in range(iters):
        if time.time() > deadline - 3.0:
            break
        if g is None or it % 50 == 0:
            g = smooth_gradient(h, c, 5000.0, n)
            med = float(np.median(g))
            lo = np.flatnonzero(g < med)
            hi = np.flatnonzero(g > med)
            if lo.size == 0 or hi.size == 0:
                break
        i = int(lo[int(rng.randint(lo.size))])
        j = int(hi[int(rng.randint(hi.size))])
        if i == j:
            continue
        md = min(1.0 - h[i], h[j])
        if md <= 1e-10:
            continue
        lb = float(c.max())
        bd = 0.0
        bv = lb
        bdt = None
        for d in tuple(fracs) + tuple(d * md for d in (1.0, 0.5, 0.25, 0.125)):
            if d <= 1e-10 or d > md:
                continue
            val = pair_value(h, c, i, j, d, n, dt, tm)
            if val + 1e-12 < bv:
                bv = val
                bd = d
                bdt = dt.copy()
        if bd > 0.0 and bdt is not None:
            c += bdt
            h[i] += bd
            h[j] -= bd
            ct = compute_c_vec(h, n)
            tb = float(ct.max())
            if tb <= lb + 1e-10:
                c = ct
                if tb < best_c:
                    best_c = tb
                    best_h = h.copy()
            else:
                c -= bdt
                h[i] -= bd
                h[j] += bd
                c = compute_c_vec(h, n)
    return best_h, best_c, compute_c_vec(best_h, len(best_h))

def make_binary(h):
    n = len(h)
    S = n * 0.5
    k = int(np.floor(S + 1e-12))
    f = S - k
    o = np.argsort(h)
    x = np.zeros(n, dtype=float)
    if k > 0:
        x[o[-k:]] = 1.0
    if f > 1e-12 and k < n:
        x[o[-k - 1]] = f
    return x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8.0
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(300, 0.5)
    n = len(h)
    if n < 10:
        h = np.full(300, 0.5)
        n = 300
    h = ensure_feasible(h, n * 0.5)
    c = compute_c_vec(h, n)
    best_c = float(c.max())
    best_h = h.copy()

    rem = dl - time.time()
    reserve = 150.0 if rem > 400.0 else 40.0 if rem > 150.0 else 0.0
    main_dl = dl - reserve

    tpg = min(time.time() + 220.0, main_dl - 250.0)
    if tpg > time.time() + 10.0:
        hp, cp = pg_phase(h, n, tpg)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
        h = best_h.copy()
        n = len(h)
        c = compute_c_vec(h, n)

    if time.time() < main_dl - 20.0:
        c = compute_c_vec(h, n)
        hp, bp, cp = pair_search(h, c, main_dl - 2.0, best_h, best_c)
        if bp < best_c:
            best_c = bp
            best_h = hp.copy()
        h = best_h.copy()
        n = len(h)
        c = compute_c_vec(h, n)

    if time.time() < main_dl - 8.0:
        c = compute_c_vec(h, n)
        hp, bp, cp = refine_pairs(h, c, main_dl - 2.0, best_h, best_c, iters=3500)
        if bp < best_c:
            best_c = bp
            best_h = hp.copy()
        h = best_h.copy()
        n = len(h)
        c = compute_c_vec(h, n)

    rem = dl - time.time()
    if rem > 110.0:
        nb = len(best_h)
        hv = 1.0 - best_h
        hv = ensure_feasible(hv, nb * 0.5)
        cv = compute_c_vec(hv, nb)
        ph = hv.copy()
        pv = float(cv.max())
        if pv < best_c:
            best_c = pv
            best_h = ph.copy()
        tpg = min(time.time() + 40.0, dl - 80.0)
        if time.time() + 5.0 < tpg:
            h1, c1 = pg_phase(ph, nb, tpg, alphas=(1000.0, 5000.0))
            if c1 < pv:
                pv = c1
                ph = h1.copy()
            if c1 < best_c:
                best_c = c1
                best_h = ph.copy()
        cv = compute_c_vec(ph, nb)
        if time.time() < dl - 15.0:
            h2, c2, cv2 = pair_search(ph, cv, dl - 4.0, best_h, best_c,
                                      steps=(0.002, 0.001, 0.0005), sweeps=1)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    rem = dl - time.time()
    if rem > 60.0:
        hb = make_binary(best_h)
        nb = len(hb)
        hb = ensure_feasible(hb, nb * 0.5)
        cb = compute_c_vec(hb, nb)
        ph = hb.copy()
        pv = float(cb.max())
        if pv < best_c:
            best_c = pv
            best_h = ph.copy()
        tpg = min(time.time() + 30.0, dl - 40.0)
        if time.time() + 5.0 < tpg:
            h1, c1 = pg_phase(ph, nb, tpg, alphas=(1000.0, 5000.0),
                              eta_list=(1.0, 0.5, 0.25, 0.1, 0.05, 0.02))
            if c1 < pv:
                pv = c1
                ph = h1.copy()
            if c1 < best_c:
                best_c = c1
                best_h = ph.copy()
        cv = compute_c_vec(ph, nb)
        if time.time() < dl - 8.0:
            h2, c2, cv2 = pair_search(ph, cv, dl - 2.0, best_h, best_c,
                                      steps=(0.005, 0.002, 0.001), sweeps=1)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    h = best_h.copy()
    n = len(h)
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12 or abs(h.sum() - n * 0.5) > 1e-10:
        h = ensure_feasible(h, n * 0.5)
    c5 = float(compute_c_vec(h, n).max())
    return h, c5, n
