import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    mc = corr.max()
    exp_c = np.exp(alpha * (corr - mc))
    s = exp_c.sum()
    if s < 1e-300:
        s = 1.0
    obj = mc + np.log(s) / alpha
    diff = x.sum() - n / 2.0
    obj += penalty * diff * diff
    w = exp_c / s
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        lo1 = max(0, n - 1 - m)
        hi1 = min(n, 2 * n - 1 - m)
        if lo1 < hi1:
            grad[lo1:hi1] += wm * (1.0 - x[lo1 + m - n + 1:hi1 + m - n + 1])
        lo2 = max(0, m - n + 1)
        hi2 = min(n, m)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * x[lo2 - m + n - 1:hi2 - m + n - 1]
    grad *= dx
    grad += 2.0 * penalty * diff
    return obj, grad

            return project_feasible(h, n)
        return project_feasible(res.x, n)

        return project_feasible(h, n)
