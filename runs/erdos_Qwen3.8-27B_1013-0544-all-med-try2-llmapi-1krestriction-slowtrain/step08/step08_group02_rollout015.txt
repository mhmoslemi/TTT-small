import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-10:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        # Remove excess ones - remove those with highest local overlap
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by some criterion and remove
        to_remove = cnt - target
        scores = np.array([h[ones_idx[i]] for i in range(len(ones_idx))])
        order = np.argsort(scores)[:to_remove]
        h_bin[ones_idx[order]] = 0
    elif cnt < target:
        zeros_idx = np.where(h_bin == 0)[0]
        to_add = target - cnt
        scores = np.array([h[zeros_idx[i]] for i in range(len(zeros_idx))])
        order = np.argsort(scores)[-to_add:]
        h_bin[zeros_idx[order]] = 1
    return h_bin

def lbfgs_run(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (2000, 20.0),
               (10000, 100.0), (50000, 500.0),
               (200000, 2000.0), (1000000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def lbfgs_run(h, n, deadline, max_iters=200):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 5.0), (5000, 50.0), (50000, 500.0),
               (500000, 5000.0), (2000000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 25})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_bin if False else h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def ils_2swap(h_bin, n, deadline, rng, n_try_per_round=15000):
    """Iterated local search with 2-swap moves."""
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    cur_h = h_bin.copy()
    
    while time.time() < deadline:
        # Greedy local search
        improved = True
        rounds = 0
        while improved and time.time() < deadline and rounds < 50:
            improved = False
            rounds += 1
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            best_move_c = compute_c5(cur_h, n)
            best_moves = None
            for _ in range(n_try_per_round):
                i1, i2 = rng.choice(ones_idx, size=2, replace=False)
                j1, j2 = rng.choice(zeros_idx, size=2, replace=False)
                cur_h[i1] = 0; cur_h[i2] = 0
                cur_h[j1] = 1; cur_h[j2] = 1
                c_new = compute_c5(cur_h, n)
                cur_h[i1] = 1; cur_h[i2] = 1
                cur_h[j1] = 0; cur_h[j2] = 0
                if c_new < best_move_c - 1e-15:
                    best_move_c = c_new
                    best_moves = (i1, i2, j1, j2)
                    improved = True
            if best_moves is not None:
                i1, i2, j1, j2 = best_moves
                cur_h[i1] = 0; cur_h[i2] = 0
                cur_h[j1] = 1; cur_h[j2] = 1
                if best_move_c < best_c:
                    best_c = best_move_c
                    best_h = cur_h.copy()
        
        # Perturb: random 1-flips
        k_p = rng.randint(3, 9)
        for _ in range(k_p):
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            cur_h[i] = 0; cur_h[j] = 1
    
    return best_h, best_c

def binary_sa(h, n, deadline, rng, max_flips=5000000):
    h_bin = to_binary(h, n, rng)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    temp = 0.008
    flips = 0
    while time.time() < deadline and flips < max_flips:
        if flips % 15000 == 0 and flips > 0:
            temp = max(temp * 0.9, 0.0005)
            if compute_c5(h_bin, n) - best_c > 0.002:
                h_bin = best_h.copy()
                temp = 0.01
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        use_2swap = rng.random() < 0.6
        if use_2swap:
            i1, i2 = rng.choice(ones_idx, size=2, replace=False)
            j1, j2 = rng.choice(zeros_idx, size=2, replace=False)
            h_bin[i1] = 0; h_bin[i2] = 0
            h_bin[j1] = 1; h_bin[j2] = 1
            c_new = compute_c5(h_bin, n)
            if c_new < best_c - 1e-15:
                best_c = c_new; best_h = h_bin.copy()
            elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-10)):
                pass
            else:
                h_bin[i1] = 1; h_bin[i2] = 1; h_bin[j1] = 0; h_bin[j2] = 0
        else:
            i1 = ones_idx[rng.randint(len(ones_idx))]
            j1 = zeros_idx[rng.randint(len(zeros_idx))]
            h_bin[i1] = 0; h_bin[j1] = 1
            c_new = compute_c5(h_bin, n)
            if c_new < best_c - 1e-15:
                best_c = c_new; best_h = h_bin.copy()
            elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-10)):
                pass
            else:
                h_bin[i1] = 1; h_bin[j1] = 0
        flips += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.zeros(n_init)
        idx = rng.choice(n_init, size=n_init // 2, replace=False)
        h_init[idx] = 1.0
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: Quick L-BFGS-B polish (40s max)
    t1 = min(t0 + 40, deadline - 550)
    if t1 > t0 + 5:
        best_h, best_c = lbfgs_run(best_h, best_n, t1)
    
    # Phase 2: ILS 2-swap on n=500 (300s)
    h_bin500 = to_binary(best_h, best_n, rng)
    remaining = deadline - time.time()
    if remaining > 100:
        t2 = time.time() + min(300, remaining * 0.5)
        h_b, c_b = ils_2swap(h_bin500, best_n, t2, rng, n_try_per_round=12000)
        if c_b < best_c:
            best_c = c_b; best_h = h_b
    
    # Phase 3: SA refinement (100s)
    remaining = deadline - time.time()
    if remaining > 60:
        t3 = time.time() + min(100, remaining * 0.4)
        h_s, c_s = binary_sa(best_h, best_n, t3, rng, max_flips=4000000)
        if c_s < best_c:
            best_c = c_s; best_h = h_s
    
    # Phase 4: Polish with L-BFGS-B (30s)
    remaining = deadline - time.time()
    if remaining > 25:
        t4 = time.time() + min(30, remaining * 0.4)
        h_p, c_p = lbfgs_run(best_h, best_n, t4)
        if c_p < best_c:
            best_c = c_p; best_h = h_p
    
    # Phase 5: Try n=250 (120s)
    remaining = deadline - time.time()
    if remaining > 60:
        n2 = 250
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n2)
        h_bin2 = to_binary(h_int, n2, rng)
        t5 = time.time() + min(80, remaining * 0.6)
        h_b2, c_b2 = ils_2swap(h_bin2, n2, t5, rng, n_try_per_round=8000)
        if c_b2 < best_c:
            best_c = c_b2; best_h = h_b2; best_n = n2
    
    # Phase 6: Try n=750 (80s)
    remaining = deadline - time.time()
    if remaining > 50:
        n3 = 750
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n3, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n3)
        h_bin3 = to_binary(h_int, n3, rng)
        t6 = time.time() + min(60, remaining * 0.6)
        h_b3, c_b3 = ils_2swap(h_bin3, n3, t6, rng, n_try_per_round=8000)
        if c_b3 < best_c:
            best_c = c_b3; best_h = h_b3; best_n = n3
    
    # Phase 7: Final SA + polish on best
    remaining = deadline - time.time()
    if remaining > 30:
        t7 = time.time() + min(40, remaining * 0.5)
        h_f, c_f = binary_sa(best_h, best_n, t7, rng, max_flips=3000000)
        if c_f < best_c:
            best_c = c_f; best_h = h_f
    
    remaining = deadline - time.time()
    if remaining > 10:
        t8 = time.time() + min(15, remaining - 3)
        h_f, c_f = lbfgs_run(best_h, best_n, t8, max_iters=100)
        if c_f < best_c:
            best_c = c_f; best_h = h_f
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
