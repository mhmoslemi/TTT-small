def dca_step(h, n, t_current, tolerance=0.01, max_iter=50):
    """DCA/CCP step: linearize quadratic terms and solve LP"""
    dx = 2.0 / n
    m = 2 * n - 1
    # Compute current correlation
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    
    for it in range(max_iter):
        t = c.max()
        if t * dx < t_current:
            break
        
        # Identify active lags (within tolerance of max)
        active = c > t - tolerance / dx
        
        # Build linearized constraints
        # For each active lag k, we need:
        # sum_i h[i] * (1 - h[i+k]) * dx <= t * dx / dx = t (in discrete units)
        # 
        # Actually let's work in discrete units: c[k] <= T where T = t

        # Linearize: h[i+k]*(1-h[i]) ≈ h[i+k]*(1-h_old[i]) + h_old[i+k]*(1-h[i]) - h_old[i+k]*(1-h_old[i])
        # = h[i+k] - h[i+k]*h_old[i] + h_old[i+k] - h_old[i+k]*h[i] - h_old[i+k] + h_old[i+k]*h_old[i]

        # = h[i+k]*(1-h_old[i]) - h_old[i+k]*h[i] + h_old[i+k]*h_old[i]
        
        # So linearized c[k] ≈ sum_i [h[i+k]*(1-h_old[i]) - h_old[i+k]*h[i] + h_old[i+k]*h_old[i]]
        # = sum_i h[i+k]*(1-h_old[i]) - sum_i h_old[i+k]*h[i] + sum_i h_old[i+k]*h_old[i]
        
        # The last term is constant. The first two are linear in h.
        
        # Let's build the constraint matrix for linprog
        # Minimize T (variable at index n)
        # Subject to:
        #   For each active k: sum_i [coeff_i] * h[i] <= T - const_k
        #   0 <= h[i] <= 1
        #   sum h[i] = n/2
        
        n_active = active.sum()
        if n_active == 0:
            break
        
        # c variables: h[0], ..., h[n-1], T
        # Objective: minimize T
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Constraints (inequality: A_ub @ x <= b_ub)
        A_rows = []
        b_rows = []
        
        m_range = np.arange(-(n-1), n)
        for ki in np.where(active)[0]:
            k = m_range[ki]  # actual lag
            # Build linearized c[k]
            # c_lin[k] = sum over valid (i, i+k) pairs of [h[i+k]*(1-h_old[i]) - h_old[i+k]*h[i] + h_old[i+k]*h_old[i]]
            row = np.zeros(n)
            const = 0.0
            if k >= 0:
                # i ranges from 0 to n-1-k
                for i in range(n - k):
                    row[i+k] += (1 - h[i])
                    row[i] -= h[i+k]
                    const += h[i+k] * h[i]
            else:
                # k < 0, let k' = -k
                kp = -k
                for i in range(n - kp):
                    row[i] += (1 - h[i+kp])
                    row[i+kp] -= h[i]
                    const += h[i] * h[i+kp]
            
            # row @ h + const <= T
            # row @ h - T <= -const
            full_row = np.zeros(n + 1)
            full_row[:n] = row
            full_row[n] = -1.0
            A_rows.append(full_row)
            b_rows.append(-const)
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # Equality: sum h = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [n / 2.0]
        
        bounds = [(0, 1)] * n + [(t * dx * 0.5, None)]  # T is bounded below
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            # Fallback: just return current h
            return h
        
        h_new = res.x[:n]
        T_new = res.x[n]
        
        if T_new * dx >= t_current - 1e-10:
            # No improvement, try with smaller tolerance
            tolerance *= 0.5
            if tolerance < 1e-10:
                return h
            continue
        
        # Check actual objective
        c_new = np.correlate(h_new, 1 - h_new, mode='full')
        if c_new.max() * dx >= t_current:
            # Linearization didn't help, reduce step
            h = 0.5 * h + 0.5 * h_new
            continue
        
        h = h_new
        t_current = c_new.max() * dx
        c = c_new
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n-1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1-h, mode='full').copy()
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2*n-1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            v0 = np.max(corr - B*h[i])
            v1 = np.max(corr + B*(1.0-h[i]))
            lo = min(v0, v1) - 1e-9
            hi = max(v0, v1) + 1e-9
            pos = B > 1e-14
            neg = B < -1e-14
            zro = ~(pos | neg)
            zmax = corr[zro].max() if zro.any() else -np.inf
            for _ in range(40):
                mid = (lo+hi)*0.5
                if zmax > mid + 1e-14:
                    lo = mid
                    continue
                lb = 0.0
                if neg.any():
                    lb = max(lb, h[i]+np.max((mid-corr[neg])/B[neg]))
                ub = 1.0
                if pos.any():
                    ub = min(ub, h[i]+np.min((mid-corr[pos])/B[pos]))
                if lb <= ub + 1e-12:
                    hi = mid
                else:
                    lo = mid
            lb = 0.0
            if neg.any():
                lb = max(lb, h[i]+np.max((hi-corr[neg])/B[neg]))
            ub = 1.0
            if pos.any():
                ub = min(ub, h[i]+np.min((hi-corr[pos])/B[pos]))
            xopt = np.clip((lb+ub)*0.5, 0.0, 1.0)
            if abs(xopt - h[i]) > 1e-15:
                corr += B*(xopt - h[i])
                h[i] = xopt
    return h, corr

def slsqp_opt(h, n, alpha, maxiter, ftol, active_idx):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0-x, mode='full')[active_idx] * dx
        mv = g.max()
        sh = alpha*(g - mv)
        sm = sh.max()
        return mv + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def obj_grad(x):
        g = np.correlate(x, 1.0-x, mode='full')[active_idx] * dx
        mv = g.max()
        sh = alpha*(g - mv)
        se = np.exp(sh - sh.max())
        w = se / se.sum()
        grad = np.zeros(n)
        for jj in range(len(g)):
            wi = w[jj]
            if wi < 1e-12:
                continue
            k = active_idx[jj] - (n-1)
            if k >= 0:
                idx = np.arange(n-k)
                grad[idx+k] += wi*dx*(1.0-x[idx])
                grad[idx] -= wi*dx*x[idx+k]
            else:
                kp = -k
                idx = np.arange(n-kp)
                grad[idx] += wi*dx*(1.0-x[idx+kp])
                grad[idx+kp] -= wi*dx*x[idx]
        return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum()-n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    c = np.correlate(h, 1-h, mode='full')
    return np.argsort(c)[::-1][:min(top_k, 2*n-1)]
