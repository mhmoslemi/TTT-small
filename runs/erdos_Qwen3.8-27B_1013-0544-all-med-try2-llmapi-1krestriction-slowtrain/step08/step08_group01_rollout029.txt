import numpy as np
from scipy.optimize import minimize
import time

def _fft_overlap(h, n):
    L = 1 << (2 * n - 2).bit_length()
    a = np.zeros(L); a[:n] = h
    b = np.zeros(L); b[:n] = 1.0 - h
    A = np.fft.rfft(a); B = np.fft.rfft(b)
    circ = np.fft.irfft(A * np.conj(B), n=L)
    return np.roll(circ, n - 1)[:2 * n - 1] * (2.0 / n)

def compute_c5(h, n):
    return float(np.max(_fft_overlap(h, n)))

def _grad_raw(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[k:] += (1.0 - h[:nv])
        g[:nv] += -h[k:nv]
    else:
        kk = -k
        nv = n - kk
        g[:nv] += (1.0 - h[kk:])
        g[kk:] += -h[:nv]
    return g

def _grad_smooth(h, n, alpha):
    c = _fft_overlap(h, n)
    cm = c.max()
    sh = alpha * (c - cm)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for m in active:
        g += w[m] * _grad_raw(h, n, m)
    return g * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-14: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, mu):
    c = _fft_overlap(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_obj(x, n, alpha, mu):
    g = _grad_smooth(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def fw_steps(h, n, n_iters, deadline):
    best_c = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        gamma = 2.0 / (it + 2)
        c = _fft_overlap(h, n)
        cm = c.max()
        alpha = 5000.0
        sh = alpha * (c - cm)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        g = np.zeros(n)
        active = np.where(w > 1e-6)[0]
        for m in active:
            g += w[m] * _grad_raw(h, n, m)
        order = np.argsort(g)
        s = np.zeros(n)
        s[order[:n // 2]] = 1.0
        h_new = (1.0 - gamma) * h + gamma * s
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            gamma *= 0.5
            h_new2 = (1.0 - gamma) * h + gamma * s
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                h = h_new2
    return h, best_c

def _lbfgs(h, n, deadline, max_alphas=8):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 0.1), (2000, 1.0), (10000, 10.0), (50000, 100.0),
               (200000, 500.0), (1000000, 2000.0), (5000000, 10000.0),
               (20000000, 50000.0)]
    for i, (alpha, mu) in enumerate(configs[:max_alphas]):
        if time.time() > deadline - 10:
            break
        remaining = deadline - time.time() - 5
        if remaining < 10:
            break
        maxiter = min(600, int(remaining / 0.05))
        try:
            res = minimize(_obj, h, jac=_grad_obj, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15,
                                   'maxfun': maxiter * 50, 'maxls': 40})
            h_new = _project(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_polish(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    pos = np.floor(np.arange(half) * phi).astype(int) % n
    h[pos] = 1.0
    return _project(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 8):
        amp = rng.uniform(0.15, 0.4)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + ph)
    return _project(h, n)

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return _project(h, n)
