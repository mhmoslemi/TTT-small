```python
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]

        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1

    best_h = h_bin.copy()
    
    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]

            i = ones_idx[rng.randint(n_ones)]
            j = zeros_idx[rng.randint(n_zeros)]
```

```python
def project_feasible(h, n):

        diff = target - h.sum()
        if abs(diff) < 1e-12:

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin
```

2. For a swap (i, j): delta = remove_from_i - remove_from_j, new_max = max(c + delta)
3. To find the best swap: try many (i,j) pairs

The key optimization: I can precompute remove_from as a 2D array of shape (n, 2n-1). Then for any (i,j), delta = remove_from[i] - remove_from[j], and max(c + delta) = max(c + remove_from[i] - remove_from[j]).

To find the best j for a fixed i: I need to minimize over j (where h[j]=0) the quantity max(c + remove_from[i] - remove_from[j]).

Let base = c + remove_from[i]. Then I need min over j of max(base - remove_from[j]).

This is not easy to vectorize directly (the max is inside the min). But I can:
- For each candidate j, compute max(base - remove_from[j])
- This is O(2n-1) per j
- Total for all j: O(n * 2n) = O(n²)

For n=500: 500 * 1000 = 500K per i. For all i: 250M. About 1-2 seconds in numpy.

Alternatively, I can just sample. For each i, sample 100 random j's, compute max(base - remove_from[j]) for each, pick the best. This is O(100 * 2n) = 100K per i, or 50M total. Much faster.

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

def precompute_remove(h, n):
    """Precompute remove_from_p for all p. Returns array of shape (n, 2n-1)."""
    nfull = 2 * n - 1
    remove = np.zeros((n, nfull))

    for p in range(n):

        m_first = p + t  # m indices

        t2 = np.arange(n)

        remove[p, m_second] += h[t2]
    return remove
```

```python
def compute_delta(h, n, i, j):
    """Compute delta to full correlation vector for swap (i: 1->0, j: 0->1)"""
    nfull = 2 * n - 1
    delta = np.zeros(nfull)
    d = np.arange(-(n-1), n)  # shifts
    
    # h[i] changes 1->0:
    # As first factor: d in [i-(n-1), i], contribution = h[i-d]-1
    mask1 = (d >= i-(n-1)) & (d <= i)
    idx1 = i - d[mask1]  # indices into h
    delta[mask1] += h[idx1] - 1.0
    
    # As second factor: d in [-i, n-1-i], contribution = h[i+d]
    mask2 = (d >= -i) & (d <= n-1-i)
    idx2 = i + d[mask2]
    delta[mask2] += h[idx2]
    
    # h[j] changes 0->1:
    # As first factor: d in [j-(n-1), j], contribution = 1-h[j-d]
    mask3 = (d >= j-(n-1)) & (d <= j)
    idx3 = j - d[mask3]
    delta[mask3] += 1.0 - h[idx3]
    
    # As second factor: d in [-j, n-1-j], contribution = -h[j+d]
    mask4 = (d >= -j) & (d <= n-1-j)
    idx4 = j + d[mask4]
    delta[mask4] -= h[idx4]
    
    return delta
```

```python
def remove_from(h, n, p):
    nfull = 2 * n - 1
    delta = np.zeros(nfull)
    d = np.arange(-(n-1), n)
    mask1 = (d >= p-(n-1)) & (d <= p)
    delta[mask1] += h[p - d[mask1]] - 1.0
    mask2 = (d >= -p) & (d <= n-1-p)
    delta[mask2] += h[p + d[mask2]]
    return delta
```

```python
    """Compute remove_from_p for multiple p values. Returns array of shape (len(p_list), 2n-1)."""
    M = len(p_list)
```
