import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_g(h, n):
    """g[i, k_idx] = dC_k/dh[i] (without dx factor)"""
    g = np.zeros((n, 2*n-1))
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        if k >= 0:
            g[0:n-k, k_idx] += (1.0 - h[k:n])
            g[k:n, k_idx] -= h[0:n-k]
        else:
            idx = -k
            g[idx:n, k_idx] += (1.0 - h[0:n-idx])
            g[0:n-idx, k_idx] -= h[idx:]
    return g

def smooth_max_obj(h, n, alpha):
    C = compute_c5_vec(h, n)
    m = C.max()
    shifted = alpha * (C - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return val

def smooth_max_grad(h, n, alpha, g_cache=None):
    C = compute_c5_vec(h, n)
    m = C.max()
    shifted = alpha * (C - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    if g_cache is None:
        g_cache = compute_g(h, n)
    grad = (g_cache.T @ (w * dx))
    return grad

def lbfgs_step(h, n, deadline, alpha, mu, max_iter=100):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        C = compute_c5_vec(x, n)
        m = C.max()
        shifted = alpha * (C - m)
        mx = shifted.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(shifted))) / alpha
        val += mu * (x.sum() - target) ** 2
        return float(val)

        C = compute_c5_vec(x, n)
        m = C.max()
        shifted = alpha * (C - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)

        dx = 2.0 / n
        g = compute_g(x, n)
        grad_val = (g.T @ (w * dx))
        grad_val += 2.0 * mu * (x.sum() - target)

def fast_pair_search(h, n, deadline, step_sizes):

    best_c = compute_c5(h, n)
    
    for d in step_sizes:

            C = compute_c5_vec(h, n)
            C_max = C.max()
            improved = False

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        if k >= 0:
            g[0:n - k, k_idx] += (1.0 - h[k:n])
            g[k:n, k_idx] -= h[0:n - k]
        else:
            idx = -k
            g[idx:n, k_idx] += (1.0 - h[0:n - idx])
            g[0:n - idx, k_idx] -= h[idx:]
    return g

def _smmax_val(C, alpha):
    m = C.max()
    sh = alpha * (C - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _smmax_weights(C, alpha):
    m = C.max()
    sh = alpha * (C - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def lbfgs_step(h, n, alpha, mu, max_iter=100):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        C = compute_c5_vec(x, n)
        return _smmax_val(C, alpha) + mu * (x.sum() - target) ** 2

    def grad(x):
        C = compute_c5_vec(x, n)
        w = _smmax_weights(C, alpha)
        dx = 2.0 / n
        g = compute_g(x, n)
        rv = (g.T @ (w * dx))
        rv += 2.0 * mu * (x.sum() - target)
        return rv

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': max_iter * 5, 'maxls': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h

def fast_pair_search(h, n, deadline, step_sizes, direction=1):
    best_c = compute_c5(h, n)
    for d in step_sizes:
        while time.time() < deadline:
            g = compute_g(h, n)
            C = compute_c5_vec(h, n)
            improved = False
            if direction == 1:
                for i in range(n):
                    if time.time() > deadline:
                        return h, best_c
                    if h[i] + d > 1.0:
                        continue
                    deltas = d * (g[i] - g)
                    feasible = h >= d - 1e-15
                    new_maxs = (C[None, :] + deltas).max(axis=1)
                    new_maxs[~feasible] = np.inf
                    new_maxs[i] = np.inf
                    bj = int(np.argmin(new_maxs))
                    if new_maxs[bj] < best_c - 1e-14:
                        h[i] += d
                        h[bj] -= d
                        best_c = compute_c5(h, n)
                        improved = True
            else:
                for j in range(n):
                    if time.time() > deadline:
                        return h, best_c
                    if h[j] + d > 1.0:
                        continue
                    deltas = d * (g[j] - g)
                    feasible = h >= d - 1e-15
                    new_maxs = (C[None, :] + deltas).max(axis=1)
                    new_maxs[~feasible] = np.inf
                    new_maxs[j] = np.inf
                    bi = int(np.argmin(new_maxs))
                    if new_maxs[bi] < best_c - 1e-14:
                        h[bi] -= d
                        h[j] += d
                        best_c = compute_c5(h, n)
                        improved = True
            if not improved:
                break
    return project_feasible(h, n), compute_c5(h, n)
