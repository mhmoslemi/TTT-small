```python
def binary_local_search(h, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    h = (h >= 0.5).astype(float)  # Binary-ize
    # Fix to have exactly n/2 ones
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        # Remove excess ones
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, n_ones - target, replace=False)
        h[to_remove] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[to_add] = 1
    
    c = np.correlate(h, 1-h, mode='full')
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # For each zero position a, find best one position b to swap
        for a in zeros:
            # Compute f_a[k] = h(a+k) + h(a-k) for all k
            # k ranges from -(n-1) to n-1
            # h(a+k) is valid when 0 <= a+k < n
            # h(a-k) is valid when 0 <= a-k < n
            fa = np.zeros(2*n - 1)
            # h(a+k) for k = 0, 1, ..., n-1-a
            for k in range(n - a):
                fa[n - 1 + k] += h[a + k]
            # h(a-k) for k = 0, -1, ..., a-(n-1) i.e. k from -(n-1) to 0 where a-k in [0,n-1]
            for k in range(a + 1):
                fa[n - 1 - k] += h[a - k]
            
            # For each one position b, compute max(c + g_b - f_a)
            # g_b[k] = h(b+k) - h(b-k)
            best_b = -1
            best_val = cur_max
            for b in ones:
                gb = np.zeros(2*n - 1)
                for k in range(n - b):
                    gb[n - 1 + k] += h[b + k]
                for k in range(b + 1):
                    gb[n - 1 - k] -= h[b - k]
                
                new_max = np.max(c + gb - fa)
                if new_max < best_val:
                    best_val = new_max
                    best_b = b
            
            if best_b >= 0 and best_val < cur_max - 1e-15:
                # Make the swap
                h[a] = 1
                h[best_b] = 0
                # Update c
                # ... (recompute or update incrementally)
                c = np.correlate(h, 1-h, mode='full')
                cur_max = c.max()
                improved = True
                break  # Restart sweep
        
        if not improved:
            break
    
    return h, cur_max * dx
```

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n, rng):
    target = n // 2
