            zeros_arr = np.where(h == 0)[0]

                best_c = cur_max * dx
                best_h = h.copy()

            i, j = int(batch_i[0]), int(batch_j[0])

```python
def batch_delta(h, i_batch, j_batch, n):
    """Compute deltas for multiple swaps at once.
    i_batch: array of positions to change 0->1
    j_batch: array of positions to change 1->0
    Returns: array of shape (len(i_batch), 2*n-1)
    """
    B = len(i_batch)
    k = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each b, delta_b[k] = (1-h[i_b - k]) - (1-h[j_b - k]) - h[i_b + k] + h[j_b + k]
    # i_batch[:, None] - k[None, :] gives shape (B, 2n-1)
    idx1 = i_batch[:, None] - k[None, :]  # (B, 2n-1)
    m1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j_batch[:, None] - k[None, :]
    m2 = (idx2 >= 0) & (idx2 < n)
    idx3 = i_batch[:, None] + k[None, :]
    m3 = (idx3 >= 0) & (idx3 < n)
    idx4 = j_batch[:, None] + k[None, :]
    m4 = (idx4 >= 0) & (idx4 < n)
    
    delta = np.zeros((B, 2*n-1))
    # For valid indices, gather h values
    safe_idx1 = np.where(m1, idx1, 0)
    safe_idx2 = np.where(m2, idx2, 0)
    safe_idx3 = np.where(m3, idx3, 0)
    safe_idx4 = np.where(m4, idx4, 0)
    
    delta += m1 * (1.0 - h[safe_idx1])
    delta -= m2 * (1.0 - h[safe_idx2])
    delta -= m3 * h[safe_idx3]
    delta += m4 * h[safe_idx4]
    
    return delta
```

def project_feasible(h, n):

    dx = 2.0 / n
    g = np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return float(m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha) * dx

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    hc = np.clip(x, 0, 1)
    g = np.correlate(hc, 1 - hc, mode='full')
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    k = np.arange(-(n - 1), n)
    mi = np.arange(n)

    im = mi[:, None] + k[None, :]
    mm = (im >= 0) & (im < n)

    grad = (dx * (mp * w[None, :] * (1 - hc[sp])).sum(axis=1)
            - dx * (mm * w[None, :] * hc[sm2]).sum(axis=1))

        hc = slsqp_refine(hc, n, alpha, 2000, p2 - 5)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c, best_h = c, hc.copy()

    # Phase 3: Binary anneal from continuous solution (15%)
    p3 = t0 + 0.90 * (budget_s - 10)
    hb2, cb2 = anneal_binary(best_h, n, p3, seed=seed + 100, batch_size=500)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()

    # Phase 4: Final SLSQP + short binary (10%)
    p4 = dl - 5
    if time.time() < p4 - 30:
        hc2 = project_feasible(best_h + 0.02 * rng.randn(n), n)
        hc2 = slsqp_refine(hc2, n, 10000000, 3000, p4 - 30)
        c = compute_c5(hc2, n)
        if c < best_c:
            best_c, best_h = c, hc2.copy()
    if time.time() < p4 - 5:
        hb3, cb3 = anneal_binary(best_h, n, p4 - 2, seed=seed + 200, batch_size=1000)
        if cb3 < best_c:
            best_c, best_h = cb3, hb3.copy()

    # Try alternate n if time allows
    n_try = [200, 300]
    if time.time() < dl - 60:
        for nh in n_try:
            if time.time() > dl - 40:
                break
            if nh == n:
                continue
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh = slsqp_refine(hh, nh, 50000, 500, dl - 50)
            ch = compute_c5(hh, nh)
            hb4, cb4 = anneal_binary(hh, nh, dl - 10, seed=seed + 300, batch_size=300)
            if cb4 < best_c:
                best_c, best_h, n = cb4, hb4.copy(), nh
            elif ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
