For a pair move (add δ to h[i], subtract δ from h[j]):
The new C5 = max_k (c_k + δ * g_k) where g_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]

This is the max of linear functions in δ, which is convex. The minimum of a convex piecewise linear function is where 0 is in the subgradient, i.e., where the slope changes from negative to positive.

We can find this by:
1. Computing g_k for all k
2. The function f(δ) = max_k (c_k + δ * g_k)
3. f'(δ) = g_k* where k* is the argmax
4. We want the smallest δ > 0 such that f stops decreasing

Actually, the minimum of max_k (c_k + δ * g_k) over δ ∈ [0, δ_max] occurs at:
- δ = 0 if all g_k* >= 0 for k* in the active set (i.e., moving in this direction increases the max)
- Or at the point where the active set changes
- Or at δ_max

A simpler approach: ternary search on δ ∈ [0, δ_max]. Since f(δ) is convex, ternary search will find the minimum in O(log(1/ε)) evaluations.

Each evaluation of f(δ) is O(n) (compute the max of n shifted correlations). So ternary search takes O(n log(1/ε)) per pair.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += diff * room / tot
        else:
            mask = h > 1e-12
            msum = h[mask].sum()
            if msum < 1e-14:
                break
            h[mask] += diff * h[mask] / msum
    return np.clip(h, 0.0, 1.0)

def obj_fn(x, n, alpha, mu):
    c = compute_all_c(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(x, n, alpha, mu):
    c = compute_all_c(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
        w /= w.sum()
    else:
        w = np.exp(sh)
        w /= w.sum()
    dx = 2.0 / n
    gh = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            gh[:nv] += wm * (1.0 - x[k:k + nv])
            gh[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            gh[kk:] += wm * (1.0 - x[:nv])
            gh[:nv] += wm * (-x[kk:kk + nv])
    gh += 2.0 * mu * (x.sum() - n / 2.0)
    return gh

def lbfgs_optimize(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 5.0), (500, 10.0), (2000, 20.0),
               (10000, 50.0), (50000, 100.0), (200000, 200.0),
               (1000000, 500.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        t_sec = min(40, deadline - time.time() - 5)
        if t_sec < 5:
            break
        try:
            res = minimize(obj_fn, h, jac=grad_fn, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-16,
                                    'gtol': 1e-13, 'maxfun': max_iters * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def line_search_pair(h, n, i, j, best_c):
    """Try moving delta from j to i and from i to j. Return best (delta, h_copy)."""
    h_orig = h.copy()
    hi_ji = min(h[j], 1.0 - h[i])  # max delta moving j->i
    hi_ij = min(h[i], 1.0 - h[j])  # max delta moving i->j
    if hi_ji <= 0 and hi_ij <= 0:
        return h, best_c, False
    improved = False
    # Try j->i
    if hi_ji > 1e-15:
        for frac in (0.01, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95, 1.0):
            d = frac * hi_ji
            h[i] = h_orig[i] + d
            h[j] = h_orig[j] - d
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                h[i] = h_orig[i]
                h[j] = h_orig[j]
    # Try i->j
    if hi_ij > 1e-15:
        for frac in (0.01, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95, 1.0):
            d = frac * hi_ij
            h[i] = h_orig[i] - d
            h[j] = h_orig[j] + d
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                h[i] = h_orig[i]
                h[j] = h_orig[j]
    if not improved:
        h[:] = h_orig
    return h, best_c, improved

def coord_descent(h, n, deadline, max_sweeps=500):
    best_c = compute_c5(h, n)
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        c_all = compute_all_c(h, n)
        m = c_all.max()
        active = c_all > m - 1e-4
        order = np.argsort(h)
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(len(order)):
                if jj == ii:
                    continue
                j = order[jj]
                if time.time() > deadline:
                    return h, best_c
                h, best_c, imp = line_search_pair(h, n, i, j, best_c)
                if imp:
                    improved = True
        if not improved:
            break
    return h, best_c

def fast_coord_descent(h, n, deadline, top_k=50):
    """Only try top-k gradient-suggested pairs per iteration."""
    best_c = compute_c5(h, n)
    alpha = 5000.0
    while time.time() < deadline:
        c_all = compute_all_c(h, n)
        m = c_all.max()
        sh = alpha * (c_all - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        gh = np.zeros(n)
        for idx in range(2 * n - 1):
            if w[idx] < 1e-14:
                continue
            k = idx - (n - 1)
            wm = w[idx] * dx
            if k >= 0:
                nv = n - k
                gh[:nv] += wm * (1.0 - h[k:k + nv])
                gh[k:] += wm * (-h[:nv])
            else:
                kk = -k
                nv = n - kk
                gh[kk:] += wm * (1.0 - h[:nv])
                gh[:nv] += wm * (-h[kk:kk + nv])
        # Find best pairs: negative gradient (should increase) with positive gradient (should decrease)
        neg_idx = np.where(gh < -1e-10)[0]
        pos_idx = np.where(gh > 1e-10)[0]
        if len(neg_idx) == 0 or len(pos_idx) == 0:
            break
        neg_idx = neg_idx[np.argsort(gh[neg_idx])][:top_k]
        pos_idx = pos_idx[np.argsort(-gh[pos_idx])][:top_k]
        improved = False
        for i in neg_idx:
            for j in pos_idx:
                if i == j:
                    continue
                if time.time() > deadline:
                    return h, best_c
                h, best_c, imp = line_search_pair(h, n, i, j, best_c)
                if imp:
                    improved = True
        if not improved:
            alpha *= 2
            if alpha > 1e6:
                break
    return h, best_c
