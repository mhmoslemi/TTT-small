import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, n, alpha):
    corr = compute_corr(h, n)
    mx = corr.max()
    sh = alpha * (corr - mx)
    mx2 = sh.max()
    w = np.exp(sh - mx2)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_penalty(x, n, alpha, mu):
    corr = compute_corr(x, n)
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_penalty(x, n, alpha, mu):
    g = grad_smooth(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s * np.ones(n)
    return g

def obj_slsqp(x, n, alpha):
    corr = compute_corr(x, n)
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alpha_list, mu_list, maxiter=800):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, (alpha, mu) in enumerate(zip(alpha_list, mu_list)):
        if time.time() > deadline - 2:
            break
        if deadline - time.time() < 4:
            break
        try:
            res = minimize(obj_penalty, h, jac=grad_penalty,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-16, 'maxfun': maxiter * 25,
                                   'maxls': 50})

            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new
        except Exception:
            pass
    return h, best_c

def topk_subgrad(h, n, deadline, k=3):
    """Optimize using subgradient from top-k active shifts with backtracking."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_step = 0.005
    step = base_step
    stall_count = 0
    for it in range(80000):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        c_cur = corr.max()
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 500:
                step *= 0.8
                if step < 1e-9:
                    break
                stall_count = 0

        # Get top-k indices
        idx = np.argsort(corr)[-k:]
        weights = np.zeros(len(corr))
        for ix in idx:
            weights[ix] = 1.0
        # Normalize
        weights /= weights.sum()

        # Compute weighted gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in idx:
            k_shift = m - (n - 1)
            wm = weights[m] * dx
            if k_shift >= 0:
                nv = n - k_shift
                grad[:nv] += wm * (1.0 - h[k_shift:])
                grad[k_shift:] += wm * (-h[:nv])
            else:
                kk = -k_shift
                nv = n - kk
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])

        # Zero-mean direction
        g = grad - grad.mean()

        # Backtracking
        accepted = False
        for _ in range(20):
            trial = h - step * g
            trial = project_feasible(trial, n)
            c_trial = compute_c5(trial, n)
            if c_trial < c_cur - 1e-17:
                h = trial
                accepted = True
                break
            step *= 0.6
        if accepted:
            step = min(step * 1.05, base_step)
        else:
            step *= 0.5
    return best_h, best_c

    return h, b_c

def pair_polish(h, n, deadline, radius=40):
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    best_c = compute_c5(h, n)
    for d in deltas:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
