    xo=(np.arange(no)+0.5)*(2.0/no)
    xn=(np.arange(n)+0.5)*(2.0/n)
    return np.interp(xn,xo,ho)

def coarse_search(ph,pn,deadline,rng,init_c):
    best_h=None; best_c=init_c; best_n=pn
    ns=[nn for nn in (96,128,160,192,224,256,320,384) if nn<pn]
    if not ns: return None,best_c,best_n
    total=max(1.0,deadline-time.time())
    per=min(40.0,max(6.0,total/(3.0*len(ns))))

      starts=[]
      st=interpolate_to(ph,pn,nn)
      starts.append(st)
      starts.append(golden_start(nn))

import time
import numpy as np
from scipy.optimize import minimize

_STATE = None

def corr_c(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5_value(h, n):
    return float(np.max(corr_c(h, n)))

def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    z = np.clip(y, 0.0, 1.0)
    if abs(z.sum() - target) < 1e-13:
        return z
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    a, b = lo, hi
    for _ in range(55):
        m = 0.5 * (a + b)
        if np.clip(y - m, 0.0, 1.0).sum() > target:
            a = m
        else:
            b = m
    return np.clip(y - 0.5 * (a + b), 0.0, 1.0)

def make_sums_exact(h, n):
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h2 = h.copy()
            h2[free] += diff / free.sum()
            if np.all((h2 >= -1e-12) & (h2 <= 1.0 + 1e-12)):
                h = np.clip(h2, 0.0, 1.0)
                diff = target - h.sum()
    if abs(diff) > 1e-13:
        for i in range(n):
            if diff > 1e-14:
                s = min(1.0 - h[i], diff)
                h[i] += s
                diff -= s
            elif diff < -1e-14:
                s = min(h[i], -diff)
                h[i] -= s
                diff += s
            if abs(diff) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def active_count(n):
    if n < 256:
        return 50
    if n < 450:
        return 90
    if n < 700:
        return 160
    return 220

def select_active(c, T):
    c = np.asarray(c)
    m = c.size
    if m <= T:
        return np.arange(m)
    return np.argpartition(c, -T)[-T:]

def obj_lbg(x, n, idx, alpha, mu):
    c = corr_c(x, n)
    v = c[idx]
    amax = float(v.max())
    val = amax + float(np.log(np.sum(np.exp(alpha * (v - amax))))) / alpha
    s = x.sum() - n / 2.0
    return float(val + mu * s * s)

def grad_lbg(x, n, idx, alpha, mu):
    c = corr_c(x, n)
    v = c[idx]
    amax = float(v.max())
    w = np.exp(alpha * (v - amax))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    off = n - 1
    for p in range(idx.size):
        wm = w[p] * dx
        if wm <= 1e-15:
            continue
        d = int(idx[p]) - off
        if d >= 0:
            nv = n - d
            if nv > 0:
                g[:nv] += wm * (1.0 - x[d:])
                g[d:] -= wm * x[:nv]
        else:
            e = -d
            nv = n - e
            if nv > 0:
                g[e:] += wm * (1.0 - x[:nv])
                g[:nv] -= wm * x[e:]
    s = x.sum() - n / 2.0
    if mu and s:
        g += 2.0 * mu * s
    return g

def record_cb(x):
    st = _STATE
    if st is None:
        return
    st['count'] += 1
    if st['count'] % 8:
        return
    h = make_sums_exact(project_box_sum(x, st['target']), st['n'])
    c = c5_value(h, st['n'])
    if c < st['c'] - 1e-15:
        st['c'] = c
        st['h'] = h

def active_lbfgs(h, n, deadline, iters=90, alphas=None, mus=None, reserve=30.0):
    global _STATE
    h = make_sums_exact(project_box_sum(h, n / 2.0), n)
    if alphas is None:
        alphas = (20.0, 80.0, 400.0, 2000.0, 10000.0, 50000.0, 200000.0, 800000.0)
    if mus is None:
        mus = (1e-4, 1e-3, 1e-2, 0.1, 1.0, 5.0, 20.0, 100.0)
    best_h = h.copy()
    best_c = c5_value(h, n)
    target = n / 2.0
    if time.time() > deadline:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    T = active_count(n)
    for ai in range(len(alphas)):
        if time.time() > deadline - reserve:
            break
        alpha = alphas[ai]
        mu = mus[ai]
        rounds = 2 if (deadline - time.time()) > 3.0 * reserve + 30.0 else 1
        for _ in range(rounds):
            if time.time() > deadline - reserve:
                break
            idx = select_active(corr_c(best_h, n), T)
            _STATE = {'n': n, 'target': target, 'c': best_c, 'h': best_h.copy(), 'count': 0}
            try:
                res = minimize(obj_lbg, best_h, jac=grad_lbg, args=(n, idx, alpha, mu),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': iters, 'ftol': 1e-13, 'gtol': 1e-10,
                                        'maxcor': 18, 'maxfun': iters * 25, 'maxls': 40},
                               callback=record_cb)
            except Exception:
                res = None
            st = _STATE
            _STATE = None
            if st is not None and st['c'] < best_c:
                best_c = st['c']
                best_h = st['h']
            x = res.x if res is not None and hasattr(res, 'x') else best_h
            h2 = make_sums_exact(project_box_sum(x, target), n)
            c2 = c5_value(h2, n)
            if c2 < best_c - 1e-15:
                best_c = c2
                best_h = h2
    return best_h, best_c

def eq_sum(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(len(x), dtype=float)

def slsqp_active(h, n, deadline, alpha=400000.0, maxiter=50):
    if time.time() > deadline:
        return h, c5_value(h, n)
    h = make_sums_exact(project_box_sum(h, n / 2.0), n)
    c = c5_value(h, n)
    idx = select_active(corr_c(h, n), 90)
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_lbg, h, args=(n, idx, alpha, 0.0), jac=grad_lbg,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return h, c
    h2 = make_sums_exact(project_box_sum(res.x, n / 2.0), n)
    c2 = c5_value(h2, n)
    return (h2, c2) if c2 < c - 1e-15 else (h, c)

def proj_grad_polish(h, n, deadline, alpha=250000.0):
    h = make_sums_exact(project_box_sum(h, n / 2.0), n)
    best = h.copy()
    bc = c5_value(best, n)
    target = n / 2.0
    T = active_count(n)
    eta = 0.02
    ac = alpha * 0.5
    stall = 0
    for _ in range(250):
        if time.time() > deadline - 1.0:
            break
        c = corr_c(best, n)
        idx = select_active(c, T)
        g = grad_lbg(best, n, idx, ac, 0.0)
        g -= g.mean()
        if float(np.linalg.norm(g)) < 1e-13:
            break
        step = eta
        improved = False
        for _ in range(25):
            if step < 1e-7:
                break
            h2 = make_sums_exact(project_box_sum(best - step * g, target), n)
            c2 = c5_value(h2, n)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                improved = True
                eta = min(0.1, step * 1.4)
                break
            step *= 0.5
        if improved:
            stall = 0
        else:
            stall += 1
            ac = min(800000.0, ac * 1.6)
            eta = max(1e-6, step)
            if stall >= 3:
                break
    return best, bc

def pair_polish(h, n, deadline, K=8, alpha=350000.0):
    best = make_sums_exact(project_box_sum(h, n / 2.0), n)
    bc = c5_value(best, n)
    T = active_count(n)
    steps = (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
             0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6)
    for _ in range(120):
        if time.time() > deadline - 1.0:
            break
        c = corr_c(best, n)
        idx = select_active(c, T)
        g = grad_lbg(best, n, idx, alpha, 0.0)
        g -= g.mean()
        order = np.argsort(g)
        lows = []
        for ii in order:
            if best[ii] < 1.0 - 1e-8:
                lows.append(int(ii))
                if len(lows) >= K:
                    break
        highs = []
        for ii in reversed(order):
            if best[ii] > 1e-8:
                highs.append(int(ii))
                if len(highs) >= K:
                    break
        if not lows or not highs:
            break
        improved = False
        for i in lows:
            if time.time() > deadline - 1.0:
                break
            room = 1.0 - best[i]
            if room <= 1e-9:
                continue
            for j in highs:
                if i == j:
                    continue
                maxs = min(room, best[j])
                if maxs <= 1e-10:
                    continue
                cands = []
                for d in steps:
                    dd = min(d, 0.8 * maxs)
                    if dd > 1e-8 and (not cands or dd < cands[-1] - 1e-15):
                        cands.append(dd)
                for d in cands:
                    best[i] += d
                    best[j] -= d
                    c2 = c5_value(best, n)
                    if c2 < bc - 1e-14:
                        bc = c2
                        improved = True
                    else:
                        best[i] -= d
                        best[j] += d
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    best = make_sums_exact(best, n)
    return best, c5_value(best, n)

def golden_start(n):
    h = np.zeros(n)
    k = n // 2
    if k:
        pos = (np.arange(k) * 1.618033988749895).astype(np.int64) % n
        h[pos] = 1.0
    return make_sums_exact(project_box_sum(h, n / 2.0), n)

def fourier_random_start(n, rng):
    m = n // 2 + 1
    f = np.zeros(m, dtype=complex)
    f[0] = 0.5 * n
    if m > 1:
        fr = np.arange(1, m)
        amp = rng.uniform(0.0, 0.40, size=m - 1) * np.exp(-fr / max(4.0, n / 12.0))
        ph = rng.uniform(0.0, 2.0 * np.pi, size=m - 1)
        f[1:] = amp * np.exp(2j * np.pi * ph)
    h = np.fft.irfft(f, n=n)
    return make_sums_exact(project_box_sum(h, n / 2.0), n)

def interpolate_to(ho, no, n):
    if no == n:
        return np.array(ho, dtype=float, copy=True)
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xn, xo, ho)

def squeeze(h, n, deadline, do_slsqp=False):
    h = make_sums_exact(project_box_sum(h, n / 2.0), n)
    c = c5_value(h, n)
    rem = deadline - time.time()
    if rem > 30.0:
        p_end = time.time() + min(rem * 0.5, rem - 30.0)
        h, c = proj_grad_polish(h, n, p_end)

        h, c = slsqp_active(h, n, s_end)
    if time.time() < deadline - 2.0:
        h, c = pair_polish(h, n, deadline - 1.0, K=8)
    return h, c

            if c1 < best_c:
                best_h = h1.copy()
                best_c = c1

    return best_h, best_c, best_n

        hv = make_sums_exact(project_box_sum(hv, n_v / 2.0), n_v)
        alloc_v = 220.0 if best_n >= 350 else 100.0
        end_v = time.time() + min(alloc_v, max(30.0, deadline - time.time() - 130.0))
        hv, cv = active_lbfgs(hv, n_v, end_v, iters=65, reserve=80.0)
        if cv < best_c + 0.003 and time.time() < deadline - 80.0:
            sq_end = min(time.time() + 55.0, deadline - 80.0)
            if sq_end > time.time() + 15.0:
                hv, cv = squeeze(hv, n_v, sq_end, do_slsqp=False)
            if cv < best_c:
                best_h, best_c, best_n = hv, cv, n_v

    if time.time() < deadline - 4.0:
        hf, cf = squeeze(best_h, best_n, deadline - 2.0, do_slsqp=(best_n <= 650))
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = make_sums_exact(best_h, best_n)
    best_c = c5_value(best_h, best_n)
    return best_h, best_c, int(best_n)
