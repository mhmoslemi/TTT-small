```python
    else:

    wsum = w.sum()
    w /= wsum
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0 - 1e-15] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def sparse_smooth(x, n, alpha, topK):
    c = np.correlate(x, 1.0 - x, mode='full')
    if topK < len(c):
        idx = np.argpartition(c, len(c) - topK)[len(c) - topK:]
    else:
        idx = np.arange(len(c))
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    wsum = w.sum()
    w /= wsum
    val = m + (smax + np.log(wsum)) / alpha
    grad = np.zeros(n)
    for j in range(len(idx)):
        if w[j] < 1e-15:
            continue
        lag = idx[j] - (n - 1)
        wj = w[j]
        if lag >= 0:
            nv = n - lag
            grad[lag:] += wj * (1.0 - x[:nv])
            grad[:nv] -= wj * x[lag:]
        else:
            L = -lag
            nv = n - L
            grad[:nv] += wj * (1.0 - x[L:])
            grad[L:] -= wj * x[:nv]
    return val, grad

def obj_pen(x, n, alpha, mu, topK):
    val, grad = sparse_smooth(x, n, alpha, topK)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def grad_pen(x, n, alpha, mu, topK):
    val, grad = sparse_smooth(x, n, alpha, topK)
    s = x.sum() - n / 2.0
    return grad + 2.0 * mu * s

def lbfgs(h, n, deadline, topK=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50,1,500),(200,5,500),(1000,20,400),(5000,100,400),
               (20000,500,300),(100000,2000,300),(500000,10000,200),(2000000,50000,200)]
    for alpha, mu, miter in configs:
        if time.time() > deadline - 5:
            break
        if deadline - time.time() < 8:
            break
        try:
            res = minimize(obj_pen, h, jac=grad_pen, args=(n, alpha, mu, topK),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': miter, 'ftol': 1e-20, 'gtol': 1e-15,
                                   'maxfun': miter * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def to_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_b.sum())
    if cnt > target:

    elif cnt < target:

        h_b[rng.choice(idx, target - cnt, replace=False)] = 1

    best_c = compute_c5(h_b, n)
    best_h = h_b.copy()
    temp = 0.05

            temp *= 0.9
            if temp < 0.001:
                break
        ones = np.where(h_b == 1)[0]

        j = zeros[rng.randint(len(zeros))]
        h_b[i] = 0; h_b[j] = 1
        c_new = compute_c5(h_b, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_b.copy()
        elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-12)):
            pass
        else:
            h_b[i] = 1; h_b[j] = 0
    return best_h, best_c

def binary_greedy(h, n, deadline, rng, max_rounds=100):
    h_b = to_binary(h, n, rng)
    best_c = compute_c5(h_b, n)
    best_h = h_b.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        best_mc = best_c
        bi, bj = -1, -1
        n_try = min(20000, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_b[i] = 0; h_b[j] = 1
            c_new = compute_c5(h_b, n)
            h_b[i] = 1; h_b[j] = 0
            if c_new < best_mc - 1e-15:
                best_mc = c_new
                bi, bj = i, j
        if bi >= 0:
            h_b[bi] = 0; h_b[bj] = 1
            best_c = best_mc
            best_h = h_b.copy()
        else:
            break
    return best_h, best_c

    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

    for f in range(1, 10):
        h += rng.uniform(0.1, 0.3) * np.cos(2 * np.pi * f * x / 2.0 + rng.uniform(0, 6.28))
    return project_feasible(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0 - 1e-15] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def sparse_smooth(x, n, alpha, topK):
    c = np.correlate(x, 1.0 - x, mode='full')
    if topK < len(c):
        idx = np.argpartition(c, len(c) - topK)[len(c) - topK:]
    else:
        idx = np.arange(len(c))
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    wsum = w.sum()
    w /= wsum
    val = m + (smax + np.log(wsum)) / alpha
    grad = np.zeros(n)
    for j in range(len(idx)):
        if w[j] < 1e-16:
            continue
        lag = idx[j] - (n - 1)
        wj = w[j]
        if lag >= 0:
            nv = n - lag
            grad[lag:] += wj * (1.0 - x[:nv])
            grad[:nv] -= wj * x[lag:]
        else:
            L = -lag
            nv = n - L
            grad[:nv] += wj * (1.0 - x[L:])
            grad[L:] -= wj * x[:nv]
    return val, grad

def obj_pen(x, n, alpha, mu, topK):
    val, grad = sparse_smooth(x, n, alpha, topK)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def grad_pen(x, n, alpha, mu, topK):
    val, grad = sparse_smooth(x, n, alpha, topK)
    s = x.sum() - n / 2.0
    return grad + 2.0 * mu * s

    h_b = (h > 0.5).astype(float)
```
