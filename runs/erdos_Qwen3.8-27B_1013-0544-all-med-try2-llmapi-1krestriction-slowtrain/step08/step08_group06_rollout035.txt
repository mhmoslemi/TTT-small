def compute_c5(h, n):

def project_box_sum(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid

def grad_lse(h, n, alpha):
    corr = get_corr(h, n)
    m = corr.max()

        hi2 = min(N, N - k)

    return grad
