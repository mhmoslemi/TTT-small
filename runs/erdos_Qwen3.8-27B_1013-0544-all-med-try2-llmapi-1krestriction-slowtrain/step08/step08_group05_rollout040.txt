def compute_smoothed_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shifted = c - c.max()
    W = np.exp(c_shifted)
    W /= W.sum()
    
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for lag in range(-(n-1), n):
            w = W[n-1+lag]
            if w < 1e-15:
                continue
            j1 = i - lag
            j2 = i + lag
            if 0 <= j1 < n:
                g += w * (1 - h[j1])
            if 0 <= j2 < n:
                g -= w * h[j2]
        grad[i] = dx * g
    return grad, c

def compute_smoothed_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

    W = np.exp(c_shifted)
    W /= W.sum()

    B = np.correlate(W, h, mode='full')[0:n][::-1]

    return grad, c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_logsumexp(c, alpha):
    m = c.max()
    return m + np.log(np.exp(c - m).sum()) / alpha

def compute_grad(c, h, n, alpha):
    dx = 2.0 / n
    cs = c - c.max()
    W = np.exp(cs)
    W /= W.sum()
    A = np.convolve(W, 1.0 - h, mode='full')[n-1:2*n-1]
    B = np.correlate(W, h, mode='full')[0:n][::-1]
    return dx * (A - B)

def pgd_optimize(h, n, deadline, alpha0=100, lr0=0.5):
    """Projected gradient descent with momentum and adaptive step size."""
    dx = 2.0 / n
    h = project_box_sum(h.copy(), n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    beta = 0.9
    alpha = alpha0
    lr = lr0
    t = 0
    
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_c = c.max()
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        grad = compute_grad(c, h, n, alpha)
        
        # Adaptive step size (barzilai-borwein style or simple backtracking)
        v = beta * v - lr * grad
        h_new = h + v
        h = project_box_sum(h_new, n)
        
        if t % 500 == 0 and t > 0:
            alpha = min(alpha * 1.5, 1e6)
            lr = max(lr * 0.95, 0.001)
        
        t += 1
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=2000):
    """Local search on binary solutions."""
    # Round to binary
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h_b.copy()
    
    ones = set(np.where(h_b == 1)[0].tolist())
    zeros = set(np.where(h_b == 0)[0].tolist())
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try each zero position
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in sorted(ones):
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2*n-1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_b[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0:
                h_b[i], h_b[best_j] = 1.0, 0.0
                corr = corr + best_delta
                cur_c = best_nc
                ones.add(i)
                ones.discard(best_j)
                zeros.discard(i)
                zeros.add(best_j)
                improved = True
                break
        if not improved:
            break
    
    if cur_c < best_c:
        best_c = cur_c
        best_h = h_b.copy()
    
    return best_h, best_c

def simulate_anneal_binary(h, n, deadline, t_start=1.0, t_end=0.01):
    """Simulated annealing on binary solutions."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h_b.copy()
    
    ones = set(np.where(h_b == 1)[0].tolist())
    zeros = set(np.where(h_b == 0)[0].tolist())
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(42)
    
    t_start_time = time.time()
    t_end_time = deadline
    total_time = t_end_time - t_start_time
    
    step = 0
    while time.time() < deadline:
        progress = (time.time() - t_start_time) / max(total_time, 0.1)
        T = t_start * (t_end / t_start) ** progress
        
        # Pick a random zero and a random one
        z_list = list(zeros)
        o_list = list(ones)
        i = z_list[rng.randint(len(z_list))]
        j = o_list[rng.randint(len(o_list))]
        
        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range
        delta = np.zeros(2*n-1)
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h_b[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h_b[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h_b[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h_b[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        delta_c = new_c - cur_c
        
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            h_b[i], h_b[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            ones.add(i)
            ones.discard(j)
            zeros.discard(i)
            zeros.add(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
        
        step += 1
    
    return best_h, best_c
