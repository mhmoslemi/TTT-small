    h = np.clip(np.array(h, float).copy(), 0, 1)
    t = 0.5 * n
    for _ in range(60):
        d = t - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            m = h < 1 - 1e-13
            if m.any():
                r = 1 - h[m]; h[m] += d * r / max(r.sum(),1e-15)
        else:
            m = h > 1e-13
            if m.any():
                v = h[m]; h[m] += d * v / max(v.sum(),1e-15)
    return np.clip(h, 0, 1)

import numpy as np, time, math
from scipy.optimize import minimize

def cc(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5(h, n):
    return float(np.max(cc(h)) * (2.0 / n))

def proj(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(70):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
            if np.any(m):
                r = 1.0 - h[m]
                tr = r.sum()
                if tr > 1e-14:
                    h[m] += d * r / tr
        else:
            m = h > 1e-13
            if np.any(m):
                v = h[m]
                tv = v.sum()
                if tv > 1e-14:
                    h[m] += d * v / tv
    return np.clip(h, 0.0, 1.0)

def resam(h, n0, n1):
    if n0 == n1:
        return proj(h, n1)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    return proj(np.interp(x1, x0, h), n1)

def topb(h, n):
    t = n // 2
    b = np.zeros(n, dtype=np.int32)
    if t:
        b[np.argsort(h, kind='mergesort')[::-1][:t]] = 1
    return b

def randb(n, rng):
    t = n // 2
    b = np.zeros(n, dtype=np.int32)
    if t:
        b[rng.choice(n, t, replace=False)] = 1
    return b

def edgeb(n):
    t = n // 2
    b = np.zeros(n, dtype=np.int32)
    if t:
        d = np.abs(np.arange(n, dtype=float) - 0.5 * (n - 1))
        b[np.argsort(d, kind='mergesort')[::-1][:t]] = 1
    return b

def ustart(n, a=0.45):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return proj(0.5 + a * np.cos(np.pi * x), n)

def fstart(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    s = np.zeros(n)
    for m in range(1, 7):
        s += rng.uniform(0.08, 0.35) * np.cos(np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi))
    return proj(0.5 + 0.5 * np.tanh(1.3 * s), n)

def lp(h, n, sg):
    g = np.fft.rfft(h - 0.5)
    f = np.fft.rfftfreq(n) * n * 0.5
    g *= np.exp(-(f * f) / (2.0 * sg * sg))
    return proj(0.5 + np.fft.irfft(g, n), n)

def smv(x, n, a):
    c = cc(x)
    if not np.all(np.isfinite(c)):
        return 1e100
    m = float(c.max())
    if a < 1e-6:
        return m
    sh = a * (c - m)
    sm = float(sh.max())
    w = np.exp(sh - sm)
    z = float(w.sum())
    if z <= 0.0 or not math.isfinite(z):
        return m
    return m + (sm + math.log(z)) / a

def smg(x, n, a):
    c = cc(x)
    g = np.zeros(n, dtype=float)
    if not np.all(np.isfinite(c)):
        return g
    m = float(c.max())
    if a < 1e-6:
        return g
    sh = a * (c - m)
    sm = float(sh.max())
    w = np.exp(sh - sm)
    zw = float(w.sum())
    if zw <= 0.0 or not math.isfinite(zw):
        return g
    w /= zw
    idx = np.nonzero(w > max(1e-12, 1e-8 * w.max()))[0]
    if idx.size > 400:
        idx = idx[np.argsort(w[idx])[-400:]]
    for mi in idx:
        wt = float(w[mi])
        k = mi - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:

                g[k:] -= wt * x[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wt * (1.0 - x[:nv])
                g[:nv] -= wt * x[kk:]
    return g

def po(x, n, a, mu):
    s = x.sum() - 0.5 * n
    return smv(x, n, a) + mu * s * s

def pg(x, n, a, mu):
    s = x.sum() - 0.5 * n
    return smg(x, n, a) + 2.0 * mu * s

def so(x, n, a):
    return smv(x, n, a)

def se(x, n):

def sj(x, n):

def eswap(h, n, dl, rng, steps=1500):
    h = proj(h, n)
    bh = h.copy()
    bc = c5(h, n)
    st = 0
    while time.time() < dl and st < steps:
        cor = cc(h)
        M = float(cor.max())
        act = np.nonzero(cor >= M - max(1.0, 0.1 * M))[0]
        act = act[act != n - 1]
        if act.size == 0:
            break
        a = int(act[rng.randint(act.size)])
        k = a - (n - 1)
        L = n - abs(k)
        if L > 1:
            if k >= 0:
                cp = h[:L] * (1.0 - h[k:])
            else:
                cp = h[-k:] * (1.0 - h[:L])
            if cp.sum() > 1e-14:
                li = int(np.argmax(cp)) if rng.random() < 0.7 else int(rng.randint(L))
                s = li if k >= 0 else li - k
                t = li + k if k >= 0 else li

                if amt > 1e-13:
                    os_ = h[s]

                    else:

    t0 = time.time()
    dl = t0 + float(budget_s) - 5.0
    if dl < t0 + 5.0:
        dl = t0 + 5.0
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        hp = None

    if hp is None or len(hp) < 20:
        n = 500
        hp = proj(0.5 * ustart(n, 0.45) + 0.5 * randb(n, rng).astype(float), n)
    else:
        n = len(hp)
        if n % 2:
            n += 1
            hp = resam(hp, n - 1, n)
        if n > 1000:
            hp = resam(hp, n, 1000)
            n = 1000

    hp = proj(hp, n)
    bh = hp.copy()
    bc = c5(bh, n)

    for th in (0.5 * (hp + hp[::-1]), lp(hp, n, 0.25), lp(hp, n, 0.8), ustart(n, 0.45)):
        th = proj(th, n)
        cr = c5(th, n)
        if cr < bc:
            bh, bc = th.copy(), cr
    for sh in (1, 3, 8):
        th = proj(0.5 * (hp + np.roll(hp, sh)), n)
        cr = c5(th, n)
        if cr < bc:
            bh, bc = th.copy(), cr

    exps = [(hp, 140, 0), (bh, 45, 1), (proj(0.5 * (hp + hp[::-1]), n), 35, 0),
            (lp(hp, n, 0.3), 25, 0), (ustart(n, 0.4), 30, 0),
            (fstart(n, rng), 25, 0), (fstart(n, rng), 20, 0)]
    stopc = max(t0 + 20.0, dl - 420.0)
    for he, ss, high in exps:
        if time.time() >= stopc:
            break
        al = min(ss, stopc - time.time())
        if al < 10.0:
            continue
        ho, co = cont_opt(proj(he, n), n, time.time() + al, bool(high))
        if co < bc:
            bh, bc = ho.copy(), co

    alt_n = ((n + 250) // 2) * 2
    if alt_n > n and alt_n < 1000 and time.time() < dl - 500.0:
        ha = resam(bh, n, alt_n)
        ca = c5(ha, alt_n)
        if ca < bc:
            n, bh, bc = alt_n, ha.copy(), ca
        if time.time() < dl - 450.0:
            h2, c2 = cont_opt(ha, n, min(dl - 450.0, time.time() + 60.0), 0)
            if c2 < bc:
                bh, bc = h2.copy(), c2

    if len(hp) != n:
        hp = resam(hp, len(hp), n)

    stopb = max(t0 + 20.0, dl - 170.0)
    bsts = [bh, hp, proj(0.5 * (hp + hp[::-1]), n), lp(hp, n, 0.3), edgeb(n).astype(float)]
    brows = [80, 40, 25, 25, 30]
    for hs, ss in zip(bsts, brows):
        if time.time() >= stopb:
            break
        al = min(ss, stopb - time.time())
        if al < 12.0:
            continue
        hb = bina(proj(hs, n), n, time.time() + al, rng)
        cr = c5(hb, n)
        if cr < bc:
            bh, bc = hb.copy(), cr

    stopf = max(dl - 3.0, t0 + 10.0)
    if time.time() < stopf - 20.0:
        hf, cf = cont_opt(bh, n, min(stopf - 5.0, time.time() + 70.0), 1)
        if cf < bc:
            bh, bc = hf.copy(), cf
    if time.time() < stopf - 15.0:
        he, hc = eswap(bh, n, min(stopf - 15.0, time.time() + 35.0), rng,
                       steps=800 if n > 600 else 2000)
        if hc < bc:
            bh, bc = he.copy(), hc
    if time.time() < stopf - 8.0:
        hs, cs = slsqp_opt(bh, n, stopf - 3.0, 10000, 80)
        if cs < bc:
            bh, bc = hs.copy(), cs

    if not np.all((bh == 0.0) | (bh == 1.0)):
        bh = proj(bh, n)
    bc = c5(bh, n)
    return bh.astype(float), float(bc), int(n)
