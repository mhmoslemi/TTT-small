    """FISTA-like projected gradient on smooth max."""
    dx = 2.0 / n
    h = h.copy()
    y = h.copy()  # momentum variable
    t = 1.0
    # Initial step size: use 1/L where L is Lipschitz constant
    # L ≈ alpha * dx * (2n-1) roughly (since each component of g has at most n terms)
    # Actually L = alpha * L_g where L_g is Lipschitz of g
    # g_s is bilinear with "operator norm" bounded by dx * max(n, n) = dx * n
    # So L_g ≈ alpha * dx * n (very rough)
    # Let's just use adaptive step size

    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = smooth_max_and_grad(y, n, alpha)
        # Gradient step
        y_new = y - step * grad
        # Project
        y_proj = project_feasible(y_new, n)
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = y_proj + (t/t_new - 1) * (y_proj - h)
        t = t_new
        h = y_proj
        
        # Backtracking (check if objective decreased)
        # (skip for simplicity, adjust step manually)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    shifted -= shifted.max()
    e = np.exp(shifted)
    return m + np.log(e.sum()) / alpha

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    shifted -= shifted.max()
    e = np.exp(shifted)
    p = e / e.sum()
    A = np.zeros(n)
    B = np.zeros(n)
    C = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        j_lo = max(0, s)
        j_hi = min(n - 1, s + n - 1)
        if j_lo <= j_hi:
            C[j_lo:j_hi + 1] += pk
            A[j_lo:j_hi + 1] += pk * x[j_lo - s:j_hi - s + 1]
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            B[j_lo2:j_hi2 + 1] += pk * x[j_lo2 + s:j_hi2 + s + 1]

        return smooth_max_obj(x, n, alpha)
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

def coord_descent(h, n, max_sweeps, deadline):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            hi_v = max(np.max(corr - B * h[i]), np.max(corr + B * (1.0 - h[i]))) + 1e-10
            lo_v = min(np.max(corr - B * h[i]), np.max(corr + B * (1.0 - h[i]))) - 1e-10
            pos_B = B > 1e-13

            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):

                    continue

                if neg_B.any():

                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():

    # Phase 2: SLSQP with analytical gradient, main ramp
    p2 = t0 + 0.55 * (dl - t0)
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000, 200000000, 500000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        remaining = max(100, int(2000 * (p2 - time.time()) / (p2 - t0)))
        h = slsqp_opt(h, n, alpha, maxiter=remaining, deadline=p2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Coord descent polish
    p3 = t0 + 0.68 * (dl - t0)
    hc, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 4: More SLSQP from polished result
    p4 = t0 + 0.80 * (dl - t0)
    h = best_h.copy()
    for alpha in [10000000, 50000000, 200000000, 1000000000]:
        if time.time() > p4 - 5:
            break
        h = slsqp_opt(h, n, alpha, maxiter=3000, deadline=p4)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Multi-start with perturbations
    p5 = t0 + 0.92 * (dl - t0)
    for trial in range(5):
        if time.time() > p5 - 10:
            break
        scale = 0.005 + 0.005 * trial
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [1000000, 10000000, 100000000]:
            if time.time() > p5 - 5:
                break
            h = slsqp_opt(h, n, alpha, maxiter=2000, deadline=p5)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 6: Try different resolutions
    p6 = dl - 5
    for nh in [300, 500, 200]:
        if time.time() > p6 - 10:
            break
        xo = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        xn = np.linspace(0, 2, nh, endpoint=False) + 1.0 / nh
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=60, deadline=p6 - 8)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
        for alpha in [10000000, 100000000]:
            if time.time() > p6 - 4:
                break
            hh = slsqp_opt(hh, nh, alpha, maxiter=1500, deadline=p6)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Final coord descent
    best_h, _ = coord_descent(best_h, n, max_sweeps=100, deadline=dl)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
