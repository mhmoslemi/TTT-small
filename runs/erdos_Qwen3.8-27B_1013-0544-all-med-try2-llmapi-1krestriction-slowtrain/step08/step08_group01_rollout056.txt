    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            if tot < 1e-15:

def smooth_max_topk(c, alpha, K):
    """Smooth max of top K values in c."""
    idx = np.argsort(c)[-K:]
    ck = c[idx]
    m = ck.max()
    sh = alpha * (ck - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    h = project_feasible(h, n)
    c = compute_c5_vec(h, n)

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    nfull = 2 * n - 1
    for m_idx in range(nfull):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-10:
        return False
    
    saved_i = h[i]
    saved_j = h[j]
    
    def f(t):
        h[i] = t
        h[j] = s - t
        c = compute_c5_vec(h, n)
        return smooth_max_val(c, alpha)
    
    try:
        res = minimize_scalar(f, bounds=(t_lo, t_hi), method='bounded',
                             options={'xatol': 1e-10, 'maxiter': 50})
        h[i] = res.x
        h[j] = s - res.x

        h[j] = saved_j
        return False

        h[j] = saved_j

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    nfull = 2 * n - 1
    for m_idx in range(nfull):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def pair_1d_opt(h, n, i, j, alpha):
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-10:
        return False

    saved_j = h[j]

        h[j] = saved_j
        return False
    if abs(h[i] + h[j] - s) > 1e-9:

def cyclic_pair(h, n, deadline, alpha=5000, radius=15):

def greedy_polish(h, n, deadline, radius=20):
    best_c = compute_c5(h, n)
    for d in [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 4):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 4):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d

                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
