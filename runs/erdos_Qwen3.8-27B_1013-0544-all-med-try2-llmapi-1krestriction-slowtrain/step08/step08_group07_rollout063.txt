Let base[t] = corr[t] - B[t]*h[i]. Then we want to minimize max_t (base[t] + B[t]*x).

This is a 1D convex optimization. The function g(x) = max_t (base[t] + B[t]*x) is convex and piecewise linear. Its minimum is at a point where two (or more) "active" constraints have the same value, or at a boundary.

A clean way to solve this: 
- g(0) = max(base)
- g(1) = max(base + B)
- If g(0) <= g(1), the minimum is at x ∈ [0, x_cross] where x_cross is where the slope changes sign
- The slope of g at x is: max_{t active}(B[t]) where active means base[t] + B[t]*x = g(x)

Actually, the simplest correct approach for 1D convex piecewise linear minimization is:
- Binary search on the answer T: is there an x ∈ [0,1] such that g(x) <= T?
- g(x) <= T iff for all t: base[t] + B[t]*x <= T
  - If B[t] > 0: x <= (T - base[t]) / B[t]
  - If B[t] < 0: x >= (T - base[t]) / B[t]
  - If B[t] = 0: base[t] <= T (must hold regardless of x)
- So the feasible x range is [max_{B<0} (T-base[t])/B[t], min_{B>0} (T-base[t])/B[t]] ∩ [0,1]
- This is non-empty iff max(lower bounds) <= min(upper bounds)

Binary search on T gives us the minimum value and the optimal x.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_with_grad(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    m = 2 * n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = c.max()
        if alpha < 1e6:
            sh = alpha * (c - cm)
            sm = sh.max()
            return (cm + (sm + np.log(np.exp(sh - sm).sum())) / alpha) * dx
        else:
            return cm * dx
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = c.max()
        if alpha < 1e6:
            sh = alpha * (c - cm)
            sm = sh.max()
            p = np.exp(sh - sm)
            p /= p.sum()
        else:
            p = np.zeros(m)
            idx = np.argmax(c)
            p[idx] = 1.0
        
        g = np.zeros(n)
        for t in range(m):
            w = p[t]
            if w < 1e-14:
                continue
            k = t - (n - 1)
            if k >= 0:
                g[:n-k] += w * (1.0 - x[k:])
                g[k:] -= w * x[:n-k]
            else:
                kk = -k
                g[kk:] += w * (1.0 - x[:n-kk])
                g[:n-kk] -= w * x[kk:]
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h_init

def coord_descent(h, n, max_sweeps=200, deadline=None):
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            base = corr - B * h[i]
            pos = B > 1e-14
            neg = B < -1e-14
            zero = ~(pos | neg)
            T_lo = np.max(base) - np.max(np.abs(B))
            T_hi = np.max(np.maximum(base, base + B))
            if T_hi - T_lo < 1e-14:
                continue
            for _ in range(28):
                T_mid = (T_lo + T_hi) / 2.0
                if zero.any() and np.max(base[zero]) > T_mid + 1e-14:
                    T_lo = T_mid
                    continue
                lb = 0.0
                if neg.any():
                    lb = max(lb, float(np.max((T_mid - base[neg]) / B[neg])))
                ub = 1.0
                if pos.any():
                    ub = min(ub, float(np.min((T_mid - base[pos]) / B[pos])))
                if lb <= ub + 1e-12:
                    T_hi = T_mid
                else:
                    T_lo = T_mid
            lb = 0.0
            if neg.any():
                lb = max(lb, float(np.max((T_hi - base[neg]) / B[neg])))
            ub = 1.0
            if pos.any():
                ub = min(ub, float(np.min((T_hi - base[pos]) / B[pos])))
            x_opt = np.clip((lb + ub) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-10:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h, corr

def subgradient_step(h, n, steps=5000, lr=0.01, momentum=0.9, deadline=None):
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for s in range(steps):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        t_star = int(np.argmax(c))
        k = t_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        else:
            kk = -k
            g[kk:] += (1.0 - h[:n - kk])
            g[:n - kk] -= h[kk:]
        g *= dx
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr = max(lr * 0.95, 1e-6)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = budget_s - 15
    dl = t0 + total_budget
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick coordinate descent
    p1 = t0 + 0.08 * total_budget
    hh, _ = coord_descent(best_h, n, max_sweeps=80, deadline=p1)
    hh = project_feasible(hh, n)
    cc = compute_c5(hh, n)
    if cc < best_c:
        best_c, best_h = cc, hh.copy()

    # Phase 2: Subgradient with momentum for quick progress
    p2 = t0 + 0.15 * total_budget
    hh, cc = subgradient_step(best_h, n, steps=8000, lr=0.005, deadline=p2)
    if cc < best_c:
        best_c, best_h = cc, hh.copy()

    # Phase 3: SLSQP with analytical gradient - main optimization
    p3 = t0 + 0.60 * total_budget
    alphas = [10, 30, 100, 300, 1000, 3000, 10000, 50000, 200000, 1000000, 5000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p3 - 15:
            break
        maxiter = 3000 if alpha < 1000 else 5000
        h = slsqp_grad(h, n, alpha, maxiter, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Coordinate descent post-processor
    p4 = t0 + 0.78 * total_budget
    hh, _ = coord_descent(best_h, n, max_sweeps=500, deadline=p4)
    hh = project_feasible(hh, n)
    cc = compute_c5(hh, n)
    if cc < best_c:
        best_c, best_h = cc, hh.copy()

    # Phase 5: Final SLSQP polish with very high alpha
    if time.time() < dl - 30:
        h = best_h.copy()
        for alpha in [10000000, 50000000, 200000000, 1000000000]:
            if time.time() > dl - 20:
                break
            h = slsqp_grad(h, n, alpha, 8000, 1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 6: Final coordinate descent
    if time.time() < dl - 5:
        hh, _ = coord_descent(best_h, n, max_sweeps=200, deadline=dl - 2)
        hh = project_feasible(hh, n)
        cc = compute_c5(hh, n)
        if cc < best_c:
            best_c, best_h = cc, hh.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
