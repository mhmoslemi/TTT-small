def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj_lbfgs(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += 0.01 * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_lbfgs(x, n, alpha):
    g = grad_smooth_max(x, n, alpha)
    g += 0.02 * (x.sum() - n / 2.0)
    return g

def _obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh)))) / alpha

def _eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, alphas, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        t_remain = deadline - time.time() - 3
        if t_remain < 5:
            break
        try:
            res = minimize(_obj_lbfgs, h, jac=_grad_lbfgs,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters * 20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def momentum_phase(h, n, deadline, alpha=500, lr_init=0.003, mom=0.85, max_iters=8000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
            h = h_new
        else:
            moved = False
            for shrink in [0.5, 0.25, 0.1]:
                h_t = project_feasible(h + shrink * v, n)
                c_t = compute_c5(h_t, n)
                if c_t < best_c - 1e-16:
                    best_c = c_t
                    best_h = h_t
                    h = h_t
                    moved = True
                    break
            if not moved:
                lr *= 0.6
                v *= 0.3
            if lr < 1e-15:
                break
    return best_h, best_c

def top_shift_phase(h, n, deadline, top_k=5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(500):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_k:]
        total_grad = np.zeros(n)
        for m in top_idx:
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                total_grad[0:nv] += (1.0 - h[k:n]) * dx
                total_grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                total_grad[kk:n] += (1.0 - h[0:nv]) * dx
                total_grad[0:nv] += (-h[kk:n]) * dx
        g_norm = np.linalg.norm(total_grad)
        if g_norm < 1e-14:
            break
        g = total_grad / g_norm
        moved = False
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                moved = True
                break
        if not moved:
            break
    return h, best_c

def coord_descent(h, n, deadline, step_sizes, rng):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in step_sizes:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            indices = rng.permutation(n)
            for i in indices:
                if time.time() > deadline:
                    return h, best_c
                for _ in range(4):
                    j = rng.randint(n)
                    if i == j:
                        continue
                    for direction in [1, -1]:
                        move = direction * d
                        if h[i] + move < -1e-12 or h[i] + move > 1 + 1e-12:
                            continue
                        if h[j] - move < -1e-12 or h[j] - move > 1 + 1e-12:
                            continue
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= move
                            h[j] += move
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP with increasing alpha (exact constraint handling)
    t1 = min(t0 + 200, deadline - 500)
    for a in [5000, 50000, 200000, 1000000]:
        if time.time() > t1:
            break
        alloc = min(50, t1 - time.time())
        if alloc < 10:
            break
        h_s, c_s = slsqp_phase(best_h, n_init, time.time() + alloc, alpha=a, maxiter=1000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    # Phase 2: L-BFGS refinement
    if time.time() < deadline - 350:
        t2 = min(time.time() + 150, deadline - 350)
        best_h, best_c = lbfgs_phase(best_h, n_init, t2,
                                      alphas=[2000, 20000, 100000, 500000], max_iters=200)
    # Phase 3: Momentum descent
    if time.time() < deadline - 250:
        t3 = min(time.time() + 100, deadline - 250)
        h3, c3 = momentum_phase(best_h, n_init, t3, alpha=1000, lr_init=0.002, mom=0.8)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    # Phase 4: Top-shift focused
    if time.time() < deadline - 180:
        t4 = min(time.time() + 80, deadline - 180)
        h4, c4 = top_shift_phase(best_h, n_init, t4, top_k=5)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    # Phase 5: SLSQP fine-tune
    if time.time() < deadline - 120:
        t5 = min(time.time() + 60, deadline - 120)
        for a in [500000, 2000000]:
            if time.time() > t5:
                break
            h5, c5 = slsqp_phase(best_h, n_init, min(time.time() + 30, t5), alpha=a, maxiter=1500)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    # Phase 6: Coordinate descent
    if time.time() < deadline - 60:
        t6 = min(time.time() + 50, deadline - 60)
        best_h, best_c = coord_descent(best_h, n_init, t6,
                                        [0.005, 0.002, 0.001, 0.0005, 0.0002], rng)
    # Phase 7: Perturbation restarts
    if time.time() < deadline - 80:
        for ps in [0.03, 0.01, 0.005]:
            if time.time() > deadline - 40:
                break
            h_p = best_h + rng.randn(n_init) * ps
            h_p = project_feasible(h_p, n_init)
            alloc = min(30, deadline - time.time() - 30)
            if alloc < 15:
                break
            h_r, c_r = lbfgs_phase(h_p, n_init, time.time() + alloc * 0.6,
                                    alphas=[5000, 50000], max_iters=100)
            h_r, c_r = top_shift_phase(h_r, n_init, time.time() + min(10, deadline - time.time() - 20), top_k=5)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    # Phase 8: Final refinement
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = lbfgs_phase(best_h, nf, deadline - 3,
                                      alphas=[100000, 500000], max_iters=50)
    if time.time() < deadline - 1:
        best_h, best_c = coord_descent(best_h, nf, deadline - 0.5,
                                        [0.001, 0.0005, 0.0002], rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
