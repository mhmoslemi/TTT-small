def lp_step(h,n,c,max_rows=200,delta=None):

    idx=np.argsort(c)[::-1]

       else:
          r=-m; row[:n-r] += 1.0-h[r:]; row[r:] -= h[:n-r]

       for i in range(n): bounds.append((max(0.0,h[i]-delta), min(1.0,h[i]+delta)))

import numpy as np
import time
from scipy.optimize import minimize, linprog

_G_N = 0
_G_ALPHA = 1.0
_G_CACHE = [None, None]

def c_counts(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def compute_c(h, n):
    return float(np.max(c_counts(h)) * (2.0 / n))

def project_feasible(x, n):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    t = n / 2.0
    for _ in range(300):
        d = t - x.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = x < 1.0 - 1e-14
            r = 1.0 - x[m]
            s = r.sum()
            if s <= 1e-15:
                break
            x[m] += d * r / s
        else:
            m = x > 1e-14
            r = x[m]
            s = r.sum()
            if s <= 1e-15:
                break
            x[m] += d * r / s
    x = np.clip(x, 0.0, 1.0)
    for _ in range(100):
        d = t - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            idx = np.flatnonzero(x < 1.0 - 1e-12)
            if idx.size == 0:
                break
            i = idx[int(np.argmax(1.0 - x[idx]))]
            x[i] += min(d, 1.0 - x[i])
        else:
            idx = np.flatnonzero(x > 1e-12)
            if idx.size == 0:
                break
            i = idx[int(np.argmax(x[idx]))]
            x[i] -= min(-d, x[i])
    return x

def resample(h, n):
    if h.size == n:
        return h.astype(float).copy()
    if h.size == 0:
        return np.full(n, 0.5)
    return np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, h.size), h)

def balance_binary(h, n):
    target = n // 2 if n % 2 == 0 else int(n * 0.5)
    cur = int(h.sum())
    if cur > target:
        idx = np.flatnonzero(h == 1.0)
        h[idx[:cur - target]] = 0.0
    elif cur < target:
        idx = np.flatnonzero(h == 0.0)
        h[idx[:target - cur]] = 1.0
    return h

def thue_binary(n):
    t = np.array([0], dtype=np.int8)
    while t.size < n:
        t = np.concatenate((t, 1 - t))
    return balance_binary((t[:n] == 1).astype(float), n)

def random_binary(n, rng):
    h = np.zeros(n)
    if n > 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    return balance_binary(h, n)

def smooth_binary(h, n, w=3):
    k = np.ones(w) / w
    return project_feasible(np.convolve(h, k, mode='same'), n)

def random_fourier(n, rng, lf=10, amp=0.35):
    h = 0.5 * np.ones(n)
    x = np.arange(n)
    for f in range(1, lf + 1):
        h += (amp / np.sqrt(float(f))) * np.cos(2.0 * np.pi * f * x / n + rng.uniform(0.0, 2.0 * np.pi))
    return project_feasible(h, n)

def construct_start(n, var, rng, best):
    if best is not None and var == 0:
        return project_feasible(resample(best, n) + 0.02 * rng.standard_normal(n), n)
    if var == 1:

    if var == 2:
        return smooth_binary(random_binary(n, rng), n, 3)
    if var == 3:
        return random_binary(n, rng)
    return random_fourier(n, rng, 10, 0.35)

def coord_descent(h, n, max_sweeps=4, deadline=None, grid=21):
    h = project_feasible(h, n).copy()
    corr = c_counts(h)
    best_h = h.copy()
    best_c = float(corr.max())
    A = 2 * n - 1
    p = n - 1
    mr = np.arange(-n + 1, n)
    xs = np.linspace(0.0, 1.0, grid)
    no_improve = 0

    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        hw = h.copy()
        cw = corr.copy()
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            old = hw[i]
            B = np.zeros(A)
            k1 = i - mr
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += 1.0 - hw[k1[m1]]
            k2 = i + mr
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= hw[k2[m2]]
            c0 = cw[p]
            f0 = old * (1.0 - old)
            base = cw - B * old
            b2 = base.copy()
            s2 = B.copy()
            b2[p] = -np.inf
            s2[p] = 0.0
            bestx = old
            bestv = float(cw.max())
            for x in xs:
                val = float(np.max(b2 + s2 * x))
                q = c0 + x * (1.0 - x) - f0
                if q > val:
                    val = q
                if val < bestv:
                    bestv = val
                    bestx = x
            if bestx != old:
                lo = min(1.0, max(0.0, bestx - 0.12))
                hi = max(0.0, min(1.0, bestx + 0.12))
                for x in np.linspace(lo, hi, 11):
                    val = float(np.max(b2 + s2 * x))
                    q = c0 + x * (1.0 - x) - f0
                    if q > val:
                        val = q
                    if val < bestv:
                        bestv = val
                        bestx = x
            if abs(bestx - old) > 1e-14:
                cw = base + B * (bestx - old)
                cw[p] = c0 + bestx * (1.0 - bestx) - f0
                hw[i] = bestx

        hp = project_feasible(hw, n)
        cp = c_counts(hp)
        cm = float(cp.max())
        if cm < best_c - 1e-13:
            h = hp
            corr = cp
            best_h = h.copy()
            best_c = cm
            no_improve = 0
        else:
            no_improve += 1
            h = best_h.copy()
            corr = c_counts(h)
            if no_improve >= 2:
                break
    return best_h, best_c

def _smooth_obj(x):
    h = np.clip(x, 0.0, 1.0)
    c = c_counts(h)
    m = float(np.max(c))
    sh = np.clip(_G_ALPHA * (c - m), -700.0, 0.0)
    z = np.exp(sh)
    s = float(z.sum())
    f = m + (np.log(s) / _G_ALPHA if s > 0.0 else 0.0)
    _G_CACHE[0] = h.copy()
    _G_CACHE[1] = c
    return f / 100.0

def _smooth_jac(x):
    n = _G_N
    h = np.clip(x, 0.0, 1.0)
    ch = _G_CACHE[0]
    if ch is None or not np.allclose(h, ch, rtol=0.0, atol=1e-12):
        c = c_counts(h)
        _G_CACHE[0] = h.copy()
        _G_CACHE[1] = c
    else:
        c = _G_CACHE[1]
    m = float(np.max(c))
    sh = np.clip(_G_ALPHA * (c - m), -700.0, 0.0)
    z = np.exp(sh)
    s = float(z.sum())
    w = z / s if s > 0.0 else np.zeros_like(z)
    d = np.zeros(n)
    p = n - 1
    w0 = float(w[p])
    if w0 > 1e-12:
        d += w0 * (1.0 - 2.0 * h)
    for r in range(1, n):
        wp = float(w[p + r])
        wn = float(w[p - r])
        if wp > 1e-12:
            d[r:] += wp * (1.0 - h[:n - r])
            d[:n - r] -= wp * h[r:]
        if wn > 1e-12:
            d[:n - r] += wn * (1.0 - h[r:])
            d[r:] -= wn * h[:n - r]
    return d / 100.0

def _eq_fun(x):
    return float(x.sum() / _G_N - 0.5)

def _eq_jac(x):
    return np.full(_G_N, 1.0 / _G_N)

def slsqp_smooth(h, n, alpha, maxiter, ftol=1e-16):
    global _G_N, _G_ALPHA
    h = project_feasible(h, n)
    _G_N = n
    _G_ALPHA = float(alpha)
    _G_CACHE[0] = None
    _G_CACHE[1] = None
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': float(ftol)})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h, n)

def lp_step(h, n, c, max_rows=200, delta=None):
    A = 2 * n - 1
    p = n - 1
    cmax = float(c.max())
    idx = np.argsort(c)[::-1]
    if max_rows < idx.size:
        idx = idx[:max_rows]
    if idx.size == 0:
        return None
    G = np.empty((idx.size, n))
    for a, q in enumerate(idx):
        row = G[a]
        row.fill(0.0)
        m = int(q - p)
        if m == 0:
            row[:] = 1.0 - 2.0 * h
        elif m > 0:
            r = m
            row[r:] += 1.0 - h[:n - r]
            row[:n - r] -= h[r:]
        else:
            r = -m
            row[:n - r] += 1.0 - h[r:]
            row[r:] -= h[:n - r]
    b = G @ h - c[idx] + 1e-9
    Au = np.empty((idx.size, n + 1))
    Au[:, :n] = G
    Au[:, n] = -1.0
    Ae = np.zeros((1, n + 1))
    Ae[0, :n] = 1.0
    bo = np.zeros(n + 1)
    bo[-1] = 1.0
    bounds = []
    if delta is None:
        for _ in range(n):
            bounds.append((0.0, 1.0))
    else:
        for i in range(n):
            bounds.append((max(0.0, h[i] - delta), min(1.0, h[i] + delta)))
    bounds.append((0.0, cmax + 1.0))
    try:
        res = linprog(c=bo, A_ub=Au, b_ub=b, A_eq=Ae, b_eq=[n / 2.0],
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.x is None or not res.success:
        return None
    h1 = project_feasible(res.x[:n], n)
    if np.linalg.norm(h1 - h) < 1e-10:
        return None
    return h1

def line_search(h0, h1, n, c0):
    best_h = h0.copy()
    best_c = c0
    for a in (1.0, 0.7, 0.5, 0.35, 0.22, 0.12, 0.06, 0.03):
        hh = (1.0 - a) * h0 + a * h1
        hh = project_feasible(hh, n)
        cc = compute_c(hh, n)
        if cc < best_c - 1e-14:
            best_c = cc
            best_h = hh.copy()
    return best_h, best_c

def lp_polish(h, n, deadline, max_steps=2):
    h = project_feasible(h, n)
    best_c = compute_c(h, n)
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        moved = False
        c = c_counts(h)
        for delta in (None, 0.25, 0.1, 0.05):
            h1 = lp_step(h, n, c, delta=delta)
            if h1 is None:
                continue
            h2, c2 = line_search(h, h1, n, best_c)
            if c2 < best_c - 1e-14:
                h = h2
                best_c = c2
                moved = True
                break
        if not moved:
            break
    return h, best_c

def optimize_task(h, n, deadline, coord_sweeps=4, lp_steps=2,
                  slsqp_alphas=(1000, 10000, 100000, 1000000)):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c(h, n)

    if n <= 800 and time.time() < deadline - 2.0:
        hd, _ = coord_descent(h, n, max_sweeps=coord_sweeps,
                              deadline=time.time() + 0.25 * max(0.0, deadline - time.time()),
                              grid=21)
        c = compute_c(hd, n)
        if c < best_c:
            best_c = c
            best_h = hd.copy()

    if time.time() < deadline - 15.0:
        hd, c2 = lp_polish(best_h, n,
                           time.time() + 0.30 * max(0.0, deadline - time.time()),
                           max_steps=lp_steps)
        if c2 < best_c:
            best_c = c2
            best_h = hd.copy()

    if time.time() < deadline - 15.0:
        if n <= 100:
            nmax = 800
        elif n <= 200:
            nmax = 500
        elif n <= 400:
            nmax = 250
        else:
            nmax = 180
        for alpha in slsqp_alphas:
            if time.time() > deadline - 8.0:
                break
            hd = slsqp_smooth(best_h, n, alpha, nmax, 1e-16)
            c = compute_c(hd, n)
            if c < best_c:
                best_c = c
                best_h = hd.copy()

    if time.time() < deadline - 5.0:
        hd, _ = coord_descent(best_h, n, max_sweeps=3,
                              deadline=deadline - 2.0, grid=18)
        c = compute_c(hd, n)
        if c < best_c:
            best_c = c
            best_h = hd.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(20.0, float(budget_s) - 30.0)
    dl = t0 + total
    rng = np.random.RandomState(seed)

    best_h = None
    best_n = 400
    best_c = np.inf

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size > 0:
            best_n = len(h0)
            best_h = project_feasible(h0, best_n)
            best_c = compute_c(best_h, best_n)
    except Exception:
        best_h = None

    if best_h is None:
        best_n = 200
        best_h = construct_start(best_n, 0, rng, None)
        best_c = compute_c(best_h, best_n)

    task_dl = t0 + 0.38 * (dl - t0)
    h2, c2 = optimize_task(best_h, best_n, task_dl,
                           coord_sweeps=6, lp_steps=2,
                           slsqp_alphas=(1000, 10000, 100000, 1000000))
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    for n in (40, 64, 96, 144, 200, 300):
        if time.time() > dl - 80.0 or n == best_n:
            continue
        rem_budget = 0.045 * max(0.0, dl - time.time())
        if rem_budget < 12.0:
            continue
        starts = 3 if n <= 120 else 2
        sub = rem_budget / float(starts)
        for var in range(starts):
            if time.time() > dl - 70.0:
                break
            hs = construct_start(n, var, rng, best_h)
            h2, c2 = optimize_task(hs, n, time.time() + sub,
                                   coord_sweeps=4, lp_steps=1,
                                   slsqp_alphas=(2000, 20000))
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = n

    if time.time() < dl - 180.0:
        hn = project_feasible(best_h + 0.01 * rng.standard_normal(best_n), best_n)
        h2, c2 = optimize_task(hn, best_n, dl - 80.0,
                               coord_sweeps=4, lp_steps=2,
                               slsqp_alphas=(50000, 500000))
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    if time.time() < dl - 30.0:
        h2, c2 = optimize_task(best_h, best_n, dl - 5.0,
                               coord_sweeps=6, lp_steps=3,
                               slsqp_alphas=(20000, 200000, 2000000))
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c(best_h, best_n)
    return best_h, best_c, best_n
