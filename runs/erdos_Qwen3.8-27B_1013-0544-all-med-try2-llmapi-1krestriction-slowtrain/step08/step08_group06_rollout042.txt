def subgradient_step(h, n, m, step_size):
    """One subgradient step using top m lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Find top m lags
    top_idx = np.argsort(corr)[::-1][:m]
    # Compute gradient of sum of top m correlation values
    grad = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - (n - 1)  # shift
        # For this shift s:
        # ∂corr[n-1+s]/∂h[j] = (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)
        # Vectorized:
        # First term: (1-h[j+s]) for j where 0<=j+s<n, i.e., j in [max(0,-s), min(n-1, n-1-s)]
        # Second term: -h[j-s] for j where 0<=j-s<n, i.e., j in [max(0,s), min(n-1, n-1+s)]
        
        if s >= 0:
            # j+s < n means j < n-s, so j in [0, n-1-s]
            grad[:n-s] += (1 - h[s:])  # (1-h[j+s]) for j=0..n-1-s, h[j+s] = h[s..n-1]
            # j-s >= 0 means j >= s, so j in [s, n-1]
            grad[s:] -= h[:n-s]  # h[j-s] for j=s..n-1, h[j-s] = h[0..n-1-s]
        else:
            t = -s  # positive
            # j+s < n means j < n-s = n+t, always true since j <= n-1 < n+t
            # j >= 0 always
            # So first term: (1-h[j+s]) for all j, where j+s = j-t
            # valid when j-t >= 0, i.e., j >= t
            grad[t:] += (1 - h[:n-t])  # (1-h[j-t]) for j=t..n-1
            # Second term: -h[j-s] = -h[j+t], valid when j+t < n, i.e., j < n-t
            grad[:n-t] -= h[t:]  # h[j+t] for j=0..n-1-t
    
    # Gradient descent step (we want to DECREASE the correlation, so go against gradient)
    h_new = h - step_size * grad
    # Project onto feasible set
    h_new = project_feasible(h_new, n)
    return h_new

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-10:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

    """Compute gradient of sum of top-m correlation values."""
    corr = np.correlate(h, 1 - h, mode='full')
    # Find top m
    if m >= len(corr):

    else:

    for idx in top_idx:

            if s < n:
                grad[s:] -= h[:n-s]

            # For j in [t, n-1]: + (1-h[j-t])

            y_vec = grad  # approximate

            sy = np.dot(s_vec, y_vec)
            if ss > 1e-20 and abs(sy) > 1e-20:

                best_h = h.copy()
        else:
            step *= 0.5
        
        prev_h = h.copy()
        prev_c = c

            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def momentum_gradient(h, n, m, deadline, max_iter=500000, lr=0.01, momentum=0.9):
    """Nesterov momentum gradient descent."""
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead
        y = h - momentum * v
        y = np.clip(y, 0.0, 1.0)
        
        # Compute gradient at y
        corr = np.correlate(y, 1 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:m]
        
        grad = np.zeros(n)
        for idx in top_idx:
            s = idx - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - y[s:])
                if s < n:
                    grad[s:] -= y[:n-s]
            else:
                t = -s
                grad[t:] += (1.0 - y[:n-t])
                grad[:n-t] -= y[t:]
        
        # Update
        v = momentum * v + lr * grad
        h = h - v
        h = project_feasible(h, n)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # Reduce learning rate if stuck
            if it % 1000 == 0:
                lr *= 0.9
    
    return best_h, best_c

**Analytical insight:**
The problem is:
min_h max_s sum_i h[i]*(1-h[i+s]) * dx
s.t. sum h[i] = n/2, 0 <= h[i] <= 1

At the optimum, by the KKT conditions (if the max is achieved at a unique s*), we'd have:
∇_h [sum_i h[i]*(1-h[i+s*])] = λ * 1 (for the equality constraint) + μ (for inequality constraints)

Where λ and μ are Lagrange multipliers.

∂/∂h[k] of sum_i h[i]*(1-h[i+s*]):
= (1-h[k+s*])*[k < n-s*] - h[k-s*]*[k >= s*] (for s* > 0)

At the optimum (interior points, 0 < h[k] < 1):
(1-h[k+s*])*[k < n-s*] - h[k-s*]*[k >= s*] = λ

This gives a system of equations that determines the optimal h.

But since the max might be achieved at multiple s values (which is likely at the optimum due to the "balancing" effect), the KKT conditions become more complex.

For a practical algorithm, the "balancing" intuition suggests: at the optimum, the top correlation values should be approximately equal. We can use this as a heuristic:
- Compute the top correlations
- If the top value is much larger than the second, focus on reducing the top
- If they're close, try to reduce them all simultaneously

This is essentially what the log-sum-exp / top-m approach does.
