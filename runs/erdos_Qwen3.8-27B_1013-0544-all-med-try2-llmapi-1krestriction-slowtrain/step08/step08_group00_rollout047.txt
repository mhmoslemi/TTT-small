    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_grad_single(h, n, k):
    """Gradient of C_k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:])
        grad[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] += (-h[kk:])
    return grad * dx

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = h.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def smooth_max_grad(h, n, alpha, mu):

    dx = 2.0 / n
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)

        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    s = h.sum() - n / 2.0
    grad += 2.0 * mu * s

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_grad_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:])
        grad[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] += (-h[kk:])
    return grad * dx

def _obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = h.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def _grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    s = h.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_step(h, n, alpha, mu, maxit, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-18,
                                'gtol': 1e-16, 'maxfun': maxit * 30,
                                'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def subgrad_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.02
    for it in range(50000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        cmax = c.max()
        tol = max(0.0005, (cmax - c.min()) * 0.05)
        active = np.where(c > cmax - tol)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += c5_grad_k(h, n, k)
        grad /= len(active)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        improved = False
        step = lr
        for _ in range(8):
            h_try = project_feasible(h - step * grad * gnorm, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                h = h_try
                improved = True
                break
            step *= 0.4
        if improved:
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.6
            if lr < 1e-7:
                break
    return h, best_c

def grad_pair_search(h, n, deadline):

    dx = 2.0 / n
    for d in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]:

                else:

                    nv = n - kk

            grad -= grad.mean()
            inc = np.argsort(grad)[:30]

def slsqp_polish(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    def eq_fun(x, nn):
        return float(x.sum() - nn / 2.0)
    def eq_jac(x, nn):
        return np.ones(nn)
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha, 0.0),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun,
                                     'jac': eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)
