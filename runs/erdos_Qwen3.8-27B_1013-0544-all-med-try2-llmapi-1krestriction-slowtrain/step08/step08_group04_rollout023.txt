def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu, top_only=None):
    c = compute_c5_vec(h, n)
    m = c.max()
    if top_only is not None:
        active = c > (m - 0.002)
        if active.sum() > top_only:
            thr = np.sort(c)[-top_only]
            active = c >= thr
    else:
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        active = w > 1e-10
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        k = k_idx - (n - 1)
        wm = 1.0
        if top_only is None:
            wm = w[k_idx]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * dx * (1.0 - h[k:n])
            grad[k:n] += wm * dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * dx * (1.0 - h[0:nv])
            grad[0:nv] += wm * dx * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, deadline, alphas, mus, maxit=250):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in zip(alphas, mus):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_val, h, jac=smooth_max_grad,
                          args=(n, alpha, mu, None), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 20,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_grad_descent(h, n, deadline, lr0=0.004, topk=8, eps=0.001):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    for it in range(30000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        m = c.max()
        active = c >= (m - eps)
        if active.sum() > topk:
            thr = np.sort(c)[-topk]
            active = c >= thr
        if not active.any():
            break
        dx = 2.0 / n
        grad = np.zeros(n)
        for k_idx in np.where(active)[0]:
            k = k_idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] += dx * (-h[kk:n])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        g = grad / gnorm
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c

def local_pair_search(h, n, deadline, deltas, radius=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    total_n = 2 * n - 1
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        si, sj = h[i], h[j]
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] = si; h[j] = sj
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        si, sj = h[i], h[j]
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] = si; h[j] = sj
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def grad_guided_pairs(h, n, deadline, step=0.003, topk=8):
    h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            m = c.max()
            active = c >= (m - 0.002)
            if active.sum() > topk:
                thr = np.sort(c)[-topk]
                active = c >= thr
            active_idx = np.where(active)[0]
            dx = 2.0 / n
            grad = np.zeros(n)
            for k_idx in active_idx:
                k = k_idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += dx * (1.0 - h[k:n])
                    grad[k:n] += dx * (-h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += dx * (1.0 - h[0:nv])
                    grad[0:nv] += dx * (-h[kk:n])
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if abs(grad[i] - grad[j]) < 1e-10:
                        continue
                    move = -step * (grad[i] - grad[j])
                    if abs(move) > 0.1:
                        move = np.sign(move) * 0.1
                    if h[i] + move >= -1e-12 and h[i] + move <= 1.0 + 1e-12 and \
                       h[j] - move >= -1e-12 and h[j] - move <= 1.0 + 1e-12:
                        si, sj = h[i], h[j]
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] = si; h[j] = sj
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def coarse_optimize(h_parent, n_parent, n_coarse, deadline):
    x_old = np.linspace(0, 2, n_parent)
    x_new = np.linspace(0, 2, n_coarse)
    h_c = np.interp(x_new, x_old, h_parent)
    h_c = project_feasible(h_c, n_coarse)
    alphas = [50, 200, 1000, 5000, 50000, 500000, 2000000]
    mus = [1.0, 1.0, 5.0, 20.0, 50.0, 200.0, 1000.0]
    h_c, c_c = lbfgs_run(h_c, n_coarse, deadline, alphas, mus, maxit=200)
    if time.time() < deadline - 5:
        h_c, c_c = active_grad_descent(h_c, n_coarse, deadline, lr0=0.005, topk=6, eps=0.001)
    return h_c, c_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS with increasing alpha
    t1 = min(t0 + 180, deadline - 350)
    alphas1 = [200, 1000, 5000, 50000, 500000, 2000000, 10000000]
    mus1 = [1.0, 1.0, 5.0, 50.0, 200.0, 1000.0, 5000.0]
    best_h, best_c = lbfgs_run(best_h, n_init, t1, alphas1, mus1, maxit=200)

    # Phase 2: Active-set gradient descent
    if time.time() < deadline - 200:
        t2 = min(time.time() + 80, deadline - 200)
        h2, c2 = active_grad_descent(best_h, n_init, t2, lr0=0.003, topk=8, eps=0.0008)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Coarse grid exploration
    if time.time() < deadline - 250:
        best_n = n_init
        for n_try in [100, 80, 60, 150]:
            if time.time() > deadline - 120:
                break
            alloc = min(70, deadline - time.time() - 60)
            if alloc < 35:
                break
            h_t, c_t = coarse_optimize(best_h, len(best_h), n_try, time.time() + alloc)
            if c_t < best_c:
                best_c = c_t
                x_old = np.linspace(0, 2, n_try)
                x_new = np.linspace(0, 2, best_n)
                best_h = np.interp(x_new, x_old, h_t)
                best_h = project_feasible(best_h, best_n)
                best_n = best_n

    # Phase 4: Refine on best grid - L-BFGS with very high alpha
    nf = len(best_h)
    if time.time() < deadline - 150:
        t4 = min(time.time() + 70, deadline - 150)
        alphas4 = [10000, 100000, 1000000, 5000000]
        mus4 = [50.0, 200.0, 1000.0, 5000.0]
        best_h, best_c = lbfgs_run(best_h, nf, t4, alphas4, mus4, maxit=200)

    # Phase 5: Active-set gradient descent again
    if time.time() < deadline - 100:
        t5 = min(time.time() + 50, deadline - 100)
        h5, c5 = active_grad_descent(best_h, nf, t5, lr0=0.002, topk=5, eps=0.0005)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Gradient-guided pair search
    if time.time() < deadline - 80:
        t6 = min(time.time() + 50, deadline - 80)
        best_h, best_c = grad_guided_pairs(best_h, nf, t6, step=0.002, topk=6)

    # Phase 7: Fine pair search with decreasing steps
    if time.time() < deadline - 40:
        t7 = min(time.time() + 40, deadline - 40)
        best_h, best_c = local_pair_search(best_h, nf, t7,
                                           [0.002, 0.001, 0.0005, 0.0002], radius=35)

    # Phase 8: Even finer tweaks
    if time.time() < deadline - 15:
        t8 = min(time.time() + 20, deadline - 15)
        best_h, best_c = local_pair_search(best_h, nf, t8,
                                           [0.0005, 0.0002, 0.0001, 0.00005], radius=25)

    # Phase 9: Final L-BFGS polish
    if time.time() < deadline - 3:
        best_h, best_c = lbfgs_run(best_h, nf, deadline - 1,
                                   [5000000, 20000000], [2000.0, 10000.0], maxit=80)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
