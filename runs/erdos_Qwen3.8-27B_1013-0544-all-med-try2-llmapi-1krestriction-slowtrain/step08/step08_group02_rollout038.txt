def fast_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    sig = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for m_idx in sig:
        s = int(m_idx - (n - 1))
        wm = w[m_idx]
        if s >= 0:
            grad[s:] += wm * (1.0 - h[:n - s])
            grad[:n - s] += -wm * h[s:]
        else:
            t = -s
            grad[:n - t] += wm * (1.0 - h[t:])
            grad[t:] += -wm * h[:n - t]
    return grad

def smooth_max_obj(h, n, alpha, mu):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    val = m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha
    s = h.sum() - n / 2.0
    return val + mu * s * s

def smooth_max_jac(h, n, alpha, mu):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    sig = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for m_idx in sig:
        s = int(m_idx - (n - 1))
        wm = w[m_idx]
        if s >= 0:
            grad[s:] += wm * (1.0 - h[:n - s])
            grad[:n - s] += -wm * h[s:]
        else:
            t = -s
            grad[:n - t] += wm * (1.0 - h[t:])
            grad[t:] += -wm * h[:n - t]
    s_sum = h.sum() - n / 2.0
    grad += 2.0 * mu * s_sum
    return grad

def lbfgs_opt(h, n, deadline, alphas=None, mu=100.0, max_iter=300):
    if alphas is None:
        alphas = [500.0, 5000.0, 50000.0, 500000.0]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_jac,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': max_iter * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_sweeps=20000):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    best_c5 = c.max() * (2.0 / n)
    n_shifts = 2 * n - 1
    s_vals = np.arange(n_shifts) - (n - 1)
    n_mid = n - 1
    concave_c = c[n_mid] - h[j0] * (1.0 - h[j0]) if False else None

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            a_lin = np.zeros(n_shifts)
            v1 = (s_vals <= j) & (s_vals >= j - (n - 1)) & (s_vals >= -(n-1)) & (s_vals <= (n-1))
            a_lin[v1] += 1.0 - h[j - s_vals[v1]]
            v2 = (s_vals >= -j) & (s_vals <= n - 1 - j) & (s_vals >= -(n-1)) & (s_vals <= (n-1))
            a_lin[v2] -= h[j + s_vals[v2]]

            concave_c = c[n_mid] - h[j] * (1.0 - h[j])
            y = h[j]
            lo, hi = 0.0, 1.0
            for _ in range(45):
                ym = 0.5 * (lo + hi)
                vm = c + (ym - y) * a_lin
                vm[n_mid] = (ym - ym * ym) + concave_c
                m = vm.max()
                am = np.argmax(vm)
                if am == n_mid:
                    sg = 1.0 - 2.0 * ym
                else:
                    sg = a_lin[am]
                if sg > 1e-14:
                    hi = ym
                elif sg < -1e-14:
                    lo = ym
                else:
                    break
            y_opt = 0.5 * (lo + hi)
            delta = y_opt - y
            if abs(delta) > 1e-14:
                c += delta * a_lin
                c[n_mid] = float((y_opt - y_opt * y_opt) + concave_c)
                h[j] = y_opt

        if sweep % 200 == 0:
            c5 = c.max() * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
            h_p = project_feasible(h, n)
            h = h_p
            c = np.correlate(h, 1 - h, mode='full')

    c5 = c.max() * (2.0 / n)
    if c5 < best_c5:
        best_c5 = c5
    return h, best_c5

def binary_sa(h, n, deadline, rng, max_flips=2000000):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    temp = 0.015
    flips = 0
    while time.time() < deadline and flips < max_flips:
        if flips % 40000 == 0 and flips > 0:
            temp *= 0.82
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = compute_c5(h_bin, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_bin.copy()
        elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-10)):
            pass
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        flips += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.full(n_init, 0.5)
        x = np.linspace(0, 2, n_init, endpoint=False)
        for freq in range(1, 10):
            h_init += 0.3 * np.cos(2 * np.pi * freq * x / 2.0 + rng.uniform(0, 2 * np.pi))

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Coordinate descent (main optimizer)
    t1 = min(time.time() + 400, deadline - 300)
    h_cd, c_cd = coord_descent(best_h, best_n, t1)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd

    # Phase 2: L-BFGS-B polish
    if time.time() < deadline - 250:
        t2 = min(time.time() + 80, deadline - 200)
        h_lb, c_lb = lbfgs_opt(best_h, best_n, t2)
        if c_lb < best_c:
            best_c = c_lb
            best_h = h_lb

    # Phase 3: Another CD round from polished solution
    if time.time() < deadline - 180:
        t3 = min(time.time() + 100, deadline - 130)
        h_cd2, c_cd2 = coord_descent(best_h, best_n, t3)
        if c_cd2 < best_c:
            best_c = c_cd2
            best_h = h_cd2

    # Phase 4: Binary SA for diverse exploration
    if time.time() < deadline - 200:
        t4a = min(time.time() + 60, deadline - 140)
        h_b, c_b = binary_sa(best_h, best_n, t4a, rng)
        if c_b < best_c * 1.001:
            t4b = min(time.time() + 40, deadline - 80)
            h_p, c_p = lbfgs_opt(h_b, best_n, t4b)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 5: Try different resolutions
    for n_try in [1000, 2000, 375]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        alloc = min(35, (deadline - time.time() - 60) * 0.6)
        if alloc < 15:
            continue
        h_t, c_t = coord_descent(h_int, n_try, time.time() + alloc)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try
        if time.time() < deadline - 40:
            h_t, c_t = lbfgs_opt(h_t, n_try, min(time.time() + 20, deadline - 30),
                                alphas=[5000.0, 50000.0, 200000.0])
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 6: Final L-BFGS-B with high alpha
    if time.time() < deadline - 10:
        t6 = min(time.time() + 30, deadline - 5)
        h_f, c_f = lbfgs_opt(best_h, best_n, t6,
                            alphas=[5000.0, 50000.0, 200000.0, 1000000.0],
                            mu=50.0, max_iter=500)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final CD polish
    if time.time() < deadline - 3:
        h_f, c_f = coord_descent(best_h, best_n, deadline - 2)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
