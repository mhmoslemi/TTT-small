from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    s = h.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def eq_fun(h, n):
    return float(h.sum() - n / 2.0)

def eq_jac(h, n):
    return np.ones(n)

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        mu = 1e-6 * alpha
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-20,
                                  'gtol': 1e-18, 'maxfun': max_iters * 30,
                                  'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=5000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_local_search(h, n, deadline, k_top=30, max_rounds=200):
    """Gradient-guided local search with adaptive steps."""
    best_c = compute_c5(h, n)
    alpha = 500000
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for step in steps:
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            g = smooth_max_grad(h, n, alpha, 0.0)
            # Find best direction: increase where grad is most negative, decrease where most positive
            # But we need sum to be preserved, so we do paired moves
            up = np.argsort(g)[:k_top]  # most negative grad -> increase h
            down = np.argsort(g)[-k_top:]  # most positive grad -> decrease h
            improved = False
            for i in up:
                if time.time() > deadline:
                    break
                for j in down:
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                        if improved:
                            break
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smo_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def _smo_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-14:
            continue
        k = ki - (n - 1)
        wm = w[ki] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    s = h.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def _eq(h, n):
    return float(h.sum() - n / 2.0)

def _eqj(h, n):
    return np.ones(n)

def _slq_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def lbfgs_opt(h, n, deadline, max_iters=1000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [200, 1000, 5000, 25000, 100000, 500000, 2000000, 10000000, 50000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        mu = max(1.0, 1e-5 * alpha)
        try:
            res = minimize(_smo_obj, h, jac=_smo_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': max_iters * 40,
                                   'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new

        except Exception:

    return h, best_c

def slsqp_opt(h, n, deadline, alpha=50000000, maxiter=10000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj, 'args': (n,)}]

def grad_local(h, n, deadline, k_top=50, max_rounds=500):

        xn = np.linspace(0, 2, n_f, endpoint=False)
        hi = project_feasible(np.interp(xn, xo, best_h), n_f)
        alloc = min(120, (deadline - time.time() - 120) * 0.45)
        if alloc > 40:
            ht, ct = lbfgs_opt(hi, n_f, time.time() + alloc, max_iters=600)
            if ct < best_c:
                best_c, best_h, best_n = ct, ht, n_f

    # Phase 5: Diverse restarts
    starts = [binary_start(best_n, rng),
              arithmetic_start(best_n, rng),
              project_feasible(best_h + rng.randn(best_n) * 0.03, best_n),
              project_feasible(best_h + rng.randn(best_n) * 0.08, best_n)]
    for s in starts:
        if time.time() > deadline - 60:
            break
        alloc = min(30, (deadline - time.time() - 40) * 0.35)
        if alloc < 12:
            break
        hs, cs = lbfgs_opt(s, best_n, time.time() + alloc, max_iters=400)
        if cs < best_c:
            best_c, best_h = cs, hs

    # Phase 6: Second L-BFGS-B pass on best (with highest alphas only)
    if time.time() < deadline - 40:
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [50000000, 200000000]:
            if time.time() > deadline - 5:
                break
            mu = max(1.0, 1e-5 * alpha)
            try:
                res = minimize(_smo_obj, best_h, jac=_smo_grad,
                            args=(best_n, alpha, mu), method='L-BFGS-B',
                            bounds=bounds,
                            options={'maxiter': 500, 'ftol': 1e-20,
                                     'gtol': 1e-18, 'maxfun': 20000,
                                     'maxls': 60})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c, best_h = c_new, h_new
            except Exception:
                pass

    # Phase 7: Final local search + SLSQP
    if time.time() < deadline - 10:
        best_h, best_c = grad_local(best_h, best_n, deadline - 8, k_top=60, max_rounds=300)
    if time.time() < deadline - 3:
        hf, cf = slsqp_opt(best_h, best_n, deadline - 2, alpha=100000000, maxiter=5000)
        if cf < best_c:
            best_c, best_h = cf, hf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
