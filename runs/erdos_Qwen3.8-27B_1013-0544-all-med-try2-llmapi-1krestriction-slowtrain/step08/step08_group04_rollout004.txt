      if tot<1e-15: break

      add=min(diff,1-h[i])

      i=int(np.argmin(h))
      add=max(diff,h[i]) # negative

  return h

import numpy as np
from scipy.optimize import minimize
import time

def cv(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def cc(h, n):
    return float(np.max(cv(h, n)))

def proj(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    t = n / 2.0
    for _ in range(60):
        s = h.sum(); d = t - s
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; q = r.sum()
            if q < 1e-15: break
            h += d * r / q
        else:
            q = h.sum()
            if q < 1e-15: break
            h += d * h / q
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum(); d = t - s
        if abs(d) < 1e-13: break
        if d > 0:
            i = int(np.argmax(1.0 - h)); e = min(d, 1.0 - h[i])
        else:
            i = int(np.argmax(h)); e = -min(-d, h[i])
        if abs(e) < 1e-15: break
        h[i] += e
    return np.clip(h, 0.0, 1.0)

def resample(h, m):
    if len(h) == m: return h.copy()
    return np.interp(np.linspace(0.0, 2.0, m), np.linspace(0.0, 2.0, len(h)), h)

def variants(h, n):

    v = proj((base + base[::-1]) * 0.5, n); c = cc(v, n)
    if c < bc: best, bc = v, c
    for a in (1.5, 2.0, 3.0):

def grad(h, n, w):
    g = np.zeros(n)
    for idx, wm in enumerate(w):
        if wm <= 1e-13: continue
        k = idx - (n - 1)
        if k == 0:
            g += wm * (1.0 - 2.0 * h)
        elif k > 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:n])
            g[k:n] -= wm * h[:nv]
        else:
            kk = -k; nv = n - kk
            g[kk:n] += wm * (1.0 - h[:nv])
            g[:nv] -= wm * h[kk:n]
    return g

def soft(h, n, a):
    L = cv(h, n); z = a * L; m = z.max()
    e = np.exp(z - m); s = e.sum()
    return (m + np.log(s)) / a, e / s

def lbfun(x, n, a, mu):
    v, g = soft(x, n, a)
    return v + mu * (x.sum() - n / 2.0) ** 2

def lbjac(x, n, a, mu):
    v, g = soft(x, n, a)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs(h, n, dl):

def bcorr(b):

    c = np.correlate(a, 1 - a, mode='full')
    return c[n - 1:].astype(np.int32).copy(), c[:n - 1][::-1].astype(np.int32).copy()

def bflip(b, lp, ln, p, one):
    n = len(b); r = n - p
    if one:
        if r > 1:
            x = b[p + 1:n]
            lp[1:r] += 1 - x
            ln[1:r] -= x
        if p > 0:
            x = b[:p]
            lp[1:p + 1] -= x
            ln[1:p + 1] += 1 - x
    else:
        if r > 1:
            x = b[p + 1:n]
            lp[1:r] -= 1 - x
            ln[1:r] += x
        if p > 0:
            x = b[:p]
            lp[1:p + 1] += x
            ln[1:p + 1] -= 1 - x
    b[p] = 1 if one else 0

def bswap(b, lp, ln, p1, p0, m, temp, rng):
