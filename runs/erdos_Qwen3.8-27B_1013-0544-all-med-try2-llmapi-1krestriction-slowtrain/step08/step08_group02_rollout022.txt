def binary_flip(h, n, rng, n_flips=10000):

    target = n // 2

        rem = rng.choice(ones, size=cnt - target, replace=False)
        hb[rem] = 0
    elif cnt < target:
        zeros = np.where(hb == 0)[0]
        add = rng.choice(zeros, size=target - cnt, replace=False)
        hb[add] = 1
    best_c = compute_c5(hb, n)
    best_h = hb.copy()
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    for _ in range(n_flips):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i] = 0
        hb[j] = 1
        c = compute_c5(hb, n)
        if c < best_c:
            best_c = c
            best_h = hb.copy()
        else:
            hb[i] = 1
            hb[j] = 0
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
    return best_h, best_c

    hb = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(hb.sum())
    if cnt > target:
        idx = np.where(hb == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        hb[rem] = 0
    elif cnt < target:
        idx = np.where(hb == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        hb[add] = 1
    best_c = compute_c5(hb, n)
    best_h = hb.copy()
    n_try = min(20000, int((deadline - time.time()) * 200))
    for _ in range(max(n_try, 100)):
        if time.time() > deadline - 2:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i], hb[j] = 0.0, 1.0
        c = compute_c5(hb, n)
        if c < best_c - 1e-15:
            best_c, best_h = c, hb.copy()
        else:
            hb[i], hb[j] = 1.0, 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.zeros(n)
        h[:n // 2] = 1.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_n = n
    # Phase 1: Adam (250s)
    t1 = min(t0 + 250, deadline - 400)
    hr, cr = adam_optimize(h, n, t1, alpha=120, topK=100, lr=0.035, mu=14.0)
    if cr < best_c:
        best_c, best_h = cr, hr
    # Phase 2: L-BFGS (70s)
    if time.time() < deadline - 320:
        t2 = min(time.time() + 60, deadline - 300)
        hr, cr = lbfgs_polish(best_h, best_n, t2, alpha=40000)
        if cr < best_c:
            best_c, best_h = cr, hr
    # Phase 3: Adam round 2 different params (120s)
    if time.time() < deadline - 220:
        t3 = min(time.time() + 100, deadline - 200)
        hr, cr = adam_optimize(best_h, best_n, t3, alpha=60, topK=150, lr=0.018, mu=10.0)
        if cr < best_c:
            best_c, best_h = cr, hr
    # Phase 4: Binary snap + re-optimize (60s)
    if time.time() < deadline - 180:
        t4 = min(time.time() + 30, deadline - 150)
        hb, cb = binary_snap_optimize(best_h, best_n, t4, rng)
        if cb < best_c + 0.003:
            h_soft = np.where(hb > 0.5, 0.85, 0.15)
            h_soft = project_feasible(h_soft, best_n)
            t4b = min(time.time() + 30, deadline - 100)
            hr, cr = adam_optimize(h_soft, best_n, t4b, alpha=100, topK=80, lr=0.02, mu=12.0)
            if cr < best_c:
                best_c, best_h = cr, hr
    # Phase 5: Upscale to 750 (150s)
    if time.time() < deadline - 180:
        h750 = upsample(best_h, best_n, 750)
        t5 = min(time.time() + 90, deadline - 100)
        h750, c750 = adam_optimize(h750, 750, t5, alpha=80, topK=100, lr=0.025, mu=12.0)
        if time.time() < deadline - 60:
            h750, c750 = lbfgs_polish(h750, 750, min(time.time() + 25, deadline - 50), alpha=60000)
        if c750 < best_c:
            best_c, best_h, best_n = c750, h750, 750
    # Phase 6: Upscale to 1000 (120s)
    if time.time() < deadline - 120:
        h1000 = upsample(best_h, best_n, 1000)
        t6 = min(time.time() + 70, deadline - 50)
        h1000, c1000 = adam_optimize(h1000, 1000, t6, alpha=70, topK=100, lr=0.02, mu=10.0)
        if time.time() < deadline - 30:
            h1000, c1000 = lbfgs_polish(h1000, 1000, min(time.time() + 15, deadline - 20), alpha=80000)
        if c1000 < best_c:
            best_c, best_h, best_n = c1000, h1000, 1000
    # Phase 7: Final Adam refinement (50s)
    if time.time() < deadline - 25:
        t7 = min(time.time() + 40, deadline - 20)
        hr, cr = adam_optimize(best_h, best_n, t7, alpha=300, topK=30, lr=0.008, mu=20.0)
        if cr < best_c:
            best_c, best_h = cr, hr
    # Phase 8: Final L-BFGS
    if time.time() < deadline - 5:
        hr, cr = lbfgs_polish(best_h, best_n, deadline - 3, alpha=200000)
        if cr < best_c:
            best_c, best_h = cr, hr
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
