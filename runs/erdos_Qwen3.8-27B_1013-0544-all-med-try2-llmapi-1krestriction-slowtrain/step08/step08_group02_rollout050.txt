import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    return float(m + np.log(np.sum(np.exp(a - a_max))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-10:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            if k == 0:
                grad[:] += wm * (1.0 - 2.0 * x)
            else:
                grad[k:] += wm * (1.0 - x[:n - k])
                grad[:n - k] += wm * (-x[k:])
        else:
            kk = -k
            if kk == 0:
                grad[:] += wm * (1.0 - 2.0 * x)
            else:
                grad[:n - kk] += wm * (1.0 - x[kk:])
                grad[kk:] += wm * (-x[:n - kk])
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        mu = alpha * 0.01
        try:
            res = minimize(pen_obj, best_h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': 5000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

    # Ensure exactly n//2 ones

    except Exception:
        n = 500
        h = rng.rand(n)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B with increasing alpha (schedule)
    t1 = min(t0 + 250, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, n, t1,
        alphas=[100, 500, 2000, 10000, 50000, 200000], maxiter=500)

    # Phase 2: Alternating binary SA + continuous polish (main search)
    n_cycles = 8
    for cyc in range(n_cycles):
        if time.time() > deadline - 80:
            break
        seg_time = min(90, (deadline - time.time() - 60) / max(n_cycles - cyc, 1))
        t_bin = time.time() + min(seg_time * 0.55, seg_time - 15)
        h_b, c_b = binary_sa(best_h, n, t_bin, rng, max_evals=400000)
        c_test = compute_c5(h_b, n)
        if c_test < best_c:
            best_c = c_test
            best_h = h_b.copy()
        if time.time() < deadline - 30:
            t_cont = min(time.time() + min(seg_time * 0.4, 30), deadline - 20)
            h_c, c_c = lbfgs_optimize(h_b, n, t_cont,
                alphas=[50000, 200000, 1000000, 5000000], maxiter=300)
            if c_c < best_c:
                best_c = c_c
                best_h = h_c

    # Phase 3: Diverse perturbations from best solution
    if time.time() < deadline - 120:
        for n_flips in [5, 10, 15, 20, 30]:
            if time.time() > deadline - 30:
                break
            h_pert = perturb_binary(to_binary(best_h, n, rng), n, rng, n_flips)
            t_alloc = min(35, (deadline - time.time() - 20) * 0.4)
            if t_alloc < 10:
                break
            h_r, c_r = lbfgs_optimize(h_pert, n, time.time() + t_alloc,
                alphas=[50000, 200000, 1000000], maxiter=300)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 4: Final high-alpha polish
    if time.time() < deadline - 8:
        t_final = deadline - 4
        best_h, best_c = lbfgs_optimize(best_h, n, t_final,
            alphas=[1000000, 5000000, 10000000], maxiter=500)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
