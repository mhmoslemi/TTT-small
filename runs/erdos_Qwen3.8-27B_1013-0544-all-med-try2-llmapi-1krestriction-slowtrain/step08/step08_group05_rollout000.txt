import numpy as np
import time
import math
from scipy.optimize import minimize

_OPT_DEADLINE = 0.0
_CONST_N = 0
_BEST_SLSQP_H = None
_BEST_SLSQP_VAL = float("inf")


class OptTimeout(Exception):
    pass


def opt_check():
    if time.time() > _OPT_DEADLINE:
        raise OptTimeout()


def overlap_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_sum_box(z, target):
    z = np.asarray(z, dtype=float)
    n = z.size
    if n == 0:
        return z
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    return adj_sum(p, target)


def adj_sum(h, target):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3):
        diff = float(target - np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            cand = np.flatnonzero(h < 1.0)
        else:
            cand = np.flatnonzero(h > 0.0)
        if cand.size == 0:
            break
        h[cand] += diff / float(cand.size)
        h = np.clip(h, 0.0, 1.0)

    diff = float(target - np.sum(h))
    if abs(diff) > 1e-10:
        if diff > 0.0:
            cand = np.argsort(1.0 - h)[::-1]
        else:
            cand = np.argsort(h)
        rem = diff
        for idx in cand:
            if abs(rem) < 1e-15:
                break
            if rem > 0.0:
                add = min(1.0 - h[idx], rem)
            else:
                add = -min(h[idx], -rem)
            h[idx] += add
            rem -= add
    return np.clip(h, 0.0, 1.0)


def score_h(h, n):
    c = overlap_counts(h)
    return float(np.max(c) * (2.0 / float(n)))


def smooth_value(c, alpha):
    m = float(np.max(c))
    if alpha <= 0.0:
        return m
    z = alpha * (c - m)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        return m
    return m + math.log(s) / alpha


def smooth_grad(h, n, alpha, c):
    g = np.zeros(n, dtype=float)
    m = float(np.max(c))
    z = alpha * (c - m)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        return g
    w = e / s
    act = np.where(w > 1e-8)[0]
    if act.size == 0:
        act = np.array([int(np.argmax(c))], dtype=int)

    h1 = 1.0 - h
    for idx in act:
        wt = float(w[idx])
        l = n - 1 - int(idx)
        if l > 0:
            a = n - l
            g[:a] += wt * h1[l:]
            g[l:] -= wt * h[:a]
        elif l < 0:
            a = -l
            b = n - a
            g[a:] += wt * h1[:b]
            g[:b] -= wt * h[a:]
        else:
            g += wt * (1.0 - 2.0 * h)
    return g


def pg_polish(h, n, alpha, deadline, maxiter=200000):
    target = 0.5 * float(n)
    h = project_sum_box(h, target)
    c = overlap_counts(h)
    ex = float(np.max(c))
    val = smooth_value(c, alpha)
    grad = smooth_grad(h, n, alpha, c)
    grad = grad - float(np.mean(grad))

    best_h = h.copy()
    best_ex = ex
    best_c = float(ex * (2.0 / float(n)))

    step = 0.10
    stall = 0
    last_val = val

    for _ in range(maxiter):
        if time.time() >= deadline:
            break
        gn = float(np.max(np.abs(grad)))
        if gn < 1e-13:
            break

        accepted = False
        t = step
        old_val = val
        old_ex = ex
        for _ in range(32):
            if time.time() >= deadline:
                break
            if t < 1e-11:
                break
            hn = project_sum_box(h - t * grad, target)
            nc = overlap_counts(hn)
            nmax = float(np.max(nc))
            nval = smooth_value(nc, alpha)
            if nval < old_val - 1e-12 and nmax <= old_ex + 1e-8:
                h = hn
                c = nc
                ex = nmax
                val = nval
                accepted = True
                if nval < old_val - 1e-8:
                    step = min(t * 1.15, 0.45)
                else:
                    step = max(t * 0.92, 1e-9)
                break
            t *= 0.5

        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break

        if val < last_val - 1e-14:
            last_val = val
            stall = 0
        else:
            stall += 1

        if ex < best_ex - 1e-11:
            best_ex = ex
            best_c = float(ex * (2.0 / float(n)))
            best_h = h.copy()

        if stall > 1200:
            break
        grad = smooth_grad(h, n, alpha, c)
        grad = grad - float(np.mean(grad))

    return best_h, best_c


def slsqp_obj(x, alpha, n):
    global _BEST_SLSQP_VAL, _BEST_SLSQP_H
    c = overlap_counts(x)
    val = smooth_value(c, alpha)
    ex = float(np.max(c) * (2.0 / float(n)))
    if ex < _BEST_SLSQP_VAL - 1e-11:
        _BEST_SLSQP_VAL = ex
        _BEST_SLSQP_H = np.array(x, dtype=float, copy=True)
    opt_check()
    return val


def slsqp_jac(x, alpha, n):
    opt_check()
    c = overlap_counts(x)
    return smooth_grad(x, n, alpha, c)


def slsqp_eq(x):
    return float(x.sum() - 0.5 * float(_CONST_N))


def slsqp_eq_jac(x):
    return np.ones_like(x, dtype=float)


def slsqp_solve(h, n, alpha, deadline, maxiter):
    global _OPT_DEADLINE, _CONST_N, _BEST_SLSQP_H, _BEST_SLSQP_VAL

    target = 0.5 * float(n)
    h0 = project_sum_box(h, target)
    c0 = score_h(h0, n)

    _BEST_SLSQP_VAL = c0
    _BEST_SLSQP_H = np.array(h0, dtype=float, copy=True)
    _OPT_DEADLINE = deadline
    _CONST_N = n

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slsqp_eq, "jac": slsqp_eq_jac}]

    bx = None
    try:
        res = minimize(
            slsqp_obj,
            h0,
            args=(alpha, n),
            method="SLSQP",
            jac=slsqp_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-16, "maxls": 50},
        )
        if res is not None and getattr(res, "x", None) is not None:
            bx = res.x
    except (OptTimeout, Exception):
        pass

    best = h0
    bestc = c0
    cand = [h0]
    if _BEST_SLSQP_H is not None:
        cand.append(_BEST_SLSQP_H)
    if bx is not None:
        cand.append(np.asarray(bx, dtype=float))

    for x in cand:
        x2 = proj_fix = project_sum_box(x, target)
        x2 = adj_sum(x2, target)
        c2 = score_h(x2, n)
        if c2 < bestc:
            bestc = c2
            best = x2
    return best


def resize_h(h, n_old, n_new):
    if n_old == n_new:
        return project_sum_box(h, 0.5 * float(n_new))
    xo = np.linspace(0.0, 2.0, int(n_old))
    xn = np.linspace(0.0, 2.0, int(n_new))
    z = np.interp(xn, xo, h)
    return project_sum_box(z, 0.5 * float(n_new))


def kronecker_h(n):
    phi = 0.5 * (1.0 + math.sqrt(5.0))
    x = np.mod(np.arange(n) * phi, 1.0)
    return (x < 0.5).astype(float)


def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=float)
    if n == 0:
        return h
    k = int(n // 2)
    if k > 0:
        h[rng.choice(n, k, replace=False)] = 1.0
    if n % 2 == 1:
        free = np.flatnonzero(h == 0.0)
        if free.size:
            h[free[0]] = 0.5
    return h
