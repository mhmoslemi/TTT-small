            if k >= 0:
                nv = n - k
                g[:nv] += wm * (1.0 - x[k:])
                g[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[kk:])
        g += 2.0 * mu * (x.sum() - target)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-17,
                               'maxfun': maxiter * 50, 'maxls': 100})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def multi_lag_refine(h, n, deadline, top_m=8, max_rounds=50):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corrs = compute_corrs(best_h, n)
        top_lags = np.argsort(corrs)[-top_m:]
        improved = False
        for lag in top_lags:
            if time.time() > deadline:
                break
            k = int(lag) - (n - 1)
            grad = corr_grad_single(best_h, n, k)
            for scale in [0.005, 0.002, 0.001, 0.0005, 0.0001, 0.01, 0.005, 0.02]:
                h_try = project_feasible(best_h - scale * grad, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-16:
                    best_c = c_try
                    best_h = h_try
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=120):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def subgradient_polish(h, n, deadline, max_iters=50000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    step = 0.01
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corrs = compute_corrs(h, n) * dx
        max_idx = int(np.argmax(corrs))
        k = max_idx - (n - 1)
        grad = corr_grad_single(h, n, k)
        improved = False
        for scale in [step, step * 0.5, step * 2, step * 0.25]:
            h_try = project_feasible(h - scale * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 30 or step < 1e-9:
                break
        else:
            stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Intensive L-BFGS with increasing alpha
    alphas = [5000, 50000, 500000, 5000000, 50000000]
    for alpha in alphas:
        if time.time() > deadline - 400:
            break
        alloc = min(100, deadline - time.time() - 300)
        if alloc < 15:
            break
        h, c = smooth_max_lbfgs(best_h, best_n, time.time() + alloc, alpha=alpha, maxiter=400)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 2: Multi-lag refinement
    if time.time() < deadline - 200:
        t2 = min(time.time() + 120, deadline - 150)
        best_h, best_c = multi_lag_refine(best_h, best_n, t2, top_m=10, max_rounds=100)

    # Phase 3: Try other n values briefly
    for n_try in [400, 600, 700]:
        if time.time() > deadline - 150:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 100)
        if alloc < 15:
            continue
        h_t, c_t = smooth_max_lbfgs(h_interp, n_try, time.time() + alloc, alpha=5000000, maxiter=300)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 4: Subgradient polish (very fine)
    if time.time() < deadline - 80:
        t4 = min(time.time() + 60, deadline - 60)
        best_h, best_c = subgradient_polish(best_h, best_n, t4, max_iters=100000)

    # Phase 5: Fine pair search
    if time.time() < deadline - 30:
        t5 = min(time.time() + 40, deadline - 20)
        best_h, best_c = fine_pair_search(best_h, best_n, t5, radius=100)

    # Phase 6: Final L-BFGS with extreme alpha
    if time.time() < deadline - 8:
        t6 = deadline - 4
        best_h, best_c = smooth_max_lbfgs(best_h, best_n, t6, alpha=100000000, maxiter=200)

    # Phase 7: Final multi-lag refinement
    if time.time() < deadline - 2:
        best_h, best_c = multi_lag_refine(best_h, best_n, deadline - 1, top_m=5, max_rounds=50)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
