- We have 2n-1 correlation values
- Only the top few matter for the objective
- A local move (changing h[i] and h[j]) affects all correlations
- For the move to improve the objective, it must not increase any of the top correlations

1. Compute all correlations
2. Identify the "active set" (top K correlations)
3. For each active correlation, compute the gradient w.r.t. h
4. For a candidate move (i, j, δ), check if it decreases all active correlations
5. If yes, accept and update

If δ < 0 (removing from i, adding to j): we need grad_m[i] > grad_m[j] for all active m.

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    m = c.max()
    sh = alpha * (c - m)

        else:
            kk = -k
            grad[kk:] += wm * (1.0 - x[:n - kk])
            grad[:n - kk] += -wm * x[kk:]
    return grad

def pen_obj(x, n, alpha, mu):
    v = smooth_max_obj(x, n, alpha)
    s = x.sum() - n / 2.0
    return v + 0.5 * mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += mu * s
    return g

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, alpha, mu, deadline):
    bounds = [(0.0, 1.0)] * n
    t = time.time()
    tr = deadline - t - 2
    if tr < 5:
        return h
    maxiter = max(50, min(3000, int(tr * 8)))
    try:
        res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxls': 40})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_phase(h, n, alpha, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]
    t = time.time()
    tr = deadline - t - 2
    if tr < 5:
        return h
    maxiter = max(50, min(3000, int(tr * 4)))
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def gradient_local_search(h, n, deadline, K=4):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            c_vec = compute_c5_vec(h, n)
            m_max = c_vec.max()
            top_K = np.argsort(c_vec)[::-1][:K]
            active_grads = np.zeros((K, n))
            for ti, m_idx in enumerate(top_K):
                k = m_idx - (n - 1)
                g = np.zeros(n)
                if k >= 0:
                    g[k:] = (1.0 - h[:n - k])
                    g[:n - k] = -h[k:]
                else:
                    kk = -k
                    g[kk:] = (1.0 - h[:n - kk])
                    g[:n - kk] = -h[kk:]
                active_grads[ti] = g
            risk = active_grads.max(axis=0)
            order = np.argsort(risk)
            n_try = min(30, n // 3)
            for idx in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[idx]
                if h[i] >= 1.0 - step + 1e-10:
                    continue
                found = False
                for j_off in range(1, n_try + 1):
                    j = order[n - j_off]
                    if j == i or h[j] < step - 1e-10:
                        continue
                    safe = True
                    for ti in range(K):
                        if active_grads[ti, i] >= active_grads[ti, j]:
                            safe = False
                            break
                    if not safe:
                        continue
                    h[i] += step
                    h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        any_improved = True
                    else:
                        h[i] -= step
                        h[j] += step
                    found = True
                    break
                if found:
                    break
    return h, best_c

def brief_lsqp_polish(h, n, deadline):
    best_c = compute_c5(h, n)
    for alpha in [10000, 50000, 200000]:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_phase(h, n, alpha, min(time.time() + 80, deadline))
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS with increasing alpha and mu
    configs = [
        (200, 5.0), (1000, 20.0), (5000, 100.0), (20000, 500.0),
        (100000, 2000.0), (500000, 10000.0), (2000000, 50000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 200:
            break
        h = lbfgs_phase(h, n, alpha, mu, deadline)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SLSQP polish
    if time.time() < deadline - 120:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 40:
                break
            h = slsqp_phase(h, n, alpha, deadline)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 3: Gradient-guided local search
    if time.time() < deadline - 30:
        h, c = gradient_local_search(h, n, deadline)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Another L-BFGS round from best, then SLSQP
    if time.time() < deadline - 60:
        h2 = lbfgs_phase(best_h, n, 1000000, 50000, deadline)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        if time.time() < deadline - 20:
            h2 = slsqp_phase(best_h, n, 2000000, deadline)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 5: Final local search
    if time.time() < deadline - 10:
        best_h, best_c = gradient_local_search(best_h, n, deadline)

    # Phase 6: Try a different resolution if time permits
    if time.time() < deadline - 150:
        for n_try in [400, 600]:
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_feasible(h_try, n_try)
            for alpha, mu in [(1000, 100), (10000, 500), (100000, 5000)]:
                if time.time() > deadline - 30:
                    break
                h_try = lbfgs_phase(h_try, n_try, alpha, mu, deadline)
            c_try = compute_c5(h_try, n_try)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
                n = n_try
            else:
                # revert
                # best_h remains unchanged

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
