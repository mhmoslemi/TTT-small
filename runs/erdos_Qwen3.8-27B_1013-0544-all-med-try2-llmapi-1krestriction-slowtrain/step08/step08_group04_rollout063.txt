import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, n_grid=17):
    h = h.copy()
    dx = 2.0 / n
    m_size = 2 * n - 1
    m_idx = np.arange(m_size)
    k_shifts = m_idx - n + 1
    grid = np.unique(np.concatenate([[0.0, 1.0], np.linspace(0.02, 0.98, n_grid)]))
    
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full').copy()
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            jkf = j + k_shifts
            jkb = j - k_shifts
            vf = (jkf >= 0) & (jkf < n)
            vb = (jkb >= 0) & (jkb < n)
            g = np.zeros(m_size)
            if vf.any():
                g[vf] = 1.0 - h[jkf[vf]]
            if vb.any():
                g[vb] -= h[jkb[vb]]
            
            best_v = old
            best_f = float(np.max(c))
            for v in grid:
                if abs(v - old) < 1e-9:
                    continue
                f = float(np.max(c + (v - old) * g))
                if f < best_f - 1e-14:
                    best_f = f
                    best_v = v
            if abs(best_v - old) > 1e-8:
                h[j] = best_v
                c += (best_v - old) * g
                improved = True
        if improved:
            h = project_feasible(h, n)
            new_c = compute_c5(h, n)
            if new_c < best_c:
                best_c = new_c
        else:
            break
    return h, best_c

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_phase(h, n, deadline, max_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in [50, 200, 1000, 5000, 20000]:
        if time.time() > deadline - 3:
            break
        t_sec = min(80, deadline - time.time() - 3)
        if t_sec < 5:
            break
        try:
            res = minimize(smax_obj, h, jac=smax_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': max_iters * 8})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, radius=25):
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for s in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] >= s and h[j] <= 1.0 - s:
                        h[i] -= s; h[j] += s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[i] += s; h[j] -= s
                    if h[j] >= s and h[i] <= 1.0 - s:
                        h[j] -= s; h[i] += s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[j] += s; h[i] -= s
    return h, best_c

def make_strat_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300
    
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Coordinate descent (primary method)
    t1 = min(t0 + 350, deadline - 250)
    h1, c1 = coord_descent(h, n, t1, n_grid=17)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: L-BFGS-B for larger moves
    if time.time() < deadline - 150:
        h2, c2 = lbfgs_phase(best_h, n, min(time.time() + 120, deadline - 120), max_iters=400)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: More coordinate descent
    if time.time() < deadline - 100:
        h3, c3 = coord_descent(best_h, n, min(time.time() + 150, deadline - 60), n_grid=21)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Pair search refinement
    if time.time() < deadline - 40:
        h4, c4 = pair_search(best_h, n, min(time.time() + 80, deadline - 15), radius=20)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Try alternative initializations
    if time.time() < deadline - 60:
        for alt_seed in [seed + 100, seed + 200]:
            if time.time() > deadline - 30:
                break
            h_alt = make_strat_init(n, alt_seed)
            # Blend with best solution
            for mix in [0.3, 0.5]:
                if time.time() > deadline - 25:
                    break
                h_blend = mix * h_alt + (1 - mix) * best_h
                h_blend = project_feasible(h_blend, n)
                alloc = min(40, deadline - time.time() - 15)
                if alloc < 15:
                    break
                h_t, c_t = coord_descent(h_blend, n, time.time() + alloc, n_grid=13)
                if c_t < best_c:
                    best_c = c_t
                    best_h = h_t
    
    # Phase 6: Try different resolutions
    if time.time() < deadline - 80:
        for n_try in [200, 400, 250]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 25)
            if alloc < 20:
                break
            h_t, c_t = coord_descent(h_interp, n_try, time.time() + alloc)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try
    
    # Phase 7: Final refinement
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, n, deadline - 5, n_grid=25)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    if time.time() < deadline - 3:
        h_p, c_p = pair_search(best_h, n, deadline - 1, radius=15)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
