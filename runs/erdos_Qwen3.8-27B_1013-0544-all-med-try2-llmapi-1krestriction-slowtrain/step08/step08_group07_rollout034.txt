def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(v, n):
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(200):
        diff = target - v.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - v
            s = room.sum()
            if s < 1e-14:
                break
            v += (diff / s) * room
        else:
            s = v.sum()
            if s < 1e-14:
                break
            v += (diff / s) * v
    return np.clip(v, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    obj = (m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha) * dx
    w = np.exp(shifted - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wl = w[k]
        if wl < 1e-14:
            continue
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wl * (1.0 - h[i_lo + s:i_hi + s])
        j_lo = max(0, s)
        j_hi = min(n, n + s)
        if j_lo < j_hi:
            grad[j_lo:j_hi] -= wl * h[j_lo - s:j_hi - s]
    grad *= dx
    return obj, grad

def lbfgs_opt(h, n, alpha, mu, maxiter):
    target = n / 2.0
    def f(x):
        o, _ = smooth_obj_grad(x, n, alpha)
        return o + mu * (x.sum() - target) ** 2
    def g(x):
        _, gr = smooth_obj_grad(x, n, alpha)
        return gr + 2.0 * mu * (x.sum() - target)
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
        return res.x
    except Exception:
        return h

def slsqp_opt(h, n, alpha, maxiter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        return (m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha) * dx
    def cons_fun(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps, deadline):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(55):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def try_grid(h_curr, best_c, n_curr, n_new, deadline, rng):
    if n_new == n_curr:
        return h_curr.copy(), best_c, n_curr
    xo = np.linspace(0, 2, n_curr)
    xn = np.linspace(0, 2, n_new)
    hh = project_feasible(np.interp(xn, xo, h_curr), n_new)
    hh, _ = coord_descent(hh, n_new, max_sweeps=100, deadline=deadline)
    hh = project_feasible(hh, n_new)
    ch = compute_c5(hh, n_new)
    best_h, best_c, best_n = h_curr.copy(), best_c, n_curr
    if ch < best_c:
        best_c, best_h, best_n = ch, hh.copy(), n_new
    for alpha in [10000, 100000, 1000000]:
        if time.time() > deadline:
            break
        hh = lbfgs_opt(hh, n_new, alpha, 100.0, 300)
        hh = project_feasible(hh, n_new)
        ch = compute_c5(hh, n_new)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), n_new
    hh, _ = coord_descent(best_h, best_n, max_sweeps=50, deadline=deadline)
    hh = project_feasible(hh, best_n)
    ch = compute_c5(hh, best_n)
    if ch < best_c:
        best_c, best_h, best_n = ch, hh.copy(), best_n
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 25
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Coordinate descent to convergence
    p1 = t0 + 0.18 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=600, deadline=p1)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 2: L-BFGS-B with increasing smoothness and penalty
    p2 = t0 + 0.42 * total
    h = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > p2 - 8:
            break
        for mu in [10.0, 100.0, 500.0]:
            if time.time() > p2 - 4:
                break
            h = lbfgs_opt(h, n, alpha, mu, 600)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 3: Coordinate descent refinement
    p3 = t0 + 0.52 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=300, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 4: SLSQP with high alpha
    p4 = t0 + 0.70 * total
    h = best_h.copy()
    for alpha in [200000, 2000000, 20000000, 200000000]:
        if time.time() > p4 - 4:
            break
        h = slsqp_opt(h, n, alpha, 3000)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Final coord descent
    p5 = t0 + 0.78 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p5)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 6: Try other grid sizes
    p6 = t0 + 0.92 * total
    for nh in [300, 200, 500, 600]:
        if time.time() > p6 - 8:
            break
        h_try, c_try, n_try = try_grid(best_h, best_c, n, nh, p6 - 4, rng)
        if c_try < best_c:
            best_c, best_h, n = c_try, h_try.copy(), n_try

    # Phase 7: Perturbation restarts on best grid
    if time.time() < dl - 8:
        for trial in range(6):
            if time.time() > dl - 3:
                break
            h = best_h + 0.004 * rng.randn(n)
            h = project_feasible(h, n)
            h = lbfgs_opt(h, n, 100000, 500.0, 300)
            h = project_feasible(h, n)
            hc2, _ = coord_descent(h, n, max_sweeps=50, deadline=dl - 2)
            hc2 = project_feasible(hc2, n)
            c = compute_c5(hc2, n)
            if c < best_c:
                best_c, best_h = c, hc2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
