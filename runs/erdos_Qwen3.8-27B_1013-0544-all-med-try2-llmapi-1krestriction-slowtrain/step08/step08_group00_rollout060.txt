```python
def delta_f_for_swap(h, n, i, j, d, active_k_indices):
    """Compute change in f_k for a swap h[i]+=d, h[j]-=d for given lags."""
    deltas = []
    for k_idx in active_k_indices:
        s = k_idx - (n-1)  # actual shift
        # g(s, i) = 1 - h[i+s] - h[i-s] (with bounds checking)
        gi = 1.0
        if 0 <= i+s < n: gi -= h[i+s]
        if 0 <= i-s < n: gi -= h[i-s]
        gj = 1.0
        if 0 <= j+s < n: gj -= h[j+s]
        if 0 <= j-s < n: gj -= h[j-s]
        deltas.append(d * (gi - gj))
    return np.array(deltas)
```

```python
import time
```

```python
    """Compute all f_k values."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_f(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """Gradient of smooth max of f_k."""
    f = compute_f(h, n)
    m = f.max()
    shifted = alpha * (f - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()

        else:

            nv = n - ss

    sh = alpha * (f - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return val

    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

    bounds = [(0.0, 1.0)] * n
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]

    """Local search using efficient delta computation."""
    best_c = compute_c5(h, n)
    f = compute_f(h, n)
    # Active indices: those near the max
    threshold = f.max() - 1e-10
    active = np.where(f >= threshold)[0]

    else:

            f = compute_f(h, n)
            threshold = f.max() - 1e-10
            active = np.where(f >= threshold)[0][:top_k]

                for j in range(n):

                        continue
                    # Check if swap improves
                    # Δf_k = d * (g(k,i) - g(k,j)) for each active k
                    # We need Δf_k <= 0 for all active k (to not increase max)
                    ok = True
                    for k_idx in active:

                        gi = 1.0
                        if 0 <= i+s < n: gi -= h[i+s]
                        if 0 <= i-s < n: gi -= h[i-s]

                        if 0 <= j+s < n: gj -= h[j+s]

                            improved = True
                            break  # restart j loop
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c
```

1. Start with some h, compute f_k, identify active set A = {k : f_k ≈ max}
2. Solve: min Σ_{k∈A} f_k(h) subject to constraints (or min max_{k∈A} f_k)
3. Update active set
4. Repeat

The second term is h^T Q h where Q is a matrix built from the active set. This is a QP!

If Q is negative semidefinite (which it likely is since it involves -h[i]h[i+k] terms), then minimizing h^T Q h is a non-convex QP. But we can still use gradient-based methods.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_f(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_f(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, n, alpha):
    f = compute_f(h, n)
    m = f.max()
    shifted = alpha * (f - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-13:
            continue
        s = k_idx - (n - 1)
        wm = w[k_idx]
        if s >= 0:
            nv = n - s
            grad[:nv] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:nv]
        else:
            ss = -s
            nv = n - ss
            grad[:nv] += wm * (1.0 - h[ss:])
            grad[ss:] -= wm * h[:nv]
    return grad

def _obj(x, n, alpha, mu):
    f = compute_f(x, n)
    m = f.max()
    sh = alpha * (f - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    f = compute_f(x, n)
    m = f.max()
    sh = alpha * (f - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _eq(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_total_iter=5000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (300, 1.0, 400), (2000, 10.0, 600), (10000, 100.0, 800),
        (50000, 500.0, 1000), (200000, 2000.0, 1200),
        (1000000, 10000.0, 1500), (5000000, 50000.0, 2000),
    ]
    for alpha, mu, mxit in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': mxit, 'ftol': 1e-17,
                                  'gtol': 1e-15, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c, h = c, h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=3000000, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c = compute_c5(h_new, n)
        return h_new, c
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_local(h, n, deadline, k_top=40):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for d in steps:
        imp = True
        while imp and time.time() < deadline:
            imp = False
            f = compute_f(h, n)
            g = grad_smooth(h, n, 5000.0)
            up = np.argsort(g)[-k_top:]
            down = np.argsort(g)[:k_top]
            for direction in [0, 1]:
                if imp or time.time() > deadline:
                    break
                if direction == 0:
                    srcs, dsts = down, up
                    check_src = lambda i: h[i] + d <= 1.0
                    check_dst = lambda j: h[j] - d >= 0.0
                    do_swap = lambda i, j: (h.__setitem__(i, h[i]+d), h.__setitem__(j, h[j]-d))
                    undo_swap = lambda i, j: (h.__setitem__(i, h[i]-d), h.__setitem__(j, h[j]+d))
                else:
                    srcs, dsts = up, down
                    check_src = lambda i: h[i] - d >= 0.0
                    check_dst = lambda j: h[j] + d <= 1.0
                    do_swap = lambda i, j: (h.__setitem__(i, h[i]-d), h.__setitem__(j, h[j]+d))
                    undo_swap = lambda i, j: (h.__setitem__(i, h[i]+d), h.__setitem__(j, h[j]-d))
                for i in srcs:
                    if time.time() > deadline:
                        return h, best_c
                    for j in dsts:
                        if i == j:
                            continue
                        if not check_src(i) or not check_dst(j):
                            continue
                        do_swap(i, j)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            imp = True
                        else:
                            undo_swap(i, j)
                        if imp:
                            break
    return h, best_c
```
