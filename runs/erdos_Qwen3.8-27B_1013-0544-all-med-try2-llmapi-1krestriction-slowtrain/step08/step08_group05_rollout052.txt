import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=200, deadline=None, seed=42):
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()
    gr = (np.sqrt(5) - 1) / 2
    
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for p in order:
            old_val = h[p]
            a = np.zeros(2*n-1)
            # a[k] = (1-h[p+k-(n-1)]) - h[p-k+(n-1)] for valid indices
            s1 = n - 1 - p  # p+k-(n-1) >= 0 => k >= n-1-p
            e1 = 2*(n-1) - p  # p+k-(n-1) <= n-1 => k <= 2(n-1)-p
            lo1 = max(0, s1)
            hi1 = min(2*n-2, e1)
            if lo1 <= hi1:
                idx = np.arange(lo1, hi1+1)
                a[idx] += 1.0 - h[idx - (n-1) + p]
            s2 = p  # p-k+(n-1) >= 0 => k <= p+n-1, and k >= 0
            e2 = p + n - 1
            lo2 = max(0, s2)
            hi2 = min(2*n-2, e2)
            if lo2 <= hi2:
                idx = np.arange(lo2, hi2+1)
                a[idx] -= h[p + n - 1 - idx]
            
            b = c - old_val * a
            # Golden section: min max(a*x+b) on [0,1]
            c1 = 1.0 - gr
            c2 = gr
            f1 = np.max(a*c1 + b)
            f2 = np.max(a*c2 + b)
            lo, hi = 0.0, 1.0
            for _ in range(30):
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr*(hi-lo)
                    f1 = np.max(a*c1 + b)
                else:
                    lo, c1, f1 = c2, f2, f1
                    c2 = lo + gr*(hi-lo)
                    f2 = np.max(a*c2 + b)
            x_opt = (lo+hi) * 0.5
            if x_opt <= old_val + 1e-12 and x_opt >= old_val - 1e-12:
                continue
            f_opt = np.max(a*x_opt + b)
            if f_opt < cur_max - 1e-10:
                h[p] = x_opt
                c += (x_opt - old_val) * a
                cur_max = c.max()
                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh-sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n/2.0

    except:
