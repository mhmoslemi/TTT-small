import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    # Find mu such that sum(clip(h - mu, 0, 1)) = target
    # Use bisection
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mu = (lo + hi) / 2
        s = np.clip(h - mu, 0, 1).sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) / 2
    return np.clip(h - mu, 0, 1)

def gradient_at_lag(h, n, L):
    """Compute gradient of c[n-1+L] w.r.t. h"""
    g = np.zeros(n)
    if L >= 0:
        # (1 - h[j+L]) for j in [0, n-1-L]
        g[:n-L] += (1 - h[L:])
        # -h[j-L] for j in [L, n-1]
        g[L:] -= h[:n-L]
    else:
        M = -L
        # (1 - h[j-M]) for j in [M, n-1]
        g[M:] += (1 - h[:n-M])
        # -h[j+M] for j in [0, n-1-M]
        g[:n-M] -= h[M:]
    return g

def projected_subgradient(h, n, steps, step_scale, top_k, deadline):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for t in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        cur_c = c.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Find top-k lags
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Average gradient
        g = np.zeros(n)
        for idx in top_idx:
            L = idx - (n - 1)
            g += gradient_at_lag(h, n, L)
        g /= top_k
        
        # Adaptive step size
        step = step_scale / (1 + 0.01 * t)
        
        h_new = h - step * g
        h = project_feasible(h_new, n)
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):

            lo = mu
        else:
            hi = mu

def lag_gradient(h, n, L):
    g = np.zeros(n)
    if L >= 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n-M])
        g[:n-M] -= h[M:]
    return g

            best_h = h.copy()
        t += 1

def direct_delta(a, b, L):
    # a goes 0→1, b goes 1→0
    # Change in sum_i h[i]*(1-h[i+L])
    delta = 0
    for i in [a, b, a-L, b-L]:
        if 0 <= i < n and 0 <= i+L < n:
            old = h[i]*(1-h[i+L])
            new_h_i = 1 if i == a else (0 if i == b else h[i])
            new_h_ipL = 1 if i+L == a else (0 if i+L == b else h[i+L])
            new = new_h_i * (1 - new_h_ipL)
            delta += new - old
    return delta

def fast_local_search(h_init, n, deadline, max_rounds=100000, n_restarts=3):
    target = n // 2
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    
    best_c = np.inf
    best_h = h.copy()
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            # Perturb
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(3 + restart * 2):
                if len(ones) > 0 and len(zeros) > 0:
                    a = np.random.choice(zeros)
                    b = np.random.choice(ones)
                    h[a], h[b] = 1.0, 0.0
        
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_max = corr.max()
        
        for rnd in range(max_rounds):
            if time.time() > deadline:
                break
            
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
            
            # Find top few lags
            top_k = 5
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
            
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            best_swap = None
            best_new_max = cur_max
            
            # Score zeros (candidates to turn on)
            z_scores = np.zeros(len(zeros))
            o_scores = np.zeros(len(ones))
            for ki in top_idx:
                L = ki - (n - 1)
                for zi, a in enumerate(zeros):
                    s = 0.0
                    if 0 <= a + L < n:
                        s += (1.0 - h[a + L])
                    if 0 <= a - L < n:
                        s -= h[a - L]
                    z_scores[zi] += s
                for oi, b in enumerate(ones):
                    s = 0.0
                    if 0 <= b + L < n:
                        s -= (1.0 - h[b + L])
                    if 0 <= b - L < n:
                        s += h[b - L]
                    o_scores[oi] += s

            # We want the most negative total (biggest decrease)

            n_cand_o = min(10, len(ones))
            z_cand = np.argpartition(z_scores, n_cand - 1)[:n_cand] if len(zeros) >= n_cand else np.arange(len(zeros))
            o_cand = np.argpartition(o_scores, n_cand_o - 1)[:n_cand_o]
