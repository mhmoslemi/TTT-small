```python
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    """Gradient of O_k with respect to h"""
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_vec(x, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    n_pts = len(x)

    top_idx = np.argsort(w)[-top_k:]

    for i in top_idx:

def active_set_descent(h, n, deadline, alpha=500.0):
    """PGD with smooth max, focusing on active set"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    prev_h = None
    prev_g = None
    stall = 0
    
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        w = smooth_max_weights(c, alpha)
        dx = 2.0 / n
        top_k = min(50, len(w))
        top_idx = np.argsort(w)[-top_k:]
        g = np.zeros(n)
        for i in top_idx:
            if w[i] < 1e-14:
                continue
            k = i - (n - 1)
            g += w[i] * dx * grad_single(h, n, k)
        
        # Barzilai-Borwein step size
        if prev_h is not None:
            s = h - prev_h
            y = g - prev_g
            ss = s @ s
            sy = s @ y
            yy = y @ y
            if yy > 1e-20:
                if sy > 1e-14:
                    step = ss / sy
                else:
                    step = ss / yy
                step = float(np.clip(step, 0.001, 0.5))
        prev_h = h.copy()
        prev_g = g.copy()
        
        # Line search
        accepted = False
        for _ in range(15):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-12:
                h = h_new
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            break
        if stall > 80:
            break
    
    return best_h, best_c
```

```python
def compute_swap_delta(h, i, j, d, n):
    """Compute change in all O_k when h[i]+=d, h[j]-=d. Returns delta_O_k for k=-(n-1)..(n-1)"""
    n_corr = 2*n - 1
    delta = np.zeros(n_corr)
    dx = 2.0/n  # we'll multiply by dx at the end
    
    # Contribution from h[i] changing by d
    # For k >= 0 (index n-1+k in full correlation):
    #   O_k = sum_{m=0}^{n-1-k} h[m](1-h[m+k])
    #   Change: d*(1-h[i+k]) if i <= n-1-k, plus -d*h[i-k] if 0 <= i-k <= n-1-k
    
    k_range = np.arange(1, n)
    
    # h[i] as first factor: valid when i <= n-1-k, i.e., k <= n-1-i
    valid1 = k_range <= (n - 1 - i)
    if np.any(valid1):
        k_valid = k_range[valid1]
        delta[n-1+k_valid] += d * (1 - h[i + k_valid])
    # h[i] as second factor: valid when i-k >= 0 and i-k <= n-1-k, i.e., k <= i and k <= n-1-i
    valid2 = (k_range <= i) & (k_range <= (n - 1 - i))
    if np.any(valid2):
        k_valid = k_range[valid2]
        delta[n-1+k_valid] += (-d) * h[i - k_valid]
    
    # For k < 0, i.e., O_{-k} = sum_{m=k}^{n-1} h[m](1-h[m-k])
    # h[i] as first factor: valid when k <= i <= n-1, i.e., 1 <= k <= i
    valid3 = k_range <= i
    if np.any(valid3):
        k_valid = k_range[valid3]
        delta[n-1-k_valid] += d * (1 - h[i - k_valid])
    # h[i] as second factor: valid when i+k <= n-1, i.e., k <= n-1-i
    valid4 = k_range <= (n - 1 - i)
    if np.any(valid4):
        k_valid = k_range[valid4]
        delta[n-1-k_valid] += (-d) * h[i + k_valid]
    
    # Repeat for h[j] with -d
    # h[j] as first factor for k >= 0:
    valid1j = k_range <= (n - 1 - j)
    if np.any(valid1j):
        k_valid = k_range[valid1j]
        delta[n-1+k_valid] += (-d) * (1 - h[j + k_valid])
    # h[j] as second factor for k >= 0:
    valid2j = (k_range <= j) & (k_range <= (n - 1 - j))
    if np.any(valid2j):
        k_valid = k_range[valid2j]
        delta[n-1+k_valid] += d * h[j - k_valid]
    # h[j] as first factor for k < 0:
    valid3j = k_range <= j
    if np.any(valid3j):
        k_valid = k_range[valid3j]
        delta[n-1-k_valid] += (-d) * (1 - h[j - k_valid])
    # h[j] as second factor for k < 0:
    valid4j = k_range <= (n - 1 - j)
    if np.any(valid4j):
        k_valid = k_range[valid4j]
        delta[n-1-k_valid] += d * h[j + k_valid]
    
    return delta
```

```python
                    if h[j] > 1 - d:
                        continue
                    
                    # Fast check: delta at top lags
                    reject = False
                    for k in top_lags_k:
                        dk = delta_at_lag(h, i, j, -d, d, k, n) * dx
                        if corrs[k + n - 1] + dk >= best_c:
                            reject = True
                            break
                    if reject:
                        continue
                    
                    # Full check
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        corrs = compute_c5_vec(h, n)
                        top_idx = np.argsort(corrs)[-top_lags:]
                        top_lags_k = top_idx - (n - 1)
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return float(smooth_max(c, alpha) + 100.0 * (x.sum() - n / 2.0) ** 2)

def lbfgs_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    dx = 2.0 / n
    top_idx = np.argsort(w)[-80:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    grad += 200.0 * (x.sum() - n / 2.0)
    return grad

def pgd_optimize(h, n, deadline, top_k=50, alpha=500.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    prev_h, prev_g = None, None
    stall = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        w = smooth_max_w(c, alpha)
        dx = 2.0 / n
        top_idx = np.argsort(w)[-top_k:]
        g = np.zeros(n)
        for i in top_idx:
            if w[i] < 1e-14:
                continue
            g += w[i] * dx * grad_single(h, n, i - (n - 1))
        if prev_h is not None:
            s, y = h - prev_h, g - prev_g
            ss, sy, yy = s @ s, s @ y, y @ y
            if yy > 1e-20:
                step = ss / sy if sy > 1e-14 else ss / yy
                step = float(np.clip(step, 0.001, 0.5))
        prev_h, prev_g = h.copy(), g.copy()
        accepted = False
        for _ in range(15):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-12:
                h = h_new
                stall = 0 if c_new < best_c - 1e-14 else stall + 1
                if c_new < best_c - 1e-14:
                    best_c, best_h = c_new, h_new.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted or stall > 80:
            break
    return best_h, best_c

def swap_delta(h, i, j, d, n):
    """Exact change in all O_k when h[i]+=d, h[j]-=d. Returns delta array (len 2n-1)."""
    nc = 2 * n - 1
    delta = np.zeros(nc)
    k_arr = np.arange(1, n)
    # For h[i] with +d:
    v1 = k_arr <= (n - 1 - i)
    if v1.any():
        kv = k_arr[v1]
        delta[n - 1 + kv] += d * (1 - h[i + kv])
    v2 = (k_arr <= i) & (k_arr <= (n - 1 - i))
    if v2.any():
        kv = k_arr[v2]
        delta[n - 1 + kv] += -d * h[i - kv]
    v3 = k_arr <= i
    if v3.any():
        kv = k_arr[v3]
        delta[n - 1 - kv] += d * (1 - h[i - kv])
    v4 = k_arr <= (n - 1 - i)
    if v4.any():
        kv = k_arr[v4]
        delta[n - 1 - kv] += -d * h[i + kv]
    # For h[j] with -d:
    v1 = k_arr <= (n - 1 - j)
    if v1.any():
        kv = k_arr[v1]
        delta[n - 1 + kv] += -d * (1 - h[j + kv])
    v2 = (k_arr <= j) & (k_arr <= (n - 1 - j))
    if v2.any():
        kv = k_arr[v2]
        delta[n - 1 + kv] += d * h[j - kv]
    v3 = k_arr <= j
    if v3.any():
        kv = k_arr[v3]
        delta[n - 1 - kv] += -d * (1 - h[j - kv])
    v4 = k_arr <= (n - 1 - j)
    if v4.any():
        kv = k_arr[v4]
        delta[n - 1 - kv] += d * h[j + kv]
    return delta

def fast_pair_search(h, n, deadline, num_top=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    corrs = compute_c5_vec(h, n)
    dx = 2.0 / n
    deltas = [0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            top_idx = np.argsort(corrs)[-num_top:]
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if h[i] < d:
                    continue
                for j in range(n):
                    if j == i or h[j] > 1.0 - d:
                        continue
                    # Fast rejection: check top lags only
                    delta = swap_delta(h, i, j, d, n)
                    if np.any(corrs[top_idx] + delta[top_idx] >= best_c - 1e-15):
                        continue
                    # Accept and update
                    h[i] += d
                    h[j] -= d
                    corrs += delta
                    new_max = float(np.max(corrs))
                    best_c = new_max
                    best_h = h.copy()
                    improved = True
            # Also try reverse: h[i]-=d, h[j]+=d
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if h[i] < d:
                    continue
                for j in range(n):
                    if j == i or h[j] > 1.0 - d:
                        continue
                    delta = swap_delta(h, i, j, -d, n)
                    if np.any(corrs[top_idx] + delta[top_idx] >= best_c - 1e-15):
                        continue
                    h[i] -= d
                    h[j] += d
                    corrs += delta
                    new_max = float(np.max(corrs))
                    best_c = new_max
                    best_h = h.copy()
                    improved = True
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:

            pass

        best_h, best_c = fast_pair_search(best_h, best_n, t3, num_top=12)
    # Phase 4: Additional L-BFGS + PGD cycle
    if time.time() < deadline - 30:

        try:

            h_new = project_feasible(res.x, best_n)

        except Exception:
            pass
    # Final tiny pair polish
    if time.time() < deadline - 1:
        best_h, best_c = fast_pair_search(best_h, best_n, deadline - 0.5, num_top=10)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
