import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def augmented_lagrangian_step(h, n, alpha, lambda_val, mu, deadline):
    """One augmented Lagrangian step using L-BFGS-B on the AL objective."""
    target = n / 2.0
    
    def al_obj(x):
        val = smooth_max_obj(x, n, alpha)
        val += lambda_val * (x.sum() - target) + 0.5 * mu * (x.sum() - target)**2
        return float(val)
    
    def al_grad(x):
        g = smooth_max_grad(x, n, alpha)
        g += lambda_val * np.ones(n) + mu * (x.sum() - target) * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    maxiter = max(10, int((deadline - time.time()) * 5))
    try:
        res = minimize(al_obj, h, jac=al_grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': min(maxiter, 500), 'ftol': 1e-16, 'gtol': 1e-14})
        return res.x
    except Exception:
        return h

def coordinate_descent(h, n, deadline):
    """Coordinate descent on individual or pairs of coordinates."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    target = n / 2.0
    
    # For each coordinate i, find optimal h[i] given others fixed
    # f_k depends on h[i] through terms involving h[i] and h[i±k]
    # For coordinate i: it appears in f_k for k where i and i+k are both valid
    #   and for k where i-k and i are both valid
    
    # This is complex to do exactly, so use a simpler approach:
    # Try small perturbations of pairs
    
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        while time.time() < deadline:
            improved = False
            # Random pairs for diversity
            rng = np.random.RandomState(int(time.time()*1000) % 10000)
            n_trials = 500
            for _ in range(n_trials):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = step * (1 if rng.rand() < 0.5 else -1)
                if h[i]+d >= 0 and h[i]+d <= 1 and h[j]-d >= 0 and h[j]-d <= 1:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d; h[j] += d
            if not improved:
                break
    
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    """Returns (objective, gradient) for smooth max of c = correlate(h,1-h)*dx."""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        w = np.exp(sh - mx)
        w /= w.sum()
        val = m + (mx + np.log(w.sum())) / alpha
    else:
        e = np.exp(sh)
        w = e / e.sum()
        val = m + np.log(e.sum()) / alpha
    # Gradient: sum_j w_j * dc_j/dh
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return float(val), grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    e = np.exp(sh - mx)
    w = e / e.sum()
    val = m + (mx + np.log(e.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return float(val), grad

def obj_pen(x, n, alpha, mu):
    v, _ = smooth_max_and_grad(x, n, alpha)
    return v + 0.5 * mu * (x.sum() - n / 2.0) ** 2

def grad_pen(x, n, alpha, mu):
    _, g = smooth_max_and_grad(x, n, alpha)
    cnt = x.sum() - n / 2.0
    g += mu * cnt * np.ones(n)
    return g

def obj_slsqp(x, n, alpha):
    return smooth_max_and_grad(x, n, alpha)[0]

def con_slsqp(x, n):
    return x.sum() - n / 2.0

def jac_slsqp(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, alpha, mu=0.01, maxiter=250):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad_pen, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def run_slsqp(h, n, deadline, alpha=50000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_slsqp, 'jac': jac_slsqp, 'args': (n,)}]
    try:
        res = minimize(obj_slsqp, h, method='SLSQP', bounds=bounds,
                     constraints=cons, args=(n, alpha),
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def pipeline(h, n, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for a in [50, 200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > deadline - 5:
            break
        tn = min(deadline - 3, time.time() + 45)
        h_new, c_new = run_lbfgs(best_h, n, tn, alpha=a, mu=0.01, maxiter=200)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 8:
        for a in [50000, 200000]:
            if time.time() > deadline - 4:
                break
            tn = min(deadline - 2, time.time() + 50)
            h_new, c_new = run_slsqp(best_h, n, tn, alpha=a, maxiter=1200)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 2:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    return best_h, best_c

def local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for s in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + s <= 1.0 and h[j] - s >= 0.0:
                        h[i] += s; h[j] -= s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= s; h[j] += s
                    if h[i] - s >= 0.0 and h[j] + s <= 1.0:
                        h[i] -= s; h[j] += s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += s; h[j] -= s
    return h, best_c

def grad_guided_search(h, n, deadline, step=0.003):
    best_c = compute_c5(h, n)
    alphas = [500, 2000, 10000]
    for a in alphas:
        while time.time() < deadline:
            improved = False
            _, g = smooth_max_and_grad(h, n, a)
            order = np.argsort(g)
            n_try = min(40, n // 3)
            for idx in range(n_try):
                if time.time() > deadline:
                    return h, best_c
                i = order[idx]
                for idx2 in range(n - n_try, n):
                    j = order[idx2]
                    d = step * (1 if g[i] < g[j] else -1)
                    if d == 0:
                        continue
                    if h[i] + d >= 0 and h[i] + d <= 1 and h[j] - d >= 0 and h[j] - d <= 1:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:

    n_list = [200, 250, 350, 400, 500]
    for n_try in n_list:

        x_old = np.linspace(0, 2, len(best_h))

        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(70, deadline - time.time() - 40)
        if alloc < 30:

        h_t, c_t = pipeline(h_interp, n_try, time.time() + alloc)

    for amt in [0.08, 0.04, 0.02]:
