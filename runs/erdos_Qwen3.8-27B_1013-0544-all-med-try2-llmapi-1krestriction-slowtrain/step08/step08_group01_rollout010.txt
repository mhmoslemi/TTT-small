import time, math
import numpy as np
from scipy.optimize import minimize

def cn(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5v(h, n):
    return float(np.max(cn(h, n)))

def project(h, n):
    h = np.asarray(h, dtype=float).ravel().copy()
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
        h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + d)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + d)
    return h

def wgrad(h, n, w):
    P = np.asarray(w, dtype=float).ravel()
    need = 2 * n - 1
    if P.size < need:
        P = np.concatenate((P, np.zeros(need - P.size)))
    else:
        P = P[:need]
    cp = np.zeros(P.size + 1)
    cp[1:] = P.cumsum()
    j = np.arange(n)
    A = cp[2 * n - 1 - j] - cp[n - 1 - j]
    B = np.convolve(P, h, mode='full')[n - 1:n - 1 + n]
    return (2.0 / n) * (A * (1.0 - h) - B)

def sobj(x, n, a):
    c = cn(x, n)
    m = float(np.max(c))
    if a <= 0.0:
        return m
    t = a * (c - m)
    t = t - t.max()
    return float(m + math.log(float(np.sum(np.exp(t)))) / a)

def sgrad(x, n, a):
    c = cn(x, n)
    m = float(np.max(c))
    if a <= 0.0:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return wgrad(x, n, w)
    t = a * (c - m)
    w = np.exp(t)
    s = w.sum()
    if s <= 0.0:
        w = np.full_like(c, 1.0 / c.size)
    else:
        w /= s
    return wgrad(x, n, w)

def peobj(x, n, a, mu):
    s = x.sum() - 0.5 * n
    return sobj(x, n, a) + 0.5 * mu * s * s

def pegrad(x, n, a, mu):
    s = x.sum() - 0.5 * n
    return sgrad(x, n, a) + mu * s

def eqf(x, n):
    return float(x.sum() - 0.5 * n)

def eqj(x, n):
    return np.ones((1, x.size), dtype=float)

def rslq(h, n, a, mi):
    h = project(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj, 'args': (n,)}]
    try:
        r = minimize(sobj, h, jac=sgrad, args=(n, a), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': int(mi), 'ftol': 1e-16,
                              'maxfun': int(4 * max(20, mi))})
        h2 = project(r.x, n)
        c2 = c5v(h2, n)
        if math.isfinite(c2):
            return h2, c2
    except Exception:
        pass
    return h.copy(), c5v(h, n)

def rlbf(h, n, a, mu, mi):
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(peobj, h, jac=pegd, args=(n, a, mu), method='L-BFGS-B',
                     bounds=bounds,

        c2 = c5v(h2, n)
        if math.isfinite(c2):
            return h2, c2
    except Exception:
        pass
    return h.copy(), c5v(h, n)

def dsq(start, n, dl, alphas, mi=220):

    best = cur.copy()
    bc = c5v(cur, n)
    for a in alphas:

def anti_h(a, m):
    h = np.empty(2 * m, dtype=float)
    h[:m] = a
    h[m:] = 1.0 - a
    return h

def anti_obj(a, m, al):

def anti_grad(a, m, al):
    h = anti_h(a, m)
    g = sgrad(h, 2 * m, al)
    return g[:m] - g[m:]

def anti_polish(a, m, dl):

def pstep(h, n, i, j, bound, rad):
    oi, oj = float(h[i]), float(h[j])
    lo = max(-rad, -oi, oj - 1.0)
    hi = min(rad, 1.0 - oi, oj)
    if hi - lo < 1e-9:
        return bound
    bv, bd = bound, 0.0
    for d in np.linspace(lo, hi, 9):
        h[i] = oi + d
        h[j] = oj - d
        v = c5v(h, n)
        if v < bv:
            bv, bd = v, d
    if bd != 0.0:
        e = max((hi - lo) / 80.0, 1e-6)
        for d in np.linspace(max(lo, bd - e), min(hi, bd + e), 5):
            h[i] = oi + d
            h[j] = oj - d
            v = c5v(h, n)
            if v < bv:
                bv, bd = v, d
    if bv < bound - 1e-15:
        h[i] = oi + bd
        h[j] = oj - bd
        return bv
    h[i], h[j] = oi, oj
    return bound

def ppol(h, n, dl):
    h = project(h, n)
    bc = c5v(h, n)
    for rad in (0.005, 0.002, 0.0008):
        for _ in range(4):

            low = ordg[:10]

            cand = []
            for i in low:

                    if i != j:
                        cand.append((float(g[j] - g[i]), int(i), int(j)))
            if not cand:

                if nv < bc:
                    bc = nv
                    imp += 1
                    if imp >= 6:

        if imp == 0:
            break
    return h, bc
