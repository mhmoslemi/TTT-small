            if tot < 1e-14:

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _sobj(x, n, alpha):
    return float(smooth_max_val(x, n, alpha))

def _seq(x, n):
    return float(x.sum() - n / 2.0)

def _seqj(x, n):
    return np.ones(n)

def run_lbfgs(h, n, alphas, deadline, max_iters=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = max(alpha * 0.005, 1.0)
        try:
            res = minimize(_obj, best_h, jac=_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-16,
                                  'gtol': 1e-14, 'maxfun': max_iters * 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_slsqp(h, n, alpha, deadline, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _seq, 'jac': _seqj, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run_slsqp_warm(h, n, deadline, maxiter=3000):
    """SLSQP with increasing alpha (warm starts)."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [1000, 10000, 100000, 1000000, 10000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new, c_new = run_slsqp(best_h, n, alpha, deadline - 3, maxiter)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def grad_informed_search(h, n, deadline, k=40, alphas=None):
    """Local search using gradient to identify promising pairs."""
    if alphas is None:
        alphas = [1000, 10000, 100000, 1000000]
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for alpha in alphas:
        while time.time() < deadline - 3:
            g = smooth_max_grad(h, n, alpha)
            up = np.argsort(g)[-k:]
            down = np.argsort(g)[:k]
            improved = False
            for i in down:
                if time.time() > deadline - 3:
                    return h, best_c
                for j in up:
                    if i == j:
                        continue
                    for step in steps:
                        if time.time() > deadline - 3:
                            return h, best_c
                        if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= step
                                h[j] += step
                            if improved:
                                break
                        if h[j] + step <= 1.0 and h[i] - step >= 0.0:
                            h[j] += step
                            h[i] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[j] -= step
                                h[i] += step
                            if improved:
                                break
            if not improved:
                break
    return h, best_c

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.arange(n)
    for freq in range(1, n // 3 + 1):
        amp = rng.uniform(0, 0.2) / np.sqrt(freq)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / n + phase)
    return project_feasible(h, n)

    best_h, best_c = run_lbfgs(best_h, best_n, alphas_1, t1, max_iters=400)

    if time.time() < deadline - 350:
        t2 = min(time.time() + 80, deadline - 350)
        h, c = run_slsqp_warm(best_h, best_n, t2, maxiter=3000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Gradient-informed local search
    if time.time() < deadline - 250:
        t3 = min(time.time() + 60, deadline - 250)
        h, c = grad_informed_search(best_h, best_n, t3, k=40)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 4: Diverse restarts
    starts = [fourier_start(best_n, rng)]
    for scale in [0.01, 0.03, 0.06]:
        s = project_feasible(best_h + rng.randn(best_n) * scale, best_n)
        starts.append(s)

    for s in starts:
        if time.time() > deadline - 100:
            break
        alloc = min(30, (deadline - time.time() - 60) * 0.4)
        if alloc < 12:
            break
        h_s, c_s = run_lbfgs(s, best_n, [200, 2000, 20000, 200000], time.time() + alloc, max_iters=300)
        if time.time() < deadline - 70:
            h_s, c_s = run_slsqp(h_s, best_n, 5000000, min(time.time() + 20, deadline - 70), maxiter=2000)
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 5: Another SLSQP warm start on best
    if time.time() < deadline - 40:
        t5 = min(time.time() + 50, deadline - 40)
        h, c = run_slsqp_warm(best_h, best_n, t5, maxiter=4000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 6: Gradient-informed search again
    if time.time() < deadline - 15:
        h, c = grad_informed_search(best_h, best_n, deadline - 10, k=50,
                                     alphas=[50000, 500000, 5000000])
        if c < best_c:
            best_c, best_h = c, h

    # Phase 7: Final SLSQP squeeze
    if time.time() < deadline - 2:
        h, c = run_slsqp(best_h, best_n, 50000000, deadline - 1, maxiter=5000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 8: Tiny final gradient search
    if time.time() < deadline - 1:
        h, c = grad_informed_search(best_h, best_n, deadline - 0.5, k=20,
                                     alphas=[10000000])
        if c < best_c:
            best_c, best_h = c, h

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
