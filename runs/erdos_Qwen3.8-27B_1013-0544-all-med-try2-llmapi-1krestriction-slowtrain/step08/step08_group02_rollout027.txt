import numpy as np
from scipy.optimize import minimize, fftconvolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            room = 1.0 - h
            room[~mask] = 0.0
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    return float(m + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha)

def smooth_max_grad_fast(x, n, alpha):
    """Compute gradient of smooth max using FFT convolutions - O(n log n)."""
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    w = np.exp(shifted - smax)
    w /= w.sum()
    
    # w is length 2n-1, w[m] corresponds to lag k = m - (n-1)
    # grad[j] = sum_k w[k] * ((1-x[j+k])*1_{0<=j+k<n} - x[j-k]*1_{0<=j-k<n})
    # = A[j] - B[j]
    # where A[j] = sum_m w[m] * (1-x)[j + m - (n-1)]
    #       B[j] = sum_m w[m] * x[j - m + (n-1)]
    
    # B[j] = sum_m w[m] * x[(j+n-1) - m] = (w * x)[j + n - 1]
    conv_b = fftconvolve(w, x)  # length (2n-1) + n - 1 = 3n-2
    B = conv_b[(n-1):(n-1)+n]
    
    # A[j] = sum_m w[m] * (1-x)[(j-(n-1)) + m]
    # = cross_correlate(w, 1-x)[j-(n-1)]
    # cross_correlate(a,b)[p] = sum_m a[m]*b[m+p]
    # = convolve(flip(w), 1-x)[p + len(w)-1]
    # = convolve(flip(w), 1-x)[j-(n-1) + 2n-2]
    # = convolve(flip(w), 1-x)[j + n - 2]
    w_flip = w[::-1]
    conv_a = fftconvolve(w_flip, 1 - x)  # length 3n-2
    A = conv_a[(n-2):(n-2)+n]
    
    return A - B

def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad_fast(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, max_iters=200):
    """L-BFGS-B with penalty for sum constraint, multiple alpha levels."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    configs = [(100, 1.0), (500, 5.0), (2000, 20.0),
               (10000, 100.0), (50000, 500.0),
               (200000, 2000.0), (1000000, 10000.0)]
    
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c

def binary_flip_delta(h, i, j, n):
    """Compute change in correlation when h[i]: 1->0 and h[j]: 0->1."""
    # For a binary vector, c[k] = sum_m h[m]*(1-h[m+k])
    # Flipping h[i] from 1 to 0:
    #   - term h[i]*(1-h[i+k]) removed: change = -(1-h[i+k]) = -h[i+k] (since binary)
    #   - term where h[i-k]=1 contributes h[i-k]*(1-h[i]) = h[i-k]*1, now h[i]=0 so becomes h[i-k]*1... 
    #     wait: (1-h[i]) was 0, now becomes 1, so change = +h[i-k]
    # Flipping h[j] from 0 to 1:
    #   - term h[j]*(1-h[j+k]) added: change = +(1-h[j+k])
    #   - term where h[j-k]=1: h[j-k]*(1-h[j]) was h[j-k]*1, now h[j]=1 so (1-h[j])=0, change = -h[j-k]
    
    dc = np.zeros(2*n-1)
    n1 = n - 1
    for k in range(-n+1, n):
        m = k + n1
        # h[i]: 1->0

            dc[m] -= (1 - h[i+k])
        if 0 <= i-k < n:

        if 0 <= j+k < n:
            dc[m] += (1 - h[j+k])
        if 0 <= j-k < n:
            dc[m] -= h[j-k]
    return dc

def compute_c5_binary_fast(h, n):
    """Compute C5 for binary h using FFT autocorrelation."""
    N = 2 * n
    fft_h = np.fft.fft(h, N)
    acf = np.fft.ifft(fft_h * np.conj(fft_h)).real[:n]  # autocorrelation at lags 0..n-1

import numpy as np
from scipy.optimize import minimize, fftconvolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad_fast(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    n1 = n - 1
    w_flip = w[::-1]
    conv_b = fftconvolve(w, x)
    B = conv_b[n1:n1 + n]
    conv_a = fftconvolve(w_flip, 1.0 - x)
    A = conv_a[n1 - 1:n1 - 1 + n]
    return A - B

def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad_fast(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def eq_con(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def slsqp_obj(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def lbfgs_optimize(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (2000, 20.0),
               (10000, 100.0), (50000, 500.0),
               (200000, 2000.0), (1000000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def to_binary(h, rng):
    h_bin = (h > 0.5).astype(float)
    n = len(h_bin)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

        j = zeros[rng.randint(len(zeros))]

        h_bin[j] = 1

            pass
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        iters += 1
    return best_h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return h

def quad_res_start(n, rng):
    h = np.zeros(n)
    for x in range(1, n):
        h[(x * x) % n] = 1.0
    return to_binary(h, rng)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)

        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
