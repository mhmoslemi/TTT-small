    max_c = c.max()
    best_h = h.copy()
    best_c = max_c
    
    # Precompute tight shifts
    # Tight shifts are those where c[k] is close to max
    tight_threshold = max_c - 1e-10  # essentially the argmax
    
    while time.time() < deadline:
        # Find tight shifts (within 2/dx of max, i.e., within 2 units of raw correlation)
        dx = 2.0 / n
        # In raw correlation units (before multiplying by dx):
        # c_raw[k] = c[k] / dx
        # slack_raw[k] = (max_c - c[k]) / dx = (max_c/dx - c_raw[k])

        # Δc_raw[k] ∈ {-2, -1, 0, 1, 2} (for binary h)

        # i.e., max_c - c[k] ≤ 2*dx

            a = ones_idx[rng.randint(len(ones_idx))]
            b = zeros_idx[rng.randint(len(zeros_idx))]

            feasible = True

                if new_max < best_c:

            a = ones_idx[rng.randint(len(ones_idx))]

                best_c = c.max()
                best_h = h.copy()
    
    return best_h, best_c
