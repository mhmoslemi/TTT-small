import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alpha_list=None, max_iters=400):
    if alpha_list is None:
        alpha_list = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        for mu in [1.0, 5.0, 20.0]:
            if time.time() > deadline:
                break
            try:
                res = minimize(_penalty_obj, h, jac=_penalty_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': max_iters, 'ftol': 1e-17,
                                       'gtol': 1e-15, 'maxfun': max_iters * 20,
                                       'maxls': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    def _sobj(x):
        return _penalty_obj(x, n, alpha, 0.0)
    def _scon(x):
        return float(x.sum() - n / 2.0)
    def _sjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _sjac}]
    try:
        res = minimize(_sobj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def target_lag_search(h, n, deadline):
    """Focus search on reducing the lag that gives the maximum"""
    best_c = compute_c5(h, n)
    for _ in range(500):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_star = int(np.argmax(c))
        k = k_star - (n - 1)
        # Identify which pairs affect c[k_star]
        # c[m] = sum_i h[i]*(1-h[i+m-n+1])*dx for valid indices
        improved = False
        # Try moving mass between pairs that affect the target lag
        if k >= 0:
            pairs = [(i, i + k) for i in range(n - k)]
        else:
            kk = -k
            pairs = [(i, i + kk) for i in range(n - kk)]
        rng_move = np.random.RandomState(42 + k_star)
        for _ in range(20):
            idx = rng_move.randint(0, len(pairs))
            i, j = pairs[idx]
            for delta in [0.005, 0.002, 0.001, 0.0005]:
                if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                    h[i] += delta; h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new; improved = True; break
                    else:
                        h[i] -= delta; h[j] += delta
                if h[j] + delta <= 1.0 and h[i] - delta >= 0.0:
                    h[j] += delta; h[i] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new; improved = True; break
                    else:
                        h[j] -= delta; h[i] += delta
        if not improved:
            break
    return h, best_c

def fine_pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_guided_search(h, n, deadline):
    """Use gradient to determine best pair moves"""
    best_c = compute_c5(h, n)
    for alpha in [200, 1000, 5000]:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            # Sort by gradient - move from high gradient to low gradient
            idx = np.argsort(g)
            for _ in range(50):
                if time.time() > deadline:
                    return h, best_c
                i = idx[-1]  # highest gradient (most to reduce)
                j = idx[0]   # lowest gradient
                for delta in [0.003, 0.001, 0.0005, 0.0002]:
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta; h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True; break
                        h[i] += delta; h[j] -= delta
                if not improved:
                    # Try next pair
                    idx = np.array([x for x in idx if not (x == i or x == j)])
                    if len(idx) < 2:
                        break
            if not improved:
                break
    return h, best_c

def fourier_init(n, K, rng, amplitude=0.3):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):

        b = rng.uniform(-amplitude, amplitude) / k
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n)
