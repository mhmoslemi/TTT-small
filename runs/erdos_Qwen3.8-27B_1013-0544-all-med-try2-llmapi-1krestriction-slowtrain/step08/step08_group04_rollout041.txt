        t3 = min(time.time() + 80, deadline - 180)
        h, c = lbfgs_opt(h, n, t3)
        if c < best_c:
            best_c = c; best_h = h.copy()
    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 120)
        h, c = grad_pair(h, n, t4)
        if c < best_c:
            best_c = c; best_h = h.copy()
    # Phase 5: Standard pair search
    if time.time() < deadline - 60:
        t5 = min(time.time() + 60, deadline - 60)
        h, c = pair_search(h, n, t5, [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=60)
        if c < best_c:
            best_c = c; best_h = h.copy()
    # Phase 6: Try other n values
    if time.time() < deadline - 70:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 35:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 20)
            if alloc < 20:
                break
            h_t, c_t = lbfgs_opt(h_interp, n_try, time.time() + alloc * 0.6)
            if time.time() < deadline - 15:
                h_t, c_t = target_opt(h_t, n_try, time.time() + min(30, deadline - time.time() - 10))
            if c_t < best_c:
                best_c = c_t; best_h = h_t.copy()
    # Phase 7: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        h, c = lbfgs_opt(best_h, nf, deadline - 8)
        if c < best_c:
            best_c = c; best_h = h.copy()
        if time.time() < deadline - 2:
            h, c = pair_search(best_h, nf, deadline - 1,
                               [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=40)
            if c < best_c:
                best_c = c; best_h = h
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
