    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    logsumexp = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    p /= p.sum()
    
    grad = np.zeros(n)

    # p[k] corresponds to shift s = k - (n-1)

            continue

        # (1-x[i+s]) for valid i: i+s in [0, n-1], i in [0, n-1]

            grad[i_lo2:i_hi2] -= w * x[i_lo2 - s:i_hi2 - s]
