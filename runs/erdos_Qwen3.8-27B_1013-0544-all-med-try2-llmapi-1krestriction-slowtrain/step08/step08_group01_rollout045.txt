            stall += 1
            step = step0
            alpha = max(alpha * 0.5, 50)
            if stall > 8:
                break
    return best_h, best_c

def target_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = best_c + 0.001
    for _ in range(30):
        if time.time() > deadline - 10:
            break
        target -= 0.0003
        alpha = max(50000, 5000.0 / max(target, 0.01))
        bounds = [(0.0, 1.0)] * n
        mu = 10.0
        try:
            res = minimize(obj_sm, best_h, jac=grad_sm,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 150, 'ftol': 1e-17,
                                   'gtol': 1e-13, 'maxfun': 4000,
                                   'maxls': 40, 'm': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def golden_ratio_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_restart(h, n, rng, n_modes=15, amp=0.06):
    h_new = h.copy()
    freqs = rng.choice(range(1, n // 2), size=min(n_modes, n // 2 - 1), replace=False)
    x = np.arange(n)
    for f in freqs:
        a = rng.uniform(-amp, amp)
        ph = rng.uniform(0, 2 * np.pi)
        h_new += a * np.cos(2 * np.pi * f * x / n + ph)
    return project_feasible(h_new, n)

def fast_pair_polish(h, n, deadline, radius=60):
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = golden_ratio_start(500)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Aggressive multi-stage L-BFGS
    t1 = min(t0 + 180, deadline - 350)
    best_h, best_c = multistage_lbfgs(best_h, best_n, t1)

    # Phase 2: Projected gradient refinement
    if time.time() < deadline - 250:
        t2 = min(time.time() + 100, deadline - 250)
        h2, c2 = proj_grad_descent(best_h, best_n, t2, alpha=300000, step0=0.008)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Target-based descent
    if time.time() < deadline - 150:
        t3 = min(time.time() + 80, deadline - 150)
        h3, c3 = target_descent(best_h, best_n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try n=400 (finer optimization on smaller grid)
    if time.time() < deadline - 120:
        n4 = 400
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n4, endpoint=False)
        h4 = np.interp(x_new, x_old, best_h)
        h4 = project_feasible(h4, n4)
        alloc4 = min(70, (deadline - time.time() - 80) * 0.5)
        if alloc4 > 25:
            h4, c4 = multistage_lbfgs(h4, n4, time.time() + alloc4)
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = n4

    # Phase 5: Diverse restarts
    starts = [
        golden_ratio_start(best_n),
        fourier_restart(best_h, best_n, rng, 12, 0.07),
        fourier_restart(best_h, best_n, rng, 25, 0.03),
    ]
    for s in starts:
        if time.time() > deadline - 60:
            break
        alloc = min(40, (deadline - time.time() - 25) * 0.5)
        if alloc < 15:
            break
        h_s, c_s = multistage_lbfgs(s, best_n, time.time() + alloc,
                                     stage_alphas=(500, 5000, 50000, 500000, 5000000))
        if time.time() < deadline - 20:
            h_s, c_s = proj_grad_descent(h_s, best_n, min(time.time() + 15, deadline - 15),
                                          alpha=500000, step0=0.004)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 6: Final high-alpha L-BFGS
    if time.time() < deadline - 20:
        bounds = [(0.0, 1.0)] * best_n
        t6 = min(time.time() + 15, deadline - 10)
        try:
            res = minimize(obj_sm, best_h, jac=grad_sm,
                          args=(best_n, 50000000, 10.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 5000, 'maxls': 40, 'm': 40})
            h_f = project_feasible(res.x, best_n)
            c_f = compute_c5(h_f, best_n)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        except Exception:
            pass

    # Phase 7: Final pair polish
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_polish(best_h, best_n, deadline - 1, radius=50)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
