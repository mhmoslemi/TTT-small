import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def build_contrib(h, n):
    """Build the contribution matrix contrib[p,k]."""
    m_range = np.arange(-(n-1), n)
    p_idx = np.arange(n)[:, None]
    m_idx = m_range[None, :]
    pm = p_idx - m_idx
    pp = p_idx + m_idx
    valid_pm = (pm >= 0) & (pm < n)
    valid_pp = (pp >= 0) & (pp < n)
    h_pm = np.where(valid_pm, h[np.clip(pm, 0, n-1)], 0)
    h_pp = np.where(valid_pp, h[np.clip(pp, 0, n-1)], 0)
    contrib = (1 - h_pm) * valid_pm.astype(float) - h_pp * valid_pp.astype(float)
    return contrib  # shape (n, 2n-1)

def greedy_best_swap(h_init, n, deadline, max_sweeps=200):
    """Greedy binary search taking the best swap each iteration."""
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    contrib = build_contrib(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        # Find best swap
        best_score = cur_c + 1e-15
        best_pair = None
        # For each zero, find best one to swap with
        for i in zeros:
            # scores for all j in ones
            c_i = corr + contrib[i]  # (2n-1,)
            # For each j: max(c_i - contrib[j])
            deltas = c_i[None, :] - contrib[ones]  # (len(ones), 2n-1)
            maxes = deltas.max(axis=1)  # (len(ones),)
            idx = maxes.argmin()
            if maxes[idx] * dx < best_score:
                best_score = maxes[idx] * dx
                best_pair = (i, ones[idx])
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i], h[j] = 1.0, 0.0
        corr = corr + contrib[i] - contrib[j]
        contrib = build_contrib(h, n)
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def build_contrib(h, n):
    m_range = np.arange(-(n - 1), n)
    p_idx = np.arange(n)[:, None]
    m_idx = m_range[None, :]
    pm = p_idx - m_idx
    pp = p_idx + m_idx
    valid_pm = (pm >= 0) & (pm < n)
    valid_pp = (pp >= 0) & (pp < n)
    pm_c = np.clip(pm, 0, n - 1)
    pp_c = np.clip(pp, 0, n - 1)
    h_pm = np.where(valid_pm, h[pm_c], 0.0)
    h_pp = np.where(valid_pp, h[pp_c], 0.0)
    contrib = (1.0 - h_pm) * valid_pm.astype(float) - h_pp * valid_pp.astype(float)
    return contrib

def greedy_best_swap(h_init, n, deadline, max_sweeps=2000):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    contrib = build_contrib(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        best_score = cur_c + 1e-18
        best_i = -1
        best_j = -1
        for i in zeros:
            c_i = corr + contrib[i]
            deltas = c_i[None, :] - contrib[ones]
            maxes = deltas.max(axis=1)
            idx = maxes.argmin()
            sc = maxes[idx] * dx
            if sc < best_score:
                best_score = sc
                best_i = i
                best_j = ones[idx]
        if best_i < 0:
            break
        h[best_i], h[best_j] = 1.0, 0.0
        corr = corr + contrib[best_i] - contrib[best_j]
        contrib = build_contrib(h, n)
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, T0=0.02, cool=0.99992):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    contrib = build_contrib(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    while time.time() < deadline:
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        delta = contrib[i] - contrib[j]
        nc = (corr + delta).max() * dx
        accept = nc <= cur_c or (T > 1e-10 and rng.random() < np.exp((cur_c - nc) / T))
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr = corr + delta
            contrib = build_contrib(h, n)
            cur_c = corr.max() * dx
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def continuous_lp(h, n, p, maxiter, ftol):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    p_idx = np.arange(n)[:, None]
    m_idx = m_range[None, :]
    pm = p_idx - m_idx
    pp = p_idx + m_idx
    valid_pm = (pm >= 0) & (pm < n).astype(float)
    valid_pp = (pp >= 0) & (pp < n).astype(float)
    pm_c = np.clip(pm, 0, n - 1)
    pp_c = np.clip(pp, 0, n - 1)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return float(np.sum(c ** p) / (2 * n - 1))
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        S = np.sum(c ** p)
        if p <= 1:
            w = np.ones(2 * n - 1)
        else:
            w = np.maximum(c, 0) ** (p - 1)
        h_pm = np.where(valid_pm > 0, x[pm_c], 0.0)
        h_pp = np.where(valid_pp > 0, x[pp_c], 0.0)
        dc = (1.0 - h_pm) * valid_pm - h_pp * valid_pp
        g = dc @ w * dx
        factor = S ** (1.0 / p - 1.0)
        return factor * g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def continuous_lse(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    p_idx = np.arange(n)[:, None]
    m_idx = m_range[None, :]
    pm = p_idx - m_idx
    pp = p_idx + m_idx
    valid_pm = (pm >= 0) & (pm < n).astype(float)
    valid_pp = (pp >= 0) & (pp < n).astype(float)
    pm_c = np.clip(pm, 0, n - 1)
    pp_c = np.clip(pp, 0, n - 1)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w = w / w.sum()
        h_pm = np.where(valid_pm > 0, x[pm_c], 0.0)
        h_pp = np.where(valid_pp > 0, x[pp_c], 0.0)
        dc = (1.0 - h_pm) * valid_pm - h_pp * valid_pp
        g = dc @ w * dx
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    x_src = np.linspace(0, 2, n_src)
