def lse_gradient(h, p, n):
    """Compute gradient of LSE softening of correlation values.
    p: probability vector of length 2n-1 (softmax of alpha*c)
    Returns gradient of length n."""
    g = np.zeros(n)
    # k >= 0 part: k = 0, 1, ..., n-1
    for k in range(n):
        pk = p[k + n - 1]
        if pk < 1e-15:
            continue
        g[k:] += pk * (1.0 - h[:n-k])
        g[:n-k] -= pk * h[k:]
    # k < 0 part: k = -1, -2, ..., -(n-1), i.e., j = 1, ..., n-1
    for j in range(1, n):
        pj = p[n - 1 - j]
        if pj < 1e-15:
            continue
        g[:n-j] += pj * (1.0 - h[j:])
        g[j:] -= pj * h[:n-j]
    return g

            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_gradient(h, p, n):
    g = np.zeros(n)
    for k in range(n):
        pk = p[k + n - 1]
        if pk < 1e-12:
            continue
        g[k:] += pk * (1.0 - h[:n - k])
        g[:n - k] -= pk * h[k:]
    for j in range(1, n):
        pj = p[n - 1 - j]
        if pj < 1e-12:
            continue
        g[:n - j] += pj * (1.0 - h[j:])
        g[j:] -= pj * h[:n - j]
    return g

def lse_optimize(h0, n, deadline, alpha0=50, alpha_mult=1.03, alpha_max=1e8, max_no_improve=300):
    h = h0.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    c = corr * dx
    best_h = h.copy()
    best_c = c.max()
    alpha = alpha0
    no_improve = 0
    lr_base = 0.008
    rng_local = np.random.RandomState(123)

    while time.time() < deadline:
        batch = 50
        for _ in range(batch):
            if time.time() > deadline:
                break
            shifted = alpha * c
            sm = shifted.max()
            exp_v = np.exp(shifted - sm)
            Z = exp_v.sum()
            if Z < 1e-300:
                break
            p = exp_v / Z
            g = lse_gradient(h, p, n)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break
            lr = lr_base / (1.0 + alpha * g_norm * 0.01)
            improved = False
            for _bt in range(25):
                h_new = project_feasible(h - lr * g, n)
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c_new = corr_new * dx
                cmax_new = c_new.max()
                if cmax_new < best_c - 1e-15:
                    h = h_new
                    c = c_new
                    best_c = cmax_new
                    best_h = h.copy()
                    no_improve = 0
                    improved = True
                    break
                elif cmax_new < c.max() - 1e-13:
                    h = h_new
                    c = c_new
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                no_improve += 1
                if no_improve > max_no_improve:
                    alpha = min(alpha * alpha_mult, alpha_max)
                    no_improve = 0
                    if alpha >= alpha_max * 0.99:
                        h = project_feasible(h + 0.02 * rng_local.randn(n), n)
                        c = np.correlate(h, 1 - h, mode='full') * dx
                elif no_improve > 100:
                    alpha = min(alpha * 1.5, alpha_max)
                    no_improve = 0
        if no_improve > max_no_improve * 2:
            break
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=2000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.012
    cool = 0.99994
    scale = 2500.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        delta[n - 1 + (j - i)] += 1.0
        delta[n - 1 + (i - j)] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def active_set_optimize(h0, n, deadline, max_iters=5000):
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c = corr * dx
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
        threshold = cmax - 0.0005
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.where(c >= cmax - 0.001)[0]
        alpha = 5000.0
        sub_c = c[active]
        sm = sub_c.max()
        exp_v = np.exp(alpha * (sub_c - sm))
        p_active = exp_v / exp_v.sum()
        p_full = np.zeros(2 * n - 1)
        p_full[active] = p_active
        g = lse_gradient(h, p_full, n)
        lr = 0.003
        improved = False
        for _bt in range(20):
            h_new = project_feasible(h - lr * g, n)
            c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
            if c_new.max() < cmax - 1e-14:
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: LSE gradient descent with increasing alpha
    p1 = t0 + 0.38 * total
    h1, c1 = lse_optimize(h0, n, p1, alpha0=30, alpha_mult=1.025, alpha_max=3e6)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary refinement
    p2 = t0 + 0.52 * total
    hb, cb = greed_binary(best_h, n, p2, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    p2b = t0 + 0.68 * total
    ha, ca = anneal_binary(best_h, n, p2b, seed=seed, n_iters=3000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 3: LSE polish from best (possibly binary) result
    p3 = t0 + 0.82 * total
    h3, c3 = lse_optimize(best_h, n, p3, alpha0=200, alpha_mult=1.02, alpha_max=1e7)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    # Phase 4: Active set focused optimization
    p4 = t0 + 0.90 * total
    h4, c4 = active_set_optimize(best_h, n, p4)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Final LSE polish with high alpha + noise restarts
    p5 = dl - 3
    for trial in range(4):
        if time.time() > p5 - 2:
            break
        if trial == 0:
            h_start = best_h.copy()
        else:
            h_start = project_feasible(best_h + 0.008 * rng.randn(n), n)
        sub_dl = min(t0 + 6, p5 - (trial * 2) * 0.5)
        if sub_dl <= time.time():
            sub_dl = p5 - 1
        hs, cs = lse_optimize(h_start, n, sub_dl, alpha0=5000 + trial * 5000,
                              alpha_mult=1.01, alpha_max=5e7)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
