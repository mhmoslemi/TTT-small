import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_pts = len(x)
    top_k = min(100, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n_pts)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 200.0 * s
    return grad

def flat_spectrum_init(h, n, n_iter=500):
    """Gerchberg-Saxton style: alternate between flat spectrum and constraint projection"""
    h = h.copy()
    for it in range(n_iter):
        H = np.fft.fft(h)
        # Target: flat magnitude (preserve DC)
        H[0] = 0  # Zero out DC for spectral shaping
        target_mag = np.sqrt(np.sum(np.abs(H)**2) / (n - 1))
        H[0] = np.fft.fft(h)[0]  # Restore DC
        mag = np.ones(n) * target_mag
        mag[0] = np.abs(np.fft.fft(h)[0])
        # Actually, let me be more careful
        H_full = np.fft.fft(h)
        dc = H_full[0]
        ac_power = np.sum(np.abs(H_full[1:])**2)
        target_ac = np.sqrt(ac_power / (n - 1))
        H_new = np.zeros(n, dtype=complex)
        H_new[0] = dc
        H_new[1:] = target_ac * np.exp(1j * np.angle(H_full[1:]))
        h = np.real(np.fft.ifft(H_new))
        h = project_feasible(h, n)
    return h

def coordinate_descent(h, n, deadline, top_pairs=200):
    """Smart coordinate descent using gradient to guide pair selection"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for _ in range(10000):
        if time.time() > deadline:
            break
        # Compute gradient to find good pairs
        c = compute_c5_vec(h, n)
        idx = np.argsort(c)[-10:]
        top_idx = c[idx]
        m = top_idx.max()
        alpha = 50.0
        sh = alpha * (top_idx - m)
        w = np.exp(sh - sh.max())
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(len(idx)):
            k = idx[i] - (n - 1)
            grad += w[i] * dx * grad_single(h, n, k)
        
        # For each i with large |grad[i]|, try to move mass to/from a good j
        # Priority: large positive grad[i] means we want to decrease h[i]
        # Large negative grad[i] means we want to increase h[i]
        
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if grad[i] > 0:  # Want to decrease h[i]
                    if h[i] < d:
                        continue
                    # Find best j to receive the mass (where grad[j] < 0)
                    # Just try a few candidates
                    candidates = np.argsort(grad)[:top_pairs]
                    for j in candidates:
                        if i == j:
                            continue
                        if h[j] + d > 1.0:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                elif grad[i] < 0:  # Want to increase h[i]
                    if h[i] + d > 1.0:
                        continue
                    candidates = np.argsort(grad)[-top_pairs:]
                    for j in candidates:
                        if i == j:
                            continue
                        if h[j] - d < 0.0:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved and d < 0.001:
                break
    return best_h, best_c

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 100.0 * (x.sum() - n/2.0)**2)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(80, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    grad += 200.0 * (x.sum() - n/2.0)
    return grad

    best_c = compute_c5(h, n)
    step = 0.05
    prev_h = h.copy()
    prev_g = None
    stall = 0
    while time.time() < deadline:

        m = c.max()
        sh = alpha * (c - m)
        sh -= sh.max()
        w = np.exp(sh)
        w /= w.sum()

            k = i - (n - 1)
            g += w[i] * dx * grad_single(h, n, k)

            y = g - prev_g
            sy = s @ y
            ss = s @ s
            if sy > 1e-20:

                    stall = 0

                    stall += 1
                accepted = True

        if stall > 100:

    return best_h, best_c

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 100.0 * (x.sum() - n / 2.0) ** 2)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(80, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    grad += 200.0 * (x.sum() - n / 2.0)
    return grad

def flat_spectrum(h, n, n_iter=600):
    h = h.copy()
    for _ in range(n_iter):
        H = np.fft.fft(h)
        energy = np.sum(h ** 2)
        dc_val = h.sum()
        target_ac = np.sqrt(max(0.0, n * energy - dc_val ** 2) / (n - 1))
        H_new = np.zeros(n, dtype=complex)
        H_new[0] = dc_val
        ang = np.angle(H[1:])
        H_new[1:] = target_ac * np.exp(1j * ang)
        h = np.real(np.fft.ifft(H_new))
        h = project_feasible(h, n)
    return h

def pgd_refine(h, n, deadline, alpha=120.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.04
    prev_h = h.copy()
    prev_g = None
    stall = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        m = c.max()
        sh = alpha * (c - m)
        sh -= sh.max()
        w = np.exp(sh)
        w /= w.sum()
        dx = 2.0 / n
        top_k = min(60, len(w))
        top_idx = np.argsort(w)[-top_k:]
        g = np.zeros(n)
        for i in top_idx:
            if w[i] < 1e-12:
                continue
            k = i - (n - 1)
            g += w[i] * dx * grad_single(h, n, k)
        if prev_g is not None:
            s = h - prev_h
            y = g - prev_g
            sy = s @ y
            ss = s @ s
            if sy > 1e-20:
                step = max(0.001, min(0.12, ss / sy))
            else:
                step = max(0.001, step * 0.9)
        prev_h = h.copy()
        prev_g = g.copy()
        accepted = False
        for _ in range(14):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-13:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.5
        if not accepted:
            stall += 1
        if stall > 80:
            break
    return best_h, best_c

def pair_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    while time.time() < deadline:

        sh -= sh.max()

        w /= w.sum()
        dx = 2.0 / n
        top_k = min(15, len(w))
        top_idx = np.argsort(w)[-top_k:]
        g = np.zeros(n)
        for i in top_idx:
            if w[i] < 1e-10:
                continue
            k = i - (n - 1)
            g += w[i] * dx * grad_single(h, n, k)
        pos_mask = g > 0
        neg_mask = g < 0
        if not np.any(pos_mask) or not np.any(neg_mask):
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        pos_sel = pos_idx[np.argsort(-g[pos_idx])][:50]
        neg_sel = neg_idx[np.argsort(g[neg_idx])][:50]
        any_improved = False
        for d in deltas:

            for i in pos_sel:

                        any_improved = True
                    else:
                        h[i] += d
                        h[j] -= d
        if not any_improved:
            break
    return best_h, best_c
