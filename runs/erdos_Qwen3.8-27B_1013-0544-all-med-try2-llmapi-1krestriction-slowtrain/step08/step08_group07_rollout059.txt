    h_neg = np.where(valid_neg, h[im], 0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(x, alpha, mu):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    exps = np.exp(shifted)
    w = exps / exps.sum()
    
    obj = c_max + np.log(exps.sum()) / alpha + mu * (x.sum() - n/2.0)**2
    
    # Gradient using broadcasting
    i_idx = np.arange(n)[:, None]
    k_vals = np.arange(-(n-1), n)[None, :]
    ik = i_idx + k_vals
    im = i_idx - k_vals
    valid_pos = (ik >= 0) & (ik < n)
    valid_neg = (im >= 0) & (im < n)
    
    h_pos = np.where(valid_pos, x[ik], 0.0)
    h_neg = np.where(valid_neg, x[im], 0.0)
    
    A = (h_pos * w[None, :]).sum(axis=1)
    C = (valid_pos.astype(np.float64) * w[None, :]).sum(axis=1)
    B = (h_neg * w[None, :]).sum(axis=1)
    
    grad = C - A - B + 2 * mu * (x.sum() - n/2.0)
    
    return obj, grad

def lbfgs_smooth(h, n, alpha, mu, maxiter=1000):
    def obj(x):
        return smooth_max_obj_grad(x, alpha, mu)[0]
    def grad(x):
        return smooth_max_obj_grad(x, alpha, mu)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def coord_descent_fast(h, n, max_sweeps=100, deadline=None):
    """Fast coordinate descent with vectorized operations"""
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    m_range = np.arange(-(n-1), n)
    
    result = h.copy()
    best_c = corr.max()
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            
            # Find optimal x that minimizes max(corr + B*x)
            # Binary search on the value v
            val0 = np.max(corr - B * 0)  # = max(corr) (x=0)
            val1 = np.max(corr + B * 1)  # (x=1)
            
            lo_v = min(val0, val1) - 1e-12
            hi_v = max(val0, val1) + 1e-9
            
            for _ in range(50):
                mid = (lo_v + hi_v) / 2.0
                # Check if there exists x in [0,1] such that corr[s] + B[s]*x <= mid for all s
                # For B[s] > 0: x <= (mid - corr[s]) / B[s]
                # For B[s] < 0: x >= (mid - corr[s]) / B[s]
                # For B[s] = 0: need corr[s] <= mid
                pos = B > 1e-14
                neg = B < -1e-14
                zero = ~(pos | neg)
                
                if zero.any() and corr[zero].max() > mid + 1e-14:
                    lo_v = mid
                    continue
                
                lb_x = 0.0
                if neg.any():
                    lb_x = max(lb_x, np.max((mid - corr[neg]) / B[neg]))
                ub_x = 1.0
                if pos.any():
                    ub_x = min(ub_x, np.min((mid - corr[pos]) / B[pos]))
                
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            
            # Find optimal x at hi_v
            pos = B > 1e-14
            neg = B < -1e-14
            lb_x = 0.0
            if neg.any():
                lb_x = max(lb_x, np.max((hi_v - corr[neg]) / B[neg]))
            ub_x = 1.0
            if pos.any():
                ub_x = min(ub_x, np.min((hi_v - corr[pos]) / B[pos]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
        
        c_now = corr.max()
        if c_now < best_c:
            best_c = c_now
            result = h.copy()
    
    return result, best_c

    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Quick coordinate descent (10%)
    t = t0 + 0.10 * budget_s
    hc, _ = coord_desc(best_h, n, 50, t)
    hc = project_feasible(hc, n)
    c = compute_c5(hc, n)
    if c < best_c:
        best_c, best_h = c, hc.copy()

    # Phase 2: L-BFGS-B main optimization (45%)
    t = t0 + 0.55 * budget_s
    alphas = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    h = best_h.copy()
    for i_a, alpha in enumerate(alphas):
        if time.time() > t - 15:
            break
        remaining = (t - 15 - time.time()) / (len(alphas) - i_a)
        maxiter = max(100, int(remaining * 300))
        maxiter = min(maxiter, 5000)
        h = lbfgs_opt(h, n, alpha, 500.0, maxiter, 1e-18)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Coordinate descent polish (15%)
    t = t0 + 0.70 * budget_s
    hc, _ = coord_desc(best_h, n, 100, t)
    hc = project_feasible(hc, n)
    c = compute_c5(hc, n)
    if c < best_c:
        best_c, best_h = c, hc.copy()

    # Phase 4: High-alpha L-BFGS-B (10%)
    t = t0 + 0.80 * budget_s
    for alpha in [5000000, 50000000, 500000000]:
        if time.time() > t - 5:
            break
        h = lbfgs_opt(best_h, n, alpha, 500.0, 3000, 1e-18)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Noise restarts (10%)
    t = t0 + 0.90 * budget_s
    for trial in range(4):
        if time.time() > t - 10:
            break
        scale = 0.005 * (trial + 1)
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        h = lbfgs_opt(h, n, 10000, 500.0, 1500, 1e-16)
        h = project_feasible(h, n)
        hc, _ = coord_desc(h, n, 30, t - 5)
        hc = project_feasible(hc, n)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c, best_h = c, hc.copy()

    # Phase 6: Multi-resolution (10%)
    t = dl
    for nh in [300, 500, 600]:
        if time.time() > t - 15:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = lbfgs_opt(hh, nh, 5000, 500.0, 1500, 1e-16)
        hh = project_feasible(hh, nh)
        hc, _ = coord_desc(hh, nh, 40, t - 8)
        hc = project_feasible(hc, nh)
        ch = compute_c5(hc, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hc.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
