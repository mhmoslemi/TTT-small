def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def precompute_masks(n):
    idx = np.arange(n)

    return j1_safe, mask1, j2_safe, mask2

def _obj(x):

    alpha = _G['alpha']
    lam = _G['lam']

    m = g.max()
    u = alpha * (g - m)
    umax = u.max()
    exp_u = np.exp(u - umax)
    sum_exp = exp_u.sum()
    return m + np.log(sum_exp) / alpha

def _grad(x):
    n = _G['n']
    alpha = _G['alpha']
    lam = _G['lam']
    j1s, m1, j2s, m2 = _G['j1s'], _G['m1'], _G['j2s'], _G['m2']
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    u = alpha * (g - m)
    umax = u.max()
    exp_u = np.exp(u - umax)
    w = exp_u / exp_u.sum()
    
    term1 = m1 @ w
    x_j1 = x[j1s]
    term2 = (x_j1 * m1) @ w
    x_j2 = x[j2s]
    term3 = (x_j2 * m2) @ w
    
    grad = dx * (term1 - term2 - term3) + 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def coord_descent(h, n, max_sweeps=100, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)

            val0 = np.max(corr - B * h[i])

            neg_B = B < -1e-14

            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf

                    continue

            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
