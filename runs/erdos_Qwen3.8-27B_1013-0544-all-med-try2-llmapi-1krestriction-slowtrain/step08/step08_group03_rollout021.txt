import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:k+nv]
        grad[k:k+nv] -= h[:nv]
    else:
        kk = -k
        grad[kk:kk+nv] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:kk+nv]
    return grad

def smooth_max_obj_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    obj = m + (mx + np.log(w.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only use top k lags for efficiency
    top_k = min(50, len(w))
    top_idx = np.argsort(w)[-top_k:]
    top_w = w[top_idx].copy()
    top_w /= top_w.sum()
    for i in top_idx:
        if top_w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += top_w[i] * dx * grad_lag(x, n, k)
    return obj, grad

def adam_optimize(h, n, deadline, alpha=200.0, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    t = 0
    last_improve = time.time()
    patience = 30  # seconds
    
    target = n / 2.0
    
    while time.time() < deadline:
        t += 1
        obj, grad = smooth_max_obj_grad(h, n, alpha)
        
        # Add sum constraint penalty to gradient
        s = h.sum() - target
        grad = grad + 50.0 * s
        
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)

            last_improve = time.time()
            h = h_new
        else:
            h = h_new  # still take the step
            if time.time() - last_improve > patience:
                # Reduce learning rate
                lr *= 0.5
                last_improve = time.time()
                if lr < 1e-7:
                    break
    
    return best_h, best_c

So the change is EXACTLY linear (first order = exact order) because the objective is bilinear! This means the analytic pair exchange is EXACT, not just approximate.

This is great! So for a pair (i, j) with move h[i] += s*d, h[j] -= s*d:
new_c_k = c_k + s*d * (grad_k[i] - grad_k[j])

And we want to minimize max_k new_c_k.

This is a 1D piecewise linear minimization problem for each pair. The optimal d is at a "breakpoint" where two lags cross.

Now, the challenge is efficiency. For n=500, there are ~999 lags and 500*499/2 ≈ 125K pairs (times 2 for direction). For each pair, finding the optimal d requires O(n_lags) work (to find the breakpoints). Total: 250K * 1000 = 250M operations per full pass. That's too slow.

Optimization: we can restrict to pairs near the diagonal (small |i-j|) or use a smarter search.

Actually, for the local polish, we don't need to try ALL pairs. We can:
1. Find the current max lag k*
2. Find indices i where grad_{k*}[i] is positive (increasing h[i] would increase the max lag - bad)
3. Find indices j where grad_{k*}[j] is negative (decreasing h[j] would decrease the max lag - good)
4. Try pairs (i, j) where i is in group 2 and j is in group 1, or vice versa

Wait, I want to DECREASE the max lag. So I want:
- s * (grad_{k*}[i] - grad_{k*}[j]) < 0
- If s = +1: grad_{k*}[i] < grad_{k*}[j], so moving mass from i to j decreases lag k*
- If s = -1: grad_{k*}[i] > grad_{k*}[j], so moving mass from j to i decreases lag k*

But we also need to not increase other lags too much. So we need to check all lags.

For efficiency, let me limit the search to a subset of promising pairs. Here's my approach:

1. Compute c and identify top K lags
2. For the top lag, find the indices with highest and lowest gradient values
3. Try a limited number of pairs (say 1000-5000) per iteration
4. For each pair, analytically find the optimal d
5. Apply the best move
6. Repeat
