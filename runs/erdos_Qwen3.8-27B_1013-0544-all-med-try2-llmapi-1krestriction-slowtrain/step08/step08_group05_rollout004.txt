def eval_counts(h):

def eval_c5(h, n):
    return float(np.max(eval_counts(h))) * (2.0 / n)

def project(h, n):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    g = n / 2.0
    if h.size != n:
        h = np.resize(h, n)
    for _ in range(10):
        d = g - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h = h + (d / s) * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (d / s) * h
    return np.clip(h, 0.0, 1.0)

def make_random_balanced(n, seed):
    rng = np.random.RandomState(seed)
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, n // 2, replace=False)] = 1.0
    return a

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    if n_old < 2:
        return np.full(n_new, 0.5)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def binarize(h, n, seed=0):
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = resample_h(h, n)
    h = np.clip(h, 0.0, 1.0) + 1e-12 * np.arange(n)
    out = np.zeros(n, dtype=float)
    out[np.argsort(h)[::-1][:n // 2]] = 1.0
    return project(out, n)

def binary_local_search(h_ini, n, deadline, seed=0, max_sweeps=12, target=None):
    h8 = binarize(h_ini, n, seed).astype(float)
    z8 = 1.0 - h8
    corr = np.correlate
    c = corr(h8, z8, mode="full")
    M = float(c.max())
    bh = h8.copy()
    bM = M
    rng = np.random.RandomState(seed)
    for _ in range(max_sweeps):
        if time.time() > deadline or (target is not None and M <= target):
            break
        z = np.where(z8 > 0.5)[0]
        o = np.where(h8 > 0.5)[0]
        bn, bi, bj = M, -1, -1
        stop = False
        if n <= 350 and len(z) * len(o) <= 60000:
            for a in range(len(z)):
                i = z[a]
                h8[i] = 1.0; z8[i] = 0.0
                for b in range(len(o)):
                    j = o[b]
                    if j == i:
                        continue
                    h8[j] = 0.0; z8[j] = 1.0
                    nm = float(corr(h8, z8, mode="full").max())
                    h8[j] = 1.0; z8[j] = 0.0
                    if nm < bn:
                        bn, bi, bj = nm, i, j
                        if target is not None and nm <= target:
                            stop = True
                            break
                h8[i] = 0.0; z8[i] = 1.0
                if stop:
                    break
        else:
            ns = min(len(z) * len(o), 12000)
            for _ in range(ns):
                i = z[rng.randint(len(z))]; j = o[rng.randint(len(o))]
                if i == j:
                    continue
                h8[i] = 1.0; z8[i] = 0.0
                h8[j] = 0.0; z8[j] = 1.0
                nm = float(corr(h8, z8, mode="full").max())
                h8[j] = 1.0; z8[j] = 0.0
                h8[i] = 0.0; z8[i] = 1.0
                if nm < bn:
                    bn, bi, bj = nm, i, j
                    if target is not None and nm <= target:
                        break
        if bn < M and bi >= 0:
            h8[bi] = 1.0; z8[bi] = 0.0
            h8[bj] = 0.0; z8[bj] = 1.0
            c = corr(h8, z8, mode="full")
            M = float(c.max())
            if M < bM:
                bM, bh = M, h8.copy()
        else:
            break
    return bh, bM

def active_lp_polish(h_ini, n, deadline, max_calls=20):
    h = project(h_ini, n)
    c = eval_counts(h)
    M = float(c.max())
    bh, bM = h.copy(), M
    calls = 0
    for sb in (0.50, 0.25, 0.12, 0.06):
        stall = 0
        while time.time() < deadline and calls < max_calls and stall < 4:
            calls += 1
            tol = max(0.4, 0.03 * M + 0.1)
            top = np.argsort(-c)[:min(180, 2 * n - 1)]
            sel = top[c[top] >= M - tol]
            if sel.size < 2:
                sel = top[:min(60, len(top))]
            A = np.empty((sel.size, n))
            for r, q in enumerate(sel):
                l = int(q - (n - 1))
                row = A[r]
                row[:] = 0.0
                if l >= 0:
                    m = n - l
                    if m > 0:
                        row[:m] += 1.0 - h[l:]
                        row[l:] -= h[:m]
                else:
                    m = -l
                    if m < n:
                        row[m:] += 1.0 - h[:n - m]
                        row[:n - m] -= h[m:]
            lo = np.maximum(-h, -sb)
            hi = np.minimum(1.0 - h, sb)
            obj = np.zeros(n + 1)
            obj[-1] = 1.0
            A_ub = np.hstack((A, -np.ones((sel.size, 1))))
            b_ub = -c[sel]
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.zeros(1)
            bnds = [(float(lo[i]), float(hi[i])) for i in range(n)] + [(None, None)]
            res = None
            try:
                res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bnds, method="highs")
            except Exception:
                try:
                    res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                  bounds=bnds, method="interior-point")
                except Exception:
                    res = None
            if res is None or res.status != 0 or not np.isfinite(res.x).all():
                stall += 1
                continue
            d = res.x[:n].astype(float)
            d -= d.sum() / n
            d = np.clip(d, lo, hi)
            d -= d.sum() / n
            if np.max(np.abs(d)) < 1e-9:
                stall += 1
                continue
            ok = False
            for a in (1.0, 0.5, 0.25, 0.125, 0.0625):
                nh = project(h + a * d, n)
                nc = eval_counts(nh)
                nm = float(nc.max())
                if nm < M - 1e-10:
                    h, c, M = nh, nc, nm
                    ok = True
                    if M < bM:
                        bM, bh = M, nh.copy()
                    break
            stall = 0 if ok else stall + 1
        if calls >= max_calls:
            break
    return bh, bM

def smooth_f(x):
    n = _SM["n"]
    al = float(_SM["alpha"])
    c = eval_counts(x)
    m = float(c.max())
    e = np.exp(al * (c - m))
    s = float(e.sum())
    if s > 0.0:
        w = e / s
        v = m + float(np.log(s)) / al
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        v = m
    _SM["x"] = x.copy()
    _SM["w"] = w
    return v

def smooth_g(x):
    if _SM.get("w") is None or _SM.get("x") is None or not np.array_equal(x, _SM["x"]):
        smooth_f(x)
    w = _SM["w"]
    n = _SM["n"]
    g = np.zeros(n)
    nz = np.nonzero(w > 1e-12)[0]
    for q in nz:
        ww = float(w[q])
        l = int(q - (n - 1))
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += ww * (1.0 - x[l:])
                g[l:] -= ww * x[:m]
        else:
            m = -l
            if m < n:
                g[m:] += ww * (1.0 - x[:n - m])
                g[:n - m] -= ww * x[m:]
    return g

def smooth_scon(x):
    return x.sum() - _SM["n"] / 2.0

def smooth_slsqp(h_ini, n, alphas, maxiter=120, deadline=None):
    h = project(h_ini, n)
    cur = h.copy()
    best = h.copy()
    bestc = eval_c5(best, n)
    for al in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _SM["n"] = n
        _SM["alpha"] = float(al)
        _SM["x"] = None
        _SM["w"] = None
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_f, cur, method="SLSQP", jac=smooth_g,
                           bounds=bnds,
                           constraints=[{"type": "eq", "fun": smooth_scon}],
                           options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            nh = project(res.x, n)
            nc = eval_c5(nh, n)
            if nc < eval_c5(cur, n):
                cur = nh
            if nc < bestc:
                best, bestc = nh.copy(), nc
        else:
            break
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dur = max(20.0, float(budget_s) - 2.0)
    deadline = t0 + dur

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = make_random_balanced(n0, seed)
    if n0 < 10:
        n0 = 250
        h0 = make_random_balanced(n0, seed)
    if n0 % 2:
        n0 -= 1
    if n0 <= 0:
        n0 = 250
        h0 = make_random_balanced(n0, seed)
    h0 = project(h0[:n0], n0)

    best = h0.copy()
    bestn = n0
    bestc = eval_c5(best, bestn)
    if not np.isfinite(bestc):
        best = make_random_balanced(bestn, seed)
        bestc = eval_c5(best, bestn)

    p1 = t0 + 0.34 * dur
    dx0 = 2.0 / bestn
    tgt0 = int(np.floor(bestc / dx0 + 1e-8))
    hb, mb = binary_local_search(best, bestn, p1, seed, 18, tgt0)
    if mb * dx0 < bestc:
        best, bestc = hb.copy(), mb * dx0

    p2 = t0 + 0.62 * dur
    p2a = min(p2, t0 + 0.50 * dur)
    if time.time() < p2a - 5:
        ha, ma = active_lp_polish(best, bestn, p2a, 10)
        if ma * (2.0 / bestn) < bestc:
            best, bestc = ha.copy(), ma * (2.0 / bestn)
    if time.time() < p2 - 5:
        ha, ma = active_lp_polish(hb, bestn, p2, 8)
        if ma * (2.0 / bestn) < bestc:
            best, bestc = ha.copy(), ma * (2.0 / bestn)

    p3 = t0 + 0.82 * dur
    if time.time() < p3 - 35:
        alphas = (10.0, 50.0, 200.0) if bestn <= 320 else (5.0, 20.0, 80.0)
        iters = 140 if bestn <= 320 else 75
        hs, cs = smooth_slsqp(best, bestn, alphas, iters, p3)
        if cs < bestc:
            best, bestc = hs.copy(), cs

    if time.time() < deadline - 170 and bestc > 0.38078:
        n_alt = 300 if bestn < 300 else (400 if bestn < 450 else 0)
        if n_alt > 0:
            ad = deadline - 110
            ha = project(resample_h(best, n_alt), n_alt)
            ca = eval_c5(ha, n_alt)
            tgt_alt = int(np.floor(bestc * (n_alt / 2.0) + 1e-8))
            ba, ma = binary_local_search(ha, n_alt, max(ad - 70, t0), seed + 17, 6, tgt_alt)
            cab = ma * (2.0 / n_alt)
            if cab < bestc:
                best, bestc, bestn = ba.copy(), cab, n_alt
            base_alt = best if bestn == n_alt else (ba if cab < ca else ha)
            if time.time() < ad - 8:
                da, daM = active_lp_polish(base_alt, n_alt, ad, 6)
                if daM * (2.0 / n_alt) < bestc:
                    best, bestc, bestn = da.copy(), daM * (2.0 / n_alt), n_alt

    finale = deadline - 5
    dx = 2.0 / bestn
    if time.time() < finale - 45:
        tgt = int(np.floor(bestc / dx + 1e-8))
        hf, mf = binary_local_search(best, bestn, finale - 25, seed + 97, 6, tgt)
        if mf * dx < bestc:
            best, bestc = hf.copy(), mf * dx
    if time.time() < finale:
        hf, mf = active_lp_polish(best, bestn, finale, 10)
        if mf * (2.0 / bestn) < bestc:
            best, bestc = hf.copy(), mf * (2.0 / bestn)

    best = project(best, bestn)
    bestc = eval_c5(best, bestn)
    return best, float(bestc), int(bestn)
