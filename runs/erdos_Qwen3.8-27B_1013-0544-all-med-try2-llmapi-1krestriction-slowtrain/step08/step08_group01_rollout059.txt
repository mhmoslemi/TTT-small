def fourier_start(n, rng):

    x = np.linspace(0, 2, n, endpoint=False)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - h[kk:])
            grad[kk:] += wm * (-h[:nv])
    return grad

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_run(h, n, deadline, max_iters=600):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 0.3), (200, 1.0), (1000, 5.0), (5000, 20.0),
               (20000, 100.0), (100000, 500.0), (500000, 2000.0),
               (2000000, 10000.0), (10000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        if deadline - time.time() < 5:
            break
        try:
            res = minimize(obj_fn, h, jac=grad_fn, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-19,
                                  'gtol': 1e-17, 'maxfun': max_iters * 60,
                                  'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_run(h, n, deadline, alpha=5000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-19})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgradient_polish(h, n, deadline, top_k=5):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.002
    for it in range(2000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        cmax = c.max()
        if cmax < best_c - 1e-15:
            best_c = cmax

        grad = np.zeros(n)

                nv = n - k

                kk = -k
                nv = n - kk
                grad[:nv] += dx * (1.0 - h[kk:])
                grad[kk:] += dx * (-h[:nv])

def golden_ratio_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 12):
        amp = rng.uniform(0.08, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)
