        return w[0]*(1.0-2.0*h[0])

import time
import numpy as np
from scipy.optimize import minimize


def fft_len(n):
    return 1 << max(0, (2 * n - 2).bit_length())


def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(1.0 - h[idx], diff)
                if add <= 0.0:
                    continue
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(h[idx], -diff)
                if sub <= 0.0:
                    continue
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return h


def compute_c_counts_fft(h, n):
    N = fft_len(n)
    H = np.fft.rfft(h, N)
    A = np.fft.rifft(H * np.conj(H), N)
    A = A[:n]
    P = np.empty(n + 1)
    P[0] = 0.0
    P[1:] = np.cumsum(h)
    total = P[n]
    c = np.empty(2 * n - 1)
    c[n - 1] = total - A[0]
    if n > 1:
        c[:n - 1] = P[n - 1:0:-1] - A[1:]
        c[n:] = total - P[1:n] - A[1:]
    return np.maximum(c, 0.0)


def compute_c5_fft(h, n):
    c = compute_c_counts_fft(h, n)
    return float(np.max(c) * (2.0 / n))


def compute_c5_np(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def grad_from_weights(h, n, w):
    g = np.zeros(n)
    if n == 1:
        return w[0] * (1.0 - 2.0 * h[0])
    neg = w[:n - 1][::-1]
    pos = w[n:]
    prefL = np.cumsum(neg)
    cpre = np.zeros(n)
    cpre[:n - 1] = prefL[::-1]
    prefR = np.cumsum(pos)
    csuf = np.zeros(n)
    csuf[1:] = prefR
    g = cpre + csuf
    D = np.zeros(n)
    D[1:] = neg + pos
    N = fft_len(n)
    H = np.fft.rfft(h, N)
    Dhat = np.fft.rfft(D, N)
    t_suf = np.fft.rifft(H * Dhat, N)[:n]
    Dr_hat = np.fft.rfft(D[::-1], N)
    C2 = np.fft.rifft(H * Dr_hat, N)
    t_pref = C2[n - 1:2 * n - 1]
    g -= t_suf + t_pref
    w0 = w[n - 1]
    if w0 != 0.0:
        g += w0 * (1.0 - 2.0 * h)
    return g


def smooth_objective(x, n, alpha, mu):
    c = compute_c_counts_fft(x, n)
    m = float(c.max())
    if alpha <= 1e-12:
        val = m
    else:
        z = alpha * (c - m)
        zm = float(z.max())
        val = m + (zm + np.log(np.sum(np.exp(z - zm)))) / alpha
    if mu:
        s = x.sum() - 0.5 * n
        val += mu * s * s
    return float(val)


def smooth_gradient(x, n, alpha, mu):
    c = compute_c_counts_fft(x, n)
    m = float(c.max())
    if alpha <= 1e-12:
        w = np.zeros_like(c)
        mask = c >= m - 1e-12
        w[mask] = 1.0
        w /= w.sum()
    else:
        z = alpha * (c - m)
        zm = float(z.max())
        w = np.exp(z - zm)
        w /= w.sum()
    g = grad_from_weights(x, n, w)
    if mu:
        g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g


def slsqp_eq(x, n):
    return float(x.sum() - 0.5 * n)


def slsqp_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, n, deadline, alpha, maxiter):
    best = compute_c5_fft(h, n)
    if time.time() > deadline - 20:
        return h, best
    bounds = [(0.0, 1.0)] * n
    cons = [
        {"type": "eq", "fun": slsqp_eq, "jac": slsqp_jac, "args": (n,)}
    ]
    try:
        res = minimize(
            smooth_objective,
            h,
            jac=smooth_gradient,
            args=(n, alpha, 0.0),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-15},
        )
        hn = project_feasible(res.x, n)
        cn = compute_c5_fft(hn, n)
        if cn < best:
            return hn, cn
    except Exception:
        pass
    return h, best


def lbfgs_optimize(h, n, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5_fft(best_h, n)
    bounds = [(0.0, 1.0)] * n
    stages = [
        (10.0, 0.05, 140),
        (40.0, 0.05, 140),
        (150.0, 0.10, 140),
        (600.0, 0.10, 120),
        (2500.0, 0.50, 100),
        (9000.0, 1.00, 80),
        (30000.0, 2.00, 60),
    ]
    for alpha, mu, mmax in stages:
        if time.time() > deadline:
            break
        mi = max(20, min(mmax, int(120.0 * 500.0 / max(1, n)) + 20))
        try:
            res = minimize(
                smooth_objective,
                best_h,
                jac=smooth_gradient,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": mi, "ftol": 1e-17, "gtol": 1e-15, "maxfun": mi * 28},
            )
            hn = project_feasible(res.x, n)
            cn = compute_c5_fft(hn, n)
            if cn < best_c - 1e-15:
                best_h, best_c = hn, cn
        except Exception:
            pass
    if n <= 512 and time.time() < deadline - 30:
        h2, c2 = slsqp_optimize(best_h, n, deadline, 2500.0, 55)
        if c2 < best_c - 1e-15:
            best_h, best_c = h2, c2
    return project_feasible(best_h, n), best_c


def build_row(i, h, n):
    row = np.zeros(2 * n - 1)
    ctr = n - 1
    row[ctr] = 1.0 - 2.0 * h[i]
    m1 = n - 1 - i
    if m1 > 0:
        d = np.arange(1, m1 + 1)
        hp = h[i + 1:i + m1 + 1]
        row[ctr - d] += 1.0 - hp
        row[ctr + d] += -hp
    m2 = i
    if m2 > 0:
        d = np.arange(1, m2 + 1)
        hb = h[i - m2:i]
        row[ctr - d] += -hb
        row[ctr + d] += 1.0 - hb
    return row

    best_c = compute_c5_fft(h, n)

def binary_round(h, n):
    if n % 2:
        return h
    half = n // 2
    out = np.zeros(n)
    idx = np.partition(h, n - half)[n - half:]
    out[idx] = 1.0
    return out


def interpolate_to(h, n, n2):
    x0 = np.linspace(0.0, 2.0, n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x1, x0, h)
    return project_feasible(h2, n2)


def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)


def golden_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    phi = 0.5 * (1.0 + np.sqrt(5.0))
    pos = np.floor(np.arange(half) * phi).astype(int) % n
    h[pos] = 1.0
    if h.sum() < half:
        rem = np.flatnonzero(h == 0.0)
        need = int(half - h.sum())
        if rem.size and need:
            h[rng.choice(rem, size=need, replace=False)] = 1.0
    return project_feasible(h, n)


def wave_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 * np.ones(n)
    for f in (1, 2, 3, 5):
        amp = rng.uniform(0.06, 0.24)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * f * x + ph)
    return project_feasible(h, n)


def cluster_start(n, rng):
    h = np.zeros(n)
    pos = 0
    bit = int(rng.randint(0, 2))
    while pos < n:
        l = int(rng.randint(1, max(2, n // 16) + 1))
        l = min(l, n - pos)
        h[pos:pos + l] = bit
        pos += l
        bit = 1 - bit
    h = (h > 0.5).astype(float)
    half = n // 2
    s = int(h.sum())
    if s < half and n % 2 == 0:
        rem = np.flatnonzero(h == 0.0)
        if rem.size:
            h[rng.choice(rem, size=half - s, replace=False)] = 1.0
    elif s > half:
        rem = np.flatnonzero(h == 1.0)
        if rem.size:
            h[rng.choice(rem, size=s - half, replace=False)] = 0.0
    return project_feasible(h, n)

            h2, c2 = lbfgs_optimize(h, n, tlb)
            if c2 < best - 1e-15:
                h, best = h2, c2

    if n % 2 == 0:
        hb = binary_round(h, n)
        cb = compute_c5_fft(hb, n)
        if cb < best - 1e-15:
            h, best = hb, cb

    if local_alloc > 8:
        tl = min(time.time() + local_alloc, deadline - 5)
        if tl > time.time():
            h2, c2 = local_search(h, n, tl, alpha_ls, K)
            if c2 < best - 1e-15:
                h, best = h2, c2

    if n % 2 == 0 and time.time() < deadline - 2:
        hb = binary_round(h, n)
        hb = project_feasible(hb, n)
        cb = compute_c5_fft(hb, n)
        if cb < best - 1e-15:
            h, best = hb, cb

    h = project_feasible(h, n)
    best = compute_c5_fft(h, n)
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 512
        h0 = random_binary_start(n0, rng)

    if n0 < 60:
        n0 = 512
        h0 = random_binary_start(n0, rng)

    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = compute_c5_np(best_h, best_n)

    if time.time() < deadline - 70:
        h, _ = refine_candidate(best_h, best_n, 120.0, 230.0, 500.0, 24, deadline)
        h = project_feasible(h, best_n)
        c = compute_c5_np(h, best_n)
        if c < best_c:
            best_h, best_c, best_n = h, c, best_n

    sized_jobs = [(512, 65.0, 140.0, 600.0, 22),
                  (768, 50.0, 105.0, 800.0, 20),
                  (1000, 35.0, 75.0, 1000.0, 18)]
    for nt, lb, loc, alp, ks in sized_jobs:
        if time.time() > deadline - 55:
            break
        if best_n == nt and best_h.size == nt:
            hstart = best_h.copy()
        else:
            hstart = interpolate_to(best_h, best_n, nt)
        h, _ = refine_candidate(hstart, nt, lb, loc, alp, ks, deadline)
        h = project_feasible(h, nt)
        c = compute_c5_np(h, nt)
        if c < best_c:
            best_h, best_c, best_n = h, c, nt

    nd = 512
    starters = (random_binary_start, cluster_start, wave_start)
    for starter in starters:
        if time.time() > deadline - 70:
            break
        hstart = starter(nd, rng)
        h, _ = refine_candidate(hstart, nd, 18.0, 45.0, 300.0, 24, deadline)
        h = project_feasible(h, nd)
        c = compute_c5_np(h, nd)
        if c < best_c:
            best_h, best_c, best_n = h, c, nd

    if time.time() < deadline - 18:
        kf = 28 if best_n <= 700 else 20
        h2, _ = local_search(best_h, best_n, deadline - 2.0, 4000.0, kf)
        h2 = project_feasible(h2, best_n)
        c2 = compute_c5_np(h2, best_n)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if best_n % 2 == 0 and time.time() < deadline - 1.0:
        hb = binary_round(best_h, best_n)
        hb = project_feasible(hb, best_n)
        cb = compute_c5_np(hb, best_n)
        if cb < best_c:
            best_h, best_c = hb, cb

    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5_np(best_h, best_n)
    return best_h, final_c, best_n
