import time
import numpy as np
from scipy.optimize import minimize

_ST = {'n': 1, 'gamma': 10.0, 'target': 0.5, 'lam': 0.0}


def corr_custom(h):
    n = len(h)
    L = 2 * n - 1
    c = np.zeros(L)
    b = 1.0 - h
    for d in range(n):
        c[d + n - 1] = float(np.dot(h[:n - d], b[d:]))
    for d in range(1, n):
        c[n - 1 - d] = float(np.dot(h[d:], b[:n - d]))
    return c


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def interp_to_n(h, n0, n):
    h = np.asarray(h, dtype=float).ravel()
    if n0 == n:
        return h.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = np.linspace(0.0, 2.0, n0)
    x1 = np.linspace(0.0, 2.0, n)
    return np.interp(x1, x0, h)


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).ravel()
    if y.size != n:
        tmp = np.full(n, 0.5)
        m = min(n, y.size)
        tmp[:m] = y[:m]
        y = tmp
    y = np.where(np.isfinite(y), y, 0.5)
    target = 0.5 * n
    p = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(p)) - target) < 1e-13:
        return p
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    p = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        diff = target - float(np.sum(p))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - p
            s = float(np.sum(room))
            if s < 1e-15:
                break
            p = np.clip(p + (diff / s) * room, 0.0, 1.0)
        else:
            s = float(np.sum(p))
            if s < 1e-15:
                break
            p = np.clip(p + (diff / s) * p, 0.0, 1.0)
    return p


def grad_c_all(h, p):
    n = len(h)
    G = np.zeros(n)
    for idx, w in enumerate(p):
        if w <= 1e-16:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            G[:m] += w * (1.0 - h[lag:lag + m])
            G[lag:lag + m] -= w * h[:m]
        else:
            l = -lag
            m = n - l
            G[l:] += w * (1.0 - h[:m])
            G[:m] -= w * h[l:]
    return G


def _smooth_obj(x):
    n = _ST['n']
    c = corr_custom(x)
    g = _ST['gamma']
    mx = float(np.max(c))
    sh = g * (c - mx)
    sh -= float(np.max(sh))
    s = float(np.sum(np.exp(sh)))
    F = mx + float(np.log(s)) / g
    if _ST['lam'] > 0.0:
        F += _ST['lam'] * (float(np.sum(x)) - _ST['target']) ** 2
    return float(F)


def _smooth_jac(x):
    n = _ST['n']
    c = corr_custom(x)
    g = _ST['gamma']
    sh = g * (c - float(np.max(c)))
    sh -= float(np.max(sh))
    p = np.exp(sh)
    s = float(np.sum(p))
    p /= s
    G = grad_c_all(x, p)
    if _ST['lam'] > 0.0:
        G += 2.0 * _ST['lam'] * (float(np.sum(x)) - _ST['target'])
    return G


def _vol_eq(x):
    return float(np.sum(x) - _ST['target'])


def slsqp_smooth(h, n, gamma, maxiter):
    _ST['n'] = n
    _ST['gamma'] = float(gamma)
    _ST['target'] = 0.5 * n
    _ST['lam'] = 0.0
    h0 = project_box_sum(h, n)
    bounds = [(0.0, 1.0)] * n
    maxiter = max(5, int(maxiter))
    try:
        res = minimize(
            _smooth_obj, h0, method='SLSQP', bounds=bounds,
            jac=_smooth_jac,
            constraints=[{'type': 'eq', 'fun': _vol_eq}],
            options={'maxiter': maxiter, 'ftol': 1e-11, 'disp': 0}
        )
        if hasattr(res, 'x') and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h0


def lbfgs_smooth(h, n, gamma, maxiter, lam):
    _ST['n'] = n
    _ST['gamma'] = float(gamma)
    _ST['target'] = 0.5 * n
    _ST['lam'] = float(lam)
    h0 = project_box_sum(h, n)
    bounds = [(0.0, 1.0)] * n
    maxiter = max(5, int(maxiter))
    try:
        res = minimize(
            _smooth_obj, h0, method='L-BFGS-B', bounds=bounds,
            jac=_smooth_jac,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': 0}
        )
        if hasattr(res, 'x') and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h0


def build_rows(h):
    n = len(h)
    L = 2 * n - 1
    lag = np.arange(L) - (n - 1)
    b = 1.0 - h
    rows = np.zeros((n, L), dtype=np.int8)
    for x in range(n):
        jd = x + lag
        m1 = (jd >= 0) & (jd < n)
        rows[x, m1] = b[jd[m1]]
        idd = x - lag
        m2 = (idd >= 0) & (idd < n)
        rows[x, m2] -= h[idd[m2]]
    return rows


def swap_new_c(c, rows, i, j, n):
    newc = c + rows[i] - rows[j]
    base = n - 1
    newc[base] -= 2.0
    newc[base + j - i] += 1.0
    newc[base + i - j] += 1.0
    return newc


def find_swap_with_target(h, n, c, rows, target, D):
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h > 0.5)[0]
    if zeros.size == 0 or ones.size == 0:
        return None
    D = np.asarray(D, dtype=int)
    if D.size == 0:
        return None
    RD = rows[:, D]
    RD_on = RD[ones]
    lim = target - c[D]
    base = n - 1
    L = 2 * n - 1
    colmap = np.full(L, -1, dtype=int)
    colmap[D] = np.arange(D.size)
    best_margin = -1e100
    best_pair = None
    for i in zeros:
        margin = lim[None, :] - RD[i][None, :] + RD_on
        c1 = colmap[base + (ones - i)]
        v1 = c1 >= 0
        if v1.any():
            margin[v1, c1[v1]] -= 1.0
        c2 = colmap[base + (i - ones)]
        v2 = c2 >= 0
        if v2.any():
            margin[v2, c2[v2]] -= 1.0
        minm = margin.min(axis=1)
        jj = int(np.argmax(minm))
        if float(minm[jj]) > best_margin:
            best_margin = float(minm[jj])
            best_pair = (int(i), int(ones[jj]))
    if best_pair is not None and best_margin >= -1e-8:
        return best_pair
    return None


def make_binary(h, n, rng=None):
    if h is None:
        arr = np.full(n, rng.rand(n) if rng is not None else 0.5)
    else:
        arr = np.asarray(h, dtype=float).ravel()
        if arr.size != n:
            arr = interp_to_n(arr, arr.size, n)
    vals = np.argsort(arr)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[vals[:n // 2]] = 1.0
    return hb


def positive_split(total, parts, rng):
    total = int(total)
    parts = int(parts)
    if parts <= 0:
        return []
    if total < parts:
        return None
    if parts == 1:
        return [total]
    if total - 1 < parts - 1:
        return None
    cuts = np.sort(rng.choice(np.arange(1, total), size=parts - 1, replace=False))
    vals = np.diff(np.concatenate(([0], cuts, [total])))
    return vals.astype(int).tolist()


def block_binary(n, nb, start, rng):
    target = n // 2
    candidates = [int(nb), max(2, int(nb) // 2), max(2, int(nb) // 4), 2]
    for B in candidates:
        if B < 2 or B > n:
            continue
        if start == 1:
            ob = (B + 1) // 2
            zb = B // 2
        else:
            ob = B // 2
            zb = (B + 1) // 2
        if ob == 0 or zb == 0:
            continue
        if target < ob or (n - target) < zb:
            continue
        olen = positive_split(target, ob, rng)
        zlen = positive_split(n - target, zb, rng)
        if olen is None or zlen is None:
            continue
        out = []
        oi = 0
        zi = 0
        for b in range(B):
            val = start if (b % 2 == 0) else (1 - start)
            if val == 1:
                l = int(olen[oi]); oi += 1
            else:
                l = int(zlen[zi]); zi += 1
            out.append(np.full(l, float(val)))
        hb = np.concatenate(out)
        if hb.size == n and abs(float(np.sum(hb)) - target) < 1e-9:
            return hb
    idx = rng.choice(n, size=target, replace=False)
    hb = np.zeros(n, dtype=float)
    hb[idx] = 1.0
    return hb

def continuous_polish(h, n, limit, seed):
    t0 = time.time()
    deadline = t0 + float(limit)
    h = project_box_sum(h, n)
    best = h.copy()
    best_c = compute_c5(best, n)
    rng = np.random.RandomState(seed)
    noise_levels = [0.0, 0.035, 0.09, 0.18]

    for nz in noise_levels:
        if time.time() > deadline - 12.0:
            break
        if nz == 0.0:
            cur = best.copy()
        else:
            cur = project_box_sum(best + nz * rng.randn(n), n)

        for gamma in [8.0, 30.0, 80.0]:
            rem = deadline - time.time()
            if rem < 8.0:
                break
            mi = 30 if rem < 70 else 90
            hp = lbfgs_smooth(cur, n, gamma, mi, 1.0e4)
            cp = compute_c5(hp, n)
            if cp < best_c - 1e-12:
                best = hp
                best_c = cp
                cur = hp

        for gamma in [40.0, 90.0, 180.0]:
            rem = deadline - time.time()
            if rem < 8.0:
                break
            mi = 25 if rem < 70 else 70
            hp = slsqp_smooth(cur, n, gamma, mi)
            cp = compute_c5(hp, n)
            if cp < best_c - 1e-12:
                best = hp
                best_c = cp
                cur = hp

    best = project_box_sum(best, n)
    return best, compute_c5(best, n)

            starts.append(make_binary(bin_h, best_n, rng))
        for sidx, hb in enumerate(starts):
            if time.time() > multi_deadline - 35.0:
                break
            alloc = min(multi_deadline - time.time() - 15.0, 45.0)
            if alloc < 12.0:
                break
            hb2, _ = binary_optimize(hb, best_n, alloc, seed + 111 + sidx)
            cb = compute_c5(hb2, best_n)
            if cb < bin_c:
                bin_h, bin_c, bin_n = hb2, cb, best_n
            if cb < best_c:
                best_h, best_c, best_n = hb2, cb, best_n
            if cb < best_c and time.time() < multi_deadline - 30.0:
                alloc2 = min(multi_deadline - time.time() - 12.0, 40.0)
                if alloc2 > 20.0:
                    hp, cp = continuous_polish(hb2, best_n, alloc2, seed + 151 + sidx)
                    if cp < best_c:
                        best_h, best_c, best_n = hp, cp, best_n

    if best_h is not None and time.time() < multi_deadline:
        for n in (150, 200, 250, 300):
            if n == best_n:
                continue
            if time.time() > multi_deadline - 35.0:
                break
            if parent is not None:
                hstart = project_box_sum(interp_to_n(parent, parent_n, n), n)
            else:
                hstart = project_box_sum(interp_to_n(best_h, best_n, n), n)
            hb = make_binary(hstart, n, rng)
            alloc_b = min(multi_deadline - time.time() - 25.0, 18.0)
            if alloc_b < 8.0:
                break
            hb, _ = binary_optimize(hb, n, alloc_b, seed + 211 + n)
            cb = compute_c5(hb, n)
            if cb < bin_c:
                bin_h, bin_c, bin_n = hb, cb, n
            if cb < best_c:
                best_h, best_c, best_n = hb, cb, n
            alloc_c = min(multi_deadline - time.time() - 10.0, 22.0)
            if alloc_c > 10.0:
                hp, cp = continuous_polish(hstart, n, alloc_c, seed + 311 + n)
                if cp < best_c:
                    best_h, best_c, best_n = hp, cp, n

    if best_h is not None and time.time() < deadline - 12.0:
        alloc = min(deadline - time.time() - 5.0, 35.0)
        hp, cp = continuous_polish(best_h, best_n, alloc, seed + 991)
        if cp < best_c:
            best_h, best_c, best_n = hp, cp, best_n

    if best_h is None:
        n = 200
        best_h = make_binary(rng.rand(n), n, rng)
        best_n = n
        best_c = compute_c5(best_h, n)

    best_h = project_box_sum(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
