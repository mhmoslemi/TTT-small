def active_set_minimax(h, n, deadline, n_lags=5):
    dx = 2.0 / n
    h = h.copy()
    t0 = time.time()
    
    while time.time() < deadline:
        corr = np.correlate(h, 1-h, 'full') * dx
        m = corr.max()
        # Find top n_lags positions
        top_idx = np.argsort(corr)[::-1][:n_lags]
        
        # Compute gradient for top lags only
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n-1)  # actual lag
            for i in range(n):
                # dc/dh[i] for this lag
                if i + lag >= 0 and i + lag < n:
                    grad[i] += (1 - h[i + lag])
                if i - lag >= 0 and i - lag < n:
                    grad[i] -= h[i - lag]
        
        grad *= dx
        grad /= n_lags  # average
        
        # Projected gradient step
        best_step = 0.01
        best_val = m
        for _ in range(20):
            h_new = project_feasible(h - best_step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_val:
                best_val = c_new
                best_h_new = h_new.copy()
            else:
                break
            best_step *= 0.7
        
        if best_step < 1e-10:
            break
        h = best_h_new
    return h, compute_c5(h, n)

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def active_set_grad(h, n, n_active=5):
    """Compute gradient of max correlation w.r.t. h, using active set."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, 'full')
    # Find active set
    top = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for idx in top:
        k = idx - (n - 1)
        # dc_k/dh[i] = (1-h[i+k])*[valid] - h[i-k]*[valid]

        m2 = (ik2 >= 0) & (ik2 < n)

def coord_descent(h, n, deadline, m=10):
    dx = 2.0 / n
    h = h.copy()
    for sweep in range(100000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1-h, 'full') * dx
        cur_max = corr.max()
        top = np.argsort(corr)[::-1][:m]
        
        improved = False
        for i in range(n):
            # Compute q_i(k) for active lags
            q = np.zeros(m)
            for t in range(m):
                k = top[t] - (n-1)
                q[t] = 0.0
                if 0 <= i+k < n: q[t] += (1-h[i+k])
                if 0 <= i-k < n: q[t] -= h[i-k]
            
            # Find v in [0,1] that minimizes max_t (corr[top[t]] + (v-h[i])*q[t])
            # This is min_v max_t (a_t + v*b_t) where a_t = corr[top[t]] - h[i]*q[t], b_t = q[t]
            a = corr[top] - h[i] * q
            b = q
            
            # The optimal v is where the top 2 lines cross, or at a boundary
            # Find all pairwise intersections v = (a_j - a_i) / (b_i - b_j) for b_i != b_j
            candidates = [0.0, 1.0]

                for j2 in range(i2+1, m):
                    if abs(b[i2] - b[j2]) > 1e-12:
                        v = (a[j2] - a[i2]) / (b[i2] - b[j2])
                        if 0 <= v <= 1:
                            candidates.append(v)

            best_v = h[i]

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
