        delta[s + n-1] -= h[i - s]
    lo3 = max(-j, -(n-1))
    hi3 = min(n-1-j, n-1)
    if lo3 <= hi3:
        s = np.arange(lo3, hi3+1)
        delta[s + n-1] -= (1.0 - h[j + s])
    lo4 = max(j-(n-1), -(n-1))
    hi4 = min(j, n-1)
    if lo4 <= hi4:
        s = np.arange(lo4, hi4+1)
        delta[s + n-1] += h[j - s]
    return delta

def binary_local_search(h_init, n, deadline, anneal_prob=0.03, anneal_tol=2.0, K=40):
    dx = 2.0 / n
    h = make_binary(h_init, n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_c = cur_max * dx
    rng = np.random.RandomState(777)
    n_ones = n // 2
    
    for it in range(200000):
        if time.time() > deadline:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        max_pos = np.argmax(corr)
        s_star = max_pos - (n - 1)
        
        # Vectorized fi computation
        i_arr = zeros_idx
        i_p = i_arr + s_star
        i_m = i_arr - s_star
        vp = (i_p >= 0) & (i_p < n)
        vm = (i_m >= 0) & (i_m < n)
        fi = np.zeros(len(i_arr))
        if np.any(vp):
            fi[vp] += (1.0 - h[i_p[vp]])
        if np.any(vm):
            fi[vm] += (-h[i_m[vm]])
        
        j_arr = ones_idx
        j_p = j_arr + s_star
        j_m = j_arr - s_star
        vp2 = (j_p >= 0) & (j_p < n)
        vm2 = (j_m >= 0) & (j_m < n)
        gj = np.zeros(len(j_arr))
        if np.any(vp2):
            gj[vp2] += (1.0 - h[j_p[vp2]])
        if np.any(vm2):
            gj[vm2] += (-h[j_m[vm2]])
        
        improved = False
        KC = min(K, len(zeros_idx))
        KJ = min(K, len(ones_idx))
        i_cands = np.argsort(fi)[:KC]
        j_cands = np.argsort(gj)[::-1][:KJ]
        
        for ia in i_cands:
            i = zeros_idx[ia]
            for ja in j_cands:
                j = ones_idx[ja]
                d = swap_delta(h, i, j, n)
                nm = (corr + d).max()
                if nm < cur_max - 1e-12:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += d
                    cur_max = nm
                    improved = True
                    if cur_max * dx < best_c:
                        best_c = cur_max * dx
                        best_h = h.copy()
                    break
            if improved:
                break
        
        if not improved:
            if rng.random() < anneal_prob:
                i = zeros_idx[rng.randint(len(zeros_idx))]
                j = ones_idx[rng.randint(len(ones_idx))]
                d = swap_delta(h, i, j, n)
                nm = (corr + d).max()
                if nm < cur_max + anneal_tol:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += d
                    cur_max = nm
            else:
                break
    
    return best_h, best_c

def slsqp_refine(h, n, deadline, alpha=100000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except:
        n = 250
        h0 = np.full(n, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Binary local search (50% of time)
    p1 = t0 + 0.45 * (budget_s - 8)
    hb, cb = binary_local_search(h0, n, p1, anneal_prob=0.05, anneal_tol=3.0, K=50)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: SLSQP continuous refinement
    p2 = t0 + 0.55 * (budget_s - 8)
    if time.time() < p2:
        hp = project_feasible(best_h + 0.04 * rng.randn(n), n)
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p2 - 5:
                break
            hp = slsqp_refine(hp, n, p2 - 3, alpha=alpha)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    
    # Phase 3: Binary from continuous
    p3 = t0 + 0.72 * (budget_s - 8)
    if time.time() < p3:
        hb, cb = binary_local_search(best_h, n, p3, anneal_prob=0.04, anneal_tol=2.5, K=60)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 4: SLSQP + binary polish
    p4 = t0 + 0.85 * (budget_s - 8)
    if time.time() < p4:
        hp = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [500000, 2000000]:
            if time.time() > p4 - 10:
                break
            hp = slsqp_refine(hp, n, p4 - 8, alpha=alpha)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    
    # Phase 5: Extended binary
    if time.time() < dl - 5:
        hb, cb = binary_local_search(best_h, n, dl - 3, anneal_prob=0.02, anneal_tol=1.5, K=80)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Final: SLSQP if time
    if time.time() < dl - 2:
        hp = slsqp_refine(project_feasible(best_h, n), n, dl - 0.5, alpha=5000000)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
