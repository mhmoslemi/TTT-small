def lag_grad_from_weights(w,h,n,threshold=0.0):
    grad=np.zeros(n)
    shift=n-1

def project_feasible(h,n):
    target=n/2.0
    h=np.clip(np.asarray(h,dtype=float).reshape(-1),0.0,1.0)
    for _ in range(500):
        diff=target-h.sum()
        if abs(diff)<1e-14: break
        if diff>0:
            room=1.0-h; tot=room.sum()
            if tot<1e-14: break
            h += (diff/tot)*room
        else:
            tot=h.sum()
            if tot<1e-14: break
            h += (diff/tot)*h
    h=np.clip(h,0,1)
    diff=target-h.sum()
    if diff>0:
       idx=np.flatnonzero(h<1.0)

          h[idx]+=diff/idx.size; h=np.clip(h,0,1)
    else:
       idx=np.flatnonzero(h>0.0)
       if idx.size:
          h[idx]+=diff/idx.size; h=np.clip(h,0,1)
    return h

def compute_S(h,n):
    S=np.empty(2*n-1)
    c=1.0-h
    sh=n-1
    S[sh]=float(np.dot(h,c))
    for k in range(1,n):
       m=n-k
       S[sh+k]=float(np.dot(h[:m],c[k:]))
       S[sh-k]=float(np.dot(h[k:],c[:m]))
    return S

def lag_grad_from_weights(w,h,n,threshold=1e-12):
    g=np.zeros(n); sh=n-1
    for m in range(len(w)):
       wm=w[m]
       if wm <= threshold: continue
       l=m-sh
       if l>=0:
          nv=n-l
          if nv>0:
             g[:nv] += wm*(1.0-h[l:])
             g[l:] -= wm*h[:nv]
       else:
          r=-l; nv=n-r
          if nv>0:
             g[r:] += wm*(1.0-h[:nv])
             g[:nv] -= wm*h[r:]
    return g

def random_fourier_h(n,rng):
    x=(np.arange(n)+0.5)*2.0/n
    u=np.zeros(n)
    for m in range(1,9):
        a=rng.normal(0.0,1.0/(0.5*m))
        b=rng.normal(0.0,1.0/(0.5*m))
        ph=rng.uniform(0.0,2.0*np.pi)

    if sc>1e-12:

import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.clip(np.asarray(h, dtype=float).reshape(-1), 0.0, 1.0)
    for _ in range(800):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = np.flatnonzero(h < 1.0 - 1e-15)
            if idx.size == 0:
                break
            h[idx] += diff / float(idx.size)
        else:
            idx = np.flatnonzero(h > 1e-15)
            if idx.size == 0:
                break
            h[idx] += diff / float(idx.size)
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_S(h, n):
    L = 2 * n - 1
    sh = n - 1
    S = np.empty(L, dtype=float)
    c = 1.0 - h
    S[sh] = float(np.dot(h, c))
    for k in range(1, n):
        m = n - k
        S[sh + k] = float(np.dot(h[:m], c[k:]))
        S[sh - k] = float(np.dot(h[k:], c[:m]))
    return S


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def lag_grad_from_weights(w, h, n, threshold=1e-12):
    g = np.zeros(n, dtype=float)
    sh = n - 1
    for m in range(len(w)):
        wm = w[m]
        if wm <= threshold:
            continue
        l = m - sh
        if l >= 0:
            nv = n - l
            if nv > 0:
                g[:nv] += wm * (1.0 - h[l:])
                g[l:] -= wm * h[:nv]
        else:
            r = -l
            nv = n - r
            if nv > 0:
                g[r:] += wm * (1.0 - h[:nv])
                g[:nv] -= wm * h[r:]
    return g


def smooth_obj(x, n, alpha, mu):
    S = compute_S(x, n)
    c = S * (2.0 / n)
    m = float(c.max())
    z = alpha * (c - m)
    M = float(z.max())
    se = np.exp(z - M)
    val = m + (M + np.log(se.sum())) / alpha
    if mu != 0.0:
        s = x.sum() - 0.5 * n
        val += mu * s * s
    return float(val)


def smooth_grad(x, n, alpha, mu):
    S = compute_S(x, n)
    c = S * (2.0 / n)
    m = float(c.max())
    z = alpha * (c - m)
    M = float(z.max())
    se = np.exp(z - M)
    w = se / se.sum()
    g = lag_grad_from_weights(w, x, n, 1e-12)
    if mu != 0.0:
        s = x.sum() - 0.5 * n
        g += 2.0 * mu * s
    return g


def lbfgs_refine(h, n, deadline, seed=42):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (20.0, 0.1),
        (100.0, 1.0),
        (500.0, 5.0),
        (2000.0, 20.0),
        (10000.0, 100.0),
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 3.0:
            break
        try:
            res = minimize(
                smooth_obj,
                h,
                jac=smooth_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 90, "ftol": 1e-16, "gtol": 1e-12,
                         "maxfun": 450, "maxls": 50},
            )
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-15:
                best_c = c2
                h = h2
                best_h = h2.copy()
        except Exception:
            pass
    return best_h, best_c


def active_gradient(h, S, n, raw_tol=1e-7):
    m = float(S.max())
    act = np.flatnonzero(S >= m - raw_tol)
    w = np.zeros(len(S), dtype=float)
    if act.size == 0:
        w[-1] = 1.0
    else:
        w[act] = 1.0 / float(act.size)
    return lag_grad_from_weights(w, h, n, 0.0)


def try_move(h, S, i, j, d, sh, n, best_raw):
    if i == j:
        return False, best_raw
    hi = h[i]
    hj = h[j]
    d = min(d, 1.0 - hi, hj)
    if d <= 1e-15:
        return False, best_raw
    si = sh - i
    sj = sh - j
    S[si:si + n] += d
    S[sj:sj + n] -= d
    selfd = 2.0 * d * (hj - hi) - 2.0 * d * d
    paird = d * (hi - hj + d)
    S[sh] += selfd
    S[sh + j - i] += paird
    S[sh + i - j] += paird
    nm = float(S.max())
    if nm < best_raw - 1e-14:
        h[i] = hi + d
        h[j] = hj - d
        return True, nm
    S[sh + i - j] -= paird
    S[sh + j - i] -= paird
    S[sh] -= selfd
    S[sj:sj + n] += d
    S[si:si + n] -= d
    return False, nm


def pair_pass(h, S, n, step, sh, best_raw, rng, deadline, g):
    order = rng.permutation(n)
    rn = range(n)
    improved = False
    cands = 0
    use_g = g is not None and step <= 1e-4
    tm = try_move
    for ipos in order:
        if time.time() > deadline:
            break
        i = int(ipos)
        hi = h[i]
        if hi >= 1.0:
            continue
        gi = g[i] if use_g else 0.0
        for j in rn:
            if j == i:
                continue
            hj = h[j]
            if hj <= 0.0:
                continue
            if use_g and g[j] < gi - 1e-12:
                continue
            ok, nm = tm(h, S, i, j, step, sh, n, best_raw)
            cands += 1
            if ok:
                best_raw = nm
                improved = True
                hi = h[i]
                if hi >= 1.0:
                    break
            if (cands & 4095) == 0:
                S = compute_S(h, n)
                best_raw = float(S.max())
                if time.time() > deadline:
                    return best_raw, improved, S
    return best_raw, improved, S


def pair_search_fast(h, n, deadline, seed=42, start_step=0.02, min_step=1e-6):
    h = project_feasible(h, n)
    S = compute_S(h, n)
    sh = n - 1
    dx = 2.0 / n
    best_raw = float(S.max())
    best_c = best_raw * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    step = start_step
    while time.time() < deadline and step >= min_step:
        old = best_raw
        maxp = 8 if step > 1e-3 else 15
        for _ in range(maxp):
            if time.time() > deadline:
                break
            if step <= 1e-4:
                g = active_gradient(h, S, n, 1e-7)
            else:
                g = None
            best_raw, imp, S = pair_pass(h, S, n, step, sh, best_raw, rng, deadline, g)
            cur_c = best_raw * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            if not imp:
                break
        S = compute_S(h, n)
        best_raw = float(S.max())
        cur_c = best_raw * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        step *= 0.5
    return best_h, best_c


def guided_pair(h, n, deadline, seed=42, base_step=0.05, K=60):
    h = project_feasible(h, n)
    S = compute_S(h, n)
    sh = n - 1
    dx = 2.0 / n
    best_raw = float(S.max())
    best_c = best_raw * dx
    best_h = h.copy()
    scales = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    while time.time() < deadline - 2.0:
        S = compute_S(h, n)
        best_raw = float(S.max())
        if best_raw * dx < best_c:
            best_c = best_raw * dx
            best_h = h.copy()
        g = active_gradient(h, S, n, 1e-6)
        i_cand = np.unique(np.concatenate((np.argsort(g)[:K], np.argsort(-h)[:K])))
        j_cand = np.unique(np.concatenate((np.argsort(-g)[:K], np.argsort(h)[:K])))
        pairs = []
        for ii in i_cand:
            i = int(ii)
            hi = h[i]
            dmax_i = 1.0 - hi
            if dmax_i <= 1e-12:
                continue
            for jj in j_cand:
                j = int(jj)
                if i == j:
                    continue
                hj = h[j]
                dmax = min(dmax_i, hj, base_step)
                if dmax <= 1e-12:
                    continue
                score = (g[j] - g[i]) + 0.2 * (hi - hj)
                if score > -1e-7:
                    pairs.append((score, i, j, dmax))
        if not pairs:
            break
        pairs.sort(reverse=True)
        improved = 0
        limit = min(len(pairs), 3500)
        for idxp in range(limit):
            if time.time() > deadline - 2.0:
                break
            _, i, j, dmax = pairs[idxp]
            dmax = min(dmax, 1.0 - h[i], h[j])
            if dmax <= 1e-12:
                continue
            for sc in scales:
                d = dmax * sc
                if d < 1e-7:
                    break
                ok, nm = try_move(h, S, i, j, d, sh, n, best_raw)
                if ok:
                    best_raw = nm
                    improved += 1
                    if best_raw * dx < best_c:
                        best_c = best_raw * dx
                        best_h = h.copy()
                    if (improved & 7) == 0:
                        S = compute_S(h, n)
                        best_raw = float(S.max())
                    break
        if improved == 0:
            break
    return best_h, best_c


def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(np.asarray(h, dtype=float).copy(), n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return project_feasible(np.interp(x_new, x_old, np.asarray(h, dtype=float)), n_new)


def random_fourier_h(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    u = np.zeros(n, dtype=float)
    for m in range(1, 9):
        sc = 1.0 / (0.5 * m)
        a = rng.normal(0.0, sc)
        b = rng.normal(0.0, sc)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        u += a * np.cos(np.pi * m * x + ph) + b * np.sin(np.pi * m * x + ph)
    sc = float(np.max(np.abs(u)))
    if sc > 1e-12:
        u = 0.5 * np.tanh(2.0 * u / (sc + 1e-12))
    return project_feasible(0.5 + u, n)


def optimize_n(h, n, deadline, seed=42):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    lb_end = min(time.time() + 100.0, deadline - 180.0)
    if lb_end > time.time() + 20.0:
        hl, cl = lbfgs_refine(best_h, n, lb_end, seed)
        if cl < best_c:
            best_c = cl
            best_h = hl

    gp_end = min(time.time() + 80.0, deadline - 80.0)
    if gp_end > time.time() + 20.0:
        hg, cg = guided_pair(best_h, n, gp_end, seed + 7, base_step=0.04, K=60)
        if cg < best_c:
            best_c = cg
            best_h = hg

    pp_end = deadline - 10.0
    if pp_end > time.time() + 20.0:
        hp, cp = pair_search_fast(best_h, n, pp_end, seed + 13, start_step=0.02, min_step=1e-6)
        if cp < best_c:
            best_c = cp
            best_h = hp

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.asarray(initial_h_values, dtype=float).reshape(-1)
        n0 = int(h_init.size)
    except Exception:
        h_init = random_fourier_h(300, rng)
        n0 = 300

    if n0 > 512:
        h_init = resample_h(h_init, 512)
        n0 = 512

    best_n = n0
    best_h = project_feasible(h_init, best_n)
    best_c = compute_c5(best_h, best_n)

    phase1 = min(deadline - 60.0, t0 + int(0.45 * float(budget_s)))
    if phase1 < t0 + 40.0:
        phase1 = min(deadline - 20.0, t0 + 40.0)
    if phase1 > time.time():
        h1, c1 = optimize_n(best_h, best_n, phase1, seed)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    if best_n < 512 and time.time() < deadline - 250.0:
        n2 = int(round(1.3 * float(best_n) / 4.0) * 4)
        if n2 < best_n + 40:
            n2 = best_n + 48
        if n2 > 512:
            n2 = 512
        end2 = deadline - 120.0
        if end2 > time.time() + 80.0:
            h2 = resample_h(best_h, n2)
            h2, c2 = optimize_n(h2, n2, end2, seed + 101)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n2

    if time.time() < deadline - 220.0:
        rand_end = min(time.time() + 200.0, deadline - 150.0)
        if rand_end > time.time() + 60.0:
            hr = random_fourier_h(best_n, rng)
            hr, cr = optimize_n(hr, best_n, rand_end, seed + 500)
            if cr < best_c:
                best_c = cr
                best_h = hr

    if time.time() < deadline - 4.0:
        hf, cf = pair_search_fast(
            best_h, best_n, deadline - 2.0, seed + 900,
            start_step=5e-5, min_step=2e-7
        )
        if cf < best_c:
            best_c = cf
            best_h = hf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
