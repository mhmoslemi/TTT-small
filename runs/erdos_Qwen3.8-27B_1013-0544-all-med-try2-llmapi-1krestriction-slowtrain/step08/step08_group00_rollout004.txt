    w /= w.sum()

            grad[kk:] += wm * (1.0 - h[:nv])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def top_m_obj(x, n, m, alpha, mu):
    c = compute_c5_vec(x, n)
    top = np.partition(c, -m)[-m:] if m < len(c) else c
    a = alpha * (top - top.max())
    val = top.max() + (a.max() + np.log(np.sum(np.exp(a - a.max())))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_m_grad(x, n, m, alpha, mu):
    c = compute_c5_vec(x, n)
    top_idx = np.argsort(c)[-m:] if m < len(c) else np.arange(len(c))
    top = c[top_idx]
    a = alpha * (top - top.max())
    w = np.exp(a - a.max())
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for ii, idx in enumerate(top_idx):
        if w[ii] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_topm(h, n, m, alpha, mu, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    def o(x):

    def g(x):
        return top_m_grad(x, n, m, alpha, mu)
    try:
        res = minimize(o, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)
