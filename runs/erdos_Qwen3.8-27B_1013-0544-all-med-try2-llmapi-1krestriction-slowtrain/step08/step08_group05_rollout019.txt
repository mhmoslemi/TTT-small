    target = n / 2.0

def slsqp_polish(h, n, maxiter=2000, deadline=None):
    dx = 2.0 / n
    alpha = 100000
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h
