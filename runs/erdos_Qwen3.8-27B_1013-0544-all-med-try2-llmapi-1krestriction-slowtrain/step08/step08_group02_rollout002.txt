import numpy as np
import math
import time
from scipy.optimize import minimize


def compute_carr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(float, copy=False)


def compute_csum(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))


def compute_c5(h, n):
    return compute_csum(h, n) * (2.0 / float(n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(1200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    if abs(h.sum() - target) > 1e-11:
        idx = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if idx.size:
            h[idx[0]] += target - h.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(vals, n, rng=None):
    vals = np.asarray(vals, dtype=float)
    target = n // 2
    h = np.zeros(n, dtype=float)
    idx = np.argsort(vals)[::-1]
    h[idx[:target]] = 1.0
    return h


def random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def quadratic_binary(n, rng):
    vals = np.zeros(n, dtype=float)
    for x in range(1, n + 1):
        vals[(x * x) % n] = 1.0
    return to_binary(vals, n, rng)


def fourier_h(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for f in range(1, 9):
        amp = rng.uniform(0.08, 0.45)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(2.0 * np.pi * f * x / 2.0 + ph)
    return project_feasible(h, n)


def flip_value(c, h, p, new, n):
    old = float(h[p])
    new = float(new)
    e = new - old
    if abs(e) < 1e-15:
        return
    c[n - 1 - p:2 * n - 1 - p] += e * (1.0 - h)
    c[p:n + p] -= e * h[::-1]
    c[n - 1] -= e * e
    h[p] = new


def apply_swap(c, h, a, b, n):
    flip_value(c, h, a, 0.0, n)
    flip_value(c, h, b, 1.0, n)


def binary_swap_eval(c, h, a, b, n):
    cc = c.copy()
    hh = h.copy()
    flip_value(cc, hh, a, 0.0, n)
    flip_value(cc, hh, b, 1.0, n)
    return float(cc.max())


def swap_lists(ones, zeros, p1, p0, n1, n0, a, b):
    pa = p1[a]
    last1 = ones[n1 - 1]
    ones[pa] = last1
    p1[last1] = pa
    p1[a] = -1
    n1 -= 1

    pb = p0[b]
    last0 = zeros[n0 - 1]
    zeros[pb] = last0
    p0[last0] = pb
    p0[b] = -1
    n0 -= 1

    p0[a] = n0
    zeros[n0] = a
    n0 += 1

    p1[b] = n1
    ones[n1] = b
    n1 += 1
    return n1, n0


def structured_binary_descent(h, n, deadline, rng):
    h = np.asarray(h, dtype=float).copy()
    c = compute_carr(h, n)
    curmax = float(c.max())
    besth = h.copy()
    bestc = c.copy()
    bestmax = curmax
    stagnant = 0

    while time.time() < deadline and stagnant < 18:
        curmax = float(c.max())
        if curmax <= bestmax:
            if curmax < bestmax - 1e-12:
                bestmax = curmax
                besth = h.copy()
                bestc = c.copy()
        active = np.where(c >= curmax - 2.0)[0]
        if active.size == 0:
            break
        rng.shuffle(active)

        max_active = 28 if n > 300 else 48
        max_pairs = 160 if n > 300 else 520
        cand_limit = max(350, min(3500, 1800000 // max(1, n)))

        cand = []
        for p in active[:max_active]:
            if len(cand) >= cand_limit:
                break
            d = int(p - (n - 1))
            if d == 0:
                continue
            kk = abs(d)
            if d >= 0:
                idx = np.nonzero((h[:n - kk] > 0.5) & (h[kk:] < 0.5))[0]
            else:
                idx = np.nonzero((h[kk:] > 0.5) & (h[:n - kk] < 0.5))[0]
            if idx.size == 0:
                continue
            if idx.size > max_pairs:
                idx = rng.choice(idx, size=max_pairs, replace=False)
            for ii in idx:
                j = int(ii + d)
                if 0 <= j < n:
                    a = int(min(ii, j))
                    b = int(max(ii, j))
                    if h[a] < 0.5:
                        a, b = b, a
                    cand.append((a, b))
                    if len(cand) >= cand_limit:
                        break
        if not cand:
            stagnant += 1
            continue

        bestmove = curmax
        bestab = None
        for a, b in cand:
            nm = binary_swap_eval(c, h, a, b, n)
            if nm < bestmove - 1e-12:
                bestmove = nm
                bestab = (a, b)
        if bestab is not None:
            apply_swap(c, h, bestab[0], bestab[1], n)
            curmax = float(c.max())
            stagnant = 0
            if curmax < bestmax - 1e-12:
                bestmax = curmax
                besth = h.copy()
                bestc = c.copy()
        else:
            stagnant += 1
            if stagnant > 6 and time.time() < deadline - 10 and n > 0:
                for _ in range(3):
                    aa = np.nonzero(h > 0.5)[0]
                    bb = np.nonzero(h < 0.5)[0]
                    if aa.size == 0 or bb.size == 0:
                        break
                    a = int(aa[rng.randint(aa.size)])
                    b = int(bb[rng.choice(bb.size)])
                    apply_swap(c, h, a, b, n)
                curmax = float(c.max())
                stagnant = 0
    return besth, bestc, bestmax


def binary_sa(h, n, deadline, rng, max_iter=2000000):
    h = np.asarray(h, dtype=float).copy()
    c = compute_carr(h, n)
    curmax = float(c.max())
    besth = h.copy()
    bestc = c.copy()
    bestmax = curmax

    idx1 = np.where(h > 0.5)[0]
    idx0 = np.where(h < 0.5)[0]
    if idx1.size == 0 or idx0.size == 0:
        return besth, bestc, bestmax

    ones_arr = np.zeros(n, dtype=int)
    zeros_arr = np.zeros(n, dtype=int)
    pos1 = np.full(n, -1, dtype=int)
    pos0 = np.full(n, -1, dtype=int)
    n1 = idx1.size
    n0 = idx0.size
    ones_arr[:n1] = idx1
    pos1[idx1] = np.arange(n1)
    zeros_arr[:n0] = idx0
    pos0[idx0] = np.arange(n0)

    temp = 1.2
    noimp = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 20000 == 1:
            temp *= 0.986
            if noimp > 120000:
                temp = max(temp, 0.8)
                noimp = 0
            if it % 40000 == 1:
                c = compute_carr(h, n)
                curmax = float(c.max())

        if n1 <= 0 or n0 <= 0:
            break
        a = int(ones_arr[rng.randint(n1)])
        b = int(zeros_arr[rng.randint(n0)])
        nm = binary_swap_eval(c, h, a, b, n)

        accept = False
        if nm < curmax - 1e-12:
            accept = True
        elif nm <= curmax + 1e-12:
            if rng.random() < 0.22:
                accept = True
        elif nm < curmax + 5.0 * temp + 1e-12:
            if rng.random() < math.exp(-(nm - curmax) / max(temp, 1e-9)):
                accept = True

        if accept:
            apply_swap(c, h, a, b, n)
            n1, n0 = swap_lists(ones_arr, zeros_arr, pos1, pos0, n1, n0, a, b)
            curmax = nm
            if curmax < bestmax - 1e-12:
                bestmax = curmax
                besth = h.copy()
                bestc = c.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
        if noimp > 180000:
            for _ in range(5):
                aa = np.nonzero(h > 0.5)[0]
                bb = np.nonzero(h < 0.5)[0]
                if aa.size == 0 or bb.size == 0:
                    break
                a = int(aa[rng.randint(aa.size)])
                b = int(bb[rng.randint(bb.size)])
                apply_swap(c, h, a, b, n)
                n1, n0 = swap_lists(ones_arr, zeros_arr, pos1, pos0, n1, n0, a, b)
            c = compute_carr(h, n)
            curmax = float(c.max())
            noimp = 0
    return besth, bestc, bestmax


def coordinate_refine(h, c, n, deadline, rng):
    h = np.asarray(h, dtype=float).copy()
    if c is None:
        c = compute_carr(h, n)
    else:
        c = np.asarray(c, dtype=float).copy()
    curmax = float(c.max())
    besth = h.copy()
    bestc = c.copy()
    bestmax = curmax
    arange = np.arange(n)
    p0 = n - 1
    passes = 0

    while time.time() < deadline:
        changed = 0
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            old = float(h[i])
            if not math.isfinite(old):
                continue
            b = c.copy()
            a = np.zeros_like(c)
            r1 = slice(p0 - i, p0 - i + n)
            b[r1] -= old * (1.0 - h)
            a[r1] += 1.0 - h
            r2 = slice(i, i + n)
            hr = h[::-1]
            b[r2] += hr * old
            a[r2] -= hr
            b[p0] = c[p0] - old * (1.0 - old)
            a[p0] = 0.0

            grid_g = 33 if n > 250 else 45
            xs = np.unique(np.concatenate((
                np.linspace(0.0, 1.0, grid_g),
                np.array([old, 0.0, 1.0], dtype=float)
            )))
            vals = b[None, :] + a[None, :] * xs[:, None]
            b0 = b[p0]
            vals[:, p0] = b0 + xs - xs * xs
            scores = np.max(vals, axis=1)
            xb = int(np.argmin(scores))
            xnew = float(np.clip(xs[xb], 0.0, 1.0))
            delta = xnew - old
            if abs(delta) < 1e-7:
                continue

            if delta > 0.0:
                mask = (h >= delta + 1e-12) & (arange != i)
            else:
                mask = (h <= -delta + 1e-12) & (arange != i)
            idx = np.where(mask)[0]
            if idx.size == 0:
                continue
            if idx.size <= 36:
                cand = idx
            else:
                if delta > 0.0:
                    part = np.argpartition(h[idx], -20)[-20:]
                else:
                    part = np.argpartition(h[idx], 20)[:20]
                rnd = rng.choice(idx, size=min(16, int(idx.size)), replace=False)
                cand = np.unique(np.concatenate((idx[part], rnd)))

            bestnm = curmax
            bestj = -1
            bestxj = old
            for j in cand:
                j = int(j)
                xj = float(h[j] - delta)
                if xj < -1e-10 or xj > 1.0 + 1e-10:
                    continue
                xj = float(np.clip(xj, 0.0, 1.0))
                cc = c.copy()
                hh = h.copy()
                flip_value(cc, hh, int(i), xnew, n)
                flip_value(cc, hh, j, xj, n)
                nm = float(cc.max())
                if nm < bestnm - 1e-12:
                    bestnm = nm
                    bestj = j
                    bestxj = xj
            if bestj >= 0:
                flip_value(c, h, int(i), xnew, n)
                flip_value(c, h, bestj, bestxj, n)
                curmax = float(c.max())
                changed += 1
                if curmax < bestmax - 1e-12:
                    bestmax = curmax
                    besth = h.copy()
                    bestc = c.copy()

        passes += 1
        c = compute_carr(h, n)
        curmax = float(c.max())
        if curmax < bestmax - 1e-12:
            bestmax = curmax
            besth = h.copy()
            bestc = c.copy()
        if changed == 0 and passes > 2:
            break
    return besth, bestc, bestmax


def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    smax = float(sh.max())
    s = float(np.log(np.sum(np.exp(sh - smax))))
    return float(m + (smax + s) / alpha)


def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    smax = float(sh.max())
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    p0 = n - 1
    for p in range(2 * n - 1):
        wm = float(w[p])
        if wm < 1e-14:
            continue
        d = p - p0
        if d >= 0:
            nv = n - d
            grad[:nv] += wm * (1.0 - x[d:d + nv])
            grad[d:d + nv] -= wm * x[:nv]
        else:
            kk = -d
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] -= wm * x[kk:kk + nv]
    return grad


def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - 0.5 * float(n)
    return float(val + mu * s * s)


def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - 0.5 * float(n)
    return g + 2.0 * mu * s


def lbfgs_smooth(h, n, deadline):
    besth = np.asarray(h, dtype=float).copy()
    bestv = compute_csum(besth, n)
    bounds = [(0.0, 1.0)] * n
    configs = ((2500.0, 8000.0, 250), (8000.0, 18000.0, 300), (30000.0, 45000.0, 350))
    for alpha, mu, iters in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                pen_obj,
                besth,
                args=(n, alpha, mu),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-15, 'maxfun': iters * 3}
            )
            hnew = project_feasible(res.x, n)
            if np.all(np.isfinite(hnew)):
                v = compute_csum(hnew, n)
                if v < bestv:
                    bestv = v
                    besth = hnew.copy()
        except Exception:
            pass
    return besth, bestv


def resample_linear(h, n_old, n_new):
    if n_old <= 1:
        return np.full(n_new, float(np.mean(h)))
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), n_new)
