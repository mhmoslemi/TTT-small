import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    C = c * dx
    max_C = C.max()
    shifted = alpha * (C - max_C)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    Z = exps.sum()
    p = exps / Z
    val = max_C + np.log(Z) / alpha
    b = 1.0 - x
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if p[m_idx] < 1e-18:
            continue
        coeff = p[m_idx] * dx
        lo1 = max(0, n - 1 - m_idx)
        hi1 = min(n, 2 * n - 1 - m_idx)
        if lo1 < hi1:
            grad[lo1:hi1] += coeff * b[lo1 + m_idx - (n - 1): hi1 + m_idx - (n - 1)]
        lo2 = max(0, m_idx - 1)
        hi2 = min(n, n + m_idx - 1)
        if lo2 < hi2:
            grad[lo2:hi2] -= coeff * x[lo2 - m_idx + 1: hi2 - m_idx + 1]
    return val, grad

def lbfgs_smooth(h, n, alpha, maxiter=500, pen=5000.0):
    target = n / 2.0
    def fun(x):
        v, g = obj_and_grad(x, n, alpha)
        d = x.sum() - target
        return v + pen * d * d
    def jac(x):
        v, g = obj_and_grad(x, n, alpha)
        d = x.sum() - target
        g = g + 2.0 * pen * d
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return project_feasible(res.x, n)

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=2000000):

    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    T = 0.02
    cool = 0.99998
    scale = 2000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
