    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, alpha_list, max_iters=500):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline - 5:
            break
        tr = deadline - time.time() - 5
        if tr < 10:
            break
        mu = alpha * 0.01
        try:
            res = minimize(penalty_obj, best_h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': max_iters * 40,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_val, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_at_shift(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] = (1.0 - h[k:])
        g[k:] += -h[:nv]
    else:
        kk = -k
        nv = n - kk
        g[kk:] = (1.0 - h[:nv])
        g[:nv] += -h[kk:]
    return g * dx

def focal_local_search(h, n, deadline, max_rounds=200):
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c = corr_vec(h, n)
            k_star_idx = np.argmax(c)
            k_star = k_star_idx - (n - 1)
            g = grad_at_shift(h, n, k_star)
            # Find best (i,j): g[i] > g[j], h[i]>=d, h[j]<=1-d
            hi_mask = h >= d
            lo_mask = h <= 1.0 - d
            # Sort indices by g descending
            order = np.argsort(-g)
            found = False
            for ii in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[ii]
                if not hi_mask[i]:
                    continue
                for jj in range(n - 1, -1, -1):
                    j = order[jj]
                    if i == j:
                        continue
                    if not lo_mask[j]:
                        continue
                    if g[i] - g[j] < 1e-8:
                        break
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        found = True
                        break
                    h[i] += d
                    h[j] -= d
                if found:
                    break
            if not found:
                break
    return h, best_c

def all_pairs_local_search(h, n, deadline, radius=80):
    best_c = compute_c5(h, n)
    d = 0.005
    c = corr_vec(h, n)
    top_k = np.argsort(-c)[:5]
    for d in [0.01, 0.005, 0.002, 0.001]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = corr_vec(h, n)
            top_k = np.argsort(-c)[:3]
            for ki in top_k:
                if time.time() > deadline:
                    return h, best_c
                k = ki - (n - 1)
                g = grad_at_shift(h, n, k)
                hi_idx = np.where(h >= d)[0]
                lo_idx = np.where(h <= 1.0 - d)[0]
                # Try top candidates: high g donors, low g receivers
                hi_sorted = hi_idx[np.argsort(-g[hi_idx])][:40]
                lo_sorted = lo_idx[np.argsort(g[lo_idx])][:40]
                for i in hi_sorted:
                    if time.time() > deadline:
                        return h, best_c
                    for j in lo_sorted:
                        if i == j:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def golden_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.array([int(np.floor(i * phi)) % n for i in range(half)])
    h[positions] = 1.0
    return project_feasible(h, n)

def random_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: L-BFGS-B with adaptive alpha (aggressive ramp)
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    t1_end = min(t0 + 250, deadline - 400)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1_end, alphas, max_iters=300)

    # Phase 2: SLSQP polish
    if time.time() < deadline - 300:
        t2_end = min(time.time() + 80, deadline - 280)
        h2, c2 = slsqp_optimize(best_h, best_n, t2_end, alpha=500000, maxiter=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Focal local search
    if time.time() < deadline - 150:
        t3_end = min(time.time() + 120, deadline - 100)
        best_h, best_c = focal_local_search(best_h, best_n, t3_end)

    # Phase 4: All-pairs local search
    if time.time() < deadline - 40:
        t4_end = min(time.time() + 60, deadline - 20)
        best_h, best_c = all_pairs_local_search(best_h, best_n, t4_end)

    # Phase 5: Try n=400
    if time.time() < deadline - 80:
        n4 = 400
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n4, endpoint=False)
        h4 = np.interp(x_new, x_old, best_h)
        h4 = project_feasible(h4, n4)
        c4 = compute_c5(h4, n4)
        alloc4 = min(100, deadline - time.time() - 30)
        if alloc4 > 30:
            h4, c4 = lbfgs_optimize(h4, n4, time.time() + alloc4 * 0.6,
                                     [100, 1000, 10000, 100000], max_iters=200)
            if time.time() < deadline - 30:
                h4, c4 = focal_local_search(h4, n4, min(time.time() + 40, deadline - 15))
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n4

    # Phase 6: Try diverse restarts at best_n
    starts = [golden_start(best_n, rng), random_start(best_n, rng)]
    for s in starts:
        if time.time() > deadline - 50:
            break
        alloc = min(40, (deadline - time.time() - 20) * 0.3)
        if alloc < 15:
            break
        hs, cs = lbfgs_optimize(s, best_n, time.time() + alloc * 0.7,
                                 [200, 2000, 20000], max_iters=150)
        if cs < best_c:
            best_c = cs
            best_h = hs
            best_n = best_n

    # Phase 7: Final SLSQP
    if time.time() < deadline - 15:
        t7 = min(time.time() + 30, deadline - 5)
        hf, cf = slsqp_optimize(best_h, best_n, t7, alpha=1000000, maxiter=2000)
        if cf < best_c:
            best_c = cf
            best_h = hf

    # Final polish
    if time.time() < deadline - 3:
        t8 = min(time.time() + 10, deadline - 1)
        best_h, best_c = focal_local_search(best_h, best_n, t8, max_rounds=50)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
