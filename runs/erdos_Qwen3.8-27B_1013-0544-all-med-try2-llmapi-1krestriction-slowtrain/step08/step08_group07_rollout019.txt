        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_linearized_con(h0, n, aux):
    # aux: list of (lag_index, offset) for active constraints
    # Returns arrays for constraint: A_con @ h <= b_con (linearized)
    dx = 2.0 / n
    n_cons = len(aux)
    A = np.zeros((n_cons, n))
    b = np.zeros(n_cons)
    for idx, (lag, offset) in enumerate(aux):
        c0 = 0.0
        if offset >= 0:
            cnt = n - offset
            if cnt > 0:
                i1 = np.arange(cnt)
                i2 = i1 + offset
                c0 = np.sum(h0[i1] * (1 - h0[i2]) - h0[i2] * h0[i1] - h0[i1] * h0[i2])
                # linearized: h0[i1]*(1-h0[i2]) - h0[i2]*(h[i1]-h0[i1]) - h0[i1]*(h[i2]-h0[i2])
                # = h0[i1] - h0[i1]*h0[i2] - h0[i2]*h[i1] + h0[i2]*h0[i1] - h0[i1]*h[i2] + h0[i1]*h0[i2]
                # = h0[i1] - h0[i2]*h[i1] - h0[i1]*h[i2]
                # Wait let me redo: h0[i1]*(1-h0[i2]) - h0[i2]*(h[i1]-h0[i1]) - h0[i1]*(h[i2]-h0[i2])
                # = h0[i1] - h0[i1]*h0[i2] - h0[i2]*h[i1] + h0[i2]*h0[i1] - h0[i1]*h[i2] + h0[i1]*h0[i2]
                # = h0[i1] + h0[i1]*h0[i2] - h0[i2]*h[i1] - h0[i1]*h[i2]
                # Hmm that doesn't simplify nicely. Let me just use the direct formula.
                c0 = np.sum(h0[i1] * (1.0 - h0[i2])
                           - h0[i2] * (h0[i1] - h0[i1])  # this is 0, wrong approach
                           )
                # Let me just numerically build the linear part
                # linearized_k(h) = corr_k(h0) + sum of partial derivatives * (h - h0)
                # partial wrt h[i1_j] = -h0[i2_j] (from -h[i1]*h0[i2] term)
                # partial wrt h[i2_j] = -h0[i1_j] (from -h0[i1]*h[i2] term)
                A[idx, i1] += -h0[i2] * dx
                A[idx, i2] += -h0[i1] * dx
                b[idx] = np.sum(h0[i1] * (1.0 - h0[i2])) * dx  # this is corr at h0
                # But we need: linearized(h) = corr(h0) + A_row @ (h - h0) = A_row @ h + corr(h0) - A_row @ h0
                # So the constraint is: A_row @ h + corr(h0) - A_row @ h0 <= t
                # => A_row @ h - t <= A_row @ h0 - corr(h0)
                # Let me store: A_row @ h - t <= rhs where rhs = A_row @ h0 - corr(h0)
                b[idx] = np.dot(A[idx], h0) - b[idx]  # rhs
        else:
            cnt = n + offset
            if cnt > 0:
                i1 = np.arange(-offset, n)
                i2 = i1 + offset
                A[idx, i1] += -h0[i2] * dx
                A[idx, i2] += -h0[i1] * dx
                corr0 = np.sum(h0[i1] * (1.0 - h0[i2])) * dx
                b[idx] = np.dot(A[idx], h0) - corr0
    return A, b

def solve_ccp_qp(h0, n, mu, active_lags, maxiter=300):
    # Solve: min t + mu/2 ||h-h0||^2 s.t. A@h - t <= b, sum(h)*dx=1, 0<=h<=1
    dx = 2.0 / n
    A, b = build_linearized_con(h0, n, active_lags)
    nc = len(active_lags)
    nv = n + 1  # h (n vars) + t
    
    def obj(v):
        h = v[:n]
        t = v[n]
        return t + mu * 0.5 * np.sum((h - h0) ** 2)
    
    def grad(v):
        h = v[:n]
        g = np.empty(nv)
        g[:n] = mu * (h - h0)
        g[n] = 1.0
        return g
    
    cons = [{'type': 'eq',
             'fun': lambda v: np.sum(v[:n]) * dx - 1.0,
             'jac': lambda v: np.concatenate([dx * np.ones(n), np.zeros(1)])}]
    for i in range(nc):
        cons.append({'type': 'ineq',
                     'fun': lambda v, i=i: -(A[i] @ v[:n] - v[n] - b[i]),
                     'jac': lambda v, i=i: np.concatenate([-A[i], np.array([1.0])])})
    
    bounds = [(0, 1)] * n + [(0, None)]
    v0 = np.concatenate([h0, [calculate_t(h0, n, active_lags)]])
    
    try:
        res = minimize(obj, v0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return h0

def calculate_t(h, n, active_lags):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    t = 0.0
    for lag, offset in active_lags:
        t = max(t, corr[lag])
    return max(t, 0.0)

def get_active_lags(h, n, top_m=30, gap=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    indices = np.argsort(corr)[::-1]
    tmax = corr[indices[0]]
    active = []
    for idx in indices:
        if corr[idx] > tmax - gap:
            active.append((int(idx), int(idx) - (n - 1)))
        if len(active) >= top_m:
            break
    return active

def ccp_optimize(h_init, n, budget, mu_start=50.0, mu_decay=0.6, max_iters=500):
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    t0 = time.time()
    
    for mu in _mu_sequence(mu_start, mu_decay, max_iters):
        if time.time() > budget - 5:
            break
        active = get_active_lags(h, n, top_m=40, gap=0.005)
        for sub_iter in range(3):
            if time.time() > budget - 5:
                break
            h_new = solve_ccp_qp(h, n, mu, active, maxiter=200)
            if np.allclose(h_new, h, atol=1e-10):
                break
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
            h = project_feasible(h_new, n)
            # Update active set
            active = get_active_lags(h, n, top_m=40, gap=0.005)
    
    return h, best_c

def _mu_sequence(start, decay, max_n):
    m = start
    for _ in range(max_n):
        yield m
        m *= decay
        if m < 0.1:
            m = 0.1

def coord_descent_fast(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = budget_s - 10
    dl = t0 + total_budget
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    best_h = project_feasible(h0.copy(), n0)
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: CCP from parent
    p1 = t0 + 0.35 * total_budget
    h_ccp, c_ccp = ccp_optimize(best_h, n, p1, mu_start=100.0, mu_decay=0.5, max_iters=300)
    if c_ccp < best_c:
        best_c, best_h = c_ccp, h_ccp.copy()
    
    # Phase 2: Coordinate descent polish
    p2 = t0 + 0.50 * total_budget
    h_cd = coord_descent_fast(best_h, n, max_sweeps=300, deadline=p2)
    h_cd = project_feasible(h_cd, n)
    c_cd = compute_c5(h_cd, n)
    if c_cd < best_c:
        best_c, best_h = c_cd, h_cd.copy()
    
    # Phase 3: Another CCP round (finer mu)
    p3 = t0 + 0.70 * total_budget
    h_ccp2, c_ccp2 = ccp_optimize(best_h, n, p3, mu_start=20.0, mu_decay=0.4, max_iters=400)
    if c_ccp2 < best_c:
        best_c, best_h = c_ccp2, h_ccp2.copy()
    
    # Phase 4: More coordinate descent
    p4 = t0 + 0.80 * total_budget
    h_cd2 = coord_descent_fast(best_h, n, max_sweeps=500, deadline=p4)
    h_cd2 = project_feasible(h_cd2, n)
    c_cd2 = compute_c5(h_cd2, n)
    if c_cd2 < best_c:
        best_c, best_h = c_cd2, h_cd2.copy()
    
    # Phase 5: Try different resolutions
    p5 = t0 + 0.92 * total_budget
    for nh in [300, 500, 600]:
        if time.time() > p5 - 20:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh, ch = ccp_optimize(hh, nh, p5 - 15, mu_start=50.0, mu_decay=0.5, max_iters=100)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
        hh2 = coord_descent_fast(hh, nh, max_sweeps=100, deadline=p5 - 10)
        hh2 = project_feasible(hh2, nh)
        ch2 = compute_c5(hh2, nh)
        if ch2 < best_c:
            best_c, best_h, n = ch2, hh2.copy(), nh
    
    # Phase 6: Final CCP + noise restarts
    if time.time() < dl - 10:
        for trial in range(3):
            if time.time() > dl - 5:
                break
            h_r = best_h + 0.002 * rng.randn(n)
            h_r = project_feasible(h_r, n)
            h_r, c_r = ccp_optimize(h_r, n, dl - 3, mu_start=30.0, mu_decay=0.5, max_iters=100)
            if c_r < best_c:
                best_c, best_h = c_r, h_r.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
