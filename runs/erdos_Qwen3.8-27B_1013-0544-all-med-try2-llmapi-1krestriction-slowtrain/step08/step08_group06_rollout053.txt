    """Project h onto {0 ≤ x ≤ 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection on shift t: h_proj[i] = clip(h[i] - t, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0, 1)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def grad_logsumexp(h, n, alpha):
    """Compute gradient of smoothed max objective. Returns gradient and objective value."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    cmax = c.max()
    # Smooth max: f = cmax + (1/alpha)*log(sum exp(alpha*(c-cmax)))
    shifted = alpha * (c - cmax)
    logsum = shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())
    fval = cmax + logsum / alpha
    
    # Softmax weights
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    
    # Gradient: (1/alpha) * sum_k p_k * grad_k
    # grad_k[j] = dx * [(1-h[j+k]) - h[j-k]] (with boundary handling)
    k_vals = np.arange(-(n-1), n)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    # Vectorized: for each j, grad[j] = dx * (F[j] - B[j])
    # F[j] = sum_k p[n-1+k] * (1-h[j+k]) * valid(j+k)
    # B[j] = sum_k p[n-1+k] * h[j-k] * valid(j-k)
    
    # Use the fact that F = correlation-like operation
    # F[j] = sum_m p[m] * (1-h)[j+m-(n-1)] where j+m-(n-1) in [0,n)
    # This is: (1-h) * p cross-correlated, shifted
    
    # For n=400, direct computation is fast enough
    for j in range(n):
        kf = k_vals[(j + k_vals >= 0) & (j + k_vals < n)]
        F = np.dot(p[n-1+kf], one_minus_h[j+kf])
        kb = k_vals[(j - k_vals >= 0) & (j - k_vals < n)]
        B = np.dot(p[n-1+kb], h[j-kb])
        grad[j] = dx * (F - B)
    
    return grad, fval, cmax * dx

def pgd_optimize(h, n, deadline, alpha_start=50, alpha_end=50000, n_phases=5,
                 lr_start=0.1, momentum=0.9):
    """Projected gradient descent with adaptive log-sum-exp."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    phase_duration = (deadline - time.time()) / n_phases
    alphas = np.logspace(np.log10(alpha_start), np.log10(alpha_end), n_phases)
    
    for phase in range(n_phases):
        alpha = alphas[phase]
        phase_deadline = time.time() + phase_duration
        lr = lr_start
        n_phase_iters = 0
        stall = 0
        
        while time.time() < phase_deadline:
            grad, fval, cval = grad_logsumexp(h, n, alpha)
            
            # Adaptive learning rate
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Step with momentum
            v = momentum * v - lr * grad
            h_new = project_feasible(h + v, n)
            c_new = compute_c5(h_new, n)
            
            if c_new <= fval + 1e-10:  # accept (using actual C for safety)
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                # Increase lr slightly if improving
                if c_new < best_c:
                    lr = min(lr * 1.05, 2.0)
            else:
                # Reject, decrease lr
                v = 0.5 * v
                lr *= 0.7
                stall += 1
            
            n_phase_iters += 1
            if stall > 200:
                # Try larger step or break
                lr *= 2.0
                stall = 0
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):

    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)

    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        for i in zeros[:]:
            if time.time() > deadline:
                break

                jm = j + m_range
                msk = (jm >= 0) & (jm < n)

                jmn = j - m_range
