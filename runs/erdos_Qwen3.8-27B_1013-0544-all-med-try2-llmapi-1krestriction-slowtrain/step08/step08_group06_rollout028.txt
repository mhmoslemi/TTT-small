```python
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())

        return h
```

Let me think about the "active-set equalization" more carefully.

After SLSQP, we have a solution where a few lags are at (or very near) the maximum. Let's say lags k1, k2, ..., km are at the max (within epsilon).

We want to find a direction d such that:
- dC_{ki}/dh * d < 0 for all active i (reduces all active correlations)
- The constraint sum(d) = 0 is maintained
- 0 ≤ h + t*d ≤ 1 for small t > 0

This is a linear programming problem! We can solve it with scipy.

But actually, a simpler approach: just take the negative sum of the gradients of the active correlations, project to the null space of the sum constraint, and do a line search.

```python
def active_set_polish(h, n, deadline, max_rounds=100):
    """Try to equalize the top correlations by moving in the gradient direction."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1-h, mode='full') * dx
        max_val = corr.max()
        
        if max_val < best_c:
            best_c = max_val
            best_h = h.copy()
        
        # Find active lags (within 1e-8 of max)
        active = np.where(corr >= max_val - 1e-8)[0]
        
        # Compute gradient of sum of active correlations
        grad = np.zeros(n)
        for k in active:
            l = k - (n - 1)
            if l >= 0:
                jl = np.arange(l, n)
                grad[jl] += (1 - h[jl - l])
                jl2 = np.arange(0, n - l)
                grad[jl2] -= h[jl2 + l]
            else:
                ll = -l
                jl2 = np.arange(0, n - ll)
                grad[jl2] += (1 - h[jl2 + ll])
                jl = np.arange(ll, n)
                grad[jl] -= h[jl - ll]
        
        # Project gradient to null space of sum constraint
        grad -= grad.mean()
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        d = -grad / gnorm
        
        # Line search
        improved = False
        step = 0.01
        for _ in range(30):
            h_new = h + step * d
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < max_val - 1e-12:
                h = h_new
                improved = True
                break
            step *= 0.7
            if step < 1e-8:
                break
        
        if not improved:
            break
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_arr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, n, k_list):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in k_list:
        l = k - (n - 1)
        if l >= 0:
            jl = np.arange(l, n)
            grad[jl] += (1.0 - h[jl - l])
            jl2 = np.arange(0, n - l)
            grad[jl2] -= h[jl2 + l]
        else:
            ll = -l
            jl2 = np.arange(0, n - ll)
            grad[jl2] += (1.0 - h[jl2 + ll])
            jl = np.arange(ll, n)
            grad[jl] -= h[jl - ll]
    return grad * dx

def slsqp_grad(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj_and_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        max_g = g.max()
        shifted = alpha * (g - max_g)
        shifted = np.clip(shifted, -500, 500)
        exp_s = np.exp(shifted - shifted.max())
        logsum = max_g + (shifted.max() + np.log(exp_s.sum())) / alpha
        p = exp_s / exp_s.sum()
        active = np.where(p > 1e-12)[0]
        grad = np.zeros(n)
        if len(active) > 0:
            for k in active:
                l = k - (n - 1)
                if l >= 0:
                    jl = np.arange(l, n)
                    grad[jl] += p[k] * (1.0 - x[jl - l])
                    jl2 = np.arange(0, n - l)
                    grad[jl2] -= p[k] * x[jl2 + l]
                else:
                    ll = -l
                    jl2 = np.arange(0, n - ll)
                    grad[jl2] += p[k] * (1.0 - x[jl2 + ll])
                    jl = np.arange(ll, n)
                    grad[jl] -= p[k] * x[jl - ll]
            grad *= dx
        return logsum, grad
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones}]
    try:
        res = minimize(obj_and_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_nograd(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -500, 500)
        return m + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_polish(h, n, deadline, max_rounds=500):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr_arr(h, n)
        max_val = corr.max()
        if max_val < best_c:
            best_c = max_val
            best_h = h.copy()
        active = np.where(corr >= max_val - 1e-9)[0]
        if len(active) == 0:
            break
        grad = corr_gradient(h, n, active)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        d = -grad / gnorm
        improved = False
        step = 0.005
        for _ in range(40):
            h_new = project_feasible(h + step * d, n)
            c_new = compute_c5(h_new, n)
            if c_new < max_val - 1e-13:
                h = h_new
                improved = True
                break
            step *= 0.6
            if step < 1e-10:
                break
        if not improved:
            # Try individual active lags
            found = False
            for k in active:
                if time.time() > deadline:
                    break
                gk = corr_gradient(h, n, [k])
                gk -= gk.mean()
                nk = np.linalg.norm(gk)
                if nk < 1e-14:
                    continue
                dk = -gk / nk
                step2 = 0.002
                for _2 in range(30):
                    h_new = project_feasible(h + step2 * dk, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < max_val - 1e-13:
                        h = h_new
                        found = True
                        break
                    step2 *= 0.6
                if found:
                    break
            if not found:
                break
    return best_h, best_c
```
