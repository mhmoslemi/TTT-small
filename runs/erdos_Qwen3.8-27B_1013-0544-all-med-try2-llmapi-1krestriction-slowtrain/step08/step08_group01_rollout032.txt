import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n

For the "full" mode with two arrays of length n, the output has length 2n-1. Element i of the output is:
out[i] = sum_j h[j] * (1-h)[j + i - (n-1)]

where the sum is over j such that both j and j+i-(n-1) are in [0, n-1].

- For k < 0: let |k| = -k. out[i] = sum_{j=|k|}^{n-1} h[j] * (1-h[j-|k|]) = sum_{j=0}^{n-1-|k|} h[j+|k|] * (1-h[j])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            g[:n - k] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:n - k])
        else:
            kk = -k
            g[kk:] += wm * (1.0 - x[:n - kk])
            g[:n - kk] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, a_lo=50, a_hi=500000, n_alpha=8, maxiter=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in np.geomspace(a_lo, a_hi, n_alpha):
        if time.time() > deadline - 2:
            break
        mu = max(0.5, alpha * 0.005)
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': maxiter * 30,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def bottleneck_pairs(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for st in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_all(h, n)
            top_idx = np.argsort(c)[-2:]
            g = np.zeros(n)
            for mi in top_idx:
                k = mi - (n - 1)
                if k >= 0:
                    g[:n - k] += (1.0 - h[k:])
                    g[k:] += (-h[:n - k])
                else:
                    kk = -k
                    g[kk:] += (1.0 - h[:n - kk])
                    g[:n - kk] += (-h[kk:])
            order = np.argsort(g)
            n_try = min(60, n)
            for ii in range(n_try):
                if time.time() > deadline:
                    return h, best_c
                i = order[ii]
                for jj in range(n - n_try, n):
                    j = order[jj]
                    if i == j:
                        continue
                    if h[i] + st <= 1.0 and h[j] - st >= 0.0:
                        h[i] += st
                        h[j] -= st
                        c2 = compute_c5(h, n)
                        if c2 < best_c - 1e-14:
                            best_c = c2
                            improved = True
                        else:
                            h[i] -= st
                            h[j] += st
    return h, best_c

def pair_polish(h, n, deadline, radius=35):
    best_c = compute_c5(h, n)
    for d in [0.002, 0.001, 0.0004, 0.0001]:
        while time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not any_imp:
                break
    return h, best_c

def fourier_init(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for f in range(1, 8):
        h += rng.uniform(0.05, 0.25) / np.sqrt(f) * np.cos(2 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, n)

def golden_init(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    pos = np.unique(np.floor(np.arange(half) * phi).astype(int) % n)
    h[pos[:half]] = 1.0
    return project_feasible(h, n)

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return project_feasible(h, n)

    # Phase 2: Try multiple n values
    for n_try in [200, 250, 300, 400]:
        if time.time() > deadline - 200:

        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, n_try, endpoint=False)
        hi = project_feasible(np.interp(xn, xo, best_h), n_try)
        alloc = min(100, max(25, (deadline - time.time() - 120) * 0.45))
        ht, ct = lbfgs_opt(hi, n_try, time.time() + alloc, 50, 500000, 8)
        if ct < best_c:
            best_c = ct

    # Phase 3: Multi-start at best_n
    starts = []
    starts.append(fourier_init(best_n, rng))
    starts.append(golden_init(best_n))
    starts.append(random_binary(best_n, rng))
    starts.append(fourier_init(best_n, rng))
    for s in starts:
        if time.time() > deadline - 80:
            break
        alloc = min(40, max(15, (deadline - time.time() - 40) * 0.35))
