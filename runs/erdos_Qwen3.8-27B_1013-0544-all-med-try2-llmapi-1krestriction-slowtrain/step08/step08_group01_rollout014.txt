    n=len(h)
    if top_k and top_k < len(c):
       idx = np.argpartition(c, -top_k)[-top_k:]
       sub = c[idx]
       s = alpha*(sub - sub.max())
       s -= s.max()
       w = np.exp(s); w /= w.sum()
    else:

def grad_top_active(h, c, alpha, top_k):

        vals = c

        idx = np.argpartition(c, -kk)[-kk:]

    s -= float(np.max(s))
    w = np.exp(s)
    tot = float(np.sum(w))
    if not np.isfinite(tot) or tot <= 0:
       w = np.zeros_like(c)
       w[int(np.argmax(c))] = 1.0
    else: w /= tot
    return w

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(500):

        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:

            t = float(r.sum())
            if t <= 1e-15:
                break
            h += (d / t) * r
        else:

                break
            h += (d / s) * h
        h = np.clip(h, 0.0, 1.0)
    for _ in range(25):

def corr_direct(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def corr_fft(h, n):
    b = (1.0 - h)[::-1]
    out = fftconvolve(h, b, mode="full")
    return out.real[:2 * n - 1]


def corr_auto(h, n):
    if n <= 1200:
        return corr_direct(h, n)
    return corr_fft(h, n)


def c5_direct(h, n):
    return float(np.max(corr_direct(h, n)) * (2.0 / float(n)))


def c5_auto(h, n):
    return float(np.max(corr_auto(h, n)) * (2.0 / float(n)))


def smooth_value(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    s = alpha * (c - m)
    s -= float(np.max(s))
    return float(m + np.log(np.sum(np.exp(s))) / alpha)


def softmax_weights(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s -= float(np.max(s))
    w = np.exp(s)
    t = float(np.sum(w))
    if not np.isfinite(t) or t <= 0.0:
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= t
    return w


def grad_fft_weights(h, w):
    n = len(h)
    L = n - 1
    a = fftconvolve(1.0 - h, w, mode="full").real
    b = fftconvolve(h, w[::-1], mode="full").real
    return a[L:L + n] - b[L:L + n]


def lbfgs_objective(x, n, alpha, mu):
    c = corr_fft(x, n)
    v = smooth_value(c, alpha)
    s = float(x.sum()) - 0.5 * float(n)
    return v + mu * s * s


def lbfgs_gradient(x, n, alpha, mu):
    c = corr_fft(x, n)
    w = softmax_weights(c, alpha)
    g = grad_fft_weights(x, w)
    s = float(x.sum()) - 0.5 * float(n)
    return g + 2.0 * mu * s

def resize_h(h_old, n_old, n_new):

    bh = h0.copy()
    bc = c5_direct(h0, n0)
    bn = n0

    if dl - time.time() > 300:
        pd = min(time.time() + 45.0, dl - 500.0)
        if pd > time.time() + 10.0:
            hp, _ = projected_descent(bh, bn, pd, rng, [(10, 48), (60, 90), (400, 140)])
            cd = c5_direct(hp, bn)
            if cd < bc:
                bc = cd
                bh = hp

    n1 = 1000
    if dl - time.time() > 150:
        hs = resize_h(bh, bn, n1)
        hs = project_feasible(hs + 0.003 * rng.normal(size=n1), n1)
        c_start = c5_direct(hs, n1)
        if c_start < bc:
            bc = c_start
            bh = hs.copy()
            bn = n1

        rem = dl - time.time()
        if rem > 250:
            ldb = min(time.time() + 210.0, dl - 320.0)
            if ldb > time.time() + 20.0:
                hl, _ = fast_lbfgs_fft(hs, n1, ldb)
                cd = c5_direct(hl, n1)
                if cd < bc:
                    bc = cd
                    bh = hl.copy()
                    bn = n1
                hs = hl.copy()

        rem = dl - time.time()
        if rem > 120:
            pdm = time.time() + 0.72 * rem
            if pdm > dl - 80.0:
                pdm = dl - 80.0
            if pdm < time.time() + 30.0:
                pdm = dl - 20.0
            seq1000 = [(8, 48), (30, 80), (120, 120), (500, 160), (2500, 190), (20000, 220)]
            hp, _ = projected_descent(hs, n1, pdm, rng, seq1000)
            cd = c5_direct(hp, n1)
            if cd < bc:
                bc = cd
                bh = hp.copy()
                bn = n1
            hs = hp.copy()

        if time.time() < dl - 25:
            ppd = min(time.time() + 50.0, dl - 10.0)
            hp, _ = pair_polish_top(hs, n1, ppd, rng)
            cd = c5_direct(hp, n1)
            if cd < bc:
                bc = cd
                bh = hp.copy()
                bn = n1
            hs = hp.copy()

        if time.time() < dl - 60:
            alt = project_feasible(hs + 0.01 * rng.normal(size=n1), n1)
            cd = c5_direct(alt, n1)
            if cd < bc:
                bc = cd
                bh = alt.copy()
                bn = n1
            hp, _ = projected_descent(alt, n1, dl - 8.0, rng, [(5000, 160), (100000, 240)])
            cd = c5_direct(hp, n1)
            if cd < bc:
                bc = cd
                bh = hp.copy()
                bn = n1

    h1 = project_feasible(bh, bn)
    c1 = c5_direct(h1, bn)
    h2 = np.clip(bh, 0.0, 1.0)
    c2 = c5_direct(h2, bn)
    if c2 < c1 and h2.min() >= 0.0 and h2.max() <= 1.0 and abs(h2.sum() - 0.5 * bn) < 1e-8:
        return h2, c2, bn
    return h1, c1, bn
