import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_wasserstein(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(x, n, alpha):
    dx = 2.0 / n
    corr = compute_corr(x, n)  # length 2n-1
    c = corr * dx  # objective values at each lag
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    w = exps / exps.sum()  # softmax weights, sum to 1
    obj = m + (sm + np.log(exps.sum())) / alpha
    
    # Gradient
    w_arr = w  # length 2n-1
    conv1 = np.convolve(w_arr, x, mode='full')
    conv2 = np.convolve(w_arr[::-1], x, mode='full')
    A = conv1[n-1:2*n-1]
    B = conv2[n-1:2*n-1]
    grad = dx * (1.0 - A - B)
    return obj, grad

def fista_smooth(h_init, n, alpha, n_iters, deadline):
    dx = 2.0 / n
    x = h_init.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_val = compute_c5(x, n)
    
    # Lipschitz constant estimate: L ≈ 2 * alpha * dx
    # Step size: 1/L
    L = 2.0 * alpha * dx
    step = 1.0 / L
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, grad = smooth_max_and_grad(y, n, alpha)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = project_wasserstein(x - step * grad, n)
        # Momentum
        y = y + ((t - 1.0) / t_new) * (y - x)
        y = project_wasserstein(y, n)
        x = y
        t = t_new
        
        if it % 200 == 0:
            val = compute_c5(x, n)
            if val < best_val:
                best_val = val
                best_x = x.copy()
    
    val = compute_c5(x, n)
    if val < best_val:
        best_val = val
        best_x = x.copy()
    return best_x, best_val

def subgradient_polish(h_init, n, deadline, n_iters=500000):
    """Polish using exact max subgradient with diminishing step size."""
    dx = 2.0 / n
    x = h_init.copy()
    best_x = x.copy()
    best_val = compute_c5(x, n)
    corr = compute_corr(x, n) * dx
    
    # Lower bound for Polyak step size
    # Theory gives C* >= some value; use a conservative lower bound
    lb = best_val - 0.01  # will be updated
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Find max lag
        k_star = np.argmax(corr)
        # Subgradient at this point
        lag = k_star - (n - 1)
        # d corr_k / d x[j] = (1-x[j-k]) - x[j+k]
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j - lag < n:
                g += (1.0 - x[j - lag])
            if 0 <= j + lag < n:
                g -= x[j + lag]
            grad[j] = g
        grad *= dx

        norm_sq = (grad ** 2).sum()
        if norm_sq < 1e-20:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_proj(y, n):
    """Exact projection onto {0<=x<=1, sum=x = n/2} via bisection."""
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def fista_smooth(h_init, n, alpha, max_iters, deadline):
    dx = 2.0 / n
    x = h_init.copy()
    y = x.copy()
    t = 1.0
    step = 1.0 / (2.0 * alpha * dx)
    best_x = x.copy()
    best_val = compute_c5(x, n)

        if time.time() > deadline:
            break
        # Compute gradient at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        c = corr * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        w = exps / exps.sum()

        conv1 = np.convolve(w, y, mode='full')
        conv2 = np.convolve(w[::-1], y, mode='full')
        A = conv1[n-1:2*n-1]
        B = conv2[n-1:2*n-1]

        x = x_new
        t = t_new

                best_x = x.copy()

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_proj(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(y, n, alpha):
    dx = 2.0 / n

    c = corr * dx
    m = c.max()

    w = exps / exps.sum()
    conv1 = np.convolve(w, y, mode='full')
    conv2 = np.convolve(w[::-1], y, mode='full')
    A = conv1[n - 1:2 * n - 1]
    B = conv2[n - 1:2 * n - 1]
    grad = dx * (1.0 - A - B)
