import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlap_grad(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    s = alpha * (c - c.max())
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    grad = overlap_grad(h, n, w)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def active_set_obj(h, n, indices, mu):
    c = compute_c5_vec(h, n)
    val = c[indices].mean() + mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(h, n, indices, mu):
    w = np.zeros(2 * n - 1)
    w[indices] = 1.0 / len(indices)
    grad = overlap_grad(h, n, w)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_opt(h, n, obj_f, grad_f, args, time_limit, max_iters=100):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_f, h, jac=grad_f, args=args,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iters, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': max_iters * 20})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, deltas, radius=70):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def gradient_guided_pairs(h, n, deadline, step=0.003, radius=80):
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000]
    for alpha in alphas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            s = alpha * (c - c.max())
            mx = s.max()
            w = np.exp(s - mx)
            w /= w.sum()
            g = overlap_grad(h, n, w)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    move = -step * (g[i] - g[j])
                    if abs(move) < 1e-10:

                    if h[i] + move >= 0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0 and h[j] - move <= 1.0:

                        if c_val < best_c - 1e-14:
                            best_c = c_val; improved = True
                        else:
                            h[i] -= move; h[j] += move
    return h, best_c
