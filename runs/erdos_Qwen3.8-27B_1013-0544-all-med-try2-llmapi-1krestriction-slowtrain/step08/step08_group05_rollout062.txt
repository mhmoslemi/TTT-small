```python
def fft_correlate(a, b):
    n = len(a) + len(b) - 1
    fa = np.fft.fft(a, n)
    fb = np.fft.fft(b, n)
    return np.fft.ifft(fa * np.conj(fb)).real
```

```python
def delta_c(h, a, b, k, n):
    """Compute change in c(k) when h[a]: 1->0 and h[b]: 0->1"""
    # c(k) = sum_i h[i] * (1-h[i+k]) for valid i, i+k
    # Change comes from positions i where h[i] or h[i+k] changes
    # Affected i values: a, b, a-k, b-k (if in range)
    
    delta = 0
    affected = set()
    for i in [a, b, a-k, b-k]:
        if 0 <= i < n and 0 <= i+k < n:
            affected.add(i)
    
    for i in affected:
        old = h[i] * (1 - h[i+k])
        new_h_i = 0 if i == a else (1 if i == b else h[i])
        new_h_i_k = 0 if (i+k) == a else (1 if (i+k) == b else h[i+k])
        new = new_h_i * (1 - new_h_i_k)
        delta += new - old
    
    return delta
```

```python
def fft_correlate(a, b, n):
    size = 2*n - 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * np.conj(fb), size)
```

```python
            valid_bmk = (bmk >= 0) & (bmk < n)
            delta_b[valid_bmk] -= h[bmk[valid_bmk]]
            
            rng.shuffle(ones_list)
            for a in ones_list:
                ak = a + k_range
                valid_ak = (ak >= 0) & (ak < n)
                delta_a = np.zeros(2*n-1)
                delta_a[valid_ak] -= (1 - h[ak[valid_ak]])
                amk = a - k_range
                valid_amk = (amk >= 0) & (amk < n)
                delta_a[valid_amk] += h[amk[valid_amk]]
                
                new_corr = corr + delta_a + delta_b
                new_max = new_corr.max()
                
                if new_max < cur_max - 1e-10:
                    # Verify with exact computation
                    h[a] = 0
                    h[b] = 1
                    corr = np.correlate(h, 1-h, mode='full').astype(float)
                    cur_max = float(corr.max())
                    ones_list.remove(a)
                    ones_list.append(b)
                    zeros_list.remove(b)
                    zeros_list.append(a)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, cur_max * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n, rng):
    target = n // 2
    h = np.round(h).astype(float)
    while h.sum() > target:
        idx = rng.choice(np.where(h == 1)[0])
        h[idx] = 0
    while h.sum() < target:
        idx = rng.choice(np.where(h == 0)[0])
        h[idx] = 1
    return h

def fast_delta(h, a, b, n, k_range):
    """Compute approximate delta in correlation for swap a(1->0), b(0->1)."""
    delta = np.zeros(2 * n - 1)
    ak = a + k_range
    m = (ak >= 0) & (ak < n)
    delta[m] -= (1 - h[ak[m]])
    amk = a - k_range
    m = (amk >= 0) & (amk < n)
    delta[m] += h[amk[m]]
    bk = b + k_range
    m = (bk >= 0) & (bk < n)
    delta[m] += (1 - h[bk[m]])
    bmk = b - k_range
    m = (bmk >= 0) & (bmk < n)
    delta[m] -= h[bmk[m]]
    return delta

def greedy_binary_search(h, n, deadline, seed=42, max_rounds=500000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = project_binary(h, n, rng)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    k_range = np.arange(-(n - 1), n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(zeros)
        for b in zeros:
            if time.time() > deadline:
                break
            delta_b = np.zeros(2 * n - 1)
            bk = b + k_range
            m = (bk >= 0) & (bk < n)
            delta_b[m] += (1 - h[bk[m]])
            bmk = b - k_range
            m = (bmk >= 0) & (bmk < n)
            delta_b[m] -= h[bmk[m]]
            
            rng.shuffle(ones)
            found = False
            for a in ones:
                ak = a + k_range
                m = (ak >= 0) & (ak < n)
                delta_a = np.zeros(2 * n - 1)
                delta_a[m] -= (1 - h[ak[m]])
                amk = a - k_range
                m = (amk >= 0) & (amk < n)
                delta_a[m] += h[amk[m]]
                
                nm = float((corr + delta_a + delta_b).max())

                    h[b] = 1
                    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)

                    ones[ones.index(a)] = b
                    zeros[zeros.index(b)] = a

                    found = True

    return h, cur_max * dx

def anneal_binary(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)

    h = project_binary(h, n, rng)

    cur_max = float(corr.max())

    best_max = cur_max
    k_range = np.arange(-(n - 1), n)
    t0 = time.time()
    total = max(deadline - t0, 1.0)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    while time.time() < deadline:
        prog = (time.time() - t0) / total
        T = 0.5 * (0.01 / 0.5) ** prog + 0.001
        a = int(rng.choice(ones))
        b = int(rng.choice(zeros))
        if a == b:
            continue
        delta = fast_delta(h, a, b, n, k_range)
        nm = float((corr + delta).max())
        d = nm - cur_max
        if d < 0 or rng.random() < np.exp(-d / T):

            h[b] = 1

            cur_max = float(corr.max())
            oi = ones.index(a)
            zi = zeros.index(b)
            ones[oi] = b
            zeros[zi] = a
            if cur_max < best_max:

                best_h = h.copy()

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return np.clip(res.x, 0, 1)
    except Exception:
        return np.clip(h, 0, 1)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):

        if abs(diff) < 1e-12:
            break
        if diff > 0:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
```
