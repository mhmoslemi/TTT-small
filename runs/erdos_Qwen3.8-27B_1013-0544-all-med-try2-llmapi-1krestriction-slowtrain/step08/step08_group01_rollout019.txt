def slsqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):

    bounds = [(0.0, 1.0)] * n

        if alloc < 20:
            continue
        h4, c4 = lbfgs_phase(hi, n_try, time.time() + alloc)
        if c4 < best_c:
            best_c, best_h, best_n = c4, h4, n_try

    # Phase 5: Diverse restarts
    starts = [fourier_start(best_n, rng, 30),
              fourier_start(best_n, rng, 60),
              random_binary_start(best_n, rng)]
    for s in starts:
        if time.time() > deadline - 60:
            break
        alloc = min(35, (deadline - time.time() - 40) * 0.5)
        if alloc < 15:
            break
        hr, cr = lbfgs_phase(s, best_n, time.time() + alloc)
        if cr < best_c:
            best_c, best_h = cr, hr

    # Phase 6: Binary pair search (key local improvement)
    if time.time() < deadline - 40:
        t6 = min(time.time() + 100, deadline - 20)
        h6, c6 = binary_pair(best_h, best_n, t6)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Final SLSQP
    if time.time() < deadline - 15:
        t7 = min(time.time() + 40, deadline - 8)
        h7, c7 = slsqp_polish(best_h, best_n, t7, alpha=3000000, maxiter=5000)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Final pair search
    if time.time() < deadline - 2:
        h8, c8 = pair_search(best_h, best_n, deadline - 1)
        if c8 < best_c:
            best_c, best_h = c8, h8

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
