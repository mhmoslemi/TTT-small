import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj_lbfgs(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_lbfgs(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def _eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def _obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def lbfgs_optimize(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 0.5), (500, 2.0), (2000, 10.0),
               (10000, 50.0), (50000, 200.0),
               (200000, 1000.0), (1000000, 5000.0),
               (5000000, 20000.0), (20000000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_obj_lbfgs, h, jac=_grad_lbfgs,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': max_iters * 50,
                                  'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=5000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_pair_search(h, n, deadline, k_top=50):
    best_c = compute_c5(h, n)
    alpha = 500000
    no_imp = 0
    while time.time() < deadline:
        if no_imp > 8:
            break
        g = grad_smooth_max(h, n, alpha)
        up = np.argsort(g)[-k_top:]
        down = np.argsort(g)[:k_top]
        any_imp = False
        for i in down:
            for j in up:
                if i == j:
                    continue
                ms = min(1.0 - h[i], h[j])
                if ms <= 1e-15:
                    continue
                for frac in [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]:
                    step = min(frac, ms)
                    h[i] += step
                    h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        any_imp = True
                        break
                    h[i] -= step
                    h[j] += step
                ms2 = min(1.0 - h[j], h[i])
                if ms2 <= 1e-15:
                    continue
                for frac in [0.001, 0.0004, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]:
                    step = min(frac, ms2)
                    h[j] += step
                    h[i] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        any_imp = True
                        break
                    h[j] -= step
                    h[i] += step
        if not any_imp:
            no_imp += 1
        else:
            no_imp = 0
    return h, best_c

def random_pair_anneal(h, n, deadline, rng, temp0=0.002, cool=0.7):
    best_c = compute_c5(h, n)
    temp = temp0
    steps_cs = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    trial_count = 0
    for d in steps_cs:
        while time.time() < deadline:
            any_imp = False
            n_trials = 2000
            for _ in range(n_trials):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                h[i] += d
                h[j] -= d
                if h[i] > 1.0 or h[j] < 0.0:
                    h[i] -= d
                    h[j] += d
                    continue
                c = compute_c5(h, n)
                delta = c - best_c
                if delta < -1e-14:
                    best_c = c
                    any_imp = True
                else:
                    accept = rng.uniform() < np.exp(-max(delta, 0.0) / max(temp, 1e-10))
                    if not accept:
                        h[i] -= d
                        h[j] += d
                trial_count += 1
                if trial_count % 100 == 0:
                    temp *= cool
            if not any_imp and temp < 1e-8:
                break
    return h, best_c
