```python
def swap_delta(h, i, j, n):
    # h[i] was 1, now 0; h[j] was 0, now 1
    s = np.arange(-(n-1), n)  # all lags
    delta = np.zeros(2*n-1)
    
    # i: 1->0
    # As first factor: -(1-h[i-s]) for s where i-s in [0,n)
    valid1 = (i - s >= 0) & (i - s < n)
    delta[valid1] -= (1 - h[i - s[valid1]])
    
    # As second factor: +h[i+s] for s where i+s in [0,n)
    valid2 = (i + s >= 0) & (i + s < n)
    delta[valid2] += h[i + s[valid2]]
    
    # j: 0->1
    # As first factor: +(1-h[j-s]) for s where j-s in [0,n)
    valid3 = (j - s >= 0) & (j - s < n)
    delta[valid3] += (1 - h[j - s[valid3]])
    
    # As second factor: -h[j+s] for s where j+s in [0,n)
    valid4 = (j + s >= 0) & (j + s < n)
    delta[valid4] -= h[j + s[valid4]]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    target = n // 2
    h = np.clip(h, 0, 1)
    # Sort by value, set top n/2 to 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def corr_delta(h, i, j, n):
    """Compute delta to correlation when h[i]:1->0 and h[j]:0->1."""
    s = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    v = (i - s >= 0) & (i - s < n)
    delta[v] -= (1 - h[i - s[v]])
    
    v = (i + s >= 0) & (i + s < n)
    delta[v] += h[i + s[v]]
    
    v = (j - s >= 0) & (j - s < n)
    delta[v] += (1 - h[j - s[v]])
    
    v = (j + s >= 0) & (j + s < n)
    delta[v] -= h[j + s[v]]
    
    return delta

def simulated_annealing(h_init, n, deadline, seed=42):
    """Simulated annealing on binary solutions."""
    dx = 2.0 / n
    target = n // 2
    
    # Start from binary version of h_init
    h = make_binary(h_init, n)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    
    best_c = cur_c
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    # Temperature schedule
    T_high = 0.005
    T_low = 1e-7
    # Total number of iterations
    n_iter = 500000  # Will be limited by deadline
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Cooling schedule (geometric)
        frac = min(1.0, it / n_iter)
        T = T_high * (T_low / T_high) ** frac
        
        # Random swap
        if not ones or not zeros:
            break
        i = rng.choice(list(zeros))  # position to set to 1
        j = rng.choice(list(ones))   # position to set to 0
        
        # Compute delta
        delta = corr_delta(h, j, i, n)  # h[j]:1->0, h[i]:0->1
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        
        # Accept/reject
        delta_c = new_c - cur_c
        if delta_c < 0 or (T > 0 and rng.random() < np.exp(-delta_c / T)):
            # Accept
            h[j] = 0.0
            h[i] = 1.0
            corr = new_corr
            cur_c = new_c
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c
```

```python
def coordinate_descent(h, n, max_iter=1000, tol=1e-10):
    dx = 2.0 / n
    h = h.copy()
    
    for it in range(max_iter):
        max_change = 0
        for i in range(n):
            # Compute corr for each k as a function of h[i]
            # corr_k = A_k * h[i] + B_k where A_k and B_k depend on other values
            # We need to find h[i] in [0,1] that minimizes max_k (A_k * h[i] + B_k)
            
            # Current corr with h[i] = 0
            h[i] = 0
            corr_0 = np.correlate(h, 1-h, mode='full')
            # Current corr with h[i] = 1
            h[i] = 1
            corr_1 = np.correlate(h, 1-h, mode='full')
            
            # corr_k(t) = corr_0[k] + t * (corr_1[k] - corr_0[k]) for t in [0,1]
            A = corr_1 - corr_0
            B = corr_0
            
            # Minimize max_k (A[k]*t + B[k]) over t in [0,1]
            # This is a 1D convex optimization
            # The maximum of linear functions is a convex piecewise linear function
            # Minimum is where the subgradient contains 0
            
            # Simple approach: ternary search or just evaluate at critical points
            # Critical points are where A[k]*t + B[k] = A[k']*t + B[k']*
            # i.e., t = (B[k'] - B[k]) / (A[k] - A[k'])
            
            # Simpler: just do a fine grid search
            t_candidates = np.concatenate([
                [0, 1],
                (B[np.argsort(B)][:5] - B) / np.where(A != 0, A, 1)
            ])
            t_candidates = t_candidates[(t_candidates >= 0) & (t_candidates <= 1)]
            t_candidates = np.unique(t_candidates)
            
            best_t = 0
            best_val = np.inf
            for t in t_candidates:
                val = np.max((A * t + B) * dx)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Also try golden section search
            # ... (for robustness)
            
            h[i] = best_t
            max_change = max(max_change, abs(best_t - h[i]))
        
        if max_change < tol:
            break
    
    return h
```

For the 1D minimization of max_k (D[k] + t*C[k]) over t ∈ [0,1]:
This is minimizing a convex piecewise-linear function. The minimum occurs where the subgradient contains 0, i.e., where the maximum is achieved by at least two "active" constraints with opposite slopes.

A simple approach: 
- The function f(t) = max_k (D[k] + t*C[k])
- f is convex, so we can use ternary search (but it's not strictly convex, so we need to be careful)
- Better: find all breakpoints t_k = -D[k]/C[k] (where C[k] ≠ 0) that are in [0,1], sort them, and evaluate f at each breakpoint plus 0 and 1.

The number of breakpoints is at most 2n-1, so this is O(n log n) per coordinate.

Total per sweep: O(n² log n). For n=250: ~250*250*8 = 500K operations. Very fast!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    h = np.clip(h, 0, 1)
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def coord_descent(h, n, max_sweeps=50, n_grid=201):
    """Coordinate descent for continuous h."""
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    lags = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)  # Random order to avoid bias
        for i in order:
            # Compute C for coordinate i
            idx1 = i + lags
            idx2 = i - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            h1 = np.zeros(2*n-1)
            h1[v1] = h[idx1[v1]]
            h2 = np.zeros(2*n-1)
            h2[v2] = h[idx2[v2]]
            C = v1.astype(float) - h1 - h2
            
            # Find best t
            h_old = h[i]
            t = np.linspace(0, 1, n_grid)
            vals = corr + (t[:, None] - h_old) * C[None, :]
            f = vals.max(axis=1)
            best_t = t[f.argmin()]
            
            if abs(best_t - h_old) > 1e-10:
                h[i] = best_t
                corr = corr + (best_t - h_old) * C
                improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    return h

def sim_anneal_binary(h_init, n, deadline, seed=42, n_iters=500000):
    """Simulated annealing on binary solutions."""
    dx = 2.0 / n
    h = make_binary(h_init, n)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    lags = np.arange(-(n-1), n)
    
    T_high = 0.004
    T_low = 1e-7
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        frac = min(1.0, it / n_iters)
        T = T_high * (T_low / T_high) ** frac
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        j = ones[rng.randint(len(ones))]
        i = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Compute delta for h[j]: 1->0, h[i]: 0->1
        # delta = C_j * (0-1) + C_i * (1-0) = -C_j + C_i
        # where C_p is computed for coordinate p
        idx1_j = j + lags
        idx2_j = j - lags
        v1j = (idx1_j >= 0) & (idx1_j < n)
        v2j = (idx2_j >= 0) & (idx2_j < n)
        h1j = np.zeros(2*n-1); h1j[v1j] = h[idx1j[v1j]]
        h2j = np.zeros(2*n-1); h2j[v2j] = h[idx2j[v2j]]
        Cj = v1j.astype(float) - h1j - h2j
        
        idx1_i = i + lags
        idx2_i = i - lags
        v1i = (idx1_i >= 0) & (idx1_i < n)
        v2i = (idx2_i >= 0) & (idx2_i < n)
        h1i = np.zeros(2*n-1); h1i[v1i] = h[idx1i[v1i]]
        h2i = np.zeros(2*n-1); h2i[v2i] = h[idx2i[v2i]]
        Ci = v1i.astype(float) - h1i - h2i
        
        delta = Ci - Cj  # h[i]: 0->1 adds Ci, h[j]: 1->0 subtracts Cj
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        
        dc = new_c - cur_c
        if dc < 0 or (T > 1e-8 and rng.random() < np.exp(-dc / T)):
            h[j] = 0.0
            h[i] = 1.0
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

Wait, I have an issue with the coordinate descent. After changing h[i], the constraint sum(h) = n/2 is violated. The `project_feasible` at the end handles this, but during the descent, the violation might accumulate and cause issues.

Let me add a periodic re-projection during coordinate descent:

Actually, the coordinate descent is unconstrained (it minimizes the max correlation without the volume constraint). The volume constraint is handled by projection at the end. This is a common approach in non-convex optimization - solve the unconstrained problem, then project.

But the unconstrained problem might drive all h[i] to 0 or 1 (since the max correlation is minimized by spreading the mass). Actually no, if all h[i] = 0, then corr = 0 and max = 0. But that violates the volume constraint.

So the unconstrained minimum is h = 0 (or h = 1), which is useless. I need to incorporate the constraint.

Options:
1. Add a penalty: f(h) = max_k corr_k * dx + penalty * (sum(h) - n/2)²
2. Use the constraint in the 1D subproblem: when optimizing h[i], also ensure the total sum stays close to n/2
3. Use a Lagrange multiplier: f(h) + lambda * (sum(h) - n/2)

Option 3 is cleanest. The Lagrangian is:
L(h) = max_k corr_k * dx + lambda * (sum(h) - n/2)

In the coordinate descent, when optimizing h[i]:
new_corr = corr + (t - h_old) * C
new_sum_change = (t - h_old)
f(t) = max_k (new_corr[k]) * dx + lambda * (t - h_old)

This is still a 1D convex problem (max of linears + linear = max of linears).

I can update lambda using a simple subgradient method:
lambda_new = max(0, lambda + eta * (sum(h) - n/2))  (for the inequality, but we have equality)

Actually for equality constraint, lambda can be positive or negative:
lambda_new = lambda + eta * (sum(h) - n/2)

```python
def coord_descent_constrained(h, n, max_sweeps=50, n_grid=201, lambda_init=0.0, eta=0.01):
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    lags = np.arange(-(n-1), n)
    dx = 2.0 / n
    target = n / 2.0
    lam = lambda_init
    
    for sweep in range(max_sweeps):
        order = np.random.permutation(n)
        for i in order:
            idx1 = i + lags
            idx2 = i - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            h1 = np.zeros(2*n-1); h1[v1] = h[idx1[v1]]
            h2 = np.zeros(2*n-1); h2[v2] = h[idx2[v2]]
            C = v1.astype(float) - h1 - h2
            
            h_old = h[i]
            t = np.linspace(0, 1, n_grid)
            # f(t) = max_k (corr[k] + (t-h_old)*C[k]) * dx + lam * (t - h_old)
            vals = (corr + (t[:, None] - h_old) * C[None, :]) * dx + lam * (t - h_old)
            f = np.maximum.reduce(vals.T).T  # max over k dimension
            # Wait, vals has shape (n_grid, 2n-1), I want max over axis=1
            f = vals.max(axis=1)
            best_t = t[f.argmin()]
            
            h[i] = best_t
            corr = corr + (best_t - h_old) * C
        
        # Update lambda
        sum_h = h.sum()
        lam = lam + eta * (sum_h - target)
        lam = np.clip(lam, -1.0, 1.0)  # Keep lambda bounded
    
    h = project_feasible(h, n)
    return h
```

```python
    lags = np.arange(-(n-1), n)

            h1 = np.zeros(2*n-1); h1[v1] = h[idx1[v1]]
            h2 = np.zeros(2*n-1); h2[v2] = h[idx2[v2]]

            h_old = h[i]
            t = np.linspace(0, 1, n_grid)
            vals = (corr + (t[:, None] - h_old) * C[None, :])
            f = vals.max(axis=1)
            best_t = t[f.argmin()]
            
            h[i] = best_t
            corr = corr + (best_t - h_old) * C
        
        # Project to constraint
        h = project_feasible(h, n)
        corr = np.correlate(h, 1-h, mode='full').astype(float)
    
    return h
```
