from scipy.optimize import minimize
import time

    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def grad_c5(h, n, alpha, top_k=80):
    c = compute_c5_vec(h, n)
    w = smooth_max_grad(c, alpha)
    top_idx = np.argsort(w)[-top_k:]
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(h, n, k)
    return grad

    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

    g += 200.0 * (x.sum() - n / 2.0)

def nesterov_pgd(h, n, deadline, alpha=500.0, mu=0.9, lr=0.02, top_k=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    v = np.zeros(n)
    prev_g = None
    while time.time() < deadline:
        g = grad_c5(x, n, alpha, top_k=top_k)
        if prev_g is not None:
            v = mu * v + g - prev_g  # Nesterov momentum term
        else:
            v = g
        prev_g = g.copy()
        # Barzilai-Borwein-like step
        if prev_g is not None:
            s = x - v
            ss = s @ s
            if ss > 1e-10:

from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_c5(h, n, alpha, top_k=80):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    top_idx = np.argsort(w)[-top_k:]
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(h, n, k)
    return grad
