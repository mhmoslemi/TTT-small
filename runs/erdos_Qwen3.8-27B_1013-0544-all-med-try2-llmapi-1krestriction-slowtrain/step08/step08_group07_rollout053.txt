import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        room = 1.0 - h
        s = room.sum()
        if s < 1e-13:
            s2 = h.sum()
            if s2 < 1e-13:
                break
            h = h * (target / s2)
        else:
            h = h + diff / s * room
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_sweeps=300, deadline=None, random_order=False, rng=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            c_prime = corr - B * h[i]
            lo_v = min(np.max(c_prime), np.max(c_prime + B)) - 1e-9
            hi_v = max(np.max(c_prime), np.max(c_prime + B)) + 1e-9
            for _ in range(32):
                mid = (lo_v + hi_v) / 2.0
                pos = B > 1e-14
                neg = B < -1e-14
                zero = ~(pos | neg)
                if zero.any() and c_prime[zero].max() > mid + 1e-12:
                    lo_v = mid
                    continue
                lb = 0.0
                if neg.any():
                    lb = max(lb, float(np.max((mid - c_prime[neg]) / B[neg])))
                ub = 1.0
                if pos.any():
                    ub = min(ub, float(np.min((mid - c_prime[pos]) / B[pos])))
                if lb <= ub + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb = 0.0
            if (B < -1e-14).any():
                lb = max(lb, float(np.max((hi_v - c_prime[B < -1e-14]) / B[B < -1e-14])))
            ub = 1.0

                ub = min(ub, float(np.min((hi_v - c_prime[B > 1e-14]) / B[B > 1e-14])))
            x_opt = np.clip((lb + ub) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-13:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n_ones]] = 1.0
    return h_b

def greedy_swap(h_init, n, deadline, max_sweeps=400):

                jm = j + m_range

def reweighting_polish(h, n, n_rounds=8, deadline=None):
    dx = 2.0 / n
    h = h.copy()
    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        n_active = max(3, min(15, (2*n-1) // 20))
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[n_active - 1] * 0.9
        active = c >= threshold
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            return float(np.dot(w, g))
        def cons_fun(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]

                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            pass
    return h
