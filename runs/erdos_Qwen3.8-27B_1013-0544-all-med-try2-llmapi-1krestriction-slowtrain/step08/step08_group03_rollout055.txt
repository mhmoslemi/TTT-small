    return best_h, best_c

def pair_polish_fast(h0, n, deadline, radius=60):
    h = h0.copy()
    bc = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in deltas:
        improved = True
        max_sweeps = 4
        sweep = 0
        while improved and sweep < max_sweeps and time.time() < deadline:
            improved = False
            sweep += 1
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15:
                            bc = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15:
                            bc = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.zeros(500)
        h_init[rng.choice(500, 250, replace=False)] = 1.0
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS with increasing alpha
    alphas = [300, 1500, 5000, 20000, 80000]
    for a in alphas:
        if time.time() > deadline - 250:
            break
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(sobj, best_h.copy(), jac=grad_fn,
                          args=(best_n, a, 100), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-13, 'maxls': 60})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Active set descent (direct max minimization)
    if time.time() < deadline - 150:
        t2 = min(time.time() + 120, deadline - 80)
        best_h, best_c = active_set_descent(best_h, best_n, t2, eps=0.0003)

    # Phase 3: Try different n values
    for n_try in [400, 600, 750, 900]:
        if time.time() > deadline - 80:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        bounds = [(0.0, 1.0)] * n_try
        try:
            res = minimize(sobj, h_interp, jac=grad_fn,
                          args=(n_try, 10000, 80), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-17, 'gtol': 1e-12})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass

    # Phase 4: High-alpha L-BFGS polish
    if time.time() < deadline - 40:
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(sobj, best_h.copy(), jac=grad_fn,
                          args=(best_n, 200000, 120), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-14})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 5: Active set final
    if time.time() < deadline - 25:
        best_h, best_c = active_set_descent(best_h, best_n, deadline - 10, eps=0.0002)

    # Phase 6: Pair polish
    if time.time() < deadline - 1:
        best_h, best_c = pair_polish_fast(best_h, best_n, deadline - 0.5, radius=50)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
