def momentum_pgd(h0, n, deadline, alpha=800, top_k=60, step0=0.015, mom=0.8):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step0
    stall = 0
    rng_local = np.random.RandomState(12345)

    while time.time() < deadline:
        g = grad_smooth(h, n, alpha, top_k)
        v = mom * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new <= best_c + 1e-13:
            h = h_new
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            step = min(step * 1.015, 0.035)
        else:
            v *= 0.5
            step *= 0.7
            stall += 1
            if stall > 60:
                h = project_feasible(h + 0.008 * rng_local.randn(n), n)
                v = np.zeros(n)
                step = step0
                stall = 0
    return best_h, best_c

def targeted_polish(h0, n, deadline):
    h = h0.copy()
    best_c = compute_c5(h, n)
    rng_local = np.random.RandomState(999)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]

    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            top_lags = np.argsort(c)[-20:]
            for trial in range(80):
                if time.time() > deadline:
                    break
                a = rng_local.randint(n)
                b = rng_local.randint(n)
                if a == b:
                    continue
                if h[a] >= d and h[b] + d <= 1.0:
                    h[a] -= d
                    h[b] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        improved = True
                    else:
                        h[a] += d
                        h[b] -= d
                elif h[b] >= d and h[a] + d <= 1.0:
                    h[b] -= d
                    h[a] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        improved = True
                    else:
                        h[b] += d
                        h[a] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    total_time = budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.zeros(500)
        h_init[::2] = 1.0
        n_init = 500

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    best_n = n_init

    # Phase 1: Momentum PGD (35% time)
    t1 = t0 + 0.35 * total_time
    if time.time() < t1:
        h1, c1 = momentum_pgd(h_init, n_init, t1, alpha=600, top_k=50, step0=0.018, mom=0.82)
        if c1 < best_c:
            best_c = c1
            best_h = h1
            best_n = n_init

    # Phase 2: L-BFGS-B refinement (20% time)
    t2 = t1 + 0.20 * total_time
    if time.time() < t2:
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(best_n, 3000, 80), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 3: Momentum PGD with different params (20% time)
    t3 = t2 + 0.20 * total_time
    if time.time() < t3:
        h3, c3 = momentum_pgd(best_h, best_n, t3, alpha=1500, top_k=80, step0=0.01, mom=0.75)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n

    # Phase 4: Try n=1000 interpolated (15% time)
    t4 = t3 + 0.15 * total_time
    if time.time() < t4:
        n2 = 1000
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n2)
        h4, c4 = momentum_pgd(h_interp, n2, t4, alpha=800, top_k=80, step0=0.012, mom=0.8)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n2

    # Phase 5: Final targeted polish (10% time)
    if time.time() < deadline:
        best_h, best_c = targeted_polish(best_h, best_n, deadline)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
