    imn = i - m_range

    jm = j + m_range

    delta[n - 1] -= 2.0

    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n-1:
        delta[k2] += 1.0

    h = h.copy()
    if time.time() > deadline:
        return h
    res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                   method='L-BFGS-B', args=(n, alpha, mu),
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 40})
    return np.clip(res.x, 0.0, 1.0)

def active_set_refine(h, n, mu, deadline):
    """Refine only the active (interior) points."""
    threshold = 0.01
    active = (h > threshold) & (h < 1.0 - threshold)
    active_idx = np.where(active)[0]
    if len(active_idx) == 0 or len(active_idx) == n:
        return h
    
    # Fix inactive points, optimize active ones
    # We'll use a parameterized approach: optimize only active_idx
    def obj_sub(x):
        h_full = h.copy()
        h_full[active_idx] = x
        return smooth_max_obj(h_full, n, 10000, mu)
    
    def grad_sub(x):
        h_full = h.copy()
        h_full[active_idx] = x
        g = smooth_max_grad(h_full, n, 10000, mu)
        return g[active_idx]
    
    x0 = h[active_idx].copy()
    res = minimize(obj_sub, x0, jac=grad_sub, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * len(active_idx),
                   options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10})
    h_new = h.copy()
    h_new[active_idx] = np.clip(res.x, 0.0, 1.0)
    return h_new

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def delta_for_swap(i, j, h, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2*n-1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n-1:
        delta[k2] += 1.0
    return delta

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = delta_for_swap(i, j, h, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def _sm_obj(x, n, alpha, mu):
    h = x
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cmax = c.max()
    sh = alpha * (c - cmax)
    val = cmax + np.log(np.exp(sh).sum()) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

def _sm_grad(x, n, alpha, mu):
    h = x
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for m in range(len(w)):
        lag = int(lags[m])
        wm = w[m] * dx
        if wm < 1e-18:
            continue
        jlo = max(0, -lag)
        jhi = min(n - 1, n - 1 - lag)
        if jlo <= jhi:
            grad[jlo:jhi+1] += wm * (1.0 - h[jlo+lag:jhi+1+lag])
        jlo2 = max(0, lag)
        jhi2 = min(n - 1, n - 1 + lag)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2+1] -= wm * h[jlo2-lag:jhi2+1-lag]
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, alpha, mu, maxiter, deadline):
    if time.time() > deadline:
        return h
    try:
        res = minimize(_sm_obj, h.copy(), jac=_sm_grad, method='L-BFGS-B',
                       args=(n, alpha, mu), bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

def active_refine(h, n, deadline):
    threshold = 0.02
    active_mask = (h > threshold) & (h < 1.0 - threshold)
    aidx = np.where(active_mask)[0]
    if len(aidx) < 2 or len(aidx) > n:
        return h
    fixed = h.copy()
    def _o(x):
        hf = fixed.copy()
        hf[aidx] = x
        return _sm_obj(hf, n, 50000, 50.0)
    def _g(x):
        hf = fixed.copy()
        hf[aidx] = x
