def project_feasible(x, n):
    target = 0.5 * n
    x = np.clip(np.asarray(x, dtype=float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - x
            room[x >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            x += (diff / tot) * room
        else:
            mass = x.copy()
            mass[x <= 0.0] = 0.0
            tot = mass.sum()
            if tot < 1e-14:
                break
            x += (diff / tot) * mass
    return np.clip(x, 0.0, 1.0)

    x = np.clip(np.asarray(x, dtype=float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - x
            room[x >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            x += (diff / tot) * room
        else:
            mass = x.copy()
            mass[x <= 0.0] = 0.0
            tot = mass.sum()
            if tot < 1e-14:
                break
            x += (diff / tot) * mass
    return np.clip(x, 0.0, 1.0)


def interpolate_to(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def top_k_binary(h, n):
    k = n // 2
    out = np.zeros(n, dtype=float)
    if k > 0:
        idx = np.argpartition(h, k - 1)[:k]
        out[idx] = 1.0
    return out


def fourier_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for f in range(1, 9):
        amp = rng.uniform(0.3, 0.9) / math.sqrt(f)
        ph = rng.uniform(0.0, 2.0 * math.pi)
        h += amp * np.cos(2.0 * math.pi * f * x / 2.0 + ph)
    h += 0.05 * rng.normal(size=n)
    return h


def binary_flip(C, h, om, p, val, n):
    if h[p] == val:
        return
    s = n - 1 - p
    if val:
        C[s:s + n] += om
        C[p:p + n] -= h[::-1]
    else:
        C[s:s + n] -= om
        C[p:p + n] += h[::-1]
    C[n - 1] -= 1
    h[p] = val
    om[p] = 1 - val


def init_C_binary(binary, n):
    h = np.zeros(n, dtype=np.uint8)
    om = np.ones(n, dtype=np.uint8)
    C = np.zeros(2 * n - 1, dtype=np.int32)
    b = np.asarray(binary) > 0.5
    for p in np.flatnonzero(b):
        binary_flip(C, h, om, p, 1, n)
    return C


def binary_hill_climb(h, om, C, cur_max, best_h, best_om, best_C, best_max,
                      n, end_time, rng, max_sweeps):
    for _ in range(max_sweeps):
        if time.time() >= end_time:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(om)
        if ones.size == 0 or zeros.size == 0:
            break
        ones = ones[rng.permutation(ones.size)]
        zeros = zeros[rng.permutation(zeros.size)]
        improved = False
        cnt = 0
        for i in ones:
            if h[i] == 0:
                continue
            for j in zeros:
                if om[j] == 0:
                    continue
                binary_flip(C, h, om, i, 0, n)
                binary_flip(C, h, om, j, 1, n)
                m = int(C.max())
                if m < cur_max:
                    cur_max = m
                    improved = True
                    if m < best_max:
                        best_max = m
                        best_h = h.copy()
                        best_om = om.copy()
                        best_C = C.copy()
                else:
                    binary_flip(C, h, om, j, 0, n)
                    binary_flip(C, h, om, i, 1, n)
                cnt += 1
                if (cnt & 1023) == 0 and time.time() >= end_time:
                    break
            if time.time() >= end_time:
                break
        if not improved:
            break
    return h, om, C, cur_max, best_h, best_om, best_C, best_max


def binary_ils(h0, n, end_time, rng):
    b = np.asarray(h0, dtype=float) > 0.5
    target = n // 2
    cnt = int(b.sum())
    if cnt < target:
        zeros = np.flatnonzero(~b)
        b[rng.choice(zeros, size=target - cnt, replace=False)] = True
    elif cnt > target:
        ones = np.flatnonzero(b)
        b[rng.choice(ones, size=cnt - target, replace=False)] = False

    h = b.astype(np.uint8)
    om = (~b).astype(np.uint8)
    C = init_C_binary(b, n)
    cur = int(C.max())
    best_h = h.copy()
    best_om = om.copy()
    best_C = C.copy()
    best_max = cur

    init_end = min(end_time, time.time() + max(5.0, 0.25 * max(0.0, end_time - time.time())))
    h, om, C, cur, best_h, best_om, best_C, best_max = binary_hill_climb(
        h, om, C, cur, best_h, best_om, best_C, best_max, n, init_end, rng, 60
    )

    k = 3
    fail = 0
    skipped = 0
    while time.time() < end_time:
        base_h = best_h.copy()
        base_om = best_om.copy()
        base_C = best_C.copy()
        cur = best_max
        if cur <= 0:
            break

        kk = min(k, max(1, n // 20))
        onesp = np.flatnonzero(base_h)
        zerosp = np.flatnonzero(base_om)
        for _ in range(kk):
            if onesp.size == 0 or zerosp.size == 0:
                break
            i = int(onesp[rng.integers(onesp.size)])
            j = int(zerosp[rng.integers(zerosp.size)])
            if i != j and base_h[i] == 1 and base_om[j] == 1:
                binary_flip(base_C, base_h, base_om, i, 0, n)
                binary_flip(base_C, base_h, base_om, j, 1, n)

        pm = int(base_C.max())
        if pm > best_max + max(3, best_max // 8):
            k = max(1, k // 2)
            skipped += 1
            if skipped > 50:
                break
            continue
        skipped = 0

        max_sweeps = min(25, 4 + k + fail // 10)
        before = best_max
        h, om, C, cur, best_h, best_om, best_C, best_max = binary_hill_climb(
            base_h, base_om, base_C, pm, best_h, best_om, best_C, best_max,
            n, end_time, rng, max_sweeps
        )
        if best_max < before:
            fail = 0
            k = max(2, k // 2)
        else:
            fail += 1
            k = min(18, k + 1)
        if fail > 250:
            break
    return best_h, best_C, best_max


def cont_flip(C, h, oh, p, delta, n):
    if delta == 0.0:
        return
    s = n - 1 - p
    C[s:s + n] += delta * oh
    C[p:p + n] += -delta * h[::-1]
    C[n - 1] -= delta * delta
    v = h[p] + delta
    h[p] = v
    oh[p] = 1.0 - v


def active_gradient(h, C, n, tol=0.55, max_lags=32):
    maxC = float(C.max())
    if maxC <= 0.0:
        return None
    act = np.where(C >= maxC - tol)[0]
    if act.size == 0:
        act = np.array([int(np.argmax(C))])
    if act.size > max_lags:
        act = act[np.argpartition(C[act], -max_lags)[-max_lags:]]
    g = np.zeros(n, dtype=float)
    nm1 = n - 1
    for t in act:
        ell = int(t - nm1)
        if ell == 0:
            g += 1.0 - 2.0 * h
        elif ell > 0:
            nv = n - ell
            g[:nv] += 1.0 - h[ell:]
            g[ell:] -= h[:nv]
        else:
            kk = -ell
            nv = n - kk
            g[kk:] += 1.0 - h[:nv]
            g[:nv] -= h[kk:]
    g -= g.mean()
    return g


def continuous_step(h, oh, C, n, rng):
    maxC = float(C.max())
    g = active_gradient(h, C, n)
    if g is None:
        return False, maxC
    d = -g
    d -= d.mean()
    pos = np.flatnonzero(h < 1.0 - 1e-8)
    neg = np.flatnonzero(h > 1e-8)
    if pos.size == 0 or neg.size == 0:
        return False, maxC
    pv = d[pos]
    nv = d[neg]
    if pv.max() <= 1e-9 or nv.min() >= -1e-9:
        return False, maxC
    kp = min(5, pos.size)
    kn = min(5, neg.size)
    ip = pos[np.argpartition(pv, -kp)[-kp:]]
    jn = neg[np.argpartition(nv, kn)[:kn]]
    ip = ip[d[ip] > 1e-9]
    jn = jn[d[jn] < -1e-9]
    if ip.size == 0 or jn.size == 0:
        return False, maxC

    best_m = maxC + 1.0
    bi = bj = -1
    be = 0.0
    for _ in range(2):
        i = int(ip[rng.integers(ip.size)])
        j = int(jn[rng.integers(jn.size)])
        mv = min(1.0 - h[i], h[j])
        if mv <= 1e-8:
            continue
        for fr in (0.05, 0.2, 0.5):
            eps = mv * fr
            if eps <= 1e-9:
                continue
            cont_flip(C, h, oh, i, eps, n)
            cont_flip(C, h, oh, j, -eps, n)
            m = float(C.max())
            cont_flip(C, h, oh, j, eps, n)
            cont_flip(C, h, oh, i, -eps, n)
            if m < best_m:
                best_m = m
                bi, bj, be = i, j, eps
    if bi >= 0 and best_m < maxC - 1e-10:
        cont_flip(C, h, oh, bi, be, n)
        cont_flip(C, h, oh, bj, -be, n)
        return True, best_m
    return False, maxC


def continuous_polish(h0, n, end_time, rng):
    h = project_feasible(h0, n)
    oh = 1.0 - h
    C = compute_C_float(h, n)
    cur = float(C.max())
    best = cur
    best_h = h.copy()
    best_oh = oh.copy()
    scale = 2.0 / n
    stall = 0
    it = 0

    while time.time() < end_time:
        ok, _ = continuous_step(h, oh, C, n, rng)
        cur = float(C.max())
        if ok and cur < best - 1e-12:
            best = cur
            best_h = h.copy()
            best_oh = oh.copy()
            stall = 0
        else:
            stall += 1

        if stall > 25:
            if cur > best + 0.2:
                h = best_h.copy()
                oh = best_oh.copy()
                C = compute_C_float(h, n)
                cur = best
            for _ in range(int(rng.integers(1, 4))):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                if h[i] < h[j]:
                    mv = min(1.0 - h[i], h[j])
                    eps = mv * rng.uniform(0.05, 0.4)
                    if eps > 1e-9:
                        cont_flip(C, h, oh, i, eps, n)
                        cont_flip(C, h, oh, j, -eps, n)
                elif h[j] < h[i]:
                    mv = min(1.0 - h[j], h[i])
                    eps = mv * rng.uniform(0.05, 0.4)
                    if eps > 1e-9:
                        cont_flip(C, h, oh, j, eps, n)
                        cont_flip(C, h, oh, i, -eps, n)
            cur = float(C.max())
            if cur > best + 1.0:
                h = best_h.copy()
                oh = best_oh.copy()
                C = compute_C_float(h, n)
                cur = best
            stall = 0

        it += 1
        if it % 200 == 0:
            h = project_feasible(h, n)
            oh = 1.0 - h
            C = compute_C_float(h, n)
            cur = float(C.max())
            if cur < best:
                best = cur
                best_h = h.copy()
                best_oh = oh.copy()
    return best_h, float(best * scale)


def optimize_blend(h_bin, h_par, n):
    hb = project_feasible(h_bin, n)
    hp = project_feasible(h_par, n)
    best = hb.copy()
    best_c = compute_c5(best, n)
    for a in np.linspace(0.0, 1.0, 41):
        hh = a * hb + (1.0 - a) * hp
        hh = project_feasible(hh, n)
        cc = compute_c5(hh, n)
        if cc < best_c - 1e-14:
            best = hh.copy()
            best_c = cc
    return best, float(best_c)


def lbfgs_pen_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    smax = float(sh.max())
    lse = np.log(np.sum(np.exp(sh - smax)))
    val = m + (smax + lse) / alpha
    s = x.sum() - 0.5 * n
    return float(val + mu * s * s)


def lbfgs_pen_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    smax = float(sh.max())
    w = np.exp(sh - smax)
    w /= w.sum()
    g = np.zeros_like(x)
    nm1 = n - 1
    for t in range(2 * n - 1):
        wt = float(w[t])
        if wt < 1e-11:
            continue
        ell = t - nm1
        if ell == 0:
            g += wt * (1.0 - 2.0 * x)
        elif ell > 0:
            nv = n - ell
            g[:nv] += wt * (1.0 - x[ell:])
            g[ell:] -= wt * x[:nv]
        else:
            kk = -ell
            nv = n - kk
            g[kk:] += wt * (1.0 - x[:nv])
            g[:nv] -= wt * x[kk:]
    s = x.sum() - 0.5 * n
    g += 2.0 * mu * s
    return g


def lbfgs_polish(h0, n, end_time):
    h0 = project_feasible(h0, n)
    best = compute_c5(h0, n)
    best_h = h0.copy()
    bounds = ((0.0, 1.0),) * n
    for alpha in (150, 1500, 12000):
        if time.time() >= end_time or (end_time - time.time()) < 12:
            break
        try:
            res = minimize(
                lbfgs_pen_obj,
                best_h,
                jac=lbfgs_pen_grad,
                args=(n, alpha, 1e5),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 120, "ftol": 1e-15, "gtol": 1e-12, "maxfun": 700},
            )
            hx = project_feasible(res.x, n)
            cx = compute_c5(hx, n)
            if cx < best:
                best = cx
                best_h = hx.copy()
        except Exception:
            pass
    return best_h, float(best)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 6.0
    rng = np.random.default_rng(seed)

    h_par = None
    try:
        h_par = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_par = None

    if h_par is None:
        h_par = fourier_start(500, rng)

    n_par = len(h_par)
    h_par = project_feasible(h_par, n_par)

    if n_par % 2 == 0 and 200 <= n_par <= 1200:
        base_n = n_par
        h_base = h_par.copy()
    else:
        base_n = 500
        h_base = project_feasible(interpolate_to(h_par, n_par, base_n), base_n)

    best_h = h_base.copy()
    best_n = int(base_n)
    best_c = compute_c5(best_h, best_n)

    h_b = top_k_binary(h_base, base_n)
    Cb = init_C_binary(h_b, base_n)
    mb = int(Cb.max())
    cb = mb * 2.0 / base_n
    if cb < best_c:
        best_h = h_b.astype(float)
        best_c = float(cb)
        best_n = int(base_n)

    if deadline - time.time() > 260:
        alloc = min((deadline - time.time() - 220) * 0.65, max(30.0, 0.34 * budget))
        if alloc > 8.0:
            h_b, Cb, mb = binary_ils(h_b, base_n, time.time() + alloc, rng)
            cb = mb * 2.0 / base_n
            if cb < best_c:
                best_h = h_b.astype(float)
                best_c = float(cb)
                best_n = int(base_n)

    if time.time() < deadline - 90:
        h_blend, cb = optimize_blend(h_b, h_base, base_n)
        if cb < best_c:
            best_h = h_blend.copy()
            best_c = float(cb)
            best_n = int(base_n)
        alloc = min((deadline - time.time() - 60) * 0.5, max(15.0, 0.14 * budget))
        if alloc > 5.0:
            h_c, cc = continuous_polish(h_blend, base_n, time.time() + alloc, rng)
            if cc < best_c:
                best_h = h_c.copy()
                best_c = float(cc)
                best_n = int(base_n)
        if time.time() < deadline - 80:
            alloc = min((deadline - time.time() - 60) * 0.35, 40.0)
            if alloc > 8.0:
                h_in = best_h if best_n == base_n else project_feasible(
                    interpolate_to(best_h, best_n, base_n), base_n
                )
                h_l, cl = lbfgs_polish(h_in, base_n, time.time() + alloc)
                if cl < best_c:
                    best_h = h_l.copy()
                    best_c = float(cl)
                    best_n = int(base_n)

    other_n = 1000 if base_n < 1000 else 500
    if other_n != base_n and time.time() < deadline - 140:
        h_start = project_feasible(interpolate_to(best_h, best_n, other_n), other_n)
        h_bo = top_k_binary(h_start, other_n)
        Cbo = init_C_binary(h_bo, other_n)
        mbo = int(Cbo.max())
        cbo = mbo * 2.0 / other_n
        if cbo < best_c:
            best_h = h_bo.astype(float)
            best_c = float(cbo)
            best_n = int(other_n)

        alloc = min((deadline - time.time() - 120) * 0.55, max(25.0, 0.20 * budget))
        if alloc > 10.0:
            h_bo, Cbo, mbo = binary_ils(h_bo, other_n, time.time() + alloc, rng)
            cbo = mbo * 2.0 / other_n
            if cbo < best_c:
                best_h = h_bo.astype(float)
                best_c = float(cbo)
                best_n = int(other_n)

        if time.time() < deadline - 60:
            h_blend2, cb2 = optimize_blend(h_bo, h_start, other_n)
            if cb2 < best_c:
                best_h = h_blend2.copy()
                best_c = float(cb2)
                best_n = int(other_n)
            alloc2 = min((deadline - time.time() - 40) * 0.45, max(10.0, 0.08 * budget))
            if alloc2 > 5.0:
                h_c2, cc2 = continuous_polish(h_blend2, other_n, time.time() + alloc2, rng)
                if cc2 < best_c:
                    best_h = h_c2.copy()
                    best_c = float(cc2)
                    best_n = int(other_n)

    if time.time() < deadline - 5:
        h_rev = best_h[::-1]
        h_com = 1.0 - best_h
        h_sym = 0.5 * (best_h + 1.0 - h_rev)
        h_pal = 0.5 * (best_h + h_rev)
        for hc in (h_rev, h_com, h_sym, h_pal):
            hc = project_feasible(hc, best_n)
            cc = compute_c5(hc, best_n)
            if cc < best_c:
                best_h = hc.copy()
                best_c = float(cc)

        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5(best_h, best_n)

        if time.time() < deadline - 10:
            h_c, cc = continuous_polish(best_h, best_n, deadline - 4.0, rng)
            if cc < best_c:
                best_h = h_c.copy()
                best_c = float(cc)

        if time.time() < deadline - 12:
            h_l, cl = lbfgs_polish(best_h, best_n, deadline - 3.0)
            if cl < best_c:
                best_h = h_l.copy()
                best_c = float(cl)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
