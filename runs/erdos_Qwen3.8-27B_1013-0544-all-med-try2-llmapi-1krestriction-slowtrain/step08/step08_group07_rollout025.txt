def slsqp_smooth(h, n, alpha, maxiter, ftol, use_penalty=False, pen_lambda=0.0):

        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        pen = 0.0

            s = x.sum() - n / 2.0
            pen = pen_lambda * s * s
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha + pen

    else:

            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Noise restarts
    if time.time() < dl - 10:
        for trial in range(6):
            if time.time() > dl - 3:
                break
            noise_scale = 0.002 + 0.002 * trial
            h = best_h + noise_scale * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [5000000, 50000000, 500000000]:
                if time.time() > dl - 2:
                    break
                h = slsqp_smooth(h, n, alpha, 4000, 1e-18)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()

    # Final: coord descent polish
    if time.time() < dl - 5:
        hf, _ = coord_descent(best_h, n, max_sweeps=200, deadline=dl - 2)
        hf = project_feasible(hf, n)
        cf = compute_c5(hf, n)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
