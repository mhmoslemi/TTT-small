import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def fast_project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def ps_grad(c, dc, h, n):
    """Compute gradient given correlation c and dc = c^(p-1)/s^(1/p)."""
    grad = np.zeros(n)
    for i in range(n):
        m_lo = max(-(n-1), i - (n-1))
        m_hi = min(n-1, i)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi + 1)
            grad[i] += np.dot(dc[ms + (n-1)], 1.0 - h[i - ms])
        m_lo2 = max(-(n-1), -i)
        m_hi2 = min(n-1, n-1-i)
        if m_lo2 <= m_hi2:
            ms2 = np.arange(m_lo2, m_hi2 + 1)
            grad[i] -= np.dot(dc[ms2 + (n-1)], h[i + ms2])
    return grad

def opt_lbfgs(h, n, p, mu, maxiter, deadline):
    """L-BFGS-B optimization with power-sum objective."""
    def fun(x):
        c = np.abs(np.correlate(x, 1-x, mode='full'))
        cp = c ** (p-1)
        s = np.sum(c ** p)
        f = s ** (1.0/p)
        diff = x.sum() - n/2.0
        return f + mu * diff * diff
    
    def jac(x):
        c = np.abs(np.correlate(x, 1-x, mode='full'))
        cp = c ** (p-1)
        s = np.sum(c ** p)
        dc = cp / (s ** (1.0/p))
        grad = ps_grad(c, dc, x, n)
        diff = x.sum() - n/2.0
        grad += 2 * mu * diff
        return grad
    
    bounds = [(0.0, 1.0)] * n
    
    def cb(xk):
        return time.time() > deadline
    
    res = minimize(fun, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                            'maxfun': 50000, 'callback': cb})
    return res.x

    # Convert to binary
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:

                msk = (im >= 0) & (im < n)
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

```python
def compute_gradient(h, n, p_power):
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    c_pos = np.maximum(c, 0)  # safety, c should be >= 0
    cp = c_pos ** (p_power - 1)
    s = np.sum(c_pos ** p_power)
    if s < 1e-30:
        return np.zeros(n)
    dc = cp / (s ** (1.0 / p_power))
    # dc[k] corresponds to p = k - (n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        # First sum: p in [max(-(n-1), -i), min(n-1, n-1-i)]
        p_lo1 = max(-(n-1), -i)
        p_hi1 = min(n-1, n-1-i)
        if p_lo1 <= p_hi1:
            ps = np.arange(p_lo1, p_hi1+1)
            grad[i] += np.dot(dc[ps + (n-1)], 1.0 - h[i + ps])
        
        # Second sum: p in [i-(n-1), min(n-1, i)]
        p_lo2 = i - (n-1)
        p_hi2 = min(n-1, i)
        if p_lo2 <= p_hi2:
            ps2 = np.arange(p_lo2, p_hi2+1)
            grad[i] -= np.dot(dc[ps2 + (n-1)], h[i - ps2])
    
    return grad
```
