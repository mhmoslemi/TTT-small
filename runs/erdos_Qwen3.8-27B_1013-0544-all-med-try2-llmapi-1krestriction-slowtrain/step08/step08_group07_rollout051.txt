            lo = max(0 - h[i], -h[j])  # delta >= -h[i], delta >= -h[j]... wait

            # h[i] + delta <= 1 => delta <= 1 - h[i]
            # h[j] - delta >= 0 => delta <= h[j]
            # h[j] - delta <= 1 => delta >= h[j] - 1
            lo = max(-h[i], h[j] - 1)
            hi = min(1 - h[i], h[j])
            if lo >= hi:
                continue
            # Minimize max_k(corr[k] + delta * D[k]) over delta in [lo, hi]
            # Same bisection as before
            f_lo = np.max(corr + lo * D)
            f_hi = np.max(corr + hi * D)
            if min(f_lo, f_hi) >= np.max(corr) - 1e-15:
                continue
            a, b = lo, hi
            for _ in range(40):
                mid = (a + b) / 2.0
                f_mid = np.max(corr + mid * D)
                active = (corr + mid * D) >= f_mid - 1e-15
                sg_min = D[active].min()
                sg_max = D[active].max()
                if sg_min > 1e-14:
                    b = mid
                elif sg_max < -1e-14:
                    a = mid
                else:
                    a = b = mid
                    break
            delta_opt = (a + b) / 2.0
            if abs(delta_opt) > 1e-14:
                new_val = np.max(corr + delta_opt * D)
                if new_val < np.max(corr) - 1e-15:
                    corr += delta_opt * D
                    h[i] += delta_opt
                    h[j] -= delta_opt
                    improved = True
        if not improved:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_B(h, n):
    m_range = np.arange(-(n-1), n)
    B = np.zeros((n, 2*n-1))
    for i in range(n):
        k1 = i - m_range
        mask1 = (k1 >= 0) & (k1 < n)
        B[i, mask1] += (1.0 - h[k1[mask1]])
        k2 = i + m_range
        mask2 = (k2 >= 0) & (k2 < n)
        B[i, mask2] -= h[k2[mask2]]
    return B

def cd_individual(h, n, max_sweeps=300, deadline=None):
    m_range = np.arange(-(n-1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2*n-1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            val0 = np.max(corr)
            val1 = np.max(corr + B)
            if val1 >= val0:
                # x=0 is better or equal
                # Check if moving from 0 helps
                if val0 - val1 < 1e-16:
                    continue
            # Binary search for optimal x in [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                f_mid = np.max(corr + B * mid)
                active = (corr + B * mid) >= f_mid - 1e-14
                sg_min = B[active].min() if active.any() else 0.0
                sg_max = B[active].max() if active.any() else 0.0
                if sg_min > 1e-14:
                    hi = mid
                elif sg_max < -1e-14:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            x_opt = np.clip((lo + hi) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h

def level_set_slsqp(h, n, max_rounds=30, deadline=None):
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    best = corr.max()
    
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        
        t_raw = corr.max()
        margin = max(0.05 * t_raw, 1.0)
        w = np.maximum(0.0, corr - (t_raw - margin))
        w = 1.0 + 50.0 * w / max(w.max(), 1e-10)
        
        # Define objective and gradient for the subproblem
        def obj_and_grad(x):
            c = np.correlate(x, 1-x, mode='full')
            B = compute_B(x, n)
            v = w * c
            grad = (2.0 * B.T @ v)
            # Add constraint gradient
            f = np.sum(w * c * c)
            return f, grad
        
        # Use L-BFGS-B with equality constraint (sum = n/2)
        # Since L-BFGS-B doesn't support constraints, we'll use SLSQP
        def obj(x):
            return obj_and_grad(x)[0]
        def grad(x):
            return obj_and_grad(x)[1]
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-12})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            h_new = h
        
        corr_new = np.correlate(h_new, 1-h_new, mode='full')
        if corr_new.max() < best - 1e-14:
            h = h_new
            corr = corr_new
            best = corr.max()
        else:
            # Try projected gradient as fallback
            B = compute_B(h, n)
            v = w * corr
            g = 2.0 * B.T @ v
            lr = 0.01 / (np.max(np.abs(g)) + 1e-10)
            h_new = project_feasible(h - lr * g, n)
            corr_new = np.correlate(h_new, 1-h_new, mode='full')
            if corr_new.max() < best - 1e-14:
                h = h_new
                corr = corr_new
                best = corr.max()
    
    return h, best * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_B(h, n):
    m_range = np.arange(-(n - 1), n)
    B = np.zeros((n, 2 * n - 1))
    for i in range(n):
        k1 = i - m_range
        mask1 = (k1 >= 0) & (k1 < n)
        B[i, mask1] += (1.0 - h[k1[mask1]])
        k2 = i + m_range
        mask2 = (k2 >= 0) & (k2 < n)
        B[i, mask2] -= h[k2[mask2]]
    return B

def cd_individual(h, n, max_sweeps=300, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            val0 = np.max(corr)
            val1 = np.max(corr + B)
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                f_mid = np.max(corr + B * mid)
                active = (corr + B * mid) >= f_mid - 1e-14
                sg_min = B[active].min() if active.any() else 0.0
                sg_max = B[active].max() if active.any() else 0.0
                if sg_min > 1e-14:
                    hi = mid
                elif sg_max < -1e-14:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            x_opt = np.clip((lo + hi) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h

def cd_pairwise(h, n, max_sweeps=30, deadline=None, n_pairs=3000, seed=0):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        idx = np.random.RandomState(sweep * 1000 + seed).choice(n * (n - 1) // 2, size=min(n_pairs, n * (n - 1) // 2))
        pair_list = np.unravel_index(idx, (n, n))
        for i, j in zip(pair_list[0], pair_list[1]):
            if i == j:
                continue
            B_i = np.zeros(2 * n - 1)
            k1 = i - m_range
            mk = (k1 >= 0) & (k1 < n)
            B_i[mk] += (1.0 - h[k1[mk]])
            k2 = i + m_range
            mk = (k2 >= 0) & (k2 < n)
            B_i[mk] -= h[k2[mk]]
            B_j = np.zeros(2 * n - 1)
            k1 = j - m_range
            mk = (k1 >= 0) & (k1 < n)
            B_j[mk] += (1.0 - h[k1[mk]])
            k2 = j + m_range
            mk = (k2 >= 0) & (k2 < n)
            B_j[mk] -= h[k2[mk]]

            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo >= hi - 1e-15:
                continue
            cur_max = np.max(corr)
            f_lo = np.max(corr + lo * D)
            f_hi = np.max(corr + hi * D)
            if min(f_lo, f_hi) >= cur_max - 1e-14:
                continue
            a, b = lo, hi
            for _ in range(40):
                mid = (a + b) / 2.0
                f_mid = np.max(corr + mid * D)
                active = (corr + mid * D) >= f_mid - 1e-14
                sg_min = D[active].min() if active.any() else 0.0

                if sg_min > 1e-14:
                    b = mid
                elif sg_max < -1e-14:
                    a = mid
                else:
                    a = b = mid
                    break

            if abs(delta) > 1e-14:
                new_max = np.max(corr + delta * D)
                if new_max < cur_max - 1e-14:
                    corr += delta * D
                    h[i] += delta
                    h[j] -= delta
                    improved = True
        if not improved:
            break
    return h

def level_set_opt(h, n, max_rounds=40, deadline=None):
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best = corr.max()
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        t_raw = corr.max()
        spread = corr.max() - corr.min()
        margin = max(spread * (0.3 ** (rnd * 0.3)), 0.001 * t_raw)
        w = np.maximum(0.0, corr - (t_raw - margin))
        w = 1.0 + 80.0 * w / max(w.max(), 1e-10)

        def obj(x):
            c = np.correlate(x, 1 - x, mode='full')
            return float(np.sum(w * c * c))

        def grad(x):
            c = np.correlate(x, 1 - x, mode='full')
            B = compute_B(x, n)
            return 2.0 * B.T @ (w * c)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-14})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            h_new = h
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        if corr_new.max() < best - 1e-14:
            h = h_new
            corr = corr_new
            best = corr.max()
    return h, best * dx

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-16, deadline=None):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
