        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()

def grad_shift(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:nv])
        grad[:nv] -= h[k:]

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()

    s = x.sum() - n / 2.0

    best_c = compute_c5(h, n)
    m = np.zeros(n)

        h_interp = np.interp(x_new, x_old, best_h, left=best_h[0], right=best_h[-1])
        h_interp = project_feasible(h_interp, n_try)
        # Also try from sine init
        h_sine = sine_init(n_try, 3 + rng.rand() * 2, rng.rand() * 6.28)
        # Pick better starting point
        c_interp = compute_c5(h_interp, n_try)
        c_sine = compute_c5(h_sine, n_try)
        h_start = h_interp if c_interp < c_sine else h_sine
        # LBFGS
        h_new, c_new = lbfgs_round(h_start, n_try, 5000, deadline - 200, maxiter=100)
        # Adam PGD
        rem = deadline - time.time() - 150
        if rem > 20:
            h_new, c_new = adam_pgd(h_new, n_try, time.time() + min(40, rem), top_k=40, alpha=80)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            best_n = n_try

    # Phase 3: More LBFGS on best
    if time.time() < deadline - 150:
        for alpha in [10000, 30000, 80000]:
            if time.time() > deadline - 120:
                break
            h_new, c_new = lbfgs_round(best_h, best_n, alpha, deadline - 120, maxiter=150)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 4: Adam PGD
    if time.time() < deadline - 80:
        t4 = min(time.time() + 60, deadline - 40)
        best_h, best_c = adam_pgd(best_h, best_n, t4, top_k=50, alpha=120)

    # Phase 5: Flip local search (for near-binary solutions)
    if time.time() < deadline - 40:
        t5 = min(time.time() + 30, deadline - 15)
        best_h, best_c = flip_local_search(best_h, best_n, t5)

    # Phase 6: Pair polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1, radius=35)

    # Final check: also try one more Adam pass
    if time.time() < deadline - 2:
        best_h, best_c2 = adam_pgd(best_h, best_n, deadline - 1, top_k=30, alpha=200)
        if best_c2 < best_c:
            best_c = best_c2
            best_h = best_h

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
