import time

    c = np.correlate(h, 1 - h, mode='full')

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:

            if tot < 1e-15:

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    logsum = smax + np.log(np.sum(np.exp(sh - smax)))
    return (m + logsum / alpha) * (2.0 / n)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:

            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    val = (m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha) * (2.0 / n)
    s = x.sum() - n / 2.0
    return val + mu * s * s

    c = np.correlate(x, 1 - x, mode='full')

    sh = alpha * (c - m)
    smax = sh.max()

    s = x.sum() - n / 2.0
    return val + mu * s * s

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def pen_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    val = (m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha) * (2.0 / n)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    df = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * df
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

                if c_new < best_c:

                    h = h_new
                mu *= 4
            except Exception:
                pass
    return h, best_c

    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]

        idx = np.where(h_bin == 0)[0]

def cosine_init(n, rng):
