def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n, rng):

    sorted_idx = np.argsort(h)[::-1]

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        # Soft max for the top values
        sh = alpha * (g - m)
        sm = sh.max()
        soft = (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return m + soft
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

So TermA delta = (1 if i+k==j else (1-h[i+k])) for valid i+k. ✓

One more consideration: the simulated annealing approach. The issue is that for a random swap, the probability of improvement is low (most swaps will increase the max). So the acceptance rate will be very low at low temperatures, and the search will essentially stop.

A better approach might be:

2. Then do a "kick" (random perturbation) and greedy again
