import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coordinate_descent(h, n, deadline, max_rounds=5):
    """1D coordinate descent: for each coordinate, find optimal value via ternary/golden search."""

def subgradient_step(h, n, step_size):
    """One projected subgradient step on the active constraint."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k_star = int(np.argmax(c))
    # k_star is the index in the full correlate output (0 to 2n-2)
    # The lag s = k_star - (n-1)
    s = k_star - (n - 1)
    # Compute gradient of c_{k_star} w.r.t. h
    grad = np.zeros(n)
    if s >= 0:
        # c_s = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
        # d/dh[i]: from h[j+s] term when j+s=i -> j=i-s (valid if 0<=i-s<=n-1-s, i.e., s<=i<=n-1)
        #          from (1-h[j]) term when j=i (valid if 0<=i<=n-1-s)
        for i in range(n):
            g = 0.0
            if i >= s and i - s <= n - 1 - s:
                g += (1.0 - h[i - s])
            if 0 <= i <= n - 1 - s:
                g -= h[i + s]
            grad[i] = g
    else:
        # s < 0, let m = -s > 0
        m = -s
        # c_{-m} = sum_{j=m}^{n-1} h[j-m]*(1-h[j])
        # d/dh[i]: from h[j-m] term when j-m=i -> j=i+m (valid if m<=i+m<=n-1, i.e., 0<=i<=n-1-m)

            g = 0.0
            if 0 <= i <= n - 1 - m and i + m <= n - 1:
                g += (1.0 - h[i + m])
            if m <= i <= n - 1:
                g -= h[i - m]
            grad[i] = g
            
    grad *= dx
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

    return h, best_c

            alloc = min(50, (deadline - time.time() - 45) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_block(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final intensive polish
    if time.time() < deadline - 3:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                         max_passes=600)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                              radius=80, max_passes=300)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
