    """Compute new C5 after swapping h[a]=1 to 0 and h[b]=0 to 1.
    corr is the current correlation array (2n-1 length, unscaled).
    Returns (new_c5_unscaled, delta_array).
    """
    delta = np.zeros(2*n-1)
    
    # Effect of removing a (h[a]: 1->0)
    # a as first index: i=a, need 0<=a+s<n where s=k-(n-1), so k in [n-1-a, 2n-2-a]
    s_min_a1 = max(0, n-1-a)
    s_max_a1 = min(2*n-2, 2*n-2-a)
    for k in range(s_min_a1, s_max_a1+1):

        if a+s == b:
            delta[k] += -1  # was 1*(1-0)=1, now 0
        elif a+s == a:  # s=0
            delta[k] += 0  # was 1*0=0, now 0
        else:
            delta[k] += -(1 - h[a+s])
