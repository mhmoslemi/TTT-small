        #              - h[j-s] (if j-s valid, from cross term where i=j-s, i+s=j)
        # Actually more carefully:
        # c[k] = sum_{i=max(0,-s)}^{min(n-1, n-1-s)} h[i]*(1-h[i+s])

        # dc/dh[j] for j in [lo+s, hi+s] (i.e., j = i+s for some valid i): -h[j-s]

                g[j] += (1.0 - h[j+s])
            for j in range(lo+s, hi+s+1):
                g[j] -= h[j-s]

            if tot < 1e-14: break
            h += (diff / tot) * h

            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > 60:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx + 1, len(steps) - 1)
            if stall > 60:
                break
    return best_h, best_c

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 15:
        for a in [100, 1000, 10000]:
            if time.time() > deadline - 12:
                break
            ht = slsqp_round(best_h, n, a, maxiter=1500)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c, best_h = c, ht

    remaining = deadline - time.time() - 12
    if remaining > 10:
        sa_time = min(remaining * 0.55, 350)
        best_h, best_c = sa_with_segments(best_h, n, time.time() + sa_time, rng)

    if time.time() < deadline - 8:
        for a in [50000, 200000, 1000000]:
            if time.time() > deadline - 4:
                break
            ht = slsqp_round(best_h, n, a, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c, best_h = c, ht

    if time.time() < deadline - 2:
        best_h, best_c = directed_polish(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        n = n_init
        alloc = (deadline - t0) * 0.42
        best_h, best_c = full_optimize(h_init, n, t0 + alloc, rng)
        x_init = np.linspace(0, 2, n)
        for n_try in [200, 300]:
            if time.time() > deadline - 60:
                break
            remaining = deadline - time.time()
            alloc = min(120, remaining * 0.22)
            if alloc < 35:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = full_optimize(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    else:
        n = 250
        h_start = project_feasible(np.full(n, 0.5), n)
        best_h, best_c = full_optimize(h_start, n, deadline - 5, rng)

    if best_h is not None and time.time() < deadline - 25:
        n = len(best_h)
        n_restarts = max(2, int((deadline - time.time() - 15) / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.004 + 0.04 * (r / max(1, n_restarts))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 12) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = full_optimize(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        best_h, best_c = directed_polish(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
