    best_h = h.copy()

    T_min = 1e-7

        i = rng.randint(n)
        j = rng.randint(n)

            continue
        d = rng.uniform(0.0001, 0.01)
        if rng.uniform() < 0.5:

        else:
            di, dj = -d, d

            continue

        h[j] += dj

                h[j] -= dj

def fast_pair_search(h, n, deadline, deltas):
    """Pair search with O(n) per pair evaluation using incremental updates."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')  # unscaled
    best_c = corr.max() * dx
    r = np.arange(2*n-1)
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    # Compute f(r) for this pair
                    # Δcorr[r] = d * [(1-h[i-r+n-1]) - (1-h[j-r+n-1]) - h[i+r-n+1] + h[j+r-n+1]]
                    f = np.zeros(2*n-1)
                    
                    idx_a = i - r + n - 1
                    va = (idx_a >= 0) & (idx_a < n)
                    if va.any():
                        f[va] += (1 - h[idx_a[va]])
                    
                    idx_b = j - r + n - 1
                    vb = (idx_b >= 0) & (idx_b < n)
                    if vb.any():

                    idx_c = i + r - n + 1
                    vc = (idx_c >= 0) & (idx_c < n)
                    if vc.any():
                        f[vc] -= h[idx_c[vc]]
                    
                    idx_d = j + r - n + 1
                    vd = (idx_d >= 0) & (idx_d < n)
                    if vd.any():
                        f[vd] += h[idx_d[vd]]

Let me also add the "optimal step" computation. For each pair, instead of using a fixed d, we can find the best d:
- The function g(d) = max(corr + d*f) is convex in d
- We want to minimize g(d) subject to d ∈ [max(-h[i], -(1-h[j])), min(1-h[i], h[j])]
- (for the direction h[i]+=d, h[j]-=d)

We can find the minimum using the fact that it's at a breakpoint: d* = (c_k2 - c_k1) / (f_k1 - f_k2) for some pair of indices (k1, k2) where f_k1 ≠ f_k2.

A practical approach: 
1. At d=0, find k1 = argmax(corr)
2. If f[k1] == 0, no improvement (the max lag doesn't change with d)
3. If f[k1] > 0, the max increases with d, so we want to decrease d (i.e., try h[i]-=d)
