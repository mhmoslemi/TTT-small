    best_c = cur_c
    stall = 0

    while time.time() < deadline:
        g = compute_g(h, n)
        top_idx = np.argsort(g)[::-1][:top_lags]

        improved = False
        for idx in top_idx:
            if time.time() > deadline:
                return best_h, best_c
            lag = int(idx) - (n - 1)
            if lag == 0:
                continue
            if lag > 0:
                nl = n - lag
                if nl <= 0:
                    continue
                cands = np.where((h[:nl] == 1) & (h[lag:] == 0))[0]
                zero_m = (h == 0)
                bad = np.zeros(n, dtype=bool)
                bad[:nl] = (h[lag:] == 1)
                valid_j = np.where(zero_m & ~bad)[0]
            else:
                to = -lag
                nl = n - to
                if nl <= 0:
                    continue
                cands = np.where((h[to:] == 1) & (h[:nl] == 0))[0] + to
                zero_m = (h == 0)
                bad = np.zeros(n, dtype=bool)
                bad[to:] = (h[:nl] == 1)
                valid_j = np.where(zero_m & ~bad)[0]

            if len(cands) == 0 or len(valid_j) == 0:
                continue

            i = int(cands[rng.randint(len(cands))])
            j = int(valid_j[rng.randint(len(valid_j))])
            if i == j:
                continue

            h[i] = 0
            h[j] = 1
            new_c = compute_c5(h, n)
            if new_c < cur_c - 1e-15:
                cur_c = new_c
                stall = 0
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] = 1
                h[j] = 0
                stall += 1

        if stall >= perturb_interval:
            nf = rng.randint(2, 7)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(nf):
                a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
                h[a], h[b] = h[b], h[a]
            cur_c = compute_c5(h, n)
            stall = 0

    return best_h, best_c

def double_swap_search(h0, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = project_binary(h0, n)
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c

    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(300):
            if time.time() > deadline:
                return best_h, best_c
            a = int(ones[rng.randint(len(ones))])
            b = int(ones[rng.randint(len(ones))])
            c = int(zeros[rng.randint(len(zeros))])
            d = int(zeros[rng.randint(len(zeros))])
            if a == b or c == d:
                continue
            h[a] = 0
            h[b] = 0
            h[c] = 1
            h[d] = 1
            new_c = compute_c5(h, n)
            if new_c < cur_c - 1e-15:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[a] = 1
                h[b] = 1
                h[c] = 0
                h[d] = 0
    return best_h, best_c

def smooth_obj(x, n, alpha):
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max()
    shifted = g - m
    return m + np.log(np.exp(shifted - shifted.max()).sum()) / alpha

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def continuous_polish(h0, n, deadline, alpha=50000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}]
    try:
        res = minimize(lambda x: smooth_obj(x, n, alpha), h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-18})
        h = project_feasible(res.x, n)
        return h, compute_c5(h, n)
    except Exception:
        return h0, compute_c5(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = project_binary(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Intensive lag-balanced search from parent (30%)
    p1 = t0 + 0.30 * (budget_s - 3)
    h, c = lag_balanced_search(best_h, n, p1, seed=seed, top_lags=6)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Multiple diverse starts (30%)
    p2 = t0 + 0.60 * (budget_s - 3)
    num_starts = 6
    per_start = (p2 - time.time()) / max(num_starts, 1)
    for s in range(num_starts):
        if time.time() > p2 - 5:
            break
        if s == 0:
            rb = quasi_random_init(n, seed + 100)
        else:
            rb = random_binary_init(n, rng)
        te = min(t0 + (s + 1) * per_start, p2 - 2)
        h, c = lag_balanced_search(rb, n, te, seed=seed + 1000 + s * 7, top_lags=4)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Double-swap search on best (10%)
    p3 = t0 + 0.70 * (budget_s - 3)
    if time.time() < p3:
        h, c = double_swap_search(best_h, n, p3, seed=seed + 5000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: More lag-balanced with wider top-lags (10%)
    p4 = t0 + 0.80 * (budget_s - 3)
    if time.time() < p4:
        h, c = lag_balanced_search(best_h, n, p4, seed=seed + 9000, top_lags=8)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Refine with smaller perturb (8%)
    p5 = t0 + 0.88 * (budget_s - 3)
    if time.time() < p5:
        h, c = lag_balanced_search(best_h, n, p5, seed=seed + 12000, top_lags=3,
                                   perturb_interval=5000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Continuous polish (8%)
    p6 = t0 + 0.96 * (budget_s - 3)
    if time.time() < p6:
        h, c = continuous_polish(best_h, n, p6, alpha=100000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Final: verify and ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
