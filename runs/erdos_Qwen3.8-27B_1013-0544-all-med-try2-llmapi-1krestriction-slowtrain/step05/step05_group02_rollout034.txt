def max_corr_gradient(h, n, num_top=3):

    c = get_corr(h, n)
    cmax = c.max()
    # Get indices of top lags

            g[:n-s] += wi * (1.0 - h[s:])

def lag_grad_raw(h, n, s):
    """Gradient of raw correlation (without dx) at shift s."""
    g = np.zeros(n)
    if s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    elif s < 0:
        sp = -s
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def smooth_grad(h, n, alpha):
    """Gradient of smooth max approximation."""
    dx = 2.0 / n
    c = get_corr(h, n)
    # Softmax weights
    c_shift = alpha * c - alpha * c.max()
    w = np.exp(c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over argmax
        w = np.zeros(len(c))
        w[c == c.max()] = 1.0
        w /= w.sum()
    else:
        w /= w_sum
    
    # Weighted gradient
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-8:
            continue
        s = k - (n - 1)
        g += w[k] * lag_grad_raw(h, n, s)
    g *= dx
    # Zero-mean projection for sum constraint
    g -= g.mean()
    return g

def smooth_value(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shift = alpha * c - alpha * c.max()
    return c.max() + np.log(np.sum(np.exp(c_shift))) / alpha

        if gn < 1e-14:
            break
        
        # Nesterov-like momentum
        v = beta * v - lr * g / gn
        h_trial = project_feasible(h + v, n)
        
        # Backtracking line search on actual C5
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:

            best_h = h_trial.copy()

        else:

            h_trial = project_feasible(h + LR * g / gn, n)
            c_trial = compute_c5(h_trial, n)

            else:

                v = np.zeros(n)
                lr = 0.1

    return best_h, best_c
