```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def find_opt_t(a, b, lam):
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        v = a * mid + b
        mx = v.max()
        act = v >= mx - 1e-9
        if (a[act] + lam).max() < -1e-13:
            lo = mid
        elif (a[act] + lam).min() > 1e-13:
            hi = mid
        else:
            break
    t = 0.5 * (lo + hi)
    f0, f1 = b.max(), (a + b).max()
    ft = (a * t + b).max()
    if f0 <= ft and f0 <= f1:
        return 0.0
    if f1 <= ft and f1 <= f0:
        return 1.0
    return t

def coord_desc(h, n, n_sweeps, lam_lr, seed, deadline):
    nc = 2 * n - 1
    lags = np.arange(nc) - (n - 1)
    rng = np.random.RandomState(seed)
    lam = 0.0
    corr = np.correlate(h, 1 - h, mode='full').copy()
    dx = 2.0 / n
    best_c = float(corr.max()) * dx
    best_h = h.copy()
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            ip = i + lags
            im = i - lags
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            hp = np.zeros(nc)
            hm = np.zeros(nc)
            hp[vp] = h[ip[vp]]
            hm[vm] = h[im[vm]]
            a = 1.0 - hp - hm
            b = corr - h[i] * a
            t_new = find_opt_t(a, b, lam)
            d = t_new - h[i]
            if abs(d) > 1e-14:
                h[i] = t_new
                corr += d * a
        lam += lam_lr * (h.sum() - 0.5 * n)
        lam = max(-5.0, min(5.0, lam))
        c = float(corr.max()) * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def prox_grad(h, n, alpha, n_iters, step, deadline):
    dx = 2.0 / n
    nc = 2 * n - 1
    lags = np.arange(nc) - (n - 1)
    i_idx = np.arange(n)[:, None]
    lag_idx = lags[None, :]
    v1 = (i_idx - lag_idx >= 0) & (i_idx - lag_idx < n)
    v2 = (i_idx + lag_idx >= 0) & (i_idx + lag_idx < n)
    w1 = np.where(v1, 1.0, 0.0)
    w2 = np.where(v2, 1.0, 0.0)
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        mc = c.max()
        ec = np.exp(alpha * (c - mc))
        sw = ec.sum()
        w = ec / sw
        s1 = (w1 * w[None, :]) @ h
        s2 = (w2 * w[None, :]) @ h
        grad = (dx / alpha) * (1.0 - s1 - s2)
        h = h - step * grad
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_fn(x, alpha, n):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_con(x, n):
    return x.sum() - 0.5 * n

def slsqp_polish(h, n, alpha, maxiter, deadline):
    res = minimize(slsqp_fn, h, method='SLSQP', args=(alpha, n),
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 250
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Proximal gradient with increasing alpha
    p1 = t0 + 0.25 * (budget_s - 5)
    for alpha in [100, 500, 2000, 10000]:
        if time.time() > p1:
            break
        step = 2.0 / (alpha * n)
        hh, cc = prox_grad(best_h.copy(), n, alpha, 500, step, p1)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
    
    # Phase 2: Coordinate descent
    p2 = t0 + 0.55 * (budget_s - 5)
    for lr in [0.1, 0.05, 0.01, 0.005, 0.001]:
        if time.time() > p2 - 10:
            break
        hh, cc = coord_desc(best_h.copy(), n, 300, lr, seed, p2 - 10)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
    
    # Phase 3: SLSQP polish
    p3 = t0 + 0.70 * (budget_s - 5)
    for alpha in [50000, 500000]:
        if time.time() > p3:
            break
        hh = slsqp_polish(best_h.copy(), n, alpha, 1000, p3)
        cc = compute_c5(hh, n)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
    
    # Phase 4: Restarts with perturbation
    p4 = t0 + 0.90 * (budget_s - 5)
    for k in range(30):
        if time.time() > p4:
            break
        hh = best_h + 0.05 * rng.randn(n)
        hh = project_feasible(hh, n)
        hh, cc = coord_desc(hh, n, 100, 0.05, seed + 100 + k, p4)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
    
    # Phase 5: Final refinement
    hh, cc = coord_desc(best_h.copy(), n, 300, 0.005, seed + 200, dl)
    if cc < best_c:
        best_c = cc
        best_h = hh.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
