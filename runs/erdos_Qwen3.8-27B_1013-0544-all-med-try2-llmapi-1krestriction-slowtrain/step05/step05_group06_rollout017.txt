        m = g.max()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_val(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha


def smooth_max_w(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    return w


def g_jac(h, n):
    nsh = 2 * n - 1
    J = np.zeros((n, nsh))
    for m in range(nsh):
        d = m - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        cnt = i_hi - i_lo
        if cnt > 0:
            J[i_lo:i_hi, m] += (1.0 - h[i_lo + d:i_hi + d])
            J[i_lo + d:i_hi + d, m] -= h[i_lo:i_hi]
    return J


def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max_val(g, alpha)

    def jac(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        w = smooth_max_w(g, alpha)
        J = g_jac(x, n)
        return (J * w[None, :]).sum(axis=1)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h


def pair_descent(h, n, deadline, K=12, noise_every=15, noise_scale=0.015):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    nsh = 2 * n - 1
    sweep = 0
    stall = 0
    while time.time() < deadline:
        g = compute_g(h, n)
        top_k = np.argsort(g)[::-1][:K]
        F = g_jac(h, n)
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                cands = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if m1 == m2:
                            continue
                        t1, t2 = top_k[m1], top_k[m2]
                        ds = s[t2] - s[t1]
                        if abs(ds) > 1e-15:
                            dc = (g[t1] - g[t2]) / ds
                            if d_lo < dc < d_hi:
                                cands.append(dc)
                best_d = 0.0
                best_val = g.max()
                for d in cands:
                    d = max(d_lo, min(d_hi, d))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-12:
                        best_val = val
                        best_d = d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    g += best_d * s
                    improved += 1
                    if improved % 50 == 0:
                        top_k = np.argsort(g)[::-1][:K]
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        sweep += 1
        if improved == 0:
            stall += 1
            if stall > 6 and noise_scale > 0.001:
                mask = np.random.random(n) < 0.08
                h[mask] += np.random.uniform(-noise_scale, noise_scale, int(mask.sum()))
                h = project_feasible(h, n)
                noise_scale *= 0.7
                stall = 0
            elif stall > 12:
                break
        else:
            stall = 0
            noise_scale = max(0.001, 0.015 * (0.95 ** (sweep // noise_every)))
    return best_h, best_c


def block_descent(h, n, deadline, alpha=100000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    sizes = [20, 15, 30, 10, 25, 12]
    si = 0
    while time.time() < deadline:
        bs = sizes[si % len(sizes)]
        si += 1
        if n - bs <= 0:
            bs = max(5, n // 4)
        start = np.random.randint(0, max(1, n - bs))
        idx = np.arange(start, start + bs)
        mask = np.ones(n, dtype=bool)
        mask[idx] = False
        target = n / 2.0 - best_h[mask].sum()
        h_fixed = best_h.copy()
        x0 = best_h[idx].copy()

        def bobj(x):
            ht = h_fixed.copy()
            ht[idx] = x
            g = np.correlate(ht, 1 - ht, mode='full') * dx
            return smooth_max_val(g, alpha)

        bounds = [(0.0, 1.0)] * bs
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(bs)}]
        try:
            res = minimize(bobj, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 80, 'ftol': 1e-13})
            h_new = h_fixed.copy()
            h_new[idx] = np.clip(res.x, 0, 1)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-11:
                best_h = h_new
                best_c = c
                stall = 0
            else:
                stall += 1
        except Exception:
            stall += 1
        if stall > 150:
            break
    return best_h, best_c
