import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(v, n):
    """Project v onto {h : 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(v - lambda, 0, 1)) = target
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Redistribute among non-bound variables
        mask = (h > 1e-12) & (h < 1 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n, alpha_smooth=100.0):
    """Compute smooth approximation of subgradient of max_k corr_k"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Soft-max weighting
    m = corr.max()
    if alpha_smooth < 1e-6:
        active = (corr == m).astype(float)
        w = active / active.sum()
    else:
        exps = np.exp(alpha_smooth * (corr - m))
        w = exps / exps.sum()
    
    # Compute weighted gradient
    grad = np.zeros(2*n - 1)  # This won't work, grad should be same size as h
    grad = np.zeros(n)
    
    # For each k in range(2n-1), j = k - (n-1)
    # d(g_k)/d(h[m]) = (1-h[m+j]) * I[0<=m+j<n] - h[m-j] * I[0<=m-j<n]
    # We need to sum over k with weights w[k]
    
    # Let's compute this efficiently
    # For a fixed m, the contribution from lag j (where j = k - (n-1)) is:
    # w[k] * [(1-h[m+j]) * I[0<=m+j<n] - h[m-j] * I[0<=m-j<n]]
    
    # Let's iterate over j values
    for j in range(-(n-1), n):
        k = j + (n-1)
        wk = w[k]
        if wk < 1e-10:
            continue
        # First term: (1-h[m+j]) for m where 0<=m<n and 0<=m+j<n
        if j >= 0:
            # m+j < n means m < n-j
            # First term: for m in [0, n-j): (1-h[m+j])
            grad[:n-j] += wk * (1 - h[j:n])
            # Second term: -h[m-j] for m where m-j >= 0, i.e., m >= j
            # and m < n (always true since m is index of h)
            grad[j:] -= wk * h[:n-j]
        else:
            l = -j  # l > 0
            # m+j < n means m < n+l (always true since m < n)
            # First term: for m in [0, n): (1-h[m-l]) but we need m-l >= 0, i.e., m >= l
            grad[l:] += wk * (1 - h[:n-l])
            # Second term: -h[m-j] = -h[m+l] for m where m+l < n, i.e., m < n-l
            grad[:n-l] -= wk * h[l:]
    
    return grad

def subgradient_descent(h, n, t_max, step_scale=0.1):
    """Subgradient descent with momentum"""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    velocity = np.zeros(n)
    momentum = 0.9
    
    for t in range(int(t_max)):
        step = step_scale / (1.0 + 0.01 * t)
        grad = subgradient(h, n, alpha_smooth=50.0)
        velocity = momentum * velocity - step * grad
        h_new = project_feasible(h + velocity, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def linearize_step(h, n, damping=1.0):
    """One step of successive linear programming"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')  # length 2n-1
    # Compute gradients
    grad = np.zeros((2*n-1, n))
    for j in range(-(n-1), n):
        k = j + (n-1)
        # d(g_k)/d(h[m]) = (1-h[m+j]) * I[0<=m+j<n] - h[m-j] * I[0<=m-j<n]
        if j >= 0:
            grad[k, :n-j] += (1 - h[j:n])
            grad[k, j:] -= h[:n-j]
        else:
            l = -j
            grad[k, l:] += (1 - h[:n-l])
            grad[k, :n-l] -= h[l:]
    
    # LP: minimize t
    # subject to: grad[k] @ delta + corr[k]*dx - t <= 0 for each k
    #             0 <= h + delta <= 1
    #             sum(delta) = 0 (since sum(h) is already n/2)
    #

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(v, n):
    target = n / 2.0
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

def compute_grads(h, n):
    """Compute gradients of all correlation components. Returns array of shape (2n-1, n)."""
    m = 2 * n - 1
    grad = np.zeros((m, n))
    for j in range(-(n - 1), n):

            l = -j

            grad[k, :n-l] -= h[l:]
    return grad

    c_obj[n] = 1.0

def subgradient_step(h, n, step_size, top_k=5):
    """One subgradient step with top-k weighting."""
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    # Get top-k lags
    top_indices = np.argsort(corr)[-top_k:]
    top_vals = corr[top_indices]
    # Soft-max weights among top-k
    weights = np.exp(100 * (top_vals - top_vals.max()))
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        j = idx - (n - 1)
        if j >= 0:
            grad[:n-j] += w * (1.0 - h[j:n])
            grad[j:] -= w * h[:n-j]
        else:
            l = -j
            grad[l:] += w * (1.0 - h[:n-l])
            grad[:n-l] -= w * h[l:]
    
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

5. Repeat until no violation

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(v, n):
    target = n / 2.0
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grads(h, n):
    m = 2 * n - 1
    grad = np.zeros((m, n))
    for j in range(-(n - 1), n):
        k = j + (n - 1)
        if j >= 0:
            grad[k, :n - j] += (1.0 - h[j:n])
            grad[k, j:] -= h[:n - j]
        else:
            l = -j
            grad[k, l:] += (1.0 - h[:n - l])
            grad[k, :n - l] -= h[l:]
    return grad

def slp_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    grad = compute_grads(h, n)
    m = 2 * n - 1
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    A_ub[:m, :n] = grad
    A_ub[:m, n] = -1.0
    b_ub[:m] = -corr * dx
    for i in range(n):
        A_ub[m + i, i] = -1.0
        b_ub[m + i] = -h[i]
        A_ub[m + n + i, i] = 1.0
        b_ub[m + n + i] = 1.0 - h[i]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * (n + 1), method='highs')
    if res.success:
        return h + res.x[:n]
    return h.copy()

def slp_optimize(h, n, deadline, max_iters=10000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_try = slp_step(h, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = project_feasible(h_try, n)
            h = best_h.copy()
            stall = 0
        else:
            h_half = project_feasible(h + 0.5 * (h_try - h), n)
            c_half = compute_c5(h_half, n)
            if c_half < best_c - 1e-14:
                best_c = c_half
                best_h = h_half.copy()
                h = h_half.copy()
                stall = 0
            else:
                stall += 1
                if stall > 200:
                    break
    return best_h, best_c

def subgrad_optimize(h, n, deadline, base_step=0.05, top_k=5):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    vel = np.zeros(n)
    mom = 0.85
    for t in range(200000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        step = base_step / (1.0 + 0.005 * t)
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        w = np.exp(50.0 * (top_vals - top_vals.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            j = idx - (n - 1)
            if j >= 0:
                grad[:n - j] += wi * (1.0 - h[j:n])
                grad[j:] -= wi * h[:n - j]
            else:
                l = -j
                grad[l:] += wi * (1.0 - h[:n - l])
                grad[:n - l] -= wi * h[l:]
        vel = mom * vel - step * grad
        h_new = project_feasible(h + vel, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def optimize_phase(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    # Subgradient
    if time.time() < deadline - 30:
        h, c = subgrad_optimize(h, n, min(time.time() + 40, deadline - 30))
        if c < best_c:
            best_c = c
    # SLP
    if time.time() < deadline - 20:
        h, c = slp_optimize(h, n, min(time.time() + 100, deadline - 20))
        if c < best_c:
            best_c = c
    # SLSQP
    for alpha in [1000, 10000, 50000, 200000]:
        if time.time() > deadline - 15:
            break
        ht = slsqp_polish(h, n, alpha, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    # More SLP after SLSQP
    if time.time() < deadline - 10:
        h, c = slp_optimize(h, n, deadline - 5)
        if c < best_c:
            best_c = c
    return h, best_c
