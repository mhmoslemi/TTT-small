    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    dx = _p['dx']
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += 1.0 - h[-k:]
        g[-k:] -= h[:n + k]
    return g * dx

def _obj_lse(x):
    n = _p['n']
    alpha = _p['alpha']
    c = get_c(x)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha + 1e5 * (x.sum() - n / 2.0) ** 2

def _jac_lse(x):
    n = _p['n']
    n1 = n - 1
    c = np.correlate(x, 1 - x, mode='full')
    k = int(np.argmax(c)) - n1
    return grad_k(x, k) + 2e5 * (x.sum() - n / 2.0)

def _obj_slsqp(x):
    alpha = _p['alpha']
    c = get_c(x)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def _con_eq(x):
    return x.sum() - _p['n'] / 2.0

def subgrad_multi(h, n, deadline, rng, K, aw):
    _p['n'] = n
    _p['dx'] = 2.0 / n
    h = project(h.copy(), n)
    bh, bc = h.copy(), compute_c5(h, n)
    lr, mom, v, stall = 0.4, 0.65, np.zeros(n), 0
    n1 = n - 1
    while time.time() < deadline:
        cdx = get_c(h)
        m = cdx.max()
        top = np.argsort(cdx)[::-1][:K]
        w = np.exp(aw * (cdx[top] - m))
        w /= w.sum()
        g = np.zeros(n)
        for idx, wi in zip(top, w):
            g += wi * grad_k(h, idx - n1)
        v = mom * v - lr * g
        hn = project(h + v, n)
        cn = compute_c5(hn, n)
        if cn < bc - 1e-14:
            bc, bh, h, stall = cn, hn.copy(), hn, 0
            lr = min(lr * 1.03, 2.5)
        else:
            h = project(h - 0.3 * v, n)
            v *= 0.4
            lr *= 0.93
            stall += 1
            if stall > 200:
                h = project(bh + 0.01 * rng.randn(n), n)
                v = np.zeros(n)
                lr = 0.2
                stall = 0
        if lr < 1e-8:
            break
    return bh, bc

def lbfgs_solve(h, n, alpha, maxit):
    _p['n'] = n
    _p['dx'] = 2.0 / n
    _p['alpha'] = alpha
    try:
        r = minimize(_obj_lse, h, method='L-BFGS-B', jac=_jac_lse,
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12})
        return project(r.x, n)
    except Exception:
        return h

def slsqp_solve(h, n, alpha, maxit):
    _p['n'] = n
    _p['dx'] = 2.0 / n
    _p['alpha'] = alpha
    try:
        r = minimize(_obj_slsqp, h, method='SLSQP', jac=_jac_lse,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_eq}],
                     options={'maxiter': maxit, 'ftol': 1e-15})
        return project(r.x, n)
    except Exception:
        return h

def cont_sa(h, n, deadline, rng):
    _p['n'] = n
    _p['dx'] = 2.0 / n
    h = project(h.copy(), n)
    bh, bc = h.copy(), compute_c5(h, n)
    temp = 0.004
    for _ in range(200000):
        if time.time() > deadline:
            break
        i, j = rng.randint(n, 2)
        d = rng.uniform(-0.04, 0.04)
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            cn = compute_c5(h, n)
            if cn < bc - 1e-15:
                bc, bh = cn, h.copy()
            elif rng.random() < np.exp((bc - cn) / max(temp, 1e-9)):
                bh, bc = h.copy(), cn
            else:
                h[i], h[j] = hi - d, hj + d
        temp *= 0.99995
        if temp < 1e-7:
            temp = 1e-7
    return bh, bc

def fine_subgrad(h, n, deadline, rng):
    _p['n'] = n
    _p['dx'] = 2.0 / n
    h = project(h.copy(), n)
    bh, bc = h.copy(), compute_c5(h, n)
    lr = 0.04
    n1 = n - 1
    for _ in range(20000):
        if time.time() > deadline:
            break
        cdx = get_c(h)
        k = int(np.argmax(cdx)) - n1
        g = grad_k(h, k)
        hn = project(h - lr * g, n)
        cn = compute_c5(hn, n)
        if cn < bc - 1e-15:
            bc, bh, h = cn, hn.copy(), hn
            lr = min(lr * 1.005, 0.2)
        else:
            lr *= 0.97
            if lr < 1e-7:
                break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Multi-lag subgradient, annealing K (30%)
    p1 = t0 + 0.30 * (budget_s - 5)
    for K, aw in [(1, 1.0), (2, 5.0), (3, 8.0), (5, 12.0)]:
        if time.time() > p1 - 30:
            break
        seg = min(p1, t0 + (0.075 * (K) * (budget_s - 5)))
        h, c = subgrad_multi(best_h.copy(), n, seg, rng, K, aw)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: L-BFGS-B with increasing alpha (30%)
    p2 = t0 + 0.60 * (budget_s - 5)
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
        if time.time() > p2 - 20:
            break
        hn = lbfgs_solve(best_h.copy(), n, alpha, 500)
        cn = compute_c5(hn, n)
        if cn < best_c:
            best_c, best_h = cn, hn.copy()

    # Phase 3: SLSQP polish (15%)
    p3 = t0 + 0.75 * (budget_s - 5)
    for alpha in [200000, 1000000]:
        if time.time() > p3 - 10:
            break
        hn = slsqp_solve(best_h.copy(), n, alpha, 800)
        cn = compute_c5(hn, n)
        if cn < best_c:
            best_c, best_h = cn, hn.copy()

    # Phase 4: Continuous SA for escape (12%)
    p4 = t0 + 0.87 * (budget_s - 5)
    hn, cn = cont_sa(best_h.copy(), n, p4, rng)
    if cn < best_c:
        best_c, best_h = cn, hn.copy()

    # Phase 5: Fine subgradient polish (8%)
    p5 = deadline
    for _ in range(3):
        if time.time() > p5 - 5:
            break
        hn, cn = fine_subgrad(best_h.copy(), n, p5, rng)
        if cn < best_c:
            best_c, best_h = cn, hn.copy()
        else:
            break

    # Final L-BFGS with very high alpha
    if time.time() < deadline - 10:
        hn = lbfgs_solve(best_h.copy(), n, 5000000, 300)
        cn = compute_c5(hn, n)
        if cn < best_c:
            best_c, best_h = cn, hn.copy()

    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
