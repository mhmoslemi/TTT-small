def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

    for _ in range(100):

        if abs(diff) < 1e-14:

            if s < 1e-14:

    active = np.where(weights > threshold)[0]
    for m in active:
        w = weights[m]
        d = m - (n - 1)
        if d >= 0:
            j_start = d
            if j_start < n:
                grad[j_start:n] += w * (1.0 - h[0:n-d])
            j_end = n - d
            if j_end > 0:
                grad[0:j_end] -= w * h[d:n]
        else:
            j_end = n + d
            if j_end > 0:
                grad[0:j_end] += w * (1.0 - h[-d:n])
            j_start = -d
            if j_start < n:
                grad[j_start:n] -= w * h[0:n+d]
    return grad

def frank_wolfe(h, n, deadline, max_iters=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_half = n // 2
    for t in range(max_iters):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        sharpness = min(20 + t * 0.05, 200)
        s = compute_subgradient(h, n, g, sharpness)
        sorted_idx = np.argsort(s)
        h_lp = np.zeros(n)
        h_lp[sorted_idx[:n_half]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * h_lp
        h_new = project_feasible(h_new, n)
        h = h_new
        if t % 100 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c
