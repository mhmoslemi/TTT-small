def compute_G(h, N):
    """G[p, idx] where idx = k + (N-1), k ranges from -(N-1) to (N-1)."""
    G = np.zeros((N, 2*N-1))
    p = np.arange(N)
    for k in range(-(N-1), N):
        idx = k + (N-1)
        if k >= 0:
            g = np.zeros(N)
            mask1 = p >= k  # (1-h[p-k]) term
            g[mask1] += (1.0 - h[p[mask1] - k])
            mask2 = p <= (N-1-k)  # -h[p+k] term
            g[mask2] -= h[p[mask2] + k]
        else:
            m = -k
            g = np.zeros(N)
            mask1 = p <= (N-1-m)  # (1-h[p+m]) term
            g[mask1] += (1.0 - h[p[mask1] + m])
            mask2 = p >= m  # -h[p-m] term
            g[mask2] -= h[p[mask2] - m]
        G[:, idx] = g
    return G
