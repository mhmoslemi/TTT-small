    """Fast local search using O(n) correlation updates."""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    c_vec = np.correlate(h, 1 - h, mode='full') * dx
    best_c = c_vec.max()
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        m_star = int(np.argmax(c_vec))
        k_star = m_star - (n - 1)
        
        # Find best i (to increase) and j (to decrease) for shift k_star
        # d[m*] = h[i-k*] + h[i+k*] - h[j-k*] - h[j+k*]
        # Maximize h[i-k*] + h[i+k*] and minimize h[j-k*] + h[j+k*]
        
        if k_star >= 0:
            scores_i = np.zeros(n)
            scores_i[k_star:] += h[:n-k_star]  # h[i-k*] for i >= k_star
            scores_i[:n-k_star] += h[k_star:]   # h[i+k*] for i < n-k_star
            scores_j = np.zeros(n)
            scores_j[k_star:] += h[:n-k_star]
            scores_j[:n-k_star] += h[k_star:]
            # We want max score_i and min score_j, with i != j
        else:
            k_abs = -k_star
            scores_i = np.zeros(n)
            scores_i[:n-k_abs] += h[k_abs:]  # h[i-k*] = h[i+k_abs] for i < n-k_abs
            scores_i[k_abs:] += h[:n-k_abs]   # h[i+k*] = h[i-k_abs] for i >= k_abs
            scores_j = np.zeros(n)
            scores_j[:n-k_abs] += h[k_abs:]
            scores_j[k_abs:] += h[:n-k_abs]
        
        i_star = int(np.argmax(scores_i))
        j_star = int(np.argmin(scores_j))
        if i_star == j_star:
            # Get second best
            sorted_i = np.argsort(scores_i)[::-1]
            sorted_j = np.argsort(scores_j)
            for candidate in sorted_i:
                if candidate != j_star:
                    i_star = candidate
                    break
            else:
                j_star = sorted_j[0] if sorted_j[0] != i_star else sorted_j[1]

            idx_plus_k = idx + k_vals
            mask2 = (idx_plus_k >= 0) & (idx_plus_k < n)
            d[mask2] += sign * h[idx_plus_k[mask2]]

        lo = max(-h[i_star], h[j_star] - 1)
        hi = min(1 - h[i_star], h[j_star])

            m2 = hi - (hi - lo) / 3

            if f1 < f2:

                sc = scores.copy()
                sc[j_star] = 2.0
                j_star = int(np.argmin(sc))
            if i_star == j_star:
                continue

            d = compute_d_vec(h, i_star, j_star, n, k_vals)
            lo = max(-h[i_star], h[j_star] - 1.0) + 1e-12
            hi = min(1.0 - h[i_star], h[j_star]) - 1e-12
            if hi <= lo:
                continue

            for _ in range(18):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if float((c_vec - m1*dx*d).max()) <= float((c_vec - m2*dx*d).max()):
                    hi = m2
                else:
                    lo = m1

            delta = (lo + hi) / 2.0
            c_new = c_vec - delta * dx * d
            c_new_max = float(c_new.max())

            if c_new_max < best_c - 1e-15:
                h[i_star] = np.clip(h[i_star] + delta, 0.0, 1.0)
                h[j_star] = np.clip(h[j_star] - delta, 0.0, 1.0)
                c_vec = c_new
                best_c = c_new_max
                best_h = h.copy()
                stuck = 0
                made_move = True
                break

        if not made_move:
            stuck += 1
            if stuck > 80:
                improved = False
                for _ in range(40):
                    ri, rj = rng.choice(n, 2, replace=False)
                    rd = rng.uniform(-0.05, 0.05)
                    if 0 <= h[ri]+rd <= 1 and 0 <= h[rj]-rd <= 1:
                        d_r = compute_d_vec(h, ri, rj, n, k_vals)
                        c_t = c_vec - rd*dx*d_r
                        ct = float(c_t.max())
                        if ct < best_c:
                            h[ri] += rd; h[rj] -= rd
                            c_vec = c_t
                            best_c = ct
                            best_h = h.copy()
                            improved = True
                            break
                if not improved:
                    for _ in range(5):
                        ri, rj = rng.choice(n, 2, replace=False)
                        rd = rng.uniform(-0.15, 0.15)
                        if 0 <= h[ri]+rd <= 1 and 0 <= h[rj]-rd <= 1:
                            h[ri] += rd; h[rj] -= rd
                            c_vec = np.correlate(h, 1-h, mode='full') * dx
                            break
                stuck = 0

    return best_h, best_c

def lbfgs_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e6 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n-k])
            g[:n-k] += -x[k:]
        else:
            g[:n+k] += (1.0 - x[-k:])
            g[-k:] += -x[:n+k]
        g *= dx
        g += 2e6 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0.copy(), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Fast local search (60%)
    p1 = t0 + 0.60 * (budget_s - 3)
    h1, c1 = fast_local_search(best_h.copy(), n, p1, rng)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: L-BFGS polish + local search (25%)
    p2 = t0 + 0.85 * (budget_s - 3)
    for alpha in [5000, 20000, 100000, 500000, 2000000]:
        if time.time() > p2 - 10:
            break
        h_new = lbfgs_smooth(best_h.copy(), n, alpha, 150)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if time.time() < p2 - 5:
            rem = min(p2 - time.time(), 30)
            h_ls, c_ls = fast_local_search(best_h.copy(), n, time.time() + rem, rng)
            if c_ls < best_c:
                best_c, best_h = c_ls, h_ls.copy()

    # Phase 3: Final local search with multi-target (15%)
    h3, c3 = fast_local_search(best_h.copy(), n, deadline, rng)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    # Final L-BFGS if time remains
    if time.time() < deadline - 15:
        h_f = lbfgs_smooth(best_h.copy(), n, 5000000, 100)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
