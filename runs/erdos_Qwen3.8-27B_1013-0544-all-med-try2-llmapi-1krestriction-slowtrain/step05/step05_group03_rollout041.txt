def compute_swap_delta(c, h, i, j, n):
    """Compute the new correlation after swapping h[i]:0->1, h[j]:1->0.
    Returns (new_c, new_max)."""
    # Affected d values
    d_lo = max(-i, -j, i-(n-1), j-(n-1), -(n-1))
    d_hi = min(n-1-i, n-1-j, i, j, n-1)
    
    if d_lo > d_hi:
        return c, c.max()
    
    # For each affected d, compute the change
    new_c = c.copy()
    for d in range(d_lo, d_hi+1):
        delta = 0.0
        idx = n - 1 + d  # index in c
        
        # p = i (valid if 0 <= i+d < n, which is guaranteed by d_lo, d_hi)
        if 0 <= i + d < n:
            # old: h[i+d] * (1-h[i]) = h[i+d] * 1
            # new: h'[i+d] * (1-h'[i]) = h'[i+d] * 0 = 0
            # h'[i+d] = h[i+d] if i+d != j, else 0
            hd = i + d
            if hd == j:
                delta += 0 - h[hd]  # new is 0*0=0, old is h[j]*1 = h[j] = 1
            else:
                delta += 0 - h[hd]
        
        # p = j (valid if 0 <= j+d < n)
        if 0 <= j + d < n:
            # old: h[j+d] * (1-h[j]) = h[j+d] * 0 = 0
            # new: h'[j+d] * (1-h'[j]) = h'[j+d] * 1
            hd = j + d
            if hd == i:
                delta += 1 - 0  # h'[i]=1, old was 0
            else:
                delta += h[hd] - 0
        
        # p = i-d (valid if 0 <= i-d < n and 0 <= i < n (always true))
        p = i - d
        if 0 <= p < n and p != i and p != j:  # avoid double counting
            # old: h[i] * (1-h[p]) = 0 * (1-h[p]) = 0
            # new: h'[i] * (1-h'[p]) = 1 * (1-h'[p])
            if p == j:
                pass  # already counted as p=j case above
            hpp = h[p]  # p is not i or j, so h'[p] = h[p]
            delta += (1 - hpp) - 0
        
        # p = j-d (valid if 0 <= j-d < n)
        p = j - d
        if 0 <= p < n and p != i and p != j:
            # old: h[j] * (1-h[p]) = 1 * (1-h[p])
            # new: h'[j] * (1-h'[p]) = 0
            hpp = h[p]
            delta += 0 - (1 - hpp)
        
        # Handle the cases where p=i-d equals j or p=j-d equals i (double counting)
        # p = i-d = j (i.e., d = i-j): already handled in p=j case
        # p = j-d = i (i.e., d = j-i): already handled in p=i case
        
        new_c[idx] = c[idx] + delta
    
    return new_c, new_c.max()

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def sa_binary(h, n, t_start, t_end, iterations, rng, deadline):
    """Simulated annealing on binary solutions."""
    c = np.correlate(h, 1.0 - h, mode='full')
    curr_max = c.max()
    best_h = h.copy()
    best_max = curr_max
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / iterations)
        
        # Random swap
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        
        if i == j:
            continue
        
        # Perform swap
        h[i] = 1.0
        h[j] = 0.0
        new_c = np.correlate(h, 1.0 - h, mode='full')
        new_max = new_c.max()
        
        delta = new_max - curr_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            curr_max = new_max
            c = new_c
            zeros[ii] = j
            ones[jj] = i
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()
        else:
            # Reject
            h[i] = 0.0
            h[j] = 1.0

When we swap h[i]:0→1 and h[j]:1→0, the new correlation is:
c'[d] = Σ_{p: valid} h'[p+d] * (1-h'[p])

The difference c'[d] - c[d] is nonzero only for p where h'[p]≠h[p] or h'[p+d]≠h[p+d], i.e., p∈{i,j} or p+d∈{i,j}.

For each such p, the change in the term h[p+d]*(1-h[p]) is:

- p=i-d = j: This means d = i-j. In this case, "p=i-d" and "p=j" are the same p value. We should only count it once.
  - As p=j: change = h'[j+d] = h'[i] = 1 (since j+d = j+(i-j) = i)
  - As p=i-d: change = 1 (as computed above)
  - These are the same term! So we should only count it once.
  - The correct change for this p value: h'[i]*(1-h'[j]) - h[i]*(1-h[j]) = 1*1 - 0*0 = 1.
  
- p=j-d = i: This means d = j-i. In this case, "p=j-d" and "p=i" are the same.
  - As p=i: change = -h[i+d] = -h[j] = -1 (since i+d = i+(j-i) = j)
  - As p=j-d: change = -1 (as computed above)
  - Same term! Correct change: h'[j]*(1-h'[i]) - h[j]*(1-h[i]) = 0*0 - 1*1 = -1.

So the algorithm is:
1. Collect the set of affected p values: {i, j, i-d, j-d} ∩ [0,n) where also p+d ∈ [0,n)
2. For each unique p in this set, compute the change as h'[p+d]*(1-h'[p]) - h[p+d]*(1-h[p])
3. Sum the changes

To compute h'[x]:
- If x = i: h'[x] = 1
- If x = j: h'[x] = 0
- Otherwise: h'[x] = h[x]

Let me code this:

```python
def swap_delta_at_d(h, i, j, d, n):
    """Compute change in c[d] when swapping h[i]:0->1, h[j]:1->0."""
    delta = 0.0
    # Affected p values
    candidates = set()
    for p in [i, j, i-d, j-d]:
        if 0 <= p < n and 0 <= p+d < n:
            candidates.add(p)
    
    for p in candidates:
        a = p + d  # index for h[a]
        # Old: h[a] * (1-h[p])
        # New: h'[a] * (1-h'[p])
        hp_old = h[p]
        hp_new = 1.0 if p == i else (0.0 if p == j else h[p])
        ha_old = h[a]
        ha_new = 1.0 if a == i else (0.0 if a == j else h[a])
        delta += ha_new * (1.0 - hp_new) - ha_old * (1.0 - hp_old)
    
    return delta
```
