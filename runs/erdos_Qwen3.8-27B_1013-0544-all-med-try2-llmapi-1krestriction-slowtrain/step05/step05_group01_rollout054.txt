    n_c = 2*n - 1
    bias = n - 1
    
    # Compute gradient matrix G[i,k] = dc[k]/dh[i]
    def compute_G(h):

        for i in range(n):

def compute_G_vec(h, n):
    n_c = 2*n - 1
    bias = n - 1
    G = np.zeros((n, n_c))
    idx = np.arange(n)
    
    for i in range(n):
        # First term: (1 - h[i+s]) for valid s where 0 ≤ i+s < n
        # s ranges from max(0, -i) to min(n-1, n-1-i) but also -bias ≤ s ≤ bias
        # i+s ∈ [0, n-1]
        lo = max(0, -i)  # actually s ≥ -i for i+s ≥ 0
        hi = min(n-1, n-1-i)  # s ≤ n-1-i for i+s ≤ n-1
        
        # h[i+s] for s in [lo, hi] → h[i+lo : i+hi+1]
        if hi >= lo:
            k_lo = lo + bias
            k_hi = hi + bias
            G[i, k_lo:k_hi+1] += (1 - h[i+lo : i+hi+1])
        
        # Second term: -h[i-s] for valid s where 0 ≤ i-s < n
        # s ranges from max(0, i-(n-1)) to min(i, bias)
        # i-s ∈ [0, n-1]
        lo2 = max(0, i-(n-1))  # i-s ≤ n-1 → s ≥ i-(n-1)
        hi2 = min(i, bias)      # i-s ≥ 0 → s ≤ i
        
        if hi2 >= lo2:
            # h[i-s] for s in [lo2, hi2] → h[i-hi2 : i-lo2+1][::-1]
            k_lo2 = lo2 + bias
            k_hi2 = hi2 + bias
            # h[i-s] for s=lo2,lo2+1,...,hi2 is h[i-lo2], h[i-lo2-1], ..., h[i-hi2]
            # = h[i-hi2 : i-lo2+1][::-1]
            G[i, k_lo2:k_hi2+1] -= h[i-hi2 : i-lo2+1][::-1]
    
    return G

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def power_obj(h, n, p):
    """Power-sum approximation to max of correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Use p-norm as approximation to max
    # (sum |c|^p)^(1/p) → max(|c|) as p→∞
    # But c can be negative, so use max(c, 0)
    c_pos = np.maximum(c, 0)
    return float((np.sum(c_pos**p)) ** (1.0/p))

def _sum_eq(x, n):
    return np.sum(x) - n/2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_power(h, n, p, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(lambda x: power_obj(x, n, p), h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_G(h, n):
    """Compute gradient matrix G[i,k] = dc[k]/dh[i] where c = correlate(h,1-h,'full')*dx"""
    n_c = 2*n - 1
    bias = n - 1
    G = np.zeros((n, n_c))
    for i in range(n):
        # Term 1: (1 - h[i+s]) for s where 0 ≤ i+s < n
        s_lo = max(0, -i)  # need i+s ≥ 0
        s_hi = min(n-1-i, bias)  # need i+s ≤ n-1 and s ≤ bias
        if s_hi >= s_lo:
            k_lo = s_lo + bias
            k_hi = s_hi + bias
            G[i, k_lo:k_hi+1] += (1.0 - h[i+s_lo : i+s_hi+1])
        # Term 2: -h[i-s] for s where 0 ≤ i-s < n
        s_lo2 = max(0, i-(n-1))  # need i-s ≤ n-1
        s_hi2 = min(i, bias)  # need i-s ≥ 0
        if s_hi2 >= s_lo2:
            k_lo2 = s_lo2 + bias
            k_hi2 = s_hi2 + bias
            G[i, k_lo2:k_hi2+1] -= h[i-s_hi2 : i-s_lo2+1][::-1]
    return G

def fast_local_search(h, n, deadline, deltas=(0.02, 0.01, 0.005, 0.002, 0.001, 0.0005)):
    h = h.copy()
    dx = 2.0 / n
    current_max = compute_c5(h, n)
    
    for delta in deltas:
        if time.time() > deadline:
            break
        G = compute_G(h, n)
        c = np.correlate(h, 1-h, 'full') * dx
        current_max = c.max()
        threshold = (current_max - c) / delta  # need G[j,k]-G[i,k] ≥ -threshold[k] for all k
        
        n_passes = 0
        while n_passes < 200 and time.time() < deadline:
            n_passes += 1
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, current_max
                # Need: min_k (G[j,k] - G[i,k] + threshold[k]) ≥ 0
                scores = np.min(G - G[i:i+1, :] + threshold[None, :], axis=1)
                # Exclude i itself
                scores[i] = -np.inf
                # Also need h[i]+delta ≤ 1 and h[j]-delta ≥ 0
                feasible = (h[i] + delta <= 1.0 + 1e-12)
                if not feasible:
                    continue
                feasible_j = scores >= -1e-10
                feasible_j[i] = False
                feasible_j[h < delta - 1e-12] = False
                if not feasible_j.any():
                    continue
                # Pick j with minimum score (most "room" for improvement)
                candidates = np.where(feasible_j)[0]
                j = candidates[np.argmin(scores[candidates])]
                # Apply move
                h[i] += delta
                h[j] -= delta
                # Recompute
                c_new = np.correlate(h, 1-h, 'full') * dx
                new_max = c_new.max()
                if new_max < current_max - 1e-14:
                    current_max = new_max
                    G = compute_G(h, n)
                    c = c_new
                    threshold = (current_max - c) / delta
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
            if not improved:
                break
    return h, current_max

def binary_init(n, rng):
    """Create a random binary initialization."""
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def halton_init(n, rng, dim=1):
    """Initialize using Halton sequence for low-discrepancy placement."""
    h = np.zeros(n)
    k = n // 2
    # Use a simple low-discrepancy sequence
    # Van der Corput in base 2
    def vdc(i, base=2):
        f = 1.0
        r = 0.0
        while i > 0:
            f /= base
            r += f * (i % base)
            i //= base
        return r
    # Place k points using vdc sequence
    pts = sorted([vdc(i+1) * n for i in range(k)])
    idx = [int(p) % n for p in pts]
    # Remove duplicates
    seen = set()
    count = 0
    for idx_val in idx:
        if idx_val not in seen:
            h[idx_val] = 1.0
            seen.add(idx_val)
            count += 1
    if count < k:
        # Fill remaining randomly
        remaining = [i for i in range(n) if h[i] == 0]
        extra = rng.choice(remaining, k-count, replace=False)
        h[extra] = 1.0
    return h

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: SLSQP with increasing power
    for p in [4, 8, 16, 32, 64, 128, 256, 512]:
        if time.time() > deadline - 30:
            break
        remaining = deadline - time.time() - 30
        maxiter = min(5000, int(remaining * 50))
        if maxiter < 100:
            break
        ht = slsqp_power(best_h, n, p, maxiter=maxiter)
        c = compute_c5(ht, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = ht.copy()
    
    # Phase 2: Fast local search
    if time.time() < deadline - 10:
        best_h, best_c = fast_local_search(best_h, n, deadline - 5)
    
    # Phase 3: Final SLSQP polish
    if time.time() < deadline - 5:

        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 4: Final local search
    if time.time() < deadline - 2:
        best_h, best_c = fast_local_search(best_h, n, deadline - 1,
                                           deltas=(0.001, 0.0005, 0.0002, 0.0001))
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def power_obj(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_pos = np.maximum(c, 0.0)
    return float((np.sum(c_pos ** p)) ** (1.0 / p))

def _sum_eq(x, n):
    return np.sum(x) - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_power(h, n, p, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(lambda x: power_obj(x, n, p), h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_G(h, n):
    n_c = 2 * n - 1
    bias = n - 1
    G = np.zeros((n, n_c))
    for i in range(n):
        s_lo = max(0, -i)
        s_hi = min(n - 1 - i, bias)
        if s_hi >= s_lo:
            k_lo = s_lo + bias
            k_hi = s_hi + bias
            G[i, k_lo:k_hi + 1] += (1.0 - h[i + s_lo:i + s_hi + 1])
        s_lo2 = max(0, i - (n - 1))
        s_hi2 = min(i, bias)
        if s_hi2 >= s_lo2:
            k_lo2 = s_lo2 + bias
            k_hi2 = s_hi2 + bias
            G[i, k_lo2:k_hi2 + 1] -= h[i - s_hi2:i - s_lo2 + 1][::-1]
    return G

def fast_local_search(h, n, deadline, deltas=(0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001)):
    h = h.copy()
    c = np.correlate(h, 1 - h, 'full') * (2.0 / n)
    current_max = c.max()
    for delta in deltas:
        if time.time() > deadline:
            break
        G = compute_G(h, n)
        threshold = (current_max - c) / delta
        idx = np.arange(n)
        for _pass in range(300):
            if time.time() > deadline:
                return h, current_max
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, current_max
                if h[i] + delta > 1.0 + 1e-12:
                    continue
                margins = np.min(G - G[i:i+1, :] + threshold[None, :], axis=1)
                gains = np.max(G - G[i:i+1, :] + threshold[None, :], axis=1)
                feasible = (margins >= -1e-10) & (idx != i) & (h >= delta - 1e-12)
                if not feasible.any():
                    continue
                candidates = np.where(feasible)[0]
                j = candidates[np.argmax(gains[candidates])]
                h[i] += delta
                h[j] -= delta
                c_new = np.correlate(h, 1 - h, 'full') * (2.0 / n)
                new_max = c_new.max()
                if new_max < current_max - 1e-14:
                    current_max = new_max
                    G = compute_G(h, n)
                    c = c_new
                    threshold = (current_max - c) / delta
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
            if not improved:
                break
    return h, current_max

def binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def halton_init(n, rng):
    h = np.zeros(n)
    k = n // 2
    def vdc(i, base=2):
        f, r = 1.0, 0.0
        while i > 0:
            f /= base
            r += f * (i % base)
            i //= base
        return r
    pts = sorted([int(vdc(i + 1) * n) % n for i in range(k)])
    seen = set()
    for p in pts:
        if p not in seen:
            h[p] = 1.0
            seen.add(p)
    count = len(seen)
    if count < k:
        remaining = [i for i in range(n) if h[i] == 0]
        h[rng.choice(remaining, k - count, replace=False)] = 1.0
    return h

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in [4, 16, 64, 256, 1024, 4096]:
        if time.time() > deadline - 20:
            break
        remaining = deadline - time.time() - 20
        maxiter = max(200, min(5000, int(remaining * 40)))
        ht = slsqp_power(best_h, n, p, maxiter=maxiter)
        c = compute_c5(ht, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 8:
        best_h, best_c = fast_local_search(best_h, n, deadline - 4)
    if time.time() < deadline - 4:
        ht = slsqp_power(best_h, n, 8192, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 2:
        best_h, best_c = fast_local_search(best_h, n, deadline - 1,
                                           deltas=(0.0005, 0.0002, 0.0001, 0.00005, 0.00002))
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
