import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:int(diff)+1]
        else:
            idx = np.argsort(h)[:int(-diff)+1]
        for i in idx:
            a = min(diff, 1.0-h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += a; diff -= a
            if abs(diff) < 1e-14: break
    return h

def slsqp_solve(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_construct(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    dx = 2.0 / n
    target = n // 2
    for _ in range(target):
        best_i = 0; best_val = 1e9
        candidates = np.where(h == 0)[0]
        for i in candidates:
            h[i] = 1
            c = np.correlate(h, 1-h, mode='full')
            v = c.max() * dx
            h[i] = 0
            if v < best_val:
                best_val = v; best_i = i
        h[best_i] = 1
    return h

def binary_local_search(h, n, max_sweeps, deadline, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').copy()
    cur_c = corr.max() * dx
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    best_h = h.copy(); best_c = cur_c
    for sweep in range(max_sweeps):
        if time.time() > deadline: break
        improved = False
        rng.shuffle(zeros); rng.shuffle(ones)
        zl = zeros[:]; ol = ones[:]
        for i in zl:
            if time.time() > deadline: break
            for j in ol:
                im = i + m_range; imn = i - m_range
                jm = j + m_range; jmn = j - m_range
                delta = np.zeros(2*n-1)
                m1 = (im >= 0) & (im < n)
                delta[m1] += 1.0 - h[im[m1]]
                m2 = (imn >= 0) & (imn < n)
                delta[m2] -= h[imn[m2]]
                m3 = (jm >= 0) & (jm < n)
                delta[m3] -= 1.0 - h[jm[m3]]
                m4 = (jmn >= 0) & (jmn < n)
                delta[m4] += h[jmn[m4]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr; cur_c = nc
                    ol.remove(j); ol.append(i)
                    zl.remove(i); zl.append(j)
                    improved = True
                    break
            if improved: break
        if not improved: break
        if cur_c < best_c:
            best_c = cur_c; best_h = h.copy()
    return best_h, best_c

def sa_binary_search(h, n, deadline, seed=42, T0=0.05, cool=0.9995, max_iter=50000):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n); h[idx[:target]] = 1.0
    dx = 2.0 / n; m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').copy()
    cur_c = corr.max() * dx
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    best_h = h.copy(); best_c = cur_c
    T = T0
    for it in range(max_iter):
        if time.time() > deadline: break
        i = rng.choice(zeros); j = rng.choice(ones)
        im = i + m_range; imn = i - m_range
        jm = j + m_range; jmn = j - m_range
        delta = np.zeros(2*n-1)
        m1 = (im >= 0) & (im < n); delta[m1] += 1.0 - h[im[m1]]
        m2 = (imn >= 0) & (imn < n); delta[m2] -= h[imn[m2]]
        m3 = (jm >= 0) & (jm < n); delta[m3] -= 1.0 - h[jm[m3]]
        m4 = (jmn >= 0) & (jmn < n); delta[m4] += h[jmn[m4]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1: delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1: delta[k2] += 1.0
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dC = nc - cur_c
        if dC < 0 or rng.random() < np.exp(-dC / max(T, 1e-10)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr; cur_c = nc
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c; best_h = h.copy()
        T = max(T * cool, 1e-8)
    return best_h, best_c
