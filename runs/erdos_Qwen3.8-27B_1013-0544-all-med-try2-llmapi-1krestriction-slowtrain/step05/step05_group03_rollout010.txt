import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha, penalty=0.0):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    sm = shifted.max()
    val = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    if penalty > 0:
        val += penalty * (h.sum() - n/2.0)**2
    return val

def smooth_grad(h, n, alpha, penalty=0.0):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= wsum
    grad = np.zeros(n)
    for ki in range(2*n - 1):
        wk = w[ki]
        if wk < 1e-15:
            continue
        lag = ki - n + 1
        if lag >= 0:
            grad[:n-lag] += wk * dx * (1.0 - h[lag:])
            grad[lag:] -= wk * dx * h[:n-lag]
        else:
            ll = -lag
            grad[ll:] += wk * dx * (1.0 - h[:n-ll])
            grad[:n-ll] -= wk * dx * h[ll:]
    if penalty > 0:
        grad += 2.0 * penalty * (h.sum() - n/2.0)
    return grad

def lbfgsb_run(h, n, alpha, penalty, maxiter, deadline):
    def obj(x):
        return smooth_obj(x, n, alpha, penalty)
    def grad(x):
        return smooth_grad(x, n, alpha, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_run(h, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_obj(x, n, alpha, 0.0)
    def grad(x):
        return smooth_grad(x, n, alpha, 0.0)
    def con_fun(x):
        return x.sum() - n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_sa(hb, n, deadline, T_max=0.05, T_min=0.001, max_iter=500000):
    # Simulated annealing on binary solutions with swap moves
    best = hb.copy()
    best_c = compute_c5(hb, n)
    cur_c = best_c
    ones = set(np.where(hb == 1.0)[0].tolist())
    zeros = set(np.where(hb == 0.0)[0].tolist())
    n_iters = 0
    rng = np.random.RandomState(123)
    
    while time.time() < deadline and n_iters < max_iter:
        # Determine temperature
        progress = min(n_iters / max_iter, 1.0)
        T = T_max * (T_min / T_max) ** progress
        
        # Choose a random swap
        i = rng.choice(sorted(zeros)) if len(zeros) > 0 else 0
        j = rng.choice(sorted(ones)) if len(ones) > 0 else 0
        
        # Perform swap
        hb[i], hb[j] = 1.0, 0.0
        new_c = compute_c5(hb, n)
        
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-10)):
            cur_c = new_c
            ones.add(i)
            ones.remove(j)
            zeros.remove(i)
            zeros.add(j)
            if new_c < best_c:
                best_c = new_c
                best = hb.copy()
        else:
            hb[i], hb[j] = 0.0, 1.0
        
        n_iters += 1
        if n_iters % 1000 == 0:
            pass  # checkpoint
    
    return best, best_c
