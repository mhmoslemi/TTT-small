def project_feasible(h, n):

            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_descent(h, n, deadline, max_sweeps=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    stall_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = np.zeros((n, 2 * n - 1))
        for m_idx_val in range(2 * n - 1):
            d = dm[m_idx_val]
            i_lo = max(0, d)
            i_hi = min(n, n + d)
            if i_lo < i_hi:
                F[i_lo:i_hi, m_idx_val] += (1.0 - h[i_lo - d:i_hi - d])
            j_lo = max(0, -d)
            j_hi = min(n, n - d)
            if j_lo < j_hi:
                F[j_lo:j_hi, m_idx_val] -= h[j_lo + d:j_hi + d]
        g = compute_g(h, n)
        current_max = g.max()
        K = min(6, 2 * n - 1)
        top_idx = np.argsort(g)[::-1][:K]
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    v1 = (i + dm >= 0) & (i + dm < n)
                    F[i + dm[v1], m_idx[v1]] -= best_delta
                    v2 = (i - dm >= 0) & (i - dm < n)
                    F[i - dm[v2], m_idx[v2]] -= best_delta
                    v3 = (j + dm >= 0) & (j + dm < n)
                    F[j + dm[v3], m_idx[v3]] += best_delta
                    v4 = (j - dm >= 0) & (j - dm < n)
                    F[j - dm[v4], m_idx[v4]] += best_delta
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved = True
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            stall_count += 1
            if stall_count > 3:
                break
        else:
            stall_count = 0
    return best_h, best_c

def subgradient_descent(h, n, deadline, step_decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    it = 0
    while time.time() < deadline:
        g = compute_g(h, n)
        k_star = int(np.argmax(g))
        d = k_star - (n - 1)
        grad = np.zeros(n)
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            grad[j_lo:j_hi] -= h[j_lo + d:j_hi + d]
        # Projected subgradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            step = max(step * 0.999, 0.001)
        else:
            step *= 0.95
        it += 1
    return best_h, best_c
