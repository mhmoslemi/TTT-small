import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _pair_delta(h, n, i, j, d, center):
    """Compute delta in corr for h[i]+=d, h[j]-=d. Returns (positions, deltas)."""
    # k range for i
    if i < n // 2:
        lo_i, hi_i = i - (n - 1), i
    else:
        lo_i, hi_i = -i, n - 1 - i
    # k range for j
    if j < n // 2:
        lo_j, hi_j = j - (n - 1), j
    else:
        lo_j, hi_j = -j, n - 1 - j
    
    lo = min(lo_i, lo_j)
    hi = max(hi_i, hi_j)
    
    ks = np.arange(lo, hi + 1)
    # Compute delta for each k
    # delta[k] = d * [(1-h[i-k]) - h[i+k] - (1-h[j-k]) + h[j+k]] + corrections
    ik = i - ks  # i-k
    ip = i + ks  # i+k
    jk = j - ks  # j-k
    jp = j + ks  # j+k
    
    valid_ik = (ik >= 0) & (ik < n)
    valid_ip = (ip >= 0) & (ip < n)
    valid_jk = (jk >= 0) & (jk < n)
    valid_jp = (jp >= 0) & (jp < n)
    
    delta = np.zeros(len(ks))
    delta[valid_ik] += d * (1.0 - h[ik[valid_ik]])
    delta[valid_ip] -= d * h[ip[valid_ip]]
    delta[valid_jk] -= d * (1.0 - h[jk[valid_jk]])
    delta[valid_jp] += d * h[jp[valid_jp]]
    
    # Corrections (exact)
    d2 = d * d
    # k=0 correction: -2*d^2
    mask0 = (ks == 0)
    delta[mask0] += -2.0 * d2
    # k=i-j correction: +d^2
    mask_ij = (ks == (i - j))
    delta[mask_ij] += d2
    # k=j-i correction: +d^2
    mask_ji = (ks == (j - i))
    delta[mask_ji] += d2
    
    positions = center + ks
    return positions, delta
