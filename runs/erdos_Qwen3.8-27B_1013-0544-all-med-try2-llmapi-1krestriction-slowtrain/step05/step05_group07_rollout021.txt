def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_project(h, n):
    """Project to binary with exactly n/2 ones"""
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target - 1]
    h_bin = (h >= threshold).astype(float)
    if h_bin.sum() > target:
        # Need to flip some
        excess = h_bin.sum() - target
        # Flip the smallest values among the ones
        one_indices = np.where(h_bin == 1)[0]
        # Sort by h value, flip the smallest
        order = np.argsort(h[one_indices])[:int(excess)]
        h_bin[one_indices[order]] = 0
    elif h_bin.sum() < target:
        deficit = target - h_bin.sum()
        zero_indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_indices])[:int(deficit)]
        h_bin[zero_indices[order]] = 1
    return h_bin

def smooth_max_opt(h, n, alpha, maxiter, deadline, method='L-BFGS-B'):

        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        val = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        val += 1e4 * (x.sum() - n/2.0)**2
        return val

def nesterov_smooth(h, n, alpha, deadline, lr=None):
    """Nesterov accelerated gradient for smooth max approximation"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    if lr is None:
        lr = 1.0 / alpha
    t = 1.0
    for it in range(10000):
        if time.time() > deadline:
            break
        # Compute gradient at y
        c = np.correlate(y, 1 - y, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
        
        # Gradient of smooth max: sum_k w_k * grad_k
        # grad_k[i] = (1-h[i+k])*dx if i+k valid, -h[i]*dx if i valid
        grad = np.zeros(n)
        for k_idx in range(2*n - 1):
            k = k_idx - (n - 1)
            w = weights[k_idx]
            if w < 1e-10:
                continue
            if k >= 0:
                grad[k:] += w * (1.0 - y[:n-k]) * dx
                grad[:n-k] -= w * y[k:] * dx
            else:
                grad[:n+k] += w * (1.0 - y[-k:]) * dx
                grad[-k:] -= w * y[:n+k] * dx
        grad += 2e4 * (y.sum() - n/2.0)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        x = project_feasible(y - lr * grad, n)
        y = beta * x + (1 - beta) * y
        y = project_feasible(y, n)
        t = t_new
        
        c_new = compute_c5(y, n)
        if c_new < best_c:
            best_c = c_new
            best_h = y.copy()
    
    return best_h, best_c

    """Compute true gradient of smooth max approximation"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()

    sm = shifted.max()
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    
    # Only keep significant weights for efficiency
    significant = weights > 1e-6
    if significant.sum() == 0:
        significant = weights > 0

        else:
            grad[:n+k] += w * (1.0 - h[-k:]) * dx

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

            if s < 1e-13:

def binary_project(h, n):

    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target - 1]
    h_bin = (h >= threshold).astype(float)

        order = np.argsort(-h[zero_idx])[:target - cnt]

def coordinate_descent(h, n, deadline, steps):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for sgn in (1, -1):
                    for frac in (1.0, 0.5, 0.25):
