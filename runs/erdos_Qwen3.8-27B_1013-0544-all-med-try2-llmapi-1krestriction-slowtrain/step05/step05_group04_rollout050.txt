    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            if not mask.any(): break
            h[mask] += diff / room[mask].sum() * room[mask]
        else:
            mask = h > 1e-14
            if not mask.any(): break
            h[mask] -= (-diff) / h[mask].sum() * h[mask]
    return np.clip(h, 0.0, 1.0)

def soft_max_val_grad(h, n, alpha, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    ac = alpha * c
    M = ac.max()
    exp_ac = np.exp(ac - M)
    s = exp_ac.sum()
    val = (M + np.log(s)) / alpha
    p = exp_ac / s
    
    grad = np.zeros(n)
    # For each output index k, lag = k - (n-1)
    # d(corr[k])/d(h[r]) = (1-h[r-lag])*[0<=r-lag<n] - h[r+lag]*[0<=r+lag<n]
    for k in range(2 * n - 1):

        w = p[k] * dx
        # First term: (1-h[r-lag]) for valid r
        r_lo = max(0, lag)
        r_hi = min(n, lag + n)
        if r_lo < r_hi:
            grad[r_lo:r_hi] += w * (1.0 - h[r_lo - lag:r_hi - lag])

        if r_lo2 < r_hi2:
            grad[r_lo2:r_hi2] -= w * h[r_lo2 + lag:r_hi2 + lag]
    
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return val, grad

                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def gradient_pair_search(h, n, deadline, max_rounds=50):
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        _, g = soft_max_val_grad(h, n, 50000, 0.0)
        pos = np.where(g > 1e-8)[0]
        neg = np.where(g < -1e-8)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        improved = False
        for i in pos:
            if time.time() > deadline:
                break
            for j in neg:
                if i == j:
                    continue
                d = 0.005
                if h[i] - d < 0:
                    d = h[i] * 0.5
                if h[j] + d > 1:
                    d = (1 - h[j]) * 0.5
                if d < 1e-6:
                    continue
                h[i] -= d
                h[j] += d
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] += d
                    h[j] -= d
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B with increasing alpha
    t1 = time.time() + min(120, (deadline - t0) * 0.15)
    h1, c1 = lbgfb_opt(best_h, n_init, t1,
                       alphas=[500, 2000, 10000, 50000])
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Gradient-guided pair search
    if time.time() < deadline - 120:
        t2 = time.time() + min(80, (deadline - time.time()) * 0.1)
        h2, c2 = gradient_pair_search(best_h, n_init, t2)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Full pair search with decreasing steps
    if n_init <= 200:
        if time.time() < deadline - 80:
            t3 = time.time() + min(60, (deadline - time.time()) * 0.15)
            best_h, best_c = pair_search(best_h, n_init, t3,
                                         [0.01, 0.005, 0.002, 0.001])
    else:
        if time.time() < deadline - 80:
            t3 = time.time() + min(60, (deadline - time.time()) * 0.15)
            best_h, best_c = near_pair(best_h, n_init, t3,
                                       [0.01, 0.005, 0.002, 0.001], radius=50)

    # Phase 4: L-BFGS-B with high alpha (near-exact max)
    if time.time() < deadline - 60:
        t4 = time.time() + min(80, (deadline - time.time()) * 0.2)
        h4, c4 = lbgfb_opt(best_h, n_init, t4,
                           alphas=[100000, 500000, 1000000])
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Fine pair search
    if time.time() < deadline - 30:
        t5 = time.time() + min(40, (deadline - time.time()) * 0.2)
        if n_init <= 200:
            best_h, best_c = pair_search(best_h, n_init, t5,
                                         [0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = near_pair(best_h, n_init, t5,
                                       [0.001, 0.0005, 0.0002], radius=40)

    # Phase 6: Try alternate n values
    if time.time() < deadline - 80:
        for n_try in [150, 200, 300]:
            if time.time() > deadline - 60:
                break
            if n_try == n_init:
                continue
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(60, (deadline - time.time()) * 0.25)
            if alloc < 20:
                break
            ht, ct = lbgfb_opt(h_interp, n_try, time.time() + alloc,
                               alphas=[2000, 20000, 100000])
            if n_try <= 200 and time.time() < deadline - 30:
                ht, ct = pair_search(ht, n_try, deadline - 30,
                                     [0.005, 0.001, 0.0005])
            if ct < best_c:
                best_c, best_h = ct, ht

    # Phase 7: Final ultra-fine polish
    if time.time() < deadline - 5:
        n = len(best_h)
        h7, c7 = lbgfb_opt(best_h, n, deadline - 3,
                           alphas=[500000, 2000000], mu=200.0)
        if c7 < best_c:
            best_c, best_h = c7, h7
        if n <= 200 and time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.0005, 0.0002, 0.0001])
        elif time.time() < deadline - 2:
            best_h, best_c = near_pair(best_h, n, deadline - 1,
                                       [0.0005, 0.0002], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
