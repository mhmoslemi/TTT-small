    c = np.correlate(h, 1 - h, mode='full') * dx

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag_k(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    s = k - (n - 1)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s]) * dx
    g[idx + s] -= h[idx] * dx
    return g

Actually wait, I realize the ternary search evaluations don't require a full correlate - they just require computing max_k(a_k * t + b_k) where a and b are precomputed arrays of length 2n-1. So each evaluation is O(n) (just a dot product and max). That's much faster than a full correlate.

So the bottleneck is the 2 correlate calls per coordinate (to compute a and b), which is 2 * 100μs = 200μs per coordinate, 50ms per sweep. We can do ~20 sweeps per second, or ~2000 sweeps in 100 seconds.

This is more than enough. Let me proceed with this approach.

Now, one concern: after the coordinate descent, the solution might be at a "saddle point" where no single coordinate can be improved, but a joint perturbation of multiple coordinates could. This is where the subgradient method and SLSQP come in.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag_k(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    s = k - (n - 1)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s]) * dx
    g[idx + s] -= h[idx] * dx
    return g

def subgrad_method(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    for it in range(10000):
        if time.time() > deadline:
            break
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c:
            best_c = curr
            best_h = h.copy()
        tol = 0.005
        active = np.where(c >= curr - tol)[0]
        g = np.zeros(n)
        for k in active:
            g += grad_lag_k(h, n, int(k))
        g /= len(active)
        h_new = project_feasible(h - step * g, n)
        new_c = compute_c5(h_new, n)
        if new_c < curr + 1e-12:
            h = h_new
            step = min(step * 1.02, 0.03)
        else:
            step = max(step * 0.95, 0.001)
    return best_h, best_c

def coord_descent(h, n, deadline, max_sweeps=200):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = 0
        for j in range(n):
            if time.time() > deadline:
                break
            c_orig = get_corr(h, n)
            h2 = h.copy()
            h2[j] += 1.0
            c2 = get_corr(h2, n)
            a = c2 - c_orig
            b = c_orig - a * h[j]
            l_t, r_t = 0.0, 1.0
            for _ in range(45):
                m1 = l_t + (r_t - l_t) / 3.0
                m2 = r_t - (r_t - l_t) / 3.0
                v1 = float(np.max(a * m1 + b))
                v2 = float(np.max(a * m2 + b))
                if v1 < v2:
                    r_t = m2
                else:
                    l_t = m1
            t_opt = (l_t + r_t) / 2.0
            if abs(t_opt - h[j]) > 1e-12:
                h[j] = t_opt
                improved += 1
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        if improved == 0 and sweep > 5:
            break
    return best_h, best_c

def slsqp_soft(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    
    def fun_sum(x):
        return float(np.sum(x) - n / 2.0)
    
    def jac_sum(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': fun_sum, 'jac': jac_sum}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_lag_polish(h, n, deadline, num_lags=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
        top_k = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += grad_lag_k(h, n, int(k))
        if g.max() < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps_list[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-12:
            step_idx += 1
            stall += 1
            if stall > 200 or step_idx >= len(steps_list):
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < curr - 1e-15:
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx + 1, len(steps_list) - 1)
            if stall > 200:
                break
    return best_h, best_c

def optimize_full(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Subgradient phase
    t_deadline_sg = min(deadline, time.time() + (deadline - time.time()) * 0.25)
    h_sg, c_sg = subgrad_method(h, n, t_deadline_sg)
    if c_sg < best_c:
        best_c = c_sg
        best_h = h_sg
    
    # Coordinate descent
    t_deadline_cd = min(deadline, time.time() + (deadline - time.time()) * 0.35)
    h_cd, c_cd = coord_descent(best_h, n, t_deadline_cd)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd
