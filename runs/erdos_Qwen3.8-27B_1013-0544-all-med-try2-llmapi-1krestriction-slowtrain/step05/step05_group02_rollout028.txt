```python
import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_smooth(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        soft = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return soft
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k_lag):
    """Gradient of c_k with respect to h, where k_lag is the index in the full correlation."""
    s = k_lag - (n - 1)  # actual lag
    g = np.zeros(n)
    # c_k = sum_i h[i] * (1-h[i+s]) for valid i
    # d c_k / d h[i] = (1-h[i+s]) - h[i-s]
    valid_first = np.arange(max(0, -s), min(n, n - s))
    g[valid_first] += (1.0 - h[valid_first + s])
    valid_second = np.arange(max(0, s), min(n, n + s))
    g[valid_second] -= h[valid_second - s]
    return g

def directed_polish_v2(h, n, deadline, top_k=3):
    """Enhanced directed polish with line search."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    stall = 0
    max_stall = 500
    
    while time.time() < deadline and stall < max_stall:
        c = get_corr(best_h, n)
        curr = float(c.max())
        
        # Get top worst lags
        top_indices = np.argsort(c)[-top_k:]
        
        # Compute combined gradient
        g = np.zeros(n)
        for k_idx in top_indices:
            g += lag_gradient(best_h, n, int(k_idx))
        
        # Find the direction: decrease where g is large positive, increase where g is large negative
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        
        # Line search for optimal step
        def line_obj(t):
            ht = best_h.copy()
            ht[i_src] -= t
            ht[i_dst] += t
            if ht[i_src] < 0 or ht[i_dst] > 1:
                return 1e10
            c_t = get_corr(ht, n)
            return float(c_t.max())
        
        t_max = min(best_h[i_src], 1.0 - best_h[i_dst])
        res = minimize_scalar(line_obj, bounds=(1e-8, t_max * 0.999), method='bounded',
                            options={'xatol': 1e-10})
        
        if res.fun < curr - 1e-14:
            best_h[i_src] -= res.x
            best_h[i_dst] += res.x
            best_c = res.fun
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c
```

∂/∂h[i] of (1/α) * log(sum_k exp(α * c_k)) = (1/α) * sum_k exp(α * c_k) / sum_k exp(α * c_k) * ∂c_k/∂h[i]

```python
import time
from scipy.optimize import minimize, minimize_scalar
```

```python
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_smooth(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k_idx):
    s = k_idx - (n - 1)
    g = np.zeros(n)
    vf = np.arange(max(0, -s), min(n, n - s))
    g[vf] = 1.0 - h[vf + s]
    vs = np.arange(max(0, s), min(n, n + s))
    g[vs] -= h[vs - s]
    return g

def pair_line_search(h, n, i_src, i_dst):
    """Find optimal t: h[i_src] -= t, h[i_dst] += t, minimizing max c_k."""
    t_max = min(h[i_src], 1.0 - h[i_dst])
    if t_max < 1e-12:
        return h, compute_c5(h, n)
    
    def obj_1d(t):
        ht = h.copy()
        ht[i_src] -= t
        ht[i_dst] += t
        c = get_corr(ht, n)
        return float(c.max())
    
    res = minimize_scalar(obj_1d, bounds=(1e-10, t_max * 0.9999), method='bounded',
                         options={'xatol': 1e-12})
    h[i_src] -= res.x
    h[i_dst] += res.x
    return h, float(res.fun)

def line_search_polish(h, n, deadline, top_k=3):
    """Polish using pair line search on worst lag gradients."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    stall = 0
    max_stall = 200
    
    while time.time() < deadline and stall < max_stall:

        curr = float(c.max())
        
        top_indices = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for ki in top_indices:
            g = g + lag_gradient(best_h, n, int(ki))

        i_dst = int(np.argmin(g))
```

```python
import numpy as np

from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_smooth(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)

        return h.copy()

        ht[i_src] -= t

    h_new[i_dst] += res.x
```
