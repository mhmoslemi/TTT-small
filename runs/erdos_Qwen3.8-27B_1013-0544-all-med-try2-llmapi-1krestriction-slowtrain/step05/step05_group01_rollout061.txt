- h[j] appears as h[i] when i=j: contributes (1-h[j+s]) if j is in the sum range
- h[j] appears as h[i+s] when i=j-s: contributes h[j-s]*(-1) = -h[j-s] if j-s is in the sum range

The sum range is i from max(0,-s) to min(n-1, n-1-s).

So:
- First term: (1-h[j+s]) if max(0,-s) <= j <= min(n-1, n-1-s) and 0 <= j+s < n
  - j+s in [0,n-1] means -s <= j <= n-1-s
  - Combined: max(0,-s) <= j <= min(n-1, n-1-s) automatically satisfies 0 <= j+s < n (since j >= max(0,-s) means j+s >= max(s,0) >= 0, and j <= n-1-s means j+s <= n-1)
  - So condition simplifies to: max(0,-s) <= j <= min(n-1, n-1-s)

- Second term: -h[j-s] if max(0,-s) <= j-s <= min(n-1, n-1-s) and 0 <= j-s < n
  - j-s >= max(0,-s) and j-s <= min(n-1, n-1-s)
  - j >= max(0,-s) + s and j <= min(n-1, n-1-s) + s
  - j >= max(s, 0) and j <= min(n-1+s, n-1)
  - For s >= 0: j >= s and j <= n-1
  - For s < 0: j >= 0 and j <= n-1+s = n-1-|s|

So the gradient is:
grad[j] = (1-h[j+s]) * I[max(0,-s) <= j <= min(n-1, n-1-s)] - h[j-s] * I[max(s,0) <= j <= min(n-1+s, n-1)]

```python
def grad_of_corr(h, s, n):
    """Gradient of sum_i h[i]*(1-h[i+s]) w.r.t. h, where s is the shift."""
    grad = np.zeros(n)
    # First term: (1-h[j+s]) for j in [max(0,-s), min(n-1, n-1-s)]
    lo1 = max(0, -s)
    hi1 = min(n-1, n-1-s)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += (1.0 - h[lo1+s:hi1+s+1])
    # Second term: -h[j-s] for j in [max(s,0), min(n-1+s, n-1)]
    lo2 = max(s, 0)
    hi2 = min(n-1+s, n-1)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2-s:hi2-s+1]
    return grad
```

```python
def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

    hi1 = min(n - 1, n - 1 - s)
    if lo1 <= hi1:

    hi2 = min(n - 1 + s, n - 1)
    if lo2 <= hi2:

    return grad

def subgradient_descent(h, n, deadline, n_iter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    momentum = np.zeros(n)
    beta = 0.9
    step = 0.01
    patience = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        lags = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(lags)
        s = max_idx - (n - 1)
        g = grad_lag(h, s, n)

        g_mean = g.sum() / n
        g = g - g_mean
        # Normalized step
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        # Step
        momentum = beta * momentum + (1 - beta) * g
        h_new = h - step * momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

            best_h = h_new.copy()
            h = h_new
            patience = 0
            step = min(step * 1.05, 0.1)
        else:
            h = h_new
            patience += 1
            if patience > 100:
                step *= 0.9
                patience = 0
                if step < 1e-5:
                    break
```

```python
def pair_search(h, n, deadline, deltas, radius=100):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        if time.time() > deadline:
            break
        n_passes = 0
        while time.time() < deadline:
            n_passes += 1
            if n_passes > 500:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                jlo = max(0, i - radius)
                jhi = min(n, i + radius + 1)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try moving mass from i to j
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c
```
