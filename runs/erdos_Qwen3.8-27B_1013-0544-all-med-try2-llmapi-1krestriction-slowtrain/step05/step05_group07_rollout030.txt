def compute_c5(h, n):

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:

            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_A(h, n):
    dx = 2.0 / n
    A = np.zeros((n, 2 * n - 1))

        else:
            s = -shift
            A[s:, k] = (1.0 - h[:n - s])
            A[:n - s, k] += (-h[s:])
    A *= dx
    return A
