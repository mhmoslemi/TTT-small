import numpy as np
import time

def compute_c5_raw(h, n):
    """Returns c_raw (without dx) and the max value * dx."""
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return c_raw, float(np.max(c_raw)) * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_q(h, j, n):
    q = np.zeros(2 * n - 1)
    mid = n - 1
    k1_lo = max(-(n-1), j - (n-1))
    k1_hi = min(n-1, j)
    for k in range(k1_lo, k1_hi + 1):
        q[k + mid] += (1.0 - h[j - k])
    k2_lo = max(-(n-1), -j)
    k2_hi = min(n-1, n - 1 - j)
    for k in range(k2_lo, k2_hi + 1):
        q[k + mid] -= h[j + k]
    return q

def line_search_1d(c, d, d_lo, d_hi, n_iter=100):
    """Minimize max(c + t*d) over t in [d_lo, d_hi]. Returns optimal t."""
    if d_hi - d_lo < 1e-15:
        return (d_lo + d_hi) / 2.0
    lo = d_lo
    hi = d_hi
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        val = c + mid * d
        s_star = np.argmax(val)
        sg = d[s_star]
        if abs(sg) < 1e-14:
            return mid
        if sg > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def sa_coordinate_descent(h, n, deadline, rng, T_init=0.01, T_min=1e-8, cooling=0.9995):
    """SA with exact 1D line search for coordinate moves."""
    dx = 2.0 / n
    c, best_c5 = compute_c5_raw(h, n)
    best_h = h.copy()
    T = T_init
    n_iters = 0
    n_accept = 0
    n_reject = 0
    
    while time.time() < deadline:
        n_iters += 1
        # Choose move type
        r = rng.random()
        if r < 0.7:
            # Gradient-guided move: move from high q[worst] to low q[worst]
            k_star = int(np.argmax(c))
            # Compute gradient at worst lag
            s = k_star - (n - 1)
            # g[j] = change in c[k_star] per unit change in h[j]
            # = q[k_star, j] = (1-h[j-s])*I(0<=j-s<n) - h[j+s]*I(0<=j+s<n)
            j_src = _argmax_grad(h, s, n)
            j_dst = _argmin_grad(h, s, n)
        elif r < 0.85:
            # Random pair
            j_src = rng.randint(n)
            j_dst = rng.randint(n)
        else:
            # Move between two random positions
            j_src = rng.randint(n)
            j_dst = rng.randint(n)
        
        if j_src == j_dst:
            continue
        
        # Exact line search
        qj = get_q(h, j_src, n)
        qm = get_q(h, j_dst, n)
        d = qj - qm
        
        d_lo = max(-h[j_src], h[j_dst] - 1.0)
        d_hi = min(1.0 - h[j_src], h[j_dst])
        
        if d_hi - d_lo < 1e-14:
            continue
        
        delta = line_search_1d(c, d, d_lo, d_hi, n_iter=80)
        
        # Evaluate new state
        h_new = h.copy()
        h_new[j_src] += delta
        h_new[j_dst] -= delta
        h_new = np.clip(h_new, 0.0, 1.0)
        
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(np.max(c_new)) * dx
        
        # Metropolis acceptance
        if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            c = c_new
            n_accept += 1
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            n_reject += 1
        
        # Cooling
        T = max(T * cooling, T_min)
        
        # Periodic re-optimization from best
        if n_iters % 5000 == 0 and best_c5 < c5_new if hasattr(c5_new, 'x') else False:
            pass
    
    return best_h, best_c5

def _argmax_grad(h, s, n):
    """Find j that maximizes the gradient contribution to lag s."""
    # g[j] = (1-h[j-s])*I(0<=j-s<n) - h[j+s]*I(0<=j+s<n)
    g = np.zeros(n)
    # First term: j-s in [0,n-1] → j in [s, s+n-1], intersect [0,n-1]
    j_lo1 = max(0, s)
    j_hi1 = min(n-1, s + n - 1)
    for j in range(j_lo1, j_hi1 + 1):
        g[j] += (1.0 - h[j - s])
    # Second term: j+s in [0,n-1] → j in [-s, n-1-s], intersect [0,n-1]
    j_lo2 = max(0, -s)
    j_hi2 = min(n-1, n - 1 - s)
    for j in range(j_lo2, j_hi2 + 1):
        g[j] -= h[j + s]
    return int(np.argmax(g))

def _argmin_grad(h, s, n):
    g = np.zeros(n)
    j_lo1 = max(0, s)
    j_hi1 = min(n-1, s + n - 1)
    for j in range(j_lo1, j_hi1 + 1):
        g[j] += (1.0 - h[j - s])
    j_lo2 = max(0, -s)
    j_hi2 = min(n-1, n - 1 - s)
    for j in range(j_lo2, j_hi2 + 1):
        g[j] -= h[j + s]
    return int(np.argmin(g))

Wait, there's a subtlety. When I do h[j] += delta, h[m] -= delta, the change in c is:
c_new[s] = c[s] + delta * (q_j[s] - q_m[s])

where q_j = get_q(h_old, j, n) and q_m = get_q(h_old, m, n) are computed BEFORE the update.

But q_j depends on h, and after the update, q_j would be different. However, for the purpose of computing c_new, the formula c_new = c_old + delta * (q_j - q_m) is exact because:
- c is a bilinear function of h
- The change in c when h[j] changes by delta (with all other h fixed) is linear in delta
- So c_new = c_old + delta * (linear coefficient)

The linear coefficient is exactly q_j (computed with the OLD h values for the other positions). So the formula is exact!

Great, so I can maintain c incrementally with O(n) updates per step.

        return (d_lo + d_hi) / 2.0
    lo, hi = d_lo, d_hi
    for _ in range(n_iter):
        mid_t = (lo + hi) / 2.0
        val = c + mid_t * d
        s_star = np.argmax(val)
        sg = d[s_star]
        if abs(sg) < 1e-14:
            return mid_t
        if sg > 0:
            hi = mid_t
        else:
            lo = mid_t
    return (lo + hi) / 2.0

def gradient_index(h, s, n, find_max=True):
    """Find index j that maximizes/minimizes gradient contribution to lag s."""
    g = np.zeros(n)
    j_lo1 = max(0, s)
    j_hi1 = min(n-1, s + n - 1)
    for j in range(j_lo1, j_hi1 + 1):
        g[j] += (1.0 - h[j - s])
    j_lo2 = max(0, -s)
    j_hi2 = min(n-1, n - 1 - s)
    for j in range(j_lo2, j_hi2 + 1):
        g[j] -= h[j + s]
    return int(np.argmax(g)) if find_max else int(np.argmin(g))

def sa_optimize(h_init, n, deadline, seed=42):

    h = project_feasible(h_init, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    dx_c = c * dx
    best_c = float(np.max(dx_c))
    best_h = h.copy()
    
    T = 0.05 * dx  # initial temperature (in units of C5)
    T_min = 1e-10
    cooling = 0.9998

        # Batch of iterations
        batch_size = 1000
        for _ in range(batch_size):
            if time.time() >= deadline:

            iter_count += 1
            
            r = rng.random()
            if r < 0.6:

                k_star = int(np.argmax(c))
                s = k_star - (n - 1)
                j_src = gradient_index(h, s, n, True)
                j_dst = gradient_index(h, s, n, False)
            elif r < 0.85:
                # Semi-random: src from gradient, dst random
                k_star = int(np.argmax(c))
                s = k_star - (n - 1)
                j_src = gradient_index(h, s, n, True)
                j_dst = rng.randint(n)
            else:
                j_src = rng.randint(n)
                j_dst = rng.randint(n)

            qm = get_q(h, j_dst, n)

            d_hi = min(1.0 - h[j_src], h[j_dst])

            delta = line_search_1d(c, d, d_lo, d_hi, n_iter=60)

            h[j_src] += delta
            h[j_dst] -= delta
            h[j_src] = np.clip(h[j_src], 0.0, 1.0)
            h[j_dst] = np.clip(h[j_dst], 0.0, 1.0)
            c = c + delta * d

            if curr_c < best_c:

                best_h = h.copy()
                stall_count = 0
            elif rng.random() < np.exp(-(curr_c - best_c) / T):
                stall_count += 1
            else:

                h[j_dst] += delta
                h[j_src] = np.clip(h[j_src], 0.0, 1.0)
                h[j_dst] = np.clip(h[j_dst], 0.0, 1.0)
                c = c - delta * d
                stall_count += 1

            stall_count = 0
    
    # Final verification
    best_c = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_q(h, j, n):
    q = np.zeros(2 * n - 1)
    mid = n - 1
    kl = max(-(n-1), j - (n-1))
    kh = min(n-1, j)
    for k in range(kl, kh + 1):
        q[k + mid] += 1.0 - h[j - k]
    kl2 = max(-(n-1), -j)
    kh2 = min(n-1, n - 1 - j)
    for k in range(kl2, kh2 + 1):
        q[k + mid] -= h[j + k]
    return q

def ls_1d(c, d, lo, hi, iters=50):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        idx = np.argmax(c + mid * d)
        sg = d[idx]
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5
