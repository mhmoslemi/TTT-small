from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def golden_init(n, seed=0):
    """Initialize using golden ratio Kronecker sequence."""
    alpha = (1 + np.sqrt(5)) / 2
    frac = np.mod(np.arange(n) * (alpha - 1), 1.0)
    idx = np.argsort(frac)[:(n // 2)]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def sa_binary(h, n, deadline, t_start=0.1, t_end=0.001, max_iters=200000, seed=777):
    """Simulated annealing on binary patterns."""
    target = n // 2
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        for i in ones[target:]:
            h[i] = 0
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        for i in zeros[:target - len(ones)]:
            h[i] = 1
    
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max_iters)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i_idx = int(zeros[rng.randint(len(zeros))])
        j_idx = int(ones[rng.randint(len(ones))])
        
        h[i_idx] = 1.0
        h[j_idx] = 0.0
        new_c = compute_c5(h, n)
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i_idx] = 0.0
            h[j_idx] = 1.0
    
    return best_h, best_c

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    """Smooth max SLSQP optimization."""
    g = np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / n)
    m = g.max()
    
    def obj(x):
        gg = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        mm = gg.max()
        s = gg - mm
        return mm + (s.max() + np.log(np.exp(s - s.max()).sum())) / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgrad_polish(h0, n, deadline, max_iter=50000):
    """Subgradient method for the max of affine functions."""
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - best_h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        
        # Backtracking line search
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:

            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:

                found = True

            if lr < 1e-13:
                break
        else:
            c = compute_c5(best_h, n)
            best_c = min(best_c, c)
    
    return best_h, best_c

        h_sa, c_sa = sa_binary(best_h.copy(), n_best, p2 - 30, 0.04, 0.0003, 250000, seed + 555)
        _upd(h_sa, c_sa)
        if time.time() < p2 - 10:
            h_sa2, c_sa2 = sa_binary(best_h.copy(), n_best, p2 - 10, 0.015, 0.0001, 150000, seed + 888)
            _upd(h_sa2, c_sa2)

    # Phase 3: More guided search (15%)
    p3 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p3 - 10:
        h_g, c_g = guided_binary_search(best_h.copy(), n_best, p3 - 10, max_sweeps=80000, seed=seed + 333)
        _upd(h_g, c_g)

    # Phase 4: Gradient polish (12%)
    p4 = t0 + 0.92 * (budget_s - 5)
    if time.time() < p4 - 10:
        for alpha in [50000, 500000, 5000000, 50000000]:
            if time.time() > p4 - 8:
                break
            h_s = slsqp_opt(best_h.copy(), n_best, alpha, 2000, p4 - 5)
            c_s = compute_c5(h_s, n_best)
            _upd(h_s, c_s)

    # Phase 5: Final subgradient (3%)
    if time.time() < dl - 3:
        h_sp, c_sp = subgrad_polish(best_h.copy(), n_best, dl - 3, max_iter=20000)
        _upd(h_sp, c_sp)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
