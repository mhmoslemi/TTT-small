                delta_min=0.001, delta_max=0.05, cooling=0.99995):
    h_curr = h.copy()
    c_curr = compute_c5(h_curr, n)
    h_best = h_curr.copy()
    c_best = c_curr
    T = t_init
    n_acc = 0
    n_tot = 0
    # Use multiple restarts within the SA phase
    n_restarts = 3
    seg_deadline = (deadline - time.time()) / n_restarts
    for r in range(n_restarts):
        seg_end = time.time() + seg_deadline
        T = t_init * (0.7 ** r)
        while time.time() < seg_end:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(delta_min, delta_max)
            forward = False
            if h_curr[i] + delta <= 1.0 and h_curr[j] - delta >= 0.0:
                h_curr[i] += delta
                h_curr[j] -= delta
                forward = True
            elif h_curr[j] + delta <= 1.0 and h_curr[i] - delta >= 0.0:
                h_curr[j] += delta
                h_curr[i] -= delta
                forward = False
            else:
                continue
            c_new = compute_c5(h_curr, n)
            dc = c_new - c_curr
            n_tot += 1
            if dc < 0 or (T > t_final and rng.random() < np.exp(-dc / T)):
                c_curr = c_new
                n_acc += 1
                if c_new < c_best:
                    c_best = c_new
                    h_best = h_curr.copy()
            else:
                if forward:
                    h_curr[i] -= delta
                    h_curr[j] += delta
                else:
                    h_curr[j] -= delta
                    h_curr[i] += delta
            T = max(T * cooling, t_final)
            # Adaptive: if acceptance too low, increase T
            if n_tot % 5000 == 0 and n_tot > 10000:
                rate = n_acc / n_tot
                if rate < 0.05 and T < t_init:
                    T = min(T * 1.5, t_init)
                    n_acc = 0
                    n_tot = 0
            # If stuck, restart from best
            if n_tot % 20000 == 0 and n_acc / max(1, n_tot) < 0.01:
                h_curr = h_best.copy()
                c_curr = c_best
                n_acc = 0
                n_tot = 0
    return h_best, c_best

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def basin_hopping(h, n, deadline, rng, n_restarts=8, per_time=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_start = time.time()
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        # Perturb
        scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        # SLSQP polish
        alloc = min(per_time, deadline - time.time() - 5)
        if alloc < 5:
            break
        ht = slsqp_opt(hp, n, 20000, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n = 100

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial SLSQP polish (fast, low alpha)
    if time.time() < deadline - 60:
        ht = slsqp_opt(best_h.copy(), n, 2000, maxiter=800)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 2: Simulated annealing (main exploration)
    sa_end = min(t0 + budget_s * 0.45, deadline - 200)
    if time.time() < sa_end:
        h_sa, c_sa = sa_optimize(best_h.copy(), n, sa_end, rng,
                                 t_init=0.015, t_final=0.00003,
                                 delta_min=0.002, delta_max=0.06,
                                 cooling=0.99995)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 3: Basin hopping with SLSQP
    bh_end = min(time.time() + 150, deadline - 100)
    if time.time() < bh_end and n <= 250:
        h_bh, c_bh = basin_hopping(best_h.copy(), n, bh_end, rng,
                                   n_restarts=10, per_time=25)
        if c_bh < best_c:
            best_c = c_bh
            best_h = h_bh

    # Phase 4: Progressive SLSQP with increasing alpha
    for alpha in [10000, 50000, 200000, 1000000]:
        if time.time() > deadline - 60:
            break
        ht = slsqp_opt(best_h.copy(), n, alpha, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    # Phase 5: Pair search (fine tuning)
    if n <= 200 and time.time() < deadline - 15:
        best_h, best_c = pair_search(best_h, n, deadline - 5,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    elif time.time() < deadline - 15:
        best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                          [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                          radius=60)

    # Phase 6: Final SLSQP + micro pair search
    if time.time() < deadline - 3:
        ht = slsqp_opt(best_h.copy(), n, 2000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if n <= 200 and time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                     [0.0001, 0.00005, 0.00002])
    elif time.time() < deadline - 1:
        best_h, best_c = near_pair_search(best_h, n, deadline - 0.5,
                                          [0.0001, 0.00005], radius=40)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
