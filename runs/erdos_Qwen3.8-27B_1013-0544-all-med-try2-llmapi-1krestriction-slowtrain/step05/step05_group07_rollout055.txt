import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_ALPHA = 0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h_p.sum()
    if diff > 1e-10:
        room = 1.0 - h_p
        s = room.sum()
        if s > 0:
            h_p += (diff / s) * room
    elif diff < -1e-10:
        s = h_p.sum()
        if s > 0:
            h_p += (diff / s) * h_p
    return np.clip(h_p, 0, 1)

def smooth_obj(x):
    n, alpha = _G_N, _G_ALPHA
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh_max = sh.max()
    val = cmax + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha
    return val

def smooth_grad(x):
    n, alpha = _G_N, _G_ALPHA
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in np.where(w > 1e-7)[0]:
        k = k_idx - (n - 1)
        wk = w[k_idx] * dx
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += wk * (1.0 - x[k:])
                grad[k:] += wk * (-x[:nk])
        else:
            ka = -k
            nk = n - ka
            if nk > 0:
                grad[ka:] += wk * (1.0 - x[:nk])
                grad[:nk] += wk * (-x[ka:])
    return grad

def coord_obj(x):
    return compute_c5(x, _G_N)

def constraint_sum(x, n):
    return x.sum() - n / 2.0

def constraint_sum_jac(x, n):
    return np.ones(n)

def coord_descent(h, n, deadline, max_sweeps=15):
    global _G_N
    _G_N = n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            old_val = best_h[i]
            lo, hi = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            best_h[i] = c_pt
            fc = compute_c5(best_h, n)
            best_h[i] = d_pt
            fd = compute_c5(best_h, n)
            for _ in range(35):
                if b - a < 1e-7:
                    break
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    best_h[i] = c_pt
                    fc = compute_c5(best_h, n)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    best_h[i] = d_pt
                    fd = compute_c5(best_h, n)
            mid = (a + b) * 0.5
            best_h[i] = mid
            cm = compute_c5(best_h, n)
            if cm < best_c - 1e-14:
                best_c = cm
                improved = True
            else:
                best_h[i] = old_val
        best_h = project_feasible(best_h, n)
        cc = compute_c5(best_h, n)
        if cc < best_c:
            best_c = cc
            improved = True
        elif not improved:
            break
    return best_h, best_c

def pair_local_search(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for _ in range(800):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for s in (1.0, -1.0):
                    hi = best_h[i] + s * step
                    hj = best_h[j] - s * step
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i] = hi
                        best_h[j] = hj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= s * step
                            best_h[j] += s * step
    return best_h
