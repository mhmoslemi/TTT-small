    dx = 2.0 / n

        w = np.exp(np.clip(alpha * (c - m), -50, 500))
        w /= w.sum()
        dc = compute_dc(x, n)
        return alpha * (dc @ w)
    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_dc(h, n):
    k_arr = np.arange(2 * n - 1) - (n - 1)
    I = np.arange(n)[:, None]
    K = k_arr[None, :]
    idx1 = I - K
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = I + K
    valid2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return valid1.astype(np.float64) - h1 - h2

def dc_row(h, n, i, k_arr):
    idx1 = i - k_arr
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i + k_arr
    valid2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return valid1.astype(np.float64) - h1 - h2

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = np.clip(alpha * (c - m), -50, 500)
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _slsqp_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    w = np.exp(np.clip(alpha * (c - m), -50, 500))
    w /= w.sum()
    dc = compute_dc(x, n)
    return alpha * (dc @ w)

def slsqp_opt(h, n, alpha, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pgd_opt(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [200, 600, 2000, 6000, 20000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        h = best_h.copy()
        lr = 3.0 / (alpha * n)
        for it in range(400):
            if time.time() > deadline - 10:
                break
            c = compute_corr(h, n)
            m = c.max()
            w = np.exp(np.clip(alpha * (c - m), -50, 500))
            w /= w.sum()
            dc = compute_dc(h, n)
            g = alpha * (dc @ w)
            step_lr = lr
            converged = False
            for _ in range(40):
                h_new = np.clip(h - step_lr * g, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < compute_c5(h, n) - 1e-15:
                    h = h_new
                    converged = True
                    break
                step_lr *= 0.5
            if not converged:
                break
            c_curr = compute_c5(h, n)
            if c_curr < best_c - 1e-13:
                best_c = c_curr
                best_h = h.copy()
    return best_h, best_c

def fast_pair(h, n, deadline):
    best_c = compute_c5(h, n)
    c = compute_corr(h, n)
    k_arr = np.arange(2 * n - 1) - (n - 1)
    dc = compute_dc(h, n)
    deltas = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        nm = np.max(c + d * (dc[i] - dc[j]))
                        if nm < best_c - 1e-14:
                            h[i] += d; h[j] -= d
                            c += d * (dc[i] - dc[j])
                            dc[i] = dc_row(h, n, i, k_arr)
                            dc[j] = dc_row(h, n, j, k_arr)
                            best_c = nm
                            improved = True
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        nm = np.max(c - d * (dc[i] - dc[j]))
                        if nm < best_c - 1e-14:
                            h[i] -= d; h[j] += d
                            c -= d * (dc[i] - dc[j])
                            dc[i] = dc_row(h, n, i, k_arr)
                            dc[j] = dc_row(h, n, j, k_arr)
                            best_c = nm
                            improved = True
            if not improved:
                break
    best_c = compute_c5(h, n)
    return h, best_c

    best_h = h.copy()
    t_slsqp = min(time.time() + 40, deadline - 30)
