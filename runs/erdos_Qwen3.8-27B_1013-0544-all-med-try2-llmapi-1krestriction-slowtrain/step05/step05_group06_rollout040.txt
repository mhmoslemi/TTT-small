            best_c = c_new

Let me try approach 1: **Greedy pair selection**.

At each step, instead of trying all pairs, try only the "most promising" pairs. A pair (i,j) is promising if moving mass from j to i (or vice versa) can significantly reduce the maximum g.

A quick criterion: compute the "descent direction" from the subgradient. The subgradient at h is:
grad[p] = Σ_{m ∈ active} w_m * F[p, m]

Then the pair (i,j) with the largest (grad[i] - grad[j]) is the most promising (for moving mass from j to i).

So the algorithm would be:
1. Compute subgradient grad
2. Find the pair (i,j) with max (grad[i] - grad[j]) (or grad[j] - grad[i])
3. For that pair, find the optimal δ by binary search
4. Make the move
5. Update F and g
6. Repeat

This is O(n²) to find the best pair (or O(n log n) with a priority queue), plus O(log * n) for the binary search. Much faster than O(n² * K² * n) for the full enumeration.

    dm = np.arange(2*n-1) - (n-1)
    m_idx = np.arange(2*n-1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    F = compute_F(h, n)
    current_f = g.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute subgradient (average over top active lags)
        active = np.argsort(g)[::-1][:min(10, 2*n-1)]
        grad = np.zeros(n)
        for m_i in active:
            grad += F[:, m_i]
        grad /= len(active)
        
        # Find best pair: maximize grad[i] - grad[j] (move mass from j to i)
        # Or minimize grad[i] - grad[j] (move mass from i to j)
        # We want to decrease max(g), so we want to decrease all active g(m)
        # Moving δ from j to i changes g by δ * (F[i] - F[j])
        # We want F[i] - F[j] to be "negative" on active lags
        
        # Simple heuristic: pair with max grad difference
        order = np.argsort(grad)
        i = order[0]  # min grad
        j = order[-1]  # max grad
        # Move from j to i: δ > 0 means h[i] += δ, h[j] -= δ
        # Change in g: δ * (F[i] - F[j])
        # We want this to decrease active g's
        
        s = F[i] - F[j]
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        
        if d_hi - d_lo < 1e-14:
            continue
        
        # Binary search for optimal δ
        best_delta = 0.0
        best_val = current_f
        lo, hi = d_lo, d_hi
        for _ in range(40):
            mid = (lo + hi) / 2
            val = np.max(g + mid * s)
            if val < best_val:
                best_val = val
                best_delta = mid
            # Subgradient of max(g + mid*s) at mid is s[argmax(g + mid*s)]
            m_star = np.argmax(g + mid * s)
            if s[m_star] < 0:
                lo = mid  # f is decreasing, try larger δ
            elif s[m_star] > 0:
                hi = mid  # f is increasing, try smaller δ
            else:
                break
        
        if best_delta != 0.0 and best_val < current_f:
            h[i] += best_delta
            h[j] -= best_delta
            g += best_delta * s
            update_F(F, i, j, best_delta, n, dm)
            current_f = g.max()
            c = current_f
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = int(dm[m_idx])
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], m_idx[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    F[j - dm[v4], m_idx[v4]] += delta

def find_best_delta(g, s, d_lo, d_hi):
    """Binary search for optimal delta that minimizes max(g + delta*s)."""
    if d_hi - d_lo < 1e-14:
        return 0.0, g.max()
    best_delta = 0.0
    best_val = float(g.max())
    lo, hi = d_lo, d_hi
    for _ in range(35):
        mid = (lo + hi) / 2.0
        val = float(np.max(g + mid * s))
        if val < best_val - 1e-15:
            best_val = val
            best_delta = mid
        gs = g + mid * s
        m_star = int(np.argmax(gs))
        sm = s[m_star]
        if sm < -1e-12:
            lo = mid
        elif sm > 1e-12:
            hi = mid
        else:
            break
    # Also check endpoints
    for d in [d_lo, d_hi]:
        val = float(np.max(g + d * s))
        if val < best_val - 1e-15:
            best_val = val
            best_delta = d
    return best_delta, best_val

def greedy_pair_descent(h, n, deadline, max_sweeps=500, top_m=5, random_frac=0.1):

    m_idx = np.arange(2 * n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    F = compute_F(h, n)
    current_f = float(g.max())
    stall = 0
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        active = np.argsort(g)[::-1][:min(10, 2 * n - 1)]
        grad = np.zeros(n)
        for m_i in active:
            grad += F[:, m_i]
        grad /= float(len(active))
        
        sweep_improved = 0
        for i in range(n):

            if rng.random() < random_frac:

            else:
                diff = grad[i] - grad

def subgrad_optimize(h, n, deadline, steps=10000):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    beta = 0.85
    f_lb = 0.35
    lr_init = 0.3
    lr_decay = 0.999
    
    for k in range(steps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_f = float(g.max())
        
        if current_f < best_c:
            best_c = current_f
            best_h = h.copy()
        
        active = np.argsort(g)[::-1][:min(5, 2 * n - 1)]
        grad = np.zeros(n)
        for m_i in active:
            d = int(dm[m_i])
            if d >= 0:
                if d < n:
                    grad[d:] += (1.0 - h[:n - d])
                    grad[:n - d] -= h[d:]
            else:
                dd = -d
                if dd < n:
                    grad[:n - dd] += (1.0 - h[dd:])
                    grad[dd:] -= h[:-dd]
        grad *= dx / float(len(active))
        
        lr = lr_init * (lr_decay ** k)
        lr = min(lr, (current_f - f_lb) / (np.dot(grad, grad) + 1e-10))
        lr = min(lr, 1.0)
        
        momentum = beta * momentum - lr * grad
        h = project_feasible(h + momentum, n)
    
    return best_h, best_c

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):

def greedy_pair_descent(h, n, deadline, max_sweeps=2000, top_m=5, random_frac=0.15):

    m_idx = np.arange(2 * n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    F = compute_F(h, n)
    current_f = float(g.max())
    stall = 0
    rng = np.random.RandomState(77)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        active = np.argsort(g)[::-1][:min(10, 2 * n - 1)]
        grad = np.zeros(n)
        for m_i in active:
            grad += F[:, m_i]
        grad /= float(len(active))
        
        sweep_improved = 0
        for i in range(n):

            if rng.random() < random_frac:

def subgrad_optimize(h, n, deadline, steps=20000):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    beta = 0.85
    f_lb = 0.35
    lr_init = 0.2
    lr_decay = 0.9995
    
    for k in range(steps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_f = float(g.max())
        if current_f < best_c:
            best_c = current_f
            best_h = h.copy()
        
        active = np.argsort(g)[::-1][:min(5, 2 * n - 1)]
        grad = np.zeros(n)
        for m_i in active:
            d = int(dm[m_i])
            if d >= 0:
                if d < n:
                    grad[d:] += (1.0 - h[:n - d])
                    grad[:n - d] -= h[d:]
            else:
                dd = -d
                if dd < n:
                    grad[:n - dd] += (1.0 - h[dd:])
                    grad[dd:] -= h[:-dd]
        grad *= dx / float(len(active))
        
        lr = lr_init * (lr_decay ** k)
        polyak_lr = (current_f - f_lb) / (np.dot(grad, grad) + 1e-10)
        lr = min(lr, polyak_lr, 1.0)
        
        momentum = beta * momentum - lr * grad
        h = project_feasible(h + momentum, n)
    return best_h, best_c

            for o in ones:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(o)
                    ones.append(z)
                    ones.sort()
                    zeros.remove(z)
                    zeros.append(o)
                    zeros.sort()
                    improved = True
                    break
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def slsqp_polish(h, n, deadline, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = 50000.0 * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / 50000.0
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
