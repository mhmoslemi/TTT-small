    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    """Compute softmax-weighted gradient of max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -100, 100)
    e = np.exp(bc)
    se = e.sum()
    w = e / se
    # Only process lags with significant weight
    g = np.zeros(n)
    threshold = 1e-8
    for j in range(len(corr)):
        if w[j] < threshold:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        else:
            k = abs(lag)
            if k < n:
                g[k:] += w[j] * (1.0 - h[:-k])
                g[:-k] -= w[j] * h[k:]
    return g * dx, m * dx, w

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -200, 200)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        else:
            k = abs(lag)
            if k < n:
                g[k:] += w[j] * (1.0 - h[:-k])
                g[:-k] -= w[j] * h[k:]
    return val, g * dx

def slsqp_opt(h, n, beta, maxiter=3000, ftol=1e-17):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[0]
    def gradf(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[1]
    def eqf(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    res = minimize(objf, np.clip(h, 0, 1), jac=gradf, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

def subgrad_refine(h, target, deadline, beta0=500, beta1=50000):
    n = len(h)
    hc = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    avg_h = np.zeros(n)
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 30000.0, 1.0)
        beta = beta0 * (beta1 / beta0) ** frac
        g, _, _ = softmax_obj_grad(hc, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.04
        improved = False
        for _ in range(7):
            ht = project_sum(hc - step * g / gn, target)
            c5t = compute_c5(ht)
            if c5t < compute_c5(hc) - 1e-15:
                hc = ht
                improved = True
                break
            step *= 0.5
        if not improved:
            hc = project_sum(hc - step * g / gn, target)
        avg_h += (hc - avg_h) / it
        c5c = compute_c5(hc)
        if c5c < best_c5:
            best_h = hc.copy()
            best_c5 = c5c
            stall = 0
        else:
            stall += 1
        if it % 2000 == 0:
            c5a = compute_c5(avg_h)
            if c5a < best_c5:
                best_h = avg_h.copy()
                best_c5 = c5a
                hc = best_h.copy()
        if stall > 5000 and it % 1000 == 0:
            perturb = 0.005 * (stall / 5000.0)
            ht = project_sum(best_h + np.random.randn(n) * perturb, target)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h = ht
                best_c5 = c5t
                hc = ht.copy()
                avg_h = ht.copy()
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() and initial_h_values is not None:

    # Phase 2: Subgradient refinement with averaging
    dl = t0 + budget_s - 200

    if remaining() > 200:

        for beta in [200, 1000, 5000]:

            g /= gn

                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-16:
                    best_h = ht
                    best_c5 = c5t
                    break

    return best_h, best_c5, len(best_h)
