import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _con_fun(h, ts):
    return h.sum() - ts

def _con_jac(h, ts):
    return np.ones_like(h)

def compute_subgrad(h, n, top_indices, weights):
    grad = np.zeros(n)
    for w, m in zip(weights, top_indices):
        k = m - (n - 1)
        if k >= 0:
            m_val = n - k
            grad[k:] += w * (1.0 - h[:m_val])
            grad[:m_val] -= w * h[k:]
        else:
            la = -k
            m_val = n - la
            grad[:m_val] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:m_val]
    return grad

def nesterov_polish(h, n, deadline, seed, n_top=15, beta_init=5.0):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    y = h.copy()
    h_prev = h.copy()
    t_iter = 1.0
    base_lr = 0.08
    stagnant = 0
    beta = beta_init

    for it in range(50000):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        w = np.exp(beta * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = compute_subgrad(y, n, top_idx, w)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
        alpha = (t_iter - 1.0) / t_new
        lr = base_lr / (np.sqrt(t_iter) + 1.0)
        y_new = y + alpha * (y - h_prev) - lr * grad
        y_new = project_feasible(y_new, ts)
        c_new = compute_c5(y_new)
        if c_new < best_c5:
            best_h = y_new.copy()
            best_c5 = c_new
            stagnant = 0
        else:
            stagnant += 1
        h_prev = y
        y = y_new
        t_iter = t_new
        if stagnant % 150 == 0 and stagnant > 0:
            beta *= 0.82
            base_lr *= 0.92
            if beta < 0.8:
                beta = beta_init
                base_lr = 0.08
                stagnant = 0
        if stagnant > 5000 and beta < 1.5:
            break
    return best_h, best_c5

def coordinate_descent(h, n, deadline, seed, n_sweeps=300):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    order = np.random.permutation(n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in order:
            if time.time() > deadline:
                break
            h_save = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            diff = c1 - c0
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(c0 + m1 * diff) < np.max(c0 + m2 * diff):
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo + hi) / 2.0
        h = project_feasible(h, ts)
        c_new = compute_c5(h)
        if c_new < c5:
            c5 = c_new
    return h, c5

def binary_rounding(h, n, deadline, seed, n_passes=50):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    order = np.random.permutation(n)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False

                break
            for val in [0.0, 1.0]:
                h_try = h.copy()
                h_try[i] = val
                h_try = project_feasible(h_try, ts)
                c_try = compute_c5(h_try)
                if c_try < c5 - 1e-12:
                    h = h_try
                    c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def slsqp_opt(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in [50, 200, 1000, 5000, 20000, 80000]:
        if time.time() > deadline:
            break
        for p in [0.0, 0.003, 0.01]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(lambda x, b=beta, d=dx: smooth_max(np.correlate(x, 1.0-x, mode='full'), b) * d,
                             x0, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 8000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5
