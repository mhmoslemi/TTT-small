def coordinate_descent(h, n, max_passes=100, deadline=None):
    """Coordinate descent: for each position, try values {0, h[i], 1} and pick best"""
    c5 = compute_c5(h)
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            best_val = old_val
            best_c5 = c5
            # Try 0
            h[i] = 0.0
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_val = 0.0
            # Try 1
            h[i] = 1.0
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_val = 1.0
            # Restore or set best
            h[i] = best_val
            if best_val != old_val:
                improved = True
                c5 = best_c5
        if not improved:
            break
    return h, c5
```

def project_feasible_fast(h, ts):
    """Fast projection: clip to [0,1] and adjust sum"""
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)
```

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def subgradient_polish(h, n, deadline, seed, max_iters=20000, top_k=5, base_lr=0.003):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = base_lr
    stagnant = 0
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top-k lags
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(15.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for j in range(len(top_idx)):
            lag = top_idx[j] - (n - 1)
            if lag >= 0:
                m = n - lag
                if m > 0:
                    # dA(lag)/dh[i]
                    # (1-h[i-lag]) for i in [lag, n-1]
                    idx1 = np.arange(lag, n)
                    grad[idx1] += w[j] * (1.0 - h[idx1 - lag])
                    # -h[i+lag] for i in [0, m-1]
                    idx2 = np.arange(m)
                    grad[idx2] -= w[j] * h[idx2 + lag]
            else:
                la = -lag
                m = n - la
                if m > 0:
                    # dB(la)/dh[i]
                    # (1-h[i+la]) for i in [0, m-1]
                    idx1 = np.arange(m)
                    grad[idx1] += w[j] * (1.0 - h[idx1 + la])
                    # -h[i-la] for i in [la, n-1]
                    idx2 = np.arange(la, n)
                    grad[idx2] -= w[j] * h[idx2 - la]
        
        velocity = 0.85 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.01, 0.008)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.92
            if stagnant > 500 and lr < 1e-6:
                break
    
    return best_h, best_c5

def coord_descent(h, n, deadline, seed, max_passes=500):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for p in range(max_passes):
        if time.time() > deadline:

                    best_c5 = c5
                    best_h = h.copy()
        if not improved and p > 10:

    return best_h, best_c5

def binary_local_search(h, n, deadline, seed, max_iters=50000):
    """Swap-based local search on near-binary solutions"""
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick random pair (one high, one low)
        high_mask = h > 0.5
        low_mask = h < 0.5
        n_high = high_mask.sum()
        n_low = low_mask.sum()
        if n_high == 0 or n_low == 0:
            break
        i = np.random.choice(np.where(high_mask)[0])
        j = np.random.choice(np.where(low_mask)[0])
        
        # Swap: h[i] and h[j]
        h[i], h[j] = h[j], h[i]
        # This preserves the sum!
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
            c5 = c5_try
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
            # Try complement instead: h[i] -> 1-h[i], h[j] -> h[j] + 2*h[i] - 1
            v_i = 1.0 - h[i]
            v_j = h[j] + 2.0 * h[i] - 1.0
            if 0.0 <= v_j <= 1.0:
                h[i] = v_i
                h[j] = v_j
                h_p = project_feasible(h, ts)
                c5_try = compute_c5(h_p)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_p.copy()
                    h = h_p
                    c5 = c5_try
                else:
                    h = best_h.copy()  # revert
    
    return best_h, best_c5

def slsqp_optimize(h, n, deadline, seed):

    ts = n / 2.0
    dx = 2.0 / n

                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12

        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

                c5 = c_try
                stagnant = 0
            else:
                h[i], h[j] = vi, vj
                stagnant += 1
        # Try pushing i toward 1, compensate at j
        if vj - (1.0 - vi) >= -1e-12:
            h[i] = 1.0
            h[j] = max(vj - (1.0 - vi), 0.0)
            h = project_feasible(h, ts)
            c_try = compute_c5(h)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h.copy()
                c5 = c_try
                stagnant = 0
            else:
                h[i], h[j] = vi, vj
                stagnant += 1
        if stagnant > 5000:
            h = project_feasible(h + np.random.randn(n) * 0.01, ts)
            c5 = compute_c5(h)
            stagnant = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    best_n = 200

    parent = None
    try:
        pv = initial_h_values
        if pv is not None:
            parent = np.array(pv, dtype=float).copy()
    except Exception:
        pass
    if parent is None:
        parent = best_h.copy()
    pn = len(parent)

    # Phase 1: Subgradient on parent (10%)
    d1 = t0 + budget_s * 0.10
    h1, c1 = subgradient_polish(parent.copy(), pn, d1, seed, top_k=5, base_lr=0.004)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, pn

    # Phase 2: SLSQP (15%)
    d2 = t0 + budget_s * 0.25
    h2, c2 = slsqp_optimize(best_h.copy(), best_n, d2, seed + 100)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2, c2, best_n

    # Phase 3: Coordinate descent - main new method (30%)
    d3 = t0 + budget_s * 0.55
    h3, c3 = coord_descent(best_h.copy(), best_n, d3, seed + 200, max_passes=3000)
    if c3 < best_c5:
        best_h, best_c5, best_n = h3, c3, best_n

    # Phase 4: Paired coordinate descent (12%)
    d4 = t0 + budget_s * 0.67
    h4, c4 = paired_coord_descent(best_h.copy(), best_n, d4, seed + 300)
    if c4 < best_c5:
        best_h, best_c5, best_n = h4, c4, best_n

    # Phase 5: Subgradient refinement (10%)
    d5 = t0 + budget_s * 0.77
    h5, c5 = subgradient_polish(best_h.copy(), best_n, d5, seed + 400, top_k=8, base_lr=0.002)
    if c5 < best_c5:
        best_h, best_c5, best_n = h5, c5, best_n

    # Phase 6: Another coordinate descent round (10%)
    d6 = t0 + budget_s * 0.87
    h6, c6 = coord_descent(best_h.copy(), best_n, d6, seed + 500, max_passes=2000)
    if c6 < best_c5:
        best_h, best_c5, best_n = h6, c6, best_n

    # Phase 7: SLSQP fine-tune (8%)
    d7 = t0 + budget_s * 0.95
    h7, c7 = slsqp_optimize(best_h.copy(), best_n, d7, seed + 600)
    if c7 < best_c5:
        best_h, best_c5, best_n = h7, c7, best_n

    # Phase 8: Final micro-polish (5%)
    d8 = t0 + budget_s * 0.99
    h8, c8 = subgradient_polish(best_h.copy(), best_n, d8, seed + 700, top_k=3, base_lr=0.001)
    if c8 < best_c5:
        best_h, best_c5, best_n = h8, c8, best_n

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
