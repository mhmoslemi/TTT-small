from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytic_grad_max(h, n, k_star):
    m = k_star - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        j_max1 = n - 1 - m
        grad[0:j_max1+1] += (1 - h[m:m+n])
        if m > 0:
            grad[m:] -= h[0:n-m]
    else:
        mp = -m
        j_min1 = mp
        grad[mp:] += (1 - h[0:n-mp])
        grad[0:n-mp] -= h[mp:]
    return grad * (2.0 / n)

def slsqp_step(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        mx = sh.max()
        return m + (mx + np.log(np.exp(sh - mx).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(2*n-1)
        # gradient of each g[k] w.r.t. x
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            grad[k] = w[k] * alpha
        # Now compute dx of g[k]/dx[j]
        gx = np.zeros(n)
        for k in range(2*n-1):
            if abs(grad[k]) < 1e-15:
                continue
            m_lag = k - (n-1)
            if m_lag >= 0:
                j1 = min(n-1-m_lag, n-1)
                gx[0:j1+1] += grad[k] * (1 - x[m_lag:m_lag+j1+1])
                if m_lag > 0:
                    gx[m_lag:] += grad[k] * (-x[0:n-m_lag])
            else:
                mp = -m_lag
                gx[mp:] += grad[k] * (1 - x[0:n-mp])
                gx[0:n-mp] += grad[k] * (-x[mp:])
        return gx * dx
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def subgrad_exact(h, n, deadline, max_iter=200000):

    lr = 0.05
    stale = 0

        m = g.max()

        grad = analytic_grad_max(h, n, ks)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:

        found = False

            step = lr * f / ng

            if ct < m - 1e-13:

                found = True
                break

            lr *= 0.9
            stale += 1
            if stale > 200 or lr < 1e-12:
                break
        else:
            stale = 0
            lr = min(lr * 1.05, 0.5)
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=500):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_hb = h_bin.copy()
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_swap = None
        best_c_swap = best_c
        best_pair = None
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                h_bin[i], h_bin[j] = 0.0, 1.0
                if c < best_c_swap - 1e-12:
                    best_c_swap = c
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h_bin[i], h_bin[j] = 1.0, 0.0
            best_c = best_c_swap
            best_hb = h_bin.copy()
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
        else:
            break
    return best_hb, best_c

def pair_refine(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                for j in range(n-1, i+1, -2):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s*d, h[j] - s*d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-13:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = hi - s*d, hj + s*d
    return h, best_c
