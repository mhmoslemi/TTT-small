    se = ev.sum()

    _target_c[0] = n / 2.0

                      jac=_grad, constraints=constraints,

        return h0.copy()

    best_c5 = compute_c5(h)
    best_h = h.copy()

from scipy.optimize import minimize
import time

_bc = [100.0]
_tc = [100.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def psum_val(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    pmax = np.power(np.maximum(corr, 0), p).sum() ** (1.0 / p)
    return float(pmax)

def psum_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cp = np.maximum(corr, 0)
    cp_p = np.power(cp, p)
    pmax = cp_p.sum() ** (1.0 / p)
    if pmax < 1e-30:
        return np.zeros(n)
    weights = (cp_p / pmax ** p) / (2 * n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)

            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag:
            grad[lag:] += wk * (1.0 - h[:n - lag])

        elif lag > -n:
            m2 = -lag
            grad[:n - m2] += wk * (1.0 - h[m2:])

    return lse_val_grad(x, _bc[0])[0]

def _grad_lse(x):
    return lse_val_grad(x, _bc[0])[1]

def _con_eq(x):
    return x.sum() - _tc[0]

def _con_jac(x):
    return np.ones(len(x))

def slsqp_solve(h0, beta, maxiter=5000):
    n = len(h0)
    _bc[0] = beta
    _tc[0] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:
        res = minimize(_obj_lse, h0, method='SLSQP', bounds=bounds,
                      jac=_grad_lse, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x, n / 2.0)
    except:
        return h0.copy()

def subgrad_opt(h, target, top_k, max_iter, time_left_fn, base_step=0.01):

    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_imp = 0
    step = base_step
    for it in range(max_iter):
