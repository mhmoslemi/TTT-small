```python
def weighted_projected_gradient(h, weights, target, n, lr):
    """One step of projected gradient on weighted sum of correlations."""
    # Compute gradient of sum_k weights[k] * c_k
    grad = np.zeros(n)
    # c_k for lag k: 
    # For k >= 0: c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # dc_k/dh[j] = (1-h[j+k]) for j < n-k, minus h[j-k] for j >= k
    for k in range(-n+1, n):
        if abs(weights[k+n-1]) < 1e-15:
            continue
        w = weights[k+n-1]
        if k >= 0:
            # dc/dh[j] for j in [0, n-1-k): (1-h[j+k])
            grad[:n-k] += w * (1.0 - h[k:])
            # dc/dh[j] for j in [k, n-1]: -h[j-k]
            grad[k:] += -w * h[:n-k]
        else:
            la = -k
            # dc/dh[j] for j in [la, n-1]: (1-h[j-la])
            grad[la:] += w * (1.0 - h[:n-la])
            # dc/dh[j] for j in [0, n-1-la]: -h[j+la]
            grad[:n-la] += -w * h[la:]
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_h(h_new, target)
    return h_new
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:

            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)

    return smooth_max(c, beta) * (2.0 / n)

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

    best_h = cur.copy()
    best_c5 = compute_c5(cur)

    eta = eta_init

    best_patience = 0

            break

            else:
                la = -k
                grad[la:] += w * (1.0 - cur[:n-la])
                grad[:n-la] += -w * cur[la:]

        c_vals = np.correlate(cur, 1.0 - cur, mode="full")

            best_h = cur.copy()

        c_max = np.max(c_vals)
        log_w = np.log(weights + 1e-30) + eta * (c_vals - c_max)
        log_w -= np.max(log_w)
        weights = np.exp(log_w)
        weights /= weights.sum()

        eta = eta_init * (1.0 + 2.0*frac)

            # Perturb and restart
            cur = project_h(best_h + 0.01 * np.random.randn(n), target)
            weights = np.ones(n_lags) / n_lags
            patience = 0
    
    return best_h, best_c5
```

```python
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * 2.0/n)

def compute_lags(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode="full") * 2.0/n

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

    return m + np.log(np.sum(np.exp(np.clip(beta*(x-m), -400, 400))/len(x))) / beta
```

Let me also think about the learning rate schedule. For projected gradient on a problem with L-Lipschitz gradient:
- Step size = 1/L gives convergent (sublinear) rate
- Nesterov acceleration gives linear rate (for strongly convex) or O(1/t²) (for convex)

Since our problem is not strongly convex (the feasible set has flat parts), we get sublinear convergence at best. But the multiplicative weights outer loop helps by focusing on the right direction.

A practical step size: lr = 0.1 initially, decreasing to 0.01 over the optimization.

For the eta (weight update sharpness): start at 5, increase to 50. Higher eta means more focus on the worst lags.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def compute_lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * 2.0 / n

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -400, 400)) / len(x))) / beta

def slsqp_obj(h_in, beta, n):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

def compute_weighted_grad(h, weights, n):
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-12:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            la = -k
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    return grad

def mw_optimize(h_init, target, n, t0, budget, max_iter=150000, lr_init=0.06, eta_init=5.0):
    cur = h_init.copy()
    best_h = cur.copy()
    best_c5 = compute_c5(cur)
    n_lags = 2 * n - 1
    weights = np.ones(n_lags) / n_lags
    stall = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 5:
            break
        frac = it / max_iter
        lr = lr_init * (1.0 - 0.5 * frac)
        eta = eta_init * (1.0 + 4.0 * frac)
        grad = compute_weighted_grad(cur, weights, n)
        cur = project_h(cur - lr * grad, target)
        c_vals = np.correlate(cur, 1.0 - cur, mode="full")
        c5 = float(np.max(c_vals) * (2.0 / n))
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = cur.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8000:
            cur = project_h(best_h, target)
            stall = 0
        c_max = np.max(c_vals)
        exp_arg = np.clip(eta * (c_vals - c_max), -50, 50)
        weights *= np.exp(exp_arg)
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.ones(n_lags) / n_lags
        else:
            weights /= ws
    return best_h, best_c5

def coord_polish(h, target, n, t0, budget, max_sweeps=200):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.025, -0.025, 0.01, -0.01, 0.005, -0.005]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 3:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 384
        best_h = np.full(n, 0.5)
        best_h[:n // 2] = 0.75
        best_h[n // 2:] = 0.25
        best_h = project_h(best_h, n / 2.0)

    n = len(best_h)
    target = n / 2.0
    best_c5 = compute_c5(best_h)

    # Phase 1: Multiplicative weights + projected gradient (main workhorse)
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        best_h, best_c5 = mw_optimize(best_h, target, n, t0, budget_s,
                                       max_iter=200000, lr_init=0.06, eta_init=5.0)

    # Phase 2: SLSQP polish with increasing sharpness
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        for beta in [500, 2000, 8000, 30000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            best_h = do_slsqp(best_h, n, target, beta, maxiter=5000)
            c_try = compute_c5(best_h)
            if c_try < best_c5:
                best_c5 = c_try

    # Phase 3: Restart with perturbation + re-optimize
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 42)
    for att in range(6):
        if remaining < 30:
            break
        scale = 0.015 * (0.6 ** att)
        hp = project_h(best_h + rng.standard_normal(n) * scale, target)
        hp, c_try = mw_optimize(hp, target, n, t0, budget_s,
                                 max_iter=60000, lr_init=0.04, eta_init=8.0)
        if c_try < best_c5:
            best_h, best_c5 = hp, c_try
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            hp = do_slsqp(hp, n, target, 10000, maxiter=3000)
            c_try = compute_c5(hp)
            if c_try < best_c5:
                best_h, best_c5 = hp, c_try

    # Phase 4: Try lower resolution for speed, then upsample
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        n_low = 192
        target_low = n_low / 2.0
        h_low = project_h(resample_fn(best_h, n_low), target_low)
        h_low, c_low = mw_optimize(h_low, target_low, n_low, t0, budget_s,
                                    max_iter=80000, lr_init=0.08, eta_init=6.0)
        if remaining > 20:
            h_low = do_slsqp(h_low, n_low, target_low, 5000, maxiter=3000)
        h_up = project_h(resample_fn(h_low, n), target)
        h_up, c_up = mw_optimize(h_up, target, n, t0, budget_s,
                                  max_iter=50000, lr_init=0.04, eta_init=10.0)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_h, best_c5 = h_up, c_up
        remaining = budget_s - (time.time() - t0)
        if c_up < best_c5 + 0.001 and remaining > 10:
            best_h = do_slsqp(best_h, n, target, 20000, maxiter=5000)
            best_c5 = compute_c5(best_h)

    # Phase 5: Final SLSQP with very high beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        for beta in [50000, 100000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            best_h = do_slsqp(best_h, n, target, beta, maxiter=10000)
            c_try = compute_c5(best_h)
            if c_try < best_c5:
                best_c5 = c_try

    # Phase 6: Coordinate descent final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = coord_polish(best_h, target, n, t0, budget_s, max_sweeps=300)

    # Final MW pass if time remains
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = mw_optimize(best_h, target, n, t0, budget_s,
                                       max_iter=100000, lr_init=0.03, eta_init=15.0)
        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h = do_slsqp(best_h, n, target, 50000, maxiter=5000)
            best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
```
