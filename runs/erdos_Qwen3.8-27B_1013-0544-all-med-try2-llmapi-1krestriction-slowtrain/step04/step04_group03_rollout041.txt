from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def full_softmax_gradient(h, beta):
    """Compute gradient of smooth max of correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # softmax weights
    corr_max = np.max(corr)
    log_weights = (corr - corr_max) / beta
    log_weights = np.clip(log_weights, -500, 500)
    weights = np.exp(log_weights)
    weights = weights / weights.sum()
    
    # q_m = weights[n-1+m] + weights[n-1-m] for m >= 1, q_0 = weights[n-1]
    q = np.zeros(n)
    q[0] = weights[n-1]
    for m in range(1, n):
        q[m] = weights[n-1+m] + weights[n-1-m]
    
    # g(j) = sum_{m=0}^{min(j,n-1)} q_m * (1-h(j-m)) - sum_{m=0}^{min(n-1-j,n-1)} q_m * h(j+m)
    g = np.zeros(n)
    for m in range(n):
        # First sum: j >= m
        if m < n:
            g[m:] += q[m] * (1.0 - h[:n-m])
        # Second sum: j <= n-1-m
        if m < n:
            g[:n-m] -= q[m] * h[m:]
    
    return g * (2.0 / n)  # multiply by dx

def slsqp_obj(h_in, beta, n):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    dx = 2.0 / n
    m = np.max(corr)
    c = np.clip((corr - m) / beta, -500, 500)
    return (m + np.log(np.sum(np.exp(c)) / len(corr)) / beta) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter=10000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except:
        return h0

def softmax_gradient_descent(h, n, target, n_iter, beta, step_init, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = step_init
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        g = full_softmax_gradient(cur, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn  # normalize gradient direction
        # We want to DECREASE the objective, so go in -g direction
        trial = project_h(cur - step * g, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            stall = 0
            step = min(step * 1.03, 0.05)
        else:
            stall += 1
            step *= 0.7
            if stall > 200:
                break
    return cur, best_c5

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    dx = 2.0 / n

    cmax = np.max(corr)

    weights = np.exp(logits)

    g += weights[n-1] * (1.0 - 2.0 * h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def full_softmax_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    logits = np.clip((corr - cmax) / beta, -500, 500)
    weights = np.exp(logits)
    weights /= weights.sum()
    g = np.zeros(n)
    g += weights[n-1] * (1.0 - 2.0 * h)
    for m in range(1, n):
        wp = weights[n-1+m]
        g[m:] += wp * (1.0 - h[:-m])
        g[:n-m] -= wp * h[m:]
        wn = weights[n-1-m]
        g[:n-m] += wn * (1.0 - h[m:])
        g[m:] -= wn * h[:-m]
    return g * dx

def smooth_max_obj(h_in, beta, n):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    dx = 2.0 / n
    m = np.max(corr)
    c = np.clip((corr - m) / beta, -500, 500)
    return (m + np.log(np.sum(np.exp(c)) / len(corr)) / beta) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except:
        return h0

def softmax_gd(h, n, target, n_iter, beta, step_init, t0, budget, momentum=0.85):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = step_init
    mom = np.zeros(n)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        g = full_softmax_gradient(cur, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        mom = momentum * mom - step * g
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            stall = 0
            step = min(step * 1.03, 0.06)
        else:
            stall += 1
            step *= 0.65
            mom *= 0.4
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:
            break
    return cur, best_c5

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.008, -0.008, 0.004, -0.004, 0.002, -0.002, 0.001, -0.001]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def binary_init_kronecker(n, seed):
    rng = np.random.default_rng(seed)
    n_ones = n // 2
    h = np.zeros(n)
    # Kronecker sequence: place ones at floor(i * n / n_ones) mod n
    idx = np.array([int(np.floor(i * n / n_ones)) % n for i in range(n_ones)])
    idx = np.unique(idx)
    while len(idx) < n_ones:
        new_idx = rng.integers(0, n)
        if new_idx not in idx:
            idx = np.append(idx, new_idx)
    h[idx[:n_ones]] = 1.0
    return h

def evolve_crossover(h1, h2, target, rng):
    n = len(h1)

        child = 0.5 * (h1 + h2)
    else:
        mask = rng.random(n) < 0.5
        child = np.where(mask, h1, h2)
    return project_h(child, target)
