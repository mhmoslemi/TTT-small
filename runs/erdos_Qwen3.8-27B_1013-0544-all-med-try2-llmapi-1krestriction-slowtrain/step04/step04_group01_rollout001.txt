import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum_box(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha < 1e6:
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        # For very large alpha, use inf-norm approximation
        return float(m + np.sum(np.exp(alpha * (c - m)) - 1.0) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < obj_smooth(h, n, alpha):
            return project_sum_box(res.x, n / 2.0)
    except Exception:
        pass
    return h.copy()

def autocorr(h, n):
    """Compute autocorrelation r(k) = dx * sum h[i]*h[i+k]"""
    dx = 2.0 / n
    c = np.correlate(h, h, mode='full') * dx
    return c

    dx = 2.0 / n
    r = np.correlate(h, h, mode='full')
    k_star = np.argmin(r)
    shift = k_star - (n - 1)
    # grad r(k*) w.r.t. h[i] = dx * (h[i+shift] + h[i-shift]) with bounds=0
    g = np.zeros(n)
    if shift > 0:

        g[shift:] += h[:n-shift]
    elif shift < 0:
        s = -shift
        g[s:] += h[:n-s]
        g[:n-s] += h[s:]
    else:
        g += h
    g *= dx
    h_new = h + lr * g
    return project_sum_box(h_new, n / 2.0), k_star

def pair_search(h, n, deadline, deltas):

    for d in deltas:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def project_sum_box(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha > 1e6:
        return float(m + (np.sum(np.exp(alpha * (c - m))) - 1.0) / alpha)
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)


def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < obj_smooth(h, n, alpha):
            return project_sum_box(res.x, n / 2.0)
    except Exception:
        pass
    return h.copy()


def subgrad_gradient(h, n):
    dx = 2.0 / n
    r = np.correlate(h, h, mode='full')
    k_min = np.argmin(r)
    shift = k_min - (n - 1)
    g = np.zeros(n)
    if shift > 0:
        g[:n - shift] += h[shift:]
        g[shift:] += h[:n - shift]
    elif shift < 0:
        s = -shift
        g[s:] += h[:n - s]
        g[:n - s] += h[s:]
    else:
        g = h.copy()
    return g * dx, k_min


def subgrad_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    while time.time() < deadline:
        g, _ = subgrad_gradient(h, n)
        v = momentum * v + g
        h_new = h + lr * v
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = project_sum_box(h - lr * v, n / 2.0)
            stall += 1
        if stall > 500:
            lr *= 0.9
            momentum = min(momentum + 0.01, 0.95)
            stall = 0
        if lr < 1e-6:
            break
    return best_h, best_c


def fast_c5_update(h, n, i, j, d):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(300):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for sgn in (1, -1):
                        hi = h[i] + sgn * d
                        hj = h[j] - sgn * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sgn * d, hj + sgn * d
                    if time.time() > deadline:
                        return h, best_c
            if not improved:
                break
    return h, best_c


def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c


def optimize_round(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for a in [5000, 50000, 500000]:
        if time.time() > deadline - 5:
            break
        ht = slsqp(best_h, n, a, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    if time.time() < deadline - 10:
        best_h, best_c = subgrad_optimize(best_h, n, time.time() + min(30, (deadline - time.time()) * 0.3))

    if time.time() < deadline - 5:
        ht = slsqp(best_h, n, 2000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

        h_interp = project_sum_box(h_interp, n_try / 2.0)
        h_opt, c_opt = optimize_round(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts at best n
    n_best = len(best_h)
    remaining = deadline - time.time()
    n_restarts = max(2, int(remaining / 50))
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        scale = 0.01 + 0.08 * (r / max(1, n_restarts - 1))
        hp = best_h + rng.normal(0, scale, n_best)
        hp = project_sum_box(hp, n_best / 2.0)
        alloc = min(40, (deadline - time.time() - 40) * 0.4)
        if alloc < 10:

        h_opt, c_opt = optimize_round(hp, n_best, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Pair search (coarse)
    n_f = len(best_h)
    if time.time() < deadline - 40:

            best_h, best_c = pair_search(best_h, n_f, time.time() + min(80, (deadline - time.time()) * 0.5),

        else:

    if time.time() < deadline - 20:

        c = compute_c5(ht, n_f)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    # Phase 6: Fine pair search
