import numpy as np
from scipy.optimize import minimize
import time

_beta = [100.0]
_target = [0.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    m = np.max(c)
    x = np.clip(beta * (c - m), -50, 50)
    ev = np.exp(x)
    se = np.sum(ev)
    w = ev / se
    val = m + np.log(se) / beta
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for k in active:
        wk = w[k]
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - h[:-lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += wk * (1.0 - h[m2:])
                grad[m2:] -= wk * h[:n - m2]
    return val, grad

def _sobj(x):
    v, _ = smooth_max_obj_grad(x, _beta[0])
    return v

def _sgrad(x):
    _, g = smooth_max_obj_grad(x, _beta[0])
    return g

def _scon(x):
    return x.sum() - _target[0]

def _scon_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h0, betas, maxiter, time_left, min_time):
    n = len(h0)
    _target[0] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _scon_jac}]
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for beta in betas:
        if time_left() < min_time:
            break
        _beta[0] = beta
        try:
            res = minimize(_sobj, best_h, method='SLSQP', bounds=bounds,
                          jac=_sgrad, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_h(res.x, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    return best_h, best_c5

def grad_descent(h, n_iter, time_left, beta_lo, beta_hi, step_init,
                 perturb_every, perturb_scale, min_time):
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_left() < min_time:
            break
        frac = min(1.0, it / max(1, int(n_iter * 0.75)))
        beta = beta_lo + (beta_hi - beta_lo) * frac
        _beta[0] = beta
        val, grad = smooth_max_obj_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_dir = -grad / gnorm
        step = step_init / (1.0 + it * 0.0005)
        found = False
        for _ in range(12):
            h_try = project_h(h + step * grad_dir, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                h = h_try
                best_c5 = c5_try
                no_imp = 0
                found = True
                break
            step *= 0.5
        if not found:
            no_imp += 1
            if no_imp % perturb_every == 0:
                sc = perturb_scale * min(4.0, 1.0 + 0.4 * (no_imp // perturb_every))

                    h = h_p
                    best_c5 = c5_p
                    no_imp = 0
            if no_imp > max(300, n_iter // 3):
                break
    return h, best_c5

def pairwise_refine(h, n_iter, time_left, n_pairs, min_time):
    """Try small transfers between pairs of positions."""
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_left() < min_time:
            break
        improved = False
        for _ in range(n_pairs):
            i = np.random.randint(n)

                continue
            for delta in [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            no_imp += 1
            if no_imp > 200:
                break
        else:
            no_imp = 0
    return h, best_c5
