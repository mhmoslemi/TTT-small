import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_corr(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500))) / len(corr)) / beta

def obj_fn(args, beta, target):
    h = args
    return softmax_corr(h, beta)

def optimize_lbfgsb(h0, n, target, beta, maxiter, tol, t0, budget):
    from scipy.optimize import minimize
    res = minimize(obj_fn, h0, args=(beta, target), method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol})
    return project_sum(res.x, target)

    h = (h0 > 0.5).astype(float)
    target = n / 2
    diff = target - h.sum()

        zeros = np.where(h == 0)[0]
        h[zeros[np.argsort(h0[zeros])[::-1]][:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        h[ones[np.argsort(h0[ones])][:int(-diff)]] = 0.0
    
    def bin_c5(h):
        corr = np.correlate(h.astype(np.float64), 1.0 - h, mode="full")
        return np.max(corr)
    
    best_h = h.copy()
    best_val = bin_c5(h)
    T = 5.0
    cooling = 0.9997
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        if it % 1000 == 0:
            T *= cooling
        # Pick a 1 and a 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_val = bin_c5(h)
        if new_val < best_val or np.random.random() < np.exp(-(new_val - best_val) / max(T, 0.01)):
            if new_val < best_val:
                best_h = h.copy()
                best_val = new_val
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_val * 2.0 / n

    n = len(h)
    corr = np.zeros(n)  # only need k=0,...,n-1 due to symmetry
    for k in range(n):
        corr[k] = np.dot(h[k:], 1.0 - h[:n-k])
    
    # Softmax probabilities
    m = np.max(corr)
    p = np.exp(np.clip(beta*(corr - m), -500, 500))
    p /= np.sum(p)
    
    # Gradient
    grad = np.zeros(n)
    for i in range(n):

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_correlate(h):
    """Compute c_k for k=0,...,n-1 using FFT."""
    n = len(h)
    N = 2 * n
    H = np.fft.fft(h, N)
    OH = np.fft.fft(1.0 - h, N)
    return np.fft.ifft(H * np.conj(OH)).real[:n]

    h = np.clip(h.copy(), 0.0, 1.0)

        if cnt == 0:
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    corr = fast_correlate(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta*(corr-m), -500, 500))) / len(corr)) / beta

def soft_max_grad(h, beta):
    n = len(h)
    corr = fast_correlate(h)
    m = np.max(corr)
    ev = np.exp(np.clip(beta*(corr-m), -500, 500))
    p = ev / ev.sum()
    A = np.convolve(p, 1.0 - h, 'full')[:n]
    B = np.correlate(h, p, 'full')[n-1:2*n-1]

    return soft_max_val(x, beta)

    return soft_max_grad(x, beta)

    return x.sum() - target

        tm[zeros[:int(diff)]] = 1.0

        ones = np.where(tm == 1)[0]
        tm[ones[:int(-diff)]] = 0.0
