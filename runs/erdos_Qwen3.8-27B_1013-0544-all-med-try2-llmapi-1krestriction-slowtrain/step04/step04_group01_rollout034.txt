def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sca_solve_lp(h0, n):
    """Solve the LP from SCA linearization around h0."""
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    for m in range(2 * n - 1):
        mu = m - (n - 1)
        if mu >= 0:
            i_start, i_end = 0, n - mu
        else:
            i_start, i_end = -mu, n
        indices = np.arange(i_start, i_end)
        j_indices = indices + mu
        
        coeff = np.zeros(n)
        const = 0.0
        for k in range(len(indices)):
            i = indices[k]
            j = j_indices[k]
            const += h0[i] * (1.0 - h0[j])
            coeff[j] += -h0[i]
            coeff[i] += -h0[j]
        
        constraints.append(cp.dot(coeff, h) + const >= t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if h.value is not None and np.all(np.isfinite(h.value)):

    """Return the full correlation array (not just the max)."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def swap_delta(h, a, b, n):
    """Compute Δc for all lags when swapping h[a]:1->0, h[b]:0->1."""
    # For each lag mu (from -(n-1) to n-1):
    # a contributes:
    #   As 'first' position: if 0<=a<n and 0<=a+mu<n: delta = -(1-h[a+mu]) = h[a+mu]-1
    #   As 'second' position: if 0<=a-mu<n (i.e., a-mu >= 0 and a-mu < n) and 0<=a<n: delta = h[a-mu]
    # b contributes:
    #   As 'first' position: if 0<=b<n and 0<=b+mu<n: delta = +(1-h[b+mu])
    #   As 'second' position: if 0<=b-mu<n and 0<=b<n: delta = -h[b-mu]
    
    mus = np.arange(-(n-1), n)  # all lags
    delta_c = np.zeros(2*n - 1)
    
    # a as first: valid when 0 <= a+mu < n, i.e., -a <= mu < n-a
    mask_a1 = (mus >= -a) & (mus < n - a)
    if np.any(mask_a1):
        j = a + mus[mask_a1]
        delta_c[mask_a1] += h[j] - 1.0  # = -(1-h[j])
    
    # a as second: valid when 0 <= a-mu < n, i.e., a-n < mu <= a
    mask_a2 = (mus > a - n) & (mus <= a) & (mus >= -(n-1)) & (mus <= n-1)
    if np.any(mask_a2):
        i = a - mus[mask_a2]
        delta_c[mask_a2] += h[i]

    mask_b1 = (mus >= -b) & (mus < n - b)
    if np.any(mask_b1):

def project_feasible(h, n):

def sca_solve_lp(h0, n):
    """Solve LP from SCA linearization."""
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    for m in range(2 * n - 1):
        mu = m - (n - 1)
        if mu >= 0:
            i_start, i_end = 0, n - mu
        else:
            i_start, i_end = -mu, n
        indices = np.arange(i_start, i_end)
        j_indices = indices + mu
        
        coeff = np.zeros(n)
        const = 0.0
        for k in range(len(indices)):
            i = int(indices[k])
            j = int(j_indices[k])
            const += h0[i] * (1.0 - h0[j])
            coeff[j] += -h0[i]
            coeff[i] += -h0[j]
        
        constraints.append(cp.dot(coeff, h) + const >= t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if h.value is not None and np.all(np.isfinite(h.value)):
            return np.clip(h.value, 0, 1)
    except Exception:
        pass
    return None
