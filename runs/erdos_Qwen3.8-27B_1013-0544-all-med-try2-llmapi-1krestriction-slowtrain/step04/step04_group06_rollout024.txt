            h_try = project_sum(best_h ** 3 + 0.3, target)
        for beta in [2000, 20000]:
            if tl() < 40:
                break
            h_new = lbfgs_optimize(h_try, n, target, beta, maxiter=1200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new

    # Phase 5: Final intensive subgradient polishing
    best_h, best_c5 = subgrad_polish(best_h, target, 6, 10000, t0, budget_s)

    # Phase 6: Ultra-fine final polish
    best_h, best_c5 = ultra_fine_polish(best_h, target, 30000, t0, budget_s)

    return best_h, best_c5, len(best_h)
