Let me also add a key trick: **rescaling the gradient to account for the constraint**. The feasible direction space is the intersection of the box [0,1] and the hyperplane sum(h) = target. The projected gradient should be the component of -grad that's in this feasible direction space.

For the equality constraint (sum = target), the projection of a vector g onto the feasible direction space (orthogonal to the constraint normal) is:
g_proj = g - (sum(g)/n) * ones(n)

Then we clip to the box direction: for coordinates where h[i] is at 0, we can only go up (g_proj[i] should be >= 0, else set to 0). For h[i] at 1, only go down.

Actually, for the projected gradient method on the constrained set, the standard approach is:
1. Compute gradient g at current point
2. Take step: h_new = h - alpha * g
3. Project h_new onto the feasible set [0,1]^n ∩ {sum = target}

The projection in step 3 is what project_h does. This is the standard approach and it works.

For the Barzilai-Borwein step size:
alpha_BB = <s, y> / ||y||^2 where s = h_k - h_{k-1}, y = g_k - g_{k-1}

But for non-smooth problems, BB step sizes can be unstable. A safer alternative is the "diminishing step with restart":
- Start with a moderate step
- If no improvement, reduce step by factor 0.5
- If improvement, increase step by factor 1.1 (capped)

Or even simpler: **backtracking line search**.
- Start with alpha = alpha_max
- While f(h - alpha*g) > f(h) - c*alpha*||g||^2 (Armijo condition), halve alpha

For the max function, the Armijo condition becomes:
C5(project(h - alpha*g)) < C5(h) - c*alpha*||g||^2 / ||g|| (something like that)

Actually, for non-smooth functions, a simpler condition is just:
C5(project(h - alpha*g)) < C5(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        step = diff / free.sum()
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h

def corr_grad(h, k):
    """Gradient of correlation value at index k w.r.t. h."""
    n = len(h)
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[lag:] += 1.0 - h[:-lag]
        g[:n-lag] -= h[lag:]
    elif -n < lag < 0:
        m = -lag
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def top_k_subgrad(h, K):
    """Average gradient of top K correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:K]
    g = np.zeros(n)
    for k in top_idx:
        g += corr_grad(h, int(k))
    g /= K
    return g, np.max(corr) * (2.0/n)

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    s = np.sum(exp_x)
    w = exp_x / s
    val = (m + np.log(s) / beta) * dx
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        g += w[k] * corr_grad(h, int(k))
    g *= dx
    return val, g

def slsqp_obj(v, beta):
    val, _ = smooth_max_and_grad(v, beta)
    return val

def slsqp_grad(v, beta):
    _, g = smooth_max_and_grad(v, beta)
    return g

def do_subgrad_nesterov(h, target, n_iter, tleft_fn, K=5, alpha0=0.01):
    """Subgradient with Nesterov momentum."""
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    v_prev = np.zeros(n)  # momentum
    f_prev = compute_c5(x)
    best_x = x.copy()
    best_f = f_prev
    alpha = alpha0
    mu = 0.9  # momentum decay
    
    for it in range(n_iter):
        if tleft_fn() < 5:
            break
        
        g, f_cur = top_k_subgrad(x, K)
        
        # Nesterov: evaluate gradient at x + mu*(x - x_prev)
        # Simplified: use momentum on the gradient
        v = mu * v_prev - alpha * g
        v_prev = v.copy()
        
        # Project the step
        new_x = project_h(x + v, target)
        f_new = compute_c5(new_x)
        
        if f_new < f_cur - 1e-14:
            x = new_x
            f_cur = f_new
            if f_cur < best_f:
                best_f = f_cur
                best_x = x.copy()
            # Increase alpha slightly
            alpha = min(alpha * 1.05, 0.1)
        else:
            # Halve alpha
            alpha = max(alpha * 0.5, 0.0001)
            if alpha <= 0.0001 and it > 100:
                # Kick
                kick = np.random.randn(n) * 0.01
                new_x = project_h(x + kick, target)
                if compute_c5(new_x) < f_cur:
                    x = new_x
                    f_cur = compute_c5(x)
                    alpha = alpha0 * 0.5
    
    return best_x, best_f

def do_slsqp_phase(h, target, betas, tleft_fn, maxiter=3000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]

            if compute_c5(h_new) < compute_c5(h_cur):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(h, k):
    n = len(h)
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[lag:] += 1.0 - h[:-lag]
        g[:n - lag] -= h[lag:]
    elif -n < lag < 0:
        m = -lag
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def top_k_subgrad(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:K]
    g = np.zeros(n)
    for k in top_idx:
        g += corr_grad(h, int(k))
    g /= K
    return g * dx, np.max(corr) * dx

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    s = np.sum(exp_x)
    w = exp_x / s
    val = (m + np.log(s) / beta) * dx
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        g += w[k] * corr_grad(h, int(k))
    g *= dx
    return val, g

_bm = [1.0]

def _sobj(v):
    v = np.clip(v, 0.0, 1.0)
    val, _ = smooth_max_and_grad(v, _bm[0])
    return val

def _sgrad(v):
    v = np.clip(v, 0.0, 1.0)
    _, g = smooth_max_and_grad(v, _bm[0])
    return g

def do_slsqp(h, target, betas, tleft, maxiter=3000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if tleft() < 15:
            break
        _bm[0] = beta
        try:
            res = minimize(_sobj, h_cur, method='SLSQP', jac=_sgrad,
                          bounds=bounds, constraints=constr,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def do_subgrad(h, target, n_iter, tleft, K=5, alpha0=0.005):
    n = len(h)
    x = h.copy()
    f_cur = compute_c5(x)
    best_x = x.copy()
    best_f = f_cur
    alpha = alpha0
    v = np.zeros(n)
    mu = 0.7
    no_imp = 0
    for it in range(n_iter):
        if tleft() < 3:
            break
        g, _ = top_k_subgrad(x, K)
        v = mu * v - alpha * g
        x_new = project_h(x + v, target)
        f_new = compute_c5(x_new)
        if f_new < f_cur - 1e-14:
            x = x_new
            f_cur = f_new
            no_imp = 0
            if f_cur < best_f:
                best_f = f_cur
                best_x = x.copy()
        else:
            no_imp += 1
            alpha = max(alpha * 0.7, 1e-5)
            v = np.zeros(n)
            if no_imp > 500:
                kick_scale = 0.005 * min(5.0, no_imp / 500.0)
                kick = np.random.randn(n) * kick_scale
                kick -= kick.mean()
                x_try = project_h(x + kick, target)
                f_try = compute_c5(x_try)
                if f_try < f_cur:
                    x = x_try
                    f_cur = f_try
                    no_imp = 0
                    alpha = alpha0 * 0.5
                elif no_imp > 2000:
                    break
    return best_x, best_f
