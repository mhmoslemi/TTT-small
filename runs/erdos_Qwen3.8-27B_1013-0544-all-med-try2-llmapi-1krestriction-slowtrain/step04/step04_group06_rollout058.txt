```python
def smooth_max_and_grad(h, beta, top_k=50):
    """
    Compute the smooth maximum of the overlap values and its gradient.
    Overlap at lag k: f_k = sum_i h[i]*(1-h[i+k]) * dx
    Returns (smooth_max_value, gradient_of_smooth_max)
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    
    # Select top_k lags
    if top_k is not None and top_k < nc:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(nc)
    
    c = corr[top_idx].astype(float)
    m = c.max()
    bc = np.clip(beta * (c - m), -500, 0)
    e = np.exp(bc)
    se = e.sum()
    
    # Smooth max value (in units of dx)
    val = (m + np.log(se) / beta) * dx
    
    # Weights
    w = e / se
    
    # Gradient
    g = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-10:
            continue
        lag = top_idx[i] - (n - 1)
        if lag == 0:
            g += w[i] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[i] * (1.0 - h[:-lag])
            g[:n - lag] -= w[i] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[i] * (1.0 - h[m2:])
            g[m2:] -= w[i] * h[:n - m2]
    
    return val, g * dx
```
