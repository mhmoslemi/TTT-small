import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    return e / e.sum()

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_val(corr, beta) * dx
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wi = w[k_idx]
        if wi < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += wi * (1.0 - h[lag:lag+m])
            grad[lag:] -= wi * h[:m]
        else:
            la = -lag
            m = n - la
            grad[:m] -= wi * h[la:la+m]
            grad[la:] += wi * (1.0 - h[:m])
    grad *= dx
    return val, grad

    return h.sum() - ts

    return np.ones_like(h)

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]

                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    shifted = np.clip(beta * (x - m), -700.0, 700.0)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = e.sum()
    return e / s

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wi = w[k_idx]
        if wi < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += wi * (1.0 - h[lag:lag+m])
            grad[lag:] -= wi * h[:m]
        else:
            la = -lag
            m = n - la
            grad[:m] += wi * (1.0 - h[la:la+m])
            grad[la:] -= wi * h[:m]
    return grad * dx

    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def obj_jac(h, dx, beta):
    return compute_grad(h, dx, beta)

def con_jac(h, ts):

def slsqp_refine(h, n, deadline, seed, betas):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.zeros_like(x)
    return e / s

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wi = w[k_idx]
        if wi < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += wi * (1.0 - h[lag:lag + m])
            grad[lag:] -= wi * h[:m]
        else:
            la = -lag
            m = n - la
            grad[:m] += wi * (1.0 - h[la:la + m])
            grad[la:] -= wi * h[:m]
    return grad * dx

def obj_func(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def obj_jac(h, dx, beta):
    return compute_grad(h, dx, beta)

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def momentum_gd(h, n, deadline, seed, alpha_init=0.01, beta_start=10.0, beta_end=500.0, n_iter=30000):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    alpha = alpha_init
    vel = np.zeros(n)
    mom = 0.85
    stagnant = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        beta_sm = beta_start * (beta_end / beta_start) ** frac
        grad = compute_grad(h, dx, beta_sm)
        vel = mom * vel - alpha * grad
        h_new = project_feasible(h + vel, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-14:
            h = h_new
            c5 = c_new
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            stagnant = 0
            alpha = min(alpha * 1.01, 0.05)
        else:
            stagnant += 1
            alpha *= 0.92
            vel *= 0.3
            if stagnant > 100:
                vel = np.zeros(n)
            if stagnant > 400 and alpha < 5e-6:
                break
    return best_h, best_c5

def slsqp_refine(h, n, deadline, seed, betas):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.004, 0.012]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_func, x0, args=(dx, beta), jac=obj_jac,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def top_lag_equalize(h, n, deadline, seed, top_k=3, n_iter=5000):
    """Focus on equalizing the top K lags while minimizing their common value."""
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    alpha = 0.005
    stagnant = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        target = top_vals.mean()
        spread = top_vals.max() - target
        if spread < 1e-10:
            break
        grad = np.zeros(n)
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            gi = np.zeros(n)
            if lag >= 0:
                m = n - lag
                gi[:m] += (1.0 - h[lag:lag + m])
                gi[lag:] -= h[:m]
            else:
                la = -lag
                m = n - la
                gi[:m] += (1.0 - h[la:la + m])
                gi[la:] -= h[:m]
            # Push down lags above target, push up lags below
            weight = 1.0 if corr[k_idx] > target else 0.3
            grad += weight * gi
        h_new = project_feasible(h - alpha * dx * grad, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-14:
            h = h_new
            c5 = c_new
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            stagnant = 0
            alpha = min(alpha * 1.01, 0.03)
        else:
            stagnant += 1
            alpha *= 0.93
            if stagnant > 300:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(200, 0.5)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: Momentum GD with smooth beta ramp (25%)
    d1 = t0 + budget_s * 0.25
    h1 = parent.copy()
    h1, c1 = momentum_gd(h1, pn, d1, seed, alpha_init=0.008, beta_start=5.0, beta_end=100.0, n_iter=20000)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: SLSQP with analytic gradient (20%)
    d2 = t0 + budget_s * 0.45
    h2 = best_h.copy()
    h2, c2 = slsqp_refine(h2, best_n, d2, seed + 1, [15, 40, 80, 200])
    if c2 < best_c5:
        best_h, best_c5 = h2.copy(), c2

    # Phase 3: Top-lag equalization (15%)
    d3 = t0 + budget_s * 0.60
    h3 = best_h.copy()
    h3, c3 = top_lag_equalize(h3, best_n, d3, seed + 50, top_k=3, n_iter=8000)
    if c3 < best_c5:
        best_h, best_c5 = h3.copy(), c3
    h3b = best_h.copy()
    h3b, c3b = top_lag_equalize(h3b, best_n, d3, seed + 60, top_k=4, n_iter=5000)
    if c3b < best_c5:
        best_h, best_c5 = h3b.copy(), c3b

    # Phase 4: Second momentum GD round with higher beta (15%)
    d4 = t0 + budget_s * 0.75
    h4 = best_h.copy()
    h4, c4 = momentum_gd(h4, best_n, d4, seed + 200, alpha_init=0.005, beta_start=50.0, beta_end=2000.0, n_iter=15000)
    if c4 < best_c5:

    # Phase 5: SLSQP polish with high beta (18%)

    h5 = best_h.copy()
    h5, c5 = slsqp_refine(h5, best_n, d5, seed + 500, [200, 800, 3000, 10000])
    if c5 < best_c5:
        best_h, best_c5 = h5.copy(), c5

    # Phase 6: Final micro-gradient (7%)
    d6 = t0 + budget_s * 0.99
    h6 = best_h.copy()
    h6, c6 = momentum_gd(h6, best_n, d6, seed + 999, alpha_init=0.002, beta_start=200.0, beta_end=1000.0, n_iter=8000)
    if c6 < best_c5:
        best_h, best_c5 = h6.copy(), c6

    # Final: one more SLSQP squeeze
    if time.time() < t0 + budget_s * 0.995:
        try:
            dx = 2.0 / best_n
            ts = best_n / 2.0
            bounds = Bounds(np.zeros(best_n), np.ones(best_n))
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
            res = minimize(obj_func, best_h, args=(dx, 5000), jac=obj_jac,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-16})
            h_final = project_feasible(res.x, ts)
            c5_final = compute_c5(h_final)
            if c5_final < best_c5:
                best_h, best_c5 = h_final, c5_final
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
