import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_grad(h, beta):
    """Compute gradient of log-sum-exp approximation of max correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return g * dx

def nesterov_descent(h, target, beta, max_iter, t0, budget, momentum=0.9):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    lr = 0.05
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 5:
            break
        h_y = h + momentum * v  # Nesterov lookahead
        g = soft_grad(h_y, beta)
        v_new = momentum * v - lr * g
        h_try = project_sum(h + v_new, target)
        c5_try = compute_c5(h_try)
        if c5_try < compute_c5(h):
            v = v_new
            h = h_try
            if c5_try < best_c5:
                best_h = h.copy()
                best_c5 = c5_try
                no_improve = 0
            else:
                no_improve += 1
        else:
            lr *= 0.7
            if lr < 1e-8:
                lr = 0.01
                no_improve += 1
        if no_improve > 500:
            break
    return best_h, best_c5, h

def slsqp_optimize(h_init, beta, maxiter=2000):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0, 1)] * n
    
    def obj(x):
        n = len(x)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        return (m + np.log(se / len(corr)) / beta) * dx
    
    def grad(x):
        return soft_grad(x, beta)
    
    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def subgrad_polish(h, target, n_active, max_iter, t0, budget):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0003]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        wj = e[j] / se
        if wj < 1e-10:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wj * (1.0 - h[:-lag])
            g[:n - lag] -= wj * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += wj * (1.0 - h[m2:])
            g[m2:] -= wj * h[:n - m2]
    return g * dx

def lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m2 = -lag
        g[:n - m2] += (1.0 - h[m2:])
        g[m2:] -= h[:n - m2]
    return g

            c5t = compute_c5(ht)

                best = ht

            lr = max(lr, 1e-7)
            if no_improve > 250:
                break
        else:
            no_improve = 0
            lr = min(lr * 1.08, 0.4)
    return best, best_c5

def slsqp_optimize(h_init, beta, maxiter=2000):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0, 1)] * n
    def obj(x):
        n2 = len(x)
        dx2 = 2.0 / n2
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        return (m + np.log(se / len(corr)) / beta) * dx2
    def grad(x):
        return soft_grad(x, beta)
    def eq(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)
