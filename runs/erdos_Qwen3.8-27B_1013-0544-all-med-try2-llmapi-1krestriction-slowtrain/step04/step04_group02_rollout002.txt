    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: SLSQP on parent resolution (30%)
    d1 = t0 + budget_s * 0.30
    h1, c1 = slsqp_round(parent.copy(), pn, d1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: Wide subgradient on best (30%)
    d2 = t0 + budget_s * 0.60
    h2, c2 = subgradient_wide(best_h.copy(), best_n, d2, seed + 1000)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2.copy(), c2, best_n

    # Phase 3: SLSQP polish with high beta (10%)
    d3 = t0 + budget_s * 0.70
    h3, c3 = slsqp_round(best_h.copy(), best_n, d3, seed + 2000)
    if c3 < best_c5:
        best_h, best_c5, best_n = h3.copy(), c3, best_n

    # Phase 4: Try other resolutions (20%)
    d4 = t0 + budget_s * 0.90
    try_res = [200, 250, 300, 400, 450, 500]
    for r in try_res:
        if time.time() > d4:
            break
        if r == best_n:
            continue
        h_r = interp_to_res(parent, r)
        c_r = compute_c5(h_r)
        h_r, c_r = slsqp_round(h_r, r, d4, seed + r)
        h_r, c_r = subgradient_wide(h_r, r, d4, seed + r + 5000)
        if c_r < best_c5:
            best_h, best_c5, best_n = h_r.copy(), c_r, r

    # Phase 5: Final subgradient + SLSQP (10%)
    d5 = t0 + budget_s * 0.98
    h5, c5f = subgradient_wide(best_h.copy(), best_n, d5, seed + 9999)
    if c5f < best_c5:
        best_h, best_c5 = h5, c5f
    h5, c5f = slsqp_round(best_h.copy(), best_n, d5, seed + 8888)
    if c5f < best_c5:
        best_h, best_c5 = h5, c5f

    if best_h is None:
        best_h = project_feasible(np.full(100, 0.5), 50.0)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
