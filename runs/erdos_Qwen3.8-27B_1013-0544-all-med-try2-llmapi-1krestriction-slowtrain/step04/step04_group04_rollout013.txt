def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    """Compute smooth max value and gradient of max_k corr_k * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def run_lbfgs(h0, betas, time_fn, maxiter_each=2000, penalty=80.0):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    ones_n = np.ones(n)
    for beta in betas:
        if time_fn() < 10:
            break
        def obj(x):
            v, _ = smooth_max_val_grad(x, beta)
            sd = x.sum() - target
            return v + penalty * sd * sd
        def grad_fn(x):
            _, g = smooth_max_val_grad(x, beta)
            sd = x.sum() - target
            return g + 2.0 * penalty * sd * ones_n
        try:
            res = minimize(obj, h_cur, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter_each,
                                                   'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def run_slsqp(h0, betas, time_fn, maxiter_each=3000):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    ones_n = np.ones(n)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return ones_n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    for beta in betas:
        if time_fn() < 10:
            break
        def obj(x):
            return smooth_max_val_grad(x, beta)[0]
        def grad_fn(x):
            return smooth_max_val_grad(x, beta)[1]
        try:
            res = minimize(obj, h_cur, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def subgradient_polish(h, time_fn, n_iter, top_k_start, top_k_end,
                       steps_list, perturb_every=300, perturb_scale=0.008):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_fn() < 3:
            break
        frac = it / max(1, n_iter)
        top_k = int(round(top_k_start + (top_k_end - top_k_start) * frac))
        top_k = max(1, min(top_k, n // 5))
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, len(corr))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n - m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        found = False
        for step in steps_list:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-15:
                h = ht
                c5_cur = c5t
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % perturb_every == perturb_every - 1:
                sc = perturb_scale * min(5.0, no_imp / float(perturb_every) + 1.0)
                ht = project_h(h + np.random.randn(n) * sc, target)
                c5t = compute_c5(ht)
                if c5t < c5_cur:
                    h = ht
                    c5_cur = c5t
                    no_imp = 0
            if no_imp > n_iter // 5:
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at moderate resolutions (faster, constrained)
    for n_res in [100, 150, 200]:
        if time_left() < 120:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        h_new, c5_new = run_slsqp(h_res, [100, 500, 2000, 10000], time_left, maxiter_each=3000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: L-BFGS-B at higher resolutions (faster per iteration)
    for n_res in [200, 300, 400, 500]:
        if time_left() < 100:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        h_new, c5_new = run_lbfgs(h_res, [200, 1000, 5000, 20000], time_left,
                                   maxiter_each=3000, penalty=100.0)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Long subgradient polish with adaptive top_k
    n_cur = len(best_h)
    tk_start = max(8, n_cur // 12)
    tk_end = 2
    steps_coarse = [0.08, 0.05, 0.03, 0.02, 0.01, 0.005]
    best_h, best_c5 = subgradient_polish(best_h, time_left, 40000,
                                          tk_start, tk_end, steps_coarse)

    # Phase 4: SLSQP refinement if we have time
    if time_left() > 60:
        n_cur = len(best_h)

    # Phase 5: Fine subgradient polish

    steps_fine = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]

                                          perturb_every=400, perturb_scale=0.005)

    # Phase 6: Ultra-fine final polish

                                              2, 1, steps_ultra,
                                              perturb_every=500, perturb_scale=0.003)
