```python
def binary_simulated_annealing(h_continuous, n, deadline, initial_temp=0.01, cooling=0.999):
    """Simulated annealing on binary solutions"""
    # Initialize from continuous solution
    target = int(n // 2)
    h_bin = (h_continuous >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    diff = target - h_bin.sum()
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        # Flip the ones with highest h_continuous values among zeros
        order = np.argsort(h_continuous[zeros])[::-1]
        h_bin[zeros[order[:int(diff)]]] = 1
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_continuous[ones])
        h_bin[ones[order[:int(-diff)]]] = 0
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    curr_c = best_c
    temperature = initial_temp
    
    while time.time() < deadline and temperature > 1e-10:
        # Generate a random move
        # Option 1: flip a random bit
        # Option 2: swap a random 1 with a random 0
        if np.random.random() < 0.3:
            # Flip
            i = np.random.randint(n)
            h_bin[i] = 1 - h_bin[i]
        else:
            # Swap
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        
        new_c = compute_c5(h_bin, n)
        delta = new_c - curr_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            # Revert
            if np.random.random() < 0.3:
                h_bin[i] = 1 - h_bin[i]
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        
        temperature *= cooling
    
    return best_h, best_c
```

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth_max(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def binary_sa(h_cont, n, deadline, seed=0):
    """Simulated annealing on binary solutions"""
    rng = np.random.RandomState(seed)
    target = int(n // 2)
    h = (h_cont >= 0.5).astype(float)
    diff = target - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        order = np.argsort(h_cont[zeros])[::-1]
        h[zeros[order[:diff]]] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        order = np.argsort(h_cont[ones])
        h[ones[order[:diff]]] = 0
    
    best_h, best_c = h.copy(), compute_c5(h, n)
    curr_c = best_c
    temp = 0.005
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    while time.time() < deadline and temp > 1e-6:

        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            curr_c = new_c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if new_c < best_c:
                best_c = new_c; best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        temp *= 0.9995
    return best_h, best_c
