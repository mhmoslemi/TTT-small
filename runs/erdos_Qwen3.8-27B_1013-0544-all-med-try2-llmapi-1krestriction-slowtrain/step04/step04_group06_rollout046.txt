    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        wj = e[j] / se
        if wj < 1e-11:
            continue
        lag = j - (n - 1)
        g += wj * corr_grad_at_lag(x, lag)
    return val, g * dx

def _obj_func(x):
    return soft_obj_val(x, _beta_global)

def _grad_func(x):
    return soft_obj_grad(x, _beta_global)[1]

def lbfgs_pass(h_init, beta, maxiter):
    global _beta_global
    n = len(h_init)
    _beta_global = float(beta)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_func, h_init, jac=_grad_func, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
    return project_sum(res.x, n / 2.0)

def subgrad_polish(h, target, n_active, max_iter, t0, budget, mom_decay):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        elapsed = time.time() - t0
        if budget - elapsed < 5:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += corr_grad_at_lag(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = mom_decay
        v = mom * v + (1.0 - mom) * g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            v = g
            vn = 1.0
        else:
            v /= vn
        improved = False
        base = 0.05 / (1.0 + 0.002 * it)
        steps = [base, base * 0.5, base * 0.25, base * 0.1, base * 0.04]
        for step in steps:
            ht = project_sum(best - step * v, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 150 and it % 75 == 74:
                scale = 0.005 + 0.01 * min(1.0, no_improve / 500.0)
                ht = project_sum(best + np.random.randn(n) * scale, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
                    v = np.zeros(n)
            if no_improve > 3000:
                break
        else:
            no_improve = 0
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n0 = len(best_h)

    for beta in [50, 300, 1500, 8000, 30000]:

            best_c5 = c5_new

    best_h, best_c5 = subgrad_polish(best_h, target_final, 6, 5000, t0, budget_s, 0.2)
