import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def sensitivity(h, n, k):
    """Compute d(c[k])/d(h[i]) for all i, where c = correlate(h,1-h,'full')."""
    shift = k - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i + shift < n:
            g += (1.0 - h[i + shift])
        if 0 <= i - shift < n:
            g -= h[i - shift]
        grad[i] = g
    return grad

def lobb_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, 'full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
    return project_feasible(res.x, n)

        # g[k] = sum_j h[j]*(1-h[j-shift_k]) * dx where shift_k = k-(n-1)
        # d g[k]/d h[i] = dx * [(1-h[i+shift]) if valid - h[i-shift] if valid]

            shifted_val = shift_k = k - (n - 1)
            for i in range(n):
                sg = 0.0
                if 0 <= i + shifted_val < n:
                    sg += (1.0 - h[i + shifted_val])
                if 0 <= i - shifted_val < n:
                    sg -= h[i - shifted_val]
                grad[i] += w[k] * sg
        grad *= dx
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
