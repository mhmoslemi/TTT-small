def smooth_max_grad(corr, h, alpha, n):
    """Compute gradient of smooth max of corr with respect to h."""
    dx = 2.0 / n
    # Soft-max weights
    shifted = alpha * corr
    shifted -= shifted.max()  # for numerical stability
    w = np.exp(shifted)
    w /= w.sum()
    
    # corr has length 2n-1
    # For output index k, shift s = k - (n-1)
    # d(output[k])/dh[j] = dx * [I(0≤j-s<n)*(1-h[j-s]) - I(0≤j+s<n)*h[j+s]]
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        wdx = w[k] * dx
        # First term: I(0 <= j-s < n) * (1-h[j-s])
        # j-s is valid when 0 <= j-s < n, i.e., s <= j < s+n
        # For j in [max(0,s), min(n, s+n)):
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_lo < j_hi:

        # Second term: -I(0 <= j+s < n) * h[j+s]

        j_lo2 = max(0, -s)

        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wdx * h[j_lo2 + s : j_hi2 + s]

Actually, the most efficient approach for a 1D piecewise-linear convex minimization is:

- When moving right, the next breakpoint is the smallest t > current where a new line becomes active
- Continue until 0 is in the subgradient

This is the "slope trick" or "invariant box" approach. It's O(K log K) per coordinate if we sort the lines by their values at t=0.

    """Coordinate descent for the Erdős problem."""
    imports time

    dx = 2.0 / n
    K = 2 * n - 1
    target_sum = n / 2.0
    
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline and time.time() > deadline:
                return h
            
            # Compute a_k and b_k for coordinate j
            s_arr = np.arange(K) - (n - 1)  # shifts, from -(n-1) to n-1
            # a[k] = dx * (1-h[j-s]) [if 0<=j-s<n] - dx * h[j+s] [if 0<=j+s<n]
            a = np.zeros(K)
            for k in range(K):
                s = s_arr[k]
                if 0 <= j - s < n:
                    a[k] += dx * (1 - h[j - s])
                if 0 <= j + s < n:
                    a[k] -= dx * h[j + s]
            
            # Compute current corr
            corr_cur = np.correlate(h, 1-h, mode='full') * dx
            b = corr_cur - a * h[j]
            
            # Binary search on C to find optimal t
            # Lower bound: min possible value of max_k(a_k*t+b_k) >= min(max_k b_k, max_k(a_k+b_k))
            # Upper bound: current value
            C_lo = min(float(np.max(b)), float(np.max(a + b)))
            C_hi = float(np.max(a * h[j] + b))  # = current corr max
            # Actually C_hi should be the current max of corr
            C_hi = float(np.max(corr_cur))
            
            # Make sure C_lo <= C_hi
            if C_lo > C_hi:
                C_lo = C_hi
            
            # Binary search
            for _ in range(80):
                C_mid = (C_lo + C_hi) / 2
                lower = 0.0
                upper = 1.0
                feasible = True
                for k in range(K):
                    if abs(a[k]) < 1e-15:
                        if b[k] > C_mid:
                            feasible = False
                            break
                    elif a[k] > 0:
                        t_ub = (C_mid - b[k]) / a[k]
                        if t_ub < upper:
                            upper = t_ub
                            if upper < lower:

                    else:

            lower = 0.0
            upper = 1.0

                improved = True

        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def paired_coord_optimize(h, n, deadline, rng, max_pairs_per_pass=2000, n_passes=1000):
    """Paired coordinate descent maintaining sum constraint."""
    dx = 2.0 / n
    K = 2 * n - 1
    target_sum = n / 2.0
    best_c = compute_c5(h, n)
    
    # Precompute shifts
    s_arr = np.arange(K) - (n - 1)
    
    for pass_num in range(n_passes):
        if time.time() > deadline:
            break
        
        # Select pairs for this pass
        if pass_num % 5 == 0:
            # Random pairs
            pairs = [(rng.randint(n), rng.randint(n)) for _ in range(max_pairs_per_pass)]
        else:
            # Sequential adjacent pairs
            pairs = [(i, (i+1) % n) for i in range(0, n, 1)]
            # Also some random
            pairs += [(rng.randint(n), rng.randint(n)) for _ in range(max_pairs_per_pass // 2)]
        
        for j, l in pairs:
            if j == l:
                continue
            if time.time() > deadline:
                break
            
            S = h[j] + h[l]  # maintain this sum
            t_lo = max(0.0, S - 1.0)
            t_hi = min(1.0, S)
            if t_lo >= t_hi:
                continue  # already at boundary
            
            # Compute coefficients
            # a[k] = dc_k/dh[j], c[k] = dc_k/dh[l]
            # corr[k] = a[k]*h[j] + c[k]*h[l] + d[k]
            # With constraint h[l] = S - h[j]:
            # corr[k] = (a[k] - c[k])*h[j] + c[k]*S + d[k]
            # = e[k]*t + f[k] where t = h[j]
            
            # Compute a[k] and c[k]
            # a[k] = dx * [(1-h[j-s_k]) if 0<=j-s_k<n] - dx * [h[j+s_k] if 0<=j+s_k<n]
            # c[k] = dx * [(1-h[l-s_k]) if 0<=l-s_k<n] - dx * [h[l+s_k] if 0<=l+s_k<n]
            
            # Vectorized computation:
            # For shift s, a contribution to a[k] is:
            #   dx*(1-h[j-s]) if 0 <= j-s < n
            #   -dx*h[j+s] if 0 <= j+s < n
            
            a_k = np.zeros(K)
            c_k = np.zeros(K)
            
            # For a_k:
            # Term 1: dx*(1-h[j-s]) for valid s where 0<=j-s<n
            # s ranges from -(n-1) to n-1
            # j-s is valid when 0 <= j-s < n, i.e., max(-(n-1), j-(n-1)) <= s <= min(n-1, j)
            # i.e., max(1-j, -(n-1)) <= s <= min(n-1, j)
            # In terms of index k = s + (n-1): max(1-j+(n-1), 0) <= k <= min(n-1+j, 2n-2)
            
            # Let me just do it with boolean masks
            # j-s is in [0, n-1] when s is in [j-(n-1), j]
            mask_j_minus = (s_arr >= j - (n-1)) & (s_arr <= j)
            # j+s is in [0, n-1] when s is in [-(n-1)+j... wait
            # j+s in [0, n-1] when s in [-j, n-1-j]
            mask_j_plus = (s_arr >= -j) & (s_arr <= n-1-j)
            
            a_k[mask_j_minus] = dx * (1 - h[j - s_arr[mask_j_minus]])
            a_k[mask_j_plus] -= dx * h[j + s_arr[mask_j_plus]]
            
            # Similarly for c_k (with l):
            mask_l_minus = (s_arr >= l - (n-1)) & (s_arr <= l)
            mask_l_plus = (s_arr >= -l) & (s_arr <= n-1-l)
            
            c_k[mask_l_minus] = dx * (1 - h[l - s_arr[mask_l_minus]])
            c_k[mask_l_plus] -= dx * h[l + s_arr[mask_l_plus]]
            
            # Effective 1D problem: minimize max_k (e[k]*t + f[k])
            # where e[k] = a[k] - c[k], f[k] = c[k]*S + d[k]
            # But I don't have d[k] directly. I can compute:
            # corr[k] = a[k]*h[j] + c[k]*h[l] + d[k]
            # So d[k] = corr[k] - a[k]*h[j] - c[k]*h[l]
            # And f[k] = c[k]*S + d[k] = c[k]*S + corr[k] - a[k]*h[j] - c[k]*h[l]
            # = corr[k] + c[k]*(S - h[l]) - a[k]*h[j]
            # = corr[k] + c[k]*h[j] - a[k]*h[j]
            # = corr[k] + (c[k] - a[k])*h[j]
            # = corr[k] - e[k]*h[j]
            
            e_k = a_k - c_k
            # f[k] = corr[k] - e[k]*h[j]
            # But I need corr[k]. Let me compute it:
            # Actually, I can avoid computing the full corr:
            # f[k] = corr[k] - e[k]*h[j]
            # And I want to minimize max_k(e[k]*t + f[k]) = max_k(e[k]*t + corr[k] - e[k]*h[j])
            # = max_k(e[k]*(t - h[j]) + corr[k])
            
            # Let u = t - h[j] (change in h[j]). Then h[l] changes by -u.
            # corr[k] becomes corr[k] + e[k]*u
            # We want to minimize max_k(corr[k] + e[k]*u)
            # Subject to: t ∈ [t_lo, t_hi], i.e., u ∈ [t_lo - h[j], t_hi - h[j]]
            
            u_lo = t_lo - h[j]
            u_hi = t_hi - h[j]
            
            # Now: minimize max_k(corr_k + e_k * u) over u ∈ [u_lo, u_hi]
            # This is a 1D convex piecewise-linear optimization
            # Binary search on C:
            # Find min C such that exists u ∈ [u_lo, u_hi] with corr_k + e_k*u <= C for all k
            # i.e., e_k*u <= C - corr_k for all k
            # If e_k > 0: u <= (C - corr_k)/e_k
            # If e_k < 0: u >= (C - corr_k)/e_k
            # If e_k = 0: need corr_k <= C
            
            corr_k = np.correlate(h, 1-h, mode='full') * dx  # O(n log n) or O(n²)
            
            # Binary search on C
            C_lo = min(float(np.max(corr_k)), float(np.max(corr_k + e_k * u_hi)), float(np.max(corr_k + e_k * u_lo)))
            # Actually C_lo should be the minimum possible value, which is at least
            # min over u of max_k(corr_k + e_k*u)
            # Lower bound: min(max(corr_k + e_k * u_lo), max(corr_k + e_k * u_hi))
            # Upper bound: min(max(corr_k + e_k * u_lo), max(corr_k + e_k * u_hi)) — same thing
            # Actually the upper bound is the current max (at u=0): max(corr_k)
            C_hi = float(np.max(corr_k))  # value at u=0 (current point)
            # But we might be able to do better than u=0
            # Let's use: C_lo = min of the two endpoints
            val_at_lo = float(np.max(corr_k + e_k * u_lo))
            val_at_hi = float(np.max(corr_k + e_k * u_hi))
            C_lo = min(val_at_lo, val_at_hi)
            
            if C_lo >= C_hi - 1e-15:
                # Can't improve
                continue
            
            # Binary search
            for _ in range(50):
                C_mid = (C_lo + C_hi) / 2
                lower = u_lo
                upper = u_hi
                # Check feasibility: exists u with e_k*u <= C_mid - corr_k for all k
                mask_e_pos = e_k > 1e-15
                mask_e_neg = e_k < -1e-15
                mask_e_zero = ~(mask_e_pos | mask_e_neg)
                
                if np.any(corr_k[mask_e_zero] > C_mid + 1e-12):
                    C_lo = C_mid
                    continue
                
                if np.any(mask_e_pos):
                    upper = min(upper, float(np.min((C_mid - corr_k[mask_e_pos]) / e_k[mask_e_pos])))
                if np.any(mask_e_neg):
                    lower = max(lower, float(np.max((C_mid - corr_k[mask_e_neg]) / e_k[mask_e_neg])))
                
                if lower <= upper + 1e-12:
                    C_hi = C_mid
                else:
                    C_lo = C_mid
            
            # Find optimal u at C = C_hi
            lower = u_lo
            upper = u_hi
            mask_e_pos = e_k > 1e-15
            mask_e_neg = e_k < -1e-15
            if np.any(mask_e_pos):
                upper = min(upper, float(np.min((C_hi - corr_k[mask_e_pos]) / e_k[mask_e_pos])))
            if np.any(mask_e_neg):
                lower = max(lower, float(np.max((C_hi - corr_k[mask_e_neg]) / e_k[mask_e_neg])))
            
            u_opt = np.clip((lower + upper) / 2, u_lo, u_hi)
            
            if abs(u_opt) > 1e-10:
                h[j] += u_opt
                h[l] -= u_opt
                h = np.clip(h, 0, 1)
        
        # Check improvement
        c_cur = compute_c5(h, n)
        if c_cur < best_c:
            best_c = c_cur
        # Re-project to maintain feasibility
        h = project_feasible(h, n)
    
    return h, best_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * c
    shifted -= shifted.max()
    return float(shifted.max() / alpha + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    K = 2 * n - 1
    shifted = alpha * c
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(K):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        wdx = w[k] * dx
        # First term: (1-h[j-s]) for j where 0<=j-s<n
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wdx * (1 - h[j_lo - s:j_hi - s])
        # Second term: -h[j+s] for j where 0<=j+s<n
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wdx * h[j_lo2 + s:j_hi2 + s]
    return grad
