        if diff > 0:

    """One full sweep of exact coordinate descent."""
    for j in order:
        t0 = h[j]
        ks = np.arange(2*n-1) - (n-1)
        jk = j + ks
        jmk = j - ks
        a = np.zeros(2*n-1)

            h, corr = coord_sweep(h, corr, n, np.random.permutation(n), ks)
            h = subgrad_step(h, corr, n, ks, lr)
            h = project_feasible(h, ts)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = float(np.max(corr) * 2.0 / n)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.006)
            else:
                no_improve += 1
                lr *= 0.95

    # Phase 4: Other resolutions (12%)
    d4 = deadline
    if time.time() < d4:
        for n_alt in [300, 350, 400, 450]:
            if time.time() > d4 or n_alt == n:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_alt, endpoint=False)
            h_alt = np.interp(x_new, x_old, best_h)
            h_alt = project_feasible(h_alt, n_alt / 2.0)
            corr_alt = np.correlate(h_alt, 1.0 - h_alt, mode='full')
            c5_alt = float(np.max(corr_alt) * 2.0 / n_alt)
            ks_alt = np.arange(2 * n_alt - 1) - (n_alt - 1)
            sub_dl = min(d4, time.time() + 0.04 * budget_s)
            for _ in range(200):
                if time.time() > sub_dl:
                    break
                h_alt, corr_alt = coord_sweep(h_alt, corr_alt, n_alt, np.random.permutation(n_alt), ks_alt)
                h_alt = project_feasible(h_alt, n_alt / 2.0)
                corr_alt = np.correlate(h_alt, 1.0 - h_alt, mode='full')
                c5_alt = float(np.max(corr_alt) * 2.0 / n_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                n = n_alt
                ts = n / 2.0
                ks = ks_alt

    # Final SLSQP micro-polish if time remains
    if time.time() < deadline - 5:
        h_f = project_feasible(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = slsqp_polish(h_f, len(best_h), deadline)
        if c5_f < best_c5:
            best_h = h_f
            best_c5 = c5_f

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
