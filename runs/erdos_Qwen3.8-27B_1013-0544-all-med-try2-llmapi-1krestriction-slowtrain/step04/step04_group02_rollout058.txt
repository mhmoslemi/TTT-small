def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    scaled = beta * (x - m)
    scaled = np.clip(scaled, -700, 700)
    return m + np.log(np.sum(np.exp(scaled))) / beta

def obj_and_grad(h, dx, beta):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    scaled = np.clip(scaled, -700, 700)

    best_h = h.copy()
    best_c5 = c5
    y = h.copy()
    t_k = 1.0

            break

        # FISTA update

            best_c5 = c5_new

            lr *= 0.8
    
    return best_h, best_c5

def slsqp_optimize(h, dx, target_sum, beta, max_iter, deadline):

    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))

        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), beta) * dx

        return x.sum() - target_sum

        return np.ones_like(x)

                      options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    scaled = np.clip(beta * (x - m), -700, 700)
    return m + np.log(np.sum(np.exp(scaled))) / beta

def grad_top_lags(h, dx, top_m, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ti = np.argsort(corr)[-top_m:]
    tv = corr[ti]
    wts = np.exp(temp * (tv - tv.max()))
    wts /= wts.sum()
    grad = np.zeros(n)
    for j in range(top_m):
        ww = wts[j] * dx
        lag = ti[j] - (n - 1)
        if lag >= 0:
            v = n - lag
            grad[:v] += ww * (1.0 - h[lag:lag+v])
            grad[lag:lag+v] -= ww * h[:v]
        else:
            la = -lag
            v = n - la
            grad[la:la+v] += ww * (1.0 - h[:v])
            grad[:v] -= ww * h[la:la+v]
    return grad

def fista_opt(h0, dx, ts, beta, max_iter, deadline):
    n = len(h0)
    h = h0.copy()
    y = h.copy()
    t_k = 1.0
    lr = 1.0 / (2.0 * beta)
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_top_lags(y, dx, min(n - 1, 200), beta)
        y_new = y - lr * g
        x_new = project_feasible(y_new, ts)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - h)
        h = x_new
        t_k = t_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best = h.copy()
            best_c5 = c5
        elif it > 100 and c5 > best_c5 + 0.01:
            lr *= 0.5
    return best, best_c5

def grad_smooth_full(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = np.clip(beta * (corr - m), -700, 700)
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for j in nz:
        ww = w[j] * dx
        lag = j - (n - 1)
        if lag >= 0:
            v = n - lag
            grad[:v] += ww * (1.0 - h[lag:lag+v])
            grad[lag:lag+v] -= ww * h[:v]
        else:
            la = -lag
            v = n - la
            grad[la:la+v] += ww * (1.0 - h[:v])
            grad[:v] -= ww * h[la:la+v]
    return grad

def active_polish(h0, dx, ts, deadline, top_m, seed):
    np.random.seed(seed)
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = 0.0008
    mom = 0.85
    stag = 0
    for it in range(50000):
        if time.time() > deadline:
            break
        grad = grad_top_lags(h, dx, top_m, 300.0)
        vel = mom * vel - lr * grad
        h_new = project_feasible(h + vel, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best = h.copy()
            best_c5 = c5_new
            stag = 0
            lr = min(lr * 1.004, 0.003)
        else:
            stag += 1
            vel *= 0.3
            lr *= 0.94
        if stag > 5000 and lr < 5e-9:
            break
    return best, best_c5

def slsqp_opt(h0, dx, ts, beta, max_iter, deadline):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def objf(x):
        return smooth_max_val(np.correlate(x, 1.0 - x, mode='full'), beta) * dx
    def jacf(x):
        return grad_smooth_full(x, dx, beta)
    def cf(x):
        return x.sum() - ts
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(objf, h0, jac=jacf, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, ts)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)
