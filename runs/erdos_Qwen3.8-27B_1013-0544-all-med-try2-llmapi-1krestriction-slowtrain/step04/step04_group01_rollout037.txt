def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)

    L = lag_idx - (n - 1)
    grad = np.zeros(n)

Let me implement a fast evaluation for 2-point changes:

When h[i] changes by δ and h[j] changes by -δ:
For each correlation c[k], the change is:
Δc[k] = (contribution of h[i] to c[k]) * δ + (contribution of h[j] to c[k]) * (-δ)

The coefficient of h[m] in c[k] is exactly the value we computed in the gradient! So contrib_m[k] = grad_k[m] (the m-th entry of the gradient of c[k] w.r.t. h).

So for a 2-point change (i, j, δ):
Δc[k] = δ * (grad_k[i] - grad_k[j])

And the new c[k] = old_c[k] + δ * (grad_k[i] - grad_k[j])

    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()

        if time.time() > deadline:
            break
        
        # Find the active (max) correlation and its gradient
        active_k = np.argmax(c_vec)
        grad = corr_gradient(h, n, active_k)

        # decrease h where grad is positive and increase where negative.
        # For a 2-point move, find i with positive grad and j with negative grad.
        pos_mask = grad > 0
        neg_mask = grad < 0
        
        if not pos_mask.any() or not neg_mask.any():
            # Try random
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = 0.005
        else:
            # Pick the position with largest positive grad
            pos_grads = np.where(pos_mask)[0]
            neg_grads = np.where(neg_mask)[0]
            i = pos_grads[np.argmax(grad[pos_grads])]
            j = neg_grads[np.argmin(grad[neg_grads])]
            delta = min(0.01, abs(grad[i]) * 0.1, h[i], 1 - h[j])
            if delta < 0.001:

        hj = h[j] - delta

            hj = h[j] + delta

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

def corr_gradient(h, n, lag_idx):

    grad = np.zeros(n)
    if L >= 0:
        grad[L:n] += (1.0 - h[0:n-L])
        grad[0:n-L] -= h[L:n]
    else:
        P = -L
        grad[0:n-P] += (1.0 - h[P:n])
        grad[P:n] -= h[0:n-P]
    return grad

def gradient_local_search(h, n, deadline, delta_range=None, max_trials=200000):

    if delta_range is None:
        delta_range = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

            break

            if time.time() > deadline:

            grad = np.zeros(n)
            for k in top3:
                grad += corr_gradient(h, n, k)
            grad /= len(top3)

            pos_candidates = np.where(grad > 0.01)[0]
            neg_candidates = np.where(grad < -0.01)[0]
            
            if len(pos_candidates) == 0 or len(neg_candidates) == 0:

                break

            i = pos_candidates[np.argmax(grad[pos_candidates])]
            j = neg_candidates[np.argmin(grad[neg_candidates])]
            
            # Determine step size
            d = min(delta_base, 0.5 * abs(grad[i]), 0.5 * abs(grad[j]))
            d = max(d, 0.0001)
            d = min(d, h[i], 1.0 - h[j])
            if d < 0.00005:
                continue
            
            h[i] += d
            h[j] -= d
            
            c_new_max = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
            if c_new_max < best_c - 1e-15:
                best_c = c_new_max
                c_vec = np.correlate(h, 1 - h, mode='full') * dx
                # Keep h as is (improved)
            else:
                h[i] -= d
                h[j] += d
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
