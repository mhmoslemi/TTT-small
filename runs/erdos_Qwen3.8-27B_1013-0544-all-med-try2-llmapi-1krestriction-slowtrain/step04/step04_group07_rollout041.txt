    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta*(x-m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, params):
    beta, dx = params
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sml_jac(h_in, params):
    beta, dx = params
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -500, 500))
    w /= w.sum()
    n = len(h_in)

def pair_descent(cur, target, n, deadline):

    rng = np.random.RandomState(77)

        if stall > 4:
