            if tl() > 30:
                best_h, best_c5 = coord_descent(best_h, target, 20, t0, budget_s)

    # Final polish
    if tl() > 5:
        best_h = project_sum(best_h, target)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n0
