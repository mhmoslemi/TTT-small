def smooth_max_obj_grad(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    exp_c = np.exp(beta * (c - m))
    s = np.sum(exp_c)
    w = exp_c / s
    obj = (m + np.log(s) / beta) * dx
    n = len(h)
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k] * dx / beta
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                g[:] += wk * (1.0 - 2.0 * h)
            else:
                g[:n-lag] += wk * (1.0 - h[lag:])
                g[lag:] -= wk * h[:n-lag]
        else:
            la = -lag
            g[la:] += wk * (1.0 - h[:n-la])
            g[:n-la] -= wk * h[la:]
    return obj, g

    n = len(h0)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

        res = minimize(smooth_max_obj, h0,
                       jac=lambda x: smooth_max_obj_grad(x, beta, dx)[1],
                       args=(beta, dx),

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

        return h0

            h_new = slsqp_opt(best_h, target, 1500, 3000, 1e-15)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c

    # Final: ensure feasibility
    target_final = len(best_h) / 2.0
    best_h = project_sum(best_h, target_final)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
