```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_at_k(h, n, k_star):
    """Exact gradient of g[k_star] w.r.t. h, in O(n)."""
    k_shift = k_star - (n - 1)
    A = np.zeros(n)
    B = np.zeros(n)
    if k_shift >= 0:
        A[k_shift:] = 1.0 - h[:n - k_shift]
        B[:n - k_shift] = h[k_shift:]
    else:
        koff = -k_shift
        A[:n - koff] = 1.0 - h[koff:]
        B[koff:] = h[:n - koff]
    return (A - B) * (2.0 / n)

def subgradient_optimize(h, n, deadline, max_iter=50000, alpha=None):
    """Projected subgradient with exact gradient. If alpha=None, use hard max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    n_full = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        
        if alpha is not None and alpha > 0:
            # Log-sum-exp gradient
            shifted = g - m
            weights = np.exp(alpha * shifted)
            wsum = weights.sum()
            # grad = sum_k weights[k]/wsum * grad_at_k(h, n, k)
            # This is expensive to compute exactly. Use a subset.
            # Top contributing k values
            top_k = np.argsort(weights)[-min(200, n_full):]
            grad = np.zeros(n)
            for k in top_k:
                w = weights[k] / wsum
                if w > 1e-10:
                    grad += w * grad_at_k(h, n, k)
        else:
            k_star = int(np.argmax(g))
            grad = grad_at_k(h, n, k_star)
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Backtracking line search
        step = lr
        found = False
        for _ in range(20):
            h_new = project_feasible(h - step * grad / ng, n)
            c_new = compute_c5(h_new, n)
            if c_new < m - 0.01 * step * ng * 0.1:  # sufficient decrease
                found = True
                break
            step *= 0.5
        
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            lr = min(lr * 1.1, 5.0)
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_at_k(h, n, k):
    k_shift = k - (n - 1)
    A = np.zeros(n)
    B = np.zeros(n)
    if k_shift >= 0:
        A[k_shift:] = 1.0 - h[:n - k_shift]
        B[:n - k_shift] = h[k_shift:]
    else:
        koff = -k_shift
        A[:n - koff] = 1.0 - h[koff:]
        B[koff:] = h[:n - koff]
    return (A - B) * (2.0 / n)

def lse_grad(h, n, g, alpha):
    m = g.max()
    p = np.exp(alpha * (g - m))
    psum = p.sum()
    if psum < 1e-300:
        return np.zeros(n)
    p = p / psum
    grad = np.zeros(n)
    for k in range(len(g)):
        if p[k] > 1e-10:
            grad += p[k] * grad_at_k(h, n, k)
    return grad

def optimize_lse(h, n, deadline, alpha_seq, max_iter_per=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        h_curr = best_h.copy()
        lr = 0.3
        for it in range(max_iter_per):
            if time.time() > deadline:
                break
            g = compute_all_g(h_curr, n)
            if alpha < 1e6:
                grad = lse_grad(h_curr, n, g, alpha)
            else:
                grad = grad_at_k(h_curr, n, int(np.argmax(g)))
            ng = np.linalg.norm(grad)
            if ng < 1e-14:
                break
            # Backtracking
            step = lr
            improved = False
            for _ in range(30):
                h_new = project_feasible(h_curr - step * grad / ng, n)
                c_new = compute_c5(h_new, n)
                c_curr = g.max()
                if c_new < c_curr - 1e-13:
                    h_curr = h_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                lr *= 0.9
                if lr < 1e-12:
                    break
            else:
                c_curr = compute_c5(h_curr, n)
                if c_curr < best_c:
                    best_c = c_curr
                    best_h = h_curr.copy()
                lr = min(lr * 1.05, 3.0)
    return best_h, best_c

def binary_local_search(h_bin, n, deadline, max_sweeps=50):
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i_idx, i in enumerate(zeros[:]):
            if time.time() > deadline:
                break
            for j_idx, j in enumerate(ones[:]):
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-12:
                    best_c = c
                    improved = True
                    ones[j_idx] = i
                    zeros[i_idx] = j
                    break
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: LSE subgradient (50% of budget)
    p1 = t0 + 0.50 * (budget_s - 5)
    alpha_seq = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    h, c = optimize_lse(best_h.copy(), n, p1, alpha_seq, max_iter_per=15000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 2: Binary local search (25% of budget)
    p2 = t0 + 0.75 * (budget_s - 5)
    # Binarize
    target = n // 2
    idx = np.argsort(best_h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    h_bin, cb = binary_local_search(h_bin.copy(), n, p2, max_sweeps=30)
    if cb < best_c:
        best_c = cb
        best_h = h_bin.copy()
    
    # Also try binarizing a perturbed version
    h_pert = project_feasible(best_h + 0.05 * rng.randn(n), n)
    idx = np.argsort(h_pert)[::-1]
    h_bin2 = np.zeros(n)
    h_bin2[idx[:target]] = 1.0
    h_bin2, cb2 = binary_local_search(h_bin2.copy(), n, p2, max_sweeps=20)
    if cb2 < best_c:
        best_c = cb2
        best_h = h_bin2.copy()
    
    # Phase 3: Try different n values (15% of budget)
    p3 = t0 + 0.90 * (budget_s - 5)
    for nh in [150, 200, 300]:
        if time.time() > p3 - 30:
            break
        if nh == n:
            continue
        # Interpolate
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        h_interp = np.interp(xn, xo, best_h)
        h_interp = project_feasible(h_interp, nh)
        h_new, c_new = optimize_lse(h_interp, nh, p3 - 20, [50, 200, 1000, 5000], max_iter_per=8000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            n = nh
    
    # Phase 4: Final polish (10% of budget)
    if time.time() < deadline - 10:
        h, c = optimize_lse(best_h.copy(), n, deadline, [10000, 50000, 100000], max_iter_per=10000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
