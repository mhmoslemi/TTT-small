    n = len(h0)

    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    dx = 2.0 / n
    C = np.correlate(h, 1-h, mode='full')  # 2n-1 values

    cooling = 0.9999

    zeros = list(np.where(h == 0)[0])

            break
        if T < T_min:
            break
        
        # Pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)

        delta_C = compute_delta_C(h, C, i, j, n)
        old_max = C.max()
        new_max = (C + delta_C).max()

        if delta < 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            C += delta_C

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_C(h, n):
    return np.correlate(h, 1-h, mode='full')

def delta_C(h, i, j, n):
    """Delta in C when h[i]:1->0, h[j]:0->1. Returns array of length 2n-1."""
    d = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    idx = n-1 + d  # index into full correlation
    
    # Term 1: -(1-h[i+d]) where i+d in [0,n-1]
    m = (d >= -i) & (d <= n-1-i)
    delta[idx[m]] -= (1.0 - h[i + d[m]])
    
    # Term 2: +(1-h'[j+d]) where j+d in [0,n-1]
    m = (d >= -j) & (d <= n-1-j)
    vals = 1.0 - h[j + d[m]]
    # Special case: if j+d == i, then h'[j+d] = 0, so 1-h'[j+d] = 1
    d_spec = i - j
    spec_idx = np.where(m & (d == d_spec))[0]
    if len(spec_idx) > 0:
        # Find position in the masked array
        masked_d = d[m]
        pos = np.where(masked_d == d_spec)[0]
        for p in pos:
            vals[p] = 1.0
    delta[idx[m]] += vals
    
    # Term 3: +h[i-d] where i-d in [0,n-1], d≠0, i-d≠j (i.e., d≠i-j)
    m = (d >= i-(n-1)) & (d <= i) & (d != 0) & (d != i-j)
    if m.any():

    # Term 4: -h[j-d] where j-d in [0,n-1], d≠0, j-d≠i (i.e., d≠j-i)
    m = (d >= j-(n-1)) & (d <= j) & (d != 0) & (d != j-i)

    cur_max = float(C.max())
    best_h = h.copy()
    best_C = C.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

            break
        it += 1

        j = zeros[jj]

    target = n // 2
    srt = np.argsort(h)[::-1]
    hb = np.zeros(n)

    return hb

    return np.correlate(h, 1.0 - h, mode='full')

def delta_C_vec(h, i, j, n):
    """O(n) vectorized computation of delta in full correlation."""
    d = np.arange(-(n-1), n)
    idx = (n-1) + d
    delta = np.zeros(2*n-1)
    
    # Term 1: p=i, valid when i+d in [0,n-1]
    m = (i + d >= 0) & (i + d <= n-1)
    if m.any():
        delta[idx[m]] -= (1.0 - h[i + d[m]])
    
    # Term 2: p=j, valid when j+d in [0,n-1]
    m = (j + d >= 0) & (j + d <= n-1)
    if m.any():
        vals = 1.0 - h[j + d[m]]
        # Correction: where j+d == i (i.e., d == i-j)
        dspec = i - j
        sm = d[m] == dspec
        if sm.any():
            vals[sm] = 1.0  # 1 - h'[i] = 1 - 0 = 1
        delta[idx[m]] += vals
    
    # Term 3: p=i-d (p+d=i), valid when i-d in [0,n-1], d≠0, i-d≠j
    m = (i - d >= 0) & (i - d <= n-1) & (d != 0) & (i - d != j)

        delta[idx[m]] += h[i - d[m]]
    
    # Term 4: p=j-d (p+d=j), valid when j-d in [0,n-1], d≠0, j-d≠i
    m = (j - d >= 0) & (j - d <= n-1) & (d != 0) & (j - d != i)

        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    dx = 2.0 / n
    h = make_binary(h0, n)
    C = full_C(h, n)
    best_c = float(C.max()) * dx
    best_h = h.copy()
    best_C = C.copy()

    # Phase 1: SA (45%)
    p1 = t0 + 0.45 * (budget_s - 5)
    h, C, cm = sa_search(h, n, p1, seed=seed)
    if cm * dx < best_c:
        best_c, best_h, best_C = cm * dx, h.copy(), C.copy()

    # Phase 2: Greedy local search (15%)
    p2 = t0 + 0.60 * (budget_s - 5)
    h, C, cm = greedy_ls(best_h, best_C, n, p2)
    if cm * dx < best_c:
        best_c, best_h, best_C = cm * dx, h.copy(), C.copy()

    # Phase 3: SA + LS from perturbed starts (remaining)
    if time.time() < dl - 30:
        rng = np.random.RandomState(seed + 99)
        n_starts = 6
        remain = dl - 30 - time.time()
        per_start = remain / n_starts
        for rs in range(n_starts):
            if time.time() > dl - 10:
                break
            h2 = best_h.copy()
            C2 = best_C.copy()
            nsw = rng.randint(8, 25)
            ones = list(np.where(h2 == 1)[0])
            zeros = list(np.where(h2 == 0)[0])
            for _ in range(nsw):
                ii = rng.randint(len(ones))
                jj = rng.randint(len(zeros))
                i, j = ones[ii], zeros[jj]
                dC = delta_C_vec(h2, i, j, n)
                h2[i] = 0.0
                h2[j] = 1.0
                C2 = C2 + dC
                ones[ii] = j
                zeros[jj] = i
            sub_dl_sa = time.time() + per_start * 0.55
            h3, C3, m3 = sa_search(h2, n, sub_dl_sa, seed=seed + 1000 + rs)
            if m3 * dx < best_c:
                best_c, best_h, best_C = m3 * dx, h3.copy(), C3.copy()
            if time.time() < dl - 10:
                sub_dl_ls = min(dl - 10, time.time() + per_start * 0.4)
                h4, C4, m4 = greedy_ls(h3, C3, n, sub_dl_ls)
                if m4 * dx < best_c:
                    best_c, best_h, best_C = m4 * dx, h4.copy(), C4.copy()

    return best_h, best_c, n
