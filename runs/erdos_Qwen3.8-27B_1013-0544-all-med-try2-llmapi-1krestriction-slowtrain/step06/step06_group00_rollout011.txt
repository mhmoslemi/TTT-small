def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(len(zeros_list)):
            if time.time() > deadline:
                break
            zi = zeros_list[i]
            for j in range(len(ones_list)):
                oj = ones_list[j]
                delta = np.zeros(2 * n - 1)
                im = zi + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = zi - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = oj + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = oj - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (oj - zi)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (zi - oj)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[zi], h[oj] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list[j] = ones_list[-1]
                    ones_list.pop()
                    zeros_list[i] = zeros_list[-1]
                    zeros_list.pop()
                    ones_list.append(zi)
                    zeros_list.append(oj)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def project_feasible(h, n):

def smooth_obj_and_grad(x, n, alpha):
    """Compute smooth max approximation and its gradient."""
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    S = exps.sum()
    weights = exps / S
    logsum = sm + np.log(S)
    obj = m + logsum / alpha
    
    # Gradient
    # g[k] = sum_i x[i] * (1-x[i-k]) for valid indices (k = k_idx - (n-1))
    # dg[k]/dx[j] = (1-x[j-k])*1{0<=j-k<n} - x[j+k]*1{0<=j+k<n}
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    js = np.arange(n)

        grad[vm] += w * (1.0 - x[jm[vm]])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    S = exps.sum()
    weights = exps / S
    obj = m + (sm + np.log(S)) / alpha
    # Gradient
    grad = np.zeros(n)
    js = np.arange(n)
    for k_idx in range(2 * n - 1):
        w = dx * weights[k_idx]
        if w < 1e-18:
            continue
        ks = k_idx - (n - 1)
        jm = js - ks
        jpl = js + ks
        vm = (jm >= 0) & (jm < n)
        vp = (jpl >= 0) & (jpl < n)
        grad[vm] += w * (1.0 - x[jm[vm]])
        grad[vp] -= w * x[jpl[vp]]
    return obj, grad

def slsqp_opt(h, n, alpha, maxiter, ftol, deadline):
    def obj(x):
        return smooth_obj_grad(x, n, alpha)[0]
    def grad(x):
        return smooth_obj_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_l = [int(x) for x in np.where(h == 1)[0]]
    zeros_l = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros_l)):
            if time.time() > deadline:
                break
            zi = zeros_l[ii]
            for jj in range(len(ones_l)):
                oj = ones_l[jj]
                im = zi + m_range; m1 = (im >= 0) & (im < n)
                imn = zi - m_range; m2 = (imn >= 0) & (imn < n)
                jm = oj + m_range; m3 = (jm >= 0) & (jm < n)
                jmn = oj - m_range; m4 = (jmn >= 0) & (jmn < n)
                delta = np.zeros(2 * n - 1)
                delta[m1] += 1.0 - h[im[m1]]
                delta[m2] -= h[imn[m2]]
                delta[m3] -= 1.0 - h[jm[m3]]
                delta[m4] += h[jmn[m4]]
                delta[n-1] -= 2.0
                k1 = n-1+(oj-zi)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n-1+(zi-oj)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[zi], h[oj] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_l[jj] = ones_l[-1]; ones_l.pop()
                    zeros_l[ii] = zeros_l[-1]; zeros_l.pop()
                    ones_l.append(zi); zeros_l.append(oj)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
