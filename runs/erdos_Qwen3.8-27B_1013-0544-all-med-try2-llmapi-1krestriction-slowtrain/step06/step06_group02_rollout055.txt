    target = n / 2.0

            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

def linearized_active_set(h, n, n_active=20, t0=None):
    c = compute_correlations(h, n)
    m = c.max()
    
    # Select active shifts
    if t0 is None:
        t0 = m + 0.01  # small slack
    active = np.where(c > m - 0.02)[0]
    if len(active) > n_active:
        idx = np.argsort(c[active])[::-1][:n_active]
        active = active[idx]
    active = list(active)
    
    # Compute gradients for active shifts
    dx = 2.0 / n
    grads = []
    for m_idx in active:
        k = m_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] += (1.0 - h[k:n]) * dx
            g[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += (1.0 - h[0:nv]) * dx
            g[0:nv] += (-h[kk:n]) * dx
        grads.append(g)
    
    # LP: min t
    # subject to: for each active k: grad_k · δ ≤ t - c_k
    #             sum(δ) = 0
    #             -h[i] ≤ δ[i] ≤ 1-h[i]
    
    # Decision variables: [δ_0, ..., δ_{n-1}, t]
    # c vector for linprog: [0, ..., 0, 1] (minimize t)
    c_lp = np.zeros(n + 1)
    c_lp[n] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active shift k: grad_k · δ - t ≤ -c_k
    A_ub = np.zeros((len(active) + 1, n + 1))
    b_ub = np.zeros(len(active) + 1)
    for i, m_idx in enumerate(active):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = -c[m_idx]
    # sum(δ) ≤ 0 and sum(δ) ≥ 0 handled separately
    
    # Actually, let's handle the sum constraint as equality
    # A_eq @ x = b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] ≤ δ[i] ≤ 1-h[i], t ≥ 0
    bounds = [(-h[i], 1-h[i]) for i in range(n)]
    bounds.append((0, None))
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        delta = res.x[:n]
        h_new = project_feasible(h + delta, n)
        return h_new
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlations(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_shift_gradient(h, n, shift_idx):
    """Gradient of correlation at a specific shift index"""
    k = shift_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n]) * dx
        g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv]) * dx
        g[0:nv] += (-h[kk:n]) * dx
    return g

def smooth_max_grad(h, n, alpha):
    c = compute_correlations(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_fn(x, n, alpha, mu):
    c = compute_correlations(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_pass(h, n, alpha, maxiter=500, deadline=None):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=(n, alpha, 1.0),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 5, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def targeted_pair_search(h, n, deadline):
    """Focus on pairs that directly affect the max-correlation shift"""
    best_c = compute_c5(h, n)
    c = compute_correlations(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        while time.time() < deadline:
            improved = False
            c = compute_correlations(h, n)
            m_idx = np.argmax(c)
            k = m_idx - (n - 1)
            
            # Try to reduce the max correlation by adjusting pairs
            # that directly affect it
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Find the partner for shift k
                if k >= 0:
                    j_partner = i + k if i + k < n else None
                else:
                    j_partner = i + (-k) if i >= (-k) else None
                
                # Also try nearby indices
                neighbors = set()
                if j_partner is not None:
                    neighbors.add(j_partner)
                for d in [-2, -1, 1, 2]:
                    for jj in [i + d]:
                        if 0 <= jj < n:
                            neighbors.add(jj)
                
                for j in neighbors:
                    if j == i:
                        continue
                    for di, dj in [(step, -step), (-step, step)]:
                        if h[i] + di < 0 or h[i] + di > 1:
                            continue
                        if h[j] + dj < 0 or h[j] + dj > 1:

                        h[i] += di
                        h[j] += dj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= di
                            h[j] -= dj
            if not improved:
                break
    return h, best_c

                if h[i] + di < 0 or h[i] + di > 1:

                if h[j] + dj < 0 or h[j] + dj > 1:

                    improved = True
                else:

When we change h[i] by δ, the correlations change as follows:
- For shift k (where m = k + n - 1):
  - If i is in the valid range for this shift:

    - The term h[i-k]*(1-h[i]) changes by -δ*h[i-k] (if i-k is valid)

This means I can compute the effect of changing h[i] on all correlations in O(n) time (since there are 2n-1 shifts, and for each shift I just need to check if i and i±k are in range).

Wait, actually for a shift k, the relevant indices are those where both i and i+k are in [0, n-1]. So for a given i:
- i contributes to correlation at shift k if i+k is in [0, n-1], i.e., -i ≤ k ≤ n-1-i
- i-k contributes to correlation at shift k if i is in [0, n-1] (always true) and i-k is in [0, n-1], i.e., k ≤ i ≤ k+n-1

So for shift k, the change due to δ in h[i] is:
Δc_k = δ * dx * [(1-h[i+k]) * I(0≤i+k<n) - h[i-k] * I(0≤i-k<n)]

I can precompute this vector for a given i in O(n) time. Then the objective as a function of δ is:
f(δ) = max_k [c_k + δ * g_k(i)]

where g_k(i) = dx * [(1-h[i+k]) * I(0≤i+k<n) - h[i-k] * I(0≤i-k<n)]

This is the max of linear functions of δ, which is convex. I can find its minimum by:
1. Computing all breakpoints (where two lines cross)
2. Or using a bisection on the derivative

The derivative f'(δ) = max of {g_k(i) : c_k + δ*g_k(i) is the max}

So f'(δ) = max{g_k(i) : k is active at δ}

The minimum is where f'(δ) changes from negative to positive. I can find this by:
1. At δ=0, find the active set A(0) = {k : c_k = max c}
2. f'(0+) = max{g_k(i) : k ∈ A(0)}
3. If f'(0+) > 0, minimum is at δ ≤ 0
4. If f'(0-) < 0, minimum is at δ > 0
5. Otherwise, the minimum is at δ = 0

More precisely, I can use the fact that f(δ) is piecewise linear and convex. The minimum is where the slope crosses zero.

For a practical implementation, I can just evaluate f at a few points and do ternary search since f is convex.

```python
def coord_gradient(h, n, i):
    """Compute g_k(i) for all k: the rate of change of c_k when h[i] increases"""
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        m = k + n - 1
        coeff = 0.0
        if 0 <= i + k < n:
            coeff += (1.0 - h[i + k])
        if 0 <= i - k < n:
            coeff += -h[i - k]
        g[m] = coeff * dx
    return g
```

But wait, I can do even better. The full matrix G[m, i] = g_k(i) has a specific structure. Let me think...

For shift k (where m = k + n - 1):
G[m, i] = dx * [(1-h[i+k]) * I(0≤i+k<n) - h[i-k] * I(0≤i-k<n)]

For the first term: (1-h[i+k]) is nonzero when 0≤i+k<n, i.e., i ∈ [-k, n-1-k]. For a fixed k, this is a contiguous range of i values.

For the second term: -h[i-k] is nonzero when 0≤i-k<n, i.e., i ∈ [k, k+n-1].

So G has a banded structure (actually, it's full but with a specific pattern).

In any case, for the coordinate descent, I don't need the full matrix. I just need, for a given i, the vector g(·, i) which tells me how each correlation changes when I change h[i].

```python
def coord_effect(h, n, i):
    """Vector of length 2n-1: effect on each correlation when h[i] increases by 1"""
    dx = 2.0 / n
    effect = np.zeros(2 * n - 1)
    k = np.arange(-(n-1), n)
    # i+k in [0, n): contributes (1-h[i+k])
    mask1 = (k >= -i) & (k < n - i)
    if np.any(mask1):
        idx = np.where(mask1)[0]
        effect[idx] += (1.0 - h[i + k[mask1]]) * dx
    # i-k in [0, n): contributes -h[i-k]
    mask2 = (k >= i - n + 1) & (k <= i)
    if np.any(mask2):
        idx = np.where(mask2)[0]
        effect[idx] += (-h[i - k[mask2]]) * dx
    return effect
```

Then for coordinate descent on coordinate i:
- Current correlations: c (length 2n-1)
- Effect vector: e = coord_effect(h, n, i)
- If we change h[i] by δ: new correlations = c + δ * e
- Objective: f(δ) = max(c + δ * e)
- Find δ that minimizes f(δ), subject to 0 ≤ h[i] + δ ≤ 1 and sum constraint

The sum constraint is tricky. If I change h[i] by δ, I need to compensate by changing some other h[j] by -δ. This makes it a 2D problem.

For simplicity, let me handle the sum constraint by projection after each coordinate update (like the previous code does).

For finding the optimal δ:
f(δ) = max_k (c[k] + δ * e[k])

This is a convex piecewise-linear function. The minimum is where the derivative changes sign.
f'(δ) = max{e[k] : c[k] + δ*e[k] = f(δ)}

I can find the minimum by:
1. Compute all "breakpoints" where two lines cross: δ_break = (c[k1] - c[k2]) / (e[k2] - e[k1]) for e[k1] ≠ e[k2]
2. Sort breakpoints
3. At each interval, the active set is fixed, so f is linear
4. Find the interval where the slope crosses zero

But this is O(n²) to compute all breakpoints, which is the same as evaluating f at n points. So it's not faster.

Alternative: since f is convex and piecewise linear, I can use bisection on the derivative:
- f'(δ) = max{e[k] : k is active at δ}
- Binary search for δ where f'(δ) = 0

But "active at δ" requires knowing which k achieves the max, which requires evaluating f(δ).

Simplest approach: just evaluate f at a few points and interpolate. Since f is piecewise linear, 2n-1 points would suffice to find the exact minimum, but in practice, the relevant breakpoints are few.

Let me just use a practical approach: evaluate f at 30 points in the range [lo, hi] and take the best. This is O(30 * n) = 9000 ops per coordinate, or 2.7M ops per sweep. Very fast.
