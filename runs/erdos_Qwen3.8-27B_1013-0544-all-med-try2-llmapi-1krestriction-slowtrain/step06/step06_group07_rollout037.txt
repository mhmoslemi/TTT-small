    h = np.array(h, dtype=float)

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_val_w(corr, beta):
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 500.0)
    es = np.exp(sh)
    se = es.sum()
    return m + np.log(se) / beta, es / se


def wgrad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if abs(w) < 1e-14:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:]) * w * dx
            grad[k:] -= h[:n - k] * w * dx
        else:
            m = -k
            grad[m:] += (1.0 - h[:n - m]) * w * dx
            grad[:n - m] -= h[m:] * w * dx
    return grad


def fista(h, n, beta, alpha, max_iter, target_sum, t0, budget_s, min_time=30):
    h_x = h.copy()
    h_y = h.copy()
    t_k = 1.0
    best_local = h_x.copy()
    best_c5 = compute_c5(h_x)
    dx = 2.0 / n
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < min_time:
            break
        corr_y = np.correlate(h_y, 1.0 - h_y, mode="full") * dx
        _, wts = smooth_max_val_w(corr_y, beta)
        g = wgrad(h_y, n, wts)
        h_new = project_box_sum(h_y - alpha * g, target_sum)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        h_y = h_new + ((t_k - 1.0) / t_new) * (h_new - h_x)
        h_x = h_new
        t_k = t_new
        if it % 300 == 299:
            c5 = compute_c5(h_x)
            if c5 < best_c5:
                best_c5 = c5
                best_local = h_x.copy()
    c5 = compute_c5(h_x)
    if c5 < best_c5:
        best_c5 = c5
        best_local = h_x.copy()
    return best_local, best_c5


def obj_fn(hx, dx, beta):
    corr = np.correlate(hx, 1.0 - hx, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 500.0)
    es = np.exp(sh)
    return (m + np.log(es.sum()) / beta) * dx


def obj_grad_fn(hx, dx, beta):
    corr = np.correlate(hx, 1.0 - hx, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 500.0)
    es = np.exp(sh)
    w = es / es.sum()
    return wgrad(hx, len(hx), w)


def con_eq(hx, ts):
    return hx.sum() - ts


def con_eq_jac(hx, ts):
    return np.ones(len(hx))


def slsqp_polish(h, n, dx, target_sum, t0, budget_s, min_time):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum,)}]
    for beta in [2000, 10000, 50000, 200000]:
        if budget_s - (time.time() - t0) < min_time:
            break
        for trial in range(2):
            if budget_s - (time.time() - t0) < min_time / 2:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.005
                x0 = project_box_sum(x0, target_sum)
            try:
                res = minimize(obj_fn, x0, args=(dx, beta),
                               jac=lambda x, d, b: obj_grad_fn(x, d, b),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_box_sum(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    return best_h, best_c5


def try_resolution(h_best, n_best, n_try, t0, budget_s, min_time):
    if n_try < 20 or budget_s - (time.time() - t0) < min_time:
        return h_best, n_best, compute_c5(h_best)
    dx_t = 2.0 / n_try
    ts_t = n_try / 2.0
    x_old = np.linspace(0, 2, n_best, endpoint=False)
    x_new = np.linspace(0, 2, n_try, endpoint=False)
    h_t = np.interp(x_new, x_old, h_best)
    h_t = project_box_sum(h_t, ts_t)
    alpha = n_try / 6.0
    for beta in [1000, 10000]:
        h_t, c5_t = fista(h_t, n_try, beta, alpha, 3000, ts_t, t0, budget_s, min_time / 2)
    h_t, c5_t = slsqp_polish(h_t, n_try, dx_t, ts_t, t0, budget_s, min_time / 3)
    c5_best = compute_c5(h_best)
    if c5_t < c5_best:
        return h_t, n_try, c5_t
    return h_best, n_best, c5_best


def coordinate_refine(h, n, target_sum, t0, budget_s, min_time):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    h_c = h.copy()
    order = np.random.permutation(n)
    for it in range(n * 3):
        if budget_s - (time.time() - t0) < min_time:
            break
        i = order[it % n]
        for delta in [-0.05, -0.02, 0.0, 0.02, 0.05, -0.01, 0.01, -0.1, 0.1]:
            h_try = h_c.copy()
            h_try[i] = h_c[i] + delta
            h_try = np.clip(h_try, 0.0, 1.0)
            s = h_try.sum()
            diff = target_sum - s
            if abs(diff) > 1e-10:
                others = np.delete(np.arange(n), i)
                ovals = h_try[others]
                if diff > 0:
                    mask = ovals < 1.0
                else:
                    mask = ovals > 0.0
                cnt = mask.sum()
                if cnt == 0:
                    continue
                h_try[others[mask]] += diff / cnt
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_box_sum(h_try, target_sum)
            c5 = compute_c5(h_try)
            if c5 < best_c5 - 1e-12:
                best = h_try.copy()
                best_c5 = c5
                h_c = h_try.copy()
                break
        h_c = best.copy()
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_box_sum(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: FISTA continuation
    for beta in [10, 50, 200, 1000, 5000, 20000, 80000]:
        if budget_s - (time.time() - t0) < 40:
            break
        alpha = n / 5.0
        h_new, c5_new = fista(best_h, n, beta, alpha, 4000, target_sum, t0, budget_s, 40)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: SLSQP polish
    if budget_s - (time.time() - t0) > 80:
        best_h, best_c5 = slsqp_polish(best_h, n, dx, target_sum, t0, budget_s, 60)

    # Phase 3: Coordinate refinement
    if budget_s - (time.time() - t0) > 60:
        best_h, best_c5 = coordinate_refine(best_h, n, target_sum, t0, budget_s, 50)

    # Phase 4: Second FISTA round from improved point
    if budget_s - (time.time() - t0) > 50:
        for beta in [20000, 80000, 300000]:
            if budget_s - (time.time() - t0) < 20:
                break
            alpha = n / 5.0
            h_new, c5_new = fista(best_h, n, beta, alpha, 3000, target_sum, t0, budget_s, 20)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Try different resolutions
    for n_try in [n + 10, n - 10, n + 20, 80, 60, 100]:
        if n_try == n or n_try < 20:
            continue
        if budget_s - (time.time() - t0) < 30:
            break
        h_try, n_new, c5_try = try_resolution(best_h, n, n_try, t0, budget_s, 40)
        if c5_try < best_c5:
            best_h = h_try
            n = n_new
            dx = 2.0 / n
            target_sum = n / 2.0
            best_c5 = c5_try

    # Phase 6: Final polish
    if budget_s - (time.time() - t0) > 15:
        best_h, best_c5 = slsqp_polish(best_h, n, 2.0 / n, n / 2.0, t0, budget_s, 10)

    if budget_s - (time.time() - t0) > 10:
        best_h, best_c5 = coordinate_refine(best_h, n, n / 2.0, t0, budget_s, 5)

    return best_h, best_c5, len(best_h)
