    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    scaled = alpha * (c - m)
    mx = scaled.max()
    if mx > 500:
        w = np.exp(scaled - mx)
    else:
        w = np.exp(scaled)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        wk = w[lag_idx]
        if wk < 1e-14:
            continue
        k = lag_idx - (n - 1)
        wm = wk * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        maxiter = min(400, max(20, int(remaining * 30)))
        if maxiter < 10:
            break
        mu = max(1.0, alpha * 0.005)
        try:
            res = minimize(smooth_max_obj, h,
                          jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxiter * 20,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def binary_local_search(h, n, deadline, max_passes=5000, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    best_c = compute_c5(h, n)
    pass_count = 0
    stuck_count = 0
    while time.time() < deadline and pass_count < max_passes:
        pass_count += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(ones) == n:
            break
        pairs = [(int(ones[rng.randint(len(ones))]), int(zeros[rng.randint(len(zeros))]))
                 for _ in range(min(200, len(ones) * len(zeros)))]
        rng.shuffle(pairs)
        # Also add all pairs if small enough
        if len(ones) * len(zeros) <= 25000:
            all_pairs = [(int(a), int(b)) for a in ones for b in zeros]
            rng.shuffle(all_pairs)
            pairs = all_pairs
        improved = False
        for i, j in pairs:
            if time.time() > deadline:
                return h, best_c
            if h[i] != 1.0 or h[j] != 0.0:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if improved:
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count >= 50 and len(ones) * len(zeros) <= 50000:
                # Perturbation: random swap
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
                best_c = compute_c5(h, n)
                stuck_count = 0
    return h, best_c

def continuous_refine(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        maxiter = min(200, max(10, int(remaining * 20)))
        if maxiter < 5:
            break
        mu = max(1.0, alpha * 0.005)
        try:
            res = minimize(smooth_max_obj, h,
                          jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxiter * 20,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = project_feasible(h_init, n_init)
    best_n = len(best_h)
    best_c = compute_c5(best_h, best_n)

    # Phase 1: Continuous L-BFGS-B on initial
    t1 = min(t0 + 120, deadline - 600)
    if t1 > t0 + 20:
        best_h, best_c = run_lbfgs(best_h, best_n, t1)
        best_n = len(best_h)

    # Phase 2: Round to binary and do local search
    h_bin = round_to_binary(best_h, best_n)
    c_bin = compute_c5(h_bin, best_n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin
    t2 = min(t0 + 500, deadline - 300)
    if t2 > time.time() + 30:
        best_h, best_c = binary_local_search(best_h, best_n, t2, seed=seed)

    # Phase 3: Continuous refinement after binary search
    t3 = min(t0 + 600, deadline - 200)
    if t3 > time.time() + 15:
        h_ref = project_feasible(best_h, best_n)
        h_ref, c_ref = continuous_refine(h_ref, best_n, t3)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    # Phase 4: More binary local search with perturbation
    t4 = min(t0 + 800, deadline - 150)
    if t4 > time.time() + 30:
        h_bin2 = round_to_binary(best_h, best_n)
        best_h, best_c = binary_local_search(h_bin2, best_n, t4, seed=seed + 1)

    # Phase 5: Try different n values
    for n_try in [100, 150, 200, 250, 400]:
        if time.time() > deadline - 100:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, deadline - time.time() - 80)
        if alloc < 40:
            break
        h_t, c_t = run_lbfgs(h_interp, n_try, time.time() + alloc * 0.4)
        if time.time() < deadline - 80:
            h_bin_t = round_to_binary(h_t, n_try)
            c_bt = compute_c5(h_bin_t, n_try)
            if c_bt < c_t:
                h_t, c_t = h_bin_t, c_bt
            h_t, c_t = binary_local_search(h_t, n_try, min(time.time() + 60, deadline - 60),
                                           seed=seed + n_try)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 6: Final rounds on the best solution
    t6 = deadline - 5
    if time.time() < t6 - 30:
        h_bin_f = round_to_binary(best_h, best_n)
        c_f = compute_c5(h_bin_f, best_n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_bin_f
        best_h, best_c = binary_local_search(best_h, best_n, t6 - 10, seed=seed + 999)

    if time.time() < t6 - 10:
        h_ref = project_feasible(best_h, best_n)
        h_ref, c_ref = continuous_refine(h_ref, best_n, t6 - 5)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
