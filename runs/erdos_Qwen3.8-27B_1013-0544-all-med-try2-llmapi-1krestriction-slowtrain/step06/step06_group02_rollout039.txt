```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def coordinate_descent(h, n, deadline, max_sweeps=200):
    """Coordinate descent exploiting the 1D structure."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        improved = False
        
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            # Compute s vector (slope of c_k with respect to h[i])
            s = np.zeros(2 * n - 1)
            # Term 1: h[i]*(1-h[i-k]) for 0 <= i-k < n
            # k ranges from i-(n-1) to i, idx = k + (n-1) ranges from i to i+n-1
            lo1 = i
            hi1 = min(i + n - 1, 2 * n - 2)
            for idx in range(lo1, hi1 + 1):
                k = idx - (n - 1)
                j = i - k
                s[idx] += (1.0 - h[j])
            
            # Term 2: -h[i+k]*h[i] for 0 <= i+k < n
            # k ranges from -i to n-1-i, idx ranges from n-1-i to 2n-2-i
            lo2 = max(n - 1 - i, 0)
            hi2 = min(2 * n - 2 - i, 2 * n - 2)
            for idx in range(lo2, hi2 + 1):
                k = idx - (n - 1)
                j2 = i + k
                s[idx] += (-h[j2])
            
            # A_k = c_k - s_k * h[i]
            A = c - s * h[i]
            
            # Find x in [0,1] minimizing max_k(A_k + s_k * x)
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(25):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(A + s * m1)
                f2 = np.max(A + s * m2)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            x_opt = (a + b) / 2.0
            val_opt = np.max(A + s * x_opt)
            
            if val_opt < c.max() - 1e-10:
                h[i] = x_opt
                improved = True
        
        # Check improvement
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = h.copy()
        elif not improved:
            break
    
    return best_h, best_c
```

Even though the problem is non-convex, we can use an iterated linearization:
1. At current h, linearize the objective around h.
2. Solve the resulting (approximately) convex problem.
3. Update h.
4. Repeat.

The linearization of c_k around h is:
c_k(h + δ) ≈ c_k(h) + ∇c_k(h) · δ

where ∇c_k(h)[i] = ∂c_k/∂h[i].

We can compute ∂c_k/∂h[i]:
c_k = ∑_n h[n+k](1-h[n])
∂c_k/∂h[i] = (1-h[i-k])*[0≤i-k<n] - h[i+k]*[0≤i+k<n] (same as s_k above!)

So the linearized problem is:
min max_k (c_k + s_k(i) * δ_i) ... wait, this doesn't quite work because s_k depends on i.

Let me write it properly. The linearized objective is:
∑_i s_k(i) * δ_i + c_k = c_k + ∑_i s_k(i) * (h[i]_new - h[i]_old)

where s_k(i) = (1-h[i-k])*[valid] - h[i+k]*[valid]

So the linearized problem is:
min t
s.t. c_k + ∑_i s_k(i) * δ_i ≤ t for all k
h[i] + δ[i] ∈ [0,1] for all i
∑(h[i] + δ[i]) = n/2

This IS a linear program! (The objective is linear, all constraints are linear.)

With n+1 variables (t and δ_0,...,δ_{n-1}) and 2n-1 + 2n + 1 constraints, for n=300 that's 301 variables and 1201 constraints. Very solvable with an LP solver!

This is essentially a "linearized min-max" approach. At each iteration:
1. Compute c_k for all k.
2. Compute the Jacobian (s_k(i) for all k, i).
3. Solve the LP to get δ.
4. Update h = h + δ (with clipping).
5. Repeat.

The Jacobian is a (2n-1) × n matrix. For n=300, that's 599 × 300 = 179,700 entries. Each entry is simple to compute.

```python
import cvxpy as cp

def linearized_min_max(h, n):
    """Solve linearized min-max LP."""
    c = np.correlate(h, 1 - h, mode='full')  # shape (2n-1,)
    
    # Build Jacobian: J[k, i] = s_k(i)
    # s_k(i) = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    # k ranges from -(n-1) to n-1, indexed 0 to 2n-2
    # For each i:
    #   Term 1: contributes (1-h[i-k]) to J[k,i] where k = i-j for j in [0,n-1]
    #     i.e., k ranges from i-(n-1) to i
    #   Term 2: contributes -h[i+k] to J[k,i] where k ranges from -i to n-1-i
    
    # Actually, let me think of it differently using the correlation structure.
    # c_k = sum_n h[n+k]*(1-h[n])
    # dc_k/dh[i] = (1-h[i-k])*[0<=i-k<n] + h[i+k]*(-1)*[0<=i+k<n]
    #            = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    
    # Build the constraint matrix A where A[k,i] = dc_k/dh[i]
    # For the LP: c + A@delta <= t
    # i.e., A@delta - t <= -c
    
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # Term 1: (1-h[i-k]) for k where 0 <= i-k < n
        # k = i-j for j in range(n), so k ranges from i-(n-1) to i
        for j in range(n):
            k = i - j  # this is the shift
            idx = k + (n-1)  # index in 0..2n-2
            if 0 <= idx < 2*n-1:
                A[idx, i] += (1.0 - h[j])
        # Term 2: -h[i+k] for k where 0 <= i+k < n
        # k ranges from -i to n-1-i
        for k in range(-i, n-i):
            idx = k + (n-1)
            if 0 <= idx < 2*n-1:
                j2 = i + k
                A[idx, i] += (-h[j2])
    
    # LP: min t
    # s.t. A @ delta - t <= -c  (i.e., c + A@delta <= t)
    #      0 <= h[i] + delta[i] <= 1
    #      sum(h + delta) = n/2
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        A @ delta - t <= -c,
        delta >= -h,  # h + delta >= 0
        delta <= 1 - h,  # h + delta <= 1
        cp.sum(h + delta) == n / 2.0
    ]
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, verbose=False)  # or CVXOPT, ECOS, etc.
    
    if t.value is not None and delta.value is not None:
        h_new = h + delta.value
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        return h_new
    return h
```

```python
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # Term 1: (1-h[j]) where j = i-k = i-(idx-n+1) = i-idx+n-1
        # Valid when 0 <= j < n, i.e., 0 <= i-idx+n-1 < n
        # → 1-n <= i-idx <= 0 → i <= idx <= i+n-1
        # Also idx in [0, 2n-2]
        lo = max(0, i)  # i <= idx, and idx >= 0
        hi = min(2*n-2, i + n - 1)  # idx <= i+n-1, and idx <= 2n-2
        for idx in range(lo, hi + 1):
            j = i - idx + n - 1
            A[idx, i] += (1.0 - h[j])
        
        # Term 2: -h[j2] where j2 = i+k = i+(idx-n+1) = i+idx-n+1
        # Valid when 0 <= j2 < n, i.e., 0 <= i+idx-n+1 < n
        # → n-1-i <= idx <= 2n-2-i
        lo2 = max(0, n - 1 - i)
        hi2 = min(2*n-2, 2*n-2-i)
        for idx in range(lo2, hi2 + 1):
            j2 = i + idx - n + 1
            A[idx, i] += (-h[j2])
    
    # LP: min t
    # s.t. A @ delta - t <= -c
    #      bounds on delta
    #      sum(delta) = 0
    
    # Variables: x = (delta_0, ..., delta_{n-1}, t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality: [A, -1] @ x <= -c
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -r)
        hi = min(1.0 - h[i], r)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is unbounded
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[:n]
        h_new = h + delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        return h_new, res.x[-1]
    return h, None
```

Wait, there's an issue. The LP minimizes t subject to c + A@delta <= t. If the linearization is good, the optimal t should be close to the current c.max(). But the linearization might not be tight, and the LP might give t < c.max() (which is impossible for the true objective). In that case, the actual c after the update might be worse.

So I need to check: after the update, compute the actual c5. If it improved, accept. If not, reject (and reduce the trust region radius).

Let me add this:

This looks good. The trust region adaptivity ensures we make progress.

Now, let me also think about the coordinate descent more carefully. The issue with my earlier implementation is that after updating h[i], the c values change, but I'm using stale c values for subsequent updates within the same sweep. This can lead to poor convergence.

A better approach: after each coordinate update, check if the overall c5 improved. If so, recompute c. If not, revert.

But that's O(n²) per coordinate (to recompute c), making the total O(n⁴) per sweep. For n=300, that's 8.1 billion operations. Too slow.

Alternative: use a "stale" c for all updates in a sweep, but verify at the end. If the sweep didn't help, try a different ordering or smaller steps.

The problem is invariant under: h → 1-h (complement). Wait, is it?
If h' = 1-h, then ∫h' = ∫(1-h) = 2-1 = 1. ✓
c_k(h') = ∑ h'[n+k](1-h'[n]) = ∑ (1-h[n+k])*h[n] = ∑ h[n](1-h[n+k])

And c_k(h) = ∑ h[n+k](1-h[n]).

These are related: c_k(h') = ∑ h[n](1-h[n+k]) = c_{-k}(h) (by changing variable).

So h and 1-h give the same C₅ (just with k → -k). This means we can WLOG assume something about the "bias" of h.

Also, the problem is invariant under reversal: h'[i] = h[n-1-i]. This gives c_k(h') = c_{-k}(h), so same C₅.

These symmetries don't directly help with optimization, but they confirm that the optimal solution should have some structure.

```python
def build_jacobian(h, n):
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # Term 1: A[idx, i] += (1-h[j]) where j = i - (idx - n + 1) = i - idx + n - 1
        # idx ranges from i to min(i+n-1, 2n-2)
        lo1 = i
        hi1 = min(i + n - 1, 2*n - 2)
        idxs = np.arange(lo1, hi1 + 1)
        js = i - idxs + n - 1  # = i - (idxs - (n-1)) = i - k
        A[idxs, i] += (1.0 - h[js])
        
        # Term 2: A[idx, i] += -h[j2] where j2 = i + (idx - n + 1) = i + idx - n + 1
        # idx ranges from max(0, n-1-i) to min(2n-2-i, 2n-2)
        lo2 = max(0, n - 1 - i)
        hi2 = min(2*n - 2 - i, 2*n - 2)
        idxs2 = np.arange(lo2, hi2 + 1)
        j2s = i + idxs2 - n + 1
        A[idxs2, i] += (-h[j2s])
    
    return A
```

This uses numpy vectorization for each column, making it O(n) numpy operations per column, total O(n²) numpy operations. Should be fast (~milliseconds).

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_jacobian(h, n):
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo1 = i
        hi1 = min(i + n - 1, 2 * n - 2)
        idxs = np.arange(lo1, hi1 + 1)
        js = i - idxs + n - 1
        A[idxs, i] += (1.0 - h[js])
        lo2 = max(0, n - 1 - i)
        hi2 = min(2 * n - 2 - i, 2 * n - 2)
        idxs2 = np.arange(lo2, hi2 + 1)
        j2s = i + idxs2 - n + 1
        A[idxs2, i] += (-h[j2s])
    return A

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    grad = np.zeros(n)
