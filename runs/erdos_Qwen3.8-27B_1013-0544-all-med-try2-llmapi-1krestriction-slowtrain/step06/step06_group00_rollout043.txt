```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
```

```python
def gradient_fft(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gmax = g.max()
    shifted = alpha * (g - gmax)
    smax = shifted.max()
    exps = np.exp(shifted - smax)
    p = exps / exps.sum()
    
    # Compute convolutions via FFT
    # p has length 2n-1, h has length n
    # Linear convolution of p (length 2n-1) and h (length n) has length 3n-2
    # We only need the first n values (for indices 0 to n-1)
    
    # For (p * h)[i] = sum_k p[k] * h[i+k], we need i+k in [0,n)
    # This is a cross-correlation-like quantity
    
    # Let me use a different approach:
    # (p * h)[i] = sum_{k} p[k] * h[i+k]
    # Let j = i+k, k = j-i:
    # = sum_{j} p[j-i] * h[j] (for valid k and j)
    # = sum_{j} p[j-i] * h[j]
    # This is the cross-correlation of p and h at lag i: (p vs h)[i]
    
    # numpy.correlate(p, h, 'full')[i + (len(p)-1)] gives sum p[m] * h[m+i] ... 
    # Actually let me just use the direct computation with FFT.

    N_fft = 2 ** int(np.ceil(np.log2(3*n)))
    P = np.fft.rfft(p, N_fft)
    H = np.fft.rfft(h, N_fft)
    
    # p * h (convolution): gives sum_k p[k]*h[i-k]... no
    # np.convolve(p, h) gives (p*h)[n] = sum_k p[k]*h[n-k]
    # So (p*h)[i] = sum_k p[k]*h[i-k] which is what I want for the first term!

    # = sum_k p[-k]*h[i-k+2k]... let me think differently
    # sum_k p[k]*h[i+k] = sum_j p[j-i]*h[j] (j=i+k)

    # Let m = -k: sum_m q[m]*h[i-m] = (q*h)[i]
```

```python
def compute_grad_direct(h, n, p):
    """p has length 2n-1, indexed by lag k from -(n-1) to (n-1)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx_k in range(2*n - 1):
        k = idx_k - (n - 1)  # actual lag
        pk = p[idx_k]
        if pk < 1e-14:
            continue
        # Term 1: p[k] * (1 - h[i-k]) for valid i-k
        # i-k in [0, n) => k <= i < k+n, intersect [0, n)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1-k:hi1-k])
        # Term 2: -p[k] * h[i+k] for valid i+k
        # i+k in [0, n) => -k <= i < n-k, intersect [0, n)
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2+k:hi2+k]
    grad *= dx
    return grad
```

```python
    g = np.correlate(x, 1 - x, mode='full') * dx
    gmax = g.max()
    shifted = alpha * (g - gmax)
    smax = shifted.max()
    return gmax + (smax + np.log(np.exp(shifted - smax).sum())) / alpha

def smooth_max_grad(x, n, alpha):
```
