```python
def smooth_grad_step(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    # gradient = weighted average of individual gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += weights[k] * dx * (1.0 - h[idx + lag])
            grad[idx + lag] -= weights[k] * dx * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += weights[k] * dx * (1.0 - h[idx])
            grad[idx] -= weights[k] * dx * h[idx + la]
    return grad
```
