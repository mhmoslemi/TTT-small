def sqp_iteration(h, n, solver='ECOS'):
    """One SQP iteration: linearize overlaps and solve LP."""
    c_all = compute_all_overlaps(h, n)
    dx = 2.0 / n
    
    # Compute gradients for all lags
    # g[k, j] = dO_k/dh[j] (without dx factor, we'll add it in the constraint)
    # G[k, j] = (1-h[j+k])*1{j+k<n} - h[j-k]*1{j-k>=0}
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        for j in range(n):
            g = 0.0
            if 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                g -= h[j - lag]
            G[k, j] = g
    
    # LP: minimize t
    # s.t. c_all[k] + dx * G[k,:] @ delta <= t for all k
    #      h - 1 <= delta <= 1 - h  (box constraints on h+delta)
    #      sum(delta) = 0
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(delta) == 0]

    constraints.append(delta >= -(h))  # h + delta >= 0
    constraints.append(delta <= 1-h)  # h + delta <= 1
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver)
        if delta.value is not None:
            h_new = h + delta.value
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)

    except Exception:
        pass
    return h, compute_c5(h, n)

import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_gradient_matrix(h, n):
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        if lag >= 0:
            G[k, :n-lag] += (1.0 - h[lag:])
        else:
            G[k, -lag:] += (1.0 - h[:n+lag])
        if lag > 0:
            G[k, lag:] -= h[:n-lag]
        elif lag < 0:
            G[k, :n+lag] -= h[-lag:]

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

            if tot < 1e-15:
                break

    m = c.max()
    shifted = alpha * (c - m)

    return np.ones(n)

    best_h = h.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_gradient_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            G[k, :n - lag] += (1.0 - h[lag:])
        else:
            G[k, -lag:] += (1.0 - h[:n + lag])
        if lag > 0:
            G[k, lag:] -= h[:n - lag]
        elif lag < 0:
            G[k, :n + lag] -= h[-lag:]
    return G

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)

        t = cp.Variable()
        constraints = [cp.sum(delta) == 0, delta >= -h, delta <= 1.0 - h]
        for k in range(2 * n - 1):
            constraints.append(c_all[k] + dx * (G[k, :] @ delta) <= t)
        prob = cp.Problem(cp.Minimize(t), constraints)
