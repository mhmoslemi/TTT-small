    target_sum = n / 2.0

        if abs(diff) < 1e-14:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room

            if tot < 1e-14: break

```python
def compute_grad_at_lag(h, n, k):
    """Compute ∂c[k]/∂h where c = correlate(h, 1-h, 'full')"""
    grad = np.zeros(n)
    offset = k - (n - 1)
    # First term: for each i, if i+offset is in [0,n), add (1-h[i+offset])
    # This is (1 - roll(h, offset)) with masking
    if offset >= 0:
        valid = np.arange(n - offset)
        grad[valid] += (1 - h[valid + offset])
    else:
        valid = np.arange(-offset, n)
        grad[valid] += (1 - h[valid + offset])
    
    # Second term: for each i, if i-offset is in [0,n), subtract h[i-offset]
    if offset >= 0:
        valid = np.arange(offset, n)
        grad[valid] -= h[valid - offset]
    else:
        valid = np.arange(0, n + offset)
        grad[valid] -= h[valid - offset]
    
    return grad
```

The gradient of the log-sum-exp of c (scaled by alpha) is:

This is a weighted average of the gradients at each lag, weighted by the softmax. It's more robust to lag switching.

```python
def smooth_grad(h, n, alpha):
    """Compute gradient of the smooth max of correlation"""
    corr = np.correlate(h, 1-h, mode='full')
    # Softmax weights
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()

    # Actually, we need: grad_h[i] = sum_m weights[m] * ∂c[m]/∂h[i]
    
    # For each m, ∂c[m]/∂h[i] = (1-h[i+m-(n-1)]) - h[i-m+(n-1)]
    # We can compute this as: for each m, the contribution to grad is
    # weights[m] * ((1 - shifted_h1[m]) - shifted_h2[m])
    # where shifted_h1 is h shifted by m-(n-1) and shifted_h2 is h shifted by -(m-(n-1))
    
    # More efficiently:
    # grad[i] = sum_m weights[m] * (1 - h[i + m - (n-1)]) - sum_m weights[m] * h[i - m + (n-1)]
    #        = sum_m weights[m] - sum_m weights[m] * h[i + m - (n-1)] - sum_m weights[m] * h[i - m + (n-1)]
    #        = 1 - conv(weights, h)[i] - conv(weights[::-1], h)[i]  (approximately)
    
    # Actually, let me just compute it directly for the lags with significant weight
    grad_h = np.zeros(n)
    # Only consider lags with non-negligible weight
    significant = np.where(weights > 1e-6)[0]
    for m in significant:
        w = weights[m]
        offset = m - (n - 1)
        if offset >= 0:
            valid1 = np.arange(n - offset)
            valid2 = np.arange(offset, n)
            grad_h[valid1] += w * (1 - h[valid1 + offset])
            grad_h[valid2] -= w * h[valid2 - offset]
        else:
            valid1 = np.arange(-offset, n)
            valid2 = np.arange(0, n + offset)
            grad_h[valid1] += w * (1 - h[valid1 + offset])
            grad_h[valid2] -= w * h[valid2 - offset]
    
    return grad_h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_smooth_grad(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * (corr - corr.max())
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad_h = np.zeros(n)
    significant = np.where(weights > 1e-8)[0]
    for m in significant:
        w = weights[m]
        offset = m - (n - 1)
        if offset >= 0:
            v1 = np.arange(n - offset)
            v2 = np.arange(offset, n)
            grad_h[v1] += w * (1 - h[v1 + offset])
            grad_h[v2] -= w * h[v2 - offset]
        else:
            v1 = np.arange(-offset, n)
            v2 = np.arange(0, n + offset)
            grad_h[v1] += w * (1 - h[v1 + offset])
            grad_h[v2] -= w * h[v2 - offset]
    return grad_h

def grad_descent(h, n, deadline, alpha=5000, step0=0.02, min_step=1e-6):
    best_c = compute_c5(h, n)
    h = h.copy()
    step = step0
    stalling = 0
    for _ in range(200000):
        if time.time() > deadline:
            break
        grad = compute_smooth_grad(h, n, alpha)
        grad -= np.mean(grad)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        direction = -grad / g_norm
        h_new = h + step * direction
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            stalling = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.8
            stalling += 1
            if stalling > 200 or step < min_step:
                break
    return h, best_c

def fine_local_search(h, n, deadline, deltas=None, max_rounds=5000):
    best_c = compute_c5(h, n)
    h = h.copy()
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved_any = False
        for _round in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for sign in (1, -1):
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                                improved_any = True
                            else:
                                h[i], h[j] = hi - sign * d, hj + sign * d
            if not improved:
                break
        if not improved_any and d < 0.0001:
            break
    return h, best_c

def triplet_search(h, n, deadline, d=0.005, max_rounds=2000):
    best_c = compute_c5(h, n)
    h = h.copy()
    rng = np.random.RandomState(123)
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(200):
            if time.time() > deadline:
                return h, best_c
            i, j, k = rng.choice(n, size=3, replace=False)
            for signs in ((1, -1, -1), (1, 1, -2), (-1, 1, 1), (-1, -1, 2)):
                hi = h[i] + signs[0] * d
                hj = h[j] + signs[1] * d
                hk = h[k] + signs[2] * d
                if 0 <= hi <= 1 and 0 <= hj <= 1 and 0 <= hk <= 1:
                    h[i], h[j], h[k] = hi, hj, hk
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j], h[k] = h[i] - signs[0]*d, h[j] - signs[1]*d, h[k] - signs[2]*d
        if not improved:
            break
    return h, best_c

def binary_push(h, n):
    h2 = np.where(h > 0.5, 1.0, 0.0).astype(float)
    h2 = project_feasible(h2, n)
    c2 = compute_c5(h2, n)
    c1 = compute_c5(h, n)
    if c2 < c1:
        return h2, c2
    return h, c1

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [200, 1000, 5000, 20000, 80000, 300000, 1000000, 3000000, 10000000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    if time.time() < deadline - 30:
        ht, ct = grad_descent(best_h, n, time.time() + min(30, (deadline - time.time())*0.4), alpha=5000)
        if ct < best_c:
            best_c = ct
            best_h = ht

    ht, ct = binary_push(best_h, n)
    if ct < best_c:
        best_c = ct
        best_h = ht
        if time.time() < deadline - 10:
            ht2 = slsqp_optimize(best_h, n, 500000, maxiter=1500)
            c2 = compute_c5(ht2, n)
            if c2 < best_c:
                best_c = c2
                best_h = ht2

    if time.time() < deadline - 5:
        best_h, best_c = fine_local_search(best_h, n, deadline - 2)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
