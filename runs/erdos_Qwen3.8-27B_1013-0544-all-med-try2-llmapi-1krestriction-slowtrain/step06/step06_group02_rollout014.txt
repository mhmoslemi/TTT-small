import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def _smooth_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def _cons_eq(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 10.0), (500, 20.0), (2000, 50.0), (10000, 100.0), (100000, 500.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return _smooth_obj(x, n, alpha) + mu * (x.sum() - n/2.0)**2
        def grad(x):
            return _smooth_grad(x, n, alpha) + 2.0*mu*(x.sum() - n/2.0)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters*10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coordinate_descent(h, n, deadline, max_passes=100):
    """Exact coordinate descent for min max_k f_k(h)."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    nm1 = n - 1
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            c = compute_c5_vec(h, n)
            mval = c.max()
            
            # Compute slopes for coordinate j
            slopes = np.zeros(2*n-1)
            # k >= 0: m = nm1 + k
            for k in range(n):
                m = nm1 + k
                s = 0.0
                if j + k < n:
                    s += (1.0 - h[j+k])
                if j >= k:
                    s -= h[j-k]
                slopes[m] = dx * s
            # k < 0: kk = -k, m = nm1 - kk
            for kk in range(1, n):
                m = nm1 - kk
                s = 0.0
                if j >= kk:
                    s += (1.0 - h[j-kk])
                if j + kk < n:
                    s -= h[j+kk]
                slopes[m] = dx * s
            
            # Find t minimizing max_m(c[m] + slopes[m]*t) for t in [-h[j], 1-h[j]]
            lo = -h[j]
            hi = 1.0 - h[j]
            
            # Bisection on subgradient
            for _ in range(80):
                mid = (lo + hi) * 0.5
                vals = c + slopes * mid
                max_val = vals.max()
                active = vals >= max_val - 1e-14
                if not np.any(active):
                    active = vals >= max_val * 0.999 + 0.001 * max_val
                min_s = slopes[active].min()
                max_s = slopes[active].max()
                if min_s > 1e-14:
                    hi = mid
                elif max_s < -1e-14:
                    lo = mid
                else:
                    break

            h[j] = h[j] + t_new
            
            new_c = compute_c5(h, n)
            if new_c < best_c - 1e-15:
                best_c = new_c
                improved = True
        
        if not improved and p > 2:
            break
    
    return h, best_c

- h[j] appears directly when i = j (if j ≥ k): contribution h[j](1-h[j-k]), ∂/∂h[j] = (1-h[j-k])
- h[j] appears in (1-h[i-k]) when i-k = j, i.e., i = j+k (if j+k ≥ k and j+k ≤ n-1, i.e., j ≥ 0 and j ≤ n-1-k): contribution h[j+k](1-h[j]), ∂/∂h[j] = -h[j+k]

So for k ≥ 0:
∂f_k/∂h[j] = (1-h[j-k]) * 1{j ≥ k} + (-h[j+k]) * 1{j ≤ n-1-k}

For k < 0 (kk = -k > 0): valid i are 0 to n-1-kk.
f_{-kk} = sum_{i=0}^{n-1-kk} h[i](1-h[i+kk])

- h[j] appears when i = j (if j ≤ n-1-kk): ∂/∂h[j] = (1-h[j+kk])
- h[j] appears in (1-h[i+kk]) when i+kk = j, i.e., i = j-kk (if j-kk ≥ 0 and j-kk ≤ n-1-kk, i.e., j ≥ kk and j ≤ n-1): ∂/∂h[j] = -h[j-kk]

So for k < 0 (kk = -k):
∂f_{-kk}/∂h[j] = (1-h[j+kk]) * 1{j ≤ n-1-kk} + (-h[j-kk]) * 1{j ≥ kk}

Now let me rewrite in terms of m (the index in the full correlation):
m = nm1 + k for k ≥ 0, m = nm1 - kk for k = -kk < 0.

For m = nm1 + k (k ≥ 0, k = 0,...,n-1):
∂f_m/∂h[j] = (1-h[j-k]) * 1{j ≥ k} + (-h[j+k]) * 1{j ≤ n-1-k}

For m = nm1 - k (k = 1,...,n-1):
∂f_m/∂h[j] = (1-h[j+k]) * 1{j ≤ n-1-k} + (-h[j-k]) * 1{j ≥ k}
