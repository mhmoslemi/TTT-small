    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    n_total = int(t_end - t_start)
    T0 = 1.0
    Tend = 0.01
    
    for step in range(n_total):
        t = time.time()
        if t > t_end:
            break
        # Linear cooling
        frac = (t - t_start) / (t_end - t_start)
        T = T0 * (Tend / T0) ** frac
        
        # Choose block size
        b = rng.randint(1, 6)  # 1 to 5
        
        # Choose positions
        i = rng.randint(0, n - b + 1)
        j = rng.randint(0, n - b + 1)
        if abs(i - j) < b:
            continue
        
        # Check if blocks are different
        block_i = h[i:i+b]
        block_j = h[j:j+b]
        if np.array_equal(block_i, block_j):
            continue
        
        # Try the swap
        h[i:i+b] = block_j
        h[j:j+b] = block_i
        
        new_corr = np.correlate(h, 1-h, mode='full')

import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_from_score(h, n):
    """Convert to binary by thresholding at the n/2-th largest value"""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh = np.clip(sh, -500, 0)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    dx = 2.0 / n

    best_c = cur_c
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        np.random.shuffle(z_list)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = list(ones)
            np.random.shuffle(o_list)
            for j in o_list:
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)

                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)

            best_h = h.copy()

In a binary sequence, a single swap changes the correlation by a specific amount. A block swap of size b changes the correlation by a more complex amount. The advantage of block moves is that they can move a "cluster" of 1s to a new location, which is a fundamentally different type of move than moving a single 1.

So yes, SA with block moves should be useful for finding better solutions.

    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = weyl_init(n0, 3)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: SA at original n
    p1 = t0 + 0.28*(budget_s-5)
    hs,cs = block_sa(best_h, n, t0, p1, seed, mb=5)
    if cs < best_c: best_c,best_h = cs,hs
    # Phase 2: Single-swap refinement
    p2 = t0 + 0.42*(budget_s-5)
    hs,cs = swap_search(best_h, n, p2, sweeps=1500)
    if cs < best_c: best_c,best_h = cs,hs
    # Phase 3: SLSQP polish
    p3 = t0 + 0.50*(budget_s-5)
    if time.time() < p3:
        hp = project_feasible(best_h + 0.08*rng.randn(n), n)
        for al in [5000, 50000, 500000]:
            if time.time() > p3-2: break
            hp = slsqp(hp, n, al, 800)
            c = compute_c5(hp, n)
            if c < best_c: best_c,best_h = c,hp
        hs,cs = swap_search(best_h, n, p3, sweeps=300)
        if cs < best_c: best_c,best_h = cs,hs
    # Phase 4: Multi-n exploration
    p4 = t0 + 0.78*(budget_s-5)
    for nh in [200, 300, 350, 400, 250]:
        if time.time() > p4-15: break
        if nh == n: continue
        xo = np.linspace(0,2,n,endpoint=False)
        xn = np.linspace(0,2,nh,endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        t_end = min(p4-8, time.time()+25)
        hs2,cs2 = block_sa(hh, nh, time.time(), t_end, seed+100+nh, mb=4)
        hs3,cs3 = swap_search(hs2, nh, p4-5, sweeps=200)
        if cs3 < best_c: best_c,best_h,n = cs3,hs3,nh
        elif cs2 < best_c: best_c,best_h,n = cs2,hs2,nh
    # Phase 5: Also try Weyl initializations at best n
    p5 = t0 + 0.88*(budget_s-5)
    for p in [3,7,11,13,17]:
        if time.time() > p5-5: break
        hw = weyl_init(n, p)
        t_end = min(p5-3, time.time()+12)
        hw2,cw2 = block_sa(hw, n, time.time(), t_end, seed+p, mb=4)
        hw3,cw3 = swap_search(hw2, n, p5-2, sweeps=100)
        if cw3 < best_c: best_c,best_h = cw3,hw3
    # Phase 6: Final refinement
    hs,cs = swap_search(best_h, n, dl-15, sweeps=2000)
    if cs < best_c: best_c,best_h = cs,hs
    if time.time() < dl-10:
        hp = slsqp(best_h + 0.03*rng.randn(n), n, 2000000, 1500)
        c = compute_c5(hp, n)
        if c < best_c: best_c,best_h = c,hp
    if time.time() < dl-3:
        hs,cs = swap_search(best_h, n, dl, sweeps=500)
        if cs < best_c: best_c,best_h = cs,hs
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
