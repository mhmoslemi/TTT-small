Actually, since the problem is non-convex, pure gradient descent might not work well. Let me use a different approach:

**Projected gradient method with adaptive step size:**
1. Compute gradient g
2. Try step sizes η in a geometric sequence
3. Take the step that gives the best improvement in C₅ (not the smooth approximation)
4. Project to feasible set

Or better: **use the gradient to guide pair search.**
- Instead of random pairs (i, j), use the gradient to determine which coordinates to adjust
- Coordinates with large positive gradient should be decreased
- Coordinates with large negative gradient should be increased
- Adjust pairs (i, j) where g[i] > 0 and g[j] < 0 (or vice versa)

This is much more targeted than random pair search.

Let me also add **momentum** to the gradient descent to speed up convergence.

Here's my revised algorithm:

```python
def gradient_descent(h, n, deadline, alpha=10000, max_iter=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    velocity = np.zeros(n)
    beta = 0.9  # momentum
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        F, g = compute_grad_h_fft(h, n, alpha)
        # Gradient step with momentum
        velocity = beta * velocity - 0.01 * g  # step size 0.01
        h_new = h + velocity
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Reduce step size
            velocity *= 0.5
    
    return best_h, best_c
```

Actually, I realize I should be more careful. The smooth max with large α approximates the true max, but the gradient of the smooth max is not the same as the subgradient of the true max. For large α, the weights w[k] concentrate on the active set (shifts where c[k] ≈ max), so the gradient points in a direction that reduces the maximum.

Let me also think about the **active set** approach:
- At the optimum, the active set is the set of shifts k* where c[k*] = C₅
- We want to find h such that all active shifts have the same value, and this value is as small as possible
- This is like a minimax problem

For the minimax problem, we can use the following approach:
1. Identify the active set A = {k : c[k] > C₅ - ε}
2. Minimize the average of c[k] for k in A (this will also reduce the max)
3. Update the active set
4. Iterate

This is similar to the method of conditional gradients or the diagonal quadratically constrained quadratic program (D-QCCP) approach.

Let me implement a simpler version:

```python
def active_set_optimize(h, n, deadline, alpha_base=1000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for outer in range(50):
        if time.time() > deadline:
            break
        # Compute current correlation
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()
        
        # Active set: shifts where c is close to max
        threshold = c_max - 0.01
        active = c > threshold
        
        # Minimize sum of active c[k] using gradient descent
        # This is equivalent to smooth max with uniform weights on active set
        for inner in range(100):
            if time.time() > deadline:
                break
            # Recompute active set
            c = np.correlate(h, 1 - h, mode='full') * dx
            c_max = c.max()
            threshold = c_max - 0.005
            active = c > threshold
            
            # Weight: uniform on active set
            w = np.zeros(2*n-1)
            w[active] = 1.0
            w[active] /= w.sum()
            
            # Compute weighted sum gradient
            # sum_k w[k] * c[k] is a bilinear form in h
            # Its gradient wrt h[i] is:
            # sum_k w[k] * dx * [(1-h[i+s_k]) - h[i-s_k]] (with boundary conditions)
            # Same as before but with uniform weights on active set
            
            grad = np.zeros(n)
            for s in range(-(n-1), n):
                w_s = w[s + (n-1)]
                if w_s < 1e-15:
                    continue
                lo1 = max(0, -s)
                hi1 = min(n, n - s)
                if lo1 < hi1:
                    idx = np.arange(lo1, hi1)
                    grad[idx] += w_s * (1.0 - h[idx + s])
                lo2 = max(0, s)
                hi2 = min(n, n + s)
                if lo2 < hi2:
                    idx2 = np.arange(lo2, hi2)
                    grad[idx2] -= w_s * h[idx2 - s]
            grad *= dx
            
            # Line search
            eta = 0.1
            for _ in range(20):
                h_try = project_feasible(h - eta * grad, n)
                c_try = compute_c5(h_try, n)
                if c_try < c_max:
                    h = h_try
                    break
                eta *= 0.5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = np.clip(alpha * (c - m), -500, 500)
    ev = np.exp(sh)
    F = m + np.log(ev.sum()) / alpha
    w = ev / ev.sum()
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[s + n - 1]
        if ws < 1e-15:
            continue
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += ws * (1.0 - h[idx + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= ws * h[idx2 - s]
    grad *= dx
    return F, grad

def slsqp_optimize(h, n, alpha, maxiter=4000):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = np.clip(alpha * (c - m), -500, 500)
        ev = np.exp(sh)
        return float(m + np.log(ev.sum()) / alpha)
    def obj_grad(x):
        _, g = compute_grad(x, n, alpha)
        return g
    def sum_fun(x):
        return float(x.sum() - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_optimize(h, n, deadline, rng, alphas=None):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        alpha_t = time.time()
        vel = np.zeros(n)
        beta = 0.85
        eta = 0.05
        stall = 0
        for it in range(20000):
            if time.time() > deadline - 2:
                break
            F, g = compute_grad(h, n, alpha)
            vel = beta * vel - eta * g
            h_new = project_feasible(h + vel, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                stall = 0
                eta = min(0.1, eta * 1.05)
            else:
                if c_new < compute_c5(h, n) - 1e-15:
                    h = h_new
                eta *= 0.7
                vel *= 0.5
                stall += 1
                if stall > 50:
                    break
        beta = min(0.95, beta + 0.05)
    return best_h, best_c

def pair_search(h, n, deadline, deltas, max_passes=100):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    hi, hj = h[i]+d, h[j]-d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi-d, hj+d
                    hi2, hj2 = h[i]-d, h[j]+d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

    t3 = t0 + (deadline - t0) * 0.85
```
