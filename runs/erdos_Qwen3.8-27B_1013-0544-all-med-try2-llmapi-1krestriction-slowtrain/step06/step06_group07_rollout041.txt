def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s

def multiplicative_weights_cd(h, n, n_iters, t0, budget_s, tau_start=1.0, tau_end=0.1):
    """Multiplicative weights + coordinate descent."""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    w = np.ones(2*n - 1)  # uniform weights initially
    dx = 2.0 / n
    
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 60:
            break
        
        # Compute current overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        
        # Update weights (multiplicative weights)
        tau = tau_start * (tau_end / tau_start) ** (it / max(1, n_iters - 1))
        w *= np.exp(overlaps / tau)
        w /= w.sum()
        
        # Compute gradient
        grad = compute_grad(h, w, n)
        
        # Binary search on lambda for Lagrangian
        lo, hi = -5.0, 5.0
        for _ in range(50):
            lam = (lo + hi) / 2
            # h[m] = 0 if grad[m] + lam > 0, else 1
            h_trial = (grad + lam <= 0).astype(float)
            s = h_trial.sum()
            if s > target_sum:
                lo = lam  # need to make more h=0, increase lam
            else:
                hi = lam
        
        h_new = (grad + (lo + hi) / 2 <= 0).astype(float)
        
        # Project to exact sum (handle ties)
        h_new = project_feasible(h_new, target_sum)
        
        # Evaluate
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        
        h = h_new
    
    return best_h, best_c5

What about using the "fair max" (generalized entropy) and doing a proper gradient descent?

The fair max of order p: M_p(x) = (∑ x_i^p)^(1/p) for p > 1, approaching max as p → ∞.

Alternatively, the "entropic max": M(x) = (1/β) * log(∑ exp(β*x_i))

Both are smooth and their gradients are well-defined. The entropic max gradient is:
∇M(x) = exp(β*x) / ∑exp(β*x) (a probability distribution over the indices)

So the gradient of the entropic max of the overlaps is:
∇_h [M(overlaps)] = ∑_j p[j] * ∂overlap_j/∂h

where p[j] = exp(β*overlap_j) / ∑exp(β*overlap_j)

This is exactly the multiplicative weights distribution! So the MW approach IS the gradient of the entropic max. The difference is:
- MW-CD: computes the exact minimizer of the weighted sum (coordinate descent)
- Gradient descent: takes a small step in the gradient direction

The MW-CD is more aggressive (goes to the exact minimizer of the surrogate) while gradient descent is more conservative.

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad(h, w, n):
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in range(2 * n - 1):
        k = j - (n - 1)
        wj = w[j]
        if wj < 1e-15:
            continue
        lo_A = max(0, k)
        hi_A = min(n, k + n)
        if lo_A < hi_A:
            idx = np.arange(lo_A, hi_A)
            grad[idx] += wj * omh[idx - k]
        lo_B = max(0, -k)
        hi_B = min(n, n - k)
        if lo_B < hi_B:
            idx = np.arange(lo_B, hi_B)
            grad[idx] -= wj * h[idx + k]
    return grad


def mw_cd(h_init, n, n_iters, t0, time_limit):
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    w = np.ones(2 * n - 1)
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * (2.0 / n)
        frac = it / max(1, n_iters)
        tau = 0.5 * (1.0 - 0.85 * frac)
        w = w * np.exp(overlaps / tau)
        w = w / w.sum()
        grad = compute_grad(h, w, n)
        lo, hi = -5.0, 5.0
        for _ in range(60):
            lam = (lo + hi) / 2.0
            s = float((grad + lam <= 0).sum())
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        lam_s = (lo + hi) / 2.0
        h_new = (grad + lam_s <= 0).astype(float)
        h_new = project_feasible(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    return best_h, best_c5


def subgrad_refine(h, n, n_iters, t0, time_limit):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_w = int(np.argmax(corr))
        w = np.zeros(2 * n - 1)
        w[k_w] = 1.0
        grad = compute_grad(h, w, n)
        lr = 0.8 / (1.0 + 0.01 * it)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        if it % 500 == 499:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    return best_h, best_c5


def con_eq(h, target_sum):
    return h.sum() - target_sum


def con_eq_jac(h, target_sum):
    return np.ones(len(h))


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def slsqp_polish(h, n, t0, time_limit):
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum,)}]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for beta in [200, 1000, 5000, 20000, 80000]:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass
    return best_h, best_c5


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: MW-CD on initial
    if rem() > 150:
        h1, c5_1 = mw_cd(best_h, n, 8000, t0, budget_s - 150)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Subgradient refine
    if rem() > 120:
        h2, c5_2 = subgrad_refine(best_h, n, 50000, t0, budget_s - 120)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: SLSQP polish
    if rem() > 60:
        h3, c5_3 = slsqp_polish(best_h, n, t0, budget_s - 60)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Try other n values
    if rem() > 150:
        for n_try in [60, 65, 75, 80, 85, 90, 95, 100]:
            if rem() < 80 or n_try == n:
                continue
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, n_try / 2.0)
            h_t2, c5_t2 = mw_cd(h_t, n_try, 3000, t0, budget_s - 80)
            if c5_t2 < best_c5:
                best_h, best_c5 = h_t2, c5_t2
                n = n_try
            if rem() > 40:
                h_t3, c5_t3 = slsqp_polish(best_h, n, t0, budget_s - 40)
                if c5_t3 < best_c5:
                    best_h, best_c5 = h_t3, c5_t3

    # Phase 5: Random restarts
    if rem() > 80:
        n_best = len(best_h)
        for _ in range(8):
            if rem() < 25:
                break
            h_r = best_h + np.random.randn(n_best) * 0.15
            h_r = project_feasible(h_r, n_best / 2.0)

        h_fin, c5_fin = slsqp_polish(best_h, n_fin, t0, budget_s - 3)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    return best_h, best_c5, len(best_h)
