        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_values(h, n):
    """Return all correlation values (not just the max)"""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_max_obj(x, n, alpha, beta):
    """Objective: smooth max of c5 values + penalty for sum constraint"""
    c = c5_values(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    pen = beta * (np.sum(x) - n / 2.0) ** 2
    return lse + pen

def smooth_max_grad(x, n, alpha, beta):
    """Analytical gradient"""
    c = c5_values(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    weights = exps / exps.sum()  # softmax weights
    
    # Sum constraint gradient
    sum_grad = 2.0 * beta * (np.sum(x) - n / 2.0) * np.ones(n)
    
    # Correlation gradient: dc[lag]/dx[j]
    # c[lag] = sum_i x[i] * (1-x[i+lag]) for valid i
    # dc[lag]/dx[j] = (1-x[j+lag]) - x[j-lag] (with boundary conditions)
    
    # For efficiency, compute using the structure
    # grad[j] = sum_lag weights[lag] * ((1-x[j+lag])*I[j+lag valid] - x[j-lag]*I[j-lag valid])
    
    grad = np.zeros(n)
    # lags range from 0 to 2n-2 in correlate output
    # lag_in_output = k, actual_lag = k - (n-1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        # Contribution: (1-x[j+lag]) for j+lag in [0,n-1] minus x[j-lag] for j-lag in [0,n-1]
        if lag >= 0:
            # (1-x[j+lag]) is valid for j = 0, ..., n-1-lag
            idx = np.arange(n - lag)
            grad[idx] += w * (1 - x[idx + lag])
            # -x[j-lag] is valid for j = lag, ..., n-1
            idx2 = np.arange(lag, n)
            grad[idx2] += -w * x[idx2 - lag]
        else:
            lag = -lag
            # (1-x[j+lag]) with original negative lag: (1-x[j-lag]) valid for j = lag, ..., n-1
            idx = np.arange(lag, n)
            grad[idx] += w * (1 - x[idx - lag])
            # -x[j+lag] valid for j = 0, ..., n-1-lag
            idx2 = np.arange(n - lag)
            grad[idx2] += -w * x[idx2 + lag]
    
    return grad + sum_grad

def optimize_lbfgsb(h, n, deadline, alphas):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        beta = 50.0 * alpha  # penalty coefficient
        
        res = minimize(
            smooth_max_obj, h, args=(n, alpha, beta),
            jac=smooth_max_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-15}
        )
        
        ht = project_feasible(res.x, n)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
            h = ht  # warm start for next alpha
    
    return best_h, best_c

def local_search(h, n, deadline, steps=None):
    """Local search with pair moves and small perturbations"""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    if steps is None:
        steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            if time.time() > deadline:
                break
            # Random pair moves
            n_tries = 5000
            for _ in range(n_tries):
                if time.time() > deadline:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                sign = 1 if np.random.rand() < 0.5 else -1
                d = step * sign
                hi = h[i] + d
                hj = h[j] - d
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = hi - d, hj + d
    return h, best_c

def block_local_search(h, n, deadline):
    """Try moving blocks of consecutive values"""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    block_sizes = [2, 3, 5, 8, 13]
    for bs in block_sizes:
        if time.time() > deadline:
            break
        n_tries = 2000
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = np.random.randint(0, n - bs + 1)
            j = np.random.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            d = np.random.uniform(-0.03, 0.03)
            block_i = h[i:i+bs].copy()
            block_j = h[j:j+bs].copy()
            new_block_i = block_i + d
            new_block_j = block_j - d
            if np.all(new_block_i >= 0) and np.all(new_block_i <= 1) and \
               np.all(new_block_j >= 0) and np.all(new_block_j <= 1):
                h[i:i+bs] = new_block_i
                h[j:j+bs] = new_block_j
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                else:
                    h[i:i+bs] = block_i
                    h[j:j+bs] = block_j
    return h, best_c

def simulated_annealing(h, n, deadline, T0=0.01, T_end=0.0001):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    n_total = 50000
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / n_total)
        
        # Generate a move
        move_type = np.random.random()
        if move_type < 0.5:
            # Single pair move
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            d = np.random.uniform(-0.05, 0.05)
            hi, hj = h[i] + d, h[j] - d
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                old_i, old_j = h[i], h[j]
                h[i], h[j] = hi, hj
                c_new = compute_c5(h, n)
