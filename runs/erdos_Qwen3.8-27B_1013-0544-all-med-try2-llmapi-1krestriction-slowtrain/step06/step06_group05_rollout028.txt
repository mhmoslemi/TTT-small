import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def overlap_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_for_shift(h, n, s):
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        a = -s
        grad[:n - a] += (1.0 - h[a:])
        grad[a:] -= h[:n - a]
    return grad

def subgradient_descent(h, n, deadline, epsilon=0.002, max_iter=100000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    eta = 0.005
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        O = overlap_vector(h, n)
        c = O.max()
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        
        thresh = c - epsilon
        active_idx = np.where(O > thresh)[0]
        if len(active_idx) == 0:
            break
        
        grad = np.zeros(n)
        for k_idx in active_idx:
            s = k_idx - (n - 1)
            grad += grad_for_shift(h, n, s)
        grad /= len(active_idx)
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break

        h_try = project_feasible(h - eta * grad, n)
        c_try = compute_c5(h_try, n)
        if c_try < c:
            h = h_try
            eta = min(eta * 1.05, 0.1)
        else:
            eta *= 0.7
            if eta < 1e-8:
                break
    
    return best_h, best_c

def coord_descent(h, n, deadline, max_steps=1000000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    n_full = 2 * n - 1

        if time.time() > deadline:

        k_max = O.argmax()
        c = O[k_max]
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        
        s = k_max - (n - 1)
        g = grad_for_shift(h, n, s)
        
        # Try to move in the direction that reduces O(s)
        # For each j, the best move is to decrease h[j] if g[j] > 0, or increase if g[j] < 0
        # But we need to maintain the sum constraint
        
        # Simple approach: pick the j with the largest |g[j]| and move it
        # And compensate at another index
        j_max = np.argmax(np.abs(g))
        direction = -np.sign(g[j_max])  # Move h[j_max] to reduce O(s)
        
        # Find a compensation index
        if direction > 0:
            # h[j_max] increases, need to decrease some other h
            comp_mask = h > 0.01
            comp_mask[j_max] = False
            if not comp_mask.any():
                break
            j_comp = np.argmin(h[comp_mask])  # Decrease the smallest positive
            # Actually, pick the one where decreasing helps most
            # For simplicity, just pick the average
            j_comp = np.where(comp_mask)[0][np.argmin(h[np.where(comp_mask)[0]])]
        else:

            comp_mask[j_max] = False
            if not comp_mask.any():
                break
            j_comp = np.where(comp_mask)[0][np.argmax(h[np.where(comp_mask)[0]])]
        
        # Try a small move

            h_try = h.copy()
            h_try[j_max] += direction * delta
            h_try[j_comp] -= direction * delta
            if 0 <= h_try[j_max] <= 1 and 0 <= h_try[j_comp] <= 1:
                c_try = compute_c5(h_try, n)
                if c_try < c - 1e-15:
                    h = h_try
                    break
    
    return best_h, best_c

Actually, I realize there might be an issue with my subgradient approach. The subgradient of max(O(s)) is a single gradient (not a convex combination) at any point where the max is achieved at a unique point. When the max is achieved at multiple points, any convex combination is a valid subgradient.

But the issue is: the subgradient points in the direction that decreases O(s*) most rapidly, but it might increase O(s) for other s. The averaged subgradient (over the active set) is better because it tries to decrease all active O(s) simultaneously.

My current approach averages over the active set, which is correct.

v = beta * v_prev - eta * g

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)
