    a_max = a.max()
    return (a_max + np.log(np.exp(a - a_max).sum())) / alpha

def grad_smooth_max(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    a = alpha * corr * dx
    a_max = a.max()
    w = np.exp(a - a_max)
    p = w / w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[s:] += p[k] * (1.0 - x[:n - s])
            g[:n - s] -= p[k] * x[s:]
        else:
            ss = -s
            g[:n - ss] += p[k] * (1.0 - x[ss:])
            g[ss:] -= p[k] * x[:n - ss]
    return dx * g

def _pen_obj(x):
    n, alpha, mu, target = _opt_params['n'], _opt_params['alpha'], _opt_params['mu'], _opt_params['target']
    return obj_smooth_max(x, n, alpha) + (x.sum() - target) ** 2 / (2.0 * mu)

def _pen_grad(x):
    n, alpha, mu, target = _opt_params['n'], _opt_params['alpha'], _opt_params['mu'], _opt_params['target']
    return grad_smooth_max(x, n, alpha) + (x.sum() - target) / mu

def pen_optimize(h, n, alpha, mu, deadline, maxiter=500):
    _opt_params['n'] = n
    _opt_params['alpha'] = alpha
    _opt_params['mu'] = mu
    _opt_params['target'] = n / 2.0
    try:
        res = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_ls(h, n, deadline, max_sweeps=200):
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def anneal(h_init, n, deadline, seed=42, max_iters=5000000):
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = 0.008
    cool = 0.99992
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i, j = zeros[ai], ones[bi]
        if i == j:
            continue
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        delta[n - 1 + (j - i)] += 1.0
        delta[n - 1 + (i - j)] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp((cur_c - nc) / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)

    # Phase 1: Quick binary local search
    dl1 = t0 + 0.05 * budget_s
    hb, cb = swap_ls(h0, n, dl1, max_sweeps=150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: L-BFGS-B with alpha and mu schedules
    dl2 = t0 + 0.50 * budget_s
    h = best_h.copy()
    alpha_mu = [(50, 1.0), (200, 0.5), (1000, 0.2), (5000, 0.1),
                (20000, 0.05), (100000, 0.02), (500000, 0.01),
                (2000000, 0.005), (10000000, 0.002)]
    for alpha, mu in alpha_mu:
        if time.time() > dl2 - 5:
            break
        h = pen_optimize(h, n, alpha, mu, dl2 - 3, maxiter=400)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        hb2, cb2 = swap_ls(h, n, min(time.time() + 3, dl2 - 4), max_sweeps=40)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
            h = project_feasible(best_h + 0.05 * rng.randn(n), n)

    # Phase 3: Annealing from best
    dl3 = t0 + 0.70 * budget_s
    ha, ca = anneal(best_h, n, dl3, seed=seed, max_iters=5000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 4: L-BFGS-B from annealed result
    dl4 = t0 + 0.85 * budget_s
    h = best_h.copy()
    for alpha, mu in [(1000, 0.1), (10000, 0.05), (100000, 0.02),
                      (1000000, 0.01), (10000000, 0.005), (50000000, 0.002)]:
        if time.time() > dl4 - 3:
            break
        h = pen_optimize(h, n, alpha, mu, dl4 - 2, maxiter=300)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Try resolution 500
    dl5 = t0 + 0.96 * budget_s
    if time.time() < dl5 - 25:
        nh = 500
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        h500 = project_feasible(np.interp(xn, xo, best_h), nh)
        dlt = min(time.time() + 10, dl5 - 20)
        hb5, cb5 = swap_ls(h500, nh, dlt, max_sweeps=60)
        if cb5 < compute_c5(h500, nh):
            h500 = hb5.copy()
        for alpha, mu in [(5000, 0.1), (50000, 0.05), (500000, 0.02)]:
            if time.time() > dl5 - 12:
                break
            h500 = pen_optimize(h500, nh, alpha, mu, dl5 - 8, maxiter=200)
            c500 = compute_c5(h500, nh)
            if c500 < best_c:
                best_c, best_h, n = c500, h500.copy(), nh

    # Phase 6: Final polish with small noise restarts
    for trial in range(4):
        if time.time() > dl - 2:
            break
        h = project_feasible(best_h + 0.003 * rng.randn(n), n)
        for alpha, mu in [(5000000, 0.002), (50000000, 0.001)]:
            if time.time() > dl - 1:
                break
            h = pen_optimize(h, n, alpha, mu, dl - 0.5, maxiter=200)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
