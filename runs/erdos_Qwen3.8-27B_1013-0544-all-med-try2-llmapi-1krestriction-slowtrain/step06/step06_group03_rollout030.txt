    best_c = compute_c5(h, n)

    it = 0

        c_raw = np.correlate(h, 1 - h, mode='full')
        c_max = c_raw.max()
        # Find all active lags (within 1% of max)
        active = np.where(c_raw >= c_max * 0.99)[0]
        # Average gradient over active lags
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            for j in range(n):
                n_val = j - k
                if 0 <= n_val < n and 0 <= n_val + k < n:
                    grad[j] += 1.0
                if 0 <= j - k < n and 0 <= (j - k) + k < n:
                    grad[j] += -h[j - k]
                if 0 <= j < n and 0 <= j + k < n:
                    grad[j] += -h[j + k]
        grad /= max(1, len(active))
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        eta *= decay
        it += 1
    return best_h, best_c

        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:

        h[j] = hj_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    if mx > 500:
        return float(c.max() + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(c.max() + np.log(np.sum(np.exp(shifted))) / alpha)

def power_obj(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_pos = np.maximum(c, 0.0)
    return float((np.mean(c_pos ** p)) ** (1.0 / p))

def _c_sum(x, n):
    return float(x.sum() - n / 2.0)

def _c_sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_sum, 'jac': _c_sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_power(h, n, p, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_sum, 'jac': _c_sum_jac, 'args': (n,)}]
    try:
        res = minimize(power_obj, h, args=(n, p), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_cd(h, n, deadline, steps):
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_cd(h, n, deadline, steps, radius=50):
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def sa_opt(h, n, deadline, rng, t_hi=0.02, t_lo=0.0003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    cur_c = best_c
    total = int(max(200, (deadline - time.time()) * 600))
    for it in range(total):
        if time.time() > deadline:
            break
        frac = it / max(1, total)
        t = t_hi * (t_lo / t_hi) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.015 * rng.rand()
        s = 1.0 if rng.rand() < 0.5 else -1.0
        ni = cur[i] + s * d
        nj = cur[j] - s * d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        cur[i] = ni; cur[j] = nj
        nc = compute_c5(cur, n)
        delta = nc - cur_c
        if delta < 0.0 or rng.rand() < np.exp(-delta / max(t, 1e-12)):
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = cur.copy()
        else:
            cur[i] = ni - s * d
            cur[j] = nj + s * d
    return best_h, best_c

def build_start(n, kind, rng):
    if kind == 0:
        return np.full(n, 0.5)
    elif kind == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        return h
    elif kind == 2:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        return np.clip(h, 0, 1)
    elif kind == 3:
        h = rng.uniform(0.2, 0.8, n)
        return h
    else:
        h = np.zeros(n)
        h[::2] = 1.0
        return h

def optimize_candidate(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    for alpha in [3000, 15000, 60000]:
        if time.time() > deadline - 20:
            break
        ht = slsqp_opt(best_h, n, alpha, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:

    if time.time() < deadline - 10:

        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if n <= 180:
        if time.time() < deadline - 5:
            steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
            best_h, best_c = pair_cd(best_h, n, deadline - 2, steps)

            steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
            best_h, best_c = near_cd(best_h, n, deadline - 2, steps, radius=55)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
