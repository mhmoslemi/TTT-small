def compute_corr(h, n):

    dx = 2.0 / n
    # Use FFT for O(n log n)
    N = 2 * n

    Fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(Fh * np.conj(Fb), N)[:2*n-1]
    return corr * dx

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

import numpy as np
from scipy.optimize import minimize
import time

def corr_fft(h, n):
    N = 2 * n
    Fh = np.fft.rfft(h, N)
    Fb = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(Fh * np.conj(Fb), N)[:2*n-1]

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = corr_fft(h, n) * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c_raw = corr_fft(h, n)
    dx = 2.0 / n
    c = c_raw * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
        w /= w.sum()
    else:
        w = np.exp(sh)
        w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _pobj(x, n, alpha, mu):
    c = corr_fft(x, n) * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def _pgrad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

def lbfgs_opt(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50,1.0),(200,5.0),(1000,20.0),(5000,100.0),
               (20000,500.0),(100000,2000.0),(500000,10000.0),
               (2000000,50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_pobj, h, jac=_pgrad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter':400,'ftol':1e-18,'gtol':1e-14,
                                   'maxfun':4000,'maxls':40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgrad_momentum(h, n, deadline, lr_init=0.02, mom=0.85):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init

        k_star = int(np.argmax(c_full))
        k = k_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = (1.0 - h[k:n])

        else:
            kk = -k
            nv = n - kk
            g[kk:n] = (1.0 - h[0:nv])
            g[0:nv] += (-h[kk:n])
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
        lr *= 0.9995
        if lr < 1e-12:

    return best_h, best_c
