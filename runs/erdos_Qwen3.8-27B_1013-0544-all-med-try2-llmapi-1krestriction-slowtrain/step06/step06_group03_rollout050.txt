                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for alpha in [500, 2000, 10000, 50000, 200000, 500000]:
        if time.time() > deadline - 5:
            break
        ht = slsqp(h, n, alpha, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if time.time() < deadline - 10:
        for alpha in [10000, 50000, 200000]:
            if time.time() > deadline - 10:
                break
            ht, c = grad_desc(h, n, alpha, maxiter=300, lr_init=0.5)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
    if time.time() < deadline - 4:
        h, best_c = guided_pair_search(h, n, deadline - 3,
                                       [0.01, 0.005, 0.002, 0.001, 0.0005], 100000)
    if time.time() < deadline - 3:
        ht = slsqp(h, n, 2000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    if time.time() < deadline - 1:
        h, best_c = pair_search(h, n, deadline - 0.5,
                                [0.001, 0.0005, 0.0002, 0.0001])
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    x_init = np.linspace(0, 2, n_init)
    best_h = None
    best_c = float('inf')
    # Phase 1: Quick evaluation at multiple n
    for n_try in [100, 120, 150, n_init]:
        if n_try == n_init:
            h_try = h_init.copy()
        else:
            x_new = np.linspace(0, 2, n_try)
            h_try = np.interp(x_new, x_init, h_init)
        h_try = project_feasible(h_try, n_try)
        c_try = compute_c5(h_try, n_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    # Phase 2: Deep optimize at best n
    n_best = len(best_h)
    alloc2 = min(300, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_at_n(best_h, n_best, t0 + alloc2)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    # Phase 3: Try additional n values
    x_best = np.linspace(0, 2, len(best_h))
    for n_try in [80, 90, 110, 130]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_new = np.interp(x_new, x_best, best_h)
        h_new = project_feasible(h_new, n_try)
        alloc = min(80, (deadline - time.time()) * 0.3)
        if alloc < 20:
            break
        h_opt, c_opt = optimize_at_n(h_new, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 4: Multi-restart perturbation
    if time.time() < deadline - 60:
        n = len(best_h)
        for r in range(6):
            if time.time() > deadline - 25:
                break
            scale = 0.015 + 0.035 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time()) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 5: Final deep optimization
    if time.time() < deadline - 15:
        n = len(best_h)
        h_opt, c_opt = optimize_at_n(best_h, n, deadline - 3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 6: Ultra-fine pair search
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005])
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
