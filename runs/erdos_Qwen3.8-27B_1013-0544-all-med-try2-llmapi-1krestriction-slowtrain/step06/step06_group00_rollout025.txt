    # softmax over lags
    shifted = corr - corr.max()
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    # Only include lags with significant weight
    threshold = weights.max() * 1e-6
    active = weights > threshold
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        k = lags[idx]
        w = weights[idx]
        if abs(w) > 1e-12:
            grad += w * grad_for_lag(h, n, k)
    return grad

    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()

            s = room.sum()
            if s < 1e-14:

            h += diff / s * room
        else:

            if s < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += d/s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d/s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def corr_full(h, n):
    return np.correlate(h, 1.0-h, mode='full')

def grad_lag(h, n, k):
    g = np.zeros(n)
    if 0 <= k < n:
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    elif -n < k < 0:
        kp = -k
        g[kp:] += 1.0 - h[:n-kp]
        g[:n-kp] -= h[kp:]
    return g

def pgd_optimize(h, n, deadline, start_step=0.5, lags_top_k=5, restart_every=10000):
    """Fast projected subgradient descent."""
    dx = 2.0 / n
    all_lags = np.arange(-(n-1), n)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = start_step
    stall = 0
    rng = np.random.RandomState(123)
    
    for it in range(10000000):
        if time.time() > deadline:
            break
        corr = corr_full(h, n)
        c_val = corr.max() * dx
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Subgradient from top-k lags
        top = np.argpartition(corr, -lags_top_k)[-lags_top_k:]
        g = np.zeros(n)
        for idx in top:
            g += grad_lag(h, n, all_lags[idx])
        g /= lags_top_k
        
        # Step and project
        h_new = project_feasible(h - step * g, n)
        h = h_new
        
        # Adaptive step
        if stall > 100:
            step *= 0.95
        elif stall == 0:
            step = min(step * 1.02, 2.0)
        step = max(step, 0.01)
        
        # Periodic restart if stuck
        if restart_every and it > 0 and it % restart_every == 0 and stall > restart_every * 3:
            h = best_h + 0.02 * rng.randn(n)
            h = project_feasible(h, n)
            step = 0.3
            stall = 0
    
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = corr_full(h, n).astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n-1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        improved = False
        for i in zeros_list:
            if time.time() > deadline: break
            for j in ones_list:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved: break
        if not improved: break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = corr_full(x, n) * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n/2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':cons}],
                   options={'maxiter':maxiter,'ftol':ftol})
    return project_feasible(res.x, n) if res.success or hasattr(res, 'x') else h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, n, k):
    g = np.zeros(n)
    if 0 <= k < n:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    elif -n < k < 0:
        kp = -k
        g[kp:] += 1.0 - h[:n - kp]
        g[:n - kp] -= h[kp:]
    return g

def pgd_optimize(h, n, deadline, beta0=10.0, beta_rate=0.02, beta_max=5000.0,
                 step0=0.8, top_lags=20, restart_every=10000):
    all_lags = np.arange(-(n - 1), n)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step0
    stall = 0
    n_total = 2 * n - 1
    rng = np.random.RandomState(777)
    for it in range(10000000):
        if time.time() > deadline:
            break
        beta = min(beta0 * (1.0 + beta_rate * it), beta_max)
        corr = corr_full(h, n)
        c_val = corr.max() * (2.0 / n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        sh = beta * (corr - corr.max())
        w = np.exp(sh)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(n_total); w[np.argmax(corr)] = 1.0
        else:
            w /= ws
        tl = min(top_lags, n_total)
        top_idx = np.argpartition(w, -tl)[-tl:]
        g = np.zeros(n)
        for idx in top_idx:
            wi = w[idx]
            if wi > 1e-10:
                g += wi * grad_lag(h, n, all_lags[idx])
        h = project_feasible(h - step * g, n)
        if stall > 300:
            step *= 0.92
        elif stall == 0:
            step = min(step * 1.005, 1.5)
        if step < 0.003:
            step = 0.003
        if restart_every and it > 0 and it % restart_every == 0 and stall > restart_every * 2:
            h = best_h + 0.02 * rng.randn(n)
            h = project_feasible(h, n)
            step = 0.4
            stall = 0
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=400):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = corr_full(h, n).astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = corr_full(x, n) * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
