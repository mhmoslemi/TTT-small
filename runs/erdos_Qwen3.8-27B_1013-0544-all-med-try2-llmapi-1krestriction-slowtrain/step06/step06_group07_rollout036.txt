    n = len(h)
    dx = 2.0 / n

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:

# Top-level helpers
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

    return h.sum() - target_sum

        T = T_start * (T_end / T_start) ** frac
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0.0
        h[j] = 1.0
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_val = float(np.max(corr_new) * dx)
        
        if new_val < curr_val or (T > 1e-10 and rng.random() < np.exp(-(new_val - curr_val) / T)):
            curr_val = new_val
            corr_curr = corr_new
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SLSQP continuation
    betas = [10, 50, 200, 1000, 5000, 20000]
    for beta in betas:
        if remaining() < 80:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for trial in range(3):
            if remaining() < 25:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += rng.randn(n) * (0.02 / trial)
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    
    # Phase 2: Binary SA from multiple starts
    if remaining() > 120:
        starts = [best_h]
        # Thue-Morse start
        tm = thue_morse(n)
        # Adjust to have exactly n//2 ones
        if tm.sum() != n // 2:
            tm = round_to_binary(tm, n)
        starts.append(tm)
        # Random start
        rand_h = np.zeros(n)
        rand_idx = rng.choice(n, n // 2, replace=False)
        rand_h[rand_idx] = 1.0
        starts.append(rand_h)
        
        for start in starts:
            if remaining() < 60:
                break
            h_bin = round_to_binary(start, n)
            # Multiple SA rounds with decreasing temperature
            for round_num in range(3):
                if remaining() < 30:
                    break
                T_s = 0.3 / (round_num + 1)
                h_sa, c5_sa = binary_sa(h_bin, n, T_s, 0.001, 150000, rng, remaining)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
            h_bin = round_to_binary(best_h, n) if c5_sa < best_c5 else h_bin
            # Update h_bin for next round
            if c5_sa < best_c5:
                h_bin = best_h.copy()
    
    # Phase 3: SLSQP from binary/SA solution
    if remaining() > 80:
        for beta in [1000, 5000, 20000]:
            if remaining() < 30:
                break
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
            x0 = best_h.copy()
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    
    # Phase 4: Multi-resolution
    if remaining() > 100:
        for n_try in [48, 56, 64, 70, 80, 88, 96, 104, 112, 120]:
            if remaining() < 50:
                break
            if n_try == n:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
            for beta in [200, 2000, 10000]:
                if remaining() < 25:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                except Exception:
                    pass
    
    # Phase 5: Final binary SA polish
    if remaining() > 60:
        h_bin = round_to_binary(best_h, n)
        h_sa, c5_sa = binary_sa(h_bin, n, 0.1, 0.0005, 200000, rng, remaining)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
    
    # Phase 6: Final SLSQP polish
    if remaining() > 20:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for beta in [20000, 100000]:
            if remaining() < 10:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
