    target = n / 2.0
    lo, hi = -1.0, 1.0
    # Ensure the range is valid
    for _ in range(60):

        p = np.clip(h - mid, 0, 1)

            p = np.clip(p, 0, 1)
    return p

def compute_gradient(h, w, dx, n):
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx in range(len(lags)):
        wm = w[idx] * dx
        if wm < 1e-18:
            continue
        lag = lags[idx]
        # (1-h[j-lag]) term
        js = max(0, int(lag))
        je = min(n, n + int(lag))
        if je > js:
            grad[js:je] += wm * (1.0 - h[js-int(lag):je-int(lag)])

        je2 = min(n, n - int(lag))
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2+int(lag):je2+int(lag)]
    return grad

    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def compute_gradient_fft(h, w, dx, n):
    """Compute gradient using FFT-based convolution."""
    # w has length 2n-1 (the softmax weights for each lag)
    # grad[j] = dx * (1 - conv(w,h,full)[j+n-1] - corr(w,h,full)[n-1-j])
    conv_wh = np.convolve(w, h, mode='full')  # length 3n-2
    corr_wh = np.correlate(w, h, mode='full')  # length 3n-2
    # second term: conv_wh[j + n - 1] for j = 0, ..., n-1
    second = conv_wh[n-1 : n-1+n]  # indices n-1 to 2n-2
    # third term: corr_wh[n-1-j] for j = 0, ..., n-1
    third = corr_wh[n-1 : n-1+n][::-1]  # indices n-1 to 2n-2, reversed
    grad = dx * (1.0 - second - third)
    return grad
