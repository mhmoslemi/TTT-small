            mask = h < 1.0 - 1e-12
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            mask = h > 1e-12
            tot = h[mask].sum()
            if tot < 1e-15: break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def soft_max_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    return float(m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha)

def soft_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    exp_sh = np.exp(sh - sh_max)
    w = exp_sh / exp_sh.sum()
    
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        m_lag = n - 1 - k
        wk = w[k] * dx
        i_lo = max(0, -m_lag)
        i_hi = min(n, n - m_lag)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wk * (1 - x[i_lo+m_lag:i_hi+m_lag])
        i_lo2 = max(0, m_lag)
        i_hi2 = min(n, n + m_lag)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wk * x[i_lo2-m_lag:i_hi2-m_lag]
    return grad

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_analytical(h, n, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(soft_max_val, h.copy(), args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      jac=lambda x: soft_max_grad(x, n, alpha),
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_grad_descent(h, n, alpha, steps=5000, lr=None):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    if lr is None:
        lr = 1.0 / (n * 0.5)  # heuristic
    for s in range(steps):

            lr *= 0.5
        elif c < best_c - 1e-5 and lr < 1.0:
            lr *= 1.1
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
