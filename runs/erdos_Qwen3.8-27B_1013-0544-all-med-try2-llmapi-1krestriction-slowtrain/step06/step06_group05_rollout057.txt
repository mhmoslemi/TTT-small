def compute_grad_max(h, n, c):
    """Compute subgradient of max_k c[k] w.r.t. h"""
    m = c.max()
    S = np.where(np.isclose(c, m, atol=1e-10))[0]
    n = len(h)
    grad = np.zeros(n)
    for k in S:
        m_k = k - (n - 1)
        # h[i + m_k] term
        if m_k >= 0:
            grad[:n-m_k] += (1 - h[m_k:])
            # -h[i - m_k] term: i - m_k in [0,n-1] means i in [m_k, n-1]
            grad[m_k:] -= h[:n-m_k]
        elif m_k < 0:
            # i + m_k in [0,n-1] means i in [0, n-1+m_k] = [0, n+m_k-1]
            idx = range(0, n + m_k)
            grad[idx] += (1 - h[m_k + idx])  # h[i + m_k] for i in [0, n+m_k-1]
            # i - m_k in [0,n-1] means i in [0, n-1+m_k]... wait m_k < 0 so i - m_k = i + |m_k|
            # i + |m_k| in [0, n-1] means i in [0, n-1-|m_k|] = [0, n+m_k-1]
            # grad[i] -= h[i + |m_k|] = h[i - m_k]
            idx2 = range(0, n + m_k)
            grad[idx2] -= h[|m_k|:]  # h[i - m_k] for i in [0, n+m_k-1]
    grad /= len(S)
    return grad

def soft_max_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Soft max
    c_shifted = alpha * c - alpha * c.max()
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()  # weights
    logsumexp = c.max() + np.log(exp_c.sum()) / alpha
    
    # Gradient: sum_k w_k * dc_k/dh
    n = len(h)
    grad = np.zeros(n)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_val_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_c = np.exp(shifted)
    exp_sum = exp_c.sum()
    obj = c_max + np.log(exp_sum) / alpha
    w = exp_c / exp_sum
    grad = np.zeros(n)
    significant = w > 1e-12
    for k in np.where(significant)[0]:
        wk = w[k]
        m_k = k - (n - 1)
        if m_k >= 0:
            grad[:n-m_k] += wk * (1.0 - h[m_k:])
            grad[m_k:] -= wk * h[:n-m_k]
        else:
            a = -m_k
            grad[a:] += wk * (1.0 - h[:n-a])
            grad[:n-a] -= wk * h[a:]
    return obj, grad

def lbfgsb_optimize(h, n, alpha, mu, maxiter=200):
    h0 = h.copy()
    def obj_grad(x):
        o, g = soft_max_val_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        o += mu * s * s
        g = g + 2.0 * mu * s
        return o, g
    res = minimize(obj_grad, h0, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return project_feasible(res.x, n)

def subgradient_descent(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for t in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        S = np.where(c >= c_max - 1e-10)[0]
        # Average subgradient over active set
        grad = np.zeros(n)
        for k in S:
            m_k = k - (n - 1)
            if m_k >= 0:
                grad[:n-m_k] += (1.0 - h[m_k:])
                grad[m_k:] -= h[:n-m_k]
            else:
                a = -m_k
                grad[a:] += (1.0 - h[:n-a])
                grad[:n-a] -= h[a:]
        grad /= len(S)
        # Step size
        eta = 0.01 / (1.0 + 0.01 * t)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def pair_refine(h, n, deadline, max_iters=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)

            break
        i = rng.randint(n)

            continue

        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

            c = compute_c5(h, n)
            if c < best_c - 1e-15:

                best_h = h.copy()
            else:

                    else:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_val_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_c = np.exp(shifted)
    exp_sum = exp_c.sum()
    obj = c_max + np.log(exp_sum) / alpha
    w = exp_c / exp_sum
    grad = np.zeros(n)
    significant = w > 1e-12
    idxs = np.where(significant)[0]
    for k in idxs:
        wk = w[k]
        m_k = k - (n - 1)
        if m_k >= 0:
            grad[:n-m_k] += wk * (1.0 - h[m_k:])
            grad[m_k:] -= wk * h[:n-m_k]
        else:
            a = -m_k
            grad[a:] += wk * (1.0 - h[:n-a])
            grad[:n-a] -= wk * h[a:]
    return obj, grad

def active_subgrad(h, n, tol=1e-9):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    S = np.where(c >= c_max - tol)[0]
    grad = np.zeros(n)
    for k in S:
        m_k = k - (n - 1)
        if m_k >= 0:
            grad[:n-m_k] += (1.0 - h[m_k:])
            grad[m_k:] -= h[:n-m_k]
        else:
            a = -m_k
            grad[a:] += (1.0 - h[:n-a])
            grad[:n-a] -= h[a:]
    grad /= len(S)
    return grad, c_max

def lbfgsb_opt(h, n, alpha, mu, maxiter=300):
    def obj_grad(x):
        o, g = soft_max_val_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        o += mu * s * s
        g = g + 2.0 * mu * s
        return o, g
    try:
        res = minimize(obj_grad, h.copy(), jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def subgrad_refine(h, n, deadline, n_iter=3000, eta0=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(n_iter):
        if time.time() > deadline:
            break
        grad, c_max = active_subgrad(h, n, tol=1e-8)
        eta = eta0 / (1.0 + 0.02 * t)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if c_max - best_c < 1e-10 and t > 100:
            break
    return best_h, best_c

def pair_refine(h, n, deadline, max_iters=50000, seed=0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            best_h = h.copy()
            best_c = compute_c5(h, n)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if it % 20 == 0:
            d = rng.uniform(0.005, 0.03) * (1 if rng.rand() < 0.5 else -1)
        else:
            d = rng.uniform(0.0001, 0.003) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return best_h, best_c

    rng = np.random.RandomState(seed)

            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = grad[i] - grad[j]
        if abs(d) < 1e-15:
            continue
        step = 0.002
        if d > 0:
            hi = h[i] - step
            hj = h[j] + step
        else:
            hi = h[i] + step
            hj = h[j] - step
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj

            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
            else:

                h[i] = hi - (0.002 if d > 0 else -0.002)
                h[j] = hj + (0.002 if d > 0 else -0.002)
                h[i], h[j] = np.clip(h[i], 0, 1), np.clip(h[j], 0, 1)
    return best_h, best_c
