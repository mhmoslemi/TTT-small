One more thought: I should also try a "subgradient" approach where I accumulate gradients over multiple iterations and take a step in the averaged direction. This can help when the max lags keep switching.

Actually, let me think about this differently. The key insight is:

The objective is f(h) = max_k c[k]. This is the maximum of (2n-1) functions. At the optimum, typically several of these functions are active (achieve the maximum). The subgradient is the average of the gradients of the active functions.

So a good approach is:
1. Find all k where c[k] > f(h) - ε (the "active" set)
2. Compute the average gradient of the active functions
3. Take a step in the negative subgradient direction
4. Project to feasibility

This is the "subgradient method" and it's known to converge (albeit slowly) for convex problems. Our problem is non-convex, but locally it should work well.

Let me implement this:

```python
def subgradient_search(h, n, deadline, eps=0.005, step_scale=0.005):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_c(h, n)
    step = step_scale
    stall = 0
    rng = np.random.RandomState(77)
    
    for it in range(1000000):
        if time.time() > deadline:
            break
        
        f = c.max()
        if f < best_c - 1e-15:
            best_c = f
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            step *= 0.98
            stall = 0
        if step < 1e-10:
            break
        
        # Find active set
        active = np.where(c >= f - eps)[0]
        
        # Compute average gradient of active functions
        avg_grad = np.zeros(n)
        for k in active:
            avg_grad += grad_c(h, n, int(k))
        avg_grad /= len(active)
        
        # We want to decrease f, so move in -avg_grad direction
        # But we need a feasible direction (preserves sum)
        # Project onto the sum-zero subspace
        avg_grad -= avg_grad.mean()
        
        # Find the pair that best approximates -avg_grad
        # We want h[i] to increase where -avg_grad[i] is large, and h[j] to decrease where -avg_grad[j] is large
        neg_grad = -avg_grad
        i = int(np.argmax(neg_grad))
        j = int(np.argmin(neg_grad))
        
        if i == j:
            break
        
        d = min(step, h[i], 1.0 - h[j])
        if d < 1e-10:
            break
        
        h[i] += d
        h[j] -= d
        c = get_c(h, n)
        
        if c.max() > best_c + 0.01:
            # Too much increase, revert
            h[i] -= d
            h[j] += d
            step *= 0.9
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c(h, n, k):
    d = k - n + 1
    g = np.zeros(n)
    if d >= 0:
        g[:n - d] += 1.0 - h[d:]
        g[d:] -= h[:n - d]
    else:
        g[-d:] += 1.0 - h[:-d]
        g[:n + d] -= h[-d:]
    return g

def soft_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(soft_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, seed=42):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_c(h, n)
    rng = np.random.RandomState(seed)
    step = 0.008
    stall = 0
    
    for it in range(2000000):
        if time.time() > deadline:
            break
        
        f = c.max()
        if f < best_c - 1e-15:
            best_c = f
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            step *= 0.96
            stall = 0
        if step < 1e-10:
            break
        
        k_max = int(np.argmax(c))
        
        # Choose strategy
        r = rng.rand()
        if r < 0.5:
            # Direct: reduce max
            g = grad_c(h, n, k_max)
        elif r < 0.75:
            # Balance: reduce max, increase min
            k_min = int(np.argmin(c))
            g = grad_c(h, n, k_max) - 0.3 * grad_c(h, n, k_min)
        elif r < 0.9:
            # Top-3 active
            top3 = np.argsort(c)[-3:]
            g = np.zeros(n)
            for k in top3:
                g += grad_c(h, n, int(k))
            g /= 3.0
        else:
            # Random pair
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                d = min(step, h[i], 1.0 - h[j])
                if d > 1e-10:
                    h[i] += d
                    h[j] -= d
                    c = get_c(h, n)
                    if c.max() < best_c - 1e-15:
                        best_c = c.max()
                        best_h = h.copy()
                continue
        
        # Find best pair from gradient
        g_proj = g - g.mean()  # project to sum-zero
        i = int(np.argmax(g_proj))
        j = int(np.argmin(g_proj))
        if i == j:
            continue
        
        d = min(step, h[i], 1.0 - h[j])
        if d < 1e-10:
            continue
        
        h[i] += d
        h[j] -= d
        c = get_c(h, n)
        
        # Accept only if improvement
        if c.max() >= best_c + 1e-14:
            h[i] -= d
            h[j] += d
            # Try half step
            d *= 0.5
            if d > 1e-10:
                h[i] += d
                h[j] -= d
                c = get_c(h, n)
                if c.max() >= best_c + 1e-14:
                    h[i] -= d
                    h[j] += d
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n = 250
    
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: SLSQP (quick)
    for a in [2000, 20000, 200000, 2000000]:
        if time.time() > deadline - 500:
            break
        ht = slsqp_run(best_h, n, a, maxiter=3000)
        cc = compute_c5(ht, n)
        if cc < best_c:
            best_c = cc
            best_h = ht
    
    # Phase 2: Main local search
    if time.time() < deadline - 80:
        best_h, best_c = local_search(best_h, n, deadline - 80, seed=seed)
    
    # Phase 3: SLSQP polish
    if time.time() < deadline - 50:
        for a in [5000000, 20000000]:
            if time.time() > deadline - 20:
                break
            ht = slsqp_run(best_h, n, a, maxiter=4000)
            cc = compute_c5(ht, n)
            if cc < best_c:
                best_c = cc
                best_h = ht
    
    # Phase 4: Final fine search
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, deadline - 1, seed=seed + 123)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
