        if cur_h[i] + d < -1e-9 or cur_h[i] + d > 1 + 1e-9:
            continue
        if cur_h[j] - d < -1e-9 or cur_h[j] - d > 1 + 1e-9:
            continue
        cur_h[i] += d
        cur_h[j] -= d
        new_c = compute_c5(cur_h, n)
        diff = new_c - cur_c
        if diff < 0 orrng.random() < np.exp(-diff / (t + 1e-14)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] -= d
            cur_h[j] += d
    return best_h, best_c

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for a in [5000, 50000, 300000, 2000000, 10000000]:
        if time.time() > deadline - 30:
            break
        ht = slsqp(h, n, a, maxiter=4000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht
    if time.time() < deadline - 8:
        if n <= 160:
            h, best_c = pair_search(h, n, deadline - 4,
                                    [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            h, best_c = near_pair_search(h, n, deadline - 4,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], radius=60)
    if time.time() < deadline - 3:
        ht = slsqp(h, n, 20000000, maxiter=5000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial (n=250)
    alloc1 = min(250, (deadline - t0) * 0.3)
    h1, c1 = full_optimize(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h1, c1

    # Phase 2: Try n=100 (faster local search, different discretization)
    if time.time() < deadline - 100:
        x_old = np.linspace(0, 2, len(best_h))
        for n_try in [100, 150]:
            if time.time() > deadline - 60:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(100, deadline - time.time() - 30)
            if alloc < 40:
                break
            h_t, c_t = full_optimize(h_interp, n_try, time.time() + alloc, rng)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 3: Iterated local search with perturbation on best
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(4, int(remaining / 50))
        scales = [0.02, 0.04, 0.06, 0.08, 0.1, 0.12, 0.15, 0.2]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            sc = scales[r % len(scales)]
            hp = best_h + rng.normal(0, sc, n)
            hp = project_feasible(hp, n)
            alloc = min(55, deadline - time.time() - 15)
            if alloc < 20:
                break
            h_r, c_r = full_optimize(hp, n, time.time() + alloc, rng)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 4: SA on best
    if best_h is not None and time.time() < deadline - 25:
        n = len(best_h)
        h_sa, c_sa = sa_optimize(best_h, n, deadline - 20, rng, n_iter=5000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 5: Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp(best_h, n, 50000000, maxiter=6000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            if n <= 160:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.0005, 0.0002, 0.0001, 0.00005], radius=40)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
