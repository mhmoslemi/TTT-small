def project_binary(h, n):

    result = np.zeros(n)

    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0]

        # Try all swaps

            if time.time() > deadline:

Let me also think about the coordinate descent more carefully.

**Coordinate descent for this problem**:
At each step, pick a coordinate i and optimize h[i] while keeping others fixed.

The objective is f(h[i]) = max_k g_k(h[i]) where g_k is linear in h[i].
Each g_k(h[i]) = a_k * h[i] + b_k (for fixed h[-i]).

The max of linear functions is a convex PL function. Its minimum on [0,1] is at:
- 0 (if the function is decreasing on [0,1])
- 1 (if the function is increasing)
- A breakpoint (where two lines cross and both are on the upper envelope)

To find the minimum:
1. Compute g_k(0) and g_k(1) for all k
2. Find the upper envelope of the lines
3. The minimum of the upper envelope on [0,1] is the answer

This can be done in O(n log n) time per coordinate update (using a convex hull trick), but for simplicity, I can just evaluate at all breakpoints.

The breakpoints of the upper envelope are where pairs of lines cross. There are O(n²) pairs, so O(n²) breakpoints in the worst case. But in practice, many of them are not on the upper envelope.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = c.max()
    sh = alpha * (c - M) - alpha * c.max() * 0  # sh = alpha*(c - M)
    # For stability
    sh_max = sh.max()
    S = np.sum(np.exp(sh - sh_max))
    return M + (sh_max + np.log(S)) / alpha, c, M

def smax_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = c.max()
    sh = alpha * (c - M)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    S = w.sum()
    w /= S
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            nv = n - s
            grad[:nv] += wm * (1.0 - x[s:])
            grad[s:] += -wm * x[:nv]
        else:
            t = -s
            nv = n - t
            grad[t:] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[t:]
    return grad

def obj_pen(x, n, alpha, mu):
    val, c, M = smax(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_pen(x, n, alpha, mu):
    g = smax_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_slsqp(x, n, alpha):
    val, c, M = smax(x, n, alpha)
    return val

def constr_slsqp(x, n):
    return x.sum() - n / 2.0

def jac_slsqp(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, configs, max_iters=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_pen, h, jac=grad_pen, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c
