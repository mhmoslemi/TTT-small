```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def analytic_gradient(h, n, alpha):

    # s = k - n + 1

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_optimize(h, n, alpha, deadline, maxiter=5000):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        exp_g = np.exp(alpha * (g - m))
        return m + np.log(exp_g.sum()) / alpha
    
    def grad(x):
        grad, _ = analytic_gradient(x, n, alpha)
        return grad
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    # Use SLSQP with analytic gradient
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        # Try L-BFGS-B with penalty
        def obj_pen(x):
            base = obj(x)
            pen = 1e6 * (x.sum() - n/2.0)**2
            return base + pen
        def grad_pen(x):
            g = grad(x)
            return g + 2e6 * (x.sum() - n/2.0)
        try:
            res = minimize(obj_pen, h.copy(), jac=grad_pen, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'gtol': 1e-10})
            return project_feasible(res.x, n)
        except Exception:
            return h

def subgrad_polish_analytic(h, n, deadline, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        # Find active lags (at max)
        active = np.where(g >= m - 1e-10)[0]
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            s = k - n + 1
            if s >= 0:
                idx1 = np.arange(0, n - s)
                grad[idx1] += (1 - h[idx1 + s])
                idx2 = np.arange(s, n)
                grad[idx2] -= h[idx2 - s]
            else:
                s2 = -s
                idx1 = np.arange(s2, n)
                grad[idx1] += (1 - h[idx1 - s2])
                idx2 = np.arange(0, n - s2)
                grad[idx2] -= h[idx2 + s2]
        grad *= dx
        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break
        # Backtracking line search
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.8
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):
    target = int(n / 2)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        best_swap = None
        best_c_try = best_c
        # Try all swaps and pick the best
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c_try:
                    best_c_try = c
                    best_swap = (i, j)
                h_bin[i] = 0.0
                h_bin[j] = 1.0
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            ones.remove(j)
            ones.add(i)
            zeros.add(j)
            zeros.remove(i)
            best_c = best_c_try
            improved = True
        if not improved:
            break
    return h_bin, best_c
```

```python
def swap_delta(h, i, j, n):
    """Compute the change in c[k] for all k when h[i]:0->1, h[j]:1->0"""
    # s = k - n + 1, k = 0..2n-2, s = -(n-1)..(n-1)
    s = np.arange(-(n-1), n)
    
    # grad at m: (1-h[m+s])*[0<=m+s<n] - h[m-s]*[0<=m-s<n]
    # For m=i:
    v1 = (i + s >= 0) & (i + s < n)
    v2 = (i - s >= 0) & (i - s < n)
    # Need h[i+s] where valid, h[i-s] where valid
    # Use safe indexing
    idx1 = np.clip(i + s, 0, n-1)
    idx2 = np.clip(i - s, 0, n-1)
    grad_i = np.where(v1, 1 - h[idx1], 0) - np.where(v2, h[idx2], 0)
    
    idx1j = np.clip(j + s, 0, n-1)
    idx2j = np.clip(j - s, 0, n-1)
    v1j = (j + s >= 0) & (j + s < n)
    v2j = (j - s >= 0) & (j - s < n)
    grad_j = np.where(v1j, 1 - h[idx1j], 0) - np.where(v2j, h[idx2j], 0)
    
    return grad_i - grad_j
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytic_grad_smo(h, n, alpha):
    """Gradient of smooth approximation (1/a)*log(sum(exp(a*g))) + max(g)"""
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    ag = alpha * (g - m)
    e = np.exp(ag)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only process lags with significant weight
    thresh = w.max() * 1e-3
    active = np.where(w > thresh)[0]
    for k in active:
        s = k - n + 1
        if s >= 0:
            idx1 = np.arange(0, n - s)
            grad[idx1] += w[k] * (1 - h[idx1 + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= w[k] * h[idx2 - s]
        else:
            s2 = -s
            idx1 = np.arange(s2, n)
            grad[idx1] += w[k] * (1 - h[idx1 - s2])
            idx2 = np.arange(0, n - s2)
            grad[idx2] -= w[k] * h[idx2 + s2]
    return grad * dx

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        ag = alpha * (g - m)
        return m + np.log(np.exp(ag).sum()) / alpha
    
    def jac(x):
        return analytic_grad_smo(x, n, alpha)
    
    def con(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(obj, h.copy(), jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except:
        return h
```

  - As first factor: (1-h[i+s]) - this is correct, h[i+s] is unchanged
  - As second factor: -h[i-s] - this is correct, h[i-s] is unchanged
  But wait, if i-s = j (i.e., s = i-j), then h[i-s] = h[j] = 1 currently. After the swap, h[j] = 0. So the linear approximation would give -h[j] = -1, but the actual change would be -0 = 0. This is a second-order effect.

For binary functions, the exact change when swapping i:0→1 and j:1→0 is:
Δc[s] = [new c[s]] - [old c[s]]

Let me compute this exactly. The terms in c[s] that involve h[i] or h[j]:
- h[i] * (1-h[i+s]): was 0*(1-h[i+s]) = 0, now 1*(1-h[i+s])
- h[j] * (1-h[j+s]): was 1*(1-h[j+s]), now 0*(1-h[j+s]) = 0
- h[m] * (1-h[i]) where m+s=i: was h[m]*1 = h[m], now h[m]*0 = 0
- h[m] * (1-h[j]) where m+s=j: was h[m]*0 = 0, now h[m]*1 = h[m]

where we interpret h[x] = 0 if x is out of [0,n).

After: h[i]=1, h[i+s] = h[j] = 0 (after swap), so h[i]*(1-h[i+s]) = 1*1 = 1.

After swap: h_new[i] = 1, h_new[j] = 0, others unchanged.

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytic_grad_smo(h, n, alpha):
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    ag = alpha * (g - m)
    e = np.exp(ag)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    thresh = w.max() * 1e-3
    active = np.where(w > thresh)[0]
    for k in active:
        s = k - n + 1
        if s >= 0:
            idx1 = np.arange(0, n - s)
            grad[idx1] += w[k] * (1 - h[idx1 + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= w[k] * h[idx2 - s]
        else:
            s2 = -s
            idx1 = np.arange(s2, n)
            grad[idx1] += w[k] * (1 - h[idx1 - s2])
            idx2 = np.arange(0, n - s2)
            grad[idx2] -= w[k] * h[idx2 + s2]
    return grad * dx

def slsqp_analytic(h, n, alpha, maxiter):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        ag = alpha * (g - m)
        emax = ag.max()
        return m + (emax + np.log(np.exp(ag - emax).sum())) / alpha
    
    def jac(x):
        return analytic_grad_smo(x, n, alpha)
    
    def con(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(obj, h.copy(), jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 1.0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        active = np.where(g >= m - 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            s = k - n + 1
            if s >= 0:
                idx1 = np.arange(0, n - s)
                grad[idx1] += (1 - h[idx1 + s])
                idx2 = np.arange(s, n)
                grad[idx2] -= h[idx2 - s]
            else:
                s2 = -s
                idx1 = np.arange(s2, n)
                grad[idx1] += (1 - h[idx1 - s2])
                idx2 = np.arange(0, n - s2)
                grad[idx2] -= h[idx2 + s2]
        grad *= dx
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:

            ht = project_feasible(h - step * grad, n)

            stall += 1

            c = np.max(np.correlate(h, 1 - h, mode='full')) * dx
```

```python
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP analytic (20% budget)
    p1 = t0 + 0.20 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [50, 200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_analytic(h, n, alpha, 400)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Subgradient (10% budget)
    p2 = t0 + 0.30 * (budget_s - 3)
    h, c = subgrad_analytic(best_h.copy(), n, p2)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 3: Binary best-swap search (35% budget)
    p3 = t0 + 0.65 * (budget_s - 3)
    hb, cb = binary_search_best(best_h.copy(), n, p3, max_sweeps=10000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 4: Simulated annealing for binary (15% budget)
    p4 = t0 + 0.80 * (budget_s - 3)
    if time.time() < p4 - 10:
        hb2, cb2 = binary_sa(best_h.copy(), n, p4, rng, max_iters=300000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
        # Also try SA from random binary start
        hrand = np.zeros(n)
        idx = rng.permutation(n)
        hrand[idx[:n//2]] = 1.0
        hb3, cb3 = binary_sa(hrand, n, p4, rng, max_iters=100000)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
    # Phase 5: SLSQP polish on binary (20% budget)
    if time.time() < dl - 10:
        h = slsqp_analytic(best_h.copy(), n, 100000, 600)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 2:
        h, c = subgrad_analytic(best_h.copy(), n, dl)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
