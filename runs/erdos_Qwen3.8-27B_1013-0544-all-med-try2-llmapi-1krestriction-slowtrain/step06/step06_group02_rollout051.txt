import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

    return val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def fast_pair_delta(h, n, d, i, j):
    """Compute delta in c vector when h[i]+=d, h[j]-=d. O(n) time."""
    g = 1.0 - h
    dc = np.zeros(2 * n - 1)
    # Contribution from h[i] += d:
    # c[m] gets +d*g[m-i+n-1] (if 0<=m-i+n-1<n)
    lo = max(0, i - n + 1)
    hi = min(2 * n - 2, i)
    if hi >= lo:
        idx = np.arange(lo, hi + 1) - i + n - 1
        dc[lo:hi+1] += d * g[idx]
    # c[m] gets -d*h[i+m-n+1] (if 0<=i+m-n+1<n)
    lo2 = max(0, n - 1 - i)
    hi2 = min(2 * n - 2, n - 1)
    if hi2 >= lo2:
        idx2 = i + np.arange(lo2, hi2 + 1) - n + 1
        dc[lo2:hi2+1] -= d * h[idx2]
    # Contribution from h[j] -= d:
    lo3 = max(0, j - n + 1)
    hi3 = min(2 * n - 2, j)
    if hi3 >= lo3:
        idx3 = np.arange(lo3, hi3 + 1) - j + n - 1
        dc[lo3:hi3+1] -= d * g[idx3]
    lo4 = max(0, n - 1 - j)
    hi4 = min(2 * n - 2, n - 1)
    if hi4 >= lo4:
        idx4 = j + np.arange(lo4, hi4 + 1) - n + 1
        dc[lo4:hi4+1] += d * h[idx4]
    return dc

def fast_pair_eval(c_base, n, d, i, j):
    """Evaluate max of (c_base + delta) * dx after h[i]+=d, h[j]-=d."""
    dc = fast_pair_delta(None, n, d, i, j)
    c_new = c_base + dc
    return float(np.max(c_new))

def lbfgs_phase(h, n, deadline, alphas):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mus = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for idx, alpha in enumerate(alphas):
        if time.time() > deadline - 3:
            break
        mu = mus[min(idx, len(mus) - 1)]
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 4000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x, nn=n: x.sum() - nn / 2.0,
             'jac': lambda x, nn=n: np.ones(nn)}]
    try:
        res = minimize(smooth_max_obj, h,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 1500, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def rapid_pair_search(h, n, deadline, top_k=500):
    """Fast pair search using O(n) incremental correlation updates."""
    h = h.copy()
    best_c = compute_c5(h, n)
    c_base = compute_c5_vec(h, n)
    dx = 2.0 / n
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    alphas = [500, 2000, 10000, 50000]
    
    for d in deltas:
        while time.time() < deadline:
            # Compute gradient to guide pair selection
            alpha = alphas[min(deltas.index(d), len(alphas)-1)] if d in deltas else 50000
            g = smooth_max_grad(h, n, alpha, 0.0)
            
            # Score all pairs by gradient difference
            # We want to decrease h where grad is high, increase where grad is low
            grad_diff = g[None, :] - g[:, None]  # [i,j] = g[j]-g[i], positive means move i->j
            np.fill_diagonal(grad_diff, 0)
            
            # Get top_k pairs with largest positive gradient diff (move from i to j)
            flat = np.argsort(grad_diff.ravel())[::-1]
            flat = flat[:top_k]
            rows = flat // n
            cols = flat % n
            
            improved = False
            for idx in range(len(rows)):
                if time.time() > deadline:
                    break
                i, j = rows[idx], cols[idx]
                # Move from i to j: h[i]-=d, h[j]+=d
                if h[i] - d >= 0 and h[j] + d <= 1:
                    h[i] -= d; h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        c_base = compute_c5_vec(h, n)
                        improved = True
                    else:
                        h[i] += d; h[j] -= d
                # Move from j to i: h[j]-=d, h[i]+=d
                if h[j] - d >= 0 and h[i] + d <= 1:
                    h[j] -= d; h[i] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        c_base = compute_c5_vec(h, n)
                        improved = True
                    else:
                        h[j] += d; h[i] -= d
            if not improved:
                break
    return h, best_c
