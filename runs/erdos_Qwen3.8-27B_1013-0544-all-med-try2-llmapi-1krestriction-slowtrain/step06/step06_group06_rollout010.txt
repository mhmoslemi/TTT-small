import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_g(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    idx = np.argsort(h)[::-1]

        g = compute_all_g(h_bin, n)
        max_k = int(np.argmax(g))
        s = max_k - (n - 1)
        
        # For the max shift, compute which swaps would help
        # Score for flipping position i from 0 to 1 (at the max shift)
        # and position j from 1 to 0
        
        ones = list(np.where(h_bin == 1)[0])
        zeros = list(np.where(h_bin == 0)[0])
        
        best_gain = 0
        best_i = -1
        best_j = -1
        
        # Efficient: for the max shift s, compute the effect of each swap
        if s == 0:
            continue
        
        if s > 0:
            # g = sum_{i=0}^{n-s-1} h[i]*(1-h[i+s])
            # Flipping i from 0 to 1: gain = (1-h[i+s]) if i < n-s, else 0
            #   (but also affects other shifts)
            # Flipping j from 1 to 0: loss = (1-h[j+s]) if j < n-s, plus h[j-s] if j >= s
            # This is complex for multi-shift, so just do pairwise evaluation for top candidates
            
        # Simpler: just try random swaps
        rng = np.random.RandomState(sweep)
        n_tries = min(2000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h_bin[i], h_bin[j] = 1.0, 0.0
            c = compute_c5(h_bin, n)
            if c < best_c - 1e-14:
                best_c = c
                improved = True
                ones.remove(j)
                ones.append(i)
                zeros = [x for x in range(n) if h_bin[x] == 0]
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        
        if not improved:
            # Try deterministic: for each zero, try best one
            best_gain = -1
            best_pair = None
            for i in zeros:
                # Compute potential gain
                for j in ones[:min(50, len(ones))]:  # limit for speed
                    h_bin[i], h_bin[j] = 1.0, 0.0
                    c = compute_c5(h_bin, n)
                    gain = best_c - c
                    h_bin[i], h_bin[j] = 0.0, 1.0
                    if gain > best_gain:
                        best_gain = gain
                        best_pair = (i, j)
            if best_pair is not None and best_gain > 1e-14:
                i, j = best_pair
                h_bin[i], h_bin[j] = 1.0, 0.0
                best_c += best_gain
                ones.remove(j)
                ones.append(i)
                zeros = [x for x in range(n) if h_bin[x] == 0]
            else:
                break
    
    return h_bin, best_c

import numpy as np
from scipy.optimize import linprog, minimize
import time

_SLQP_N = 0
_SLQP_ALPHA = 1.0

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_g(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x):
    n = _SLQP_N
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    a = _SLQP_ALPHA
    sh = a * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / a

def _slsqp_cons(x):
    return x.sum() - _SLQP_N / 2.0

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    global _SLQP_N, _SLQP_ALPHA
    _SLQP_N = n
    _SLQP_ALPHA = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
