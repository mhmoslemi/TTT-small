import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def analytical_grad(h, n, k_star):
    """Compute gradient of g(k_star) w.r.t. h[i] for all i."""
    dx = 2.0 / n
    m = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        # h[i] appears as the "h" part when j=i, m_valid: i+m in [0,n-1]
        if 0 <= i + m < n:
            if m == 0:
                s += (1 - 2*h[i]) * dx  # d/dh[i] of h[i]*(1-h[i])*dx
            else:
                s += (1 - h[i+m]) * dx
        # h[i] appears as the "1-h" part when j+m=i, i.e., j=i-m
        if 0 <= i - m < n:
            s -= h[i-m] * dx
        grad[i] = s
    return grad

def subgrad_method(h, n, deadline, max_iter=10000):
    """Projected subgradient with analytical gradient and backtracking."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline: break
        g = compute_g(h, n)
        m = g.max()
        k_star = int(np.argmax(g))
        grad = analytical_grad(h, n, k_star)
        ng = np.linalg.norm(grad)
        if ng < 1e-15: break
        # Backtracking line search
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-12:
                h = ht
                lr = min(lr * 1.1, 10.0)
                found = True
                break
        if not found:
            lr *= 0.7
            if lr < 1e-12: break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        # Gradient of smooth max
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        # d/dx of sum_k weights_k * g(k)
        # g(k) = sum_j x[j]*(1-x[j+m])*dx where m = k-(n-1)
        # d/dx[i] g(k) = (1-x[i+m])*dx - x[i-m]*dx (with boundary handling)
        grad = np.zeros(n)
        m_vals = np.arange(2*n-1) - (n-1)
        for idx, mv in enumerate(m_vals):
            w = weights[idx]
            if w < 1e-15: continue
            for i in range(n):
                s = 0.0
                if 0 <= i + mv < n:
                    if mv == 0: s += (1 - 2*x[i]) * dx
                    else: s += (1 - x[i+mv]) * dx
                if 0 <= i - mv < n:
                    s -= x[i-mv] * dx
                grad[i] += w * s
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

def grad_for_lag(x, mv, dx):
    n = len(x)
    contrib = np.zeros(n)
    if mv == 0:
        contrib[:] = (1 - 2*x) * dx
    else:
        if mv > 0:
            # i+mv in [0,n-1] → i in [0, n-1-mv]
            contrib[:n-mv] += (1 - x[mv:]) * dx
            # i-mv in [0,n-1] → i in [mv, n-1]
            contrib[mv:] -= x[:n-mv] * dx
        else:
            mv_neg = -mv
            # i+mv in [0,n-1] → i+(-mv_neg) in [0,n-1] → i in [mv_neg, n-1]
            contrib[mv_neg:] += (1 - x[:n-mv_neg]) * dx
            # i-mv in [0,n-1] → i+mv_neg in [0,n-1] → i in [0, n-1-mv_neg]
            contrib[:n-mv_neg] -= x[mv_neg:] * dx
    return contrib
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(x, mv, dx):
    n = len(x)
    contrib = np.zeros(n)
    if mv == 0:
        contrib[:] = (1.0 - 2.0 * x) * dx
    elif mv > 0:
        contrib[:n - mv] += (1.0 - x[mv:]) * dx
        contrib[mv:] -= x[:n - mv] * dx
    else:
        mv_neg = -mv
        contrib[mv_neg:] += (1.0 - x[:n - mv_neg]) * dx
        contrib[:n - mv_neg] -= x[mv_neg:] * dx
    return contrib

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    sw = weights.sum()
    if sw < 1e-300:
        weights = np.zeros_like(g)
        weights[int(np.argmax(g))] = 1.0
    else:
        weights /= sw
    grad = np.zeros(n)
    m_vals = np.arange(2 * n - 1) - (n - 1)
    for idx in range(len(m_vals)):
        w = weights[idx]
        if w < 1e-12:
            continue
        grad += w * grad_for_lag(x, m_vals[idx], dx)
    return grad

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_method(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline: break
        g = compute_g(h, n)
        m = g.max()
        k_star = int(np.argmax(g))
        mv = k_star - (n - 1)
        dx = 2.0 / n
        grad = grad_for_lag(h, mv, dx)
        ng = np.linalg.norm(grad)
        if ng < 1e-15: break

        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)

            c = compute_c5(h, n)

                best_h = h.copy()

def binary_search_multiflip(h, n, deadline, max_sweeps=200):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1.0)[0])
    zeros = set(np.where(h_bin == 0.0)[0])
    ones_list = list(ones)
    zeros_list = list(zeros)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline: break
        improved = False
        # Single flip
        for i in zeros_list:
            if time.time() > deadline: break
            for j in ones_list:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.add(i); ones.remove(j)
                    zeros.remove(i); zeros.add(j)
                    ones_list = list(ones)
                    zeros_list = list(zeros)
                    improved = True
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try 2-for-2 flips
            ones_list = list(ones)
            zeros_list = list(zeros)
            for i in range(len(zeros_list)):
                if time.time() > deadline: break
                for j in range(i+1, len(zeros_list)):
                    if time.time() > deadline: break
                    ii, jj = zeros_list[i], zeros_list[j]
                    for k in range(len(ones_list)):
                        if time.time() > deadline: break
                        for l in range(k+1, len(ones_list)):
                            kk, ll = ones_list[k], ones_list[l]

    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)
    ones = list(np.where(hb == 1.0)[0])
    zeros = list(np.where(hb == 0.0)[0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng_s = np.random.RandomState(sweep * 7 + 3)
        rng_s.shuffle(ones)
        rng_s.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    jpos = ones.index(j)
                    ipos = zeros.index(i)
                    ones[jpos] = zeros[ipos]
                    zeros[ipos] = j
                    improved = True
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c

def coord_descent_polish(h, n, deadline, max_sweeps=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            best_val = old
            best_local_c = compute_c5(h, n)
            for candidate in [0.0, 1.0, 0.5, old * 0.5, old * 1.5]:
                candidate = max(0.0, min(1.0, candidate))
                if abs(candidate - old) < 1e-12:
                    continue
                h[i] = candidate
                ht = project_feasible(h, n)
                c = compute_c5(ht, n)
                if c < best_local_c:
                    best_local_c = c
                    best_val = candidate
            if abs(best_val - old) > 1e-12:
                h = project_feasible(h, n)
                h[i] = best_val
                h = project_feasible(h, n)
                improved = True
            else:
                h[i] = old
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
