```python
                g[t:] -= h[:n-t]           # -h[j-t] for j=t..n-1
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgradient(h, n, active_indices):
    g = np.zeros(n)
    for m in active_indices:
        s = m - (n - 1)
        if s >= 0:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
        else:
            t = -s
            g[:n - t] += (1.0 - h[t:])
            g[t:] -= h[:n - t]
    return g

def subgradient_optimize(h, n, deadline, top_k=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.05
    stall = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = float(corr.max() * 2.0 / n)
        
        if c5 < best_c - 1e-12:
            best_c = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            alpha *= 0.5
            stall = 0
            if alpha < 1e-6:
                break
        
        # Active set: top_k lags
        idx = np.argpartition(corr, -top_k)[-top_k:]
        g = compute_subgradient(h, n, idx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g /= g_norm
        
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new <= c5 + 1e-10:
            h = h_new
            alpha *= 1.02
        else:
            alpha *= 0.7
        
        if alpha < 1e-7:
            break
    
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_h, best_c = h, c
    return best_h, best_c

def _obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, deadline, alphas=None):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [500, 2000, 10000, 50000]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    
    for a in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_obj_smooth, h, args=(n, a), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-14})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
        except Exception:
            pass
    
    return h, best_c

def coord_descent(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    indices = np.arange(n)
    
    while time.time() < deadline:
        improved = False
        perm = indices.copy()
        rng.shuffle(perm)
        
        for i in perm:
            if time.time() > deadline:
                break
            # Pick a few partner indices
            partners = [perm[(i + 1) % n], perm[(i + n//3) % n], perm[(i + 2*n//3) % n]]
            for j in partners:
                if j == i:
                    continue
                for direction in [1, -1]:
                    # Quick feasibility check
                    t0_val = 0.005
                    if direction == 1:
                        if h[i] + t0_val > 1.0 or h[j] - t0_val < 0.0:
                            continue
                    else:
                        if h[i] - t0_val < 0.0 or h[j] + t0_val > 1.0:
                            continue
                    
                    # Test if direction helps
                    h[i] += t0_val * direction
                    h[j] -= t0_val * direction
                    c = compute_c5(h, n)
                    h[i] -= t0_val * direction
                    h[j] += t0_val * direction
                    
                    if c >= best_c:
                        continue
                    
                    # Binary search for max step
                    lo, hi = 0.0, 0.5
                    for _ in range(15):
                        mid = (lo + hi) / 2
                        t = mid * direction
                        if h[i] + t < 0 or h[i] + t > 1 or h[j] - t < 0 or h[j] - t > 1:
                            hi = mid
                            continue
                        h[i] += t
                        h[j] -= t
                        c = compute_c5(h, n)
                        h[i] -= t
                        h[j] += t
                        if c < best_c:
                            lo = mid
                        else:
                            hi = mid
                    
                    t = lo * direction
                    if abs(t) > 1e-10:
                        h[i] += t
                        h[j] -= t
                        best_c = compute_c5(h, n)
                        improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
```
