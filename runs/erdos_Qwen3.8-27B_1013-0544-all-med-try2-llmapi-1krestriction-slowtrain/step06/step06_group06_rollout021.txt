```python
            grad[:] -= h

        grad[s:] -= h[:n-s]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def grad_lag(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        s = -lag
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    return grad * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smoothed_max_gradient(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sg = alpha * (g - m)
    exp_sg = np.exp(sg)
    w = exp_sg / exp_sg.sum()
    grad = np.zeros(n)
    # Only compute for active lags
    active = np.where(w > 1e-8)[0]
    for k in active:
        lag = k - (n - 1)
        grad += w[k] * grad_lag(h, n, lag)
    return grad, g, m

def projected_gradient(h, n, alpha, n_iters, step_size, decay, momentum, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    step = step_size
    for it in range(n_iters):
        if time.time() > deadline:
            break
        grad, g, m = smoothed_max_gradient(h, n, alpha)
        v = momentum * v - step * grad
        h_new = project(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        step *= decay
    return best_h, best_c

def binary_local_search_fast(h, n, deadline, max_sweeps=100):
    # Make binary
    idx = np.argsort(h)[::-1]
    target = int(n / 2)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    g = compute_all_g(h_bin, n)
    max_k = int(np.argmax(g))
    
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Recompute g and max
        g = compute_all_g(h_bin, n)
        max_k = int(np.argmax(g))
        max_g = g[max_k]
        lag = max_k - (n - 1)
        
        # Try all swaps: move a 1 to a 0
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):
                # Compute change in g[max_k]
                # Δg = dx * [h[i-lag] + h[i+lag] - h[j-lag] - h[j+lag]]
                # with boundary handling (out of range = 0)
                def get(idx):
                    return h_bin[idx] if 0 <= idx < n else 0.0
                delta = dx * (get(i-lag) + get(i+lag) - get(j-lag) - get(j+lag))
                if max_g + delta < best_c - 1e-12:
                    # Accept
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
                    ones.remove(j)
                    ones.add(i)
                    zeros.remove(i)
                    zeros.add(j)
                    best_c = max_g + delta
                    improved = True
                    break  # restart sweep
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
```

            if s > 1e-12:
                h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sg = alpha * (g - m)
    ex = np.exp(sg)
    w = ex / ex.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-9)[0]
    for k in active:
        lag = k - (n - 1)
        grad += w[k] * grad_lag(h, n, lag)
    return grad, m

def proj_grad_descent(h, n, deadline, alpha=1000, n_max=5000,
                      step0=0.04, momentum=0.9, decay=0.9995):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    step = step0
    for it in range(n_max):
        if time.time() > deadline:
            break
        grad, m = smooth_max_grad(h, n, alpha)
        v = momentum * v - step * grad
        h_new = project(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        if it % 50 == 0:
            step *= decay
    return best_h, best_c

def exact_subgrad(h, n, deadline, n_max=20000, step0=0.02, momentum=0.85):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    step = step0
    for it in range(n_max):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        k_star = int(np.argmax(g))
        lag = k_star - (n - 1)
        grad = grad_lag(h, n, lag)
        v = momentum * v - step * grad
        h_new = project(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        step *= 0.9999
    return best_h, best_c

def binary_ls(h, n, deadline, max_sweeps=800):
    idx = np.argsort(h)[::-1]
    target = n // 2
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_all_g(h_bin, n)
        max_k = int(np.argmax(g))
        max_val = g[max_k]
        lag = max_k - (n - 1)
        ones = np.where(h_bin == 1)[0].tolist()
        zeros = np.where(h_bin == 0)[0].tolist()
        best_delta = 0.0
        best_pair = None
        for i in zeros:
            vi = h_bin[i - lag] if 0 <= i - lag < n else 0.0
            vj = h_bin[i + lag] if 0 <= i + lag < n else 0.0
            for j in ones:
                wj = h_bin[j - lag] if 0 <= j - lag < n else 0.0
                wj2 = h_bin[j + lag] if 0 <= j + lag < n else 0.0
                delta = dx * (wj + wj2 - vi - vj)
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        improved = False
        if best_pair is not None and best_delta < -1e-12:
            i, j = best_pair
            h_bin[i] = 1.0
