```
def fista_optimize(h0, n, alpha, max_iter, deadline):
    # FISTA (Nesterov accelerated gradient)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    L = 4.0  # Lipschitz constant estimate (needs to be bounded)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Gradient step
        grad = soft_max_grad(y, n, alpha)
        z = y - grad / L
        # Projection
        x_new = project_feasible(z, n)
        # Update t
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = x_new + ((t - 1) / t_new) * (x_new - x)
        y = project_feasible(y, n)
        t = t_new
        x = x_new
    
    return x
```

```python
import numpy as np
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_grad(h, n, alpha, topk=15):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    if topk < len(g):
        top_idx = np.argpartition(g, -topk)[-topk:]
    else:
        top_idx = np.arange(len(g))
    shifted = alpha * (g[top_idx] - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    grad = np.zeros(n)
    nr = np.arange(n)
    for ii, idx in enumerate(top_idx):
        s = int(idx - (n - 1))
        ws = w[ii]
        if ws < 1e-12:
            continue
        ip = nr + s
        im = nr - s
        m1 = (ip >= 0) & (ip < n)
        m2 = (im >= 0) & (im < n)
        fv = np.ones(n)
        fv[m1] = 1.0 - h[ip[m1]]
        sv = np.zeros(n)
        sv[m2] = h[im[m2]]
        grad += ws * (fv - sv)
    return g.max() * dx, grad
```
