    e = np.exp(shifted)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_and_weights(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + np.log(np.exp(alpha * (c - m) - (alpha * (c - m)).max()).sum()) / alpha
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    return float(val), c, w

def compute_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    w_pos = w[n-1:2*n-1]
    one_minus_h = 1.0 - h
    part1 = np.convolve(w_pos, one_minus_h, mode='full')[:n]
    part2 = np.correlate(h, w_pos, mode='full')[n-1:2*n]
    return part1 - part2

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_smooth(h, n, alpha, maxiter=500, ftol=1e-16, deadline=None):
    dx = 2.0 / n
    def obj(x):
        if deadline is not None and time.time() > deadline:
            return np.inf
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        shifted -= shifted.max()
        val = m + np.log(np.exp(shifted).sum()) / alpha
        return float(val)
    def grad(x):
        if deadline is not None and time.time() > deadline:
            return np.zeros(n)
        return compute_gradient(x, n, alpha) * dx
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h

def anneal_binary(h_init, n, deadline, seed=42, n_iters=5000000, T0=0.01, cool=0.99997):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            T *= cool
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        if i == j:
            continue
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        d = j - i
        k1 = n - 1 + d
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n - 1 - d
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp((cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
