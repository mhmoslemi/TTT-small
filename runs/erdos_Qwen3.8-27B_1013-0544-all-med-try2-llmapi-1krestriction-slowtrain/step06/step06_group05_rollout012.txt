```python
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_full(h, n):
    """Compute all correlation values."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def subgradient_step(h, n, k_indices, step):
    """Take a subgradient step using gradients from specified k indices."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in k_indices:
        r = k - (n - 1)
        if r >= 0:
            # (1-h[j+r]) for j in [0, n-1-r]
            js = np.arange(0, n - r)
            grad[js] += (1 - h[js + r])
            if r > 0:
                js2 = np.arange(r, n)
                grad[js2] += -h[js2 - r]
        else:
            s = -r
            js = np.arange(s, n)
            grad[js] += (1 - h[js - s])
            js2 = np.arange(0, n - s)
            grad[js2] += -h[js2 + s]
    grad *= dx / len(k_indices)
    h_new = h - step * grad
    return h_new

        # Get top-k indices
        k_indices = np.argsort(c)[-top_k:]
        # Also always include the argmax

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

            h = h_new

        else:
```

```python
def sg_direction(h, n, k_indices):
    """Subgradient direction using multiple active lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in k_indices:
        r = k - (n - 1)
        if r >= 0:
            i1 = np.arange(0, n - r)

            grad[i1] += (1.0 - h[i1 - s])
            i2 = np.arange(0, n - s)
```

Actually wait, for each pair (i,j), I'm computing the full correlation every time. That's the bottleneck. Can I do better?

**Optimization**: Instead of recomputing the full correlation after each move, I can compute the CHANGE in the correlation due to the move.

When I change h[i] by δ and h[j] by -δ:
- The change in c[k] = Σ h[m](1-h[m+r]) where r = k-(n-1)
- = δ·(1-h[i+r]) - δ·h[j-r] (approximately, considering both positions)
- Plus second-order terms (which are O(δ²) and negligible for small δ)

So I can estimate the change in c[k] in O(1) per k, or O(n) for all k. This is much faster than O(n²)!

```python
def c5_change(h, n, i, j, di):
    """Estimate change in max correlation when h[i]+=di, h[j]-=di."""
    dx = 2.0 / n
    # For each lag k (shift r), the change in c[k] is:
    # di * (1 - h[i+r]) - di * h[j+r] (if i+r and j+r are valid)
    # - di * h[i-r] + di * h[j-r] (if i-r and j-r are valid)
    # Actually let me think more carefully.
    
    # c[k] = sum_m h[m](1-h[m+r]) for valid m
    # After changing h[i] += di and h[j] -= di:
    # Change in c[k]:
    #   From h[i] as first factor: di * (1-h[i+r]) if i+r valid
    #   From h[i] as second factor: -h[i-r] * di if i-r valid (h[i-r] is first factor, h[i] is second)
    #   Wait, h[m](1-h[m+r]): if m+r = i, i.e., m = i-r, then the term is h[i-r](1-h[i])
    #   Change: -h[i-r] * di
    #   Similarly for h[j]:
    #   From h[j] as first factor: -di * (1-h[j+r]) if j+r valid
    #   From h[j] as second factor: -h[j-r] * (-di) = h[j-r]*di if j-r valid
    
    # Total change in c[k]:
    # di * (1-h[i+r])·1[i+r valid] - h[i-r]·di·1[i-r valid]
    # - di * (1-h[j+r])·1[j+r valid] + h[j-r]·di·1[j-r valid]
    
    # For the max, we estimate: new_max ≈ old_max + max_k (change in c[k])
    # But this is an upper bound. For a better estimate, we check the current argmax.
    
    # Simpler: just compute the change at the current argmax
    c = c5_all(h, n)
    k_max = np.argmax(c)
    r = k_max - (n-1)
    
    delta = 0
    # i as first factor
    if 0 <= i+r < n:
        delta += di * (1 - h[i+r])
    # i as second factor (m = i-r)
    if 0 <= i-r < n:
        delta -= h[i-r] * di
    # j as first factor  
    if 0 <= j+r < n:
        delta -= di * (1 - h[j+r])
    # j as second factor (m = j-r)
    if 0 <= j-r < n:
        delta += h[j-r] * di
    
    return c[k_max] * dx + delta * dx
```

```python
                        new_c = c.copy()

                            # c[k] where k = r + (n-1)
                            # h[i] contributes: di*(1-h[i+r]) if 0<=i+r<n
                            # h[i] as second: -h[i-r]*di if 0<=i-r<n
                            # h[j] contributes: -di*(1-h[j+r]) if 0<=j+r<n
                            # h[j] as second: h[j-r]*di if 0<=j-r<n
                            delta = 0
                            if 0 <= i+r < n:
                                delta += di * (1 - h[i+r])
                            if 0 <= i-r < n:
                                delta -= h[i-r] * di
                            if 0 <= j+r < n:
                                delta -= di * (1 - h[j+r])
                            if 0 <= j-r < n:
                                delta += h[j-r] * di
                            new_c[r + (n-1)] += delta
                        new_max = float(np.max(new_c) * dx)
                        if new_max < best_c - 1e-15:
                            h[i], h[j] = hi, hj
                            best_c = new_max
                            c = new_c  # Update
                            improved = True
                        # Note: we don't undo the move since we already have new_c
            if not improved: break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def sg_direction(h, n, k_indices):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in k_indices:
        r = k - (n - 1)
        if r >= 0:
            i1 = np.arange(0, n - r)
            grad[i1] += (1.0 - h[i1 + r])
            if r > 0:
                i2 = np.arange(r, n)
                grad[i2] -= h[i2 - r]
        else:
            s = -r
            i1 = np.arange(s, n)
            grad[i1] += (1.0 - h[i1 - s])
            i2 = np.arange(0, n - s)
            grad[i2] -= h[i2 + s]
    return grad * dx / len(k_indices)

def subgradient_opt(h, n, deadline, top_k=5, momentum=0.85):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    vel = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        c = c5_all(h, n)
        k_idx = np.argsort(c)[-top_k:]
        d = sg_direction(h, n, k_idx)
        vel = momentum * vel - step * d
        h_new = project_feasible(h + vel, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        elif c_new < compute_c5(h, n):
            h = h_new
            stall += 1
        else:
            h = best_h + 0.005 * np.random.randn(n)
            h = project_feasible(h, n)
            step *= 0.8
            vel *= 0.5
            stall += 1
        if stall > 300:
            break
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        for _pass in range(2000):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for sgn in (1, -1):
                        a, b = h[i] + sgn * d, h[j] - sgn * d
                        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
                            h[i], h[j] = a, b
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = a - sgn * d, b + sgn * d
            if not improved:
                break
    return h, best_c

def try_resolution(h_src, n_src, n_tgt, deadline):
    if n_tgt == n_src:
        return h_src.copy(), compute_c5(h_src, n_src)
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_tgt)
    h = np.interp(x_tgt, x_src, h_src)
    h = project_feasible(h, n_tgt)
    t_sub = min(20, max(5, (deadline - time.time()) * 0.3))
    h, c = subgradient_opt(h, n_tgt, time.time() + t_sub, top_k=5)
    h = slsqp(h, n_tgt, 50000, maxiter=1500)
    c = compute_c5(h, n_tgt)
    return h, c
```
