```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_step(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
```

```python
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        # Analytical gradient
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exp_shifted = np.exp(shifted - sm)
        weights = exp_shifted / exp_shifted.sum()  # softmax weights, length 2n-1
        
        # Gradient of g = correlate(x, 1-x, 'full') * dx w.r.t. x
        # g[m] = dx * sum_i x[i] * (1-x[i+m-(n-1)])
        # dg/dx[i] = dx * [(1-x[i+m-(n-1)]) - x[i-m+(n-1)-2*(n-1)+...]]
        # More carefully:
        # g[m] = dx * sum_j x[j] * (1-x[j+lag]) where lag = m-(n-1)
        # dg/dx[i] = dx * [(1-x[i+lag]) * I(valid) - x[i-lag] * I(valid')]
        
        # Vectorized: for each m, the contribution is weights[m] * dg[m]/dx[i]
        # This is O(n * (2n-1)) which is O(n^2)
        
        # Alternative: use the structure
        # dg/dx[i] = dx * sum_m weights[m] * [(1-x[i+m-(n-1)]) * I(0<=i+m-(n-1)<n) - x[i-m+(n-1)] * I(0<=i-m+(n-1)<n)]
        
        # Let's compute this using convolutions
        # First term: sum_m weights[m] * (1-x[i+m-(n-1)]) * I(valid)
        # = sum_k weights[k+(n-1)] * (1-x[i+k]) * I(0<=i+k<n) for k = -(n-1)..(n-1)
        
        w = weights  # length 2n-1, index m corresponds to lag m-(n-1)
        
        # For position i, the first term sum:
        # sum over valid k where 0 <= i+k < n: w[k+(n-1)] * (1-x[i+k])
        # This is a convolution of w with (1-x)
        # Let's use: for each i, sum_k w[k+(n-1)] * (1-x[i+k]) where k ranges such that i+k in [0,n)
        # = weighted sum of (1-x) centered at i with weights w
        
        # Second term: sum_m weights[m] * x[i-(m-(n-1))] * I(0<=i-(m-(n-1))<n)
        # = sum_k w[k+(n-1)] * x[i-k] where i-k in [0,n)
        # = weighted sum of x centered at i with weights w (but reversed)
        
        # Actually, let me just do it the naive O(n^2) way for n=400
        grad = np.zeros(n)
        for i in range(n):
            # First part: contribution from x[i] being in the first factor
            # For each lag k (where k = m-(n-1)), x[i] appears in the first factor
            # if 0 <= i < n and 0 <= i+k < n
            # contribution: w[k+(n-1)] * (1-x[i+k])
            # Second part: x[i] appears in the second factor (as 1-x[i+k] where i+k = i means k=0... no)
            # x[i] appears in (1-x[j+k]) when j+k = i, i.e., j = i-k
            # contribution: -w[k+(n-1)] * x[i-k] (if 0<=i-k<n and 0<=i<n)
            
            # Combine: for each valid k:
            val = 0.0
            for k in range(-(n-1), n):
                m = k + (n-1)
                wk = w[m]
                if wk < 1e-18:
                    continue
                # First factor: x[i] * (1-x[i+k])
                if 0 <= i+k < n:
                    val += wk * (1.0 - x[i+k])
                # Second factor: x[j] * (1-x[j+k]) where j+k = i => j = i-k
                # -x[j+k] = -x[i], so contribution to dx[i] is -x[j] * w[m]
                if 0 <= i-k < n:
                    val -= wk * x[i-k]
            grad[i] = dx * val
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        # Fall back to no gradient
        try:
            def obj2(x):
                g = np.correlate(x, 1.0 - x, mode='full') * dx
                m = g.max()
                shifted = alpha * (g - m)
                sm = shifted.max()
                return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
            bounds2 = [(0.0, 1.0)] * n
            cons2 = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
            res = minimize(obj2, h, method='SLSQP', bounds=bounds2,
                           constraints=cons2, options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def softmax_obj_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    p = exp_s / exp_s.sum()  # length 2n-1
    obj_val = m + (sm + np.log(exp_s.sum())) / alpha
    
    # Gradient
    j_i = np.arange(n)[None, :] - np.arange(n)[:, None]
    valid = j_i >= 0
    M_idx = n - 1 + np.where(valid, j_i, 0)
    M_vals = p[M_idx] * valid
    A = 1.0 - x
    grad = dx * (M_vals @ A - M_vals.T @ x)
    return obj_val, grad

def slsqp_smooth(h, n, alpha, maxiter, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_g(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exp_s = np.exp(shifted - sm)
        p = exp_s / exp_s.sum()
        j_i = np.arange(n)[None, :] - np.arange(n)[:, None]
        valid = j_i >= 0
        M_idx = n - 1 + np.where(valid, j_i, 0)
        M_vals = p[M_idx] * valid
        return dx * (M_vals @ (1.0 - x) - M_vals.T @ x)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_g,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(h_init, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                h[i], h[j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):

    m_range = np.arange(-(n - 1), n)

    best_c = cur_c

    zeros_arr = np.where(h == 0)[0].copy()

            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
```
