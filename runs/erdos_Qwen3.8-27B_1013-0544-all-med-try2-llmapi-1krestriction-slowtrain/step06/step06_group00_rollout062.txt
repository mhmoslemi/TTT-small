            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:

            h += diff / s * h

def swap_delta(h, i, j, n, corr):
    """Compute delta in correlation when swapping h[i]: 0->1 and h[j]: 1->0"""
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)

    delta[n - 1] -= 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def swap_delta(h, i, j, n):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_search(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi, i in enumerate(zeros_list):
            if time.time() > deadline:
                break
            for oi, j in enumerate(ones_list):
                if time.time() > deadline:
                    break
                delta = swap_delta(h, i, j, n)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list[oi] = i
                    zeros_list[zi] = j
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def proj_grad_lse(h, n, dx, alpha, deadline, step=0.15, patience=5000):
    h = h.copy()
    best_local = h.copy()
    best_local_f = compute_c5(h, n)
    no_improve = 0
    cur_step = step
    while time.time() < deadline:
        C = np.correlate(h, 1.0 - h, mode='full')
        g = C * dx
        m = g.max()
        s = alpha * (g - m)
        s_max = s.max()
        s -= s_max
        w = np.exp(s)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum
        W_full = w
        first_sum = np.convolve(1.0 - h, W_full, mode='full')[n - 1:2 * n - 1]
        second_sum = np.correlate(W_full, h, mode='full')[n - 1:2 * n - 1][::-1]
        grad = dx * (first_sum - second_sum)
        h_new = project_feasible(h - cur_step * grad, n)
        c = compute_c5(h_new, n)
        if c < best_local_f - 1e-16:
            best_local_f = c
            best_local = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            h = h_new
            no_improve += 1
            if no_improve > patience:
                h = best_local.copy()
                cur_step *= 0.7
                no_improve = 0
                if cur_step < 1e-9:
                    break
    return best_local

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.025
    cool = 0.99997
    scale = 5000.0
    while time.time() < deadline:
        batch = 3000
        for _ in range(batch):
            if time.time() > deadline:
                break
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = int(zeros_arr[ai])
            j = int(ones_arr[bi])
            delta = swap_delta(h, i, j, n)
            nc = (corr + delta).max() * dx
            accept = nc <= cur_c
            if not accept and T > 1e-10:
                ratio = min(0.0, scale * (cur_c - nc) / T)
                accept = rng.random() < np.exp(ratio)
            if accept:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool
    return best_h, best_c
