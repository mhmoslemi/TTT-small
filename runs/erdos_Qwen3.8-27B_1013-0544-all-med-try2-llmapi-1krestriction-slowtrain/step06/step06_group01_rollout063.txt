import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def make_binary(h, n, seed=42):
    """Convert h to binary with exactly n/2 ones"""
    target = n // 2
    # Sort by value, take top target
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_delta(h, i, j, n, m_range, corr):
    """Compute new max correlation after swapping h[i]=0->1, h[j]=1->0
    Returns the new max value (not multiplied by dx)"""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    
    # Cross terms
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2*n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n - 1:
        delta[k2] += 1.0
    
    return np.max(corr + delta)

    alpha = np.zeros(2*n-1)
