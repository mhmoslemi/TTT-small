def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        sm = max_s + np.log(np.exp(shifted - max_s).sum())
        return m + sm / alpha
    def grad(x):
        m, g = softmax_gradient(x, n, alpha)
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgsb_optimize(h, n, alpha, maxiter, ftol, penalty_mu=1000.0):
    """L-BFGS-B with analytic gradient and penalty for constraint."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        sm = max_s + np.log(np.exp(shifted - max_s).sum())
        penalty = penalty_mu * (x.sum() - target) ** 2
        return m + sm / alpha + penalty
    
    def grad(x):
        m, g = softmax_gradient(x, n, alpha)
        g += 2 * penalty_mu * (x.sum() - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8,
                                'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def softmax_value_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    shifted = alpha * (corr - m)
    ms = shifted.max()
    exp_s = np.exp(shifted - ms)
    p = exp_s / exp_s.sum()
    sm = ms + np.log(exp_s.sum())
    # Gradient via convolution
    c1 = np.convolve(p, 1 - h, 'full')
    c2 = np.convolve(p[::-1], h, 'full')
    idx = np.arange(n) + (n - 1)
    grad = (c1[idx] - c2[idx]) * dx
    val = (m + sm / alpha) * dx
    return val, grad

                k1 = n - 1 + (j - i)
                if k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=8000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.02
    cool = 0.99993
    min_T = 1e-9
    scale = 8000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        if dc <= 0 or (T > min_T and rng.random() < np.exp(-scale * dc / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T = max(min_T, T * cool)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick greedy
    p1 = t0 + 0.05 * total
    hb, cb = greedy_binary(best_h, n, p1, max_sweeps=150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: L-BFGS-B main (fastest optimizer with analytic grad)
    p2 = t0 + 0.40 * total
    h = best_h.copy()
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    mus = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for i in range(len(alphas)):
        if time.time() > p2 - 3:
            break
        h = lbfgsb_opt(h, n, alphas[i], 4000, mus[i])
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: SLSQP refinement
    p3 = t0 + 0.52 * total
    h = best_h.copy()
    for alpha in [50000, 500000, 5000000]:
        if time.time() > p3 - 3:
            break
        h = slsqp_opt(h, n, alpha, 2500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Annealing for better binary
    p4 = t0 + 0.72 * total
    ha, ca = anneal_binary(best_h, n, p4, seed=seed, n_iters=8000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 5: L-BFGS-B from annealed
    p5 = t0 + 0.87 * total
    h = best_h.copy()
    for i, (a, mu) in enumerate(zip([5000, 50000, 500000, 5000000], [1000, 5000, 20000, 100000])):
        if time.time() > p5 - 2:
            break
        h = lbfgsb_opt(h, n, a, 3000, mu)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Final polish with noise restarts
    p6 = dl - 3
    if time.time() < p6:
        for trial in range(6):
            if time.time() > p6 - 1:
                break
            h = project_feasible(best_h + 0.003 * rng.randn(n), n)
            for a, mu in [(1000000, 50000), (10000000, 200000)]:
                if time.time() > p6 - 0.5:
                    break
                h = lbfgsb_opt(h, n, a, 2000, mu)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
            # Also try SLSQP from noise-perturbed
            if time.time() < p6 - 1:
                h2 = project_feasible(best_h + 0.001 * rng.randn(n), n)
                h2 = slsqp_opt(h2, n, 5000000, 2000)
                c2 = compute_c5(h2, n)
                if c2 < best_c:
                    best_c, best_h = c2, h2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
