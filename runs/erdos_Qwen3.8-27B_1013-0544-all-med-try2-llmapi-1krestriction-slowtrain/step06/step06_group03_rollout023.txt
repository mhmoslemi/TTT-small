from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_delta(h, j, n):
    m = 2 * n - 1
    base = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute coefficient c[idx] for each lag
    # L = idx - (n-1), ranges from -(n-1) to (n-1)
    c = np.zeros(m)
    for L in range(-(n-1), n):
        idx = L + (n - 1)
        coeff = 0.0
        i1 = j - L
        if 0 <= i1 < n:
            coeff += 1.0 - h[i1]
        i2 = j + L
        if 0 <= i2 < n:
            coeff -= h[i2]
        c[idx] = coeff
    
    d_lo = -h[j]
    d_hi = 1.0 - h[j]
    
    # Evaluate at boundaries
    best_val = min(np.max(base + c * d_lo), np.max(base + c * d_hi))
    best_d = d_lo if np.max(base + c * d_lo) <= np.max(base + c * d_hi) else d_hi
    
    # Interior: find all pairwise crossings in (d_lo, d_hi)
    # delta_pq = (base[q] - base[p]) / (c[p] - c[q])
    b_col = base.reshape(-1, 1)
    c_col = c.reshape(-1, 1)
    num = b_col.T - b_col
    den = c_col - c_col.T
    valid = np.abs(den) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        cr = np.where(valid, num / np.where(valid, den, 1.0), np.inf)
    
    mask = np.isfinite(cr) & (cr > d_lo + 1e-12) & (cr < d_hi - 1e-12)
    if mask.any():
        pts = np.sort(cr[mask])
        # Remove near-duplicates
        keep = np.concatenate(([True], np.diff(pts) > 1e-10))
        pts = pts[keep]
        if len(pts) > 500:
            idx_sub = np.linspace(0, len(pts)-1, 500).astype(int)
            pts = pts[idx_sub]
        vals = (base[:, None] + c[:, None] * pts[None, :]).max(axis=0)
        mi = np.argmin(vals)
        if vals[mi] < best_val - 1e-13:
            best_val = vals[mi]
            best_d = pts[mi]
    
    return best_d

def cd_sweep(h, n, order):
    """One coordinate descent sweep. Returns h and whether improved."""
    improved = False
    for j in order:
        d = coord_delta(h, j, n)
        if abs(d) > 1e-13:
            new_val = h[j] + d
            if -1e-15 <= new_val <= 1.0 + 1e-15:
                h[j] = np.clip(new_val, 0.0, 1.0)
                improved = True
    return h, improved

def obj_sle(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_delta(h, j, n):
    m = 2 * n - 1
    base = np.correlate(h, 1.0 - h, mode='full')
    c = np.zeros(m)

        idx = L + (n - 1)
        coeff = 0.0
        i1 = j - L
        if 0 <= i1 < n:
            coeff += 1.0 - h[i1]
        i2 = j + L
        if 0 <= i2 < n:
            coeff -= h[i2]
        c[idx] = coeff
    d_lo = -h[j]
    d_hi = 1.0 - h[j]
    if d_hi - d_lo < 1e-14:
        return 0.0
    v_lo = np.max(base + c * d_lo)
    v_hi = np.max(base + c * d_hi)
    best_val = min(v_lo, v_hi)
    best_d = d_lo if v_lo <= v_hi else d_hi
    b_col = base.reshape(-1, 1)
    c_col = c.reshape(-1, 1)
    num = b_col.T - b_col
    den = c_col - c_col.T
    valid = np.abs(den) > 1e-14
    if not valid.any():
        return best_d
    with np.errstate(divide='ignore', invalid='ignore'):

    mask = np.isfinite(cr) & (cr > d_lo + 1e-12) & (cr < d_hi - 1e-12)

        mi = np.argmin(vals)
        if vals[mi] < best_val - 1e-13:
            best_val = vals[mi]
            best_d = pts[mi]

def cd_sweep(h, n, order):
    improved = False
    for j in order:
        if h[j] > 1e-12 and h[j] < 1.0 - 1e-12:
            d = coord_delta(h, j, n)
            if abs(d) > 1e-13:
                nv = h[j] + d
                if -1e-12 <= nv <= 1.0 + 1e-12:
                    h[j] = np.clip(nv, 0.0, 1.0)
                    improved = True
    return improved

def obj_sle(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_sle, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
