            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def grad_smooth_max(h, n, alpha):
    """Exact gradient of alpha * logsumexp(c/alpha) where c = correlate(h, 1-h, 'full')."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    logits = c / alpha
    mx = logits.max()
    exp_l = np.exp(logits - mx)
    w = exp_l / exp_l.sum()
    w_rev = w[::-1]
    off = n - 1
    corr1 = np.correlate(w, 1.0 - h, mode='full')
    corr2 = np.correlate(w_rev, h, mode='full')
    return (corr1[off:off+n] - corr2[off:off+n]) * dx

def prox_gradient(h, n, alpha, lr_init, max_iter, deadline, use_momentum=True):
    """Proximal gradient with optional momentum and backtracking."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    momentum = np.zeros(n)
    lr = lr_init
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad = grad_smooth_max(h, n, alpha)
        if use_momentum:
            step = lr * grad + momentum
        else:
            step = lr * grad
        
        # Backtracking line search
        improved = False
        for bs in range(10):
            h_new = project_feasible(h - step, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 1e-12:  # allow tiny increases
                h = h_new
                momentum = (step - lr * grad) * 0.9 if use_momentum else np.zeros(n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                improved = True
                break
            else:
                step = step * 0.5
        
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
            stall_count += 1
            if stall_count > 50:
                break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, T_init=0.005, T_final=0.0001, max_iter=10000):
    """SA with structured perturbations."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_cur = best_c
    rng = np.random.RandomState(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_final / T_init) ** (it / max(1, max_iter - 1))
        
        # Perturbation: move mass between random pairs
        n_perturb = rng.randint(1, 4)
        for _ in range(n_perturb):
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(0.01, 0.05) * (1 if rng.random() < 0.5 else -1)
            if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                h[i] += d
                h[j] -= d
        
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(T, 1e-12)):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Revert (approximately)
            h = project_feasible(h, n)  # can't easily revert, so just keep
    
    return best_h, best_c

def pair_polish(h, n, deadline, deltas=None):
    """Pair-based coordinate descent for final polishing."""
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
