    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def anneal_search(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    T = 0.0005
    for it in range(500000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_5 = np.argsort(c)[-5:]
        k_star = top_5[rng.randint(5)]
        lag = int(k_star) - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            break
        i = rng.randint(0, n - m)
        j = i + m
        delta = 0.005 * (2.0 * rng.random() - 1.0)
        if delta == 0:
            continue
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h[i] += delta
        h[j] -= delta
        c_new = compute_c5(h, n)
        if c_new < best_c or rng.random() < np.exp((best_c - c_new) / max(T, 1e-10)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        T *= 0.9995
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 30:
        h_new = slsqp_optimize(best_h, n, 50000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 40:
        sg_end = min(deadline - 25, time.time() + 80)
        h_sg, c_sg = subgrad_momentum(best_h, n, sg_end)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    if time.time() < deadline - 15:
        h_new = slsqp_optimize(best_h, n, 200000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 10:
        h_ls, c_ls = local_search_greedy(best_h, n, min(deadline - 5, time.time() + 30))
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    if time.time() < deadline - 8:
        h_ls, c_ls = local_search_fractional(best_h, n, min(deadline - 5, time.time() + 25))
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial n
    if time.time() < deadline - 150:
        alloc = min(120, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_at_n(best_h, n_init, t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Try nearby n values
    n_candidates = sorted(set([n_init - 2, n_init, n_init + 2, 200, 400]))
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if n_try == n_init and len(best_h) == n_try:
            continue
        if n_try < 50 or n_try % 2 != 0:
            continue
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(80, remaining * 0.25)
        if len(best_h) > 1:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Concurrent and anneal search on best
    n = len(best_h)
    if time.time() < deadline - 40:
        cc_end = min(deadline - 20, time.time() + 40)
        h_cc, c_cc = concurrent_search(best_h, n, cc_end)
        if c_cc < best_c:
            best_c = c_cc
            best_h = h_cc

    if time.time() < deadline - 30:
        an_end = min(deadline - 15, time.time() + 40)
        h_an, c_an = anneal_search(best_h, n, an_end, seed=seed)
        if c_an < best_c:
            best_c = c_an
            best_h = h_an

    # Phase 4: Restart with perturbations
    if time.time() < deadline - 50:
        n_rest = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.008 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n_rest)
            hp = project_feasible(hp, n_rest)
            alloc = min(40, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n_rest, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final polish
    n = len(best_h)
    if time.time() < deadline - 10:
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 3:
        h_ls, c_ls = local_search_greedy(best_h, n, deadline - 1)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
