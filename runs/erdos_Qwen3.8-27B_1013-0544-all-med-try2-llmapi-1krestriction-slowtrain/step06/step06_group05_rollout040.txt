```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def corr_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = corr_all(x, n)
    m = c.max()
    if alpha < 1e6:
        shifted = alpha * (c - m)
        logsum = np.log(np.sum(np.exp(shifted))) 
        return float(m + logsum / alpha)
    else:
        return float(m)

def active_set_gradient(h, n, c_vals, tol=1e-4):
    """Compute gradient using active set (shifts within tol of max)"""
    m = c_vals.max()
    active = np.where(c_vals >= m - tol)[0]
    # Convert from correlation index to shift
    # np.correlate(h, 1-h, mode='full') has index k corresponding to shift k-(n-1)
    n1 = n - 1
    grad = np.zeros(n)
    for idx in active:
        shift = idx - n1
        grad += grad_shift(h, n, shift)
    return grad / len(active)

def grad_shift(h, n, shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[0:n-shift] += (1 - h[shift:n])
            grad[shift:n] += -h[0:n-shift]
    else:
        s = -shift
        if s < n:
            grad[s:n] += (1 - h[0:n-s])
            grad[0:n-s] += -h[s:n]
    return grad * dx

def projected_subgradient(h, n, deadline, max_iter=50000):
    """Projected subgradient method with adaptive step size"""
    h = h.copy()
    c_vals = corr_all(h, n)
    best_c = c_vals.max()
    best_h = h.copy()
    
    # Adaptive step size
    step = 0.01
    tol_active = 1e-3
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vals = corr_all(h, n)
        cur_c = c_vals.max()
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
        
        # Compute active gradient
        m = cur_c
        active = np.where(c_vals >= m - tol_active)[0]
        n1 = n - 1
        grad = np.zeros(n)
        for idx in active:
            shift = idx - n1
            grad += grad_shift(h, n, shift)
        grad /= len(active)
        
        # Projected gradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c_new = corr_all(h_new, n).max()
        if c_new < cur_c:
            h = h_new
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.8
            if step < 1e-5:
                tol_active = max(1e-5, tol_active * 0.7)
    
    return best_h, best_c

def pair_search_local(h, n, deadline, deltas, radius=50):
    """Local pair search within a radius"""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(100):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000):

    dx = 2.0 / n
    
    def obj(x):
        c = corr_all(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        max_exp = shifted.max()
        if max_exp > 500:
            return float(m + (max_exp + np.log(np.sum(np.exp(shifted - max_exp)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        c = corr_all(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        max_exp = shifted.max()
        
        if max_exp > 500:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def corr_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_with_grad(x, n, alpha):
    """Compute smooth max of correlations and its gradient."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    
    if max_s > 500:
        exp_s = np.exp(shifted - max_s)
        log_sum = max_s + np.log(exp_s.sum())
        val = m + log_sum / alpha
    else:
        exp_s = np.exp(shifted)
        log_sum = np.log(exp_s.sum())
        val = m + log_sum / alpha
    
    w = exp_s / exp_s.sum()  # weights (softmax)
    
    # Gradient computation using convolutions
    # A[i] = sum of w_k for k in [N-1-i, N-1+i]
    # Use cumsum for sliding window
    w_csum = np.concatenate([[0], np.cumsum(w)])
    N = n
    # A[i] = w_csum[min(2N-2, N-1+i)] - w_csum[max(0, N-1-i)]
    lo = np.maximum(0, N - 1 - np.arange(N))
    hi = np.minimum(2*N-2, N - 1 + np.arange(N))
    A = w_csum[hi+1] - w_csum[lo]
    
    # B[i] = sum_k w_k * x[i+s_k] where s_k = k-(N-1), valid when 0<=i+s_k<N
    # = conv(x, w_rev, 'full')[i+1]
```

```python
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_val_grad(x, n, alpha):
    dx = 2.0 / n
    N = n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        exp_s = np.exp(shifted - max_s)
        log_sum = max_s + np.log(exp_s.sum())
        val = m + log_sum / alpha
    else:
        exp_s = np.exp(shifted)
        log_sum = np.log(exp_s.sum())
        val = m + log_sum / alpha
    w = exp_s / exp_s.sum()
    w_csum = np.empty(2 * N)
    w_csum[0] = 0
    w_csum[1:] = np.cumsum(w)
    idx = np.arange(N)
    lo = np.maximum(0, N - 1 - idx)
    hi = np.minimum(2 * N - 2, N - 1 + idx)
    A = w_csum[hi + 1] - w_csum[lo]
    B_full = np.convolve(x, w[::-1], mode='full')
    B = B_full[N - 1:2 * N - 1]
    C_full = np.convolve(w, x, mode='full')
    C = C_full[N - 1:2 * N - 1]
    grad = dx * (A - B - C)
    return val, grad

def slsqp_fast(h, n, alpha, maxiter=2000):
    N = n
    def obj(x):
        return smooth_val_grad(x, N, alpha)[0]
    def obj_grad(x):
        return smooth_val_grad(x, N, alpha)[1]
    def con_fun(x):
        return x.sum() - N / 2.0
    def con_jac(x):
        return np.ones(N)
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, N)
    except Exception:
        return project_feasible(h, N)

def compute_g(h, n):
    N = n
    i_idx = np.arange(N)[:, None]
    k_idx = np.arange(2 * N - 1)[None, :]
    idx1 = i_idx + k_idx - (N - 1)
    idx2 = i_idx - k_idx + (N - 1)
    mask1 = (idx1 >= 0) & (idx1 < N)
    mask2 = (idx2 >= 0) & (idx2 < N)
    idx1c = np.clip(idx1, 0, N - 1)
    idx2c = np.clip(idx2, 0, N - 1)
    g = (1.0 - h[idx1c]) * mask1.astype(float) - h[idx2c] * mask2.astype(float)
    return g

def fast_pair_search(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                return h, best_c
            c = corr_all(h, n)
            g = compute_g(h, n)
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        delta_c = dx * d * (g[i] - g[j])
                        c_new = c + delta_c
                        new_max = c_new.max()
                        if new_max < best_c - 1e-15:
                            h[i], h[j] = hi, hj
                            best_c = float(new_max)
                            c = corr_all(h, n)
                            g = compute_g(h, n)
                            improved = True
                            continue
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        delta_c = dx * (-d) * (g[i] - g[j])
                        c_new = c + delta_c
                        new_max = c_new.max()
                        if new_max < best_c - 1e-15:
                            h[i], h[j] = hi2, hj2
                            best_c = float(new_max)
                            c = corr_all(h, n)
                            g = compute_g(h, n)
                            improved = True
            if not improved:
                break
    return h, best_c

def active_subgrad(h, n, deadline, alpha_act=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.08
    N = n
    for it in range(50000):
        if time.time() > deadline:
            break
        c = corr_all(h, N)
        cur_c = float(c.max())
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
        tol = 5e-4
        m = cur_c
        active = c >= m - tol
        if active.sum() == 0:
            break
        c_act = c[active]
        sh = alpha_act * (c_act - c_act.max())
        w_act = np.exp(sh)
        w_act /= w_act.sum()
        w = np.zeros(2 * N - 1)
        w[active] = w_act
        dx = 2.0 / N
        w_csum = np.empty(2 * N)
        w_csum[0] = 0
        w_csum[1:] = np.cumsum(w)
        idx = np.arange(N)
        lo = np.maximum(0, N - 1 - idx)
        hi = np.minimum(2 * N - 2, N - 1 + idx)
        A = w_csum[hi + 1] - w_csum[lo]
        B = np.convolve(h, w[::-1], mode='full')[N - 1:2 * N - 1]
        C = np.convolve(w, h, mode='full')[N - 1:2 * N - 1]
        grad = dx * (A - B - C)
        h_new = project_feasible(h - step * grad, N)
        c_new = compute_c5(h_new, N)
        if c_new < cur_c:
            h = h_new
            step = min(step * 1.1, 0.3)
        else:
            step *= 0.75
            if step < 1e-5:
                break
    return best_h, best_c

def binarize_try(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    sorted_idx = np.argsort(h)
    for frac in np.arange(0.44, 0.56, 0.02):
        n_ones = int(frac * n)
        h_try = np.zeros(n)
        h_try[sorted_idx[n - n_ones:]] = 1.0
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    if time.time() < deadline:
        best_h, best_c = fast_pair_search(best_h, n, deadline,
                                          [0.01, 0.005, 0.002, 0.001], radius=80)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SLSQP with fast gradient, increasing alpha
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 100:
            break
        alloc = min(60, (deadline - time.time() - 100) * 0.3)
        if alloc < 10:
            break
        ht = slsqp_fast(best_h, n_init, a, maxiter=3000)
        c = compute_c5(ht, n_init)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Phase 2: Active-set subgradient
    if time.time() < deadline - 60:
        ht, ct = active_subgrad(best_h, n_init, deadline - 60, alpha_act=200)
        if ct < best_c:
            best_c = ct
            best_h = ht

    # Phase 3: Binarization + local search
    if time.time() < deadline - 60:
        ht, ct = binarize_try(best_h, n_init, deadline - 60)
        if ct < best_c:
            best_c = ct
            best_h = ht
        else:
            best_h, best_c = fast_pair_search(best_h, n_init, deadline - 60,
                                              [0.005, 0.002, 0.001, 0.0005], radius=80)

    # Phase 4: Multi-start perturbations
    if time.time() < deadline - 50:
        scales = [0.02, 0.05, 0.1]
        for sc in scales:
            if time.time() > deadline - 40:
                break
            hp = project_feasible(best_h + rng.normal(0, sc, n_init), n_init)
            alloc = min(30, (deadline - time.time() - 35) * 0.4)
            if alloc < 10:
                break
            ht = slsqp_fast(hp, n_init, 20000, maxiter=1500)
            c = compute_c5(ht, n_init)
            if time.time() < deadline - 30:
                ht, c = fast_pair_search(ht, n_init, deadline - 30,
                                         [0.005, 0.002, 0.001], radius=70)
            if c < best_c:
                best_c = c
                best_h = ht

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 25:
        ht = slsqp_fast(best_h, n_init, 500000, maxiter=3000)
        c = compute_c5(ht, n_init)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Final fine local search
    if time.time() < deadline - 3:
        best_h, best_c = fast_pair_search(best_h, n_init, deadline - 2,
                                          [0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=100)

    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
```
