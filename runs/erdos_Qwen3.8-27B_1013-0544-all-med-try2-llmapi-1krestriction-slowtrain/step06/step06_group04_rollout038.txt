def project_feasible(h, n):

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += -h[:n-m]
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += -h[m:]
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def soft_max_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_slsqp(h, n, top_m, alpha, maxiter, ftol):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    
    def obj(x):
        return soft_max_obj(x, n, alpha, top_idx)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent(h, n, deadline, max_iters=2000):

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h, n)
        m = c.max()
        # Soft weights based on how close each lag is to the max
        weights = np.exp(50 * (c - m))
        weights /= weights.sum()
        
        # Only use top lags for efficiency
        top_k = 5
        top_idx = np.argsort(weights)[-top_k:]
        top_w = weights[top_idx]
        top_w /= top_w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for w, idx in zip(top_w, top_idx):
            lag = int(idx) - (n - 1)
            grad += w * lag_gradient(h, n, lag)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] += -h[:n - m]
    else:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += -h[m:]
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _soft_max_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_slsqp(h, n, top_m, alpha, maxiter, ftol):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_soft_max_obj, h, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent(h, n, deadline, max_iters=3000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        m = c.max()
        weights = np.exp(80.0 * (c - m))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        top_k = 5
        top_idx = np.argsort(weights)[-top_k:]
        top_w = weights[top_idx]
        top_w /= top_w.sum()
        grad = np.zeros(n)
        for w, idx in zip(top_w, top_idx):
            lag = int(idx) - (n - 1)
            grad += w * lag_gradient(h, n, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        step = 0.05
        for _ in range(25):
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                break
            step *= 0.5
        if step < 1e-10:
            break
    return best_h, best_c

def pair_local_search(h, n, deadline, max_rounds=1000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        grad = lag_gradient(h, n, lag)
        pos_idx = np.argsort(-grad)[:15]
        neg_idx = np.argsort(grad)[:15]
        best_improve = 0
        best_move = None
        deltas = [0.02, 0.008, 0.003]
        for d in deltas:
            for i in pos_idx:
                if h[i] < 1e-10:
                    continue
                for j in neg_idx:
                    if j == i or 1.0 - h[j] < 1e-10:
                        continue
                    delta = min(d, h[i] * 0.5, (1.0 - h[j]) * 0.5)
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    h[i] += delta
                    h[j] -= delta
                    if c_new < best_c - 1e-14:
                        imp = best_c - c_new
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j,(delta))
            if best_move is not None:
                break
        if best_move is None:
            for d in deltas:
                for i in neg_idx:
                    if 1.0 - h[i] < 1e-10:
                        continue
                    for j in pos_idx:
                        if j == i or h[j] < 1e-10:
                            continue
                        delta = min(d, (1.0 - h[i]) * 0.5, h[j] * 0.5)
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        h[i] -= delta
                        h[j] += delta
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            best_move = (i, j, -delta)
                            break
                    if best_move is not None:
                        break
                if best_move is not None:
                    break
        if best_move is None:
            break
        i, j, delta = best_move
        h[i] += delta
        h[j] -= delta
        h = project_feasible(h, n)
        best_c = compute_c5(h, n)
        best_h = h.copy()
    return best_h, best_c

def optimize_round(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 25:
        h_new = active_slsqp(best_h, n, 5, 30000, 4000, 1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

    if time.time() < deadline - 20:
        h_new = active_slsqp(best_h, n, 3, 150000, 5000, 1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 15:
        gd_end = min(deadline - 12, time.time() + 30)
        h_new, c_new = gradient_descent(best_h, n, gd_end)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 12:
        h_new = active_slsqp(best_h, n, 3, 400000, 6000, 1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 5:
        ls_end = min(deadline - 2, time.time() + 25)
        h_new, c_new = pair_local_search(best_h, n, ls_end)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 3:
        h_new = active_slsqp(best_h, n, 3, 1000000, 5000, 1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 1:
        h_new, c_new = pair_local_search(best_h, n, deadline - 0.5, max_rounds=500)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
