        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def subgrad_fast(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_eq(h, target_sum):
    return h.sum() - target_sum


def con_eq_jac(h, target_sum):
    return np.ones(len(h))


def resample_h(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def slsqp_solve(h0, n, dx, target_sum, bounds, beta, maxiter):
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0


def simulated_annealing(h_start, n, dx, target_sum, bounds, iters, time_limit_fn, current_best):
    h = h_start.copy()
    T = 0.02
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    for it in range(iters):
        if time_limit_fn() < 20:
            break
        # Random perturbation
        scale = T * 0.5
        h_trial = h + np.random.randn(n) * scale
        h_trial = project_feasible(h_trial, target_sum)
        c5_trial = compute_c5(h_trial)
        c5_current = compute_c5(h)
        # Accept with probability
        if c5_trial < c5_current or np.random.rand() < np.exp(-(c5_trial - c5_current) / T):
            h = h_trial
            if c5_trial < best_c5_local:
                best_c5_local = c5_trial
                best_local = h.copy()
                if c5_trial < current_best[0]:
                    current_best[0] = c5_trial
                    current_best[1] = h.copy()
        T *= 0.9995
    return best_local, best_c5_local


def momentum_subgrad(h_start, n, dx, target_sum, iters, time_limit_fn, current_best):
    h = h_start.copy()
    vel = np.zeros(n)
    mu = 0.9
    lr = 0.005
    for it in range(iters):
        if time_limit_fn() < 25:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_worst = np.argmax(corr)
        grad = subgrad_fast(h, k_worst, n)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad /= gnorm
        vel = mu * vel - lr * grad
        h = h - vel
        h = project_feasible(h, target_sum)
        if it % 500 == 499:
            c5 = compute_c5(h)
            if c5 < current_best[0]:
                current_best[0] = c5
                current_best[1] = h.copy()
            # Adaptive lr
            if c5 < current_best[0] + 1e-6:
                lr = min(lr * 1.05, 0.03)
            else:
                lr = max(lr * 0.9, 0.0005)
    return h, current_best[0]


def binary_push(h_start, n, dx, target_sum, time_limit_fn, current_best):
    h = h_start.copy()
    c5_cur = compute_c5(h)
    improved = True
    while improved and time_limit_fn() > 30:
        improved = False
        for i in range(n):
            if h[i] > 0.03 and h[i] < 0.97:
                for direction in [-1, 1]:
                    h_try = h.copy()
                    h_try[i] += direction * 0.15
                    h_try = project_feasible(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur - 1e-10:
                        h = h_try
                        c5_cur = c5_try
                        improved = True
                        if c5_try < current_best[0]:
                            current_best[0] = c5_try
                            current_best[1] = h.copy()
                        break
    return h, c5_cur


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    current_best = [best_c5, best_h.copy()]

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation (refine starting point)
    for beta in [10, 50, 200, 1000, 5000]:
        if tl() < 120:
            break
        h_new = slsqp_solve(best_h, n, dx, target_sum, bounds, beta, 3000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
            current_best = [best_c5, best_h.copy()]

    # Phase 2: Momentum subgradient
    if tl() > 150:
        h_out, _ = momentum_subgrad(best_h, n, dx, target_sum, 60000, tl, current_best)
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1].copy()

    # Phase 3: Simulated annealing for exploration
    if tl() > 120:
        for restart in range(3):
            if tl() < 60:
                break
            if restart == 0:
                h_sa = best_h.copy()
            else:
                h_sa = best_h + np.random.randn(n) * 0.05
                h_sa = project_feasible(h_sa, target_sum)
            h_out, c5_out = simulated_annealing(h_sa, n, dx, target_sum, bounds,
                                                20000, tl, current_best)
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1].copy()

    # Phase 4: Binary push heuristic
    if tl() > 60:
        h_out, c5_out = binary_push(best_h, n, dx, target_sum, tl, current_best)
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1].copy()

    # Phase 5: High-precision SLSQP polish
    if tl() > 50:
        for beta in [10000, 50000, 200000]:
            if tl() < 20:
                break
            h_new = slsqp_solve(best_h, n, dx, target_sum, bounds, beta, 8000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
                current_best = [best_c5, best_h.copy()]

    # Phase 6: Multi-resolution search
    if tl() > 150:
        for n_try in [50, 60, 80, 90, 100, 120, 140]:
            if tl() < 50:
                break
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [500, 5000, 50000]:
                if tl() < 25:
                    break
                h_new = slsqp_solve(h_t, n_try, dx_t, ts_t, bnds_t, beta, 2000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                    current_best = [best_c5, best_h.copy()]
                    # Refine subgradient at this resolution
                    h_out, c5_out = momentum_subgrad(h_new, n_try, dx_t, ts_t,
                                                     20000, tl, current_best)
                    if current_best[0] < best_c5:
                        best_c5 = current_best[0]
                        best_h = current_best[1].copy()

    # Phase 7: Final polish at best resolution
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    if tl() > 30:
        for beta in [50000, 200000]:
            if tl() < 15:
                break
            h_new = slsqp_solve(best_h, n, dx, target_sum, bounds, beta, 10000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    return best_h, best_c5, len(best_h)
