    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        lag = int(idx) - (n - 1)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] += w * (1.0 - h[i + lag])
            grad[i + lag] -= w * h[i]
        else:
            la = -lag
            i = np.arange(n - la)
            grad[i + la] += w * (1.0 - h[i])
            grad[i] -= w * h[i + la]
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    ts0 = n0 / 2.0

    # Build diverse starting points
    starts = [h_init.copy()]
    starts.append(make_sinusoidal(n0, 1, 0.0))
    starts.append(make_sinusoidal(n0, 2, np.pi / 3))
    starts.append(make_sinusoidal(n0, 3, 0.5))
    starts.append(make_piecewise(n0, [(0, 0.3, 1.0), (0.3, 0.5, 0.0), (0.5, 0.8, 1.0), (0.8, 1.0, 0.0)]))
    starts.append(make_piecewise(n0, [(0, 0.25, 0.8), (0.25, 0.5, 0.2), (0.5, 0.75, 0.8), (0.75, 1.0, 0.2)]))
    starts.append(make_binary_random(n0, rng))
    starts.append(make_binary_random(n0, rng))
    starts.append(project_feasible(h_init + rng.randn(n0) * 0.08, ts0))
    starts.append(project_feasible(h_init + rng.randn(n0) * 0.15, ts0))

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    def optimize_single(h_start, n_opt, time_budget_fn):
        dx_o = 2.0 / n_opt
        ts_o = n_opt / 2.0
        h_c = project_feasible(h_start.astype(float).copy(), ts_o)
        c5_c = compute_c5(h_c)
        bnds = Bounds(np.zeros(n_opt), np.ones(n_opt))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_o, n_opt)}]

        # Phase A: SLSQP continuation
        for beta in [8, 40, 150, 600, 2500]:
            if time_budget_fn() < 12:
                break
            for trial in range(2):
                if time_budget_fn() < 8:
                    break
                x0 = h_c.copy()
                if trial > 0:
                    x0 += rng.randn(n_opt) * 0.01
                    x0 = project_feasible(x0, ts_o)
                try:
                    res = minimize(obj_smooth, x0, args=(dx_o, beta),
                                   method='SLSQP', bounds=bnds, constraints=cons,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_o)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_c:
                        h_c = h_new
                        c5_c = c5_new
                except Exception:
                    pass

        # Phase B: Multi-lag subgradient with adaptive step
        if time_budget_fn() > 25:
            h_s = h_c.copy()
            for it in range(60000):
                if time_budget_fn() < 15:
                    break
                grad = multi_lag_subgrad(h_s, n_opt, top_k=5, beta_lag=10)
                lr = 0.004 / (1.0 + it * 0.0004)
                h_s = h_s - lr * grad
                h_s = project_feasible(h_s, ts_o)
                if it % 2500 == 2499:
                    c5_ch = compute_c5(h_s)
                    if c5_ch < c5_c:
                        h_c = h_s.copy()
                        c5_c = c5_ch
            c5_ch = compute_c5(h_s)
            if c5_ch < c5_c:
                h_c = h_s.copy()
                c5_c = c5_ch

        # Phase C: High-beta SLSQP polish
        if time_budget_fn() > 10:
            for beta in [5000, 20000]:
                if time_budget_fn() < 8:
                    break
                try:
                    res = minimize(obj_smooth, h_c, args=(dx_o, beta),
                                   method='SLSQP', bounds=bnds, constraints=cons,
                                   options={'maxiter': 6000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_o)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_c:
                        h_c = h_new
                        c5_c = c5_new
                except Exception:
                    pass

        return h_c, c5_c

    # Optimize each start with time allocation
    n_starts = len(starts)
    time_per = (budget_s * 0.55) / n_starts

    for i, hs in enumerate(starts):
        if remaining() < 25:
            break
        if remaining() < 10:
            break
        h_opt, c5_opt = optimize_single(hs, n0, remaining)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            n0 = len(h_opt)

    # Try different resolutions from best
    if remaining() > 50:
        for dn in [5, -5, 10, -10, 15, -15, 20, -20]:
            if remaining() < 25:
                break
            n_try = n0 + dn
            if n_try < 20:
                continue
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_res = np.interp(x_new, x_old, best_h)
            h_res = project_feasible(h_res, n_try / 2.0)
            h_opt, c5_opt = optimize_single(h_res, n_try, remaining)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                n0 = n_try

    # Final aggressive polish
    if remaining() > 12:
        df = 2.0 / len(best_h)
        nf = len(best_h)
        tf = nf / 2.0
        bnds_f = Bounds(np.zeros(nf), np.ones(nf))
        cons_f = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (tf, nf)}]
        for beta in [10000, 40000, 100000]:
            if remaining() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(df, beta),
                               method='SLSQP', bounds=bnds_f, constraints=cons_f,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, tf)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Very final subgradient squeeze
    if remaining() > 10:
        h_s = best_h.copy()
        nf = len(h_s)
        ts_f = nf / 2.0
        for it in range(20000):
            if remaining() < 5:
                break
            grad = multi_lag_subgrad(h_s, nf, top_k=8, beta_lag=15)
            lr = 0.002 / (1.0 + it * 0.001)
            h_s = h_s - lr * grad
            h_s = project_feasible(h_s, ts_f)
            if it % 3000 == 2999:
                c5_ch = compute_c5(h_s)
                if c5_ch < best_c5:
                    best_h = h_s.copy()
                    best_c5 = c5_ch

    return best_h, best_c5, len(best_h)
