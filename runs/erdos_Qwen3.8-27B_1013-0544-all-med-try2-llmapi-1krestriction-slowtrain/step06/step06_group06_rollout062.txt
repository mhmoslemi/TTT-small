```python
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, max_sweeps=5000):
    """Efficient binary local search using single swaps."""
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_improvement = 1e-10
        best_swap = None
        best_new_c = best_c
        
        # Try all single swaps
        for a in zeros:
            for b in ones:
                # Swap: a goes 0->1, b goes 1->0
                h_bin[a] = 1.0
                h_bin[b] = 0.0
                c = compute_c5(h_bin, n)
                h_bin[a] = 0.0
                h_bin[b] = 1.0
                
                if best_c - c > best_improvement:
                    best_improvement = best_c - c
                    best_swap = (a, b)
                    best_new_c = c
        
        if best_swap is None:
            break
        
        a, b = best_swap
        h_bin[a] = 1.0
        h_bin[b] = 0.0
        ones.remove(b)
        ones.add(a)
        zeros.remove(a)
        zeros.add(b)
        best_c = best_new_c
    
    return h_bin, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smooth(h, n, alpha, maxiter=500, deadline=None):
    """SLSQP optimization with smooth max approximation."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def grad(x):
        # Numerical gradient (finite differences)
        eps = 1e-6
        g0 = obj(x)
        grad = np.zeros(n)
        for i in range(n):
            x[i] += eps
            grad[i] = (obj(x) - g0) / eps
            x[i] -= eps
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

But actually, I realize I should use a closed-form or semi-analytical gradient for the smooth max. Let me derive it properly.

For g_k = dx * ∑_i h[i] * (1 - h[i+k-(n-1)]):

∂g_k/∂h[i] = dx * [(1 - h[j(i,k)]) - h[i'] * δ] where:
- j(i,k) = i+k-(n-1) (the index in the (1-h) factor)
- i' = i-k+(n-1) (the index in the h factor where i appears in (1-h))

More precisely:
g_k = dx * ∑_m h[m] * (1 - h[m+k-(n-1)])

∂g_k/∂h[i]:
- h[i] appears in the first factor (h[m]) when m = i: contributes dx * (1 - h[i+k-(n-1)]) if i+k-(n-1) is valid
- h[i] appears in the second factor (1-h[m+k-(n-1)]) when m+k-(n-1) = i, i.e., m = i-k+(n-1): contributes dx * h[m] * (-1) = -dx * h[i-k+(n-1)] if i-k+(n-1) is valid

So:
∂g_k/∂h[i] = dx * (1 - h[i+k-(n-1)]) * 1_{valid(i+k-(n-1))} - dx * h[i-k+(n-1)] * 1_{valid(i-k+(n-1))}

where valid(x) means 0 ≤ x < n.

For the smooth max: S(α) = max_k g_k + (1/α) * log(∑_k exp(α * g_k))

∂S/∂h[i] = ∑_k w_k * (∂g_k/∂h[i])

where w_k = exp(α * g_k) / ∑_j exp(α * g_j) (softmax weights)

So the gradient is a weighted sum of the individual ∂g_k/∂h[i] with weights w_k.

For large α, w_k is concentrated on the k that maximizes g_k, so this becomes approximately the subgradient.

This gradient computation is O(n * (2n-1)) = O(n^2) for the full sum, or O(n) if we only use the single active k (subgradient approximation).

Let me implement the full smooth gradient:

```python
def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    total_len = 2*n - 1
    g = np.correlate(h, 1-h, mode='full') * dx
    m = g.max()
    exp_arg = alpha * (g - m)
    exp_vals = np.exp(exp_arg - exp_arg.max())
    w = exp_vals / exp_vals.sum()
```

```python
    m = g.max()
    exp_arg = alpha * (g - m)
    exp_vals = np.exp(exp_arg - exp_arg.max())
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    h_one = 1.0 - x
    for k in range(total_len):
        if w[k] < 1e-12:
            continue
        offset = k - (n - 1)
        i_s = max(0, -offset)
        i_e = min(n, n - offset)
        if i_e > i_s:
            grad[i_s:i_e] += w[k] * dx * h_one[i_s + offset:i_e + offset]
        i_s2 = max(0, offset)
        i_e2 = min(n, n + offset)
        if i_e2 > i_s2:
            grad[i_s2:i_e2] -= w[k] * dx * x[i_s2 - offset:i_e2 - offset]
    return grad

def proj_grad_desc(h, n, alpha, deadline, max_iter=10000, lr_init=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    obj_curr = smooth_obj(h, n, alpha)
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        accepted = False
        for ft in range(25):
            h_new = project_feasible(h - lr * grad, n)
            obj_new = smooth_obj(h_new, n, alpha)
            if obj_new <= obj_curr - 1e-4 * lr * gn * gn:
                h = h_new
                obj_curr = obj_new
                accepted = True
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                break
            lr *= 0.5
        if not accepted:
            lr *= 0.5
            if lr < 1e-16:
                break
            stagnant += 1
        if stagnant > 500:
            break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=300, deadline=None):
    class Pb:
        def __init__(self, nn, aa):
            self.n = nn
            self.alpha = aa
        def obj(self, x):
            return smooth_obj(x, self.n, self.alpha)
        def grad(self, x):
            return smooth_grad(x, self.n, self.alpha)
        def cons(self, x):
            return x.sum() - self.n / 2.0
    pb = Pb(n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': pb.cons}]
    try:
        res = minimize(pb.obj, h, jac=pb.grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_bin, n, deadline, max_sweeps=5000):
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_improve = 1e-13
        best_pair = None
        for a in zeros:
            for b in ones:
                h_bin[a] = 1.0
                h_bin[b] = 0.0
                c = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
                h_bin[a] = 0.0
                h_bin[b] = 1.0
                if best_c - c > best_improve:
                    best_improve = best_c - c
                    best_pair = (a, b)
        if best_pair is None:
            break
        a, b = best_pair
        h_bin[a] = 1.0
        h_bin[b] = 0.0
        ones.remove(b)
        ones.add(a)
        zeros.remove(a)
        zeros.add(b)
        best_c -= best_improve
    return h_bin, best_c

def binary_sa(h_bin, n, deadline, T_init=0.02, T_factor=0.9995, max_iters=100000, seed=99):
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    c = best_c
    T = T_init
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    rng = np.random.RandomState(seed)
    ones_list = sorted(ones)
    zeros_list = sorted(zeros)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 50 == 0:
            T *= T_factor
            if T < 1e-12:
                break
        a = zeros_list[rng.randint(len(zeros_list))]
        b = ones_list[rng.randint(len(ones_list))]
        h_bin[a] = 1.0
        h_bin[b] = 0.0
        c_new = compute_c5(h_bin, n)
        delta = c_new - c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            c = c_new
            ones.remove(b)
            ones.add(a)
            zeros.remove(a)
            zeros.add(b)
            ones_list = sorted(ones)
            zeros_list = sorted(zeros)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h_bin.copy()
        else:
            h_bin[a] = 0.0
            h_bin[b] = 1.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Projected gradient with analytical smooth max gradient
    p1_dl = t0 + 0.30 * (budget_s - 10)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000, 5000000]:
        if time.time() > p1_dl:
            break
        h = proj_grad_desc(h, n, alpha, p1_dl, max_iter=8000, lr_init=0.003)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: SLSQP polish with very large alpha
    p2_dl = t0 + 0.45 * (budget_s - 10)
    if time.time() < p2_dl:
        h = slsqp_smooth(best_h.copy(), n, alpha, maxiter=500, deadline=p2_dl)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 3: Binary local search from multiple starts
    p3_dl = t0 + 0.85 * (budget_s - 10)
    rng = np.random.RandomState(seed)
    h_bin = to_binary(best_h, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    starts = [h_bin.copy()]
    for _ in range(4):
        h_pert = best_h + 0.12 * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        hb = to_binary(h_pert, n)
        starts.append(hb)
    for i, sb in enumerate(starts):
        if time.time() > p3_dl:
            break
        mx = 3000 if i == 0 else 1500
        hb, cb = binary_local_search(sb.copy(), n, p3_dl, max_sweeps=mx)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 4: Simulated annealing
    p4_dl = dl
    if time.time() < p4_dl - 5:
        h_bin = to_binary(best_h, n)
        hb, cb = binary_sa(h_bin, n, p4_dl, T_init=0.03, T_factor=0.9998,
                          max_iters=80000, seed=seed + 7)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Final polish
    if time.time() < dl - 3:
        h_bin = to_binary(best_h, n)
        hb, cb = binary_local_search(h_bin, n, dl, max_sweeps=300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
