= (t-a) * [sum_{q: i,q in S_k}(1-h[q]) - sum_{p: p,i in S_k} h[p]] 
  + (a-t) * [sum_{q: j,q in S_k}(1-h[q]) - sum_{p: p,j in S_k} h[p]]
= (t-a) * slope_i_k - (t-a) * slope_j_k
= (t-a) * (slope_i_k - slope_j_k)

where slope_i_k is the derivative of c_k w.r.t. h[i] (computed as before).

So c_k(t) = c_k_old + (t-a)*(slope_i_k - slope_j_k)

This is still linear in t! So the objective max_k c_k(t) is the max of linear functions of t, and we can minimize it with binary search just like before.

This is great! So the algorithm is:
1. Pick a pair (i,j)
2. Compute slope_i_k and slope_j_k for all k
3. Set a = h[i], b = h[j], c = a+b
4. For t in [max(0, c-1), min(1, c)]: minimize max_k (c_k_old + (t-a)*(slope_i_k - slope_j_k)) * dx
5. Update h[i] = t_opt, h[j] = c - t_opt

The feasible range for t is [max(0, c-1), min(1, c)].

This is a clean 1D optimization that preserves the sum constraint exactly!

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_slopes(h, n):
    """Compute slope[i, k] = d(corr[k])/d(h[i]) for all i, k."""
    m = n - 1
    slopes = np.zeros((n, 2*n-1))
    for i in range(n):
        # h[i] as first index: corr[k] = sum_j h[j]*(1-h[j-k+m])
        # h[i] appears when j=i: term is h[i]*(1-h[i-k+m]), valid if 0 ≤ i-k+m < n
        # i.e., k-m ≤ i ≤ k, i.e., i-m ≤ k ≤ i
        k_lo = max(0, i - m)
        k_hi = min(2*n-2, i)
        for k in range(k_lo, k_hi+1):
            idx = i - k + m  # = i - k + n - 1
            if 0 <= idx < n:
                slopes[i, k] += (1 - h[idx])
        
        # h[i] as second index: j-k+m = i, so j = i+k-m
        # valid if 0 ≤ i+k-m < n, i.e., m-i ≤ k ≤ m-i+n-1 = 2n-2-i
        # Also need j = i+k-m to be in [0, n-1] (already ensured)
        # And k in [0, 2n-2]
        k_lo2 = max(0, m - i)
        k_hi2 = min(2*n-2, m - i + n - 1)  # = min(2n-2, 2n-2-i)
        for k in range(k_lo2, k_hi2+1):
            j = i + k - m  # = i + k - n + 1
            if 0 <= j < n:
                slopes[i, k] -= h[j]
    return slopes

        max_s = diff_slopes[active_mask].max()
        if min_s > 1e-14:
            hi = mid
        elif max_s < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2

def coordinate_descent_pair(h, n, slopes, max_sweeps=100, deadline=None):
    """Coordinate descent on pairs."""

            break
        corr = np.correlate(h, 1-h, mode='full')
        improved = False

        np.random.shuffle(pairs)

            t_opt = optimize_pair(h, i, j, slopes, corr, 0)
            a = h[i]

                h[j] = a + h[j] - t_opt
                # Update corr

                corr += (t_opt - a) * (slopes[i] - slopes[j])
                # Update slopes (since h changed)
                # slopes[i,k] depends on h values, slopes[j,k] depends on h values
                # This is the expensive part...
                # Actually, let's recompute slopes periodically
                new_c = float(np.max(corr) * (2.0/n))
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    improved = True
    
    return h, best_c

    """
    For a batch of pairs, find optimal t.
    intercepts: (B, 2n-1) array
    diffs: (B, 2n-1) array
    t_lo: (B,) array
    t_hi: (B,) array
    Returns: (B,) array of optimal t values

    for i in range(n):
        # Case 1: i is the first index, contributes (1-h[i-k+n-1]) for k in [i, min(i+n-1, 2n-2)]
        k_lo1 = i  # = max(0, i) since i >= 0
        k_hi1 = min(i + n - 1, 2*n - 2)
        for k in range(k_lo1, k_hi1 + 1):
            idx = i - k + n - 1  # should be in [0, n-1]
            assert 0 <= idx < n
            slopes[i, k] += (1.0 - h[idx])
        
        # Case 2: i is the second index (i-k+n-1 = i means... wait no)

        # Contributes -h[j] = -h[i+k-n+1] for k in [max(n-1-i, 0), min(2n-2-i, 2n-2)]
