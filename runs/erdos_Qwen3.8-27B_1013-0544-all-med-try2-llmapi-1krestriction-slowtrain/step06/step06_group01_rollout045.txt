        T = t0 * ratio ** (it / max(1, n_iters))
        
        # Pick random zero and one
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta
        d = np.zeros(2 * n - 1)
        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range
        msk = (im >= 0) & (im < n)
        d[msk] += 1.0 - h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        d[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        d[msk] -= 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        d[msk] += h[jmn[msk]]
        d[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2*n-1:
            d[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2*n-1:
            d[k2] += 1.0
        
        new_corr = corr + d
        nc = new_corr.max() * dx
        
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(T, 1e-10)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c

def double_swap_search(h, n, deadline, seed=42, n_trials=100000):
    """Try random double swaps (swap two 0s with two 1s)."""
    rng = np.random.RandomState(seed)
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h)[::-1][:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        # Pick two random 0s and two random 1s
        i1, i2 = rng.choice(len(zeros), size=2, replace=False)
        i1, i2 = zeros[i1], zeros[i2]
        j1, j2 = rng.choice(len(ones), size=2, replace=False)
        j1, j2 = ones[j1], ones[j2]
        
        # Compute combined delta: swap i1→1, j1→0 AND i2→1, j2→0
        # Delta for (i1,j1) + Delta for (i2,j2) + cross terms
        # For simplicity, compute the new h and recompute correlation
        # (slower but correct for double swaps)
        h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        nc = new_corr.max() * dx
        if nc < cur_c - 1e-15:
            corr = new_corr
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            ones.remove(j1); ones.remove(j2)
            ones.append(i1); ones.append(i2)
            zeros.remove(i1); zeros.remove(i2)
            zeros.append(j1); zeros.append(j2)
        else:
            # Revert
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

            if cb_sa < cb2:
                cb2, hb2 = cb_sa, hb_sa
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh

    # Phase 6: Final intensive search
    p6 = dl - 8
    hb, cb = binary_descent(best_h, n, min(p6 - 20, t0 + 120), max_sweeps=10000, seed=seed + 500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < p6 - 20:
        hb_sa, cb_sa = sa_binary(best_h, n, p6 - 20, seed=seed + 600, t_high=6.0, t_low=0.2, n_iters=300000)
        if cb_sa < best_c:
            best_c = cb_sa
            best_h = hb_sa.copy()
    if time.time() < p6 - 5:
        hb_d, cb_d = binary_descent(best_h, n, p6, max_sweeps=2000, seed=seed + 601)
        if cb_d < best_c:
            best_c = cb_d
            best_h = hb_d.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
