import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_eq_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, n, alpha, maxiter, ftol, top_idx):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, top_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def frank_wolfe_phase(h, n, max_iters, top_m, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        grad = np.zeros(n)
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            grad += lag_gradient(h, n, lag)
        grad /= len(top_idx)
        v = np.zeros(n)
        idx = np.argsort(grad)
        v[idx[:n // 2]] = 1.0
        if it % 50 == 0:
            ts = np.linspace(0.0, 1.0, 30)
        else:
            ts = np.array([0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0])
        best_t = 0.0
        for t in ts:
            if t == 0.0:
                continue
            h_try = (1.0 - t) * h + t * v
            h_try = project_feasible(h_try, n)
            f_try = compute_c5(h_try, n)
            if f_try < best_c - 1e-14:
                best_c = f_try
                best_h = h_try.copy()
                best_t = t
        if best_t > 0:
            h = best_h.copy()
        else:
            if it > 10:
                break
    return best_h, best_c

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=150):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.3
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.002, trust * 0.92)
    return best_h, best_c

def gradient_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    lrs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for lr in lrs:
        if time.time() > deadline - 1:
            break
        for _ in range(200):
            if time.time() > deadline - 1:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            grad = lag_gradient(h, n, lag)
            grad_cent = grad - grad.mean()
            h_try = h - lr * grad_cent
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                h = h_try
            else:
                break
        if time.time() > deadline - 1:
            break
    # Pairwise refinement
    for d in [0.001, 0.0005, 0.0002]:
        if time.time() > deadline - 1:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        for _ in range(500):
            if time.time() > deadline - 1:
                break
            improved = False
            for i in range(n - m):
                j = i + (m if lag > 0 else 0)
                j2 = (i - m) if lag < 0 else (i + m)
                a = i
                b = i + m
                hi = h[a]
                hj = h[b]
                if hi > hj + 0.01:
                    delta = min(d, (hi - hj) * 0.3)
                    h[a] -= delta
                    h[b] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[a] += delta
                        h[b] -= delta
                elif hj > hi + 0.01:
                    delta = min(d, (hj - hi) * 0.3)
                    h[a] += delta
                    h[b] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[a] -= delta
                        h[b] += delta
            if not improved:
                break
    return h, best_c

def optimize_single(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Outer approximation
    if time.time() < deadline - 60:

        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    # Phase 2: Frank-Wolfe

    # Phase 3: SLSQP polish

            h_new = slsqp_solve(best_h, n, alpha, 5000, 1e-16, None)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 4: Active set SLSQP
    if time.time() < deadline - 20:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-4:]
        h_new = slsqp_solve(best_h, n, 500000, 5000, 1e-16, top_idx)

    # Phase 6: Final SLSQP
