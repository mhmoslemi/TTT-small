def swap_delta(h, i, p, n):
    """Compute the change in correlation when h[i] goes 0->1 and h[p] goes 1->0."""
    # r = m - (n-1), so m = n-1+r
    # valid r range: -(n-1) to n-1
    r = np.arange(-(n-1), n)
    m = r + (n-1)  # = np.arange(0, 2*n-1)
    
    # Term 1: h[p+r] - h[i+r] (where valid)
    idx1_p = p + r
    idx1_i = i + r
    term1 = np.zeros(2*n-1)
    msk_p = (idx1_p >= 0) & (idx1_p < n)
    msk_i = (idx1_i >= 0) & (idx1_i < n)
    term1[msk_p] += h[idx1_p[msk_p]]
    term1[msk_i] -= h[idx1_i[msk_i]]
    
    # Term 2: h[p-r] - h[i-r] (where valid)
    idx2_p = p - r
    idx2_i = i - r
    term2 = np.zeros(2*n-1)
    msk_p = (idx2_p >= 0) & (idx2_p < n)
    msk_i = (idx2_i >= 0) & (idx2_i < n)
    term2[msk_p] += h[idx2_p[msk_p]]
    term2[msk_i] -= h[idx2_i[msk_i]]
    
    # Term 3: sparse correction
    term3 = np.zeros(2*n-1)
    term3[n-1] += -2  # m = n-1
    kp = n-1 + (p-i)  # m = n-1+(p-i)
    ki = n-1 + (i-p)  # m = n-1+(i-p)
    if 0 <= kp < 2*n-1:
        term3[kp] += 1
    if 0 <= ki < 2*n-1:
        term3[ki] += 1
    
    return term1 + term2 + term3
