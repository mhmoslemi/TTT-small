import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

        stalled = 0
        while time.time() < deadline and stalled < 200:
            h = best_h.copy()
            c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            # Get top M lags (positive ones)
            c_pos = c[n:]  # Positive lags (k=0, 1, ..., n-1)
            # Actually c has 2n-1 entries, c[m] corresponds to lag m-(n-1)
            # Positive lags: m = n-1, n, ..., 2n-2, corresponding to k = 0, 1, ..., n-1
            # We want k >= 1 for meaningful shifts
            lags = np.arange(1, n)
            c_for_lags = c[n-1+lags]  # c for lags 1 to n-1
            top_idx = np.argsort(c_for_lags)[-top_m:][::-1]
            
            improved = False
            for ti in top_idx:
                k = lags[ti]
                # Try balancing this lag
                h_try = h.copy()
                for i in range(n - k):
                    diff = h_try[i] - h_try[i + k]
                    if abs(diff) < 1e-9:

                    d = -eps * np.sign(diff)
                    hi = h_try[i] + d

                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break

            if tot < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_balance(h, n, deadline, top_m=6):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    epsilons = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for eps in epsilons:
        if time.time() > deadline:
            break
        no_improve = 0
        while time.time() < deadline and no_improve < 40:
            h_cur = best_h.copy()
            c_arr = np.correlate(h_cur, 1 - h_cur, mode='full') * (2.0 / n)
            top_idx = np.argsort(c_arr)[-top_m:][::-1]
            improved = False
            for m in top_idx:
                lag = abs(int(m) - (n - 1))
                h_try = h_cur.copy()
                if lag == 0:
                    signs = np.sign(h_try - 0.5)
                    signs[signs == 0] = 1
                    h_try = np.clip(h_try + eps * signs, 0.0, 1.0)
                else:
                    for i in range(n - lag):
                        diff = h_try[i] - h_try[i + lag]
                        if abs(diff) < 1e-10:
                            continue
                        d = -eps * np.sign(diff)
                        hi = h_try[i] + d
                        hj = h_try[i + lag] - d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h_try[i] = hi
                            h_try[i + lag] = hj
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-13:
                    best_c = c_try
                    best_h = h_try.copy()
                    h_cur = best_h.copy()
                    improved = True
            no_improve = 0 if improved else no_improve + 1
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    for ds in [1, -1]:
                        dd = d * ds
                        hi = best_h[i] + dd
                        hj = best_h[j] - dd
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            oi, oj = best_h[i], best_h[j]
                            best_h[i], best_h[j] = hi, hj
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                best_h[i], best_h[j] = oi, oj
            if not improved:
                break
    return best_h, best_c
