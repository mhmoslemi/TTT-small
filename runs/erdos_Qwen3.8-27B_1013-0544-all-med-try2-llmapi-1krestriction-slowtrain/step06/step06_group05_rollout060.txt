    h = np.clip(h.copy(), 0.0, 1.0)

    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(c.max())
    
    # Try to reduce the max by moving mass
    # Strategy: for the lag k* that achieves the max,
    # move mass from positions i where h[i]=1 and h[i+k*]=0
    # to positions where this doesn't increase other lags

            if time.time() > deadline:
                return h, best_c
            improved = False
            # Focus on the active lags
            active = np.where(c > best_c - 1e-6)[0]
            for k_idx in active:
                k = k_idx - (n - 1)  # actual lag

                    return h, best_c

                for i in range(n):

                        return h, best_c

                    hi = h[i] - d
                    if hi < 0:

                    # Find where to move the mass
                    for j in range(n):
                        if j == i:
                            continue
                        hj = h[j] + d
                        if hj > 1:

def subgradient_step(h, n, step_size, n_active=10):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    # Find active lags
    active = np.where(c > m - step_size * 0.1)[0]
    active = active[:n_active]  # Take top n_active
    # Compute average subgradient
    lags = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for k in active:
        lag = lags[k]
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:]) / len(active)
            grad[lag:] += (-h[:n-lag]) / len(active)
        else:
            j = -lag
            grad[j:] += (1.0 - h[:n-j]) / len(active)
            grad[:n-j] += (-h[j:]) / len(active)
    grad *= dx
    # Remove mean to maintain sum constraint
    grad -= grad.mean()
    # Take step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sg(x, n, alpha):
    dx = 2.0/n
    c = np.correlate(x, 1-x, mode='full')*dx
    m = c.max()
    sh = alpha*(c-m)
    ex = np.exp(sh)
    val = m + np.log(ex.sum())/alpha
    sm = ex/ex.sum()
    g = np.zeros(n)
    lags = np.arange(2*n-1)-(n-1)
    for k in range(2*n-1):
        wk = sm[k]
        if wk < 1e-12: continue
        lag = lags[k]
        if lag >= 0:
            g[:n-lag] += wk*(1.0-x[lag:])
            g[lag:] += wk*(-x[:n-lag])
        else:
            j = -lag
            g[j:] += wk*(1.0-x[:n-j])
            g[:n-j] += wk*(-x[j:])
    g *= dx
    return val, g

    v, _ = _sg(x, n, alpha)
    return v

def _df(x, n, alpha):
    _, g = _sg(x, n, alpha)
    return g

def _c1(x, n): return x.sum()-n/2.0
def _c1j(x, n): return np.ones(n)

def slsqp(h, n, alpha, mi=5000):
    try:
        r = minimize(_f, h, jac=_df, args=(n,alpha), method='SLSQP',
                     bounds=[(0,1)]*n,

                     options={'maxiter':mi,'ftol':1e-18})
        return project_feasible(r.x, n)
    except: return h.copy()

                        c = compute_c5(h,n)
                        if c < best-1e-15: best=c; imp=True
            if not imp: break
    return h, best

This is a 1D convex optimization! We can solve it with ternary search in O(log(1/ε)) evaluations, each O(n).

```python
def optimal_delta(h, n, i, j, c, dx):
    """Find optimal delta for pair (i,j) to minimize max correlation."""
    # c_k(delta) = c[k] + delta * s_k
    # where s_k = dx * ((1-h[i+lag_k]-h[i-lag_k]) - (1-h[j+lag_k]-h[j-lag_k]))
    lags = np.arange(2*n-1) - (n-1)
    s = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = lags[k]
        # ∂c_k/∂h[i]
        di = 0.0
        if 0 <= i+lag < n: di += 1.0 - h[i+lag]
        if 0 <= i-lag < n: di -= h[i-lag]
        # ∂c_k/∂h[j]
        dj = 0.0
        if 0 <= j+lag < n: dj += 1.0 - h[j+lag]
        if 0 <= j-lag < n: dj -= h[j-lag]
        s[k] = dx * (di - dj)
    
    # Constraints: 0 ≤ h[i]+δ ≤ 1, 0 ≤ h[j]-δ ≤ 1
    # => -h[i] ≤ δ ≤ 1-h[i], and h[j]-1 ≤ δ ≤ h[j]
    delta_lo = max(-h[i], h[j]-1)
    delta_hi = min(1-h[i], h[j])
    
    if delta_lo >= delta_hi:
        return 0.0
    
    # Minimize max_k (c[k] + delta*s[k]) over delta ∈ [delta_lo, delta_hi]
    # This is a convex 1D problem. Use ternary search.
    for _ in range(60):  # ~1e-18 precision
        m1 = delta_lo + (delta_hi - delta_lo) / 3
        m2 = delta_hi - (delta_hi - delta_lo) / 3
        f1 = np.max(c + m1 * s)
        f2 = np.max(c + m2 * s)
        if f1 < f2:
            delta_hi = m2
        else:
            delta_lo = m1
    
    return (delta_lo + delta_hi) / 2
```

This is O(n) per evaluation of the ternary search, and we do 60 evaluations, so O(60n) = O(15000) per pair. For 31,000 pairs, that's 465 million operations per pass. At ~1ns per operation, that's 0.46 seconds per pass. Much better!

```python
def compute_sensitivity(h, n):
    """Compute S[i,k] = dc_k/dh[i] for all i, k."""
    dx = 2.0 / n
    lags = np.arange(2*n-1) - (n-1)
    S = np.zeros((n, 2*n-1))
    one_minus_h = 1.0 - h
    for i in range(n):
        for k in range(2*n-1):
            lag = lags[k]
            val = 0.0
            if 0 <= i+lag < n:
                val += one_minus_h[i+lag]
            if 0 <= i-lag < n:
                val -= h[i-lag]
            S[i, k] = dx * val
    return S
```

This is O(n²) = 62,500 operations, very fast.

def fast_pair_search(h, n, deadline, c, S, n_passes=500):

        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                s = S[i] - S[j]
                # Find optimal delta
                dlo = max(-h[i], h[j]-1)

                    m2 = dhi - (dhi-dlo)/3

                        dhi = m2
                    else:
                        dlo = m1

                    h[j] += delta

4. Every K moves, recompute S from scratch (O(n²))
5. Track the best solution
