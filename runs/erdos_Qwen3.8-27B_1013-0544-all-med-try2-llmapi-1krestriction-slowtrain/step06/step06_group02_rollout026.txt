def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def grad_focal(h, n, focal_idx, w, dx):
    """Compute gradient focusing on specific lags with weights w."""
    grad = np.zeros(n)

        mi = focal_idx[i]

            grad[k:] += wm * (-h[:nv])

        mi = int(focal_idx[i])
        wi = weights[i]

            grad[-k:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[-k:])

    m = c.max()

    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_all(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = grad_focal(x, n, np.arange(2 * n - 1), w)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            active = room > 1e-12
            if not active.any():
                break

        else:
            # Need to decrease sum: remove from elements with value > 0
            active = h > 1e-12
            if not active.any():
                break
            h[active] += diff / active.sum() * h[active]
