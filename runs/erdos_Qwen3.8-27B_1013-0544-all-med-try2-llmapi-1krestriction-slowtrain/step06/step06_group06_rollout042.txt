def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12: break
            h = h + diff / s * h

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    shifted = g - g_max
    exps = np.exp(alpha * shifted)
    return g_max + np.log(exps.sum()) / alpha

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    shifted = g - g_max
    exps = np.exp(alpha * shifted)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = n - 1 - k
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * dx * (1.0 - h[idx + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= wk * dx * h[idx2 - lag]
    return grad

def projected_gradient_descent(h, n, alpha, deadline, max_iter=2000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = smooth_gradient(h, n, alpha)
        # Projected gradient step with backtracking
        for _ in range(30):
            h_new = project_feasible(h - lr * grad, n)
            c_new = smooth_objective(h_new, n, alpha)
            c_old = smooth_objective(h, n, alpha)
            if c_new < c_old:
                h = h_new
                break
            lr *= 0.5
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def binary_targeted_swap(h, n, deadline, max_iters=10000):
    """For binary h, try targeted swaps to reduce max overlap."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        k_star = int(np.argmax(g))
        lag = n - 1 - k_star
        # Find i in ones where h[i+lag] is also 1 (these don't contribute to d_{k_star})

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:

            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    sh = g - g_max
    return g_max + np.log(np.exp(alpha * sh).sum()) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    sh = g - g_max
    exps = np.exp(alpha * sh)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = n - 1 - k
        lo = max(0, -lag)
        hi = min(n - 1, n - 1 - lag)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wk * dx * (1.0 - h[idx + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= wk * dx * h[idx2 - lag]
    return grad

def pgd_step(h, n, alpha, deadline):
    """One phase of projected gradient descent."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 2.0 / (alpha * 2.0 / n + 1e-6)  # initial learning rate
    for it in range(5000):
        if time.time() > deadline:
            break
        grad = smooth_grad(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Backtracking line search
        step = lr
        for _ in range(40):
            h_new = project_feasible(h - step * grad, n)
            obj_new = smooth_obj(h_new, n, alpha)
            obj_old = smooth_obj(h, n, alpha)
            if obj_new < obj_old - 1e-15:
                h = h_new
                break
            step *= 0.5
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    sh = g - g_max
    return g_max + np.log(np.exp(alpha * sh).sum()) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    sh = g - g_max
    exps = np.exp(alpha * sh)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = n - 1 - k
        lo = max(0, -lag)
        hi = min(n - 1, n - 1 - lag)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wk * dx * (1.0 - h[idx + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= wk * dx * h[idx2 - lag]
    return grad
