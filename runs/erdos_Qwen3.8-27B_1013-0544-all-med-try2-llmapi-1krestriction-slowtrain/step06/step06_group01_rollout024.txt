def fast_local_search(h, n, max_sweeps, deadline, seed):
    dx = 2.0 / n
    target = n // 2
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    # Work with binary h
    h = (h >= 0.5).astype(float)
    # Ensure exactly target ones
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) > target:
        # Remove excess ones
        excess = ones_idx[len(ones_idx)-target:]
        h[excess] = 0
    elif len(ones_idx) < target:
        # Add more ones
        needed = target - len(ones_idx)
        h[zeros_idx[:needed]] = 1
    
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    
    corr = np.correlate(h, 1-h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    # Precompute solo deltas
    # solo_delta[i] = change in corr when h[i] goes from current to flipped
    # For h[i]=0 → 1: adds (1-h) at positions i+k, subtracts h at positions i-k
    # For h[i]=1 → 0: subtracts (1-h) at positions i+k, adds h at positions i-k
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Compute solo deltas for all positions
        # For zero positions: delta_0[i] = solo effect of setting h[i] to 1
        # For one positions: delta_1[j] = solo effect of setting h[j] to 0
        
        # delta for h[i]: 0→1 (h[j] unchanged):
        # corr[n-1+k] += (1-h[i+k]) for valid i+k
        # corr[n-1+k] -= h[i-k] for valid i-k
        
        improved = False
        z_list = sorted(zeros_set)
        o_list = sorted(ones_set)
        
        # For efficiency, try random order
        rng.shuffle(z_list)
        rng.shuffle(o_list)
        
        # Limit to top candidates
        n_try = min(len(z_list), 50)
        m_try = min(len(o_list), 50)
        
        best_delta_max = cur_c + 1  # threshold for improvement
        best_pair = None
        best_new_corr = None
        
        for i in z_list[:n_try]:
            # Compute solo delta for setting h[i] to 1
            di = np.zeros(2*n-1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            di[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            di[msk2] -= h[imn[msk2]]
            
            for j in o_list[:m_try]:
                # Cross term
                d = di.copy()
                d[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:

                if 0 <= k2 < 2*n-1:
                    d[k2] += 1.0
                
                new_corr = corr + d
                nc = new_corr.max() * dx
                if nc < best_delta_max:
                    best_delta_max = nc
                    best_pair = (i, j)
                    best_new_corr = new_corr
        
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = 1.0, 0.0
            corr = best_new_corr
            cur_c = best_delta_max
            ones_set.discard(j)
            ones_set.add(i)
            zeros_set.discard(i)
            zeros_set.add(j)
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                h = h + diff / n
            else:
                h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                h = h - diff / n
            else:
                h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]

def fast_binary_local_search(h, n, deadline, max_sweeps=2000, kick_size=3, seed=42):

    cur_c = float(corr.max() * dx)

    total_sweeps = 0
    stuck_count = 0
    
    while time.time() < deadline and total_sweeps < max_sweeps * 10:
        # Local search phase
        improved = False
        sweeps_since_improve = 0
        
        for _ in range(max_sweeps):
            if time.time() > deadline:
                return best_h, best_c
            total_sweeps += 1
            
            z_list = list(zeros_set)
            o_list = list(ones_set)
            rng.shuffle(z_list)
            rng.shuffle(o_list)

            best_pair = None
            best_new_corr = None

                di = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)

                imn = i - m_range

                di[msk2] -= h[imn[msk2]]
                
                # Quick check: if max of (corr + di) is already >= cur_c, 
                # the swap can't help (unless cross term saves it, which is unlikely)
                # But let's not skip, just proceed
                
                for j in o_list[:m_try]:
                    d = di.copy()
                    d[n-1] -= 2.0
                    k1 = n-1 + (j-i)
                    if 0 <= k1 < 2*n-1:
                        d[k1] += 1.0
                    k2 = n-1 + (i-j)
                    if 0 <= k2 < 2*n-1:
                        d[k2] += 1.0

                    if nc < best_delta_max:
                        best_delta_max = nc
                        best_pair = (i, j)
                        best_new_corr = corr + d

                i, j = best_pair
                h[i], h[j] = 1.0, 0.0
                corr = best_new_corr
                cur_c = best_delta_max
                ones_set.discard(j)
                ones_set.add(i)
                zeros_set.discard(i)
                zeros_set.add(j)
                improved = True
                sweeps_since_improve = 0

                if sweeps_since_improve >= 3:

                j = rng.choice(o_list)
                h[i], h[j] = 1.0, 0.0
                ones_set.discard(j)
                ones_set.add(i)
                zeros_set.discard(i)
                zeros_set.add(j)
                z_list.remove(i)
                o_list.remove(j)
            
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = float(corr.max() * dx)
    
    return best_h, best_c

def gradient_optimize(h, n, deadline, maxiter=5000):
    """Optimize using gradient descent with log-sum-exp smoothing."""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline:
            break
        
        def obj(x):
            g = np.correlate(x, 1-x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            # Log-sum-exp for numerical stability
            sm = shifted.max()
            log_sum = sm + np.log(np.exp(shifted - sm).sum())
            return m + log_sum / alpha
        
        def grad(x):
            g = np.correlate(x, 1-x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            weights = np.exp(shifted - sm)
            weights /= weights.sum()
            # dg/dx: derivative of correlate(x, 1-x) w.r.t. x
            # corr[k] = sum_m x[m]*(1-x[m+k])
            # d corr[n-1+l] / d x[i] = (1-x[i+l]) - x[i-l] (for valid indices)
            # = delta_i (same as solo delta computed before)
            # grad = sum_l weights[l] * d(corr[l]*dx)/d(x)
            dcorr_dx = np.zeros(n)
            for l in range(2*n-1):
                if weights[l] < 1e-15:
                    continue
                k = l - (n-1)  # lag
                # d corr[l] / d x[i] = (1-x[i+k]) for valid i+k, minus x[i-k] for valid i-k
                # For all i: contribution from (1-x[i+k]) term
                im = np.arange(n) + k
                msk = (im >= 0) & (im < n)
                dcorr_dx[msk] += weights[l] * (1.0 - x[im[msk]])
                im2 = np.arange(n) - k
                msk2 = (im2 >= 0) & (im2 < n)
                dcorr_dx[msk2] -= weights[l] * x[im2[msk2]]
            return dcorr_dx * dx
        
        # Use projected gradient / L-BFGS-B with penalty
        # Since we can't handle equality constraints with L-BFGS-B, use penalty
        best_h_local = h.copy()
        best_c_local = compute_c5(h, n)
        
        for sub_iter in range(maxiter // 5):
            if time.time() > deadline:
                break
            
        # Actually, let's just use scipy minimize with SLSQP but fewer iterations
        # to keep it fast
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 200, 'ftol': 1e-16, 'maxls': 50})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        
        c = compute_c5(h, n)
        if c < best_c_local:
            best_c_local = c
            best_h_local = h.copy()
    
    return best_h_local, best_c_local

Let me also think about the simulated annealing approach for the binary domain:
- Current state: binary h with n/2 ones
- Move: swap a random (i,j) pair where h[i]=0, h[j]=1
- Cost: new max correlation
- Accept with probability exp(-(new_cost - old_cost) / T)
- T decreases over time

This can escape local optima that greedy search can't.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_solo_deltas(h, n, positions):
    """Compute solo delta for setting each position in 'positions' to 1 (from 0)."""
    m_range = np.arange(-(n-1), n)
    deltas = {}
    for i in positions:
        di = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        di[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk2 = (imn >= 0) & (imn < n)
        di[msk2] -= h[imn[msk2]]
        deltas[i] = di
    return deltas

def apply_swap_delta(corr, h, i, j, n, m_range, di):
    """Apply the swap: h[i] 0->1, h[j] 1->0. Returns new corr."""
    d = di.copy()
    d[n-1] -= 2.0
    k1 = n-1 + (j-i)
    if 0 <= k1 < 2*n-1:
        d[k1] += 1.0
    k2 = n-1 + (i-j)
    if 0 <= k2 < 2*n-1:
        d[k2] += 1.0
    return corr + d

def binary_local_search(h, n, deadline, max_iters=50000, seed=0):
    """Fast binary local search with adaptive kicks."""
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    h = binarize(h, n).copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    
    it = 0
    no_improve = 0
    kick_size = 2
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        z_list = list(zeros)
        o_list = list(ones)
        rng.shuffle(z_list)
        rng.shuffle(o_list)
        
        # Limit candidates for speed
        nz = min(len(z_list), 30)
        no = min(len(o_list), 30)
        
        best_nc = cur_c
        best_i = -1
        best_j = -1
        best_new_corr = None
        
        # Compute solo deltas for the top zeros
        zero_candidates = z_list[:nz]
        solo_deltas = {}
        for i in zero_candidates:
            di = np.zeros(2*n-1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            di[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            di[msk2] -= h[imn[msk2]]
            solo_deltas[i] = di
        
        one_candidates = o_list[:no]
        
        for i in zero_candidates:
            di = solo_deltas[i]
            for j in one_candidates:
                new_corr = apply_swap_delta(corr, h, i, j, n, m_range, di)
                nc = float(np.max(new_corr) * dx)
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_i = i
                    best_j = j
                    best_new_corr = new_corr
        
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            corr = best_new_corr
            cur_c = best_nc
            ones.discard(best_j)
            ones.add(best_i)
            zeros.discard(best_i)
            zeros.add(best_j)
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 5:
                # Kick
                k = min(kick_size + no_improve // 5, 12)
                for _ in range(k):
                    if not zeros or not ones:
                        break
                    i = rng.choice(list(zeros))
                    j = rng.choice(list(ones))
                    h[i], h[j] = 1.0, 0.0
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur_c = float(corr.max() * dx)
                no_improve = 0
                kick_size = min(kick_size + 1, 15)
    
    return best_h, best_c

def slsqp_optimize(h, n, deadline, maxiter=500):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in [1000, 10000, 100000, 500000]:
        if time.time() > deadline:
            break
        
        def obj(x):
            g = np.correlate(x, 1-x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n/2.0
        
        try:
            res = minimize(obj, best_h, method='SLSQP',
                         bounds=[(0.0, 1.0)]*n,
                         constraints=[{'type': 'eq', 'fun': cons_fun}],
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def slsqp_optimize(h, n, deadline, alphas=None, maxiter=400):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        def _obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        def _cons(x):
            return x.sum() - n / 2.0
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _cons}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn
                best_h = hn.copy()
        except Exception:
            pass
    return best_h, best_c

def binary_local_search(h_init, n, deadline, max_iters=100000, seed=0, nz_max=35, no_max=35):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    nz = 2 * n - 1
    rng = np.random.RandomState(seed)
    h = binarize(h_init, n).copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    it = 0
    no_imp = 0
    kick_sz = 2
    while time.time() < deadline and it < max_iters:
        it += 1
        zl = list(zeros)
        ol = list(ones)
        rng.shuffle(zl)
        rng.shuffle(ol)
        nz_cand = zl[:min(nz_max, len(zl))]
        no_cand = ol[:min(no_max, len(ol))]
        best_nc = cur_c + 1e-15
        best_i = -1
        best_j = -1
        best_nc_arr = None
        for i in nz_cand:
            di = np.zeros(nz)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            di[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            di[msk2] -= h[imn[msk2]]
            base = corr + di
            base[n - 1] -= 2.0
            for j in no_cand:
                k1 = n - 1 + (j - i)
                k2 = n - 1 + (i - j)
                nc = base.max()
                if 0 <= k1 < nz:
                    nc2 = base.copy()
                    nc2[k1] += 1.0
                    nc = max(nc, nc2.max())
                if 0 <= k2 < nz:
                    nc3 = base.copy()
                    nc3[k2] += 1.0
                    nc = max(nc, nc3.max())
                nc_val = nc * dx
                if nc_val < best_nc:
                    best_nc = nc_val
                    best_i = i
                    best_j = j
                    full_delta = di.copy()
                    full_delta[n - 1] -= 2.0
                    if 0 <= k1 < nz:
                        full_delta[k1] += 1.0
                    if 0 <= k2 < nz:
                        full_delta[k2] += 1.0
                    best_nc_arr = corr + full_delta
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            corr = best_nc_arr
            cur_c = best_nc
            ones.discard(best_j)
            ones.add(best_i)
            zeros.discard(best_i)
            zeros.add(best_j)
            no_imp = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            no_imp += 1
            if no_imp >= 4:
                k = min(kick_sz + no_imp // 4, 14)
                for _ in range(k):
                    if not zeros or not ones:
                        break
                    ii = rng.choice(tuple(zeros))
                    jj = rng.choice(tuple(ones))
                    h[ii], h[jj] = 1.0, 0.0
                    ones.discard(jj)
                    ones.add(ii)
                    zeros.discard(ii)
                    zeros.add(jj)
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = float(corr.max() * dx)
                no_imp = 0
                kick_sz = min(kick_sz + 1, 16)
    return best_h, best_c

def sa_binary_search(h_init, n, deadline, seed=0, iters=50000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = binarize(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    T0 = 0.005
    Tend = 0.0001
    it = 0
    while time.time() < deadline and it < iters:
        it += 1
        T = T0 * (Tend / T0) ** (it / iters)
        oi = list(np.where(h == 1)[0])
        zi = list(np.where(h == 0)[0])
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        h[i], h[j] = 1.0, 0.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c = float(new_corr.max() * dx)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            corr = new_corr
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_c
