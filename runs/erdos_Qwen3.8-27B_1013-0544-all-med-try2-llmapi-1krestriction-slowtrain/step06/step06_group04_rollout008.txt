Now, for the local search, the key insight is: if we move a small amount δ from h[a] to h[b], the change in f_m is:
δ * (∂f_m/∂h[b] - ∂f_m/∂h[a])

We want this to be negative (reduce f_m), so we need ∂f_m/∂h[b] < ∂f_m/∂h[a].

But we also need to check that this doesn't increase other critical lags! This is the key difficulty.

A practical approach:
1. Find the single worst lag m*
2. Compute the gradient g = ∂f_{m*}/∂h
3. Find the pair (a, b) that gives the largest negative change: g[b] - g[a] is most negative, with h[a] > 0 and h[b] < 1
4. Move a small amount from a to b
5. Check if the overall C_5 decreased

This is essentially a coordinate descent / steepest descent on the feasible set.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(c)
    shifted = alpha * (c - m)
    mx = np.max(shifted)
    s = shifted - mx
    exps = np.exp(s)
    lse = mx + np.log(np.sum(exps))
    val = m + lse / alpha
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if weights[i] > 1e-12:
            grad += weights[i] * grad_lag(x, n, i - (n - 1)) * dx
    return val, grad

def lbfgsb_optimize(h, n, alpha, mu=1e8, maxiter=20000):
    def pen_obj(x):
        v, _ = smooth_obj_grad(x, n, alpha)
        return v + mu * (np.sum(x) - n / 2.0) ** 2
    def pen_grad(x):
        _, g = smooth_obj_grad(x, n, alpha)
        g += 2.0 * mu * (np.sum(x) - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_obj, h, jac=pen_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_optimize(h, n, alpha, maxiter=5000):
    def obj(x):
        v, _ = smooth_obj_grad(x, n, alpha)
        return v
    def jac(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0), 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def cvx_step(h_curr, n, trust, n_active=20):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    active = np.argsort(c)[-n_active:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    if trust is not None:
        cons.append(cp.sum((h_var - h_curr) ** 2) <= trust ** 2)
    for ki in active:
        lag = ki - (n - 1)
        g = grad_lag(h_curr, n, lag)
        cons.append(c[ki] + dx * (g @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLOP, cp.CBC]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

def greedy_local(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in steps:

            lag = k_star - (n - 1)
            if lag == 0:

            g = grad_lag(h, n, lag) * dx
            h_pos = h > 1e-6
            h_neg = h < 1 - 1e-6
            if not h_pos.any() or not h_neg.any():
                break
            g_pos = np.where(h_pos, g, -np.inf)
            g_neg = np.where(h_neg, g, np.inf)
            a = int(np.argmax(g_pos))
            b = int(np.argmin(g_neg))
            if a == b or g[a] <= g[b] + 1e-12:
                break
            delta = min(d, h[a] * 0.4, (1.0 - h[b]) * 0.4)
            if delta < 1e-9:

            h[a] -= delta
            h[b] += delta
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[a] += delta
                h[b] -= delta
                h = project_feasible(h, n)
                break
    return h, best_c
