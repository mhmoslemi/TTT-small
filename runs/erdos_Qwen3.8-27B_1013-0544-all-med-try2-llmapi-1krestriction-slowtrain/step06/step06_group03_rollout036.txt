import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    s = alpha * (corr - m)
    sm = s.max()
    if sm > 500:
        obj = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
        w = np.exp(s - sm)
    else:
        obj = m + np.log(np.sum(np.exp(s))) / alpha
        w = np.exp(s)
    w /= w.sum()
    A = np.convolve(x, w, mode='full')[n - 1:2 * n - 1]
    B = np.correlate(w, x, mode='full')[n - 1:2 * n - 1][::-1]
    ones_n = np.ones(n)
    first = np.convolve(w, ones_n, mode='valid')
    grad = dx * (first - A - B)
    return obj, grad

def lbfgs_solve(h, n, alpha, maxiter=1000):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    res = minimize(c5_obj_grad, h, args=(n, alpha), method='L-BFGS-B',
                   jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-15})
    return project_feasible(res.x, n)

def multi_alpha_optimize(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = lbfgs_solve(h, n, a, maxiter=800)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    return h, best_c

    dx = 2.0 / n
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

            if time.time() > deadline:

            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def local_search_near(h, n, deadline, radius=60):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def spectral_init(n, rng, freq_scale=2.0):
    m = np.arange(n)
    k = 1 + rng.randint(0, 5)
    phase = rng.uniform(0, 2 * np.pi)
    g = np.zeros(n)
    for harmonic in range(1, 6):
        amp = rng.uniform(0.1, 0.4) / harmonic
        ph = rng.uniform(0, 2 * np.pi)
        g += amp * np.cos(2 * np.pi * harmonic * m / n + ph)
    g -= g.mean()
    g = np.clip(g * freq_scale, -0.5, 0.5)
    h = 0.5 + g
    return project_feasible(h, n)

def symmetric_init(n, rng):
    half = n // 2
    g = rng.uniform(-0.3, 0.3, half + 1)
    if n % 2 == 0:
        g_full = np.concatenate([g, g[:0:-1]])
    else:
        g_full = np.concatenate([g, g[:0:-1]])
    g_full[:half + 1] = g
    g_full[half + 1:] = g[:half][::-1]
    g_full -= g_full.mean()
    g_full = np.clip(g_full, -0.5, 0.5)
    h = 0.5 + g_full
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize the initial construction with multi-alpha L-BFGS
    alloc = min(120, (deadline - t0) * 0.2)
    h1, c1 = multi_alpha_optimize(best_h, n_init, t0 + alloc)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Try multiple resolutions
    for n_try in [128, 160, 200]:
        if time.time() > deadline - 120:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.15)
        if alloc < 20:
            break
        ht, ct = multi_alpha_optimize(h_interp, n_try, time.time() + alloc)
        if ct < best_c:
            best_c = ct
            best_h = ht

    # Phase 3: Multi-restart with diverse initial conditions
    n_best = len(best_h)
    remaining = deadline - time.time() - 80
    if remaining > 40:
        n_starts = max(3, int(remaining / 50))
        for r in range(n_starts):
            if time.time() > deadline - 70:
                break
            if r == 0:
                hp = best_h + rng.normal(0, 0.05, n_best)
            elif r == 1:
                hp = spectral_init(n_best, rng)
            elif r == 2:
                hp = symmetric_init(n_best, rng)
            else:
                scale = 0.03 + 0.1 * (r / max(1, n_starts - 3))
                hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 60) * 0.3)
            if alloc < 10:
                break
            ht, ct = multi_alpha_optimize(hp, n_best, time.time() + alloc)
            if ct < best_c:
                best_c = ct
                best_h = ht

    # Phase 4: Local search refinement
    n_best = len(best_h)
    if n_best <= 160:
        best_h, best_c = local_search(best_h, n_best, deadline - 30, n_passes=2)
    else:
        best_h, best_c = local_search_near(best_h, n_best, deadline - 30, radius=50)

    # Phase 5: Final L-BFGS polish with high alpha
    if time.time() < deadline - 20:
        ht = lbfgs_solve(best_h, n_best, 500000, maxiter=1500)
        c = compute_c5(ht, n_best)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Final fine local search
    if time.time() < deadline - 5:
        if n_best <= 160:
            best_h, best_c = local_search(best_h, n_best, deadline - 2, n_passes=2)
        else:
            best_h, best_c = local_search_near(best_h, n_best, deadline - 2, radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
