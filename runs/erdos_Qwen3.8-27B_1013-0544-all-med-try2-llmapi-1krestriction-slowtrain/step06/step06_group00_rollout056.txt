    # (1-h[j+k]) for j where j+k in [0,n-1], i.e., j in [-k, n-1-k]

        grad[:n-k] += (1 - h[k:n])

    # -h[j-k] for j where j-k in [0,n-1], i.e., j in [k, k+n-1]
    if k >= 0:
        grad[k:] -= h[:n-k]
    else:
        grad[:n+k] -= h[-k:]
    return grad

import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n-kk])
        grad[:n-kk] -= h[kk:]
    return grad

def proj_grad(h, n, deadline, initial_lr=0.05, top_m=3, min_lr=1e-6):
    """Projected gradient descent with adaptive step size."""
    h = h.copy()
    lr = initial_lr
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    stalled = 0
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        max_c = c.max()
        # Get top m lags
        idx = np.argsort(c)[::-1][:top_m]
        grad = np.zeros(n)
        for idx_i in idx:
            k = idx_i - (n - 1)
            grad += lag_gradient(h, k, n)
        grad /= top_m
        # Step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stalled = 0
            # Maybe increase lr slightly
            lr = min(lr * 1.05, 0.5)
        else:
            h = h_new  # still accept to avoid getting stuck
            lr *= 0.7
            stalled += 1
            if c_new > max_c * dx:
                # Revert
                pass  # keep h_new anyway for exploration
        if lr < min_lr:
            lr = initial_lr * 0.5
            # Perturb
            h = project_feasible(h + 0.01 * np.random.randn(n), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_fast(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 1e-14:
            h += diff / s * room
    else:
        s = h.sum()
        if s > 1e-14:
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n - kk])
        grad[:n - kk] -= h[kk:]
    return grad

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def proj_gradient(h, n, deadline, initial_lr=0.04, top_m=3, restart_every=5000):
    h = project_feasible(h, n).copy()
    lr = initial_lr
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    it = 0
    no_improve = 0
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        mc = float(c.max()) * dx
        if mc < best_c:
            best_c = mc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_idx = np.argpartition(c, -top_m)[-top_m:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        grad = np.zeros(n)
        for idx_i in top_idx:
            k = int(idx_i) - (n - 1)
            grad += lag_gradient(h, k, n)
        grad /= top_m
        h_new = h - lr * grad
        h_new = project_fast(h_new, n)
        c_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))) * dx
        if c_new <= mc + 1e-14:
            h = h_new
            lr = min(lr * 1.05, 0.25)
        else:
            lr *= 0.5
            if lr < 1e-7:
                h = project_feasible(h + 0.03 * np.random.randn(n), n)
                lr = initial_lr
                no_improve = 0
        it += 1
        if it % restart_every == 0 and no_improve > 2000:
            h = project_feasible(h + 0.02 * np.random.randn(n), n)
            lr = initial_lr
            no_improve = 0
    return best_h, best_c

def sparse_obj(x, fixed, free_idx, n):
    h_full = fixed.copy()
    h_full[free_idx] = x
    c = np.correlate(h_full, 1.0 - h_full, mode='full')
    return float(np.max(c)) * (2.0 / n)

def sparse_con(x, free_target):
    return x.sum() - free_target

def sparse_slsqp(h, n, n_free=12):
    not_at_boundary = (h > 0.005) & (h < 0.995)
    free_indices = np.where(not_at_boundary)[0]
    if len(free_indices) == 0:
        return h, compute_c5(h, n)
    if len(free_indices) > n_free:
        fracs = np.abs(h[free_indices] - 0.5)
        keep = np.argsort(fracs)[:n_free]
        free_indices = free_indices[keep]
    free_mask = np.zeros(n, dtype=bool)
    free_mask[free_indices] = True
    fixed = h.copy()
    for i in range(n):
        if not free_mask[i]:
            fixed[i] = 1.0 if h[i] >= 0.5 else 0.0
    fixed_sum = float(fixed.sum())
    free_target = n / 2.0 - fixed_sum
    nf = len(free_indices)
    x0 = h[free_indices].copy()
    bounds = [(0.0, 1.0)] * nf
    cons = [{'type': 'eq', 'fun': sparse_con, 'args': (free_target,)}]
    try:
        res = minimize(sparse_obj, x0, args=(fixed, free_indices, n), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': 8000, 'ftol': 1e-18})
        h_new = fixed.copy()
        h_new[free_indices] = res.x
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def binary_local_search(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
