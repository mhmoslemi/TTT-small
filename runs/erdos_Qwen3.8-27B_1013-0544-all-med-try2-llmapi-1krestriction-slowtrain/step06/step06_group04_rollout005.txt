import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    mid = 0.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - mid, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += (-h[:n-m])
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += (-h[m:])
    return grad

def smooth_max_obj_grad(h, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    emax = shifted.max()
    shifted = shifted - emax
    w = np.exp(shifted)
    wsum = w.sum()
    w = w / wsum
    obj = m + emax / alpha + np.log(wsum) / alpha
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-14:
            continue
        lag = top_idx[i] - (n - 1)
        grad += w[i] * lag_gradient(h, n, lag)
    return obj, grad

def slsqp_analytic(h0, n, top_idx, alpha, maxiter=3000, ftol=1e-14):
    def fun(x):
        return smooth_max_obj_grad(x, n, top_idx, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, n, top_idx, alpha)[1]
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def projected_grad(h0, n, deadline, top_m=20, alpha=5000, max_iter=5000):
    h = h0.copy()
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    prev_grad = None
    prev_x = None
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 150 == 0:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
        obj, grad = smooth_max_obj_grad(h, n, top_idx, alpha)
        gnrm = np.linalg.norm(grad)
        if gnrm < 1e-12:
            break
        if prev_grad is not None:
            s = h - prev_x
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            step = sy / yy if (yy > 1e-16 and sy > 1e-16) else 0.5 / (gnrm + 1e-10)
        else:
            step = 0.5 / (gnrm + 1e-10)
        step = np.clip(step, 1e-8, 1.0)
        h_new = project_feasible(h - step * grad, n)
        if np.max(np.abs(h_new - h)) < 1e-15:
            break
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        prev_grad = grad.copy()
        prev_x = h.copy()
        h = h_new
    return best_h, best_c

def local_search(h0, n, deadline):
    h = h0.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with growing active set
    for top_m in [10, 20, 30, 50]:
        if time.time() > deadline - 10:
            break
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-top_m:]
        alpha = 20000
        h_new = slsqp_analytic(best_h, n, top_idx, alpha, maxiter=2000, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 2: Projected gradient
    if time.time() < deadline - 20:
        pg_end = min(deadline - 15, time.time() + 60)
        h_pg, c_pg = projected_grad(best_h, n, pg_end, top_m=25, alpha=8000, max_iter=4000)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    # Phase 3: High-alpha SLSQP polish
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-40:]
        h_new = slsqp_analytic(best_h, n, top_idx, 200000, maxiter=4000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 4: Local search
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
