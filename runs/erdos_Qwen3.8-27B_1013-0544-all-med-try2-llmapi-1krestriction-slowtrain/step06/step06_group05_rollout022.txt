def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
