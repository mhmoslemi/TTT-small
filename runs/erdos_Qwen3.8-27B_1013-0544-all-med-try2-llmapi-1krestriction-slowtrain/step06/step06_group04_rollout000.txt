        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
        def _obj_act(x):
            dx = 2.0 / n
            cc = np.correlate(x, 1 - x, mode='full')
            act = cc[top_idx]
            m = act.max()
            sh = alpha * (act - m)
            mx = sh.max()
            if mx > 500:
                return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
        try:
            res = minimize(_obj_act, h_curr, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new
        except Exception:
            pass
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c_arr = compute_all_lags(h, n)
            top_lags = np.argsort(c_arr)[-3:]
            for k_star in top_lags:
                if time.time() > deadline:
                    break
                lag = int(k_star) - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                if lag > 0:
                    pairs = [(i, i + m) for i in range(n - m)]
                else:
                    pairs = [(i - m, i) for i in range(m, n)]
                # Sort pairs by gradient magnitude (top 100)
                if lag > 0:
                    diffs = h[:n-m] - h[m:]
                    order = np.argsort(-np.abs(diffs))[:100]
                else:
                    diffs = h[m:] - h[:n-m]
                    order = np.argsort(-np.abs(diffs))[:100]
                for idx in order:
                    if time.time() > deadline:
                        break
                    i, j = pairs[idx]
                    if lag > 0:
                        if h[i] > h[j]:
                            delta = min(d, (h[i]-h[j])*0.4, h[i]*0.5, (1-h[j])*0.5)
                        elif h[j] > h[i]:
                            delta = min(d, (h[j]-h[i])*0.4, h[j]*0.5, (1-h[i])*0.5)
                        else:
                            continue
                    else:
                        if h[j] > h[i]:
                            delta = min(d, (h[j]-h[i])*0.4, h[j]*0.5, (1-h[i])*0.5)
                        elif h[i] > h[j]:
                            delta = min(d, (h[i]-h[j])*0.4, h[i]*0.5, (1-h[j])*0.5)
                        else:
                            continue
                    if delta < 1e-9:
                        continue
                    if lag > 0:
                        if h[i] > h[j]:
                            h[i] -= delta; h[j] += delta
                        else:
                            h[i] += delta; h[j] -= delta
                    else:
                        if h[j] > h[i]:
                            h[i] += delta; h[j] -= delta
                        else:
                            h[i] -= delta; h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        if lag > 0:
                            if h[i] < h[j]:
                                h[i] += delta; h[j] -= delta
                            else:
                                h[i] -= delta; h[j] += delta
                        else:
                            if h[i] < h[j]:
                                h[i] -= delta; h[j] += delta
                            else:
                                h[i] += delta; h[j] -= delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: outer approximation
    if time.time() < deadline - 60:
        oo_dead = min(deadline - 60, time.time() + 120)
        h_oo, c_oo = outer_approx(best_h, n, oo_dead)
        if c_oo < best_c:
            best_c, best_h = c_oo, h_oo
    # Phase 2: SLSQP full smooth
    if time.time() < deadline - 30:
        for alpha in [50000, 200000]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_polish(best_h, n, alpha, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c, best_h = c_new, h_new
    # Phase 3: active set
    if time.time() < deadline - 20:
        as_dead = min(deadline - 20, time.time() + 60)
        h_as, c_as = active_slsqp(best_h, n, as_dead)
        if c_as < best_c:
            best_c, best_h = c_as, h_as
    # Phase 4: final SLSQP high alpha
    if time.time() < deadline - 10:
        h_new = slsqp_polish(best_h, n, 500000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    # Phase 5: local search
    if time.time() < deadline - 2:
        ls_dead = deadline - 1
        best_h, best_c = local_search(best_h, n, ls_dead)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    # Phase A: optimize at initial n
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    # Phase B: try nearby n values
    n_candidates = [200, 250, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    # Phase C: restarts at best n
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, size=n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_polish(best_h, n, 1000000, maxiter=6000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
