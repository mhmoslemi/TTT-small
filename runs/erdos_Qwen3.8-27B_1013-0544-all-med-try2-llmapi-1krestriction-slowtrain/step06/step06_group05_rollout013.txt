import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining in tiny amounts
        step = diff / n
        h = np.clip(h + step, 0, 1)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Log-sum-exp for stability
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + lse / alpha)

def sum_con_fun(x, n):
    return float(x.sum() - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lbfgsb_optimize(h, n, alpha, maxiter=5000):
    # L-BFGS-B with penalty for sum constraint
    def penalized_obj(x):
        base = smooth_obj(x, n, alpha)
        penalty = 1000.0 * (x.sum() - n/2.0)**2
        return base + penalty
    
    def penalized_grad(x):

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def _fc(x, n):
    return float(x.sum() - n / 2.0)

def _fj(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _fc, 'jac': _fj, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_round(h, n, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for a in [500, 2000, 10000, 50000, 200000, 1000000]:
        if time.time() > deadline - 3:
            break
        ht = slsqp_opt(best_h, n, a)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c, best_h = c, ht
    return best_h, best_c

def targeted_search(h, n, deadline, rng):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    step = 0.004
    plateau = 0
    top_k = 6
    n_lags = 4

    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        worst_idx = np.argsort(corr)[-n_lags:]

        improved = False
        for ki in worst_idx:
            if time.time() > deadline:
                break
            lag = ki - (n - 1)
            if lag == 0:
                continue
            if lag > 0:
                valid = np.arange(n - lag)
            else:
                valid = np.arange(-lag, n)
            if len(valid) == 0:
                continue
            contrib = h[valid] * (1.0 - h[valid + lag])
            top_pos = valid[np.argsort(contrib)[-top_k:]]

            for ii in range(len(top_pos)):
                if time.time() > deadline:
                    break
                i = int(top_pos[ii])
                j = i + lag
                for sgn in (1.0, -1.0):
                    d = step
                    hi = h[i] + sgn * d
                    hj = h[j] - sgn * d
                    if hi >= 0.0 and hi <= 1.0 and hj >= 0.0 and hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c, best_h = c, h.copy()
                            improved = True
                            plateau = 0
                        else:
                            h[i], h[j] = hi - sgn * d, hj + sgn * d

        if not improved:
            plateau += 1
            if plateau >= 300:
                step *= 0.8
                if step < 0.00005:
                    h = project_feasible(best_h + rng.normal(0, 0.005, n), n)
                    step = 0.003
                plateau = 0
        elif plateau > 0:
            plateau = max(0, plateau - 1)

    return best_h, best_c

def global_pair_search(h, n, deadline, deltas):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, n):
                    for sgn in (d, -d):
                        hi = h[i] + sgn
                        hj = h[j] - sgn
                        if hi >= 0.0 and hi <= 1.0 and hj >= 0.0 and hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c, best_h = c, h.copy()
                                improved = True
                            else:
                                h[i], h[j] = hi - sgn, hj + sgn
            if not improved:
                break
    return best_h, best_c
