    dx = 2.0 / n

        c = np.correlate(h, 1-h, mode='full')

            if time.time() > deadline:
                return h, best_c

            # Find top lags
            c_sorted = np.sort(c)[::-1]
            top_k = min(10, len(c))
            top_vals = c_sorted[:top_k]

            # c[m] = a[m] + h[j]*b[m] (after removing current h[j] contribution)

            candidates = [0.0, 1.0, h[j]]

                b_val = 0.0

                if 0 <= j-k < n:

                a_val = c[m] - h[j] * b_val

                # to minimize: if b_val > 0, t=0; if b_val < 0, t=1
                # but we want the max to be small, so this isn't quite right
                # Instead, find t where this lag equals the current max
                # a_val + t*b_val = best_c/dx
                if abs(b_val) > 1e-10:

                    if 0.0 <= t_star <= 1.0:
                        candidates.append(t_star)

            for delta in [-0.1, -0.05, -0.01, 0.01, 0.05, 0.1]:
                t = h[j] + delta
                if 0.0 <= t <= 1.0:
                    candidates.append(t)

            best_cv = best_c
            for t in candidates:
                t = np.clip(t, 0.0, 1.0)
                if abs(t - h[j]) < 1e-12:
                    continue
                h[j] = t
                # Quick recompute of max c
                # We can update c incrementally
                # c[m] changes by (t - old_h_j) * b[m]
                c_new_max = compute_c5(h, n)
                if c_new_max < best_cv:
                    best_cv = c_new_max
                    best_t = t
                else:
                    h[j] = best_t  # revert
            
            if h[j] != best_t:
                h[j] = best_t
            if best_cv < best_c:
                best_c = best_cv
                improved = True
        
        if not improved:
            break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline):
    """Fast coordinate descent with exact 1D subproblems (approximate)."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').copy()
    n_c = 2 * n - 1
    
    for _ in range(500):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            
            # Compute b_j: how c changes when h[j] changes
            b_j = np.zeros(n_c)
            for m in range(n_c):
                k = m - (n - 1)
                val = 0.0
                jp = j + k
                jm = j - k
                if 0 <= jp < n:
                    val += (1.0 - h[jp])
                if 0 <= jm < n:
                    val -= h[jm]
                b_j[m] = val
            
            # Generate candidates
            candidates = [0.0, 1.0]
            # From top lags
            top_idx = np.argsort(c)[::-1][:8]
            for m in top_idx:
                if abs(b_j[m]) > 1e-10:
                    t = h[j] + (best_c / dx - c[m]) / b_j[m]
                    if 0.0 < t < 1.0:
                        candidates.append(t)
            
            # Around current
            for d in (-0.05, -0.02, -0.01, 0.01, 0.02, 0.05):
                t = h[j] + d
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
            
            old = h[j]
            best_t = old
            for t in candidates:
                t = float(np.clip(t, 0.0, 1.0))
                if abs(t - old) < 1e-12:
                    continue
                dv = t - old
                c_max = float(np.max(c + dv * b_j)) * dx
                if c_max < best_c - 1e-14:
                    best_t = t
                    best_c = c_max
            
            if best_t != old:
                h[j] = best_t
                c += (best_t - old) * b_j
                improved = True
        
        if not improved:
            break
    return h, best_c

def lbfgs_smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    m = c_scaled.max()
    sh = alpha * (c_scaled - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
        w = w / w.sum()
    else:
        w = np.exp(sh)
        w = w / w.sum()
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def lbfgs_phase(h, n, deadline, max_iters=200):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        mu = 1.0
        try:
            res = minimize(lbfgs_smooth_obj, h, jac=lbfgs_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters * 20,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def perturb(h, n, rng, strength=0.1):
    """Random perturbation that maintains feasibility."""
    h_new = h.copy()
    # Change a random subset of coordinates
    n_change = max(5, n // 10)
    idx = rng.choice(n, n_change, replace=False)
    h_new[idx] = np.clip(h_new[idx] + rng.randn(n_change) * strength, 0, 1)
    return project_feasible(h_new, n)

        (lambda h: block_perturb(h, n_init, rng)),
        (lambda h: perturb(h, n_init, rng, 0.3)),
        (lambda h: perturb(h, n_init, rng, 0.15)),
        (lambda h: block_perturb(h, n_init, rng)),
    ]
    for r in range(len(perturbations)):
        if time.time() > deadline - 120:
            break
        h_pert = perturbations[r](best_h)
        remaining = deadline - 120 - time.time()
        if remaining < 20:
            break
        alloc_cd = min(50, remaining * 0.4)
        h_pert, c_pert = coord_descent(h_pert, n_init, time.time() + alloc_cd)
        if time.time() < deadline - 110:
            h_pert, c_pert = lbfgs_phase(h_pert, n_init, time.time() + min(25, deadline - time.time() - 105), max_iters=100)
        if time.time() < deadline - 100:
            h_pert, c_pert = coord_descent(h_pert, n_init, time.time() + min(15, deadline - time.time() - 95))
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
    # Phase 5: Try different n
    for n_try in [200, 350, 400, 150]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_t, c_t = coord_descent(h_interp, n_try, min(time.time() + 20, deadline - 50))
        if time.time() < deadline - 40:
            h_t, c_t = lbfgs_phase(h_t, n_try, time.time() + min(10, deadline - time.time() - 35), max_iters=80)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    # Phase 6: Upsample and refine
    if time.time() < deadline - 40:
        n_up = len(best_h) * 2
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, n_up)
        h_u, c_u = coord_descent(h_up, n_up, min(time.time() + 25, deadline - 30))
        if time.time() < deadline - 20:
            h_u, c_u = lbfgs_phase(h_u, n_up, time.time() + min(10, deadline - time.time() - 15), max_iters=80)
        if c_u < best_c:
            best_c = c_u
            best_h = h_u
    # Phase 7: Final polish
    if time.time() < deadline - 2:
        best_h, best_c = coord_descent(best_h, len(best_h), deadline - 1)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
