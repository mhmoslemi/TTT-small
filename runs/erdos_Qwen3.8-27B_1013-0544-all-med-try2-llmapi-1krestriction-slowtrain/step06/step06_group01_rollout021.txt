After: h[i]=1, h[j]=0: term = 1*(1-0) = 1
Change: +1

Similarly if j = m and i = m+k (i.e., i-j=k):
Before: h[j]=1, h[i]=0: term = 1*(1-0) = 1
After: h[j]=0, h[i]=1: term = 0*(1-1) = 0
Change: -1

So:
delta_corr[k] = (1-h[i+k]) - h[i-k] + h[j-k] - (1-h[j+k])
                + (1 if k=j-i else 0) - (1 if k=i-j else 0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def sa_binary(h_init, n, t_start, t_end, n_iter_max, deadline, seed=42):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = float(corr.max())
    best_c = cur_max * dx
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    m_range = np.arange(-(n-1), n)
    
    n_iter = min(n_iter_max, int((deadline - time.time()) * 5000))
    n_iter = max(n_iter, 1000)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Temperature
        frac = it / max(n_iter - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        # Propose swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta_corr
        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range
        
        delta = np.zeros(2*n - 1)
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        
        # Cross terms
        delta[n-1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        new_corr = corr + delta
        new_max = float(new_corr.max())
        delta_energy = new_max - cur_max
        
        if delta_energy < 0 or rng.uniform() < np.exp(-delta_energy / max(T, 1e-10)):

            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_max = new_max

            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
    
    return best_h, best_c

    best_h = h_init.copy()

            m = corr.max()
            sh = alpha * (corr - m)
            smax = sh.max()
            exp_sh = np.exp(sh - smax)
            p = exp_sh / exp_sh.sum()
            w = alpha * p  # length 2n-1
            
            # Compute gradient
            grad = np.zeros(n)
            for j in range(n):
                # First term: sum_m w[j-m+n-1] * (1-h[m])
                # k = j-m, so index into w is k+n-1 = j-m+n-1
                # m ranges 0..n-1, k = j-m ranges j-(n-1)..j
                lo = max(0, j - (n-1))

                if lo <= hi:
                    ms = np.arange(lo, hi+1)
                    ks = j - ms
                    grad[j] += np.sum(w[ks + n - 1] * (1 - h[ms]))
                
                # Second term: -sum_m w[m-j+n-1] * h[m]
                # k = m-j, so index into w is k+n-1 = m-j+n-1
                # m ranges 0..n-1, k = m-j ranges -j..n-1-j
                lo2 = max(0, j)
                hi2 = min(n-1, n-1+j)
                if lo2 <= hi2:
                    ms = np.arange(lo2, hi2+1)
                    ks = ms - j
                    grad[j] -= np.sum(w[ks + n - 1] * h[ms])
            
            # Projected gradient step
            h = h - lr * grad
            h = project_feasible(h, n)
            
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            
            lr *= decay
        
        # After smooth optimization, threshold to binary and re-optimize
        if time.time() < deadline:
            h_bin = (h >= 0.5).astype(float)
            # Fix to have exactly n/2 ones
            if h_bin.sum() > n/2:
                # Remove excess ones (those with lowest h values)
                vals = h[(h_bin == 1)]
                order = np.argsort(vals)
                excess = int(h_bin.sum() - n/2)
                to_remove = np.where(h_bin == 1)[0][order[:excess]]
                h_bin[to_remove] = 0
            elif h_bin.sum() < n/2:
                vals = h[(h_bin == 0)]
                order = np.argsort(vals)[::-1]
                deficit = int(n/2 - h_bin.sum())
                to_add = np.where(h_bin == 0)[0][order[:deficit]]
                h_bin[to_add] = 1
            c_bin = compute_c5(h_bin, n)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin.copy()
    
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def smooth_step(h, n, alpha, steps, deadline, lr_init):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(steps):

            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = corr.max()
        sh = alpha * (corr - m)
        smax = sh.max()
        exp_sh = np.exp(sh - smax)
        w = alpha * exp_sh / exp_sh.sum()
        grad = np.zeros(n)

            best_h = h.copy()
