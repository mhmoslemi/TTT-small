    c = np.correlate(h, 1-h, mode='full')  # length 2n-1, without dx
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # k=0 (index n-1 in c)
    grad += w[n-1] * dx * (1 - 2*h)
    
    # k > 0 (indices n to 2n-2)
    for k_idx in range(n, 2*n-1):
        k = k_idx - (n-1)
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        # For lag k: c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # ∂/∂h[j]: 
        #   as first factor (i=j): need j+k < n, contributes (1-h[j+k])*dx
        #   as second factor (i+k=j, i=j-k): need j-k >= 0, contributes -h[j-k]*dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Projected gradient with alpha annealing
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for a in alphas:
        if time.time() > deadline - 100:
            break
        h_opt, c_opt = proj_grad(best_h, n_init, a, deadline - 100, max_iter=6000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 2: Greedy local search
    if time.time() < deadline - 50:
        best_h, best_c = greedy_local_search(best_h, n_init, deadline - 50)
    # Phase 3: Try different resolutions
    x_grid = np.linspace(0, 2, n_init)
    for n_try in [150, 200, 300, 400]:
        if time.time() > deadline - 70:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_grid, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = proj_grad(h_interp, n_try, 5000, deadline - 70, max_iter=3000)
        if c_opt < best_c and time.time() < deadline - 25:
            h_opt, c_opt = greedy_local_search(h_opt, n_try, deadline - 25)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            x_grid = np.linspace(0, 2, n_try)
    # Phase 4: Perturbation restarts on best solution
    n = len(best_h)
    if time.time() < deadline - 40:
        for r in range(5):
            if time.time() > deadline - 20:
                break
            scale = 0.02 + 0.05 * (r / 4.0)
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = project_feasible(h_pert, n)
            h_opt, c_opt = proj_grad(h_pert, n, 50000, deadline - 20, max_iter=3000)
            if time.time() < deadline - 8:
                h_opt, c_opt = greedy_local_search(h_opt, n, deadline - 8)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 5: Final high-alpha polish + local search
    if time.time() < deadline - 5:
        n = len(best_h)
        h_opt, c_opt = proj_grad(best_h, n, 2000000, deadline - 3, max_iter=3000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        best_h, best_c = greedy_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
