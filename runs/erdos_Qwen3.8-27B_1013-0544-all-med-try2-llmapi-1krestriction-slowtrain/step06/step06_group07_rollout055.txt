    n = len(h_init)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: Differential evolution for global exploration
    if tl() > 250:
        bounds_list = [(0.0, 1.0)] * n
        beta_de = 30
        pen_de = 50.0
        try:
            de_res = differential_evolution(
                de_penalized_obj, bounds_list, args=(ts, beta_de, pen_de),
                seed=seed, maxiter=150, tol=1e-6, popsize=12,
                mutation=(0.4, 1.0), recombination=0.6, polish=False,
                init='latinhypercube')
            h_de = project_feasible(de_res.x, ts)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_h = h_de.copy()
                best_c5 = c5_de
        except Exception:
            pass

    # Phase 2: SLSQP continuation
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': jac_sum, 'args': (ts,)}]
    if tl() > 200:
        for beta in [10, 50, 200, 1000, 5000, 20000, 100000]:
            if tl() < 30:
                break
            for trial in range(4):
                if tl() < 20:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * (0.02 / (trial + 1))
                    x0 = project_feasible(x0, ts)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 6000, 'ftol': 1e-16})
                    h_new = project_feasible(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 3: Multi-worst subgradient with adaptive step
    if tl() > 80:
        h = best_h.copy()
        n_lags = 2 * n - 1
        k_top = 5
        for it in range(100000):
            if tl() < 30:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            P = corr * dx
            worst_idx = np.argsort(P)[-k_top:]
            grad = subgrad_multi(h, worst_idx, n)
            grad_norm = np.sqrt(np.sum(grad ** 2)) + 1e-15
            # Polyak-like step: scale by current gap
            current_c5_approx = np.max(P)
            gap = max(current_c5_approx - 0.3805, 0.001)
            lr = 1.0 * gap / (grad_norm ** 2 * n)
            lr = min(lr, 0.05)
            h = h - lr * grad
            h = project_feasible(h, ts)
            if it % 5000 == 4999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 4: Perturbation + re-optimize to escape local minima
    if tl() > 120:
        for perturb_scale in [0.03, 0.05, 0.08, 0.12]:
            if tl() < 30:
                break
            x0 = best_h + np.random.randn(n) * perturb_scale
            x0 = project_feasible(x0, ts)
            for beta in [500, 5000]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 5: Multi-resolution search
    if tl() > 80:
        for n_try in [50, 60, 80, 100, 120]:
            if tl() < 25:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_sum, 'jac': jac_sum, 'args': (ts_t,)}]
            for beta in [200, 2000, 20000]:
                if tl() < 10:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 6: Final high-precision polish
    n = len(best_h)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': jac_sum, 'args': (ts,)}]

    if tl() > 15:
        for beta in [100000, 500000]:
            if tl() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-18})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Quick coordinate descent polish
    if tl() > 10:
        h = best_h.copy()
        n = len(h)
        dx = 2.0 / n
        for sweep in range(200):
            if tl() < 3:
                break
            improved = False
            for i in range(n):
                # Try small adjustments to h[i]
                old_val = h[i]
                # Binary search for best h[i] in [0,1] keeping sum constraint
                # adjusting one other element
                for delta in [0.001, -0.001, 0.005, -0.005]:
                    h[i] = np.clip(old_val + delta, 0, 1)
                    # Adjust a neighbor to maintain sum
                    diff = h.sum() - ts
                    if abs(diff) > 1e-12:
                        j = (i + 1) % n
                        h[j] = np.clip(h[j] - diff, 0, 1)
                        if abs(h.sum() - ts) > 1e-8:
                            h[i] = old_val
                            h[j] = h[j] + diff
                            continue
                    c5_trial = compute_c5(h)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] = old_val + delta
                    h[j] = h[j]
                if improved:
                    break
            if not improved:
                break

    return best_h, best_c5, len(best_h)
