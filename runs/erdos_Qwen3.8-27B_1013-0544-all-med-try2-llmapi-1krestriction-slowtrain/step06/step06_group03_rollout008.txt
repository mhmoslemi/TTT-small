import numpy as np
from scipy.optimize import minimize
from cvxpy import cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
              break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_slp_matrix(h_cur, n):
    """Precompute the A matrix and b vector for the SLP.
    Returns A (shape (2n-1, n)), b (shape (2n-1,)), such that
    the constraint is A[m,:] @ h + b[m] <= t for each shift m.
    Shift m corresponds to row m + (n-1) in the matrix.
    """
    A = np.zeros((2*n - 1, n))
    b = np.zeros(2*n - 1)
    
    for m in range(-(n-1), n):
        row = m + n - 1
        if m >= 0:
            # f_m = sum_{n=0}^{n-1-m} h[n+m](1-h[n])
            # Linearized: coeff[j] @ h + const
            # coeff[m:] += 1
            # coeff[:n-m] -= h_cur[m:]
            # coeff[m:] -= h_cur[:n-m]
            # const = dot(h_cur[:n-m], h_cur[m:])
            A[row, m:] += 1.0
            A[row, :n-m] -= h_cur[m:]
            A[row, m:] -= h_cur[:n-m]
            b[row] = np.dot(h_cur[:n-m], h_cur[m:])
        else:
            p = -m
            # f_{-p} = sum_{j=0}^{n-1-p} h[j](1-h[j+p])
            # coeff[:n-p] += 1
            # coeff[:n-p] -= h_cur[p:]
            # coeff[p:] -= h_cur[:n-p]
            # const = dot(h_cur[:n-p], h_cur[p:])
            A[row, :n-p] += 1.0
            A[row, :n-p] -= h_cur[p:]
            A[row, p:] -= h_cur[:n-p]
            b[row] = np.dot(h_cur[:n-p], h_cur[p:])
    
    return A, b

def slp_iteration(h_cur, n, solver='CLARABEL', damping=1.0):
    """One SLP iteration."""
    A, b = build_slp_matrix(h_cur, n)
    dx = 2.0 / n
    A_scaled = A * dx  # scale to get the actual C5 contribution
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        A_scaled @ h_var + b * dx <= t
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver)
        if h_var.value is not None:
            h_new = damping * np.array(h_var.value).flatten() + (1 - damping) * h_cur
            return project_feasible(h_new, n)
    except Exception:
        pass
    
    # Try fallback solvers
    for s in ['SCS', 'ECOS', 'GLPK']:
        try:
            prob.solve(solver=s)
            if h_var.value is not None:
                h_new = damping * np.array(h_var.value).flatten() + (1 - damping) * h_cur
                return project_feasible(h_new, n)
        except Exception:
            continue
    
    return h_cur.copy()

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_range = range(i + 1, n)
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    j_range = range(lo, hi)
                for j in j_range:

from cvxpy import cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_slp_matrices(h_cur, n):
    A = np.zeros((2 * n - 1, n))
    b = np.zeros(2 * n - 1)
    for m in range(-(n - 1), n):
        row = m + n - 1
        if m >= 0:
            A[row, m:] += 1.0
            A[row, :n - m] -= h_cur[m:]
            A[row, m:] -= h_cur[:n - m]
            b[row] = np.dot(h_cur[:n - m], h_cur[m:])
        else:
            p = -m
            A[row, :n - p] += 1.0
            A[row, :n - p] -= h_cur[p:]
            A[row, p:] -= h_cur[:n - p]
            b[row] = np.dot(h_cur[:n - p], h_cur[p:])
    return A, b

def solve_lp(A, b, n, dx):
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    constraints.append(A * dx @ h_var + b * dx <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for s in ['CLARABEL', 'SCS', 'ECOS', 'GLPK']:
        try:
            prob.solve(solver=s)
            if h_var.value is not None:
                return np.array(h_var.value).flatten()
        except Exception:
            continue
    return None

    if h_lp is None:
        return h_cur.copy()
    h_new = damping * h_lp + (1.0 - damping) * h_cur
    return project_feasible(h_new, n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
