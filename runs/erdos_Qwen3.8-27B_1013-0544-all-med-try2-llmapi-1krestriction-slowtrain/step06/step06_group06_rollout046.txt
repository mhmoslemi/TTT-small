import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def build_coeff_matrix(n, v):
    """A[s, i] = v[j] where j = i + s - (n-1), if valid; else 0."""
    A = np.zeros((2*n-1, n))
    for s in range(2*n-1):
        offset = s - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        j_lo = i_lo + offset
        j_hi = i_hi + offset
        A[s, i_lo:i_hi] = v[j_lo:j_hi]
    return A

def solve_lp(n, fixed_vec, deadline=None):
    """Solve: min t s.t. correlate(x, fixed_vec, 'full')*dx <= t, sum(x)=n/2, 0<=x<=1"""
    dx = 2.0 / n
    A_corr = build_coeff_matrix(n, fixed_vec)
    A_corr *= dx  # scale to get the integral
    
    # Variables: [x_0, ..., x_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.hstack([A_corr, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        return res.x[:n].copy()
    return None

def alternating_lp(h_init, n, max_iters=50, deadline=None, relax=1.0):
    h = np.clip(h_init, 0, 1).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        # Step 1: Fix v=1-h, optimize h
        v = 1.0 - h
        h_new = solve_lp(n, v, deadline)
        if h_new is None:
            break
        h = (1.0 - relax) * h + relax * h_new
        h = np.clip(h, 0, 1)
        
        # Step 2: Fix h, optimize v (=1-h_new)
        v_new = solve_lp(n, h, deadline)
        if v_new is None:
            break
        h_from_v = 1.0 - v_new
        h = (1.0 - relax) * h + relax * h_from_v
        h = np.clip(h, 0, 1)
        
        # Project to satisfy sum constraint approximately
        # (LP already ensures this, but numerical errors might creep in)
        diff = n/2.0 - h.sum()
        if abs(diff) > 1e-10:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff / s * room if diff > 0 else diff / h.sum() * h
            h = np.clip(h, 0, 1)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_project(h, n):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_init, n, deadline, max_sweeps=200):
    h = binary_project(h_init, n)
    best_c = compute_c5(h, n)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(range(n)) - ones
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try moving each zero to one, paired with a one going to zero
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in sorted(ones):
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-12:
                    best_c = c
                    ones.remove(j)
                    ones.add(i)
                    zeros.remove(i)
                    zeros.add(j)
                    improved = True
                    break
                else:
                    h[i], h[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    
    return h, best_c

    c_obj[n] = 1.0
    A_ub = np.hstack([A_corr, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return np.clip(res.x[:n], 0.0, 1.0)
    return None

def alternating_lp(h_init, n, max_iters=500, deadline=None, relax=1.0):
    h = np.clip(h_init, 0.0, 1.0).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    prev_c = best_c
    stall = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        v = 1.0 - h
        h_new = solve_lp(n, v)
        if h_new is None:
            break
        h = (1.0 - relax) * h + relax * h_new
        h = project_feasible(h, n)
        v_new = solve_lp(n, h)
        if v_new is None:
            break
        h2 = 1.0 - v_new
        h = (1.0 - relax) * h + relax * h2
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50 and relax == 1.0:
                relax = 0.5
                stall = 0
        if abs(prev_c - c) < 1e-12 and it > 10:
            break
        prev_c = c
    return best_h, best_c

def subgradient_descent(h, n, deadline, max_iter=5000, lr_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_shifts(h, n)
        m = g.max()
        k_act = int(np.argmax(g))
        grad = get_subgradient(h, n, k_act)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f / ng
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < m - 1e-13:
                h = h_try
                found = True
                break
        if not found:
            lr *= 0.92
            if lr < 1e-12:
                break
        else:
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
    return best_h, best_c

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_init, n, deadline, max_sweeps=400):
    h = binary_project(h_init, n)
    best_c = compute_c5(h, n)
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_list = sorted(set(range(n)) - ones_set)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = None
            best_c_try = best_c
            for j in ones_set:
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_c_try:
                    best_c_try = c
                    best_j = j
                h[i], h[j] = 0.0, 1.0
            if best_j is not None:
                h[i], h[best_j] = 1.0, 0.0
                best_c = best_c_try
                ones_set.remove(best_j)
                ones_set.add(i)
                zeros_list.remove(i)
                zeros_list.append(best_j)
                zeros_list.sort()
                improved = True
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)

    # Phase 1: Alternating LP from parent (30% budget)
    p1 = t0 + 0.30 * (budget_s - 3)
    h, c = alternating_lp(best_h.copy(), n0, max_iters=500, deadline=p1, relax=1.0)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Subgradient refinement (12%)
    p2 = t0 + 0.42 * (budget_s - 3)
    h, c = subgradient_descent(best_h.copy(), n0, p2, max_iter=5000, lr_init=0.05)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Alternating LP with relaxation (13%)
    p3 = t0 + 0.55 * (budget_s - 3)
    h, c = alternating_lp(best_h.copy(), n0, max_iters=300, deadline=p3, relax=0.6)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 4: Random restarts with alternating LP (15%)
    p4 = t0 + 0.70 * (budget_s - 3)
    for trial in range(5):
        if time.time() > p4 - 10:
            break
        rem = p4 - time.time()
        if rem < 20:
            break
        x = np.linspace(0, 2, n0)
        if trial == 0:
            hh = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) * rng.choice([1, -1]), n0)
        elif trial == 1:
            hh = project_feasible(0.5 + 0.4 * np.cos(2 * np.pi * 2 * x / 2), n0)
        elif trial == 2:
            hh = project_feasible(rng.uniform(0.2, 0.8, n0), n0)
        elif trial == 3:
            hh = project_feasible(best_h + 0.15 * rng.randn(n0), n0)
        else:
            hh = project_feasible(np.interp(x, x, best_h) + 0.1 * rng.randn(n0), n0)
        hh, ch = alternating_lp(hh, n0, max_iters=200, deadline=time.time() + min(rem * 0.7, 80), relax=1.0)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()

    # Phase 5: Multi-resolution (13%)
    p5 = t0 + 0.83 * (budget_s - 3)
    n_cur = len(best_h)
    for nh in [150, 200, 300, 400]:
        if time.time() > p5 - 15:
            break
        rem = p5 - time.time()
        if rem < 15:
            break
        xo = np.linspace(0, 2, n_cur)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, ch = alternating_lp(hh, nh, max_iters=150, deadline=time.time() + min(rem * 0.8, 60), relax=1.0)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n_cur = nh

    # Phase 6: Binary local search (10%)
    p6 = deadline
    n_final = len(best_h)
    hb, cb = binary_local_search(best_h.copy(), n_final, p6, max_sweeps=500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Final subgradient polish
    if time.time() < deadline - 2:
        hf, cf = subgradient_descent(best_h.copy(), len(best_h), deadline, max_iter=2000, lr_init=0.01)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
