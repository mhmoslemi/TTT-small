def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad_multi(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k_worst in idx:
        lag = int(k_worst) - (n - 1)
        if lag >= 0:
            idx2 = np.arange(n - lag)
            grad[idx2] += (1.0 - h[idx2 + lag])
            grad[idx2 + lag] -= h[idx2]
        else:
            la = -lag
            idx2 = np.arange(n - la)
            grad[idx2 + la] += (1.0 - h[idx2])
            grad[idx2] -= h[idx2 + la]
    return grad / top_k

                    improved = True
                else:
                    h[i] = val
                    h = h2
