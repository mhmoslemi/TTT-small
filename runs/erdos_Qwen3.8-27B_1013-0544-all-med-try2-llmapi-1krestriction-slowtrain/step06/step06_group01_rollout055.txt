    mask_i_t = (idx_i_t >= 0) & (idx_i_t < n)
    # h[i-t] where 0<=i-t<n

    mask_it = (idx_it >= 0) & (idx_it < n)
    # h[j+t] where 0<=j+t<n
    idx_j_t = j + t
    mask_j_t = (idx_j_t >= 0) & (idx_j_t < n)
    # h[j-t] where 0<=j-t<n
    idx_jt = j - t
    mask_jt = (idx_jt >= 0) & (idx_jt < n)
    
    delta = np.zeros(2*n-1)
    delta[mask_i_t] += (1 - h[idx_i_t[mask_i_t]])
    delta[mask_it] -= h[idx_it[mask_it]]
    delta[mask_j_t] -= (1 - h[idx_j_t[mask_j_t]])
    delta[mask_jt] += h[idx_jt[mask_jt]]
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def swap_local_search(h, n, deadline, max_iters=50000, seed=42, stall_limit=300):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64).copy()
    cur_max = corr.max()
    best_h = h.copy()
    best_c = cur_max * dx
    t_arr = np.arange(-(n - 1), n)
    ones_list = np.where(h > 0.5)[0].copy()
    zeros_list = np.where(h <= 0.5)[0].copy()
    rng = np.random.RandomState(seed)
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        ni = len(zeros_list)
        nj = len(ones_list)
        if ni == 0 or nj == 0:
            break
        ii = zeros_list[:, None]
        ii_t = ii + t_arr[None, :]
        m1 = (ii_t >= 0) & (ii_t < n)
        delta_i0 = np.zeros((ni, 2 * n - 1))
        delta_i0[m1] += (1.0 - h[ii_t[m1]])
        ii_m_t = ii - t_arr[None, :]
        m2 = (ii_m_t >= 0) & (ii_m_t < n)
        delta_i0[m2] -= h[ii_m_t[m2]]
        jj = ones_list[:, None]
        jj_t = jj + t_arr[None, :]
        m3 = (jj_t >= 0) & (jj_t < n)
        delta_j1 = np.zeros((nj, 2 * n - 1))
        delta_j1[m3] -= (1.0 - h[jj_t[m3]])
        jj_m_t = jj - t_arr[None, :]
        m4 = (jj_m_t >= 0) & (jj_m_t < n)
        delta_j1[m4] += h[jj_m_t[m4]]
        best_val = cur_max + 1.0
        best_ii = -1
        best_jj = -1
        best_nc = None
        for a in range(ni):
            base = corr + delta_i0[a]
            nm = (base[None, :] + delta_j1).max(axis=1)
            b = int(np.argmin(nm))
            if nm[b] < best_val:
                best_val = nm[b]
                best_ii = a
                best_jj = b
                best_nc = base + delta_j1[b]
        if best_ii < 0 or best_val >= cur_max - 1e-15:
            no_improve += 1
            if no_improve > stall_limit:
                for _ in range(5):
                    a = rng.randint(ni)
                    b = rng.randint(nj)
                    h[zeros_list[a]], h[ones_list[b]] = 1.0, 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                cur_max = corr.max()
                ones_list = np.where(h > 0.5)[0].copy()
                zeros_list = np.where(h <= 0.5)[0].copy()
                no_improve = 0
            continue
        i_pos = zeros_list[best_ii]
        j_pos = ones_list[best_jj]
        h[i_pos] = 1.0
        h[j_pos] = 0.0
        corr = best_nc.copy()
        cur_max = best_val
        zeros_list = np.delete(zeros_list, best_ii)
        zeros_list = np.append(zeros_list, j_pos)
        ones_list = np.delete(ones_list, best_jj)
        ones_list = np.append(ones_list, i_pos)
        no_improve = 0
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        c = np.correlate(x, 1.0 - x, mode='full')
        dc = np.zeros(2 * n - 1)
        # subgradient of max approximation
        idx = np.argmax(g)
        t = idx - (n - 1)
        for i in range(n):
            s = 0.0
            if 0 <= i + t < n:
                s += (1.0 - x[i + t])
            if 0 <= i - t < n:
                s -= x[i - t]
            dc[i] = s
        gd = np.zeros(n)
        for i in range(n):
            gd[i] = (1.0 - x[i + t]) * (0 <= i + t < n) - x[i - t] * (0 <= i - t < n) if t != 0 else 0.0
        return gd * dx
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h
