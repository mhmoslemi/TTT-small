def gradient_binary_search(h_init, n, deadline, alpha=10000, n_iterations=100000, seed=42):

    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    temperature = 0.001
    cool_rate = 0.999
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break

        grad = compute_smoothed_gradient(h, n, alpha)

        # Find best one to flip to 0 (most positive gradient)
        if len(ones) == 0 or len(zeros) == 0:
            break

        g_ones = grad[ones]

        # and flip one with most positive grad to 0
        best_z = zeros[np.argmin(g_zeros)]

                best_h = h.copy()

            h[best_o] = 1.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = corr.max()
    ec = np.exp(alpha * (corr - mc))
    w = ec / ec.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        lo1 = max(0, -l)
        hi1 = min(n - 1, n - 1 - l)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx + l])
        lo2 = max(0, l)
        hi2 = min(n - 1, n - 1 + l)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= wk * h[idx2 - l]
    return grad
