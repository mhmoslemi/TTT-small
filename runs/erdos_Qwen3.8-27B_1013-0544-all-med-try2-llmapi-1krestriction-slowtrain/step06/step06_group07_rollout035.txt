```python
import numpy as np
from scipy.optimize import minimize, Bounds, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_maxlines(a_k, b_k):
    """Solve min_{x in [0,1]} max_k (a_k*x + b_k)"""
    def f(x):
        return float(np.max(a_k * x + b_k))
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return float(res.x)

def coord_descent(h, n, n_sweeps, rng):
    """Coordinate descent with exact 1D minimization."""
    dx = 2.0 / n
    target_sum = n / 2.0
    order = rng.permutation(n)
    
    for sweep in range(n_sweeps):
        if sweep % 10 == 0:
            order = rng.permutation(n)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for i in order:
            h_i = h[i]
            a_k = np.zeros(2 * n - 1)
            # For each lag k, find coefficient of h[i]
            # corr[k] = sum_j h[j] * (1 - h[j - k + n - 1])
            # h[i] appears as h[j] when j=i: coeff = (1 - h[i - k + n - 1]) if valid
            # h[i] appears as h[j - k + n - 1] when j - k + n - 1 = i, j = i + k - n + 1: coeff = -h[j]
            for k in range(2 * n - 1):
                j2 = i - k + n - 1
                if 0 <= j2 < n:
                    a_k[k] += (1.0 - h[j2])
                j1 = i + k - n + 1
                if 0 <= j1 < n:
                    a_k[k] -= h[j1]
            
            b_k = corr - a_k * h_i
            x_opt = solve_1d_maxlines(a_k, b_k)
            h[i] = x_opt
            corr = corr + a_k * (x_opt - h_i)
        
        h = project_feasible(h, target_sum)
    
    return h

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def sum_constraint_fun(x, target_sum):
    return x.sum() - target_sum

def sum_constraint_jac(x, target_sum):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    best_c5 = np.inf
    best_h = None

        # Get initial point

        elif has_parent:
            x_old = np.linspace(0, 2, parent_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, np.array(initial_h_values, dtype=float))
            h = project_feasible(h, target_sum)
        else:
            h = np.full(n, 0.5)

        n_sweeps = max(100, int(remaining() * 2))
        h = coord_descent(h, n, n_sweeps, rng)
        
        c5 = compute_c5(h)
        if c5 < best_c5:

            cons = [{'type': 'eq', 'fun': sum_constraint_fun, 'jac': sum_constraint_jac,
                     'args': (target_sum,)}]

                    if c5_new < best_c5:

                        best_h = h_new.copy()

        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_maxlines(a_k, b_k):
    tol = 1e-12
    f0 = np.max(b_k)
    active0 = np.where(b_k >= f0 - tol)[0]
    if np.min(a_k[active0]) >= 0:
        return 0.0
    f1 = np.max(a_k + b_k)
    active1 = np.where(a_k + b_k >= f1 - tol)[0]
    if np.max(a_k[active1]) <= 0:
        return 1.0
    x_lo, x_hi = 0.0, 1.0
    for _ in range(200):
        x_mid = (x_lo + x_hi) / 2.0
        vals = a_k * x_mid + b_k
        f_mid = np.max(vals)
        threshold = f_mid - tol * max(1.0, abs(f_mid))
        active = np.where(vals >= threshold)[0]
        a_min = np.min(a_k[active])
        a_max = np.max(a_k[active])
        if a_min <= 0.0 <= a_max:
            return x_mid
        elif a_min > 0.0:
            x_hi = x_mid
        else:
            x_lo = x_mid
        if x_hi - x_lo < 1e-14:
            break
    return (x_lo + x_hi) / 2.0


def coord_descent(h, n, n_sweeps, rng):
    target_sum = n / 2.0
    order = rng.permutation(n)
    for sweep in range(n_sweeps):
        if sweep % 15 == 0:
            order = rng.permutation(n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in order:
            h_i = h[i]
            a_k = np.zeros(2 * n - 1)
            a_k[i:i + n] += (1.0 - h)[::-1]
            a_k[n - 1 - i:2 * n - 1 - i] -= h
            b_k = corr - a_k * h_i
            x_opt = solve_1d_maxlines(a_k, b_k)
            h[i] = x_opt
            corr = corr + a_k * (x_opt - h_i)
        h = project_feasible(h, target_sum)
    return h


def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx


def sum_con_fun(x, ts):
    return x.sum() - ts


def sum_con_jac(x, ts):
    return np.ones(len(x))


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_c5 = np.inf
    best_h = None

    def remaining():
        return budget_s - (time.time() - t0)

    has_parent = 'initial_h_values' in globals() and initial_h_values is not None
    parent_h = np.array(initial_h_values, dtype=float).copy() if has_parent else None
    parent_n = len(parent_h) if has_parent else 0

    n_candidates = [50, 60, 70, 80, 90, 100, 120, 150]
    if has_parent:
        if parent_n in n_candidates:
            n_candidates.remove(parent_n)
        n_candidates = [parent_n] + n_candidates

    for n in n_candidates:
        if remaining() < 120:
            break
        dx = 2.0 / n
        target_sum = n / 2.0

        if has_parent and parent_n == n:
            h = project_feasible(parent_h.copy(), target_sum)
        elif has_parent:
            x_old = np.linspace(0, 2, parent_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, parent_h)
            h = project_feasible(h, target_sum)
        else:
            h = np.full(n, 0.5)

        # Main: coordinate descent with exact 1D subproblem
        n_sweeps = min(400, max(80, int(remaining() * 2.5)))
        h = coord_descent(h, n, n_sweeps, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Perturb and redo
        if remaining() > 80:
            for pert_scale in [0.08, 0.03]:
                if remaining() < 40:
                    break
                h2 = h + rng.randn(n) * pert_scale
                h2 = project_feasible(h2, target_sum)
                h2 = coord_descent(h2, n, 150, rng)
                c5_2 = compute_c5(h2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    h = h2.copy()

        # SLSQP polish
        if remaining() > 30:
            bounds = Bounds(np.zeros(n), np.ones(n))
            cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac,
                     'args': (target_sum,)}]
            for beta in [3000, 15000, 80000]:
                if remaining() < 15:
                    break
                try:
                    res = minimize(smooth_max_obj, h, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                except Exception:
                    pass

        # Additional coordinate descent after SLSQP
        if remaining() > 30:
            h = coord_descent(h, n, 100, rng)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
