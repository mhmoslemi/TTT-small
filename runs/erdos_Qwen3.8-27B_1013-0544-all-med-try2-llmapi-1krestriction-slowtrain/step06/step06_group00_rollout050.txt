   - When n = j: the term is h[j+k]*(1-h[j]), changes by +d*h[j+k] (if 0 ≤ j+k < n)

So: Δcorr[p] = d * [(1-h[i-k]) * I(0≤i-k<n) - (1-h[j-k]) * I(0≤j-k<n) - h[i+k]*I(0≤i+k<n) + h[j+k]*I(0≤j+k<n)]

Let me define:
A[p] = (1-h[i-k])*I(0≤i-k<n) - (1-h[j-k])*I(0≤j-k<n) - h[i+k]*I(0≤i+k<n) + h[j+k]*I(0≤j+k<n)

where k = p - (n-1).

Then corr_new[p] = corr[p] + d * A[p].

We want to minimize max_p corr_new[p] = max_p (corr[p] + d*A[p]) subject to:
- h[i]+d ∈ [0,1] → d ∈ [-h[i], 1-h[i]]
- h[j]-d ∈ [0,1] → d ∈ [h[j]-1, h[j]]
- Combined: d ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]

This is a 1D convex minimization (max of linear functions is convex). We can solve it with:
- Ternary search: ~50 iterations, each O(2n) to compute the max
- Or: find the breakpoint where the subgradient changes sign

For efficiency, I'll use a bisection on the subgradient:
- Subgradient of g(d) = max_p (corr[p] + d*A[p]) at d is: A[p*] where p* = argmax
- At the optimum, 0 is in the subgradient, meaning there exist p1, p2 with A[p1] < 0 and A[p2] > 0 (or A[p*] = 0)
- We can binary search for d where the "switching" happens

Actually, the simplest efficient approach: since g(d) is convex and piecewise linear, the minimum is at a point where two (or more) of the linear functions corr[p] + d*A[p] are tied for the maximum. We can find this by:

1. Evaluate g at d_lo and d_hi (the bounds)
2. If g(d_lo) ≤ g(d_hi), minimum is at d_lo or somewhere in [d_lo, d_hi]
3. Use the fact that the subgradient is A[argmax] to do bisection

    A = np.zeros(2*n-1)
    # (1-h[i-k]) for p in [i, n-1+i]: k = p-(n-1), so i-k = i-p+(n-1)
    p_lo, p_hi = i, n-1+i
    k_vals = np.arange(p_lo, p_hi+1) - (n-1)
    idx = i - k_vals  # = i - (p - (n-1)) = i + n - 1 - p
    A[p_lo:p_hi+1] += (1 - h[idx])
    
    # -h[i+k] for p in [n-1-i, 2n-2-i]: k = p-(n-1)

    k_vals = np.arange(p_lo, p_hi+1) - (n-1)

    A[p_lo:p_hi+1] -= h[idx]

    k_vals = np.arange(p_lo, p_hi+1) - (n-1)

    A[p_lo:p_hi+1] += h[idx]
    
    return A

4. Use the rate of change to estimate the next d
5. Iterate

This is essentially a subgradient method but with a smarter step size. It might converge faster than ternary search for this specific problem.

            if s < 1e-13: break
            h += diff / s * h

    # (1-h[i-k]) for p in [i, n-1+i]

    # i-k = i - (p-(n-1)) = i+n-1-p, p from i to n-1+i
    # = (i+n-1-i) to (i+n-1-(n-1+i)) = n-1 to 0

    # p from n-1-i to 2n-2-i
    # i+k = i + p - (n-1), p from n-1-i to 2n-2-i
    # = i+n-1-i to i+2n-2-i-n+1 = 0 to n-1

            f2 = np.max(corr + m2 * A)
            if f1 < f2:
                hi = m2
            else:

            cur_c = corr.max()
            if cur_c * dx < best_c:
                best_c = cur_c * dx
                best_h = h.copy()
        
        T *= cool

        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def continuous_anneal(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    idx_desc = np.arange(n - 1, -1, -1)
    idx_asc = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    T = 0.004
    cool = 0.99993
    scale = 4000.0
    for it in range(8000000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-9:
            continue
        A = compute_A(h, i, j, n, idx_desc, idx_asc)
        d_opt, f_opt = find_opt_d(corr, A, d_lo, d_hi)
        if f_opt <= cur_c / dx or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c / dx - f_opt) / T)):
            h[i] = np.clip(h[i] + d_opt, 0.0, 1.0)
            h[j] = np.clip(h[j] - d_opt, 0.0, 1.0)
            corr += d_opt * A
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary greedy + anneal (15%)
    p1 = t0 + 0.10 * total
    hb, cb = greed_binary(h0, n, p1, max_sweeps=150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    p1b = t0 + 0.15 * total
    hb, cb = binary_anneal(best_h, n, p1b, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Continuous pair-swap annealing (45%)
    p2 = t0 + 0.60 * total
    idx_desc = np.arange(n - 1, -1, -1)
    idx_asc = np.arange(n)
    h_cont = project_feasible(best_h, n)
    ha, ca = continuous_anneal(h_cont, n, p2, seed=seed)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Second round of continuous anneal with different params
    p2b = t0 + 0.70 * total
    ha2, ca2 = continuous_anneal(best_h, n, p2b, seed=seed + 1)
    if ca2 < best_c:
        best_c, best_h = ca2, ha2.copy()

    # Phase 3: SLSQP polish (20%)
    p3 = t0 + 0.90 * total
    h = best_h.copy()
    for alpha in [5000, 50000, 500000, 5000000, 50000000]:
        if time.time() > p3 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Final continuous polish (10%)
    p4 = dl - 2
    if time.time() < p4:
        ha3, ca3 = continuous_anneal(best_h, n, p4, seed=seed + 2)
        if ca3 < best_c:
            best_c, best_h = ca3, ha3.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
