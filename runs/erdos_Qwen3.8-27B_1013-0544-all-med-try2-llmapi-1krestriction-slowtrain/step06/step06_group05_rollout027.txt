- The minimum is at a "vertex" (intersection of two lines) or at an endpoint
- We can find it by: for each pair of lines, compute their intersection, check if it's in [0,1], evaluate g there
- But that's O(n²) pairs * O(n) evaluation = O(n³) which is too slow

Better: use the fact that the minimum of a convex PL function over [0,1] can be found by:
- Checking t=0 and t=1
- Finding where the "slope" changes from negative to positive
- The slope of g at t is in [min_{active} a_k, max_{active} a_k]
- We want the leftmost t where max_{active} a_k ≥ 0 (i.e., the slope is non-negative)

This can be done with a single pass:
1. Sort the lines by their value at t=0 (i.e., by b_k)
2. The upper envelope changes at intersection points
3. Find the point where the slope of the upper envelope crosses 0

Actually, the simplest efficient approach for the 1D problem:
- Binary search on t
- At each t, compute g(t) and the subgradient [min_active_a, max_active_a]
- If max_active_a < 0: minimum is to the left
- If min_active_a > 0: minimum is to the right
- Otherwise: t is optimal (or very close)

Each step is O(n), and we need O(log(1/ε)) steps. Total: O(n log(1/ε)).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-10)[0]
            if len(idx) > 0:
                h[idx] = np.minimum(h[idx] + diff / len(idx), 1.0)
        else:
            idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] = np.maximum(h[idx] + diff / len(idx), 0.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    scaled = alpha * c * dx
    scaled_max = scaled.max()
    exps = np.exp(scaled - scaled_max)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        pi = p[i]
        if abs(pi) < 1e-14:
            continue
        m_lo1 = max(0, k)
        m_hi1 = min(n - 1, k + n - 1)
        if m_lo1 <= m_hi1:
            idx = np.arange(m_lo1, m_hi1 + 1)
            grad[idx] += pi * (1.0 - h[idx - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, -k + n - 1)
        if m_lo2 <= m_hi2:
            idx = np.arange(m_lo2, m_hi2 + 1)
            grad[idx] += pi * (-h[idx + k])
    return grad * dx, c, p

def proj_grad(h, n, alpha, n_iter, deadline, lr0=0.5):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad, c, p = smooth_max_grad(h, n, alpha)
        lr = lr0 / (1.0 + 0.01 * it)
        h_new = h - lr * grad
        h = project_feasible(h_new, n)
        cv = compute_c5(h, n)
        if cv < best_c - 1e-15:
            best_c = cv
            best_h = h.copy()
    return best_h, best_c

def golden_1d(alphas, betas, n_iters=40):
    a_lb, a_ub = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = a_ub - gr * (a_ub - a_lb)
    c2 = a_lb + gr * (a_ub - a_lb)
    f1 = np.max(alphas * c1 + betas)
    f2 = np.max(alphas * c2 + betas)
    for _ in range(n_iters):
        if f1 < f2:
            a_ub = c2
            c2 = c1
            f2 = f1
            c1 = a_ub - gr * (a_ub - a_lb)
            f1 = np.max(alphas * c1 + betas)
        else:
            a_lb = c1
            c1 = c2
            f1 = f2
            c2 = a_lb + gr * (a_ub - a_lb)
            f2 = np.max(alphas * c2 + betas)
    return (a_lb + a_ub) / 2.0

def coord_descent(h, n, n_passes, deadline, mu=0.0):
    h = h.copy()
    c_full = np.correlate(h, 1 - h, mode='full')
    best_c = np.max(c_full) * (2.0 / n)
    best_h = h.copy()
    for _pass in range(n_passes):
        if time.time() > deadline:
            break

                return best_h, best_c
            old = h[i]
            alphas = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = idx - (n - 1)
                a = 0.0
                ik = i - k
                if 0 <= ik < n:
                    a += (1.0 - h[ik])
                ik2 = i + k
                if 0 <= ik2 < n:
                    a += (-h[ik2])
                alphas[idx] = a
            betas = c_full - alphas * old
            if mu > 0:
                S_oth = h.sum() - old
                gamma = n / 2.0 - S_oth
                a_lb, a_ub = 0.0, 1.0
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = a_ub - gr * (a_ub - a_lb)
                c2 = a_lb + gr * (a_ub - a_lb)
                f1 = np.max(alphas * c1 + betas) + mu * (c1 - gamma) ** 2
                f2 = np.max(alphas * c2 + betas) + mu * (c2 - gamma) ** 2
                for _ in range(40):
                    if f1 < f2:
                        a_ub = c2; c2 = c1; f2 = f1
                        c1 = a_ub - gr * (a_ub - a_lb)
                        f1 = np.max(alphas * c1 + betas) + mu * (c1 - gamma) ** 2
                    else:
                        a_lb = c1; c1 = c2; f1 = f2
                        c2 = a_lb + gr * (a_ub - a_lb)
                        f2 = np.max(alphas * c2 + betas) + mu * (c2 - gamma) ** 2
                t_opt = (a_lb + a_ub) / 2.0
            else:
                t_opt = golden_1d(alphas, betas, 35)
            delta = t_opt - old
            h[i] = t_opt
            c_full += delta * alphas
        c_full = np.correlate(h, 1 - h, mode='full')
        cv = np.max(c_full) * (2.0 / n)
        if cv < best_c - 1e-15:
            best_c = cv
            best_h = h.copy()
    return best_h, best_c

def binary_round(h, n):
    target = int(n / 2)
    h_copy = h.copy()
    # Sort by value, round top 'target' to 1
    order = np.argsort(-h_copy)
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb

        for i in range(2 * n - 1):

            pi = p[i]
            if abs(pi) < 1e-14:

            m_hi1 = min(n - 1, k + n - 1)
            if m_lo1 <= m_hi1:

                grad[idx] += pi * (1.0 - h[idx - k])
            m_lo2 = max(0, -k)
            m_hi2 = min(n - 1, -k + n - 1)
            if m_lo2 <= m_hi2:
                idx = np.arange(m_lo2, m_hi2 + 1)

        lr = 0.3 / (1.0 + 0.05 * it)

        h = project_feasible(h_new, n)
