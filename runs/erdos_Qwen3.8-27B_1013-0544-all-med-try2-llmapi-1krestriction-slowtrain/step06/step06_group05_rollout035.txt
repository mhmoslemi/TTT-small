import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_A(b, n):
    """Build correlation coefficient matrix. A[k,p] = b[p-k+(n-1)] if valid."""
    m = 2 * n - 1
    m_idx = np.arange(m)
    p_idx = np.arange(n)
    indices = p_idx[None, :] - m_idx[:, None] + (n - 1)
    mask = (indices >= 0) & (indices < n)
    A = np.zeros((m, n))
    A[mask] = b[indices[mask]]
    return A

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Fast local search using analytic correlation changes."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    h = h.copy()
    n_corr = 2 * n - 1
    rng = np.random.RandomState(42)

    for _round in range(10000):

            break
        n_rounds += 1

        best_c = c.max()

        order = rng.permutation(n)
        
        for ii in range(n):
            i = int(order[ii])
            if time.time() > deadline:

            for j in range(n):
                if j == i:
                    continue
                if j <= i:
                    continue
                
                d_max_pos = min(1.0 - h[i], h[j])
                d_max_neg = min(h[i], 1.0 - h[j])

                    continue

                k_vals = np.arange(n_corr)
                a = i + k_vals - (n - 1)
                b = j + k_vals - (n - 1)
                p = i - k_vals + (n - 1)
                q = j - k_vals + (n - 1)
                
                g = np.zeros(n_corr)
                ma = (a >= 0) & (a < n)
                mb = (b >= 0) & (b < n)

                mq = (q >= 0) & (q < n)
                g[ma] += (1.0 - h[a[ma]])
                g[mb] -= (1.0 - h[b[mb]])
                g[mp] -= h[p[mp]]
                g[mq] += h[q[mq]]

                if d_max_pos > tol:
                    candidates.append(d_max_pos * 0.7)
                    candidates.append(d_max_pos * 0.3)

                    candidates.append(-d_max_neg * 0.7)

                        ki = c_sorted_idx[ai]
                        kj = c_sorted_idx[bi]

                            if -d_max_neg <= dc <= d_max_pos:
                                candidates.append(float(dc))

                    h[j] -= best_d
                    improved = True
                    best_c = best_f

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_A(b, n):
    m = 2 * n - 1
    m_idx = np.arange(m)
    p_idx = np.arange(n)
    indices = p_idx[None, :] - m_idx[:, None] + (n - 1)
    mask = (indices >= 0) & (indices < n)
    A = np.zeros((m, n))
    A[mask] = b[indices[mask]]
    return A

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

            if abs(dg) > 1e-12:
                dc = (c[kj] - c[ki]) / dg
                if -d_max_neg <= dc <= d_max_pos:
                    candidates.append(float(dc))
    best_d = 0.0
    best_val = cur_max
    for d in candidates:
        d = float(np.clip(d, -d_max_neg, d_max_pos))
        if abs(d) < 1e-13:
            continue
        val = (c + d * g).max()
        if val < best_val - 1e-13:
            best_val = val
            best_d = d
    return best_d, best_val

def fast_local_search(h, n, deadline, tol=1e-11, seed=42):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    h = h.copy()
    n_corr = 2 * n - 1
    rng = np.random.RandomState(seed)
    for _round in range(100000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        best_c = c.max()
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return project_feasible(h, n), best_c
            for j in range(i + 1, n):
                d_max_pos = min(1.0 - h[i], h[j])
                d_max_neg = min(h[i], 1.0 - h[j])
                if d_max_pos < tol and d_max_neg < tol:
                    continue
                k_vals = np.arange(n_corr)
                a = i + k_vals - (n - 1)
                b = j + k_vals - (n - 1)
                p = i - k_vals + (n - 1)
                q = j - k_vals + (n - 1)
                g = np.zeros(n_corr)
                ma = (a >= 0) & (a < n)
                mb = (b >= 0) & (b < n)
                mp = (p >= 0) & (p < n)
                mq = (q >= 0) & (q < n)
                if ma.any():
                    g[ma] += (1.0 - h[a[ma]])
                if mb.any():
                    g[mb] -= (1.0 - h[b[mb]])
                if mp.any():
                    g[mp] -= h[p[mp]]
                if mq.any():
                    g[mq] += h[q[mq]]
                bd, bv = best_d_for_pair(c, g, d_max_pos, d_max_neg)
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    improved = True
                    best_c = bv
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def optimize_multi(h_start, n, deadline, seed=42):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # SLSQP phase
    for alpha in [1000, 10000, 100000, 1000000]:
        if time.time() > deadline - 20:
            break
        if time.time() > deadline - 5:
            break
        ht = slsqp_optimize(best_h, n, alpha, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    # Alternating LP
    if time.time() < deadline - 15:
        ht, ct = alternating_lp(best_h, n, deadline - 15)
        if ct < best_c:
            best_c = ct
            best_h = ht
    # Local search
    if time.time() < deadline - 3:
        best_h, best_c = fast_local_search(best_h, n, deadline - 1, seed=seed)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = None
    best_c = float('inf')
    x_init = np.linspace(0, 2, n_init)
    # Phase 1: Optimize at initial n
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_multi(h_init, n_init, t0 + alloc1, seed=seed)
    best_h, best_c = h_opt, c_opt
    # Phase 2: Try smaller n values
    for n_try in [150, 100]:
        if n_try >= n_init:
            continue
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time()
        alloc = min(150, remaining * 0.35)
        if alloc < 50:
            break
        h_interp = np.interp(np.linspace(0, 2, n_try), x_init, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_multi(h_interp, n_try, time.time() + alloc, seed=seed + n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: Restarts
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 55))
        scales = [0.02, 0.04, 0.06, 0.01]
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 35) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_multi(hp, n, time.time() + alloc, seed=seed + 100 + r)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 4: Final local search polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = fast_local_search(best_h, n, deadline - 1, seed=seed + 999)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
