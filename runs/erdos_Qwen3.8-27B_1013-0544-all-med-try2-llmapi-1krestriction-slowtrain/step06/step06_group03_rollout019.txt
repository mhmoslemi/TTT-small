def incremental_update(corr, h, n, i, j, d):
    """Update correlation after h[i]+=d, h[j]-=d. Returns new corr array."""
    corr = corr.copy()
    # For each lag k' (from -(n-1) to n-1), the index in 'full' mode is k' + (n-1)
    # Change at position p (lag k' = p - (n-1)):
    # δ * [(1-h[i+k']) - h[i-k'] - (1-h[j+k']) + h[j-k']]
    # where i+k', i-k', j+k', j-k' must be in [0, n-1]
    
    dx = 2.0 / n
    # We need to update corr for all lags where any of these terms are valid
    
    # Terms involving i:
    # (1-h[i+k']) valid when 0 <= i+k' < n, i.e., -i <= k' < n-i
    # h[i-k'] valid when 0 <= i-k' < n, i.e., i-n+1 < k' <= i, i.e., -(n-1-i) <= k' <= i... 
    
    # Let me just do it with a loop over the relevant range
    min_k = max(-(n-1), -i, i-n+1, -j, j-n+1)
    max_k = min(n-1, n-1-i, i, n-1-j, j)
    
    for k in range(max(min_k, -(n-1)), min(max_k, n-1) + 1):
        p = k + (n - 1)  # position in 'full' output
        delta = 0.0
        # (1-h[i+k])
        if 0 <= i + k < n:
            delta += (1 - h[i + k])
        # -h[i-k]
        if 0 <= i - k < n:
            delta -= h[i - k]
        # -(1-h[j+k])
        if 0 <= j + k < n:
            delta -= (1 - h[j + k])
        # +h[j-k]
        if 0 <= j - k < n:
            delta += h[j - k]
        corr[p] += d * delta
    
    return corr

def gradient_descent(h, n, deadline):
    dx = 2.0 / n
    corr = compute_correlate(h, n)
    best_c = float(corr.max())
    best_h = h.copy()
    
    lr = 0.1
    patience = 0
    
    while time.time() < deadline:
        k_star = int(corr.argmax())
        lag = k_star - (n - 1)
        
        # Compute gradient of f_{k_star} w.r.t. h
        grad = np.zeros(n)
        if lag >= 0:
            # f = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j]: from h[j]*(1-h[j+lag]) → (1-h[j+lag]) if j+lag < n
            #           from h[j-lag]*(1-h[j]) → -h[j-lag] if j-lag >= 0

            valid2 = np.arange(n) - lag >= 0

            grad[valid2] -= h[np.arange(n)[valid2] - l]
        
        grad *= dx

        j_dec = int(np.argmax(grad))
        j_inc = int(np.argmin(grad))

            break  # No improvement possible

        max_step = min(h[j_dec], 1.0 - h[j_inc])
        step = min(lr * max_step, max_step)

        h[j_inc] += step

        c = float(corr.max())
        
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            patience = 0
            lr = min(lr * 1.1, 0.5)
        else:
            patience += 1
            lr *= 0.9
            if patience > 100:
                break
    
    return best_h, best_c

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlate(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def incr_update(corr, h, n, i, j, d):
    kmin = max(-(n-1), -i, i-n+1, -j, j-n+1)
    kmax = min(n-1, n-1-i, i, n-1-j, j)
    for k in range(kmin, kmax + 1):
        p = k + n - 1
        delta = 0.0
        a = i + k
        if 0 <= a < n:
            delta += 1.0 - h[a]
        a = i - k
        if 0 <= a < n:
            delta -= h[a]
        a = j + k
        if 0 <= a < n:
            delta -= 1.0 - h[a]
        a = j - k
        if 0 <= a < n:
            delta += h[a]
        corr[p] += d * delta

def quick_delta(h, n, i, j, d, max_pos):
    k = max_pos - (n - 1)
    delta = 0.0
    a = i + k
    if 0 <= a < n:
        delta += 1.0 - h[a]
    a = i - k
    if 0 <= a < n:
        delta -= h[a]
    a = j + k
    if 0 <= a < n:
        delta -= 1.0 - h[a]
    a = j - k
    if 0 <= a < n:
        delta += h[a]
    return d * delta

def fast_local_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    corr = compute_correlate(h, n)
    best_c = float(corr.max())
    best_h = h.copy()
    mp = int(corr.argmax())
    for d in deltas:
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:
                    break
                hi = h[i]
                for j in range(n):
                    if i == j:
                        continue
                    if hi + d <= 1.0 and h[j] - d >= 0.0:
                        if quick_delta(h, n, i, j, d, mp) < 0:
                            nc = corr.copy()
                            incr_update(nc, h, n, i, j, d)
                            c2 = float(nc.max())
                            if c2 < best_c - 1e-15:
                                h[i] += d; h[j] -= d
                                corr = nc; best_c = c2
                                best_h = h.copy(); mp = int(corr.argmax())
                                progress = True
                    if hi - d >= 0.0 and h[j] + d <= 1.0:
                        if quick_delta(h, n, i, j, -d, mp) < 0:
                            nc = corr.copy()
                            incr_update(nc, h, n, i, j, -d)
                            c2 = float(nc.max())
                            if c2 < best_c - 1e-15:
                                h[i] -= d; h[j] += d
                                corr = nc; best_c = c2
                                best_h = h.copy(); mp = int(corr.argmax())
                                progress = True
    return best_h, best_c

def grad_search(h, n, deadline):
    dx = 2.0 / n
    corr = compute_correlate(h, n)
    best_c = float(corr.max())
    best_h = h.copy()
    lr = 0.15
    patience = 0
    while time.time() < deadline and patience < 300:
        mp = int(corr.argmax())
        lag = mp - (n - 1)
        grad = np.zeros(n)
        idx = np.arange(n)
        if lag >= 0:
            m = n - lag
            grad[:m] += 1.0 - h[lag:]
            grad[lag:] -= h[:m]
        else:
            l = -lag
            m = n - l
            grad[l:] += 1.0 - h[:m]
            grad[:m] -= h[l:]
        j_dec = int(np.argmax(grad))
        j_inc = int(np.argmin(grad))
        if grad[j_dec] + grad[j_inc] >= -1e-14:
            patience += 1
            continue
        mx = min(h[j_dec], 1.0 - h[j_inc])
        if mx < 1e-12:
            patience += 1
            continue
        step = min(lr, 1.0) * mx
        nd = -step
        nc = corr.copy()
        incr_update(nc, h, n, j_dec, j_inc, nd)
        c2 = float(nc.max())
        if c2 < best_c - 1e-15:
            h[j_dec] += nd
            h[j_inc] -= nd
            corr = nc; best_c = c2; best_h = h.copy()
            patience = 0
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.92
            patience += 1
    return best_h, best_c
