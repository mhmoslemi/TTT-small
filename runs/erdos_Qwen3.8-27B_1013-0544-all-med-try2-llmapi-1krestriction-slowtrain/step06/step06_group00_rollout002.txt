def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lbbf_optimize(h, n, alpha, top_k, maxiter=500, pen=5.0):
    dx = 2.0 / n
    target = n / 2.0
    nl = len(h)
    
    def obj_and_grad(x):
        o1 = 1.0 - x
        corr = np.correlate(x, o1, mode='full') * dx
        if top_k < len(corr):
            idx = np.argpartition(corr, -top_k)[-top_k:]
            vals = corr[idx]
        else:
            idx = np.arange(len(corr))
            vals = corr
        sm = vals.max()
        exps = np.exp(vals - sm)
        Z = exps.sum()
        logsum = sm + np.log(Z)
        obj = logsum / alpha + pen * (x.sum() - target) ** 2
        
        weights = exps * alpha / Z
        g = np.zeros(nl)
        for wi, ki in zip(weights, idx):
            k = ki - (nl - 1)
            if k >= 0:
                g[:nl-k] += wi * (1.0 - x[k:])
                g[k:] += wi * (-x[:nl-k])
            else:
                k2 = -k
                g[:nl-k2] += wi * (-x[k2:])
                g[k2:] += wi * (1.0 - x[:nl-k2])
