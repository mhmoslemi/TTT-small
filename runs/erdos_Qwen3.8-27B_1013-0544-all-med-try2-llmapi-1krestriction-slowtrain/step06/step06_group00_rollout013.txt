import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_lse(h, n, alpha, top_n=10):
    """Compute gradient of log-sum-exp approximation"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    shifted = alpha * corr - np.max(alpha * corr)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Only use top few for efficiency
    top_idx = np.argsort(weights)[::-1][:top_n]
    w_sum = weights[top_idx].sum()
    if w_sum < 1e-15:
        return np.zeros(n), corr
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        w = weights[idx] / w_sum
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]
    return grad, corr

def projected_gradient(h, n, deadline, alpha_start=100, alpha_end=500000, n_alphas=15, top_n=15):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    for alpha in alphas:
        for it in range(5000):
            if time.time() > deadline:
                return best_h, best_c
            grad, corr = grad_lse(h, n, alpha, top_n)
            # Project gradient: make sum zero, respect bounds
            g = grad.copy()
            # Zero out components where h is at bound and gradient pushes outward
            at_zero = h < 1e-10
            at_one = h > 1.0 - 1e-10
            g[at_zero & (g < 0)] = 0
            g[at_one & (g > 0)] = 0
            # Make sum zero
            g -= g.mean()
            # Further: if h is at 0, g should be >= 0; if h is at 1, g should be <= 0
            # After making sum zero, re-check
            # Actually, let's just do a step and project
            # Line search
            base_f = np.log(np.sum(np.exp(alpha * (corr - corr.max())))) / alpha + corr.max()
            best_t = 0.0
            best_f = base_f
            for t in [0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
                h_new = project_feasible(h - t * g, n)
                c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full') * dx)
                # Use LSE objective
                corr_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
                sm = alpha * corr_new - np.max(alpha * corr_new)
                f_new = np.max(alpha * corr_new) / alpha + np.log(np.exp(sm).sum()) / alpha
                if f_new < best_f:
                    best_f = f_new
                    best_t = t
                    h = h_new
            if best_t == 0.0:
                break
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_val(corr, alpha):
    a = alpha * corr
    m = a.max()
    return m / alpha + np.log(np.exp(a - m).sum()) / alpha

def lse_grad_from_corr(h, n, corr, alpha, n_active=25):
    a = alpha * corr
    m = a.max()
    p = np.exp(a - m)
    ps = p.sum()
    if ps < 1e-300:
        return np.zeros(n)
    p /= ps
    active = np.argsort(p)[::-1][:n_active]
    w_tot = p[active].sum()
    if w_tot < 1e-15:
        return np.zeros(n)
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - (n - 1)
        wt = p[idx] / w_tot
        if k >= 0:
            grad[k:] += wt * (1.0 - h[:n - k])
            grad[:n - k] -= wt * h[k:]
        else:
            kp = -k
            grad[:n - kp] += wt * (1.0 - h[kp:])
            grad[kp:] -= wt * h[:n - kp]
    return grad

def pgd(h_init, n, deadline, alphas, n_active=25, max_iter=30000):

                f1 = lse_val(corr_new, alpha)

                    corr = corr_new

def binary_steepest(h_init, n, deadline, max_rounds=150, rng=None):

    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n - 1), n)
    n_ones = n // 2

        best_val = cur_c

            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)

                delta[k2] += 1.0
