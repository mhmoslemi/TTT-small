from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def sa_binary(h_binary, n, max_iters, deadline, T0=1.0, T1=0.01):
    """Simulated annealing on binary vectors."""
    best_h = h_binary.copy()
    best_c = compute_c5(h_binary, n)
    cur_h = h_binary.copy()
    cur_c = best_c
    ones = set(np.where(cur_h == 1)[0].tolist())
    zeros = set(np.where(cur_h == 0)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / max_iters)
        # Pick random swap
        i = list(zeros)[np.random.randint(len(zeros))]
        j = list(ones)[np.random.randint(len(ones))]
        cur_h[i], cur_h[j] = 1.0, 0.0
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_c = new_c
            ones.remove(j); ones.add(i)
            zeros.remove(i); zeros.add(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 0.0, 1.0
    return best_h, best_c

def local_search_binary(h_binary, n, deadline, max_rounds=50):
    """Full local search: try all possible swaps."""
    best_h = h_binary.copy()
    best_c = compute_c5(best_h, n)
    ones = list(np.where(best_h == 1)[0])
    zeros = list(np.where(best_h == 0)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            best_swap = None
            best_new_c = best_c
            for j in ones:
                best_h[i], best_h[j] = 1.0, 0.0
                c = compute_c5(best_h, n)
                if c < best_new_c:

                    best_swap = j
                best_h[i], best_h[j] = 0.0, 1.0
            if best_swap is not None:

                ones.remove(best_swap); ones.append(i)
                zeros.remove(i)
                best_c = best_new_c
                improved = True
                break  # restart inner loop
        if not improved:

    return best_h, best_c

    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(n):
                if i == j:
                    continue
                if h[i] >= 1.0 or h[j] <= 0.0:
                    continue
                # Direction: increase h[i], decrease h[j]
                # g_k(δ) = g_k + δ * (dg_k_di - dg_k_dj)
                # Find δ* that minimizes max_k g_k(δ)

                # This requires knowing the correlation structure
                # For full correlation c[k] = sum_p h[p]*(1-h[p-k+n-1])
                # dc/dh[i] = (1-h[i-k+n-1]) if i-k+n-1 in [0,n-1]

                h[j] = h_old_j - eps

                h[j] = h_old_j

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_bin, n, deadline, max_rounds=200):
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        best_delta = 0.0
        best_swap = None
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                best_h[i], best_h[j] = 1.0, 0.0
                c = compute_c5(best_h, n)
                best_h[i], best_h[j] = 0.0, 1.0
                delta = best_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        if best_swap is None:
            break
        best_h[best_swap[0]], best_h[best_swap[1]] = 1.0, 0.0
        best_c = best_c - best_delta
    return best_h, best_c

def pair_coord_descent(h, n, deadline, max_rounds=500):
    best_h = h.copy()

    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx

            if time.time() > deadline:

                hi, hj = h[i], h[j]

                    continue
                eps = 1e-6
                h[i], h[j] = hi + eps, hj - eps
                g_plus = np.correlate(h, 1 - h, mode='full') * dx
                h[i], h[j] = hi, hj
                d = (g_plus - g) / eps
                d_min = -min(hi, 1.0 - hj)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-10:
                    continue
                a, b = d_min, d_max
                gr = (np.sqrt(5) - 1.0) / 2.0
                c1 = b - gr * (b - a)
                d1 = a + gr * (b - a)
                fc = (g + c1 * d).max()
                fd = (g + d1 * d).max()

def sa_binary(h0, n, max_iters, deadline, T0, T1):
    cur_h = h0.copy()
    best_h = h0.copy()
    cur_c = compute_c5(cur_h, n)
    best_c = cur_c
    rng = np.random.RandomState(42)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        cur_h[i], cur_h[j] = 1.0, 0.0
        new_c = compute_c5(cur_h, n)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 0.0, 1.0
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
