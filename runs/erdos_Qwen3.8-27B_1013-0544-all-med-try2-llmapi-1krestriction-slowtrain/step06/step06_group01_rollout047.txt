    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Vectorized binary search from initial
    p1 = t0 + 0.35 * (budget_s - 5)
    hb, cb = vbin_search(best_h, n, p1, max_sweeps=8000, n_restarts=5, seed=seed, multi_swap=2)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 2: SLSQP polish
    p2 = t0 + 0.50 * (budget_s - 5)
    hc = project_feasible(np.clip(best_h + 0.04 * rng.randn(n), 0, 1), n)
    for alpha in [5000, 20000, 100000, 500000]:
        if time.time() > p2:
            break
        hc = slsqp_smooth(hc, n, alpha, 1500, p2)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c = c
            best_h = hc.copy()
    # Phase 3: Binary search from SLSQP
    p3 = t0 + 0.65 * (budget_s - 5)
    hb, cb = vbin_search(best_h, n, p3, max_sweeps=5000, n_restarts=4, seed=seed + 50, multi_swap=3)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 4: Multi-resolution and diverse starts
    p4 = dl - 40
    starts = [best_h.copy()]
    gh = greedy_construct(n, seed=seed + 1)
    starts.append(gh)
    rh = np.zeros(n)
    ri = rng.permutation(n)
    rh[ri[:n // 2]] = 1.0
    starts.append(rh)
    for si, sh in enumerate(starts):
        if time.time() > p4 - 20:
            break
        hb, cb = vbin_search(sh, n, p4 - 10, max_sweeps=4000, n_restarts=3, seed=seed + 200 + si * 7, multi_swap=2)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Try other resolutions
    for nh in [200, 300, 350]:
        if time.time() > p4 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hb2, cb2 = vbin_search(hh, nh, p4, max_sweeps=2000, n_restarts=3, seed=seed + 500 + nh, multi_swap=2)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh
    # Phase 5: Final polish
    if time.time() < dl - 20:
        hp = slsqp_smooth(project_feasible(np.clip(best_h + 0.02 * rng.randn(n), 0, 1), n), n, 2000000, 2000, dl - 12)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()
        hb, cb = vbin_search(best_h, n, dl, max_sweeps=3000, n_restarts=3, seed=seed + 999, multi_swap=3)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
