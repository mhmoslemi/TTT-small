def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c_shift = alpha * (c - c.max())
