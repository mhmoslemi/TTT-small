            improved = False
            lags = compute_all_lags(h, n)
            k_star = int(np.argmax(lags))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            # Try to reduce the overlap at the critical lag
            # For lag > 0: overlap = sum h[i+m]*(1-h[i])
            # To reduce: decrease h[i+m] where h[i] is small, or increase h[i] where h[i+m] is large
            # Simple heuristic: swap h[i] and h[i+m] if it helps
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] == h[j]:
                    continue
                # Try swapping
                h_swap = h.copy()
                h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
                h_swap = project_feasible(h_swap, n)
                c_new = compute_c5(h_swap, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = h_swap
                    improved = True
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Subgradient method
    if time.time() < deadline - 30:
        h_sg, c_sg = subgradient_method(h, n, n_iter=300, top_k=5, alpha=500, lr_init=0.02)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    # Phase 2: SLSQP with smooth max
    if time.time() < deadline - 20:
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_with_grad(best_h, n, alpha, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 3: Local search
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, deadline - 2, delta_init=0.01)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
