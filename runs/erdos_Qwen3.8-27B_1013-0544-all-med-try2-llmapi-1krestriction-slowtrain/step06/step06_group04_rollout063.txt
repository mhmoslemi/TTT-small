import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, n, m):
    """Compute gradient of c_m with respect to h, where m is the lag (can be negative)."""
    grad = np.zeros(n)
    dx = 2.0 / n
    if m == 0:
        # c_0 = sum h[i]*(1-h[i])
        grad = dx * (1 - 2*h)
        return grad
    if m > 0:
        # c_m = sum_{t=0}^{n-1-m} h[t+m]*(1-h[t])
        # d/dh[j] = dx * [(1-h[j-m]) if m<=j<=n-1] + dx * [-h[j+m] if 0<=j<=n-1-m]
        # First term: for j in [m, n-1], add (1-h[j-m])
        grad[m:] += (1.0 - h[:n-m])
        # Second term: for j in [0, n-1-m], subtract h[j+m]
        grad[:n-m] -= h[m:]
        return dx * grad
    else:
        # m < 0, let m' = -m
        # c_{-m'} = sum_{t=0}^{n-1-m'} h[t]*(1-h[t+m'])
        # d/dh[j] = dx * [(1-h[j+m']) if 0<=j<=n-1-m'] + dx * [-h[j-m'] if m'<=j<=n-1]
        mp = -m
        grad[:n-mp] += (1.0 - h[mp:])
        grad[mp:] -= h[:n-mp]
        return dx * grad

def projected_subgradient(h, n, max_iters=10000, deadline=None):
    """Subgradient method for min max_k c_k(h)"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    # Learn rate schedule
    # ||∇f|| <= L where L ~ 4*sqrt(2)/sqrt(n) * sqrt(n_active)
    # Use step_t = alpha / sqrt(t+1)
    alpha = 0.5
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h_curr.copy()
        
        # Find active set
        threshold = cmax - 1e-6
        active = np.where(c >= threshold)[0]
        
        # Subgradient = average of gradients of active lags
        subgrad = np.zeros(n)
        for idx in active:
            m = idx - (n - 1)
            subgrad += compute_lag_gradient(h_curr, n, m)
        subgrad /= len(active)
        
        # Step size
        step = alpha / np.sqrt(it + 1)
        
        # Projected step
        h_trial = h_curr - step * subgrad
        h_trial = project_feasible(h_trial, n)
        
        h_curr = h_trial
        # Decay alpha
        alpha *= 0.999
    
    return best_h, best_c

def subgradient_method(h, n, deadline, max_iters=50000, alpha0=0.05):
    """Projected subgradient descent for min max_k c_k(h)."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h_curr.copy()
        
        # Active set: lags within small tolerance of max
        tol = max(1e-8, 0.001 * cmax)
        active = np.where(c >= cmax - tol)[0]
        
        # Subgradient
        sg = np.zeros(n)
        for idx in active:
            m = idx - (n - 1)
            sg += lag_gradient(h_curr, n, m)
        sg /= len(active)
        
        # Step size
        alpha = alpha0 / np.sqrt(it + 1)

        h_trial = h_curr - alpha * sg
        h_curr = project_feasible(h_trial, n)

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, m):
    dx = 2.0 / n
    g = np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    if m > 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        mp = -m
        g[:n - mp] += (1.0 - h[mp:])
        g[mp:] -= h[:n - mp]
    return dx * g

def subgradient_method(h, n, deadline, max_iters=100000, alpha0=0.03):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h_curr.copy()
        tol = max(1e-9, 0.005 * cmax)
        active = np.where(c >= cmax - tol)[0]
        sg = np.zeros(n)
        for idx in active:
            m = idx - (n - 1)
            sg += lag_gradient(h_curr, n, m)
        sg /= len(active)
        alpha = alpha0 / np.sqrt(it + 1)
        h_curr = project_feasible(h_curr - alpha * sg, n)
    return best_h, best_c

def _smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_at(h, n, alpha, top_idx, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def outer_lp_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.nnls.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag) / dx
        cons.append(c[k] + (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=40):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_lp_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def focused_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        for _round in range(80):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            m_star = k_star - (n - 1)
            if m_star == 0:
                break
            if m_star > 0:
                a_idx = np.arange(n - m_star)
                b_idx = a_idx + m_star
                contrib = h[b_idx] * (1.0 - h[a_idx])
            else:
                mp = -m_star
                a_idx = np.arange(n - mp)
                b_idx = a_idx + mp
                contrib = h[a_idx] * (1.0 - h[b_idx])
            top = np.argsort(contrib)[::-1][:30]
            any_improved = False
            for t in top:
                if time.time() > deadline:
                    break
                a, b = a_idx[t], b_idx[t]
                if m_star > 0:
                    if h[b] > h[a]:
                        delta = min(d, (h[b] - h[a]) * 0.35)
                        h[b] -= delta
                        h[a] += delta
                    else:
                        delta = min(d, (h[a] - h[b]) * 0.35)
                        h[a] -= delta
                        h[b] += delta
                else:
                    if h[a] > h[b]:
                        delta = min(d, (h[a] - h[b]) * 0.35)
                        h[a] -= delta
                        h[b] += delta
                    else:
                        delta = min(d, (h[b] - h[a]) * 0.35)
                        h[b] -= delta
                        h[a] += delta
                h = np.clip(h, 0, 1)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    any_improved = True
            if not any_improved:
                h = best_h.copy()
                break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: outer approximation
    if time.time() < deadline - 20:
        oob = min(50, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_approx(best_h, n, time.time() + oob)
        if c_oo < best_c:
            best_c, best_h = c_oo, h_oo
    # Phase 2: subgradient
    if time.time() < deadline - 15:
        sg_end = min(40, (deadline - time.time()) * 0.3)
        h_sg, c_sg = subgradient_method(best_h, n, time.time() + sg_end,
                                        max_iters=80000, alpha0=0.02)
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg
    # Phase 3: SLSQP
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top = np.argsort(c)[-4:]
        for alpha in [20000, 100000]:
            if time.time() > deadline - 8:
                break
            h_new = slsqp_at(best_h, n, alpha, top, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
            c = compute_all_lags(best_h, n)
            top = np.argsort(c)[-4:]
    # Phase 4: local search
    if time.time() < deadline - 3:
        h_new, c_new = focused_local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    # Main optimization at n_init
    alloc1 = min(350, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_at_n(best_h, n_init, t0 + alloc1)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try higher resolutions
    for n_try in [400, 500, 200]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 40:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_old, best_h)
        h_start = project_feasible(h_start, n_try)
        alloc = min(100, remaining * 0.35)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    # Random restarts on best resolution
    n = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = max(1, int((deadline - time.time() - 30) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 30) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top = np.argsort(c)[-3:]
        h_new = slsqp_at(best_h, n, 200000, top, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 3:
        h_new, c_new = focused_local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
