def binary_local_search(h, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improvement = 0
        best_pair = None
        
        # For efficiency, only check a subset of pairs per sweep
        # Or check all pairs but vectorize
        n_o = len(ones)
        n_z = len(zeros)
        
        # Sample pairs if too many
        max_pairs = 5000
        if n_o * n_z > max_pairs:
            # Random sampling
            rng = np.random.RandomState(sweep)
            pairs = [(zeros[rng.randint(n_z)], ones[rng.randint(n_o)]) for _ in range(max_pairs)]
        else:
            pairs = [(z, o) for z in zeros for o in ones]
        
        for i, j in pairs:  # i is zero→one, j is one→zero
            delta = np.zeros(2*n-1)
            # Term 1: (1-h[i+d]) for valid i+d
            ip = i + m_range
            mask = (ip >= 0) & (ip < n)
            delta[mask] += (1 - h[ip[mask]])
            # Term 2: -(1-h[j+d]) for valid j+d
            jp = j + m_range
            mask = (jp >= 0) & (jp < n)
            delta[mask] -= (1 - h[jp[mask]])
            # Term 3: -h[i-d] for valid i-d
            im = i - m_range
            mask = (im >= 0) & (im < n)
            delta[mask] -= h[im[mask]]
            # Term 4: +h[j-d] for valid j-d
            jm = j - m_range
            mask = (jm >= 0) & (jm < n)
            delta[mask] += h[jm[mask]]
            
            new_corr = corr + delta
            nc = new_corr.max() * dx
            improvement = cur_c - nc
            if improvement > best_improvement + 1e-15:
                best_improvement = improvement
                best_pair = (i, j, new_corr)

            i, j, new_corr = best_pair
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = corr.max() * dx
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lq_smooth(h, n, alpha, deadline, maxiter=500):
    dx = 2.0 / n
    def obj_and_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        exp_sh = np.exp(sh - sm)
        w = exp_sh / exp_sh.sum()
        val = m + (sm + np.log(exp_sh.sum())) / alpha
        # Gradient
        one_minus_x = 1 - x
        grad = np.zeros(n)
        d = np.arange(-(n-1), n)
        for i in range(n):
            ip = i + d
            im = i - d
            mask_a = (ip >= 0) & (ip < n)
            mask_b = (im >= 0) & (im < n)
            grad[i] = dx * (np.sum(w[mask_a] * one_minus_x[ip[mask_a]]) - np.sum(w[mask_b] * x[im[mask_b]]))
        return val, grad
    bounds = [(0.0, 1.0)] * n
    h0 = project_feasible(h, n)
    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lq_smooth(h, n, alpha, deadline, maxiter=400):
    dx = 2.0 / n
    def obj_and_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        exp_sh = np.exp(sh - sm)
        w = exp_sh / exp_sh.sum()
        val = m + (sm + np.log(exp_sh.sum())) / alpha
        one_minus_x = 1 - x
        d = np.arange(-(n - 1), n)
        k_idx = np.arange(2 * n - 1)
        grad = np.zeros(n)
        for i in range(n):
            ip = i + d
            im = i - d
            ma = (ip >= 0) & (ip < n)
            mb = (im >= 0) & (im < n)
            grad[i] = dx * (np.dot(w[ma], one_minus_x[ip[ma]]) - np.dot(w[mb], x[im[mb]]))
        return val, grad
    h0 = project_feasible(h, n)
    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9,
                                'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def vec_binary_search(h, n, deadline, max_sweeps=5000, sa_temp=0.0):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    h_pad = np.zeros(3 * n)
    h_pad[n:2 * n] = h
    omh_pad = 1.0 - h_pad

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        n_z = len(zeros)
        n_o = len(ones)
        if n_z == 0 or n_o == 0:
            break

        h_pad[n:2 * n] = h
        omh_pad = 1.0 - h_pad

        max_pairs = 120000
        if n_z * n_o > max_pairs:
            rng = np.random.RandomState(sweep * 104729 + 42)
            tz = min(n_z, 400)
            to = min(n_o, 400)
            zeros = zeros[rng.choice(n_z, tz, replace=False)]
            ones = ones[rng.choice(n_o, to, replace=False)]
            n_z, n_o = len(zeros), len(ones)

        z = zeros[:, None, None]
        o = ones[None, :, None]
        d = m_range[None, None, :]
        ip = z + d
        jp = o + d
        im = z - d
        jm = o - d

        mi_ip = (ip >= 0) & (ip < n)
        mi_jp = (jp >= 0) & (jp < n)
        mi_im = (im >= 0) & (im < n)
        mi_jm = (jm >= 0) & (jm < n)

        ip_s = np.clip(ip, 0, n - 1)
        jp_s = np.clip(jp, 0, n - 1)
        im_s = np.clip(im, 0, n - 1)
        jm_s = np.clip(jm, 0, n - 1)

        t1 = mi_ip * omh_pad[n + ip_s].astype(np.float64)
        t2 = mi_jp * omh_pad[n + jp_s].astype(np.float64)
        t3 = mi_im * h_pad[n + im_s].astype(np.float64)
        t4 = mi_jm * h_pad[n + jm_s].astype(np.float64)
        delta = t1 - t2 - t3 + t4

        new_vals = (corr[None, None, :] + delta).max(axis=2) * dx
        flat_idx = new_vals.argmin()
        best_idx = np.unravel_index(flat_idx, new_vals.shape)
        best_val = new_vals[best_idx]

        accept = False
        if best_val < cur_c - 1e-12:
            accept = True
        elif sa_temp > 0 and np.random.random() < np.exp((cur_c - best_val) / sa_temp):
            accept = True

        if accept:
            i = int(zeros[best_idx[0]])
            j = int(ones[best_idx[1]])
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta[best_idx]
            cur_c = best_val
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    return best_h, best_c
