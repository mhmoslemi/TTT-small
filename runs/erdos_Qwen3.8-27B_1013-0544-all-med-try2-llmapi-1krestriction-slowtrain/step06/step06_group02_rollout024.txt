import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-15: break
            h += (d / r) * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

    c = c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

    m = c.max()

def c5_val(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-15:
                break
            h += (d / r) * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sm_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def sm_obj(x, n, alpha, mu):
    c = c5_vec(x, n)
    val = sm_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def sm_grad(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n

import time

def c5_val(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-15:
                break
            h += (d / r) * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sm_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def sm_obj(x, n, alpha, mu):
    c = c5_vec(x, n)
    val = sm_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def sm_grad(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wdx = wi * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wdx * (1.0 - x[k:])
            g[k:] += wdx * (-x[:nv])
        else:

            nv = n - kk
            g[kk:] += wdx * (1.0 - x[:nv])
            g[:nv] += wdx * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n / 2.0)

    h = project(h, n)
    best_c = c5_val(h, n)
    bounds = [(0.0, 1.0)] * n
