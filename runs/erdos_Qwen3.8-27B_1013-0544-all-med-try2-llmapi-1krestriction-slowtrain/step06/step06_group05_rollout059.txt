        m_hi2 = min(n-1, 2*n-2-k)

            end_idx2 = m_hi2 - n + 1 + k
            grad_k[m_lo2:m_hi2+1] -= h[start_idx2:end_idx2+1]

        # But we need to maintain sum constraint, so we do paired moves

        # and j where grad_k is most positive (decrease h[j])
        i = int(np.argmin(grad_k))
        j = int(np.argmax(grad_k))

            # Try second best
            sorted_idx = np.argsort(grad_k)
            i = int(sorted_idx[0])
            j = int(sorted_idx[-1])

            h[i] += d
            h[j] -= d

            if c_new < best_c - 1e-14:
                best_c = c_new
                iters += 1
                break
            else:
                h[i] -= d
                h[j] += d

            j = np.random.randint(n)

                continue
            d = 0.001
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:

                else:
                    h[i] -= d

                h[i] -= d
                h[j] += d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                else:
                    h[i] += d
                    h[j] -= d

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full') * dx
    c_shift = np.clip(c - c.max(), -100.0 / max(alpha, 1.0), 0.0)
    exp_c = np.exp(alpha * c_shift)
    p = exp_c / exp_c.sum()
    obj = c.max() + np.log(exp_c.sum()) / alpha
    obj += penalty * (h.sum() - n / 2.0) ** 2
    conv_A = np.convolve(p, g, mode='full')
    A = conv_A[n - 1:2 * n - 1]
    corr_B = np.correlate(h, p, mode='full')
    B = corr_B[n - 1:2 * n - 1]
    grad = dx * (A - B) + 2.0 * penalty * (h.sum() - n / 2.0)
    return obj, grad

def lbfgs_solve(h, n, deadline, alpha_list, penalty=50.0):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:

            break
        def fun(x):
            v, _ = obj_grad(x, n, alpha, penalty)
            return v
        def gfun(x):
            _, gr = obj_grad(x, n, alpha, penalty)
            return gr
        try:
            res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 5000, 'maxfun': 200000,
                                    'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            h = best_h.copy()
        except Exception:
            pass
    return best_h, best_c

            m_lo2 = max(0, n - 1 - k)

            grad_k -= np.mean(grad_k)
            gn = np.linalg.norm(grad_k)
