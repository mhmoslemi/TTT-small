def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_at_shift(h, n, s):
    """Compute d(overlap(s))/dh[j] for all j."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        # j-s valid: j in [s, n)
        grad[s:] += (1 - h[:n-s])  # (1-h[j-s]) for j in [s,n)
        # j+s valid: j in [0, n-s)
        grad[:n-s] -= h[s:]  # -h[j+s] for j in [0,n-s)
    else:
        t = -s
        # j-s = j+t valid: j in [0, n-t)
        grad[:n-t] += (1 - h[t:])  # (1-h[j+t]) for j in [0,n-t)
        # j+s = j-t valid: j in [t, n)
        grad[t:] -= h[:n-t]  # -h[j-t] for j in [t,n)
    return grad * dx

def subgradient_optimize(h, n, deadline, top_k=3):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.1
    no_improve = 0
    
    while time.time() < deadline:
        c = np.correlate(h, 1-h, mode='full') * dx
        c_max = c.max()
        
        if c_max < best_c - 1e-13:
            best_c = c_max
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.1, 0.5)
        else:
            no_improve += 1
            if no_improve > 500:
                step *= 0.8
                if step < 1e-7:
                    break
                no_improve = 0
        
        # Get top-k shifts and average their gradients
        top_idx = np.argsort(c)[-top_k:]
        weights = c[top_idx]
        if weights.sum() > 0:
            weights = weights / weights.sum()
        else:
            weights = np.ones(top_k) / top_k
        
        grad = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            s = idx - (n - 1)
            grad += w * gradient_at_shift(h, n, s)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new <= c_max + 0.01 * step:
            h = h_new
        else:
            step *= 0.5
    
    return best_h, best_c

            if tot < 1e-14:

def shift_gradient(h, n, s):
    """Gradient of overlap at shift s with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        # (1-h[j-s]) for j in [s, n): indices j-s in [0, n-s)
        grad[s:] += (1 - h[:n-s])
        # -h[j+s] for j in [0, n-s): indices j+s in [s, n)
        grad[:n-s] -= h[s:]
    else:
        t = -s
        # (1-h[j+t]) for j in [0, n-t): indices j+t in [t, n)
        grad[:n-t] += (1 - h[t:])
        # -h[j-t] for j in [t, n): indices j-t in [0, n-t)
        grad[t:] -= h[:n-t]
    return grad * dx

            break

        m = c.max()
        exp_c = np.exp(c - m)
        weights = exp_c / (exp_c.sum() * alpha)

        # Only compute for significant weights to save time
        threshold = 1e-8
        significant = weights > threshold
        for k in np.where(significant)[0]:
            s = k - (n - 1)
            grad += weights[k] * shift_gradient(h, n, s)

        v = momentum * v - 0.01 * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)

        if c_new < c_max + 0.001:
            h = h_new

            h_new = h + v
            h_new = project_feasible(h_new, n)
            if compute_c5(h_new, n) < c_max:
                h = h_new
            else:
                v = 0
    
    h = project_feasible(h, n)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

Here's another approach: **coordinate descent with optimal 1D search**

For each coordinate i, find the optimal value of h[i] (given all other h[j] fixed) that minimizes the max overlap. This is a 1D optimization problem (minimize a max of linear functions of h[i]), which can be solved exactly!

When we fix all h[j] for j ≠ i and vary h[i] = t:
overlap_k(t) = c[k] + t * (contribution of h[i] to c[k])

where the contribution is the gradient component (linear in t).

So overlap_k(t) = a_k * t + b_k (linear in t for each k).
We want to minimize max_k (a_k * t + b_k) over t ∈ [0, 1].

This is a convex optimization in 1D (min of max of linear functions = max of linear functions, which is convex). The optimum is where the two "active" constraints intersect, or at a boundary.

We can solve this by:
1. Computing a_k and b_k for all k
2. Finding the t that minimizes max_k(a_k*t + b_k)
3. This is equivalent to finding the intersection of the upper envelope of lines

This is O(n²) in the worst case (to build the upper envelope) but in practice much faster.

Actually, simpler: I can just use a ternary search or bisection on the 1D convex function f(t) = max_k(a_k*t + b_k).

But wait, I also need to maintain the sum constraint. If I change h[i] to t, I need to adjust another h[j] to compensate. So it's not a pure 1D problem.

Let me think differently. What if I do:
1. For each i, compute the "marginal effect" of changing h[i]: g_i[k] = ∂c[k]/∂h[i]
2. The new c after changing h[i] by δ (and adjusting h[j] by -δ) is:
   c_new[k] = c[k] + δ*g_i[k] - δ*g_j[k]
3. I want to choose (i, j, δ) to minimize max(c_new)

This is still combinatorial. But I can simplify:
- Fix the direction: for the worst shift s*, the gradient tells me which i to decrease and which j to increase.
- Then optimize δ in 1D.

Here's the refined algorithm:
1. Compute c, find worst shift s*
2. Compute g_i = gradient of overlap(s*) w.r.t. h_i (this is a vector)
3. For each i with positive g_i[i] (decreasing h[i] helps), find the best j with negative g_i[j] (increasing h[j] helps)
4. For the best (i,j) pair, find the optimal δ by 1D search

This is O(n²) per step (to find the best pair) but each step makes a meaningful improvement.
