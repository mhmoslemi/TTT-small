    #   grads[m] @ h - t <= c_vals[m] - grads[m] @ h_current  for all m
    #   sum(h) = n/2
    #   0 <= h <= 1
    
    # Variable: x = [h(0), ..., h(n-1), t]
    # Objective: minimize t -> c_obj = [0, ..., 0, 1]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A_ub @ x <= b_ub
    # For each m: grads[m] @ h - t <= c_vals[m] - grads[m] @ h_current
    # Row: [grads[m,0], ..., grads[m,n-1], -1]

    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    
    b_ub = rhs
    
    # Box constraints: h >= 0 -> -h <= 0; h <= 1 -> h <= 1
    A_box = np.zeros((2*n, n+1))

    A_box[n:2*n, :n] = np.eye(n)  # h <= 1
    b_box = np.concatenate([np.zeros(n), np.ones(n)])
    
    A_all = np.vstack([A_ub, A_box])
    b_all = np.concatenate([b_ub, b_box])
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    res = linprog(c_obj, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)]*n + [(None, None)], method='highs')
    
    if res.success:
        return res.x[:n], res.fun
    return h_current, None

from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_grads(h, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    grads = np.zeros((nl, n))
    for m in range(nl):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grads[m, 0:nv] = dx * (1.0 - h[k:n])
            grads[m, k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grads[m, kk:n] = dx * (1.0 - h[0:nv])
            grads[m, 0:nv] += dx * (-h[kk:n])
    return grads

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def sca_solve(h, n, max_iters=300, tol=1e-11):

    best_c = compute_c5(h, n)
    prev_c = best_c
    stagnant = 0
    
    for it in range(max_iters):
        c_vals = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        grads = get_grads(h, n)
        
        # LP: min t
        # s.t. grads[m] @ h_new - t <= c_vals[m] - grads[m] @ h
        #      sum(h_new) = n/2
        #      0 <= h_new <= 1
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        rhs = c_vals - grads @ h
        A_ub = np.hstack([grads, -np.ones((2*n-1, 1))])
        b_ub = rhs
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-13:
            best_c = c_new
            h = h_new
            stagnant = 0
        else:

            improved = False
            for alpha in [0.5, 0.25, 0.1]:
                h_t = h + alpha * (h_new - h)

                    h = h_t
                    improved = True
                    break
            if not improved:

                if stagnant >= 5:
                    break
    
    return h, best_c

def fast_pair_search(h, n, deadline, deltas, radius=60):

                            best_c = c; improved = True
                        else:

            if not improved:

def active_set_sca(h, n, max_rounds=10):
    """After basic SCA, fix variables at bounds and re-solve"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for _ in range(max_rounds):
        # Identify free variables (not at bounds)
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        n_free = free.sum()
        if n_free < 2:
            break
        
        free_idx = np.where(free)[0]
        # Solve LP with only free variables
        # Fixed variables keep their values
        h_new = h.copy()
        
        c_vals = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        grads = get_grads(h, n)

        #           sum(h_free) = target_free

from scipy.optimize import linprog
import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_grads(h, n):

    nl = 2 * n - 1
    g = np.zeros((nl, n))
    for m in range(nl):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            g[m, 0:nv] = dx * (1.0 - h[k:n])
            g[m, k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[m, kk:n] = dx * (1.0 - h[0:nv])
            g[m, 0:nv] += dx * (-h[kk:n])
    return g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15: break
            h += (d / t) * r
        else:
            if s < 1e-15: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sca_one(h, n):
    """One SCA iteration. Returns new h."""
    dx = 2.0 / n
    cv = np.correlate(h, 1 - h, mode='full') * dx
    G = get_grads(h, n)
    c_obj = np.zeros(n + 1); c_obj[-1] = 1.0
    rhs = cv - G @ h
    A = np.hstack([G, -np.ones((2 * n - 1, 1))])
    Aeq = np.zeros((1, n + 1)); Aeq[0, :n] = 1.0
    beq = np.array([n / 2.0])
    bd = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A, b_ub=rhs, A_eq=Aeq, b_eq=beq,
                 bounds=bd, method='highs')
    return res

def sca_solve(h, n, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    stag = 0
    for _ in range(max_iters):
        res = sca_one(h, n)
        if not res.success:
            break
        h_new = np.clip(res.x[:n], 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new; h = h_new; stag = 0
        else:

                ct = compute_c5(ht, n)
                if ct < best_c - 1e-13:
                    best_c = ct; h = ht; ok = True; break
            if not ok:
                stag += 1
                if stag >= 8: break
    return h, best_c

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_grads(h, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    g = np.zeros((nl, n))
    for m in range(nl):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            g[m, 0:nv] = dx * (1.0 - h[k:n])
            g[m, k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[m, kk:n] = dx * (1.0 - h[0:nv])
            g[m, 0:nv] += dx * (-h[kk:n])
    return g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h += (d / t) * r
        else:
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sca_solve(h, n, max_iters=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    stag = 0
    dx = 2.0 / n
    for _ in range(max_iters):
        cv = np.correlate(h, 1 - h, mode='full') * dx
        G = get_grads(h, n)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        rhs = cv - G @ h
        A = np.hstack([G, -np.ones((2 * n - 1, 1))])
        Aeq = np.zeros((1, n + 1))
        Aeq[0, :n] = 1.0
        beq = np.array([n / 2.0])
        bd = [(0.0, 1.0)] * n + [(None, None)]
        res = linprog(c_obj, A_ub=A, b_ub=rhs, A_eq=Aeq, b_eq=beq,
                     bounds=bd, method='highs')
        if not res.success:
            break
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            h = h_new
            stag = 0
        else:
            ok = False
            for a in [0.5, 0.25, 0.1]:

                    best_c = ct
                    h = ht
                    ok = True
                    break
            if not ok:
                stag += 1
                if stag >= 10:
                    break
    return h, best_c
