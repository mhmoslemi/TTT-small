```python
    # Sort gradient components
    sorted_grad = np.sort(grad)
    # The solution sets z_i = 1 for the n/2 smallest grad components
    # and z_i = 0 for the rest, with one potential fractional value
    target = n / 2.0

    small_idx = np.argsort(grad)[:threshold_idx]

    return z
```

```python
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c
```

The issue with basic FW is slow convergence (O(1/k)). But we can improve with:
- **Strongly convex FW** (not applicable here since f is not strongly convex)
- **Pairwise FW** (Freund-Sra): Move along 2-point lines instead of full FW steps
- **Step size backtracking**: Use the exact line search

Actually, let me use a **hybrid approach**:
1. FW for the main optimization (fast, O(n log n) per iter)
2. SLSQP for polishing (expensive but accurate)

- Move along the line segment from z' to z

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _compute_lag_gradient(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] += (-h[:n - m])
        else:
            grad[:n - m] += (1.0 - h[-m:])
            grad[-m:] += (-h[:-m])
    return grad

def _fw_vertex(grad, n):
    idx = np.argsort(grad)[:n // 2]
    z = np.zeros(n)
    z[idx] = 1.0
    return z

def fw_optimize(h, n, deadline, top_m=3, max_iters=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_cur, n)
        top_idx = np.argsort(c)[-top_m:]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            grad += _compute_lag_gradient(h_cur, n, lag)
        grad /= top_m
        z = _fw_vertex(grad, n)
        t = 2.0 / (it + 2)
        h_new = (1.0 - t) * h_cur + t * z
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        h_cur = h_new
        if stagnant > 5000:
            break
    return best_h, best_c

def _smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _active_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_solve(h, n, alpha, top_idx=None, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        try:
            res = minimize(_active_obj, h, args=(n, top_idx, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(_smooth_max_obj, h, args=(n, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
        except Exception:
            return h.copy()
    return project_feasible(res.x, n)

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            if lag > 0:
                for i in range(n - m):
                    j = i + m
                    if h[i] > h[j] + 1e-10:
                        delta = min(d, (h[i] - h[j]) * 0.4)
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
            else:
                for i in range(m):
                    j = i + m
                    if h[j] > h[i] + 1e-10:
                        delta = min(d, (h[j] - h[i]) * 0.4)
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
    return h, best_c

def optimize_round(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    t_fw = min(deadline - 20, time.time() + 40)
    h_fw, c_fw = fw_optimize(best_h, n, t_fw, top_m=3, max_iters=150000)
    if c_fw < best_c:
        best_c = c_fw
        best_h = h_fw
    t_fw2 = min(deadline - 15, time.time() + 30)
    h_fw2, c_fw2 = fw_optimize(best_h, n, t_fw2, top_m=5, max_iters=80000)
    if c_fw2 < best_c:
        best_c = c_fw2
        best_h = h_fw2
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-8:]
        for alpha in [5000, 20000, 100000]:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_solve(best_h, n, alpha, top_idx, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 5:
        h_new = slsqp_solve(best_h, n, 200000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h, best_c = local_search(best_h, n, deadline)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    n_candidates = [200, 300, 400, 500]
```
