The function f(λ) = sum(clip(z - λ, 0, 1)) is continuous and piecewise linear, decreasing in λ.
f(+∞) = 0, f(-∞) = n.
We need f(λ) = n/2.

Bisection works well here. The range of λ is [min(z)-1, max(z)] (or a bit wider).

def project_box_sum(z, target):
    """Project z onto {w: sum(w)=target, 0<=w<=1}."""

Now for the projected gradient method:
1. Compute gradient g of the objective at h
2. Direction: d = -g (steepest descent)
3. Project d to the tangent of the sum constraint: d = d - mean(d) (so that sum(d)=0, maintaining the sum constraint to first order)
4. Line search: find α such that f(h + α*d) is minimized (after projection to [0,1] and re-projection to sum constraint)

Actually, a cleaner approach:
1. Compute gradient g at h
2. Take step: z = h - α * g for some α
3. Project z to {sum=n/2, 0≤w≤1} using project_box_sum
4. Check if objective decreased; if not, reduce α and retry

For the smooth max objective, the gradient is:
∇f(h) = Σ_k softmax_α(k) * ∇c(k)

where softmax_α(k) = α·exp(α·(c(k)-max_c)) / Σ_j exp(α·(c(j)-max_c))

This is a convex combination of the subgradients, weighted by how "active" each lag is.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (target / tot)
    return np.clip(h, 0, 1)

def project_box_sum(z, target):
    n = len(z)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        w = np.clip(z - mid, 0.0, 1.0)
        if w.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def lag_gradient_vec(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    m = abs(k)
    s = slice(m, n)
    t = slice(0, n - m)
    grad[s] += (1.0 - h[t])
    grad[t] -= h[s]
    return dx * grad

def smooth_max_obj_and_grad(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c_full = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c_full[top_idx]
        offset = top_idx[0]
    else:
        c = c_full
        offset = 0
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    weights = alpha * np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
    grad = np.zeros(n)
    if top_idx is not None:
        for w, idx in zip(weights, top_idx):
            lag = idx - (n - 1)
            grad += w * lag_gradient_vec(x, n, lag)
    else:
        for j in range(len(c)):
            lag = j - (n - 1)
            grad += weights[j] * lag_gradient_vec(x, n, lag)
    return float(val), grad

def slsqp_smooth(h, n, alpha, top_idx=None, maxiter=5000, ftol=1e-15):
    def obj(x):
        v, _ = smooth_max_obj_and_grad(x, n, alpha, top_idx)
        return v
    def jac(x):
        _, g = smooth_max_obj_and_grad(x, n, alpha, top_idx)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient(h, n, alpha, deadline, n_iter=5000):
    h_curr = h.copy().astype(float)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    step = 0.1
    for it in range(n_iter):
        if time.time() > deadline - 2:
            break
        c_full = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c_full)[::-1]
        # Focus on top lags for sharper gradient
        use_top = min(20, len(top_idx))
        active = top_idx[:use_top]
        _, grad = smooth_max_obj_and_grad(h_curr, n, alpha, active)
        # Project gradient to maintain sum constraint
        grad = grad - grad.mean()
        # Backtracking line search
        f_curr = compute_c5(h_curr, n)
        alpha_ls = step
        found = False
        for _ in range(20):
            z = h_curr - alpha_ls * grad
            h_try = project_box_sum(z, n / 2.0)
            f_try = compute_c5(h_try, n)
            if f_try < f_curr - 1e-16:
                h_curr = h_try
                found = True
                if f_try < best_c:
                    best_c = f_try
                    best_h = h_curr.copy()
                break
            alpha_ls *= 0.5
        if not found:
            step *= 0.7
            if step < 1e-8:
                break
        else:
            step = min(step * 1.1, 0.5)
    return best_h, best_c

def active_set_slsqp(h, n, deadline, top_m=5, max_rounds=15):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for rnd in range(max_rounds):

        top_idx = np.argsort(c_full)[-top_m:]
        # Minimize max of active lags via t variable
        def obj_t(v):
            x, t = v[:n], v[n]
            c_a = c_full_active(x, n, top_idx)

        def c_full_active(x, n2, idx):
            c = np.correlate(x, 1 - x, mode='full') * (2.0 / n2)
            return c[idx]
        h_curr = slsqp_smooth(h_curr, n, 50000, top_idx=top_idx, maxiter=3000)
        c_new = compute_c5(h_curr, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_curr.copy()
        else:
            # Try with slightly larger top_m

    return best_h, best_c

def coordinate_descent(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    for _ in range(50):
        if time.time() > deadline - 2:
            break
        c_full = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c_full))
        lag = k_star - (n - 1)
        m = abs(lag)
        improved = False
        if m == 0 or m >= n:
            break
        # Try moving mass from high to low along the active lag
        if lag > 0:
            pairs = [(i, i + m) for i in range(n - m)]
        else:
            pairs = [(i + m, i) for i in range(n - m)]
        for i, j in pairs:
            if time.time() > deadline - 2:
                break
            if h_curr[i] > h_curr[j]:
                delta = min(h_curr[i] - h_curr[j], 0.01) * 0.5
                if delta < 1e-10:
                    continue
                h_curr[i] -= delta
                h_curr[j] += delta
                c_new = compute_c5(h_curr, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h_curr[i] += delta
                    h_curr[j] -= delta
            elif h_curr[j] > h_curr[i]:
                delta = min(h_curr[j] - h_curr[i], 0.01) * 0.5
                if delta < 1e-10:
                    continue
                h_curr[i] += delta
                h_curr[j] -= delta
                c_new = compute_c5(h_curr, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h_curr[i] -= delta
                    h_curr[j] += delta
        if not improved:
            break
    return h_curr, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with increasing alpha
    for alpha in [500, 2000, 5000, 20000]:
        if time.time() > deadline - 30:
            break
        h_new = slsqp_smooth(best_h, n, alpha, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 2: Projected gradient
    if time.time() < deadline - 40:
        h_new, c_new = projected_gradient(best_h, n, 5000, min(deadline - 40, time.time() + 120))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 30:
        h_new, c_new = active_set_slsqp(best_h, n, min(deadline - 30, time.time() + 100))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 4: Final SLSQP + coordinate descent
    if time.time() < deadline - 15:
        h_new = slsqp_smooth(best_h, n, 100000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = coordinate_descent(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
