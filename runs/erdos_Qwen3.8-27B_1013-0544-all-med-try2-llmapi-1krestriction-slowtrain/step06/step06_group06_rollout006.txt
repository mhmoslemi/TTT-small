        else:

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = int(np.argmax(c))

        for i in range(k, n):
            grad[i] += dx * (1 - h[i - k])
            grad[i - k] += -dx * h[i]

            best_h = x.copy()
        
        # Adaptive step size (simple geometric decay + occasional reset)

    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()

        # Avoid overflow
        return g.sum()
    return m + np.log(np.exp(alpha * (g - m)).sum()) / alpha + mu * (h.sum() - n/2)**2

    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    w = np.exp(alpha * (g - m))
    w = w / w.sum()

    grad = np.zeros(n)

            grad[k2:] += w[k] * dx * (1 - h[:n-k2])
            grad[:n-k2] += -w[k] * dx * h[k2:]
    grad += 2 * mu * (h.sum() - n/2)

    rng = np.random.RandomState(777)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        c = compute_c5(h, n)
        if c < cur_c - 1e-15:
            cur_c = c
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    return h, best_c

def make_binary(h_cont, n):
    hb = np.zeros(n)
    srt = np.argsort(h_cont)[::-1]
    hb[srt[:n // 2]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS smooth (25%)
    p1 = t0 + 0.25 * (budget_s - 5)
    h, c = lbfgs_opt(best_h.copy(), n, p1, [200, 1000, 5000, 20000, 100000])
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 2: FISTA subgradient (15%)
    p2 = t0 + 0.40 * (budget_s - 5)
    h, c = fista_subgrad(best_h.copy(), n, p2, max_iter=40000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 3: Binary SA (25%)
    p3 = t0 + 0.65 * (budget_s - 5)
    hb = make_binary(best_h, n)
    hb, cb = binary_sa(hb, n, p3, max_iter=400000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 4: Binary local search (20%)
    p4 = t0 + 0.85 * (budget_s - 5)
    hb = make_binary(best_h, n)
    hb, cb = binary_local_search(hb, n, p4, max_iters=3000000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 5: Continuous refinement from best binary (10%)
    if time.time() < dl - 10:
        h_cont = best_h.copy()
        h, c = lbfgs_opt(h_cont, n, dl - 5, [5000, 50000, 200000])
        if c < best_c:
            best_c = c
            best_h = h.copy()
        hb = make_binary(best_h, n)
        cb = compute_c5(hb, n)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
