```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h


def grad_worst_lag(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    p_star = np.argmax(corr)
    k = p_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        idx1 = np.arange(k, n)
        grad[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        grad[idx2] -= h[idx2 + k]
    else:
        la = -k
        idx1 = np.arange(0, n - la)
        grad[idx1] += (1.0 - h[idx1 + la])
        idx2 = np.arange(la, n)
        grad[idx2] -= h[idx2 - la]
    return grad


def grad_top_lags(h, n, m=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-m:]
    grad = np.zeros(n)
    for p in top:
        k = p - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            idx1 = np.arange(k, n)
            g[idx1] += (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            g[idx2] -= h[idx2 + k]
        else:
            la = -k
            idx1 = np.arange(0, n - la)
            g[idx1] += (1.0 - h[idx1 + la])
            idx2 = np.arange(la, n)
            g[idx2] -= h[idx2 - la]
        grad += g
    return grad / m


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def projected_gradient(h, n, target_sum, dx, t0, budget, max_time):
    h = h.copy()
    lr = 0.02
    best_local = compute_c5(h)
    for it in range(200000):
        elapsed = time.time() - t0
        if elapsed > budget - max_time:
            break
        if it % 5000 == 0 and it > 0:
            c5_now = compute_c5(h)
            if c5_now < best_local:
                best_local = c5_now
            if it % 20000 == 0:
                h += np.random.randn(n) * 0.01
                h = project_feasible(h, target_sum)
        if it % 2000 == 0:
            grad = grad_worst_lag(h, n)
        else:
            grad = grad_top_lags(h, n, m=2)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        grad = grad / grad_norm
        best_c5 = compute_c5(h)
        for _ in range(30):
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
                break
            lr *= 0.5
        else:
            lr = max(lr * 0.5, 1e-7)
            continue
        lr = min(lr * 1.05, 0.1)
    c5_final = compute_c5(h)
    if c5_final < best_local:
        best_local = c5_final
    return h, best_local


def slsqp_optimize(h, n, target_sum, dx, t0, budget, max_time):
    h = h.copy()
    best_c5 = compute_c5(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(len(x))}]
    for beta in [50, 200, 800, 3000]:
        elapsed = time.time() - t0
        if elapsed > budget - max_time:
            break
        for trial in range(2):
            elapsed = time.time() - t0
            if elapsed > budget - max_time:
                break
            x0 = h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.005
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(
                    lambda x: smooth_max(compute_corr(x), beta) * dx,
                    x0, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    return h, best_c5

        return time.time() - t0

        target_sum = n / 2.0

        candidates.append(np.clip(np.random.randn(n) * 0.3 + 0.5, 0, 1))
        # Sidelobe pattern

        h = best_cand
        c5_cur = best_cand_c5

        # Phase A: Projected gradient
        time_left = budget_s - elapsed()
        time_alloc = min(60, time_left * 0.4)
        if time_alloc > 10:
            h, c5_cur = projected_gradient(h, n, target_sum, dx, t0, budget_s,
                                           budget_s - elapsed() - time_alloc)

        # Phase B: SLSQP
        time_left = budget_s - elapsed()
        time_alloc = min(50, time_left * 0.3)
        if time_alloc > 10:
            h, c5_cur = slsqp_optimize(h, n, target_sum, dx, t0, budget_s,
                                       budget_s - elapsed() - time_alloc)

        # Phase C: Second projected gradient round (shorter)
        time_left = budget_s - elapsed()
        if time_left > 20:
            time_alloc = min(30, time_left * 0.3)
            if time_alloc > 5:
                h, c5_cur = projected_gradient(h, n, target_sum, dx, t0, budget_s,
                                               budget_s - elapsed() - time_alloc)

        # Phase D: Final SLSQP polish
        time_left = budget_s - elapsed()
        if time_left > 15:
            h, c5_cur = slsqp_optimize(h, n, target_sum, dx, t0, budget_s,
                                       budget_s - elapsed() - 10)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Final verification and polish at best resolution
    if best_h is not None and elapsed() < budget_s - 15:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        best_h = project_feasible(best_h, target_sum)
        # One more SLSQP at very high beta
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                 'jac': lambda x: np.ones(len(x))}]
        try:
            res = minimize(
                lambda x: smooth_max(compute_corr(x), 10000) * dx,
                best_h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
```
