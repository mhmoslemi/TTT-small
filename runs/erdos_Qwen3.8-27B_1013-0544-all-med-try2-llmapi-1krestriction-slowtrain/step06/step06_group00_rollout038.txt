import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not mask.any():
            break
        if diff > 0:
            s = (1.0 - h)[mask].sum()
            if s < 1e-14: break
            h[mask] += (diff / s) * (1.0 - h)[mask]
        else:
            s = h[mask].sum()
            if s < 1e-14: break
            h[mask] += (diff / s) * h[mask]
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_corr_k(h, n, k):
    m = k - (n - 1)
    grad = np.zeros(n)
    lo1, hi1 = max(0, m), min(n-1, n-1+m)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += 1.0 - h[lo1-m:hi1-m+1]
    lo2, hi2 = max(0, -m), min(n-1, n-1-m)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2+m:hi2+m+1]
    return grad
