import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_active(h, n, k_active):
    """Analytical gradient of g[k_active] w.r.t. h, vectorized O(n)."""
    s = k_active - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        p = -s
        grad[:n - p] += (1.0 - h[p:])
        grad[p:] -= h[:n - p]
    return grad

def grad_smooth(h, n, alpha):
    """Gradient of the log-sum-exp smoothed max."""
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    weights = np.exp(sh - sm)
    weights /= weights.sum()
    grad = np.zeros(n)
    active_ks = np.where(weights > 1e-10)[0]
    for k in active_ks:
        grad += weights[k] * grad_active(h, n, k)
    return grad, m

def smooth_obj(h, n, alpha):
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def projected_gradient(h, n, deadline, alphas, max_iter_per_alpha=2000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        lr = 0.5 / alpha  # learning rate
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            grad, m = grad_smooth(h, n, alpha)
            ng = np.linalg.norm(grad)

                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                lr *= 0.5
                if lr < 1e-15:
                    break
    return best_h, best_c

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        g_max = g.max()
        sig = np.zeros((n, 2 * n - 1))
        for p in range(n):
            sv = np.arange(-(n - 1), n)
            i1 = p - sv
            v1 = (i1 >= 0) & (i1 < n)
            sig[p, v1] += (1.0 - h[i1[v1]])
            i2 = p + sv
            v2 = (i2 >= 0) & (i2 < n)
            sig[p, v2] -= h[i2[v2]]
        improved = False
        for a in zeros:
            if time.time() > deadline:
                return h, best_c
            nm = (g[np.newaxis, :] + sig[a][np.newaxis, :] - sig[ones][:]).max(axis=1)
            bi = int(nm.argmin())
            if nm[bi] < g_max - 1e-12:
                h[a] = 1.0
                h[ones[bi]] = 0.0
                best_c = compute_c5(h, n)
                improved = True
                break
        if not improved:
            break
    return h, best_c

def smooth_from_binary(h_bin, n, deadline, seed=0):
    rng = np.random.RandomState(seed)
    h = project_feasible(h_bin + 0.05 * rng.randn(n), n)
    h, c = subgrad_fast(h, n, deadline, max_iter=50000)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Fast subgradient (30%)
    p1 = t0 + 0.30 * (budget_s - 3)
    h, c = subgrad_fast(best_h.copy(), n, p1)
    if c < best_c:
        best_c, best_h = c, h

    # Phase 2: Top-k subgradient (10%)
    p2 = t0 + 0.40 * (budget_s - 3)
    h, c = subgrad_topk(best_h.copy(), n, p2, topk=3)
    if c < best_c:
        best_c, best_h = c, h

    # Phase 3: Binary optimization (20%)
    p3 = t0 + 0.60 * (budget_s - 3)
    hb, cb = binary_optimize(best_h.copy(), n, p3)
    if cb < best_c:
        best_c, best_h = cb, hb

    # Phase 4: Smooth from binary (10%)
    p4 = t0 + 0.70 * (budget_s - 3)
    hr, cr = smooth_from_binary(best_h.copy(), n, p4, seed=1)
    if cr < best_c:
        best_c, best_h = cr, hr

    # Phase 5: Multi-start continuous (20%)
    p5 = t0 + 0.90 * (budget_s - 3)
    for s in range(6):
        if time.time() > p5 - 10:
            break
        perturb = 0.12 * rng.randn(n)
        hp = project_feasible(best_h + perturb, n)
        h_new, c_new = subgrad_fast(hp, n, time.time() + (p5 - time.time()) / 2.0, max_iter=30000)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 6: Final binary + refine (10%)
    if time.time() < dl - 40:
        hb, cb = binary_optimize(best_h.copy(), n, dl - 30)
        if cb < best_c:
            best_c, best_h = cb, hb
        if time.time() < dl - 10:
            hr, cr = smooth_from_binary(best_h.copy(), n, dl, seed=42)
            if cr < best_c:
                best_c, best_h = cr, hr

    # Final polish
    if time.time() < dl - 2:
        h, c = subgrad_fast(best_h.copy(), n, dl, max_iter=100000)
        if c < best_c:
            best_c, best_h = c, h

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
