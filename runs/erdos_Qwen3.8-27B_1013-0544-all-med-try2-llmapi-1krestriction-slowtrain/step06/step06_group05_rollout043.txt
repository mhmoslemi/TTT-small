import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sc(x, n):
    return float(x.sum() - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, iters=4000):
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        r = minimize(soft_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=b, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-18})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

    mi = int(cr.argmax())

            g[i + lag] -= h[i]
    else:
        L = -lag
        for i in range(n - L):
            g[i + L] += 1.0 - h[i]
            g[i] -= h[i + L]
    return g * dx
