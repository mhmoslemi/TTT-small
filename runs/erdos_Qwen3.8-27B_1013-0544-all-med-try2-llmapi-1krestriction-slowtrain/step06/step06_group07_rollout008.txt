For Phase 3 (pairwise local search), the idea is:
- For each pair (i, j) with i < j:
  - Try transferring δ from i to j (or vice versa)
  - δ is chosen to either make h[i] or h[j] hit a boundary, or to be a small step
  - Accept if C5 decreases
- This is like a steepest descent in the "pairwise transfer" basis

The feasible set {h : sum(h) = n/2, 0 ≤ h ≤ 1} is a polytope, and the "pairwise transfers" (e_i - e_j directions) are the edge directions of this polytope. So this is essentially a simplex-like method on the feasible polytope.

At each step, for the current h, the worst constraint k* gives a linear function f_{k*}(h) = sum_valid h[i+k*]*(1-h[i]).
In the direction e_i - e_j (transfer from j to i):
Δf = [∂f/∂h[i] - ∂f/∂h[j]] * δ

So we want to move in the direction that most decreases f_{k*}. The step size is limited by the box constraints.

This is essentially a projected gradient method where the projection is onto the feasible polytope, and we use the "natural" direction (pairwise transfer).

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def subgrad_active(h, k_set, n):
    grad = np.zeros(n)
    cnt = len(k_set)
    if cnt == 0:
        return grad
    for kw in k_set:
        lag = kw - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad / cnt


def con_fun(h, ts):
    return h.sum() - ts


def con_jac(h, ts):
    return np.ones_like(h)


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def pairwise_local_search(h, n, target_sum, max_passes=50, time_limit=60):
    t0 = time.time()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    h = h.copy()
    order = np.random.permutation(n)
    for p in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        grad = subgrad_active(h, [k_worst], n)
        improved = False
        for j in order:
            if time.time() - t0 > time_limit:
                break
            gj = grad[j]
            if abs(gj) < 1e-12:
                continue
            # Find best i to pair with j
            if gj > 0:
                # h[j] should decrease; find i where grad[i] < gj (transferring j->i decreases f)
                best_delta = 0
                best_i = -1
                for i in order:
                    if i == j:
                        continue
                    gi = grad[i]
                    if gi < gj:
                        delta = min(h[j], 1.0 - h[i])
                        if delta > 0:
                            best_delta = delta
                            best_i = i
                            break
                if best_i >= 0 and best_delta > 0:
                    step = best_delta * 0.5
                    h[best_i] += step
                    h[j] -= step
                    h = np.clip(h, 0.0, 1.0)
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        improved = True
                    else:
                        h[best_i] -= step
                        h[j] += step
            else:
                # h[j] should increase; find i where grad[i] > gj (transferring i->j decreases f)
                best_delta = 0
                best_i = -1
                for i in order:
                    if i == j:

                    gi = grad[i]
                    if gi > gj:

                    h[j] += step

                    h = np.clip(h, 0.0, 1.0)

def slsqp_refine(h, n, target_sum, beta, maxiter=5000, ftol=1e-15):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h, compute_c5(h)


def subgradient_phase(h, n, target_sum, max_iter=50000, time_limit=120):
    t0 = time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    avg_h = h.copy()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        thresh = max_val * 0.001
        active = np.where(corr >= max_val - thresh)[0]
        grad = subgrad_active(h, active, n)
        lr = 0.003 / (1.0 + 0.001 * it)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        avg_h = (it * avg_h + h) / (it + 1)
        if it % 5000 == 4999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            c5_avg = compute_c5(project_feasible(avg_h, target_sum))
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = project_feasible(avg_h, target_sum).copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
