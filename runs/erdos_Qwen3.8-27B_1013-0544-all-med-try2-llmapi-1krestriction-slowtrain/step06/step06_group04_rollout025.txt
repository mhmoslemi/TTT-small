    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        g += w[k] * dx * _lag_grad(x, n, lag)
    return g

def _act_obj(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return _smooth_max_val(c[idx], alpha)

def _act_grad(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ca = c[idx]
    m = ca.max()
    s = alpha * (ca - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        lag = idx[ii] - (n - 1)
        g += w[ii] * dx * _lag_grad(x, n, lag)
    return g

def _sc(x, n):
    return x.sum() - n / 2.0

def _scj(x, n):
    return np.ones(n)

def slsqp_full(h, n, alpha, maxit=6000):
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    bounds = [(0, 1)] * n
    try:
        res = minimize(_full_obj, h, jac=_full_grad, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_act(h, n, idx, alpha, maxit=6000):
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    bounds = [(0, 1)] * n
    try:
        res = minimize(_act_obj, h, jac=_act_grad, args=(n, idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent_step(h, n, c, alpha, step_size):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        g += w[k] * dx * _lag_grad(h, n, lag)
    g_p = g - g.mean()
    h_new = h - step_size * g_p
    return project_feasible(h_new, n)

def active_loop(h, n, deadline, top_m=8, n_rounds=12):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [5000, 20000, 100000, 500000, 2000000]
    for rnd in range(n_rounds):
        if time.time() > deadline - 10:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_act(h, n, top_idx, alpha, maxit=5000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    return best_h, best_c

def local_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(3000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_st = int(np.argmax(c))
        lag = k_st - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        gc = _lag_grad(h, n, lag) * dx
        gc_p = gc - gc.mean()
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = project_feasible(h - step * gc_p, n)
            c_check = compute_c5(h_new, n)
            if c_check < best_c - 1e-15:
                best_c = c_check
                h = h_new
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 20:
        h_new = slsqp_full(best_h, n, 50000, maxit=6000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new

    if time.time() < deadline - 15:
        h_oo, c_oo = active_loop(best_h, n, min(deadline - 10, time.time() + 60), top_m=8, n_rounds=15)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
            h = h_oo

    if time.time() < deadline - 8:
        h_new = slsqp_full(best_h, n, 500000, maxit=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new

    if time.time() < deadline - 3:
        h, best_c = local_refine(best_h, n, min(deadline - 1, time.time() + 20))
        best_h = project_feasible(h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Use initial values as primary starting point
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Try other n values
    n_candidates = [100, 150, 200, 250, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(80, remaining * 0.22)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Random restarts at best n
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scales = [0.01, 0.02, 0.03, 0.05]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Structured initialization attempts
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        x = np.linspace(0, 2, n)
        for freq in [2, 3, 5, 7, 11]:
            if time.time() > deadline - 25:
                break
            h_struct = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / 2.0 + rng.uniform(0, 2*np.pi))
            h_struct = project_feasible(h_struct, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(h_struct, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_full(best_h, n, 2000000, maxit=8000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 2:
            best_h, best_c = local_refine(best_h, n, deadline - 0.5)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
