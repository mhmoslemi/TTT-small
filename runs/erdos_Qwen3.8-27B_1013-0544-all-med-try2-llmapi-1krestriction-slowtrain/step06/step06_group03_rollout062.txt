```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h[mask] += diff / tot * room[mask]
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += diff / tot * h[mask]
    return np.clip(h, 0.0, 1.0)

def active_lag_gradient(h, n, k):
    """Gradient of correlation at lag k with respect to h."""
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        grad[-k:] += (1.0 - h[:n+k])
        grad[:n+k] -= h[-k:]
    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # log-sum-exp trick
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

        # Best: maximize grad[j] - grad[i]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * room[mask]
        else:
            mask = h > 1e-15
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Gradient of correlation at lag k (k can be negative)."""
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
    else:
        m = -k
        if m < n:
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_fn(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fista(h0, n, alpha, max_iter=5000, lr=None):
    """Accelerated projected gradient (FISTA) for smoothed max."""
    h = h0.copy()
    x = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    dx = 2.0 / n
    
    if lr is None:
        # Estimate Lipschitz constant
        # grad of smooth obj has norm bounded by ~ alpha * max_grad * dx
        # Just use a conservative step
        lr = 0.02 / alpha * n  # heuristic
    
    for it in range(max_iter):
        # Gradient of log-sum-exp at x
        c_raw = np.correlate(x, 1 - x, mode='full')
        c = c_raw * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        
        # Weighted sum of lag gradients
        grad = np.zeros(n)
        for k_idx in range(len(w)):
            if w[k_idx] < 1e-12:
                continue
            k = k_idx - (n - 1)
            grad += w[k_idx] * lag_gradient(x, n, k)
        grad *= dx
        
        # Gradient step
        y = x - lr * grad
        y = project_feasible(y, n)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x = y + ((t - 1.0) / t_new) * (y - h)
        x = project_feasible(x, n)
        h = y
        t = t_new
        
        cc = compute_c5(h, n)
        if cc < best_c:
            best_c = cc
            best_h = h.copy()
    
    return best_h, best_c

def grad_pair_search(h, n, deadline, max_rounds=100):
    """Gradient-guided pair local search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        cmax = corr.max()

        eps = 0.005 * cmax
        active = np.where(corr > cmax - eps)[0]

        count = 0
        for idx in active:
            k = idx - (n - 1)
            if abs(k) < n:
                grad += lag_gradient(best_h, n, k)
                count += 1
        if count > 0:
            grad /= count
        
        # Find best pair: max grad[j] - grad[i] for i,j with grad[j] > grad[i]
        # We want to move weight FROM high-grad TO low-grad
        # i.e., decrease h[high_grad], increase h[low_grad]
        # Improvement ≈ d * (grad[low] - grad[high]) < 0
        
        # Sort by grad
        order = np.argsort(grad)
        # Best source: highest grad (last in order)
        # Best target: lowest grad (first in order)
        n_try = min(30, n)
        sources = order[-n_try:]   # high grad
        targets = order[:n_try]    # low grad

            if time.time() > deadline:
                return best_h, best_c
            for t_idx in targets:
                if s == t_idx:
                    continue
                # Move from s to t_idx
                max_d = min(best_h[s], 1.0 - best_h[t_idx])
                if max_d <= 0:
                    continue
                # Try geometric sequence of step sizes
                for d in [max_d * 0.8, max_d * 0.5, max_d * 0.3, 0.05, 0.02, 0.01, 0.005, 0.001]:
                    if d > max_d:

                    best_h[t_idx] += d
                    best_h[s] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[t_idx] -= d
                        best_h[s] += d
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_c

def random_pair_search(h, n, deadline, deltas):
```

```python
def compute_deltas(h, n, i, j):
    """Compute Δ_k = ∂R[k]/∂h[i] - ∂R[k]/∂h[j] for all lags k."""
    # For each lag k (represented by index in 'full' correlation),
    # Δ_k = grad_k[i] - grad_k[j]
    # where grad_k[i] = (1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n]
    
    # There are 2n-1 lags, indexed from -(n-1) to (n-1)
    # Let's compute for each k:
    num_lags = 2 * n - 1
    deltas = np.zeros(num_lags)
    
    for idx in range(num_lags):
        k = idx - (n - 1)
        # grad_k[i]
        gi = 0.0
        if 0 <= i + k < n:
            gi += 1.0 - h[i + k]
        if 0 <= i - k < n:
            gi -= h[i - k]
        # grad_k[j]
        gj = 0.0
        if 0 <= j + k < n:
            gj += 1.0 - h[j + k]
        if 0 <= j - k < n:
            gj -= h[j - k]
        deltas[idx] = gi - gj
    
    return deltas
```

```python
                    vals = R + mid * deltas
                    vm = vals.max()
                    act = deltas[vals >= vm - 1e-10]
                    if act.size == 0:
                        lo = hi = mid
                        break
                    if act.min() > 1e-10:
                        lo = mid
                    elif act.max() < -1e-10:
                        hi = mid
                    else:
                        lo = hi = mid
                        break
                
                d_opt = (lo + hi) / 2.0
                if abs(d_opt) < 1e-15:
                    continue
                
                # Evaluate improvement
                new_R = R + d_opt * deltas
                new_c = float(np.max(new_R) * dx)
                
                if new_c < bc - 1e-15:
                    # Apply move
                    h[t] += d_opt
                    h[s] -= d_opt
                    R = new_R
                    bc = new_c
                    found = True
                    break
            if found:
                break
        
        if not found:
            break
    
    return h, bc
```

This is the key function. Let me estimate its speed:
- Per pair: computing deltas is O(n) = 499 iterations (Python loop), binary search is 50 × O(n) = 12,475 operations
- So per pair: ~13,000 operations
- Per round (30×30 = 900 potential pairs, but we break early): maybe 10-50 pairs tried
- Per round: ~500,000 to 650,000 operations
- Can do ~1000 rounds in reasonable time

Actually, the inner loop over 499 lags in Python is the bottleneck. Let me vectorize it.

For a given pair (t, s), I need:
deltas[idx] for idx in 0..2n-2, where k = idx - (n-1)

deltas[idx] = [(1-h[t+k])*[0≤t+k<n] - h[t-k]*[0≤t-k<n]] - [(1-h[s+k])*[0≤s+k<n] - h[s-k]*[0≤s-k<n]]

Let me define arrays:
For position p and lag k: contribution is (1-h[p+k])*[valid] - h[p-k]*[valid]

I can precompute for each position p:
g[p, k] = (1-h[p+k])*[0≤p+k<n] - h[p-k]*[0≤p+k<n]

But this is an n × (2n-1) matrix, which for n=250 is 250 × 499 ≈ 125,000 entries. Storing it is fine but computing it is O(n²) = 62,500 per position, or O(n²) total if I'm smart about it.

Actually, let me just do it in a vectorized way:

```python
def compute_deltas_vec(h, n, t, s):
    """Vectorized computation of Δ_k for pair (t, s)."""
    # k ranges from -(n-1) to (n-1)
    ks = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For position t:
    # (1-h[t+k]) where 0 ≤ t+k < n
    t_plus = t + ks  # t+k for each k
    t_minus = t - ks  # t-k for each k
    
    valid_tp = (t_plus >= 0) & (t_plus < n)
    valid_tm = (t_minus >= 0) & (t_minus < n)
    
    # Need h[t+k] and h[t-k]
    h_tp = np.where(valid_tp, h[np.clip(t_plus, 0, n-1)], 0.0)
    h_tm = np.where(valid_tm, h[np.clip(t_minus, 0, n-1)], 0.0)
    
    grad_t = (1.0 - h_tp) * valid_tp - h_tm * valid_tm
    
    # For position s:
    s_plus = s + ks
    s_minus = s - ks
    
    valid_sp = (s_plus >= 0) & (s_plus < n)
    valid_sm = (s_minus >= 0) & (s_minus < n)
    
    h_sp = np.where(valid_sp, h[np.clip(s_plus, 0, n-1)], 0.0)
    h_sm = np.where(valid_sm, h[np.clip(s_minus, 0, n-1)], 0.0)
    
    grad_s = (1.0 - h_sp) * valid_sp - h_sm * valid_sm
    
    return grad_t - grad_s
```

This is O(n) in vectorized NumPy, which is much faster than a Python loop.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def _sum_fn(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def do_slsqp(h, n, alpha, maxiter=2000):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n,)}]
    try:
        r = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                     bounds=b, constraints=c, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def compute_deltas_vec(h, n, t, s):
    ks = np.arange(-(n-1), n)
    tau = t + ks; tum = t - ks
    vtp = (tau >= 0) & (tau < n)
    vtm = (tum >= 0) & (tum < n)
    htp = np.where(vtp, h[np.clip(tau, 0, n-1)], 0.0)
    htm = np.where(vtm, h[np.clip(tum, 0, n-1)], 0.0)
    gt = (1.0 - htp) * vtp - htm * vtm
    sas = s + ks; sam = s - ks
    vsp = (sas >= 0) & (sas < n)
    vsm = (sam >= 0) & (sam < n)
    hsp = np.where(vsp, h[np.clip(sas, 0, n-1)], 0.0)
    hsm = np.where(vsm, h[np.clip(sam, 0, n-1)], 0.0)
    gs = (1.0 - hsp) * vsp - hsm * vsm
    return gt - gs

def optimal_move_search(h, n, deadline, max_rounds=5000, top_k=25):
    dx = 2.0 / n
    R = np.correlate(h, 1 - h, mode='full')
    bc = float(np.max(R) * dx)
    ks = np.arange(-(n-1), n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        cmax = R.max()
        thresh = cmax - max(1e-10, 5e-4 * cmax)
        active = np.where(R >= thresh)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if 0 <= k < n:
                grad[:n-k] += 1.0 - h[k:]
                grad[k:] -= h[:n-k]
            elif -n < k < 0:
                m = -k
                grad[m:] += 1.0 - h[:n-m]
                grad[:n-m] -= h[m:]
        grad /= len(active)
        order = np.argsort(grad)
        ntr = min(top_k, n // 3)
        targets = order[:ntr]
        sources = order[-ntr:]
        found = False
        for s in sources:
            if time.time() > deadline:
                break
            for t in targets:
                if s == t:
                    continue
                d_min = max(-h[t], h[s] - 1.0)
                d_max = min(1.0 - h[t], h[s])
                if d_min >= d_max - 1e-14:
                    continue
                deltas = compute_deltas_vec(h, n, t, s)
                lo_d, hi_d = d_min, d_max
                for _ in range(50):
                    if hi_d - lo_d < 1e-14:
                        break
                    mid_d = (lo_d + hi_d) / 2.0
                    vals = R + mid_d * deltas
                    vm = vals.max()
                    act_d = deltas[vals >= vm - 1e-10]
                    if act_d.size == 0:
                        lo_d = hi_d = mid_d
                        break
                    if act_d.min() > 1e-12:
                        lo_d = mid_d
                    elif act_d.max() < -1e-12:
                        hi_d = mid_d
                    else:
                        lo_d = hi_d = mid_d
                        break
                d_opt = (lo_d + hi_d) / 2.0
                if abs(d_opt) < 1e-14:
                    continue
                new_R = R + d_opt * deltas
                new_c = float(np.max(new_R) * dx)
                if new_c < bc - 1e-15:
                    h[t] += d_opt
                    h[s] -= d_opt
                    R = new_R
                    bc = new_c
                    found = True
                    break
            if found:
                break
        if not found:
            break
    return project_feasible(h, n), bc

def random_pair(h, n, deadline, deltas_steps):
    bh = h.copy()
    bc = compute_c5(bh, n)
    for d in deltas_steps:
        imp = True
        while imp and time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return bh, bc
                for j in range(i+1, n):
                    if bh[i]+d <= 1.0 and bh[j]-d >= 0.0:
                        bh[i]+=d; bh[j]-=d
                        c = compute_c5(bh, n)
                        if c < bc - 1e-14:
                            bc = c; imp = True
                        else:
                            bh[i]-=d; bh[j]+=d
                    if bh[i]-d >= 0.0 and bh[j]+d <= 1.0:
                        bh[i]-=d; bh[j]+=d
                        c = compute_c5(bh, n)
                        if c < bc - 1e-14:
                            bc = c; imp = True
                        else:
                            bh[i]+=d; bh[j]-=d
    return bh, bc

def try_resolution(h_best, best_c, n_target, deadline, rng):
    n_cur = len(h_best)
    x_old = np.linspace(0, 2, n_cur, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_target)
    c_interp = compute_c5(h_interp, n_target)
    for a in [1000, 10000, 50000]:
        if time.time() > deadline - 30:
            break
        ht = do_slsqp(h_interp, n_target, a, maxiter=1500)
        ct = compute_c5(ht, n_target)
        if ct < c_interp:
            c_interp = ct
            h_interp = ht
    if time.time() < deadline - 20:
        h_interp, c_interp = optimal_move_search(h_interp, n_target, deadline - 5, max_rounds=2000)
    return h_interp, c_interp, n_target

    rng = np.random.RandomState(seed)

    # Phase 4: More optimal moves

    for n_try in [200, 300]:

        alloc = min(80, deadline - time.time() - 30)

            best_c = ct
            best_h = ht
        if time.time() < deadline - 1:
            best_h, best_c = optimal_move_search(best_h, n, deadline, max_rounds=5000, top_k=50)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
