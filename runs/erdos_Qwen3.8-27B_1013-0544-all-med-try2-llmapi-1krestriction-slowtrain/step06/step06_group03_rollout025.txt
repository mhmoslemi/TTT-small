def subgradient_descent(h, n, deadline, eta=0.01):
    """Subgradient method for the max of correlations."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        # Active set: lags within 1% of max
        active = np.where(c > 0.999 * c_max)[0]
        
        # Compute average subgradient
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - h[s:n])
                grad[s:] -= h[:n-s]
            else:
                t = -s
                grad[t:] += (1.0 - h[:n-t])
                grad[:n-t] -= h[t:]
        grad /= len(active)
        
        # Project gradient onto sum(d) = 0
        grad -= np.mean(grad)
        
        # Scale step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        d = -eta * grad / grad_norm
        
        # Find max step before hitting bounds
        t_max = 1.0
        for i in range(n):
            if d[i] > 0:
                t_max = min(t_max, (1.0 - h[i]) / d[i])
            elif d[i] < 0:
                t_max = min(t_max, -h[i] / d[i])
        t_max = max(t_max, 0.0)
        
        # Line search
        best_t = 0
        best_val = best_c
        for t in np.linspace(0, min(t_max, 1.0), 20):
            h_new = h + t * d
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
            val = compute_c5(h_new, n)
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t > 0 and best_val < best_c:

            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            best_c = best_val
            best_h = h.copy()
            # Decrease step size
            eta *= 0.95
        else:
            eta *= 0.9

One more thought: instead of trying fixed delta values, I can try the "optimal" delta for each pair. For a pair (i,j) with the move h[i] += d, h[j] -= d, the objective as a function of d is:

where a_{k,ij} = ∂c_k/∂h[i] - ∂c_k/∂h[j].

2. Finding the minimum of max_k (c_k + a_{k,ij} * d) over d ∈ [d_min, d_max]

- Evaluate at d_min, d_max, and the midpoint

- Repeat (bisection) until the interval is small enough

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    ms = shifted.max()
    if ms > 500:
        lse = ms + np.log(np.sum(np.exp(shifted - ms)))
    else:
        lse = np.log(np.sum(np.exp(shifted)))
    return float(m + lse / alpha)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _ones_jac(x):
    return np.ones_like(x)

def slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _ones_jac, 'args': (n,)}]
    try:
        res = minimize(lambda x: smooth_obj(x, n, alpha), h,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
