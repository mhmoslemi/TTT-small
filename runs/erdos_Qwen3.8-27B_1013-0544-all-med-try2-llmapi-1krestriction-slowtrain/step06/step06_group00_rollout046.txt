            best_c, best_h = c, h.copy()

    # Phase 6: Try slightly different resolution
    if time.time() < dl - 30:
        for nh in [500, 600]:
            if time.time() > dl - 35:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh = fast_pg(hh, nh, 10000, dl - 32)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                hqb, cqb = greed_binary(hh, nh, dl - 28, max_sweeps=50)
                if cqb < ch:
                    hh, ch = hqb, cqb
                hh = fast_pg(hh, nh, 500000, dl - 25)
                ch = compute_c5(hh, nh)
                if ch < best_c:
                    best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
