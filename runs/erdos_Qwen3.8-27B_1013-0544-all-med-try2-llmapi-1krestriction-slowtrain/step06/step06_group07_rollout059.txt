y_k = x_k + (t_{k-1}/t_k - 1)(x_k - x_{k-1})
x_{k+1} = y_k - (1/t_k) ∇f(y_k)
t_{k+1} = (1 + sqrt(1 + 4t_k^2)) / 2

This is for a specific step size that works for L-Lipschitz gradients. For our problem, the gradient Lipschitz constant depends on beta (higher beta → larger Lipschitz constant).

For the smooth max with parameter beta, the Hessian is bounded by beta in the "corr" space, and the mapping from h to corr involves products, so the Lipschitz constant of ∇(smooth_max * dx) is roughly bounded by something like C * beta * dx where C depends on the problem structure.

For n=70, dx = 2/70 ≈ 0.0286. The gradient of the correlation w.r.t. h has entries bounded by 1 (since each entry is a sum of at most n terms each bounded by 1). So the Hessian of the smooth max composed with corr is bounded by beta * n * dx = beta * 2 approximately.

The optimal Nesterov step size for an L-Lipschitz gradient is 1/sqrt(L). So step ≈ 1/sqrt(2*beta).

For beta=200: step ≈ 1/sqrt(400) = 0.05
For beta=1000: step ≈ 1/sqrt(2000) ≈ 0.022
For beta=5000: step ≈ 1/sqrt(10000) = 0.01

The Nesterov 1/t_k schedule starts at 1/t_0 = 2 (for t_0=0.5) and decays as 2/k. This might be too large initially.

Let me use a simpler approach: fixed step size with backtracking, or just use the standard 2/(k+2) step size which is the classic Nesterov schedule.

lr_initial = 0.02  # tuned for beta=200

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def softmax_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(shifted)
    return w / w.sum()


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def gradient_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    weights = softmax_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad * dx


def smooth_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def constraint_fun(h, target_sum):
    return h.sum() - target_sum


def constraint_jac(h, n):
    return np.ones(n)


def subgrad_for_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def avg_active_subgrad(h, n, tol):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    active = np.where(corr > max_c * (1.0 - tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        grad += subgrad_for_lag(h, n, lag)
    return grad / len(active)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation from initial
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in [5, 20, 100, 500, 2000]:
        if tl() < 20:
            break
        cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac,
                 'args': (target_sum, n)}]
        for trial in range(2):
            if tl() < 10:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += rng.randn(n) * 0.005
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(smooth_obj, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Nesterov gradient with smooth max (main workhorse)
    if tl() > 100:
        h = best_h.copy()
        h_prev = h.copy()
        beta = 500
        for it in range(200000):
            if tl() < 20:
                break
            if it > 0:
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
                t_curr = (1.0 + np.sqrt(1.0 + 4.0 * it ** 2)) / 2.0
                h_nest = h + (t_prev - 1.0) / t_curr * (h - h_prev)
                h_nest = np.clip(h_nest, 0.0, 1.0)
            else:
                h_nest = h.copy()
                t_curr = 0.5
            grad = gradient_smooth(h_nest, dx, beta)
            step = 1.0 / t_curr
            h_new = h_nest - step * grad
            h_new = project_feasible(h_new, target_sum)
            if it % 200 == 199:
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            h_prev = h
            h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 3: Active-set subgradient with adaptive step
    if tl() > 80:
        h = best_h.copy()
        lr = 0.003
        stagnation = 0
        for it in range(100000):
            if tl() < 15:
                break
            grad = avg_active_subgrad(h, n, tol=0.15)
            h_new = project_feasible(h - lr * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new
                lr = min(lr * 1.05, 0.01)
                stagnation = 0
            else:
                h = h_new
                stagnation += 1
                if stagnation > 500:
                    lr *= 0.7
                    if lr < 1e-5:
                        break
                    stagnation = 0
                    h = best_h.copy()

    # Phase 4: Multi-resolution SLSQP
    if tl() > 60:
        resolutions = [n - 10, n + 10, n + 15, n + 20, n + 30, n + 40, n + 50]
        for n_try in resolutions:
            if n_try < 35 or n_try > 200 or tl() < 25:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac,
                       'args': (ts_t, n_try)}]
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            for beta in [200, 2000, 10000]:
                if tl() < 10:
                    break
                try:
                    res = minimize(smooth_obj, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bounds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bounds_t
                except Exception:
                    pass

    # Phase 5: Nesterov on best resolution with high beta
    if tl() > 30:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        h = best_h.copy()
        h_prev = h.copy()
        for beta in [2000, 10000]:
            if tl() < 10:
                break
            for it in range(50000):
                if tl() < 5:
                    break
                if it > 0:
                    t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
                    t_curr = (1.0 + np.sqrt(1.0 + 4.0 * it ** 2)) / 2.0
                    h_nest = h + (t_prev - 1.0) / t_curr * (h - h_prev)
                    h_nest = np.clip(h_nest, 0.0, 1.0)
                else:
                    h_nest = h.copy()
                    t_curr = 0.5
                grad = gradient_smooth(h_nest, dx, beta)
                step = 1.0 / t_curr
                h_new = h_nest - step * grad
                h_new = project_feasible(h_new, target_sum)
                h_prev = h
                h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Phase 6: Final SLSQP polish
    if tl() > 8:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [5000, 30000]:
            if tl() < 4:
                break
            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac,
                     'args': (target_sum, n)}]
            try:
                res = minimize(smooth_obj, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-17})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Quick local subgradient swipe
    if tl() > 5:
        h = best_h.copy()
        n = len(h)
        target_sum = n / 2.0
        for it in range(20000):
            if tl() < 2:
                break
            grad = avg_active_subgrad(h, n, tol=0.05)
            lr = 0.001 / (1.0 + it * 1e-5)
            h = project_feasible(h - lr * grad, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return best_h, best_c5, len(best_h)
