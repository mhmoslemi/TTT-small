            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            for beta in [1000, 10000, 100000]:
                if remaining_budget() < 25:
                    break
                h_new = slsqp_optimize(h_t, n_try, dx_t, ts_t, beta, 4000)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    best_h = h_new
                    best_c5 = c5
                    n = n_try
                    dx = dx_t
                    target_sum = ts_t

    # Phase 5: Binary SA at best resolution
    if remaining_budget() > 150:
        h_bin = project_binary(best_h, n // 2)
        h_sa, sa_c5 = binary_sa(h_bin, n, t0, remaining_budget)
        if sa_c5 < best_c5:
            best_h = h_sa
            best_c5 = sa_c5
        # Refine
        if remaining_budget() > 60:
            for beta in [10000, 100000, 500000]:
                if remaining_budget() < 20:
                    break
                h_new = slsqp_optimize(best_h, n, dx, target_sum, beta, 10000)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    best_h = h_new
                    best_c5 = c5

    # Phase 6: Final polish
    if remaining_budget() > 20:
        for beta in [50000, 500000]:
            if remaining_budget() < 10:
                break
            h_new = slsqp_optimize(best_h, n, dx, target_sum, beta, 15000)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5

    return best_h, best_c5, len(best_h)
