def grad_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    # (1-h[j+lag]) part
    if lag >= 0:
        grad[:n-lag] += (1 - h[lag:])
    else:
        grad[-lag:] += (1 - h[:n+lag])
    # -h[j-lag] part
    if lag >= 0:
        grad[lag:] -= h[:n-lag]
    else:
        grad[:n+lag] -= h[-lag:]
    return grad

    """Compute change in correlation when h[i]+=delta, h[j]-=delta."""
    k = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    ipk = i + k  # shape (2n-1,)
    imk = i - k
    jpk = j + k

    def safe_h(idx):

        mask = (idx >= 0) & (idx < n)
        out[mask] = h[idx[mask]]
        return out
    
    val = (1 - safe_h(ipk)) - safe_h(imk) - (1 - safe_h(jpk)) + safe_h(jmk)

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def make_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').copy()

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Precompute lag indices
def make_lags(n):
    return np.arange(2*n-1) - (n-1)

def safe_get(h_arr, idx):
    """Get h_arr[idx] with 0 for out-of-bounds."""
    out = np.zeros(len(idx), dtype=float)
    mask = (idx >= 0) & (idx < len(h_arr))
    out[mask] = h_arr[idx[mask]]
    return out

def delta_correlation(h, lags, i, j, delta):
    """Compute change in full correlation when h[i]+=delta, h[j]-=delta."""
    n = len(h)
    ipk = i + lags
    imk = i - lags
    jpk = j + lags
    jmk = j - lags
    val = (1 - safe_get(h, ipk)) - safe_get(h, imk) - (1 - safe_get(h, jpk)) + safe_get(h, jmk)
    return delta * val

def fast_local_search(h, n, deadline, deltas, radius=None):
    """Fast local search with incremental correlation updates."""
    dx = 2.0 / n
    corr = make_corr(h, n)
    lags = make_lags(n)
    best_c = float(np.max(corr) * dx)
    
    for delta in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            pairs_done = 0
            for i in range(n):
                if time.time() > deadline:
                    return h, corr, best_c
                if radius is not None:
                    j_start = max(0, i + 1)
                    j_end = min(n, i + 1 + radius)
                else:
                    j_start = i + 1
                    j_end = n
                for j in range(j_start, j_end):
                    if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                        continue
                    dc = delta_correlation(h, lags, i, j, delta)
                    new_max = float(np.max(corr + dc))
                    if new_max < best_c - 1e-12:
                        # Accept
                        h[i] += delta
                        h[j] -= delta
                        corr += dc
                        best_c = new_max * dx
                        improved = True
                        improved_any = True
                        pairs_done += 1
                        if pairs_done > n:  # Limit per pass
                            break
                if radius is not None and pairs_done > n:
                    break
            if not improved:
                break

Let me use a different approach: **two-variable moves**. I'll always move h[i] and h[j] together (h[i] += δ, h[j] -= δ), which automatically preserves the sum.

For a two-variable move, the objective as a function of δ is:
max_k (c[k] + δ * D[k])

where D[k] = (1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k] (the change per unit δ).

This is a piecewise-linear function of δ, and its minimum is at a kink or boundary.

The boundaries are: δ = -h[i] (making h[i]=0), δ = 1-h[i] (making h[i]=1), δ = h[j] (making h[j]=0), δ = -h[j] (making h[j]=1).

The kinks are where c[k1] + δ*D[k1] = c[k2] + δ*D[k2], i.e., δ = (c[k2] - c[k1]) / (D[k1] - D[k2]).

So for each pair (i,j), I can:
1. Compute D[k] for all k (or just active k)
2. Find all kink points in the valid range
3. Evaluate the max at each kink and at boundaries
4. Choose the best δ

This is O(n) to compute D, O(M²) to find kinks (M = number of active lags), and O(n) to evaluate the max at each candidate. Total per pair: O(n + M² * n) = O(n*M²).

For n=250, M=10: 250*100 = 25,000 per pair. With n²=62,500 pairs: 1.56 * 10⁹. That's too slow per pass.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def make_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').copy()

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def safe_h(h_arr, idx):
    out = np.zeros(len(idx), dtype=float)
    mask = (idx >= 0) & (idx < len(h_arr))
    out[mask] = h_arr[idx[mask]]
    return out

def delta_corr(h, n, lags, i, j, d):
    ipk = i + lags
    imk = i - lags
    jpk = j + lags
    jmk = j - lags
    return d * ((1 - safe_h(h, ipk)) - safe_h(h, imk) - (1 - safe_h(h, jpk)) + safe_h(h, jmk))

def fast_search(h, n, deadline, deltas, radius=0):
    """Local search with O(n) per move evaluation using incremental updates."""
    dx = 2.0 / n
    corr = make_corr(h, n)
    lags = np.arange(2 * n - 1) - (n - 1)
    best_raw = float(corr.max())
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = i + 1
                j_hi = n if radius == 0 else min(n, i + 1 + radius)
                for j in range(j_lo, j_hi):
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    dc = delta_corr(h, n, lags, i, j, d)
                    new_max = float(np.max(corr + dc))
                    if new_max < best_raw - 1e-13:
                        h[i] += d
                        h[j] -= d
                        corr = corr + dc
                        best_raw = new_max
                        improved = True
            if not improved:
                break
    return h, best_raw * dx
