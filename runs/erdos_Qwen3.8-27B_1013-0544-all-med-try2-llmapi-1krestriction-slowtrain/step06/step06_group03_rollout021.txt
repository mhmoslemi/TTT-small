    best_c = compute_c5(h, n)
    corr = compute_full_corr(h, n)
    offset = n - 1  # to convert index to lag
    
    while time.time() < deadline:
        max_idx = np.argmax(corr)
        K = max_idx - offset
        g = grad_corr_at_lag(h, n, K)

        # So we want pairs where g[j] > g[i]

            # Sort indices by gradient
            order = np.argsort(g)
            # Best moves: from highest g (decrease) to lowest g (increase)
            # Try top pairs

            for a in range(min(20, n)):  # lowest gradient indices

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, K):
    g = np.zeros(n)
    if K >= 0:
        K = min(K, n - 1)
        g[K:] += (1.0 - h[:n - K])
        g[:n - K] -= h[K:]
    else:
        Kp = min(-K, n - 1)
        g[:n - Kp] += (1.0 - h[Kp:])
        g[Kp:] -= h[:n - Kp]
    return g

def pair_line_search(h, n, i, j, direction, best_c, max_iter=40):
    """Binary search for optimal step moving h[j]->h[i] (direction=1) or h[i]->h[j] (direction=-1)."""
    if direction == 1:
        d_max = min(1.0 - h[i], h[j])
    else:
        d_max = min(h[i], 1.0 - h[j])
    if d_max < 1e-15:
        return h, best_c
    lo, hi = 0.0, d_max
    best_d = 0.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        h[i] += direction * mid
        h[j] -= direction * mid
        hc = compute_c5(h, n)
        h[i] -= direction * mid
        h[j] += direction * mid

            lo = mid
        else:

        h[j] -= direction * best_d
        best_c = compute_c5(h, n)
    return h, best_c

        order_desc = np.argsort(-g)  # highest g first (best to decrease)

                j = order_desc[b]

                new_h, new_c = pair_line_search(h, n, i, j, 1, best_c)
                if new_c < best_c:
                    h, best_c = new_h, new_c
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reverse: move from i to j
            for a in range(min(top_k, n)):
                for b in range(min(top_k, n)):
                    i = order_desc[a]
                    j = order_asc[b]
                    if i == j:

                    new_h, new_c = pair_line_search(h, n, i, j, -1, best_c)
                    if new_c < best_c:

                        break

            # Perturb to escape
            mask = h > 0.01

                h = project_feasible(h, n)
                if compute_c5(h, n) > best_c + 0.001:
                    h = project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def full_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, K):
    g = np.zeros(n)
    if K >= 0:
        if K < n:
            g[K:] += (1.0 - h[:n - K])
            g[:n - K] -= h[K:]
    else:
        Kp = min(-K, n - 1)
        g[:n - Kp] += (1.0 - h[Kp:])
        g[Kp:] -= h[:n - Kp]
    return g

def guided_local_search(h, n, deadline, top_k=25):
    best_c = compute_c5(h, n)

    stall = 0

        max_idx = int(np.argmax(corr))
        K = max_idx - n_off
        g = grad_lag(h, n, K)
        order_asc = np.argsort(g)[:top_k]
        order_desc = np.argsort(-g)[:top_k]
        improved = False
        for idx_i in order_asc:
            if time.time() > deadline:
                break
            for idx_j in order_desc:
                if idx_i == idx_j:
                    continue
                i, j = int(idx_i), int(idx_j)
                d_max = min(1.0 - h[i], h[j])
                if d_max < 1e-10:

                lo, hi = 0.0, d_max

                for _ in range(28):

                    h[i] += mid
                    h[j] -= mid
                    c = compute_c5(h, n)
                    h[i] -= mid
                    h[j] += mid
                    if c < best_c - 1e-15:
                        best_d = mid
                        lo = mid
                    else:
                        hi = mid
                if best_d > 1e-12:
                    h[i] += best_d
                    h[j] -= best_d
                    best_c = compute_c5(h, n)
                    improved = True

            if improved:
                break

                    h[idx] += d
            h = project_feasible(h, n)
            if compute_c5(h, n) > best_c + 0.01:

def gradient_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    n_off = n - 1
    step = 0.02
    while time.time() < deadline:
        corr = full_corr(h, n)
        K = int(np.argmax(corr)) - n_off
        g = grad_lag(h, n, K)
        g = g - g.mean()
        h_new = project_feasible(h - step * g, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-15:
            best_c = c
            h = h_new
            step = min(step * 1.1, 0.08)
        else:
            step *= 0.5
            if step < 1e-7:
                break
    return h, best_c

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_optimize(h, n, alpha, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': np.ones(n), 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def spectral_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, min(10, n // 3)):
        amp = rng.uniform(0.04, 0.15) / (k ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
    return project_feasible(h, n)
