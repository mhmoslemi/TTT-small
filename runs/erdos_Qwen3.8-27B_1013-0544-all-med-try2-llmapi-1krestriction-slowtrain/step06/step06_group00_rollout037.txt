    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_v = np.exp(shifted - sm)
    Z = exp_v.sum()
    logsum = sm + np.log(Z)
    obj = (m + logsum / alpha) * dx
    w = exp_v / Z
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            grad[lo:hi] += wk * (1.0 - x[lo+k:hi+k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2-k:hi2-k]

    return obj, grad

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_v = np.exp(shifted - sm)
    Z = exp_v.sum()
    obj = (m + (sm + np.log(Z)) / alpha) * dx
    w = exp_v / Z
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            grad[lo:hi] += wk * (1.0 - x[lo + k:hi + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 - k:hi2 - k]
    grad *= dx
    return obj, grad

        self.n = n
        self.alpha = alpha

        return obj_grad(x, self.n, self.alpha)[0]

            grad[lo:hi] += wk * (1.0 - x[lo + k:hi + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 - k:hi2 - k]
    grad *= dx
    return obj, grad

class OptW:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha
    def fun(self, x):
        return obj_grad(x, self.n, self.alpha)[0]
    def jac(self, x):
        return obj_grad(x, self.n, self.alpha)[1]
    def cf(self, x):
        return x.sum() - self.n / 2.0
    def cj(self, x):
        return np.ones(self.n)

def slsqp(h, n, alpha, maxiter):
    w = OptW(n, alpha)
    try:
        res = minimize(w.fun, h, jac=w.jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': w.cf, 'jac': w.cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Progressive SLSQP with analytical gradient
    h = best_h.copy()
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000, 1e6, 5e6, 2e7, 1e8, 5e8, 2e9]
    for alpha in alphas:
        if time.time() > dl - 180:
            break
        h = slsqp(h, best_n, alpha, 4000)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Perturbation restarts to escape local optima
    for trial in range(8):
        if time.time() > dl - 100:
            break
        h = best_h + 0.008 * rng.randn(best_n)
        h = project_feasible(h, best_n)
        for alpha in [5000, 50000, 500000, 5e6, 5e7]:
            if time.time() > dl - 80:
                break
            h = slsqp(h, best_n, alpha, 2500)
            c = compute_c5(h, best_n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 3: Try higher resolutions for finer discretization
    for nh in [500, 600, 800]:
        if time.time() > dl - 30:
            break
        if nh <= best_n:
            continue
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [2000, 20000, 200000, 2e6, 2e7, 2e8, 2e9]:
            if time.time() > dl - 15:
                break
            hh = slsqp(hh, nh, alpha, 2500)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, best_n = ch, hh.copy(), nh

    # Phase 4: Final high-alpha polish
    if time.time() < dl:
        h = best_h.copy()
        for alpha in [1e10, 1e11, 1e12]:
            if time.time() > dl:
                break
            h = slsqp(h, best_n, alpha, 6000)
            c = compute_c5(h, best_n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
