def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()
    sh = alpha * (c - m)

    g += 2.0 * mu * (x.sum() - n / 2.0)

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    target = n / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _pobj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:

    return float(val)

    configs = [(100,1.0),(500,5.0),(2000,20.0),(10000,100.0),
               (50000,500.0),(200000,2000.0),(1000000,10000.0),
               (5000000,50000.0),(20000000,200000.0),(100000000,1e6)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_pobj, h, jac=_pgrad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_set_refine(h, n, deadline, top_k=8):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(5000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        active_idx = np.argsort(c)[-top_k:][::-1]
        active_k = active_idx - (n - 1)
        m_lp = n + 1
        c_lp = np.zeros(m_lp)
        c_lp[-1] = 1.0
        A_ub = np.zeros((top_k, m_lp))
        b_ub = np.zeros(top_k)
        for idx, k in enumerate(active_k):
            if k >= 0:
                nv = n - k
                f_k = dx * (h[0:nv].sum() - np.dot(h[0:nv], h[k:n]))
                gk = np.zeros(n)
                gk[0:nv] = dx * (1.0 - h[k:n])
                gk[k:n] -= dx * h[0:nv]
            else:
                kk = -k
                nv = n - kk
                f_k = dx * (h[kk:n].sum() - np.dot(h[0:nv], h[kk:n]))
                gk = np.zeros(n)
                gk[kk:n] = dx * (1.0 - h[0:nv])
                gk[0:nv] -= dx * h[kk:n]
            b_interp = f_k - np.dot(gk, h)
            A_ub[idx, :n] = gk
            A_ub[idx, -1] = -1.0
            b_ub[idx] = -b_interp
        bounds_lp = [(0, 1)] * n + [(0, None)]
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
            if res.status != 0:
                break
            h_lp = res.x[:n]
            h_new = project_feasible(h_lp, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
            else:
                improved = False
                for a in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                    h_s = project_feasible((1 - a) * h + a * h_lp, n)
                    c_s = compute_c5(h_s, n)
                    if c_s < best_c - 1e-16:
                        best_c = c_s
                        h = h_s
                        improved = True
                        break
                if not improved:
                    break
        except Exception:
            break
    return h, best_c

def pair_search_basic(h, n, deadline, deltas, radius=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def grad_guided_pairs(h, n, deadline, step=0.002, top_pairs=3000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for alpha in [50, 200, 1000, 5000, 20000]:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            diffs = np.abs(g[:, None] - g[None, :])
            np.fill_diagonal(diffs, 0)
            flat_idx = np.argsort(diffs.flatten())[::-1][:top_pairs]
            for fi in flat_idx:
                if time.time() > deadline:
                    return h, best_c
                i, j = fi // n, fi % n
                dg = g[i] - g[j]
                if abs(dg) < 1e-12:
                    continue
                if dg > 0:
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                else:
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n_best = n_init
    # Phase 1: L-BFGS-B
    t1 = min(t0 + 280, deadline - 300)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=200)
    # Phase 2: Active set LP
    if time.time() < deadline - 220:
        h2, c2 = active_set_refine(best_h, n_init, min(time.time() + 120, deadline - 200), top_k=10)
        if c2 < best_c:
            best_c, best_h = c2, h2
    # Phase 3: L-BFGS-B again
    if time.time() < deadline - 150:
        h3, c3 = lbfgs_phase(best_h, n_init, min(time.time() + 100, deadline - 130), max_iters=200)
        if c3 < best_c:
            best_c, best_h = c3, h3
    # Phase 4: Active set LP (tighter)
    if time.time() < deadline - 100:
        h4, c4 = active_set_refine(best_h, n_init, min(time.time() + 60, deadline - 80), top_k=5)
        if c4 < best_c:
            best_c, best_h = c4, h4
    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 60:
        h5, c5 = grad_guided_pairs(best_h, n_init, min(time.time() + 40, deadline - 40), step=0.001, top_pairs=2000)
        if c5 < best_c:
            best_c, best_h = c5, h5
    # Phase 6: Fine pair search
    if time.time() < deadline - 15:
        h6, c6 = pair_search_basic(best_h, n_init, min(time.time() + 30, deadline - 5),
                                   [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002], radius=40)
        if c6 < best_c:
            best_c, best_h = c6, h6
    # Phase 7: Try other n values
    if time.time() < deadline - 80:
        for n_try in [200, 250, 400, 500]:
            if time.time() > deadline - 50:
                break
            x_old = np.linspace(0, 2, n_best)
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(70, deadline - time.time() - 30)
            if alloc < 25:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.4, max_iters=150)
            if time.time() < deadline - 20:
                h_t, c_t = active_set_refine(h_t, n_try, min(time.time() + 25, deadline - 10), top_k=8)
            if time.time() < deadline - 8:
                h_t, c_t = pair_search_basic(h_t, n_try, deadline - 5,
                                             [0.001, 0.0005, 0.0002], radius=30)
            if c_t < best_c:
                best_c, best_h, n_best = c_t, h_t, n_try
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
