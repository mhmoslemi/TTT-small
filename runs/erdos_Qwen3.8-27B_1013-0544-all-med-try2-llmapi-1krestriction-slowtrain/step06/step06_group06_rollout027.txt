        hp, cp = binary_local_search(hp, n, p5, max_rounds=1000)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    
    # Phase 6: Final polish
    if time.time() < dl - 15:
        h, c = subgrad_descent(best_h, n, dl - 10, max_iter=50000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    if time.time() < dl - 5:
        hb, cb = binary_local_search(best_h, n, dl - 2, max_rounds=2000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
