        h_fine, c_fine = random_pair_swap(h_fine, n_fine, deadline, rng, steps=15000)
    return h_fine, c_fine

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 200
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Coarse optimization first
    n_coarse = max(40, n // 4)
    x_full = np.linspace(0, 2, n)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = np.interp(x_coarse, x_full, best_h)
    h_coarse = project_feasible(h_coarse, n_coarse)
    t1 = min(t0 + 120, deadline - 500)
    h_coarse, c_coarse = lbfgs_opt(h_coarse, n_coarse, t1, alphas=(50, 500, 5000, 50000))
    if time.time() < deadline - 450:
        h_coarse, c_coarse = random_pair_swap(h_coarse, n_coarse, min(time.time() + 60, deadline - 450), rng, steps=20000)
    # Upsample refined coarse back
    h_up = np.interp(x_full, x_coarse, h_coarse)
    h_up = project_feasible(h_up, n)
    c_up = compute_c5(h_up, n)
    if c_up < best_c:
        best_c = c_up
        best_h = h_up

    # Phase 2: L-BFGS on full resolution
    t2 = min(time.time() + 150, deadline - 350)
    best_h, best_c = lbfgs_opt(best_h, n, t2, alphas=(50, 200, 1000, 5000, 50000))

    # Phase 3: Focused pair swap
    if time.time() < deadline - 250:
        t3 = min(time.time() + 120, deadline - 250)
        best_h, best_c = focused_pair_swap(best_h, n, t3, rng)

    # Phase 4: L-BFGS with high alpha
    if time.time() < deadline - 200:
        t4 = min(time.time() + 80, deadline - 200)
        best_h, best_c = lbfgs_opt(best_h, n, t4, alphas=(50000, 200000, 1000000))

    # Phase 5: Random pair swap
    if time.time() < deadline - 150:
        t5 = min(time.time() + 80, deadline - 150)
        best_h, best_c = random_pair_swap(best_h, n, t5, rng, steps=40000)

    # Phase 6: Block optimization
    if time.time() < deadline - 120:
        t6 = min(time.time() + 60, deadline - 120)
        best_h, best_c = block_opt(best_h, n, t6, rng, bsize=5)

    # Phase 7: Try different resolutions
    if time.time() < deadline - 100:
        for n_try in [int(n * 0.6), int(n * 1.4)]:
            if n_try < 30 or n_try > 600 or time.time() > deadline - 80:
                break
            ht, ct = coarse_to_fine(best_h, n_try, deadline - 50, rng)
            if ct < best_c - 1e-14:
                best_c = ct
                best_h = ht
                n = n_try

    # Phase 8: Final L-BFGS polish
    nf = len(best_h)
    if time.time() < deadline - 20:
        best_h, best_c = lbfgs_opt(best_h, nf, deadline - 10, alphas=(100000, 500000, 2000000))

    # Phase 9: Final pair swap
    if time.time() < deadline - 2:
        best_h, best_c = random_pair_swap(best_h, nf, deadline - 1, rng, steps=10000)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
