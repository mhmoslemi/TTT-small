def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-15:

def binary_local_search(h, n, deadline, max_sweeps=2000, seed=42):
    """Greedy binary local search - swap to reduce max correlation"""
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max()
    best_h = h.copy()
    best_c = cur_c
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        np.random.RandomState(seed + sweep).shuffle(z_list)
        for i in z_list:
            o_list = list(ones)
            np.random.RandomState(seed + sweep + i).shuffle(o_list)
            for j in o_list:
                # Compute delta for swapping i(0->1) and j(1->0)
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max()

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    h = np.zeros(n)

    rng = np.random.RandomState(seed)

            h[pos] = 0.0
        h[best_pos] = 1.0
        available.remove(best_pos)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

        # Binary local search
        hb = to_binary(hh, nh)
        hb, cb = greedy_binary_ls(hb, nh, min(time.time() + 40, p4 - 10), max_sweeps=300, seed=seed + nh)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            n = nh
        # SA on this resolution
        if time.time() < p4 - 5:
            hb2, cb2 = sa_binary(hb, nh, min(time.time() + 30, p4 - 2), n_iter=200000, seed=seed + nh + 500)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()
                n = nh

    # Phase 5: Final aggressive SA
    if time.time() < deadline - 20:
        hb, cb = sa_binary(best_h, n, deadline - 15, n_iter=600000, seed=seed + 9000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 6: Final greedy + SLSQP polish
    if time.time() < deadline - 5:
        hb, cb = greedy_binary_ls(best_h, n, deadline - 3, max_sweeps=300, seed=seed + 9500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    if time.time() < deadline - 2:
        hp = project_feasible(best_h + 0.02 * rng.randn(n), n)
        hp = slsqp_optimize(hp, n, 1000000, 1500, deadline - 1)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
