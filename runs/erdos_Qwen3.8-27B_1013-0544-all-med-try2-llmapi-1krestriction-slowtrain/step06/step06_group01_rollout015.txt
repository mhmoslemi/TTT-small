def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

    try:

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

- Fix "1-h", solve for the best h (linear!)

This is the **alternating minimization** approach (like EM algorithm). At each step:
- Given h, the optimal "1-h" is fixed (it's just 1-h)
- Given "1-h" (call it g), solve: minimize corr_{k*} = ∑ h[i] * g[i+k*] subject to constraints

This is a linear program! And we can solve it with scipy.optimize.linprog or cvxpy.

But wait, we need ALL correlations to be small, not just one. So the LP would be:
minimize t
subject to: ∑_i h[i] * g[i+k] ≤ t for all k
            ∑_i h[i] = n/2
            0 ≤ h[i] ≤ 1

This is a linear program with (2n-1) + 1 + n variables and 2n-1 + 1 + 2n constraints. For n=250, that's about 749 variables and 749 constraints. Very manageable!

But the issue is that g = 1-h, so after solving for h, we update g = 1-h and solve again. This alternating minimization might converge to a local optimum, but it's worth trying.

def alternating_lp(h_init, n, max_iters=100, deadline=None):
    """Alternating LP minimization."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    h = h_init.copy()
    g = 1.0 - h
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        
        # Build LP: minimize t
        # Variables: h[0], h[1], ..., h[n-1], t
        # h_i >= 0, h_i <= 1, t >= 0
        # sum(h) = n/2
        # For each k: sum_i h[i] * g[i+k] <= t
        
        # c = [0, 0, ..., 0, 1] (minimize t)
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each shift k (0 to 2n-2): sum_i h[i] * g[i+k-(n-1)] <= t
        # i.e., sum_i h[i] * g[i+k-(n-1)] - t <= 0
        
        n_shifts = 2 * n - 1
        A_ub = np.zeros((n_shifts, n + 1))
        b_ub = np.zeros(n_shifts)
        
        for k in range(n_shifts):
            shift = k - (n - 1)  # actual lag
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    A_ub[k, i] = g[j]
            A_ub[k, n] = -1.0  # -t
        
        # Equality constraint: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            h_new = project_feasible(h_new, n)
            g = 1.0 - h_new
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        else:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        if m < 1e-15:
            return 0.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def alternating_lp(h_init, n, max_iters=2000, deadline=None, seed=42):
    """Alternating LP: alternate between solving for h given g=1-h_prev."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Precompute the constraint structure
    n_shifts = 2 * n - 1
    # For shift k (0 to 2n-2), the actual lag is k - (n-1)
    # A[k, i] = 1 if i + (k-(n-1)) is in [0, n), else 0
    # The constraint is: sum_i A[k,i] * h[i] * g[i + lag] <= t
    # But g depends on h, so we can't precompute A*g.
    # Instead, for each iteration, we build A_ub with g values.
    
    shifts = np.arange(n_shifts) - (n - 1)  # lags from -(n-1) to n-1
    # For each shift s and position i, the relevant g index is i+s
    # Precompute which (i, g_idx) pairs are valid for each shift
    # shift_indices[k] = (i_list, g_idx_list) where g_idx = i + shifts[k]
    shift_data = []
    for k in range(n_shifts):
        s = shifts[k]
        i_min = max(0, -s)
        i_max = min(n - 1, n - 1 - s)
        if i_min <= i_max:
            i_list = np.arange(i_min, i_max + 1)
            g_idx_list = i_list + s
            shift_data.append((k, i_list, g_idx_list))
        else:
            shift_data.append((k, np.array([], dtype=int), np.array([], dtype=int)))
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        g = 1.0 - h
        # Build LP
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        A_ub = np.zeros((n_shifts, n + 1))
        b_ub = np.zeros(n_shifts)
        
        for k, i_list, g_idx_list in shift_data:
            if len(i_list) > 0:
                A_ub[k, i_list] = g[g_idx_list]
            A_ub[k, n] = -1.0
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_new = np.clip(res.x[:n], 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                # Update h for next iteration (with slight noise to avoid cycling)
                h = h_new + 1e-8 * rng.randn(n)
                h = project_feasible(h, n)
            else:
                break
        except Exception:
            break
    
    return best_h, best_c
