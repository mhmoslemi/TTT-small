```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def grad_single_shift(h, n, m_idx):
    """Gradient of c[m_idx] where c = correlate(h, 1-h, 'full') * dx"""
    dx = 2.0 / n
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        # c[m] = dx * sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        # grad[j] = dx * [(1-h[j-k]) if j>=k,  -h[j+k] if j<=n-1-k]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] += -h[k:]
    else:
        # k < 0, let kk = -k
        kk = -k
        # c[m] = dx * sum_{i=0}^{n-1-kk} h[i]*(1-h[i+kk])
        # grad[j] = dx * [(1-h[j+kk]) if j<=n-1-kk,  -h[j-kk] if j>=kk]
        g[:n-kk] += (1.0 - h[kk:])
        g[kk:] += -h[:n-kk]
    return g * dx

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        g_m = grad_single_shift(h, n, m_idx)
        grad += w[m_idx] * g_m
    return grad

def obj_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return float(val)

def lbfgs_optimize(h, n, deadline, alphas=(100, 500, 2000, 10000, 50000)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        maxiter = min(500, int((deadline - time.time()) * 50))
        if maxiter < 10:
            break
        try:
            res = minimize(obj_smooth_max, best_h, jac=lambda x: grad_smooth_max(x, n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          args=(n, alpha) if False else None,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c
```

This is more expensive per pair than a fixed step, but each step is guaranteed to be optimal for that pair. So fewer steps are needed.

One idea: **block coordinate descent**. Instead of updating individual components or pairs, update blocks of consecutive components. For example, update h[0:10] while keeping h[10:] fixed. This gives more degrees of freedom per update but is still tractable.

For a block of size b, the sub-problem is:
min_{h[0:b]} max_k O_k
subject to: sum_{i=0}^{b-1} h[i] = target_block_sum (to maintain total sum)
            0 ≤ h[i] ≤ 1

This is a b-dimensional optimization problem. For b=5-10, we can solve it with scipy minimize in a few milliseconds.

```python
def block_cd_phase(h, n, deadline, block_size=10, step=1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        improved = False
        for start in range(0, n - block_size + 1, step):
            if time.time() > deadline:
                return best_h, best_c
            
            # Extract block
            block = best_h[start:start+block_size].copy()
            block_sum = block.sum()
            rest = best_h.copy()
            
            def block_obj(x):
                rest[start:start+block_size] = x
                return compute_c5(rest, n)
            
            def block_constraint(x):
                return x.sum() - block_sum
            
            try:
                from scipy.optimize import minimize as min_fn
                res = min_fn(block_obj, block, method='SLSQP',
                           constraints=[{'type': 'eq', 'fun': block_constraint}],
                           bounds=[(0,1)]*block_size,
                           options={'maxiter': 100, 'ftol': 1e-12})
                if res.success:
                    best_h[start:start+block_size] = np.clip(res.x, 0, 1)
                    # Re-project to maintain total sum
                    best_h = project_feasible(best_h, n)
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
            except:
                pass
        
        if not improved:
            break
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff /_tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def grad_single_shift(h, n, m_idx):
    dx = 2.0 / n
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[k:n] += (1.0 - h[0:nv])
        g[0:nv] += (-h[k:n])
    else:
        kk = -k
        nv = n - kk
        g[0:nv] += (1.0 - h[kk:n])
        g[kk:n] += (-h[0:nv])
    return g * dx

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        grad += w[m_idx] * grad_single_shift(h, n, m_idx)
    return grad

def _sm_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def _sm_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def _pen_obj(x, n, alpha, mu):
    val = _sm_obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def _pen_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alphas=(200, 1000, 5000, 20000, 100000)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        maxiter = min(400, max(20, int((deadline - time.time()) * 30)))
        if maxiter < 10:
            break
        try:
            res = minimize(_pen_obj, best_h, jac=_pen_grad, args=(n, alpha, 0.01),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def alternating_phase(h, n, deadline, top_m=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.15
    momentum = np.zeros(n)
    beta = 0.85
    no_improve = 0
    rng = np.random.RandomState(12345)

    while time.time() < deadline:
        c = compute_c5_vec(best_h, n)
        top_idx = np.argsort(c)[::-1][:top_m]
        grad_sum = np.zeros(n)
        for idx in top_idx:
            grad_sum += grad_single_shift(best_h, n, int(idx))
        grad_sum /= float(len(top_idx))
        momentum = beta * momentum + (1.0 - beta) * grad_sum
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_new = project_feasible(best_h - trial_lr * momentum, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                break
        if improved:

            lr = min(lr * 1.05, 0.5)

            lr *= 0.7
            if no_improve > 30:

                    best_h = h_try
                momentum *= 0.5
    return best_h, best_c

def pair_search_phase(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(67890)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    radius = min(80, n // 3)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            n_trials = 200
            for _ in range(n_trials):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                if abs(i - j) > radius:
                    continue
                if h_try_move(best_h, n, i, j, d) < best_c - 1e-14:
                    best_h[i] += d
                    best_h[j] -= d
                    best_h = project_feasible(best_h, n)
                    best_c = compute_c5(best_h, n)
                    improved = True
                elif h_try_move(best_h, n, i, j, -d) < best_c - 1e-14:
                    best_h[i] -= d
                    best_h[j] += d
                    best_h = project_feasible(best_h, n)
                    best_c = compute_c5(best_h, n)
                    improved = True
            if not improved:
                break
    return best_h, best_c

def h_try_move(h, n, i, j, d):
    hi = h[i] + d
    hj = h[j] - d
    if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
        return 1e10
    h[i] += d
    h[j] -= d
    c = compute_c5(h, n)
    h[i] -= d
    h[j] += d
    return c

def gradient_pair_phase(h, n, deadline, base_step=0.003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 10000]
    rng = np.random.RandomState(4242)
    for alpha in alphas:
        while time.time() < deadline:
            g = grad_smooth_max(best_h, n, alpha)
            improved = False
            n_trials = 300
            for _ in range(n_trials):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                move = -base_step * (g[i] - g[j])
                if abs(move) < 1e-12:
                    continue
                hi = best_h[i] + move
                hj = best_h[j] - move
                if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                    continue
                best_h[i] = hi
                best_h[j] = hj
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    best_h[i] -= move
                    best_h[j] += move
            if not improved:
                break
    return best_h, best_c

    if time.time() < deadline - 1:
        best_h, best_c = pair_search_phase(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
