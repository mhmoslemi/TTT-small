def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def vectorized_local_search(h, n, deadline, use_sa=False, T_init=0.01, T_final=0.001):
    """Fast vectorized binary local search with optional simulated annealing."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    lags = np.arange(-(n-1), n)  # (2n-1,)
    best_h = h.copy()
    best_c = cur_c
    
    no_sweeps = 0
    T = T_init
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        
        # Compute d_add for all zeros: (nz, 2n-1)
        im = zeros[:, None] - lags[None, :]
        ipm = zeros[:, None] + lags[None, :]
        d_add = np.zeros((nz, 2*n-1))
        mask_im = (im >= 0) & (im < n)
        d_add[mask_im] += 1.0 - h[im[mask_im]]
        mask_ipm = (ipm >= 0) & (ipm < n)
        d_add[mask_ipm] -= h[ipm[mask_ipm]]
        
        # Compute d_rem for all ones: (no, 2n-1)
        jm = ones[:, None] - lags[None, :]
        jpm = ones[:, None] + lags[None, :]
        d_rem = np.zeros((no, 2n-1))
        mask_jm = (jm >= 0) & (jm < n)
        d_rem[mask_jm] -= 1.0 - h[jm[mask_jm]]
        mask_jpm = (jpm >= 0) & (jpm < n)
        d_rem[mask_jpm] += h[jpm[mask_jpm]]

            new_vals = base[None, :] + d_rem  # (no, 2n-1)
            new_maxes = new_vals.max(axis=1) * dx
            
            if use_sa:

                best_j = np.argmin(new_maxes)
                best_new_c = new_maxes[best_j]
                if best_new_c < cur_c:

                T = T_final
        
        if not improved and not use_sa:

    return best_h, best_c
