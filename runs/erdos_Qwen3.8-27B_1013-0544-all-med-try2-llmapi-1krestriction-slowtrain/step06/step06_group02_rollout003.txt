    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for sh in range(2 * n - 1):
        if w[sh] < 1e-12:
            continue
        k = sh - (n - 1)
        wt = w[sh] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wt * (1.0 - h[k:])
            grad[k:] += wt * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wt * (1.0 - h[:nv])
            grad[:nv] += wt * (-h[kk:])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    v = smooth_max_val(c, alpha)
    if mu > 0:
        v += mu * (x.sum() - n / 2.0) ** 2
    return v

def _obj_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_once(h, n, alpha, maxiter, mu):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, jac=_obj_grad, args=(n, alpha, mu),
                  method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                           'maxfun': maxiter * 5, 'maxls': 25})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def progressive_optimize(h, n, deadline, maxiter=250):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [5, 20, 80, 300, 1000, 5000, 20000, 80000, 300000, 1000000, 5000000]
    n_alphas = len(alphas)
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 2:
            break
        remaining = deadline - time.time()
        frac = remaining / max(1, n_alphas - i)
        max_t = max(3, min(60, frac))
        mu = max(10.0, alpha * 0.005)
        mi = max(30, int(maxiter * min(1.0, max_t / 60.0)))
        h_new, c_new = lbfgs_once(best_h, n, alpha, mi, mu)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def local_search(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        improved = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for s in (1, -1):
                        vi = h[i] + s * d
                        vj = h[j] - s * d
                        if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
                            h[i] = vi
                            h[j] = vj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] += s * d
                                h[j] -= s * d
            if not improved:
                break
    return h, best_c

def grad_local_search(h, n, deadline, step=0.005):
    best_c = compute_c5(h, n)
    alpha = 5000
    rng = np.random.RandomState(12345)
    while time.time() < deadline:
        improved = False
        g = smooth_max_grad(h, n, alpha)
        for _ in range(500):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            move = step * (g[j] - g[i])
            if abs(move) < 1e-8:
                continue
            vi = h[i] + move
            vj = h[j] - move
            if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
                h[i] = vi
                h[j] = vj
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    h[i] -= move
                    h[j] += move
        if not improved:
            step *= 0.5
            alpha = min(alpha * 2, 50000)
            if step < 1e-5:
                break
    return h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def make_start(n, rng, mode=0):
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        x = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x / n)
    elif mode == 2:
        x = np.arange(n)
        h = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x / n)
    elif mode == 3:
        h = rng.rand(n)
        thr = np.percentile(h, 50)
        h = np.where(h > thr, 0.8, 0.2)
    elif mode == 4:
        h = rng.rand(n)
        np.clip(h, 0.2, 0.8, out=h)
    else:
        x = np.arange(n)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * 3 * x / n) + 0.15 * np.cos(2 * np.pi * 5 * x / n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Progressive optimization on parent
    t1 = t0 + min(200, (deadline - t0) * 0.22)
    h1, c1 = progressive_optimize(best_h, n_init, t1, maxiter=250)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Gradient-guided local search
    if time.time() < deadline - 80:
        t2 = min(time.time() + 60, deadline - 80)
        best_h, best_c = grad_local_search(best_h, best_n, t2, step=0.004)

    # Phase 3: Pair local search
    if time.time() < deadline - 60:
        t3 = min(time.time() + 50, deadline - 60)
        best_h, best_c = local_search(best_h, best_n, t3, radius=40)

    # Phase 4: Try alternative n values with diverse starts
    n_trials = [150, 200, 250, 350, 400]
    n_trials = [x for x in n_trials if abs(x - best_n) > 30]
    for n_try in n_trials:
        if time.time() > deadline - 50:
            break
        alloc = min(70, (deadline - time.time()) * 0.15)
        if alloc < 25:
            break
        for mode in [1, 3, 4]:
            if time.time() > deadline - 20:
                break
            h_s = make_start(n_try, rng, mode)
            # Also try interpolation from best
            if mode == 1:
                h_s = interp_h(best_h, best_n, n_try)
            sub_dead = time.time() + alloc
            h_s, c_s = progressive_optimize(h_s, n_try, sub_dead, maxiter=150)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
                best_n = n_try
            if c_s < best_c + 0.0005 and time.time() < deadline - 15:
                h_s2, c_s2 = local_search(h_s, n_try, min(time.time() + 12, deadline - 12), radius=30)
                if c_s2 < best_c:
                    best_c = c_s2
                    best_h = h_s2
                    best_n = n_try

    # Phase 5: Final high-alpha refinement
    if time.time() < deadline - 15:
        t5 = min(time.time() + 30, deadline - 12)
        alphas_hi = [100000, 500000, 2000000, 10000000]
        for a in alphas_hi:
            if time.time() > t5:
                break
            mu = max(50.0, a * 0.003)
            h5, c5 = lbfgs_once(best_h, best_n, a, 150, mu)
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 6: Final fine local search
    if time.time() < deadline - 2:
        best_h, best_c = local_search(best_h, best_n, deadline - 1, radius=30)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
