```python
                    val -= (1.0 - h[ip])  # remove first-index contribution
                im = i - l
                if 0 <= im < n:
                    val += h[im]  # remove second-index contribution (negate the -h[im] term)
                R[i, l + n - 1] = val
        if h[i] == 0:
            # A[i]: h[i] 0->1
            for l in range(-n+1, n):
                if l == 0:
                    continue
                val = 0.0
                ip = i + l
                if 0 <= ip < n:
                    val += (1.0 - h[ip])  # add first-index contribution
                im = i - l
                if 0 <= im < n:
                    val -= h[im]  # add second-index contribution
                A[i, l + n - 1] = val
    return R, A
```

For candidate generation, I'll focus on:
- Moving 1s from positions that are "active" at lag ℓ* (h[i]=1 and h[i+ℓ*]=0 for ℓ*>0)

```python
def swap_effect(h, n, i, j):
    """Compute the change in corr vector for swapping h[i]:1->0, h[j]:0->1."""
    # corr index = lag + n - 1
    # For h[i] 1->0:
    #   As first index: for lag l with 0<=i+l<n: corr[i+l - i + n-1]... 
    #   Let me just use the formula: delta at corr position p
    #   p corresponds to lag l = p - (n-1)
    #   h[i] as first index contributes to corr[p] when i + (p-(n-1)) is valid
    #   i.e., p = (n-1) + l where l ranges from -i to (n-1-i)
    #   The contribution is h[i]*(1-h[i+l]) = 1*(1-h[i+l])
    #   When h[i] goes to 0, this contribution disappears: delta -= (1-h[i+l])
    #   
    #   h[i] as second index: contributes to corr[p] when (p-(n-1)) + i is valid as first index
    #   i.e., the first index is i+l where l = p-(n-1), and we need 0<=i+l<n
    #   The contribution is h[i+l]*(1-h[i]) = h[i+l]*1 = h[i+l]
    #   When h[i] goes to 0, (1-h[i]) goes to 1, so the contribution becomes h[i+l]*1 = h[i+l]
    #   Old contribution was h[i+l]*0 = 0
    #   So delta += h[i+l]
    
    # Let me just compute it directly for correctness
    delta = np.zeros(2*n-1)
    # Remove contribution of h[i]=1
    for p in range(2*n-1):
        l = p - (n-1)
        # h[i] as first index: need i+l valid
        if 0 <= i+l < n and l != 0:
            delta[p] -= (1.0 - h[i+l])
        # h[i] as second index: need i-l valid (first index is i-l, second is i)
        if 0 <= i-l < n and l != 0:
            delta[p] += h[i-l]
    # Add contribution of h[j]=1 (was 0)
    for p in range(2*n-1):
        l = p - (n-1)
        if 0 <= j+l < n and l != 0:
            delta[p] += (1.0 - h[j+l])
        if 0 <= j-l < n and l != 0:
            delta[p] -= h[j-l]
    return delta
```

This is O(n) per evaluation. For n=300, that's 600 operations. With 100K candidate swaps, that's 60M operations per full scan. Too slow in Python.

```python
def precompute_influence(h, n):
    """Precompute influence vectors.
    infl[i] = change in corr when h[i] increases by 1 (from 0 to 1).
    So for h[i] 1->0, the change is -infl[i].
    For h[j] 0->1, the change is +infl[j].
    """
    infl = np.zeros((n, 2*n-1))
    for i in range(n):
        for p in range(2*n-1):
            l = p - (n-1)
            if l == 0:
                continue
            val = 0.0
            # h[i] as first index
            if 0 <= i+l < n:
                val += (1.0 - h[i+l])
            # h[i] as second index  
            if 0 <= i-l < n:
                val -= h[i-l]
            infl[i, p] = val
    return infl
```

This is O(n²) to compute. For n=300, that's 180K operations. Fine.

Then for a swap (i: 1→0, j: 0→1): delta = -infl[i] + infl[j]

I can vectorize the evaluation of many swaps at once:
- For a fixed i, the delta for all j is: -infl[i] + infl[j] for all j
- This is a matrix operation: infl[j_all] - infl[i] (broadcasting)

So I can evaluate all swaps for a fixed i in O(n * 2n) = O(n²) = O(180K) for n=300.

And for all i: O(n³) = 27M. Still too slow for pure Python, but with numpy it should be fast (a few ms per full scan).

```python
def best_swap(corr, infl, h, n, top_k=5):
    """Find the best swap to reduce max corr."""
    M = corr.max()
    # Critical lags: those with corr >= M - 2
    crit_mask = corr >= M - 2.5
    crit_idx = np.where(crit_mask)[0]
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_score = M
    best_i = -1
    best_j = -1
    
    for i in ones:
        # Delta for all j: -infl[i] + infl[j]
        # New corr at critical lags: corr[crit_idx] + (-infl[i, crit_idx] + infl[j, crit_idx])
        base = corr[crit_idx] - infl[i, crit_idx]  # shape: (|crit|,)
        # For each j: new_vals = base + infl[j, crit_idx], score = max(new_vals)
        # We want to minimize max over all lags, but as a proxy, minimize max over critical lags
        # (lags outside critical set can increase by at most 2, so if they're < M-2, they'll be < M)
        new_crit = base[None, :] + infl[zeros, crit_idx]  # shape: (|zeros|, |crit|)
        scores = new_crit.max(axis=1)
        # Also need to check that non-critical lags don't exceed the score
        # For simplicity, use the critical max as proxy
        j_local = np.argmin(scores)
        if scores[j_local] < best_score - 1e-10:
            best_score = scores[j_local]
            best_i = i
            best_j = zeros[j_local]
    
    return best_i, best_j, best_score
```

This is O(n_ones * n_zeros * |crit|) per call. For n=300, n_ones=150, n_zeros=150, |crit|~20: 450K operations. Fast enough!

But wait, I should also verify that non-critical lags don't blow up. The max possible increase at any non-critical lag is 2 (since infl values are at most 1 in absolute value for binary h, and we have two changes). So if a non-critical lag has corr ≤ M-3, it can't exceed M-1 after the swap. If it has corr = M-2, it could reach M. So I should include lags with corr ≥ M-3 in the critical set to be safe.

Actually, let me be more precise. For a binary sequence, the influence at any single lag from a single position change is at most 1 (since h values are 0 or 1, so (1-h) is 0 or 1). For a swap (two position changes), the total change at any lag is at most 2.

So if corr[lag] ≤ M - 3, then after any swap, corr[lag] + delta ≤ M - 3 + 2 = M - 1 < M. Safe.
If corr[lag] = M - 2, it could reach M. 
If corr[lag] = M - 1, it could reach M + 1.

So I should include lags with corr ≥ M - 2 in the critical set. Let me use M - 2.5 to be safe (include lags within 3 of the max).
