        ak = a + k
        ak_valid = (ak >= 0) & (ak < n)
        safe_ak = np.clip(ak, 0, n-1)

        ak2_valid = (ak2 >= 0) & (ak2 < n)
        safe_ak2 = np.clip(ak2, 0, n-1)
        term2_a = np.where(ak2_valid, h[safe_ak2], 0.0)
        prof_a = (term1_a - term2_a) * (2.0/n)
        
        bk = b + k

        bk2 = b - k

def profile_at(h, idx, n):
    """Compute the profile vector for position idx.
    Returns array of length 2n-1 where entry m corresponds to shift k=m-(n-1).
    prof[k] = (1 - h[idx+k]) - h[idx-k] (with h=0 out of range), times dx."""
    k = np.arange(-(n-1), n)
    dx = 2.0 / n
    ak = idx + k
    akv = (ak >= 0) & (ak < n)
    term1 = np.where(akv, 1.0 - h[np.clip(ak, 0, n-1)], 1.0)
    ak2 = idx - k
    ak2v = (ak2 >= 0) & (ak2 < n)
    term2 = np.where(ak2v, h[np.clip(ak2, 0, n-1)], 0.0)
    return (term1 - term2) * dx

def find_best_delta(c, diff_prof, max_delta, iters=15):
    """Find delta in [0, max_delta] that minimizes max(c + delta * diff_prof)."""
    if max_delta < 1e-12:
        return 0.0
    lo, hi = 0.0, max_delta
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (c + m1 * diff_prof).max()
        f2 = (c + m2 * diff_prof).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
