from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def gradient_c5(h, n):
    """Compute gradient of max g_k with respect to h.
    Returns (gradient, max_value, active_k)
    """
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    ks = int(np.argmax(g))
    # ks is index into the full correlation array of length 2n-1
    # lag = ks - (n-1)
    lag = ks - (n - 1)
    
    # For the active constraint g_lag:
    # ∂g_lag/∂h[j] = (1 - h[j+lag]) - h[j-lag] (with boundary handling)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        # Term 1: h[j] appears as h[i] with i=j, need j+lag in [0,n-1]
        if 0 <= j + lag < n:
            val += (1.0 - h[j + lag])
        # Term 2: h[j] appears as h[i+lag] with i+lag=j, i.e., i=j-lag, need j-lag in [0,n-1]
        if 0 <= j - lag < n:
            val -= h[j - lag]
        grad[j] = val
    return grad * dx, m, ks

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    """Gradient descent on smoothed max with projection."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    alpha = alpha_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        
        # Soft max gradient: use top-k constraints
        # Find constraints close to max
        threshold = m - 0.01
        active_mask = g >= threshold
        n_active = active_mask.sum()
        
        if n_active == 0:
            active_mask = np.zeros(2*n-1, dtype=bool)
            active_mask[np.argmax(g)] = True
        
        # Average gradient over active constraints
        grad = np.zeros(n)
        for ks in np.where(active_mask)[0]:
            lag = ks - (n - 1)
            for j in range(n):
                val = 0.0
                if 0 <= j + lag < n:
                    val += (1.0 - h[j + lag])
                if 0 <= j - lag < n:
                    val -= h[j - lag]
                grad[j] += val
        grad *= dx / n_active
        
        # Gradient step with line search
        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break
        
        # Try decreasing step sizes
        for f in [0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]:
            step = f * dx / ng  # scale by dx for consistency
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                break
        else:
            # No improvement, reduce alpha and try smoothed
            alpha *= 0.9
            if alpha < 1e-6:
                break
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

    """Compute gradient of g_lag with respect to h."""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[:m] += (1.0 - h[lag:])
        grad[lag:] -= h[:m]
    else:
        l = -lag
        m = n - l
        grad[l:] += (1.0 - h[:m])
        grad[:m] -= h[l:]
    return grad

def project_feasible(h, n):

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    """Subgradient method with analytical gradient."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = initial_lr
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full')
        m = g.max()
        c = m * dx

        tol = max(1e-4, 0.005 * m)
        active_ks = np.where(g >= m - tol)[0]

        grad = np.zeros(n)
        for ks in active_ks:

            grad[:m] += 1.0 - h[lag:]

        m = n - l
        if m > 0:
            grad[l:] += 1.0 - h[:m]

    m = g.max()
    active = np.where(g >= m - tol)[0]
    grad = np.zeros(n)
    for ks in active:

    return grad / len(active), m * (2.0 / n)

def lbfgs_smoothed(h, n, deadline, alpha, maxiter=5000):

    def func(x):

        m = g.max()
        s = alpha * (g - m)
        ms = s.max()
        return m + (ms + np.log(np.exp(s - ms).sum())) / alpha
    def func_grad(x):

        ms = s.max()

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def grad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += 1.0 - h[lag:]
            grad[lag:] -= h[:m]
    else:
        l = -lag
        m = n - l
        if m > 0:
            grad[l:] += 1.0 - h[:m]
            grad[:m] -= h[l:]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1-x, mode='full') * dx
    m = g.max()
    s = alpha * (g - m)
    ms = s.max()
    return m + (ms + np.log(np.exp(s - ms).sum())) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1-x, mode='full') * dx
    m = g.max()
    s = alpha * (g - m)
    ms = s.max()
    es = np.exp(s - ms)
    w = es / es.sum()
    grad = np.zeros(n)
    for ks in range(2*n-1):
        wk = w[ks]
        if wk > 1e-10:
            grad += wk * grad_for_lag(x, ks - (n-1), n)
    return grad

def slsqp_solve(h, n, deadline, alpha, maxiter=2000):
    def obj(x):
        return smooth_obj(x, n, alpha)
    def obj_grad(x):
        return smooth_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n/2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)]*n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def subgrad_polish(h, n, deadline, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    dx = 2.0 / n
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g_raw = np.correlate(h, 1-h, mode='full')
        m_raw = g_raw.max()
        c = m_raw * dx
        # Active constraints
        tol = max(dx * 0.05, 0.001 * m_raw)
        active = np.where(g_raw >= m_raw - tol)[0]
        grad = np.zeros(n)
        for ks in active:
            grad += grad_for_lag(h, ks - (n-1), n)
        grad /= (len(active) * dx)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        improved = False
        for f in [1.0, 0.5, 0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < c - 1e-15:

                improved = True
                break
        if improved:

            if c2 < best_c - 1e-15:

                best_h = h.copy()
                stagnant = 0
            else:

                lr *= 0.8
        else:
            lr *= 0.9
            stagnant += 1
        if stagnant > 5000 or lr < 1e-14:
            break
    return best_h, best_c
