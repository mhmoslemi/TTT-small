```python
def simulated_annealing(h, n, t_start, t_end, n_iter, seed=42):
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1-h, mode='full')
    cur_val = corr.max()
    best_val = cur_val
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    shift = np.arange(-(n-1), n)
    
    for it in range(n_iter):
        T = t_start * (t_end/t_start) ** (it / n_iter)
        # Pick random i from zeros, j from ones
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta
        delta = np.zeros(2*n-1)
        ms = (i + shift >= 0) & (i + shift < n)
        delta[ms] += (1 - h[i + shift[ms]])
        ms = (i - shift >= 0) & (i - shift < n)
        delta[ms] -= h[i - shift[ms]]
        ms = (j + shift >= 0) & (j + shift < n)
        delta[ms] -= (1 - h[j + shift[ms]])
        ms = (j - shift >= 0) & (j - shift < n)
        delta[ms] += h[j - shift[ms]]
        
        new_corr = corr + delta
        new_val = new_corr.max()
        delta_E = new_val - cur_val
        
        if delta_E < 0 or rng.rand() < np.exp(-delta_E / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_val = new_val
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
    
    return best_h, best_val
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n, seed=42):
    """Round to binary with exactly n/2 ones, using stochastic rounding"""
    rng = np.random.RandomState(seed)
    target = n // 2
    # Sort by value descending, pick top n/2 with some randomness
    h = h.copy()
    # Add small noise to break ties
    h += rng.uniform(0, 0.01, n)
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def grad_step(h, n, alpha, lr):
    """One gradient step to maximize min autocorrelation"""
    a = np.correlate(h, h, mode='full')
    a_min = a.min()
    exp_neg = np.exp(-alpha * (a - a_min))
    w = exp_neg / exp_neg.sum()
    # g[k] = correlate(w, h, 'full')[k] for k=0..n-1
    g = np.correlate(w, h, mode='full')[:n]
    h_new = h + 2.0 * lr * g
    return project(h_new, n)

def grad_descent(h, n, deadline, alpha_start=1.0, alpha_end=500.0, 
                 lr_start=0.1, n_phase=5):
    """Multi-phase gradient descent with increasing alpha"""
    h = h.copy()
    phase_time = (deadline - time.time()) / n_phase
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for phase in range(n_phase):
        alpha = alpha_start * (alpha_end / alpha_start) ** (phase / max(n_phase-1, 1))
        phase_deadline = time.time() + phase_time
        lr = lr_start / (1 + 0.1 * phase)
        
        while time.time() < phase_deadline:
            h = grad_step(h, n, alpha, lr)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

def simulated_annealing(h, n, t_start=0.01, t_end=0.0001, 
                        n_iter=200000, deadline=None, seed=42):
    """Simulated annealing on binary solutions"""
    rng = np.random.RandomState(seed)
    h = binary_round(h, n, seed)
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_val = corr.max()
    best_val = cur_val
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    shift = np.arange(-(n-1), n)
    
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / n_iter)
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta
        delta = np.zeros(2*n-1)
        ms = (i + shift >= 0) & (i + shift < n)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
```
```python
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def binary_project(h, n, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    scored = h + rng.uniform(0, 0.05, n)
    idx = np.argsort(scored)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

    h_new = h + 2.0 * lr * g
    return project(h_new, n)
```

    alpha = alpha_start
    for phase in range(n_phases):
        pdl = time.time() + phase_t
        while time.time() < pdl:
            h = grad_step(h, n, alpha, lr)
            alpha *= 1.003
        alpha *= 3.0
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def sa_run(h, n, deadline, t_start=3.0, t_end=0.005, seed=42):
    rng = np.random.RandomState(seed)
    h = binary_project(h, n, seed).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_m = float(corr.max())
    best_m = cur_m
    best_h = h.copy()
    half = n // 2
    one_pos = np.where(h == 1)[0].copy()
    zero_pos = np.where(h == 0)[0].copy()
    shift = np.arange(-(n - 1), n)
    d_buf = np.zeros(2 * n - 1)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 3000000.0)
        T = t_start * (t_end / t_start) ** frac
        ii = rng.randint(half)
        jj = rng.randint(half)
        i = int(zero_pos[ii])
        j = int(one_pos[jj])
        d_buf[:] = 0.0
        ip = i + shift
        m = (ip >= 0) & (ip < n)
        d_buf[m] += 1.0 - h[ip[m]]
        im = i - shift
        m = (im >= 0) & (im < n)
        d_buf[m] -= h[im[m]]
        jp = j + shift
        m = (jp >= 0) & (jp < n)
        d_buf[m] -= 1.0 - h[jp[m]]
        jm = j - shift
        m = (jm >= 0) & (jm < n)
        d_buf[m] += h[jm[m]]
        nc = corr + d_buf
        nv = float(nc.max())
        dE = nv - cur_m
        if dE < 0 or (T > 1e-8 and rng.random_sample() < np.exp(-dE / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = nc
            cur_m = nv
            zero_pos[ii] = j
            one_pos[jj] = i
            if cur_m < best_m:
                best_m = cur_m
                best_h = h.copy()
    return best_h, best_m

def greedy_run(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = binary_project(h, n, seed).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_m = float(corr.max())
    best_m = cur_m
    best_h = h.copy()
    half = n // 2
    shift = np.arange(-(n - 1), n)
    one_list = np.where(h == 1)[0].tolist()
    zero_list = np.where(h == 0)[0].tolist()
    d_buf = np.zeros(2 * n - 1)
    while time.time() < deadline:
        improved = False
        z_order = rng.permutation(len(zero_list)).tolist()
        for zi in z_order:
            if time.time() > deadline:
                break
            i = zero_list[zi]
            d_buf[:] = 0.0
            ip = i + shift
            m = (ip >= 0) & (ip < n)
            d_buf[m] += 1.0 - h[ip[m]]
            im = i - shift
            m = (im >= 0) & (im < n)
            d_buf[m] -= h[im[m]]
            base = corr + d_buf
            best_j = -1
            best_v = cur_m
            for j in one_list:
                d_buf[:] = 0.0
                jp = j + shift
                m = (jp >= 0) & (jp < n)
                d_buf[m] += h[jp[m]]
                jm = j - shift
                m = (jm >= 0) & (jm < n)
                d_buf[m] += h[jm[m]]
                d_buf -= 1.0
                nv = float((base + d_buf).max())
                if nv < best_v:
                    best_v = nv
                    best_j = j
            if best_j >= 0 and best_v < cur_m - 1e-12:
                j = best_j
                d_buf[:] = 0.0
                ip = i + shift
                m = (ip >= 0) & (ip < n)
                d_buf[m] += 1.0 - h[ip[m]]
                im = i - shift
                m = (im >= 0) & (im < n)
                d_buf[m] -= h[im[m]]
                jp = j + shift
                m = (jp >= 0) & (jp < n)
                d_buf[m] -= 1.0 - h[jp[m]]
                jm = j - shift
                m = (jm >= 0) & (jm < n)
                d_buf[m] += h[jm[m]]
                h[i] = 1.0
                h[j] = 0.0
                corr += d_buf
                cur_m = best_v
                zero_list[zi] = j
                one_list[one_list.index(j)] = i
                improved = True
                if cur_m < best_m:
                    best_m = cur_m
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_m

def make_pulsed_init(n, seed, n_pulses=5):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    centers = rng.choice(n, n_pulses, replace=False)
    width = max(1, n // (n_pulses * 4))
    for c in centers:
        for d in range(-width, width + 1):
            idx = (c + d) % n
            h[idx] = 1.0
    return project(h, n)
