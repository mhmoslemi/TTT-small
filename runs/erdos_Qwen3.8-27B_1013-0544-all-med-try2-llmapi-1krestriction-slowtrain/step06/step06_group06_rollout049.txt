def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_at_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    j_start = max(0, k - (n - 1))
    j_end = min(n - 1, k)
    if j_start > j_end:
        return grad
    j_idx = np.arange(j_start, j_end + 1)
    kmj = k - j_idx
    grad[j_idx] = dx * (1.0 - 2.0 * h[kmj])
    return grad

def smooth_val_and_grad(x, n, alpha):
    g = compute_all_g(x, n)

    ag = alpha * g
    ag_max = ag.max()
    exp_ag = np.exp(ag - ag_max)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_at_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    j_start = max(0, k - (n - 1))
    j_end = min(n - 1, k)
    if j_start > j_end:
        return grad
    j_idx = np.arange(j_start, j_end + 1)
    kmj = k - j_idx
    grad[j_idx] = dx * (1.0 - 2.0 * h[kmj])
    return grad

def smooth_val_and_grad(x, n, alpha):
    g = compute_all_g(x, n)
    dx = 2.0 / n
    ag = alpha * g
    ag_max = ag.max()
    exp_ag = np.exp(ag - ag_max)
    w = exp_ag / exp_ag.sum()
    val = ag_max / alpha + np.log(exp_ag.sum()) / alpha
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            kmj = k - j_idx
            grad[j_idx] += alpha * w[k] * dx * (1.0 - 2.0 * x[kmj])
    return val, grad

def _slsqp_obj(x, n, alpha):
    v, _ = smooth_val_and_grad(x, n, alpha)
    return v

def _slsqp_grad(x, n, alpha):
    _, g = smooth_val_and_grad(x, n, alpha)
    return g

def _slsqp_cons(x, n, alpha):
    return x.sum() - n / 2.0

def smooth_slsqp(h, n, alpha, deadline, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'args': (n, alpha)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                      bounds=bounds, constraints=cons, args=(n, alpha),
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_subgrad(h, n, deadline, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        # Collect all indices achieving (near) max for averaged subgradient
        thresh = m - 1e-10
        k_indices = np.where(g >= thresh)[0]
        grad = np.zeros(n)
        for k in k_indices:
            grad += subgradient_at_k(h, n, k)
        grad /= len(k_indices)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-12:
                h = ht
                found = True
                break
        if found:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 5.0)
        else:
            lr *= 0.85
            stall += 1
            if lr < 1e-14 or stall > 500:
                break
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=500):
    target = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = sorted(np.where(h_bin == 1)[0].tolist())
    zeros = sorted(np.where(h_bin == 0)[0].tolist())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(i)
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin.copy(), best_c

    for it in range(max_iters):
        if time.time() > deadline:
            break
        temp = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        ri = rng.randint(len(zeros))
        rj = rng.randint(len(ones))
        zi = zeros[ri]

            ones[rj] = zi
            zeros[ri] = oj
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[zi], h_bin[oj] = 0.0, 1.0

    best_h = h.copy()
