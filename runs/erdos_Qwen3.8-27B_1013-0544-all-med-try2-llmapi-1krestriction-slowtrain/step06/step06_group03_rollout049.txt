def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_onto_box_sum(h, n, target=None):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    if target is None:
        target = n / 2.0
    h = h.copy()
    # Use bisection to find tau such that sum(clamp(h - tau, 0, 1)) = target
    lo, hi = -1.0, 2.0  # range for tau
    for _ in range(200):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - projected.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-boundary elements
        mask = (projected > 1e-10) & (projected < 1.0 - 1e-10)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
    return np.clip(projected, 0.0, 1.0)

            grad[l:] -= h[:n-l]

    return grad, max_val

def compute_subgradient(h, n, active_mask):
    """Compute subgradient of max_k f_k at h, given active_mask over lags."""
    grad = np.zeros(n)
    n_active = active_mask.sum()
    if n_active == 0:
        return grad
    for k in range(2*n-1):
        if not active_mask[k]:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # valid i: 0 <= i < n-lag (so i+lag < n)
            # ∂f/∂h[i+lag] += (1-h[i])  -> positions lag, lag+1, ..., n-1 get += (1-h[0..n-lag-1])
            # ∂f/∂h[i] += -h[i+lag]    -> positions 0, 1, ..., n-lag-1 get += -h[lag..n-1]
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
        else:
            l = -lag
            # valid i: l <= i < n (so i+lag = i-l >= 0)
            # ∂f/∂h[i+lag] += (1-h[i])  -> positions 0, ..., n-l-1 get += (1-h[l..n-1])
            # ∂f/∂h[i] += -h[i+lag]    -> positions l, ..., n-1 get += -h[0..n-l-1]
            grad[:n-l] += (1.0 - h[l:])
            grad[l:] -= h[:n-l]
    return grad / n_active

        else:
            lo_T = mid_T
    T_opt = hi_T
    lower_b = delta_min
    upper_b = delta_max
    if pos_m.any():
        upper_b = min(upper_b, float(np.min((T_opt - f[pos_m]) / d[pos_m])))
    if neg_m.any():
        lower_b = max(lower_b, float(np.max((T_opt - f[neg_m]) / d[neg_m])))
    if lower_b <= upper_b:
        delta = (lower_b + upper_b) / 2.0
    else:
        delta = 0.0
    return delta, T_opt

def exact_pair_descent(h, n, deadline, K=8, max_epochs=200):
    h = h.copy()
    best_h = h.copy()
    c5, corr = compute_c5(h, n)
    best_c = c5
    dx = 2.0 / n
    for epoch in range(max_epochs):
        if time.time() > deadline:
            break
        improved = False
        grad = subgradient_direction(h, n, corr)
        neg_idx = np.argsort(grad)[:K]
        pos_idx = np.argsort(grad)[-K:][::-1]
        for i in neg_idx:
            if time.time() > deadline:
                return best_h, best_c
            for j in pos_idx:
                if i == j:
                    continue
                if grad[i] >= -1e-10 or grad[j] <= 1e-10:
                    continue
                delta, T_opt = pair_optimal_delta(h, n, i, j, corr)
                if T_opt * dx < best_c - 1e-14 and abs(delta) > 1e-14:
                    h[i] += delta
                    h[j] -= delta
                    h = project_box_sum(h, n)
                    best_c = T_opt * dx
                    best_h = h.copy()
                    improved = True
                    c5, corr = compute_c5(h, n)
                    grad = subgradient_direction(h, n, corr)
                    break
        if not improved:
            break
    return best_h, best_c

def subgradient_descent(h, n, deadline, init_step=0.05, cooling=0.998, max_iter=5000):
    h = h.copy()
    best_h = h.copy()
    c5, corr = compute_c5(h, n)
    best_c = c5
    dx = 2.0 / n
    step = init_step
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = subgradient_direction(h, n, corr)
        h_new = h - step * grad
        h_new = project_box_sum(h_new, n)
        new_c5, new_corr = compute_c5(h_new, n)
        if new_c5 < c5:
            h = h_new
            corr = new_corr
            c5 = new_c5
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        else:
            h_half = h - 0.5 * step * grad
            h_half = project_box_sum(h_half, n)
            half_c5, half_corr = compute_c5(h_half, n)
            if half_c5 < c5:
                h = h_half
                corr = half_corr
                c5 = half_c5
                if c5 < best_c:
                    best_c = c5
                    best_h = h.copy()
        step *= cooling
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_box_sum(h_start.copy(), n)
    best_h, best_c = h, compute_c5(h, n)[0]
    t1 = min(time.time() + 60, deadline - 80)
    h2, c2 = subgradient_descent(h, n, t1)
    if c2 < best_c:
        best_h, best_c = h2, c2
    t2 = min(time.time() + 120, deadline - 20)
    h3, c3 = exact_pair_descent(best_h, n, t2, K=10, max_epochs=300)
    if c3 < best_c:
        best_h, best_c = h3, c3
    h4, c4 = exact_pair_descent(best_h, n, deadline - 5, K=15, max_epochs=500)
    if c4 < best_c:
        best_h, best_c = h4, c4
    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)[0]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(120, 0.5)
        n_init = 120

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)[0]

    # Phase 1: Optimize at initial resolution
    alloc1 = min(250, (deadline - t0) * 0.3)
    h1, c1 = optimize_at_n(best_h, n_init, t0 + alloc1)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Try other resolutions
    x_full = np.linspace(0, 2.0, n_init + 1)
    for n_try in sorted(set([80, 100, 120, 150, 200, 250, 300, 400])):
        if time.time() > deadline - 120:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2.0, n_try + 1)
        h_interp = np.interp(x_new[:-1], x_full[:-1], h_init)
        h_interp = project_box_sum(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc < 40:
            break
        ht, ct = optimize_at_n(h_interp, n_try, time.time() + alloc)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 3: Perturbation restarts on best
    n_best = len(best_h)
    if time.time() < deadline - 100:
        n_restarts = max(3, int((deadline - time.time() - 60) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.04 + 0.12 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_box_sum(hp, n_best)
            alloc = min(60, (deadline - time.time() - 30) * 0.3)
            if alloc < 25:
                break
            ht, ct = optimize_at_n(hp, n_best, time.time() + alloc)
            if ct < best_c:
                best_h, best_c = ht, ct

    # Phase 4: Final heavy pair descent
    if time.time() < deadline - 15:
        ht, ct = exact_pair_descent(best_h, n_best, deadline - 3, K=20, max_epochs=1000)
        if ct < best_c:
            best_h, best_c = ht, ct

    best_h = project_box_sum(best_h, n_best)
    best_c = compute_c5(best_h, n_best)[0]
    return best_h, best_c, n_best
