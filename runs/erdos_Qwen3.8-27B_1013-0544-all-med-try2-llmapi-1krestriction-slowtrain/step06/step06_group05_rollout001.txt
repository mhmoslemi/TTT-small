        if abs(diff) < 1e-14:
            break

            tot = room.sum()
            if tot < 1e-14: break

            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h

    # c[k] = sum_j h[j] * (1 - h[j + k - (n-1)])
    # S[i, k] = (1 - h[i+k-(n-1)]) if valid - h[i-k+(n-1)] if valid
    m = 2 * n - 1

    for i in range(n):

        k_lo = max(0, n-1-i)
        k_hi = min(m-1, 2*n-2-i)
        j = np.arange(k_lo, k_hi+1) - (n-1) + i  # = i + k - (n-1)
        S[i, k_lo:k_hi+1] += (1.0 - h[j])
        
        # Second term: h[i] as second factor (in 1-h), valid when 0 <= i-k+(n-1) < n
        k_lo2 = max(0, i-n+1)
        k_hi2 = min(m-1, i)
        j2 = i - (np.arange(k_lo2, k_hi2+1)) + (n-1)  # = i - k + (n-1)
        S[i, k_lo2:k_hi2+1] -= h[j2]
    return S

    if d_hi < d_lo:
        return 0.0, c.max()

        return float(np.max(c + delta * T))
    
    # Quick check: if the optimal is at a boundary
    f_lo = f(d_lo)

        return d_hi, f_hi
