```python
def gradient_step(h, n, alpha, step):
    """Take one gradient descent step on smooth max, then project."""
    target_sum = n / 2.0
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        weights = np.exp(shifted - mx)
    else:
        weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    idx = np.arange(2*n - 1)
    for w, k in zip(weights, idx):
        lag = k - (n-1)
        m_abs = abs(lag)
        if m_abs == 0:
            grad += w * (1 - 2*h)
        elif m_abs < n:
            if lag > 0:
                grad[:n-m_abs] += w * (1 - h[m_abs:])
                grad[m_abs:] += w * (-h[:n-m_abs])
            else:
                grad[m_abs:] += w * (1 - h[:n-m_abs])
                grad[:n-m_abs] += w * (-h[m_abs:])
    
    grad *= dx  # scale by dx since C5 = max(c)*dx
    
    # Take step
    h_new = h - step * grad
    h_new = project_box_sum(h_new, target_sum)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if lag > 0:
        g[:n-m] = 1.0 - h[m:]
        g[m:] = -h[:n-m]
    else:
        g[m:] = 1.0 - h[:n-m]
        g[:n-m] = -h[m:]
    return g

def smooth_max_grad(h, n, alpha, top_m=20):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    c_top = c[top_idx]
    m = c_top.max()
    shifted = alpha * (c_top - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        lse = mx + np.log(w.sum())
    else:
        w = np.exp(shifted)
        lse = np.log(w.sum())
    val = m + lse / alpha
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        lag = ki - (n - 1)
        grad += wi * lag_gradient(h, n, lag)
    grad *= dx
    return val, grad

def gd_optimize(h, n, deadline, alphas=[100, 500, 2000, 10000, 50000]):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    top_m = 20
    for alpha in alphas:
        if time.time() > deadline:
            break
        step = 0.1
        for it in range(200):
            if time.time() > deadline:
                break
            val, grad = smooth_max_grad(h, n, alpha, top_m)
            # Backtracking line search
            c_curr = compute_c5(h, n)
            accepted = False
            for bs in range(20):
                h_new = project_box_sum(h - step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new <= c_curr - 1e-14 or c_new < best_c:
                    h = h_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                step = max(step * 2, 0.01)  # increase if stuck
            if step < 1e-10:
                break
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 1e-5]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            n_pairs = n - m
            for i in range(n_pairs):
                if time.time() > deadline:
                    return h, best_c
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-10:

                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-10:

                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000):
    target = n / 2.0
    def obj(x):
        return smooth_max_grad(x, n, alpha, top_m=50)[0]
    def obj_sum(x):
        return x.sum() - target
    def obj_sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj_sum, 'jac': obj_sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_box_sum(res.x, n)
    except:
        return h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if lag > 0:
        g[:n - m] = 1.0 - h[m:]
        g[m:] = -h[:n - m]
    else:
        g[m:] = 1.0 - h[:n - m]
        g[:n - m] = -h[m:]
    return g

def smooth_max_grad(h, n, alpha, top_m=20):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    c_top = c[top_idx]
    mx_shift = alpha * (c_top - c_top.max())
    pk = mx_shift.max()
    if pk > 500:
        w = np.exp(mx_shift - pk)
        lse = pk + np.log(w.sum())
    else:
        w = np.exp(mx_shift)
        lse = np.log(w.sum())
    val = c_top.max() + lse / alpha
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        grad += wi * lag_gradient(h, n, ki - (n - 1))
    return val, grad * dx

def gd_optimize(h, n, deadline, alphas=(200, 1000, 5000, 20000, 100000, 200000)):

    best_h = h.copy()
    best_c = compute_c5(h, n)

        if time.time() > deadline:

        step = 0.05
        stall = 0

            if time.time() > deadline:

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:

            for _ in range(30):
                h_new = project_box_sum(h - step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:

                    best_c = c_new
                    best_h = h.copy()
                    accepted = True
                    break
                if c_new < compute_c5(h, n) + 1e-14:
                    h = h_new
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                step = max(step * 1.5, 0.01)
                stall += 1
                if stall > 20:
                    break
            else:
                stall = 0
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter=3000):
    target = n / 2.0
    def obj(x):
        return smooth_max_grad(x, n, alpha, top_m=30)[0]
    def eq_fun(x):
        return x.sum() - target
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_box_sum(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 1e-5, 5e-6]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                return h, best_c
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            for i in range(n - m):
                if time.time() > deadline:
                    return h, best_c
                j = i + m
                if h[i] > h[j] + 1e-12:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta > 1e-10:
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:

                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                elif h[j] > h[i] + 1e-12:
```
