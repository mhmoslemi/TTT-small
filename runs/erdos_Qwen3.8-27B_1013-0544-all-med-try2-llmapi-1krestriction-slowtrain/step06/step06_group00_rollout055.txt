    cool = 0.99998
    scale = 5000.0

        if time.time() > deadline:
            break

        bi = rng.randint(len(ones_arr))

        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0

            if cur_c < best_c:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_lse_value_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    S = exps.sum()
    lse = m + (sm + np.log(S)) / alpha
    w = exps / S
    A = np.zeros(n)
    B = np.zeros(n)
    for lag in range(-(n - 1), n):
        wval = w[n - 1 + lag]
        if wval < 1e-15:
            continue
        lo_a = max(0, -lag)
        hi_a = min(n, n - lag)
        if hi_a > lo_a:
            A[lo_a:hi_a] += wval * h[lo_a + lag:hi_a + lag]
        lo_b = max(0, lag)
        hi_b = min(n, n + lag)
        if hi_b > lo_b:
            B[lo_b:hi_b] += wval * h[lo_b - lag:hi_b - lag]
    grad = dx * alpha * (1.0 - A - B)
    return lse, grad

        gp = 2.0 * pen_coeff * (s - target)
        return val + pen, grad + gp

        if time.time() > deadline:
            return h
        res = minimize(func, h.copy(), jac=True, method='L-BFGS-B',
