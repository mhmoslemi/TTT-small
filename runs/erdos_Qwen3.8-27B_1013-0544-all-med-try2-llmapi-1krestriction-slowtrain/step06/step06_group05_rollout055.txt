def subgradient(h, n, w=None):
    """Compute subgradient of max_k c(k) w.r.t. h"""
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    if w is None:
        # Use argmax
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
    else:
        w = np.exp(alpha * (c - c.max()))
        w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(2*n-1):
        if w[i] == 0:
            continue
        # dc_i/dh[k] = (1-h[k+i-n+1]) * 1[valid] - h[k-i+n-1] * 1[valid]
        # First term: k+i-n+1 in [0,n-1] => n-1-i ≤ k ≤ n-1-i+n ... wait
        # k+i-n+1 >= 0 => k >= n-1-i
        # k+i-n+1 <= n-1 => k <= 2n-2-i
        # So k is in [max(0, n-1-i), min(n-1, 2n-2-i)]
        k_lo = max(0, n-1-i)
        k_hi = min(n-1, 2*n-2-i)
        if k_lo <= k_hi:
            # (1-h[k+i-n+1]) for k in [k_lo, k_hi]
            idx = k_lo + i - n + 1  # k+i-n+1 when k=k_lo
            # idx ranges from 0 to (k_hi-k_lo)
            length = k_hi - k_lo + 1
            grad[k_lo:k_hi+1] += w[i] * (1 - h[idx:idx+length])
        
        # Second term: k-i+n-1 in [0,n-1] => i-n+1 ≤ k ≤ n-1+i
        k_lo2 = max(0, i - n + 1)
        k_hi2 = min(n-1, n-1 + i)
        if k_lo2 <= k_hi2:
            idx2 = k_lo2 - i + n - 1  # k-i+n-1 when k=k_lo2
            length2 = k_hi2 - k_lo2 + 1
            grad[k_lo2:k_hi2+1] -= w[i] * h[idx2:idx2+length2]

            idx = np.arange(k_lo + i - n + 1, k_hi + i - n + 2)
            grad[k_lo:k_hi + 1] += wi * (1.0 - h[idx])
        k_lo2 = max(0, i - n + 1)
        if k_lo2 <= n - 1:
            idx2 = np.arange(k_lo2 - i + n - 1, 2 * n - i - 1)
            grad[k_lo2:n] -= wi * h[idx2]
    grad *= dx
    return grad

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= lr_ratio
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
