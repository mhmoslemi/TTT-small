        if abs(diff) < 1e-14:

            if tot < 1e-14:

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def _sm_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    if zm > 500:
        val = m + (zm + np.log(np.sum(np.exp(z - zm)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(z))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _sm_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()

        grad[L:n] += w[shift] * (-x[0:nv])
