def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0, 1)
    return h

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

    # Phase 2: Fine-tune with smaller alpha
    if time.time() < deadline - 350:
        t2 = min(time.time() + 100, deadline - 350)
        bounds = [(0.0, 1.0)] * n_init
        for alpha in [5000, 20000, 100000, 500000, 2000000]:
            if time.time() > t2:
                break
            try:
                res = minimize(_obj, best_h, jac=_grad, args=(n_init, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 100, 'ftol': 1e-17,
                                       'gtol': 1e-15})
                h_new = project_feasible(res.x, n_init)
                c_new = compute_c5(h_new, n_init)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 3: Fast pair search (gradient-guided)
    if time.time() < deadline - 200:
        t3 = min(time.time() + 120, deadline - 200)
        best_h, best_c = fast_pair_search(best_h, n_init, t3,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    # Phase 4: Naive pair search with very small steps
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 120)
        best_h, best_c = naive_pair_search(best_h, n_init, t4,
                                           [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                           radius=40)

    # Phase 5: Perturbation search to escape local optima
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        h_p, c_p = perturbation_search(best_h, n_init, t5, rng)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
            # Re-optimize after perturbation
            if time.time() < deadline - 60:
                best_h, best_c = lbfgs_opt(best_h, n_init, deadline - 60, max_iters=100)
                if time.time() < deadline - 30:
                    best_h, best_c = naive_pair_search(best_h, n_init, deadline - 30,
                                                       [0.0005, 0.0002, 0.0001], radius=30)

    # Phase 6: Try other n values
    if time.time() < deadline - 60:
        for n_try in [200, 250, 350, 400]:
            if time.time() > deadline - 40:
                break
            if n_try == n_init:
                continue
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(40, deadline - time.time() - 30)
            if alloc < 20:
                break
            h_t, c_t = lbfgs_opt(h_interp, n_try, time.time() + alloc * 0.5, max_iters=100)
            if time.time() < deadline - 20:
                h_t, c_t = naive_pair_search(h_t, n_try, deadline - 20,
                                             [0.001, 0.0005, 0.0002], radius=30)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    # Phase 7: Final polishing
    nf = len(best_h)
    if time.time() < deadline - 8:
        best_h, best_c = lbfgs_opt(best_h, nf, deadline - 8, max_iters=100)
    if time.time() < deadline - 3:
        best_h, best_c = naive_pair_search(best_h, nf, deadline - 2,
                                           [0.0002, 0.0001, 0.00005, 0.00002], radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
