    exp_shifted = np.exp(shifted - mx)
    weights = exp_shifted / exp_shifted.sum()
    
    f = cmax + (mx + np.log(exp_shifted.sum())) / alpha + penalty * sp * sp
    
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w > 1e-10:
            offset = idx - (n - 1)
            g += w * grad_corr_offset(x, offset)
    g *= dx
    g += 2.0 * penalty * sp
    return f, g

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [10, 30, 100, 300, 1000, 3000, 10000]
    h_curr = h.copy()
    penalty = 100.0
    for a in alphas:
        if time.time() > deadline - 5:
            break
        h_curr = project_feasible(h_curr, n)
        res = minimize(obj_grad, h_curr, args=(n, a, penalty),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-14,
                                'maxfun': 2000})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) < compute_c5(h_curr, n):
            h_curr = h_new
        if a >= 3000:
            penalty *= 10
    return h_curr

def gradient_local_search(h, n, deadline, eps_list=None):
    if eps_list is None:
        eps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    for eps in eps_list:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            kmax = int(np.argmax(c))
            offset = kmax - (n - 1)
            g = grad_corr_offset(h, offset)
            order = np.argsort(g)
            n_tries = 0
            for ii in range(min(30, n)):
                i = int(order[ii])
                if g[i] >= 0:
                    break
                for jj in range(min(30, n)):
                    j = int(order[n - 1 - jj])
                    if g[j] <= 0:
                        break
                    if i == j:
                        continue
                    if h[i] + eps > 1.0 or h[j] - eps < 0.0:
                        continue
                    h[i] += eps
                    h[j] -= eps
                    n_tries += 1
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        order = np.argsort(grad_corr_offset(h, offset))
                    else:
                        h[i] -= eps
                        h[j] += eps
                    if n_tries > 500:
                        break
                if n_tries > 500:
                    break
    return h, best_c

def pair_search_fast(h, n, deadline, eps_list):
    h = h.copy()
    best_c = compute_c5(h, n)
    for eps in eps_list:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + eps > 1.0:
                    continue
                for j in range(n):
                    if i == j or h[j] < eps:
                        continue
                    h[i] += eps
                    h[j] -= eps
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= eps
                        h[j] += eps
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B on initial resolution
    alloc1 = min(120, (deadline - t0) * 0.2)
    if time.time() < deadline - 30:
        h_opt = lbfgs_optimize(best_h, n_init, t0 + alloc1)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Gradient-informed local search
    if time.time() < deadline - 30:
        alloc2 = min(120, (deadline - t0) * 0.15)
        best_h, best_c = gradient_local_search(best_h, n_init, t0 + alloc2)

    # Phase 3: Try different resolutions
    x_init = np.linspace(0, 2, n_init)
    for n_try in [100, 150, 200, 300, 400]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(60, (deadline - time.time()) * 0.1)
        if alloc < 20:
            break
        h_opt = lbfgs_optimize(h_interp, n_try, time.time() + alloc,
                               alphas=[50, 200, 1000, 5000])
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            if time.time() < deadline - 30:
                best_h, best_c = gradient_local_search(best_h, n_try,
                                                       time.time() + min(60, (deadline - time.time()) * 0.1))

    # Phase 4: Multi-restart with structured perturbations
    n_best = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = max(3, int((deadline - time.time() - 30) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            pert_type = r % 3
            if pert_type == 0:
                hp = best_h + rng.normal(0, 0.02 + 0.03 * r / max(1, n_restarts), n_best)
            elif pert_type == 1:
                shift = rng.randint(1, max(2, n_best // 20))
                hp = np.roll(best_h, shift)
            else:
                freq = rng.randint(2, 20)
                phase = rng.uniform(0, 2 * np.pi)
                wave = 0.03 * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                hp = best_h + wave
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt = lbfgs_optimize(hp, n_best, time.time() + alloc,
                                   alphas=[100, 500, 2000, 10000])
            c_opt = compute_c5(h_opt, n_best)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Extended local search on best
    if time.time() < deadline - 10:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search_fast(best_h, n, deadline - 3,
                                              [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            best_h, best_c = gradient_local_search(best_h, n, deadline - 3,
                                                   [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    # Phase 6: Final L-BFGS-B polish
    if time.time() < deadline - 3:
        n = len(best_h)
        h_opt = lbfgs_optimize(best_h, n, deadline - 2, alphas=[5000, 20000, 100000])
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
