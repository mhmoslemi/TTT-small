def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            sr = room.sum()
            if sr < 1e-14:
                break
            h += diff * room / sr
        else:
            if s < 1e-14:
                break
            h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, L):
    grad = np.zeros(n)
    if L >= 0:
        if L < n:
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
    else:
        Lp = -L
        if Lp < n:
            grad[:n - Lp] += (1.0 - h[Lp:])
            grad[Lp:] -= h[:n - Lp]
    return grad

def compute_soft_grad(h, n, alpha):
    g = compute_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-30:
        ks = int(np.argmax(g))
        w = np.zeros(len(g))
        w[ks] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for ki in nz:
        L = ki - (n - 1)
        grad += w[ki] * lag_gradient(h, n, L)
    return grad, g, m

            break

        zeros.add(p)
