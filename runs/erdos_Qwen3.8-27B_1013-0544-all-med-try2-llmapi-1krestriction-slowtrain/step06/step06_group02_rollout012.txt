import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def c5_gradient(h, n, alpha):
    """Gradient of smooth max (log-sum-exp) of C5 values."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    # Phase 6: Random perturbation restart (escape local min)
    if time.time() < deadline - 80:
        nf = len(best_h)
        for _ in range(3):
            if time.time() > deadline - 50:
                break
            h_pert = best_h + rng.randn(nf) * 0.02
            h_pert = project_feasible(h_pert, nf)
            t_r = min(time.time() + 25, deadline - 30)
            h_r, c_r = lbfgs_optimize(h_pert, nf, t_r,
                                      configs=[(10000, 100.0), (100000, 1000.0),
                                               (1000000, 5000.0)])
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 7: Final aggressive local refinement
    if time.time() < deadline - 5:
        nf = len(best_h)
        h_l, c_l = local_pair(best_h, nf, deadline - 2)
        if c_l < best_c:
            best_c = c_l
            best_h = h_l

    # Final L-BFGS polish
    if time.time() < deadline - 2:
        nf = len(best_h)
        h_f, c_f = lbfgs_optimize(best_h, nf, deadline - 1,
                                  configs=[(5000000, 10000.0)])
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
