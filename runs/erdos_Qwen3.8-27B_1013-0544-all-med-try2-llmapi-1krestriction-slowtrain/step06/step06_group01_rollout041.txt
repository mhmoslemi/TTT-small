```python
def vectorized_binary_search(h, n, deadline, n_iter=1000):
    dx = 2.0 / n
    target = n // 2
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        a, b = len(zeros), len(ones)
        
        # Precompute delta for each zero position
        # DI[i][p+n-1] = delta from changing zeros[i] from 0 to 1
        DI = np.zeros((a, 2*n-1))
        for idx, i in enumerate(zeros):
            # i+p valid: p in [-i, n-1-i]
            p_range = np.arange(-i, n-i)
            DI[idx, p_range + n - 1] += 1 - h[i + p_range]
            # i-p valid: p in [i-(n-1), i]
            p_range2 = np.arange(-(n-1-i), i+1)
            DI[idx, p_range2 + n - 1] -= h[i - p_range2]
        
        # Precompute delta for each one position
        DJ = np.zeros((b, 2*n-1))
        for idx, j in enumerate(ones):
            p_range = np.arange(-j, n-j)
            DJ[idx, p_range + n - 1] -= 1 - h[j + p_range]
            p_range2 = np.arange(-(n-1-j), j+1)
            DJ[idx, p_range2 + n - 1] += h[j - p_range2]
        
        # Compute max correlation for all swaps
        # new_vals[i,j,k] = corr[k] + DI[i,k] + DJ[j,k]
        # max over k
        max_vals = (corr[None, None, :] + DI[:, None, :] + DJ[None, :, :]).max(axis=2)
        
        # Find best swap
        ii, jj = np.unravel_index(max_vals.argmin(), max_vals.shape)
        
        if max_vals[ii, jj] * dx < cur_c - 1e-15:
            h[zeros[ii]] = 1
            h[ones[jj]] = 0
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # No improving swap found
            break
    
    return best_h, best_c
```

def vectorized_binary_search(h_init, n, deadline, max_iter=5000):
    dx = 2.0 / n
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        a, b = len(zeros), len(ones)
        if a == 0 or b == 0:
            break
        
        # Precompute DI: delta from each zero going to 1
        # DI[i_idx, k] = change in corr[k] when zeros[i_idx] changes 0->1
        # Change: for lag s=k-(n-1):
        #   + (1-h[i+s]) if i+s valid (first factor)
        #   - h[i-s] if i-s valid (second factor, because h[i] was 0, now 1, 
        #     so the term h[i-s]*(1-h[i]) changes from h[i-s]*1 to h[i-s]*0)
        
        # Let me compute this directly
        # For zero position i:
        # delta[k] for k = s + (n-1):
        #   + (1-h[i+s]) if 0 <= i+s < n
        #   - h[i-s] if 0 <= i-s < n
        
        DI = np.zeros((a, 2*n-1))
        for idx_i, i in enumerate(zeros):
            s_vals = np.arange(-(n-1), n)
            # First part: i+s valid
            mask1 = (i + s_vals >= 0) & (i + s_vals < n)
            DI[idx_i, mask1] += (1 - h[i + s_vals[mask1]])
            # Second part: i-s valid
            mask2 = (i - s_vals >= 0) & (i - s_vals < n)
            DI[idx_i, mask2] -= h[i - s_vals[mask2]]
        
        DJ = np.zeros((b, 2*n-1))
        for idx_j, j in enumerate(ones):
            s_vals = np.arange(-(n-1), n)
            # For one position j going to 0:
            # delta[k] = -DI_for_j (because it's the reverse change)
            # Previously h[j]=1, now h[j]=0
            # First factor: was h[j]*(1-h[j+s]) = 1*(1-h[j+s]), now 0
            #   So change: -(1-h[j+s]) if j+s valid
            # Second factor: was h[j-s]*(1-h[j]) = h[j-s]*0 = 0, now h[j-s]*1 = h[j-s]
            #   So change: +h[j-s] if j-s valid
            mask1 = (j + s_vals >= 0) & (j + s_vals < n)
            DJ[idx_j, mask1] -= (1 - h[j + s_vals[mask1]])
            mask2 = (j - s_vals >= 0) & (j - s_vals < n)
            DJ[idx_j, mask2] += h[j - s_vals[mask2]]
        
        # For swap (i, j): new_corr = corr + DI[i] + DJ[j]
        # We want: max(corr + DI[i] + DJ[j]) minimized over all (i,j)
        
        # Vectorized: max over k of (corr[k] + DI[i,k] + DJ[j,k])
        # = max over k of (corr[k] + DI[i,k] + DJ[j,k])
        
        # This is a 3D array: (a, b, 2n-1)
        # For memory efficiency, process in batches
        batch = 200
        best_val = cur_c / dx + 1  # in units of raw correlation
        best_ii, best_jj = -1, -1
        
        for i_start in range(0, a, batch):
            i_end = min(i_start + batch, a)
            # shape: (batch, b, 2n-1)
            vals = corr[None, None, :] + DI[i_start:i_end, None, :] + DJ[None, :, :]
            max_vals = vals.max(axis=2)  # shape (batch, b)
            local_min = max_vals.min()
            if local_min < best_val:
                li, lj = np.unravel_index(max_vals.argmin(), max_vals.shape)
                best_val = local_min
                best_ii = i_start + li
                best_jj = lj
        
        if best_ii >= 0 and best_val < cur_c / dx - 1e-12:
            h[zeros[best_ii]] = 1
            h[ones[best_jj]] = 0
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break  # No improving swap
    
    return best_h, best_c

    target = n / 2.0

def slsqp_opt(h, n, alpha, maxiter, deadline):
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = c.max()
        if alpha < 1e8:
            s = alpha * (c - m)
            sm = s.max()
            return m + (sm + np.log(np.exp(s-sm).sum())) / alpha
        return m
    def grad(x):
        c = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = c.max()
        if alpha < 1e8:
            s = alpha * (c - m)
            sm = s.max()
            w = np.exp(s - sm)
            w /= w.sum()
        else:
            idx = np.where(np.isclose(c, m, atol=1e-12))[0]
            w = np.zeros(2*n-1)
            w[idx] = 1.0 / len(idx)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-16: continue
            s = k - (n-1)
            wk = w[k] * (2.0/n)
            if s >= 0:

                t2[s:] += wk * x[:n-s]
            else:

                t1[t:] += wk * (1-x[:n-t])
                t2[:n-t] += wk * x[t:]

                       options={'maxiter':maxiter, 'ftol':1e-16})

    except Exception:
        return h
