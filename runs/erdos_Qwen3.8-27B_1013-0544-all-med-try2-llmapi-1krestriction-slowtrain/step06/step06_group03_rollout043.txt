```python
def smooth_max_topk(c, alpha, top_k):
    k = min(top_k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    ck = c[idx]
    m = ck.max()
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
```

```python
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    # Add penalty for constraint violation
    viol = x.sum() - n / 2.0
    penalty = 50.0 * viol * viol
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + penalty)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha + penalty)
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

    return x.sum() - n / 2.0

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        ht = slsqp_opt(best_h.copy(), n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 3:
            if n <= 300:
                best_h, best_c = pair_search(best_h, n, deadline - 2,
                                             [0.0001, 0.00005, 0.00002, 0.00001])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 2,
                                                  [0.0001, 0.00005, 0.00002], radius=120)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
