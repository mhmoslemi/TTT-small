```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_of_c_at_s(h, n, s):
    """Gradient of c(s) = sum_i h[i]*(1-h[i+s])*dx with respect to h.
    s = m - (n-1) where m is the index in the 'full' correlation array."""
    g = np.zeros(n)
    dx = 2.0 / n
    # c(s) = sum_i h[i] * (1-h[i+s]) * dx for valid i
    # d/dh[i] of the term at position i: (1-h[i+s])*dx (if i+s valid)
    # d/dh[j] of the term at position j-s: h[j-s]*(-dx) = -h[j-s]*dx (if j-s valid)
    
    # First term: for each valid i (0<=i<n, 0<=i+s<n), derivative wrt h[i] is (1-h[i+s])*dx
    # Second term: for each valid i (0<=i<n, 0<=i+s<n), derivative wrt h[i+s] is -h[i]*dx
    
    if s >= 0:
        # i ranges from 0 to n-1-s
        for i in range(n - s):
            g[i] += (1 - h[i+s]) * dx
            g[i+s] -= h[i] * dx
    else:
        # s < 0, let s' = -s > 0
        # i+s valid means i >= s' and i < n
        # i ranges from -s to n-1
        for i in range(-s, n):
            g[i] += (1 - h[i+s]) * dx
            g[i+s] -= h[i] * dx
    
    return g
```

```python
    best_h = h.copy()

                    best_c = c_try
                    best_h = h.copy()
                break

        else:
            # No improvement, try different step
```

5. Check if this reduces the overall max (not just c(m*)).
6. If yes, accept and repeat. If no, try next best pair.

This is O(n) per iteration (for the correlate + gradient + argmax/argmin), which is very fast!

But we might get stuck because reducing c(m*) might increase other c(m'). To handle this, we can:
- After a few iterations on m*, re-check all c(m) and switch to the new max
- Or use a smoothed objective

Let me implement this "greedy gradient" approach:

```python
def greedy_gradient_descent(h, n, deadline, d=0.01):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1 - h, mode='full') * dx
        m_val = c.max()
        
        if m_val < best_c - 1e-12:
            best_c = m_val
            best_h = h.copy()
            continue
        
        m_star = int(np.argmax(c))
        s = m_star - (n - 1)
        
        # Gradient of c(m_star)
        g = gradient_of_c_at_s(h, n, s)
        
        # Best move: decrease h where g is positive, increase where g is negative
        # Find i with max g (most to decrease) and j with min g (most to increase)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        
        if g[i] <= -g[j] + 1e-12:  # No clear improvement direction
            break
        
        # Try the move
        if h[i] - d >= 0 and h[j] + d <= 1:
            h[i] -= d
            h[j] += d
            new_c = compute_c5(h, n)
            if new_c < m_val - 1e-12:
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # Revert
                h[i] += d
                h[j] -= d
                # Try smaller step
                d_new = d * 0.5
                if d_new < 1e-6:
                    break
                d = d_new
                if h[i] - d >= 0 and h[j] + d <= 1:
                    h[i] -= d
                    h[j] += d
                    new_c = compute_c5(h, n)
                    if new_c < m_val - 1e-12:
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                    else:
                        h[i] += d
                        h[j] -= d
                        break
        else:
            # Can't make the move, try next best
            # Remove i and j from consideration
            g[i] = -np.inf
            g[j] = np.inf
            i2 = int(np.argmax(g))
            j2 = int(np.argmin(g))
            if g[i2] <= -g[j2] + 1e-12 or h[i2] - d < 0 or h[j2] + d > 1:
                break
            h[i2] -= d
            h[j2] += d
            new_c = compute_c5(h, n)
            if new_c < m_val - 1e-12:
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i2] += d
                h[j2] -= d
                break
    
    return best_h, best_c
```

```python
def compute_c5(h, n):

def project_feasible(h, n):

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_s(h, n, s):
    """Gradient of c(s) where s = m-(n-1) for 'full' correlate index m."""
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        idx = np.arange(n - s)
        g[idx] += (1 - h[idx + s]) * dx
        g[idx + s] -= h[idx] * dx
    else:
        idx = np.arange(-s, n)
        g[idx] += (1 - h[idx + s]) * dx
        g[idx + s] -= h[idx] * dx
    return g

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

    return np.ones(n)
```
