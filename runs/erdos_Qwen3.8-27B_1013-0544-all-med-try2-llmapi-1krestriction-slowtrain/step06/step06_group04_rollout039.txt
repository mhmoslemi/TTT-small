            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def subgradient_descent(h, n, deadline, max_iter=50000):

    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h -= (-diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, top_k=6, max_iter=300000):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        mc = c.max()
        if mc < best_c - 1e-13:
            best_c = mc
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15000:
                break
        k = int(np.argmax(c))
        lag = k - (n - 1)
        if lag == 0:
            break
        ma = abs(lag)
        if lag > 0:
            diff = h[:n - ma] - h[ma:]
        else:
            diff = h[ma:] - h[:n - ma]
        if diff.max() < 1e-9:
            break
        top_idx = np.argsort(diff)[-top_k:][::-1]
        best_mv = mc
        best_mh = None
        for i in top_idx:
            if diff[i] < 1e-9:
                continue
            for frac in [0.5, 0.25, 0.1, 0.03]:
                d = diff[i] * frac
                hs = h.copy()
                if lag > 0:
                    hs[i] -= d
                    hs[i + ma] += d

                    hs[i + ma] -= d
                    hs[i] += d
                hs = project_feasible(hs, n)

        if best_mh is None:

def gradient_flow(h, n, deadline, max_iter=50000):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    stall = 0
