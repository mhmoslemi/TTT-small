    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
    return h, best_c

def adaptive_pair_search(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        grad = active_grad(h, n, 0.004)
        improved_this_round = False
        for _ in range(50):
            if time.time() > deadline:
                return h, best_c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            direction = 1 if grad[i] < grad[j] else -1
            for d in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                di = direction * d
                if h[i] + di < -1e-12 or h[i] + di > 1 + 1e-12:
                    continue
                if h[j] - di < -1e-12 or h[j] - di > 1 + 1e-12:
                    continue
                h[i] += di; h[j] -= di
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved_this_round = True
                    break
                h[i] -= di; h[j] += di
        if not improved_this_round:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 250
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP with increasing alpha (120s)
    t1 = t0 + 120
    for alpha in [5000, 20000, 80000, 200000]:
        if time.time() > t1:
            break
        ht = slsqp_run(h, n, alpha, 3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht
            h = ht

    # Phase 2: Iterative subgradient kick + SLSQP reconverge (450s)
    t2 = t0 + 570
    iterations = 0
    while time.time() < t2 - 30:
        iterations += 1
        kick_dur = min(25, (t2 - time.time()) * 0.3)
        if kick_dur < 5:
            break
        h_kick, c_kick = subgrad_kick(h, n, kick_dur, rng)
        if c_kick < best_c:
            best_c = c_kick
            best_h = h_kick.copy()
            h = h_kick
        # SLSQP reconverge from kicked position
        slsqp_budget = min(60, (t2 - time.time()) * 0.5)
        maxit = max(1000, int(slsqp_budget * 40))
        ht = slsqp_run(h, n, 100000, maxit)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()
            h = ht
        # Alternate: also try from best_h with noise
        if iterations % 3 == 0:
            h_noisy = best_h + 0.01 * rng.randn(n)
            h_noisy = project_feasible(h_noisy, n)
            ht2 = slsqp_run(h_noisy, n, 50000, 2000)
            c2 = compute_c5(ht2, n)
            if c2 < best_c - 1e-15:
                best_c = c2
                best_h = ht2.copy()
                h = ht2

    # Phase 3: Adaptive pair search (150s)
    t3 = t0 + 720
    h_opt, c_opt = adaptive_pair_search(best_h, n, min(t3, deadline - 60), rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 4: Fine pair search (180s)
    t4 = t0 + 900
    h_opt, c_opt = pair_search(best_h, n, min(t4, deadline - 30),
                                [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 15:
        ht = slsqp_run(best_h, n, 500000, 5000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Tiny pair search
    if time.time() < deadline - 3:
        h_opt, c_opt = pair_search(best_h, n, deadline - 1,
                                    [0.0001, 0.00005, 0.00002, 0.00001])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
