def compute_delta(h, i, j, n):
    """Compute delta in correlation when h[i]: 1->0 and h[j]: 0->1."""
    # lag ranges from -(n-1) to n-1, indexed as 0 to 2n-2
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each lag, compute the change
    # Terms involving position i as first factor: valid if 0 <= i+lag < n
    valid_i1 = (i + lags >= 0) & (i + lags < n)
    # Terms involving position i as second factor: valid if 0 <= i-lag < n
    valid_i2 = (i - lags >= 0) & (i - lags < n)
    # Terms involving position j as first factor: valid if 0 <= j+lag < n
    valid_j1 = (j + lags >= 0) & (j + lags < n)
    # Terms involving position j as second factor: valid if 0 <= j-lag < n
    valid_j2 = (j - lags >= 0) & (j - lags < n)

    delta[valid_i2] += h[i - lags[valid_i2]]

    delta[valid_j1] += (1.0 - h[j + lags[valid_j1]])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -50, 50)
    se = np.exp(shifted).sum()
    return m + np.log(se) / alpha + lam * (x.sum() - n/2.0)**2

def smooth_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -50, 50)
    es = np.exp(shifted)
    w = es / es.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        lag = k - (n-1)
        wk = w[k] * dx
        lo = max(0, -lag)
        hi = min(n-1, n-1-lag)
        if lo <= hi:
            js = np.arange(lo, hi+1)
            grad[js] += wk * (1.0 - x[js+lag])
        lo = max(0, lag)
        hi = min(n-1, n-1+lag)
        if lo <= hi:
            js = np.arange(lo, hi+1)
            grad[js] += wk * (-x[js-lag])
    grad += 2.0 * lam * (x.sum() - n/2.0)
    return grad

    best_c = compute_c5(h, n)

                smooth_obj, h, jac=smooth_grad,

    return best_h, best_c

    target = n // 2
    h = np.zeros(n, dtype=float)

    cur_c = corr.max() * dx

    best_h = h.copy()

    zeros = np.where(h == 0)[0].copy()

    T1 = 0.001

    step = 0

        total_est = 60  # estimate total SA time in seconds
        frac = min(elapsed / total_est, 1.0)
        T = T0 * (T1/T0) ** frac

        j = zeros[rng.randint(len(zeros))]

        h[j] = 1.0
        new_corr = np.correlate(h, 1-h, mode='full')

            corr = new_corr
            cur_c = new_c
            ones[i] = j

        else:

            h[j] = 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()

    se = np.exp(shifted).sum()
    return m + np.log(se) / alpha + lam * (x.sum() - n/2.0)**2
