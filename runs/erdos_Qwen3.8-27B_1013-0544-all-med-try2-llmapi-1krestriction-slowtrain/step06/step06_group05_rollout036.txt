def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute to non-saturated elements
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    log_sum = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + log_sum / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def slsqp_sum(x, n):
    return x.sum() - n / 2.0

def slsqp_sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':slsqp_sum,'jac':slsqp_sum_jac,'args':(n,)}],
                       options={'maxiter':maxiter,'ftol':1e-18})
        return project_feasible(res.x, n)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def _slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=4000):
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def full_pair_search(h, n, deadline, deltas, max_passes=500):
    h = h.copy()
    g = 1.0 - h
    dx = 2.0 / n
    c = np.correlate(h, g, mode='full').astype(np.float64)
    best_c = float(c.max())
    best_h = h.copy()
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                return best_h, best_c * dx
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c * dx
                hi_p = h[i] + d
                hi_m = h[i] - d
                skip_p = hi_p > 1.0
                skip_m = hi_m < 0.0
                if skip_p and skip_m:
                    continue
                for j in range(i + 1, n):
                    if not skip_p and h[j] >= d:
                        h[i] = hi_p; h[j] -= d
                        g[i] -= d; g[j] += d
                        c = np.correlate(h, g, mode='full')
                        m = float(c.max())
                        if m < best_c - 1e-16:
                            best_c = m
                            best_h = h.copy()
                            g = 1.0 - h
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                            g = 1.0 - h
                    if not skip_m and h[j] <= 1.0 - d:
                        h[i] = hi_m; h[j] += d
                        g[i] += d; g[j] -= d
                        c = np.correlate(h, g, mode='full')
                        m = float(c.max())
                        if m < best_c - 1e-16:
                            best_c = m
                            best_h = h.copy()
                            g = 1.0 - h
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
                            g = 1.0 - h
            if not improved:
                break
    return best_h, best_c * dx

                    if hi_p <= 1.0 and h[j] >= d:

                        c = np.correlate(h, g, mode='full')

def perturb_and_refine(h, n, deadline, rng, best_c_current):
    h_copy = h.copy()
    c_current = best_c_current
    scales = [0.01, 0.02, 0.05, 0.1]
    for scale in scales:
        if time.time() > deadline:
            break
        hp = h_copy + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        c = compute_c5(hp)
        if c < c_current:
            h_copy = hp
            c_current = c
            full_pair_search(h_copy, n, min(time.time() + 30, deadline), [0.005, 0.001])
    return h_copy, c_current
