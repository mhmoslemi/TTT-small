        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur = float(np.max(c) * dx)
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        if max_change < 1e-10:
            break
    return best_h, best_c

def _obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sfun(x, n):
    return float(x.sum() - n / 2.0)

def _sjac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha=50000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sfun, 'jac': _sjac, 'args': (n,)}]
    try:
        res = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_local(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    center = n - 1
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = np.correlate(h, 1.0 - h, mode='full')
            ks = int(np.argmax(c))
            lag = ks - center
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > 0.005:
                    delta = min(d, h[i] * 0.4)
                    h[i] -= delta
                    h[j] = min(1.0, h[j] + delta)
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-13:
                        best_c = cn
                        improved = True
                    else:
                        h[i] += delta
                        h[j] = max(0.0, h[j] - delta)
                if h[j] < 0.995:
                    delta = min(d, (1.0 - h[j]) * 0.4)
                    h[j] += delta
                    h[i] = max(0.0, h[i] - delta)
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-13:
                        best_c = cn
                        improved = True
                    else:
                        h[j] -= delta
                        h[i] = min(1.0, h[i] + delta)
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 20:
        h2, c2 = cd_optimize(h, n, deadline - 20, max_rounds=300)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 10:
        h3 = slsqp_optimize(best_h, n, alpha=50000, maxiter=2000)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3
    if time.time() < deadline - 3:
        h4, c4 = cd_optimize(best_h, n, deadline - 1, max_rounds=100)
        if c4 < best_c:
            best_c, best_h = c4, h4
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    if h_init is not None:
        alloc = time.time() + min(180, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_at_n(h_init, n_init, alloc)
        best_h, best_c = h_opt, c_opt
    n_candidates = [200, 250, 300, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and abs(n_try - len(best_h)) <= 20:
            continue
        rem = deadline - time.time() - 50
        if rem < 50:
            break
        alloc = min(120, rem * 0.25)
        if alloc < 35:
            break
        if best_h is not None:
            xb = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n_try)
            h_start = np.interp(xn, xb, best_h)
        elif h_init is not None:
            xi = np.linspace(0, 2, n_init)
            xn = np.linspace(0, 2, n_try)
            h_start = np.interp(xn, xi, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        rem = deadline - time.time() - 40
        n_restarts = max(1, int(rem / 55))
        for r in range(min(n_restarts, 6)):
            if time.time() > deadline - 35:
                break
            scale = 0.015 * (1.0 + 0.4 * r)
            hp = best_h + np.random.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = cd_optimize(hp, n, time.time() + alloc, max_rounds=150)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h_f, c_f = pair_local(best_h, n, deadline - 8)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
        if time.time() < deadline - 5:
            h_f2, c_f2 = cd_optimize(best_h, n, deadline - 2, max_rounds=200)
            if c_f2 < best_c:
                best_c, best_h = c_f2, h_f2
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
