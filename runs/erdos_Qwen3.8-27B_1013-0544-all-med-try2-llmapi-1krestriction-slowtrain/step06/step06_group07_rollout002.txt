    """Find optimal h[i] that minimizes max correlation, given other h fixed."""
    # Compute correlations with h[i] = 0 and h[i] = 1
    h0 = h.copy()
    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    h1 = h.copy()
    h1[i] = 1.0
    corr1 = np.correlate(h1, 1.0 - h1, mode='full')
    
    # corr(x) = (1-x)*corr0 + x*corr1 for each component
    # So corr(x)[m] = corr0[m] + x*(corr1[m] - corr0[m])
    # We want to minimize max_m corr0[m] + x*(corr1[m] - corr0[m])
    # subject to 0 <= x <= 1

    # At x, the value is corr0[m] + x * diff[m]
    # We want min_x max_m (corr0[m] + x * diff[m])
    
    # This is a 1D convex optimization (max of linear functions)
    # The minimum is where the "upper envelope" changes slope from negative to positive
    
    # Simple approach: ternary search or bisection on the derivative
    # The subgradient at x is {sum of diff[m] for m where corr0[m] + x*diff[m] = max}

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = mask.sum()
        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    return m + np.log(np.sum(np.exp(shifted))) / beta

import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            return h
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_pen(h, dx, beta, target_sum):
    corr = np.correlate(h, 1.0 - h, mode='full')
    pen = 50.0 * (h.sum() - target_sum) ** 2
    return smooth_max(corr, beta) * dx + pen


def coord_step(h, i, n):
    h0 = h.copy()
    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    h1 = h.copy()
    h1[i] = 1.0
    corr1 = np.correlate(h1, 1.0 - h1, mode='full')
    diff = corr1 - corr0

    def f(x):
        return np.max(corr0 + x * diff)

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    h[i] = res.x
    return h


def coord_descent(h, n, target_sum, max_time, rng):

    return np.interp(x_new, x_old, h_src)

        return h
