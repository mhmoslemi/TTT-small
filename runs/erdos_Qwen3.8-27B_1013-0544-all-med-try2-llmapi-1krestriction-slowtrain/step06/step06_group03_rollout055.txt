def simulated_annealing(h, n, t_start, t_end, max_iters, deadline, rng):
    """Simulated annealing with constraint-preserving swap moves."""

    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    
    n_evals = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Every 1000 iters, check time and adjust
        T = t_start * (t_end / t_start) ** (it / max(1, max_iters))
        
        # Propose a move: pick i, j, and delta
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.01, 0.1)
        
        # Try both directions
        if h[i] >= delta and h[j] + delta <= 1.0:
            h[i] -= delta
            h[j] += delta
            new_c = compute_c5(h, n)
            n_evals += 1
            if new_c < curr_c:
                curr_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            elif rng.random() < np.exp(-(new_c - curr_c) / max(T, 1e-10)):
                curr_c = new_c
            else:
                h[i] += delta
                h[j] -= delta
        elif h[j] >= delta and h[i] + delta <= 1.0:
            h[j] -= delta
            h[i] += delta
            new_c = compute_c5(h, n)

        else:

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):

    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)
