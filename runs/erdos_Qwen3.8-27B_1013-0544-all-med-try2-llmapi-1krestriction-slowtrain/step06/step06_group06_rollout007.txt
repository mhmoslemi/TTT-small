import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_gk(h, n, k):
    if k >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if k > 0:
        grad[k:] = 1.0 - h[:n-k]
    else:
        grad[:] = 1.0 - h[:]
    grad[:n-k] -= h[k:]
    return grad

def softmax_obj_and_grad(x, n, alpha):
    h = x
    g = np.correlate(h, 1 - h, mode='full')[:n]  # only first n matter
    dx = 2.0 / n
    gs = g * alpha
    mx = gs.max()
    exps = np.exp(gs - mx)
    sm = exps.sum()
    obj = dx * (mx + np.log(sm)) / alpha
    weights = exps / sm
    grad = np.zeros(n)
    for k in range(n):
        if weights[k] > 1e-8:
            grad += weights[k] * grad_gk(h, n, k)
    grad *= dx
    return obj, grad

    def obj(x):
        return softmax_obj_and_grad(x, n, alpha)[0]
    def grad(x):
        return softmax_obj_and_grad(x, n, alpha)[1]

        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n

                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_gk(h, n, k):
    if k >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if k > 0:
        grad[k:] = 1.0 - h[:n - k]
    else:
        grad[:] = 1.0 - h[:]
    grad[:n - k] -= h[k:]
    return grad

def make_slsqp_funcs(n, alpha):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full')[:n]
        gs = g * alpha
        mx = gs.max()
        exps = np.exp(gs - mx)
        return dx * (mx + np.log(exps.sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full')[:n]
        gs = g * alpha
        mx = gs.max()
        exps = np.exp(gs - mx)
        w = exps / exps.sum()
        grad = np.zeros(n)
        for k in range(n):
            if w[k] > 1e-8:
                grad += w[k] * grad_gk(x, n, k)
        return grad * dx
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    return obj, obj_grad, cons_fun, cons_jac

def binary_local_search(h_init, n, deadline, max_iter=500):
    target_idx = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target_idx]] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full')[:n]
        k_star = int(np.argmax(g))
        g_max = g[k_star]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        best_swap = None
        best_new_max = g_max
        for i in zeros:
            delta_i = np.zeros(n)
            if i - k_star >= 0:
                delta_i[i - k_star] += 1 - h[i - k_star]
            if i + k_star < n:
                delta_i[i + k_star] -= h[i + k_star]
            for j in ones:
                delta_j = np.zeros(n)
                if j - k_star >= 0:
                    delta_j[j - k_star] -= 1 - h[j - k_star]
                if j + k_star < n:
                    delta_j[j + k_star] += h[j + k_star]
                new_g_ks = g_max + delta_i[k_star] + delta_j[k_star]
                if new_g_ks < best_new_max - 1e-12:
                    h[i], h[j] = 1.0, 0.0
                    c = compute_c5(h, n)
                    if c < best_new_max * (2.0 / n):
                        best_new_max = c / (2.0 / n)
                        best_swap = (i, j)
                    else:
                        h[i], h[j] = 0.0, 1.0
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = 1.0, 0.0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c

def sa_binary(h_init, n, deadline, max_iter=50000):
    target_idx = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target_idx]] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = np.correlate(h, 1 - h, mode='full')[:n]
    cur_c = float(np.max(g) * 2.0 / n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = 0.5
    T_min = 0.001
    cool = (T_min / T) ** (1.0 / max_iter)
    rng = np.random.RandomState(123)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        g_new = np.correlate(h, 1 - h, mode='full')[:n]
        c_new = float(np.max(g_new) * 2.0 / n)
        if c_new < cur_c or rng.random() < np.exp((cur_c - c_new) / T):
            zeros.remove(i)
            ones.append(i)
            ones.remove(j)
            zeros.append(j)
            g = g_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
            zeros.remove(j)
            ones.append(j)
            ones.remove(i)
            zeros.append(i)
        T *= cool
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: SLSQP with analytic gradient
    p1 = t0 + 0.45 * (budget_s - 3)
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    for alpha in alphas:
        if time.time() > p1 - 10:
            break
        obj, obj_grad, cons_fun, cons_jac = make_slsqp_funcs(n, alpha)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h.copy(), jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
    # Phase 2: Subgradient with analytic gradient, multi-active
    p2 = t0 + 0.65 * (budget_s - 3)
    h = best_h.copy()
    dx = 2.0 / n
    lr = 0.3
    for it in range(30000):
        if time.time() > p2:
            break
        g = np.correlate(h, 1 - h, mode='full')[:n]
        top_k = np.argsort(g)[::-1][:5]
        grad = np.zeros(n)
        for k in top_k:
            grad += grad_gk(h, n, k)
        grad *= dx / 5.0
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        c_cur = compute_c5(h, n)
        step = lr / ng
        ht = project_feasible(h - step * grad, n)
        ct = compute_c5(ht, n)
        if ct < best_c:
            h = ht
            best_c = ct
            best_h = h.copy()
            lr = min(lr * 1.05, 2.0)
        elif ct < c_cur * (1 - 1e-8):
            h = ht
            lr = min(lr * 1.02, 2.0)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    # Phase 3: Binary local search
    p3 = t0 + 0.82 * (budget_s - 3)
    if time.time() < p3 - 5:
        hb, cb = binary_local_search(best_h, n, p3, max_iter=200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 4: Try other n values
    p4 = t0 + 0.93 * (budget_s - 3)
    for nh in [100, 150, 200, 300, 400]:
        if time.time() > p4 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [5000, 50000, 500000]:
            if time.time() > p4 - 5:
                break
            obj, obj_grad, cons_fun, cons_jac = make_slsqp_funcs(nh, alpha)
            bounds = [(0.0, 1.0)] * nh
            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            try:
                res = minimize(obj, hh, jac=obj_grad, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 1500, 'ftol': 1e-14})
                hh = project_feasible(res.x, nh)
                c = compute_c5(hh, nh)
                if c < best_c:
                    best_c = c
                    best_h = hh.copy()
                    n = nh
            except Exception:
                pass
    # Phase 5: Final SLSQP polish
    if time.time() < dl - 5:
        obj, obj_grad, cons_fun, cons_jac = make_slsqp_funcs(n, 50000000)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h.copy(), jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
