    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    val = m + np.log(exp_vals.sum()) / beta

    """Gradient of smooth max of corr values w.r.t. h."""
    # p has length 2n-1, corresponding to lags -(n-1) to (n-1)
    # corr[m] = sum_j h[j] * (1-h[j+lag]) where lag = m-(n-1)
    # d(corr[m])/dh[i] = (1-h[i+lag])*[valid] - h[i-lag]*[valid]

            continue
        lag = m - (n - 1)

            grad[idx] += p[m] * (1.0 - h[idx + lag])
