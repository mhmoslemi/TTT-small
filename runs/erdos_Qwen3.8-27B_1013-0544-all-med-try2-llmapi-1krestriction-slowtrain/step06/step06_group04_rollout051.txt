    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        h_new = slsqp(h, n, alpha, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def gradient_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        for _ in range(300):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-3:]
            grad = np.zeros(n)
            na = 0
            for k in top_idx:
                m = abs(k - (n - 1))
                if m == 0:
                    grad += 1.0 - 2.0 * h
                    na += 1
                elif m < n:
                    if k > n - 1:
                        grad[:n - m] += (1.0 - h[m:])
                        grad[m:] -= h[:n - m]
                    else:
                        grad[:n - m] += h[m:]
                        grad[m:] -= (1.0 - h[:n - m])
                    na += 1
            if na == 0:
                break
            grad /= na
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                break
    return best_h, best_c

def pair_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            m = abs(k_star - (n - 1))
            if m == 0 or m >= n:
                break
            if k_star > n - 1:
                a = h[:n - m].copy()
                b = h[m:].copy()
                diffs = a - b
                candidates = np.argsort(-np.abs(diffs))[:30]
                for i in candidates:
                    if time.time() > deadline:
                        break
                    if diffs[i] > delta * 2:
                        d = min(delta, diffs[i] * 0.5)
                        h[i] -= d
                        h[i + m] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[i + m] -= d
                    elif diffs[i] < -delta * 2:
                        d = min(delta, -diffs[i] * 0.5)
                        h[i] += d
                        h[i + m] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[i + m] += d
    return best_h, best_c

def outer_lp_step(h, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    nn = 2 * n - 1
    A_ineq_rows = []
    b_ineq_vals = []
    for k in range(nn):
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            if m == 0:
                grad = 1.0 - 2.0 * h
            else:
                continue
        else:
            grad = np.zeros(n)
            if lag > 0:
                grad[:n - m] -= h[m:]
                grad[m:] += (1.0 - h[:n - m])
            elif lag < 0:
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] -= h[:n - m]
            else:
                grad[:] = 1.0 - 2.0 * h
        row = np.zeros(n + 1)
        row[:n] = dx * grad
        row[-1] = -1.0
        A_ineq_rows.append(row)
        b_ineq_vals.append(c[k] - dx * (grad @ h))
    for i in range(n):
        r1 = np.zeros(n + 1)
        r1[i] = 1.0
        A_ineq_rows.append(r1)
        b_ineq_vals.append(h[i] + trust)
        r2 = np.zeros(n + 1)
        r2[i] = -1.0
        A_ineq_rows.append(r2)
        b_ineq_vals.append(h[i] - trust)
    A_ineq = np.array(A_ineq_rows)
    b_ineq = np.array(b_ineq_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return h.copy()

def outer_loop(h, n, deadline, max_iters=80):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    trust = 0.12
    for _ in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_lp_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() < deadline - 60:
        h2, c2 = outer_loop(h, n, min(deadline - 50, time.time() + 80))
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 30:
        h2, c2 = adaptive_slsqp(best_h, n, deadline - 20)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 10:
        h2, c2 = gradient_search(best_h, n, deadline - 5)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 3:
        h2, c2 = pair_search(best_h, n, deadline - 1)
        if c2 < best_c:
            best_c, best_h = c2, h2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(300, 0.5)
        n0 = 300
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Main optimization at initial n
    alloc1 = min(400, (deadline - t0) * 0.45)
    h1, c1 = optimize_n(best_h, n0, t0 + alloc1)
    if c1 < best_c:
        best_c, best_h = c1, h1
    # Try other n values
    for n_try in [250, 350, 400]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.25)
        if alloc < 40:
            break
        h2, c2 = optimize_n(h_interp, n_try, time.time() + alloc)
        if c2 < best_c:
            best_c, best_h = c2, h2
    # Random restarts at best n
    n_best = len(best_h)
    remaining = deadline - time.time() - 30
    if remaining > 60:
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            noise = 0.01 + 0.005 * r
            hp = best_h + rng.normal(0, noise, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 25) * 0.35)
            if alloc < 20:
                break
            h3, c3 = optimize_n(hp, n_best, time.time() + alloc)
            if c3 < best_c:
                best_c, best_h = c3, h3
    # Final polish
    if time.time() < deadline - 8:
        h_fin = slsqp(best_h, len(best_h), 3000000, maxiter=5000, ftol=1e-18)
        c_fin = compute_c5(h_fin, len(best_h))
        if c_fin < best_c:
            best_c, best_h = c_fin, h_fin
    if time.time() < deadline - 2:
        h_fin, c_fin = pair_search(best_h, len(best_h), deadline - 1)
        if c_fin < best_c:
            best_c, best_h = c_fin, h_fin
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
