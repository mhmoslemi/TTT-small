                q = n // 4
                x0 = np.zeros(n)
                x0[:q] = 1.0
                x0[2 * q:3 * q] = 1.0
                x0 = project_feasible(x0, target_sum)
                x0 = 0.5 * x0 + 0.5 * best_h
                x0 = project_feasible(x0, target_sum)
            else:
                # Quasi-random
                x0 = 0.5 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n))
                x0 = project_feasible(x0, target_sum)

            for beta in [3000, 15000]:
                if budget_s - (time.time() - t0) < 12:
                    break
                cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum,)}]
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 4: Multi-resolution exploration
    if budget_s - (time.time() - t0) > 90:
        for n_try in [n + 10, n - 10, n + 20, n + 30, n - 20, n + 40, n - 30]:
            if n_try < 30 or budget_s - (time.time() - t0) < 40:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (ts_t,)}]
            for beta in [2000, 10000]:
                if budget_s - (time.time() - t0) < 18:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 2000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        break
                except Exception:
                    pass

    # Phase 5: Intensive subgradient polish on best
    if budget_s - (time.time() - t0) > 50:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        h = best_h.copy()
        velocity = np.zeros(n)
        lr = 0.003
        momentum = 0.8
        for it in range(40000):
            if budget_s - (time.time() - t0) < 20:
                break
            grad = combined_gradient(h, n, 5, 40.0)
            velocity = momentum * velocity - lr / (1.0 + 0.0001 * it) * grad
            h = project_feasible(h + velocity, target_sum)
            if it % 2000 == 1999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                    velocity = np.zeros(n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 6: Ultra-high beta final polish
    if budget_s - (time.time() - t0) > 15:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds_final = Bounds(np.zeros(n), np.ones(n))
        cons_final = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum,)}]
        for beta in [50000, 200000]:
            if budget_s - (time.time() - t0) < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds_final, constraints=cons_final,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
