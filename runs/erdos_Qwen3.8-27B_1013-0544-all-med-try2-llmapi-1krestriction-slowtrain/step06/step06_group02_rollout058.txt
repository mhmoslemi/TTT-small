    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(np.max(corr))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n, deadline, max_rounds=1000, grid1=21, grid2=31):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(np.max(corr))
    best_h = h.copy()
    ns = 2 * n - 1
    k0 = n - 1

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        any_improved = False
        for i in range(n):
            hi = h[i]
            slopes = np.zeros(ns)
            lo1 = max(n - 1 - i, 0)
            hi1 = min(2 * n - 2 - i, ns - 1)
            if lo1 <= hi1:
                idxs = np.arange(lo1, hi1 + 1)
                slopes[idxs] += (1.0 - h[i + idxs - (n - 1)]) * dx
            lo2 = max(i, 0)
            hi2 = min(n - 1 + i, ns - 1)
            if lo2 <= hi2:
                idxs = np.arange(lo2, hi2 + 1)
                slopes[idxs] -= h[i - (idxs - (n - 1))] * dx

            def eval_t(t):
                delta = t - hi
                if abs(delta) < 1e-15:
                    return float(np.max(corr))
                nm = np.max(corr + delta * slopes)
                ck = corr[k0] + dx * delta * (1.0 - 2.0 * hi - delta)
                if ck > nm:
                    nm = ck
                return nm

            t_grid = np.linspace(0.0, 1.0, grid1)
            t_grid = np.unique(np.append(t_grid, hi))
            best_t = hi
            best_val = float(np.max(corr))
            for t in t_grid:
                v = eval_t(t)
                if v < best_val - 1e-15:
                    best_val = v
                    best_t = t

            # Refine around best_t
            lo_r = max(0.0, best_t - 0.06)
            hi_r = min(1.0, best_t + 0.06)
            t_fine = np.linspace(lo_r, hi_r, grid2)
            t_fine = np.unique(np.append(t_fine, hi))
            for t in t_fine:
                v = eval_t(t)
                if v < best_val - 1e-15:
                    best_val = v
                    best_t = t

            if best_t != hi:
                delta = best_t - hi
                h[i] = best_t
                corr += delta * slopes
                corr[k0] -= dx * delta * delta
                any_improved = True
                if best_val < best_c:
                    best_c = best_val
                    best_h = h.copy()
        if not any_improved:
            break
    return best_h, best_c

def _smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += 0.1 * (x.sum() - n / 2.0) ** 2
    return float(val)

def _smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 0.2 * (x.sum() - n / 2.0)
    return grad

def lbfgs_adaptive(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        t_alloc = min(40, deadline - time.time() - 3)
        if t_alloc < 5:
            break
        try:
            res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 3000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 1500, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def block_perturb(h, n, rng, block_size=10, magnitude=0.03):
    h_new = h.copy()
    nb = n // block_size
    for b in range(nb):
        offset = rng.uniform(-magnitude, magnitude)
        h_new[b*block_size:(b+1)*block_size] += offset
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n)
