```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    cor = np.correlate(x, 1.0 - x, mode='full')
    m = cor.max()
    shifted = alpha * (cor - m)
    shifted -= shifted.max()
    logsum = np.log(np.exp(shifted).sum())
    return (m + logsum / alpha) * dx

def smooth_grad_func(x, n, alpha):
    dx = 2.0 / n
    cor = np.correlate(x, 1.0 - x, mode='full')
    m = cor.max()
    shifted = alpha * (cor - m)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    offsets = (n - 1) - np.arange(2 * n - 1)
    idx_A = np.arange(n)[:, None] + offsets[None, :]
    mask_A = (idx_A >= 0) & (idx_A < n)
    A = np.where(mask_A, x[np.clip(idx_A, 0, n - 1)], 0.0) @ p
    idx_B = np.arange(n)[:, None] - offsets[None, :]
    mask_B = (idx_B >= 0) & (idx_B < n)
    B = np.where(mask_B, x[np.clip(idx_B, 0, n - 1)], 0.0) @ p
    return dx * (1.0 - A - B)

def lbfgs_smooth(h, n, alpha, penalty, maxiter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        f = smooth_obj(x, n, alpha)
        c = x.sum() - target
        return f + penalty * c * c
    def grad(x):
        g = smooth_grad_func(x, n, alpha)
        g = g + 2.0 * penalty * (x.sum() - target)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
```

- +1 at position n-1+(i-j)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    cor = np.correlate(x, 1.0 - x, mode='full')
    m = cor.max()
    shifted = alpha * (cor - m)
    shifted -= shifted.max()
    return (m + np.log(np.exp(shifted).sum()) / alpha) * dx

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    cor = np.correlate(x, 1.0 - x, mode='full')
    m = cor.max()
    shifted = alpha * (cor - m)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    offsets = (n - 1) - np.arange(2 * n - 1)
    idx_A = np.arange(n)[:, None] + offsets[None, :]
    mask_A = (idx_A >= 0) & (idx_A < n)
    A = np.where(mask_A, x[np.clip(idx_A, 0, n - 1)], 0.0) @ p
    idx_B = np.arange(n)[:, None] - offsets[None, :]
    mask_B = (idx_B >= 0) & (idx_B < n)
    B = np.where(mask_B, x[np.clip(idx_B, 0, n - 1)], 0.0) @ p
    return dx * (1.0 - A - B)

def lbfgs_smooth(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0
    pen = 100.0
    def obj(x):
        f = smooth_obj(x, n, alpha)
        c = x.sum() - target
        return f + pen * c * c
    def grad(x):
        g = smooth_grad(x, n, alpha)
        return g + 2.0 * pen * (x.sum() - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
