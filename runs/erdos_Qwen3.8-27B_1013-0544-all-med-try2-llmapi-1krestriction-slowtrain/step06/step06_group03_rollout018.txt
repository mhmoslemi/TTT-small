```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        room = 1.0 - h
        idx = np.where(room > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)

    sh = alpha * (c - m)
```

```python
def compute_coefs(h, n):
    n_lags = 2 * n - 1
    coefs = np.zeros((n, n_lags))
    for k in range(-(n-1), n):
        if k == 0:
            continue
        p = k + n - 1  # lag index in the full correlation
        # For each j: coefs[j, p] = (1-h[j+k])*I(0≤j+k<n) - h[j-k]*I(0≤j-k<n)
        # j+k < n → j < n-k; j+k >= 0 → j >= -k
        # j-k >= 0 → j >= k; j-k < n → j < n+k
        # So first term valid for j in [max(0, -k), min(n-1, n-1-k)]
        # Second term valid for j in [max(0, k), min(n-1, n-1+k)]
        if k > 0:
            # First term: j from 0 to n-1-k
            js1 = np.arange(0, n - k)
            coefs[js1, p] += (1 - h[js1 + k])
            # Second term: j from k to n-1
            js2 = np.arange(k, n)
            coefs[js2, p] -= h[js2 - k]
        else:
            # First term: j from -k to n-1
            js1 = np.arange(-k, n)
            coefs[js1, p] += (1 - h[js1 + k])
            # Second term: j from 0 to n-1+k = n-1+|k|
            js2 = np.arange(0, n + k)  # n+k = n-|k|
            coefs[js2, p] -= h[js2 - k]
    return coefs
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_overlaps(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        room = 1.0 - h
        idx = np.where(room > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_coefs(h, n):
    n_lags = 2 * n - 1
    coefs = np.zeros((n, n_lags))
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        p = k + n - 1
        if k > 0:
            js1 = np.arange(0, n - k)
            coefs[js1, p] += (1.0 - h[js1 + k])
            js2 = np.arange(k, n)
            coefs[js2, p] -= h[js2 - k]
        else:
            js1 = np.arange(-k, n)
            coefs[js1, p] += (1.0 - h[js1 + k])
            js2 = np.arange(0, n + k)
            coefs[js2, p] -= h[js2 - k]
    return coefs

def coord_descent_vec(h, n, deadline, grid_size=80):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    overlaps = compute_overlaps(h, n)
    best_c = overlaps.max()
    grid = np.linspace(0.0, 1.0, grid_size)

    for sweep in range(50000):
        if time.time() > deadline:
            break
        coefs = compute_coefs(h, n)
        improved = False
        for j in range(n):
            old = h[j]
            old_ov = overlaps
            old_c = best_c
            deltas = grid - old
            new_ovals = old_ov[None, :] + dx * deltas[:, None] * coefs[j, None, :]
            new_ovals[:, n - 1] = old_ov[n - 1] + dx * deltas * (1.0 - 2.0 * old - deltas)
            maxes = new_ovals.max(axis=1)
            best_idx = maxes.argmin()
            if maxes[best_idx] < old_c - 1e-14:

                overlaps = new_ovals[best_idx]
                best_c = maxes[best_idx]
                improved = True
        if not improved:
            break

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def pair_search_fine(h, n, deadline, deltas):

    for d in deltas:
```
