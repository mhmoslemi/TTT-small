import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sobj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh = np.minimum(sh, 500)
    return float(m + np.log(np.exp(sh).sum()) / alpha)

def _sgrad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh = np.minimum(sh, 500)
    es = np.exp(sh)
    sm = es / es.sum()
    n_full = len(c)
    g = np.zeros(n)
    for k in range(n_full):
        w = sm[k] * dx
        if w < 1e-18:
            continue
        lag = k - (n - 1)
        lo = max(0, lag)
        hi = min(n, n + lag)
        if hi > lo:
            g[lo:hi] += w * (1.0 - x[lo - lag:hi - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            g[lo2:hi2] += w * (-x[lo2 + lag:hi2 + lag])
    return g

def _con_f(x, n):
    return x.sum() - n / 2.0

def _con_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_f, 'jac': _con_j, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, args=(n, alpha), jac=_sgrad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            hx = project_feasible(res.x, n)
            if compute_c5(hx, n) <= compute_c5(h, n) + 1e-10:
                return hx
    except Exception:
        pass
    return h.copy()

def local_search(h, n, deadline, deltas, radius):
    h = h.copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n):
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
        if not improved:
            break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sobj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    return float(m + np.log(np.exp(sh).sum()) / alpha)

def _sgrad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    es = np.exp(alpha * (c - m))
    sm = es / es.sum()
    nfull = 2 * n - 1
    g = np.zeros(n)
    for k in range(nfull):
        w = sm[k] * dx
        if w < 1e-18:
            continue
        lag = k - (n - 1)
        lo = max(0, lag)
        hi = min(n, n + lag)
        if hi > lo:
            g[lo:hi] += w * (1.0 - x[lo - lag:hi - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            g[lo2:hi2] += w * (-x[lo2 + lag:hi2 + lag])
    return g

def _cf(x, n):
    return x.sum() - n / 2.0

def _cj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, args=(n, alpha), jac=_sgrad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hx = project_feasible(res.x, n)
        if compute_c5(hx, n) <= compute_c5(h, n) + 1e-12:
            return hx
    except Exception:
        pass
    return h.copy()

def local_search(h, n, deadline, deltas, radius):
    h = h.copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n):
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
        if not improved:
            break
    return h, best_c
