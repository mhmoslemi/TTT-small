import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def _obj_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(cc[top_idx], alpha)

def slsqp(h, n, alpha, top_idx=None, maxiter=4000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        try:
            res = minimize(_obj_active, h, args=(n, top_idx, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(_obj_full, h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=40):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def multilag_gradient_descent(h, n, deadline, top_m=6, max_iters=200):
    """Multi-lag gradient descent with backtracking line search."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for it in range(max_iters):
        if time.time() > deadline - 1:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        # Weighted gradient: weight each top lag by its value
        weights = c[top_idx]
        w_max = weights.max()
        if w_max < 1e-15:
            break
        weights = weights / w_max
        flow = np.zeros(n)
        for w, k_idx in zip(weights, top_idx):
            lag = k_idx - (n - 1)
            flow += w * _lag_gradient(h_curr, n, lag)
        flow -= np.mean(flow)
        # Backtracking line search
        f_curr = float(np.max(c[top_idx]))
        step = 0.5
        improved = False
        for _ in range(20):
            h_trial = project_feasible(h_curr - step * flow, n)
            c_trial = compute_all_lags(h_trial, n)
            f_trial = float(np.max(c_trial[top_idx]))
            if f_trial < f_curr - 1e-13:
                h_curr = h_trial
                c_new = compute_c5(h_curr, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_curr.copy()
                improved = True
                break
            step *= 0.6
        if not improved:
            # Try even smaller steps or break
            step = 0.001
            h_trial = project_feasible(h_curr - step * flow, n)
            c_new = compute_c5(h_trial, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_trial.copy()
                h_curr = h_trial
            else:
                break
    return best_h, best_c

    """Two-phase local search: multi-lag flow + pairwise."""
    h = h.copy()
    best_c = compute_c5(h, n)

                    if delta < 1e-7:

                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:

                        found = True
                    else:
