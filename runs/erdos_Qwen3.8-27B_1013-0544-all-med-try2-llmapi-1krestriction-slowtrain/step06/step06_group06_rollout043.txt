def compute_delta(h, n, i, j):
    delta = np.zeros(2*n-1)
    
    # Term 1: m=i direct
    k_lo = max(0, n-1-i)
    k_hi = min(2*n-2, n-1+i)
    a = np.arange(k_lo, k_hi+1) + i - (n-1)

    # Term 2: m=j direct

    k_hi = min(2*n-2, n-1+j)
    a = np.arange(k_lo, k_hi+1) + j - (n-1)
    vals = 1 - h[a]
    # Fix for a==j (k=n-1): set to 0
    mask_j = (a == j)
    vals[mask_j] = 0
    # Fix for a==i: set to 1
    mask_i = (a == i)
    vals[mask_i] = 1
    delta[k_lo:k_hi+1] += vals
    
    # Term 3: m where m+k-(n-1)=i, i.e., k=i-m+(n-1)
    # m ranges over [0,n-1] excluding i and j
    # k = i - m + (n-1), valid when 0 <= k <= 2n-2
    # n-1+i <= 2n-2 always (since i<=n-1), and i-(n-1) >= 0 requires i >= n-1...

    # wait: m=0 → k=i+n-1, m=n-1 → k=i-(n-1)
    # So k ranges from i-(n-1) to i+n-1

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_delta(h, n, i, j):
    """Compute change in c vector when swapping h[i]:1->0 and h[j]:0->1."""
    delta = np.zeros(2*n-1)
    
    # Term 1: m=i, direct contribution
    k_lo = max(0, n - 1 - i)
    k_hi = min(2*n-2, n - 1 + i)
    if k_lo <= k_hi:
        a = np.arange(k_lo, k_hi+1) + i - (n-1)
        delta[k_lo:k_hi+1] -= (1.0 - h[a])
    
    # Term 2: m=j, direct contribution
    k_lo = max(0, n - 1 - j)
    k_hi = min(2*n-2, n - 1 + j)
    if k_lo <= k_hi:

        vals[a == i] = 1.0  # special case

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP with soft-max approximation."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
