    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_val_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    w = exps / exps.sum()
    val = m + (mx + np.log(exps.sum())) / alpha
    grad = np.zeros(n)
    nm1 = n - 1
    for s in range(-nm1, nm1 + 1):
        ws = w[s + nm1]
        if ws < 1e-13:
            continue
        wd = ws * dx
        if s >= 0:
            if s < n:
                grad[:n-s] += wd * (1.0 - x[s:])
                grad[s:] -= wd * x[:n-s]
        else:
            sp = -s
            grad[sp:] += wd * (1.0 - x[:n-sp])
            grad[:n-sp] -= wd * x[sp:]
    return val, grad

    return x.sum() - n / 2.0

    return np.ones(n)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

        return h.copy()

            best_h = h_new.copy()

        else:

def pair_refine(h, n, deadline, deltas, max_passes=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        for _ in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for sgn in (1, -1):
                        hi, hj = h[i] + sgn*d, h[j] - sgn*d
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sgn*d, hj + sgn*d
            if not improved:
                break
    return h, best_c
