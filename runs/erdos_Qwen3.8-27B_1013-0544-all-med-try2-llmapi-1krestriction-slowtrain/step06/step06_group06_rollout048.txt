from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = (1.0 - h)
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_gradient(h, n, alpha):
    """Compute gradient of soft-max objective using convolution."""
    dx = 2.0 / n
    g = compute_all_g(h, n)  # length 2n-1
    m = g.max()
    shifted = alpha * (g - m)
    max_shifted = shifted.max()
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()

    V = weights  # length 2n-1
    V_rev = V[::-1]  # length 2n-1

    # We only want the sum over valid m (0 to n-1)
    
    conv1 = np.convolve(1.0 - h, V, mode='full')  # length n + 2n-2 = 3n-2
    conv2 = np.convolve(h, V_rev, mode='full')  # length 3n-2
    
    grad = dx * (conv1[:n] - conv2[:n])
    return grad, m, weights

    best_h = h.copy()
    best_c = compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj_and_grad(h, n, alpha):
    """Compute soft-max objective and gradient."""
    dx = 2.0 / n
    g = compute_all_g(h, n)  # length 2n-1
    m = g.max()
    shifted = alpha * (g - m)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    E = exp_sh.sum()
    obj = m + np.log(E) / alpha
    weights = exp_sh / E  # softmax weights, length 2n-1
    
    # Gradient via convolution
    V = weights
    V_rev = V[::-1]
    conv1 = np.convolve(1.0 - h, V, mode='full')[:n]
    conv2 = np.convolve(h, V_rev, mode='full')[:n]
    grad = dx * (conv1 - conv2)
    return obj, grad, m

def proj_grad_descent(h, n, deadline, alpha_start=500, alpha_end=50000, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_start
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad, m = soft_max_obj_and_grad(h, n, alpha)
        
        # Backtracking line search
        lr = 2.0 / (alpha * (2.0/n) + 1.0)
        # Actually, use a simpler initial lr
        lr = 1.0 / (alpha * 2.0 / n + 0.1)
        for attempt in range(40):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new <= m - 1e-15:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                break
            lr *= 0.5
        else:
            # No improvement, try adjusting alpha
            if alpha > 1000:
                alpha = max(100.0, alpha * 0.8)
            else:
                break
        
        # Periodically increase alpha
        if it % 1000 == 999 and alpha < alpha_end:
            alpha = min(alpha_end, alpha * 1.5)
    
    return best_h, best_c

def binary_search_fast(h_bin, n, deadline, max_sweeps=500):
    """Fast binary local search using vectorized operations."""
    dx = 2.0 / n
    g = compute_all_g(h_bin, n)
    best_c = float(g.max())
    h = h_bin.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Precompute flip effects: effect[i][p] = change in g[p] when h[i] goes 0->1
        # effect[i][p] = dx * [(1-h[i+k_p])*1_{valid} - h[i-k_p]*1_{valid}]
        # where k_p = p - (n-1)
        
        # Vectorized computation:
        # For all i at once, we need h[i+k] and h[i-k] for all k
        # This is a Toeplitz-like structure
        
        # Use the convolution approach:
        # effect[i] = dx * (convolve(1-h, delta_i, 'full')[offset] - convolve(h, delta_i_rev, 'full')[offset])
        # But that's O(n^2) per i, total O(n^3) - same as before
        
        # Alternative: build the effect matrix directly
        k = np.arange(-(n-1), n)  # length 2n-1
        # For each i, k_p is the lag
        # h[i + k_p] valid when 0 <= i + k_p < n
        # h[i - k_p] valid when 0 <= i - k_p < n
        
        # Build index matrices
        # idx_plus[i, p] = i + k[p], valid when 0 <= i+k[p] < n
        # idx_minus[i, p] = i - k[p], valid when 0 <= i-k[p] < n
        
        idx_plus = np.arange(n)[:, None] + k[None, :]  # (n, 2n-1)
        idx_minus = np.arange(n)[:, None] - k[None, :]
        
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        
        # h at valid positions
        h_plus = np.zeros_like(idx_plus, dtype=float)
        h_minus = np.zeros_like(idx_minus, dtype=float)
        h_plus[mask_plus] = h[idx_plus[mask_plus]]
        h_minus[mask_minus] = h[idx_minus[mask_minus]]
        
        # effect[i, p] = dx * ((1-h[i+k_p])*mask_plus[i,p] - h[i-k_p]*mask_minus[i,p])
        effect = dx * ((1.0 - h_plus) * mask_plus - h_minus * mask_minus)
        
        # For swap (i from zeros, j from ones): Δg = effect[i] - effect[j]
        # new_g = g + effect[i] - effect[j]
        # Find pair (i,j) that minimizes max(new_g)
        
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        
        best_pair = None
        best_val = best_c
        
        # For each zero i, find the best one j
        for i in zeros_idx:
            if time.time() > deadline:
                break
            # new_g for all j: g + effect[i] - effect[j] for j in ones_idx
            # = (g + effect[i]) - effect[j]
            base = g + effect[i]  # (2n-1,)
            # max(base - effect[j]) for j in ones_idx
            # = max over p of (base[p] - effect[j][p]) for each j
            # Vectorized: -effect[ones_idx] has shape (|ones|, 2n-1)
            # base[None, :] - effect[ones_idx] has shape (|ones|, 2n-1)
            # max along axis=1 gives (|ones|,)
            new_maxes = np.max(base[None, :] - effect[ones_idx], axis=1)
            j_best = ones_idx[np.argmin(new_maxes)]
            val = float(new_maxes.min())
            if val < best_val - 1e-14:
                best_val = val
                best_pair = (i, j_best)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i] = 1.0
        h[j] = 0.0
        g = g + effect[i] - effect[j]
        best_c = best_val
    
    return h, best_c
