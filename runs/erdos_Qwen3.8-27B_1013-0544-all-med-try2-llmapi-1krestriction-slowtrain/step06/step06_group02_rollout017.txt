import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def thue_morse_init(n):
    """Generate a Thue-Morse-like binary initialization with exactly n/2 ones."""
    t = np.zeros(n)
    # Use popcount mod 2
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    # Fix to have exactly n//2 ones
    target = n // 2
    cur = int(t.sum())
    if cur > target:
        idx = np.where(t > 0.5)[0]
        # Remove from the end
        for k in range(cur - target):
            t[idx[-1-k]] = 0
    elif cur < target:
        idx = np.where(t < 0.5)[0]
        for k in range(target - cur):
            t[idx[-1-k]] = 1
    return t

def binary_local_search(h, n, deadline, max_passes=100):
    """Local search on binary sequences using pair swaps."""
    best_c = compute_c5(h, n)
    c_vec = np.correlate(h, 1-h, mode='full') * (2.0/n)
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in ones:
            if time.time() > deadline:
                return h, best_c
            for j in zeros:
                # Try swapping: h[i] 1→0, h[j] 0→1
                h[i] = 0; h[j] = 1
                c_new = np.max(np.correlate(h, 1-h, mode='full')) * (2.0/n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    improved = True
                else:
                    h[i] = 1; h[j] = 0
        if not improved:
            break

    best_n = n_parent

    # Phase 1: Continuous L-BFGS from parent
    if time.time() < deadline - 400:
        t1 = min(time.time() + 180, deadline - 400)
        h1, c1 = lbfgs_optimize(best_h, best_n, t1)
        if c1 < best_c:
            best_c = c1
            best_h = h1
            best_n = best_n

    # Phase 2: Binary local search from rounded continuous solution
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, best_n)
    if time.time() < deadline - 300:
        t2 = min(time.time() + 200, deadline - 300)
        h2, c2 = binary_local_search(h_bin, best_n, t2, max_passes=40, rng=rng)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Thue-Morse binary local search at multiple n
    for n_try in [best_n, 256, 200, 128]:
        if time.time() > deadline - 200:
            break
        h_tm = thue_morse_init(n_try, rng)
        t3 = min(time.time() + 120, deadline - 200)
        h3, c3 = binary_local_search(h_tm, n_try, t3, max_passes=40, rng=rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_try

    # Phase 4: Simulated annealing on best binary solution
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        h_sa_init = (best_h > 0.5).astype(float)
        h_sa_init = project_feasible(h_sa_init, n_cur)
        t4 = min(time.time() + 150, deadline - 100)
        h4, c4 = binary_sa(h_sa_init, n_cur, t4, rng, t_start=0.05, t_end=0.005)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_cur

    # Phase 5: Local search after SA
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        h5_init = (best_h > 0.5).astype(float)
        h5_init = project_feasible(h5_init, n_cur)
        t5 = min(time.time() + 60, deadline - 40)
        h5, c5 = binary_local_search(h5_init, n_cur, t5, max_passes=20, rng=rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Try continuous refinement on best
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        h6 = best_h.copy()
        t6 = min(time.time() + 40, deadline - 20)
        h6, c6 = lbfgs_optimize(h6, n_cur, t6, alphas=(10000, 50000, 200000))
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Final: try rounding and one more binary pass
    n_final = len(best_h)
    h_round = (best_h > 0.5).astype(float)
    h_round = project_feasible(h_round, n_final)
    if compute_c5(h_round, n_final) < best_c:
        best_h = h_round
    if time.time() < deadline - 2:
        best_h, best_c = binary_local_search(best_h, n_final, deadline - 1, max_passes=5, rng=rng)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
