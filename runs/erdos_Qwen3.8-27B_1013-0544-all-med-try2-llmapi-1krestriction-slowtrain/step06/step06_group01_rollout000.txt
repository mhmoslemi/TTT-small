import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def greedy_search(h, n, m_range, deadline, max_sweeps=2000, seed=42):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = make_delta(h, n, i, j, m_range)
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c

def sa_search(h, n, m_range, deadline, seed=42):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    n_ones = len(ones)
    n_zeros = len(zeros)
    rng = np.random.RandomState(seed)
    t = 1.5 * dx
    est_time = deadline - time.time()
    est_steps = max(int(est_time * 60000), 10000)
    cooling = (0.005 * dx / t) ** (1.0 / est_steps)
    no_improve = 0
    patience = 500000
    while time.time() < deadline:
        idx_i = rng.randint(n_zeros)
        idx_j = rng.randint(n_ones)
        i = zeros[idx_i]
        j = ones[idx_j]
        delta = make_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        d = new_c - cur_c
        if d < 0 or rng.rand() < np.exp(-d / max(t, 1e-12)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = new_c
            zeros[idx_i] = j
            ones[idx_j] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > patience:
            no_improve = 0
            h = best_h.copy()
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            t = min(t * 3.0, 1.5 * dx)
        t *= cooling
    return best_h, best_c
