def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_c(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return val, c

def grad_smooth_max(h, n, alpha):

                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; round_imp = True
                        else:
                            h[i] += d; h[j] -= d
                    if h[i] <= 1.0 - d and h[j] >= d:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; round_imp = True
                        else:
                            h[i] -= d; h[j] += d
            if not round_imp:
                break
    return h, best_c

def restart_exploration(h, n, deadline, scale=0.04, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    per = max(25, (deadline - time.time()) / n_starts - 5)
    for _ in range(n_starts):
        if time.time() > deadline - 20:
            break
        h_t = h.copy() + np.random.randn(n) * scale
        h_t = project_feasible(h_t, n)
        h_t, c_t = lbfgs_optimize(h_t, n, time.time() + per * 0.6, 5000, 20.0, 200)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Multi-alpha L-BFGS (coarse to fine)
    schedule = [(100, 50), (500, 70), (2000, 80), (10000, 90), (50000, 90)]
    for alpha, alloc in schedule:
        remaining = deadline - time.time() - 350
        if remaining < 30:
            break
        end = min(time.time() + alloc, deadline - 300)
        if end - time.time() < 15:
            end = time.time() + 15
        h_new, c_new = lbfgs_optimize(h, n, end, alpha, 20.0, 400)
        if c_new < best_c:
            best_c = c_new; best_h = h_new; h = h_new

    # Phase 2: Nesterov projected gradient
    if time.time() < deadline - 300:
        end = min(time.time() + 180, deadline - 250)
        h_new, c_new = nesterov_projected(h, n, end, alpha=5000, lr_init=0.003)
        if c_new < best_c:
            best_c = c_new; best_h = h_new; h = h_new

    # Phase 3: Gradient-directed pair search
    if time.time() < deadline - 250:
        end = min(time.time() + 180, deadline - 200)
        h_new, c_new = grad_pair_search(h, n, end, [0.005, 0.002, 0.001, 0.0005])
        if c_new < best_c:
            best_c = c_new; best_h = h_new; h = h_new

    # Phase 4: General pair search (broader neighborhood)
    if time.time() < deadline - 180:
        end = min(time.time() + 130, deadline - 130)
        h_new, c_new = general_pair(h, n, end, [0.002, 0.001, 0.0005, 0.0002], stride=2, radius=60)
        if c_new < best_c:
            best_c = c_new; best_h = h_new; h = h_new

    # Phase 5: Restart exploration to escape local optima
    if time.time() < deadline - 130:
        end = min(time.time() + 100, deadline - 80)
        h_new, c_new = restart_exploration(h, n, end, scale=0.03, n_starts=3)
        if c_new < best_c:
            best_c = c_new; best_h = h_new; h = h_new

    # Phase 6: Final L-BFGS polish
    if time.time() < deadline - 40:
        end = min(time.time() + 40, deadline - 15)
        h_new, c_new = lbfgs_optimize(best_h, n, end, 200000, 100.0, 300)
        if c_new < best_c:
            best_c = c_new; best_h = h_new

    # Phase 7: Ultra-fine pair search
    if time.time() < deadline - 10:
        end = deadline - 5
        h_new, c_new = grad_pair_search(best_h, n, end, [0.0005, 0.0002, 0.0001, 0.00005])
        if c_new < best_c:
            best_c = c_new; best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
