            s = h.sum()
            if s < 1e-14: break

def compute_P(h, n):
    """Compute P[p, m] = change in corr[m] when h[p] goes 0→1."""
    # corr[m] = sum_i h[i] * (1-h[i+k]) where k = m - (n-1)

    #   corr[m] += (1-h[p+k]) [if p+k valid] - h[p-k] [if p-k valid]
    P = np.zeros((n, 2*n-1))

        pk = p + k
        v1 = (pk >= 0) & (pk < n)
        P[p, v1] += (1.0 - h[pk[v1]])
        pk2 = p - k
        v2 = (pk2 >= 0) & (pk2 < n)
        P[p, v2] -= h[pk2[v2]]
    return P

    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:target]] = 1.0
    rng = np.random.RandomState(seed)
    
    best_c = np.inf
    best_h = h.copy()
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    P = compute_P(h, n)
    
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    
    improved_total = True
    while improved_total and time.time() < deadline:
        improved_total = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # For each zero, find best one to swap with
        # Heuristic: try zeros in random order, for each try a subset of ones
        z_order = rng.permutation(len(zeros))

            if time.time() > deadline:
                return best_h, best_c
            i = zeros[idx_z]
            V_i = corr + P[i]
            vi_max = V_i.max()

            # Quick bound: if vi_max - max(P[j]) >= cur_c/dx, might still work
            # Better: just compute for top candidates
            
            # Heuristic: sort ones by P[j] at the critical position
            k_crit = np.argmax(V_i)
            j_scores = P[ones, k_crit]
            j_order = ones[np.argsort(-j_scores)[:min(50, len(ones))]]
            
            best_j = -1
            best_val = cur_c / dx
            
            for j in j_order:
                val = (V_i - P[j]).max()
                if val < best_val:
                    best_val = val
                    best_j = j
                if val * dx < cur_c * 0.99:  # good enough, break
                    break
            
            if best_j >= 0 and best_val * dx < cur_c - 1e-12:
                # Accept
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + P[i] - P[best_j]
                cur_c = best_val * dx
                # Update P
                # P changes because h changed at positions i and best_j
                # P[p, m] depends on h at positions p±(m-(n-1))

                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break  # restart zero loop with updated P

def update_P_swap(P, n, i, j):
    """Update P after h[i] goes 0→1 and h[j] goes 1→0."""
    m = np.arange(2*n-1)
    # h[i] 0→1: P decreases
    p1 = i - m + n - 1
    v1 = (p1 >= 0) & (p1 < n)
    P[p1[v1], m[v1]] -= 1.0
    p2 = i + m - n + 1
    v2 = (p2 >= 0) & (p2 < n)

    v2 = (p2 >= 0) & (p2 < n)
    P[p2[v2], m[v2]] += 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_P(h, n):
    P = np.zeros((n, 2 * n - 1))
    N1 = n - 1
    for p in range(n):
        m = np.arange(2 * n - 1)
        shift = m - N1
        pk = p + shift
        v1 = (pk >= 0) & (pk < n)
        P[p, v1] += (1.0 - h[pk[v1]])
        pk2 = p - shift
        v2 = (pk2 >= 0) & (pk2 < n)
        P[p, v2] -= h[pk2[v2]]
        P[p, N1] -= 1.0
    return P

def binary_search_fast(h_init, n, deadline, max_sweeps=20000, K=30, seed=42):

    target = n // 2

    h[np.argsort(h_init)[::-1][:target]] = 1.0

    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    P = compute_P(h, n)
    best_c = cur_c
    best_h = h.copy()
    N1 = n - 1
    m_idx = np.arange(2 * n - 1)

    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        found = False
        z_order = rng.permutation(len(zeros))

            if time.time() > deadline:
                return best_h, best_c
            i = zeros[idx_z]

            top_k = min(K, len(ones))

            best_j = -1

                j = best_j

                h[j] = 0.0

                v1 = (p1 >= 0) & (p1 < n)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    p6 = dl - 5
    if time.time() < p6:
        hb, cb = binary_search_fast(best_h, n, p6, max_sweeps=50000, K=60, seed=seed + 100)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 7: Final SA polish
    if time.time() < dl - 2:
        hb, cb = sa_binary_search(best_h, n, dl, K=15, T0=0.01, T_end=0.0005, seed=seed + 200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
