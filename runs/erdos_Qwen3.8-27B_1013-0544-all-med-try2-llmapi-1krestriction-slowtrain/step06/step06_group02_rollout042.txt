def _smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    if sm > 500:
        return m + (sm + np.log(np.exp(sm - sm).sum() ... )) / alpha
    # Actually simpler:
    return m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha

def _w_from_alpha(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    c = compute_c5_all(h, n)
    w = _w_from_alpha(c, alpha)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grad = np.zeros(n)

        if s >= 0:

            else:

    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    t_k = 1.0

    alpha = alpha_start
    stall = 0
    
    while time.time() < deadline:
        c_y = compute_c5_all(y, n)
        f_y = _smooth_max(c_y, alpha)
        g_y = grad_smooth_max(y, n, alpha)

            c_t = compute_c5_all(x_trial, n)

            if f_t <= rhs + 1e-12:

                valid = True
                break
            L = L_trial * 1.5

        y = x_new + ((t_k - 1.0) / t_new) * (x_new - h)
        y = project_box_sum(y, target)
        h = x_new
        t_k = t_new

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            stall = 0
        else:

        alpha = min(alpha * 1.05, alpha_end)
        
        # If stalled, increase L

            L *= 1.3
            stall = 0
