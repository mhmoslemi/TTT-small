```
def best_swap_search(h, c, current_max, n):
    # h: current binary solution
    # c: current correlation vector (length 2n-1)
    # current_max: c.max()
    # Returns: best (i₀, j₀) swap or None
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_improvement = 0
    best_pair = None
    
    for i₀ in zeros:
        # Compute p(s) = j₀-independent part
        s_range = np.arange(-(n-1), n)  # all shifts
        s_valid_A = (i₀ + s_range >= 0) & (i₀ + s_range < n)
        s_valid_C = (i₀ - s_range >= 0) & (i₀ - s_range < n)
        
        p = np.zeros(2*n-1)
        p[s_valid_A] += 1 - h[i₀ + s_range[s_valid_A]]
        p[s_valid_C] += -h[i₀ - s_range[s_valid_C]]
        
        for j₀ in ones:
            # Compute q(s, j₀) = j₀-dependent part
            s_valid_B = (j₀ + s_range >= 0) & (j₀ + s_range < n)
            s_valid_D = (j₀ - s_range >= 0) & (j₀ - s_range < n)
            
            q = np.zeros(2*n-1)
            q[s_valid_B] -= (1 - h[j₀ + s_range[s_valid_B]])
            q[s_valid_D] += h[j₀ - s_range[s_valid_D]]
            
            delta = p + q
            
            # Apply corrections
            # s=0 corresponds to index n-1 in s_range
            delta[n-1] += -2  # for s=0
            
            # s = j₀-i₀: index = (j₀-i₀) + (n-1)
            idx_pos = (j₀ - i₀) + (n - 1)
            if 0 <= idx_pos < 2*n-1:
                delta[idx_pos] += -1
            
            # s = i₀-j₀: index = (i₀-j₀) + (n-1)
            idx_neg = (i₀ - j₀) + (n - 1)
            if 0 <= idx_neg < 2*n-1:
                delta[idx_neg] += 1
            
            # Check improvement
            new_c = c + delta
            new_max = new_c.max()
            if new_max < current_max - best_improvement:
                best_improvement = current_max - new_max
                best_pair = (i₀, j₀)
    
    return best_pair
```

This is O(n³) per full pass but with small constants. For n=250, one full pass is about 250 * 125 * 500 ≈ 15.6M numpy element operations. Should take ~1s.

```python
def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_delta(i0, j0, h, n, s_range):
    """Compute delta correlation vector for swap (i0: 0->1, j0: 1->0)"""
    delta = np.zeros(2*n-1)
    # A: i0 as first index
    valid_A = (i0 + s_range >= 0) & (i0 + s_range < n)
    delta[valid_A] += 1.0 - h[i0 + s_range[valid_A]]
    # B: j0 as first index
    valid_B = (j0 + s_range >= 0) & (j0 + s_range < n)
    delta[valid_B] -= 1.0 - h[j0 + s_range[valid_B]]
    # C: i0 as second index
    valid_C = (i0 - s_range >= 0) & (i0 - s_range < n)
    delta[valid_C] -= h[i0 - s_range[valid_C]]
    # D: j0 as second index
    valid_D = (j0 - s_range >= 0) & (j0 - s_range < n)
    delta[valid_D] += h[j0 - s_range[valid_D]]
    # Corrections for double counting
    # s=0 at index n-1
    delta[n-1] += -2.0
    # s = j0-i0 at index (j0-i0)+(n-1)

    if 0 <= idx < 2*n-1:

    # s = i0-j0 at index (i0-j0)+(n-1)
    idx = (i0 - j0) + (n - 1)
    if 0 <= idx < 2*n-1:
        delta[idx] += 1.0
    return delta

def best_swap_pass(h, n, s_range, max_passes=100, deadline=None):
    """Greedily find and apply best improving swap until no improvement."""
    c = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = c.max()
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 0
        best_pair = None
        best_delta = None
        
        for i0 in zeros:
            for j0 in ones:
                delta = compute_delta(i0, j0, h, n, s_range)
                new_max = (c + delta).max()
                improve = cur_max - new_max
                if improve > best_improve + 1e-15:
                    best_improve = improve
                    best_pair = (i0, j0)
                    best_delta = delta
        
        if best_pair is None:
            break
        
        # Apply the best swap
        i0, j0 = best_pair
        h[i0], h[j0] = 1.0, 0.0
        c = c + best_delta
        cur_max = c.max()
    
    return h, cur_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_delta(i0, j0, h, n, s_range):
    delta = np.zeros(2 * n - 1)
    vA = (i0 + s_range >= 0) & (i0 + s_range < n)
    delta[vA] += 1.0 - h[i0 + s_range[vA]]
    vB = (j0 + s_range >= 0) & (j0 + s_range < n)
    delta[vB] -= 1.0 - h[j0 + s_range[vB]]
    vC = (i0 - s_range >= 0) & (i0 - s_range < n)
    delta[vC] -= h[i0 - s_range[vC]]
    vD = (j0 - s_range >= 0) & (j0 - s_range < n)
    delta[vD] += h[j0 - s_range[vD]]
    idx_AD = (j0 - i0) + (n - 1)
    if 0 <= idx_AD < 2 * n - 1:
        delta[idx_AD] += 1.0
    delta[n - 1] += -2.0
    if 0 <= idx_AD < 2 * n - 1:
        delta[idx_AD] += -1.0
    idx_BC = (i0 - j0) + (n - 1)
    if 0 <= idx_BC < 2 * n - 1:
        delta[idx_BC] += 1.0
    return delta

def best_swap_local(h, n, s_range, deadline, max_iters=5000):
    c = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = c.max()
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_improve = 0.0
        best_pair = None
        best_delta = None
        for i0 in zeros:
            if time.time() > deadline:
                break
            for j0 in ones:
                d = make_delta(i0, j0, h, n, s_range)
                nm = (c + d).max()
                imp = cur_max - nm
                if imp > best_improve + 1e-15:
                    best_improve = imp
                    best_pair = (i0, j0)
                    best_delta = d
                    if best_improve > 10.0:
                        break
            if best_improve > 10.0:
                break
        if best_pair is None:
            break
        i0, j0 = best_pair
        h[i0], h[j0] = 1.0, 0.0
        c = c + best_delta
        cur_max = c.max()
    return h, cur_max

    best_h = h.copy()

    for r in range(n_restarts):

        nperts = rng.randint(1, 5)
        ones = np.where(cur == 1)[0]

            b = rng.choice(zeros)
            cur[a], cur[b] = 0.0, 1.0
        cur, cm = best_swap_local(cur, n, s_range, deadline, max_iters=3000)
        if cm < best_c:
            best_c = cm
            best_h = cur.copy()
```
