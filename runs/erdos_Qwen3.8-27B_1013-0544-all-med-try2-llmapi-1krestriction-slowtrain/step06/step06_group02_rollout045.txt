import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_desc(h, n, max_passes=80, deadline=None):
    """Coordinate descent exploiting linearity of each f_k in each h[i]."""
    dx = 2.0 / n
    f = np.correlate(h, 1 - h, mode='full') * dx
    best_c = f.max()
    k_arr = np.arange(2 * n - 1) - (n - 1)
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            slope = np.zeros(2 * n - 1)
            idx1 = i + k_arr
            v1 = (idx1 >= 0) & (idx1 < n)
            slope[v1] += (1.0 - h[idx1[v1]])
            idx2 = i - k_arr
            v2 = (idx2 >= 0) & (idx2 < n)
            slope[v2] -= h[idx2[v2]]
            a = f - h[i] * slope

def _smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mlag in range(2 * n - 1):
        if w[mlag] < 1e-14:
            continue
        k = mlag - (n - 1)
        wm = w[mlag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _slsqp_obj(x, n, alpha):
    return _smooth_max_obj(x, n, alpha)

def _slsqp_eq(x, n):
    return x.sum() - n / 2.0

def _slsqp_eq_jac(x, n):

def lbfgs_phase(h, n, deadline, max_iters=200):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

            h_new = project_feasible(res.x, n)

            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=100000, maxiter=1000):
    if time.time() > deadline - 2:
        return h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def quantize_and_search(h, n, deadline):
    """Push toward binary then do pair-swap local search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    # Quantize: threshold near 0.5, adjust to maintain sum
    rb = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(rb.sum())
    target = n // 2
    if n_ones > target:
        # Remove (n_ones - target) ones that are closest to 0.5
        ones_idx = np.where(rb == 1)[0]
        sorted_idx = ones_idx[np.argsort(h[ones_idx])]
        for idx in sorted_idx[:n_ones - target]:
            rb[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(rb == 0)[0]
        sorted_idx = zeros_idx[np.argsort(-h[zeros_idx])]
        for idx in sorted_idx[:target - n_ones]:
            rb[idx] = 1
    # Evaluate binary
    c_bin = compute_c5(rb, n)
    # Pair swap local search on binary
    dx = 2.0 / n
    f = np.correlate(rb, 1 - rb, mode='full') * dx
    best_c = max(best_c, c_bin)
    if c_bin < best_c:
        h = rb.copy()
        best_c = c_bin
    # Local search: swap a 1 with a 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    max_rounds = 200
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in range(len(ones)):
            if time.time() > deadline:
                break
            for zi in range(len(zeros)):
                i, j = ones[oi], zeros[zi]
                # Swap: h[i] 1->0, h[j] 0->1
                # Change in f[m]: for each lag m, delta = -[(1-h[i+k])-h[i-k]] + [(1-h[j+k])-h[j-k]]
                # Just recompute for speed with n=300
                h[i] = 0; h[j] = 1
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                    # Update ones/zeros lists
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
                else:
                    h[i] = 1; h[j] = 0
            if improved:
                break
        if not improved:
            break
    return h, best_c

def try_other_n(h_best, c_best, n_cur, deadline, rng):
    """Try interpolating to different n and optimizing."""
    for n_try in [200, 400, 150]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 30)
        if alloc < 20:
            break
        h_t, c_t = coord_desc(h_interp, n_try, max_passes=30,
                             deadline=time.time() + alloc * 0.5)
        if time.time() < deadline - 25:
            h_t, c_t = lbfgs_phase(h_t, n_try, deadline - 25, max_iters=100)
        if c_t < c_best:
            c_best = c_t
            h_best = h_t
            n_cur = n_try
    return h_best, c_best, n_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B
    t1 = min(t0 + 60, deadline - 400)
    h, best_c = lbfgs_phase(h, n, t1, max_iters=150)
    best_h = h.copy()

    # Phase 2: Coordinate descent (key new method)
    t2 = min(time.time() + 180, deadline - 250)
    h, c_cd = coord_desc(h, n, max_passes=60, deadline=t2)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h.copy()

    # Phase 3: L-BFGS-B refinement after CD
    t3 = min(time.time() + 50, deadline - 200)
    h, c_l = lbfgs_phase(best_h, n, t3, max_iters=100)
    if c_l < best_c:
        best_c = c_l
        best_h = h.copy()

    # Phase 4: More coordinate descent
    t4 = min(time.time() + 150, deadline - 150)
    h, c_cd2 = coord_desc(best_h, n, max_passes=60, deadline=t4)
    if c_cd2 < best_c:
        best_c = c_cd2
        best_h = h.copy()

    # Phase 5: SLSQP for final polish
    if time.time() < deadline - 60:
        h, c_s = slsqp_phase(best_h, n, deadline - 50, alpha=500000, maxiter=500)
        if c_s < best_c:
            best_c = c_s
            best_h = h.copy()

    # Phase 6: Quantize and binary local search
    if time.time() < deadline - 40:
        h, c_q = quantize_and_search(best_h, n, deadline - 30)
        if c_q < best_c:
            best_c = c_q
            best_h = h.copy()

    # Phase 7: Final coordinate descent
    if time.time() < deadline - 10:
        h, c_f = coord_desc(best_h, n, max_passes=40, deadline=deadline - 5)
        if c_f < best_c:
            best_c = c_f
            best_h = h.copy()

    # Phase 8: Try different n
    if time.time() < deadline - 30:
        best_h, best_c, n = try_other_n(best_h, best_c, n, deadline, rng)

    # Final polish
    if time.time() < deadline - 3:
        h, c_fin = coord_desc(best_h, n, max_passes=20, deadline=deadline - 1)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
