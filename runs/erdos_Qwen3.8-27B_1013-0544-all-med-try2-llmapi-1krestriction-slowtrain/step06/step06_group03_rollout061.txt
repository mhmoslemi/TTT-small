import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Compute smooth max value and gradient."""
    c = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * c - alpha * c.max()
    exps = np.exp(shifted)
    total = exps.sum()
    W = exps / total  # weights for each lag
    value = (c.max() + np.log(total) / alpha)
    
    dx = 2.0 / n

                    continue
                for j in dec_idx:
                    if i == j or h[j] - d < 0.0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return best_h, best_c

def full_pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if h[i] + d > 1.0:
                    continue
                for j in range(n):
                    if i == j or h[j] - d < 0.0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return best_h, best_c

def structured_init(n, rng, freq):
    x = np.linspace(0, 2, n)
    phase = rng.uniform(0, 2 * np.pi)
    A = rng.uniform(0.1, 0.4)
    h = 0.5 + A * np.sin(np.pi * freq * x + phase)
    h2 = 0.5 + A * 0.7 * np.cos(2 * np.pi * freq * x + phase * 1.3)
    h = 0.5 * h + 0.5 * h2
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    h_init = project_feasible(h_init, n_init)

    # Phase 1: PGD on initial
    alloc1 = min(250, (deadline - t0) * 0.30)
    h1, c1 = pgd(h_init, n_init, t0 + alloc1, alpha_start=50.0, alpha_end=100000.0, n_iters=4000)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Second PGD with different alpha schedule
    if time.time() < deadline - 100:
        alloc2 = min(200, (deadline - time.time()) * 0.25)
        h2, c2 = pgd(h1, n_init, time.time() + alloc2, alpha_start=200.0, alpha_end=500000.0, n_iters=3000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Gradient-informed local search
    if best_h is not None and time.time() < deadline - 80:
        alloc3 = min(200, (deadline - time.time()) * 0.25)
        best_h, best_c = gradient_pair_search(best_h, len(best_h), time.time() + alloc3, top_k=25)

    # Phase 4: Multi-restart with diverse initializations
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        n_restarts = 4
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                h_start = structured_init(n, rng, rng.choice([1, 2, 3]))
            elif r == 1:
                h_start = 1.0 - best_h
                h_start = project_feasible(h_start, n)
            elif r == 2:
                h_start = best_h + rng.normal(0, 0.05, n)
                h_start = project_feasible(h_start, n)
            else:
                h_start = structured_init(n, rng, rng.choice([2, 3, 4, 5]))
            alloc_r = min(120, (deadline - time.time() - 20) * 0.3)
            if alloc_r < 20:
                break
            hr, cr = pgd(h_start, n, time.time() + alloc_r, alpha_start=100.0, alpha_end=200000.0, n_iters=2000)
            if time.time() < deadline - 15:
                hr, cr = gradient_pair_search(hr, n, time.time() + min(40, deadline - time.time() - 15), top_k=20)
            if cr < best_c:
                best_c, best_h = cr, hr

    # Phase 5: Try coarser grid n=100
    if best_h is not None and time.time() < deadline - 60:
        n_coarse = 100
        x_fine = np.linspace(0, 2, len(best_h))
        x_coarse = np.linspace(0, 2, n_coarse)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        h_coarse = project_feasible(h_coarse, n_coarse)
        alloc5 = min(150, (deadline - time.time()) * 0.3)
        if alloc5 > 30:
            h5, c5 = pgd(h_coarse, n_coarse, time.time() + alloc5 * 0.6, alpha_start=50.0, alpha_end=200000.0, n_iters=4000)
            if time.time() < deadline - 10:
                h5, c5 = full_pair_search(h5, n_coarse, time.time() + min(60, deadline - time.time() - 5),
                                          [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            if c5 < best_c:
                best_c, best_h = c5, h5

    # Phase 6: Try n=200
    if best_h is not None and time.time() < deadline - 50:
        n_mid = 200
        x_cur = np.linspace(0, 2, len(best_h))
        x_mid = np.linspace(0, 2, n_mid)
        h_mid = np.interp(x_mid, x_cur, best_h)
        h_mid = project_feasible(h_mid, n_mid)
        alloc6 = min(100, (deadline - time.time()) * 0.25)
        if alloc6 > 25:
            h6, c6 = pgd(h_mid, n_mid, time.time() + alloc6 * 0.6, alpha_start=100.0, alpha_end=300000.0, n_iters=2500)
            if time.time() < deadline - 8:
                h6, c6 = gradient_pair_search(h6, n_mid, time.time() + min(50, deadline - time.time() - 5), top_k=20)
            if c6 < best_c:
                best_c, best_h = c6, h6

    # Phase 7: Final gradient-informed polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        best_h, best_c = gradient_pair_search(best_h, n, deadline - 3, top_k=40,
                                              deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    # Phase 8: Final PGD polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        hpol, cpol = pgd(best_h, n, deadline - 2, alpha_start=5000.0, alpha_end=1000000.0, n_iters=1000)
        if cpol < best_c:
            best_c, best_h = cpol, hpol

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
