import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    s = s - s_max
    return float(m + (s_max + np.log(np.sum(np.exp(s)))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    return w

def lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        grad[-m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[-m:]
    return grad

def grad_smooth_max(h, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    if top_idx is not None:
        c_sub = c[top_idx]
        w = smooth_max_weights(c_sub, alpha)
        idx_list = top_idx
        w_list = w
    else:
        c_sub = c
        w = smooth_max_weights(c, alpha)
        idx_list = np.arange(len(c))
        w_list = w
    grad = np.zeros(n)
    for i in range(len(idx_list)):
        if w_list[i] < 1e-10:
            continue
        lag = idx_list[i] - (n - 1)
        grad += w_list[i] * lag_grad(h, n, lag)
    return dx * grad

def nesterov_descent(h, n, alpha, deadline, n_steps=2000, momentum=0.9):
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    step = 1.0 / (alpha * 3.0)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        y = h_curr + momentum * v
        y = project_feasible(y, n)
        g = grad_smooth_max(y, n, alpha)
        v = momentum * v - step * g
        h_new = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def lp_relaxation(h, n):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = lag_grad(h, n, lag)
        cons.append(c[k] + dx * (g @ (h_var - h)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h.copy()
    h_new = h_var.value
    if h_new is None:
        return h.copy()
    return project_feasible(h_new, n)

def slsqp_polish(h, n, top_m=5, alpha=50000, maxiter=5000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    dx = 2.0 / n

    def obj(x):
        cc = compute_all_lags(x, n)
        return smooth_max_val(cc[top_idx], alpha)

    def obj_grad(x):
        cc = compute_all_lags(x, n)
        w = smooth_max_weights(cc[top_idx], alpha)
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-10:
                continue
            lag = top_idx[i] - (n - 1)
            grad += w[i] * lag_grad(x, n, lag)
        return dx * grad

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_exchange(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(best_h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if lag > 0:
                contribs = best_h[:n - m] * (1 - best_h[m:])
            else:
                contribs = best_h[-m:] * (1 - best_h[:n - m])
            i_loc = int(np.argmax(contribs))
            if lag > 0:
                i, j = i_loc, i_loc + m
            else:
                i, j = n - m + i_loc, i_loc
            if best_h[i] > best_h[j]:
                delta = min(d, 0.4 * (best_h[i] - best_h[j]), best_h[i], 1 - best_h[j])
            else:
                delta = min(d, 0.4 * (best_h[j] - best_h[i]), best_h[j], 1 - best_h[i])
            if delta < 1e-9:
                continue
            if best_h[i] > best_h[j]:
                best_h[i] -= delta
                best_h[j] += delta
            else:
                best_h[i] += delta
                best_h[j] -= delta
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
            else:
                if best_h[i] < best_h[j]:
                    best_h[i] += delta
                    best_h[j] -= delta
                else:
                    best_h[i] -= delta
                    best_h[j] += delta
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def optimize_sequence(h, n, deadline):
    best_h = h.copy()
