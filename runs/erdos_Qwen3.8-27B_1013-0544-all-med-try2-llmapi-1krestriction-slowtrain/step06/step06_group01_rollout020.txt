```python
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    x = c * dx * alpha

    log_sum = x_max / alpha + np.log(exp_x.sum()) / alpha

        s2 = 0.0

            i2 = j - k + n - 1
            wk = w[k]

                continue

            if 0 <= i2 < n:
                s2 += wk * h[i2]

        integral = h.sum() * dx
        val += penalty * (integral - 1.0)**2
        grad += 2 * penalty * (integral - 1.0) * dx

        return v

        return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def precompute(n):
    j = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :]
    idx_A = j + k - n + 1
    idx_B = j - k + n - 1
    mask_A = (idx_A >= 0) & (idx_A < n)
    mask_B = (idx_B >= 0) & (idx_B < n)
    idx_A = np.where(mask_A, idx_A, 0)
    idx_B = np.where(mask_B, idx_B, 0)
    return idx_A, mask_A, idx_B, mask_B

def smooth_max_val_grad(h, alpha, precomp, penalty=0.0):
    n = len(h)
    dx = 2.0 / n
    idx_A, mask_A, idx_B, mask_B = precomp
    c = np.correlate(h, 1 - h, mode='full')
    x = c * dx * alpha
    xm = x.max()
    ex = np.exp(x - xm)
    val = (xm + np.log(ex.sum())) / alpha
    p = ex / ex.sum()
    A = (h[idx_A] * mask_A) @ p
    B = (h[idx_B] * mask_B) @ p
    grad = dx * (1.0 - A - B)
    if penalty > 0:
        integral = h.sum() * dx
        val += penalty * (integral - 1.0) ** 2
        grad += 2.0 * penalty * (integral - 1.0) * dx
    return val, grad

def lbfgs_step(h, n, precomp, alpha, penalty, maxiter, deadline):
    def fun(x):
        v, _ = smooth_max_val_grad(x, alpha, precomp, penalty)
        return v
    def jac(x):
        _, g = smooth_max_val_grad(x, alpha, precomp, penalty)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_grad_descent(h, n, precomp, alpha, deadline, max_iter=2000):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        c5 = c.max() * dx
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        _, grad = smooth_max_val_grad(h, alpha, precomp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c5 - 1e-14:
            h = h_new
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.75
            if lr < 1e-10:
                break
    return best_h, best_c

def alternating_min(h_init, n, precomp, deadline, max_rounds=50):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        k_star = int(np.argmax(c))
        for inner in range(200):
            if time.time() > deadline:
                break
            _, grad = smooth_max_val_grad(h, 1.0, precomp)
            # Focus gradient on the active k
            dx = 2.0 / n
            # Compute gradient of just f_{k_star}
            g_active = np.zeros(n)
            for j in range(n):
                i1 = j + k_star - n + 1
                i2 = j - k_star + n - 1
                v1 = (1.0 - h[i1]) if 0 <= i1 < n else 1.0
                v2 = h[i2] if 0 <= i2 < n else 0.0
                g_active[j] = dx * (v1 - v2)
            # Projected gradient step
            lr = 0.5 / (np.linalg.norm(g_active) + 1e-10)
            h_new = h - lr * g_active
            h_new = project_feasible(h_new, n)
            c_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if c_new < c.max() - 1e-14:
                h = h_new
            else:
                break
        c5 = compute_c5(h, n)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    precomp = precompute(n)

    # Phase 1: L-BFGS-B with increasing alpha (35% of time)
    p1 = t0 + 0.35 * (budget_s - 5)
    h = best_h.copy()
    alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    penalties = [100, 50, 20, 10, 5, 1, 0.5, 0.1, 0.0]
    for i, (alpha, pen) in enumerate(zip(alphas, penalties)):
        if time.time() > p1:
            break
        maxit = max(50, int(200 * (p1 - time.time()) / (budget_s * 0.35)))
        h = lbfgs_step(h, n, precomp, alpha, pen, maxit, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Projected gradient with adaptive alpha (20% of time)
    p2 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p2:
        for alpha in [200, 500, 1000, 2000]:
            if time.time() > p2:
                break
            h_pg, c_pg = projected_grad_descent(best_h, n, precomp, alpha, p2, max_iter=500)
            if c_pg < best_c:
                best_c = c_pg
                best_h = h_pg.copy()

    # Phase 3: Alternating minimization (15% of time)
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3:
        h_alt, c_alt = alternating_min(best_h, n, precomp, p3, max_rounds=100)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 4: L-BFGS-B polish from best (15% of time)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4:
        h = best_h + 0.01 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha, pen in [(5000, 0.1), (10000, 0.05), (20000, 0.01), (50000, 0.0)]:
            if time.time() > p4:
                break
            h = lbfgs_step(h, n, precomp, alpha, pen, 300, p4)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 5: Random restarts + quick optimization (15% of time)
    p5 = deadline
    if time.time() < p5 - 10:
        for restart in range(8):
            if time.time() > p5 - 5:
                break
            h_r = best_h + 0.05 * rng.randn(n)
            h_r = project_feasible(h_r, n)
            h_r = lbfgs_step(h_r, n, precomp, 5000, 0.1, 100, p5 - 5)
            c_r = compute_c5(h_r, n)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
            h_r2 = lbfgs_step(h_r, n, precomp, 20000, 0.0, 200, p5 - 5)
            c_r2 = compute_c5(h_r2, n)
            if c_r2 < best_c:
                best_c = c_r2
                best_h = h_r2.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
