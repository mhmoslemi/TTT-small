import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Gradient of smooth max approximation of C5."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _obj_pen(h, n, alpha, mu):
    val = _obj(h, n, alpha)
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

def _obj_pen_grad(h, n, alpha, mu):
    g = grad_smooth_max(h, n, alpha)
    s = h.sum() - n / 2.0
    g += 2.0 * mu * s * np.ones(n)
    return g

def lbfgs_opt(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [10, 50, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    mus = [1e-3, 0.01, 0.1, 1.0, 10.0, 100.0, 1000.0, 10000.0, 100000.0, 1000000.0]
    for a, mu in zip(alphas, mus):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_obj_pen, h, jac=_obj_pen_grad, args=(n, a, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 5, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return float(x.sum() - n / 2.0)

        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,

def nesterov_descent(h, n, deadline, alpha_init=500.0, lr_init=0.005):
    """Nesterov-accelerated gradient descent with backtracking."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)  # momentum
    lr = lr_init
    alpha = alpha_init
    beta = 0.9  # momentum coefficient
    for it in range(50000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        v = beta * v + (1 - beta) * g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(30):
            h_try = project_feasible(h - step * v, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
            alpha *= 2.0
    return best_h, best_c
