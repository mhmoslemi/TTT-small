import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def _obj_active(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c[top_idx], alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lag_grad(h, n, lag):
    """Gradient of c_lag = dx * sum h_i(1-h_{i+lag}) w.r.t. h"""
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1 - 2*h)
    grad = np.zeros(n)
    if lag > 0:
        # c = dx * sum_{i=0}^{n-1-m} h_i (1 - h_{i+m})
        # d/dh_i (for i < n-m): dx * (1 - h_{i+m})
        # d/dh_{i+m} (for i < n-m): -dx * h_i
        grad[:n-m] += dx * (1 - h[m:])
        grad[m:] -= dx * h[:n-m]
    else:
        # c = dx * sum_{i=-m}^{n-1} h_i (1 - h_{i+m}) where m=|lag|
        # = dx * sum_{i=0}^{n-1-m} h_{i+m} (1 - h_i)
        # d/dh_i (for i < n-m): -dx * h_{i+m}  (from 1-h_i term)
        # d/dh_{i+m} (for i < n-m): dx * (1 - h_i)
        grad[:n-m] -= dx * h[m:]
        grad[m:] += dx * (1 - h[:n-m])
    return grad

def gradient_step(h, n, top_m, step_size, deadline=None):
    """Take a projected gradient step on the worst top_m lags."""
    t0 = time.time()
    if deadline and t0 > deadline:
        return h.copy()
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    # Combined gradient
    nrm = len(top_idx)
    grad = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        grad += -_lag_grad(h, n, lag)  # negative because we want to minimize
    grad /= nrm
    # Project gradient: set to 0 where h is at bound
    grad[h <= 1e-10] = np.maximum(grad[h <= 1e-10], 0)
    grad[h >= 1 - 1e-10] = np.minimum(grad[h >= 1 - 1e-10], 0)
    # Make sum-preserving: subtract mean
    grad -= grad.mean()
    # Line search
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for lr in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
        h_new = h + lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
            break
    return best_h

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, alpha, top_idx, maxiter=3000, ftol=1e-14):

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_active, h, args=(n, alpha, top_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

def outer_lp_step(h_curr, n, trust):
    """One step of outer approximation: LP with linearized constraints."""
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, 2) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        # Linearization: c(k) + grad . (h - h_curr) <= t
        linearized_val = c[k] + float(grad @ (h_var - h_curr))
        cons.append(linearized_val <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)
