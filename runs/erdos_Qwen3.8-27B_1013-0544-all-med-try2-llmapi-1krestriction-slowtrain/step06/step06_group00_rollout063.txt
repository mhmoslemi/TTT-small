    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sh_max = shifted.max()
        logsum = sh_max + np.log(np.exp(shifted - sh_max).sum())
        return m + logsum / alpha
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()

        sh_max = shifted.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def make_delta(h, i, j, n, m_range):
    """Correct delta for correlation when h[i]:0->1, h[j]:1->0"""
    delta = np.zeros(2 * n - 1)
    # Term A: (1-h[i-k]) for valid i-k
    ik = i - m_range
    mask = (ik >= 0) & (ik < n)
    delta[mask] += 1.0 - h[ik[mask]]
    # Term B: -h[i+k] for valid i+k
    ikp = i + m_range
    mask = (ikp >= 0) & (ikp < n)
    delta[mask] -= h[ikp[mask]]
    # Term C: -(1-h[j-k]) for valid j-k
    jk = j - m_range
    mask = (jk >= 0) & (jk < n)
    delta[mask] -= 1.0 - h[jk[mask]]
    # Term D: +h[j+k] for valid j+k
    jkp = j + m_range
    mask = (jkp >= 0) & (jkp < n)
    delta[mask] += h[jkp[mask]]
    # Cross terms
    delta[n - 1] -= 2.0
    kp = n - 1 + (j - i)
    if 0 <= kp < 2 * n - 1:
        delta[kp] += 1.0
    km = n - 1 + (i - j)
    if 0 <= km < 2 * n - 1:
        delta[km] += 1.0
    return delta

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sh_max = shifted.max()
        logsum = sh_max + np.log(np.exp(shifted - sh_max).sum())
        return m + logsum / alpha
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sh_max = shifted.max()
        p = np.exp(shifted - sh_max)
        p /= p.sum()
        # grad[m] = dx * sum_k p[k] * [(1-x[m-k])*1[valid] - x[m+k]*1[valid]]
        grad = np.zeros(n)
        # For efficiency, only compute for significant p values
        thresh = 1e-8
        for k_idx in range(len(m_range)):
            if p[k_idx] < thresh:
                continue
            k = m_range[k_idx]
            # (1-x[m-k]) where 0<=m-k<n → m in [k, k+n] ∩ [0,n]
            lo = max(0, k)
            hi = min(n, k + n)
            if lo < hi:
                grad[lo:hi] += p[k_idx] * (1.0 - x[lo - k:hi - k]) * dx
            # -x[m+k] where 0<=m+k<n → m in [-k, n-k] ∩ [0,n]
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if lo2 < hi2:
                grad[lo2:hi2] -= p[k_idx] * x[lo2 + k:hi2 + k] * dx
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

Similarly, the second part: Σ_k p[k]·h[m+k]·1[0≤m+k<n]
Let j = m+k, so k = j-m:
= Σ_j h[j]·p[j-m]·1[0≤j<n]·1[j-m is valid lag]
= h ⊛ p (cross-correlation)

So grad = dx * [correl(1-h, p_ext) - correl(h, p_ext)] (with appropriate indexing)

This is O(n log n) via FFT! Much faster than O(n²).

Let me implement this:

```python
def compute_gradient_fft(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sh_max = shifted.max()
    p = np.exp(shifted - sh_max)
    p /= p.sum()
    
    # p has length 2n-1, indexed by k + (n-1) for lag k
    # We need: grad[m] = dx * [sum_k p[k]·(1-h[m-k]) - sum_k p[k]·h[m+k]]
    
    # First term: sum_k p[k]·(1-h[m-k]) where 0≤m-k<n
    # = sum_j (1-h[j])·p[m-j] where 0≤j<n and m-j is valid lag
    # This is cross-correlation of (1-h) with p (in the sense of summing over j)
    
    # Using FFT:
    # correlation(a, b)[m] = sum_j a[m-j]·b[j] (with appropriate bounds)
    
    # Actually, let me use np.correlate directly:
    # np.correlate(1-h, p, mode='full') gives sum_j (1-h)[j]·p[j+k] for various k
    # That's not quite what I want.
    
    # Let me think differently. I want:
    # A[m] = sum_k p[k]·(1-h[m-k])·1[0≤m-k<n]
    # where k ranges over -(n-1) to n-1
    
    # Let's define p_full as an array of length 2n-1 where p_full[k+n-1] = p[k]
    # Then A[m] = sum_k p_full[k+n-1]·(1-h[m-k])·1[0≤m-k<n]
    # = sum_j (1-h[j])·p_full[m-j+n-1]·1[0≤j<n]
    # (where j = m-k, so k = m-j, and p is at index k+n-1 = m-j+n-1)
    
    # This is the cross-correlation: (1-h) ⊛ p_full
    # = np.correlate(p_full, 1-h, mode='full') at appropriate index
    # or np.correlate(1-h, p_full, mode='full')
    
    # Actually, np.correlate(a, v, 'full')[m + len(v) - 1] = sum_n a[n]·v[n+m]
    # Hmm, this is getting confusing. Let me just use the direct O(n²) loop but with
    # the optimization of skipping small p values.
    
    # For now, use the loop:
    m_range = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        if p[k_idx] < 1e-12:
            continue
        k = m_range[k_idx]
        # (1-h[m-k]) for valid m: 0≤m-k<n → m in [k, k+n]∩[0,n]
        lo = max(0, k)
        hi = min(n, k + n)
        if lo < hi:
            grad[lo:hi] += p[k_idx] * (1.0 - h[lo-k:hi-k]) * dx
        # -h[m+k] for valid m: 0≤m+k<n → m in [-k, n-k]∩[0,n]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= p[k_idx] * h[lo2+k:hi2+k] * dx
    return grad
```

For n=400, this is at most 799 iterations of the loop, each doing O(n) vector work. Total: O(n²) = 160,000 vector element operations per gradient. In numpy, this is fast (microseconds per vector op).

But the Python loop overhead (799 iterations) might add up. Let me estimate: 799 iterations × ~1μs overhead = ~0.8ms per gradient. With 5000 iterations, that's 4 seconds. That's fine!

Actually, for small alpha, most p values are significant, so we do all 799 iterations. For large alpha, only a few. So the average is much less than 799.

Let me now write the complete solution. I'll use:
1. Binary annealing (corrected delta) - Phase 1
2. Projected gradient with log-sum-exp - Phase 2
3. Multi-restart - Phase 3

One more optimization: for the projected gradient, I can use the fact that the objective is convex (log-sum-exp of affine functions is convex) to use a fixed step size based on the Lipschitz constant.

The Lipschitz constant of the gradient of the log-sum-exp objective is bounded by:
L ≤ α · ||A||² where A is the "matrix" defining the quadratics c[k](h).

This is hard to bound tightly, so I'll use backtracking line search.

Actually, for the projected gradient method, I can use the FISTA (accelerated) variant for faster convergence. But let's keep it simple first.

Let me also think about the step size. The objective is:
f(h) = max(c) + (1/α)·log(Σ exp(α(c[k]-max(c))))

The gradient is Lipschitz continuous with constant L. For the log-sum-exp of linear functions, L = α·λ_max where λ_max is the largest eigenvalue of the "Gram matrix" of the linear functions.

For our problem, each c[k] is a quadratic in h (not linear!), so this doesn't directly apply. The log-sum-exp of quadratics is not convex in general...

Wait, is our objective actually convex? Let's see:
c[k] = Σ_n h[n+k]·(1-h[n]) = Σ_n h[n+k] - Σ_n h[n+k]·h[n]

The first term is linear in h. The second term is a quadratic form (specifically, it's h^T Q_k h for some matrix Q_k, but actually it's a sum of products h[i]h[j] which is not a standard quadratic form).

So c[k] is a quadratic function of h (with cross terms), and the log-sum-exp of quadratics is not necessarily convex.

This means projected gradient might not converge in general. But in practice, for the range of h values we're working with (0 to 1), and for the specific structure of these quadratics, it might work well enough.

Alternatively, I can use a penalty method or a sequential quadratic programming approach.

Let me just go with the projected gradient and see how it works. If it gets stuck, the SLSQP (which handles non-convexity better) can be used as a backup.
