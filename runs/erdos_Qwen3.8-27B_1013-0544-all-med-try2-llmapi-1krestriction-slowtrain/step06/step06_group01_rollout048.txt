One optimization: we only need to check if max(corr + delta) < cur_c. We can compute an upper bound:
max(corr + delta) <= corr.max() + delta.max()

If this upper bound >= cur_c, we can skip the pair without computing the full delta. But this bound might be too loose.

A better approach: identify the "critical" shifts where corr[k] is close to the maximum. Only check if the delta at these critical shifts is negative enough.

Let me implement this:

```python
def binary_search_fast(h_init, n, deadline, max_sweeps=2000, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        o_list = list(ones)
        rng.shuffle(z_list)
        rng.shuffle(o_list)
        
        for i in z_list:
            if time.time() > deadline:
                break
            for j in o_list:
                # Quick rejection: compute delta and check
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                k2 = n - 1 + (i - j)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    o_list = list(ones)
                    z_list = list(zeros)
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
```

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-10:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-10:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    M = c.max()
    shifted = alpha * (c - M)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p = p / p.sum()
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        w = p[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:n])
            grad[k:] -= w * h[:n-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:n-kp])
            grad[:n-kp] -= w * h[kp:n]
    return grad * dx

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        M = c.max()
        shifted = alpha * (c - M)
        sm = shifted.max()
        return M + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def grad(x):
        return compute_gradient(x, n, alpha)
    bounds = [(0.0, 1.0)] * n

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
