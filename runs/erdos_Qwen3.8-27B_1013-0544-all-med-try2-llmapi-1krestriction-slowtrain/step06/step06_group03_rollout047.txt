import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent_sweep(h, n, max_sweeps=100, deadline=None):
    """Coordinate descent with exact per-coordinate optimization."""
    corr = compute_corr(h, n)
    best_c = float(np.max(corr))
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        improved = False
        for i in range(n):
            old_val = h[i]

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=200, deadline=None, order=None):
    corr = compute_corr(h, n)
    best_c = float(np.max(corr))
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        if order is not None:
            indices = order
        else:
            indices = range(n)
        improved = False
        for i in indices:
            old_val = h[i]
            k_lo = i
            k_hi = min(2 * n - 1, i + n)
            if k_hi <= k_lo:
                continue
            length = k_hi - k_lo
            offsets = np.arange(length)
            slopes = 1.0 - 2.0 * h[offsets]
            base_slice = corr[k_lo:k_hi]
            M_out = 0.0
            if k_lo > 0:
                M_out = max(M_out, float(np.max(corr[:k_lo])))
            if k_hi < 2 * n - 1:
                M_out = max(M_out, float(np.max(corr[k_hi:])))
            intercepts = base_slice - old_val * slopes
            lo_t, hi_t = 0.0, 1.0
            for _ in range(45):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = max(M_out, float(np.max(intercepts + m1 * slopes)))

                break
    return h, best_c

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Multiple rounds of coordinate descent with different orderings
    remaining = deadline - time.time()
    n_rounds = max(1, int(remaining / 15))
    for r in range(n_rounds):
        if time.time() > deadline - 5:
            break
        if r % 3 == 0:
            order = list(range(n))
        elif r % 3 == 1:
            order = list(range(n - 1, -1, -1))
        else:
            order = list(rng.permutation(n))
        h, c = coord_descent(h, n, max_sweeps=150, deadline=deadline - 5, order=order)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize initial at its native resolution
    h, c = full_optimize(h_init, n_init, t0 + 80, rng)
    h = project_feasible(h, n_init)
    c = compute_c5(h, n_init)
    if c < best_c:
        best_c, best_h = c, h

    # Phase 2: Coarse-to-fine multi-resolution
    x_best = np.linspace(0, 2, len(best_h))
    resolutions = [50, 80, 100, 150, 200, 300]
    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, (deadline - time.time()) * 0.15)
        if alloc < 20:
            break
        h, c = full_optimize(h_interp, n_try, time.time() + alloc, rng)
        h = project_feasible(h, n_try)
        c = compute_c5(h, n_try)
        if c < best_c:
            best_c, best_h = c, h
            x_best = np.linspace(0, 2, n_try)

    # Phase 3: More coordinate descent on best
    n_best = len(best_h)
    h, c = full_optimize(best_h, n_best, deadline - 60, rng)
    h = project_feasible(h, n_best)
    c = compute_c5(h, n_best)
    if c < best_c:
        best_c, best_h = c, h

    # Phase 4: Multi-restart with perturbation
    remaining = deadline - time.time() - 40
    if remaining > 20:
        n_restarts = max(2, int(remaining / 25))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 + 0.08 * (r + 1) / (n_restarts + 1)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(20, (deadline - time.time() - 20) * 0.4)
            if alloc < 10:
                break
            h, c = full_optimize(hp, n_best, time.time() + alloc, rng)
            h = project_feasible(h, n_best)
            c = compute_c5(h, n_best)
            if c < best_c:
                best_c, best_h = c, h

    # Phase 5: Local refinement
    if time.time() < deadline - 8:
        n_best = len(best_h)
        if n_best <= 120:
            best_h, best_c = pair_refine(best_h, n_best, deadline - 2,
                                         [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            best_h, best_c = near_pair_refine(best_h, n_best, deadline - 2,
                                              [0.02, 0.01, 0.005, 0.002, 0.001], radius=50)

    # Phase 6: Final coordinate descent polish
    if time.time() < deadline - 2:
        n_best = len(best_h)
        h, c = coord_descent(best_h, n_best, max_sweeps=100, deadline=deadline - 1)
        h = project_feasible(h, n_best)
        c = compute_c5(h, n_best)
        if c < best_c:
            best_c, best_h = c, h

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
