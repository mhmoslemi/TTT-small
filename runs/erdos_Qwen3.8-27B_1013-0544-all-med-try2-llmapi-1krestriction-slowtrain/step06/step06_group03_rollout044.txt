def pgd_step(h, n, alpha_smooth, deadline):
    """One step of projected gradient descent."""
    dx = 2.0 / n
    # Compute gradient of smooth objective
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha_smooth * (c - m)
    mx = sh.max()
    weights = np.exp(sh - mx) / np.sum(np.exp(sh - mx))  # softmax weights
    
    # Gradient: d/dh[i] of sum_k weights[k] * c[k]
    # c[k] = sum_i h[i] * (1-h)[i+k_offset] * dx (approximately)
    # This is complex; let me use finite differences for the gradient
    
    eps = 1e-5
    grad = np.zeros(n)
    for i in range(n):
        if time.time() > deadline:
            break
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + eps)
        # Renormalize
        h_plus = project_feasible(h_plus, n)
        c_plus = compute_c5(h_plus, n)
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - eps)
        h_minus = project_feasible(h_minus, n)
        c_minus = compute_c5(h_minus, n)
        grad[i] = (c_plus - c_minus) / (2 * eps)
    
    # Projected gradient step
    step = 0.01
    for _ in range(20):
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new, step
        step *= 0.5

    corr = np.correlate(h, 1.0 - h, mode='full')

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            if room.sum() > 1e-15:
                h += (diff / room.sum()) * room
        else:
            if h.sum() > 1e-15:
                h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _con_fun(x):
    return float(x.sum() - len(x) / 2.0)

def _con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)

    """Numerical gradient of smooth objective."""
    grad = np.zeros(n)
    base = smooth_obj(h, n, alpha)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, h[i] + eps)
        hp = project_feasible(hp, n)
        hm = h.copy()
        hm[i] = max(0.0, h[i] - eps)
        hm = project_feasible(hm, n)
        if hp[i] != hm[i]:
            grad[i] = (smooth_obj(hp, n, alpha) - smooth_obj(hm, n, alpha)) / (hp[i] - hm[i])
        else:
            grad[i] = 0.0
    return grad

def pgd_optimize(h, n, deadline, alpha_start=5000, alpha_end=200000, n_steps=2000):
    """Projected gradient descent with adaptive parameters."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    alpha = alpha_start
    
    for s in range(n_steps):
        if time.time() > deadline:
            break
        # Interpolate alpha
        frac = s / max(1, n_steps - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Compute gradient
        grad = grad_smooth(h, n, alpha, eps=1e-4)
        
        # Projected gradient step with backtracking
        c_curr = compute_c5(h, n)
        for _ in range(30):
            h_try = h - step * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_curr - 1e-15:
                h = h_try
                c_curr = c_try
                if c_curr < best_c:
                    best_c = c_curr
                    best_h = h.copy()
                break
            step *= 0.5
        else:
            # No improvement, try smaller step or reset
            step *= 0.5
            if step < 1e-8:
                # Try a random perturbation
                pert = np.random.randn(n) * 0.01
                h_try = project_feasible(h + pert, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    h = h_try
                    best_c = c_try
                    best_h = h.copy()
                step = 0.01
        
        # Adapt step size
        if c_curr < best_c:
            step = min(step * 1.2, 0.1)
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        old_i, old_j = h[i], h[j]
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i], h[j] = old_i, old_j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        old_i, old_j = h[i], h[j]
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i], h[j] = old_i, old_j
            if not improved:
                break
    return h, best_c
