import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c

def c5_val(c, n):
    return float(np.max(c) * (2.0/n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return h
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15: break
            h = h + (diff/rs)*room
        else:
            hs = h.sum()
            if hs < 1e-15: break
            h = h + (diff/hs)*h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    t = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:t]] = 1.0
    return b

def make_binary_from_continuous(hc, n):
    return to_binary(hc, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c


def c5_val(corr, n):
    return float(np.max(corr) * (2.0 / n))


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if h.sum() < 1e-15:
        return h
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h = h + (diff / rs) * room
        else:
            hs = h.sum()
            if hs < 1e-15:
                break
            h = h + (diff / hs) * h
    return np.clip(h, 0.0, 1.0)


def to_binary(h, n):
    t = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:t]] = 1.0
    return b


def best_improve_once(h, corr, n, deadline):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, corr, False
    c = 2 * n - 1
    kk = np.arange(c)
    ii = zeros[:, None]
    d1 = ii - kk[None, :]
    m1 = (d1 >= 0) & (d1 < n)
    term1 = np.where(m1, (1.0 - h)[np.clip(d1, 0, n - 1)], 0.0)
    d2 = ii + kk[None, :]
    m2 = (d2 < n)
    term2 = np.where(m2, h[np.clip(d2, 0, n - 1)], 0.0)
    rows = np.arange(len(zeros))
    cur_max = float(corr.max())
    best_val = cur_max
    best_i = -1
    best_j = -1
    best_new = None
    for jb in ones:
        jb = int(jb)
        d1b = jb - kk
        b1 = np.where((d1b >= 0) & (d1b < n), (1.0 - h)[np.clip(d1b, 0, n - 1)], 0.0)
        d2b = jb + kk
        b2 = np.where(d2b < n, h[d2b], 0.0)
        t3 = 2.0 * np.zeros_like(term1)
        t3[rows, 0] = 2.0
        kim = zeros - jb
        kjm = jb - zeros
        va = (kim >= 0) & (kim < c)
        t3[rows[va], kim[va]] -= 1.0
        vb = (kjm >= 0) & (kjm < c)
        t3[rows[vb], kjm[vb]] -= 1.0
        new = corr[None, :] + (term1 - b1[None, :]) - (term2 - b2[None, :]) - t3
        pm = new.max(axis=1)
        bw = int(np.argmin(pm))
        bv = float(pm[bw])
        if bv < best_val - 1e-14:
            best_val = bv
            best_i = int(zeros[bw])
            best_j = jb
            best_new = new[bw]
        if time.time() > deadline:
            break
    if best_i >= 0:
        h = h.copy()
        h[best_i] = 1.0
        h[best_j] = 0.0
        return h, best_new, True
    return h, corr, False


def best_improve_to_converge(h, corr, n, deadline, max_passes=50000):
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, corr, imp = best_improve_once(h, corr, n, deadline)
        if not imp:
            break
    return h, corr


def apply_kick(h, rng, k):
    h = h.copy()
    for _ in range(int(k)):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = int(rng.choice(ones))
        b = int(rng.choice(zeros))
        h[a], h[b] = 0.0, 1.0
    return h


def ils(h, n, deadline, seed=42, kick_sizes=(2, 3, 4, 5, 6)):

    cur = to_binary(h, n)

    best_c = c5_val(best_corr, n)
    ks = list(kick_sizes)
    while time.time() < deadline:

            best_corr = corr.copy()
