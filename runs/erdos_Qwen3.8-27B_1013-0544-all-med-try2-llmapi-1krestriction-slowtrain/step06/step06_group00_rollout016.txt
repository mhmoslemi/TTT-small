import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def make_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    mk = (im >= 0) & (im < n)
    delta[mk] += 1.0 - h[im[mk]]
    imn = i - m_range
    mk = (imn >= 0) & (imn < n)
    delta[mk] -= h[imn[mk]]
    jm = j + m_range
    mk = (jm >= 0) & (jm < n)
    delta[mk] -= 1.0 - h[jm[mk]]
    jmn = j - m_range
    mk = (jmn >= 0) & (jmn < n)
    delta[mk] += h[jmn[mk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def greed_binary(h_init, n, deadline):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_h, best_c = h.copy(), cur_c
    ones = list(np.where(h == 1)[0].astype(int))
    zeros = list(np.where(h == 0)[0].astype(int))
    for _ in range(1000):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[ii]
            for jj in range(len(ones)):
                j = ones[jj]
                d = make_delta(h, n, i, j, m_range)
                nc = (corr + d).max()
                if nc < cur_c:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.pop(jj)
                    ones.append(i)
                    zeros.pop(ii)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c * dx

def anneal(h_init, n, deadline, seed=42, max_iters=10000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_h, best_c = h.copy(), cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = 0.003
    cool = 0.99995
    stall = 0
    acc_count = 0
    check_int = 5000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80000:
            h = best_h.copy()
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max()
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            T = 0.003
            stall = 0
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i, j = zeros[ai], ones[bi]
        d = make_delta(h, n, i, j, m_range)
        nc = (corr + d).max()
        diff = (nc - cur_c) * dx
        accepted = False
        if diff <= 0:
            accepted = True
        elif T > 1e-12:
            if rng.random() < np.exp(-diff / T):
                accepted = True
        if accepted:
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            acc_count += 1
            if cur_c * dx < best_c:
                best_c = cur_c * dx
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if it % check_int == 0 and it > 0:
            rate = acc_count / check_int
            if rate > 0.5:
                T *= 0.8
            elif rate < 0.05:
                T *= 1.3
            acc_count = 0
        T *= cool
        T = max(T, 1e-14)
    return best_h, best_c

def refine_continuous(h, n, alpha):
    dx = 2.0 / n
    target = n / 2.0
    def objective(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(g)
        shifted = alpha * (g - m)
        sm = np.max(shifted)
        lse = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        return lse + 1e10 * (x.sum() - target) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-22, 'maxfun': 20000})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def refine_multi_alpha(h, n, deadline):
    best = h.copy()
    best_c = compute_c5(h, n)
    alphas = [5000, 20000, 100000, 500000, 2000000]
    for a in alphas:
        if time.time() > deadline - 2:
            break
        h = refine_continuous(best, n, a)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best = h.copy()
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dur = budget_s - 10
    dl = t0 + dur
    rng = np.random.RandomState(seed)

        h = best_h + 0.004 * rng.randn(n)
        h = project_feasible(h, n)

        t_mid = t0 + (p4 - t0) * 0.5 + time.time() - t0
