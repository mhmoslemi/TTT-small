def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n-s] += (1.0 - h[s:])
        if s > 0:
            g[s:] += -h[:n-s]
    else:
        sa = -s
        if sa < n:
            g[sa:] += (1.0 - h[:n-sa])
            g[:n-sa] += -h[sa:]
    return g

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    m = cc.max()
    sh = alpha * (cc - m)
    if sh.max() > 500:
        sh -= sh.max()
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    m = cc.max()
    sh = alpha * (cc - m)
    if sh.max() > 500:
        sh -= sh.max()
    w = np.exp(sh)
    ws = w.sum()
    w /= ws
    top_k = min(40, len(cc))

        s = int(k) - (n - 1)

    return grad * dx
