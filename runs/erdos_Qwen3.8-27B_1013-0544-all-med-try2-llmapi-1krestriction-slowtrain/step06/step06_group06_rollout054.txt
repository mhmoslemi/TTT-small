import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    """Compute objective value and analytic gradient of smooth max."""
    g = compute_all_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    shifted -= shifted.max()
    exps = np.exp(shifted)
    w = exps / exps.sum()
    obj = m + np.log(exps.sum()) / alpha
    
    dx = 2.0 / n
    # Compute gradient: ∇f[p] = dx * (1 - A[p] - B[p])
    # A[p] = sum_j w[j] * h[p-(j-n+1)] = sum_t w[n-1-t] * h[p+t]  (cross-corr of flip(w) and h)
    # B[p] = sum_j w[j] * h[p+(j-n+1)] = sum_t w[n-1-t] * h[p-t]  (conv of flip(w) and h)
    
    fw = w[::-1]  # flip(w), length 2n-1
    
    # A[p] = sum_t fw[t] * h[p+t], valid when 0 <= p+t < n
    # B[p] = sum_t fw[t] * h[p-t], valid when 0 <= p-t < n
    A = np.zeros(n)
    B = np.zeros(n)
    
    for t in range(2 * n - 1):
        wt = fw[t]
        if wt < 1e-16:
            continue
        # A[p]: need 0 <= p+t < n, so p in [-t, n-1-t] ∩ [0, n-1]
        p_lo = max(0, -t)
        p_hi = min(n - 1, n - 1 - t)
        if p_lo <= p_hi:
            A[p_lo:p_hi+1] += wt * h[t:p_hi+1-t]
        # B[p]: need 0 <= p-t < n, so p in [t, n-1+t] ∩ [0, n-1]
        p_lo_b = max(0, t)
        p_hi_b = min(n - 1, n - 1 + t)
        if p_lo_b <= p_hi_b:
            B[p_lo_b:p_hi_b+1] += wt * h[p_lo_b-t:p_hi_b+1-t]
    
    grad = dx * (1.0 - A - B)
    return obj, grad

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    """Optimize smooth max using L-BFGS-B with projection."""
    def obj_and_grad(x):
        # Project to feasible before computing
        x = project_feasible(x, n)
        val, grad = smooth_max_grad(x, n, alpha)
        return val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B', 
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

            if c < best_c:
                best_c = c
                best_h = h.copy()

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytic_smooth_max(h, n, alpha):
    """Return (value, grad) of smooth max of g(h) with parameter alpha."""
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()
    ex = np.exp(sh)
    w = ex / ex.sum()
    val = m + np.log(ex.sum()) / alpha
    
    dx = 2.0 / n
    fw = w[::-1]
    A = np.zeros(n)
    B = np.zeros(n)
    for t in range(2 * n - 1):
        wt = fw[t]
        if wt < 1e-17:
            continue
        lo = max(0, -t)
        hi = min(n - 1, n - 1 - t)
        if lo <= hi:
            A[lo:hi+1] += wt * h[t:hi+1-t]
        lo2 = max(0, t)
        hi2 = min(n - 1, n - 1 + t)
        if lo2 <= hi2:
            B[lo2:hi2+1] += wt * h[lo2-t:hi2+1-t]
    grad = dx * (1.0 - A - B)
    return val, grad

def local_subgrad(h, n, deadline, alpha=1000, max_iter=5000):
    """Subgradient descent using analytic smooth-max gradient."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, gr = analytic_smooth_max(h, n, alpha)
        ng = np.linalg.norm(gr)
        if ng < 1e-15:
            break
        found = False
        step = lr
        for _ in range(40):
            ht = project_feasible(h - step * gr, n)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-14:
                h = ht
                best_c = ct
                best_h = h.copy()
                found = True
                break
            step *= 0.5
        if not found:
            lr *= 0.95
            if lr < 1e-14:
                break
    return best_h, best_c

def slsqp_run(h, n, alpha, maxiter, deadline):
    """SLSQP with analytic gradient."""
    def fn(x):
        return analytic_smooth_max(x, n, alpha)[0]
    def fprime(x):
        return analytic_smooth_max(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fn, h, jac=fprime, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
