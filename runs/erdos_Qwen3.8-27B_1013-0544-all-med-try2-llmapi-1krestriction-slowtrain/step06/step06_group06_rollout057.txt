import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n, max_iter=1000):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_smooth(h, n, alpha):
    """Compute gradient of smooth approximation f_alpha(h)"""
    g = compute_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    # Numerical stability
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    # W_k where k = -(n-1), ..., n-1, stored in w of length 2n-1
    # w[k + n - 1] = W_k
    # grad_i = dx * [sum_k W_k * (1-h[i-k]) * I(0<=i-k<n) - sum_k W_k * h[i+k] * I(0<=i+k<n)]
    dx = 2.0 / n
    
    grad = np.zeros(n)
    # For each k, add w[k+n-1] * grad_g_k
    for idx in range(len(w)):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-12:
            continue
        # grad_g_k[i] = dx * [(1-h[i-k]) * I(0<=i-k<n) - h[i+k] * I(0<=i+k<n)]
        # First term: (1-h[i-k]) for i-k in [0, n-1], i.e., i in [k, k+n-1] intersect [0, n-1]
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:
            # i ranges from lo1 to hi1-1, i-k ranges from lo1-k to hi1-1-k
            grad[lo1:hi1] += wi * dx * (1.0 - h[lo1-k:hi1-k])
        # Second term: -h[i+k] for i+k in [0, n-1], i.e., i in [-k, n-1-k] intersect [0, n-1]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            # i ranges from lo2 to hi2-1, i+k ranges from lo2+k to hi2-1+k
            grad[lo2:hi2] -= wi * dx * h[lo2+k:hi2+k]
    
    return grad

def adam_optimize(h, n, alpha, max_iter, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for the smooth objective with projection"""
    h = h.copy()
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        grad = compute_gradient_smooth(h, n, alpha)
        
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad**2
        m_hat = m_t / (1 - beta1**t)
        v_hat = v_t / (1 - beta2**t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step, n)
        
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def subgradient_descent(h, n, deadline, max_iter=10000, lr_init=0.01):
    """Project subgradient descent on the exact max objective"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()
        
        # Find all k that are close to max (handle ties)
        thresh = m - 1e-8
        max_ks = np.where(g >= thresh)[0]
        
        # Average gradient over max k's
        grad = np.zeros(n)
        for idx in max_ks:
            k = idx - (n - 1)
            dx = 2.0 / n
            lo1 = max(0, k)
            hi1 = min(n, k + n)
            if lo1 < hi1:
                grad[lo1:hi1] += dx * (1.0 - h[lo1-k:hi1-k])
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if lo2 < hi2:
                grad[lo2:hi2] -= dx * h[lo2+k:hi2+k]
        
        grad /= len(max_ks)
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Line search
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * factor / ng
            h_new = project_feasible(h - step * grad, n)
            c = compute_c5(h_new, n)
            if c < m - 1e-10:
                h = h_new
                found = True
                break
        
        if found:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9
            if lr < 1e-12:
                break
    
    return best_h, best_c

                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:

        if not improved:

def smooth_gradient(h, n, alpha):
    """Gradient of the smooth approximation f_alpha = m + (1/alpha)*log(sum exp(alpha*(g-m)))"""
    g = compute_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-10:
            continue
        # d(g_k)/d(h[i]) = dx * [(1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)]
        # First part: (1-h[i-k]) for i in [max(0,k), min(n, k+n))
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:

        # Second part: -h[i+k] for i in [max(0,-k), min(n, n-k))
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * dx * h[lo2+k:hi2+k]
    return grad
