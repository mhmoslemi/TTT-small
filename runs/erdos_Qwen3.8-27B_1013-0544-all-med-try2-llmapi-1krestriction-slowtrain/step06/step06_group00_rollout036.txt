import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_eq(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_single_lag(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] = 1.0
        g[lag:] -= h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] = 1.0
        g[:n - m] -= h[m:]
        g[m:] -= h[:n - m]
    return g

def frank_wolfe(h, n, deadline, max_iters=500000, tol_active=1e-12):
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur = corr.max()
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        active = np.where(corr >= cur - tol_active)[0]
        g = np.zeros(n)
        for k in active:
            lag = int(k - (n - 1))
            g += grad_single_lag(h, n, lag)
        g /= len(active)
        idx = np.argsort(g)
        h_s = np.zeros(n)
        h_s[idx[:int(target)]] = 1.0
        t = 2.0 / (it + 2)
        h = (1.0 - t) * h + t * h_s
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_box_eq(res.x, n / 2.0)

    h = best_h.copy()
    fw2_h, fw2_c = frank_wolfe(h, n, p6 - 60, max_iters=200000)
    if fw2_c < best_c:
        best_c, best_h = fw2_c, fw2_h.copy()
    h = best_h.copy()
    sg2_h, sg2_c = subgrad_optimize(h, n, p6, max_iters=500000, step0=0.02, decay=0.99998)
    if sg2_c < best_c:
        best_c, best_h = sg2_c, sg2_h.copy()

    # Phase 7: Final polish with SLSQP (100s)
    p7 = dl - 5
    h = best_h.copy()
    for alpha in [1000000, 10000000, 100000000]:
        if time.time() > p7 - 8:
            break
        try:
            h = slsqp_smooth(h, n, alpha, 1000, 1e-18)
        except Exception:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 8: Noise restart + quick optimize (remaining)
    if time.time() < dl - 15:
        for trial in range(3):
            if time.time() > dl - 15:
                break
            h = best_h + 0.003 * rng.randn(n)
            h = project_box_eq(h, n / 2.0)
            for alpha in [5000000, 50000000]:
                if time.time() > dl - 16:
                    break
                try:
                    h = slsqp_smooth(h, n, alpha, 500, 1e-18)
                except Exception:
                    break
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()

    best_h = project_box_eq(best_h, n / 2.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
