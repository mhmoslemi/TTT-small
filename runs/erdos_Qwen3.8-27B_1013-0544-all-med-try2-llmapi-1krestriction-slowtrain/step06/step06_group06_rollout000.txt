import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_active_lag(j, h, n):
    dx = 2.0 / n
    k = j - (n - 1)
    m = np.arange(n)
    grad = np.zeros(n)
    zk = m - k
    v1 = (zk >= 0) & (zk < n)
    grad[v1] += (1.0 - h[zk[v1]])
    znk = m + k
    v2 = (znk >= 0) & (znk < n)
    grad[v2] -= h[znk[v2]]
    return grad * dx

def proj_grad_descent(h, n, deadline, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.8
    stuck = 0
    tol_active = 1e-5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        if m < best_c - 1e-14:
            best_c = m
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 1000:
                lr *= 0.92
                stuck = 0
                if lr < 1e-12:
                    break
        active = np.where(g >= m - tol_active)[0]
        grad = np.zeros(n)
        for j in active:
            grad += grad_active_lag(j, h, n)
        grad /= len(active)
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.002]:
            ht = project_feasible(h - lr * f * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.95
            if lr < 1e-13:
                break
    return best_h, best_c

def slsqp_smoothed(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_swap(h, n, deadline, max_sweeps=80):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    dx = 2.0 / n
    k_vals = np.arange(2 * n - 1) - (n - 1)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_all_g(h_bin, n)
        g_max = g.max()
        improved = False
        best_delta_max = 0
        best_swap = None
        for z in zeros:
            if time.time() > deadline:
                break
            zk = z - k_vals
            zpk = z + k_vals
            A = np.zeros(2 * n - 1)
            v1 = (zk >= 0) & (zk < n)
            A[v1] += (1.0 - h_bin[zk[v1]])
            v2 = (zpk >= 0) & (zpk < n)
            A[v2] -= h_bin[zpk[v2]]
            for o in ones:
                ok = o - k_vals
                opk = o + k_vals
                B = np.zeros(2 * n - 1)
                v3 = (ok >= 0) & (ok < n)
                B[v3] -= (1.0 - h_bin[ok[v3]])
                v4 = (opk >= 0) & (opk < n)
                B[v4] += h_bin[opk[v4]]
                delta = dx * (A + B)
                new_max = (g + delta).max()
                if new_max < g_max + best_delta_max - 1e-14:
                    best_delta_max = new_max - g_max
                    best_swap = (z, o)
        if best_swap is not None:
            z, o = best_swap
            h_bin[z] = 1.0
            h_bin[o] = 0.0
            ones.remove(o)
            ones.append(z)
            zeros.remove(z)
            zeros.append(o)
            best_c = compute_c5(h_bin, n)
            improved = True
        if not improved:
            break
    return h_bin, best_c
