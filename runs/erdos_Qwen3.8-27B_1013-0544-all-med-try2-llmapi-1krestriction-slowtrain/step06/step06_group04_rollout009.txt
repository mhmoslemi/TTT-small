def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1e-15)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    s = np.sum(np.exp(shifted))
    return float(m + np.log(s) / alpha)

def slsqp_run(h, n, alpha, top_idx, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        return smooth_max_obj(x, n, alpha, top_idx)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = np.zeros(n)
        m = abs(lag)
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr

            grad[m:] += (1.0 - h_curr[:n - m])

            grad[:n - m] += (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n - m]
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy(), False
    if prob.status not in ['optimal', 'optimal_inaccurate']:

    return float(np.max(c) * dx)

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0

                break
            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

    mx = shifted.max()
    if mx > 50:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
