def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:

        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)
