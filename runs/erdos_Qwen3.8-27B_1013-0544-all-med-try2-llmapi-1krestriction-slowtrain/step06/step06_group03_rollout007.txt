    if m < 1e-15:
        return 0.0
    return float(m * (np.mean((c / m) ** alpha)) ** (1.0 / alpha))

def slsqp(h, n, alpha, maxiter=3000, use_power=False):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    obj = obj_power if use_power else obj_lse
    try:
        res = minimize(obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def bottleneck_grad(h, n):
    """Compute gradient of the bottleneck lag's correlation w.r.t. h."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1 - h[i + lag])
            grad[i + lag] += -h[i]
    else:
        lag = -lag
        for i in range(n - lag):
            grad[i] += -h[i + lag]
            grad[i + lag] += (1 - h[i])
    return grad, k_star

def grad_informed_search(h, n, deadline, max_rounds=500):
    """Use bottleneck gradient to guide paired moves."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.01
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c
            best_h_ref = best_h.copy()
        grad, k_star = bottleneck_grad(best_h, n)
        
        # Find best paired move: increase h[i] where grad[i]<0, decrease h[j] where grad[j]>0
        neg_idx = np.where(grad < -0.01)[0]
        pos_idx = np.where(grad > 0.01)[0]
        if len(neg_idx) == 0 or len(pos_idx) == 0:
            break
        
        best_move_c = best_c
        best_move = None
        # Try a few random pairs
        n_try = min(20, len(neg_idx) * len(pos_idx))
        used = set()
        for _ in range(n_try):
            i = neg_idx[rng.startswith(0)[0] % len(neg_idx)] if False else neg_idx[rng.randint(len(neg_idx))]
            j = pos_idx[rng.randint(len(pos_idx))]
            if (i, j) in used:
                continue
            used.add((i, j))
            trial = best_h.copy()
            trial[i] = min(1.0, trial[i] + lr)
            trial[j] = max(0.0, trial[j] - lr)
            trial = project_feasible(trial, n)
            tc = compute_c5(trial, n)
            if tc < best_move_c - 1e-12:
                best_move_c = tc
                best_move = trial.copy()
        
        if best_move is not None:
            best_h = best_move
            best_c = best_move_c
        else:
            lr *= 0.8
            if lr < 1e-6:
                break
    
    return best_h, best_c

def binary_simulated_annealing(h, n, deadline, t_start=1.0, t_end=0.01):
    """Optimize binary structure via simulated annealing on bit flips."""
    # Start from thresholded version
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    # Adjust to have exactly target_ones ones
    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]
    if n_ones > target_ones:
        for idx in ones_idx[n_ones - target_ones:]:
            h_bin[idx] = 0
    elif n_ones < target_ones:
        for idx in zeros_idx[:target_ones - n_ones]:
            h_bin[idx] = 1
    
    # Soften slightly for optimization
    h_cur = h_bin * 0.9 + 0.05
    h_cur = project_feasible(h_cur, n)
    best_h = h_cur.copy()
    best_c = compute_c5(best_h, n)
    cur_c = best_c
    
    n_steps = 5000
    for step in range(n_steps):
        if time.time() > deadline:
            break
        temp = t_start * (t_end / t_start) ** (step / max(1, n_steps - 1))
        i = rng.randint(n)
        if h_cur[i] > 0.5:
            j_candidates = np.where(h_cur < 0.5)[0]
        else:
            j_candidates = np.where(h_cur > 0.5)[0]
        if len(j_candidates) == 0:
            break
        j = j_candidates[rng.randint(len(j_candidates))]
        
        trial = h_cur.copy()
        if h_cur[i] > h_cur[j]:
            trial[i] -= 0.1
            trial[j] += 0.1
        else:
            trial[i] += 0.1
            trial[j] -= 0.1
        trial = project_feasible(trial, n)
        tc = compute_c5(trial, n)
        delta = tc - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            h_cur = trial
            cur_c = tc
            if cur_c < best_c:
                best_c = cur_c
                best_h = trial.copy()
    
    return best_h, best_c

def multi_scale_optimize(h_fine, n_fine, deadline, rng):
    """Coarse-to-fine optimization."""
    best_h = h_fine.copy()
    best_c = compute_c5(best_h, n_fine)
    x_fine = np.linspace(0, 2, n_fine)
    
    # Coarosen to n_coarse
    for n_coarse in [25, 50, 100]:
        if time.time() > deadline - 30:
            break
        x_coarse = np.linspace(0, 2, n_coarse)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        h_coarse = project_feasible(h_coarse, n_coarse)
        
        # Optimize at coarse level
        for alpha in [2000, 10000, 50000]:
            if time.time() > deadline - 20:
                break
            h_opt = slsqp(h_coarse, n_coarse, alpha, maxiter=1500)
            c_opt = compute_c5(h_opt, n_coarse)
            if c_opt < compute_c5(h_coarse, n_coarse):
                h_coarse = h_opt
        
        # Upsample back
        h_up = np.interp(x_fine, x_coarse, h_coarse)
        h_up = project_feasible(h_up, n_fine)
        c_up = compute_c5(h_up, n_fine)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h_start = np.array(initial_h_values, dtype=float).copy()
        n = len(h_start)
    except Exception:
        n = 250
        h_start = np.full(n, 0.5)
    
    best_h = project_feasible(h_start, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP with log-sum-exp (quick pass)
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 400:
            break
        h_opt = slsqp(best_h, n, alpha, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: SLSQP with power smooth max (different landscape)
    for alpha in [200, 500, 1000]:
        if time.time() > deadline - 350:
            break
        h_opt = slsqp(best_h, n, alpha, maxiter=1500, use_power=True)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Gradient-informed local search
    if time.time() < deadline - 200:
        h_opt, c_opt = grad_informed_search(best_h, n, deadline - 200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Multi-scale coarse-to-fine
    if time.time() < deadline - 150:
        h_opt, c_opt = multi_scale_optimize(best_h, n, deadline - 150, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Binary simulated annealing from current best
    if time.time() < deadline - 100:
        h_opt, c_opt = binary_simulated_annealing(best_h, n, deadline - 100)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: SLSQP polish from best
    if time.time() < deadline - 50:
        for alpha in [100000, 500000, 1000000]:
            if time.time() > deadline - 45:
                break
            h_opt = slsqp(best_h, n, alpha, maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c - 1e-14:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 7: Fine-grained pair search on bottleneck
    if time.time() < deadline - 20:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for d in deltas:
            if time.time() > deadline - 10:
                break
            improved = True
            while improved and time.time() < deadline - 10:
                improved = False
                for i in range(n):
                    if time.time() > deadline - 10:
                        break
                    for j in [i - 1, i + 1, i - 2, i + 2, i + 10, i - 10,
                              i + 25, i - 25] if 0 <= max(0, min) < n else []:
                        pass
                    # Check a few nearby and far indices
                    for j in range(n):
                        if j == i:
                            continue
                        if (i * 31 + j) % 17 != 0:  # subsample for speed
                            continue
                        if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                            best_h[i] += d; best_h[j] -= d
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True
                            else:
                                best_h[i] -= d; best_h[j] += d
                        if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                            best_h[i] -= d; best_h[j] += d
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True
                            else:
                                best_h[i] += d; best_h[j] -= d
    
    # Phase 8: Final SLSQP
    if time.time() < deadline - 5:
        h_opt = slsqp(best_h, n, 500000, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
