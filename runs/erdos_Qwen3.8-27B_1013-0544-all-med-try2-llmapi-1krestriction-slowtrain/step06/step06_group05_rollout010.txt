So the move would be: take mass from positions with the most positive gradient and give it to positions with the most negative gradient.

This is essentially a "topological sort" of the gradient - move from max to min.

But we can be smarter: instead of just the active constraint, consider all constraints that are within ε of the maximum.

def gradient_local_search(h, n, deadline, max_iters=10000):
    """Gradient-guided local search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_all = np.correlate(h, 1-h, mode='full') * (2.0/n)
        k_star = int(np.argmax(c_all))
        s = abs(k_star - (n-1))
        
        if s == 0:
            grad = (1 - 2*h)
        else:
            grad = np.zeros(n)
            grad[s:] += (1 - h[:n-s])
            grad[:n-s] -= h[s:]
        
        # Find best move: take from max grad, give to min grad
        # But we need to respect the constraint ∑h = n/2 (which moves preserve)
        
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        
        if grad[i_src] - grad[i_dst] < 1e-10:
            break
        
        # Try a move
        max_d = min(h[i_src], 1 - h[i_dst])
        # Binary search for the step size that minimizes the objective
        # Or just try a few step sizes
        best_step = 0
        best_c_try = best_c
        
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            d = max_d * frac
            if d < 1e-10:
                continue
            h[i_src] -= d
            h[i_dst] += d
            c_try = compute_c5(h, n)
            if c_try < best_c_try:
                best_c_try = c_try
                best_step = d
            h[i_src] += d
            h[i_dst] -= d
        
        if best_step > 0:
            h[i_src] -= best_step
            h[i_dst] += best_step
            best_c = best_c_try
        else:
            # Try next best pair
            # Remove the top and bottom, try again
            # Or break
            # Let's try a few random pairs guided by gradient
            found = False
            for _ in range(50):
                # Sample source from top of gradient, destination from bottom
                top_idx = np.argsort(grad)[-10:]
                bot_idx = np.argsort(grad)[:10]
                i_src = top_idx[np.random.randint(len(top_idx))]
                i_dst = bot_idx[np.random.randint(len(bot_idx))]
                if i_src == i_dst:
                    continue
                max_d = min(h[i_src], 1 - h[i_dst])
                if max_d < 1e-10:
                    continue
                for frac in [0.1, 0.5, 0.9]:
                    d = max_d * frac
                    h[i_src] -= d
                    h[i_dst] += d
                    c_try = compute_c5(h, n)
                    if c_try < best_c:
                        best_c = c_try
                        found = True
                        break
                    h[i_src] += d
                    h[i_dst] -= d
                if found:
                    break
            if not found:
                break
    
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target / tot), 0, 1)
    return h

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return m + np.log(s.sum() + 1e-30) / alpha + s.max() / alpha

def lse_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    s_shifted = s - s_max
    w = np.exp(s_shifted)
    w_sum = w.sum()
    w /= w_sum
    
    # Gradient of each c[k] with respect to x
    # For correlation index k (0..2n-2), shift = k - (n-1)

        return np.ones(n)

        return lse_obj(v, n, alpha)

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    f, g = make_slsqp_obj(n, alpha)
    def cfun(v):
        return v.sum() - target
    def cjac(v):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(f, x0, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n), compute_c5(res.x, n)
    except Exception:
        return x0.copy(), compute_c5(x0, n)

def gradient_local_search(h, n, deadline, max_iters=80000):
    h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 500:
            break
        c_all = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        k_star = int(np.argmax(c_all))
        s = abs(k_star - (n - 1))
        if s == 0:
            grad = 1 - 2 * h
        else:
            grad = np.zeros(n)
            grad[s:] += (1 - h[:n - s])
            grad[:n - s] -= h[s:]
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if grad[i_src] - grad[i_dst] < 1e-10:
            no_improve += 1
            continue
        improved = False
        max_d = min(h[i_src], 1.0 - h[i_dst])
        if max_d > 1e-12:
            for frac in [0.5, 0.25, 0.75, 0.1, 0.9]:
                d = max_d * frac
                h[i_src] -= d
                h[i_dst] += d
                c_try = compute_c5(h, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    improved = True
                    break
                h[i_src] += d
                h[i_dst] -= d
        if not improved:
            found = False
            top = np.argsort(grad)[-20:]
            bot = np.argsort(grad)[:20]
            for i_s in top:
                if found or time.time() > deadline:
                    break
                for i_d in bot:
                    if i_s == i_d:
                        continue
                    md = min(h[i_s], 1.0 - h[i_d])
                    if md < 1e-12:
                        continue
                    for frac in [0.3, 0.6, 0.9]:
                        d = md * frac
                        h[i_s] -= d
                        h[i_d] += d
                        c_try = compute_c5(h, n)
                        if c_try < best_c - 1e-15:
                            best_c = c_try
                            found = True
                            break
                        h[i_s] += d
                        h[i_d] -= d
                    if found:
                        break
            if found:
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve = 0
    return h, best_c

def anneal_refine(h, n, deadline, rng, n_iter=5000, T0=0.003, Tend=0.0001):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (it / n_iter)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.02) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + d
        hj = h[j] - d
        if not (0 <= hi <= 1 and 0 <= hj <= 1):
            continue
        h[i], h[j] = hi, hj
        c_new = compute_c5(h, n)
        delta = c_new - best_c
        if delta < 0 or (T > 0 and np.log(rng.rand() + 1e-30) < -delta / T):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = hi - d, hj + d
    return best_h, best_c

def optimize_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [1000, 5000, 20000, 100000, 500000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        hr, cr = slsqp_opt(best_h, n, a, 3000, deadline - 20)
        if cr < best_c - 1e-14:
            best_c = cr
            best_h = hr
    if time.time() < deadline - 10:
        best_h, best_c = gradient_local_search(best_h, n, deadline - 10)
    if time.time() < deadline - 5:
        ha, ca = anneal_refine(best_h, n, deadline - 5, rng, n_iter=3000)
        if ca < best_c:
            best_c = ca
            best_h = ha
    if time.time() < deadline - 3:
        best_h, best_c = gradient_local_search(best_h, n, deadline - 3)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    x_init = np.linspace(0, 2, n_init)
    # Phase 1: optimize at initial resolution
    alloc1 = min(250, (deadline - t0) * 0.22)
    if t0 + alloc1 < deadline:
        best_h, best_c = optimize_resolution(h_init, n_init, t0 + alloc1, rng)
    # Phase 2: try other resolutions
    if time.time() < deadline - 80:
        for n_try in [100, 150, 200, 300, 400, 600]:
            if time.time() > deadline - 70:
                break
            if n_try == n_init:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(90, (deadline - time.time() - 50) * 0.22)
            if alloc < 25:
                break
            hr, cr = optimize_resolution(h_interp, n_try, time.time() + alloc, rng)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                x_init = x_new.copy()
                n_init = n_try
    # Phase 3: multi-restart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50:
        n_restarts = max(1, int((deadline - time.time() - 30) / 55))
        scales = [0.008, 0.02, 0.04, 0.06, 0.1]
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 25) * 0.35)
            if alloc < 18:
                break
            hr, cr = optimize_resolution(hp, n_best, time.time() + alloc, rng)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
    # Phase 4: final intensive polish
    if time.time() < deadline - 4:
        best_h, best_c = gradient_local_search(best_h, n_best, deadline - 2)
        best_h = project_feasible(best_h, n_best)
        best_c = compute_c5(best_h, n_best)
    return best_h, best_c, len(best_h)
