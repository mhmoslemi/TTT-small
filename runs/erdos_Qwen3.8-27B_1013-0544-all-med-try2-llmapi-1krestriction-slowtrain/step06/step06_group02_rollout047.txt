    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - mx))) / alpha
    return float(val)

def smooth_obj_pen(x, n, alpha, mu):
    val = smooth_obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_pen(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=200, alpha_list=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [(50, 1.0), (200, 5.0), (1000, 20.0),
                      (5000, 100.0), (20000, 500.0),
                      (100000, 2000.0), (500000, 10000.0), (2000000, 50000.0)]
    for alpha, mu in alpha_list:
        if time.time() > deadline - 1:
            break
        remaining = deadline - time.time()
        if remaining < 2:
            break
        try:
            res = minimize(smooth_obj_pen, h, jac=smooth_grad_pen,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': max_iters * 5,
                                    'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def pair_exchange_ls(h, n, deadline):
    """Pair exchange with line search for optimal step."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            radius = 80 if d > 0.005 else (40 if d > 0.001 else n)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                if d <= 0.002:
                    lo, hi = 0, n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
            if not improved:
                break
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def gradient_descent_step(h, n, deadline, lr_init=0.02, alpha=5000):
    """Projected gradient descent with backtracking."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(5000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            alpha = min(alpha * 2, 2000000)
        if lr < 1e-14:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n = 250
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: L-BFGS from parent (moderate alpha)
    t1 = min(t0 + 180, deadline - 350)
    best_h, best_c = lbfgs_optimize(best_h, n, t1, max_iters=150)
    
    # Phase 2: Pair exchange (coarse to fine)
    if time.time() < deadline - 250:
        t2 = min(time.time() + 180, deadline - 250)
        best_h, best_c = pair_exchange_ls(best_h, n, t2)
    
    # Phase 3: L-BFGS with high alpha (focus on critical shifts)
    if time.time() < deadline - 180:
        t3 = min(time.time() + 80, deadline - 180)
        h3, c3 = lbfgs_optimize(best_h, n, t3, max_iters=100,
                                alpha_list=[(50000, 500.0), (200000, 5000.0),
                                            (1000000, 50000.0)])
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Gradient descent for fine-tuning
    if time.time() < deadline - 120:
        t4 = min(time.time() + 60, deadline - 120)
        h4, c4 = gradient_descent_step(best_h, n, t4, lr_init=0.01, alpha=20000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Fine pair exchange
    if time.time() < deadline - 80:
        t5 = min(time.time() + 80, deadline - 80)
        best_h, best_c = pair_exchange_ls(best_h, n, t5)
    
    # Phase 6: Try alternative n values
    if time.time() < deadline - 60:
        x_old = np.linspace(0, 2, len(best_h))
        for n_try in [200, 400]:
            if time.time() > deadline - 30:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(40, deadline - time.time() - 25)
            if alloc < 15:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5,
                                       max_iters=80,
                                       alpha_list=[(1000, 10.0), (10000, 100.0),
                                                   (100000, 1000.0)])
            if time.time() < deadline - 15:
                h_t, c_t = pair_exchange_ls(h_t, n_try, time.time() + min(15, deadline - time.time() - 8))
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
    
    # Phase 7: Final L-BFGS + pair polish
    if time.time() < deadline - 10:
        nf = len(best_h)
        if time.time() < deadline - 5:
            best_h, best_c = lbfgs_optimize(best_h, nf, deadline - 5, max_iters=80,
                                            alpha_list=[(100000, 1000.0), (500000, 5000.0)])
        if time.time() < deadline - 1:
            best_h, best_c = pair_exchange_ls(best_h, nf, deadline)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
