```python
def slsqp_with_grad(h, n, alpha, maxiter=5000, ftol=1e-16):
    dx = 2.0 / n
    mn = n - 1
    
    def obj(x):
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        C_max = C.max()
        shifted = alpha * (C - C_max)
        return C_max + np.log(np.exp(shifted).sum()) / alpha
    
    def grad(x):
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        C_max = C.max()
        shifted = alpha * (C - C_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        g = np.zeros(n)
        for ki in range(2*n-1):
            wi = w[ki]
            if wi < 1e-15:
                continue
            k = ki - mn
            if k >= 0:
                g[k:] += wi * (1.0 - x[:n-k])
                g[:n-k] -= wi * x[k:]
            else:
                k2 = -k
                g[:n-k2] += wi * (1.0 - x[k2:])
                g[k2:] -= wi * x[:n-k2]
        return g * dx
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h
```
