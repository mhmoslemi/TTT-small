def adam_optimize(h0, n, deadline, alpha=10000, lr=0.01, beta1=0.9, beta2=0.999):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        t += 1
        # Compute objective and gradient
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        mval = corr.max()
        shifted = alpha * (corr - mval)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        w = alpha * weights / dx  # gradient weights (undo the dx)
        
        # Compute gradient: grad[i] = sum_m w[m] * [(1-h[i+m-(n-1)]) - h[i-m+(n-1)]] * dx
        # = dx * (part1 - part2)

            hi2 = min(2*n-2, i+(n-1))
            j2 = i - np.arange(lo2, hi2+1) + (n-1)
            grad[i] -= np.dot(w[lo2:hi2+1], h[j2])

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)

        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c
