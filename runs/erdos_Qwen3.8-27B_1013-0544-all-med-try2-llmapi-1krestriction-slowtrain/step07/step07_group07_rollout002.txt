def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * float(n)

        h = np.clip(h, 0.0, 1.0)

import numpy as np
import time
import math


def compute_all_lags(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))


def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * float(n)

    mn = float(np.min(y))
    mx = float(np.max(y))
    lo = min(-1.0, mn - 1.0)
    hi = max(1.0, mx + 1.0)

    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

    idx = int(n // 2)
    for _ in range(20):
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-15:
            break
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
        idx = (idx + 1) % int(n)

    return h


def raw_sine(n, phase=0.0):
    i = np.arange(n, dtype=float)
    x = 2.0 * i / float(n)
    return 0.5 + 0.5 * np.sin(math.pi * x + phase)
