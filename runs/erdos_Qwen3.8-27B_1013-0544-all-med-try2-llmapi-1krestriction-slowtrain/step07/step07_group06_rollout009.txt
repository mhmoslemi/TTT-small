a_m = (1-h[j-lag_m])*dx [if 0 ≤ j-lag_m < n] + (-h[j+lag_m])*dx [if 0 ≤ j+lag_m < n]

And the constant term (b_m) is C[m] - a_m * h[j] (computed with h[j] set to 0).

Then we want to minimize max_m (a_m * h[j] + b_m) over h[j] ∈ [0,1].

This is a 1D convex optimization problem that can be solved exactly!

The function f(t) = max_m (a_m * t + b_m) for t ∈ [0,1] is the upper envelope of lines. We can find its minimum by:
- Computing all intersection points of pairs of lines within [0,1]
- Checking f at 0, 1, and all intersection points
- Returning the minimum

But there are O(n²) pairs and O(2n) lines, so this is O(n²) per coordinate, and O(n³) total. For n=300, that's 27*10⁶ operations per sweep, which should be fast enough.

    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

            break
        t += 1

        lr_cur = lr_init * (1.0 + 0.5 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def compute_d_pair(h, n, i, j):
    dx = 2.0 / n
    ks = np.arange(-(n-1), n)
    d = np.zeros(2*n-1)
    # a_i,m terms
    m1 = (ks >= i-n+1) & (ks <= i)
    if m1.any():
        d[m1] += (1.0 - h[i - ks[m1]]) * dx
    m2 = (ks >= -i) & (ks <= n-1-i)
    if m2.any():
        d[m2] += (-h[i + ks[m2]]) * dx
    # a_j,m terms (subtracted)
    m3 = (ks >= j-n+1) & (ks <= j)
    if m3.any():
        d[m3] -= (1.0 - h[j - ks[m3]]) * dx
    m4 = (ks >= -j) & (ks <= n-1-j)
    if m4.any():
        d[m4] -= (-h[j + ks[m4]]) * dx
    return d

def exact_pair_update(C, h, n, i, j):
    d = compute_d_pair(h, n, i, j)
    # delta in [delta_lo, delta_hi]
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    if delta_lo >= delta_hi:
        return 0.0
    # Bisection on subgradient
    lo, hi = delta_lo, delta_hi
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = C + mid * d
        mx = vals.max()
        active = np.where(np.abs(vals - mx) <= 1e-13)[0]
        g_lo = d[active].min()
        g_hi = d[active].max()
        if g_lo > 0:
            hi = mid
        elif g_hi < 0:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def adam_optimize(h, n, deadline, alpha_init=50.0, alpha_final=200000.0, lr_init=0.012):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 6000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.5 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def lbfgs_phase(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 10.0), (1000, 50.0), (5000, 200.0),
               (20000, 1000.0), (100000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 4000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def compute_d_for_i(h, n, i):
    dx = 2.0 / n
    ks = np.arange(-(n - 1), n)
    a_i = np.zeros(2 * n - 1)
    m1 = (ks >= i - n + 1) & (ks <= i)
    a_i[m1] += (1.0 - h[i - ks[m1]]) * dx
    m2 = (ks >= -i) & (ks <= n - 1 - i)
    a_i[m2] += (-h[i + ks[m2]]) * dx
    a_j_mat = np.zeros((n, 2 * n - 1))
    jarr = np.arange(n)
    for ki in range(len(ks)):
        k = ks[ki]
        mt1 = (jarr - k >= 0) & (jarr - k < n)
        if mt1.any():
            a_j_mat[mt1, ki] += (1.0 - h[jarr[mt1] - k]) * dx
        mt2 = (jarr + k >= 0) & (jarr + k < n)
        if mt2.any():
            a_j_mat[mt2, ki] += (-h[jarr[mt2] + k]) * dx
    return a_i, a_j_mat

def vectorized_pair_search(h, n, deadline, max_sweeps=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    C = compute_c5_vec(h, n)
    n_active = 2 * n - 1
    idx_all = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        g = grad_smooth_max(h, n, 5000)
        order = np.argsort(-np.abs(g))[:150]
        for ii in order:
            if time.time() > deadline:
                return best_h, best_c
            i = int(ii)
            a_i, a_j_mat = compute_d_for_i(h, n, i)
            d_mat = a_i[None, :] - a_j_mat
            jarr = idx_all
            mask_valid = (jarr != i) & (h[jarr] + h[i] > 0.01) & (h[jarr] + h[i] < 1.99)
            if not mask_valid.any():
                continue
            jv = jarr[mask_valid]
            dv = d_mat[mask_valid]
            dlo = np.maximum(-h[i], h[jv] - 1.0)
            dhi = np.minimum(1.0 - h[i], h[jv])
            valid = dhi > dlo + 1e-10
            if not valid.any():
                continue
            jv = jv[valid]
            dv = dv[valid]
            dlo = dlo[valid]
            dhi = dhi[valid]
            lo = dlo.copy()
            hi = dhi.copy()
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = C[None, :] + mid[:, None] * dv
                fvals = vals.max(axis=1)
                active = (np.abs(vals - fvals[:, None]) < 1e-12)
                g_lo = np.where(active, dv, np.inf).min(axis=1)
                g_hi = np.where(active, dv, -np.inf).max(axis=1)
                go_l = g_lo > 1e-12
                go_r = g_hi < -1e-12
                hi[go_l] = mid[go_l]
                lo[go_r] = mid[go_r]
            dopt = (lo + hi) / 2.0
            fvals = (C[None, :] + dopt[:, None] * dv).max(axis=1)
            bidx = fvals.argmin()
            if fvals[bidx] < best_c - 1e-13:
                delta = dopt[bidx]
                j = int(jv[bidx])
                h[i] += delta
                h[j] -= delta
                C += delta * dv[bidx]
                best_c = float(C.max())
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def pair_search_basic(h, n, deadline, deltas, radius=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, 5000)
            order = np.argsort(-np.abs(g))[:200]
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 5

        h_init = np.array(initial_h_values, dtype=float).copy()

    start_h = h_sine if c_sine < best_c else best_h

    # Phase 1: L-BFGS
    t1 = min(t0 + 100, deadline - 500)
    h1, c1 = lbfgs_phase(start_h, n, t1)
    if c1 < best_c:
        best_c, best_h = c1, h1
    # Phase 2: Adam
    if time.time() < deadline - 350:
        t2 = min(time.time() + 280, deadline - 350)
        ha, ca = adam_optimize(best_h, n, t2)
        if ca < best_c:
            best_c, best_h = ca, ha
    # Also try from sine separately
    if time.time() < deadline - 300 and c_sine < best_c + 0.01:
        t2b = min(time.time() + 150, deadline - 300)
        hs, cs = adam_optimize(h_sine, n, t2b, lr_init=0.01)
        if cs < best_c:
            best_c, best_h = cs, hs
    # Phase 3: Vectorized exact pair search

        t3 = min(time.time() + 250, deadline - 200)
        best_h, best_c = vectorized_pair_search(best_h, n, t3, max_sweeps=40)
    # Phase 4: Basic pair search with small steps

        best_h, best_c = pair_search_basic(best_h, n, t4,

    if time.time() < deadline - 30:
        t5 = deadline - 3
        best_h, best_c = vectorized_pair_search(best_h, n, t5, max_sweeps=20)
    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n)
    c_flip = compute_c5(h_flip, n)
    if c_flip < best_c:
        best_c, best_h = c_flip, h_flip
    # Final verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
