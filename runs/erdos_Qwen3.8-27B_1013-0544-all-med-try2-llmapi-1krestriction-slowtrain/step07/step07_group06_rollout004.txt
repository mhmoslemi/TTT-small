import numpy as np
import time

BP = np.array([1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2])
BE = np.array([1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3])
EF = np.array([0.25, 0.5])
EL = np.array([0.8, 0.98, 1.0])


def proj(h, n):
    h = np.clip(np.asarray(h, float).copy(), 0.0, 1.0)
    t = n / 2.0
    for _ in range(200):
        d = t - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)


def c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * 2.0 / n


def ms(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def com(h, n):
    h = proj(h, n)
    c0 = float(np.dot(h, 1.0 - h))
    cA = np.zeros(n)
    cB = np.zeros(n)
    EA = np.zeros((n, n))
    EB = np.zeros((n, n))
    for m in range(1, n):
        cp = n - m
        hL = h[:cp]
        hR = h[m:m + cp]
        cA[m] = float(np.dot(hL, 1.0 - hR))
        cB[m] = float(np.dot(hR, 1.0 - hL))
        EA[:cp, m] = 1.0 - hR
        EB[:cp, m] = -hR
        EA[m:, m] = EA[m:, m] - hL
        EB[m:, m] = EB[m:, m] + 1.0 - hL
    mx = max(c0, float(cA.max()), float(cB.max()))
    return h, c0, cA, cB, EA, EB, float(mx)


def updE(EA, EB, p, dl, n):
    if dl == 0:
        return
    if p < n - 1:
        m = np.arange(1, n - p)
        q = p + m
        EA[q, m] = EA[q, m] - dl
        EB[q, m] = EB[q, m] - dl
    if p > 0:
        m = np.arange(1, p + 1)
        q = p - m
        EA[q, m] = EA[q, m] - dl
        EB[q, m] = EB[q, m] - dl


def mkD(dm, ex):
    if dm <= 1e-12:
        return np.empty(0)
    base = BE if ex else BP
    D = base[base <= dm]
    extra = EF * dm
    if ex and dm > 0.5:
        extra = np.concatenate((extra, EL * dm))
    D = np.concatenate((D, extra))
    D = D[(D > 1e-12) & (D <= dm + 1e-12)]
    D = np.unique(D)
    if D.size > 10:
        D = np.unique(np.concatenate((D[:6], D[-4:])))
    return D


def ev(a, b, D, h, c0, cA, cB, EA, EB):
    dA = EA[a] - EA[b]
    dB = EB[a] - EB[b]
    d = D.reshape(-1, 1)
    maxA = np.max(cA[None, :] + d * dA[None, :], axis=1)
    maxB = np.max(cB[None, :] + d * dB[None, :], axis=1)
    mm = abs(a - b)
    if mm > 0:
        dd = D * D
        maxA = np.maximum(maxA, cA[mm] + D * dA[mm] + dd)
        maxB = np.maximum(maxB, cB[mm] + D * dB[mm] + dd)
    root = c0 + 2.0 * D * (h[b] - h[a]) - 2.0 * D * D
    return np.maximum(np.maximum(maxA, maxB), root)


def mv(a, b, d, h, c0, cA, cB, EA, EB, n):
    d = min(d, max(0.0, 1.0 - h[a]), max(0.0, h[b]))
    if d <= 1e-15:
        return h, c0, cA, cB
    dA = EA[a] - EA[b]
    dB = EB[a] - EB[b]
    cA = cA + d * dA
    cB = cB + d * dB
    mm = abs(a - b)
    if mm > 0:
        cA[mm] += d * d
        cB[mm] += d * d
    c0 = c0 + 2.0 * d * (h[b] - h[a]) - 2.0 * d * d
    h[a] += d
    h[b] -= d
    if h[a] > 1.0:
        h[a] = 1.0
    if h[b] < 0.0:
        h[b] = 0.0
    updE(EA, EB, a, d, n)
    updE(EA, EB, b, -d, n)
    return h, c0, cA, cB


def smW(c0, cA, cB, al):
    n = cA.size
    z = np.empty(2 * n - 1)
    z[0] = c0
    z[1:n] = cA[1:]
    z[n:] = cB[1:]
    mx = z.max()
    w = np.exp(al * (z - mx))
    w /= w.sum()
    wA = np.zeros(n)
    wB = np.zeros(n)
    w0 = float(w[0])
    wA[1:] = w[1:n]
    wB[1:] = w[n:]
    return w0, wA, wB


def sc(h, w0, wA, wB, EA, EB):
    return w0 * (1.0 - 2.0 * h) + EA.T.dot(wA) + EB.T.dot(wB)


def sel(score, h, K):
    inc = np.flatnonzero(h < 1.0 - 1e-10)
    dec = np.flatnonzero(h > 1e-10)
    if inc.size == 0 or dec.size == 0:
        return inc, dec
    K1 = min(K, inc.size)
    K2 = min(K, dec.size)
    si = score[inc]
    lo = inc[np.argpartition(si, K1 - 1)[:K1]]
    sd = score[dec]
    hi = dec[np.argpartition(sd, dec.size - K2)[-K2:]]
    return lo, hi

            if time.time() > dl - 4:
                break
        if time.time() < dl - 4:
            inc = np.flatnonzero(h < 1.0 - 1e-10)
            dec = np.flatnonzero(h > 1e-10)
            nr = 150 if bc is not None else 400
            for _ in range(nr):
                if time.time() > dl - 4 or inc.size == 0 or dec.size == 0:
                    break
                a = int(inc[rng.integers(inc.size)])
                b = int(dec[rng.integers(dec.size)])
                if a == b:
                    continue
                if not ex and score[a] - score[b] > 1e-6:
                    continue
                dmd = min(1.0 - h[a], h[b])
                if dmd <= 1e-12:
                    continue
                D = mkD(dmd, ex)
                if D.size:
                    vals = ev(a, b, D, h, c0, cA, cB, EA, EB)
                    idx = int(np.argmin(vals))
                    if vals[idx] < cm - 1e-12:
                        cm = float(vals[idx])
                        bc = (a, b, float(D[idx]))
        if bc is not None:
            a, b, d = bc
            h, c0, cA, cB = mv(a, b, d, h, c0, cA, cB, EA, EB, n)
            mx = max(c0, float(cA.max()), float(cB.max()))
            if mx < bm - 1e-12:
                bm = mx
                bh = h.copy()
            noi = 0
            acc += 1
        else:
            noi += 1
            ai += 1
            if noi == 2:
                al2 = min(200.0, max(10.0, al))
                w0g, wAg, wBg = smW(c0, cA, cB, al2)
                sg = sc(h, w0g, wAg, wBg, EA, EB)
                sg = sg - sg.mean()
                found = False
                for lr in (1e-4, 3e-4, 1e-3, 3e-3, 1e-2):
                    if time.time() > dl - 4:
                        break
                    hx = proj(h - lr * sg, n)
                    mxg = ms(hx, n)
                    if mxg < mx - 1e-12:
                        h, c0, cA, cB, EA, EB, mx = com(hx, n)
                        if mx < bm - 1e-12:
                            bm = mx
                            bh = h.copy()
                        found = True
                        break
                if found:
                    noi = 0
                    acc = 0
                    ai = 0
            if noi >= 5:
                pert = False
                for amp in (1e-3, 3e-3, 1e-2):
                    hx = proj(h + amp * rng.normal(size=n), n)
                    h2, c02, cA2, cB2, EA2, EB2, mx2 = com(hx, n)
                    if mx2 < mx + 0.02:
                        h, c0, cA, cB, EA, EB, mx = h2, c02, cA2, cB2, EA2, EB2, mx2
                        pert = True
                        break
                if pert:
                    noi = 0
                    acc = 0
                    ai = 0
                else:
                    break
    re = com(bh, n)
    return re[0], float(re[6])


def binv(h, n):
    h = np.clip(h, 0.0, 1.0)
    k = int(n / 2)
    b = np.zeros(n)
    if k > 0:
        idx = np.argpartition(h, n - k)[-k:]
        b[idx] = 1.0
    return b


def rsmp(h, nn):
    lo = len(h)
    if nn % lo == 0:
        h = np.repeat(h, nn // lo)
    else:
        xo = np.linspace(0.0, 2.0, lo, endpoint=False)
        xn = np.linspace(0.0, 2.0, nn, endpoint=False)
        h = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return proj(h, nn)


def alloc(t, dl, use, reserve):
    rem = dl - t
    if rem <= reserve + 20:
        return t
    return t + min(use, rem - reserve)


def run(seed=42, budget_s=1100, **kwargs):
    seed = int(seed)
    t0 = time.time()
    dl = t0 + max(5.0, float(budget_s) - 5.0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 300
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        h0 = 0.5 + 0.48 * np.cos(np.pi * x)

    h0 = proj(h0, n0)
    bh = h0.copy()
    bc = c5(bh, n0)
    bn = n0

    if dl - time.time() > 100:
        ta = alloc(time.time(), dl, 420, 300)
        if ta > time.time() + 30:
            ha, mxa = opt(h0, n0, ta, False, 35, seed)
            ca = mxa * 2.0 / n0
            if ca < bc:
                bh, bc, bn = ha, ca, n0

    if dl - time.time() > 100:
        hb = binv(bh, bn)
        tb = alloc(time.time(), dl, 180, 200)
        if tb > time.time() + 30:
            hbm, mxb = opt(hb, bn, tb, True, 30, seed + 1)
            cb = mxb * 2.0 / bn
            if cb < bc:
                bh, bc, bn = hbm, cb, bn

    if dl - time.time() > 150 and bn < 600:
        nc = min(600, max(400, 2 * bn))
        hc = rsmp(bh, nc)
        tc = alloc(time.time(), dl, 240, 120)
        if tc > time.time() + 40:
            hcm, mxc = opt(hc, nc, tc, False, 25, seed + 2)
            cc = mxc * 2.0 / nc
            if cc < bc:
                bh, bc, bn = hcm, cc, nc

    if dl - time.time() > 50:
        tf = alloc(time.time(), dl, 900, 10)
        hfm, mxf = opt(bh, bn, tf, False, 40, seed + 3)
        cf = mxf * 2.0 / bn
        if cf < bc:
            bh, bc, bn = hfm, cf, bn

    bh = proj(bh, bn)
    bc = c5(bh, bn)
    return bh, bc, bn
