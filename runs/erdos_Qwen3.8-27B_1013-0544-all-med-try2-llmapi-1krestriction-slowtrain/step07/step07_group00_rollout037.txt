Now let me also think about the "active set" approach more carefully. The optimal solution satisfies the KKT conditions:
- At the optimum, there exist dual variables μ_k ≥ 0 for each shift k such that:
  - Σ_k μ_k = 1
  - μ_k > 0 only for active shifts (where I_k = C₅)
  - For each j: Σ_k μ_k * ∂I_k/∂h[j] + λ + ν_j^+ - ν_j^- = 0
    where λ is the dual for the sum constraint, ν_j^+ ≥ 0 for h[j] ≤ 1, ν_j^- ≥ 0 for h[j] ≥ 0

This means the optimal solution is where the weighted average of the active gradients is "balanced" (the projection onto the tangent space of the feasible set is zero).

My subgradient method implicitly does this by averaging the top-k gradients.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        wm = w[s + n - 1]
        if wm < 1e-15:
            continue
        if s == 0:
            grad += wm * (1.0 - 2.0 * x)
        elif s > 0:
            nv = n - s
            grad[:nv] += wm * (1.0 - x[s:])
            grad[s:] += -wm * x[:nv]
        else:
            kk = -s
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[kk:]
    return grad * dx

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    mx = c.max()
    sh = alpha * (c - mx)
    smx = sh.max()
    return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)

def lbfgs_opt(h, n, alpha, deadline, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    mu = 500.0
    def obj(x):
        return smooth_max_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
    def grad(x):
        return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgrad_polish(h, n, deadline, k=5):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.002
    stuck_count = 0
    for _ in range(500000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-k:]
        g = np.zeros(n)
        for m in top_idx:
            s = m - (n - 1)
            if s == 0:
                g += 1.0 - 2.0 * h
            elif s > 0:
                nv = n - s
                g[:nv] += 1.0 - h[s:]
                g[s:] += -h[:nv]
            else:
                kk = -s
                nv = n - kk
                g[kk:] += 1.0 - h[:nv]
                g[:nv] += -h[kk:]
        h_new = h - step * g * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            step = min(step * 1.03, 0.01)
            stuck_count = 0
        else:
            stuck_count += 1
            step *= 0.97
            if step < 1e-7 or stuck_count > 200:
                step = 0.002
                stuck_count = 0
    return h, best_c

def fourier_global(h_init, n, deadline, rng):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for M in [15, 30, 50]:
        if time.time() > deadline - 15:
            break
        H = np.fft.fft(h_init - 0.5)
        coeffs = np.zeros(2 * M)
        for k2 in range(1, M + 1):
            coeffs[2*(k2-1)] = 2.0 * np.real(H[k2]) / n
            coeffs[2*(k2-1)+1] = -2.0 * np.imag(H[k2]) / n
        def f2h(c2):
            hh = np.full(n, 0.5)
            for kk in range(M):
                hh += c2[2*kk] * np.cos(np.pi * (kk+1) * x)
                hh += c2[2*kk+1] * np.sin(np.pi * (kk+1) * x)
            return hh
        def f_obj(c2):
            hh = f2h(c2)
            cv = compute_c5_vec(hh, n)
            mx = cv.max()
            a = 300.0
            sh = a * (cv - mx)
            smx = sh.max()
            val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / a
            val += 80.0 * (np.sum(np.maximum(0, -hh)**2) + np.sum(np.maximum(0, hh-1)**2))
            return float(val)
        try:

            h_new = f2h(res.x)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c
