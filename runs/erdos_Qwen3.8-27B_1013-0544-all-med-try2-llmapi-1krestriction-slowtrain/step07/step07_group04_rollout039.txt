from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:

            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room

            s = h.sum()
            if s < 1e-14: break

                    imp = True
                    break
                else:
                    h[i], h[j] = 0.0, 1.0
            if imp:
                break
        if not imp:
            break
    return h, cur

def anneal(h_init, n, deadline, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_init.copy()
    cur = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
    best_c = cur
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = 0.003
    Tmin = 5e-7
    cool = 0.99998
    last_imp = 0
    it = 0
    while time.time() < deadline:
        it += 1
        r = rng.random()
        if r < 0.75:
            ii = rng.randint(len(zeros))
            ji = rng.randint(len(ones))
            i, j = zeros[ii], ones[ji]
            h[i], h[j] = 1.0, 0.0
            nc = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
            d = nc - cur
            if d < 0 or (T > Tmin and rng.random() < np.exp(-d / T)):
                cur = nc
                zeros[ii] = j; ones[ji] = i
                if cur < best_c:
                    best_c = cur; best_h = h.copy(); last_imp = it
            else:
                h[i], h[j] = 0.0, 1.0
        elif r < 0.92:
            if len(zeros) < 2 or len(ones) < 2:
                continue
            ii = rng.randint(len(zeros)); ji = rng.randint(len(ones))
            ii2 = rng.randint(len(zeros)); ji2 = rng.randint(len(ones))
            if ii == ii2 or ji == ji2:
                continue
            i1, i2, j1, j2 = zeros[ii], zeros[ii2], ones[ji], ones[ji2]
            h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
            nc = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
            d = nc - cur
            if d < 0 or (T > Tmin and rng.random() < np.exp(-d / T)):
                cur = nc
                zeros[ii] = j1; zeros[ii2] = j2
                ones[ji] = i1; ones[ji2] = i2
                if cur < best_c:
                    best_c = cur; best_h = h.copy(); last_imp = it
            else:
                h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
        else:
            for _ in range(rng.randint(2, 6)):
                ii = rng.randint(len(zeros)); ji = rng.randint(len(ones))
                i, j = zeros[ii], ones[ji]
                h[i], h[j] = 1.0, 0.0
                zeros[ii] = j; ones[ji] = i
            nc = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
            d = nc - cur
            if d < 0 or (T > Tmin and rng.random() < np.exp(-d / T)):
                cur = nc
                if cur < best_c:
                    best_c = cur; best_h = h.copy(); last_imp = it
        T = max(T * cool, Tmin)
        if it - last_imp > 20000 and T < Tmin * 5:
            T = 0.005
    return best_h, best_c

def sturmian_binary(n, alpha, seed=0):
    h = np.array([int(np.floor((i + 1) * alpha) - np.floor(i * alpha)) for i in range(n)], dtype=float)
    target = n // 2
    rng = np.random.RandomState(seed)
    if h.sum() > target:
        idx = np.where(h == 1)[0]; rng.shuffle(idx)
        for i in idx:
            if h.sum() <= target:
                break
            h[i] = 0
    elif h.sum() < target:
        idx = np.where(h == 0)[0]; rng.shuffle(idx)
        for i in idx:
            if h.sum() >= target:
                break
            h[i] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = sturmian_binary(n, 0.6180339887498949, 0)

    hb = (h0 >= 0.5).astype(float)
    target = n // 2
    if hb.sum() != target:
        order = np.argsort(-h0)
        for idx in order:
            if hb.sum() <= target:
                break
            hb[idx] = 0
        for idx in order:
            if hb.sum() >= target:
                break
            hb[idx] = 1

    best_h = hb.copy()
    best_c = compute_c5(best_h, n)

    p1 = t0 + 0.10 * (budget_s - 5)
    hc = project_feasible(hb + 0.05 * rng.randn(n), n)
    for a in [200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > p1:
            break
        hc = slsqp_opt(hc, n, a, 1000, p1)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c = c; best_h = hc.copy()

    p2 = t0 + 0.13 * (budget_s - 5)
    hg, cg = greedy_improve((best_h > 0.5).astype(float), n, p2, 100)
    if cg < best_c:
        best_c = cg; best_h = hg.copy()

    p3 = t0 + 0.58 * (budget_s - 5)
    ha, ca = anneal(best_h, n, p3, seed)
    if ca < best_c:
        best_c = ca; best_h = ha.copy()

    p4 = t0 + 0.70 * (budget_s - 5)
    hg, cg = greedy_improve(best_h, n, p4, 200)
    if cg < best_c:
        best_c = cg; best_h = hg.copy()

    p5 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p5:
        for s_init in [0.6180339887498949, 0.7071067811865476, 0.5436890126920762]:
            if time.time() > p5:
                break
            hs = sturmian_binary(n, s_init, seed + 100)
            hs = project_feasible(hs + 0.1 * rng.randn(n), n)
            for a in [5000, 50000]:
                if time.time() > p5:
                    break
                hs = slsqp_opt(hs, n, a, 500, p5)
                hsb = project_feasible((hs > 0.5).astype(float), n)
                cs = compute_c5(hsb, n)
                if cs < best_c:
                    best_c = cs; best_h = hsb.copy()

    p6 = t0 + 0.88 * (budget_s - 5)
    ha, ca = anneal(best_h, n, p6, seed + 50)
    if ca < best_c:
        best_c = ca; best_h = ha.copy()

    if time.time() < dl - 10:
        hc = project_feasible(best_h + 0.03 * rng.randn(n), n)
        hc = slsqp_opt(hc, n, 200000, 2000, dl - 8)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c = c; best_h = hc.copy()

    if time.time() < dl - 3:
        hg, cg = greedy_improve((best_h > 0.5).astype(float), n, dl, 100)
        if cg < best_c:
            best_c = cg; best_h = hg.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
