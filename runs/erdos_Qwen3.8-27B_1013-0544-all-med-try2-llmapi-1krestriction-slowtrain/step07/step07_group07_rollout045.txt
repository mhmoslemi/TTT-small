    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def soft_max_obj(x, n, alpha, top_k):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c = c[idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, top_k, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha, top_k), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_step(h_curr, n, trust, top_lags):
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-top_lags:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k_idx in top_idx:
        lag = k_idx - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k_idx] + (2.0/n) * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def proj_grad_descent(h, n, deadline, max_iters=200):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for ss in step_sizes:
        for _ in range(max_iters):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            g = lag_gradient(h, n, lag)
            g_centered = g - g.mean()
            h_new = project_feasible(h - ss * g_centered, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            h = h_new
            if np.max(np.abs(h_new - h)) < 1e-12:
                break
    return best_h, best_c

def optimize_round(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Outer approximation with tightening trust region
    if time.time() < deadline - 40:
        h_curr = best_h.copy()
        trust = 0.12
        for it in range(30):
            if time.time() > deadline - 40:
                break
            h_new = outer_approx_step(h_curr, n, trust, top_lags=5)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
            step = np.max(np.abs(h_new - h_curr))
            if step < 1e-9:
                break
            h_curr = h_new
            trust = max(0.003, trust * 0.82)

    # Phase 2: SLSQP with escalating alpha
    alpha_schedule = [20000, 100000, 500000, 2000000, 10000000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 20:
            break
        top_k = 3 if alpha < 100000 else 2
        h_new = slsqp_optimize(best_h, n, alpha, top_k, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new

    # Phase 3: Projected gradient descent on active lags
    if time.time() < deadline - 5:
        h_curr = best_h.copy()
        for ss in [0.005, 0.001, 0.0003, 0.0001]:
            if time.time() > deadline - 5:
                break
            for _ in range(100):
                if time.time() > deadline - 5:
                    break
                c = compute_all_lags(h_curr, n)
                k_star = int(np.argmax(c))
                lag = k_star - (n - 1)
                if lag == 0:
                    break
                g = lag_gradient(h_curr, n, lag)
                g_c = g - g.mean()
                h_new = project_feasible(h_curr - ss * g_c, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                h_curr = h_new
                if np.max(np.abs(h_new - h_curr)) < 1e-13:
                    break

    # Phase 4: Final SLSQP polish
    if time.time() < deadline - 2:
        h_new = slsqp_optimize(best_h, n, 50000000, 2, maxiter=5000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    return project_feasible(best_h, n), best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    # Optimize at the initial resolution
    if h_init is not None:
        n = n_init
        h_start = project_feasible(h_init, n)
        alloc = min(350, (deadline - t0) * 0.38)
        h_opt, c_opt = optimize_round(h_start, n, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    # Try a few other resolutions
    n_candidates = [250, 350]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 80:
            break
        alloc = min(120, remaining * 0.35)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_round(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Multi-start with perturbations at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        scales = [0.008, 0.015, 0.025, 0.04, 0.06]
        for s in scales:
            if time.time() > deadline - 60:
                break
            remaining = deadline - time.time() - 40
            if remaining < 50:
                break
            hp = best_h + rng.normal(0, s, n)
            hp = project_feasible(hp, n)
            alloc = min(60, remaining * 0.4)
            h_opt, c_opt = optimize_round(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Structured perturbation: try circular shifts and segment flips
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        for shift in [1, 2, 3]:
            if time.time() > deadline - 50:
                break
            hp = np.roll(best_h, shift)
            hp = project_feasible(hp, n)
            h_opt, c_opt = optimize_round(hp, n, time.time() + min(40, (deadline - time.time() - 40) * 0.4))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final intensive polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 100000000, 2, maxiter=8000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final projected gradient
        if time.time() < deadline - 2:
            h_curr = best_h.copy()
            for ss in [0.0001, 0.00005, 0.00001]:
                for _ in range(50):
                    if time.time() > deadline - 2:
                        break
                    c = compute_all_lags(h_curr, n)
                    k_star = int(np.argmax(c))
                    lag = k_star - (n - 1)
                    if lag == 0:
                        break
                    g = lag_gradient(h_curr, n, lag)
                    g_c = g - g.mean()
                    h_new2 = project_feasible(h_curr - ss * g_c, n)
                    c_new2 = compute_c5(h_new2, n)
                    if c_new2 < best_c - 1e-16:
                        best_c = c_new2
                        best_h = h_new2
                    h_curr = h_new2
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
