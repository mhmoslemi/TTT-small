def project_sum(h, target, n):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on offset."""
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    m = corr.max()
