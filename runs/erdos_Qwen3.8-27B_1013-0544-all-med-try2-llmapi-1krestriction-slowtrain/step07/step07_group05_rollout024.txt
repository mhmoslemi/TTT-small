import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, n_sweeps=50, tol=1e-12, time_limit=float('inf'), randomize=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    m_arr = np.arange(2 * n - 1)
    order = list(range(n))
    for sweep in range(n_sweeps):
        if time.time() > time_limit:
            break
        if randomize and rng is not None:
            rng.shuffle(order)
        max_change = 0.0
        for i in order:
            old_val = h[i]
            s_arr = m_arr - n + 1
            i_plus_s = i + s_arr
            valid_plus = (i_plus_s >= 0) & (i_plus_s <= n - 1)
            i_minus_s = i - s_arr
            valid_minus = (i_minus_s >= 0) & (i_minus_s <= n - 1)
            coef = np.zeros(2 * n - 1)
            if valid_plus.any():
                coef[valid_plus] += (1.0 - h[i_plus_s[valid_plus]])
            if valid_minus.any():
                coef[valid_minus] -= h[i_minus_s[valid_minus]]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                c_new = c + (mid - old_val) * coef
                V = c_new.max()
                active = c_new >= V - 1e-14
                min_c = coef[active].min()
                max_c = coef[active].max()
                if min_c <= 0.0 <= max_c:
                    break
                elif min_c > 0.0:
                    hi = mid
                else:
                    lo = mid
            new_val = (lo + hi) * 0.5
            new_val = min(max(new_val, 0.0), 1.0)
            if abs(new_val - old_val) > 1e-15:
                c = c + (new_val - old_val) * coef
                h[i] = new_val
                max_change = max(max_change, abs(new_val - old_val))
        if max_change < tol:
            break
    return h

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=1500):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            weights = np.exp(sh - mx)
            weights /= weights.sum()
            grad_c = np.zeros(2 * n - 1)
            grad_c += weights
            dh = np.zeros(n)
            for k in range(2 * n - 1):
                s = k - n + 1
                for j in range(max(0, -s), min(n, n - s)):
                    dh[j] += weights[k] * (1 - x[j + s]) * dx
                    if 0 <= j - s < n:
                        pass
                    dh[j] -= weights[k] * x[j + s] * dx if 0 <= j + s < n else 0
            # Numerical gradient fallback for correctness
            eps = 1e-7
            g = np.zeros(n)
            for i in range(n):
                xp = x.copy(); xp[i] += eps
                xm = x.copy(); xm[i] -= eps
                g[i] = (obj(xp) - obj(xm)) / (2 * eps)
            return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
