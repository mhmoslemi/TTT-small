def smooth_max_grad(h, alpha, c_vals=None):
    n=len(h); dx=2.0/n
    if c_vals is None: c_vals=np.correlate(h,1-h,'full')*dx
    m=float(c_vals.max())
    vals=alpha*(c_vals-m)
    v=float(vals.max())
    e=np.exp(vals-v)
    s=e.sum()
    obj=m+(v+np.log(s))/alpha
    w=e/s
    g=np.zeros(n)
    # threshold
    for idx, wt in enumerate(w):

          l=-lag; a=n-l
          g[l:] += aw*(1.0-h[:a])
          g[:a] -= aw*h[l:]
    return float(obj), g, c_vals

```
import numpy as np
from scipy.optimize import minimize
import time

YB = 14.0
EM = 1e-6
PAIR_FRACS = np.exp(np.linspace(np.log(0.001), 0.0, 20))[::-1]

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -60.0, 60.0)))

def project_feasible(q):
    q = np.asarray(q, dtype=float)
    n = q.size
    if n == 0:
        return q
    S = 0.5 * n
    lo = float(q.min()) - 1.0
    hi = float(q.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(q - mid, 0.0, 1.0))) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(q - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(100):
        diff = S - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmin(h))
            adj = min(diff, 1.0 - h[i])
        else:
            i = int(np.argmax(h))
            adj = max(diff, -h[i])
        if abs(adj) < 1e-15:
            break
        h[i] += adj
    return np.clip(h, 0.0, 1.0)

def lags_of(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5_of(h):
    return float(np.max(lags_of(h)))

def smooth_grad(h, alpha, idxs=None, cvals=None):
    n = len(h)
    if cvals is None:
        cvals = lags_of(h)
    g = np.zeros(n, dtype=float)
    if idxs is None:
        sel = np.asarray(cvals, dtype=float)
        m = float(sel.max())
        vals = float(alpha) * (sel - m)
        v = float(vals.max())
        e = np.exp(vals - v)
        s = float(e.sum())
        if s <= 0.0:
            return m, g, cvals
        w = e / s
        obj = m + (v + float(np.log(s))) / float(alpha)
        for gi in range(2 * n - 1):
            wt = float(w[gi])
            if wt < 1e-15:
                continue
            lag = gi - (n - 1)
            if lag == 0:
                g += wt * (1.0 - 2.0 * h)
            elif lag > 0:
                a = n - lag
                g[:a] -= wt * h[lag:]
                g[lag:] += wt * (1.0 - h[:a])
            else:
                l = -lag
                a = n - l
                g[:a] += wt * (1.0 - h[l:])
                g[l:] -= wt * h[:a]
        return obj, g, cvals
    idx_arr = np.asarray(idxs, dtype=int)
    sel = cvals[idx_arr]
    m = float(sel.max())
    vals = float(alpha) * (sel - m)
    v = float(vals.max())
    e = np.exp(vals - v)
    s = float(e.sum())
    if s <= 0.0:
        return m, g, cvals
    w = e / s
    obj = m + (v + float(np.log(s))) / float(alpha)
    for r in range(idx_arr.size):
        gi = int(idx_arr[r])
        wt = float(w[r])
        if wt < 1e-15:
            continue
        lag = gi - (n - 1)
        if lag == 0:
            g += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            a = n - lag
            g[:a] -= wt * h[lag:]
            g[lag:] += wt * (1.0 - h[:a])
        else:
            l = -lag
            a = n - l
            g[:a] += wt * (1.0 - h[l:])
            g[l:] -= wt * h[:a]
    return float(obj), g, cvals

def find_offset(y, n):
    y = np.asarray(y, dtype=float)
    S = 0.5 * n
    lo = float(y.max()) - 50.0
    hi = float(y.min()) + 50.0
    for _ in range(8):
        if float(np.sum(sigmoid(y + lo))) < S and float(np.sum(sigmoid(y + hi))) > S:
            break
        lo -= 30.0
        hi += 30.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.sum(sigmoid(y + mid))) < S:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def h_to_y(h):
    h = np.clip(h, EM, 1.0 - EM)
    y = np.log(h / (1.0 - h))
    return np.clip(y, -YB, YB)

def y_to_h(y):
    c = find_offset(y, len(y))
    return project_feasible(sigmoid(y + c))

def _objective_y(y, alpha, n, idxs):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        return 1e100
    c = find_offset(y, n)
    h = sigmoid(y + c)
    obj, _, _ = smooth_grad(h, float(alpha), idxs)
    return float(obj)

def _jacobian_y(y, alpha, n, idxs):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        return np.zeros_like(y)
    c = find_offset(y, n)
    h = sigmoid(y + c)
    _, gh, _ = smooth_grad(h, float(alpha), idxs)
    p = h * (1.0 - h)
    P = float(p.sum())
    if P > 1e-18:
        gy = p * (gh - float(np.dot(gh, p)) / P)
    else:
        gy = gh - float(gh.mean())
    return gy

def focused(h, alpha, K, maxiter, max_calls, deadline):
    h = project_feasible(h.copy())
    bc = c5_of(h)
    cur = h.copy()
    cur_c = bc
    n = len(cur)
    y = h_to_y(cur)
    bounds = [(-YB, YB)] * n
    options = {'maxiter': int(max(1, maxiter)), 'ftol': 1e-16, 'gtol': 1e-11,
               'maxls': 20, 'maxfun': int(max(100, 10 * maxiter))}
    fails = 0
    calls = 0
    while calls < max_calls and time.time() < deadline - 30:
        cvals = lags_of(cur)
        L = 2 * n - 1
        if K is None or K >= L:
            idxs = None
        else:
            idxs = tuple(np.argsort(cvals)[-int(K):])
        try:
            res = minimize(_objective_y, y, args=(float(alpha), n, idxs),
                           method='L-BFGS-B', jac=_jacobian_y, bounds=bounds, options=options)
        except Exception:
            break
        yb = np.asarray(res.x, dtype=float)
        if not np.all(np.isfinite(yb)):
            break
        hnew = y_to_h(yb)
        cnew = c5_of(hnew)
        calls += 1
        if cnew < cur_c - 1e-12:
            cur = hnew
            cur_c = cnew
            y = h_to_y(cur)
            fails = 0
        else:
            fails += 1
            if fails >= 5:
                break
        if cnew < bc - 1e-12:
            bc = cnew
            h = hnew
    return h, bc

def grad_pairs(g, h, top=4):
    src_all = np.flatnonzero(h > 1e-8)
    dst_all = np.flatnonzero(h < 1.0 - 1e-8)
    if src_all.size == 0 or dst_all.size == 0:
        return []
    top = int(min(top, src_all.size, dst_all.size))
    s_top = src_all[np.argsort(g[src_all])[-top:]][::-1]
    d_top = dst_all[np.argsort(g[dst_all])[:top]]
    pairs = []
    seen = set()
    for j in s_top:
        for i in d_top:
            if int(i) != int(j):
                key = (int(j), int(i))
                if key not in seen:
                    seen.add(key)
                    pairs.append(key)
    return pairs

def active_pairs(h, cvals=None, top_lags=4, K=5):
    if cvals is None:
        cvals = lags_of(h)
    n = len(h)
    L = 2 * n - 1
    top_lags = min(int(top_lags), L)
    order = np.argsort(cvals)[-top_lags:][::-1]
    pairs = []
    seen = set()
    for gi in order:
        lag = int(gi - (n - 1))
        m = abs(lag)
        if m == 0:
            continue
        a = n - m
        if lag > 0:
            pot = h[m:] * (1.0 - h[:a])
            if pot.size == 0:
                continue
            edges = np.argsort(pot)[-min(K, pot.size):]
            for e in edges:
                key = (int(e + m), int(e))
        else:
            pot = h[:a] * (1.0 - h[m:])
            if pot.size == 0:
                continue
            edges = np.argsort(pot)[-min(K, pot.size):]
            for e in edges:
                key = (int(e), int(e + m))
        if key[0] != key[1] and key not in seen:
            seen.add(key)
            pairs.append(key)
    return pairs

def pair_line_search(h, dst, src, bc):
    if dst == src:
        return h, bc
    dmax = min(float(h[src]), 1.0 - float(h[dst]))
    if dmax <= 1e-10:
        return h, bc
    od = float(h[dst])
    osrc = float(h[src])
    bestd = 0.0
    bestc = bc
    for fr in PAIR_FRACS:
        d = float(fr) * dmax
        if d <= 1e-11:
            continue
        h[dst] = min(1.0, od + d)
        h[src] = max(0.0, osrc - d)
        c = c5_of(h)
        if c < bestc:
            bestc = c
            bestd = d
        h[dst] = od
        h[src] = osrc
    if bestd > 0.0:
        h[dst] = od + bestd
        h[src] = osrc - bestd
        h = project_feasible(h)
        bestc = c5_of(h)
    return h, bestc

def pair_local(h, deadline, max_attempts=300):
    h = project_feasible(h.copy())
    bc = c5_of(h)
    n = len(h)
    alphas = (20000.0, 8000.0, 3000.0)
    ai = 0
    fails = 0
    attempts = 0
    while time.time() < deadline - 2 and attempts < max_attempts and fails < 160:
        cvals = lags_of(h)
        L = 2 * n - 1
        top_idx = np.argsort(cvals)[-25:] if L > 25 else None
        try:
            _, g, _ = smooth_grad(h, alphas[ai], top_idx, cvals)
        except Exception:
            break
        pairs = grad_pairs(g, h, 4)
        pairs.extend(active_pairs(h, cvals, 4, 5))
        improved = False
        for src, dst in pairs:
            if time.time() > deadline - 1:
                break
            attempts += 1
            h2, c2 = pair_line_search(h, dst, src, bc)
            if c2 < bc - 1e-13:
                h = h2
                bc = c2
                improved = True
                fails = 0
                break
        if not improved:
            fails += 1
            if fails == 60:
                ai = (ai + 1) % 3
                fails = 0
    return h, bc

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n)
    f1 = int(rng.randint(1, 7))
    f2 = int(rng.randint(f1 + 1, 11))
    u1 = rng.uniform(0.0, 2.0 * np.pi)
    u2 = rng.uniform(0.0, 2.0 * np.pi)
    s = 0.6 * np.sin(2.0 * np.pi * f1 * x + u1) + 0.4 * np.sin(2.0 * np.pi * f2 * x + u2)
    s = s + 0.15 * rng.standard_normal(n)
    h = 0.5 + 0.5 * np.tanh(2.0 * s)
    return project_feasible(h)

def optimize_run(h, deadline, quick=False):
    h = project_feasible(h.copy())
    bc = c5_of(h)
    if not quick:
        h, bc = focused(h, 1000.0, 120, 150, 2, deadline)
        h, bc = focused(h, 5000.0, 50, 150, 2, deadline)
    h, bc = focused(h, 20000.0, 18, 120, 3, deadline)
    h, bc = focused(h, 80000.0, 8, 100, 3, deadline)
    if time.time() < deadline - 40:
        h, bc = focused(h, 30000.0, None, 80, 1, deadline)
    if time.time() < deadline - 15:
        h, bc = pair_local(h, deadline, 100 if quick else 300)
    return h, bc

import numpy as np
from scipy.optimize import minimize
import time

YB = 14.0
EM = 1e-6
PAIR_FRACS = np.exp(np.linspace(np.log(0.001), 0.0, 20))[::-1]

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -60.0, 60.0)))

def project_feasible(q):
    q = np.asarray(q, dtype=float)
    n = q.size
    if n == 0:
        return q
    S = 0.5 * n
    lo = float(q.min()) - 1.0
    hi = float(q.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(q - mid, 0.0, 1.0))) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(q - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(100):
        diff = S - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmin(h))
            adj = min(diff, 1.0 - h[i])
        else:
            i = int(np.argmax(h))
            adj = max(diff, -h[i])
        if abs(adj) < 1e-15:
            break
        h[i] += adj
    return np.clip(h, 0.0, 1.0)

def lags_of(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5_of(h):
    return float(np.max(lags_of(h)))

def _apply_weight(g, h, gi, wt, n):
    lag = gi - (n - 1)
    if lag == 0:
        g += wt * (1.0 - 2.0 * h)
    elif lag > 0:
        a = n - lag
        g[:a] -= wt * h[lag:]
        g[lag:] += wt * (1.0 - h[:a])
    else:
        l = -lag
        a = n - l
        g[:a] += wt * (1.0 - h[l:])
        g[l:] -= wt * h[:a]

def smooth_grad(h, alpha, idxs=None, cvals=None):
    n = len(h)
    if cvals is None:
        cvals = lags_of(h)
    g = np.zeros(n, dtype=float)
    if idxs is None:
        sel = np.asarray(cvals, dtype=float)
    else:
        sel = np.asarray(cvals[np.asarray(idxs, dtype=int)], dtype=float)
    m = float(sel.max())
    vals = float(alpha) * (sel - m)
    v = float(vals.max())
    e = np.exp(vals - v)
    s = float(e.sum())
    if s <= 0.0:
        return m, g, cvals
    w = e / s
    obj = m + (v + float(np.log(s))) / float(alpha)
    if idxs is None:
        for gi in range(2 * n - 1):
            wt = float(w[gi])
            if wt > 1e-15:
                _apply_weight(g, h, gi, wt, n)
    else:
        idx_arr = np.asarray(idxs, dtype=int)
        for r in range(idx_arr.size):
            wt = float(w[r])
            if wt > 1e-15:
                _apply_weight(g, h, int(idx_arr[r]), wt, n)
    return float(obj), g, cvals

def find_offset(y, n):
    y = np.asarray(y, dtype=float)
    S = 0.5 * n
    lo = float(y.max()) - 50.0
    hi = float(y.min()) + 50.0
    for _ in range(8):
        if float(np.sum(sigmoid(y + lo))) < S and float(np.sum(sigmoid(y + hi))) > S:
            break
        lo -= 30.0
        hi += 30.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.sum(sigmoid(y + mid))) < S:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def h_to_y(h):
    h = np.clip(h, EM, 1.0 - EM)
    return np.clip(np.log(h / (1.0 - h)), -YB, YB)

def y_to_h(y):
    c = find_offset(y, len(y))
    return project_feasible(sigmoid(y + c))

def _objective_y(y, alpha, n, idxs):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        return 1e100
    c = find_offset(y, n)
    h = sigmoid(y + c)
    obj, _, _ = smooth_grad(h, float(alpha), idxs)
    return float(obj)

def _jacobian_y(y, alpha, n, idxs):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        return np.zeros_like(y)
    c = find_offset(y, n)
    h = sigmoid(y + c)
    _, gh, _ = smooth_grad(h, float(alpha), idxs)
    p = h * (1.0 - h)
    P = float(p.sum())
    if P > 1e-18:
        return p * (gh - float(np.dot(gh, p)) / P)
    return gh - float(gh.mean())

def focused(h, alpha, K, maxiter, max_calls, deadline):
    h = project_feasible(h.copy())
    bc = c5_of(h)
    cur = h.copy()
    cur_c = bc
    n = len(cur)
    y = h_to_y(cur)
    bounds = [(-YB, YB)] * n
    options = {'maxiter': int(max(1, maxiter)), 'ftol': 1e-16, 'gtol': 1e-11,
               'maxls': 20, 'maxfun': int(max(100, 10 * maxiter))}
    fails = 0
    calls = 0
    while calls < max_calls and time.time() < deadline - 30:
        cvals = lags_of(cur)
        L = 2 * n - 1
        idxs = None if K is None or K >= L else tuple(np.argsort(cvals)[-int(K):])
        try:
            res = minimize(_objective_y, y, args=(float(alpha), n, idxs),
                           method='L-BFGS-B', jac=_jacobian_y, bounds=bounds, options=options)
        except Exception:
            break
        yb = np.asarray(res.x, dtype=float)
        if not np.all(np.isfinite(yb)):
            break
        hnew = y_to_h(yb)
        cnew = c5_of(hnew)
        calls += 1
        if cnew < cur_c - 1e-12:
            cur, cur_c, y, fails = hnew, cnew, h_to_y(hnew), 0
        else:
            fails += 1
            if fails >= 5:
                break
        if cnew < bc - 1e-12:
            bc, h = cnew, hnew
    return h, bc

def grad_pairs(g, h, top=4):
    src_all = np.flatnonzero(h > 1e-8)
    dst_all = np.flatnonzero(h < 1.0 - 1e-8)
    if src_all.size == 0 or dst_all.size == 0:
        return []
    top = int(min(top, src_all.size, dst_all.size))
    s_top = src_all[np.argsort(g[src_all])[-top:]][::-1]
    d_top = dst_all[np.argsort(g[dst_all])[:top]]
    pairs, seen = [], set()
    for j in s_top:
        for i in d_top:
            if int(i) != int(j):
                key = (int(j), int(i))
                if key not in seen:
                    seen.add(key)
                    pairs.append(key)
    return pairs

def active_pairs(h, cvals=None, top_lags=4, K=5):
    if cvals is None:
        cvals = lags_of(h)
    n = len(h)
    L = 2 * n - 1
    order = np.argsort(cvals)[-min(int(top_lags), L):][::-1]
    pairs, seen = [], set()
    for gi in order:
        lag = int(gi - (n - 1))
        m = abs(lag)
        if m == 0:
            continue
        a = n - m
        if lag > 0:
            pot = h[m:] * (1.0 - h[:a])
            if pot.size == 0:
                continue
            for e in np.argsort(pot)[-min(K, pot.size):]:
                key = (int(e + m), int(e))
        else:
            pot = h[:a] * (1.0 - h[m:])
            if pot.size == 0:
                continue
            for e in np.argsort(pot)[-min(K, pot.size):]:
                key = (int(e), int(e + m))
        if key[0] != key[1] and key not in seen:
            seen.add(key)
            pairs.append(key)
    return pairs

def pair_line_search(h, dst, src, bc):
    if dst == src:
        return h, bc
    dmax = min(float(h[src]), 1.0 - float(h[dst]))
    if dmax <= 1e-10:
        return h, bc
    od, osrc = float(h[dst]), float(h[src])
    bestd, bestc = 0.0, bc
    for fr in PAIR_FRACS:
        d = float(fr) * dmax
        if d <= 1e-11:
            continue
        h[dst] = min(1.0, od + d)
        h[src] = max(0.0, osrc - d)
        c = c5_of(h)
        if c < bestc:
            bestc, bestd = c, d
        h[dst], h[src] = od, osrc
    if bestd > 0.0:
        h[dst], h[src] = od + bestd, osrc - bestd
        h = project_feasible(h)
        bestc = c5_of(h)
    return h, bestc

def pair_local(h, deadline, max_attempts=300):
    h = project_feasible(h.copy())
    bc = c5_of(h)
    n = len(h)
    alphas = (20000.0, 8000.0, 3000.0)
    ai, fails, attempts = 0, 0, 0
    while time.time() < deadline - 2 and attempts < max_attempts and fails < 160:
        cvals = lags_of(h)
        L = 2 * n - 1
        top = np.argsort(cvals)[-25:] if L > 25 else None
        try:
            _, g, _ = smooth_grad(h, alphas[ai], top, cvals)
        except Exception:
            break
        pairs = grad_pairs(g, h, 4)
        pairs.extend(active_pairs(h, cvals, 4, 5))
        improved = False
        for src, dst in pairs:
            if time.time() > deadline - 1:
                break
            attempts += 1
            h2, c2 = pair_line_search(h, dst, src, bc)
            if c2 < bc - 1e-13:
                h, bc, improved, fails = h2, c2, True, 0
                break
        if not improved:
            fails += 1
            if fails == 60:
                ai, fails = (ai + 1) % 3, 0
    return h, bc

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n)
    f1, f2 = int(rng.randint(1, 7)), int(rng.randint(2, 11))
    u1, u2 = rng.uniform(0.0, 2.0 * np.pi), rng.uniform(0.0, 2.0 * np.pi)
    s = 0.6 * np.sin(2.0 * np.pi * f1 * x + u1) + 0.4 * np.sin(2.0 * np.pi * f2 * x + u2)
    s = s + 0.15 * rng.standard_normal(n)
    return project_feasible(0.5 + 0.5 * np.tanh(2.0 * s))

def optimize_run(h, deadline, quick=False):
    h = project_feasible(h.copy())
    bc = c5_of(h)
    if not quick:
        h, bc = focused(h, 1000.0, 120, 150, 2, deadline)
        h, bc = focused(h, 5000.0, 50, 150, 2, deadline)
    h, bc = focused(h, 20000.0, 18, 120, 3, deadline)
    h, bc = focused(h, 80000.0, 8, 100, 3, deadline)
    if time.time() < deadline - 40:
        h, bc = focused(h, 30000.0, None, 80, 1, deadline)
    if time.time() < deadline - 15:
        h, bc = pair_local(h, deadline, 100 if quick else 300)
    return h, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 25.0
    rng = np.random.RandomState(seed)
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = random_start(300, rng)
    parent = project_feasible(parent)
    best_h, best_c, best_n = parent.copy(), c5_of(parent), int(parent.size)
    if time.time() < deadline - 160:
        end = min(deadline - 160.0, t0 + 240.0)
        try:
            h1, c1 = optimize_run(parent, end, quick=(best_n > 600))
        except Exception:
            h1, c1 = parent, best_c
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, int(h1.size)
    for nc in [400, 500, 700, 900]:
        if time.time() > deadline - 100 or nc <= best_n or nc > 1000:
            continue
        xb, xn = np.linspace(0.0, 2.0, best_n), np.linspace(0.0, 2.0, nc)
        hs = project_feasible(np.interp(xn, xb, best_h))
        alloc = max(25.0, min(80.0, 0.10 * max(10.0, deadline - time.time())))
        try:
            hn, cn = optimize_run(hs, time.time() + alloc, quick=(nc > 500))
        except Exception:
            hn, cn = hs, c5_of(hs)
        if cn < best_c - 1e-12:
            best_h, best_c, best_n = hn, cn, int(hn.size)
    tries = 0
    while time.time() < deadline - 70:
        if tries % 6 == 5:
            hp = random_start(best_n, rng)
        else:
            scale = 0.05 + 0.07 * (tries % 8) + 0.02 * rng.rand()
            hp = project_feasible(best_h + rng.normal(0.0, min(0.6, scale), best_n))
        alloc = max(25.0, min(70.0, 0.08 * max(10.0, deadline - time.time())))
        if time.time() + alloc > deadline - 30:
            break
        try:
            hr, cr = optimize_run(hp, time.time() + alloc, quick=True)
        except Exception:
            hr, cr = hp, c5_of(hp)
        if cr < best_c - 1e-12:
            best_h, best_c, best_n = hr, cr, int(hr.size)
        tries += 1
        if tries >= 120:
            break
    if time.time() < deadline - 60:
        try:
            hr, cr = optimize_run(best_h, deadline, quick=True)
        except Exception:
            hr, cr = best_h, best_c
        if cr < best_c:
            best_h, best_c = hr, cr
    if time.time() < deadline - 2:
        hr, cr = pair_local(best_h, deadline, 120)
        if cr < best_c:
            best_h, best_c = hr, cr
    best_h = project_feasible(best_h)
    return best_h, float(c5_of(best_h)), int(best_h.size)
