def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(x, alpha, n, mu):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    Z = exp_c.sum()
    p = exp_c / Z
    obj = c_max + np.log(Z) / alpha + mu * (h.sum() - n / 2.0) ** 2
    
    m_range = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for i in range(n):
        k1 = i + m_range
        mask1 = (k1 >= 0) & (k1 < n)
        k2 = i - m_range
        mask2 = (k2 >= 0) & (k2 < n)
        t1 = 0.0
        if mask1.any():
            t1 = np.dot(p[mask1], 1.0 - h[k1[mask1]])
        t2 = 0.0
        if mask2.any():
            t2 = np.dot(p[mask2], h[k2[mask2]])
        grad[i] = dx * (t1 - t2) + 2 * mu * (h.sum() - n / 2.0)
    
    return obj, grad

result[3] = h0*d + h1*c + h2*b
result[4] = h0*e + h1*d + h2*c
result[5] = h1*e + h2*d
result[6] = h2*e

t2[0] = sum_j h[j] * p[0-j+2] = h0*p[2] + h1*p[1] + h2*p[0] = h0*c + h1*b + h2*a = result[2]. ✓
t2[1] = sum_j h[j] * p[1-j+2] = h0*p[3] + h1*p[2] + h2*p[1] = h0*d + h1*c + h2*b = result[3]. ✓
t2[2] = sum_j h[j] * p[2-j+2] = h0*p[4] + h1*p[3] + h2*p[2] = h0*e + h1*d + h2*c = result[4]. ✓



So the vectorized gradient is:
```python
t1 = np.convolve(1.0 - h, p[::-1], mode='full')[n-1:2*n-1]
t2 = np.convolve(h, p, mode='full')[n-1:2*n-1]
grad = dx * (t1 - t2)
```

This is O(n log n) using FFT (numpy uses FFT for convolve when inputs are large enough). Much faster than the O(n^2) loop!

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, alpha, n, mu):
    h = x  # already in [0,1] due to bounds
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    Z = exp_c.sum()
    obj = c_max + np.log(Z) / alpha
    penalty = mu * (h.sum() - n / 2.0) ** 2
    return obj + penalty

def lse_gradient(x, alpha, n, mu):
    h = x
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    Z = exp_c.sum()
    p = exp_c / Z
    
    t1 = np.convolve(1.0 - h, p[::-1], mode='full')[n-1:2*n-1]
    t2 = np.convolve(h, p, mode='full')[n-1:2*n-1]
    grad = dx * (t1 - t2) + 2 * mu * (h.sum() - n / 2.0)
    return grad

            val1 = np.max(corr + B * (1.0 - h[i]))

            hi_v = max(val0, val1) + 1e-10
