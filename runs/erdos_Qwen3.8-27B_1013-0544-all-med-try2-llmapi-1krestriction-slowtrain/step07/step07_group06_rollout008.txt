This is O(n log n) per coordinate update (for the sorting in each bisection step, or O(n) per step × log(precision) steps).

    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Precompute the structure
    # For shift k >= 0 (corresponding to correlate index m = k + n - 1):
    # g_k = sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])*dx
    # For coordinate i, the contribution is:
    #   h[i]*(1-h[i+k])*dx if i < n-k (i appears as first factor)
    #   h[i-k]*(1-h[i])*dx if i >= k (i appears as second factor, i.e., h[i] = h[(i-k)+k])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        overlaps = compute_overlaps(h, n)
        # Gradient of total max... actually for coordinate descent we just need per-coordinate
        # Let's just cycle through all coordinates
        
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            
            # Compute a_k, b_k for all shifts m
            # For shift k >= 0 (correlate index m = n-1+k):
            #   If i < n-k: h[i] appears as first factor: contribution = (1-h[i+k])*dx * h[i]
            #                → a_k gets (1-h[i+k])*dx from this term, but wait...
            #   
            # Let me think differently. For a fixed coordinate i with value v:
            # g_k(v) = [terms not involving h[i]] + [terms involving h[i]]
            #
            # For shift k >= 0 (m = n-1+k):
            #   g_k = sum_{j=0}^{n-k-1} h[j]*(1-h[j+k])*dx
            #   h[i] appears when: j=i (if i < n-k) → term is h[i]*(1-h[i+k])*dx = v*(1-h[i+k])*dx
            #   h[i] appears when: j+k=i (if i >= k, i.e., j=i-k) → term is h[i-k]*(1-h[i])*dx = h[i-k]*(1-v)*dx
            #

            #   a_k = (sum of all terms not involving h[i]) * 1  [constant part]
            #   b_k = (1-h[i+k])*dx if i < n-k, plus -h[i-k]*dx if i >= k
            #   Wait, more carefully:
            #   The terms involving v are:
            #     v*(1-h[i+k])*dx if i < n-k  → contributes (1-h[i+k])*dx to b_k
            #     h[i-k]*(1-v)*dx if i >= k    → contributes -h[i-k]*dx to b_k
            #                            and h[i-k]*dx to a_k
            #

            #       a_k = g_k(h) - b_k * h[i]  (reconstructing the constant)

            # g_k(h) = a_k + b_k * h[i]
            # a_k = g_k(h) - b_k * h[i]
            # where b_k = coefficient of v in g_k when h[i]=v
            
            v = h[i]
            b = np.zeros(2*n-1)

                m = n - 1 + k
                # shift k >= 0
                bk = 0.0
                if i < n - k:
                    bk += (1 - h[i+k]) * dx
                if i >= k:
                    bk -= h[i-k] * dx
                b[m] = bk

            # g_{-k} = sum_{j=k}^{n-1} h[j]*(1-h[j-k])*dx
            # h[i] appears when j=i (if i >= k) → h[i]*(1-h[i-k])*dx = v*(1-h[i-k])*dx
            # h[i] appears when j-k=i (if i+k < n, j=i+k) → h[i+k]*(1-h[i])*dx = h[i+k]*(1-v)*dx
            for k in range(1, n):
                m = n - 1 - k
                bk = 0.0
                if i >= k:
                    bk += (1 - h[i-k]) * dx
                if i + k < n:
                    bk -= h[i+k] * dx
                b[m] = bk
            
            # Now g_m = a_m + b_m * v for all m
            overlaps = compute_overlaps(h, n)  # these are g_m at current v
            a = overlaps - b * v  # a_m = g_m - b_m * v
            
            # Minimize max_m (a_m + b_m * v) over v in [0, 1]
            # Use binary search on c: find smallest c such that ∃ v ∈ [0,1] with a_m + b_m*v <= c for all m
            lo, hi = 0.0, 1.0
            # Actually binary search on c:
            c_lo = np.max(a)  # at v=0, f = max(a)
            c_hi = np.max(a + b)  # at v=1, f = max(a+b)

            for _ in range(40):

                # Check: is there v ∈ [0,1] with a_m + b_m*v <= c_mid for all m?
                # For b_m > 0: v <= (c_mid - a_m) / b_m
                # For b_m < 0: v >= (c_mid - a_m) / b_m
                # For b_m = 0: need a_m <= c_mid
                feasible = True
                v_lo = 0.0
                v_hi = 1.0
                for m in range(2*n-1):
                    if b[m] > 1e-15:
                        v_hi = min(v_hi, (c_mid - a[m]) / b[m])
                    elif b[m] < -1e-15:
                        v_lo = max(v_lo, (c_mid - a[m]) / b[m])
                    else:
                        if a[m] > c_mid + 1e-15:

                            break
                if v_lo > v_hi + 1e-15:
                    feasible = False
                if feasible:
                    c_hi = c_mid
                else:
                    c_lo = c_mid
            
            # Find optimal v (any feasible v at c = c_hi)
            c_opt = c_hi
            v_opt = 0.5  # default
            # Find a v that works
            v_lo = 0.0
            v_hi_bound = 1.0
            for m in range(2*n-1):

            v_opt = np.clip((v_lo + max(v_hi_bound, v_lo)) / 2, 0.0, 1.0)
            
            h[i] = v_opt
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

    """Find v in [0,1] minimizing max_m (a[m] + b[m]*v)."""
    # Binary search on c
    c_lo = max(a.max(), (a + b).min())  # lower bound
    c_hi = max(a.max(), (a + b).max())  # upper bound
    
    for _ in range(50):

        # b[m] > 0: v <= (c_mid - a[m]) / b[m]
        # b[m] < 0: v >= (c_mid - a[m]) / b[m]  [dividing by negative]

        neg = b < -1e-15

        v_lo_bound = 0.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_overlaps(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            continue

        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()

    mx = sh.max()
    if mx > 500:

def coord_b_vector(i, h, n):
    """Compute b[m] for all shifts m where g_m = a_m + b_m * h[i]."""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    b = np.zeros(n_shifts)
    
    # Positive shifts: m = n-1+k, k = 0..n-1
    for k in range(n):
        m = n - 1 + k
        val = 0.0
        if i + k < n:
            val += (1.0 - h[i + k]) * dx
        if i >= k:
            val -= h[i - k] * dx
        b[m] = val
    
    # Negative shifts: m = n-1-k, k = 1..n-1
    for k in range(1, n):
        m = n - 1 - k
        val = 0.0
        if i >= k:
            val += (1.0 - h[i - k]) * dx
        if i + k < n:
            val -= h[i + k] * dx
        b[m] = val
    
    return b

def coordinate_descent(h, n, deadline, max_sweeps=300, random_order=False):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        indices = np.random.permutation(n) if random_order else np.arange(n)
        
        for i in indices:
            if time.time() > deadline:
                return best_h, best_c
            
            v = h[i]
            b = coord_b_vector(i, h, n)
            overlaps = compute_overlaps(h, n)
            a = overlaps - b * v
            
            # Ternary search for optimal v in [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(a + b * m1) < np.max(a + b * m2):
                    hi = m2
                else:
                    lo = m1
            
            v_opt = (lo + hi) / 2.0
            h[i] = v_opt
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, max_iters=500):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 10.0), (200, 50.0), (1000, 200.0),
               (5000, 1000.0), (20000, 5000.0),
               (100000, 20000.0), (500000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search_exact(h, n, deadline, max_pairs=50000):
    """Try all pairs (i,j) with a small perturbation, using exact 2D optimization."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    
    # For each pair (i, j), compute the 2D linear coefficients
    # g_m(v_i, v_j) = a_m + b_mi * v_i + b_mj * v_j
    # Minimize max_m g_m over (v_i, v_j) in [0,1]^2
    # This is a 2D LP: min c s.t. a_m + b_mi*v_i + b_mj*v_j <= c for all m
    #   0 <= v_i <= 1, 0 <= v_j <= 1
    
    # Process pairs in order of gradient magnitude
    c_arr = compute_overlaps(h, n)
    # Simple: just try all pairs with a fixed small step
    deltas = [0.01, 0.005, 0.001, 0.0005, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best_h, best_c
                # Try moving mass from i to j and vice versa
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; best_h = h.copy()
                    else:
                        h[i] -= d; h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d; h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; best_h = h.copy()
                    else:
                        h[i] += d; h[j] -= d
    return best_h, best_c
