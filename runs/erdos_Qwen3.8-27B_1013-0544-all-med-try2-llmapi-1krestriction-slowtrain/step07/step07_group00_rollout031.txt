def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def soft_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_stop in range(2 * n - 1):
        if w[k_stop] < 1e-15:
            continue
        k = k_stop - (n - 1)
        wm = w[k_stop] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.exp(s - mx).sum())) / alpha)

def sm_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] < 1e-16:
            continue
        k = i - (n - 1)
        wi = w[i] * dx
        if k >= 0:
            L = n - k
            g[:L] += wi * (1.0 - x[k:])
            g[k:] -= wi * x[:L]
        else:
            j = -k
            L = n - j
            g[j:] += wi * (1.0 - x[:L])
            g[:L] -= wi * x[j:]
    return g

def lbfgs_run(h, n, deadline, max_alphas=None):
    best_c = compute_c5(h, n)
    best = h.copy()
    bs = [(0.0, 1.0)] * n
    alphas = max_alphas or [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(sm_obj, best, jac=lambda x: sm_grad(x, n, a),
                          args=(n, a), method='L-BFGS-B', bounds=bs,
                          options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14})
            h2 = project(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c, best = c2, h2
        except Exception:
            pass
    return best, best_c

def fista_run(h, n, eta0, n_iters, alpha, deadline):

        hp, hc = hc, hn

    best = h.copy()
    if steps is None:

def grad_pair_search(h, n, deadline, alpha=5000, step=0.001, radius=30):
    best_c = compute_c5(h, n)
    best = h.copy()
    for _ in range(100):  # max outer iterations
        if time.time() > deadline:
            break
        g = sm_grad(best, n, alpha)
        order = np.argsort(g)  # most negative first
        improved = False
        for idx in range(n):
            if time.time() > deadline:
                return best, best_c
            i = order[idx]
            if g[i] > -1e-12:
                break
            lo = max(0, idx - radius)
            hi = min(n, idx + radius + 1)
            for jidx in range(lo, hi):
                j = order[jidx]
                if i == j or g[j] - g[i] < 1e-10:
                    continue
                # Try h[i] += step, h[j] -= step
                if best[i] + step <= 1.0 and best[j] - step >= 0.0:
                    best[i] += step
                    best[j] -= step
                    c = compute_c5(best, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        best[i] -= step
                        best[j] += step
        if not improved:
            # Try smaller step
            step *= 0.5
            if step < 1e-6:
                break
            alpha = max(100, alpha * 0.5)
    return best, best_c

            if s < 1e-14:

    return m + (mx + np.log(np.exp(s - mx).sum())) / a
