```
import time, math
import numpy as np
from scipy.optimize import minimize, differential_evolution

_CACHE = {'on': False, 'sig': None, 'w': None}

def c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def c5(h, n):
    return float(np.max(c5_vec(h, n)))

def smooth_max(f, alpha):
    f = np.asarray(f, dtype=float)
    fmax = float(f.max())
    if alpha <= 0:
        return fmax, np.full(f.shape, 1.0 / f.size)
    sh = alpha * (f - fmax)
    mx = float(sh.max())
    ex = np.exp(sh - mx)
    s = float(ex.sum())
    if s <= 0:
        s = 1.0
        ex = np.zeros_like(f); ex[int(np.argmax(f))] = 1.0
    w = ex / s
    val = fmax + (mx + math.log(s)) / alpha
    return val, w

def combine_weights(f, alpha):
    _, w = smooth_max(f, alpha)
    mask = w > 1e-9
    if not mask.all():
        w = w.copy()
        w[~mask] = 0.0
        ss = float(w.sum())
        if ss > 0:
            w /= ss
        else:
            w = np.full(w.shape, 1.0 / w.size)
    return w

def weighted_f_grad(x, w, n):
    g = np.zeros(n, dtype=float)
    nz = np.nonzero(w > 1e-12)[0]
    if nz.size == 0:
        return g
    center = n - 1
    for m in nz:
        wm = float(w[m])
        if wm <= 0: continue
        k = int(m) - center
        if k >= 0:
            nv = n - k
            g[k:n] += wm * (1.0 - x[:nv])
            g[:nv] += -wm * x[k:n]
        else:
            kk = -k
            nv = n - kk
            g[:nv] += wm * (1.0 - x[kk:n])
            g[kk:n] += -wm * x[:nv]
    return g

def full_objective(x, n, alpha, mu):
    f = c5_vec(x, n)
    val, _ = smooth_max(f, alpha)
    w = combine_weights(f, alpha)
    _CACHE['on'] = True
    _CACHE['sig'] = (int(n), float(alpha), float(mu), float(x.sum()), float(x[0]), float(x[-1]))
    _CACHE['w'] = w
    s = float(x.sum() - 0.5 * n)
    val += mu * s * s
    return float(val)

def full_gradient(x, n, alpha, mu):
    s = float(x.sum() - 0.5 * n)
    sig = (int(n), float(alpha), float(mu), float(x.sum()), float(x[0]), float(x[-1]))
    if _CACHE['on'] and _CACHE['sig'] == sig:
        w = _CACHE['w']
    else:
        f = c5_vec(x, n)
        w = combine_weights(f, alpha)
    g = weighted_f_grad(x, w, n)
    g += 2.0 * mu * s
    return g

def finish_exact(h, target):
    rem = float(target - h.sum())
    if abs(rem) > 1e-14:
        if rem > 0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if abs(rem) <= 1e-14: break
                add = min(rem, 1.0 - h[idx])
                h[idx] += add
                rem -= add
        else:
            order = np.argsort(h)
            rem = -rem
            for idx in order:
                if rem <= 1e-14: break
                sub = min(rem, h[idx])
                h[idx] -= sub
                rem -= sub
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
```

```
def dct_basis(n,m):
    if m <=0: return np.empty((0,n))
    j = np.arange(n, dtype=float)
    k = np.arange(1, m+1, dtype=float)[:,None]
    B = np.sqrt(2.0/n) * np.cos(np.pi * k * (j[None,:] + 0.5) / n)
    B -= B.mean(axis=1, keepdims=True)
    norms = np.linalg.norm(B, axis=1)
    B = B / np.maximum(norms, 1e-12)[:,None]
    return B
```

```python
import time
import math
import numpy as np
from scipy.optimize import minimize, differential_evolution


def c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))


def c5(h, n):
    return float(np.max(c5_vec(h, n)))


def smooth_max(f, alpha):
    f = np.asarray(f, dtype=float)
    fmax = float(f.max())
    if alpha <= 0.0:
        return fmax, np.full(f.shape, 1.0 / float(f.size))
    sh = alpha * (f - fmax)
    mx = float(sh.max())
    ex = np.exp(sh - mx)
    s = float(ex.sum())
    if s <= 0.0:
        s = 1.0
        ex = np.zeros_like(f)
        ex[int(np.argmax(f))] = 1.0
    w = ex / s
    val = fmax + (mx + math.log(s)) / alpha
    return val, w


def gradient_weights(f, alpha):
    _, w = smooth_max(f, alpha)
    w = np.where(w > 1e-10, w, 0.0)
    ss = float(w.sum())
    if ss > 0.0:
        w = w / ss
    else:
        w = np.full(w.shape, 1.0 / float(w.size))
    if w.size > 250:
        k = 250
        idx = np.argpartition(w, w.size - k)[-k:]
        mask = np.zeros(w.shape, dtype=bool)
        mask[idx] = True
        w = np.where(mask, w, 0.0)
        ss = float(w.sum())
        if ss > 0.0:
            w = w / ss
    return w


def weighted_f_grad(x, w, n):
    g = np.zeros(n, dtype=float)
    nz = np.nonzero(w > 1e-12)[0]
    if nz.size == 0:
        return g
    center = n - 1
    for m in nz:
        wm = float(w[m])
        if wm <= 0.0:
            continue
        k = int(m) - center
        if k >= 0:
            nv = n - k
            g[k:n] += wm * (1.0 - x[:nv])
            g[:nv] += -wm * x[k:n]
        else:
            kk = -k
            nv = n - kk
            g[:nv] += wm * (1.0 - x[kk:n])
            g[kk:n] += -wm * x[:nv]
    return g


def full_objective(x, n, alpha, mu):
    f = c5_vec(x, n)
    val, _ = smooth_max(f, alpha)
    s = float(x.sum() - 0.5 * n)
    val += mu * s * s
    return float(val)


def full_gradient(x, n, alpha, mu):
    f = c5_vec(x, n)
    w = gradient_weights(f, alpha)
    g = weighted_f_grad(x, w, n)
    s = float(x.sum() - 0.5 * n)
    g += 2.0 * mu * s
    return g


def finish_exact(h, target):
    rem = float(target - h.sum())
    if rem > 1e-14:
        order = np.argsort(h)
        for idx in order:
            if rem <= 1e-14:
                break
            add = min(rem, 1.0 - h[idx])
            h[idx] += add
            rem -= add
    elif rem < -1e-14:
        rem = -rem
        order = np.argsort(h)[::-1]
        for idx in order:
            if rem <= 1e-14:
                break
            sub = min(rem, h[idx])
            h[idx] -= sub
            rem -= sub
    return np.clip(h, 0.0, 1.0)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = float(h.sum())
    if abs(s - target) > 1e-13:
        if s < target:
            lo = 0.0
            hi = float(np.max(1.0 - h)) if n else 0.0
            if hi <= 0.0:
                hi = 1.0
            for _ in range(60):
                mid = 0.5 * (lo + hi)
                s2 = float(np.clip(h + mid, 0.0, 1.0).sum())
                if s2 < target:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
        else:
            lo = 0.0
            hi = float(np.max(h)) if n else 0.0

                    lo = mid
                else:
                    hi = mid
            h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return finish_exact(h, target)


def lift_h(h, n_old, n_new):

def sym_candidate(h, n):

def golden_start(n):

    phi = (1.0 + math.sqrt(5.0)) / 2.0

    return project_feasible(h, n)

    if ones > target:

        order = np.argsort(-h[idx])[:target - ones]
        hb[idx[order]] = 1.0
    return project_feasible(hb, n)


def dct_basis(n, m):
    if m <= 0:
        return np.empty((0, n), dtype=float)
    j = np.arange(n, dtype=float)
    k = np.arange(1, m + 1, dtype=float)[:, None]
    B = np.sqrt(2.0 / n) * np.cos(np.pi * k * (j[None, :] + 0.5) / n)
    B -= B.mean(axis=1, keepdims=True)
    norms = np.linalg.norm(B, axis=1)
    B = B / np.maximum(norms, 1e-12)[:, None]
    return B


def lowdim_objective(w, n, B, fixed, mu):
    h = 0.5 + fixed + w @ B
    hc = np.clip(h, 0.0, 1.0)
    s = float(h.sum() - 0.5 * n)
    over = np.maximum(h - 1.0, 0.0)
    under = np.maximum(-h, 0.0)
    pen = float((over * over + under * under).sum())
    return float(c5(hc, n) + mu * pen + mu * s * s)
```
