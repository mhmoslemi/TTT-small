    best_c = compute_c5(best_h, n)
    
    # Phase 1: coarse with low alpha
    if time.time() < deadline - 40:
        for alpha in [500, 2000]:
            if time.time() > deadline - 40:
                break
            h_new = optimize_lbfgs(best_h, n, alpha, 200, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 2: medium alpha
    if time.time() < deadline - 30:
        for alpha in [10000, 50000]:
            if time.time() > deadline - 30:
                break
            h_new = optimize_lbfgs(best_h, n, alpha, 100, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 3: high alpha SLSQP
    if time.time() < deadline - 20:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > deadline - 20:
                break
            h_new = optimize_slsqp(best_h, n, alpha, 50, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 4: even higher alpha
    if time.time() < deadline - 10:
        h_new = optimize_slsqp(best_h, n, 5000000, 20, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 5: local search
    if time.time() < deadline - 2:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase A: Optimize at initial n
    if h_init is not None:
        alloc = min(450, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase B: Try other n values
    n_candidates = [200, 250, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(70, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase C: Restart with perturbations
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase D: Final high-precision polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = optimize_slsqp(best_h, n, 10000000, 10, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
