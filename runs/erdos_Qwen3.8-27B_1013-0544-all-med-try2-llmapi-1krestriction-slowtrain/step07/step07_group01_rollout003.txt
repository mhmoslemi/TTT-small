    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 20, 5.0), (500, 20, 10.0), (2000, 10, 20.0),
               (10000, 10, 50.0), (50000, 5, 200.0), (200000, 5, 500.0),
               (1000000, 3, 2000.0)]
    for alpha, top_k, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_smo_obj, best_h, jac=_smo_grad,
                          args=(n, alpha, top_k, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'gtol': 1e-14,
                                   'ftol': 1e-16, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=100000, top_k=5, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _smo_eq, 'jac': _smo_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smo_obj_nop, h, args=(n, alpha, top_k), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_descent(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [200, 1000, 5000, 20000, 100000]
    for alpha in alphas:
        lr = 0.05
        for _ in range(3000):
            if time.time() > deadline:
                return best_h, best_c
            g = active_gradient(h, n, alpha, top_k=10)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            g = g / g_norm
            improved = False
            for trial in range(50):
                h_try = project_feasible(h - lr * g, n)
                c_try = compute_c5(h_try, n)
                if c_try <= best_c - 1e-16:
                    h = h_try
                    best_c = c_try
                    best_h = h.copy()
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                lr *= 0.3
                if lr < 1e-14:
                    break
    return best_h, best_c

def local_search(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # Gradient-directed moves
    alphas = [500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = active_gradient(h, n, alpha, top_k=8)
            for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_try = project_feasible(h - step * g, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-16:
                    h = h_try
                    best_c = c_try
                    best_h = h.copy()
                    improved = True
                    break
            # Random perturbation
            for scale in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
                h_try = project_feasible(h + rng.randn(n) * scale, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-16:
                    h = h_try
                    best_c = c_try
                    best_h = h.copy()
                    improved = True
            # Pair moves
            for _ in range(50):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c_t = compute_c5(h, n)
                        if c_t < best_c - 1e-16:
                            best_c = c_t; best_h = h.copy(); improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c_t = compute_c5(h, n)
                        if c_t < best_c - 1e-16:
                            best_c = c_t; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    # Directed pair search with small steps
    while time.time() < deadline:
        improved = False
        for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            g = active_gradient(h, n, 10000, top_k=5)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(max(0, i-30), min(n, i+31)):
                    if i == j:
                        continue
                    move = -d * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    hi_v = h[i] + move
                    hj_v = h[j] - move
                    if 0.0 <= hi_v <= 1.0 and 0.0 <= hj_v <= 1.0:
                        h[i] = hi_v; h[j] = hj_v
                        c_t = compute_c5(h, n)
                        if c_t < best_c - 1e-16:
                            best_c = c_t; best_h = h.copy(); improved = True
                        else:
                            h[i] -= move; h[j] += move
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B on initial grid
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=300)

    # Phase 2: SLSQP on initial grid
    if time.time() < deadline - 400:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 400:
                break
            h2, c2 = slsqp_phase(best_h, n_init, deadline - 400, alpha=alpha, top_k=5, maxiter=800)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient descent + local search on initial grid
    if time.time() < deadline - 300:
        h3, c3 = gradient_descent(best_h, n_init, min(time.time() + 60, deadline - 300), rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    if time.time() < deadline - 250:
        best_h, best_c = local_search(best_h, n_init, min(time.time() + 150, deadline - 250), rng)

    # Phase 4: Upscale to finer grid
    n_fine = 600
    t4_start = time.time()
    x_old = np.linspace(0, 2, n_init)
    x_new = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_new, x_old, best_h)
    h_fine = project_feasible(h_fine, n_fine)
    c_fine = compute_c5(h_fine, n_fine)
    if c_fine < best_c:
        best_c = c_fine
        best_h = h_fine.copy()

    # Phase 5: Optimize on fine grid
    t5_end = deadline - 80
    if time.time() < t5_end - 60:
        h5, c5 = lbfgs_phase(h_fine, n_fine, min(time.time() + 150, t5_end - 100), max_iters=200)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < t5_end - 80:
            h5b, c5b = slsqp_phase(best_h, n_fine, min(time.time() + 60, t5_end - 80),
                                   alpha=200000, top_k=5, maxiter=600)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
        if time.time() < t5_end - 40:
            best_h, best_c = local_search(best_h, n_fine, t5_end - 40, rng)

    # Phase 6: Try n=400 as alternative
    n_alt = 400
    if time.time() < deadline - 60:
        x_alt = np.linspace(0, 2, n_alt)
        src_n = len(best_h)
        x_src = np.linspace(0, 2, src_n)
        h_alt = np.interp(x_alt, x_src, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        h_alt, c_alt = lbfgs_phase(h_alt, n_alt, min(time.time() + 40, deadline - 40), max_iters=150)
        if time.time() < deadline - 30:
            h_alt, c_alt = local_search(h_alt, n_alt, deadline - 30, rng)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt

    # Phase 7: Final polish on best
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = local_search(best_h, nf, deadline - 2, rng)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
