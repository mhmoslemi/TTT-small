        shifted = alpha * (g - m)
        sm = shifted.max()

            zeros[ai] = j; ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                stall = 0
        else:
            stall += 1
        if stall > 3000:
            T = min(T * 2.5, 0.08)
            stall = 0
        T *= cool
    return best_h, best_c

def slsqp_refine(h, n, deadline, alphas=(100, 1000, 10000, 100000, 1000000)):
    dx = 2.0 / n
    best = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            logsum = sm + np.log(np.exp(shifted - sm).sum())
            return m + logsum / alpha
        def cfun(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cfun}]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best = h.copy()
        except Exception:
            pass
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    # Phase 1: Greedy convergence
    p1 = t0 + 0.04 * total
    hb, cb = greedy_binary(to_binary(best_h, best_n), best_n, p1)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: Main annealing (long, adaptive T)
    p2 = t0 + 0.38 * total
    hb, cb = anneal_binary(best_h, best_n, p2, seed=seed, T0=0.05, cool=0.99999)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 3: SLSQP refinement
    p3 = t0 + 0.52 * total
    best_h, best_c = slsqp_refine(best_h, best_n, p3, (100, 1000, 10000, 100000, 1000000))
    best_n = len(best_h)
    # Phase 4: Second anneal from refined
    p4 = t0 + 0.72 * total
    hb, cb = anneal_binary(to_binary(best_h, best_n), best_n, p4, seed=seed + 777, T0=0.02, cool=0.99999)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 5: SLSQP polish
    p5 = t0 + 0.82 * total
    best_h, best_c = slsqp_refine(best_h, best_n, p5, (100000, 1000000, 10000000))
    best_n = len(best_h)
    # Phase 6: Lower resolutions
    p6 = dl - 5
    for nh in [200, 250, 300]:
        if time.time() > p6 - 15:
            break
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        sub_dl = min(time.time() + 25, p6 - 5)
        hb2, cb2 = anneal_binary(to_binary(hh, nh), nh, sub_dl, seed=seed + 3000 + nh, T0=0.05, cool=0.99998)
        if cb2 < best_c:
            best_c, best_h, best_n = cb2, hb2.copy(), nh
        if time.time() < p6 - 5:
            hh = best_h if best_n == nh else hh
            hh, cb2 = slsqp_refine(hh, nh, p6 - 3, (100000, 1000000))
            if cb2 < best_c:
                best_c, best_h, best_n = cb2, hh.copy(), nh
    # Phase 7: Final ILS-like restart
    if time.time() < dl - 8:
        h_try = best_h + 0.01 * rng.randn(best_n)
        h_try = project_feasible(h_try, best_n)
        h_try, c_try = anneal_binary(to_binary(h_try, best_n), best_n, dl - 3, seed=seed + 9999, T0=0.03, cool=0.99999)
        if c_try < best_c:
            best_h, best_c, best_n = h_try.copy(), c_try, best_n
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
