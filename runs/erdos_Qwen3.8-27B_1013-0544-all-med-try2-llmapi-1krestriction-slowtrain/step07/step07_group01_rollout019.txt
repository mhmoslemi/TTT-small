                if i == j:
                    continue
                const = h[i] + h[j]
                t_lo = max(0.0, const - 1.0)
                t_hi = min(1.0, const)
                if t_hi - t_lo < 1e-13:
                    continue
                old_i, old_j = h[i], h[j]
                t_opt = exact_pair_optimal(h, i, j, n)
                if abs(t_opt - old_i) < 1e-11:
                    h[i], h[j] = old_i, old_j
                    continue
                h[i] = t_opt
                h[j] = const - t_opt
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved_total = True
                else:
                    h[i], h[j] = old_i, old_j
    return h, best_c

def lbfgs_refine(h, n, deadline, alphas=None, maxiter=300):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000, 200000]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = 0.1
        try:
            res = minimize(_smooth_obj, best_h, jac=_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': maxiter * 8,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B (moderate, to get a good starting point)
    if time.time() < deadline - 400:
        t1 = min(time.time() + 150, deadline - 400)
        h, c = lbfgs_refine(h, n, t1, alphas=[50, 200, 1000, 5000, 50000], maxiter=200)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Exact pair descent - cascading windows (main phase)
    h = best_h.copy()
    windows = [n, 150, 80, 40, 20, 10, 5, 3, 2, 1]
    for w in windows:
        if time.time() > deadline - 100:
            break
        h, c = pair_descent(h, n, deadline - 100, rng, max_sweeps=500, window=w)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: L-BFGS-B with high alpha to exploit smooth max more sharply
    if time.time() < deadline - 80:
        t3 = min(time.time() + 120, deadline - 80)
        h = best_h.copy()
        h, c = lbfgs_refine(h, n, t3, alphas=[50000, 200000, 1000000], maxiter=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: More pair descent on refined solution
    if time.time() < deadline - 30:
        h = best_h.copy()
        for w in [30, 10, 5, 3, 2, 1]:
            if time.time() > deadline - 30:
                break
            h, c = pair_descent(h, n, deadline - 30, rng, max_sweeps=300, window=w)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 5: Final L-BFGS-B + pair
    if time.time() < deadline - 10:
        h = best_h.copy()
        h, c = lbfgs_refine(h, n, deadline - 10, alphas=[500000], maxiter=200)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < deadline - 3:
            h, c = pair_descent(h, n, deadline - 2, rng, max_sweeps=100, window=3)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 6: Try slightly different n
    if time.time() < deadline - 40:
        for n_try in [n - 2, n + 2, n + 4]:
            if n_try < 100 or time.time() > deadline - 20:
                continue
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_try, c_try = lbfgs_refine(h_interp, n_try, min(time.time() + 20, deadline - 10),
                                        alphas=[5000, 50000], maxiter=150)
            if time.time() < deadline - 5:
                h_try, c_try = pair_descent(h_try, n_try, deadline - 3, rng, max_sweeps=50, window=5)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                n = n_try

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
