def binary_anneal(h_init,n,deadline,seed,max_iters=200000, T0=8.0, cool=0.99999):
    h=to_binary(h_init,n)
    inv=1-h
    cur=float(np.correlate(h,inv,'full').max())
    best_corr=cur; best_h=h.copy()
    ones=np.nonzero(h==1)[0]; zeros=np.nonzero(h==0)[0]
    rng=np.random.RandomState(seed)
    T=T0
    no_improve=0; stall_limit=20000
    for it in range(max_iters):
       if it % 1000 ==0:
          if time.time()>deadline: break
          if no_improve > stall_limit: break
       ai=rng.randint(len(zeros)); bi=rng.randint(len(ones))
       i=zeros[ai]; j=ones[bi]
       if i==j: continue
       h[i]=1.0; inv[i]=0.0; h[j]=0.0; inv[j]=1.0
       new=float(np.correlate(h,inv,'full').max())
       if new <= cur or (T>1e-12 and rng.random() < np.exp((cur-new)/T)):
           cur=new; zeros[ai]=j; ones[bi]=i
           if new < best_corr - 1e-12:
                best_corr=new; best_h=h.copy(); no_improve=0
           else: no_improve += 1
       else:
           h[i]=0.0; inv[i]=1.0; h[j]=1.0; inv[j]=0.0
           no_improve += 1
       T*=cool
    return best_h, 2.0*best_corr/n

def smooth_grad(h):
    alpha = _SM_ALPHA
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    p = np.exp(z)
    S = float(p.sum())
    if S <= 0.0:
       return np.zeros(n)
    r = p / S
    if n == 1:
       return r[0]*(1.0-2.0*h)  # c0=h(1-h). But n>1.
    P = r[n:]
    Mr = r[:n-1][::-1]
    grad = np.zeros(n)
    r0 = r[n-1]
    if r0 != 0.0:
       grad += r0 * (1.0 - 2.0 * h)
    if n > 1:
       prefixP = np.cumsum(P)
       prefixM = np.cumsum(Mr)
       pos_w = np.zeros(n)
       pos_w[1:] = prefixP
       neg_pref = np.empty(n)
       neg_pref[0] = 0.0
       neg_pref[1:] = prefixM
       idx = n - 1 - np.arange(n)

          b = h[d:]
          grad[:L] -= qd * b

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15: break
            h = h + (diff/s) * room
        else:
            s = h.sum()
            if s <= 1e-15: break
            h = h - (-diff/s) * h  # h + diff/s * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) <= 1e-13: break
        if diff > 0:
            idx = np.flatnonzero(h < 1.0)
        else:
            idx = np.flatnonzero(h > 0.0)
        if idx.size == 0: break
        step = diff / float(idx.size)
        h[idx] += step
        np.clip(h, 0.0, 1.0, out=h)
    return h

import numpy as np
import time
from scipy.optimize import minimize

_SM_ALPHA = 1.0

ALPHAS_MAIN = (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0, 500000.0, 2000000.0)
MAXITERS_MAIN = (120, 160, 220, 320, 420, 520, 650, 800, 1100)

ALPHAS_SHORT = (50.0, 500.0, 5000.0, 50000.0, 500000.0)
MAXITERS_SHORT = (90, 130, 200, 280, 380)

ALPHAS_FINAL = (50000.0, 500000.0, 5000000.0, 50000000.0)
MAXITERS_FINAL = (300, 500, 700, 1000)


def compute_c(h, n):
    if n <= 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h = h + (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            idx = np.flatnonzero(h < 1.0)
        else:
            idx = np.flatnonzero(h > 0.0)
        if idx.size == 0:
            break
        h[idx] += diff / float(idx.size)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    hb = np.zeros(n)
    k = n // 2
    order = np.argsort(h)
    if k > 0:
        hb[order[-k:]] = 1.0
    if n % 2 == 1:
        if k == 0:
            hb[order[0]] = 0.5
        else:
            hb[order[-k - 1]] = 0.5
    return hb


def random_binary(n, seed):
    h = np.zeros(n)
    k = n // 2
    if k > 0:
        rng = np.random.RandomState(seed)
        h[rng.choice(n, k, replace=False)] = 1.0
    if n % 2 == 1:
        zeros = np.flatnonzero(h == 0.0)
        if zeros.size > 0:
            h[np.random.RandomState(seed + 1).choice(zeros)] = 0.5
    return h


def sum_constraint(x):
    return x.sum() - 0.5 * float(len(x))


def smooth_value(h):
    alpha = float(_SM_ALPHA)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    p = np.exp(z)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return m
    return m + float(np.log(s)) / alpha


def smooth_grad(h):
    alpha = float(_SM_ALPHA)
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    p = np.exp(z)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n)
    r = p / s
    grad = np.zeros(n)

    r0 = float(r[n - 1])
    if r0 != 0.0:
        grad += r0 * (1.0 - 2.0 * h)

    if n > 1:
        P = r[n:]
        M = r[:n - 1][::-1]

        prefixP = np.cumsum(P)
        prefixM = np.cumsum(M)

        pos_w = np.zeros(n)
        pos_w[1:] = prefixP

        neg_pref = np.empty(n)
        neg_pref[0] = 0.0
        neg_pref[1:] = prefixM
        idx = n - 1 - np.arange(n)
        grad += pos_w + neg_pref[idx]

        q = P + M
        nz = np.nonzero(q > 1e-18)[0]
        for ii in nz:
            d = int(ii) + 1
            qd = float(q[ii])
            L = n - d
            if L <= 0:
                continue
            grad[:L] -= qd * h[d:]
            grad[d:] -= qd * h[:L]
    return grad


def slsqp_smooth(h, alpha, maxiter, ftol=1e-15):
    global _SM_ALPHA
    h = project_feasible(h, len(h))
    n = len(h)
    _SM_ALPHA = float(alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_value, h, method='SLSQP', jac=smooth_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': float(ftol), 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def polish(h, n, deadline, alphas, maxiters):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c(h, n)
    for ai, alpha in enumerate(alphas):
        if time.time() > float(deadline) - 1.0:
            break
        maxit = int(maxiters[min(ai, len(maxiters) - 1)])
        h_new = slsqp_smooth(h, float(alpha), maxit, 1e-15)
        c_new = compute_c(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c, h


def binary_anneal(h_init, n, deadline, seed, max_iters=200000):
    h = to_binary(np.asarray(h_init, dtype=float), n)
    inv = 1.0 - h
    best_corr = float(np.max(np.correlate(h, inv, mode='full')))
    best_h = h.copy()
    ones = np.flatnonzero(h == 1.0)
    zeros = np.flatnonzero(h == 0.0)
    if ones.size == 0 or zeros.size == 0:
        return best_h, 2.0 * best_corr / float(n)

    rng = np.random.RandomState(seed)
    cur = best_corr
    T = 10.0
    cool = 0.99999

    for it in range(int(max_iters)):
        if (it & 1023) == 0 and time.time() > float(deadline):
            break
        ai = int(rng.randint(zeros.size))
        bi = int(rng.randint(ones.size))
        i = int(zeros[ai])
        j = int(ones[bi])

        h[i] = 1.0
        inv[i] = 0.0
        h[j] = 0.0
        inv[j] = 1.0
        new = float(np.max(np.correlate(h, inv, mode='full')))

        if new <= cur or (T > 1e-12 and rng.random() < np.exp((cur - new) / T)):
            cur = new
            zeros[ai] = j
            ones[bi] = i
            if new < best_corr - 1e-12:
                best_corr = new
                best_h = h.copy()
        else:
            h[i] = 0.0
            inv[i] = 1.0
            h[j] = 1.0
            inv[j] = 0.0
        T *= cool

    return best_h, 2.0 * best_corr / float(n)


def resample_h(h, no, nn):
    h = np.asarray(h, dtype=float)
    if no == nn:
        return project_feasible(h.copy(), nn)
    xo = (np.arange(no) + 0.5) * (2.0 / float(no))
    xn = (np.arange(nn) + 0.5) * (2.0 / float(nn))
    return project_feasible(np.interp(xn, xo, h), nn)

    hb = None

    if hb is not None:

    for st in starts:
        if time.time() > p2 - 90.0:
