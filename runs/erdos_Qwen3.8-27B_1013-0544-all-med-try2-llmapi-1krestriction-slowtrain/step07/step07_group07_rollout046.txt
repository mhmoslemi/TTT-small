Given the current h and the active lag k*, the direction that most rapidly decreases c_{k*} is -grad_{k*}. But we need to stay feasible (0 ≤ h ≤ 1, sum = n/2). We can do a projected line search:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _active_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    s = alpha * (active - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp_opt(h, n, alpha, maxiter, ftol, deadline):
    if time.time() > deadline:
        return h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_slsqp(h, n, top_m, alpha, maxiter, ftol, deadline):
    if time.time() > deadline:
        return h.copy()
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_active_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += 1.0 - h[:n - m]
    else:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] -= h[:n - m]
    return grad

def targeted_local_search(h, n, deadline, max_rounds=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        for _ in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            c_all = compute_all_lags(h, n)
            k_star = int(np.argmax(c_all))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if lag > 0:
                contrib = h[:n - m] * (1.0 - h[m:])
                order = np.argsort(-contrib)
            else:
                contrib = (1.0 - h[:n - m]) * h[m:]
                order = np.argsort(-contrib)
            improved = False
            for idx in order[:300]:
                if time.time() > deadline:
                    return h, best_c
                i = int(idx)
                j = i + m
                if lag > 0:
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4)
                        if delta < 1e-9:
                            continue

                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:

                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.4)
                        if delta < 1e-9:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                else:
                    if h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.4)
                        if delta < 1e-9:

                    elif h[i] > h[j]:

            if not improved:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Multi-round SLSQP + local search interleaving
    alphas = [5000, 50000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        alloc = min(100, (deadline - time.time()) * 0.3)
        h_new = slsqp_opt(h, n, alpha, maxiter=4000, ftol=1e-15, deadline=time.time() + alloc)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if time.time() < deadline - 15:
            h_ls, c_ls = targeted_local_search(h, n, time.time() + 30)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls.copy()
            h = h_ls
    # Active set optimization
    if time.time() < deadline - 15:
        for top_m in [3, 5]:
            if time.time() > deadline - 10:
                break
            h_new = active_slsqp(h, n, top_m, 200000, 3000, 1e-15, time.time() + 40)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    # Final local search
    if time.time() < deadline - 5:
        best_h, best_c = targeted_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
