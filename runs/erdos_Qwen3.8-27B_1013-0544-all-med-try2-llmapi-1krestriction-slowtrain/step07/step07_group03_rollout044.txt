    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'factr': 1e8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_delta(h, i, j, m_range, n):
    """Compute correlation change for swapping h[i]=0 -> 1 and h[j]=1 -> 0."""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < len(delta):
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < len(delta):
        delta[k2] += 1.0
    return delta

def binary_local_search(h_init, n, deadline, max_sweeps=1000):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0].astype(int))
    zeros_list = list(np.where(h == 0)[0].astype(int))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = compute_delta(h, i, j, m_range, n)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-16:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=8000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].astype(np.int32).copy()
    zeros_arr = np.where(h == 0)[0].astype(np.int32).copy()
    T = 0.01
    cool = 0.99998
    scale = 8000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = compute_delta(h, i, j, m_range, n)
        nc = (corr + delta).max() * dx
        if nc <= cur_c:
            accept = True
        elif T > 1e-14:
            arg = min(60.0, scale * (cur_c - nc) / T)
            accept = (rng.random() < np.exp(arg))
        else:
            accept = False
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr = corr + delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Aggressive log-sum-exp SLSQP
    p1 = t0 + 0.30 * total
    h = best_h.copy()
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 100000000]
    for alpha in alphas:
        if time.time() > p1 - 15:
            break
        h = lse_slsqp(h, n, alpha, 4000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: p-norm optimization (different landscape)
    p2 = t0 + 0.45 * total
    h = best_h.copy()
    for p in [5, 20, 100, 500, 2000]:
        if time.time() > p2 - 10:
            break
        h = pnorm_slsqp(h, n, p, 3000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: L-BFGS-B refinement (faster per iter)
    p3 = t0 + 0.55 * total
    h = best_h.copy()
    for alpha in [100000, 1000000, 10000000]:
        if time.time() > p3 - 10:
            break
        h = lbfgs_lse(h, n, alpha, 5000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Binary local search
    p4 = t0 + 0.65 * total
    hb, cb = binary_local_search(best_h, n, p4)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 5: Annealing
    p5 = t0 + 0.80 * total
    ha, ca = anneal_binary(best_h, n, p5, seed=seed)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 6: Re-optimize continuous from best binary
    p6 = t0 + 0.92 * total
    h = best_h.copy()
    for alpha in [100000, 1000000, 10000000, 100000000]:
        if time.time() > p6 - 8:
            break
        h = lse_slsqp(h, n, alpha, 4000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    for p in [100, 1000]:
        if time.time() > p6 - 5:
            break
        h = pnorm_slsqp(h, n, p, 2000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Final perturb + polish
    if time.time() < dl - 3:
        for trial in range(4):
            if time.time() > dl - 3:
                break
            for noise_scale in [0.003, 0.01]:
                if time.time() > dl - 3:
                    break
                h = project_feasible(best_h + noise_scale * rng.randn(n), n)
                h = lse_slsqp(h, n, 100000000, 5000, 1e-18)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
