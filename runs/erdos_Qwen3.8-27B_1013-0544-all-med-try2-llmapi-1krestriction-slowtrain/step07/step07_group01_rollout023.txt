        if abs(diff) < 1e-14:

            room = 1.0 - h
            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    if mx > 500:

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    hx = np.clip(x, 0, 1)
    c = compute_c5_vec(hx, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    hx = np.clip(x, 0, 1)
    c = compute_c5_vec(hx, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx) if mx > 500 else np.exp(sh)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nlags = 2 * n - 1
    for llag in range(nlags):
        if w[llag] < 1e-14:
            continue
        k = llag - (n - 1)

        if k >= 0:

            grad[:nv] += wm * (-hx[kk:])
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new

        except:
            pass
    return h, best_c

def solve_1d_minmax(c, v, d_lo, d_hi, n_iter=50):
    """Solve min d in [d_lo,d_hi] of max_m (c[m] + d*v[m])"""
    best_d = 0.0
    best_val = np.max(c)
    lo = d_lo
    hi = d_hi
    for _ in range(n_iter):
        m1 = lo + (hi - lo) * 0.4

        val = np.max(c + d * v)
        if val < best_val:
            best_val = val
            best_d = d
    return best_d, best_val

            tot = h.sum()

                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_smooth_alpha(c, target):
    """Adaptive alpha: larger when spread is small"""
    spread = c.max() - c.min()
    if spread < 1e-10:
        return 1e6
    alpha = target / (spread * len(c))
    return min(alpha, 1e7)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nlags = 2 * n - 1
    for llag in range(nlags):
        if w[llag] < 1e-14:
            continue
        k = llag - (n - 1)
        wm = w[llag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    hx = np.clip(x, 0, 1)
    c = compute_c5_vec(hx, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(np.clip(x, 0, 1), n, alpha)
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

def lbfgs_run(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50,5,100),(200,20,100),(1000,100,150),(5000,500,200),
               (20000,2000,200),(100000,10000,300),(500000,50000,400),
               (2000000,200000,400)]
    for alpha, mu, niter in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': niter, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': niter*20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except:
            pass
    return h, best_c

def precompute_grads(h, n):
    """Precompute grad_i[m] for all i, m. Returns array of shape (n, 2n-1)."""
    nlags = 2 * n - 1
    grads = np.zeros((n, nlags))
    for i in range(n):
        for llag in range(nlags):
            k = llag - (n - 1)
            val = 0.0
            # h[i] as first factor: partner is i-k
            pk = i - k
            if 0 <= pk < n:
                val += (1.0 - h[pk])
            # h[i] as second factor (1-h[i]): partner is i+k
            pk = i + k
            if 0 <= pk < n:
                val -= h[pk]
            grads[i, llag] = val
    return grads

def vectorized_pair_search(h, n, deadline, c_vec, d_values):
    """
    For each pair (i,j), try moving h[i]+=d, h[j]-=d for d in d_values.
    Use vectorized operations.
    Returns (h, best_c, c_vec)
    """
    best_c = float(c_vec.max())
    nlags = 2 * n - 1
    dx = 2.0 / n
    
    # Precompute grads
    grads = precompute_grads(h, n)  # shape (n, nlags)
    
    improved = True
    sweep = 0
    while improved and time.time() < deadline:
        improved = False
        sweep += 1
        best_delta = 0.0
        best_ijk = None
        
        for i in range(n):
            if time.time() > deadline:
                break
            # v_j = grads[i] - grads[j] for all j simultaneously
            v = grads[i] - grads  # shape (n, nlags), row j is v_j
            
            # Bounds: d must satisfy h[i]+d <= 1 and h[j]-d >= 0
            # d <= 1-h[i] and d <= h[j]
            d_max_i = 1.0 - h[i]
            d_max_j = h  # d <= h[j] for each j
            
            for d in d_values:
                if d > d_max_i:
                    break
                # For each j, check if d <= h[j]
                valid = d <= h + 1e-12
                if not valid.any():
                    continue
                # c_new[j] = max_m (c_vec[m] + d * v[j, m])
                # = max_m (c_vec[m] + d * (grads[i,m] - grads[j,m]))
                c_new = c_vec + d * (grads[i][np.newaxis, :] - grads)  # shape (n, nlags)
                c_new = np.max(c_new, axis=1)  # shape (n,)
                c_new[~valid] = np.inf
                # Also exclude j == i
                c_new[i] = np.inf
                
                j_best = np.argmin(c_new)
                if c_new[j_best] < best_c - 1e-14:
                    best_delta = best_c - c_new[j_best]
                    best_ijk = (i, j_best, d)
        
        if best_ijk is not None:
            i, j, d = best_ijk
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            c_vec = compute_c5_vec(h, n)
            best_c = float(c_vec.max())
            improved = True
            # Recompute grads
            grads = precompute_grads(h, n)
    
    return h, best_c, c_vec

OK let me also think about an alternative: instead of the vectorized pair search, use a simpler "coordinate descent" approach:

For each coordinate i:
- Try h[i] += d (and redistribute -d proportionally to all other coordinates)
- Try h[i] -= d (and redistribute +d proportionally)
- The change in c is linear in d
- Solve 1D min-max

This is O(n) per coordinate (for the 1D solve) + O(n²) for the gradient computation. Total per sweep: O(n² + n*n) = O(n²). Much faster!

The "proportional redistribution" means: if h[i] changes by +d, then h[j] changes by -d * h[j] / (total - h[i]) for j ≠ i.

The change in c[m] is:
Δc[m] = d * (d/dh[i] c[m]) + sum_{j≠i} (-d * h[j]/(total-h[i])) * (d/dh[j] c[m])
