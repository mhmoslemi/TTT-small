def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h, n)) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_lse(h, n, alpha):
    dx = 2.0 / n
    corr = compute_corr(h, n)
    g = corr * dx
    shifted = alpha * g
    shifted -= shifted.max()
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)

        wk = w[k + n - 1]
        if wk < 1e-13:
            continue
        lo1 = max(0, k)
        hi1 = n if k >= 0 else n
        hi1 = min(n, n + k) if k >= 0 else n
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1-k:hi1-k])

        hi2 = n if k < 0 else min(n, n - k)
        if hi2 > lo2:
