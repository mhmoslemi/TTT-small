```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def adam_optimize(h, n, deadline, alpha_init=100.0, alpha_final=500000.0, lr_init=0.01):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 5000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.5 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def topk_pair_search(h, n, deadline, deltas, K=50, topJ=15):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, 10000)
            # Top K by |gradient|
            idx = np.argsort(-np.abs(g))[:K]
            for i in idx:
                if time.time() > deadline:
                    return h, best_c
                # Try increase h[i]
                if h[i] + d <= 1.0:
                    cands = np.where((h >= d) & (np.arange(n) != i))[0]
                    if len(cands) > 0:
                        # Prefer candidates with negative gradient (or low |grad|)
                        cands = cands[np.argsort(g[cands])][:topJ]
                        for j in cands:
                            h[i] += d; h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c; improved = True
                            else:
                                h[i] -= d; h[j] += d
                # Try decrease h[i]
                if h[i] - d >= 0.0:
                    cands = np.where((h <= 1.0 - d) & (np.arange(n) != i))[0]
                    if len(cands) > 0:
                        cands = cands[np.argsort(-g[cands])][:topJ]
                        for j in cands:
                            h[i] -= d; h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c; improved = True
                            else:
                                h[i] += d; h[j] -= d
            if not improved:

    return h, best_c

def binary_sa(h, n, deadline):
    """Simulated annealing on binary version."""
    # Round to binary while maintaining sum
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    target = n // 2
    current_ones = int(h_bin.sum())
    if current_ones > target:
        # Remove some ones (those with lowest "importance" - use h value as proxy)
        one_idx = np.where(h_bin == 1)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[one_idx])
        for idx in order[:current_ones - target]:
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_active(h, n, thresh_frac=0.002):

                    continue
                pairs.sort(key=lambda x: -x[2])
                for i, j, _ in pairs[:15]:
                    if i == j:
                        continue
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-15:
                            best_c = cval
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[j] - d >= -1e-12 and h[i] + d <= 1.0 + 1e-12:
                        h[j] -= d
                        h[i] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-15:
                            best_c = cval
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Adam baseline (100s)
    t1 = min(t0 + 100, deadline - 600)
    best_h, best_c = adam_optimize(best_h, n, t1, alpha_init=50, alpha_final=300000, lr_init=0.012)

    # Phase 2: Top-K active gradient search (300s)
    t2 = min(t1 + 300, deadline - 350)
    best_h, best_c = topk_search(best_h, n, t2,
                                  [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                  K=60, topJ=20, use_active=True)

    # Phase 3: Targeted lag search (150s)
    t3 = min(t2 + 150, deadline - 200)
    h_t, c_t = targeted_lag_search(best_h, n, t3,
                                    [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    if c_t < best_c:
        best_h, best_c = h_t, c_t

    # Phase 4: Adam refinement (80s)
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 120)
        h4, c4 = adam_optimize(best_h, n, t4, alpha_init=200, alpha_final=500000, lr_init=0.005)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Fine top-K search (150s)
    if time.time() < deadline - 80:
        t5 = min(time.time() + 150, deadline - 80)
        best_h, best_c = topk_search(best_h, n, t5,
                                      [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
                                      K=80, topJ=25, use_active=True)

    # Phase 6: Multi-start perturbation (80s)
    if time.time() < deadline - 50:
        rng = np.random.RandomState(seed)
        remaining = deadline - time.time() - 10
        n_starts = max(1, int(remaining // 40))
        for s in range(n_starts):
            if time.time() > deadline - 20:
                break
            h_pert = best_h + rng.uniform(-0.03, 0.03, n)
            h_pert = project_feasible(h_pert, n)
            h_opt, c_opt = adam_optimize(h_pert, n, min(time.time() + 35, deadline - 20),
                                          alpha_init=200, alpha_final=300000, lr_init=0.008)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # Phase 7: Final polish with very fine steps
    if time.time() < deadline - 3:
        best_h, best_c = topk_search(best_h, n, deadline - 2,
                                      [0.0001, 0.00005, 0.00002, 0.00001],
                                      K=100, topJ=30, use_active=True)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
