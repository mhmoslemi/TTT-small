```python
def project_water(h, target):
    """Find λ such that sum(clip(h - λ, 0, 1)) = target, then return clip(h-λ, 0, 1)."""
    # h - λ clipped to [0,1]
    # If λ is very negative, all are 1, sum = n
    # If λ is very positive, all are 0, sum = 0
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0
        grad[m:] -= h[:n-m]
    else:
        grad[:n-m] += 1.0
        grad[:n-m] -= h[m:]
    return grad

def smooth_max_and_grad(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_use = c[idx]
        lags = idx - (n - 1)
    else:
        c_use = c
        lags = np.arange(len(c)) - (n - 1)
    
    m = c_use.max()
    shifted = np.clip(alpha * (c_use - m), -500, 500)
    exps = np.exp(shifted)
    logsum_val = m + np.log(exps.sum()) / alpha
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for w, lag in zip(weights, lags):
        if w < 1e-12:
            continue
        grad += w * _lag_gradient(h, n, int(lag))
    
    return logsum_val, grad

def nesterov_pg(h, n, deadline, alpha_start=100, alpha_end=100000, top_k=10):
    """Nesterov accelerated projected gradient on smoothed max."""
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)

    n_total = int((deadline - time.time()) / 0.01)  # rough estimate
    eta = 0.5 / alpha_start
    
    it = 0
    while time.time() < deadline:

        # Nesterov momentum

        beta = (t - 1) / t_new

        y = np.clip(y, 0, 1)

        h_new = project_feasible(h_new, n)

        if c_new < best_c:

            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
        t = t_new
        it += 1
        
        # Adaptive step size: decrease if not improving
        if it % 1000 == 0:

    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += 1.0
        grad[m:] -= h[:n - m]
    else:
        grad[:n - m] += 1.0
        grad[:n - m] -= h[m:]
    return grad

def smooth_max_grad(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        cu = c[idx]
        lags = idx - (n - 1)
    else:
        cu = c
        lags = np.arange(len(c)) - (n - 1)
    m = cu.max()
    sh = np.clip(alpha * (cu - m), -400, 400)
    ex = np.exp(sh)
    val = m + np.log(ex.sum()) / alpha
    w = ex / ex.sum()
    grad = np.zeros(n)
    for wi, li in zip(w, lags):
        if wi < 1e-12:
            continue
        grad += wi * _lag_grad(h, n, int(li))
    return val, grad

def nesterov_optimize(h, n, deadline, alpha_start=200, alpha_end=500000, top_k=15):
    h_c = h.copy()
    h_p = h.copy()
    t = 1.0
    best_h = h_c.copy()
    best_c = compute_c5(h_c, n)
    eta = 1.0 / alpha_start
    stalled = 0
    it = 0
    while time.time() < deadline:
        alpha = alpha_start * (alpha_end / alpha_start) ** min(1.0, it / 30000.0)
        val, grad = smooth_max_grad(h_c, n, alpha, top_k)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        y = h_c + beta * (h_c - h_p)
        y = np.clip(y, 0, 1)
        h_new = project_feasible(y - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h_p = h_c
        h_c = h_new
        t = t_new
        it += 1
        if it % 500 == 0:
            eta *= 0.92
        if stalled > 2000 and it > 5000:
            eta *= 0.5
            stalled = 0
            if eta < 1e-10:
                break
    return best_h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16, top_k=None):
    def obj(x):
        v, _ = smooth_max_grad(x, n, alpha, top_k)
        return v
    def obj_grad(x):
        _, g = smooth_max_grad(x, n, alpha, top_k)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=obj_grad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None:
        return None
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=200):
    h_c = h.copy()
    best_h = h_c.copy()
    best_c = compute_c5(h_c, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_c, n, trust)
        if h_new is None:
            break
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_c))
        if step < 1e-9:
            break
        h_c = h_new
        trust = max(0.002, trust * 0.85)
    return best_h, best_c

def gradient_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for s in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-5:]
            thr = c[top_idx[0]] - 0.005
            active = top_idx[c[top_idx] > thr]
            lags = active - (n - 1)
            comp_grad = np.zeros(n)
            for li in lags:
                comp_grad += _lag_grad(h, n, int(li))
            comp_grad /= max(len(lags), 1)
            h_try = project_feasible(h - s * comp_grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                improved = True
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    remaining = deadline - time.time()
    if remaining > 60:
        t1 = time.time() + min(40, remaining * 0.3)
        h_oo, c_oo = outer_approx(best_h, n, t1)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    remaining = deadline - time.time()
    if remaining > 40:
        t2 = time.time() + min(remaining * 0.4, 120)
        h_ng, c_ng = nesterov_optimize(best_h, n, t2)
        if c_ng < best_c:
            best_c = c_ng
            best_h = h_ng
    remaining = deadline - time.time()
    if remaining > 20:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 10:
                break
            h_s = slsqp_optimize(best_h, n, alpha, maxiter=3000, ftol=1e-15)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    if time.time() < deadline - 5:
        best_h, best_c = gradient_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
```
