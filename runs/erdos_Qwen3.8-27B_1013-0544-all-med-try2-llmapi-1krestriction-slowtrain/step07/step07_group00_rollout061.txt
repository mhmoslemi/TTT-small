    dx = 2.0 / n
    grad = np.zeros(n)
    for m_i in range(len(w)):
        if w[m_i] < 1e-12:
            continue
        k = m_i - (n - 1)
        wm = w[m_i] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_optimize(h, n, deadline, alpha=100000, mu=500.0, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_jac,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-17,
                               'gtol': 1e-15, 'maxfun': maxiter * 30})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def coord_descent(h, n, deadline, t_grid=150, seed=42):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    f = compute_c5_vec(h, n)
    t_vals = np.linspace(0, 1, t_grid)
    s_vals = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    n_sweeps = 100000
    recompute_every = 50
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        if sweep > 0 and sweep % recompute_every == 0:
            f = compute_c5_vec(h, n)
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                return best_h, best_c
            old = h[i]
            ik = i + s_vals
            ikk = i - s_vals
            valid_ik = (ik >= 0) & (ik < n)
            valid_ikk = (ikk >= 0) & (ikk < n)
            ik_safe = np.where(valid_ik, ik, 0)
            ikk_safe = np.where(valid_ikk, ikk, 0)
            b = np.where(valid_ik, 1.0 - h[ik_safe], 0.0) - np.where(valid_ikk, h[ikk_safe], 0.0)
            b[n-1] = 0.0
            
            f0 = f[n-1]
            t_col = (t_vals - old)[:, None]
            g_mat = f[None, :] + t_col * b[None, :]
            g_mat[:, n-1] = f0 - old * (1.0 - old) + t_vals * (1.0 - t_vals)
            g_vals = g_mat.max(axis=1)
            
            best_ti = np.argmin(g_vals)
            best_t = t_vals[best_ti]
            
            if best_t == old:
                continue
            
            delta = best_t - old
            h[i] = best_t
            f += delta * b
            f[n-1] = f0 - old * (1.0 - old) + best_t * (1.0 - best_t)
        
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
    
    return best_h, best_c

def binary_snap(h, n):
    """Try setting each coordinate to 0 or 1, whichever is better"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for i in range(n):
        for val in [0.0, 1.0]:
            old = best_h[i]
            best_h[i] = val
            best_h = project_feasible(best_h, n)

            if c < best_c - 1e-14:

            else:
                best_h[i] = old
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
