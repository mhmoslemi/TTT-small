            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def adam_optimize(h, n, deadline, alpha_init=100.0, alpha_final=800000.0, lr_init=0.012, max_steps=8000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.15 * t) ** (-0.75)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def lbfgs_phase(h, n, deadline, alphas=None):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for alpha in alphas:
        if time.time() > deadline - 1:
            break
        def obj(x, a=alpha):
            return smooth_max_value(x, n, a) + 200.0 * (x.sum() - n / 2.0) ** 2
        def gr(x, a=alpha):
            g = grad_smooth_max(x, n, a)
            g += 400.0 * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': 2000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def pair_search_fast(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, 5000)
            order = np.argsort(-np.abs(g))
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline, threshold=0.02):
    h = h.copy()
    best_c = compute_c5(h, n)
    for iteration in range(500):
        if time.time() > deadline:
            break
        improved = False
        dist = np.minimum(h, 1.0 - h)
        order = np.argsort(dist)
        for i in order:
            if time.time() > deadline:
                break
            if dist[i] > threshold:
                break
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-14:
                continue
            mask = (h > abs(delta) * 1.01) & (h < 1.0 - abs(delta) * 1.01)
            mask[i] = False
            if not mask.any():
                mask = (h > 0) & (h < 1)
                mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-14:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def multi_res(h_best, n_best, deadline):
    best_h = h_best.copy()
    best_c = compute_c5(h_best, n_best)
    x_best = np.linspace(0, 2, n_best)
    for n_try in [150, 200, 250, 350, 400, 500]:
        if time.time() > deadline - 25:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        alloc_lb = min(25, (deadline - time.time()) * 0.35)
        if alloc_lb < 8:
            break
        h_t, c_t = lbfgs_phase(h_int, n_try, time.time() + alloc_lb,
                               alphas=[5000, 50000, 500000])
        if time.time() < deadline - 15:
            h_t, c_t = adam_optimize(h_t, n_try, min(time.time() + 20, deadline - 12),
                                     alpha_init=5000, alpha_final=500000, lr_init=0.008, max_steps=2000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            x_best = np.linspace(0, 2, n_try)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B baseline
    t1 = min(t0 + 100, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1,
                                  alphas=[200, 1000, 5000, 20000, 100000, 500000])
    # Phase 2: Adam with aggressive ramp
    if time.time() < deadline - 320:
        t2 = min(time.time() + 180, deadline - 320)
        h_a, c_a = adam_optimize(best_h, n_init, t2, alpha_init=100, alpha_final=800000, lr_init=0.012)
        if c_a < best_c:
            best_c, best_h = c_a, h_a
    # Phase 3: Very high alpha L-BFGS
    if time.time() < deadline - 220:
        t3 = min(time.time() + 70, deadline - 220)
        best_h, best_c = lbfgs_phase(best_h, n_init, t3,
                                      alphas=[500000, 2000000, 10000000])
    # Phase 4: Binary push
    if time.time() < deadline - 180:
        t4 = min(time.time() + 50, deadline - 180)
        h_b, c_b = binary_push(best_h, n_init, t4, threshold=0.05)
        if c_b < best_c:
            best_c, best_h = c_b, h_b
    # Phase 5: Medium pair search
    if time.time() < deadline - 130:
        t5 = min(time.time() + 70, deadline - 130)
        best_h, best_c = pair_search_fast(best_h, n_init, t5,
                                          [0.005, 0.002, 0.001, 0.0005], radius=50)
    # Phase 6: Multi-resolution
    if time.time() < deadline - 100:
        t6 = min(time.time() + 70, deadline - 100)
        best_h, best_c = multi_res(best_h, len(best_h), t6)
    # Phase 7: Binary push again (tighter)
    if time.time() < deadline - 70:
        t7 = min(time.time() + 30, deadline - 70)
        h_b2, c_b2 = binary_push(best_h, len(best_h), t7, threshold=0.02)
        if c_b2 < best_c:
            best_c, best_h = c_b2, h_b2
    # Phase 8: Fine pair search
    if time.time() < deadline - 50:
        t8 = min(time.time() + 45, deadline - 50)
        best_h, best_c = pair_search_fast(best_h, len(best_h), t8,
                                          [0.001, 0.0005, 0.0002, 0.0001], radius=40)
    # Phase 9: Final Adam polish with very high alpha
    if time.time() < deadline - 25:
        t9 = min(time.time() + 18, deadline - 12)
        h_f, c_f = adam_optimize(best_h, len(best_h), t9,
                                  alpha_init=500000, alpha_final=3000000, lr_init=0.003, max_steps=1500)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    # Phase 10: Final tiny pair search
    if time.time() < deadline - 3:
        best_h, best_c = pair_search_fast(best_h, len(best_h), deadline - 2,
                                          [0.0001, 0.00005, 0.00002, 0.00001], radius=30)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
