```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, lag):
    """Gradient of c_lag with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        return grad
    else:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += (1.0 - h[:n-m])
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def smooth_max(c, alpha):
    """Log-sum-exp approximation of max."""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_active(h, n, top_idx, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        return smooth_max(c[top_idx], alpha)

        c = np.correlate(x, 1-x, mode='full') * dx
        active = c[top_idx]
        m = active.max()
        w = np.exp(alpha * (active - m))
        w /= w.sum()
        grad = np.zeros(n)

            lag = idx - (n-1)
            grad += wi * dx * grad_lag(x, n, lag)
        return grad
    
    bounds = [(0.0, 1.0)] * n
```

Actually, I want to also try a different approach: **subgradient descent with averaging**. The idea is:
- At each iteration, compute the subgradient of the max (which is the gradient of the worst lag)
- Take a step in the negative subgradient direction
- Maintain a running average of iterates
- The average converges to a point that's "on average" good for all lags

This is like the classical approach to min-max problems.

```python
def subgradient_search(h, n, deadline, max_iters=10000):
    h_curr = h.copy()
    h_avg = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t = 1.0
    for it in range(max_iters):
        if time.time() > deadline: break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n-1)
        g = lag_grad(h_curr, n, lag)
        # Projected subgradient step
        step_size = 1.0 / (t * np.linalg.norm(g))
        h_new = h_curr - step_size * g
        h_new = project_feasible(h_new, n)
        # Update average
        t += 1
        h_avg = ((t-1)/t) * h_avg + (1/t) * h_new
        # Check both current and average
        c_curr = compute_c5(h_new, n)
        c_avg = compute_c5(h_avg, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_new.copy()
        if c_avg < best_c:
            best_c = c_avg
            best_h = h_avg.copy()
        h_curr = h_new
    return best_h, best_c
```
