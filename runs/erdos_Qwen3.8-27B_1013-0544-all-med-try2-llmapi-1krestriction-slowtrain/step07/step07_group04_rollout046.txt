def adam_smooth(h, n, deadline, alpha=10000, lr=0.01, beta1=0.9, beta2=0.999):
    """Adam optimizer for smooth max of correlations."""
    dx = 2.0 / n
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    best_h = h.copy()
    best_c = np.inf
    
    while time.time() < deadline:
        t += 1
        corr = np.correlate(h, 1-h, mode='full')
        g = corr * dx
        gs = g - g.max()
        p = np.exp(alpha * gs) / np.exp(alpha * gs).sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k_idx in range(2*n-1):
            if p[k_idx] < 1e-12:
                continue
            s = k_idx - (n-1)
            i1, i2 = max(0, s), min(n, s+n)
            if i1 < i2:
                grad[i1:i2] += p[k_idx] * (1.0 - h[i1-s:i2-s])
            i3, i4 = max(0, -s), min(n, n-s)
            if i3 < i4:
                grad[i3:i4] -= p[k_idx] * h[i3+s:i4+s]
        grad *= dx
        
        # Center gradient for sum constraint
        grad = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h, n)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    gs = g - g.max()
    p = np.exp(alpha * gs)
    ps = p.sum()
    if ps < 1e-300:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        p = p / ps
    obj = g.max() + np.log(ps + 1e-300) / alpha
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if p[k_idx] < 1e-13:
            continue
        s = k_idx - (n - 1)
        i1, i2 = max(0, s), min(n, s + n)
        if i1 < i2:
            grad[i1:i2] += p[k_idx] * (1.0 - h[i1 - s:i2 - s])
        i3, i4 = max(0, -s), min(n, n - s)
        if i3 < i4:
            grad[i3:i4] -= p[k_idx] * h[i3 + s:i4 + s]
    grad *= dx
    return obj, grad, corr

def adam_optimize(h, n, deadline, alpha=5000.0, lr=0.05, max_iter=50000):
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = np.inf
    patience = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        obj, grad, corr = smooth_objective_and_grad(h, n, alpha)
        grad = grad - grad.mean()
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        mh = m / (1 - 0.9 ** t)
        vh = v / (1 - 0.999 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h_new = project_feasible(h_new, n)
        h = h_new
        c = np.max(corr) * dx
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 3000:
                h = project_feasible(h + 0.05 * np.random.randn(n), n)
                patience = 0
    return best_h, best_c

def binary_flip_search(h, n, deadline, max_passes=2000):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        np.random.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            best_nc = cur_c
            best_j = -1
            for j in ones:
                if j == i:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                mn = i - m_range
                jm = j + m_range
                jn = j - m_range
                mask = (im >= 0) & (im < n)
                delta[mask] += 1.0 - h[im[mask]]
                mask = (mn >= 0) & (mn < n)
                delta[mask] -= h[mn[mask]]
                mask = (jm >= 0) & (jm < n)
                delta[mask] -= 1.0 - h[jm[mask]]
                mask = (jn >= 0) & (jn < n)
                delta[mask] += h[jn[mask]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + (np.correlate(h, 1 - h, mode='full') - corr)
                corr = np.correlate(h, 1 - h, mode='full').copy()
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
        if not improved:
            break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def shake_and_search(h, n, deadline, rng, n_shakes=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for s in range(n_shakes):
        if time.time() > deadline:
            break
        h2 = h.copy()
        n_swaps = rng.randint(3, 12)
        ones = np.where(h2 == 1)[0].tolist()
        zeros = np.where(h2 == 0)[0].tolist()
        for _ in range(n_swaps):
            if not ones or not zeros:
                break
            a = rng.choice(ones)
            b = rng.choice(zeros)
            h2[a], h2[b] = 0.0, 1.0
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
        search_dl = min(deadline, time.time() + (deadline - time.time()) / (n_shakes - s + 1))
        h3, c3 = binary_flip_search(h2, n, search_dl, max_passes=200)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
            h = h3.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: Adam optimization (continual)
    p1 = t0 + 0.10 * budget_s
    h = project_feasible(best_h + 0.01 * rng.randn(n), n)
    hires = [2000, 5000, 20000, 50000]
    for a in hires:
        if time.time() > p1:
            break
        h, c = adam_optimize(h, n, p1, alpha=a, lr=0.03, max_iter=20000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: SLSQP polish
    h = project_feasible(best_h, n)
    for a in [50000, 200000, 1000000]:
        if time.time() > p1 + 30:
            break
        h = slsqp_smooth(h, n, a, 1000, p1 + 30)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 3: Binarize + binary local search
    p3 = t0 + 0.50 * budget_s
    target = n // 2
    idx = np.argsort(best_h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    hb, cb = binary_flip_search(hb, n, p3, max_passes=3000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 4: Shake and search
    p4 = t0 + 0.72 * budget_s
    h2 = best_h.copy()
    if h2.sum() != n / 2:
        h2 = project_feasible(h2, n)
        idx2 = np.argsort(h2)[::-1]
        hb2 = np.zeros(n)
        hb2[idx2[:n // 2]] = 1.0
        h2 = hb2
    h2, c2 = shake_and_search(h2, n, p4, rng, n_shakes=30)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    # Phase 5: Extended binary search
    p5 = t0 + 0.88 * budget_s
    hb, cb = binary_flip_search(best_h, n, p5, max_passes=5000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 6: Final Adam + binary
    if time.time() < dl - 20:
        h = project_feasible(best_h.astype(float) + 0.02 * rng.randn(n), n)
        h, c = adam_optimize(h, n, dl - 15, alpha=100000, lr=0.02, max_iter=10000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        idx = np.argsort(best_h)[::-1]
        hb = np.zeros(n)
        hb[idx[:n // 2]] = 1.0
        hb, cb = binary_flip_search(hb, n, dl, max_passes=1000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
