import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def active_set_obj(x, n, top_k, mu):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    sh = 500.0 * (c_top - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / 500.0
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(x, n, top_k, mu):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    dx = 2.0 / n
    for m in idx:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            g[0:nv] += dx * (1.0 - x[k:n])
            g[k:n] += dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += dx * (1.0 - x[0:nv])
            g[0:nv] += dx * (-x[kk:n])
    g /= float(top_k)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(500, 5.0), (5000, 50.0), (50000, 200.0), (500000, 1000.0)]:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': max_iters * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c, best_h = c_new, h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_opt(h, n, deadline):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

def pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d; best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            best_h[i] -= d; best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d; best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            best_h[i] += d; best_h[j] -= d
    return best_h, best_c

    cur = h.copy()
    cur_c = best_c
    T = T_init
    while time.time() < deadline:
        T = max(T * cool, T_min)
        improved = False
        n_tries = 3000
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-d_max, d_max)
            if cur[i] + d < 0 or cur[i] + d > 1 or cur[j] - d < 0 or cur[j] - d > 1:
                continue
            cur[i] += d; cur[j] -= d
            c_new = compute_c5(cur, n)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = cur.copy()
                    improved = True
            elif rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-10)):
                cur_c = c_new
            else:
                cur[i] -= d; cur[j] += d
        if not improved and T < T_init * 0.01:
            cur = best_h.copy()
            cur_c = best_c
            T = T_init * 0.3
    return best_h, best_c

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
