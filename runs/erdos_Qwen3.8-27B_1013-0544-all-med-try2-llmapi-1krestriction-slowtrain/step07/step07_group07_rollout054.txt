The problem is: min max_k f_k(h) where f_k(h) = dx * Σ_i h_i(1-h_{i+k})

Frank-Wolfe: 
- Maintain h_t
- Solve: min_{s ∈ S} ∇f_{k_t}(h_t) · s where S = {h: 0≤h≤1, Σh=n/2} and k_t is the active constraint
- Update: h_{t+1} = (1-t)h_t + t*s

The linear subproblem: min ∇f_k(h) · s subject to 0≤s≤1, Σs=n/2
This is: assign s_i = 1 where ∇f_k_i is most negative, s_i = 0 where most positive, with the constraint that exactly n/2 units are assigned. Since s is continuous in [0,1], we need to find a threshold.

Actually for the linear subproblem with box constraints and sum constraint:
- Sort ∇f_k_i values
- Set s_i = 1 for the smallest values, s_i = 0 for the largest
- One value gets a fractional assignment to satisfy the sum constraint

This is very fast! O(n log n) per iteration.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of f_lag with respect to h."""
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1 - 2*h)
    grad = np.zeros(n)
    if lag > 0:
        # f = dx * sum_{j=0}^{n-1-m} h_j (1-h_{j+m})
        grad[:n-m] += dx * (1 - h[m:])
        grad[m:] -= dx * h[:n-m]
    else:
        # f = dx * sum_{j=m}^{n-1} h_j (1-h_{j-m})
        grad[m:] += dx * (1 - h[:n-m])
        grad[:n-m] -= dx * h[m:]
    return grad

def fw_step(grads, n):
    """Frank-Wolfe linear step: min grad·s s.t. 0≤s≤1, sum(s)=n/2"""
    order = np.argsort(grads)
    s = np.zeros(n)
    target = n / 2.0
    if target == int(target):
        s[order[:int(target)]] = 1.0
    else:
        k = int(target)
        s[order[:k]] = 1.0
        s[order[k]] = target - k
    return s

def frank_wolfe_momentum(h, n, deadline, n_iter=5000):
    """Frank-Wolfe with momentum and adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    v = np.zeros(n)  # momentum
    dx = 2.0 / n
    
    no_improve = 0
    for it in range(n_iter):
        if time.time() > deadline: break
        if it % 500 == 0 and no_improve > 2000:
            break
        
        c = compute_all_lags(h, n)
        # Use top-5 lags for gradient
        top_idx = np.argsort(c)[-5:]
        weights = np.exp(100 * (c[top_idx] - c[top_idx].max()))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = idx - (n-1)
            grad += w * lag_gradient(h, n, lag)
        
        # FW step
        s = fw_step(grad, n)
        
        # Momentum
        alpha = 2.0 / (it + 2.0)
        v = 0.9 * v + alpha * (s - h)
        h_new = h + v
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
    
    return best_h, best_c

   New: (h[j]-δ)*(1-h[j+k*]-δ) ≈ (smaller) * (smaller) ✓

3. Try all such moves, pick the best one (smallest resulting max lag)
4. Repeat

            else:
                if direction == 1:
                    diffs = h_curr[:n - m] - h_curr[m:]
                else:
                    diffs = h_curr[m:] - h_curr[:n - m]
            best_j = int(np.argmax(diffs))
            if diffs[best_j] < 1e-7:
                continue
            d = min(step, diffs[best_j] * 0.3)
            if d < 1e-8:
                continue
            if lag > 0 and direction == 1:
                h_curr[best_j] -= d
                h_curr[best_j + m] += d
            elif lag > 0 and direction == -1:
                h_curr[best_j + m] -= d
                h_curr[best_j] += d
            elif lag < 0 and direction == 1:
                h_curr[best_j] -= d
                h_curr[best_j + m] += d
            else:
                h_curr[best_j + m] -= d
                h_curr[best_j] += d
            h_curr = project_feasible(h_curr, n)
            c_new = compute_c5(h_curr, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_curr.copy()
                stall = 0
                improved = True
                break
            else:
                if lag > 0 and direction == 1:
                    h_curr[best_j] += d
                    h_curr[best_j + m] -= d
                elif lag > 0 and direction == -1:
                    h_curr[best_j + m] += d
                    h_curr[best_j] -= d
                elif lag < 0 and direction == 1:
                    h_curr[best_j] += d
                    h_curr[best_j + m] -= d
                else:
                    h_curr[best_j + m] += d
                    h_curr[best_j] -= d
                h_curr = project_feasible(h_curr, n)
        if not improved:
            stall += 1
            if stall % 300 == 0:
                step = max(0.0001, step * 0.85)
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # FW phase
    if time.time() < deadline - 20:
        fw_dead = min(deadline - 10, time.time() + (deadline - time.time()) * 0.3)
        h_fw, c_fw = frank_wolfe(best_h, n, fw_dead, top_m=3)
        if c_fw < best_c:
            best_c, best_h = c_fw, h_fw
    # SLSQP phase
    for alpha in [20000, 100000]:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_opt(best_h, n, alpha, top_m=15, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    # Directed local search
    if time.time() < deadline - 3:
        h_ls, c_ls = directed_local_search(best_h, n, deadline - 2)
        if c_ls < best_c:
            best_c, best_h = c_ls, h_ls
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    # Try other resolutions
    n_candidates = [200, 250, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.2)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    # Intensive optimization on best
    if best_h is not None:
        n = len(best_h)
        # More FW with different params
        if time.time() < deadline - 60:
            fw_dead = min(deadline - 40, time.time() + 80)
            h_fw, c_fw = frank_wolfe(best_h, n, fw_dead, top_m=5)
            if c_fw < best_c:
                best_c, best_h = c_fw, h_fw
        # Bidirectional local search
        if time.time() < deadline - 30:
            h_bd, c_bd = bidirectional_ls(best_h, n, min(deadline - 15, time.time() + 100))
            if c_bd < best_c:
                best_c, best_h = c_bd, h_bd
        # Restart with perturbation
        if time.time() < deadline - 50:
            for sp in [0.003, 0.006, 0.01]:
                if time.time() > deadline - 20:
                    break
                hp = best_h + rng.normal(0, sp, n)
                hp = project_feasible(hp, n)
                alloc = min(40, (deadline - time.time() - 15) * 0.3)
                if alloc < 15:
                    break
                h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt
        # Final SLSQP polish
        if time.time() < deadline - 10:
            h_new = slsqp_opt(best_h, n, 500000, top_m=30, maxiter=6000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
        # Final directed local search
        if time.time() < deadline - 2:
            h_ls, c_ls = directed_local_search(best_h, n, deadline - 1)
            if c_ls < best_c:
                best_c, best_h = c_ls, h_ls
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    return best_h, best_c, len(best_h)
