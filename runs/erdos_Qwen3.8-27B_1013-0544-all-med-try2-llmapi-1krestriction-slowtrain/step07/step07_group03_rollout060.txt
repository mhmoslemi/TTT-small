```python
def logsumexp_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    m = np.max(corr)
    shifted = alpha * (corr - m)
    z = np.exp(shifted)
    z_norm = z / z.sum()
    obj = m + np.log(z.sum()) / alpha
    
    # Gradient
    grad = np.zeros(n)
    # For each lag k (index in corr), the contribution to grad is:
    # w_k * (2/n) * [(1-h[i+l]) - h[i-l]] (with boundary handling)
    # where l = k - (n-1)
    
    # This is O(n^2) which is 160000 for n=400, fine.
    for k in range(2 * n - 1):
        wk = z_norm[k]
        if wk < 1e-15:
            continue
        l = k - (n - 1)
        # dc_l/dh_i = (1-h[i+l])*[0<=i+l<n] - h[i-l]*[0<=i-l<n]
        # For i in range(n):
        #   term1: (1-h[i+l]) if 0 <= i+l < n
        #   term2: -h[i-l] if 0 <= i-l < n
        
        # term1: i+l in [0, n-1] => i in [-l, n-1-l] => i in [max(0,-l), min(n-1, n-1-l)]
        # term2: i-l in [0, n-1] => i in [l, l+n-1] => i in [max(0,l), min(n-1, l+n-1)]
        
        lo1 = max(0, -l)
        hi1 = min(n - 1, n - 1 - l)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * dx * (1.0 - h[lo1+l:hi1+l+1])
        
        lo2 = max(0, l)
        hi2 = min(n - 1, l + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * dx * h[lo2-l:hi2-l+1]
    
    return obj, grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    z = np.exp(shifted)
    lse = m + np.log(z.sum()) / alpha
    pen = lam * (x.sum() - n / 2.0) ** 2
    return lse + pen

def smooth_grad(x, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    z = np.exp(shifted)
    z /= z.sum()
    grad = np.zeros(n)
    # For each lag k (offset from center), compute contribution
    # m_idx ranges from 0 to 2n-2, lag k = m_idx - (n-1)
    for m_idx in range(2 * n - 1):
        w = z[m_idx]
        if w < 1e-15:
            continue
        k = m_idx - (n - 1)  # lag
        # dc[k]/dh[i] = (1-h[i-k])*[valid] - h[i+k]*[valid]
        # First term: (1-h[i-k]), valid when i-k in [0,n-1] AND k>=0 -> i in [k, n-1]
        #              or k<0 -> i in [0, n-1]
        # Actually: first term valid when 0 <= i-k < n, i.e., k <= i < n+k
        #   Combined with 0 <= i < n: max(0,k) <= i < min(n, n+k)
        # Second term: h[i+k], valid when 0 <= i+k < n, i.e., -k <= i < n-k
        #   Combined with 0 <= i < n: max(0,-k) <= i < min(n, n-k)
        
        # First term: i in [max(0,k), min(n, n+k) - 1]
        lo1 = max(0, k)
        hi1 = min(n, n + k) - 1
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * dx * (1.0 - x[lo1-k:hi1-k+1])
        
        # Second term: i in [max(0,-k), min(n, n-k) - 1]
        lo2 = max(0, -k)
        hi2 = min(n, n - k) - 1
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * dx * x[lo2+k:hi2+k+1]
    
    # Penalty gradient
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def lbfgs_step(h, n, alpha, lam, maxiter, ftol):
    def obj(x):
        return smooth_obj(x, n, alpha, lam)
    def grad(x):
        return smooth_grad(x, n, alpha, lam)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def to_binary(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def anneal_binary(h_init, n, deadline, seed=42, n_iters=2000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)

    best_c = cur_c
    ones_arr = np.where(h == 1)[0]

    scale = 2000.0
    for it in range(n_iters):

        ai = rng.randint(len(zeros_arr))

def anneal_binary(h_init, n, deadline, seed=42, n_iters=5000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.008
    cool = 0.99998
    scale = 3000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 2000 == 0 and T > 1e-12:
            T *= cool ** 2000
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        accept = nc <= cur_c
        if not accept and T > 1e-14:
            log_p = -scale * (nc - cur_c) / T
            accept = (log_p > -700.0) and (rng.random() < np.exp(log_p))
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 8
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP smooth optimization (main workhorse)
    p1_end = t0 + 0.40 * total
    h = best_h.copy()
    alphas = [100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000, 3000000]
    for alpha in alphas:
        if time.time() > p1_end - 15:
            break
        maxit = 3000
        h = slsqp_run(h, n, alpha, maxit, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary greedy
    p2_end = t0 + 0.52 * total
    hb, cb = greed_binary(best_h, n, p2_end, max_swaps=30000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: Annealing
    p3_end = t0 + 0.65 * total
    ha, ca = anneal_binary(best_h, n, p3_end, seed=seed, n_iters=5000000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()
    # Second anneal from best with different seed
    if time.time() < p3_end:
        ha2, ca2 = anneal_binary(best_h, n, p3_end, seed=seed+1, n_iters=2000000)
        if ca2 < best_c:
            best_c = ca2
            best_h = ha2.copy()

    # Phase 4: SLSQP polish from best (using frozen for high alpha)
    p4_end = t0 + 0.82 * total
    h = best_h.copy()
    polish_alphas = [50000, 200000, 1000000, 5000000]
    for alpha in polish_alphas:
        if time.time() > p4_end - 10:
            break
        if alpha >= 200000:
            h = slsqp_frozen(h, n, alpha, 2000, 1e-16, thresh=0.05)
        else:
            h = slsqp_run(h, n, alpha, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Try higher resolution
    p5_end = t0 + 0.93 * total
    for nh in [480, 560]:
        if time.time() > p5_end - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, cch = greed_binary(hh, nh, min(time.time() + 15, p5_end - 15), max_swaps=5000)
        for alpha in [50000, 500000, 2000000]:
            if time.time() > p5_end - 5:
                break
            hh = slsqp_run(hh, nh, alpha, 1500, 1e-15)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n = nh
                x0_old = None

    # Phase 6: Final polish on best solution
    if time.time() < dl - 5:
        h = best_h.copy()
        nn = len(h)
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > dl - 2:
                break
            h = slsqp_frozen(h, nn, alpha, 2000, 1e-17, thresh=0.03)
            c = compute_c5(h, nn)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        # Tiny noise restart
        if time.time() < dl - 3:
            h = best_h + 0.003 * rng.randn(nn)
            h = project_feasible(h, nn)
            for alpha in [5000000, 50000000]:
                if time.time() > dl - 1:
                    break
                h = slsqp_run(h, nn, alpha, 1000, 1e-16)
                c = compute_c5(h, nn)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
