    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=3000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = compute_corr(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha

    def con(x):
        return x.sum() - n / 2.0

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, steps, radius):
    best_c = compute_c5(h, n)
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Nesterov GD on original grid
    t1 = min(t0 + 150, deadline - 500)
    h1, c1 = nesterov_gd(best_h, n, t1, topk=25, lr=0.003, beta=0.85)
    if c1 < best_c:
        best_c = c1; best_h = h1

    # Phase 2: SLSQP on original grid
    if time.time() < deadline - 400:
        t2 = min(time.time() + 60, deadline - 400)
        h2, c2 = slsqp_phase(best_h, n, t2, alpha=3000, maxiter=1200)
        if c2 < best_c:
            best_c = c2; best_h = h2

    # Phase 3: Pair search on original grid
    if time.time() < deadline - 300:
        t3 = min(time.time() + 100, deadline - 300)
        h3, c3 = pair_search(best_h, n, t3,
                             [0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)
        if c3 < best_c:
            best_c = c3; best_h = h3

    # Phase 4: Upscale to 600 points
    n2 = 600
    xo = np.linspace(0, 2, n)
    xn = np.linspace(0, 2, n2)
    h_up = np.interp(xn, xo, best_h)
    h_up = project_feasible(h_up, n2)
    base_600_c = compute_c5(h_up, n2)
    if base_600_c < best_c:
        best_c = base_600_c; best_h = h_up

    if time.time() < deadline - 250:
        t4 = min(time.time() + 120, deadline - 250)
        h4, c4 = nesterov_gd(h_up, n2, t4, topk=30, lr=0.002, beta=0.85)
        if c4 < best_c:
            best_c = c4; best_h = h4

    # Phase 5: SLSQP on 600 grid
    if time.time() < deadline - 150:
        t5 = min(time.time() + 70, deadline - 150)
        nf = len(best_h)
        h5, c5 = slsqp_phase(best_h, nf, t5, alpha=5000, maxiter=1000)
        if c5 < best_c:
            best_c = c5; best_h = h5

    # Phase 6: Nesterov GD again on best grid with different params
    if time.time() < deadline - 100:
        t6 = min(time.time() + 50, deadline - 100)
        nf = len(best_h)
        h6, c6 = nesterov_gd(best_h, nf, t6, topk=15, lr=0.002, beta=0.8)
        if c6 < best_c:
            best_c = c6; best_h = h6

    # Phase 7: Final fine pair search
    if time.time() < deadline - 3:
        nf = len(best_h)
        h7, c7 = pair_search(best_h, nf, deadline - 2,
                             [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                             radius=30)
        if c7 < best_c:
            best_c = c7; best_h = h7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
