def init_c_array(h,n):
    c=np.empty(2*n-1,dtype=float)
    one=1.0-h
    c[n-1]=float(np.dot(h,one))
    for d in range(1,n):
       nv=n-d
       c[n-1+d]=float(np.dot(h[:nv], one[d:]))
       c[n-1-d]=float(np.dot(h[d:], one[:nv]))

import numpy as np
import time

FR = (1.0, 0.5, 0.25, 0.08)

def exact_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj(h, n):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0).copy()
    t = 0.5 * n
    for _ in range(40):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            idx = np.flatnonzero(h < 1.0 - 1e-14)
            if idx.size:
                h[idx] += d / idx.size
        else:
            idx = np.flatnonzero(h > 1e-14)
            if idx.size:
                h[idx] += d / idx.size
        h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.flatnonzero((h > 1e-13) & (h < 1.0 - 1e-13))
        if idx.size:
            h[idx] += d / idx.size
            h = np.clip(h, 0.0, 1.0)
    return h

def fill_c(c, h, n):
    o = 1.0 - h
    c[n - 1] = float(np.dot(h, o))
    for d in range(1, n):
        nv = n - d
        c[n - 1 + d] = float(np.dot(h[:nv], o[d:]))
        c[n - 1 - d] = float(np.dot(h[d:], o[:nv]))
    return c

def make_c(h, n):
    return fill_c(np.empty(2 * n - 1, dtype=float), h, n)

def build_data(n):
    ct = n - 1
    fi, fp, si, sp = [], [], [], []
    e = np.empty(0, dtype=np.intp)
    for p in range(n):
        a, b = [], []
        if p:
            m = np.arange(-p, 0, dtype=np.intp)
            a.append(m + ct)
            b.append(p + m)
        if n - p > 1:
            m = np.arange(1, n - p, dtype=np.intp)
            a.append(m + ct)
            b.append(p + m)
        if a:
            fi.append(np.concatenate(a).astype(np.intp))
            fp.append(np.concatenate(b).astype(np.intp))
        else:
            fi.append(e)
            fp.append(e)
        a, b = [], []
        if p - n + 1 < 0:
            m = np.arange(p - n + 1, 0, dtype=np.intp)
            a.append(m + ct)
            b.append(p - m)
        if p:
            m = np.arange(1, p + 1, dtype=np.intp)
            a.append(m + ct)
            b.append(p - m)
        if a:
            si.append(np.concatenate(a).astype(np.intp))
            sp.append(np.concatenate(b).astype(np.intp))
        else:
            si.append(e)
            sp.append(e)
    return fi, fp, si, sp, ct

def apply_delta(c, h, p, d, dat):
    if d == 0.0:
        return
    fi, fp, si, sp, ct = dat
    f = fi[p]
    if f.size:
        c[f] += d * (1.0 - h[fp[p]])
    s = si[p]
    if s.size:
        c[s] -= d * h[sp[p]]
    old = h[p]
    c[ct] += d * (1.0 - 2.0 * old - d)
    val = old + d
    if val < 0.0 and val > -1e-15:
        val = 0.0
    elif val > 1.0 and val < 1.0 + 1e-15:
        val = 1.0
    h[p] = val

def try_pair(c, h, i, j, dat, before):
    md = min(1.0 - h[i], h[j])
    if md <= 1e-14:
        return before, False
    hi, hj = h[i], h[j]
    bv, bd = before, 0.0
    for fr in FR:
        d = fr * md
        if d <= 1e-15:
            continue
        apply_delta(c, h, i, d, dat)
        apply_delta(c, h, j, -d, dat)
        v = float(np.max(c))
        if v < bv:
            bv, bd = v, d
        apply_delta(c, h, j, d, dat)
        apply_delta(c, h, i, -d, dat)
        h[i], h[j] = hi, hj
    if bd > 0.0 and bv < before - 1e-9:
        m0 = float(np.max(c))
        if abs(m0 - before) > 1e-7:
            fill_c(c, h, dat[4] + 1)
            m0 = float(np.max(c))
        apply_delta(c, h, i, bd, dat)
        apply_delta(c, h, j, -bd, dat)
        v = float(np.max(c))
        if v < m0 - 1e-9:
            return v, True
        apply_delta(c, h, j, bd, dat)
        apply_delta(c, h, i, -bd, dat)
    h[i], h[j] = hi, hj
    return before, False

def smooth_w(c, alpha):
    sh = alpha * (c - float(np.max(c)))
    sh = np.clip(sh, -60.0, 0.0)
    w = np.exp(sh)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w.fill(1.0 / len(w))
    return w

def grad_w(w, h, n):
    g = np.zeros(n)
    ct = n - 1
    for ii in range(n - 1):
        wv = w[ii]
        if wv > 1e-14:
            d = ct - ii
            nv = n - d
            g[d:] += wv * (1.0 - h[:nv])
            g[:nv] -= wv * h[d:]
    for m in range(1, n):
        wv = w[ct + m]
        if wv > 1e-14:
            nv = n - m
            g[:nv] += wv * (1.0 - h[m:])
            g[m:] -= wv * h[:nv]
    if w[ct] > 1e-14:
        g += w[ct] * (1.0 - 2.0 * h)
    return g

def cont(h, n, dl, seed):
    h = proj(h, n).copy()
    c = make_c(h, n)
    dat = build_data(n)
    bc = float(np.max(c))
    bh = h.copy()
    rng = np.random.RandomState(seed)
    if n <= 130:
        k = max(24, n // 3)
    elif n <= 220:
        k = max(16, n // 4)
    else:
        k = 16
    alpha = 40.0
    no = 0
    ep = 0
    nrand = 40 + n // 10
    while time.time() < dl - 2:
        ep += 1
        if ep % 6 == 0:
            fill_c(c, h, n)
        if time.time() > dl - 3:
            break
        w = smooth_w(c, alpha)
        g = grad_w(w, h, n)
        order = np.argsort(g)
        low = order[:k]
        high = order[-k:]
        imp = False
        for i in low:
            if h[i] >= 1.0 - 1e-12:
                continue
            for j in high:
                if i == j or h[j] <= 1e-12:
                    continue
                before = float(np.max(c))
                v, ok = try_pair(c, h, i, j, dat, before)
                if ok:
                    imp = True
                    if v < bc:
                        bc, bh = v, h.copy()
                if time.time() > dl - 3:
                    break
            if time.time() > dl - 3:
                break
        for _ in range(nrand):
            if time.time() > dl - 3:

            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:

            if g[i] <= g[j]:
                ai, aj = i, j
            else:
                ai, aj = j, i
            if h[ai] >= 1.0 - 1e-12 or h[aj] <= 1e-12:

            before = float(np.max(c))

        if imp:
            no = 0
            if alpha < 600.0:

        else:
            no += 1

    h1 = proj(bh, n)

    h2 = proj(h, n)
    v2 = exact_c5(h2, n)
    if v2 < v1:
        return h2, float(v2), n
    return h1, float(v1), n

def resamp(h, n):
    if len(h) == n:
        return h.copy()
    x = np.linspace(0.0, 2.0, len(h))
    y = np.linspace(0.0, 2.0, n)
    return np.interp(y, x, h)

def make_start(n, mode, rng, src, srcn):
    m = n // 2
    if mode == 0 and src is not None and srcn:
        v = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, srcn), src)
        h = np.zeros(n)
        h[np.argsort(v)[-m:]] = 1.0
        return h
    if mode == 1:
        h = np.zeros(n)

    base = max(3, int(n * (0.22 + 0.05 * max(0, mode - 2))))

    pos = 0
    bit = 0 if mode == 3 else 1

        lo = max(2, base // 2)
        hi = max(lo + 1, base * 2)
        ln = int(rng.randint(lo, hi))
        if pos + ln > n:
            ln = n - pos
        h[pos:pos + ln] = float(bit)
        pos += ln
        bit = 1 - bit
    cnt = int(h.sum())
    if cnt > m:
        idx = np.flatnonzero(h == 1.0)
        h[idx[:cnt - m]] = 0.0
    elif cnt < m:
        idx = np.flatnonzero(h == 0.0)
        h[idx[:m - cnt]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
