            if s < 1e-14:

                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def anneal_binary(h_b, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_b.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = 0.008
    cool = 0.99995
    scale = 4000.0
    while time.time() < deadline:
        batch = min(200, max(1, int(0.02 / (1.0 - cool))))
        for _ in range(batch):
            if time.time() > deadline:
                break
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = int(zeros_arr[ai])
            j = int(ones_arr[bi])
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            delta[n - 1 + (j - i)] += 1.0
            delta[n - 1 + (i - j)] += 1.0
            nc = (corr + delta).max() * dx
            if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 8
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(400, 0.5)
    n = len(h0)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: binary local search
    dl1 = t0 + 0.07 * total
    hb, cb = binary_local_search(to_binary(best_h, n), n, dl1, max_sweeps=1500)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: projected gradient with increasing alpha (main)
    dl2 = t0 + 0.55 * total
    h = project_feasible(best_h, n)
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > dl2 - 3:
            break
        h = proj_gradient(h, n, alpha, dl2 - 2, max_iter=2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: binary anneal from current best
    dl3 = t0 + 0.68 * total
    ha, ca = anneal_binary(to_binary(best_h, n), n, dl3, seed=seed)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 4: re-smooth from annealed
    dl4 = t0 + 0.82 * total
    h = project_feasible(best_h, n)
    for alpha in [5000, 50000, 500000, 5000000, 50000000]:
        if time.time() > dl4 - 2:
            break
        h = proj_gradient(h, n, alpha, dl4 - 1, max_iter=1500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: multi-restart with noise
    dl5 = deadline - 3
    for trial in range(8):
        if time.time() > dl5 - 2:
            break
        noise_scale = 0.02 * (1.0 - trial / 10.0)
        h = project_feasible(best_h + noise_scale * rng.randn(n), n)
        for alpha in [20000, 500000, 5000000]:
            if time.time() > dl5 - 1:
                break
            h = proj_gradient(h, n, alpha, dl5 - 1, max_iter=1000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 6: try lower resolution
    if time.time() < deadline - 10:
        try:
            n2 = 200
            xn = np.linspace(0, 2, n2)
            xo = np.linspace(0, 2, n)
            h2 = project_feasible(np.interp(xn, xo, best_h), n2)
            h2b = to_binary(h2, n2)
            h2b, c2b = binary_local_search(h2b, n2, min(deadline - 5, t0 + total * 0.92), max_sweeps=500)
            for alpha in [5000, 200000, 5000000]:
                if time.time() > deadline - 4:
                    break
                h2b = proj_gradient(project_feasible(h2b, n2), n2, alpha, deadline - 3, max_iter=800)
                c2 = compute_c5(h2b, n2)
                if c2 < best_c:
                    best_c, best_h, n = c2, h2b.copy(), n2
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
