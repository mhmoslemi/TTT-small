def make_thue(n,rng,scale=1.0):
    pi=np.zeros(n,dtype=np.uint8)
    for i in range(1,n): pi[i]=pi[i>>1] ^ (i & 1)
    p=pi.astype(float)
    target=n/2.0

    k=int(round(target-cnt))
    if k>0:
        idx=np.flatnonzero(p==0.0)
        if idx.size:
            rng.shuffle(idx)

            p[idx[:k]]=1.0

            k=min(-k, idx.size)
            p[idx[:k]]=0.0

import time

from scipy.optimize import minimize

def compute_all_c(h):
    dx = 2.0 / len(h)
    return dx * np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    return float(np.max(compute_all_c(h)))

def project_feasible(x, target=None):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if target is None:
        target = 0.5 * n
    target = float(np.clip(target, 0.0, float(n)))
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = 0.5 * np.ones(n)
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    mid = 0.5 * (lo + hi)
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(x - mid, 0.0, 1.0)
    res = target - z.sum()
    if abs(res) > 1e-13:
        mask = (z > 1e-12) & (z < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            z[mask] += res / cnt
            z = np.clip(z, 0.0, 1.0)
            res = target - z.sum()
            if abs(res) > 1e-15:
                z[mask] += res / cnt
                z = np.clip(z, 0.0, 1.0)
    return z

def lag_value(h, n, p):
    dx = 2.0 / n
    lag = p - (n - 1)
    if lag >= 0:
        m = lag
        if m == 0:
            return dx * float(np.sum(h * (1.0 - h)))
        return dx * float(np.sum(h[:n - m] * (1.0 - h[m:])))
    m = -lag
    return dx * float(np.sum(h[m:] * (1.0 - h[:n - m])))

def lag_gradient(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    m = abs(int(lag))
    if m >= n:
        return g
    if lag > 0:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    else:
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    return g

def build_gradients(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.empty((L, n), dtype=float)
    base = n - 1
    for p in range(L):
        G[p] = dx * lag_gradient(h, n, p - base)
    return G

def top_indices(c, m):
    if c.size <= m:
        return np.arange(c.size)
    return np.argsort(c)[::-1][:m]

def _sum_eq_z(z, n):
    return float(np.sum(z[:n]) - 0.5 * n)

def _sum_eq_jac(z, n):
    jac = np.zeros(n + 1)
    jac[:n] = 1.0
    return jac

def _mm_obj(z, n):
    return float(z[n])

def _mm_obj_jac(z, n):
    jac = np.zeros(n + 1)
    jac[n] = 1.0
    return jac

def _mm_inq_fun(z, n, p):
    h = z[:n]
    return float(z[n] - lag_value(h, n, p))

def _mm_inq_jac(z, n, p):
    h = z[:n]
    dx = 2.0 / n
    jac = np.zeros(n + 1)
    jac[:n] = -dx * lag_gradient(h, n, p - (n - 1))
    jac[n] = 1.0
    return jac

def slsqp_once(h, M, maxiter):
    n = h.size
    c = compute_all_c(h)
    idx = top_indices(c, M)
    near = np.where(c >= c.max() - 1e-7)[0]
    idx = np.unique(np.concatenate((idx, near)))
    if idx.size > M + 4:
        idx = idx[np.argsort(c[idx])[::-1][:M + 4]]
    z0 = np.concatenate((h, [float(c.max()) + 1e-10]))
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': _sum_eq_z, 'jac': _sum_eq_jac, 'args': (n,)}]
    for p in idx:
        cons.append({'type': 'ineq', 'fun': _mm_inq_fun, 'jac': _mm_inq_jac, 'args': (n, int(p))})
    try:
        res = minimize(_mm_obj, z0, args=(n,), jac=_mm_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': 1e-12})
    except Exception:
        return h.copy(), compute_c5(h)
    if res.x is None:
        return h.copy(), compute_c5(h)
    hnew = project_feasible(res.x[:n])
    cnew = compute_c5(hnew)
    return hnew, cnew

def slsqp_polish(h, deadline, configs):
    h = h.copy()
    best = compute_c5(h)
    for m, it in configs:
        if time.time() > deadline - 3:
            break
        fails = 0
        cur = int(it)
        while time.time() < deadline - 3 and fails < 2:
            hn, cn = slsqp_once(h, m, cur)
            if cn + 1e-12 < best:
                h = hn
                best = cn
                fails = 0
                cur = max(30, cur // 2)
            else:
                fails += 1
    return h, best

def _pair_line_value(a, b, x, i1, i2, dx):
    vals = a + b * x
    q = dx * x * x
    if i1 >= 0:
        vals[i1] += q
    if i2 >= 0 and i2 != i1:
        vals[i2] += q
    return float(np.max(vals))

def _pair_line_info(a, b, x, i1, i2, dx):
    vals = a + b * x
    q = dx * x * x
    if i1 >= 0:
        vals[i1] += q
    if i2 >= 0 and i2 != i1:
        vals[i2] += q
    mx = float(np.max(vals))
    sel = vals >= mx - 1e-12
    slopes = b.copy()
    t = 2.0 * dx * x
    if i1 >= 0:
        slopes[i1] += t
    if i2 >= 0 and i2 != i1:
        slopes[i2] += t
    return mx, float(np.min(slopes[sel])), float(np.max(slopes[sel]))

def pair_delta_opt(a, b, maxd, dm, n, dx):
    if maxd <= 1e-12:
        return 0.0, float(np.max(a))
    L = a.size
    base = n - 1
    i1 = i2 = -1
    if dm != 0:
        i1 = dm + base
        i2 = -dm + base
        if i1 < 0 or i1 >= L:
            i1 = -1
        if i2 < 0 or i2 >= L:
            i2 = -1
        if i1 == i2:
            i2 = -1
    lo = 0.0
    hi = float(maxd)
    for _ in range(32):
        x = 0.5 * (lo + hi)
        val, smin, smax = _pair_line_info(a, b, x, i1, i2, dx)
        if smin > 1e-10:
            hi = x
        elif smax < -1e-10:
            lo = x
        else:
            break
    mid = 0.5 * (lo + hi)
    bestv = float('inf')
    bestx = 0.0
    for x in (0.0, lo, hi, mid):
        v = _pair_line_value(a, b, x, i1, i2, dx)
        if v < bestv - 1e-15:
            bestv = v
            bestx = float(x)
    for ix in (i1, i2):
        if ix >= 0:
            x = float(np.clip(-0.5 * b[ix] / dx, 0.0, maxd))
            v = _pair_line_value(a, b, x, i1, i2, dx)
            if v < bestv - 1e-15:
                bestv = v
                bestx = x
    return bestx, bestv

def pair_descent(h, end_time, rng, max_cands=140):
    n = h.size
    if n < 2:
        return h, compute_c5(h)
    h = h.copy()
    best = compute_c5(h)
    rounds = 0
    while time.time() < end_time - 1 and rounds < 250:
        c_all = compute_all_c(h)
        G = build_gradients(h, n)
        cands = generate_candidates(h, n, c_all, rng)[:max_cands]
        dx = 2.0 / n
        moves = []
        for _, s, d in cands:
            if time.time() > end_time - 1:
                break
            maxd = float(min(h[s], 1.0 - h[d]))
            if maxd <= 1e-8:
                continue
            bvec = G[:, d] - G[:, s]
            delta, pred = pair_delta_opt(c_all, bvec, maxd, int(d) - int(s), n, dx)
            if delta > 1e-9 and pred < best - 1e-13:
                moves.append((pred, float(delta), int(s), int(d)))
        if not moves:
            break
        moves.sort()
        improved = False
        for pos in range(min(12, len(moves))):
            pred, delta, s, d = moves[pos]
            hp = h.copy()
            hp[s] -= delta
            hp[d] += delta
            np.clip(hp, 0.0, 1.0, out=hp)
            c_new = compute_c5(hp)
            if c_new + 1e-12 < best:
                h = hp
                best = c_new
                improved = True
                rounds += 1
                break
        if not improved:
            break
    return h, best

def make_thue(n, rng, scale=1.0):
    pi = np.zeros(n, dtype=np.uint8)
    for i in range(1, n):
        pi[i] = pi[i >> 1] ^ (i & 1)
    p = pi.astype(float)
    target = 0.5 * n
    cnt = float(p.sum())
    k = int(round(target - cnt))
    if k > 0:
        idx = np.flatnonzero(p == 0.0)
        if idx.size:
            rng.shuffle(idx)
            k = min(k, idx.size)
            p[idx[:k]] = 1.0
    elif k < 0:
        idx = np.flatnonzero(p == 1.0)
        if idx.size:
            rng.shuffle(idx)
            k = min(-k, idx.size)
            p[idx[:k]] = 0.0
    if scale < 1.0:
        p = 0.5 + 0.5 * (2.0 * p - 1.0) * scale
    return project_feasible(p)
