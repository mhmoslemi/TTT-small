        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h_star, n):
    """Build G matrix where G[ki, j] = d(c_k)/d(h[j]) / dx"""
    m_range = np.arange(-(n-1), n)
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))
    for ki, k in enumerate(m_range):
        jpk = np.arange(n) + k
        mask_p = (jpk >= 0) & (jpk < n)
        G[ki, mask_p] += (1 - h_star[jpk[mask_p]])
        jmk = np.arange(n) - k
        mask_m = (jmk >= 0) & (jmk < n)
        G[ki, mask_m] -= h_star[jmk[mask_m]]
    return G

def loa_iteration(h_star, n, solver='SCS'):
    """One LOA iteration: solve LP with linearized constraints"""
    dx = 2.0 / n
    corr_star = np.correlate(h_star, 1 - h_star, mode='full')
    G = build_gradient_matrix(h_star, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # Linearized: corr_star[ki] + G[ki,:] @ (h - h_star) <= t
    # => G[ki,:] @ h - t <= corr_star[ki] - G[ki,:] @ h_star
    rhs = corr_star - G @ h_star
    constraints.append(G @ h_var - t_var <= rhs)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                return None
    
    if h_var.value is None:
        return None
    return np.clip(h_var.value, 0, 1)
