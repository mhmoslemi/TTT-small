def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

        if abs(s - target) < 1e-12:

def lbfgs_smooth(h_init, n, alpha, maxiter, deadline, lam=100.0):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        sh = alpha * c
        sm = sh.max()
        log_sum_exps = sm + np.log(np.exp(sh - sm).sum())
        f = log_sum_exps / alpha
        s = x.sum()
        f += lam * (s - target) ** 2

                grad[kn:] += wk * (1.0 - x[:kn])
                grad[:n-kn] += -wk * x[kn:]

        grad += 2.0 * lam * (s - target)

            B[jj] = 0.0
            B[jj, vp] = -(1.0 - h[ip[vp]])
            B[jj, vm] += h[im[vm]]
        return A, B

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            h = best_h.copy()
            n_flips = rng.randint(5, 20)
            for _ in range(n_flips):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    h[rng.choice(oi)] = 0.0
                    h[rng.choice(zi)] = 1.0
            idx = np.argsort(h)[::-1]
            h = np.zeros(n)
            h[idx[:target]] = 1.0
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur_c = c.max() * dx
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            nz, no = len(zeros), len(ones)
            if nz == 0 or no == 0:
                break
            A, B = compute_AB(h, ones, zeros)
            best_imp = 0.0
            best_ii, best_jj = -1, -1
            for ii in range(nz):
                base = c + A[ii]
                ns = (base[None, :] + B).max(axis=1)
                sp = n - 1 - (ones - zeros[ii])
                sv = np.full(no, -1e9)
                valid = (sp >= 0) & (sp < 2 * n - 1)
                if valid.any():
                    vv = np.arange(no)[valid]
                    sv[valid] = base[sp[valid]] + B[vv, sp[valid]] + 2.0
                tot = np.maximum(ns, sv)
                jj = int(tot.argmin())
                imp = cur_c - tot[jj] * dx
                if imp > best_imp:
                    best_imp = imp
                    best_ii, best_jj = ii, jj
            if best_ii < 0:
                break
            h[zeros[best_ii]] = 1.0
            h[ones[best_jj]] = 0.0
            c = np.correlate(h, 1.0 - h, mode='full').astype(float)
            cur_c = c.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: L-BFGS-B continuous
    p1 = t0 + 0.12 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000]:
        if time.time() > p1:
            break
        h = lbfgs_run(h, n0, alpha, 150, 50.0)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Binary local search
    p2 = t0 + 0.45 * (budget_s - 5)
    hb, cb = bin_search(best_h, n0, p2, max_sweeps=800, n_restarts=5, seed=seed)
    if cb < best_c:
        best_c, best_h, best_n = cb, hb.copy(), n0

    # Phase 3: Continuous polish
    p3 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p3:
        h = best_h + 0.05 * rng.randn(n0)
        h = project_feasible(h, n0)
        for alpha in [5000, 50000, 200000]:
            if time.time() > p3:
                break
            h = lbfgs_run(h, n0, alpha, 200, 100.0)
            c = compute_c5(h, n0)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 4: Multi-resolution
    p4 = t0 + 0.65 * (budget_s - 5)
    for nh in [200, 300, 350]:
        if time.time() > p4 - 30:
            break
        if nh == best_n:
            continue
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh = lbfgs_run(hh, nh, 5000, 150, 50.0)
        ch = compute_c5(hh, nh)
        hb2, cb2 = bin_search(hh, nh, p4 - 10, max_sweeps=150, n_restarts=2, seed=seed + 1)
        if cb2 < best_c:
            best_c, best_h, best_n = cb2, hb2.copy(), nh
        elif ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), nh

    # Phase 5: Extended binary search
    p5 = dl - 30
    hb, cb = bin_search(best_h, best_n, p5, max_sweeps=2000, n_restarts=10, seed=seed + 10)
    if cb < best_c:
        best_c, best_h, best_n = cb, hb.copy(), best_n

    # Phase 6: Final polish
    if time.time() < dl - 5:
        hb, cb = bin_search(best_h, best_n, dl, max_sweeps=200, n_restarts=3, seed=seed + 30)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), best_n

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
