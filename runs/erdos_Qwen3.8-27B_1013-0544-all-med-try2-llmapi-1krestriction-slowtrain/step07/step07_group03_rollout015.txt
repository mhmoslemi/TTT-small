from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_objective(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    lse = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    return lse + penalty * (x.sum() - n / 2.0) ** 2

def lse_gradient(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    g = alpha * (c - m)
    gm = g.max()
    w = np.exp(g - gm)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        s_lo = max(-(n-1), -i)
        s_hi = min(n-1, n-1-i)
        if s_hi >= s_lo:
            kr = np.arange(s_lo + n - 1, s_hi + n)
            ir = np.arange(i + s_lo, i + s_hi + 1)
            grad[i] += (1.0 - x[ir]).dot(w[kr])
        s_lo2 = max(-(n-1), i - n + 1)
        s_hi2 = min(n-1, i)
        if s_hi2 >= s_lo2:

            grad[i] -= x[ir2].dot(w[kr2])
    grad *= alpha * dx
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_value(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    lse = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    return lse + penalty * (x.sum() - n / 2.0) ** 2

def lse_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    g = alpha * (c - m)
    gm = g.max()
    w = np.exp(g - gm)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        s_lo = max(-(n - 1), -i)
        s_hi = min(n - 1, n - 1 - i)
        if s_hi >= s_lo:
            kr = np.arange(s_lo + n - 1, s_hi + n)
            ir = np.arange(i + s_lo, i + s_hi + 1)
            grad[i] += (1.0 - x[ir]).dot(w[kr])
        s_lo2 = max(-(n - 1), i - n + 1)
        s_hi2 = min(n - 1, i)
        if s_hi2 >= s_lo2:
            kr2 = np.arange(s_lo2 + n - 1, s_hi2 + n)
            ir2 = np.arange(i - s_lo2, i - s_hi2 - 1, -1)
            grad[i] -= x[ir2].dot(w[kr2])
    grad *= alpha * dx
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return grad

def lbfgs_step(h, n, alpha, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_value, h, method='L-BFGS-B', jac=lse_grad,
                   bounds=bounds, args=(n, alpha, penalty),
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
    return project_feasible(res.x, n)

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                h[i], h[j] = 1.0, 0.0
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]] + (1.0 - h[im[msk]]) * 0
                # Recompute delta properly after swap
                h2 = h.copy()
                corr_new = np.correlate(h2, 1 - h2, mode='full').astype(float)
                corr = corr_new
                cur_c = corr.max() * dx
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    T = 0.02
    cool = 0.99998
    scale = 3000.0
    while time.time() < deadline:
        ai = rng.randint(n_zeros)
        bi = rng.randint(n_ones)
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
