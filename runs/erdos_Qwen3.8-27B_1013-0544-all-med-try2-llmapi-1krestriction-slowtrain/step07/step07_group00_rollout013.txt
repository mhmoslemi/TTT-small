    return np.correlate(h, 1 - h, mode='full') * dx

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:

            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_topk(h, n, k, alpha):
    """Gradient of smooth max over top-k correlations"""
    c = compute_corr(h, n)
    total = 2 * n - 1
    if k < total:
        top_idx = np.argpartition(c, -k)[-k:]
    else:
        top_idx = np.arange(total)
    top_c = c[top_idx]
    m = top_c.max()
    sh = alpha * (top_c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    # w is now weights on the top-k terms
    # Build full weight vector
    full_w = np.zeros(total)
    full_w[top_idx] = w
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in top_idx:
        wm = full_w[m_idx] * dx
        if wm < 1e-14:
            continue
        k_shift = m_idx - (n - 1)
        if k_shift >= 0:
            nv = n - k_shift
            grad[:nv] += wm * (1.0 - h[k_shift:])
            grad[k_shift:] += wm * (-h[:nv])
        else:
            kk = -k_shift
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

    """Local search: move mass from high-gradient to low-gradient positions"""

    if k < total:
        top_idx = np.argpartition(c, -k)[-k:]
        top_c = c[top_idx]
    else:
        top_idx = np.arange(total)
        top_c = c
    m = top_c.max()
    sh = alpha * (top_c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    val = m + (mx + np.log(w.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx_pos, m_idx in enumerate(top_idx):
        wm = w[idx_pos] * dx
        if wm < 1e-14:
            continue
        k_shift = m_idx - (n - 1)
        if k_shift >= 0:
            nv = n - k_shift
            grad[:nv] += wm * (1.0 - h[k_shift:])
            grad[k_shift:] += wm * (-h[:nv])
        else:
            kk = -k_shift
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return val, grad

def obj_pen(x, n, k, alpha, mu):
    c = compute_corr(x, n)
    total = 2 * n - 1
    if k < total:
        top_idx = np.argpartition(c, -k)[-k:]
        top_c = c[top_idx]
    else:
        top_c = c
    m = top_c.max()
    sh = alpha * (top_c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

    val, g = topk_grad_and_val(x, n, k, alpha)

def cooldown_search(h, n, deadline, k=10):
    """Simulated annealing style local search"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    temperature = 0.005
    steps = [0.01, 0.005, 0.002]
    while time.time() < deadline:
        step = steps[int(temperature * 2000)]
        if step <= 0:
            step = 0.001
        _, g = topk_grad_and_val(h, n, k, 2000)
        order = np.argsort(g)
        i = order[np.random.randint(0, min(20, n))]
        j = order[np.random.randint(max(0, n - 20), n)]
        if i == j:
            continue
        if np.random.rand() < 0.5:
            if h[i] + step > 1.0 or h[j] - step < 0.0:
                continue
            old_hi, old_hj = h[i], h[j]
            h[i] += step
            h[j] -= step
        else:
            if h[i] - step < 0.0 or h[j] + step > 1.0:
                continue
            old_hi, old_hj = h[i], h[j]
            h[i] -= step
            h[j] += step
        c = compute_c5(h, n)
        if c < cur_c or np.random.rand() < np.exp(-(c - cur_c) / temperature):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] = old_hi
            h[j] = old_hj
        temperature *= 0.995
    return best_h, best_c

    target = n / 2.0

        diff = target - h.sum()

            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

def _topk_weights(c, k, alpha):

        alloc = min(110, (deadline - time.time() - 180) * 0.5)
        if alloc < 25:
            continue
        h_t, c_t = lbfgs_opt(h_interp, n_try, time.time() + alloc, k=8)
        if time.time() < deadline - 50:
            h_t, c_t = slsqp_refine(h_t, n_try,
                                    min(time.time() + 40, deadline - 40),
                                    alpha=1000000, k=4)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 3: Diverse restarts at best_n
    restarts = []
    restarts.append(project_feasible(1.0 - best_h, best_n))
    restarts.append(project_feasible(best_h[::-1].copy(), best_n))
    restarts.append(project_feasible(best_h + rng.randn(best_n) * 0.07, best_n))
    freq = rng.randint(3, 9)
    amp = rng.uniform(0.04, 0.12)
    ph = rng.uniform(0, 2 * np.pi)
    xv = np.linspace(0, 2, best_n, endpoint=False)
    restarts.append(project_feasible(best_h + amp * np.cos(2 * np.pi * freq * xv / 2.0 + ph), best_n))
    restarts.append(project_feasible(best_h + amp * 0.7 * np.sin(2 * np.pi * (freq + 2) * xv / 2.0 + ph), best_n))

    for s in restarts:
        if time.time() > deadline - 80:
            break
        alloc = min(35, (deadline - time.time() - 25) * 0.4)
        if alloc < 12:
            break
        h_s, c_s = lbfgs_opt(s, best_n, time.time() + alloc * 0.75, k=8)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 4: Gradient-directed local search
    if time.time() < deadline - 15:
        best_h, best_c = gradient_local(best_h, best_n, deadline - 10, k=10)

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 8:
        best_h, best_c = slsqp_refine(best_h, best_n, deadline - 4,
                                      alpha=2000000, k=3, maxiter=3000)

    # Phase 6: Quick pair polish
    if time.time() < deadline:
        best_h, best_c = pair_polish_fast(best_h, best_n, deadline - 1, k=8, radius=50)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
