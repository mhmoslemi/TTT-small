import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of sum_i h[i]*(1-h[i+lag]) w.r.t. h"""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        # sum_{i=0}^{n-1-m} h[i](1-h[i+lag])
        grad[:n-m] += (1.0 - h[lag:])
        grad[lag:] -= h[:n-m]
    else:
        # sum_{i=-lag}^{n-1} h[i](1-h[i+lag]) = sum_{j=0}^{n-1+m} h[j](1-h[j+lag])
        # Let's just use: for lag<0, sum_i h[i](1-h[i+lag]) where i+lag is valid
        # i ranges from -lag to n-1
        # grad[i] = (1-h[i+lag]) for i in range(-lag, n)
        # grad[i+lag] = -h[i] for i in range(-lag, n)
        start = -lag  # m
        grad[start:] += (1.0 - h[:m])
        grad[:m] -= h[m:]
    return grad

def projected_gradient_descent(h, n, deadline, max_iter=5000):
    """PGD with momentum on the max of lag objectives"""
    h_curr = project_feasible(h, n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    momentum = np.zeros(n)
    alpha = 0.01  # step size
    beta = 0.9    # momentum coefficient
    top_k = 5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_k:]
        
        # Average gradient of top lags
        grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            grad += lag_gradient(h_curr, n, lag)
        grad /= len(top_idx)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h_curr - alpha * momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            # Increase step size slightly
            alpha = min(0.1, alpha * 1.02)
        else:
            # Decrease step size
            alpha = max(0.001, alpha * 0.8)
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def lags_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(200):
        tau = 0.5 * (lo + hi)
        proj = np.clip(h - tau, 0.0, 1.0)
        if proj.sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_obj_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    e = np.exp(sh - mx)
    ps = e.sum()
    lse_val = m + (mx + np.log(ps)) / alpha
    p = e / ps
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += p[k] * (1.0 - 2.0 * x)
        elif s > 0:
            grad[:n - s] += p[k] * (1.0 - x[s:])
            grad[s:] -= p[k] * x[:n - s]
        else:
            t = -s
            grad[t:] += p[k] * (1.0 - x[:t])
            grad[:t] -= p[k] * x[t:]
    grad *= dx
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    pen = lam * (x.sum() - n / 2.0) ** 2
    return lse_val + pen, grad

def lbfgs_run(h, n, alpha, lam, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj_grad, h, args=(n, alpha, lam), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_box_sum(res.x, n)
    except Exception:
        return project_box_sum(h, n)

def pgd_step(h, n, alpha, lr, momentum, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    p = np.exp(sh - mx)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += p[k] * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += p[k] * (1.0 - h[s:])
            grad[s:] -= p[k] * h[:n - s]
        else:
            t = -s
            grad[t:] += p[k] * (1.0 - h[:t])
            grad[:t] -= p[k] * h[t:]
    grad *= dx
    momentum = beta * momentum - lr * grad
    h_new = project_box_sum(h + momentum, n)
    return h_new, momentum

def pgd_optimize(h, n, deadline, alpha_init=20, alpha_end=5000, max_iter=8000):
    h_curr = project_box_sum(h, n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.15 * (2.0 / n)
    stagnant = 0
    total_iters = max_iter
    for it in range(total_iters):
        if time.time() > deadline:
            break
        frac = it / total_iters
        alpha = alpha_init * (alpha_end / alpha_init) ** frac
        h_new, momentum = pgd_step(h_curr, n, alpha, lr, momentum, beta)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            lr = min(0.3 * (2.0 / n), lr * 1.01)
            stagnant = 0
        else:
            h_curr = h_new
            lr *= 0.95
            stagnant += 1
            if stagnant > 100:
                momentum = np.zeros(n)
                stagnant = 0
    return best_h, best_c

def local_search_multi(h, n, deadline, top_k=3):
    h = project_box_sum(h, n)
    best_c = compute_c5(h, n)
    step = 0.02
    for _ in range(5):
        if time.time() > deadline:
            break
        c = lags_array(h, n)
        top_idx = np.argsort(c)[-top_k:]
        improved_any = False
        for s_idx in top_idx:
            if time.time() > deadline:
                break
            s = s_idx - (n - 1)
            if s == 0:
                continue
            m = abs(s)
            if s > 0:
                for i in range(0, n - m, 3):
                    j = i + m
                    if h[j] > h[i]:
                        delta = min(step, 0.4 * (h[j] - h[i]), h[j] - h[i])
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        h[i] = min(h[i], 1.0)
                        h[j] = max(h[j], 0.0)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved_any = True
                        else:
                            h[i] -= delta
                            h[j] += delta
            else:
                t = -s
                for i in range(t, n, 3):
                    j = i - t
                    if h[i] > h[j]:
                        delta = min(step, 0.4 * (h[i] - h[j]), h[i] - h[j])
                        if delta < 1e-10:
                            continue
                        h[j] += delta
                        h[i] -= delta
                        h[j] = min(h[j], 1.0)
                        h[i] = max(h[i], 0.0)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved_any = True
                        else:
                            h[j] -= delta
                            h[i] += delta
        if not improved_any:
            if step < 0.001:
                break
            step *= 0.5
    return project_box_sum(h, n), best_c

def refine_at_n(h, n, deadline, rng):
    h_curr = project_box_sum(h, n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    t_remaining = deadline - time.time()
    if t_remaining < 5:
        return best_h, best_c
    alpha_schedule = [10, 30, 100, 300, 1000, 3000, 10000]
    lam = n * 50.0
    for alpha in alpha_schedule:
        if time.time() > deadline - 3:
            break
        alloc = min(40.0, (deadline - time.time()) * 0.3)
        if alloc < 3:
            break
        h_new = lbfgs_run(h_curr, n, alpha, lam, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
    if time.time() < deadline - 10:
        pgd_end = min(deadline - 8, time.time() + min(40, (deadline - time.time()) * 0.4))
        h_pgd, c_pgd = pgd_optimize(h_curr, n, pgd_end, alpha_init=50, alpha_end=20000)
        if c_pgd < best_c - 1e-15:
            best_c = c_pgd
            best_h = h_pgd
    if time.time() < deadline - 3:
        h_ls, c_ls = local_search_multi(best_h, n, deadline - 1, top_k=5)
        if c_ls < best_c - 1e-15:
            best_c = c_ls
            best_h = h_ls
    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
