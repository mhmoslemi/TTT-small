            if s < 1e-14:

            if s < 1e-14:

        m = c.max()

        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = np.zeros(2 * n - 1)
                a1 = i + m_range
                m1 = (a1 >= 0) & (a1 < n)
                delta[m1] += (1.0 - h[a1[m1]])
                a2 = i - m_range
                m2 = (a2 >= 0) & (a2 < n)
                delta[m2] -= h[a2[m2]]
                b1 = j + m_range
                m3 = (b1 >= 0) & (b1 < n)
                delta[m3] -= (1.0 - h[b1[m3]])
                b2 = j - m_range
                m4 = (b2 >= 0) & (b2 < n)
                delta[m4] += h[b2[m4]]
                delta[n - 1] -= 2.0
                jmi = n - 1 + (j - i)
                imj = n - 1 + (i - j)
                if 0 <= jmi < 2 * n - 1:
                    delta[jmi] += 1.0
                if 0 <= imj < 2 * n - 1:
                    delta[imj] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    masks = make_masks(n)

    # Phase 1: Binary greedy
    p1 = t0 + 0.05 * total
    hb, cb = greedy_binary(best_h, n, p1, max_sweeps=80)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: L-BFGS-B with increasing alpha
    p2 = t0 + 0.50 * total
    h = best_h.copy()
    alphas = [50, 200, 800, 3000, 12000, 50000, 200000, 800000, 3000000,
              12000000, 50000000, 200000000, 800000000]
    for alpha in alphas:
        if time.time() > p2 - 3:
            break
        pen = 50.0 * alpha
        h = lbfgs_opt(h, n, alpha, pen, masks, maxiter=3000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Coordinate descent
    p3 = t0 + 0.62 * total
    hc = coord_descent(best_h, n, max_sweeps=400, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 4: More L-BFGS-B with very high alpha
    p4 = t0 + 0.80 * total
    h = best_h.copy()
    for alpha in [5000000, 25000000, 100000000, 500000000, 2000000000]:
        if time.time() > p4 - 3:
            break
        pen = 200.0 * alpha
        h = lbfgs_opt(h, n, alpha, pen, masks, maxiter=5000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Final coord descent
    p5 = t0 + 0.88 * total
    hc = coord_descent(best_h, n, max_sweeps=500, deadline=p5)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 6: Try smaller resolutions
    p6 = t0 + 0.96 * total
    for nh in [200, 300, 500]:
        if time.time() > p6 - 10:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        msk_n = make_masks(nh)
        hh = coord_descent(hh, nh, max_sweeps=60, deadline=p6 - 15)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        for alpha in [50000, 500000, 5000000]:
            if time.time() > p6 - 5:
                break
            hh = lbfgs_opt(hh, nh, alpha, 50.0 * alpha, msk_n, maxiter=1000)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
                masks = msk_n

    # Phase 7: Noise restart
    if time.time() < dl - 5:
        for trial in range(3):
            if time.time() > dl - 2:
                break
            scale = 0.005 * (trial + 1)
            h = best_h + scale * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [50000000, 500000000]:
                if time.time() > dl - 1:
                    break
                h = lbfgs_opt(h, n, alpha, 100.0 * alpha, masks, maxiter=2000)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
