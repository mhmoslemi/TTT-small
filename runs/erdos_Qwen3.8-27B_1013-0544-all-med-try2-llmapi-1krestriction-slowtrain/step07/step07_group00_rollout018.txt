def pair_polish(h, n, deadline, radius=30):
    """Coordinate descent pair updates."""
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(0, n, 1):  # every point
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            # Also try smaller steps on the best candidates
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    """Corrected gradient of smooth max of shift integrals."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-16:
            continue
        k_shift = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k_shift >= 0:
            nv = n - k_shift
            grad[k_shift:k_shift + nv] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[k_shift:k_shift + nv])
        else:
            kk = -k_shift
            nv = n - kk
            grad[:nv] += wm * (1.0 - h[kk:kk + nv])
            grad[kk:kk + nv] += wm * (-h[:nv])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_opt(h, n, deadline, alpha, mu, max_iter=400):
    bounds = [(0.0, 1.0)] * n
    tlim = deadline - time.time() - 3
    if tlim < 5:
        return h, compute_c5(h, n)
    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                      bounds=bounds, args=(n, alpha, mu),
                      options={'maxiter': max_iter, 'ftol': 1e-19,
                               'gtol': 1e-16, 'maxfun': max_iter * 50,
                               'maxls': 100, 'm': 100})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def multi_alpha_optimize(h, n, deadline):
    best_c = compute_c5(h, n)
    configs = [(100, 0.5), (500, 2.0), (2000, 10.0), (10000, 50.0),
               (50000, 200.0), (200000, 1000.0), (1000000, 5000.0),
               (3000000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        alloc = min(20, deadline - time.time() - 8)
        if alloc < 5:
            break
        h, c = lbfgs_opt(h, n, time.time() + alloc, alpha, mu, max_iter=300)
        if c < best_c:
            best_c = c
    return h, best_c

def chebyshev_start(n, rng, n_terms=20):
    """Generate a starting point from a random Chebyshev expansion."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    t = x - 1.0
    h = np.zeros(n)
    for k in range(n_terms):
        amp = rng.uniform(-0.3, 0.3) / (1.0 + 0.3 * k)
        if k == 0:
            amp = 1.0
        h += amp * np.cos(k * np.arccos(np.clip(t, -1, 1)))
    return project_feasible(h, n)

def paired_gradient_search(h, n, deadline, step=0.001, window=40):
    """Use corrected gradient to guide pair updates."""
    best_c = compute_c5(h, n)
    for alpha in [500, 5000, 50000, 200000]:
        while time.time() < deadline - 5:
            g = smooth_max_grad(h, n, alpha)
            order = np.argsort(g)
            improved = False
            for idx in range(0, n, 3):
                if time.time() > deadline - 5:
                    return h, best_c
                i = order[idx]
                for j_idx in range(idx + 1, min(idx + 1 + window, n)):
                    j = order[j_idx]
                    if i == j or g[j] - g[i] < 1e-12:
                        continue
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
            if not improved:
                break
    return h, best_c

def pair_polish(h, n, deadline, radius=30):
    """Fine coordinate descent pair polish."""
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
