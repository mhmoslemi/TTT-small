    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_all(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_shift in range(2 * n - 1):
        if w[k_shift] < 1e-14:
            continue
        k = k_shift - (n - 1)
        wm = w[k_shift] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def lbfgs_run(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxfun': maxiter * 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_flow(h, n, deadline, alpha0=500, alpha1=500000, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha0
    step = 0.01
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        g = smooth_max_grad(h, n, alpha)
        step = 0.5 / (it + 1)
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.5
    return best_h, best_c

def binary_pair_polish(h, n, deadline, radius=100, n_trials=600):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    trials = 0
    while trials < n_trials and time.time() < deadline:
        trials += 1
        i = rng.randint(n)
        lo = max(0, i - radius)
        hi = min(n, i + radius + 1)
        j = lo + rng.randint(hi - lo)
        if i == j:
            continue
        # Binary search for best transfer amount (positive means h[i]->h[j])
        for direction in [1, -1]:
            if direction == 1:
                if h[i] < 1e-8 or h[j] > 1.0 - 1e-8:
                    continue
                lo_t, hi_t = 1e-6, min(h[i], 1.0 - h[j])
            else:
                if h[j] < 1e-8 or h[i] > 1.0 - 1e-8:
                    continue
                lo_t, hi_t = 1e-6, min(h[j], 1.0 - h[i])
            best_t = 0.0
            best_cv = best_c
            for _ in range(18):
                mid_t = 0.5 * (lo_t + hi_t)
                if direction == 1:
                    h[i] += mid_t
                    h[j] -= mid_t
                else:
                    h[i] -= mid_t
                    h[j] += mid_t
                h = project_feasible(h, n)
                cv = compute_c5(h, n)
                if cv < best_cv:
                    best_cv = cv
                    best_t = mid_t
                    lo_t = mid_t
                else:
                    hi_t = mid_t
                # Undo
                if direction == 1:
                    h[i] -= mid_t
                    h[j] += mid_t
                else:
                    h[i] += mid_t
                    h[j] -= mid_t
            if best_t > 0:
                if direction == 1:
                    h[i] += best_t
                    h[j] -= best_t
                else:
                    h[i] -= best_t
                    h[j] += best_t
                h = project_feasible(h, n)
                cv = compute_c5(h, n)
                if cv < best_c:
                    best_c = cv
                    best_h = h.copy()
    return best_h, best_c

def fine_pair_polish(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS with penalty constraint, geometric alpha schedule
    configs = [
        (1000, 10.0, 300),
        (5000, 50.0, 300),
        (20000, 200.0, 300),
        (100000, 1000.0, 400),
        (500000, 5000.0, 400),
        (2000000, 20000.0, 500),
        (5000000, 50000.0, 500),
        (20000000, 100000.0, 600),
    ]
    for alpha, mu, maxit in configs:
        if time.time() > deadline - 180:
            break
        remaining = deadline - time.time() - 150
        if remaining < 30:
            break
        alloc = min(90, remaining * 0.4)
        h_new, c_new = lbfgs_run(h, n, alpha, mu, maxit, time.time() + alloc)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new

    # Phase 2: Gradient flow with Nesterov-like restarts
    if time.time() < deadline - 120:
        h_g, c_g = gradient_flow(h, n, time.time() + 70,
                                  alpha0=5000, alpha1=500000, n_iter=3000)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g
            h = h_g

    # Phase 3: Another round of L-BFGS starting from improved point
    if time.time() < deadline - 100:
        for alpha, mu, maxit in [(3000000, 30000, 500), (10000000, 100000, 500)]:
            if time.time() > deadline - 50:
                break
            h_new, c_new = lbfgs_run(h, n, alpha, mu, maxit, time.time() + 50)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 4: Binary search pair polish (key innovation)
    if time.time() < deadline - 30:
        alloc = min(120, deadline - time.time() - 20)
        best_h, best_c = binary_pair_polish(best_h, n, time.time() + alloc,
                                             radius=150, n_trials=800)

    # Phase 5: Fine pair polish
    if time.time() < deadline - 3:
        best_h, best_c = fine_pair_polish(best_h, n, deadline - 2, radius=40)

    # Final: one more L-BFGS squeeze
    if time.time() < deadline - 2:
        h_new, c_new = lbfgs_run(best_h, n, 10000000, 100000, 200, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
