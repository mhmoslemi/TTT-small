def compute_delta(h, n, i, j):
    """Compute the change in correlation when h[i]:0->1, h[j]:1->0."""
    L = 2*n - 1  # length of correlation
    delta = np.zeros(L)
    
    # Contribution of h[i] going 0->1
    # (1-h[i+s]) for valid s: i+s in [0,n) -> m in [n-1-i, 2n-2-i]
    lo1 = max(0, n-1-i)
    hi1 = min(L, 2*n-1-i)
    if lo1 < hi1:
        # i+s = i + (m-(n-1)) = i+m-n+1, for m in [lo1, hi1)
        idx = i + (np.arange(lo1, hi1) - (n-1))
        delta[lo1:hi1] += (1.0 - h[idx])
    
    # -h[i-s] for valid s: i-s in [0,n) -> m in [i, i+n-1] (clamped to [0,L))
    lo2 = max(0, i)
    hi2 = min(L, i+n)
    if lo2 < hi2:
        # i-s = i-(m-(n-1)) = i-m+n-1, for m in [lo2, hi2)
        idx = i - (np.arange(lo2, hi2) - (n-1))
        delta[lo2:hi2] -= h[idx]
    
    # Contribution of h[j] going 1->0 (subtract)
    lo1 = max(0, n-1-j)
    hi1 = min(L, 2*n-1-j)
    if lo1 < hi1:
        idx = j + (np.arange(lo1, hi1) - (n-1))
        delta[lo1:hi1] -= (1.0 - h[idx])
    
    lo2 = max(0, j)
    hi2 = min(L, j+n)
    if lo2 < hi2:
        idx = j - (np.arange(lo2, hi2) - (n-1))
        delta[lo2:hi2] += h[idx]
    
    # Cross corrections
    # s=0 (m=n-1): if i≠j, correction is -2
    # (This handles the double-counting when i-s=i and j-s=j for s=0)
    # Actually, for s=0, the m=i-s=i and m=i terms coincide, and m=j-s=j and m=j coincide.
    # The naive formula gives 2*(h[j]-h[i]) = 2 for our case. True value is 0.
    # So correction = -2.
    if i != j:
        delta[n-1] -= 2.0
    
    # s=j-i (m=n-1+j-i): correction is +1 (handles m=i = m=j-s coincidence)
    m1 = n - 1 + (j - i)
    if 0 <= m1 < L:
        delta[m1] += 1.0
    
    # s=i-j (m=n-1+i-j): correction is +1 (handles m=j = m=i-s coincidence)
    m2 = n - 1 + (i - j)
    if 0 <= m2 < L:
        delta[m2] += 1.0
    
    return delta
```

            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softobj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, 'full') * dx
    cs = c - c.max()
    ec = np.exp(alpha * cs)
    sm = ec.sum()
    return c.max() + np.log(sm)/alpha + mu*(x.sum()-n/2.0)**2

def softobj_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, 'full') * dx
    cs = c - c.max()
    ec = np.exp(alpha * cs)
    p = ec / ec.sum()
    
    grad = np.zeros(n)
    for m in range(len(p)):
        if p[m] < 1e-13:
            continue
        k = m - (n-1)
        if k >= 0:
            grad[:n-k] += p[m] * (1.0 - x[k:])
            grad[k:] -= p[m] * x[:n-k]
        else:
            grad[-k:] += p[m] * (1.0 - x[:n+k])
            grad[:n+k] -= p[m] * x[-k:]
    grad *= dx
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad
```

This should be fast. The gradient computation is O(n²) in the worst case (looping over all m), but in practice with the alpha-softmax, only the top few p[m] values are significant, so it's much faster.

```python
def binary_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    target = n // 2
    dx = 2.0 / n
    L = 2*n - 1
    rng = np.random.RandomState(seed)
    
    # Start with binary version
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1-h, 'full').astype(float)
    cur_c = corr.max() * dx
    
    best_h = h.copy()
    best_c = cur_c
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            # Random perturbation
            for _ in range(rng.randint(3, 8)):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = rng.choice(ones)
                b = rng.choice(zeros)
                h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1-h, 'full').astype(float)
            cur_c = corr.max() * dx
        
        for sweep in range(max_sweeps):
            if time.time() > deadline - 5:
                break
            improved = False
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # Try to find improving flip
            for i in zeros:
                found = False
                for j in ones:
                    # Compute delta
                    delta = np.zeros(L)
                    # h[i] 0->1 contribution
                    lo = max(0, n-1-i)  # start of (1-h[i+s]) range
                    hi = min(L, 2*n-1-i)
                    if lo < hi:
                        idx_arr = i + (np.arange(lo, hi) - (n-1))
                        delta[lo:hi] += (1.0 - h[idx_arr])
                    lo2 = max(0, i)
                    hi2 = min(L, i+n)
                    if lo2 < hi2:
                        idx_arr = i - (np.arange(lo2, hi2) - (n-1))
                        delta[lo2:hi2] -= h[idx_arr]
                    
                    # h[j] 1->0 contribution (subtract)
                    lo = max(0, n-1-j)
                    hi = min(L, 2*n-1-j)
                    if lo < hi:
                        idx_arr = j + (np.arange(lo, hi) - (n-1))
                        delta[lo:hi] -= (1.0 - h[idx_arr])
                    lo2 = max(0, j)
                    hi2 = min(L, j+n)
                    if lo2 < hi2:
                        idx_arr = j - (np.arange(lo2, hi2) - (n-1))
                        delta[lo2:hi2] += h[idx_arr]
                    
                    # Cross corrections
                    delta[n-1] -= 2.0
                    m1 = n - 1 + (j - i)
                    if 0 <= m1 < L:
                        delta[m1] += 1.0
                    m2 = n - 1 + (i - j)
                    if 0 <= m2 < L:
                        delta[m2] += 1.0
                    
                    nc = (corr + delta).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + delta
                        cur_c = nc
                        improved = True
                        found = True
                        break
                if found:
                    break
            if not improved:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
```

This is O(n³) per sweep but with low constant factors thanks to numpy. For n=250, each (i,j) evaluation involves maybe 4 numpy array operations of length ~250, so about 1000 operations. With 62500 (i,j) pairs, that's about 62.5M operations per sweep. In numpy, this should take about 1-3 seconds per sweep. So we can do 100+ sweeps.

But actually, most of the time we'll find an improving flip quickly (before trying all pairs), so the average will be much faster.
