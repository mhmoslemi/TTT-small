    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h

            tot = h.sum()
            if tot < 1e-14:

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2

    bounds = [(0.0, 1.0)] * n
    configs = [(500, 1.0), (5000, 5.0), (50000, 20.0), (200000, 100.0), (1000000, 500.0)]

            break
        try:
            res = minimize(sm_objective, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-14, 'maxfun': max_iter * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new
        except Exception:

    return h, best_c

def simulated_annealing(h, n, deadline, rng):

    current_c = best_c
    T = 0.01
    T_min = 1e-7
    cool_rate = 0.995

            h[j] -= delta

def block_shift_search(h, n, deadline, rng):
    """Try shifting blocks of consecutive values"""
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        # Pick a random block
        block_len = rng.randint(2, 8)
        start = rng.randint(0, n - block_len + 1)
        shift = rng.choice([-2, -1, 1, 2])
        end = start + block_len
        
        # Try shifting the block
        new_h = h.copy()
        # Move the "mass" in the block by shift positions
        src = slice(start, end)
        dst_start = start + shift
        dst_end = end + shift
        if dst_start < 0 or dst_end > n:

        # Swap: move h[src] to dst and h[dst] to src

        block_val = h[src].copy()
        dst_val = h[dst_start:dst_end].copy()
        # We can't just swap because the sum constraint

        new_h[dst_start:dst_end] = block_val

        new_c = compute_c5(new_h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(w)):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs(h, n, deadline, max_iter=200):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 1.0), (5000, 5.0), (50000, 20.0), (500000, 200.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iter * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def guided_sa(h, n, deadline, rng):
    """Simulated annealing with gradient-guided move selection"""
    best_c = compute_c5(h, n)
    current_c = best_c
    T = 0.005
    T_min = 1e-8
    cool = 0.998
    delta = 0.01
    alpha = 5000
    
    while time.time() < deadline and T > T_min:
        # Compute gradient to guide moves
        g = sm_grad(h, n, alpha, 0.0)
        # Bias: pick i from high gradient, j from low gradient
        # Use top/bottom 20% for biased selection
        threshold_hi = np.percentile(g, 80)
        threshold_lo = np.percentile(g, 20)
        hi_candidates = np.where((g >= threshold_hi) & (h < 0.95))[0]
        lo_candidates = np.where((g <= threshold_lo) & (h > 0.05))[0]
        
        if len(hi_candidates) == 0 or len(lo_candidates) == 0:
            i = rng.randint(n)
            j = rng.randint(n)
        else:
            i = hi_candidates[rng.randint(len(hi_candidates))]
            j = lo_candidates[rng.randint(len(lo_candidates))]
        
        if i == j:
            continue
        
        moved = False
        if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
            h[i] += delta
            h[j] -= delta
            new_c = compute_c5(h, n)
            dc = new_c - current_c
            if dc < 0 or T > 0 and rng.rand() < np.exp(-dc / T):
                current_c = new_c
                if new_c < best_c:
                    best_c = new_c
            else:
                h[i] -= delta
                h[j] += delta
            moved = True
        elif h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
            h[i] -= delta
            h[j] += delta
            new_c = compute_c5(h, n)
            dc = new_c - current_c
            if dc < 0 or T > 0 and rng.rand() < np.exp(-dc / T):
                current_c = new_c
                if new_c < best_c:
                    best_c = new_c
            else:
                h[i] += delta
                h[j] -= delta
            moved = True
        
        if moved:
            T *= cool
        # Adapt delta
        if rng.rand() < 0.005:
            delta *= rng.choice([0.6, 1.5])
            delta = np.clip(delta, 0.003, 0.08)
    
    return h, best_c

def pair_polish(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (-x[k:k+nv])
            grad[k:] += wm * (1.0 - x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:kk+nv])
            grad[kk:] += wm * (-x[:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs(h, n, deadline, max_iter=200):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 1.0), (5000, 5.0), (50000, 20.0), (500000, 200.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iter * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def guided_sa(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    current_c = best_c
    # Epochs: (temperature, num_iters, delta)
    epochs = [
        (0.02, 40000, 0.03),
        (0.01, 50000, 0.02),
        (0.005, 60000, 0.01),
        (0.002, 70000, 0.005),
        (0.001, 80000, 0.003),
        (0.0005, 80000, 0.002),
        (0.0002, 80000, 0.001),
    ]
    for T, iters, delta in epochs:
        for _ in range(iters):
            if time.time() > deadline:
                return h, best_c
            c = compute_c5_vec(h, n)
            k_star = int(np.argmax(c))
            k = k_star - (n - 1)
            # Find top contributing positions for this lag
            if k >= 0:
                nv = n - k
                contrib = h[k:k+nv] * (1 - h[:nv])
            else:
                kk = -k
                nv = n - kk
                contrib = h[:nv] * (1 - h[kk:kk+nv])
            # Pick a random top contributor
            top_n = min(5, len(contrib))
            top_idx = np.argpartition(contrib, -top_n)[-top_n:]
            idx = top_idx[rng.randint(top_n)]
            if k >= 0:
                i = idx  # position in (1-h) factor
                j = k + idx  # position in h factor
            else:
                i = idx + (-k)  # position in (1-h) factor  
                j = idx  # position in h factor
            # We want to increase h[i] or decrease h[j]
            # Try increase h[i]
            moved = False
            if h[i] + delta <= 1.0:
                # Find where to take from
                j_src = rng.randint(n)
                if j_src == i:
                    j_src = (i + 1) % n
                if h[j_src] - delta >= 0.0:
                    h[i] += delta
                    h[j_src] -= delta
                    new_c = compute_c5(h, n)
