    h=np.asarray(x,float)
    c=all_lags_hx(h,n)

       L=idx-center

  center=n-1
  if p>0:

     ks=np.arange(1,p+1)

def project_to_polytope(y, target):
    y = np.asarray(y, dtype=float)
    target = float(target)
    n = y.size
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):

        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-15:
                h += (diff/tot) * room
                h = np.clip(h,0,1)
        else:
            tot = float(h.sum())
            if tot > 1e-15:

                h = np.clip(h,0,1)
    diff = target - float(np.sum(h))
    if diff > 0:
        idx = int(np.argmax(1.0 - h))

import numpy as np
import time
from scipy.optimize import minimize

def project_to_polytope(y, target):
    y = np.asarray(y, dtype=float)
    n = y.size
    target = float(target)
    if target <= 0.0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-15:
                h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = float(h.sum())
            if tot > 1e-15:
                h = np.clip(h + diff / tot * h, 0.0, 1.0)
    for _ in range(4):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + diff)
    return h

def full_lags(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def compute_c5(h, n):
    return float(np.max(full_lags(h, n)))

def resample(prev, n):
    x0 = np.arange(prev.size) * (2.0 / prev.size)
    x1 = np.arange(n) * (2.0 / n)
    return np.interp(x1, x0, prev)

def select_active(c, m, gamma):
    m = max(1, min(int(m), c.size))
    if m == c.size:
        idx = np.arange(c.size)
    else:
        idx = np.argpartition(c, c.size - m)[-m:]
    valt = float(c[idx].max())
    if gamma > 1e-12:
        z = gamma * (c[idx] - valt)
        z = z - float(z.max())
        w = np.exp(z)
    else:
        w = np.ones(m)
    s = float(w.sum())
    if s <= 0.0:
        w = np.ones(m)
    else:
        w = w / s
    return idx.astype(int), w.astype(float)

def lag_grad(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    m = abs(int(lag))
    if m >= n:
        return g
    if lag > 0:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    else:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def active_fun(x, act, w, n):
    return float(np.dot(w, full_lags(np.asarray(x, dtype=float), n)[act]))

def active_jac(x, act, w, n):
    h = np.asarray(x, dtype=float)
    g = np.zeros(n)
    ctr = n - 1
    for i, wt in zip(act, w):
        if wt > 1e-15:
            g += wt * lag_grad(h, int(i) - ctr, n)
    return (2.0 / n) * g

def sum_fun(x, target):
    return float(np.sum(x) - target)

def sum_jac(x, target):
    return np.ones_like(x, dtype=float)

def slsqp_active(x0, n, target, act, w, iters):
    x0 = np.asarray(x0, dtype=float).copy()
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (target,)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(active_fun, x0, args=(act, w, n), jac=active_jac,
                       method="SLSQP", bounds=bnds, constraints=cons,
                       options={"maxiter": int(iters), "ftol": 1e-12})
        if res.x is not None:
            return project_to_polytope(res.x, target)
    except Exception:
        pass
    return x0

def polish(h0, n, stop, target):
    h = project_to_polytope(np.asarray(h0, dtype=float), target)
    best = compute_c5(h, n)
    ctr = n - 1
    for _ in range(250):
        if time.time() > stop:
            break
        c = full_lags(h, n)
        act, w = select_active(c, 12, 150.0)
        g = np.zeros(n)
        for i, wt in zip(act, w):
            g += wt * lag_grad(h, int(i) - ctr, n)
        g = (g - g.mean()) * (2.0 / n)
        sc = float(np.max(np.abs(g)))
        if sc < 1e-8:
            break
        g /= sc
        hit = False
        for st in (0.06, 0.025, 0.01, 0.004):
            hp = project_to_polytope(h - st * g, target)
            c2 = compute_c5(hp, n)
            if c2 < best - 1e-12:
                h = hp
                best = c2
                hit = True
                break
        if not hit:
            break
    return h, best

def binarize(h, n):
    s = n // 2
    b = np.zeros(n, dtype=float)
    if s > 0:
        b[np.argpartition(h, -s)[-s:]] = 1.0
    return b

def flip_indices(n):
    emp = np.empty(0, dtype=int)
    lp, lm, rp, rm = [], [], [], []
    ctr = n - 1
    for p in range(n):
        if p:
            k = np.arange(1, p + 1, dtype=int)
            lp.append(ctr + k); lm.append(ctr - k)
        else:
            lp.append(emp); lm.append(emp)
        if p < n - 1:
            k = np.arange(1, n - p, dtype=int)
            rp.append(ctr + k); rm.append(ctr - k)
        else:
            rp.append(emp); rm.append(emp)
    return tuple(lp), tuple(lm), tuple(rp), tuple(rm)

def flip_bit(bits, cnt, p, old, idx):
    lp, lm, rp, rm = idx
    if p:
        left = bits[:p][::-1]
        if old > 0.5:
            cnt[lp[p]] -= 1.0 - left
            cnt[lm[p]] += left
        else:
            cnt[lp[p]] += 1.0 - left
            cnt[lm[p]] -= left
    if p < bits.size - 1:
        right = bits[p + 1:]
        if old > 0.5:
            cnt[rp[p]] += right
            cnt[rm[p]] -= 1.0 - right
        else:
            cnt[rp[p]] -= right
            cnt[rm[p]] += 1.0 - right
    bits[p] = 1.0 - old

def do_swap(bits, cnt, a, b, idx):
    oa = float(bits[a]); ob = float(bits[b])
    flip_bit(bits, cnt, a, oa, idx)
    flip_bit(bits, cnt, b, ob, idx)
    return oa, ob

def undo_swap(bits, cnt, a, b, oa, ob, idx):
    flip_bit(bits, cnt, b, 1.0 - ob, idx)
    flip_bit(bits, cnt, a, 1.0 - oa, idx)

def binary_greedy(bits, cnt, n, idx, stop, rnds):
    ctr = n - 1
    for _ in range(rnds):
        if time.time() > stop:
            break
        mx = float(cnt.max())
        tc = min(24, cnt.size)
        top = np.argpartition(cnt, -tc)[-tc:] if tc < cnt.size else np.arange(cnt.size)
        top = top[np.argsort(cnt[top])[::-1]]
        top_sum = float(cnt[top].sum())
        found = False
        for k in top:
            lag = int(k) - ctr
            m = abs(lag)
            if m == 0:
                continue
            low = bits[:n - m]; high = bits[m:]
            if lag > 0:
                cand = np.nonzero((low < 0.5) & (high > 0.5))[0]
            else:
                cand = np.nonzero((low > 0.5) & (high < 0.5))[0]
            if cand.size == 0:
                continue
            ba, bb = -1, -1
            bnm = mx + 1e-9
            bnt = top_sum + 1e-9
            for a in cand:
                if time.time() > stop:
                    return
                a = int(a); b = a + m
                oa, ob = do_swap(bits, cnt, a, b, idx)
                nm = float(cnt.max())
                nt = float(cnt[top].sum())
                undo_swap(bits, cnt, a, b, oa, ob, idx)
                ok = nm < mx - 1e-9 or (nm <= mx + 1e-9 and nt < top_sum - 1e-6)
                if ok and (nm < bnm - 1e-9 or (abs(nm - bnm) <= 1e-9 and nt < bnt - 1e-6)):
                    ba, bb, bnm, bnt = a, b, nm, nt
            if ba >= 0:
                do_swap(bits, cnt, ba, bb, idx)
                found = True
                break
        if not found:
            break

def binary_optimize(b0, n, stop, rng, target_val=0.3807):
    b = binarize(np.asarray(b0, dtype=float), n)
    cnt = np.correlate(b, 1.0 - b, mode="full")
    best_b = b.copy()
    best_cnt = cnt.copy()
    best_mx = float(best_cnt.max())
    idx = flip_indices(n)
    cyc = 0
    while time.time() < stop - 3:
        b = best_b.copy(); cnt = best_cnt.copy()
        binary_greedy(b, cnt, n, idx, min(stop - 4, time.time() + 25), 700)
        mx = float(cnt.max())
        if mx < best_mx - 1e-9:
            best_mx = mx; best_b = b.copy(); best_cnt = cnt.copy()
        if best_mx * 2.0 / n < target_val - 1e-12:
            break
        if time.time() > stop - 6:
            break
        imp = False
        for _ in range(8):
            if time.time() > stop - 4:
                break
            b = best_b.copy(); cnt = best_cnt.copy()
            for _ in range(3):
                ones = np.flatnonzero(b > 0.5); zeros = np.flatnonzero(b < 0.5)
                if ones.size == 0 or zeros.size == 0:
                    break
                a = int(ones[int(rng.randint(ones.size))])
                c = int(zeros[int(rng.randint(zeros.size))])
                do_swap(b, cnt, a, c, idx)
            binary_greedy(b, cnt, n, idx, min(stop - 4, time.time() + 8), 250)
            mx = float(cnt.max())
            if mx < best_mx - 1e-9:
                best_mx = mx; best_b = b.copy(); best_cnt = cnt.copy(); imp = True
                if best_mx * 2.0 / n < target_val - 1e-12:
                    return best_b, best_mx * 2.0 / n
        cyc = 0 if imp else cyc + 1
        if cyc > 7 and not imp:
            break
    final_cnt = np.correlate(best_b, 1.0 - best_b, mode="full")
    return best_b, float(final_cnt.max()) * 2.0 / n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        base = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(base.size)
    except Exception:
        n0 = 300
        x = np.arange(n0) * (2.0 / n0)
        base = 0.5 - 0.5 * np.cos(np.pi * x)
    if n0 < 16:
        n0 = 300
        base = project_to_polytope(base * 0.0 + 0.5, 0.5 * n0)
    base = project_to_polytope(base, 0.5 * n0)
    best_h = base.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    stop = min(deadline - 320, time.time() + 230)
    if stop < time.time() + 20:
        stop = time.time() + 20
    if stop > deadline - 2:
        stop = deadline - 2
    if stop > time.time():
        h, c = improve_h(best_h, best_n, stop, rng)
        if c < best_c:
            best_h, best_c, best_n = h, c, h.size

    for ntr in (best_n + 60, best_n + 120, best_n - 60, 360, 420, 480):
        ntr = int(ntr)
        if time.time() > deadline - 150:
            break
        if not (120 <= ntr < 900 and ntr % 2 == 0 and ntr != best_n):
            continue
        h0 = project_to_polytope(resample(best_h, ntr), 0.5 * ntr)
        stop = min(deadline - 130, time.time() + 75)
        if stop < time.time() + 15:
            stop = time.time() + 15
        if stop > deadline - 2:
            stop = deadline - 2
        if stop <= time.time():
            break
        h, c = improve_h(h0, ntr, stop, rng)
        if c < best_c:
            best_h, best_c, best_n = h, c, ntr

    for nbin in (best_n, 300, 360, 420, 480):
        nbin = int(nbin)
        if time.time() > deadline - 50:
            break
        if not (120 <= nbin < 900 and nbin % 2 == 0):
            continue
        h0 = best_h if nbin == best_n else resample(best_h, nbin)
        stop = min(deadline - 20, time.time() + 45)
        if stop < time.time() + 10:
            stop = time.time() + 10
        if stop > deadline - 1:
            stop = deadline - 1
        if stop <= time.time():
            break
        b, c = binary_optimize(h0, nbin, stop, rng, 0.3807)
        if c < best_c:
            best_h, best_c, best_n = b, c, nbin
        if best_c < 0.3807:
            break

    if time.time() < deadline - 10:
        act, w = select_active(full_lags(best_h, best_n), 2 * best_n - 1, 400.0)
        h = slsqp_active(best_h, best_n, 0.5 * best_n, act, w, 80 if best_n <= 350 else 50)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_h, best_c = h, c
    if time.time() < deadline - 2:
        h, c = polish(best_h, best_n, deadline - 1, 0.5 * best_n)
        if c < best_c:
            best_h, best_c = h, c

    best_h = project_to_polytope(best_h, 0.5 * best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
