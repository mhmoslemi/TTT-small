    g = np.zeros(n)
    if k >= 0:

        g[:n-k] += (1 - h[k:])
        # h[i-k] term: exists for i in [k, n-1]

        # h[i-k] = h[i+kk] term: exists for i in [0, n-kk-1]
        g[:n-kk] -= h[kk:]
    return g * dx

def penalty_objective(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    # Smoothed max
    m = c.max()
    logsumexp = m + np.log(np.sum(np.exp(c - m)))
    smooth_max = (logsumexp) / alpha
    # Penalty
    penalty = lam * (np.sum(x) - n/2.0)**2
    return smooth_max + penalty

def penalty_gradient(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()

    p = exp_c / exp_c.sum()  # softmax weights
    # Gradient of smoothed max: sum_k p_k * grad C_k
    g = np.zeros(n)
    for k_idx in range(len(c)):
        if p[k_idx] < 1e-10:

        k = k_idx - (n - 1)
        g += p[k_idx] * gradient_of_lag(x, n, k)
    g /= alpha
    # Gradient of penalty
    g += 2 * lam * (np.sum(x) - n/2.0) * np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n-kk])
        g[:n-kk] -= h[kk:]
    return g * dx

def smoothed_objective(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    return float(m + np.log(np.sum(np.exp(c - m))) / alpha)

def smoothed_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    e = np.exp(c - m)
    p = e / e.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-13: continue
        k = idx - (n - 1)
        if k >= 0:
            g[:n-k] += p[idx] * (1.0 - h[k:])
            g[k:] -= p[idx] * h[:n-k]
        else:
            kk = -k
            g[kk:] += p[idx] * (1.0 - h[:n-kk])
            g[:n-kk] -= p[idx] * h[kk:]
    return g * (2.0/n) / alpha

def sum_constraint(h, n):
    return np.sum(h) - n/2.0

def slsqp_smooth(h, n, alpha, maxiter=3000):
    bounds = [(0,1)]*n
    cons = [{'type':'eq','fun': lambda x: sum_constraint(x,n),'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smoothed_objective, h, args=(n,alpha), method='SLSQP',
                      bounds=bounds, constraints=cons, options={'maxiter':maxiter,'ftol':1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def two_point_local_search(h, n, deadline):

    best_c = compute_c5(h, n)
    recent_moves = {}
    
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = grad_lag(h, n, lag)
        
        # Find best pair using gradient
        # We want to decrease h[i] where g[i] > 0 and increase h[j] where g[j] < 0
        # The gain from transferring δ from i to j is approximately δ * (g[j] - g[i])
        # So we want g[i] - g[j] to be large
        
        # Sort by gradient
        sort_idx = np.argsort(g)
        # Best to decrease: largest g → sort_idx[-1], sort_idx[-2], etc.
        # Best to increase: smallest g → sort_idx[0], sort_idx[1], etc.
        
        found = False
        for a_rank in range(5):
            for b_rank in range(5):
                i = int(sort_idx[-1 - a_rank])  # high g (decrease)
                j = int(sort_idx[b_rank])  # low g (increase)
                if i == j: continue
                if g[i] <= g[j]: continue
                
                # Check recent moves to avoid cycling
                move_key = (min(i,j), max(i,j))
                if recent_moves.get(move_key, 0) > 10: continue
                
                # Determine direction
                if h[i] > h[j]:
                    max_step = min(h[i], 1.0 - h[j])
                else:
                    max_step = min(1.0 - h[i], h[j])
                    # In this case we'd increase i and decrease j
                    # But that goes AGAINST the gradient for this lag
                    # So skip
                    continue
                
                if max_step < 1e-8: continue
                
                # Binary search on step
                lo, hi = 0.0, max_step
                for _ in range(25):
                    mid = (lo + hi) / 2
                    h_test = h.copy()
                    h_test[i] -= mid
                    h_test[j] += mid
                    c_test = compute_c5(h_test, n)
                    if c_test < best_c - 1e-14:

                    else:
                        hi = mid

                    step = lo
                    if step > 1e-8:
                        h[i] -= step
                        h[j] += step
                        best_c = compute_c5(h, n)
                        recent_moves[move_key] = recent_moves.get(move_key, 0) + 1
                        found = True
                        break
            if found: break

                    h[j] += max_step

                        found = True

                    else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    return g * dx

def smoothed_obj(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    return float(m + np.log(np.sum(np.exp(c - m))) / alpha)

def smoothed_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    e = np.exp(c - m)
    p = e / e.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[:n - k] += w * (1.0 - h[k:])
            g[k:] -= w * h[:n - k]
        else:
            kk = -k
            g[kk:] += w * (1.0 - h[:n - kk])
            g[:n - kk] -= w * h[kk:]
    return g * (2.0 / n) / alpha

def sum_con_fun(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smoothed_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_grad_descend(h, n, alpha, n_steps=200, step_init=0.5):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    step = step_init
    for it in range(n_steps):
        g = smoothed_grad(h_curr, n, alpha)
        h_new = h_curr - step * g
        lo = h_new.min() - 1.0
        hi = h_new.max() + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            hp = np.clip(h_new - mid, 0.0, 1.0)
            if hp.sum() > n / 2.0:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2.0, 0.0, 1.0)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.1, 2.0)
        else:
            step *= 0.7
        h_curr = h_new
        if step < 1e-6:
            break
    return best_h, best_c

def two_point_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    move_count = {}
    stall = 0
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = grad_lag(h, n, lag)
        sort_idx = np.argsort(g)
        found = False
        for a_r in range(10):
            if found:
                break
            for b_r in range(10):
                i = int(sort_idx[n - 1 - a_r])
                j = int(sort_idx[b_r])
                if i == j or g[i] <= g[j] + 1e-10:
                    continue
                key = (min(i, j), max(i, j))
                if move_count.get(key, 0) > 20:
                    continue
                if h[i] <= h[j]:
                    continue
                max_step = min(h[i], 1.0 - h[j])
                if max_step < 1e-8:
                    continue
                lo_s, hi_s = 0.0, max_step

                    mid = (lo_s + hi_s) / 2.0
                    ht = h.copy()
                    ht[i] -= mid
                    ht[j] += mid
                    if compute_c5(ht, n) < best_c - 1e-14:
                        lo_s = mid
                    else:
                        hi_s = mid
                if hi_s - lo_s > 1e-10:
                    step = lo_s
                    if step > 1e-9:
                        h[i] -= step
                        h[j] += step
                        best_c = compute_c5(h, n)
                        move_count[key] = move_count.get(key, 0) + 1
                        stall = 0
                        found = True
                        break
        if not found:
            stall += 1
            rng = np.random.RandomState(int(time.time() * 1000) % 100000)
            improved = False
            for _ in range(100):
                a = rng.randint(n)
                b = rng.randint(n)
                if a == b:
                    continue
                if h[a] > h[b] + 1e-6:
                    d = min(h[a] - h[b], 1.0 - h[b], h[a]) * 0.3
                    if d < 1e-8:
                        continue
                    h[a] -= d
                    h[b] += d
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                        break
                    h[a] += d
                    h[b] -= d
                elif h[b] > h[a] + 1e-6:
                    d = min(h[b] - h[a], 1.0 - h[a], h[b]) * 0.3
                    if d < 1e-8:
                        continue
                    h[a] += d
                    h[b] -= d
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                        break
                    h[a] -= d
                    h[b] += d
            if not improved and stall > 50:
                break
    return h, best_c

def multi_lag_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        top_k = np.argsort(c)[-3:]

                    if hi_s - lo_s > 1e-10 and lo_s > 1e-9:
                        h[i] -= lo_s
                        h[j] += lo_s
                        best_c = compute_c5(h, n)

def interp_1d(x_old, y_old, x_new):
    n_new = len(x_new)
    y_new = np.empty(n_new)
    idx = 0
    for i in range(n_new):
        while idx < len(x_old) - 2 and x_old[idx + 1] < x_new[i]:
            idx += 1
        if x_new[i] <= x_old[idx]:
            y_new[i] = y_old[idx]
        elif x_new[i] >= x_old[-1]:
            y_new[i] = y_old[-1]
        else:
            t = (x_new[i] - x_old[idx]) / (x_old[idx + 1] - x_old[idx])
            y_new[i] = y_old[idx] * (1 - t) + y_old[idx + 1] * t
    return y_new
