    target = n // 2

    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    
    best_c = cur_c
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    
    T0 = 0.1  # initial temperature
    T1 = 0.001  # final temperature
    # We'll use adaptive cooling based on time
    
    t_start = time.time()
    n_iters = 0
    
    while time.time() < deadline:
        # Adaptive temperature
        elapsed = time.time() - t_start
        total_time = deadline - t_start
        if elapsed > total_time * 0.9:
            break
        progress = elapsed / total_time
        T = T0 * (T1/T0) ** progress  # geometric cooling
        
        # Choose a random swap
        i = rng.choice(sorted(zeros_set))
        j = rng.choice(sorted(ones_set))
        
        delta = compute_delta(h, i, j, n)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        delta_c = new_c - cur_c
        
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            ones_set.discard(j)
            ones_set.add(i)
            zeros_set.discard(i)
            zeros_set.add(j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        n_iters += 1
    
    return best_h, best_c

For the negative correction at n-1: since it decreases the value, the max can only stay the same or decrease. But since we're taking max, it doesn't matter (the max is at least as large as before at other positions).

Wait, actually: new = base + corrections. The max of new is:
max over all positions of (base[pos] + correction[pos])
= max( max over positions not in {n-1, n-1+(j-i), n-1+(i-j)} of base[pos],
         base[n-1] - 2,
         base[n-1+(j-i)] + 1,
         base[n-1+(i-j)] + 1 )
= max( max over "other" positions of base[pos],
         base[n-1] - 2,
         base[n-1+(j-i)] + 1,
         base[n-1+(i-j)] + 1 )

The "max over other positions" is either base_max (if the argmax is not one of the 3 special positions) or the second-largest value of base (if the argmax is at one of the 3 special positions).

This is getting complicated. For simplicity, let me just compute the full max each time:
new_corr = corr + D_j - D_i
new_corr[n-1] -= 2
if 0 <= n-1+(j-i) < 2n-1: new_corr[n-1+(j-i)] += 1
if 0 <= n-1+(i-j) < 2n-1: new_corr[n-1+(i-j)] += 1
new_c = new_corr.max() * dx

This is O(n) per evaluation (one addition, two scalar modifications, one max). In NumPy, this is a few operations on 499-element arrays (for n=250), which should be very fast (~1-2μs per operation).

Let me precompute D:

```python
def precompute_D(h, n):
    """Precompute D_p = h[p-s] + h[p+s] for all p, as a 2D array of shape (n, 2n-1)"""
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    D = np.zeros((n, 2*n-1))
    for p in range(n):
        # h[p+s] for valid s
        valid_plus = (p + s >= 0) & (p + s < n)
        D[p, valid_plus] += h[p + s[valid_plus]]
        # h[p-s] for valid s
        valid_minus = (p - s >= 0) & (p - s < n)
        D[p, valid_minus] += h[p - s[valid_minus]]
    return D
```

This takes O(n^2) time and O(n^2) memory. For n=250, that's 62500 values in memory (500KB) and 62500 operations. Very fast.

But we need to update D when h changes! When we swap h[i]:0→1 and h[j]:1→0:
- D_p changes for all p where p±s = i or p±s = j for some s
- Specifically, D_p[i-related] changes for all p (since for any p, there exists s such that p+s=i or p-s=i)
- Actually, D_p[s] = h[p+s] + h[p-s]. The value h[i] appears in D_p when p+s=i (s=i-p) or p-s=i (s=p-i).
- So D_p changes at positions s=i-p and s=p-i (in the s-indexing), i.e., at array indices (n-1)+(i-p) and (n-1)+(p-i).

So when h[i] changes by Δh_i (+1 for i, -1 for j):
D_p[(n-1)+(i-p)] += Δh_i (if in range)
D_p[(n-1)+(p-i)] += Δh_i (if in range)
D_p[(n-1)+(j-p)] += Δh_j (if in range)
D_p[(n-1)+(p-j)] += Δh_j (if in range)

Where Δh_i = +1 and Δh_j = -1.

So I need to update 4 positions in each of the n rows of D. That's O(n) per swap, which is the same as before.

Hmm, so the update is O(n) and the evaluation is O(n), giving O(n) per iteration. That's the best we can do without more complex data structures.

For n=250, O(n) = 250 operations per iteration. At ~1μs per operation (NumPy scalar ops on array elements), that's ~250μs per iteration. In 1000 seconds, that's ~4M iterations. That should be enough for SA.

Actually, I realize I can be even smarter. Since I only need to check if the max improves, and the max is achieved at a specific position, I can:
1. Track the current max position
2. When evaluating a swap, only need to compute the new values at the old max position and check if any other position exceeds it

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        residual = 0
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def precompute_D(h, n):
    s = np.arange(-(n-1), n)
    D = np.zeros((n, 2*n-1))
    for p in range(n):
        vp = (p + s >= 0) & (p + s < n)
        D[p, vp] += h[p + s[vp]]
        vm = (p - s >= 0) & (p - s < n)
        D[p, vm] += h[p - s[vm]]
    return D

def update_D(D, h, i, j, n):
    """Update D after swapping h[i]:0->1 and h[j]:1->0"""
    # h[i] changes by +1, h[j] changes by -1
    for p in range(n):
        # Contribution of h[i] to D[p]: at s=i-p and s=p-i
        idx1 = (n-1) + (i - p)
        if 0 <= idx1 < 2*n-1:
            D[p, idx1] += 1.0
        idx2 = (n-1) + (p - i)
        if 0 <= idx2 < 2*n-1:
            D[p, idx2] += 1.0
        # Contribution of h[j] to D[p]: at s=j-p and s=p-j
        idx3 = (n-1) + (j - p)
        if 0 <= idx3 < 2*n-1:
            D[p, idx3] -= 1.0
        idx4 = (n-1) + (p - j)
        if 0 <= idx4 < 2*n-1:
            D[p, idx4] -= 1.0

def greedy_binary(h_init, n, deadline, max_sweeps=200, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    D = precompute_D(h, n)
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                # delta = D[j] - D[i] + cross terms
                new_corr = corr + D[j] - D[i]
                new_corr[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    new_corr[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    new_corr[k2] += 1.0
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    # Accept swap
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    update_D(D, h, i, j, n)
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    return best_h, best_c
