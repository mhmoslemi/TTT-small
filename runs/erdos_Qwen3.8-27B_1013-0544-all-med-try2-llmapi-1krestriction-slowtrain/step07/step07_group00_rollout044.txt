import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def smooth_max_and_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    val = m + (mx + np.log(w.sum())) / alpha
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wk = w[lag]
        if k >= 0:
            nv = n - k
            grad[:nv] += wk * (1.0 - h[k:])
            grad[k:] += wk * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wk * (1.0 - h[:nv])
            grad[:nv] += wk * (-h[kk:])
    return val, grad

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def projected_grad_desc(h, n, alpha, deadline, momentum=0.9):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = 0.01
    for step in range(50000):
        if time.time() > deadline:
            break
        _, g = smooth_max_and_grad(h, n, alpha)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            h = best_h.copy()
            v *= 0.5
            lr *= 0.8
    return h, best_c

def lbfgs_smooth(h, n, alpha, deadline, maxiter=500):
    val, g0 = smooth_max_and_grad(h, n, alpha)
    def obj(x):
        x_p = np.clip(x, 0, 1)
        v, _ = smooth_max_and_grad(x_p, n, alpha)
        s = x_p.sum() - n / 2.0
        return v + 1e4 * s * s
    def jac(x):
        x_p = np.clip(x, 0, 1)
        _, g = smooth_max_and_grad(x_p, n, alpha)
        s = x_p.sum() - n / 2.0
        g += 2e4 * s
        mask = (x > 0) & (x < 1)
        g = np.where(mask, g, 0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def slsqp_refine(h, n, alpha, deadline, maxiter=3000):
    def obj(x):
        _, _ = smooth_max_and_grad(x, n, alpha)
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    def eq(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': eq, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new

        return h.copy(), compute_c5(h, n)

def pair_polish(h, n, deadline):

    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

def interpolate_h(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)

def fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for k in range(1, 9):
        amp = rng.uniform(0.05, 0.3)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + ph)
    return project_feasible(h, n)

def binary_init(n, seed):

    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h = np.zeros(n)

    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Optimize parent intensively with multi-alpha
    if time.time() < deadline - 300:
        t1 = min(t0 + 250, deadline - 300)
        best_h, best_c = multi_alpha_optimize(best_h, best_n, t1)

    # Phase 2: SLSQP refinement at high alpha
    if time.time() < deadline - 200:
        t2 = min(time.time() + 80, deadline - 180)
        h2, c2 = slsqp_refine(best_h, best_n, 200000, t2, maxiter=5000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Try different resolutions
    for n_try in [400, 600, 800]:
        if time.time() > deadline - 150:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        alloc = min(120, (deadline - time.time() - 100) * 0.5)
        if alloc < 40:
            continue
        h_t, c_t = multi_alpha_optimize(h_interp, n_try, time.time() + alloc * 0.6)
        if time.time() < deadline - 80:
            h_t, c_t2 = slsqp_refine(h_t, n_try, 500000,
                                     min(time.time() + 40, deadline - 70), maxiter=3000)
            if c_t2 < c_t:
                c_t = c_t2
                h_t = h_t
        if c_t < best_c - 1e-6:
            best_c, best_h, best_n = c_t, h_t, n_try

    # Phase 4: Diverse restarts
    inits = [fourier_init(best_n, seed + 1),
             binary_init(best_n, seed + 2),
             fourier_init(best_n, seed + 3)]
    for init in inits:
        if time.time() > deadline - 100:
            break
        alloc = min(50, (deadline - time.time() - 60) * 0.5)
        if alloc < 20:
            break
        h_s, c_s = projected_grad_desc(init, best_n, 5000, time.time() + alloc * 0.5)
        h_s, c_s = lbfgs_smooth(h_s, best_n, 50000, time.time() + min(alloc * 0.4, max(10, deadline - time.time() - 60)))
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 5: Perturbation cycle
    for _ in range(3):
        if time.time() > deadline - 60:
            break
        noise = rng.randn(best_n) * 0.03
        h_p = project_feasible(best_h + noise, best_n)
        alloc = min(40, (deadline - time.time() - 30) * 0.4)
        if alloc < 15:
            break
        h_p, c_p = projected_grad_desc(h_p, best_n, 20000, time.time() + alloc * 0.6)
        if c_p < best_c:
            best_c, best_h = c_p, h_p

    # Phase 6: Final SLSQP
    if time.time() < deadline - 30:
        t6 = min(time.time() + 50, deadline - 15)
        h6, c6 = slsqp_refine(best_h, best_n, 500000, t6, maxiter=5000)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Pair polish
    if time.time() < deadline - 5:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 2)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
