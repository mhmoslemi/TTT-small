def coord_descent(h, n, deadline=None, max_sweeps=10000, seed=1, allowed=8.0, bi_iters=30):
    rng = np.random.RandomState(seed)
    h = project_feasible(h.astype(float).copy(), n)
    target = 0.5*n
    corr = eval_c_counts(h)
    best_c = float(corr.max())
    best_h = h.copy()
    m = 2*n - 1
    s = np.arange(-(n-1), n)
    S = float(h.sum())
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        prev = float(corr.max())
        order = rng.permutation(n)
        for i in order:
            if deadline is not None and (i & 127) == 0 and time.time() > deadline:
                break
            B = np.zeros(m)
            kf = i + s
            mask = (kf >= 0) & (kf < n)
            B[mask] += 1.0
            B[mask] -= h[kf[mask]]
            kb = i - s
            mask = (kb >= 0) & (kb < n)
            B[mask] -= h[kb[mask]]
            x = float(h[i])
            if S > target + allowed:
                yl = max(0.0, target - S - allowed)
                yu = min(1.0, target - S + allowed)
                # if current x outside? include x
                yl = min(yl, x); yu = max(yu, x)
            else:
                yl = max(0.0, target - S - allowed)
                yu = min(1.0, target - S + allowed)
            # also ensure yl<=yu
            if yl > yu:
                # swap or set [min,max]
                yl, yu = min(yl, yu), max(yl, yu)
                yl = max(0.0, yl); yu = min(1.0, yu)
                if yl > yu: yl = yu = x
            if yl >= yu:
                yl = min(max(x,0.0),1.0); yu=yl
            f_l = corr + B*(yl - x)
            f_u = corr + B*(yu - x)
            Bpos = B > 1e-14; Bneg = B < -1e-14; Bzero = ~(Bpos|Bneg)
            min_ep = np.where(Bpos, f_l, np.where(Bneg, f_u, corr))
            lo = float(np.max(min_ep)) - 1e-8
            hi = float(max(np.max(f_l), np.max(f_u))) + 1e-8

def exact_sum(h,n):
    target=0.5*n
    h=np.clip(h,0.0,1.0)
    for _ in range(4):
        diff=float(target-h.sum())
        if abs(diff)<1e-13: break
        if diff>0:
            idxs=np.flatnonzero(h < 1.0-1e-12)
        else:
            idxs=np.flatnonzero(h > 1e-12)
        if idxs.size==0: break
        if diff>0:
            caps=1.0-h[idxs]
        else:
            caps=h[idxs]
        for idx,cap in zip(idxs,caps):
            if diff>0:
                add=min(float(cap), diff)
                if add<=0: continue
                h[idx]+=add; diff-=add
            else:
                sub=min(float(cap), -diff)
                if sub<=0: continue
                h[idx]-=sub; diff+=sub
            if abs(diff)<1e-13: break
    return h

def weighted_gradient(h,P,n):

import numpy as np
from scipy.optimize import minimize
import time


def eval_c_counts(h):
    n = len(h)
    total = float(np.sum(h))
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    aut = np.correlate(h, h, mode="full")
    s = np.arange(2 * n - 1) - (n - 1)
    L = np.empty(2 * n - 1)
    pos = s >= 0
    L[pos] = pref[n - s[pos]]
    L[~pos] = total - pref[-s[~pos]]
    return L - aut


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def exact_sum(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(6):
        diff = float(target - h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idxs = np.flatnonzero(h < 1.0 - 1e-12)
            if idxs.size == 0:
                break
            caps = 1.0 - h[idxs]
            for idx, cap in zip(idxs, caps):
                add = min(float(cap), diff)
                if add <= 0.0:
                    continue
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            idxs = np.flatnonzero(h > 1e-12)
            if idxs.size == 0:
                break
            caps = h[idxs]
            for idx, cap in zip(idxs, caps):
                sub = min(float(cap), -diff)
                if sub <= 0.0:
                    continue
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return h


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).reshape(n).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    if abs(float(np.sum(x)) - target) < 1e-12:
        return exact_sum(x, n)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x))
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        f = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if f > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - hi, 0.0, 1.0)
    return exact_sum(p, n)


def make_thue(n):
    a = np.ones(1, dtype=float)
    while len(a) < n:
        a = np.concatenate((a, -a))
    return 0.5 + 0.5 * a[:n]


def random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    half = n // 2
    if 0 < half < n:
        h[rng.choice(n, half, replace=False)] = 1.0
    elif half > 0:
        h[:half] = 1.0
    return h


def soft_objective(c, beta):
    m = float(np.max(c))
    if beta <= 1e-12:
        return m, np.full(len(c), 1.0 / len(c)), m
    t = beta * (c - m)
    if t.max() > 0.0:
        t = np.minimum(t, 0.0)
    e = np.exp(t)
    s = float(e.sum())
    val = float(m + np.log(s) / beta)
    return val, e / s, m


def weighted_gradient(h, P, n):
    m = 2 * n - 1
    cp = np.empty(m + 1)
    cp[0] = 0.0
    cp[1:] = np.cumsum(P)
    i = np.arange(n)
    lin = cp[2 * n - 1 - i] - cp[n - 1 - i]
    Q = P + P[::-1]
    conv = np.convolve(h, Q, mode="full")
    quad = conv[n - 1:n - 1 + n]
    return lin - quad

def lbfgs_pair(x, n, beta, rho):
    c = eval_c_counts(x)
    G, P, _ = soft_objective(c, beta)
    sd = float(np.sum(x) - 0.5 * n)
    f = float(G + rho * sd * sd)
    g = weighted_gradient(x, P, n) + (2.0 * rho * sd) * np.ones(n)
    return f, g


def polish_lbfgs(h, n, beta, iters, deadline):
    best_h = project_feasible(h, n).copy()
    best_c = float(np.max(eval_c_counts(best_h)))
    bounds = [(0.0, 1.0)] * n
    for rho in (100.0, 20000.0):
        if time.time() > deadline:
            break
        try:
            res = minimize(
                lbfgs_pair,
                best_h,
                args=(n, beta, rho),
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": iters,
                    "ftol": 1e-12,
                    "maxfun": max(1000, iters * 30),

        cc = float(np.max(eval_c_counts(x)))
        if cc < best_c:
            best_c = cc
            best_h = x.copy()
    return best_h, best_c

                h = cand
                G = G2
                m = m2
                g = weighted_gradient(h, P2, n)
                alpha = max(a, 1e-8)
                accepted = True
                bad = 0
                break
            a *= 0.5
            if a < 1e-9:
                break
        if not accepted:
            bad += 1
            if bad > 15:
                break
            alpha *= 2.0

    h = project_feasible(h, n)
    m = float(np.max(eval_c_counts(h)))
    if m < best_c:
        best_c = m
        best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(1.0, float(budget_s) - 15.0)
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h_in = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if h_in.size == 0:
            raise ValueError("empty")
        n = int(h_in.size)
        h0 = project_feasible(h_in.copy(), n)
    except Exception:
        n = 400
        h0 = project_feasible(make_thue(n), n)

    best_h = h0.copy()
    best_count = float(np.max(eval_c_counts(best_h)))
    candidates = [
        best_h,
        make_thue(n),
        random_balanced(n, rng),
        project_feasible(best_h + 0.03 * rng.randn(n), n),
        np.roll(best_h, max(1, n // 3)),
        np.flip(best_h),
    ]
    for cand in candidates:
        cand = project_feasible(cand, n)
        cc = float(np.max(eval_c_counts(cand)))
        if cc < best_count:
            best_count = cc
            best_h = cand.copy()

    allowed = max(2.0, 0.025 * n)

    dl1 = t0 + 0.28 * total
    hc, cc = coord_descent(best_h, n, deadline=dl1, seed=seed, allowed=allowed, bi_iters=28)
    if cc < best_count:
        best_count = cc
        best_h = hc.copy()

    if float(budget_s) > 120.0 and time.time() < dl1 + 0.30 * total - 30.0:
        dl2 = t0 + 0.62 * total
        lb_iters = max(20, min(70, int(float(budget_s) / 18.0)))
        for beta in (40.0, 2000.0, 40000.0):
            if time.time() > dl2 - 5.0:
                break
            hp, cc = polish_lbfgs(best_h, n, beta, lb_iters, dl2)
            if cc < best_count:
                best_count = cc
                best_h = hp.copy()

    dl3 = t0 + 0.80 * total
    hc, cc = coord_descent(
        best_h,
        n,
        deadline=dl3,
        seed=seed + 1,
        allowed=max(2.0, 0.012 * n),
        bi_iters=28,
    )
    if cc < best_count:
        best_count = cc
        best_h = hc.copy()

    dl4 = t0 + 0.90 * total
    for beta in (500.0, 6000.0, 60000.0):
        if time.time() > dl4 - 5.0:
            break
        hg, cc = projected_gradient(best_h, n, beta, 180, dl4)
        if cc < best_count:
            best_count = cc
            best_h = hg.copy()

    while time.time() < dl - 25.0:
        cand = project_feasible(best_h + 0.005 * rng.randn(n), n)
        sub_dl = min(dl - 5.0, time.time() + 45.0)
        hc, cc = coord_descent(
            cand,
            n,
            deadline=sub_dl,
            seed=int(rng.randint(1000000)),
            allowed=allowed,
            bi_iters=24,
        )
        if cc < best_count:
            best_count = cc
            best_h = hc.copy()
        if time.time() < dl - 12.0:
            hp, cc = polish_lbfgs(best_h, n, 4000.0, max(20, 30), dl - 3.0)
            if cc < best_count:
                best_count = cc
                best_h = hp.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = erdos_c5(best_h, n)
    return best_h, best_c5, n
