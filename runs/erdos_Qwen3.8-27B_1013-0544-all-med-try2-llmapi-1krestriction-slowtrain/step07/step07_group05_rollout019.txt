    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_c(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_j(x, n):
    return np.ones(len(x))

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, max_sweeps=200, tol=1e-12):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    lags = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        c = compute_corrs(h, n)
        max_change = 0.0
        for p in range(n):
            idx1 = p - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            h1 = np.where(v1, h[np.clip(idx1, 0, n - 1)], 0.0)
            idx2 = p + lags
            v2 = (idx2 >= 0) & (idx2 < n)
            h2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
            A = dx * ((1.0 - h1) * v1.astype(float) - h2 * v2.astype(float))
            B = c - h[p] * A
            v_lo, v_hi = 0.0, 1.0
            for _ in range(45):
                v_mid = 0.5 * (v_lo + v_hi)
                ms = np.argmax(A * v_mid + B)
                if A[ms] > tol:
                    v_hi = v_mid
                elif A[ms] < -tol:
                    v_lo = v_mid
                else:
                    break
            v_new = 0.5 * (v_lo + v_hi)
            v_best = min(
                (0.0, np.max(B)),
                (1.0, np.max(A + B)),
                (v_new, np.max(A * v_new + B)),
                key=lambda t: t[1]
            )[0]
            if abs(v_best - h[p]) > 1e-14:
                d = v_best - h[p]
                h[p] = v_best
                c += d * A
                max_change = max(max_change, abs(d))
        h = project_feasible(h, n)
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
        if max_change < 1e-10:
            break
    return h, best_c

def full_optimize(h, n, deadline, rng):
