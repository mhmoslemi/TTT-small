def smooth_value_grad(h, alpha):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = np.flip(corr) * dx
    maxf = float(np.max(f))
    alpha = float(alpha)
    if alpha <= 1e-8:
       weights = np.zeros(2*n-1)
       weights[int(np.argmax(f))]=1.0
       val=maxf
    else:
       z = alpha * (f - maxf)
       # z max 0, but numeric maybe 1e-14; clip to <=0
       z = np.minimum(z, 0.0)
       tmp = np.exp(z)
       s = float(tmp.sum())
       if not math.isfinite(s) or s <= 1e-300:
          weights=np.zeros(2*n-1); weights[int(np.argmax(f))]=1.0; val=maxf
       else:
          weights = tmp / s
          val = maxf + math.log(s) / alpha
    grad = np.zeros(n)
    active = np.flatnonzero(weights > 1e-16)
    for t_idx in active:

        if k >= 0:

            grad[:L] += w * (1.0 - h[k:k+L])
            grad[k:] -= w * h[:L]

            r = -k
            L = n - r
            grad[r:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[r:r+L]
    grad *= dx
    return val, grad

import numpy as np
import time
import math
from scipy.optimize import minimize

_G_N = 0
_G_ALPHA = 1.0
_G_CACHE_X = None
_G_CACHE_V = 0.0
_G_CACHE_G = None

_AL_N = 0
_AL_ALPHA = 1.0
_AL_RHO = 1000.0
_AL_CACHE_X = None
_AL_CACHE_V = 0.0
_AL_CACHE_G = None


def compute_c5(h, n=None):
    n = len(h) if n is None else int(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_feasible(h, n, target=None):
    target = 0.5 * n if target is None else float(target)
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, h.size), h)
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        lo, hi = 0.0, 1.0 - float(np.min(h))
        for _ in range(60):
            t = 0.5 * (lo + hi)
            hs = np.clip(h + t, 0.0, 1.0)
            if float(hs.sum()) < target:
                lo = t
            else:
                hi = t
        return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    lo, hi = 0.0, float(np.max(h))
    for _ in range(60):
        t = 0.5 * (lo + hi)
        hs = np.clip(h - t, 0.0, 1.0)
        if float(hs.sum()) > target:
            lo = t
        else:
            hi = t
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_value_grad(h, alpha):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = np.flip(corr) * dx
    maxf = float(np.max(f))
    alpha = float(alpha)
    if alpha <= 1e-8:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(f))] = 1.0
        val = maxf
    else:
        z = np.minimum(alpha * (f - maxf), 0.0)
        tmp = np.exp(z)
        s = float(tmp.sum())
        if not math.isfinite(s) or s <= 1e-300:
            weights = np.zeros(2 * n - 1)
            weights[int(np.argmax(f))] = 1.0
            val = maxf
        else:
            weights = tmp / s
            val = maxf + math.log(s) / alpha
    grad = np.zeros(n)
    active = np.flatnonzero(weights > 1e-16)
    for t_idx in active:
        w = float(weights[t_idx])
        k = int(t_idx) - (n - 1)
        if k >= 0:
            L = n - k
            grad[:L] += w * (1.0 - h[k:k + L])
            grad[k:] -= w * h[:L]
        else:
            r = -k
            L = n - r
            grad[r:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[r:r + L]
    grad *= dx
    return val, grad


def get_slsqp_vg(x):
    global _G_CACHE_X, _G_CACHE_V, _G_CACHE_G
    if _G_CACHE_X is not None and x.shape == _G_CACHE_X.shape and (x == _G_CACHE_X).all():
        return _G_CACHE_V, _G_CACHE_G
    v, g = smooth_value_grad(x, _G_ALPHA)
    _G_CACHE_X = np.array(x, copy=True)
    _G_CACHE_V = float(v)
    _G_CACHE_G = np.array(g, copy=True)
    return v, g


def slsqp_obj(x):
    return get_slsqp_vg(x)[0]


def slsqp_jac(x):
    return get_slsqp_vg(x)[1]


def slsqp_cons(x):
    return float(np.sum(x) - 0.5 * _G_N)


def slsqp_cons_jac(x):
    return np.ones(len(x), dtype=float)


def slsqp_step(h_init, n, alpha, maxiter, ftol):
    global _G_N, _G_ALPHA, _G_CACHE_X, _G_CACHE_V, _G_CACHE_G
    _G_N = int(n)
    _G_ALPHA = float(alpha)
    _G_CACHE_X = None
    x0 = project_feasible(h_init, _G_N)
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac}]
    try:
        res = minimize(slsqp_obj, x0, jac=slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': float(ftol)})
        out = res.x
    except Exception:
        out = x0
    return project_feasible(out, _G_N)


def slsqp_sequence(h_init, n, alphas, maxiters, ftol, deadline):
    h_cur = project_feasible(h_init, n)
    c_cur = compute_c5(h_cur, n)
    best_h = h_cur.copy()
    best_c = c_cur
    for idx, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        mi = maxiters[idx] if idx < len(maxiters) else maxiters[-1]
        h_new = slsqp_step(h_cur, n, alpha, mi, ftol)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new < c_cur + 0.001:
            h_cur, c_cur = h_new, c_new
    return best_h, best_c


def get_al_vg(x):
    global _AL_CACHE_X, _AL_CACHE_V, _AL_CACHE_G
    if _AL_CACHE_X is not None and x.shape == _AL_CACHE_X.shape and (x == _AL_CACHE_X).all():
        return _AL_CACHE_V, _AL_CACHE_G

    _AL_CACHE_X = np.array(x, copy=True)

def lbfgs_step(h_init, n, alpha, rho, maxiter, ftol):

    _AL_RHO = float(rho)

def to_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    k = int(math.floor(0.5 * n))
    out[idx[:k]] = 1.0
    rem = 0.5 * n - k
    if rem != 0.0 and k < n:
        out[idx[k]] = rem
    return out


def random_binary_search(h_init, n, deadline, seed, max_iters=20000):

    h = to_binary(h_init, n).copy()

    zeros = np.flatnonzero(h == 0.0)

            cur = new
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0

def generate_wave_binary(n, seed):

    x1 = np.linspace(0.0, 1.0, n_new)
    return project_feasible(np.interp(x1, x0, h), n_new)

    cur = compute_c5(h, n)
    best_h = h.copy()

        if time.time() > deadline:

                    best_h = h.copy()

        if time.time() > deadline - 90:
            break
        hw = generate_wave_binary(best_n, int(seed) + 1000 + 100 * tr)
        h, c = lbfgs_sequence(hw, best_n,
                              (20.0, 100.0, 1000.0, 5000.0),
                              (1000.0,), 150, 1e-12, deadline - 30)
        if c < best_c:
            best_c, best_h = c, h
            if time.time() < deadline - 30:
                h2, c2 = slsqp_sequence(h, best_n,
                                        (2500.0, 10000.0),
                                        (70, 90), 1e-12, deadline - 10)
                if c2 < best_c:
                    best_c, best_h = c2, h2

    for sigma in (0.02, 0.05, 0.10, 0.25):
        if time.time() > deadline - 60:
            break
        hp = project_feasible(best_h + sigma * rng.normal(size=best_n), best_n)
        h, c = lbfgs_sequence(hp, best_n,
                              (50.0, 500.0, 5000.0, 25000.0),
                              (1000.0,), 140, 1e-12, deadline - 20)
        if c < best_c:
            best_c, best_h = c, h
            if time.time() < deadline - 20:
                h2, c2 = slsqp_sequence(h, best_n,
                                        (5000.0, 25000.0),
                                        (60, 80), 1e-12, deadline - 5)
                if c2 < best_c:
                    best_c, best_h = c2, h2

    if time.time() < deadline - 25:
        h, c = gradient_line_polish(best_h, best_n, deadline - 3, 250)
        if c < best_c:
            best_c, best_h = c, h

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
