```python
def compute_c5_pw(b, v):
    """
    b: array of M+1 breakpoints [0, p1, ..., pM] where pM=2
    v: array of M values
    Returns the exact C5 for this piecewise constant function
    """
    M = len(v)
    # Candidate shifts: all differences b[i+1]-b[j] and b[j]-b[i+1] etc.
    # Actually, the pieces change when x crosses a breakpoint of h or h(·+k)
    # h has breakpoints at b[0],...,b[M]
    # h(x+k) has breakpoints at b[0]-k, ..., b[M]-k
    # So the product changes at max(0, min(b[i], b[j]+k, 2-k)) for various i,j
    # 
    # Simpler: the integral as a function of k is piecewise linear with
    # breakpoints at k = b[i] - b[j] for all i,j
    # So enumerate all such k values in [0, 2]
    
    candidates = set()
    candidates.add(0)
    candidates.add(2)
    for i in range(M+1):
        for j in range(M+1):
            k = b[j] - b[i]
            if 0 <= k <= 2:
                candidates.add(k)
    
    best = 0
    for k in candidates:
        if k == 0:
            # integral of h(x)(1-h(x)) dx = integral h - integral h^2
            integral = 0
            for m in range(M):
                length = b[m+1] - b[m]
                integral += v[m] * (1 - v[m]) * length
            best = max(best, integral)
        else:
            # Need to compute integral of h(x)(1-h(x+k)) dx over [0, 2-k]
            # h(x) is v[m] on [b[m], b[m+1]]
            # h(x+k) is v[n] on [b[n]-k, b[n+1]-k] (clipped to [0, 2-k])
            # Overlap intervals: intersection of [b[m], b[m+1]] with [b[n]-k, b[n+1]-k] ∩ [0, 2-k]
            integral = 0
            for m in range(M):
                x_lo = b[m]
                x_hi = b[m+1]
                for n in range(M):
                    # h(x+k) = v[n] when x+k in [b[n], b[n+1]], i.e., x in [b[n]-k, b[n+1]-k]
                    y_lo = max(x_lo, b[n] - k, 0)
                    y_hi = min(x_hi, b[n+1] - k, 2 - k)
                    if y_hi > y_lo:
                        integral += v[m] * (1 - v[n]) * (y_hi - y_lo)
            best = max(best, integral)
    return best
```

```python
import numpy as np
from scipy.optimize import minimize, fmin
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_c5_pw(b, v):
    M = len(v)
    cands = set([0.0, 1.999])
    for i in range(M+1):
        for j in range(M+1):
            k = b[j] - b[i]
            if 0.0 <= k <= 2.0:
                cands.add(round(k, 10))
    best = 0.0
    for k in cands:
        if k > 1.999: continue
        integral = 0.0
        if k < 1e-9:
            for m in range(M):
                integral += v[m] * (1.0 - v[m]) * (b[m+1] - b[m])
        else:
            for m in range(M):
                for n_i in range(M):
                    lo = max(b[m], b[n_i] - k, 0.0)
                    hi = min(b[m+1], b[n_i+1] - k, 2.0 - k)
                    if hi > lo:
                        integral += v[m] * (1.0 - v[n_i]) * (hi - lo)
        best = max(best, integral)
    return best

def obj_pw(params, M):
    u = params[:M]
    s = params[M:2*M]
    w = np.exp(u - u.max())
    w = w / w.sum() * 2.0
    b = np.zeros(M+1)
    b[1:] = np.cumsum(w)
    b[-1] = 2.0
    v = 1.0 / (1.0 + np.exp(-np.clip(s, -50, 50)))
    area = float(np.dot(v, w))
    penalty = 100.0 * (area - 1.0)**2
    return compute_c5_pw(b, v) + penalty

    M = len(v)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    h = np.zeros(n)
    for m in range(M):
        mask = (x >= b[m]) & (x < b[m+1])
        h[mask] = v[m]

    # Simpler: use searchsorted
    idx = np.searchsorted(b, x, side='right') - 1
    idx = np.clip(idx, 0, M-1)
    h = v[idx]
    return project_feasible(h, n)

def optimize_pw(M, rng, time_limit):
    """Optimize piecewise constant with M pieces."""
    t0 = time.time()
    best_c = float('inf')
    best_params = None
    
    for restart in range(20):
        if time.time() - t0 > time_limit:
            break
        # Random init
        u = rng.uniform(-2, 2, M)
        # Initialize values: try binary-ish patterns
        if restart % 3 == 0:
            v_init = np.ones(M) * 0.5
        elif restart % 3 == 1:
            v_init = (np.arange(M) % 2).astype(float)
        else:
            v_init = rng.uniform(0, 1, M)
        # Convert to logit
        v_init = np.clip(v_init, 0.01, 0.99)
        s_init = np.log(v_init / (1 - v_init))
        
        params0 = np.concatenate([u, s_init])
        
        res = fmin(obj_pw, params0, args=(M,), maxiter=500, disp=False,
                   xtol=1e-8, ftol=1e-8)
        c = obj_pw(res, M)
        if c < best_c:

            best_params = res
    
    if best_params is None:
        return None, float('inf')
    
    # Extract best solution
    u = best_params[:M]
    s = best_params[M:2*M]
    w = np.exp(u - u.max())
    w = w / w.sum() * 2.0
    b = np.zeros(M+1)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_range = range(i+1, n)
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    j_range = range(lo, hi)
                for j in j_range:
                    if i == j: continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp_opt(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def jac(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w = w / w.sum()
        # gradient of smooth max w.r.t. x
        # d/dx_i of c[k] for each k, weighted by w
        # This is complex, skip analytical grad
        return None
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return float(x.sum() - n/2.0)
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h.copy()
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_c5_pw(b, v):
    M = len(v)
    cands = set()
    cands.add(0.0)
    for i in range(M + 1):
        for j in range(M + 1):
            k = b[j] - b[i]
            if 0.0 <= k <= 2.0:
                cands.add(k)
    best = 0.0
    for k in cands:
        if k > 1.9999:
            continue
        integral = 0.0
        if k < 1e-9:

                integral += v[m] * (1.0 - v[m]) * (b[m + 1] - b[m])
        else:
            for m in range(M):

                    hi = min(xhi, b[ni + 1] - k, 2.0 - k)
                    if hi > lo:
                        integral += v[m] * (1.0 - v[ni]) * (hi - lo)
        if integral > best:

    return best
```
