            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha, penalty=0.0):
    """Compute smooth max objective and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    obj = m + (max_s + np.log(sum_exp)) / alpha
    p = exp_s / sum_exp  # softmax weights
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag > 0:
            grad[:n - lag] += pk * dx * (1.0 - h[lag:])
            grad[lag:] += pk * dx * (-h[:n - lag])
        elif lag < 0:
            m2 = -lag
            grad[m2:] += pk * dx * (1.0 - h[:n - m2])
            grad[:n - m2] += pk * dx * (-h[m2:])
        else:
            grad += pk * dx * (1.0 - 2.0 * h)
    
    # Add penalty gradient for sum constraint
    if penalty > 0:
        sum_deficit = h.sum() - n / 2.0
        grad += 2.0 * penalty * sum_deficit
        obj += penalty * sum_deficit ** 2
    
    return float(obj), grad
