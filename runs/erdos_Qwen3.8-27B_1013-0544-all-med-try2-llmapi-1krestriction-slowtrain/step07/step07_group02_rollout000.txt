import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / float(n_points)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def enforce_feasible(h, n_points):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n_points)
    for _ in range(30):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        h += diff / float(n_points)
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if diff != 0.0:
       if diff > 0:
          idx = int(np.argmax(1.0 - h))
          add = min(diff, 1.0 - h[idx])
          h[idx] += add
       else:
          idx = int(np.argmax(h))
          sub = min(-diff, h[idx])
          h[idx] -= sub
    return h

def make_triangle_exact(n_points):
    n = int(n_points)
    if n < 4:
       n = 512
    N = n // 2
    if n != 2 * N or N < 2:
       x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
       h = np.minimum(x, 2.0 - x)
       return np.clip(h,0,1)
    left = (2.0 * np.arange(N, dtype=float) + 1.0) / float(n)
    right = (float(n) - 2.0 * np.arange(N, dtype=float) - 1.0) / float(n)
    h = np.empty(n, dtype=float)
    h[:N] = left
    h[N:] = right
    return h

import numpy as np
import time


def compute_c5(h, n_points):
    dx = 2.0 / float(n_points)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def enforce_feasible(h, n_points):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n_points)

    for _ in range(50):
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-13:
            break
        h += diff / float(n_points)
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(np.sum(h))
    if diff > 0.0:
        idx = int(np.argmax(1.0 - h))
        delta = min(float(diff), float(1.0 - h[idx]))
        h[idx] += delta
    elif diff < 0.0:
        idx = int(np.argmax(h))
        delta = min(float(-diff), float(h[idx]))
        h[idx] -= delta

    return h


def make_triangle_exact(n_points=256):
    n = int(n_points)
    if n < 4:
        n = 256

    N = n // 2
    if n != 2 * N:
        x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
        h = np.minimum(x, 2.0 - x)
        return np.clip(h, 0.0, 1.0)

    # Exact power-of-two denominators when n is a power of two.
    # h_i = (i + 1/2) / N on [0, 1], and h_{i+N} = 1 - h_i.
    left = (2.0 * np.arange(N, dtype=float) + 1.0) / float(n)
    right = (float(n) - 2.0 * np.arange(N, dtype=float) - 1.0) / float(n)

    h = np.empty(n, dtype=float)
    h[:N] = left
    h[N:] = right
    return h
