```python
import time
import numpy as np
from scipy.optimize import minimize


def c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def c5_value(h, n):
    return float(np.max(c5_vec(h, n)))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    y = np.clip(x - mu, 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - y))
            y[idx] = min(1.0, y[idx] + diff)
        else:
            idx = int(np.argmax(y))
            y[idx] = max(0.0, y[idx] + diff)
    return y


def smooth_value_grad(x, n, beta, mu):
    dx = 2.0 / n
    C = np.correlate(x, 1.0 - x, mode="full") * dx
    M = float(C.max())
    if not np.isfinite(M):
        return 1.0e300, np.zeros(n, dtype=float)
    z = beta * (C - M)
    p = np.exp(z)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(C)
        p[np.isclose(C, M, rtol=0.0, atol=1e-15)] = 1.0
        s = float(p.sum())
        if s <= 0.0:
            p[C == M] = 1.0
            s = 1.0
    p = p / s
    val = M + np.log(s) / beta + mu * (float(np.sum(x)) - 0.5 * n) ** 2
    A = 1.0 - x
    c1 = np.convolve(A, p, mode="full")
    g1 = c1[n - 1:n - 1 + n]
    hr = x[::-1]
    c2 = np.convolve(hr, p, mode="full")
    g2 = c2[2 * n - 2:n - 2:-1]
    grad = dx * (g1 - g2)
    sd = float(np.sum(x)) - 0.5 * n
    if mu != 0.0:
        grad = grad + (2.0 * mu * sd) * np.ones(n, dtype=float)
    return float(val), grad


def lsq_obj(x, n, beta, mu):
    return smooth_value_grad(x, n, beta, mu)[0]


def lsq_grad(x, n, beta, mu):
    return smooth_value_grad(x, n, beta, mu)[1]


def optimize_lsq(h, n, deadline, betas, mu_list, maxiter):
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5_value(best_h, n)
    h_cur = h.copy()
    c_cur = best_c
    bounds = [(0.0, 1.0)] for _ in range(n)][ : ]
    for beta in betas:
        for mu in mu_list:
            if time.time() > deadline - 1.0:
                break
            it = maxiter
            if n > 600:
                it = int(0.7 * maxiter)
            try:
                res = minimize(
                    lsq_obj,
                    h_cur,
                    jac=lsq_grad,
                    args=(n, beta, mu),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": it, "ftol": 1.0e-14, "gtol": 1.0e-9},
                    disp=False,
                )
                if hasattr(res, "x") and res.x is not None:
                    ht = project_box_sum(res.x, 0.5 * n)
                else:
                    ht = h_cur
            except Exception:
                ht = h_cur
            ct = c5_value(ht, n)
            if np.isfinite(ct) and ct < best_c - 1.0e-15:
                best_c = ct
                best_h = ht.copy()
            if np.isfinite(ct) and ct <= c_cur + 1.0e-12:
                h_cur = ht
                c_cur = ct
            else:
                h_cur = best_h.copy()
                c_cur = best_c
    best_h = project_box_sum(best_h, 0.5 * n)
    return best_h, c5_value(best_h, n)


def derivative_matrix(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    D = np.zeros((n, M), dtype=float)
    h1 = 1.0 - h
    ks = np.arange(n)
    for i in range(n):
        idx = n - 1 + i - ks
        msk = (idx >= 0) & (idx < M)
        D[i, idx[msk]] += h1[ks[msk]]
        idx = n - 1 + ks - i
        msk = (idx >= 0) & (idx < M)
        D[i, idx[msk]] -= h[ks[msk]]
    D *= dx
    return D


def eval_pair_f(C, L, idx1, idx2, d, dx):
    tmp = C + L * d
    q = d * d * dx
    tmp[idx1] += q
    if idx2 != idx1:
        tmp[idx2] += q
    return float(np.max(tmp))


def try_pair(h, C, D, i, j, n, cur_c, dx, golden_iters):
    hi_i = float(h[i])
    hi_j = float(h[j])
    s = hi_i + hi_j
    lo_t = max(0.0, s - 1.0) - hi_i
    hi_t = min(1.0, s) - hi_i
    if hi_t - lo_t < 1.0e-13:
        return h, C, D, False, cur_c
    L = D[i] - D[j]
    idx1 = n - 1 + (j - i)
    idx2 = n - 1 + (i - j)
    Mv = 2 * n - 1
    if idx1 < 0 or idx1 >= Mv or idx2 < 0 or idx2 >= Mv:
        return h, C, D, False, cur_c
    eq = -0.5 * (hi_i - hi_j)
    ds = [0.0, lo_t, hi_t, float(np.clip(eq, lo_t, hi_t)), float(np.clip(0.5 * eq, lo_t, hi_t))]
    best_d = 0.0
    best_f = float(np.max(C))
    for d in ds:
        f = eval_pair_f(C, L, idx1, idx2, d, dx)
        if f < best_f - 1.0e-14:
            best_f = f
            best_d = d
    if best_f >= cur_c - 1.0e-13:
        return h, C, D, False, cur_c
    a, b = lo_t, hi_t
    gr = 0.6180339887498949
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1 = eval_pair_f(C, L, idx1, idx2, x1, dx)
    f2 = eval_pair_f(C, L, idx1, idx2, x2, dx)
    for _ in range(golden_iters):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - gr * (b - a)
            f1 = eval_pair_f(C, L, idx1, idx2, x1, dx)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + gr * (b - a)
            f2 = eval_pair_f(C, L, idx1, idx2, x2, dx)
    try_ds = [0.0, best_d, a, b, x1, x2]
    best_f = float(np.max(C))
    best_d = 0.0
    for d in try_ds:
        f = eval_pair_f(C, L, idx1, idx2, d, dx)
        if f < best_f - 1.0e-14:
            best_f = f
            best_d = d
    if best_f >= cur_c - 1.0e-13:
        return h, C, D, False, cur_c
    d = float(best_d)
    new_i = hi_i + d
    if new_i < 0.0:
        new_i = 0.0
    elif new_i > 1.0:
        new_i = 1.0
    new_j = s - new_i
    if new_j < 0.0:
        new_i = s
        new_j = 0.0
    elif new_j > 1.0:
        new_i = s - 1.0
        new_j = 1.0
    if new_i < 0.0:
        new_i = 0.0
    elif new_i > 1.0:
        new_i = 1.0
    new_j = s - new_i
    delta = new_i - hi_i
    if abs(delta) < 1.0e-15:
        return h, C, D, False, cur_c
    est = eval_pair_f(C, L, idx1, idx2, delta, dx)
    if est >= cur_c - 1.0e-14:
        return h, C, D, False, cur_c
    C += L * delta
    q = delta * delta * dx
    C[idx1] += q
    if idx2 != idx1:
        C[idx2] += q
    h[i] = new_i
    h[j] = new_j
    step = delta * dx
    k = np.arange(n)
    idx = n - 1 + k - i
    msk = (idx >= 0) & (idx < Mv)
    D[k[msk], idx[msk]] -= step
    idx = n - 1 + i - k
    msk = (idx >= 0) & (idx < Mv)
    D[k[msk], idx[msk]] -= step
    idx = n - 1 + k - j
    msk = (idx >= 0) & (idx < Mv)
    D[k[msk], idx[msk]] += step
    idx = n - 1 + j - k
    msk = (idx >= 0) & (idx < Mv)
    D[k[msk], idx[msk]] += step
    return h, C, D, True, float(np.max(C))


def candidate_pairs(h, n, m, dx, top, rng):
    d = m - (n - 1)
    if d == 0:
        hi_idx = np.argsort(h)[::-1][:40]
        lo_idx = np.argsort(h)[:40]
        if hi_idx.size == 0 or lo_idx.size == 0:
            return []
        pairs = []
        tries = max(12, 8 * top)
        for _ in range(tries):
            a = int(hi_idx[int(rng.randint(hi_idx.size))])
            b = int(lo_idx[int(rng.randint(lo_idx.size))])
            if a == b:
                continue
            if a > b:
                a, b = b, a
            pairs.append((a, b))
        a = int(hi_idx[0])
        b = int(lo_idx[0])
        if a != b:
            pairs.append((a, b) if a < b else (b, a))
        if pairs:
            rng.shuffle(pairs)
        return pairs
    if d >= 0:
        lows = np.arange(0, n - d)
        highs = lows + d
    else:
        highs = np.arange(-d, n)
        lows = highs + d
    lp = int(lows.size)
    if lp <= 0:
        return []
    contrib = h[highs] * (1.0 - h[lows]) * dx
    if contrib.max() <= 1.0e-15:
        p = int(rng.randint(lp))
        return [(int(lows[p]), int(highs[p]))]
    if lp <= max(8, 4 * top):
        positions = np.argsort(contrib)[::-1][:2 * top]
    else:
        top_small = 5 * top
        if top_small >= lp:
            positions = np.argsort(contrib)[::-1][:2 * top]
        else:
            pool = np.argpartition(contrib, lp - top_small)[-top_small:]
            take = min(2 * top, int(pool.size))
            positions = pool[rng.randint(int(pool.size), size=take)]
    pos_list = [int(p) for p in positions]
    for _ in range(3):
        pos_list.append(int(rng.randint(lp)))
    pairs = [(int(lows[p]), int(highs[p])) for p in pos_list]
    rng.shuffle(pairs)
    return pairs


def targeted_pair_cd(h, n, deadline, seed=42, golden_iters=24, top_count=8, active_tol=1.0e-8):
    h = project_box_sum(h, 0.5 * n).copy()
    C = c5_vec(h, n)
    D = derivative_matrix(h, n)
    best_h = h.copy()
    best_c = float(np.max(C))
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    updates = 0
    failures = 0
    no_improve = 0
    recompile_every = 40
    atol = active_tol
    while time.time() < deadline:
        if updates > 0 and updates % recompile_every == 0:
            C = c5_vec(h, n)
            D = derivative_matrix(h, n)
        cur_c = float(np.max(C))
        if cur_c < best_c - 1.0e-15:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 12000:
            break
        if failures > 80:
            atol = 2.0e-5
        elif failures > 25:
            atol = 5.0e-6
        else:
            atol = active_tol
        active_tol_eff = max(atol, 1.0e-9)
        active = np.where(C >= cur_c - active_tol_eff)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(C))])
        if failures > 20:
            m = n - 1
        else:
            m = int(active[int(rng.randint(int(active.size)))])
        pairs = candidate_pairs(h, n, m, dx, top_count, rng)
        if not pairs:
            failures += 1
            continue
        improved = False
        for i, j in pairs:
            h, C, D, ok, new_c = try_pair(h, C, D, i, j, n, float(np.max(C)), dx, golden_iters)
            if ok:
                updates += 1
                failures = 0
                cur_c = new_c
                if cur_c < best_c - 1.0e-15:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                improved = True
                break
            else:
                failures += 1
        if not improved:
            failures += 1
            if failures > 30000:
                break
    best_h = project_box_sum(best_h, 0.5 * n)
    return best_h, c5_value(best_h, n)


def resample_h(h, n, new_n):
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    y = np.interp(xn, xo, h)
    return project_box_sum(y, 0.5 * new_n)


def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    y = np.zeros(n, dtype=float)
    y[idx[: n // 2]] = 1.0
    return project_box_sum(y, 0.5 * n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(20.0, float(budget_s)) - 18.0
    deadline = t0 + total
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
        if n0 < 4:
            raise ValueError
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    if n0 > 1000:
        h0 = resample_h(h0, n0, 800)
        n0 = 800
    h0 = project_box_sum(h0, 0.5 * n0)
    best_h = h0.copy()
    best_c = c5_value(best_h, n0)
    best_n = n0
    rng = np.random.RandomState(seed)

    ps_dl = min(deadline, t0 + min(320.0, 0.45 * total))
    ph, pc = optimize_lsq(
        h0,
        n0,
        ps_dl,
        [20, 100, 500, 2500, 12000, 60000, 300000, 1500000],
        [1000.0],
        900,
    )
    if pc < best_c - 1.0e-15:
        best_c, best_h, best_n = pc, ph.copy(), n0

    pp_dl = min(deadline, time.time() + min(650.0, 0.75 * (deadline - time.time())))
    ph, pc = targeted_pair_cd(ph, n0, pp_dl, seed=seed + 1, golden_iters=26, top_count=9)
    if pc < best_c - 1.0e-15:
        best_c, best_h, best_n = pc, ph.copy(), n0

    candidates = []
    hh = to_binary(best_h, best_n)
    candidates.append((hh, best_n))
    hh = project_box_sum(best_h + 0.015 * rng.randn(best_n), 0.5 * best_n)
    candidates.append((hh, best_n))
    for nh in (300, 450, 600):
        if 0 < nh < 1000 and nh != best_n:
            hh = resample_h(best_h, best_n, nh)
            candidates.append((hh, nh))

    for hh, nh in candidates:
        rem = deadline - time.time()
        if rem < 120.0:
            break
        hh = project_box_sum(hh, 0.5 * nh)
        c0 = c5_value(hh, nh)
        if not np.isfinite(c0):
            continue
        smooth_dl = min(deadline, time.time() + min(180.0, 0.4 * rem))
        hopt, cout = optimize_lsq(hh, nh, smooth_dl, [200, 2000, 20000, 200000], [1000.0], 450)
        rem = deadline - time.time()
        if rem > 80.0:
            pair_dl = min(deadline, time.time() + min(260.0, 0.75 * rem))
            hp, cp = targeted_pair_cd(hopt, nh, pair_dl, seed=seed + int(nh) + 7, golden_iters=24, top_count=8)
            if cp < cout:
                hopt, cout = hp, cp
        if cout < best_c - 1.0e-15:
            best_c, best_h, best_n = cout, hopt.copy(), nh

    rem = deadline - time.time()
    if rem > 100.0:
        final_dl = min(deadline, time.time() + min(320.0, 0.9 * rem))
        ph, pc = targeted_pair_cd(best_h, best_n, final_dl, seed=seed + 101, golden_iters=26, top_count=9)
        if pc < best_c - 1.0e-15:
            best_c, best_h, best_n = pc, ph.copy(), best_n

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = c5_value(best_h, best_n)
    return best_h, best_c, int(best_n)
```
