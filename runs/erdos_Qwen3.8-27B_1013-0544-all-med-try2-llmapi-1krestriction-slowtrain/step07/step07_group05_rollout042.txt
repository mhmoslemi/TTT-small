def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def pnorm_obj(x, n, p):

    c = np.maximum(np.correlate(x, 1 - x, mode='full') * dx, 0.0)
    m = c.max()
    if m < 1e-15:
        return 0.0
    return float(m * (np.sum((c/m)**p))**(1.0/p))
