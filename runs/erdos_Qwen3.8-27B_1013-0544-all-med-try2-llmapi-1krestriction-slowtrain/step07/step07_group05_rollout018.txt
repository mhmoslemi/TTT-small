def cut_lp(h_current, n, cuts, t_current, deadline):
    """Solve LP with cutting planes"""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # Add cuts
    for c_shift, c_coeff, c_const in cuts:
        # c_coeff @ h_var + c_const <= t_var
        constraints.append(c_coeff @ h_var + c_const <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, max_iters=10000)
    
    if h_var.value is not None:
        return h_var.value, t_var.value
    return h_current, t_current

        if r < 0.55:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-step, step)
            if cur_h[i] + d < 0 or cur_h[i] + d > 1:
                continue
            if cur_h[j] - d < 0 or cur_h[j] - d > 1:
                continue
            cur_h[i] += d
            cur_h[j] -= d
        elif r < 0.75:
            center = rng.randint(n)
            w = rng.randint(2, 5)
            lo, hi = max(0, center - w), min(n, center + w + 1)
            old = cur_h[lo:hi].copy()
            cur_h[lo:hi] = np.clip(cur_h[lo:hi] + rng.normal(0, step * 0.4, hi - lo), 0, 1)
            cur_h[lo:hi] += (old.sum() - cur_h[lo:hi].sum()) / max(1, hi - lo)
            cur_h[lo:hi] = np.clip(cur_h[lo:hi], 0, 1)
        elif r < 0.9:
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            d = rng.uniform(-step * 2, step * 2)
            if cur_h[i] + d < 0 or cur_h[i] + d > 1:
                continue
            if cur_h[j] - d < 0 or cur_h[j] - d > 1:
                continue
            cur_h[i] += d
            cur_h[j] -= d
        else:
            center = rng.randint(n)
            w = rng.randint(3, 8)
            lo, hi = max(0, center - w), min(n, center + w + 1)
            old = cur_h[lo:hi].copy()
            nuisance = np.zeros(n)
            nuisance[lo:hi] = 1.0
            comp_idx = np.where(nuisance == 0)[0]
            if len(comp_idx) == 0:
                continue
            ci = comp_idx[rng.randint(len(comp_idx))]
            d = rng.uniform(-step, step)
            cur_h[lo:hi] = np.clip(cur_h[lo:hi] + d * 0.3, 0, 1)
            actual = cur_h[lo:hi].sum() - old.sum()
            if cur_h[ci] + actual < 0 or cur_h[ci] + actual > 1:
                continue
            cur_h[ci] += actual
        cur_h = project_feasible(cur_h, n)
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or (t > t_min and rng.random() < np.exp(-delta / t)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        t = max(t * cool, t_min)
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; imp = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; imp = True
                        else:
                            h[i] += d; h[j] -= d
            if not imp:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; imp = True
                        else:
                            h[i] -= d; h[j] += d
            if not imp:
                break
    return h, best_c

def ils_slsqp(best_h, n, deadline, rng, max_iters=30):
    best_c = compute_c5(best_h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scale = 0.02 + 0.06 * (it % 5) / 4.0
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        for alpha in [5000, 50000]:
            if time.time() > deadline:
                break
            ht = slsqp_wrap(hp, n, alpha, maxiter=800)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(250, 0.5)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Phase 1: Initial SLSQP (70s)
    tp1 = time.time() + min(70, budget_s * 0.06)
    for a in [2000, 10000, 50000, 200000]:
        if time.time() > tp1 - 5:
            break
        ht = slsqp_wrap(best_h, n0, a, maxiter=1000)
        c = compute_c5(ht, n0)
        if c < best_c:
            best_c, best_h = c, ht
    # Phase 2: SA exploration (300s)
    if time.time() < deadline - 150:
        tp2 = time.time() + min(300, budget_s * 0.28)
        ht, c = sa_optimize(best_h, n0, tp2, rng, t_init=0.04, cool=0.9998)
        if c < best_c:
            best_c, best_h = c, ht
    # Phase 3: ILS with SLSQP (250s)
    if time.time() < deadline - 100:
        n = len(best_h)
        tp3 = time.time() + min(250, budget_s * 0.22)
        best_h, best_c = ils_slsqp(best_h, n, tp3, rng, max_iters=25)
    # Phase 4: SLSQP polish + pair search (150s)
    if time.time() < deadline - 40:
        n = len(best_h)
        ht = slsqp_wrap(best_h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
        n = len(best_h)
        if n <= 150:
            if time.time() < deadline - 20:
                best_h, best_c = pair_search(best_h, n, time.time() + 60,
                    [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            if time.time() < deadline - 20:
                best_h, best_c = near_pair_search(best_h, n, time.time() + 60,
                    [0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)
    # Phase 5: Try smaller grids (120s)
    for n_try in [100, 130]:
        if time.time() > deadline - 50:
            break
        xi = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(xn, xi, best_h), n_try)
        if time.time() < deadline - 40:
            ht, c = sa_optimize(hi, n_try, time.time() + 35, rng, t_init=0.02)
            if time.time() < deadline - 15:
                ht, c = pair_search(ht, n_try, time.time() + 15,
                    [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
            if c < best_c:
                best_c, best_h = c, ht
    # Phase 6: Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_wrap(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
        n = len(best_h)
        if n <= 130 and time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        elif time.time() < deadline - 2:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                [0.005, 0.002, 0.001, 0.0005], radius=30)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
