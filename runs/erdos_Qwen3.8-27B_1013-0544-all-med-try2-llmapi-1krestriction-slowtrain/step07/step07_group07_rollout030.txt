import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target_sum / tot), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max(obj, alpha):
    m = obj.max()
    shifted = alpha * (obj - m)
    mx = shifted.max()
    if mx > 350:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def obj_lagset(x, n, idx_set, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c[idx_set], alpha)

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_jac(x, n):
    return np.ones_like(x)

def slsqp(h, n, alpha, idx_set=None, maxiter=4000):
    if idx_set is None:
        fun = obj_full
        args = (n, alpha)
    else:
        fun = obj_lagset
        args = (n, idx_set, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(fun, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def worst_lag_coefficients(h, n):
    """For the worst lag, compute coefficient of each h_j in the lag's objective.
    Returns (lag, coeffs) where coeffs[j] = 1 - h_{j+m} - h_{j-m} (boundary adjusted)."""
    c = compute_all_lags(h, n)
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    m = abs(lag)
    if m == 0 or m >= n:
        return lag, None
    coeffs = np.ones(n)
    if lag > 0:
        # f = sum_{i=0}^{n-1-m} h_i*(1-h_{i+m})
        # coeff of h_j: from linear term +1 if j < n-m
        # from -h_j*h_{j+m} if j < n-m: -h_{j+m}
        # from -h_{j-m}*h_j if j >= m: -h_{j-m}
        coeffs[:] = 0.0
        coeffs[:n-m] += 1.0
        coeffs[:n-m] -= h[m:]
        coeffs[m:] -= h[:n-m]
    else:
        # Same formula by symmetry (correlate is symmetric in a useful way)
        coeffs[:] = 0.0
        coeffs[:n-m] += 1.0
        coeffs[:n-m] -= h[m:]
        coeffs[m:] -= h[:n-m]
    return lag, coeffs

def targeted_binary_search(h, n, deadline, max_rounds=2000):
    """Binary coordinate descent on worst lag, respecting sum constraint via swaps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        lag, coeffs = worst_lag_coefficients(h, n)
        if lag == 0 or coeffs is None:
            break
        # We want to set h_j=1 where coeff<0, h_j=0 where coeff>0
        # But must maintain sum(h)=n/2, so do swaps
        neg_indices = np.where(coeff < -1e-6)[0]
        pos_indices = np.where(coeffs > 1e-6)[0]
        if len(neg_indices) == 0 or len(pos_indices) == 0:
            break
        # Find best swap: maximize decrease in worst lag
        # Moving mass from pos i to neg j changes the objective by:
        # delta = -coeff_pos[i] * delta - coeff_neg[j] * delta (for small delta)
        # = -(coeff[i] + coeff[j]) * delta
        # We want this to be negative (decrease), so coeff[i] + coeff[j] > 0
        # i.e., coeff[i] > -coeff[j] (since coeff[j]<0)
        best_gain = 0
        best_i = -1
        best_j = -1
        # Sample for efficiency
        ni_sample = neg_indices[:50] if len(neg_indices) > 50 else neg_indices
        pi_sample = pos_indices[:50] if len(pos_indices) > 50 else pos_indices
        for i in pi_sample:
            for j in ni_sample:
                gain = coeffs[i] + (-coeffs[j])
                # Also check feasibility
                h_i, h_j = h[i], h[j]
                max_delta = min(h_i, 1.0 - h_j)
                if max_delta < 1e-8:
                    continue
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        delta = min(0.1, h[best_i], 1.0 - h[best_j])
        h[best_i] -= delta
        h[best_j] += delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
        else:
            h[best_i] += delta
            h[best_j] -= delta
            h = project_feasible(h, n)
    return h, best_c

def local_search_pairs(h, n, deadline):
    """General local search: move mass between pairs to reduce max lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(5000):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            # Find the pair with largest contribution to this lag
            if lag > 0:
                contrib = h[:n-m] * (1 - h[m:])
            else:
                contrib = h[m:] * (1 - h[:n-m])
            idx = int(np.argmax(contrib))
            if lag > 0:
                i, j = idx, idx + m
            else:
                i, j = idx + m, idx
            # Move mass from the "source" to the "sink"
            if h[i] > h[j]:
                delta = min(d, 0.4 * (h[i] - h[j]), h[i], 1.0 - h[j])
            else:
                i, j = j, i
                delta = min(d, 0.4 * (h[i] - h[j]), h[i], 1.0 - h[j])
            if delta < 1e-8:
                break
            h[i] -= delta
            h[j] += delta
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[i] += delta
                h[j] -= delta
    return h, best_c
