def c5_and_gradient(h, n, alpha):

    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    c5 = corr.max()

    exp_sh = np.exp(shifted - max_sh)
    weights = exp_sh / exp_sh.sum()

            grad[Ka:] -= w * h[:-Ka]
        else:
            grad += w * (1.0 - 2.0 * h)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_feasible(v, n):
    target = n / 2.0
    lo = v.max() + 1.0
    hi = v.min() - 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def get_weights(corr, alpha=None):
    if alpha is None:
        c5 = corr.max()
        active = corr >= (c5 - 1e-10)
        if not active.any():
            ind = np.zeros(len(corr), dtype=bool)
            ind[np.argmax(corr)] = True
            active = ind
        w = active.astype(float)
        return w / w.sum()
    shifted = alpha * (corr - corr.max())
    mx = shifted.max()
    e = np.exp(shifted - mx)
    return e / e.sum()

def compute_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        L = m - center
        if L > 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        elif L < 0:
            Ka = -L
            grad[:n-Ka] += w * (1.0 - h[Ka:])
            grad[Ka:] -= w * h[:n-Ka]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def gd_optimize(h, n, deadline, alpha_start, alpha_end, max_iter=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = corr.max()
        weights = get_weights(corr, alpha)
        grad = compute_grad(h, n, weights)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = min(1.0 / (alpha * gnorm * dx + 1e-10), 1.0)
        
        improved = False
        for _ in range(20):
            h_new = project_to_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
                improved = True
                break
            elif c_new < c5 + 1e-8:
                h = h_new
                no_improve += 1
                break
            step *= 0.5

        if no_improve > 300:
            h = project_to_feasible(h + 0.005 * rng.randn(n), n)
            no_improve = 0
    
    return best_h, best_c
