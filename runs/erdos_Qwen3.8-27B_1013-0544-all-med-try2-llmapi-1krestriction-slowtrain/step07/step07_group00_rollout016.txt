import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def grad_smooth_max(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(top_k):
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single(h, n, k)
    return grad

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 50.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_pts = len(x)
    top_k = min(80, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n_pts)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 100.0 * s
    return grad

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(len(x))

def pgd_optimize(h, n, deadline, top_k=40, alpha=80.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    prev_h = None
    prev_g = None
    stall = 0
    while time.time() < deadline:
        g = grad_smooth_max(h, n, top_k, alpha)
        if prev_h is not None:
            s = h - prev_h
            y = g - prev_g
            sy = s @ y
            ss = s @ s
            yy = y @ y
            if yy > 1e-20:
                if sy > 1e-20:
                    step = ss / sy
                else:
                    step = ss / yy
                step = float(np.clip(step, 0.0005, 0.15))
        prev_h = h.copy()
        prev_g = g.copy()
        accepted = False
        for _ in range(12):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-13:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.4
        if not accepted:
            h_new = project_feasible(h - 0.0005 * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:

    return best_h, best_c

def pair_polish(h, n, deadline, radius=60):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)
