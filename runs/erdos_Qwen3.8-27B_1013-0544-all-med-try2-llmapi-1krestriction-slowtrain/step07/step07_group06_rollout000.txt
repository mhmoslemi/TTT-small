```
def smooth_grad(h, n, alpha, mu=0.0):
    c = c5_vec(h,n)
    cm = c.max()
    z = alpha * (c - cm)
    zm = z.max()
    ez = np.exp(z - zm)
    ws = ez.sum()
    w = ez / ws
    dx = 2.0/n
    g = np.zeros(n, dtype=float)
    off = n - 1
    for m in range(2*n - 1):
        ww = w[m]
        if ww <= 1e-15: continue
        k = m - off
        wm = ww * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wm * (1.0 - h[k:])
                g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wm * (1.0 - h[:nv])
                g[:nv] += wm * (-h[kk:])
    if mu:
       g += mu * (h.sum() - 0.5*n)
    return g
```

```
def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) <= 1e-13:
        return h
    lo = -1.0
    hi = 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0))) - target
        if val > 0.0:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h2.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = np.where(h2 < 1.0)[0]
        else:
            idx = np.where(h2 > 0.0)[0]

            h2[i] += diff
            h2 = np.clip(h2, 0.0, 1.0)
    return h2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GAL = 1.0
_GMU = 0.0
_GF_N = 0
_GB_N = 0
_GB_BASIS = None


def c5_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_val(h, n):
    return float(np.max(c5_vec(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) <= 1e-13:
        return h

    lo = -1.0
    hi = 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0))) - target
        if val > 0.0:
            lo = mid
        else:
            hi = mid

    h2 = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        diff = target - float(h2.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            idx = np.where(h2 < 1.0)[0]
        else:
            idx = np.where(h2 > 0.0)[0]
        if idx.size == 0:
            break
        i = int(idx[idx.size // 2])
        h2[i] += diff
        h2 = np.clip(h2, 0.0, 1.0)
    return h2


def better_candidate(h, c, n, best_h, best_c, best_n):
    if h is None or not np.isfinite(c):
        return best_h, best_c, best_n
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if h.size != n:
        if h.size == 0:
            return best_h, best_c, best_n
        h = h[:n]
        if h.size != n:
            h = np.resize(h, n)
    if float(c) < best_c:
        return h.copy(), float(c), n
    return best_h, best_c, best_n


def smooth_value(h, n, alpha, mu):
    c = c5_vec(h, n)
    cm = float(np.max(c))
    if not np.isfinite(cm):
        return cm
    z = alpha * (c - cm)
    zm = float(np.max(z))
    val = cm + (zm + float(np.log(np.sum(np.exp(z - zm))))) / alpha
    if mu != 0.0:
        s = float(h.sum()) - 0.5 * float(n)
        val += 0.5 * mu * s * s
    return float(val)


def smooth_grad(h, n, alpha, mu):
    c = c5_vec(h, n)
    cm = float(np.max(c))
    if not np.isfinite(cm):
        return np.zeros(n, dtype=float)

    z = alpha * (c - cm)
    zm = float(np.max(z))
    nz = np.exp(z - zm)
    nz /= nz.sum()

    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    off = n - 1
    for m in range(2 * n - 1):
        wm = float(nz[m]) * dx
        if wm <= 1e-16:
            continue
        k = m - off
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wm * (1.0 - h[k:])
                g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wm * (1.0 - h[:nv])
                g[:nv] += wm * (-h[kk:])

    if mu != 0.0:
        g += mu * (float(h.sum()) - 0.5 * float(n))
    return g


def _sm_obj(x):
    n = int(_GN)
    h = np.asarray(x, dtype=float).reshape(n)
    h = np.clip(h, 0.0, 1.0)
    return smooth_value(h, n, float(_GAL), float(_GMU))


def _sm_grad(x):
    n = int(_GN)
    h = np.asarray(x, dtype=float).reshape(n)
    h = np.clip(h, 0.0, 1.0)
    return smooth_grad(h, n, float(_GAL), float(_GMU))


def lbfgs_refine(h, n, deadline, max_iter=60):
    global _GN, _GAL, _GMU
    n = int(n)
    h = project_feasible(np.asarray(h, dtype=float), n)
    best = h.copy()
    bestc = c5_val(h, n)
    if not np.isfinite(bestc):
        return best, float("inf")

    bounds = [(0.0, 1.0)] * n
    _GN = n
    configs = [
        (50.0, 1.0, 70),
        (200.0, 1.0, 90),
        (1000.0, 1.0, 110),
        (5000.0, 10.0, 130),
    ]

    for alpha, mu, maxfun in configs:
        if time.time() > deadline - 2.0:
            break
        _GAL = float(alpha)
        _GMU = float(mu)
        try:
            res = minimize(
                _sm_obj,
                h,
                jac=_sm_grad,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(max_iter),
                    "maxfun": int(maxfun),
                    "ftol": 1e-16,
                    "gtol": 1e-10,
                    "maxls": 30,
                },
            )
            h_try = project_feasible(res.x, n)
            c_try = c5_val(h_try, n)
            if np.isfinite(c_try) and c_try < bestc:
                bestc = c_try
                best = h_try.copy()
                h = h_try
        except Exception:
            pass

    return best, float(bestc)


def ensure_basis(n):
    global _GB_N, _GB_BASIS
    n = int(n)
    if _GB_BASIS is not None and _GB_N == n:
        return
    x = np.arange(n, dtype=float) * (2.0 / float(n))
    basis = []
    for j in range(6):
        basis.append(np.cos(np.pi * float(2 * j + 1) * x))
    _GB_N = n
    _GB_BASIS = basis


def build_fourier(n, coef):
    n = int(n)
    ensure_basis(n)
    h = np.full(n, 0.5, dtype=float)
    c = np.asarray(coef, dtype=float).reshape(-1)
    c = np.where(np.isfinite(c), c, 0.0)
    for j in range(min(len(c), len(_GB_BASIS))):
        a = float(c[j])
        if a != 0.0:
            h += a * _GB_BASIS[j]
    return project_feasible(h, n)


def fourier_obj(coef):
    n = int(_GF_N)
    if n <= 0:
        return 1.0
    h = build_fourier(n, coef)
    c = c5_val(h, n)
    return c if np.isfinite(c) else 1.0


def fourier_optimize(n, coef0, deadline, max_iter=120):
    global _GF_N
    n = int(n)
    _GF_N = n
    c0 = np.asarray(coef0, dtype=float).reshape(-1)
    c0 = np.where(np.isfinite(c0), c0, 0.0)

    h = build_fourier(n, c0)
    c = c5_val(h, n)
    if not np.isfinite(c):
        c = 1.0

    try:
        res = minimize(
            fourier_obj,
            c0,
            method="Nelder-Mead",
            options={
                "maxiter": int(max_iter),
                "maxfev": int(max_iter * 2),
                "xatol": 1e-7,
                "fatol": 1e-10,
                "adaptive": True,
            },
        )
        if hasattr(res, "x") and res.x is not None:
            try:
                cx = np.asarray(res.x, dtype=float).reshape(-1)
                if cx.size == c0.size:
                    cx = np.where(np.isfinite(cx), cx, 0.0)
                    h2 = build_fourier(n, cx)
                    c2 = c5_val(h2, n)
                    if np.isfinite(c2) and c2 < c:
                        h = h2
                        c = c2
            except Exception:
                pass
    except Exception:
        pass

    h = project_feasible(h, n)
    return h, c5_val(h, n)


def random_fourier_candidate(n, rng, scale=0.10, dim=3):
    n = int(n)
    scale = float(scale)
    dim = int(dim)
    coef = rng.normal(0.0, scale, size=dim)
    h = build_fourier(n, coef)
    return h, c5_val(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(1.0, budget_s) - 2.5
    if deadline <= t0:
        deadline = t0 + 1.0

    rng = np.random.RandomState(int(seed) % (2**31 - 1))

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        p = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if p.size >= 30:
            p = project_feasible(p, p.size)
            cp = c5_val(p, p.size)
            best_h, best_c, best_n = better_candidate(p, cp, p.size, best_h, best_c, best_n)
    except Exception:
        pass

    for n_fix in (300, 400):
        if time.time() > deadline - 5.0:
            break
        if n_fix == 300:
            a_vals = np.linspace(0.25, 0.45, 41)
        else:
            a_vals = np.linspace(0.28, 0.42, 29)
        for a in a_vals:
            if time.time() > deadline - 5.0:
                break
            h = build_fourier(n_fix, [float(a)])
            c = c5_val(h, n_fix)
            best_h, best_c, best_n = better_candidate(h, c, n_fix, best_h, best_c, best_n)

    if best_h is None:
        n_fix = 300
        h = build_fourier(n_fix, [0.35])
        best_h, best_c, best_n = better_candidate(h, c5_val(h, n_fix), n_fix, best_h, best_c, best_n)

    if time.time() < deadline - 120.0:
        for _ in range(8):
            if time.time() > deadline - 60.0:
                break
            h, c = random_fourier_candidate(300, rng, scale=0.10, dim=3)
            best_h, best_c, best_n = better_candidate(h, c, 300, best_h, best_c, best_n)

    if best_h is not None and time.time() < deadline - 200.0:
        n_f = int(best_n) if best_n in (300, 400) else 300
        starts = [
            [0.36, 0.0, 0.0],
            [0.33, 0.10, 0.0],
            [0.32, -0.10, 0.05],
            [0.30, 0.14, 0.05],
        ]
        for co in starts:
            if time.time() > deadline - 80.0:
                break
            tf = min(time.time() + 80.0, deadline - 80.0)
            hf, cf = fourier_optimize(n_f, co, tf, max_iter=120)
            best_h, best_c, best_n = better_candidate(hf, cf, n_f, best_h, best_c, best_n)

    if best_h is not None and time.time() < deadline - 80.0:
        tn = int(best_n)
        if tn > 500:
            tn = 300
        if int(best_n) != tn:
            x_old = np.linspace(0.0, 2.0, int(best_n), endpoint=False)
            x_new = np.linspace(0.0, 2.0, tn, endpoint=False)
            h_init = project_feasible(np.interp(x_new, x_old, best_h), tn)
        else:
            h_init = best_h
        tl = min(time.time() + 300.0, deadline - 20.0)
        hr, cr = lbfgs_refine(h_init, tn, tl, max_iter=60)
        best_h, best_c, best_n = better_candidate(hr, cr, tn, best_h, best_c, best_n)

    if best_h is None:
        best_n = 300
        best_h = build_fourier(best_n, [0.35])
        best_c = c5_val(best_h, best_n)

    best_h = project_feasible(best_h, best_n)
    for _ in range(5):
        target = 0.5 * float(best_n)
        diff = target - float(best_h.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            idx = np.where(best_h < 1.0)[0]
        else:
            idx = np.where(best_h > 0.0)[0]
        if idx.size == 0:
            break
        i = int(idx[idx.size // 2])
        best_h[i] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c = c5_val(best_h, best_n)
    if not np.isfinite(best_c):
        best_c = 1.0
    return best_h, float(best_c), int(best_n)
```
