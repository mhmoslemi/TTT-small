```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad
```

Let me vectorize the inner computation:

```python
# Precompute for the current h:
# For each position p, compute:
#   f1[p, s] = h[p+s] if 0<=p+s<n else 0, for s in range(-(n-1), n)
#   f2[p, s] = h[p-s] if 0<=p-s<n else 0
# Then vec_i = f1[i] - f2[i], vec_j = f1[j] + f2[j]
# delta_j = dx * (vec_i - vec_j + 1)
```

I can precompute f1 and f2 as (n, 2n-1) arrays at the start and update them after each swap.

The update after changing h[i] and h[j]:
- f1[p, s] = h[p+s]: affected when p+s = i or p+s = j, i.e., p = i-s or p = j-s
- f2[p, s] = h[p-s]: affected when p-s = i or p-s = j, i.e., p = i+s or p = j+s

For each s (which is m - (n-1)), the affected p values are: i-s, j-s, i+s, j+s (that are in [0, n)).

So per swap, I update at most 4*(2n-1) entries in f1 and f2. That's about 4800 entries for n=300. Very fast.

Let me rewrite with this optimization:

```python
def discrete_search_fast(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    overlap = compute_c5_vec(h, n)
    current_max = float(overlap.max())
    dx = 2.0 / n
    L = 2 * n - 1
    s_vals = np.arange(L) - (n - 1)  # actual shifts
    
    # Precompute f1[p, m] = h[p + s_m] (0 if OOB)
    # and f2[p, m] = h[p - s_m] (0 if OOB)
    f1 = np.zeros((n, L))
    f2 = np.zeros((n, L))
    for p in range(n):
        for m in range(L):
            s = s_vals[m]
            if 0 <= p + s < n:
                f1[p, m] = h[p + s]
            if 0 <= p - s < n:
                f2[p, m] = h[p - s]
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    max_rounds = 500
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in ones.copy():
            if time.time() > deadline:
                return h, best_c
            if i not in ones:
                continue
            vec_i = f1[i] - f2[i]  # shape (L,)
            # For each j in zeros: delta = dx * (vec_i - (f1[j] + f2[j]) + 1)
            vec_j = f1[zeros] + f2[zeros]  # shape (len(zeros), L)
            deltas = dx * (vec_i[None, :] - vec_j + 1.0)  # shape (len(zeros), L)
            new_maxes = (overlap[None, :] + deltas).max(axis=1)
            best_idx = new_maxes.argmin()
            if new_maxes[best_idx] < current_max - 1e-14:
                j = zeros[best_idx]
                # Do the swap
                h[i] = 0.0
                h[j] = 1.0
                # Update overlap
                overlap = compute_c5_vec(h, n)
                current_max = float(overlap.max())
                best_c = current_max
                improved = True
                # Update f1 and f2
                for m in range(L):
                    s = s_vals[m]
                    # f1[p, m] = h[p+s]: affected for p = i-s, j-s
                    for p in [i - s, j - s]:
                        if 0 <= p < n:
                            f1[p, m] = h[p + s] if 0 <= p + s < n else 0.0
                    # f2[p, m] = h[p-s]: affected for p = i+s, j+s
                    for p in [i + s, j + s]:
                        if 0 <= p < n:
                            f2[p, m] = h[p - s] if 0 <= p - s < n else 0.0
                # Update ones and zeros
                ones = np.delete(ones, np.where(ones == i)[0][0])
                zeros = np.delete(zeros, best_idx)
        if not improved:
            break
    return h, best_c
```

The update of f1 and f2 is O(L) per swap = O(n). The main computation per i is O(|zeros| * L) = O(n²). Total per round: O(|ones| * n²) = O(n³) = 27M for n=300. That's fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_smooth(h, n, deadline, alphas, max_iters_each=300):
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha)
        def gr(x):
            return grad_smooth_max(x, n, alpha)
        try:
            res = minimize(obj, h_cur, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters_each, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters_each * 10, 'maxls': 40})
            h_candidate = project_feasible(res.x, n)
            c_test = compute_c5(h_candidate, n)
            if c_test < best_c:
                best_c = c_test
                h_cur = h_candidate
        except Exception:
            pass
    return h_cur, best_c

def discrete_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    overlap = compute_c5_vec(h, n)
    current_max = float(overlap.max())
    dx = 2.0 / n
    L = 2 * n - 1
    s_vals = np.arange(L) - (n - 1)
    
    f1 = np.zeros((n, L))
    f2 = np.zeros((n, L))
    for p in range(n):
        ps = p + s_vals
        pm = p - s_vals
        mask_p = (ps >= 0) & (ps < n)
        mask_m = (pm >= 0) & (pm < n)
        f1[p, mask_p] = h[ps[mask_p]]
        f2[p, mask_m] = h[pm[mask_m]]
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    max_rounds = 300
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(ones)):
            i = ones[ii]
            vec_i = f1[i] - f2[i]
            z_arr = np.array(zeros)
            vec_j = f1[z_arr] + f2[z_arr]
            deltas = dx * (vec_i[None, :] - vec_j + 1.0)
            new_maxes = (overlap[None, :] + deltas).max(axis=1)
            best_idx = int(new_maxes.argmin())
            if new_maxes[best_idx] < current_max - 1e-14:
                j = z_arr[best_idx]
                h[i] = 0.0
                h[j] = 1.0
                overlap = compute_c5_vec(h, n)
                current_max = float(overlap.max())
                best_c = current_max
                improved = True
                # Update f1, f2
                for m in range(L):
                    s = s_vals[m]
                    p1 = i - s
                    if 0 <= p1 < n:
                        f1[p1, m] = h[p1 + s] if 0 <= p1 + s < n else 0.0
                    p2 = j - s
                    if 0 <= p2 < n:
                        f1[p2, m] = h[p2 + s] if 0 <= p2 + s < n else 0.0
                    p3 = i + s
                    if 0 <= p3 < n:
                        f2[p3, m] = h[p3 - s] if 0 <= p3 - s < n else 0.0
                    p4 = j + s
                    if 0 <= p4 < n:
                        f2[p4, m] = h[p4 - s] if 0 <= p4 - s < n else 0.0
                ones.pop(ii)
                zeros.pop(best_idx)
                ii -= 1  # since we removed one, re-check same index
        if not improved:
            break
    return h, best_c

def pair_search_fine(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            order = np.arange(n)
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

    h_bin = np.where(best_h > 0.5, 1.0, 0.0)

        h_dis, c_dis = discrete_search(h_bin, n, t2)
        if c_dis < best_c:
            best_c = c_dis
            best_h = h_dis
        else:

            h_push = best_h.copy()
            for _ in range(50):

                if c_push <= best_c + 0.001:

    if time.time() < deadline - 200:

    # Phase 4: Another round of discrete search (in case continuous moved us off binary)
    if time.time() < deadline - 150:
        h_bin2 = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin2 = project_feasible(h_bin2, n)
        t4 = min(time.time() + 60, deadline - 150)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m
        wm = w[m] * dx
        nv = n - k
        if nv > 0:
            grad[0:nv] += wm * (1.0 - h[0:nv])
            grad[k:n] += wm * (-h[0:nv])
    return grad

def lbfgs_smooth(h, n, deadline, alphas, max_iters_each=200):
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha)
        def gr(x):
            return grad_smooth_max(x, n, alpha)
        try:
            res = minimize(obj, h_cur, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters_each, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters_each * 10, 'maxls': 40})
            h_candidate = project_feasible(res.x, n)
            c_test = compute_c5(h_candidate, n)
            if c_test < best_c:
                best_c = c_test
                h_cur = h_candidate
        except Exception:
            pass
    return h_cur, best_c

def make_binary(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def build_D(h, n):
    L = 2 * n - 1
    D = np.zeros((n, L))
    for p in range(n):
        pm = p - np.arange(L)
        pp = p + np.arange(L)
        mask_pm = (pm >= 0) & (pm < n)
        mask_pp = (pp >= 0) & (pp < n)
        D[p, mask_pp] += (1.0 - h[pm[mask_pp]])
        D[p, mask_pp] -= h[pp[mask_pp]]
    return D

def discrete_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    current_max = best_c
    dx = 2.0 / n
    L = 2 * n - 1
    D = build_D(h, n)
    overlap = compute_c5_vec(h, n)
    current_max = float(overlap.max())
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    for rnd in range(500):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(ones)):
            i = ones[ii]
            z_arr = np.array(zeros)
            if len(z_arr) == 0:
                break
            deltas = dx * (-D[i][None, :] + D[z_arr])
            new_maxes = (overlap[None, :] + deltas).max(axis=1)
            best_idx = int(new_maxes.argmin())
            if new_maxes[best_idx] < current_max - 1e-14:
                j = z_arr[best_idx]
                h[i] = 0.0
                h[j] = 1.0
                overlap = compute_c5_vec(h, n)
                current_max = float(overlap.max())
                best_c = current_max
                improved = True
                for p in range(n):
                    pm = p - np.arange(L)
                    pp = p + np.arange(L)
                    mask_pm = (pm >= 0) & (pm < n)
                    mask_pp = (pp >= 0) & (pp < n)
                    D[p, :] = 0.0
                    D[p, mask_pp] += (1.0 - h[pm[mask_pp]])
                    D[p, mask_pp] -= h[pp[mask_pp]]
                ones.pop(ii)
                zeros.pop(best_idx)
                ii -= 1
        if not improved:
            break
    return h, best_c

def pair_search_fine(h, n, deadline, deltas, radius=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS smooth max
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_smooth(best_h, n, t1,
                                   alphas=[20, 50, 100, 500, 2000, 10000],
                                   max_iters_each=250)
    # Phase 2: Discrete search with restarts
    t2 = deadline - 200
    h_bin = make_binary(best_h, n)
    for restart in range(8):
        if time.time() > t2 - 20:
            break
        if restart > 0:
            k = rng.randint(3, 15)
            pos_1 = np.where(h_bin > 0.5)[0]
            pos_0 = np.where(h_bin < 0.5)[0]
            for _ in range(k):
                if len(pos_1) == 0 or len(pos_0) == 0:
                    break
                a = pos_1[rng.randint(len(pos_1))]
                b = pos_0[rng.randint(len(pos_0))]
                h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
        alloc = max(15, min(100, t2 - time.time() - 15))
        h_d, c_d = discrete_search(h_bin.copy(), n, time.time() + alloc)
        if c_d < best_c:
            best_c = c_d
            best_h = h_d.copy()
            h_bin = h_d.copy()
    # Phase 3: Continuous refinement
    if time.time() < deadline - 100:
        t3 = min(time.time() + 80, deadline - 100)
        best_h, best_c = lbfgs_smooth(best_h, n, t3,
                                       alphas=[5000, 20000, 100000, 500000],
                                       max_iters_each=300)
    # Phase 4: Discrete again
    if time.time() < deadline - 60:
        h_bin2 = make_binary(best_h, n)
        h_d2, c_d2 = discrete_search(h_bin2, n, min(time.time() + 40, deadline - 60))
        if c_d2 < best_c:
            best_c = c_d2
            best_h = h_d2
        # Also try the continuous version
        h_d3, c_d3 = discrete_search(make_binary(best_h, n), n, min(time.time() + 30, deadline - 55))
        if c_d3 < best_c:
            best_c = c_d3
            best_h = h_d3
    # Phase 5: Fine pair search
    if time.time() < deadline - 10:
        best_h, best_c = pair_search_fine(best_h, n, deadline - 5,
                                           [0.01, 0.005, 0.001, 0.0005, 0.0001],
                                           radius=120)
    # Phase 6: Final polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_search_fine(best_h, n, deadline - 1,
                                           [0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                           radius=60)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
