import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    val = c.max() + (mx + np.log(w.sum())) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return val, grad

class SMObj:
    def __init__(self, n, alpha, mu):
        self.n = n
        self.alpha = alpha
        self.mu = mu
        self.target = n / 2.0
    def __call__(self, x):
        v, _ = smooth_max_grad(x, self.n, self.alpha)
        return v + self.mu * (x.sum() - self.target) ** 2
    def grad(self, x):
        _, g = smooth_max_grad(x, self.n, self.alpha)
        g += 2.0 * self.mu * (x.sum() - self.target)
        return g

        t += 1
        frac = t / max_steps
