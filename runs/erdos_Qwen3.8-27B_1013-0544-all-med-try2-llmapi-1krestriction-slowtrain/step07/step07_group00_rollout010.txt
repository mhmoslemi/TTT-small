import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def _np2(n):
    p = 1
    while p < n:
        p <<= 1
    return p

def fft_corr(a, b):
    n = len(a)
    N = _np2(2 * n - 1)
    return np.real(ifft(fft(a[::-1], N) * fft(b, N)))[:2*n-1]

def c5v(h):
    n = len(h)
    return fft_corr(h, 1.0 - h) * (2.0 / n)

def c5m(h):
    return float(np.max(c5v(h)))

def proj(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15: break
            h += (d/s) * r
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (d/s) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = c5v(x)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    v = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    v += mu * (x.sum() - n * 0.5) ** 2
    return v

def sm_grad(x, n, alpha, mu):
    c = c5v(x)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    act = np.where(w > 1e-13)[0]
    for mi in act:
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

    return x.sum() - n * 0.5

def eq_jac(x, n):
    return np.ones(n)

def sm_obj_eq(x, n, alpha):
    c = c5v(x)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def slsqp(h, n, deadline, alpha=500000, maxiter=5000):

def _np2(n):
    p = 1
    while p < n:
        p <<= 1
    return p

def fft_corr(a, b):
    n = len(a)
    N = _np2(2 * n - 1)
    c = np.real(ifft(fft(a[::-1], N) * fft(b, N)))[:2 * n - 1]
    return c[::-1]

def c5v(h):
    n = len(h)
    return fft_corr(h, 1.0 - h) * (2.0 / n)

def c5m(h):
    return float(np.max(c5v(h)))

def proj(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = c5v(x)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    v = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    v += mu * (x.sum() - n * 0.5) ** 2
    return v

def sm_grad(x, n, alpha, mu):
    c = c5v(x)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    act = np.where(w > 1e-12)[0]
    for mi in act:
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

            best_c = c_f
            best_h = h_f
    # Phase 7: Final local search
    if time.time() < deadline - 5:
        best_h, best_c = grad_local(best_h, n, deadline - 2)
    best_h = proj(best_h, n)
    best_c = c5m(best_h)
    return best_h, best_c, n
