import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        lse = (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
    else:
        lse = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + lse)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sa_run(h, n, deadline, rng, t0=0.008, cool=0.9998, min_t=5e-7):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    temp = t0
    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.1)
        moved = False
        if h[i] >= d and h[j] <= 1.0 - d:
            h[i] -= d; h[j] += d
            nc = compute_c5(h, n)
            dc = nc - cur_c
            if dc < 0 or (dc < 0.005 and rng.rand() < np.exp(-dc / temp)):
                cur_c = nc
                if nc < best_c - 1e-15:
                    best_c = nc; best_h = h.copy(); stall = 0
                else:
                    stall += 1
            else:
                h[i] += d; h[j] -= d
            moved = True
        elif h[j] >= d and h[i] <= 1.0 - d:
            h[j] -= d; h[i] += d
            nc = compute_c5(h, n)
            dc = nc - cur_c
            if dc < 0 or (dc < 0.005 and rng.rand() < np.exp(-dc / temp)):

                if nc < best_c - 1e-15:

                else:
                    stall += 1

                h[j] += d; h[i] -= d
            moved = True
        if not moved:
            continue
        temp *= cool
        if stall > 2000:
            temp = min(temp * 2.0, 0.015)
            stall = 0
        if temp < min_t:
