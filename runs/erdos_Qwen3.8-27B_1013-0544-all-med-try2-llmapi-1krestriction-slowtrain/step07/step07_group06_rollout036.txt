import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_a_vec(i, h, n):
    """Compute coefficient vector a[m] for coordinate i."""
    # a[m] = (1 - h[i+m-(n-1)]) - h[i-m+(n-1)] with zero boundary
    m = np.arange(2 * n - 1)
    j1 = i + m - (n - 1)
    j2 = i - m + (n - 1)
    c1 = np.where((j1 >= 0) & (j1 < n), 1.0 - h[j1], 1.0)
    c2 = np.where((j2 >= 0) & (j2 < n), h[j2], 0.0)
    return c1 - c2

def solve_1d_convex(base, a, lam, max_iter=80):
    """Minimize max_m(base_m + t*a_m) + lam*t over t in [0,1]."""
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        vals = base + mid * a
        f_val = vals.max()
        tol = 1e-10
        active = vals >= f_val - tol
        if not active.any():
            break
        min_a = a[active].min() + lam
        max_a = a[active].max() + lam
        if min_a <= 0 <= max_a:
            break
        elif max_a < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5

def coord_descent_lagrangian(h, n, deadline, max_sweeps=200, max_bisect=20, max_iter_1d=60):
    """Coordinate descent with Lagrangian relaxation for sum constraint."""
    h = project_feasible(h.copy(), n)
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Bisection on lambda
        lam_lo, lam_hi = -2.0, 2.0
        converged = False
        for b in range(max_bisect):
            if time.time() > deadline:
                break
            lam = (lam_lo + lam_hi) * 0.5
            h_trial = h.copy()
            corr = np.correlate(h_trial, 1 - h_trial, mode='full')
            
            for i in range(n):
                a_m = compute_a_vec(i, h_trial, n)
                base = corr - h_trial[i] * a_m
                t_opt = solve_1d_convex(base, a_m, lam, max_iter_1d)
                delta = t_opt - h_trial[i]
                if abs(delta) > 1e-15:
                    corr += delta * a_m
                    h_trial[i] = t_opt
            
            s = h_trial.sum()
            if s > target:
                lam_lo = lam
            else:
                lam_hi = lam
            if abs(lam_hi - lam_lo) < 1e-4:
                converged = True
                break
        
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial <= best_c + 1e-12:
            best_c = c_trial
            best_h = h_trial.copy()
            h = h_trial
        else:
            # Regress: try smaller step
            h_best_trial = project_feasible(0.5 * h + 0.5 * h_trial, n)
            c_rt = compute_c5(h_best_trial, n)
            if c_rt < best_c:
                best_c = c_rt
                best_h = h_best_trial.copy()
                h = h_best_trial
            if not converged:
                continue
    
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    corr = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = corr * dx
    sh = alpha * (c - c.max())
    mx = sh.max()
    return c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):
    corr = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = corr * dx
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    # grad[m] = sum over pairs (i,j) with i-j = k corresponding to shift m
    # corr[m] = sum_i x[i] * (1-x[i+m-(n-1)])
    # d(corr[m])/d(x[i]) = (1-x[i+m-(n-1)]) - x[i-m+(n-1)]
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        wm = w[m] * dx
        j1 = np.arange(n) + m - (n - 1)
        j2 = np.arange(n) - m + (n - 1)
        c1 = np.where((j1 >= 0) & (j1 < n), 1.0 - x[j1[(j1 >= 0) & (j1 < n)]], 1.0)
        c2 = np.where((j2 >= 0) & (j2 < n), x[j2[(j2 >= 0) & (j2 < n)]], 0.0)
        grad += wm * (c1 - c2)
    return grad

def lbfgs_polish(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 500, 5000, 50000, 500000]
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_max_obj(x, n, alpha) + 10.0 * (x.sum() - n/2.0)**2
        def grad(x):
            return smooth_max_grad(x, n, alpha) + 20.0 * (x.sum() - n/2.0) * np.ones(n)
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(x, n, alpha):
    corr = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = corr * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
