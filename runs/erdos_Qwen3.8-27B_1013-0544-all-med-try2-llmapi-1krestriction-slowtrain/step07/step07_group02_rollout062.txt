                if pos_B.any():
                    ub_x = min(ub_x, h[i]+np.min((mid-corr[pos_B])/B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i]+np.max((hi_v-corr[neg_B])/B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i]+np.min((hi_v-corr[pos_B])/B[pos_B]))
            x_opt = np.clip((lb_x+ub_x)/2.0, 0.0, 1.0)
            if abs(x_opt-h[i]) > 1e-15:
                corr += B*(x_opt-h[i])
                h[i] = x_opt
    return h, corr

def smooth_optim(h, n, alpha, maxiter=2000, deadline=None):
    dx = 2.0 / n
    target = n / 2.0
    def obj_and_grad(x):
        g = np.correlate(x, 1.0-x, 'full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        w = exps / exps.sum()
        val = m + (sm + np.log(exps.sum())) / alpha
        wc = np.convolve(x, w, 'full')
        wcr = np.correlate(x, w, 'full')
        grad = dx * (w.sum() - wc[n-1:2*n-1] - wcr[n-1:2*n-1])
        return val, grad
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(obj_and_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h, n)

def binary_search(h_init, n, deadline, max_sweeps=50):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, 'full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            bi = np.zeros(2*n-1)
            k1 = i - m_range
            m1 = (k1>=0)&(k1<n)
            bi[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2>=0)&(k2<n)
            bi[m2] -= h[k2[m2]]
            best_j = -1
            best_val = cur_c
            for j in ones:
                bj = np.zeros(2*n-1)
                k1 = j - m_range
                m1 = (k1>=0)&(k1<n)
                bj[m1] += (1.0 - h[k1[m1]])
                k2 = j + m_range
                m2 = (k2>=0)&(k2<n)
                bj[m2] -= h[k2[m2]]
                nc = (corr + bi - bj).max() * dx
                if nc < best_val:
                    best_val = nc
                    best_j = j
            if best_j >= 0 and best_val < cur_c - 1e-15:
                h[i], h[best_j] = 1.0, 0.0
                corr += bi
                bj = np.zeros(2*n-1)
                k1 = best_j - m_range
                m1 = (k1>=0)&(k1<n)
                bj[m1] += (1.0 - h[k1[m1]])
                k2 = best_j + m_range
                m2 = (k2>=0)&(k2<n)
                bj[m2] -= h[k2[m2]]
                corr += bj
                cur_c = best_val
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Initial coord descent
    p1 = t0 + 0.08 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=100, deadline=p1)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 2: Low-res optimization pipeline
    p2 = t0 + 0.35 * total
    for nh in [100, 150, 200, 250, 300]:
        if time.time() > p2 - 20:
            break
        if nh >= n:
            continue
        xh = interp_to(best_h, n, nh)
        xh = project_feasible(xh, nh)
        # Coord descent at low res (fast)
        xh, _ = coord_descent(xh, nh, max_sweeps=300, deadline=p2 - 15)
        xh = project_feasible(xh, nh)
        # SLSQP at low res
        for al in [200, 1000, 5000, 20000, 100000, 500000]:
            if time.time() > p2 - 10:
                break
            xh = smooth_optim(xh, nh, al, maxiter=1500, deadline=p2 - 8)
            ch = compute_c5(xh, nh)
            # Interpolate back up
            xh_up = interp_to(xh, nh, n)
            xh_up = project_feasible(xh_up, n)
            ch_up = compute_c5(xh_up, n)
            if ch_up < best_c:
                best_c, best_h = ch_up, xh_up.copy()
            # Also check low-res directly (it's a valid solution too)
            if ch < best_c and nh < n:
                pass  # We report at n resolution

    # Phase 3: SLSQP at full resolution
    p3 = t0 + 0.65 * total
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p3 - 8:
            break
        h = smooth_optim(h, n, alpha, maxiter=3000, deadline=p3 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Coord descent polish
    p4 = t0 + 0.80 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p4)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 5: More SLSQP from coord descent
    p5 = t0 + 0.92 * total
    h = best_h.copy()
    for alpha in [5000000, 50000000, 200000000]:
        if time.time() > p5 - 5:
            break
        h = smooth_optim(h, n, alpha, maxiter=4000, deadline=p5 - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Noise restarts
    if time.time() < dl - 5:
        for trial in range(6):
            if time.time() > dl - 2:
                break
            h = best_h + 0.002 * rng.randn(n)
            h = project_feasible(h, n)
            h = smooth_optim(h, n, 50000000, maxiter=2000, deadline=dl - 1)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
