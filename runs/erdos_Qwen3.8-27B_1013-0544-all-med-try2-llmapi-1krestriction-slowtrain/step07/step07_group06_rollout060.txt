def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-13:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])

def nesterov_opt(h, n, deadline, alpha=300, lr=0.015, num_steps=2000):

    momentum = 0.9
    for t in range(num_steps + 1):

        lr_cur = lr * (1.0 + m) ** (-0.5)

        vel = momentum * vel - lr_cur * g
        x_new = y + vel

        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()

    # Phase 3: Pair descent (250s)
    if time.time() < deadline - 500:
        t3 = min(time.time() + 250, deadline - 500)
        h3, c3 = pair_descent(best_h, n, t3, alpha=200, top_k=100, n_t=9)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Asbinarize again (40s)
    if time.time() < deadline - 400:
        t4 = min(time.time() + 40, deadline - 400)
        h4, c4 = asbinarize(best_h, n, t4, max_rounds=300, top_candidates=10)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Nesterov momentum (80s)
    if time.time() < deadline - 300:
        t5 = min(time.time() + 80, deadline - 300)
        h5, c5 = nesterov_opt(best_h, n, t5, alpha=300, lr=0.012, num_steps=1500)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Pair descent with higher alpha (200s)
    if time.time() < deadline - 120:
        t6 = min(time.time() + 200, deadline - 120)
        h6, c6 = pair_descent(best_h, n, t6, alpha=800, top_k=120, n_t=11)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Asbinarize final (30s)
    if time.time() < deadline - 60:
        t7 = min(time.time() + 30, deadline - 60)
        h7, c7 = asbinarize(best_h, n, t7, max_rounds=200)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: L-BFGS polish (30s)
    if time.time() < deadline - 30:
        t8 = min(time.time() + 30, deadline - 30)
        h8, c8 = lbfgs_run(best_h, n, t8, max_iters=100)
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Phase 9: Try other n (60s)
    if time.time() < deadline - 15:
        t9 = min(time.time() + 60, deadline - 15)
        best_h, best_c, n = try_interpolate(best_h, best_c, n, t9)

    # Phase 10: Final pair polish
    if time.time() < deadline - 1:
        h10, c10 = pair_descent(best_h, n, deadline - 1, alpha=1000, top_k=60, n_t=7)
        if c10 < best_c:
            best_c, best_h = c10, h10

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
