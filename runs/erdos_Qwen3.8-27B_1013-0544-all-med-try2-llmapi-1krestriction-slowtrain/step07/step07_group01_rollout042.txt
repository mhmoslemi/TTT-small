            if tot < 1e-14: break

                                  alpha=1000000, maxit=400)
            if cr < best_c:
                best_c = cr
                best_h = hr

    # Phase 5: SLSQP final push
    if time.time() < deadline - 130:
        best_h, best_c = slsqp_opt(best_h, n, min(time.time() + 35, deadline - 130),
                                  alpha=5000000, maxit=600)

    # Phase 6: Block search
    if time.time() < deadline - 80:
        best_h, best_c = block_search(best_h, n, min(time.time() + 50, deadline - 80))

    # Phase 7: Fine pair search
    if time.time() < deadline - 30:
        best_h, best_c = pair_search(best_h, n, min(time.time() + 80, deadline - 30), radius=50)

    # Phase 8: Try other grid sizes
    if time.time() < deadline - 50:
        best_h, best_c, n = multi_grid_opt(best_h, n, deadline - 5,
                                          [200, 250, 400], rng)

    # Final: one more pair search pass
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 2, radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
