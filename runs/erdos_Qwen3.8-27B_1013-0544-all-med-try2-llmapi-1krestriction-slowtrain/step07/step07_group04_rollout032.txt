import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(g, alpha):
    m = g.max()
    if alpha > 1e6:
        return m
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_max_weights(g, alpha):
    m = g.max()
    if alpha > 1e6:
        w = np.zeros_like(g)
        idx = np.where(g == m)[0]
        w[idx] = 1.0 / len(idx)
        return w
    sh = alpha * (g - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    return w

def compute_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    G = corr * dx
    w = smooth_max_weights(G, alpha)
    
    # For each j, grad[j] = dx * sum_l w[l] * (A[j,l] - B[j,l])
    # where l = idx - (n-1), A[j,l] = (1-h[j-l]) if 0<=j-l<n else 0
    # B[j,l] = h[j+l] if 0<=j+l<n else 0
    # 
    # Equivalently, for output[n-1+l]:
    # grad[j] += dx * w[n-1+l] * ((1-h[j-l]) * 1[0<=j-l<n] - h[j+l] * 1[0<=j+l<n])
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        l = idx - (n - 1)
        # A[j,l] = (1-h[j-l]) for j in [l, n-1] (when l>=0) or [0, n-1+l] (when l<0)
        # Actually: 0 <= j-l < n means l <= j < n+l
        # For l >= 0: j in [l, n-1] (since j < n always)
        # For l < 0: j in [0, n-1+l] (since j >= 0 always)
        # B[j,l] = h[j+l] for 0 <= j+l < n means -l <= j < n-l
        # For l >= 0: j in [0, n-1-l]
        # For l < 0: j in [-l, n-1]
        
        if l >= 0:
            jA = np.arange(l, n)  # A is nonzero
            grad[jA] += dx * w[idx] * (1.0 - h[jA - l])
            jB = np.arange(0, n - l) if n - l > 0 else np.empty(0, dtype=int)
            grad[jB] -= dx * w[idx] * h[jB + l]
        else:
            # l < 0
            jA = np.arange(0, n + l) if n + l > 0 else np.empty(0, dtype=int)
            grad[jA] += dx * w[idx] * (1.0 - h[jA - l])
            jB = np.arange(-l, n)
            grad[jB] -= dx * w[idx] * h[jB + l]
    
    return grad

def lbgfB_run(h0, n, alpha, mu, maxiter, deadline):
    """Run L-BFGS-B with penalty for equality constraint."""
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        G = corr * dx
        sm = smooth_max_val(G, alpha)
        penalty = mu * (x.sum() - target) ** 2
        return sm + penalty
    
    def grad_func(x):
        g = compute_gradient(x, n, alpha)
        g += 2 * mu * (x.sum() - target)
        return g
    
    res = minimize(obj, h0, jac=grad_func, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                            'maxls': 50, 'maxfun': maxiter * 10})
    return res.x

def binary_local_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    """Fast binary local search with pair swaps."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    if n % 2 == 1:
        h[idx[target]] = 1.0  # for odd n
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    current_h = h.copy()

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(3, 7)):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = sorted(zeros)
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = sorted(ones)
                for j in o_list:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2*n-1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2*n-1:
                        delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: L-BFGS-B with increasing alpha (penalty method)
    p1 = t0 + 0.20 * (budget_s - 3)
    h = best_h.copy()

        if time.time() > p1:
            break
        mu = 100.0
        h = lbgfB_run(h, n, alpha, mu, 3000, p1)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary local search
    p2 = t0 + 0.40 * (budget_s - 3)
    hb, cb = binary_local_search(best_h, n, p2, max_sweeps=600, n_restarts=5, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: L-BFGS-B polish from binary solution
    p3 = t0 + 0.55 * (budget_s - 3)

        for alpha in [10000, 50000, 200000, 1000000]:
            if time.time() > p3:

            hp = lbgfB_run(hp, n, alpha, 100.0, 3000, p3)
            hp = project_feasible(hp, n)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 4: Multi-resolution
    p4 = t0 + 0.70 * (budget_s - 3)
    for nh in [200, 300]:
        if time.time() > p4 - 30:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [1000, 10000, 100000]:
            if time.time() > p4 - 20:
                break
            hh = lbgfB_run(hh, nh, alpha, 100.0, 2000, p4 - 15)
            hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh

    # Phase 5: Extended binary search
    p5 = dl - 20
    if time.time() < p5:
        hb, cb = binary_local_search(best_h, n, p5, max_sweeps=2000, n_restarts=5, seed=seed+100)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 6: Final L-BFGS-B polish
    if time.time() < dl - 5:
        h = lbgfB_run(best_h + 0.01 * rng.randn(n), n, 1000000, 100.0, 5000, dl - 3)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(g, alpha):
    m = g.max()
    if alpha > 1e6:
        return m
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    G = corr * dx
    m = G.max()
    if alpha > 1e6:
        w = np.zeros_like(G)
        idx = np.where(G == m)[0]
        w[idx] = 1.0 / len(idx)
    else:
        sh = alpha * (G - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        l = idx - (n - 1)
        if l >= 0:
            if l < n:
                jA = np.arange(l, n)
                grad[jA] += dx * w[idx] * (1.0 - h[jA - l])
            jB = np.arange(0, n - l)
            grad[jB] -= dx * w[idx] * h[jB + l]
        else:
            nl = n + l
            if nl > 0:
                jA = np.arange(0, nl)
                grad[jA] += dx * w[idx] * (1.0 - h[jA - l])
            jB = np.arange(-l, n)
            grad[jB] -= dx * w[idx] * h[jB + l]
    return grad

    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        G = corr * (2.0 / n)
        sm = smooth_max_val(G, alpha)
        p = mu * (x.sum() - target) ** 2
        return sm + p

        g = compute_gradient(x, n, alpha)
        g += 2.0 * mu * (x.sum() - target)
        return g
    try:
        res = minimize(obj, h0, jac=gradf, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxls': 40, 'maxfun': maxiter * 5})
        return res.x
    except Exception:
        return h0

    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    current_h = h.copy()
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(3, 8)):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    current_h[rng.choice(oi)] = 0.0
                    current_h[rng.choice(zi)] = 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            for i in sorted(zeros):
                if time.time() > deadline:
                    break
                for j in sorted(ones):
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    nc = (corr + delta).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr += delta
                        cur_c = nc
                        ones.add(i)
                        ones.discard(j)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(g, alpha):
    m = g.max()
    if alpha > 1e6:
        return m
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    G = corr * dx
    m = G.max()
    if alpha > 1e6:
        w = np.zeros_like(G)
        idx = np.where(G == m)[0]
        w[idx] = 1.0 / len(idx)
    else:
        sh = alpha * (G - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-18:
            continue
        l = idx - (n - 1)
        if l >= 0:
            if l < n:
                jA = np.arange(l, n)
                grad[jA] += dx * w[idx] * (1.0 - h[jA - l])
            if n - l > 0:
                jB = np.arange(0, n - l)
                grad[jB] -= dx * w[idx] * h[jB + l]
        else:
            nl = n + l
            if nl > 0:
                jA = np.arange(0, nl)
                grad[jA] += dx * w[idx] * (1.0 - h[jA - l])
            jB = np.arange(-l, n)
            grad[jB] -= dx * w[idx] * h[jB + l]
    return grad

def lbgfB_opt(h0, n, alpha, mu, maxiter, deadline):
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        G = corr * (2.0 / n)
        return smooth_max_val(G, alpha) + mu * (x.sum() - target) ** 2
    def gradf(x):
        g = compute_gradient(x, n, alpha)
        g += 2.0 * mu * (x.sum() - target)
        return g
    try:
        res = minimize(obj, h0, jac=gradf, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxls': 40, 'maxfun': maxiter * 5})
        return res.x
    except Exception:
        return h0

def binary_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)

    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()

    rng = np.random.RandomState(seed)
    current_h = h.copy()
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(3, 8)):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    current_h[rng.choice(oi)] = 0.0
                    current_h[rng.choice(zi)] = 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            for i in sorted(zeros):
                if time.time() > deadline:
                    break
                for j in sorted(ones):
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    nc = (corr + delta).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr += delta
                        cur_c = nc
                        ones.add(i)
                        ones.discard(j)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

    half = period // 2
    for i in range(n):
        if (i % period) < half:
            h[i] = 1.0
    return project_feasible(h, n)
