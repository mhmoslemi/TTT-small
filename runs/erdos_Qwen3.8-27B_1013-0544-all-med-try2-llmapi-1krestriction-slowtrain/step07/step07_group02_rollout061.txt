def subgradient_step(h, n, f_lb=0.3, step_scale=1.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr.max() * dx
    
    # Active set: correlations within tolerance of max
    tol = 0.01 * f
    active = np.where(corr >= corr.max() - tol)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    m_range = np.arange(-(n-1), n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual shift
        # d(c_k)/dh[i] = 1 - h[i+k] - h[i-k]
        for i in range(n):
            val = 1.0
            if 0 <= i + k < n:
                val -= h[i + k]
            if 0 <= i - k < n:
                val -= h[i - k]
            g[i] += val * dx
    
    # Polyak step
    g_norm_sq = np.dot(g, g)
    if g_norm_sq < 1e-20:
        return h, f
    eta = step_scale * (f - f_lb) / g_norm_sq
    h_new = h - eta * g
    h_new = project_feasible(h_new, n)
    return h_new, f

    # This is a cross-correlation

    A_neg = np.zeros(n_corr)
    for j in range(len(active_indices)):
        k_val = active_indices[j] - (n-1)
        neg_idx = (n-1) - k_val  # index for -k
        if 0 <= neg_idx < n_corr:
            A_neg[neg_idx] += 1.0
    
    F_A_neg = np.fft.rfft(A_neg, N_fft)
    cross_neg = np.fft.irfft(H * np.conj(F_A_neg), N_fft)
    
    g = dx * (len(active_indices) - cross[:n] - cross_neg[:n])
    
    # Polyak step
    g_norm_sq = np.dot(g, g)
    if g_norm_sq < 1e-20:
        return h, f
    eta = step_scale * (f - f_lb) / g_norm_sq
    h_new = h - eta * g
    h_new = project_feasible(h_new, n)
    return h_new, f

            order = rng.permutation(n)

            # B = 1 - h[i+m] - h[i-m]
            B = np.ones(n_corr)
            B[valid_im[i]] -= h[im_all[i][valid_im[i]]]
            B[valid_im2[i]] -= h[im2_all[i][valid_im2[i]]]
            
            d = corr - B * h[i]
            
            # Bisection (25 iterations)
            lo_x, hi_x = 0.0, 1.0
            for _ in range(25):
                mid = (lo_x + hi_x) * 0.5
                if np.max(d + B * mid) <= np.max(d + B * lo_x):
                    hi_x = mid
                else:
                    lo_x = mid
            x_opt = (lo_x + hi_x) * 0.5
            
            if abs(x_opt - h[i]) > 1e-13:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    
    return h, corr.max() * dx

Let me also add a **binary search on T** approach:

For a target T, define the penalty: P(h) = max_k (c_k * dx - T). We want P(h) ≤ 0.

Use SLSQP to minimize P(h) (or a smooth approximation) subject to constraints.

Actually, a cleaner approach: minimize the log-sum-exp:
L(h) = (1/alpha) * log(sum_k exp(alpha * c_k * dx))

As alpha → ∞, this approaches max_k c_k * dx.

The gradient of L with respect to h[i]:
∂L/∂h[i] = (1/alpha) * sum_k exp(alpha*c_k*dx) * alpha*dx * ∂c_k/∂h[i] / sum_k exp(alpha*c_k*dx)
= dx * sum_k p_k * ∂c_k/∂h[i]

where p_k = exp(alpha*c_k*dx) / sum_j exp(alpha*c_j*dx)

This is a weighted average of the active set subgradients, with softmax weights.

This is actually a nice smooth objective that we can minimize with projected gradient or SLSQP.

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-14):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    m = np.arange(n_corr) - (n - 1)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        shifted = alpha * (c - c.max())  # numerical stability
        return c.max() + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0, 1)

def projected_gradient(h, n, alpha, steps=10000, deadline=None):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h = h.astype(float).copy()
    
    for step in range(steps):
        if deadline and time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        shifted = alpha * (c - c.max())
        weights = np.exp(shifted - shifted.max())
        weights /= weights.sum()
        
        # Gradient: dx * sum_k weights[k] * ∂c_k/∂h
        # ∂c_k/∂h[i] = 1 - h[i+k] - h[i-k] (with b.c.)
        # This is: dx * sum_k weights[k] * (1 - h[i+k] - h[i-k])
        # = dx * (1 - sum_k weights[k]h[i+k] - sum_k weights[k]h[i-k])
        # = dx * (1 - (w * h)[i] - (w_rev * h)[i])
        # where (w * h)[i] = sum_k w[k]h[i+k] is a convolution
        
        # Use FFT for convolution
        N = 2 * n_corr
        H = np.fft.rfft(h, N)
        W = np.fft.rfft(weights, N)
        conv1 = np.fft.irfft(H * np.conj(W), N)[:n]  # sum_k w[k] h[i+k]
        # For sum_k w[k] h[i-k], we need the correlation:
        # sum_k w[k] h[i-k] = sum_k w[-k] h[i+k]
        w_neg = np.zeros(n_corr)
        for j in range(n_corr):
            w_neg[n_corr - 1 - j] = weights[j]
        W_neg = np.fft.rfft(w_neg, N)
        conv2 = np.fft.irfft(H * np.conj(W_neg), N)[:n]
        
        grad = dx * (1.0 - conv1 - conv2)
        
        # Adaptive step size (backtracking)
        obj_val = alpha * (c.max() - c.max()) + np.log(np.exp(shifted).sum()) / alpha  # just for tracking
        # Actually let's just track the max directly
        cur_f = c.max()
        
        eta = 0.1
        for _ in range(30):
            h_new = project_feasible(h - eta * grad, n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
            if c_new.max() < cur_f:
                h = h_new
                break
            eta *= 0.5
    
    return h

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_shift_masks(n):
    n_corr = 2 * n - 1
    m = np.arange(n_corr) - (n - 1)
    im_all = np.arange(n)[:, None] + m[None, :]
    im2_all = np.arange(n)[:, None] - m[None, :]
    valid_im = (im_all >= 0) & (im_all < n)
    valid_im2 = (im2_all >= 0) & (im2_all < n)
    return im_all, im2_all, valid_im, valid_im2

def coord_descent(h, n, max_sweeps=3000, deadline=None, n_bisect=25, seed=0):
    im_all, im2_all, valid_im, valid_im2 = make_shift_masks(n)
    n_corr = 2 * n - 1
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        order = rng.permutation(n)
        any_change = False
        for i in order:
            B = np.ones(n_corr)
            vi = valid_im[i]
            B[vi] -= h[im_all[i][vi]]
            v2 = valid_im2[i]
            B[v2] -= h[im2_all[i][v2]]
            d = corr - B * h[i]
            lo, hi = 0.0, 1.0
            for _ in range(n_bisect):
                mid = (lo + hi) * 0.5
                if np.max(d + B * mid) <= np.max(d + B * lo):
                    hi = mid
                else:
                    lo = mid
            x = (lo + hi) * 0.5
            if abs(x - h[i]) > 1e-13:
                corr += B * (x - h[i])
                h[i] = x
                any_change = True
        if not any_change:
            break
    return h, corr.max() * (2.0 / n)

def projected_grad_ls(h, n, alphas, iters_per_alpha=2000, deadline=None):
    """Projected gradient with log-sum-exp objective"""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    m = np.arange(n_corr) - (n - 1)
    h = h.astype(float).copy()
    
    for alpha in alphas:
        for it in range(iters_per_alpha):
            if deadline and time.time() > deadline:
                return h
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            shifted = alpha * c
            shifted -= shifted.max()
            w = np.exp(shifted)
            w /= w.sum()
            
            # Gradient via FFT
            N = 2 * n_corr
            H = np.fft.rfft(h, N)
            W = np.fft.rfft(w, N)
            conv1 = np.fft.irfft(H * np.conj(W), N)[:n]
            w_neg = w[::-1].copy()
            Wn = np.fft.rfft(w_neg, N)
            conv2 = np.fft.irfft(H * np.conj(Wn), N)[:n]
            
            grad = dx * (1.0 - conv1 - conv2)
            
            # Adaptive step
            eta = 1.0 / (alpha * dx + 1.0)
            for _ in range(20):
                h_new = project_feasible(h - eta * grad, n)
                c_new = np.correlate(h_new, 1.0 - h_new, mode='full').max() * dx
                if c_new < c.max() - 1e-15:
                    h = h_new
                    break
                eta *= 0.5
    return h

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-15):

    def obj(x):

        return (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def precompute(n):
    n_corr = 2 * n - 1
    m = np.arange(n_corr) - (n - 1)
    im = np.arange(n)[:, None] + m[None, :]
    im2 = np.arange(n)[:, None] - m[None, :]

    v2 = (im2 >= 0) & (im2 < n)
    s1 = np.clip(im, 0, n - 1)

def coord_desc(h, n, m, v1, v2, s1, s2, sweeps=3000, deadline=None, nb=22, seed=0):
    n_corr = 2 * n - 1
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    rng = np.random.RandomState(seed)
    for sw in range(sweeps):
        if deadline and time.time() > deadline:
            break
        order = rng.permutation(n)
        changed = False
        for i in order:
            B = np.ones(n_corr)
            B[v1[i]] -= h[s1[i][v1[i]]]
            B[v2[i]] -= h[s2[i][v2[i]]]
            d = corr - B * h[i]
            lo, hi = 0.0, 1.0
            for _ in range(nb):
                mid = (lo + hi) * 0.5
                if np.max(d + B * mid) <= np.max(d + B * lo):
                    hi = mid
                else:
                    lo = mid
            x = (lo + hi) * 0.5
            if abs(x - h[i]) > 1e-13:
                corr += B * (x - h[i])
                h[i] = x
                changed = True
        if not changed:
            break
    return h, corr.max() * dx

def proj_grad(h, n, m, v1, v2, s1, s2, alphas, iters=3000, deadline=None):
    n_corr = 2 * n - 1
    dx = 2.0 / n
    h = h.astype(float).copy()
    for alpha in alphas:
        for it in range(iters):
            if deadline and time.time() > deadline:
                return h
            c = np.correlate(h, 1 - h, mode='full') * dx
            sh = alpha * c - alpha * c.max()
            w = np.exp(sh)
            w /= w.sum()
            # grad[i] = dx * (1 - sum_k w[k]h[i+k] - sum_k w[k]h[i-k])
            H1 = h[s1] * v1
            s1v = H1 @ w
            H2 = h[s2] * v2
            s2v = H2 @ w
            grad = dx * (1.0 - s1v - s2v)
            eta = 1.0 / (alpha * dx)
            for _ in range(25):

                cn = np.correlate(hn, 1 - hn, mode='full').max() * dx
                if cn < c.max() - 1e-15:
                    h = hn
                    break
                eta *= 0.5
    return h

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        sh = alpha * (c - c.max())
        return (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha
    def g(x):

        sh = alpha * c - alpha * c.max()

        w /= w.sum()

                if 0 <= i - k < nn:
                    s -= w[kk] * x[i - k]

        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h, method='SLSQP', bounds=bounds,

    return project_feasible(np.clip(res.x, 0, 1), n)
