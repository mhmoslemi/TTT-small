        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-10
            lo_v = min(val0, val1) - 1e-10
            pos_B = B > 1e-13
            neg_B = B < -1e-13
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(45):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-13:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-11:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved and sweep > 5:
            break
    return h, corr

def slsqp_solve(h, n, alpha, maxiter=5000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def gradient_step(h, n, alpha, steps=20000, deadline=None):
    """Projected gradient with softmax weighting."""
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 1.0 / n
    
    for s in range(steps):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1-h, mode='full')
        c_vals = corr * dx
        shifted = alpha * (c_vals - c_vals.max())
        p = np.exp(shifted)
        p /= p.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            pk = p[k]
            if pk < 1e-12:
                continue
            sh = k - (n-1)
            if sh >= 0:
                grad[:n-sh] += pk * dx * (1.0 - h[sh:n])
                if sh > 0:
                    grad[sh:] += -pk * dx * h[:n-sh]
            else:
                s2 = -sh
                grad[s2:] += pk * dx * (1.0 - h[:n-s2])
                grad[:n-s2] += -pk * dx * h[s2:]
        
        # Step with projection
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.99995, 1e-5)
    
    return best_h, best_c
