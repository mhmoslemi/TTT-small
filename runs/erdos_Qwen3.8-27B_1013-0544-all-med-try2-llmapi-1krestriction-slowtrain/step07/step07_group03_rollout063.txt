```python
def local_search_1swap(h, n, deadline, max_sweeps=500, rng=None):

    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Randomize order to avoid cycling
        if rng is not None:
            rng.shuffle(zeros_list)
            rng.shuffle(ones_list)
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                d = swap_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def pnorm_gradient(h, n, p):
    """Compute gradient of (sum_k corr_k^p)^(1/p) with respect to h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    S = np.sum(corr**p)
    if S < 1e-30:
        return np.zeros(n)
    scale = (1.0/p) * S**(1.0/p - 1.0)
    w = corr**(p-1)  # Will multiply by scale later
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        wk = w[k] * scale
        if abs(wk) < 1e-30:
            continue
        # First factor: grad[i] += wk * (1-h[i-lag]) for valid i-lag in [0,n)
        i_start = max(0, lag)
        i_end = min(n, n + lag)
        if i_start < i_end:
            m_idx = np.arange(i_start, i_end) - lag
            grad[i_start:i_end] += wk * (1.0 - h[m_idx])
        # Second factor: grad[i] += -wk * h[i+lag] for valid i+lag in [0,n)
        i_start2 = max(0, -lag)
        i_end2 = min(n, n - lag)
        if i_start2 < i_end2:
            m_idx2 = np.arange(i_start2, i_end2) + lag
            grad[i_start2:i_end2] -= wk * h[m_idx2]
    return grad

def projected_gradient_descent(h, n, deadline, p=100, lr=0.01, momentum=0.9, max_iter=50000):
    """Projected gradient descent with momentum on p-norm objective"""
    h = h.copy()
    target = n / 2.0
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = pnorm_gradient(h, n, p)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum constraint
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def anneal_2swap(h, n, deadline, seed=42, n_iters=1000000):
    """Simulated annealing with 2-swap moves"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.9999
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if len(ones_arr) < 2 or len(zeros_arr) < 2:
            break
        # Pick 2 random ones and 2 random zeros
        a1, b1 = rng.randint(len(zeros_arr)), rng.randint(len(ones_arr))
        a2, b2 = rng.randint(len(zeros_arr)), rng.randint(len(ones_arr))
        if a1 == a2 or b1 == b2:
            continue
        i1, i2 = zeros_arr[a1], zeros_arr[a2]
        j1, j2 = ones_arr[b1], ones_arr[b2]
        d = double_swap_delta(h, n, i1, j1, i2, j2)
        nc = (corr + d).max() * dx
        accept = nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(5000*(cur_c-nc)/T))
        if accept:
            h[i1], h[j1] = 1.0, 0.0
            h[i2], h[j2] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros_arr[a1], zeros_arr[a2] = j1, j2
            ones_arr[b1], ones_arr[b2] = i1, i2
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
```

Wait, there's an issue with the projected gradient descent. The p-norm objective might not have a good gradient at the boundary (when some corr_k = 0 and p is large, corr_k^(p-1) = 0). Also, the gradient computation might be slow for large p because corr_k^(p-1) can underflow or overflow.

Let me use a log-sum-exp style approach instead, which is more numerically stable:
f_alpha(h) = (1/alpha) * log(sum_k exp(alpha * corr_k))

This is a smooth approximation to max_k corr_k, and the gradient is:
∂f_alpha/∂h[i] = (1/alpha) * sum_k alpha * exp(alpha*corr_k) / sum_j exp(alpha*corr_j) * ∂corr_k/∂h[i]
= sum_k (exp(alpha*corr_k) / Z) * ∂corr_k/∂h[i]
= sum_k p_k * ∂corr_k/∂h[i]

where p_k = exp(alpha*corr_k) / Z is a probability distribution (softmax).

This is much more numerically stable! The gradient is a weighted average of the individual correlation gradients, weighted by the softmax probabilities.

Let me rewrite:

```python
def softmax_gradient(h, n, alpha):
    """Compute gradient of (1/alpha)*log(sum exp(alpha*corr_k))"""
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    Z = exp_c.sum()
    p = exp_c / Z  # softmax weights
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-15:
            continue
        lag = k - (n-1)
        pk = p[k]
        # First factor
        i_start = max(0, lag)
        i_end = min(n, n + lag)
        if i_start < i_end:
            m_idx = np.arange(i_start, i_end) - lag
            grad[i_start:i_end] += pk * (1.0 - h[m_idx])
        # Second factor
        i_start2 = max(0, -lag)
        i_end2 = min(n, n - lag)
        if i_start2 < i_end2:
            m_idx2 = np.arange(i_start2, i_end2) + lag
            grad[i_start2:i_end2] -= pk * h[m_idx2]
    return grad
```

This is O(n²) per gradient evaluation (n lags, each with O(n) vector work). For n=400, that's 160K operations per gradient. We can do ~10K iterations in a reasonable time.

The learning rate needs to be tuned. Since the gradient is a convex combination of ±1 type values, its magnitude is at most 1 (actually at most the number of contributing terms, but the softmax weights sum to 1, so the gradient magnitude is bounded by max(|1-h|, |h|) = 1).

So lr should be in the range [0.001, 0.1] or so.

Let me also think about the projected gradient step. After computing h_new = h - lr * grad, we need to project to [0,1]^n ∩ {sum = n/2}. The project_feasible function does this, but it might not be the optimal projection.

A better projection for this constraint set (the hypersimplex) is:
1. Clip to [0,1]
2. Find the unique lambda such that sum(clip(h - lambda, 0, 1)) = n/2

This can be found by bisection on lambda.

Let me implement this:

```python
def project_hypersimplex(h, n):
    """Project onto {x : 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to decrease: subtract from entries not at 0
            mask = h > 0
            if not mask.any():
                break
            # Proportional decrease
            scale = 1.0 - diff / h[mask].sum()
            h[mask] *= scale
        else:
            # Need to increase: add to entries not at 1
            mask = h < 1
            if not mask.any():
                break
            room = (1.0 - h[mask]).sum()
            h[mask] += (-diff) / room * (1.0 - h[mask])
    return np.clip(h, 0.0, 1.0)
```

Actually, the exact projection onto the hypersimplex {x: 0≤x_i≤1, sum(x)=r} is:
x_i = clip(v_i - lambda, 0, 1) where lambda is chosen so sum(x) = r.

This can be found by sorting v and finding the right threshold. Let me implement it properly:

```python
def project_hypersimplex(v, r):
    """Project v onto {x: 0<=x_i<=1, sum(x)=r}
    The projection is x_i = min(max(v_i - lambda, 0), 1) for the right lambda."""
    n = len(v)
    # Sort v in decreasing order
    v_sorted = np.sort(v)[::-1]
    # Find the range of lambda
    # If sum(clip(v_i - lambda, 0, 1)) = r
    # For lambda between v_sorted[k] and v_sorted[k+1]:
    #   sum = sum_{i: v_i > lambda} (v_i - lambda) + 0 * (rest clamped to 0)
    #         + 1 * (count of v_i - lambda > 1, i.e., v_i > lambda + 1)
    # This is complex. Let me use bisection.
    
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > r:
            lo = mid  # Need to increase lambda to decrease sum
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2, 0, 1)
    return x
```

This is O(n log(n)) per bisection step (if we use the sorted array) or O(n) per step with the simple approach. For 100 bisection steps and n=400, that's 40K operations, which is fast.

1. Starting from initial_h_values
2. Binary local search (1-swap, first improvement with randomization)
3. Softmax-gradient descent (smooth optimization)
4. 2-swap annealing
5. Resolution adaptation
6. Final polish

Let me also think about what makes this problem special. The function h represents a "set" (approximately) of measure 1 in [0,2]. The overlap integral measures how much h(x) and h(x+k) "agree" (one is 1 and the other is... wait, it's h(x)*(1-h(x+k)), so it measures where h is 1 at x but 0 at x+k).

The maximum over k is what we want to minimize. So we want a "well-distributed" set where no shift k has too much "1 followed by 0" pattern.

This is related to the concept of "difference sets" in combinatorics. A (v, k, lambda) difference set has the property that every non-zero difference occurs exactly lambda times. In our continuous setting, we want the "difference" to be as uniform as possible.

The theoretical minimum for a (0,1)-valued function with measure 1/2 (since ∫h = 1 on [0,2] means the set has measure 1, which is half the domain) would have corr[k] ≈ 1/4 for all k (by the "equidistribution" heuristic), giving C5 ≈ 1/4 * 2 = 0.5. Wait, that's not right.

Let me recalculate. If h is (0,1)-valued with ∫h = 1 on [0,2], then the "density" of ones is 1/2 (measure 1 out of total 2). If the set is "random", then for a typical shift k, the expected overlap ∫h(x)(1-h(x+k))dx would be:
- P(h(x)=1 and h(x+k)=0) = P(h(x)=1) * P(h(x+k)=0) = (1/2)(1/2) = 1/4
- But this is the pointwise probability; the integral over x gives (1/4) * 2 = 1/2.

Wait, that doesn't seem right either. Let me think more carefully.

If h is a random {0,1}-valued function with E[h(x)] = 1/2 (in the appropriate sense), then:
E[∫ h(x)(1-h(x+k)) dx] = ∫ E[h(x)(1-h(x+k))] dx

For a "random" h with independent values, E[h(x)(1-h(x+k))] = E[h(x)] * E[1-h(x+k)] = (1/2)(1/2) = 1/4.
So the expected integral is (1/4) * 2 = 1/2.

But we're looking at the MAX over k, not the average. The average of corr*k dx over all k would be:
(1/(2n-1)) * sum_k corr[k] * dx = (dx/(2n-1)) * sum_k corr[k]

And sum_k corr[k] = sum_k sum_x h[x]*(1-h[x+k]) = ... this is getting complicated.

The point is that the "random" baseline gives C5 ≈ 0.5, and we're trying to get below that. The current best is 0.38, which is significantly below random.

This suggests the optimal solution has significant structure - it's not random at all. The structure likely involves some kind of quasi-periodic or arithmetic progression pattern.

For my algorithm, the key question is: what's the best way to find this structure?

I think the most promising approach is:
1. Use the softmax gradient to find a good continuous solution
2. Round to binary
3. Use local search to refine

The softmax gradient with large alpha essentially focuses on the "worst" lags (those with maximum correlation) and tries to reduce them. This is like a "bottleneck" optimization.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_hypersimplex(v, r):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > r:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0
    k1 = n-1 + (j-i)
    if 0 <= k1 < 2*n-1:
        delta[k1] += 1.0
    k2 = n-1 + (i-j)
    if 0 <= k2 < 2*n-1:
        delta[k2] += 1.0
    return delta

def softmax_grad(h, n, alpha):
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    mc = corr.max()
    exp_c = np.exp(alpha * (corr - mc))
    Z = exp_c.sum()
    if Z < 1e-300:
        return np.zeros(n), mc
    p = exp_c / Z
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n-1)
        i_s = max(0, lag)
        i_e = min(n, n + lag)
        if i_s < i_e:
            gi = np.arange(i_s, i_e) - lag
            grad[i_s:i_e] += pk * (1.0 - h[gi])
        i_s2 = max(0, -lag)
        i_e2 = min(n, n - lag)
        if i_s2 < i_e2:
            gi2 = np.arange(i_s2, i_e2) + lag
            grad[i_s2:i_e2] -= pk * h[gi2]
    return grad, mc

def pgd_softmax(h, n, deadline, alpha=100, lr=0.01, momentum=0.9, max_iter=20000):
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g, _ = softmax_grad(h, n, alpha)
        v = momentum * v - lr * g
        h_new = project_hypersimplex(h + v, n/2.0)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def local_search(h, n, deadline, max_sweeps=300, rng=None):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        if rng:
            rng.shuffle(zeros)
            rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def anneal(h, n, deadline, seed=42, n_iters=500000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.99995
    for it in range(n_iters):

        a = rng.randint(len(zeros))
        b = rng.randint(len(ones))

        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(3000*(cur_c-nc)/T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros[a] = j
            ones[b] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_hypersimplex(v, r):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > r:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def softmax_grad(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    mc = corr.max()
    exp_c = np.exp(alpha * (corr - mc))
    Z = exp_c.sum()
    if Z < 1e-300:
        return np.zeros(n)
    p = exp_c / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        i_s = max(0, lag)
        i_e = min(n, n + lag)
        if i_s < i_e:
            gi = np.arange(i_s, i_e) - lag
            grad[i_s:i_e] += pk * (1.0 - h[gi])
        i_s2 = max(0, -lag)
        i_e2 = min(n, n - lag)
        if i_s2 < i_e2:
            gi2 = np.arange(i_s2, i_e2) + lag
            grad[i_s2:i_e2] -= pk * h[gi2]
    return grad

def pgd_softmax(h, n, deadline, alpha=200, lr=0.008, beta=0.9, max_iter=15000):
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = softmax_grad(h, n, alpha)
        t += 1
        tt = (1 + np.sqrt(1 + 4 * t * t)) / 2
        v_new = ((t - 1) / tt) * v
        h_new = h + (v_new / tt - lr * g)
        h_new = project_hypersimplex(h_new, n / 2.0)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        v = v_new
        h = h_new
    return best_h, best_c

def local_search(h, n, deadline, max_sweeps=400, rng=None):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        if rng:
            rng.shuffle(zeros)
            rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def anneal_2swap(h, n, deadline, seed=42, n_iters=800000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = 0.012
    cool = 0.99993
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        a1 = rng.randint(len(zeros))
        b1 = rng.randint(len(ones))
        a2 = rng.randint(len(zeros))
        b2 = rng.randint(len(ones))
        if a1 == a2 or b1 == b2:
            continue
        i1, i2 = zeros[a1], zeros[a2]
        j1, j2 = ones[b1], ones[b2]
        d = swap_delta(h, n, i1, j1)
        h2 = h.copy()
        h2[i1], h2[j1] = 1.0, 0.0
        d += swap_delta(h2, n, i2, j2)
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(4000 * (cur_c - nc) / T)):
            h[i1], h[j1] = 1.0, 0.0
            h[i2], h[j2] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros[a1], zeros[a2] = j1, j2
            ones[b1], ones[b2] = i1, i2
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
```
