import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def proj(h, n):
    t = n / 2.0
    h = np.clip(h, 0, 1).copy()
    for _ in range(2000):
        d = t - h.sum()
        if abs(d) < 1e-15: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h
    return np.clip(h, 0, 1)

def _gk(h, n, k):
    """Gradient of shift-k correlation, includes dx factor."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[k:] += 1.0 - h[:nv]
        g[:nv] -= h[k:]
    else:
        a = -k; nv = n - a
        g[:nv] += 1.0 - h[a:]
        g[a:] -= h[:nv]
    return g * dx

def fobj(h, n, alpha, mu):
    c = c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m); mx = s.max()
    v = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    ss = h.sum() - n / 2.0
    return float(v + mu * ss * ss)

def fgrad(h, n, alpha, mu):
    c = c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m); mx = s.max()
    w = np.exp(s - mx); w /= w.sum()
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] > 1e-15:
            g += w[i] * _gk(h, n, i - n + 1)
    g += 2 * mu * (h.sum() - n / 2.0)
    return g

def fobj_f(h, n, M, alpha, mu):
    c = c5_vec(h, n)
    if M < len(c):
        idx = np.argpartition(-c, M-1)[:M]; cf = c[idx]
    else:
        cf = c
    m = cf.max()
    s = alpha * (cf - m); mx = s.max()
    v = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    ss = h.sum() - n / 2.0
    return float(v + mu * ss * ss)

def fgrad_f(h, n, M, alpha, mu):
    c = c5_vec(h, n)
    if M < len(c):
        idx = np.argpartition(-c, M-1)[:M]
    else:
        idx = np.arange(len(c))
    cf = c[idx]

    s = alpha * (cf - m); mx = s.max()
    w = np.exp(s - mx); w /= w.sum()

    g += 2 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, alpha_schedule, M=None, mu=100.0, maxiter=200):
    best_c = compute_c5(h, n)
    bounds = [(0,1)] * n
    for alpha in alpha_schedule:
        if time.time() > deadline - 3: break
        remaining = deadline - time.time() - 3
        if remaining < 5: break
        if M is not None:
            res = minimize(fobj_f, h, jac=fgrad_f, args=(n, M, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
        else:
            res = minimize(fobj, h, jac=fgrad, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
        h = proj(res.x, n)
        c = compute_c5(h, n)
        if c < best_c: best_c = c
    return h, best_c

        return h.copy(), compute_c5(h, n)

def local_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    c = c5_vec(h, n)
    M = min(20, len(c))
    hot = np.argpartition(-c, M - 1)[:M]
    hot_k = [int(i) - n + 1 for i in hot]
    for d in [0.003, 0.001, 0.0005, 0.0002]:
        imp = True
        while imp and time.time() < deadline:
            imp = False
            for k in hot_k:
                if time.time() > deadline:
                    return h, best_c
                if k > 0:
                    nv = n - k
                    for i in range(0, nv, 4):
                        if h[i + k] > h[i] + 0.01:
                            h[i + k] -= d
                            h[i] += d
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-14:
                                best_c = cn
                                imp = True
                            else:
                                h[i + k] += d
                                h[i] -= d
                elif k < 0:
                    a = -k
                    nv = n - a
                    for i in range(0, nv, 4):
                        if h[i] > h[i + a] + 0.01:
                            h[i] -= d
                            h[i + a] += d
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-14:
                                best_c = cn
                                imp = True
                            else:
                                h[i] += d
                                h[i + a] -= d
    return h, best_c

def thue_morse_h(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return proj(h, n)

def interp_h(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return proj(np.interp(xn, xo, h), n_new)
