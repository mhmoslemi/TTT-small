    dx = 2.0 / n

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    """Gradient of smooth max of correlation values."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        s = 1.0
    w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

            best_h = h.copy()

def local_search_hill(h, n, deadline, alpha=10000):
    """Steepest descent local search: always take the best single-pair move."""

    best_c = compute_c5(h, n)

        g = grad_c5(h, n, alpha)
        # Find the best direction: increase where gradient is most negative, decrease where most positive
        i_add = np.argmin(g)  # most negative gradient -> adding here helps most
        i_sub = np.argmax(g)  # most positive gradient -> subtracting here helps most
        
        if i_add == i_sub:
            break
        if g[i_add] >= g[i_sub]:
            break
            
        # Try moving mass from i_sub to i_add
        # Binary search for the best amount to move
        lo, hi = 0.0, min(h[i_sub], 1.0 - h[i_add])

            else:
                h[i_add] -= mid
                h[i_sub] += mid
                hi = mid

            break

    for idx in range(len(w)):

        if k >= 0:

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

        g = grad_c5(h, n, alpha)

        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                        if c < best_c - 1e-15:

                        else:

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
