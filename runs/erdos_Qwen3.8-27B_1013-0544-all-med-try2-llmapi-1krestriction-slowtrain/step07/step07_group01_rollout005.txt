```
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
```

```
def grad_lag_weights(h,n,w):
    grad=np.zeros(n)
    dx=2.0/float(n)
    center=n-1
    # w may be list
    for m in range(2*n-1):
       wm=float(w[m])
       if wm <= 1e-16: continue
       wm *= dx
       d=m-center
       if d >= 0:
          nv=n-d
          if nv > 0:
             grad[d:n] += wm * (1.0 - h[0:nv])
             grad[0:nv] -= wm * h[d:n]
       else:
          r=-d
          nv=n-r
          if nv > 0:
             grad[0:nv] += wm * (1.0 - h[r:n])
             grad[r:n] -= wm * h[0:nv]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_CB_DEADLINE = 0.0

def _cb(x):
    return time.time() > _CB_DEADLINE

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    for _ in range(n + 5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            add = min(1.0 - h[idx], diff)
            if add <= 0:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(h[idx], -diff)
            if sub <= 0:
                break
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)

def compute_corr_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))

def eval_smooth(h, n, alpha):
    c = compute_corr_c(h, n)
    m = float(c.max())
    if not np.isfinite(m):
        return m, None, c, m
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    val = float(m + np.log(s) / alpha)
    w = e / s
    return val, w, c, m

def grad_lag_weights(h, n, w):
    g = np.zeros(n)
    dx = 2.0 / float(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = float(w[m])
        if wm <= 1e-16:
            continue
        wm *= dx
        d = m - center
        if d >= 0:
            nv = n - d
            if nv > 0:
                g[d:n] += wm * (1.0 - h[0:nv])
                g[0:nv] -= wm * h[d:n]
        else:
            r = -d
            nv = n - r
            if nv > 0:
                g[0:nv] += wm * (1.0 - h[r:n])
                g[r:n] -= wm * h[0:nv]
    return g

def smooth_grad(h, n, alpha):
    val, w, c, m = eval_smooth(h, n, alpha)
    if w is None or not np.all(np.isfinite(w)):
        return val, np.zeros(n)
    g = grad_lag_weights(h, n, w)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return float(val), g

def _lbfgs_obj(x, n, alpha, mu):
    val, w, c, m = eval_smooth(x, n, alpha)
    if not np.isfinite(val):
        return 1e100
    diff = x.sum() - 0.5 * n
    return float(val + mu * diff * diff)

def _lbfgs_grad(x, n, alpha, mu):
    val, g = smooth_grad(x, n, alpha)
    if not np.all(np.isfinite(g)):
        g = np.zeros(len(x))
    diff = x.sum() - 0.5 * n
    return g + 2.0 * mu * diff

def _slsqp_obj(x, n, alpha):
    val, w, c, m = eval_smooth(x, n, alpha)
    return 1e100 if not np.isfinite(val) else float(val)

def _slsqp_grad(x, n, alpha):
    val, g = smooth_grad(x, n, alpha)
    return g if np.all(np.isfinite(g)) else np.zeros(len(x))

def _slsqp_cons(x, n):
    return float(x.sum() - 0.5 * n)

def _slsqp_jac(x, n):
    return np.ones(len(x))

def build_basis(n, m):
    i = np.arange(n, dtype=float)
    b = np.zeros((n, 2 * m))
    for k in range(1, m + 1):
        b[:, 2 * k - 1] = np.cos(2.0 * np.pi * k * i / n)
        b[:, 2 * k] = np.sin(2.0 * np.pi * k * i / n)
    return b

def _basis_obj(p, b, n, alpha, mu_s, mu_b):
    h = 0.5 + b.dot(p)
    if not np.all(np.isfinite(h)):
        return 1e100
    val, w, c, m = eval_smooth(h, n, alpha)
    if not np.isfinite(val):
        return 1e100
    diff = h.sum() - 0.5 * n
    pen = mu_s * diff * diff
    low = h < 0.0
    if np.any(low):
        pen += mu_b * np.sum((-h[low]) ** 2)
    high = h > 1.0
    if np.any(high):
        pen += mu_b * np.sum((h[high] - 1.0) ** 2)
    return float(val + pen)

def _basis_grad(p, b, n, alpha, mu_s, mu_b):
    h = 0.5 + b.dot(p)
    if not np.all(np.isfinite(h)):
        return np.zeros(b.shape[1])
    val, g = smooth_grad(h, n, alpha)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    diff = h.sum() - 0.5 * n
    g = g + 2.0 * mu_s * diff
    low = h < 0.0
    if np.any(low):
        g[low] -= 2.0 * mu_b * h[low]
    high = h > 1.0
    if np.any(high):
        g[high] += 2.0 * mu_b * (h[high] - 1.0)
    out = b.T.dot(g)
    if not np.all(np.isfinite(out)):
        out = np.zeros(b.shape[1])
    return out

def optimize_basis(n, deadline, seed=0, m=5, b=None):
    global _CB_DEADLINE
    if b is None:
        b = build_basis(n, m)
    dim = b.shape[1]
    rng = np.random.RandomState(seed + 123)
    starts = []
    p0 = np.zeros(dim)
    p0[0] = 0.35
    starts.append(p0)
    p1 = np.zeros(dim)
    p1[0] = -0.30
    p1[1] = 0.18
    starts.append(p1)
    starts.append(rng.normal(0.0, 0.12, dim))
    best_h = None
    best_c = np.inf
    bounds = [(-0.5, 0.5)] * dim
    for p in starts:
        if time.time() > deadline:
            break
        _CB_DEADLINE = deadline
        try:
            res = minimize(
                _basis_obj, p, args=(b, n, 500.0, 1e4, 1e4),
                jac=_basis_grad, method="L-BFGS-B", bounds=bounds,
                callback=_cb,
                options={"maxiter": 700, "ftol": 1e-14, "gtol": 1e-9,
                         "maxfun": 3500, "maxls": 40}
            )
            x = res.x
        except Exception:
            x = p
        h = project_feasible(0.5 + b.dot(x), n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if best_h is None:
        best_h = project_feasible(0.5 + b.dot(starts[0]), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c

def make_cosine(n, amp, phase):
    i = np.arange(n, dtype=float)
    h = 0.5 + amp * np.cos(np.pi * (2.0 * i / n) + phase)
    return project_feasible(h, n)

def add_candidate(cands, h, n):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    cands.append((c, h, n))
    return cands

def _cand_key(item):
    return item[0]

def lbfgs_improve(h, n, deadline, quick=False):
    global _CB_DEADLINE
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if quick:
        alphas = [50.0, 500.0, 5000.0, 50000.0]
        maxiter = 120
    else:
        alphas = [25.0, 100.0, 500.0, 2500.0, 12000.0, 60000.0,
                  250000.0, 800000.0]
        maxiter = 220
    for alpha in alphas:
        if time.time() > deadline:
            break
        _CB_DEADLINE = deadline
        start = best_h.copy()
        try:
            res = minimize(
                _lbfgs_obj, start, args=(n, alpha, 1000.0),
                jac=_lbfgs_grad, method="L-BFGS-B", bounds=bounds,
                callback=_cb,
                options={"maxiter": maxiter, "ftol": 1e-16, "gtol": 1e-12,
                         "maxfun": maxiter * 10, "maxls": 50}
            )
            x = res.x
        except Exception:
            x = start
        h2 = project_feasible(x, n)
        c2 = compute_c5(h2, n)
        if c2 + 1e-14 < best_c:
            best_c = c2
            best_h = h2.copy()
    return best_h, best_c

def slsqp_improve(h, n, deadline, alphas=(5000.0, 20000.0, 100000.0), maxiter=260):
    global _CB_DEADLINE
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _slsqp_cons, "jac": _slsqp_jac, "args": (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        _CB_DEADLINE = deadline
        try:
            res = minimize(
                _slsqp_obj, best_h, args=(n, alpha),
                jac=_slsqp_grad, method="SLSQP", bounds=bounds,
                constraints=cons, callback=_cb,
                options={"maxiter": maxiter, "ftol": 1e-13, "maxls": 25}
            )
            x = res.x
        except Exception:
            x = best_h
        h2 = project_feasible(x, n)
        c2 = compute_c5(h2, n)
        if c2 + 1e-14 < best_c:
            best_c = c2
            best_h = h2.copy()
    return best_h, best_c

def _count_range(pref, lo, hi, length):
    if lo > hi:
        return 0
    if lo < 0:
        lo = 0
    if hi >= length:
        hi = length - 1
    if lo > hi:
        return 0
    return int(pref[hi] - (pref[lo - 1] if lo > 0 else 0))

def pair_search(h, n, deadline, deltas=(0.001, 0.0005, 0.0002, 0.0001)):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    passes = 3
    for d in deltas:
        if time.time() > deadline or d < 1e-7:
            break
        for _ in range(passes):
            if time.time() > deadline:
                break
            c = compute_corr_c(h, n)
            mnow = float(c.max())
            if mnow < best_c:
                best_c = mnow
            active = c >= mnow - 1e-10
            tot = int(active.sum())
            pref = np.cumsum(active)
            length = 2 * n - 1
            improved = False
            for i in range(n):
                if improved or time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        break
                    if tot > 0:
                        a1 = n - 1 - i
                        b1 = n - 1 + i
                        a2 = n - 1 - j
                        b2 = n - 1 + j
                        inside = (_count_range(pref, a1, b1, length)
                                  + _count_range(pref, a2, b2, length)
                                  - _count_range(pref, max(a1, a2), min(b1, b2), length))
                        if inside < tot:
                            continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        m2 = compute_c5(h, n)
                        if m2 + 1e-13 < best_c:
                            best_c = m2
                            improved = True
                            break
                        h[i] -= d
                        h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        m2 = compute_c5(h, n)
                        if m2 + 1e-13 < best_c:
                            best_c = m2
                            improved = True
                            break
                        h[i] += d
                        h[j] -= d
            if not improved:
                break
    h2 = project_feasible(h, n)
    return h2, compute_c5(h2, n)

def optimize_direct(h, n, deadline, quick=False):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 12:
        return best_h, best_c
    if quick:
        lb_frac, sl_frac = 0.72, 0.18
        tuple_alphas = (20000.0, 100000.0)
        pair_deltas = (0.0005, 0.0002, 0.0001)
    else:
        lb_frac, sl_frac = 0.68, 0.18
        tuple_alphas = (5000.0, 20000.0, 100000.0)
        pair_deltas = (0.001, 0.0005, 0.0002, 0.0001)
    t_lb = time.time() + max(20.0, rem * lb_frac)
    t_lb = min(t_lb, deadline - max(25.0, rem * 0.18))
    if t_lb > time.time() + 5:
        h2, c2 = lbfgs_improve(h, n, t_lb, quick=(quick or n > 600 or rem < 100))
        if c2 + 1e-14 < best_c:
            best_c = c2
            best_h = h2.copy()
    t_sl = time.time() + max(15.0, rem * sl_frac)
    t_sl = min(t_sl, deadline - 8.0)
    if t_sl > time.time() + 5:
        h2, c2 = slsqp_improve(best_h, n, t_sl, alphas=tuple_alphas,
                                maxiter=(140 if quick else 260))
        if c2 + 1e-14 < best_c:
            best_c = c2
            best_h = h2.copy()
    if time.time() < deadline - 2:
        h2, c2 = pair_search(best_h, n, deadline, deltas=pair_deltas)
        if c2 + 1e-14 < best_c:
            best_c = c2
            best_h = h2.copy()
    final_h = project_feasible(best_h, n)
    return final_h, compute_c5(final_h, n)

def final_polish(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t1 = min(deadline - 60.0, time.time() + 70.0)
    if t1 > time.time() + 5:
        h2, c2 = lbfgs_improve(h, n, t1, quick=True)
        if c2 + 1e-14 < best_c:
            best_h, best_c = h2, c2
    t2 = min(deadline - 25.0, time.time() + 55.0)
    if t2 > time.time() + 5:
        h2, c2 = slsqp_improve(best_h, n, t2, alphas=(20000.0, 100000.0), maxiter=240)
        if c2 + 1e-14 < best_c:
            best_h, best_c = h2, c2
    if time.time() < deadline - 2:
        h2, c2 = pair_search(best_h, n, deadline,
                             deltas=(0.0005, 0.0002, 0.0001, 5e-5))
        if c2 + 1e-14 < best_c:
            best_h, best_c = h2, c2
    final_h = project_feasible(best_h, n)
    return final_h, compute_c5(final_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s) - 8.0)
    rng = np.random.RandomState(seed)

    parent = None
    n0 = 300
    try:
        hv = np.asarray(initial_h_values, dtype=float).copy()
        if hv.ndim != 1:
            hv = hv.reshape(-1)
        if len(hv) >= 2 and np.all(np.isfinite(hv)):
            n0 = len(hv)
            parent = project_feasible(hv, n0)
    except Exception:
        parent = None

    cands = []
    if parent is None:
        parent = make_cosine(300, 0.35, 0.0)
        n0 = len(parent)
    add_candidate(cands, parent, n0)

    m_basis = 4 if n0 > 700 else 5
    b = build_basis(n0, m_basis)
    dim = b.shape[1]

    if time.time() < deadline - 110:
        tb = min(deadline - 210.0, time.time() + 85.0)
        try:
            hb, cb = optimize_basis(n0, tb, seed=seed, m=m_basis, b=b)
            add_candidate(cands, hb, n0)
        except Exception:
            pass
        for _ in range(6):
            if time.time() > deadline - 100:
                break
            p = rng.normal(0.0, 0.15, dim)
            add_candidate(cands, 0.5 + b.dot(p), n0)

    if time.time() < deadline - 80:
        amps = [0.26, 0.31, 0.36, 0.41, 0.46, 0.49]
        phases = np.linspace(0.0, 2.0 * np.pi, 15, endpoint=False)
        ai = 0
        while ai < len(amps):
            a = amps[ai]
            pi = 0
            while pi < len(phases):
                if time.time() > deadline - 80:
                    break
                add_candidate(cands, make_cosine(n0, a, float(phases[pi])), n0)
                pi += 1
            ai += 1

    if cands:
        h0 = cands[0][1]
        add_candidate(cands, 0.5 * (h0 + 1.0 - h0[::-1]), n0)
        add_candidate(cands, 0.5 * (h0 + h0[::-1]), n0)

    cands.sort(key=_cand_key)
    top = cands[:8]
    best_h = top[0][1].copy()
    best_c = top[0][0]

    reserve = 150.0
    avail = deadline - time.time() - reserve
    weights = [0.50, 0.30, 0.20]
    for idx in range(min(3, len(top))):
        if time.time() > deadline - reserve:
            break
        alloc = max(25.0, avail * weights[idx])
        t_end = min(deadline - reserve, time.time() + alloc)
        if t_end <= time.time():
            break
        hi = top[idx][1].copy()
        ho, co = optimize_direct(hi, n0, t_end,
                                 quick=(n0 > 650 or alloc < 90))
        if co + 1e-14 < best_c:
            best_c = co
            best_h = ho.copy()

    if time.time() < deadline - 10:
        fh, fc = final_polish(best_h, n0, deadline)
        if fc + 1e-14 < best_c:
            best_h, best_c = fh, fc

    final_h = project_feasible(best_h, n0)
    final_c = compute_c5(final_h, n0)
    if final_c + 1e-14 < best_c:
        best_h, best_c = final_h, final_c
    else:
        final_h = project_feasible(best_h, n0)
        final_c = compute_c5(final_h, n0)
    return final_h, best_c if best_c < final_c else final_c, len(final_h)
```
