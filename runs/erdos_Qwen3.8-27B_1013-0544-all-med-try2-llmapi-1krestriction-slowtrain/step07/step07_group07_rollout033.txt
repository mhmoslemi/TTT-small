import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def compute_all_lags(h, n):
    return np.correlate(h, 1-h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target / tot), 0.0, 1.0)
    return h

def _sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def _sum_jacobian(x, n):
    return np.ones_like(x)

def _smoobj(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = alpha * (c - m)
    s2 = s - s.max()
    val = (m + s.max() / alpha + np.log(np.sum(np.exp(s2))) / alpha) * (2.0 / n)
    return val

def _smograd(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = alpha * (c - m)
    s2 = s - s.max()
    w = np.exp(s2)
    w /= w.sum()
    grad = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-16:
        grad += w0 * (1.0 - 2.0 * x)
    for lag in range(1, n):
        wl = w[n-1+lag] + w[n-1-lag]
        if wl < 1e-16:
            continue
        grad[lag:] += wl * (1.0 - x[:-lag])
        grad[:-lag] += -wl * x[lag:]
    return grad * (2.0 / n)

def _mean_active_obj(x, n, active_idx):
    c = compute_all_lags(x, n)
    return float(np.mean(c[active_idx]) * (2.0 / n))

def _mean_active_grad(x, n, active_idx):
    grad = np.zeros(n)
    for idx in active_idx:
        l = idx - (n - 1)
        if l == 0:
            grad += (1.0 - 2.0 * x)
        else:
            m = abs(l)
            grad[m:] += (1.0 - x[:-m])
            grad[:-m] += -x[m:]
    return grad * (2.0 / n) / len(active_idx)

def slsqp(h, n, alpha, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(_smoobj, h, args=(n, alpha), jac=_smograd,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def slsqp_active(h, n, active_idx, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(_mean_active_obj, h, args=(n, active_idx),
                       jac=_mean_active_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for ss in step_sizes:
        for _ in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                # Try to make h more bimodal
                break
            m = abs(lag)
            improved = False
            for i in range(0, n - m, max(1, (n-m)//50)):
                if time.time() > deadline:
                    return h, best_c
                j = i + m

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target / tot), 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones_like(x)

def _smooth_obj(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = alpha * (c - m)
    s2 = s - s.max()
    return (m + s.max() / alpha + np.log(np.sum(np.exp(s2))) / alpha) * (2.0 / n)

def _smooth_grad(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = alpha * (c - m)
    s2 = s - s.max()
    w = np.exp(s2)
    w /= w.sum()
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-16:
        grad += w0 * (1.0 - 2.0 * x)
    for lag in range(1, n):
        wl = w[n - 1 + lag] + w[n - 1 - lag]
        if wl < 1e-16:
            continue
        grad[lag:] += wl * (1.0 - x[:-lag])
        grad[:-lag] += -wl * x[lag:]
    return grad * (2.0 / n)

def _active_obj(x, n, idx):
    c = compute_all_lags(x, n)
    return float(np.mean(c[idx]) * (2.0 / n))

def _active_grad(x, n, idx):
    grad = np.zeros(n)
    for i in idx:
        l = i - (n - 1)
        if l == 0:
            grad += (1.0 - 2.0 * x)
        else:
            m = abs(l)
            grad[m:] += (1.0 - x[:-m])
            grad[:-m] += -x[m:]
    return grad * (2.0 / n) / len(idx)

def slsqp_full(h, n, alpha, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), jac=_smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, idx, maxiter=3000, ftol=1e-18):
