def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_mc_matrix(h, n):
    dx = 2.0 / n
    mc = np.zeros((2*n-1, n))
    mc[n-1, :] = (1.0 - 2.0 * h) * dx
    for m in range(1, n):
        mc[n-1+m, m:] += (1.0 - h[:n-m]) * dx
        mc[n-1+m, :n-m] -= h[m:] * dx
        mc[n-1-m, :n-m] += (1.0 - h[m:]) * dx
        mc[n-1-m, m:] -= h[:n-m] * dx
    return mc

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_mc_matrix(h, n):
    dx = 2.0 / n
    mc = np.zeros((2 * n - 1, n))
    mc[n - 1, :] = (1.0 - 2.0 * h) * dx
    for m in range(1, n):
        mc[n - 1 + m, m:] += (1.0 - h[:n - m]) * dx
        mc[n - 1 + m, :n - m] -= h[m:] * dx
        mc[n - 1 - m, :n - m] += (1.0 - h[m:]) * dx
        mc[n - 1 - m, m:] -= h[:n - m] * dx
    return mc

def pair_search(h, n, deadline, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    n_lags = 2 * n - 1

    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break

        mc = compute_mc_matrix(h, n)
        found = False

        # Generate pairs with sufficient difference
        j_grid, l_grid = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
        pmask = j_grid < l_grid
        pj_all = j_grid[pmask]
        pl_all = l_grid[pmask]
        diff = np.abs(h[pj_all] - h[pl_all])
        sel = diff > 0.01
        pj = pj_all[sel]
        pl = pl_all[sel]

        if len(pj) == 0:
            break

        batch_size = 4000
        deltas = np.array([0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05,
                          -0.002, -0.005, -0.01, -0.015, -0.02, -0.03, -0.05])

        for bs in range(0, len(pj), batch_size):
            if time.time() > deadline - 2:
                break
            J = pj[bs:bs + batch_size]
            L = pl[bs:bs + batch_size]
            B = len(J)

            d = mc[:, J] - mc[:, L]

            lo = np.maximum(-h[J], h[L] - 1.0)
            hi = np.minimum(1.0 - h[J], h[L])
            valid = (deltas[:, None] > lo[None, :]) & (deltas[:, None] < hi[None, :])

            best_val = np.full(B, best_c)
            best_t = np.full(B, -1, dtype=int)

            for t in range(len(deltas)):
                if not valid[t].any():
                    continue
                vi = np.where(valid[t])[0]
                vals = (c[None, :] + deltas[t] * d[:, vi]).max(axis=1)
                better = vals < best_val[vi]
                if better.any():
                    vi2 = vi[better]
                    best_val[vi2] = vals[better]
                    best_t[vi2] = t
