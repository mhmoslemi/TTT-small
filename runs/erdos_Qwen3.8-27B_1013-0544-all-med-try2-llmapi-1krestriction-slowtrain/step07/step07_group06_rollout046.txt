def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-15:
            break
        if diff > 0:

            tot = room.sum()

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    return m + np.log(np.sum(np.exp(sh))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def sm_val(h, n, alpha):
    c = c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def sm_grad(h, n, alpha):
    c = c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def adam_phase(h, n, deadline, a0=80.0, a1=800000.0, lr0=0.018):
    h = project(h.copy(), n)
    bc, bh = compute_c5(h, n), h.copy()
    ma, va = np.zeros(n), np.zeros(n)
    t = 0
    for _ in range(20000):
        if time.time() > deadline: break
        t += 1
        f = min(t / 12000.0, 1.0)
        alpha = a0 * (a1 / a0) ** (2.0 * f * (1.0 - f) + f * f)
        lr = lr0 / (1.0 + 0.05 * t) ** 0.7
        g = sm_grad(h, n, alpha)
        ma = 0.9 * ma + 0.1 * g
        va = 0.999 * va + 0.001 * g * g
        mh = ma / (1 - 0.9 ** t)
        vh = va / (1 - 0.999 ** t)
        h_new = project(h - lr * mh / (np.sqrt(vh) + 1e-8), n)
        c = compute_c5(h_new, n)
        if c < bc - 1e-16:
            bc, bh, h = c, h_new.copy(), h_new
    return bh, bc

def lbfgs_phase(h, n, deadline, maxit=200):
    h = project(h.copy(), n)
    bc = compute_c5(h, n)
    bnd = [(0.0, 1.0)] * n
    for alpha in [500, 5000, 50000, 500000, 5000000]:
        if time.time() > deadline - 3: break
        def ob(x): return sm_val(x, n, alpha)
        def gr(x): return sm_grad(x, n, alpha)
        try:
            r = minimize(ob, h, jac=gr, method='L-BFGS-B', bounds=bnd,
                        options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxit * 8})
            h2 = project(r.x, n)
            c2 = compute_c5(h2, n)
            if c2 < bc - 1e-15:
                bc, h = c2, h2
        except: pass
    return h, bc

def bin_push(h, n, deadline, rounds=400):
    h = h.copy()
    bc = compute_c5(h, n)
    for _ in range(rounds):
        if time.time() > deadline: break
        imp = False
        order = np.argsort(np.where(h < 0.5, h, 1.0 - h))
        for i in order[:min(80, n)]:
            if time.time() > deadline: return h, bc
            if 0.08 < h[i] < 0.92: continue
            tgt = 0.0 if h[i] < 0.5 else 1.0
            d = tgt - h[i]
            if abs(d) < 1e-10: continue
            mask = (h > abs(d) * 1.01) & (h < 1.0 - abs(d) * 1.01)
            mask[i] = False
            if not mask.any():
                mask = (h > 0.01) & (h < 0.99)
                mask[i] = False
            if not mask.any(): continue
            h2 = h.copy()
            h2[i] = tgt
            h2[mask] -= d / mask.sum()
            h2 = project(h2, n)
            c = compute_c5(h2, n)
            if c < bc - 1e-15:
                bc, h, imp = c, h2, True
        if not imp: break
    return h, bc

def pair_search(h, n, deadline, deltas, radius=50, alpha=8000):
    h = h.copy()
    bc = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            imp = False
            g = sm_grad(h, n, alpha)
            order = np.argsort(-np.abs(g))
            for i in order:
                if time.time() > deadline: return h, bc
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15: bc, imp = c, True
                        else: h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15: bc, imp = c, True
                        else: h[i] += d; h[j] -= d
            if not imp: break
    return h, bc

def try_n(h, n, tg, deadline, alloc=30):
    xo = np.linspace(0, 2, n)
    xn = np.linspace(0, 2, tg)
    hi = project(np.interp(xn, xo, h), tg)
    t1 = min(time.time() + alloc * 0.5, deadline - 5)
    h1, c1 = adam_phase(hi, tg, t1, a0=100, a1=300000, lr0=0.015)
    t2 = min(time.time() + alloc * 0.3, deadline - 3)
    if time.time() < t2:
        h1, c1 = pair_search(h1, tg, t2, [0.005, 0.001, 0.0001], radius=35)
    return h1, c1
