import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0
_G_ALPHA = 0.0

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / float(n))

def compute_F(h, n):
    n = int(n)
    off = n - 1
    F = np.empty(2 * n - 1, dtype=float)
    hsum = float(np.sum(h))
    F[off] = hsum - float(np.dot(h, h))
    if n == 1:
        return F
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    for d in range(1, n):
        L = n - d
        p = float(np.dot(h[:L], h[d:]))
        F[off + d] = pref[L] - p
        F[off - d] = (hsum - pref[d]) - p
    return F

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    n = int(n)
    if n == 0:
        return h
    target = 0.5 * float(n)
    lo = -float(np.max(h))
    hi = 1.0 - float(np.min(h))
    if hi <= lo:
        lam = 0.0
    else:
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
            if s < target:
                lo = mid
            else:
                hi = mid
        lam = 0.5 * (lo + hi)
    h = np.clip(h + lam, 0.0, 1.0)
    for _ in range(120):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = float(np.sum(room))
        else:
            room = h
            s = float(np.sum(room))
        if s < 1e-15:
            break
        h = np.clip(h + diff * room / s, 0.0, 1.0)
    return h

def make_B(i, h, n):
    n = int(n)
    off = n - 1
    B = np.zeros(2 * n - 1, dtype=float)
    lo = max(-i, 1 - n)
    hi = min(n - 1 - i, n - 1)
    if lo <= hi:
        p0 = i + lo
        p1 = i + hi
        B[off + lo:off + hi + 1] += 1.0 - h[p0:p1 + 1]
    lo = max(i - (n - 1), 1 - n)
    hi = min(i, n - 1)
    if lo <= hi:
        q0 = i - lo
        q1 = i - hi
        B[off + lo:off + hi + 1] -= h[q1:q0 + 1][::-1]
    return B

def _eq_fun(x):
    return float(np.sum(x)) - 0.5 * float(_G_N)

def _eq_jac(x):
    return np.ones(_G_N, dtype=float)

def _obj(x):
    al = float(_G_ALPHA)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    y = al * (c - m)
    return m + float(np.log(np.sum(np.exp(y)))) / al

def _grad(x):
    n = int(_G_N)
    h = np.asarray(x, dtype=float)
    al = float(_G_ALPHA)
    F = compute_F(h, n)
    m = float(np.max(F))
    w = np.exp(al * (F - m))
    ws = float(np.sum(w))
    if ws <= 1e-300:
        return np.zeros(n, dtype=float)
    w /= ws
    G = np.zeros(n, dtype=float)
    off = n - 1
    G += w[off] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wd = float(w[off + d])
        wm = float(w[off - d])
        if wd <= 1e-15 and wm <= 1e-15:
            continue
        L = n - d
        a = h[:L]
        b = h[d:]
        if wd > 1e-15:
            G[:L] += wd * (1.0 - b)
            G[d:] -= wd * a
        if wm > 1e-15:
            G[d:] += wm * (1.0 - a)
            G[:L] -= wm * b
    return G

def slsqp_round(start_h, n, alphas, maxiter, ftol, deadline, best_h, best_c):
    n = int(n)
    h = project_feasible(np.asarray(start_h, dtype=float), n).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _eq_fun, "jac": _eq_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        _G_N = n
        _G_ALPHA = float(alpha)
        try:
            res = minimize(
                _obj, h, jac=_grad, method="SLSQP",
                bounds=bounds, constraints=cons,
                options={"maxiter": int(maxiter), "ftol": float(ftol)}
            )
            x = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
        except Exception:
            continue
        x = project_feasible(x, n)
        c = compute_c5(x, n)
        old = best_c
        if c < best_c:
            best_c = c
            best_h = x.copy()
        if c <= old + 1e-9:
            h = x.copy()
    return h, best_h, best_c

def update_pair(h, F, i, j, n, grid, refine, cur):
    n = int(n)
    hi_val = float(h[i])
    hj_val = float(h[j])
    u_lo = max(-hi_val, hj_val - 1.0)
    u_hi = min(1.0 - hi_val, hj_val)
    if u_hi - u_lo < 1e-13:
        return False, cur
    Bi = make_B(i, h, n)
    Bj = make_B(j, h, n)
    L = Bi - Bj
    Q = np.zeros(2 * n - 1, dtype=float)
    off = n - 1
    Q[off] = -2.0
    diff = j - i
    if diff != 0:
        Q[off + diff] = 1.0
        Q[off - diff] = 1.0
    grid = int(max(5, grid))
    refine = int(max(3, refine))
    best_u = 0.0
    best_val = cur
    U = np.linspace(u_lo, u_hi, grid)
    Uv = U[:, None]
    M = F[None, :] + L[None, :] * Uv + Q[None, :] * (Uv * Uv)
    r = np.max(M, axis=1)
    k = int(np.argmin(r))
    if r[k] < best_val - 1e-12:
        best_u = float(U[k])
        best_val = float(r[k])
    left = float(U[max(0, k - 1)])
    right = float(U[min(grid - 1, k + 1)])
    if right - left > 1e-12:
        U2 = np.linspace(left, right, refine)
        U2v = U2[:, None]
        M2 = F[None, :] + L[None, :] * U2v + Q[None, :] * (U2v * U2v)
        r2 = np.max(M2, axis=1)
        k2 = int(np.argmin(r2))
        if r2[k2] < best_val - 1e-12:
            best_u = float(U2[k2])
            best_val = float(r2[k2])
    if best_u == 0.0:
        return False, cur
    u = float(np.clip(best_u, u_lo, u_hi))
    x = float(np.clip(hi_val + u, 0.0, 1.0))
    y = float(np.clip(hj_val - u, 0.0, 1.0))
    new_cur = float(np.max(F + L * u + Q * u * u))
    if new_cur < cur - 1e-12:
        h[i] = x
        h[j] = y
        F += L * u + Q * u * u
        return True, new_cur
    return False, cur

def pair_search(h, n, max_sweeps, deadline, offsets, random_pairs, grid, refine, seed):
    n = int(n)
    h = project_feasible(np.asarray(h, dtype=float), n).copy()
    F = compute_F(h, n)
    cur = float(np.max(F))
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c = cur * dx
    rng = np.random.RandomState(seed)
    for _ in range(int(max_sweeps)):
        if time.time() > deadline:
            break
        improved = False
        for d in offsets:
            d = int(d)
            if d < 1:
                continue
            for i in range(n - d):
                if time.time() > deadline:
                    break
                ok, nc = update_pair(h, F, i, i + d, n, grid, refine, cur)
                if ok:
                    cur = nc
                    improved = True
                    c = cur * dx
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
        if random_pairs:
            cnt = int(min(random_pairs, 4 * n))
            for _ in range(cnt):
                if time.time() > deadline:
                    break
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                ok, nc = update_pair(h, F, i, j, n, grid, refine, cur)
                if ok:
                    cur = nc
                    improved = True
                    c = cur * dx
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
