```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    """Compute full correlation vector: corr[m] for m=0..2n-2"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max of correlation and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = corr  # shape (2n-1,)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    smooth_val = m + (mx + np.log(exp_s.sum())) / alpha
    
    # Gradient
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m_idx in range(n_lags):
        if w[m_idx] < 1e-16:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            # (1-h[j+k]) for j < nv
            grad[:nv] += wm * (1.0 - h[k:])
            # -h[j-k] for j >= k
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            # (1-h[j-kk]) for j >= kk  (since j >= k' = kk)
            grad[kk:] += wm * (1.0 - h[:nv])
            # -h[j+kk] for j < nv (since j < n-kk = nv)
            grad[:nv] += wm * (-h[kk:])
    return smooth_val, grad

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def objective_and_grad(x, n, alpha, mu):
    val, grad = smooth_max_and_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    val += mu * s * s
    grad += 2.0 * mu * s
    return val, grad

def run_lbfgs(h, n, deadline, max_iters=500):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    mus = [1.0, 10.0, 100.0, 1000.0, 1e6, 1e8, 1e10]
    
    for i in range(len(alphas)):
        if time.time() > deadline - 5:
            break
        alpha = alphas[i]
        mu = mus[i]
        try:
            res = minimize(lambda x: objective_and_grad(x, n, alpha, mu)[0],
                         h, jac=lambda x: objective_and_grad(x, n, alpha, mu)[1],
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    smooth_val = m + (mx + np.log(exp_s.sum())) / alpha
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-16:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return smooth_val, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lbfgs_obj(x, n, alpha, mu):
    val, _ = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def _lbfgs_jac(x, n, alpha, mu):
    _, grad = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    return grad + 2.0 * mu * s

def optimize_lbfgs(h, n, deadline):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    stages = [(50, 10.0, 300), (200, 100.0, 400), (1000, 1000.0, 400),
              (5000, 1e4, 400), (20000, 1e6, 500), (100000, 1e8, 500),
              (500000, 1e10, 600), (2000000, 1e12, 600)]
    for alpha, mu, maxiter in stages:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time() - 3
        if remaining < 5:
            break
        try:
            res = minimize(_lbfgs_obj, h, jac=_lbfgs_jac, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-19,
                                  'gtol': 1e-17, 'maxfun': maxiter * 25, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, radius=60):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 1e-5]
    for d in steps:
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            progress = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            progress = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def resample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def spectral_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    n_fourier = min(n // 2, 20)
    for freq in range(1, n_fourier + 1):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def structured_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    pattern = rng.choice(['random', 'block', 'alternating'], p=[0.5, 0.3, 0.2])
    if pattern == 'random':
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
    elif pattern == 'block':
        centers = rng.choice(n, size=2, replace=False)
        for c in centers:
            block = rng.randint(n // 8, n // 3)
            for i in range(-block, block + 1):
                h[(c + i) % n] = 1.0
    else:
        period = max(2, n // (half // 2 + 1))
        for i in range(0, n, period):
            if h.sum() < half:
                h[i] = 1.0
    return project_feasible(h, n)

def gradient_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    lr = 0.01
    for step in range(2000):
        if time.time() > deadline:
            break
        alpha = 5000 * (1 + step / 200.0)
        _, grad = smooth_max_grad(h, n, alpha)
        s = h.sum() - n / 2.0
        grad += 2.0 * 1e6 * s
        step_size = lr / (1.0 + step * 0.01)
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            lr *= 0.8
    return h, best_c
```
