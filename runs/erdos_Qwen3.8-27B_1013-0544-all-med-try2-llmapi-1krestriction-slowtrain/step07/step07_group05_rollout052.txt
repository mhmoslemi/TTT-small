    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    k_worst = np.argmax(corr)
    k_shift = k_worst - (n - 1)

    # For lag k_shift, the overlap involves pairs (i, i+k_shift)
    # We want to reduce h at these positions

    elif k_shift < 0:

    else:
        return None, 0.0

    # and transfer to a position with minimum h
    h_valid = np.array([h[i] for i in valid_i])
    i_max = valid_i[np.argmax(h_valid)]

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
