    """Find a single paired move that reduces the worst overlap."""
    overlaps = get_overlaps(h, n)
    m_star = np.argmax(overlaps)
    k = m_star - (n - 1)
    dx = 2.0 / n
    
    # Compute gradient of c[m_star] w.r.t. h
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1 - h[k:])  # from h[i] term
        grad[k:] += (-h[:nv])     # from h[i+k] term
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1 - h[:nv])  # from h[i+kk] term
        grad[:nv] += (-h[kk:])     # from h[i] term

    # Best single move: find (a, b) where grad[a] > 0, grad[b] < 0

    a = np.argmax(grad)
    b = np.argmin(grad)

        best_h, best_c = shift_local_search(best_h, best_n, t2, top_k=4, max_rounds=80000)

    # Phase 3: Binary relaxation (80s)
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 150)
        h3, c3 = binary_relax(best_h, best_n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Different n values (150s)
    for n_try in [400, 600, 480]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(60, (deadline - time.time() - 80) * 0.55)
        if alloc < 20:
            continue
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc,
                                [1000, 10000, 60000, 300000])
        if time.time() < deadline - 25:
            h_t, c_t = shift_local_search(h_t, n_try,
                                          min(time.time() + 30, deadline - 15),
                                          top_k=2, max_rounds=8000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 5: Diverse starts (80s)
    if time.time() < deadline - 80:
        starts = diverse_starts(best_n, rng)
        for s in starts:
            if time.time() > deadline - 25:
                break
            alloc = min(22, (deadline - time.time() - 15) * 0.6)
            if alloc < 8:
                break
            h_s, c_s = lbfgs_phase(s, best_n, time.time() + alloc,
                                   [2000, 15000, 80000])
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 6: Perturbation restart
    if time.time() < deadline - 35:
        h_p = best_h + rng.randn(best_n) * 0.03
        h_p = project_feasible(h_p, best_n)
        alloc = min(25, deadline - time.time() - 10)
        if alloc > 8:
            h_p, c_p = lbfgs_phase(h_p, best_n, time.time() + alloc,
                                    [5000, 50000, 300000])
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 7: Final local search
    if time.time() < deadline - 8:
        t7 = min(time.time() + 30, deadline - 3)
        best_h, best_c = shift_local_search(best_h, best_n, t7, top_k=2, max_rounds=30000)

    # Final L-BFGS
    if time.time() < deadline - 2:
        best_h, best_c = lbfgs_phase(best_h, best_n, deadline - 1, [1000000])

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
