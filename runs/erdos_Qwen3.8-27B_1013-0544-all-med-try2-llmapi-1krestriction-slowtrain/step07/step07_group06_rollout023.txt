def c5_vec(h, n):

def bvals(b):
    a = b.astype(np.int32, copy=False)
    return np.correlate(a, 1 - a, mode='full')

def bscore(vals, T):
    m = int(np.max(vals))
    ex = 0.0
    if T < m:
        ex = float(np.sum(vals[vals > T] - T))
    k = min(8, vals.size)
    top = int(np.partition(vals, vals.size - k)[-k:].sum()) if k else 0
    return ex + 0.001 * m + 1e-5 * top + 1e-8 * float(np.sum(vals))

def offending_pairs(b, mi, n):
    k = mi - (n - 1)
    if k > 0:
        idx = np.nonzero((b[:-k] == 1) & (b[k:] == 0))[0]
        return np.column_stack((idx, idx + k))
    if k < 0:
        kk = -k
        idx = np.nonzero((b[kk:] == 1) & (b[:-kk] == 0))[0] + kk
        return np.column_stack((idx, idx - kk))
    return np.empty((0, 2), dtype=int)

def adjust_weight(b, W, rng):
    cnt = int(b.sum())
    if cnt > W:
        idx = np.nonzero(b == 1)[0]
        rem = cnt - W
        if rem:
            b[rng.choice(idx, rem, replace=False)] = 0
    elif cnt < W:
        idx = np.nonzero(b == 0)[0]
        rem = W - cnt
        if rem:
            b[rng.choice(idx, rem, replace=False)] = 1
    return b

def block_binary(n, rng):
    b = np.empty(n, dtype=np.int8)
    val = 1 if rng.random() < 0.5 else 0
    i = 0
    while i < n:
        r = int(rng.integers(2, 6))
        e = min(n, i + r)
        b[i:e] = val
        val ^= 1
        i = e
    return adjust_weight(b, n // 2, rng)

def thue_binary(n, rng):
    m = 1
    while m < n:
        m <<= 1
    t = np.zeros(m, dtype=np.int8)
    for i in range(1, m):
        t[i] = (t[i >> 1] + (i & 1)) % 2
    t = t[:n]
    return adjust_weight(t, n // 2, rng)

def binary_starts(n, ref_h, rng):
    starts = []
    if ref_h.size:

    b = np.zeros(n, dtype=np.int8)
    b[rng.choice(n, n // 2, replace=False)] = 1
    starts.append(b)
    starts.append(block_binary(n, rng))
    starts.append(thue_binary(n, rng))
    return starts

                if dd < -1e-12 or (temp > 1e-9 and rng.random() < np.exp(-max(0.0, dd) / temp)):
                    sc = ns
                    acc = True
                    break
                b[i] = 1
                b[j] = 0
        if acc:
            no = 0
            continue
        no += 1
        temp *= 0.998
        if no > (1500 if T < target_T else 8000):
            break
        if no > 700 and ones.size and zeros.size:
            for _ in range(12):
                i = int(ones[rng.integers(ones.size)])
                j = int(zeros[rng.integers(zeros.size)])
                if b[i] == 1 and b[j] == 0:
                    b[i] = 0
                    b[j] = 1
                    nv = bvals(b)
                    ns = bscore(nv, T)
                    if ns <= sc + 0.3:
                        sc = ns
            no = 0
            temp = max(temp, 1.0)
    return bb, float(2 * bm) / float(n), bm

def binary_stage(n, ref, rng, end):
    if n <= 0 or n % 2:
        return ref, c5_val(ref, int(ref.size)), 0
    TT = int(np.floor(0.3808 * n / 2.0 + 1e-9))
    st = binary_starts(n, ref, rng)
    bb = None
    bc = 1.0
    fr = [1.0] if len(st) == 1 else [0.4] + [0.3] * (len(st) - 1)
    for idx, b in enumerate(st):
        if time.time() >= end - 2:
            break
        sub = min(end, time.time() + max(6.0, (end - time.time()) * fr[idx]))
        h, c, m = binary_optimize(b, n, TT, sub, rng)
        if c < bc:
            bc = c
            bb = h
    if bb is None:
        return ref, c5_val(ref, int(ref.size)), 0
    if time.time() < end - 4:
        cnt = int(np.floor(bc * n / 2.0 - 1e-9)) - 1
        cnt = max(0, cnt)
        h, c, m = binary_optimize(bb, n, cnt, end, rng)
        if c < bc:
            bc = c
            bb = h
    return bb, bc, m

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 2.0
    rng = np.random.default_rng(seed)
    try:
        base = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        base = np.full(240, 0.5)
    if base.size < 2:
        base = np.full(240, 0.5)
    n0 = int(base.size)
    n0 -= n0 % 2
    if n0 < 2:
        base = np.full(240, 0.5)
        n0 = 240
    else:
        base = base[:n0]
    base = project_feasible(base, n0)
    best_h = base
    best_c = c5_val(best_h, n0)
    total = max(30.0, dl - time.time())
    stages = [('c', n0, 1.2), ('b', 300, 3.5), ('b', 256, 2.2), ('b', 200, 1.8),
              ('b', 160, 1.2), ('b', 128, 1.0), ('b', 100, 0.8), ('b', 80, 0.6),
              ('b', 64, 0.5), ('c', -1, 1.5)]
    tw = 14.3
    for kind, n, w in stages:
        if time.time() >= dl - 4:
            break
        end = min(dl - 2.0, time.time() + max(8.0, total * w / tw))
        if kind == 'c':
            nn = n if n != -1 else int(best_h.size)
            hh, cc = cont_local(best_h, nn, end, rng)
            if cc < best_c:
                best_c = cc
                best_h = hh
        else:
            hh, cc, mm = binary_stage(n, best_h, rng, end)
            if cc < best_c:
                best_c = cc
                best_h = hh
    nn = int(best_h.size)
    best_h = project_feasible(best_h, nn)
    best_c = c5_val(best_h, nn)
    return best_h, float(best_c), nn
