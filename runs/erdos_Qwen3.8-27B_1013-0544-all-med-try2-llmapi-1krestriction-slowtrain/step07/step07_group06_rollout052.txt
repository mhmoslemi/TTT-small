            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def get_top_lags(h, n, K=10, eps=1e-5):
    c = compute_all_c(h, n)
    m = c.max()
    active = np.where(c >= m - eps)[0]
    if len(active) > K:
        active = np.argsort(-c)[:K]
    elif len(active) < 3:
        active = np.argsort(-c)[:3]
    return active

def grad_of_lag(h, n, k_idx):
    dx = 2.0 / n
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:])
        g[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv])
        g[:nv] += (-h[kk:])
    return g * dx

def active_set_lbfgs(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [500, 2000, 10000, 50000, 200000]
    Ks = [5, 8, 12, 20]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        for K in Ks:
            if time.time() > deadline - 5:
                break
            lags = get_top_lags(h, n, K=K, eps=1e-4)
            def obj(x):
                return smooth_max_val(x, n, lags, alpha)
            def gr(x):
                return smooth_max_grad(x, n, lags, alpha)
            try:
                res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 150, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def subgradient_exact(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    stuck = 0
    while time.time() < deadline - 2:
        c = compute_all_c(h, n)
        cmax = c.max()
        if cmax < best_c - 1e-14:
            best_c = cmax
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 50:
            h = best_h + np.random.normal(0, 0.005, n)
            h = project_feasible(h, n)
            stuck = 0
            continue
        # Subgradient: average gradient over near-active lags
        thresh = cmax - 1e-4
        active = np.where(c >= thresh)[0]
        g = np.zeros(n)
        for ai in active:
            g += grad_of_lag(h, n, ai)
        g /= len(active)
        # Line search
        improved = False
        for ss in steps:
            h_new = project_feasible(h - ss * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                h = h_new
                best_c = c_new
                best_h = h_new.copy()
                improved = True
                break
        if not improved and stuck > 15:
            # Try anti-grad direction briefly (escape)
            h_try = project_feasible(h + 0.001 * g, n)
            if compute_c5(h_try, n) < compute_c5(h, n):
                h = h_try
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius or n))
                hi = min(n, i + (radius or n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def grad_guided_pair(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            c = compute_all_c(h, n)
            active = np.where(c >= c.max() - 1e-4)[0]
            g = np.zeros(n)
            for ai in active:
                g += grad_of_lag(h, n, ai)
            order = np.argsort(-np.abs(g))
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def random_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def quick_lbfgs(h, n, t_limit):
    c = compute_all_c(h, n)
    lags = np.argsort(-c)[:15]
    alpha = 2000
    def obj(x):
        return smooth_max_val(x, n, lags, alpha)
    def gr(x):
        return smooth_max_grad(x, n, lags, alpha)
    try:
        res = minimize(obj, h, jac=gr, method='L-BFGS-B',
                      bounds=[(0, 1)] * n,
                      options={'maxiter': 80, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Multi-start global exploration (~150s)
    t1 = min(t0 + 150, deadline - 500)
    candidates = [(h.copy(), best_c)]
    # Try several random starts
    n_starts = 12
    per_start = max(5, (t1 - t0 - 10) / n_starts)
    for si in range(n_starts):
        if time.time() > t1 - 5:
            break
        slimit = min(time.time() + per_start, t1 - 3)
        rng_h = random_start(n, seed + si * 1000)
        hc, cc = quick_lbfgs(rng_h, n, slimit)
        # Also try perturbed version of initial
        if si % 2 == 1:
            rng_h2 = project_feasible(h_init + np.random.normal(0, 0.1, n), n)
            hc2, cc2 = quick_lbfgs(rng_h2, n, slimit)
            if cc2 < cc:
                hc, cc = hc2, cc2
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
        candidates.append((hc, cc))

    # Phase 2: Active-set L-BFGS-B on best (~250s)
    t2 = min(time.time() + 250, deadline - 350)
    if time.time() < t2:
        best_h, best_c = active_set_lbfgs(best_h, n, t2)

    # Phase 3: Subgradient on exact C5 (~250s)
    t3 = min(time.time() + 250, deadline - 250)
    if time.time() < t3:
        h3, c3 = subgradient_exact(best_h, n, t3)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Gradient-guided pair search (~250s)
    t4 = min(time.time() + 250, deadline - 120)
    if time.time() < t4:
        deltas1 = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
        best_h, best_c = grad_guided_pair(best_h, n, t4, deltas1, radius=60)

    # Phase 5: Try different resolutions (~150s)
    if time.time() < deadline - 60:
        for n_try in [200, 250, 400, 500, 600]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(30, deadline - time.time() - 30)
            if alloc < 15:
                break
            h_t, c_t = active_set_lbfgs(h_interp, n_try, time.time() + alloc * 0.5)
            if time.time() < deadline - 15:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 10, deadline - 10),
                                       [0.001, 0.0001, 0.00005], radius=40)
            if c_t < best_c:
                best_c, best_h = c_t, h_t

    # Phase 6: Final fine pair search
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = grad_guided_pair(best_h, nf, deadline - 2,
                                          [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                          radius=40)

    # Phase 7: Final active-set polish
    if time.time() < deadline - 1:
        best_h, best_c = active_set_lbfgs(best_h, len(best_h), deadline)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
