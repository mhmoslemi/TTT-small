def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    """Smooth max of scaled correlation, with gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    p = exp_s / sum_exp
    obj = max_c + np.log(sum_exp) / alpha
    
    # Gradient
    n_corr = len(corr)
    lags = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jm = j - lags
        jp = j + lags
        m1 = (jm >= 0) & (jm < n)
        m2 = (jp >= 0) & (jp < n)
        t1 = np.where(m1, 1.0 - h[jm], 0.0)
        t2 = np.where(m2, h[jp], 0.0)
        grad[j] = np.dot(p, t1 - t2)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = np.clip(alpha * (corr - max_c), -500, 0)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    p = exp_s / sum_exp
    obj = max_c + np.log(sum_exp) / alpha
    n_corr = len(corr)
    lags = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jm = j - lags
        jp = j + lags
        m1 = (jm >= 0) & (jm < n)
        m2 = (jp >= 0) & (jp < n)
        t1 = np.where(m1, 1.0 - h[jm], 0.0)
        t2 = np.where(m2, h[jp], 0.0)
        grad[j] = dx * np.dot(p, t1 - t2)
    return obj, grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj_fn(x):
        o, _ = smooth_obj_grad(x, n, alpha)
        return o
    def obj_grad(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})

        return h

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_nc = cur_c
            best_j = -1
            h[i] = 1.0
            for j in ones:
                h[j] = 0.0
                nc = np.correlate(h, 1 - h, mode='full').max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                h[j] = 1.0
            h[i] = 0.0
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                cur_c = best_nc
                improved = True
                break
        if not improved:
            break
    return h, cur_c

def simulated_annealing(h_init, n, deadline, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_total = max(1.0, deadline - t_start)
    T0, T1 = 0.05, 0.0005
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    while time.time() < deadline:
        progress = min(1.0, (time.time() - t_start) / t_total)
        T = T0 * (T1 / T0) ** progress
        i = rng.choice(zeros_list)
        j = rng.choice(ones_list)
        h[i], h[j] = 1.0, 0.0
        nc = np.correlate(h, 1 - h, mode='full').max() * dx
        delta = nc - cur_c
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_c = nc
            ones_list.remove(j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP continuous optimization
    p1 = t0 + 0.10 * (budget_s - 10)
    h = best_h.copy()
    for alpha in [200, 500, 2000, 10000, 50000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, 3000, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Iterated local search on binary
    p2 = t0 + 0.35 * (budget_s - 10)
    hb, cb = iterated_local_search(best_h, n, p2, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 3: SLSQP from best binary with perturbation
    p3 = t0 + 0.45 * (budget_s - 10)
    if time.time() < p3:
        hp = project_feasible(best_h + 0.08 * rng.randn(n), n)
        for alpha in [2000, 10000, 50000, 200000]:
            if time.time() > p3:
                break
            hp = slsqp_optimize(hp, n, alpha, 2000, p3)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    # Phase 4: Simulated annealing (exploit global search)
    p4 = t0 + 0.65 * (budget_s - 10)
    if time.time() < p4:
        ha, ca = simulated_annealing(best_h, n, p4, seed=seed + 7)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()
        # Greedy polish after annealing
        hg, cg = _greedy_from_binary(ha, n, p4)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
    # Phase 5: Multi-resolution
    p5 = t0 + 0.80 * (budget_s - 10)
    for nh in [200, 300, 350]:
        if time.time() > p5 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh = slsqp_optimize(hh, nh, 5000, 1500, p5 - 15)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        hb2, cb2 = iterated_local_search(hh, nh, p5 - 10, seed=seed + nh)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh
        elif ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh
    # Phase 6: Final iterated local search
    p6 = dl - 15
    hb, cb = iterated_local_search(best_h, n, p6, seed=seed + 200)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 7: Final SLSQP polish
    if time.time() < dl - 5:
        hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
        hp = slsqp_optimize(hp, n, 100000, 3000, dl - 3)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
