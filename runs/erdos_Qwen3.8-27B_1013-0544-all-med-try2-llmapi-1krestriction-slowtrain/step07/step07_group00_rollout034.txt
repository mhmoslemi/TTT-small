def c5_grad_fft(h, n, N, alpha):
    dx = 2.0 / n
    corr = fast_corr_full(h, n, N)
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()

    cumW = np.zeros(2*n)
    cumW[1:] = np.cumsum(W)
    A1 = cumW[2*n-1 - np.arange(n)] - cumW[n-1 - np.arange(n)]
    # A1[i] = cumW[2n-1-i] - cumW[n-1-i]
    
    # A2: cross-correlation
    Fh = np.fft.rfft(h, N)
    Fw = np.fft.rfft(W, N)
    A2_full = np.fft.irfft(Fh * np.conj(Fw), N)
    A2 = A2_full[(np.arange(n) - (n-1)) % N]
    
    # B: convolution
    B_full = np.fft.irfft(Fh * Fw, N)
    B = B_full[(n-1 + np.arange(n)) % N]
    
    grad = A1 - A2 - B
    return grad
