import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_i in range(2 * n - 1):
        if w[m_i] < 1e-15:
            continue
        k = m_i - (n - 1)
        wm = w[m_i] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj(x, n, alpha):
    return smooth_max_obj(x, n, alpha)

def _grad(x, n, alpha):
    return smooth_max_grad(x, n, alpha)

def lbfgs_phase(h, n, deadline, alphas, max_iter=600):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        remaining = deadline - time.time() - 10
        if remaining < 15:
            break
        try:
            res = minimize(_obj, best_h, jac=_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iter * 50,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=200000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def _eq(x):
        return [x.sum() - target]
    def _eq_jac(x):
        return [np.ones(n)]
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coord_descent(h, n, deadline, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for ds in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, 100000)
            order = np.argsort(g)
            for idx in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[idx]
                lo = max(0, idx - 80)
                hi = min(n, idx + 80)
                for j_idx in range(lo, hi):
                    j = order[j_idx]
                    if i == j:
                        continue
                    if g[j] - g[i] < 1e-12:
                        continue
                    if h[i] + ds <= 1.0 and h[j] - ds >= 0.0:
                        h[i] += ds
                        h[j] -= ds
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= ds
                            h[j] += ds
                    if h[i] - ds >= 0.0 and h[j] + ds <= 1.0:
                        h[i] -= ds
                        h[j] += ds
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += ds
                            h[j] -= ds
    return h, best_c

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 8):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

    t0 = time.time()
