import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-13:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += -wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[kk:]
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def eq_c(x, n):
    return float(x.sum() - n / 2.0)

def eq_j(x, n):
    return np.ones(n)

def optimize_lbgfs(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [(50, 5.0), (200, 20.0), (1000, 100.0), (5000, 500.0),
              (20000, 2000.0), (100000, 10000.0)]
    for alpha, mu in alphas:
        if time.time() > deadline - 5:
            break
        maxiter = max(50, min(400, int((deadline - time.time()) * 4)))
        try:
            res = minimize(pen_obj, best_h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def optimize_slsqp(h, n, deadline, alpha=50000, maxiter=1000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    try:

        c_new = compute_c5(h_new, n)
        if c_new < best_c:

            h = h_new
    except Exception:
        pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=80):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                hi_j = hi
                for j in range(lo, hi_j):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def grad_pairs(h, n, deadline, step=0.002):
    best_c = compute_c5(h, n)
    for alpha in [500, 2000, 10000]:

            break

                    return h, best_c
                gi = g[i]
                for j in range(n):
                    if i == j:
                        continue
                    move = -step * (gi - g[j])
                    if abs(move) < 1e-11:
                        continue
                    hi = h[i] + move
                    hj = h[j] - move
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi; h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] = hi - move; h[j] = hj + move
            if not improved:
                break
    return h, best_c

    return project_feasible(h_new, n_fine)
