import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_smooth_max(h, n, alpha, M=5):
    """Compute smooth max over top M shifts only."""
    c = compute_c5_vec(h, n)
    # Get top M indices
    idx = np.argpartition(c, -M)[-M:]
    c_active = c[idx]
    m = c_active.max()
    sh = alpha * (c_active - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    val = m + np.log(w.sum()) / alpha  # = m since w sums to 1
    # Actually: log-sum-exp of alpha*(c_active - m) / alpha + m
    # = m + (mx + log(sum(exp(sh-mx))))/alpha
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return val, idx, w, c

def active_grad(h, n, alpha, M=5):
    """Gradient of active smooth max."""
    c = compute_c5_vec(h, n)
    idx = np.argpartition(c, -M)[-M:]

            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

            alloc = min(18, deadline - time.time() - 10)
            if alloc < 8:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=200, gap=0.02)
            if time.time() < deadline - 4:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 4, deadline - 3),
                                       [0.005, 0.001, 0.0005], radius=40)
            if c_t < best_c:
                best_c, best_h, nf = c_t, h_t, n_try

    # Phase 9: Final polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
