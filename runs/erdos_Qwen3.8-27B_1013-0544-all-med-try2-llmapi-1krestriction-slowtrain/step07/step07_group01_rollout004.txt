import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-13:
            continue
        k = mm - (n - 1)
        wm_dx = wm * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm_dx * (1.0 - h[k:])
            grad[k:] += wm_dx * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm_dx * (1.0 - h[:nv])
            grad[:nv] += wm_dx * (-h[kk:])
    return grad

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    g = grad_c5(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def sm_obj_no_pen(x, n, alpha):

    mx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slq_const(x, n):
    return float(x.sum() - n / 2.0)

def slq_jac(x, n):
    return np.ones(n)

def lbfgs_run(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu, maxit in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': maxit * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_run(h, n, deadline, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_const, 'jac': slq_jac, 'args': (n,)}]
    try:
        res = minimize(sm_obj_no_pen, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius if radius else n))
                hi = min(n, i + (radius if radius else n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def grad_pair(h, n, deadline, step_size=0.003):
    alphas = [100, 500, 2000, 10000]
    best_c = compute_c5(h, n)
    for alpha in alphas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = grad_c5(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= move; h[j] += move
    return h, best_c
