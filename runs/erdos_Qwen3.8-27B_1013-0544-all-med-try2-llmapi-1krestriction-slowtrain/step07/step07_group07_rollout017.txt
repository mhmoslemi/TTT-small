    return np.correlate(h, 1 - h, mode='full') * dx

    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        return float(m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

    def obj(x):
        return smooth_max(compute_all_lags(x, n), alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=5000, ftol=1e-16):
    def obj(x):
        return smooth_max(compute_all_lags(x, n), alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def top_lag_search(h, n, deadline, n_top=10, steps=None):

            if time.time() > deadline:
                break
            lag = ki - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            if lag > 0:
                indices = [(i, i + m) for i in range(n - m)]
            else:
                indices = [(i, i - m) for i in range(m, n)]
            for i, j in indices:
                if time.time() > deadline:
                    break
                for step in steps:
                    for direction in [1, -1]:
                        if direction > 0:
                            d = min(step, best_h[i] * 0.4, 1.0 - best_h[j])
                        else:
                            d = min(step, best_h[j] * 0.4, 1.0 - best_h[i])
                        if d < 1e-12:
                            continue
                        nh = best_h.copy()
                        if direction > 0:
                            nh[i] -= d
                            nh[j] += d
                        else:
                            nh[i] += d
                            nh[j] -= d
                        nh = project_feasible(nh, n)
                        c_new = compute_c5(nh, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = nh
                            improved = True
        if not improved:
            break
    return best_h, best_c

def random_pair_search(h, n, deadline, rng, n_samples=3000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    steps = [0.003, 0.001, 0.0005]
    for _ in range(n_samples):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step = steps[rng.randint(3)]

        nh = best_h.copy()

            nh[i] -= d
            nh[j] += d
        else:
            nh[i] += d
            nh[j] -= d

        c_new = compute_c5(nh, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = nh
    return best_h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start, n)
    best_c = compute_c5(h, n)
    # Progressive SLSQP
    alphas = [2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        alloc_min = 3
        h_new = slsqp_opt(h, n, alpha, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            h = h_new  # still use as starting point
    # Top-lag local search
    ls_end = time.time() + min(80, (deadline - time.time()) * 0.5)
    h_ls, c_ls = top_lag_search(h, n, min(ls_end, deadline - 5), n_top=12)
    if c_ls < best_c:

        h = h_ls
    # Random pair search

        if c_rs < best_c:

            h = h_rs
    # Final polish with very fine steps

        if c_fin < best_c:
            best_c = c_fin
            h = h_fin
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
