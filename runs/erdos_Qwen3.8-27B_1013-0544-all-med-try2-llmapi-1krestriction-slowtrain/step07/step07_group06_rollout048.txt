    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    diff = target - h.sum()
    if abs(diff) > 1e-15:

        idx = np.where((h > 1e-10) & (h < 1 - 1e-10))[0]
        if len(idx) > 0:

def coord_descent_optimal(h, n, i, j, deadline):
    """Find optimal transfer from j to i (or i to j) using ternary search."""
    best_c = compute_c5(h, n)
    # Try transfer i -> j and j -> i
    for sign in [1, -1]:
        # h[i] += sign*delta, h[j] -= sign*delta
        lo = 0.0
        hi = min(h[i] if sign > 0 else h[j], 1.0 - h[j] if sign > 0 else 1.0 - h[i])
        if hi <= 0:

        # Ternary search for optimal delta
        for _ in range(40):
            if time.time() > deadline:

            m2 = hi - (hi - lo) / 3
            h[i] += sign * m1
            h[j] -= sign * m1
            c1 = compute_c5(h, n)
            h[i] += sign * (m2 - m1)
            h[j] -= sign * (m2 - m1)

The integral approximation: c[m] * dx where dx = 2/n.

So C₅ = max_m c[m] * (2/n).

This is correct.

Now, one important observation: the maximum is often at small |k| values (near 0). The k=0 term is ∑h[i](1-h[i]) which is 0 for binary h and 0.25n for h=0.5.

For near-binary h, the max is likely at small non-zero k values.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += min(diff / tot, 1.0) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha=1000):
    """Gradient of smooth max approximation of C5."""
    c = compute_c5_vec(h, n)
    c_shift = c - c.max()
    w = np.exp(alpha * c_shift)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def adam_opt(h, n, deadline, alpha_init=100, alpha_end=500000, lr=0.01, max_steps=5000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max(float(max_steps)), 1.0)
        a = alpha_init * (alpha_end / alpha_init) ** frac
        lr_c = lr / (1.0 + 0.3 * t) ** 0.5
        g = grad_c5(h, n, a)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g ** 2
        m_hat = m_arr / (1 - b1 ** t)
        v_hat = v_arr / (1 - b2 ** t)
        h_new = project_feasible(h - lr_c * m_hat / (np.sqrt(v_hat) + eps), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def coord_descent(h, n, deadline, radius=50, use_grad=True):
    """Adaptive coordinate descent: try transferring mass between pairs."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    while time.time() < deadline:
        improved = False
        if use_grad:
            g = grad_c5(h, n, alpha=5000)
            # Sort by |gradient| descending
            order = np.argsort(-np.abs(g))
        else:
            order = np.random.permutation(n)
        
        for i in order:
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                for d in deltas:
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
        if not improved:
            break
    return h, best_c

def binary_push(h, n, deadline):
    """Push values toward 0 or 1."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(500):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            v = h[i]
            if 0.05 < v < 0.95:
                continue
            target = 0.0 if v < 0.5 else 1.0
            delta = target - v
            if abs(delta) < 1e-12:
                continue
            # Find where to put the mass
            mask = (np.abs(h - 0.5) > abs(delta)) & (h != i)
            idx = np.where((h > 0) & (h < 1) & (np.arange(n) != i))[0]
            if len(idx) == 0:
                continue
            # Try pushing to nearest elements
            h_new = h.copy()
            h_new[i] = target
            # Redistribute
            total_needed = -delta  # amount to distribute
            if total_needed > 0:
                # Need to add mass to other elements
                room = 1.0 - h_new
                room[i] = 0
                tot = room.sum()
                if tot < 1e-14:
                    continue
                h_new[idx] += total_needed * room[idx] / room[idx].sum()
            else:
                # Need to remove mass from other elements
                avail = h_new
                avail[i] = 0
                tot = avail.sum()
                if tot < 1e-14:
                    continue
                h_new[idx] += total_needed * avail[idx] / avail[idx].sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-14:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def lbfgs_smooth(h, n, deadline):
    """L-BFGS-B on smooth max approximation."""
    h0 = project_feasible(h.copy(), n)
    best_c = compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [200, 2000, 20000, 200000, 2000000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_max(x, n, a)
        def gr(x):
            return grad_c5(x, n, a)
        try:
            res = minimize(obj, h0, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                h0 = h_new
        except Exception:
            pass
    return h0, best_c

def make_structured_init(n, seed=0):
    """Create structured initial points."""
    rng = np.random.RandomState(seed)
    # Option 1: alternating blocks
    n_blocks = rng.randint(3, 10)
    h = np.zeros(n)
    block_size = n // n_blocks

        if b % 2 == 0:

    h3 = project_feasible(h3, n)

def refine_multi_resolution(h_start, n_start, n_target, deadline):

    h = h_start.copy()

One more idea: **targeted pair search**. Instead of searching all pairs, identify which pairs (i,j) would most reduce the C₅ at the active k values, and focus there.

For a given active shift k, the term f_k(h) = ∑ h[i](1-h[i+k]). To decrease this:
- If h[i] and h[i+k] disagree (one is 1, other is 0), changing one of them helps
- Specifically, if h[i]=1 and h[i+k]=0, decreasing h[i] or increasing h[i+k] helps
- If h[i]=0 and h[i+k]=1, increasing h[i] or decreasing h[i+k] helps

So the "best" pairs to modify are those where h[i] and h[i+k] disagree, for active k values.

def targeted_coord_descent(h, n, deadline, max_sweeps=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        # Find active shifts (within 5% of max)
        active_threshold = c_max * 0.95
        active_shifts = np.where(c_vec >= active_threshold)[0]
        # Convert to shifts
        active_k = active_shifts - (n - 1)
        
        # Identify pairs to try
        pairs_to_try = set()
        for k in active_k:
            k = int(k)
            if k >= 0:
                for i in range(n - k):
                    if h[i] * (1 - h[i+k]) > 0.01:  # disagreeing pair
                        pairs_to_try.add((i, i+k))
            else:
                kk = -k
                for i in range(n - kk):
                    if h[i+kk] * (1 - h[i]) > 0.01:
                        pairs_to_try.add((i, i+kk))
        
        if len(pairs_to_try) == 0:
            pairs_to_try = set()
            for i in range(0, n, max(1, n // 100)):
                for j in range(i + 1, min(i + 30, n)):
                    pairs_to_try.add((i, j))
        
        improved = False
        for (i, j) in pairs_to_try:
            if time.time() > deadline:
                return h, best_c
            for d in deltas:
                # Try i -> j
                if h[i] >= d and h[j] <= 1 - d:
                    h[i] -= d
