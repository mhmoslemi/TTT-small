    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        n = n_init
        alloc1 = min(250, (deadline - t0) * 0.45)
        h1, c1 = optimize_one(h_init, n, t0 + alloc1, rng)
        best_h, best_c = h1, c1
        if time.time() < deadline - 60:
            for n_try in [200, 250, 400]:
                if time.time() > deadline - 80:
                    break
                x_best = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_try)
                h_interp = np.interp(x_new, x_best, best_h, period=2)
                h_interp = project_feasible(h_interp, n_try)
                alloc = min(80, (deadline - time.time()) * 0.3)
                h2, c2 = optimize_one(h_interp, n_try, time.time() + alloc, rng)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
        if time.time() < deadline - 50:
            n = len(best_h)
            for r in range(3):
                if time.time() > deadline - 40:
                    break
                scale = 0.08 * (1.0 + 0.5 * r)
                hp = best_h + rng.normal(0, scale, n)
                hp = project_feasible(hp, n)
                alloc = min(60, (deadline - time.time()) * 0.3)
                if alloc < 30:
                    break
                h3, c3 = optimize_one(hp, n, time.time() + alloc, rng)
                if c3 < best_c:
                    best_c = c3
                    best_h = h3
        if best_h is not None and time.time() < deadline - 15:
            n = len(best_h)
            h4, c4 = active_set_refine(best_h, n, deadline - 3, top_m=12, max_rounds=8)
            if c4 < best_c:
                best_c = c4
                best_h = h4
    else:
        n = 300
        h_init = project_feasible(np.full(n, 0.5) + rng.randn(n) * 0.1, n)
        best_h, best_c = optimize_one(h_init, n, deadline, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
