import numpy as np
import time

TAU_LONG = (8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.125, 0.06)
TAU_MED = (2.0, 1.0, 0.5, 0.25, 0.125, 0.06)
TAU_SHORT = (1.0, 0.5, 0.25, 0.125)


def project_box_sum(z, target):
    z = np.array(z, dtype=float, copy=False)
    target = float(target)
    n = z.size
    if n == 0:
        return z
    lo = float(np.min(z))
    hi = float(np.max(z))
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        diff = target - float(np.sum(x))
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            room = 1.0 - x
            s = float(np.sum(room))
            if s > 1e-15:
                x = x + diff * room / s
            else:
                x = x + diff / n
        else:
            s = float(np.sum(x))
            if s > 1e-15:
                x = x + diff * x / s
            else:
                x = x + diff / n
        x = np.clip(x, 0.0, 1.0)
    return x


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5_from_corr(corr, n):
    return float(np.max(corr)) * (2.0 / float(n))


def compute_c5(h, n):
    return compute_c5_from_corr(compute_corr(h), n)


def smooth_value(corr, tau):
    m = float(np.max(corr))
    if tau <= 1e-12:
        return m
    z = (corr - m) / float(tau)
    z = z - float(np.max(z))
    return m + float(tau) * float(np.log(np.sum(np.exp(z))))


def smooth_weights(corr, tau):
    p = np.zeros(corr.size, dtype=float)
    m = float(np.max(corr))
    if tau <= 1e-12:
        p[int(np.argmax(corr))] = 1.0
        return p
    z = (corr - m) / float(tau)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        p[int(np.argmax(corr))] = 1.0
        return p
    p[:] = e / s
    return p


def gradient_corr(h, p, n):
    off = n - 1
    L = 2 * n - 1
    cum = np.empty(L + 1, dtype=float)
    cum[0] = 0.0
    np.cumsum(p, out=cum[1:])

    ar = np.arange(n)
    lo = np.maximum(-off, ar - off)
    hi = np.minimum(off, ar)
    G = cum[hi + 1] - cum[lo + off]

    r = np.arange(-off, off + 1, dtype=int)
    idx = r + off
    negidx = -r + off
    w = p[idx] + p[negidx]

    need = 4 * n
    M = 1
    while M < need:
        M <<= 1
    pw = np.zeros(M, dtype=float)
    pw[np.mod(r, M)] = w
    ph = np.zeros(M, dtype=float)
    ph[:n] = h
    cross = np.fft.irfft(np.fft.rfft(ph) * np.fft.rfft(pw), M)[:n]

    g = G - cross
    g -= float(np.mean(g))
    return g


def continuous_optimize(start, n, end_time, tau_list, step0, rng):
    target = 0.5 * float(n)
    h = project_box_sum(np.clip(start, 0.0, 1.0), target)
    corr = compute_corr(h)
    best_h = h.copy()
    best_c = compute_c5_from_corr(corr, n)
    current = h.copy()
    dx = 2.0 / float(n)

    for tau in tau_list:
        step = max(0.03, float(step0) * 0.8)
        improved = False
        for _ in range(100000):
            if time.time() >= end_time - 0.003:
                break
            corr = compute_corr(current)
            cmax = float(np.max(corr))
            C = cmax * dx
            if C < best_c - 1e-15:
                best_c = C
                best_h = current.copy()

            p = smooth_weights(corr, float(tau))
            g = gradient_corr(current, p, n)
            if not np.all(np.isfinite(g)):
                break

            F0 = smooth_value(corr, float(tau))
            accepted = False
            for __ in range(26):
                cand = project_box_sum(current - step * g, target)
                c2 = compute_corr(cand)
                F2 = smooth_value(c2, float(tau))
                C2 = float(np.max(c2)) * dx
                if F2 <= F0 - 1e-13 or (float(tau) <= 0.35 and C2 <= C - 1e-12):
                    current = cand
                    accepted = True
                    improved = True
                    break
                step *= 0.5
                if step < 1e-7:
                    break
            if not accepted:
                break
            step = min(0.45, step * 1.12)

    return best_h, float(best_c)


def binary_local_search(start, n, rng, deadline, max_sweeps=400, restarts=5):
    target = n // 2
    h = np.zeros(n, dtype=float)
    order = np.argsort(start)[::-1]
    h[order[:target]] = 1.0

    corr = compute_corr(h)
    cur_c = float(np.max(corr))
    best_h = h.copy()
    best_cmax = cur_c
    best_c = best_cmax * (2.0 / float(n))

    off = n - 1
    L = 2 * n - 1
    dx = 2.0 / float(n)
    buf = np.empty(L, dtype=float)
    sweep = 0
    stall = 0

    while time.time() < deadline and sweep < max_sweeps:
        A = np.zeros((n, L), dtype=float)
        a = 1.0 - h[::-1]
        b = -h
        for v in range(n):
            A[v, v:v + n] += a
            s2 = off - v
            A[v, s2:s2 + n] += b

        zeros = np.where(h < 0.5)[0].tolist()
        ones = np.where(h > 0.5)[0].tolist()
        rng.shuffle(zeros)
        rng.shuffle(ones)

        improved = False
        stop = False
        for i in zeros:
            if stop or time.time() >= deadline:
                break
            Ai = A[i]
            for j in ones:
                np.subtract(Ai, A[j], out=buf)
                buf += corr
                buf[off] -= 2.0
                k1 = off + i - j
                k2 = off + j - i
                buf[k1] += 1.0
                buf[k2] += 1.0
                nc = float(np.max(buf))
                if nc < cur_c - 1e-12:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = buf.copy()
                    cur_c = nc
                    if cur_c < best_cmax:
                        best_cmax = cur_c
                        best_c = cur_c * dx
                        best_h = h.copy()
                    improved = True
                    stop = True
                    break

        sweep += 1
        if not improved:
            stall += 1
            if stall >= 2 and restarts > 0:
                restarts -= 1
                h = best_h.copy()
                corr = compute_corr(h)
                cur_c = best_cmax
                for _ in range(int(rng.randint(2, 6))):
                    oi = np.where(h > 0.5)[0]
                    zi = np.where(h < 0.5)[0]
                    if oi.size == 0 or zi.size == 0:

                    ai = int(rng.choice(oi))
                    bi = int(rng.choice(zi))
                    h[ai] = 0.0
                    h[bi] = 1.0
                corr = compute_corr(h)
                cur_c = float(np.max(corr))
                stall = 0
        if stall > 6 and restarts <= 0:
            break

    return best_h, float(best_c)


def resample_h(h, n_old, n_new):
    h = np.asarray(h, dtype=float)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return np.interp(x_new, x_old, h)


def make_transforms(h, n):

    out = []

    return out

                best_c = cc

    if time.time() < deadline - 60.0:
        end3 = min(deadline - 60.0, time.time() + min(0.05 * B, 70.0))
        if end3 > time.time():
            hh, cc = continuous_optimize(best_h, best_len, end3, TAU_MED, 0.07, rng)
            if cc < best_c - 1e-15:
                best_h = hh.copy()
                best_c = cc

    base_ns = [best_len]
    for cand in (252, 244, 260, 272, 288, 300):
        if cand % 2 == 0 and 160 <= cand <= 360 and cand not in base_ns:
            base_ns.append(cand)

    basin_end = deadline - max(30.0, 0.16 * B)
    trial = 0
    slice_s = max(10.0, min(30.0, 0.025 * B))

    while time.time() < basin_end and trial < 120:
        trial += 1
        if best_len not in base_ns:
            base_ns.insert(0, best_len)

        trans = make_transforms(best_h, best_len)
        base_h = trans[int(rng.randint(len(trans)))]
        n2 = base_ns[trial % len(base_ns)]

        if n2 == best_len:
            amp = max(0.004, 0.04 * (0.93 ** (trial // max(1, len(base_ns)))))
            hstart = project_box_sum(base_h + amp * rng.randn(best_len), 0.5 * float(best_len))
        else:
            hres = resample_h(base_h, best_len, n2)
            hstart = project_box_sum(hres + 0.015 * rng.randn(n2), 0.5 * float(n2))

        if trial % 12 == 6 and n2 % 2 == 0:
            hr = np.zeros(n2, dtype=float)
            idx = rng.permutation(n2)
            hr[idx[:n2 // 2]] = 1.0
            hstart = hr

        endt = min(basin_end - 1.0, time.time() + slice_s)
        if endt <= time.time():
            break
        hh, cc = continuous_optimize(hstart, n2, endt, TAU_SHORT, 0.06, rng)
        if cc < best_c - 1e-15:
            best_h = hh.copy()
            best_len = n2
            best_c = cc

    if time.time() < deadline - 12.0:
        endf = min(deadline - 10.0, time.time() + max(20.0, 0.05 * B))
        if endf > time.time():
            hh, cc = continuous_optimize(best_h, best_len, endf, TAU_MED, 0.05, rng)
            if cc < best_c - 1e-15:
                best_h = hh.copy()
                best_c = cc

    if best_len % 2 == 0 and time.time() < deadline - 15.0:
        endb = min(deadline - 6.0, time.time() + max(12.0, 0.04 * B))
        if endb > time.time():
            hh, cc = binary_local_search(best_h, best_len, rng, endb, 600, 6)
            if cc < best_c - 1e-15:
                best_h = hh.copy()
                best_c = cc
            if time.time() < deadline - 3.0:
                hh, cc = continuous_optimize(best_h, best_len, deadline - 2.0, TAU_SHORT, 0.05, rng)
                if cc < best_c - 1e-15:
                    best_h = hh.copy()
                    best_c = cc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, 0.5 * float(best_len))
    best_c = compute_c5(best_h, best_len)
    return best_h, float(best_c), int(best_len)
