import numpy as np
from scipy.optimize import minimize
import time
import math

OBJ_ALPHA = 1.0
OBJ_MU = 0.0


def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    return float(raw_corr(h).max() * (2.0 / n))


def project_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * len(h)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)


def interp_h(h, n):
    if len(h) == n:
        return project_feasible(h.copy())
    x0 = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(x1, x0, h))


def antisym_project(h):
    return project_feasible(0.5 * (h + 1.0 - h[::-1]))


def balanced_binary(h):
    n = len(h)
    out = np.zeros(n, dtype=float)
    out[np.argsort(h)[::-1][:n // 2]] = 1.0
    return out


def rand_binary(n, rng, mode=0):
    m = n // 2
    a = np.arange(m)
    if mode == 0:
        b = rng.random(m) < 0.5
    elif mode == 1:
        b = (np.floor(a * 0.6180339887498949 + rng.random()) % 1.0) < 0.5
    else:
        b = np.sin(a * (1.0 + rng.random() * 3.0) + rng.random() * 6.283) > 0.0
    h = np.empty(n, dtype=float)
    h[:m] = b.astype(float)
    h[m:] = 1.0 - b[::-1].astype(float)
    return h


def perturb_h(h, rng, steps=24, frac=0.25):
    n = len(h)
    h = h.copy()
    for _ in range(steps):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            hi = min(1.0 - h[i], h[j])
            if hi > 0.0:
                t = hi * frac * float(rng.random())
                h[i] += t
                h[j] -= t
    return project_feasible(h)


def grad_smooth_vec(h, C, alpha):
    n = len(h)
    off = n - 1
    dx = 2.0 / n
    c = C * dx
    cm = float(c.max())
    sh = alpha * (c - cm)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for r in range(len(C)):
        wr = w[r]
        if wr < 1e-12:
            continue
        m = r - off
        wm = wr * dx
        if m >= 0:
            L = n - m
            g[:L] += wm * (1.0 - h[m:])
            g[m:] -= wm * h[:L]
        else:
            d = -m
            L = n - d
            g[d:] += wm * (1.0 - h[:L])
            g[:L] -= wm * h[d:]
    return g


def smooth_value(h, C, alpha):
    dx = 2.0 / len(h)
    c = C * dx
    cm = float(c.max())
    sh = alpha * (c - cm)
    mx = float(sh.max())
    return cm + (mx + math.log(float(np.sum(np.exp(sh - mx))))) / alpha


def obj_global(x):
    n = len(x)
    C = raw_corr(x)
    v = smooth_value(x, C, OBJ_ALPHA)
    s = x.sum() - 0.5 * n
    return v + OBJ_MU * s * s


def grad_global(x):
    n = len(x)
    C = raw_corr(x)
    g = grad_smooth_vec(x, C, OBJ_ALPHA)
    g += 2.0 * OBJ_MU * (x.sum() - 0.5 * n)
    return g


def lbfgs_phase(h, deadline):
    global OBJ_ALPHA, OBJ_MU
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    mi = 40 if n > 250 else 80
    for alpha in (50.0, 500.0, 5000.0):
        if time.time() >= deadline:
            break
        OBJ_ALPHA = float(alpha)
        OBJ_MU = 1.0
        try:
            res = minimize(obj_global, h, jac=grad_global, method="L-BFGS-B",
                           bounds=bounds,
                           options={"maxiter": mi, "ftol": 1e-14, "gtol": 1e-8})
            h2 = project_feasible(res.x)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_h, h, best_c = h2.copy(), h2, c2
        except Exception:
            pass
    return best_h, best_c


def transfer_max(C, h, omh, hrev, off, n, dx, i, j, t):
    Ct = C.copy()
    t2 = t * t
    Ct[off - i:off - i + n] += t * omh
    Ct[i:i + n] -= t * hrev
    Ct[off - j:off - j + n] -= t * omh
    Ct[j:j + n] += t * hrev
    Ct[off] -= 2.0 * t2
    Ct[off + j - i] += t2
    Ct[off + i - j] += t2
    return float(Ct.max()) * dx


def apply_transfer(h, C, off, n, i, j, t):
    omh = 1.0 - h
    hrev = h[::-1]
    t2 = t * t
    C[off - i:off - i + n] += t * omh
    C[i:i + n] -= t * hrev
    C[off - j:off - j + n] -= t * omh
    C[j:j + n] += t * hrev
    C[off] -= 2.0 * t2
    C[off + j - i] += t2
    C[off + i - j] += t2
    h[i] += t
    h[j] -= t


def line_search_pair(C, h, omh, hrev, off, n, dx, i, j, hi, f0, detail):
    bt = 0.0
    bf = f0
    if detail <= 0:
        for q in (0.3333333333, 0.6666666666, 1.0):
            t = hi * q
            ft = transfer_max(C, h, omh, hrev, off, n, dx, i, j, t)
            if ft < bf:
                bf = ft
                bt = t
        return bt, bf

    m = max(4, int(detail))
    bq = 0
    for q in range(1, m + 1):
        t = hi * q / float(m)
        ft = transfer_max(C, h, omh, hrev, off, n, dx, i, j, t)
        if ft < bf:
            bf = ft
            bt = t
            bq = q

    if 1 <= bq <= m and hi > 1e-8:
        if bq == 1:
            a = 0.0
            b = hi * 2.0 / float(m)
        elif bq == m:
            a = hi * (m - 1) / float(m)
            b = hi
        else:
            a = hi * (bq - 1) / float(m)
            b = hi * (bq + 1) / float(m)
        if b > a:

                    break

                    if ft < bf:
                        bf = ft
                        bt = t
                w = (b - a) / 10.0
                if w <= 1e-10:
                    break
                a = max(0.0, bt - w)
                b = min(hi, bt + w)
    return bt, bf


def coordinate_phase(h, C, deadline, alpha=2000, k=35, rp=200, fc=40,

    n = len(h)
    off = n - 1
    dx = 2.0 / n

        g = grad_smooth_vec(h, C, alpha)
        recv = np.flatnonzero(h < 1.0 - 1e-12)

        if recv.size == 0 or give.size == 0:
            break

                    h[i] = oi

        nn = len(best_h)
        kk = 60 if nn <= 350 else 45
        det = 20 if nn > 500 else 24
        hf, _, cf = coordinate_phase(best_h.copy(), raw_corr(best_h), deadline - 1.0,
                                     alpha=30000, k=kk, rp=500, fc=150,
                                     detail=det, seed=seed, tag=999)
        hf = project_feasible(hf)
        cf = compute_c5(hf)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
