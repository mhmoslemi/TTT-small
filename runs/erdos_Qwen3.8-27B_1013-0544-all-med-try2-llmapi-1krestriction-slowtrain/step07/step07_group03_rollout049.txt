def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum=h_target."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s_room = room.sum()
            if s_room < 1e-14:
                break
            frac = diff / s_room
            h = np.clip(h + frac * room, 0.0, 1.0)
        else:
            s_h = h.sum()
            if s_h < 1e-14:
                break
            frac = diff / s_h
            h = np.clip(h * (1 + frac), 0.0, 1.0)
    return h

- This is equivalent to finding τ such that h_proj[i] = clip(h[i] - τ, 0, 1) and sum = target
- If sum(clip(h-τ,0,1)) is monotonically decreasing in τ (which it is), we can binary search for τ

```python
def project_feasible_exact(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Binary search for tau
    if s > target:
        # Need to decrease: h_proj = clip(h - tau, 0, 1)
        lo, hi = 0.0, 1.0
    else:
        # Need to increase: h_proj = clip(h + tau, 0, 1)
        lo, hi = -1.0, 0.0
    
    for _ in range(60):
        mid = (lo + hi) / 2
        if s > target:
            candidate = np.clip(h - mid, 0.0, 1.0)
            if candidate.sum() > target:
                lo = mid
            else:
                hi = mid
        else:
            candidate = np.clip(h + mid, 0.0, 1.0)
            if candidate.sum() < target:
                lo = mid
            else:
                hi = mid
    
    if s > target:
        return np.clip(h - (lo+hi)/2, 0.0, 1.0)
    else:
        return np.clip(h + (lo+hi)/2, 0.0, 1.0)
```

def c5_and_corr(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max()), corr

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            if np.clip(h - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            if np.clip(h + mid, 0.0, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    """Returns (smoothed_objective, gradient)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    shifted = alpha * (corr - cmax)
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    obj = cmax + log_sum / alpha
    w = exp_s / exp_s.sum()
    
    # Gradient: grad[j] = dx * (1 - S1[j] - S2[j])
    # S1[j] = corr1[3n-3-j] where corr1 = np.correlate(h, w, 'full')
    # S2[j] = corr2[3n-3-j] where corr2 = np.correlate(h, w[::-1], 'full')
    corr1 = np.correlate(h, w, 'full')
    corr2 = np.correlate(h, w[::-1], 'full')
    idx = 3 * n - 3 - np.arange(n)
    S1 = corr1[idx]
    S2 = corr2[idx]
    grad = dx * (1.0 - S1 - S2)
    return obj, grad

def projected_gradient(h, n, alpha, deadline, n_iters=5000, lr=None, momentum=0.9):
    """Nesterov accelerated projected gradient method."""
    h = h.copy()
    h_prev = h.copy()
    v = np.zeros(n)
    
    if lr is None:
        lr = n / 4.0  # initial learning rate
    
    best_c = compute_c5(h, n)
    h_best = h.copy()
    
    for t in range(n_iters):
        if time.time() > deadline:
            break
        
        # Nesterov step
        v_new = momentum * v
        y = h + v_new  # extrapolation point (approximate Nesterov)
        # Actually, standard Nesterov: y = h + momentum * (h - h_prev)
        # Then grad at y, then h_new = project(y - lr * grad(y))
        
        obj, grad_y = smooth_obj_grad(y, n, alpha)
        
        h_new = project_feasible(y - lr * grad_y, n)
        v = h_new - h
        h = h_new
        h_prev = h
        
        c = compute_c5(h, n)
        if c < best_c:

            h_best = h.copy()

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h, n, deadline, max_sweeps=200):
    """Greedy swap local search on binary solutions."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_delta_c = 0
            best_j = -1
            for j in ones_list:
                # Compute delta for swapping i (0->1) and j (1->0)
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-12 and nc - cur_c < best_delta_c:
                    best_delta_c = nc - cur_c
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                # Update corr
                delta = np.zeros(2 * n - 1)
                # Recompute for the actual swap
                # Actually, let me just recompute
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
