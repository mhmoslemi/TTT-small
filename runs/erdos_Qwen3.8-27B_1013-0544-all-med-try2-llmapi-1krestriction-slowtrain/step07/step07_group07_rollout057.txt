def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(len(x))

            zeros = np.append(zeros, a)

            stagnant = 0
        else:
            stagnant += 1
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def smooth_obj(x, n, alpha):
    c = compute_all_lags(x, n)
    return smooth_max(c, alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_round(h, n):
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, n_trials=300, max_rounds=8000):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    rng = np.random.RandomState(77)
    stagnant = 0
    for rnd in range(max_rounds):
        if time.time() > deadline or stagnant > 300:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        m_star = abs(k_star - (n - 1))
        best_improve = 0
        best_ab = None
        for _ in range(n_trials):
            if m_star > 0 and m_star < n and len(ones) > 0:
                if m_star > 0:
                    mask = np.array([ones[i] + m_star < n and h[ones[i] + m_star] < 0.5 for i in range(len(ones))])
                    bad = [ones[i] for i in range(len(ones)) if mask[i]]
                    if bad:
                        a = bad[rng.randint(len(bad))]
                    else:
                        a = ones[rng.randint(len(ones))]
                else:
                    a = ones[rng.randint(len(ones))]
            else:
                a = ones[rng.randint(len(ones))] if ones else 0
            b = zeros[rng.randint(len(zeros))] if zeros else 0
            if a == b:
                continue
            h[a] = 0.0
            h[b] = 1.0
            c_new = compute_c5(h, n)
            h[a] = 1.0
            h[b] = 0.0
            if c_new < best_c - 1e-15:
                if best_c - c_new > best_improve:
                    best_improve = best_c - c_new
                    best_ab = (a, b)
        if best_ab is not None:
            a, b = best_ab
            h[a] = 0.0
            h[b] = 1.0
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            best_c = compute_c5(h, n)
            stagnant = 0
        else:
            stagnant += 1
    return h, best_c

def subgradient_polish(h, n, deadline, n_iters=2000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    step = 0.01
    momentum = np.zeros(n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        m = abs(lag)
        grad = np.zeros(n)
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        elif m < n:
            if lag > 0:
                grad[:n - m] += (1.0 - h_curr[m:])
                grad[m:] -= h_curr[:n - m]
            else:
                grad[m:] += (1.0 - h_curr[:n - m])
                grad[:n - m] -= h_curr[m:]
        grad -= grad.mean()
        momentum = 0.9 * momentum + 0.1 * grad
        h_try = h_curr - step * momentum
        h_try = project_feasible(h_try, n)
        c_new = compute_c5(h_try, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_try.copy()
            h_curr = h_try
            step = min(step * 1.05, 0.1)
        else:
            step = max(step * 0.95, 0.001)
            h_curr = h_try
    return best_h, best_c

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h_src)

def optimize_pipeline(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP
    if time.time() < deadline - 30:
        for alpha in [5000, 30000, 100000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 2: Subgradient
    if time.time() < deadline - 15:
        h_sg, c_sg = subgradient_polish(best_h, n, min(deadline - 5, time.time() + 30))
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    # Phase 3: Binary round + search
    if time.time() < deadline - 10:
        h_bin = binary_round(best_h, n)
        h_bin, c_bin = binary_local_search(h_bin, n, min(deadline - 3, time.time() + (deadline - 10) * 0.5))
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    # Phase 4: Polish best
    if time.time() < deadline - 3:
        h_new = slsqp_optimize(best_h, n, 200000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
