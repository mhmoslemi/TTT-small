    dx = 2.0 / n
    old_v = h[j]
    
    # Precompute B: the "slope" of corr[ci] with respect to h[j]
    # For k≠0: B[ci] = (1-h[j+k])*1_{valid} - h[j-k]*1_{valid}
    # For k=0: the relationship is quadratic, not linear
    
    # Compute B (for k≠0 terms)
    B = np.zeros(2*n-1)

        b = 0.0
        if j+k >= 0 and j+k < n:
            b += (1.0 - h[j+k]) if k != 0 else 0.0

        for _ in range(n_swap):
            i, j = rng.choice(n, 2, replace=False)
            h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project_feasible(h_try, n)
        h_t, c_t = coord_descent_func(h_try, n, p4, max_sweeps=50000, seed=seed + 300 + trial)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()

    # Phase 5: Final coordinate descent
    if time.time() < dl - 5:
        h5, c5 = coord_descent_func(best_h.copy(), n, dl - 3, max_sweeps=100000, seed=seed + 500)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
