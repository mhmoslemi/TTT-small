def fast_corr(h, n):
    """Linear correlation of h with 1-h using FFT."""
    g = 1.0 - h
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    H = rfft(h, N)
    G = rfft(g, N)
    return irfft(np.conj(H) * G, N)[:size]

def project_feasible(h, n):

            h += (diff / tot) * room

def structured_start(n, rng):
    """Create a structured starting point."""
    # Use a "golden ratio" like placement
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.sort(np.floor(np.arange(half) * phi).astype(int) % n)
    h[positions] = 1.0
    return project_feasible(h, n)

def random_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 8):
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def pair_move_eval(h, c, i, j, d, n):
    """Evaluate the new max correlation after h[i]+=d, h[j]-=d.
    Returns (new_max, valid) where valid is True if the move is feasible."""
    if h[i] + d > 1.0 + 1e-12 or h[j] - d < -1e-12:
        return c.max(), False
    
    g = 1.0 - h
    # Δc[m] from h[i] += d:
    # = d * g[i+m-n+1] (if 0 ≤ i+m-n+1 < n) - d * h[i-m+n-1] (if 0 ≤ i-m+n-1 < n)
    # Δc[m] from h[j] -= d:
    # = -d * g[j+m-n+1] (if 0 ≤ j+m-n+1 < n) + d * h[j-m+n-1] (if 0 ≤ j-m+n+n-1 < n)
    
    # m ranges from 0 to 2n-2
    delta = np.zeros(2*n-1)
    # For h[i] += d:
    # g[i+m-n+1] valid when 0 ≤ i+m-n+1 < n => n-1-i ≤ m ≤ 2n-2-i
    m_lo = max(0, n-1-i)
    m_hi = min(2*n-2, 2*n-2-i)
    if m_lo <= m_hi:
        idx = i + np.arange(m_lo, m_hi+1) - n + 1
        delta[m_lo:m_hi+1] += d * g[idx]
    # h[i-m+n-1] valid when 0 ≤ i-m+n-1 < n => 1+i-n ≤ m ≤ i... 
    # wait: 0 ≤ i-m+n-1 < n => -i-1+n ≤ m ≤ i... 
    # i-m+n-1 ≥ 0 => m ≤ i+n-1
    # i-m+n-1 < n => m > i-1 => m ≥ i
    # So m ranges from i to min(2n-2, i+n-1), but m ≤ 2n-2 always
    # Actually m ≥ i and m ≤ i+n-1, but m ≤ 2n-2
    m_lo2 = i  # m ≥ i (from i-m+n-1 < n => m > i-1, so m ≥ i)
    m_hi2 = min(2*n-2, i + n - 1)
    if m_lo2 <= m_hi2:
        idx = i - np.arange(m_lo2, m_hi2+1) + n - 1
        delta[m_lo2:m_hi2+1] -= d * h[idx]
    
    # For h[j] -= d (opposite signs):
    m_lo = max(0, n-1-j)
    m_hi = min(2*n-2, 2*n-2-j)
    if m_lo <= m_hi:
        idx = j + np.arange(m_lo, m_hi+1) - n + 1
        delta[m_lo:m_hi+1] -= d * g[idx]
    m_lo2 = j
    m_hi2 = min(2*n-2, j + n - 1)
    if m_lo2 <= m_hi2:
        idx = j - np.arange(m_lo2, m_hi2+1) + n - 1
        delta[m_lo2:m_hi2+1] += d * h[idx]
    
    new_c = c + delta
    return new_c.max(), True
