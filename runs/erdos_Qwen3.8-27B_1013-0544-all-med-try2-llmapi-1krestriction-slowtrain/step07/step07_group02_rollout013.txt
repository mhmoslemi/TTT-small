            break

        else:
            s = h.sum()
            if s < 1e-14:
                break

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            new_corr0 = corr - B * h[i]
            new_corr1 = corr + B * (1.0 - h[i])
            val0 = np.max(new_corr0)
            val1 = np.max(new_corr1)
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(50):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            # hi_v is the minimum achievable max when varying h[i]
            # The optimal h[i] is at the boundary of the feasible interval
            # For the best x, we want the max to be exactly hi_v
            # x should be at one endpoint of the feasible interval
            x_opt = np.clip(hi_v, 0.0, 1.0)  # This isn't right...
            # Actually, we want to find x that minimizes max(corr + B*(x-h[i]))
            # The minimum is hi_v, achieved at some x in the feasible interval
            # Let's just use the midpoint of the feasible interval at level hi_v
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B] + 1e-15))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B] - 1e-15))
            if lb_x > ub_x:
                x_opt = np.clip((val0 - val1) / (val0 + val1 + 1e-30) if val0 != val1 else h[i], 0, 1)
            else:
                x_opt = (lb_x + ub_x) / 2.0
            x_opt = np.clip(x_opt, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        s = shifted.max()
        return m + (s + np.log(np.exp(shifted - s).sum())) / alpha
    def gfun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': gfun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.x is not None:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

```python
import numpy as np
from scipy.optimize import minimize
import time

# Global state for gradient computation
_gs = {'n': 0, 'alpha': 0}

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    n = _gs['n']
    alpha = _gs['alpha']
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    c = corr * dx
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    return m + (s + np.log(np.exp(shifted - s).sum())) / alpha

def _grad(x):
    n = _gs['n']
    alpha = _gs['alpha']
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    c = corr * dx
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    w = np.exp(shifted - s)
    w /= w.sum()
    b = 1.0 - x
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(w[n-1 : 2*n-1-j], b[0 : n-j])
        if j > 0:
            grad[j] -= np.dot(w[n-1 : j+n][::-1], x[0 : j+1])
    grad *= dx
    return grad

def _cons_fun(x):
    return x.sum() - x.size / 2.0

def _cons_jac(x):
    return np.ones_like(x)

def slsqp_with_grad(h, n, alpha, maxiter=10000, ftol=1e-18):
    _gs['n'] = n
    _gs['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(50):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            if lb_x <= ub_x:
                x_opt = (lb_x + ub_x) / 2.0
            else:
                x_opt = h[i]
            x_opt = np.clip(x_opt, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h

    total = budget_s - 20
    dl = t0 + total
    rng = np.random.RandomState(seed)

    p1 = t0 + 0.03 * total
    hc = coord_descent(best_h, n, max_sweeps=50, deadline=p1)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:

    # Phase 2: Progressive SLSQP with analytical gradient
    p2 = t0 + 0.60 * total
    alphas = [20, 100, 500, 2000, 10000, 50000, 200000,
              1000000, 5000000, 20000000, 100000000, 500000000, 2000000000]
    h = best_h.copy()
    for i, alpha in enumerate(alphas):

            break
        remaining = p2 - 5 - time.time()
        n_left = len(alphas) - i

        maxit = int(min(20000, max(2000, time_per * 500)))
        h = slsqp_with_grad(h, n, alpha, maxit, 1e-18)
        c = compute_c5(h, n)

    hc = coord_descent(best_h, n, max_sweeps=400, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)

        c = compute_c5(h, n)
