def active_gradient(h, c, n, alpha=2500.0, active_num=48):
    dx=2.0/n
    g=np.zeros(n)
    L=len(c)
    k=min(active_num,L)
    # use argpartition if k<L else all
    if k < L:
       idx = np.argpartition(c, L-k)[L-k:]  # top k? Actually np.argpartition(c, -k)[-k:]
    else: idx=np.arange(L)
    ct = c[idx]
    shift = ct - ct.max()
    w = np.exp(alpha*shift)
    wsum = w.sum()
    if wsum <=0: w[:] = 1.0/k
    else: w /= wsum
    for p, wt in zip(idx, w):

          N = n - lag
          if N <=0: continue
          g[lag:n] += wt*dx*(1.0 - h[:N])

          if N <=0: continue

          g[Lg:n] -= wt*dx*h[:N]

def golden_start(n):
    k = n // 2
    phi = 0.6180339887498948482
    t = np.arange(n)
    frac = (t * phi) % 1.0
    out = (frac < 0.5).astype(float)
    cnt = int(out.sum())
    if cnt != k:
       # adjust closest to 0.5
       diff = np.abs(frac - 0.5)
       order = np.argsort(diff)
       i=0
       while cnt < k and i < len(order):
          idx=order[i]
          if out[idx] == 0.0:
             out[idx]=1.0; cnt+=1
          i+=1
       i=0
       while cnt > k and i < len(order):
          idx=order[i]
          if out[idx] == 1.0:
             out[idx]=0.0; cnt-=1
          i+=1
    return out
```

def active_gradient(h,c,n,alpha=2500.0,active_num=48):
    dx=2.0/n
    g=np.zeros(n,float)
    m=len(c)
    k=min(active_num,m)
    if k <=0: return g
    if k < m:
       idx=np.argpartition(c, -k)[-k:]
    else:
       idx=np.arange(m)
    ct=c[idx]
    w=np.exp(alpha*(ct - ct.max()))
    sw=w.sum()

import numpy as np
import time

SMALL_STEPS = (1e-4, 1e-3, 1e-2, 2e-2, 5e-2, 1e-1, 2e-1, 5e-1)
GOLDEN_R = 0.6180339887498949


def compute_c_array(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def compute_c5(h, n):
    return float(np.max(compute_c_array(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def interpolate_h(h, n_old, n_new):
    h = np.asarray(h, dtype=float).ravel()
    n_old = min(n_old, len(h))
    if n_old == n_new:
        return h.copy()
    if n_old <= 0:
        return golden_start(n_new)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(xn, xo, h[:n_old])


def golden_start(n):
    k = n // 2
    phi = 0.6180339887498948482
    frac = (np.arange(n) * phi) % 1.0
    out = (frac < 0.5).astype(float)
    cnt = int(out.sum())
    if cnt != k:
        order = np.argsort(np.abs(frac - 0.5))
        for idx in order:
            if cnt <= k:
                break
            if out[idx] > 0.5:
                out[idx] = 0.0
                cnt -= 1
        for idx in order:
            if cnt >= k:
                break
            if out[idx] < 0.5:
                out[idx] = 1.0
                cnt += 1
    return out


def random_binary_start(n, rng):
    out = np.zeros(n)
    k = n // 2
    if k > 0 and k < n:
        out[rng.choice(n, size=k, replace=False)] = 1.0
    else:
        out[:k] = 1.0
    return out


def round_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    out = np.zeros(n)
    k = n // 2
    if k <= 0:
        return out
    if k >= n:
        out.fill(1.0)
        return out
    out[np.argpartition(-h, k)[:k]] = 1.0
    return out


def smooth_start(n, rng):

    h = np.full(n, 0.5)

def active_gradient(h, c, n, alpha=2000.0, active_num=64):
    dx = 2.0 / n
    g = np.zeros(n, dtype=float)
    m = len(c)
    k = min(int(active_num), m)
    if k <= 0:
        return g
    if k < m:
        idx = np.argpartition(c, -k)[-k:]
    else:
        idx = np.arange(m)
    ct = c[idx]
    w = np.exp(alpha * (ct - ct.max()))
    sw = w.sum()
    if sw > 0.0:
        w = w / sw
    else:
        w = np.full(k, 1.0 / k)
    mid = n - 1
    for pos in range(k):
        p = idx[pos]
        wt = w[pos]
        if wt == 0.0:
            continue
        lag = int(p) - mid
        if lag >= 0:
            N = n - lag
            if N <= 0:
                continue
            g[lag:n] += wt * dx * (1.0 - h[:N])
            g[:N] -= wt * dx * h[lag:n]
        else:
            Lg = -lag
            N = n - Lg
            if N <= 0:
                continue
            g[:N] += wt * dx * (1.0 - h[Lg:n])
            g[Lg:n] -= wt * dx * h[:N]
    return g


def pair_eval(delta, c, A, p0, p1, p2, dx, temp):
    np.multiply(A, delta, out=temp)
    temp += c
    d2 = dx * delta * delta
    temp[p0] -= 2.0 * d2
    temp[p1] += d2
    temp[p2] += d2
    return float(np.max(temp))


def golden_min_delta(lo, hi, c, A, p0, p1, p2, dx, temp, iters):
    if hi <= lo:
        return lo, pair_eval(lo, c, A, p0, p1, p2, dx, temp)
    c1 = hi - GOLDEN_R * (hi - lo)
    c2 = lo + GOLDEN_R * (hi - lo)
    f1 = pair_eval(c1, c, A, p0, p1, p2, dx, temp)
    f2 = pair_eval(c2, c, A, p0, p1, p2, dx, temp)
    for _ in range(int(iters)):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - GOLDEN_R * (hi - lo)
            f1 = pair_eval(c1, c, A, p0, p1, p2, dx, temp)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + GOLDEN_R * (hi - lo)
            f2 = pair_eval(c2, c, A, p0, p1, p2, dx, temp)
    d = 0.5 * (lo + hi)
    return d, pair_eval(d, c, A, p0, p1, p2, dx, temp)


def pair_line_search(h, c, i, j, n, dx, base_cmax, ones_rev, A, temp, grid_pts, refine_iters):
    a = h[i]
    b = h[j]
    lo = -a
    hi = 1.0 - b
    if hi <= lo + 1e-12:
        return None
    if lo < -1.0:
        lo = -1.0
    if hi > 1.0:
        hi = 1.0
    if hi <= lo:
        return None

    A.fill(0.0)
    A[i:i + n] += dx * ones_rev
    A[n - 1 - i:n - 1 - i + n] -= dx * h
    A[j:j + n] -= dx * ones_rev
    A[n - 1 - j:n - 1 - j + n] += dx * h

    p0 = n - 1
    p1 = n - 1 + i - j
    p2 = n - 1 + j - i

    best_v = pair_eval(0.0, c, A, p0, p1, p2, dx, temp)
    best_d = 0.0

    for s in SMALL_STEPS:
        for d in (s, -s):
            if lo <= d <= hi:
                v = pair_eval(d, c, A, p0, p1, p2, dx, temp)
                if v < best_v - 1e-15:
                    best_d = d
                    best_v = v

    if hi - lo > 1e-12 and grid_pts > 2:
        ds = np.linspace(lo, hi, int(grid_pts))
        for d in ds:
            d = float(d)
            v = pair_eval(d, c, A, p0, p1, p2, dx, temp)
            if v < best_v - 1e-15:
                best_d = d
                best_v = v

        gsize = max(1e-4, (hi - lo) / float(max(2, int(grid_pts))))
        left = max(lo, best_d - 2.0 * gsize)
        right = min(hi, best_d + 2.0 * gsize)
        if right - left > 1e-12:
            gd, gv = golden_min_delta(left, right, c, A, p0, p1, p2, dx, temp, refine_iters)
            if gv < best_v:
                best_d = gd
                best_v = gv

    if best_v < base_cmax - 1e-13:
        return float(best_d), float(best_v)
    return None


def apply_pair_move(h, c, i, j, delta, n, cmax):
    old_i = h[i]
    old_j = h[j]
    hi = old_i + delta
    hj = old_j - delta
    if hi < 0.0:
        hi = 0.0
    elif hi > 1.0:
        hi = 1.0
    if hj < 0.0:
        hj = 0.0
    elif hj > 1.0:
        hj = 1.0
    h[i] = hi
    h[j] = hj
    c_new = compute_c_array(h, n)
    cnew = float(c_new.max())
    if cnew < cmax - 1e-13:
        return True, c_new, cnew
    h[i] = old_i
    h[j] = old_j
    return False, c, cnew

        b = min(rem - 60.0, max(30.0, 250.0))
        try:
            h1, c1 = optimize_pair_cd(h2, n2, time.time() + b, seed + 2, 200, 64, 31, 18)
            if c1 < best_c:
                best_h, best_n, best_c = h1, n2, c1
        except Exception:
            pass

    # Try a finer grid; this often reduces the discretization quantization.
    rem = deadline - time.time()
    if best_n < 990 and rem > 260.0:
        n3 = 990
        h3 = project_feasible(interpolate_h(best_h, best_n, n3), n3)
        b = min(rem - 120.0, max(60.0, 300.0))
        try:
            h1, c1 = optimize_pair_cd(h3, n3, time.time() + b, seed + 3, 180, 64, 31, 18)
            if c1 < best_c:
                best_h, best_n, best_c = h1, n3, c1
        except Exception:
            pass

    # Basin hopping from diverse deterministic / randomized starts.
    starts = []
    starts.append(round_binary(best_h, best_n))
    starts.append(golden_start(best_n))
    starts.append(random_binary_start(best_n, rng))
    starts.append(project_feasible(best_h + 0.05 * rng.randn(best_n), best_n))
    starts.append(project_feasible(best_h + 0.10 * rng.randn(best_n), best_n))

    for s in starts:
        rem = deadline - time.time()
        if rem < 80.0:
            break
        b = min(rem - 40.0, 70.0)
        if b < 25.0:
            break
        try:
            h1, c1 = optimize_pair_cd(s, best_n, time.time() + b, seed + len(starts) + 50, 140, 56, 25, 15)
            if c1 < best_c:
                best_h, best_n, best_c = h1, best_n, c1
        except Exception:
            pass

    # Final polish.
    rem = deadline - time.time()
    if rem > 25.0:
        try:
            h1, c1 = optimize_pair_cd(best_h, best_n, deadline - 1.0, seed + 999, 180, 64, 33, 20)
            if c1 < best_c:
                best_h, best_n, best_c = h1, best_n, c1
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
