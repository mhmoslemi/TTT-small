def local_search(h, n, deadline, top_m=3):
    h = h.copy()
    best_c = compute_c5(h, n)

    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d_base in deltas:
        if time.time() > deadline:
            break
        improved_in_pass = True
        while improved_in_pass:
            if time.time() > deadline:
                break
            improved_in_pass = False
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-top_m:]

            for k_idx in top_idx:
                if time.time() > deadline:
                    break
                lag = k_idx - (n - 1)
                m = abs(lag)
                if m == 0 or m >= n:
                    continue

                if lag > 0:
                    # Move mass from i to i+m (decrease h[i], increase h[i+m])
                    for i in range(n - m):
                        if time.time() > deadline:
                            break
                        j = i + m
                        if h[i] > h[j]:
                            delta = min(d_base, (h[i] - h[j]) * 0.5, h[i] * 0.8, (1 - h[j]) * 0.8)
                            if delta < 1e-9:
                                continue

                            h[j] += delta

                                h[j] -= delta
