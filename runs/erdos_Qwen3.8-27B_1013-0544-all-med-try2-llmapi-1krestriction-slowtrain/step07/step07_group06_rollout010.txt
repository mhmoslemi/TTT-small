def smooth_max_and_grad(h, n, alpha, top_M=None):
    """Compute smooth max value and gradient. Optionally only over top M."""
    c = compute_c5_vec(h, n)
    if top_M is not None and top_M < len(c):
        top_idx = np.argpartition(c, -top_M)[-top_M:]
        c_work = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_work = c
    
    m = c_work.max()
    shifted = alpha * (c_work - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    log_sum_exps = mx + np.log(np.sum(exps))
    value = m + log_sum_exps / alpha
    w = exps / np.sum(exps)

            return g

    c_vec = compute_c5_vec(h_bin, n)

        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]

        h_test = h_bin.copy()
        h_test[i] = 1
        h_test[j] = 0

            best_h = h_test
            h_bin = h_test
            c_vec = compute_c5_vec(h_bin, n)

            T *= 0.95

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.
