            if tot < 1e-15:

    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_smooth(h, n, deadline, alphas, max_iters=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_val(x, n, alpha)
        def gr(x):
            return grad_smooth(x, n, alpha)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 40, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_smooth(h, n, deadline, alpha, lr_init=0.005, max_steps=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h_prev = h.copy()
    t = 1
    for step in range(max_steps):
        if time.time() > deadline:
            break
        momentum = (t - 1) / (t + 2)
        y = h + momentum * (h - h_prev)
        g = grad_smooth(y, n, alpha)
        lr = lr_init / (1.0 + step * 1e-4)
        h_new = project_feasible(y - lr * g, n)
        h_prev = h
        h = h_new
        t += 1
        c = compute_c5(h, n)
        if c < best_c - 1e-16:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, tol=1e-7, max_rounds=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        def obj(x):
            cx = compute_c5_vec(x, n)
            return float(cx[active].max())
        def gr(x):
            cx = compute_c5_vec(x, n)
            cm = cx[active].max()
            act = cx[active] >= cm - 1e-10
            dx = 2.0 / n
            g = np.zeros(n)
            for idx in np.where(act)[0]:
                mi = active[idx]
                k = mi - (n - 1)
                if k >= 0:
                    nv = n - k
                    g[:nv] += (1.0 - x[k:])
                    g[k:] += (-x[:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:] += (1.0 - x[:nv])
                    g[:nv] += (-x[kk:])
            return g * dx / act.sum()
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 100, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
            tol *= 0.5
        except Exception:
            break
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=60):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_pair_search(h, n, deadline, deltas, alpha=10000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth(h, n, alpha)
            order = np.argsort(-np.abs(g))
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                neighbors = list(range(max(0, i-40), min(n, i+41)))
                neighbors.remove(i)
                for j in neighbors:
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def random_restart_optimize(h0, n, deadline, rng, n_restarts=3):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    for _ in range(n_restarts):
        if time.time() > deadline - 60:
            break
        pert = rng.random(n) * 0.1
        h_pert = project_feasible(h0 + pert - 0.05, n)
        alloc = max(30, deadline - 60 - time.time())
        h_r, c_r = lbfgs_smooth(h_pert, n, time.time() + alloc * 0.5,
                                 [1000, 10000, 100000], max_iters=100)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(250, 0.5)
        n = 250
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Progressive L-BFGS-B
    t1 = min(t0 + 200, deadline - 500)
    best_h, best_c = lbfgs_smooth(best_h, n, t1,
                                   [50, 200, 1000, 5000, 50000, 200000, 1000000],
                                   max_iters=400)

    # Phase 2: Nesterov acceleration with high alpha
    if time.time() < deadline - 400:
        t2 = min(time.time() + 150, deadline - 400)
        h2, c2 = nesterov_smooth(best_h, n, t2, alpha=500000, lr_init=0.003, max_steps=30000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Active set refinement
    if time.time() < deadline - 300:
        t3 = min(time.time() + 60, deadline - 300)
        h3, c3 = active_set_lbfgs(best_h, n, t3, tol=1e-7, max_rounds=80)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Very high alpha L-BFGS-B
    if time.time() < deadline - 250:
        t4 = min(time.time() + 80, deadline - 250)
        h4, c4 = lbfgs_smooth(best_h, n, t4,
                               [5000000, 20000000, 100000000],
                               max_iters=500)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Gradient-guided pair search (coarse)
    if time.time() < deadline - 150:
        t5 = min(time.time() + 80, deadline - 150)
        h5, c5v = gradient_pair_search(best_h, n, t5,
                                        [0.01, 0.005, 0.001], alpha=50000)
        if c5v < best_c:
            best_c, best_h = c5v, h5

    # Phase 6: Random restarts
    if time.time() < deadline - 100:
        t6 = min(time.time() + 60, deadline - 100)
        h6, c6 = random_restart_optimize(best_h, n, t6, rng, n_restarts=3)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Fine pair search
    if time.time() < deadline - 60:
        t7 = min(time.time() + 40, deadline - 60)
        h7, c7 = pair_search(best_h, n, t7,
                              [0.001, 0.0005, 0.0002, 0.0001], radius=50)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Ultra-fine pair search
    if time.time() < deadline - 20:
        t8 = min(time.time() + 15, deadline - 18)
        h8, c8 = pair_search(best_h, n, t8,
                              [0.00005, 0.00002, 0.00001], radius=30)
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Phase 9: Try other n
    if time.time() < deadline - 15:
        for n_try in [250, 400]:
            if time.time() > deadline - 12:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = max(8, deadline - 12 - time.time())
            h_t, c_t = lbfgs_smooth(h_interp, n_try, time.time() + alloc * 0.6,
                                     [50000, 500000, 5000000], max_iters=200)
            if time.time() < deadline - 5:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 4, deadline - 4),
                                        [0.001, 0.0001], radius=30)
            if c_t < best_c:
                best_c, best_h = c_t, h_t

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
