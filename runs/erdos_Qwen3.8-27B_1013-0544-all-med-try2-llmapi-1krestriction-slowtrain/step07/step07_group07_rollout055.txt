                grad[:n - m] += w[i] * (2.0 / n) * (1.0 - h_curr[m:])
                grad[m:] += -w[i] * (2.0 / n) * h_curr[:n - m]
        step = eta / np.sqrt(t + 1)
        h_new = project_feasible(h_curr - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
        else:
            h_curr = project_feasible(h_curr - 0.5 * step * grad, n)
            c_try = compute_c5(h_curr, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_curr.copy()
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_rounds=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            if lag == 0:
                break
            m = 0
        m = abs(lag)
        if m >= n:
            break
        improved = False
        if lag > 0:
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-6:
                    for delta in [0.01, 0.005, 0.001, 0.0005]:
                        d = min(delta, h[i] - h[j])
                        if d < 1e-8:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        h[i] += d
                        h[j] -= d
                elif h[j] > h[i] + 1e-6:
                    for delta in [0.01, 0.005, 0.001, 0.0005]:
                        d = min(delta, h[j] - h[i])
                        if d < 1e-8:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        h[i] -= d
                        h[j] += d
        else:
            for i in range(m, n):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-6:
                    for delta in [0.01, 0.005, 0.001, 0.0005]:
                        d = min(delta, h[i] - h[j])
                        if d < 1e-8:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        h[i] += d
                        h[j] -= d
                elif h[j] > h[i] + 1e-6:
                    for delta in [0.01, 0.005, 0.001, 0.0005]:
                        d = min(delta, h[j] - h[i])
                        if d < 1e-8:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        h[i] -= d
                        h[j] += d
        if not improved:
            if time.time() > deadline:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline - 20:
            break
        alloc = min(120, (deadline - time.time()) * 0.3)
        alloc_end = time.time() + alloc
        h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 30:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 25:
                break
            h_new = active_set_slsqp(best_h, n, 1000000, top_m, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 20:
        h_sub, c_sub = subgradient_polish(best_h, n, min(deadline - 10, time.time() + 60))
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub
    if time.time() < deadline - 5:
        h_cd, c_cd = coordinate_descent(best_h, n, min(deadline - 1, time.time() + 80))
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = None
    best_c = float('inf')
    n_try_list = [n_init]
    for n_alt in [200, 250, 300, 350, 400]:
        if n_alt != n_init and n_alt not in n_try_list:
            n_try_list.append(n_alt)
    n_try_list.sort()
    for n_try in n_try_list:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h) and best_c < 0.381:
            continue
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif n_try == n_init:
            h_start = h_init.copy()
        else:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(70, (deadline - time.time() - 50) * 0.5)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 2000000, maxiter=5000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 10:
            h_as = active_set_slsqp(best_h, n, 5000000, 3, maxiter=5000)
            c_as = compute_c5(h_as, n)
            if c_as < best_c:
                best_c = c_as
                best_h = h_as
        if time.time() < deadline - 5:
            best_h, best_c = coordinate_descent(best_h, n, deadline - 1)
    if best_h is None:
        best_h = project_feasible(h_init.copy(), n_init)
        best_c = compute_c5(best_h, n_init)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
