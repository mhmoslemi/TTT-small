def smooth_max(values, alpha):
    """Compute smooth max using log-sum-exp."""
    shifted = alpha * values - alpha * values.max()
    return values.max() + np.log(np.exp(shifted).sum()) / alpha

def project_feasible(h, n):
    """Project h to satisfy 0≤h≤1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            room = 1.0 - h

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    target = n / 2.0

    def obj(x):
        h_local = np.clip(x, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    w_rev = w[::-1]  # reversed w, length 2n-1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad_fft(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exp_vals = np.exp(shifted - sm)
    w = exp_vals / exp_vals.sum()
    u = 1.0 - h
    w_rev = w[::-1]
    L = 1
    while L < 3 * n + 2:
        L *= 2
    W = np.fft.rfft(w, L)
    Wr = np.fft.rfft(w_rev, L)
    U = np.fft.rfft(u, L)
    H = np.fft.rfft(h, L)
    conv_wu = np.fft.irfft(W * U, L)
    conv_wrh = np.fft.irfft(Wr * H, L)
    idx = np.arange(n) + n - 1
    A = conv_wu[idx]
    B = conv_wrh[idx]
    grad = dx * (A - B)
    f = m + np.log(exp_vals.sum()) / alpha
    return grad, f

def optimize_lbfgs(h, n, alpha, deadline, maxiter=3000, pen_coeff=100.0):
    target = n / 2.0

    def obj_grad(x):
        h_clip = np.clip(x, 0.0, 1.0)
        grad, f = smooth_max_grad_fft(h_clip, n, alpha)
        sd = h_clip.sum() - target
        grad = grad + 2.0 * pen_coeff * sd
        f = f + pen_coeff * sd * sd
        return f, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxls': 40, 'maxfun': maxiter * 4})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)

    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    for _ in range(max_sweeps):

            break
        improved = False

                imn = i - m_range

                delta[msk] -= h[imn[msk]]
                jm = j + m_range
