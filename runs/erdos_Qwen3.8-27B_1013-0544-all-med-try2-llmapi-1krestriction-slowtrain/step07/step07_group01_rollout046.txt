def _lbfgs_obj(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    a = alpha * (c - m)
    mx = a.max()
    if mx > 500:
        val = m + (mx + np.log(np.exp(a - mx).sum())) / alpha
    else:
        val = m + np.log(np.exp(a).sum()) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    a = alpha * (c - m)
    mx = a.max()
    if mx > 500:
        return m + (mx + np.log(np.exp(a - mx).sum())) / alpha
    return m + np.log(np.exp(a).sum()) / alpha

def _slsqp_cons(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

    configs = [(100, 1.0), (500, 5.0), (2000, 20.0), (10000, 100.0),

    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n

def c_new_check(h, n):
    return compute_c5(h, n)
