```python
    mx = shifted.max()

    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(idx)):
        mi = idx[ii]
        wm = w[ii] * dx
        shift = mi - (n - 1)
        if shift >= 0:
            # d/dh[j] = (1-h[j-shift]) for j in [shift, n-1] - h[j+shift] for j in [0, n-1-shift]

                grad[ss:] -= wm * h[:n-ss]
    return grad, c_raw * dx
```

The term sum_i h[i+shift]*h[i] is quadratic (a bilinear form). So f_k(h) = dx * (A_k - h^T Q_k h) where A_k is linear and Q_k is a matrix.

Actually: sum_i h[i+shift] * (1-h[i]) = sum_i h[i+shift] - sum_i h[i+shift]*h[i]

The first term is linear. The second term h^T Q h where Q has Q[i,i+shift] = 0.5 (symmetrized).

So f_k(h) is actually linear minus a quadratic form. For shift >= 0:
sum_i h[i+shift]*(1-h[i]) = sum_{i=0}^{n-1-shift} h[i+shift] - sum_{i=0}^{n-1-shift} h[i+shift]*h[i]

The quadratic part is -sum h[i+shift]*h[i], which is -h^T (Q_k) h where Q_k has (Q_k)[i, i+shift] = (Q_k)[i+shift, i] = 0.5.

So f_k(h) = (linear) - h^T Q_k h.

For the constraint f_k(h) <= t, this is: (linear) - h^T Q_k h <= t, or equivalently h^T Q_k h - (linear) + t >= 0.

This is not a standard convex constraint in general (depends on whether Q_k is PSD or NSD).

Q_k has entries 0.5 at positions (i, i+shift) and (i+shift, i). The eigenvalues of such a matrix depend on the structure. For shift=1, Q_1 is like a weighted adjacency matrix of a path graph, which has both positive and negative eigenvalues. So the constraint is non-convex in general.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha, topk=None):
    c_raw = np.correlate(h, 1 - h, mode='full')
    num = 2 * n - 1
    if topk is not None and topk < num:
        idx = np.argsort(c_raw)[::-1][:topk]
    else:
        idx = np.arange(num)
    ck = c_raw[idx]
    m = ck.max()
    shifted = alpha * (ck - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(idx)):
        mi = idx[ii]
        wm = w[ii] * dx
        shift = mi - (n - 1)
        if shift >= 0:
            if shift == 0:
                grad[:] += wm * (1.0 - 2.0 * h)
            else:
                grad[shift:] += wm * (1.0 - h[:n - shift])
                grad[:n - shift] -= wm * h[shift:]
        else:
            ss = -shift
            grad[:n - ss] += wm * (1.0 - h[ss:])
            grad[ss:] -= wm * h[:n - ss]
    return grad, c_raw * dx

def smooth_max_obj(h, n, alpha, topk=None):
    c = compute_corr(h, n)
    if topk is not None and topk < len(c):
        ck = np.sort(c)[::-1][:topk]
    else:
        ck = c
    m = ck.max()
    shifted = alpha * (ck - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def adam_adapt(h, n, deadline, alpha=200.0, topk_start=80, topk_end=5, lr=0.01):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 6000
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        topk = int(topk_start + (topk_end - topk_start) * frac)
        topk = max(topk, 3)
        g, _ = grad_smooth_max(h, n, alpha, topk)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        lr_cur = lr * (1.0 + 0.3 * t) ** (-0.5)
        h_new = project_feasible(h - lr_cur * m_hat / (np.sqrt(v_hat) + eps), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # restart with perturbation
                h = best_h + 0.01 * np.random.randn(n)
                h = project_feasible(h, n)
                m_arr[:] = 0.0
                v_arr[:] = 0.0
                stall = 0
    return best_h, best_c

def grad_pair_search(h, n, deadline, deltas, use_line_search=True):
    h = h.copy()
    best_c = compute_c5(h, n)
    np.random.seed(42)
    while time.time() < deadline:
        g, _ = grad_smooth_max(h, n, 500.0, topk=10)
        sorted_idx = np.argsort(g)
        n_try = min(40, n)
        top_neg = sorted_idx[:n_try // 2]
        top_pos = sorted_idx[-(n_try // 2):]
        improved = False
        for d in deltas:
            if time.time() > deadline:
                break
            for ii in range(len(top_neg)):
                if time.time() > deadline:
                    return h, best_c
                i = top_neg[ii]
                for jj in range(len(top_pos)):
                    j = top_pos[jj]
                    if i == j:
                        continue
                    if g[i] > g[j]:
                        break
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                        if use_line_search and improved:
                            # Try larger d
                            d2 = d * 2
                            while d2 < 1.0 and h[i] + d2 <= 1.0 and h[j] - d2 >= 0.0 and time.time() < deadline:
                                h[i] = h[i] - (d - d2) if False else h[i]
                                h[i] += d2 - d
                                h[j] -= d2 - d
                                c2 = compute_c5(h, n)
                                if c2 < best_c - 1e-14:
                                    best_c = c2
                                    d = d2
                                else:
                                    h[i] -= d2 - d
                                    h[j] += d2 - d
                                    break
                                d2 *= 2
            if not improved:
                break
    return h, best_c

def block_flip(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    np.random.seed(123)
    block_sizes = [3, 5, 7, 10, 15, 20]
    for _ in range(5000):
        if time.time() > deadline:
            break
        L = block_sizes[np.random.randint(len(block_sizes))]
        a = np.random.randint(0, max(1, n - L))
        block = h[a:a+L]
        if block.sum() < 0.1 * L or block.sum() > 0.9 * L:
            continue
        h_new = h.copy()
        h_new[a:a+L] = 1.0 - block
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-14:
            best_c = c
            h = h_new
    return h, best_c

def lbfgs_smooth(h, n, deadline, alpha=200.0, topk=None, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, alpha, topk)
    def gr(x):
        g, _ = grad_smooth_max(x, n, alpha, topk)
        return g
    try:
        res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iters, 'gtol': 1e-12, 'maxls': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def fine_pair_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            g, _ = grad_smooth_max(h, n, 1000.0, topk=8)
            sorted_idx = np.argsort(g)
            for i in sorted_idx[:25]:
                if time.time() > deadline:
                    return h, best_c
                if g[i] >= 0:
                    break
                for j in sorted_idx[-25:]:
                    if i == j:
                        continue
                    if g[j] <= -g[i]:
                        if g[i] - g[j] > 0:
                            break
                        continue
                    if g[j] < g[i]:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c
```
