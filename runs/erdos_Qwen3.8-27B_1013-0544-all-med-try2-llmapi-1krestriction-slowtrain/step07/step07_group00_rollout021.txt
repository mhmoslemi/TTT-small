LBFGS_CONFIGS = ((50.0, 0.01), (200.0, 0.1), (1000.0, 1.0), (5000.0, 10.0), (20000.0, 100.0), (100000.0, 500.0))

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(25):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += diff * room / tot
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += diff * h / tot
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 0.0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        v = h[idx] + diff
        if 0.0 <= v <= 1.0:
            h[idx] = v
    return h

def softmax_weights(c, cmax, alpha):
    t = alpha * (c - cmax)
    tmax = float(np.max(t))
    e = np.exp(t - tmax)
    s = float(e.sum())
    if s <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def weighted_grad(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n, dtype=float)
    w0 = w[n - 1]
    if w0 > 1e-20:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        nv = n - d
        wp = w[n - 1 + d]
        wn = w[n - 1 - d]
        if wp > 1e-20:
            g[:nv] += wp * (1.0 - h[d:])
            g[d:] -= wp * h[:nv]
        if wn > 1e-20:
            g[d:] += wn * (1.0 - h[:nv])
            g[:nv] -= wn * h[d:]
    g *= dx
    return g

def smooth_obj(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, n)
    cmax = float(np.max(c))
    t = alpha * (c - cmax)
    tmax = float(np.max(t))
    e = np.exp(t - tmax)
    s = float(e.sum())
    if s <= 1e-300:
        val = cmax
    else:
        val = cmax + (tmax + np.log(s)) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, n)
    cmax = float(np.max(c))
    w = softmax_weights(c, cmax, alpha)
    g = weighted_grad(h, n, w)
    return g + 2.0 * mu * (h.sum() - n / 2.0)

def fast_lbfgs(h0, n, deadline, configs=None):
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    if configs is None:
        configs = LBFGS_CONFIGS
    if time.time() > deadline - 3:
        return h, best_c
    bounds = [(0.0, 1.0)] * n
    cap = 700 if n <= 600 else 420
    for alpha, mu in configs:
        if time.time() > deadline - 6:
            break
        try:
            tt = time.time()
            smooth_obj(h, n, alpha, mu)
            smooth_grad(h, n, alpha, mu)
            dt = max(1e-4, time.time() - tt)
            per = max(0.008, 1.6 * dt)
            maxit = int(max(25, min(cap, (deadline - time.time() - 5.0) / per * 0.45)))
            if maxit < 25:
                break
            res = minimize(smooth_obj, h, args=(n, alpha, mu), jac=smooth_grad,
                           method="L-BFGS-B", bounds=bounds,
                           options={"maxiter": maxit, "maxfun": max(200, 12 * maxit),
                                    "ftol": 1e-17, "gtol": 1e-11, "maxls": 40})
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                h = h2
                best_c = c2
        except Exception:
            continue
    return h, best_c

def local_pair(h0, n, deadline, pool=80, window=60, alphas=(2000.0, 20000.0, 100000.0),
               max_accept=20, fracs=(0.5, 0.25, 0.75, 1.0, 0.125, 0.875)):
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    pool = min(pool, max(5, n - 1))
    window = max(5, window)
    for alpha in alphas:
        while time.time() < deadline - 1.0:
            cvec = compute_c5_vec(h, n)
            m = float(np.max(cvec))
            if m < best_c:
                best_c = m
            w = softmax_weights(cvec, m, alpha)
            g = weighted_grad(h, n, w)
            order = np.argsort(g)
            accepted = 0
            improved = False
            for a in range(pool):
                if time.time() > deadline - 1.0:
                    break
                i = int(order[a])
                gi = g[i]
                hi = h[i]
                if hi >= 1.0:
                    continue
                b0 = a + 1
                b1 = min(n, b0 + window)
                for b in range(b0, b1):
                    if time.time() > deadline - 1.0:
                        break
                    j = int(order[b])
                    gj = g[j]
                    if gj - gi < 1e-14:
                        continue
                    tmax = min(1.0 - hi, h[j])
                    if tmax < 1e-9:
                        continue
                    oi = h[i]
                    oj = h[j]
                    local_best = best_c
                    bt = 0.0
                    for f in fracs:
                        t = f * tmax
                        h[i] = oi + t
                        h[j] = oj - t
                        cv = compute_c5(h, n)
                        if cv < local_best - 1e-14:
                            local_best = cv
                            bt = t
                        if time.time() > deadline - 1.0:
                            break
                    if bt == 0.0 and tmax > 0.08:
                        for ts in (0.01, 0.02):
                            if ts > tmax:
                                break
                            h[i] = oi + ts
                            h[j] = oj - ts
                            cv = compute_c5(h, n)
                            if cv < local_best - 1e-14:
                                local_best = cv
                                bt = ts
                            if time.time() > deadline - 1.0:
                                break
                    if bt > 0.0:
                        h[i] = oi + bt
                        h[j] = oj - bt
                        best_c = local_best
                        hi = h[i]
                        improved = True
                        accepted += 1
                        if accepted >= max_accept:
                            break
                    else:
                        h[i] = oi
                        h[j] = oj
                if accepted >= max_accept:
                    break
            if not improved:
                break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def fourier_start(n, rng, modes=5):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, modes + 1):
        amp = rng.uniform(0.02, 0.32)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * m * x + ph)
    return project_feasible(h, n)

def sine_scan(n, deadline):
    best_h = None
    best_c = 1e100
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for freq in (1, 2, 3):
        w = np.pi * freq * x
        for amp in (0.5, 0.75, 1.0):
            for ph in np.linspace(0.0, 2.0 * np.pi, 20, endpoint=False):
                if time.time() > deadline:
                    break
                h = project_feasible(0.5 + 0.5 * amp * np.sin(w + ph), n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h
    s1 = np.sin(np.pi * x)
    s2 = np.sin(2.0 * np.pi * x)
    for a1 in (-1.0, 0.0, 1.0):
        for a2 in (-1.0, 0.0, 1.0):
            if a1 == 0.0 and a2 == 0.0:
                continue
            if time.time() > deadline:
                break
            h = project_feasible(0.5 + 0.25 * (a1 * s1 + a2 * s2), n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h
    if best_h is None:
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c

def pwls_obj(ctrl, n, knots, xgrid):
    ctrl = np.clip(ctrl, 0.0, 1.0)
    h = np.interp(xgrid, knots, ctrl)
    h = project_feasible(h, n)
    return compute_c5(h, n)

def run_pwls(h0, n, rng, deadline):
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    if time.time() > deadline:
        return h, best_c
    K = int(min(14, max(8, n // 50 + 6)))
    xg = np.linspace(0.0, 2.0, n, endpoint=False)
    kn = np.linspace(0.0, 2.0, K, endpoint=True)
    starts = [np.interp(kn, xg, h)]
    starts.append(rng.uniform(0.0, 1.0, K))
    starts.append(0.5 + 0.5 * np.sin(np.pi * np.arange(K) / float(max(1, K - 1))))
    for s in starts:
        if time.time() > deadline - 2:
            break
        s = np.clip(s, 0.0, 1.0)
        tt = time.time()
        pwls_obj(s, n, kn, xg)
        per = max(1e-3, 1.5 * (time.time() - tt))
        maxit = int(max(40, min(350, (deadline - time.time() - 2.0) / per * 0.40)))
        try:
            res = minimize(pwls_obj, s, args=(n, kn, xg), method="Nelder-Mead",
                           options={"maxiter": maxit, "fatol": 1e-9, "xatol": 1e-5})
        except Exception:
            continue
        ctrl = np.clip(res.x, 0.0, 1.0)
        h2 = project_feasible(np.interp(xg, kn, ctrl), n)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h, best_c = h2, c2
    return h, best_c

    t0 = time.time()

        h_init = fourier_start(n0, rng)

    best_h = project_feasible(h_init, n0)
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    reserve_final = max(60, int(0.22 * T) + 30)
    reserve_other = max(50, int(0.25 * T) + 20)

    try:
        end1 = max(t0 + 5.0, min(t0 + 0.30 * T, deadline - reserve_final - reserve_other - 20.0))
        best_h, best_c = fast_lbfgs(best_h, best_n, end1, LBFGS_CONFIGS)

        end2 = max(time.time() + 5.0, min(time.time() + 0.15 * T, deadline - reserve_final - reserve_other))
        best_h, best_c = local_pair(best_h, best_n, end2)

        end3 = max(time.time() + 5.0, min(time.time() + 0.25 * T, deadline - reserve_final))

        if time.time() < end3 - 25.0:
            h3, c3 = run_pwls(best_h, best_n, rng, min(time.time() + 70.0, end3 - 60.0))
            if c3 < best_c:
                best_c, best_h = c3, h3

        if time.time() < end3 - 20.0:
            h4, c4 = sine_scan(best_n, min(time.time() + 35.0, end3 - 15.0))
            if c4 < best_c:
                best_c, best_h = c4, h4

        for scale in (0.8, 1.2):
            if time.time() > end3 - 25.0:
                break
            n_try = int(max(220, min(900, round(best_n * scale))))
            if n_try == best_n:
                continue
            h_try = interpolate_h(best_h, best_n, n_try)
            alloc = min(65.0, end3 - time.time() - 15.0)
            if alloc < 12.0:
                break
            h_try, c_try = fast_lbfgs(h_try, n_try, time.time() + alloc, LBFGS_CONFIGS[:4])
            if c_try < best_c + 0.004 and time.time() < end3 - 5.0:
                h_try, c_try = local_pair(h_try, n_try, min(time.time() + 25.0, end3 - 4.0),
                                          pool=60, window=50, max_accept=12)
            if c_try < best_c:
                best_c, best_h, best_n = c_try, h_try, n_try

        if time.time() < deadline - 2.0:
            best_h, best_c = local_pair(best_h, best_n, deadline - 2.0)
    except Exception:
        pass

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
