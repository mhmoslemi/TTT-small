def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

def binary_round(h, n):
    """Round h to nearest binary with sum = n/2."""
    target = int(n / 2)
    # Sort by value, set top 'target' to 1
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def binary_local_search(b, n, deadline, T_init=0.01, T_final=0.0001):
    """Simulated annealing on binary pattern."""
    best_b = b.copy()
    best_c = compute_c5(b, n)
    cur_c = best_c
    T = T_init
    n_iter = 0
    total_iter = 50000
    while time.time() < deadline:
        n_iter += 1
        T = T_init * (T_final / T_init) ** (n_iter / total_iter)
        # Pick random position with h=1 to flip to 0
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        b[i] = 0
        b[j] = 1
        new_c = compute_c5(b, n)
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_phase(h, n, deadline, alphas=None, max_iters=200):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha)
        def gr(x):
            return grad_smooth_max(x, n, alpha)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 25, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def adam_optimize(h, n, deadline, alpha_init=100.0, alpha_final=500000.0, lr_init=0.02):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 10000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.3 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def binary_round(h, n):
    target = int(n / 2)
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def binary_local_search(b, n, deadline, n_candidates=20, T_init=0.05, T_final=0.001, alpha=10000):
    b = b.copy()
    best_b = b.copy()
    best_c = compute_c5(b, n)
    cur_c = best_c
    rng = np.random.default_rng(42)
    t_count = 0
    total_steps = 200000
    while time.time() < deadline:
        t_count += 1
        T = T_init * (T_final / T_init) ** (t_count / total_steps)
        if t_count % 500 == 1:
            g = grad_smooth_max(b, n, alpha)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Score candidates: for 1->0, want high g; for 0->1, want low g
        i_scores = g[ones]
        j_scores = -g[zeros]
        # Pick top candidates
        i_top = np.argsort(-i_scores)[:n_candidates]
        j_top = np.argsort(-j_scores)[:n_candidates]
        best_delta = 0
        best_move = None
        for ii in i_top:
            i = ones[ii]
            for jj in j_top:
                j = zeros[jj]
                b[i] = 0
                b[j] = 1
                new_c = compute_c5(b, n)
                delta = new_c - cur_c
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)
                b[i] = 1
                b[j] = 0
        if best_move is not None:
            i, j = best_move
            b[i] = 0
            b[j] = 1
            new_c = compute_c5(b, n)
            delta = new_c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[i] = 1
                b[j] = 0
    return best_b, best_c

def pair_search(h, n, deadline, deltas=None, radius=None, alpha=5000):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            order = np.argsort(-np.abs(g))
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 100) if radius is None else max(0, i - radius)
                hi = min(n, i + 101) if radius is None else min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(300):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if 0.02 < h[i] < 0.98:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-12:
                continue
            mask = (h > abs(delta)) & (h < 1.0 - abs(delta))
            mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n = n_init
    # Phase 1: L-BFGS with moderate alphas
    t1 = min(t0 + 80, deadline - 600)
    best_h, best_c = lbfgs_phase(best_h, n, t1, alphas=[100, 500, 2000], max_iters=200)
    # Phase 2: Adam optimizer
    if time.time() < deadline - 400:
        t2 = min(time.time() + 200, deadline - 400)
        h_a, c_a = adam_optimize(best_h, n, t2, alpha_init=100, alpha_final=300000, lr_init=0.02)
        if c_a < best_c:
            best_c, best_h = c_a, h_a
    # Phase 3: L-BFGS with high alphas
    if time.time() < deadline - 300:
        t3 = min(time.time() + 120, deadline - 300)
        h3, c3 = lbfgs_phase(best_h, n, t3, alphas=[10000, 100000, 500000], max_iters=300)
        if c3 < best_c:
            best_c, best_h = c3, h3
    # Phase 4: Binary local search (simulated annealing on 0/1 pattern)
    if time.time() < deadline - 200:
        t4 = min(time.time() + 120, deadline - 200)
        b_round = binary_round(best_h, n)
        b_best, c_b = binary_local_search(b_round, n, t4, n_candidates=15,
                                           T_init=0.05, T_final=0.0005, alpha=50000)
        if c_b < best_c:
            best_c, best_h = c_b, b_best
            # Polish the binary solution with L-BFGS
            if time.time() < deadline - 150:
                t4b = min(time.time() + 60, deadline - 150)
                h4, c4 = lbfgs_phase(best_h, n, t4b, alphas=[50000, 200000], max_iters=200)
                if c4 < best_c:
                    best_c, best_h = c4, h4
    # Phase 5: Pair search
    if time.time() < deadline - 120:
        t5 = min(time.time() + 80, deadline - 120)
        best_h, best_c = pair_search(best_h, n, t5,
                                     [0.01, 0.005, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=80, alpha=10000)
    # Phase 6: Binary push
    if time.time() < deadline - 80:
        t6 = min(time.time() + 40, deadline - 80)
        h6, c6 = binary_push(best_h, n, t6)
        if c6 < best_c:
            best_c, best_h = c6, h6
    # Phase 7: Try alternative n
    if time.time() < deadline - 60:
        for n_try in [200, 400]:
            if time.time() > deadline - 25:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(30, deadline - time.time() - 20)
            if alloc < 10:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6,
                                    alphas=[5000, 50000, 500000], max_iters=200)
            if time.time() < deadline - 10:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 10, deadline - 8),
                                       [0.005, 0.001, 0.0001], radius=50)
            if c_t < best_c:
                best_c, best_h = c_t, h_t
    # Phase 8: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 2,
                                     [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                     radius=40, alpha=50000)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
