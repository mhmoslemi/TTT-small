    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c

def greed_binary(h_init, n, deadline, max_sweeps=200):

            return np.zeros(n)
        w = w / wsum
        active = np.where(w > 1e-14)[0]
        if len(active) == 0:
            active = np.array([np.argmax(g)])
        grad_total = np.zeros(n)
        for m in active:
            grad_total += w[m] * grad_single_lag(x, n, m)
        return grad_total * dx / alpha
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_gradient(h, n, max_steps, step_init, deadline):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    step = step_init
    for it in range(max_steps):
        if time.time() > deadline:
            break
        m_star = int(np.argmax(corr))
        g = grad_single_lag(h, n, m_star) * dx
        h_new = project_feasible(h - step * g, n)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_new = corr_new.max() * dx
        if c_new < cur_c - 1e-15:
            h = h_new
            corr = corr_new
            cur_c = c_new
            step = min(step * 1.02, 0.05)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            step *= 0.6
            if step < 1e-9:
                break
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=150):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_pipeline(h, n, deadline, alphas, maxiter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_smooth(best_h, n, alpha, maxiter, 1e-15, deadline)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    deadline = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    # Phase 1: binary greedy
    p1 = t0 + 0.04 * total
    hb, cb = greed_binary(h0, n0, p1, max_sweeps=80)
    if cb < best_c:
        best_c, best_h, best_n = cb, hb.copy(), n0
    # Phase 2: SLSQP at original resolution
    p2 = t0 + 0.40 * total
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    best_h, best_c = slsqp_pipeline(best_h, n0, p2, alphas, maxiter=1500)
    best_n = n0
    # Phase 3: projected gradient polish
    p3 = t0 + 0.52 * total
    hp, cp = projected_gradient(best_h, n0, 15000, 0.008, p3)
    if cp < best_c:
        best_c, best_h, best_n = cp, hp.copy(), n0
    # Phase 4: try n=200
    p4 = t0 + 0.72 * total
    n_alt = 200
    xo = np.linspace(0, 2, best_n)
    xa = np.linspace(0, 2, n_alt)
    h_alt = project_feasible(np.interp(xa, xo, best_h), n_alt)
    h_alt_b, c_alt_b = greed_binary(h_alt, n_alt, t0 + 0.06 * total, max_sweeps=40)
    if c_alt_b < compute_c5(h_alt, n_alt):
        h_alt = h_alt_b
    alphas_alt = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    h_alt, c_alt = slsqp_pipeline(h_alt, n_alt, p4, alphas_alt, maxiter=3000)
    hp, cp = projected_gradient(h_alt, n_alt, 8000, 0.005, p4 + 0.03 * total)
    if cp < c_alt:
        c_alt, h_alt = cp, hp
    if c_alt < best_c:
        best_c, best_h, best_n = c_alt, h_alt.copy(), n_alt
    # Phase 5: multi-restart
    p5 = deadline - 4
    for trial in range(6):
        if time.time() > p5 - 2:
            break
        noise = 0.008 * (trial + 1)
        h_try = best_h + noise * rng.randn(best_n)
        h_try = project_feasible(h_try, best_n)
        alphas_r = [100000, 1000000, 10000000]
        h_try, c_try = slsqp_pipeline(h_try, best_n, p5 - 1, alphas_r, maxiter=2000)
        hp2, cp2 = projected_gradient(h_try, best_n, 5000, 0.003, p5 - 0.5)
        if cp2 < c_try:
            c_try, h_try = cp2, hp2
        if c_try < best_c:
            best_c, best_h, best_n = c_try, h_try.copy(), best_n
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
