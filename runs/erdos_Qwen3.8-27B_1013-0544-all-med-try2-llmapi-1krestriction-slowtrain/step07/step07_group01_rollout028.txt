import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    val = m + (s.max() + np.log(np.exp(s).sum())) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Vectorized gradient computation
    # For lag index li (0 to 2n-2), shift k = li - (n-1)
    # c_li = sum_{valid i} x[i] * (1-x[i+k]) * dx
    # We need d(c_li)/dx[j]
    # This is complex to vectorize directly, use loop over significant weights
    significant = w > 1e-10
    for li in np.where(significant)[0]:
        k = li - (n - 1)
        wm = w[li] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def run_lbfgs(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = alpha * 0.01
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 3000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_slsqp(h, n, deadline, alphas=None):

    h = project_feasible(h, n)

            if tot < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        if w[li] < 1e-10:
            continue
        k = li - (n - 1)
        wm = w[li] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
    return grad

def obj_pen(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    val = m + np.log(np.exp(s).sum()) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    return float(m + np.log(np.exp(s).sum()) / alpha)
