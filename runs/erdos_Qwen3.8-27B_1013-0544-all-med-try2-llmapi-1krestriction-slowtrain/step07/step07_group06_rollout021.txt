def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()

        k = m - (n - 1)

            nv = n - k
            grad[:nv] += w[m] * (1.0 - h[k:])
            grad[k:] += -w[m] * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w[m] * (1.0 - h[:nv])
            grad[:nv] += -w[m] * h[kk:]

**New idea: "Shift-aware" random search**

Instead of random perturbations, generate perturbations that are specifically designed to reduce the overlap at the worst shift:
1. Find the worst shift k*
2. Generate a random "anti-pattern" that would reduce h(x)h(x+k*) 
3. Apply it with a small step

For shift k*, the overlap is ∑h[i](1-h[i+k*]). To reduce this, we want to:
- Decrease h[i] where h[i+k*] is also high (reduce the product)
- Or increase h[i+k*] where h[i] is high (make 1-h[i+k*] smaller)

So a good perturbation is: pick a random i, and move mass from i to i+k* (or vice versa).

Let me implement this:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += w[m] * (1.0 - h[k:])
            grad[k:] += -w[m] * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w[m] * (1.0 - h[:nv])
            grad[:nv] += -w[m] * h[kk:]
    return grad * (2.0 / n)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def compute_c5_approx(h, n, top_shifts):
    dx = 2.0 / n
    max_val = 0.0
    for m in top_shifts:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            val = float(np.sum(h[:nv] * (1.0 - h[k:]))) * dx
        else:
            kk = -k
            nv = n - kk
            val = float(np.sum(h[kk:] * (1.0 - h[:nv]))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            continue
        k = m - (n - 1)

            nv = n - k
            grad[:nv] += w[m] * (1.0 - h[k:])
            grad[k:] += -w[m] * h[:nv]
        else:

            nv = n - kk
