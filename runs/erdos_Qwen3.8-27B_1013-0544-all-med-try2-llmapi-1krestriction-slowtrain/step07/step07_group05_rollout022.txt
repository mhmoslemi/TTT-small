```python
def softmax_gradient(h, n, alpha):
    """Compute soft-max approximation and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # shape (2n-1,)
    
    # Soft max
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    weights = exp_c / exp_c.sum()
    soft_max_val = c_max + np.log(exp_c.sum()) / alpha

    # grad_k[i] = [(1 - h[i+s_k]) - h[i-s_k]] * dx where s_k = k - n + 1
```

```python
def softmax_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    weights = exp_c / exp_c.sum()
    soft_max_val = c_max + np.log(exp_c.sum()) / alpha
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = k - n + 1
        # grad_k[i] = dx * [(1 - h[i+s]) - h[i-s]]
        # Vectorize: for i in range(n), i+s and i-s
        # i+s valid when 0 <= i+s < n, i.e., max(0,-s) <= i < min(n, n-s)
        # i-s valid when 0 <= i-s < n, i.e., max(0,s) <= i < min(n, n+s)
        
        # (1 - h[i+s]) contribution: valid for i where 0 <= i+s < n
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += w * dx * (1.0 - h[s + lo1:s + hi1])
        
        # -h[i-s] contribution: valid for i where 0 <= i-s < n
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * dx * h[lo2 - s:hi2 - s]
    
    return soft_max_val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        s = k - n + 1
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * dx * (1.0 - h[s + lo1:s + hi1])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * dx * h[lo2 - s:hi2 - s]
    return c_max, grad

def nesterov_optimize(h, n, deadline, alpha_start=50.0, alpha_end=1000.0,
                      lr0=0.05, momentum=0.9):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    n_steps = 0
    total_steps = 50000
    for it in range(total_steps):
        if time.time() > deadline:
            break
        frac = it / total_steps
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = lr0 / (1.0 + it * 0.001)
        c_max, grad = softmax_grad(h, n, alpha)
        v = momentum * v + lr * grad
        h_new = h - v
        h = project_feasible(h_new, n)
        n_steps += 1
        if it % 500 == 0:
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def pair_search_fast(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n + 1
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000):
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    def obj_jac(x):
        dx_val = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx_val
        km = np.argmax(c)
        s = km - n + 1
        grad = np.zeros(n)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += dx_val * (1.0 - x[s + lo1:s + hi1])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= dx_val * x[lo2 - s:hi2 - s]
        return grad

    def _sum_c(x):
        return float(x.sum() - n / 2.0)

    def _sum_j(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
```
