def term_gradient(h,n,term_id):
  g=np.zeros(n)
  omh=1.0-h
  if term_id==0: return 1-2*h
  if term_id %2 ==1: # A
     k=(term_id+1)//2
     g[:-k] += omh[k:]
     g[k:] += -h[:-k]
  else:
     k=term_id//2
     g[k:] += omh[:-k]
     g[:-k] += -h[k:]

def sgrad(h, n, alpha):
    t = terms(h, n)
    m = t.max()
    z = alpha * (t - m)
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    o = 1.0 - h
    if w[0] > 1e-14:
        g += w[0] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wa = w[2*k - 1]
        if wa > 1e-14:
            g[:n-k] += wa * o[k:]
            g[k:] -= wa * h[:n-k]
        wb = w[2*k]
        if wb > 1e-14:
            g[k:] += wb * o[:n-k]
            g[:n-k] -= wb * h[k:]
    return g

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog

OPT = {'n': 1, 'target': 0.5, 'alpha': 10.0, 'mu': 1.0}

def project(h, n):
    h = np.nan_to_num(np.asarray(h, float).copy(), nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(20):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - h
            s = r.sum()
        else:
            r = h
            s = h.sum()
        if s < 1e-15:
            break
        h = np.clip(h + (d / s) * r, 0.0, 1.0)
    return h

def fast(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def terms(h, n):
    t = np.empty(2 * n - 1, dtype=float)
    o = 1.0 - h
    t[0] = np.dot(h, o)
    for k in range(1, n):
        t[2 * k - 1] = np.dot(h[:n - k], o[k:])
        t[2 * k] = np.dot(h[k:], o[:n - k])
    return t

def sval(h, n, alpha):
    t = terms(h, n)
    m = t.max()
    z = alpha * (t - m)
    z = z - z.max()
    return float(m + np.log(np.sum(np.exp(z))) / alpha)

def sgrad(h, n, alpha):
    t = terms(h, n)
    m = t.max()
    z = alpha * (t - m)
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    o = 1.0 - h
    if w[0] > 1e-14:
        g += w[0] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wa = w[2 * k - 1]
        if wa > 1e-14:
            g[:n - k] += wa * o[k:]
            g[k:] -= wa * h[:n - k]
        wb = w[2 * k]
        if wb > 1e-14:
            g[k:] += wb * o[:n - k]
            g[:n - k] -= wb * h[k:]
    return g

def tgrad(h, n, ids):
    g = np.zeros(n, dtype=float)
    if ids == 0:
        return 1.0 - 2.0 * h
    o = 1.0 - h
    if ids & 1:
        k = (ids + 1) // 2
        g[:n - k] += o[k:]
        g[k:] -= h[:n - k]
    else:
        k = ids // 2
        g[k:] += o[:n - k]
        g[:n - k] -= h[k:]
    return g

def obj(x):
    v = sval(x, OPT['n'], OPT['alpha'])
    mu = OPT['mu']
    if mu:
        s = x.sum() - OPT['target']
        v += mu * s * s
    return float(v)

def jac(x):
    g = sgrad(x, OPT['n'], OPT['alpha'])
    mu = OPT['mu']
    if mu:
        s = x.sum() - OPT['target']
        g = g + 2.0 * mu * s
    return g

def lbfgs(h, n, dl, alphas=(20.0, 100.0, 1000.0, 10000.0, 50000.0, 200000.0), mu=80.0, it=250):
    h = project(h.copy(), n)
    bc = fast(h, n)
    OPT['n'] = n
    OPT['target'] = 0.5 * n
    bd = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > dl - 1.0:
            break
        OPT['alpha'] = float(a)
        OPT['mu'] = float(mu)
        try:
            r = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bd,
                         options={'maxiter': it, 'ftol': 1e-14, 'gtol': 1e-10,
                                  'maxfun': it * 20, 'maxls': 60})
            h2 = project(r.x, n)
            c = fast(h2, n)
            if c < bc - 1e-15:
                h = h2
                bc = c
        except Exception:
            pass
    return h, bc

def adam(h, n, dl, alphas=(10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0), lr=0.05):
    h = project(h.copy(), n)
    bc = fast(h, n)
    max_steps = 1500 if n > 400 else 3000
    for a in alphas:
        if time.time() > dl:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        while time.time() < dl:
            if t > max_steps:
                break
            t += 1
            g = sgrad(h, n, float(a))
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * g * g
            lr_t = lr * (1.0 + 0.01 * t) ** (-0.5)
            step = m / (1.0 - 0.9 ** t)
            step = step / (np.sqrt(v / (1.0 - 0.999 ** t)) + 1e-8)
            h2 = project(h - lr_t * step, n)
            c = fast(h2, n)
            if c < bc - 1e-14:
                h = h2
                bc = c
                stall = 0
                continue
            if stall < 10:
                h2 = project(h - 0.5 * lr_t * step, n)
                c = fast(h2, n)
                if c < bc - 1e-14:
                    h = h2
                    bc = c
                    stall = 0
                    continue
            stall += 1
            if stall > 60:
                break
    return h, bc

def lp(h, n, dl, tol=1.0, r=0.08):
    h = project(h.copy(), n)
    bc = fast(h, n)
    no = 0
    mq = int(min(120, max(35, n // 4)))
    while time.time() < dl and no < 100:
        T = terms(h, n)
        m = T.max()
        order = np.argsort(-T)
        act = order[:mq]
        keep = T[act] >= m - tol
        if keep.sum() >= 8:
            act = act[keep]
        if act.size < 8:
            act = order[:min(8, mq)]
        q = act.size
        G = np.zeros((q, n), dtype=float)
        for i in range(q):
            G[i] = tgrad(h, n, int(act[i]))
        cvec = np.zeros(n + 1)
        cvec[-1] = 1.0
        A = np.empty((q, n + 1), dtype=float)
        A[:, :n] = G
        A[:, n] = -1.0
        b = np.zeros(q)
        Ae = np.ones((1, n), dtype=float)
        be = np.zeros(1)
        bd = [(-r, r)] * n + [(-100.0, 100.0)]
        try:
            res = linprog(cvec, A_ub=A, b_ub=b, A_eq=Ae, b_eq=be, bounds=bd, method='highs')
        except Exception:
            no += 1
            if no > 50:
                break
            continue
        if res.status != 0:
            no += 1
            if no > 50:
                break
            continue
        p = res.x[:n]
        z = res.x[-1]

            no += 1

        ok = False
        for f in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):

def pair(h, n, dl, deltas=None, rad=90, top_m=130, seed=1):

                    return h, bc
                gi = int(ii)

                                imp = True
                            else:
                                h[gi] = oi
                                h[j] = oj
                        elif minus_ok:
                            h[gi] = oi - d
                            h[j] = oj + d
                            c = fast(h, n)
                            if c < bc - 1e-15:
                                bc = c
                                imp = True
                            else:
                                h[gi] = oi
                                h[j] = oj
                    else:
                        if minus_ok:
                            h[gi] = oi - d
                            h[j] = oj + d
                            c = fast(h, n)
                            if c < bc - 1e-15:
                                bc = c
                                imp = True
                            else:
                                h[gi] = oi
                                h[j] = oj
                        elif plus_ok:
                            h[gi] = oi + d
                            h[j] = oj - d
                            c = fast(h, n)
                            if c < bc - 1e-15:
                                bc = c
                                imp = True
                            else:
                                h[gi] = oi
                                h[j] = oj
            if not imp:
                break
    return h, bc

def scale(h, n, dl, scales=(1.2, 1.6, 2.2, 3.0)):
    h = project(h.copy(), n)
    bc = fast(h, n)
    for s in scales:
        if time.time() > dl:
            break
        h2 = project(0.5 + float(s) * (h - 0.5), n)
        c = fast(h2, n)
        if c < bc - 1e-15:
            h = h2
            bc = c
    return h, bc

def interp(h, m):
    h = np.asarray(h, float)
    n = len(h)
    if n == m:
        return project(h.copy(), m)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, m)
    return project(np.interp(y, x, h), m)

def defh(n, seed=123):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.35 * np.sin(4.0 * np.pi * x) + 0.2 * np.sin(9.0 * np.pi * x + 0.7)
    h = h + 0.1 * (rng.random(n) - 0.5)
    return project(np.clip(h, 0.0, 1.0), n)

def polish(h, n, dl, seed=1):
    h = project(h.copy(), n)
    best = h.copy()
    bc = fast(h, n)
    if float(np.max(h) - np.min(h)) < 1e-8:
        best = defh(n, int(seed))
        bc = fast(best, n)
    bh = best.copy()
    rem = dl - time.time()
    if rem > 20:
        d = min(time.time() + max(2.0, 0.04 * rem), dl - 2.0)
        hv, cv = scale(bh, n, d)
        if cv < bc - 1e-12:
            bc, bh = cv, hv
    rem = dl - time.time()
    if rem > 40:
        d = min(time.time() + 0.45 * rem, dl - 2.0)
        hv, cv = lp(bh, n, d)
        if cv < bc - 1e-12:
            bc, bh = cv, hv
    rem = dl - time.time()
    if rem > 80:
        d = min(time.time() + 0.18 * rem, dl - 2.0)
        hv, cv = adam(bh, n, d)
        if cv < bc - 1e-12:
            bc, bh = cv, hv
    rem = dl - time.time()
    if rem > 60:
        d = min(time.time() + 0.25 * rem, dl - 2.0)
        it = 120 if n > 500 else 200
        hv, cv = lbfgs(bh, n, d, it=it)
        if cv < bc - 1e-12:
            bc, bh = cv, hv
    rem = dl - time.time()
    if rem > 20:
        d = min(time.time() + 0.9 * rem, dl - 2.0)
        hv, cv = pair(bh, n, d, seed=seed)
        if cv < bc - 1e-12:
            bc, bh = cv, hv
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 5.0
    try:
        h0 = np.asarray(initial_h_values, float).copy()
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 300
    if h0 is None or n0 < 20:
        n0 = 300
        h0 = defh(n0, seed)
    if n0 <= 700:
        bh = project(h0, n0)
        best_n = n0
        bc = fast(bh, best_n)
    else:
        best_n = 600
        bh = interp(h0, best_n)
        bc = fast(bh, best_n)
    h_work = bh.copy()
    n_work = best_n
    if dl - time.time() > 350.0:
        dmain = min(time.time() + 800.0, dl - 250.0)
        if dmain > time.time() + 30.0:
            hv, cv = polish(h_work, n_work, dmain, seed)
            if cv < bc - 1e-12:
                bc = cv
                bh = hv
                best_n = n_work
                h_work = hv
    if time.time() < dl - 120.0 and n_work < 600:
        nt = int(min(600, max(400, 2 * n_work)))
        if 200 <= nt <= 600 and nt != n_work:
            ht = interp(bh, nt)
            dnew = min(time.time() + 200.0, dl - 100.0)
            if dnew > time.time() + 60.0:
                hv, cv = polish(ht, nt, dnew, seed + 7)
                if cv < bc - 1e-12:
                    bc = cv
                    bh = hv
                    best_n = nt
                    h_work = hv
                    n_work = nt
    if time.time() < dl - 15.0:
        nh = best_n
        d = min(time.time() + 0.5 * (dl - time.time()), dl - 5.0)
        if nh <= 600 and d > time.time() + 10.0:
            hv, cv = lp(bh, nh, d)
            if cv < bc - 1e-12:
                bc = cv
                bh = hv
        d = min(time.time() + 0.9 * (dl - time.time()), dl - 2.0)
        if d > time.time() + 5.0:
            hv, cv = pair(bh, nh, d,
                          deltas=(0.01, 0.004, 0.001, 0.0004, 1e-4),
                          rad=70, top_m=100, seed=seed + 3)
            if cv < bc - 1e-12:
                bc = cv
                bh = hv
    bh = project(bh, best_n)
    bc = fast(bh, best_n)
    return bh, bc, best_n
```
