def exact_coord_step(h, i, n):
    dx = 2.0 / n
    c_old = np.correlate(h, 1-h, mode='full') * dx
    # c_old has 2n-1 entries, indexed from -(n-1) to (n-1)
    # c_old[k + n - 1] corresponds to shift k
    
    # Compute w_k for valid k
    # w_k = (1-h[i+k])*I(0<=i+k<n) - h[i-k]*I(0<=i-k<n)
    # k ranges from -(n-1) to (n-1)
    
    nv = 2*n - 1
    w = np.zeros(nv)
    for offset, k in enumerate(range(-(n-1), n)):
        ik = i + k
        imk = i - k
        wk = 0.0
        if 0 <= ik < n:
            wk += (1 - h[ik])
        if 0 <= imk < n:
            wk -= h[imk]
        w[offset] = wk
    
    # Now f(Δ) = max_k [c_old[k] + Δ * w[k] * dx]
    # Minimize over Δ ∈ [-h[i], 1-h[i]]
    
    lo = -h[i]
    hi = 1.0 - h[i]
    
    # Ternary search (f is convex)
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c_old + m1 * w * dx)
        f2 = np.max(c_old + m2 * w * dx)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2

    for offset, k in enumerate(range(-(n-1), n)):
        ik = i + k
        imk = i - k
        wk = 0.0
        if 0 <= ik < n:
            wk += (1.0 - h[ik])
        if 0 <= imk < n:
            wk -= h[imk]
        w[offset] = wk
    
    lo = -h[i]
    hi = 1.0 - h[i]
    if hi - lo < 1e-13:
        return 0.0
    
    # Ternary search
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c + m1 * w * dx)
        f2 = np.max(c + m2 * w * dx)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    delta = (lo + hi) / 2.0
    if abs(delta) < 1e-13:
        return 0.0
    h[i] += delta
    c += delta * w * dx
    return delta

def coord_descent(h, n, order, n_passes=10, tol=1e-12):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = np.max(c)
    
    for _ in range(n_passes):
        total_change = 0.0
        for i in order:
            total_change += abs(coord_step(h, i, c, n, dx))
        curr_c = np.max(c)
        if curr_c < best_c:
            best_c = curr_c
        if total_change < tol:
            break
    
    return h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for direction in [1, -1]:
                        ni = h[i] + direction * d
                        nj = h[j] - direction * d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i] = ni; h[j] = nj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= direction * d
                                h[j] += direction * d
    return h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_step(h, i, c, n, dx, k_range):
    nv = 2 * n - 1
    ik = i + k_range
    imk = i - k_range
    w = np.zeros(nv)
    mask_ik = (ik >= 0) & (ik < n)
    mask_imk = (imk >= 0) & (imk < n)
    if np.any(mask_ik):
        w[mask_ik] += (1.0 - h[ik[mask_ik]])
    if np.any(mask_imk):

        if f1 < f2:

    delta = (lo + hi) / 2.0
    if abs(delta) < 1e-13:
        return 0.0
    h[i] += delta
    c += delta * w * dx
    return delta
