```python
def compute_grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max of correlation"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # softmax weights
    m = c.max()
    w = np.exp(alpha * (c - m)) / np.exp(alpha * (c - m)).sum()
    # w has length 2n-1, indexed by m = n-1+k for k in [-(n-1), n-1]
    # grad[i] = dx * sum_k w[n-1+k] * ((1-h[i+k])*1[valid] - h[i-k]*1[valid])
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = n - 1 + k  # index into w
        wk = w[idx] * dx
        if wk < 1e-15:
            continue
        # (1-h[i+k]) for valid i: i in [max(0,-k), min(n-1, n-1-k)]
        i_low = max(0, -k)
        i_high = min(n-1, n-1-k)
        if i_low <= i_high:
            grad[i_low:i_high+1] += wk * (1 - h[i_low+k:i_high+k+1])
        # -h[i-k] for valid i: i in [max(0,k), min(n-1, n-1+k)]
        i_low2 = max(0, k)
        i_high2 = min(n-1, n-1+k)
        if i_low2 <= i_high2:
            grad[i_low2:i_high2+1] -= wk * h[i_low2-k:i_high2-k+1]
    return grad
```

```python
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    w = exp_c / exp_c.sum()
```

```python
    dx = 2.0 / n
    cur_val = smooth_max_val(h, n, alpha)
```

```python
    stall_count = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        g = compute_grad(h, n, alpha)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        g = g / gnorm * gnorm
        accepted = False
        s = step
        best_trial = None
        best_val = cur_val
        for _ls in range(12):
            h_trial = project_feasible(h - s * g, n)
            val_trial = smooth_max_val(h_trial, n, alpha)
            if val_trial < best_val:
                best_val = val_trial
                best_trial = h_trial
            if val_trial <= cur_val + 1e-15:
                accepted = True
                break
            s *= 0.5
        if best_trial is not None:
            h = best_trial
            cur_val = best_val
            if accepted:
                step = min(s * 1.1, 1.0)
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        if stall_count > 200:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Progressive alpha projected gradient
    alphas = [5, 15, 50, 150, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for i, alpha in enumerate(alphas):
        frac = (i + 1) / len(alphas)
        phase_dl = t0 + (deadline - t0) * frac * 0.72
        if time.time() > phase_dl:
            continue
        h = pg_optimize(h, n, alpha, max_iters=100000, deadline=phase_dl)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Polish with highest alphas
    polish_dl = t0 + (deadline - t0) * 0.82
    for alpha in [500000, 2000000, 10000000]:
        if time.time() > polish_dl:
            break
        h = pg_optimize(h, n, alpha, max_iters=50000, deadline=polish_dl)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Multi-restart with perturbations
    restart_dl = t0 + (deadline - t0) * 0.90
    for trial in range(4):
        if time.time() > restart_dl:
            break
        h = best_h + (rng.randn(n) * 0.01)
        h = project_feasible(h, n)
        for alpha in [5000, 50000, 500000]:
            if time.time() > restart_dl:
                break
            h = pg_optimize(h, n, alpha, max_iters=30000, deadline=restart_dl)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 4: Try different resolutions
    res_dl = deadline
    for nh in [200, 300, 500, 600]:
        if time.time() > res_dl - 20:
            break
        remaining = res_dl - time.time()
        if remaining < 15:
            break
        xn = np.linspace(0, 2, nh, endpoint=False)
        xo = np.linspace(0, 2, n, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        for alpha in [1000, 10000, 100000, 1000000]:
            if time.time() > res_dl - 5:
                break
            hh = pg_optimize(hh, nh, alpha, max_iters=20000, deadline=res_dl - 5)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n = nh

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
