def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:

        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _pen_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _pen_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slq_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slq_con(x, n):
    return float(x.sum() - n / 2.0)

def _slq_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, deadline, maxit=200):
    h = project_feasible(h, n)
    bc = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cfgs = [(50, 1.0), (200, 5.0), (1000, 20.0), (5000, 100.0),
            (20000, 500.0), (100000, 2000.0), (500000, 10000.0), (2000000, 50000.0)]
    for alpha, mu in cfgs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-17,
                                   'gtol': 1e-15, 'maxfun': maxit * 10, 'maxls': 50})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < bc - 1e-15:
                bc = cn
                h = hn
        except Exception:
            pass
    return h, bc

def slsqp_opt(h, n, deadline, alpha=500000, maxit=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con, 'jac': _slq_jac, 'args': (n,)}]
    try:
        res = minimize(_slq_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-16})
        hn = project_feasible(res.x, n)
        cn = compute_c5(hn, n)
        return hn, cn
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_desc(h, n, deadline, lr0=0.005, a0=500.0):
    h = project_feasible(h, n)
    bc = compute_c5(h, n)
    bh = h.copy()
    lr = lr0
    alpha = a0
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        imp = False
        for _ in range(30):
            hn = project_feasible(h - lr * g, n)
            cn = compute_c5(hn, n)
            if cn <= bc - 1e-15:
                h = hn; bc = cn; bh = h.copy(); imp = True
                break
            lr *= 0.5
        if not imp:
            alpha *= 1.3
            if lr < 1e-13:
                break
    return bh, bc

def pair_srch(h, n, deadline, deltas, radius=50):
    bc = compute_c5(h, n)
    step = max(1, n // 200)
    for d in deltas:
        while time.time() < deadline:
            imp = False
            for i in range(0, n, step):
                if time.time() > deadline:
                    return h, bc
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < bc - 1e-14:
                            bc = c; imp = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < bc - 1e-14:
                            bc = c; imp = True
                        else:
                            h[i] += d; h[j] -= d
            if not imp:
                break
    return h, bc

def multi_opt(h, n, deadline, tag=''):
    """Run L-BFGS, SLSQP, gradient descent in sequence."""
    h, c = lbfgs_opt(h, n, min(deadline, time.time() + (deadline - time.time()) * 0.5))
    if time.time() < deadline - 10:
        h2, c2 = slsqp_opt(h, n, min(deadline - 10, time.time() + (deadline - time.time()) * 0.3),
                          alpha=800000, maxit=1500)
        if c2 < c:
            c = c2; h = h2
    if time.time() < deadline - 5:
        h3, c3 = grad_desc(h, n, deadline - 2, lr0=0.002, a0=2000.0)
        if c3 < c:
            c = c3; h = h3
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(300, 0.5)
        n0 = 300
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Optimize base grid
    t1 = min(t0 + 260, dl - 380)
    best_h, best_c = multi_opt(best_h, n0, t1)

    # Phase 2: Upscale to fine grid and optimize
    nf = min(2 * n0, 600)
    if nf > n0 and time.time() < dl - 200:
        xo = np.linspace(0, 2, n0)
        xn = np.linspace(0, 2, nf)
        hf = np.interp(xn, xo, best_h)
        hf = project_feasible(hf, nf)
        t2 = min(time.time() + 250, dl - 200)
        hf, cf = multi_opt(hf, nf, t2)
        if cf < best_c:
            best_c = cf; best_h = hf; n0 = nf

    # Phase 3: Pair search on best
    if time.time() < dl - 20:
        nn = len(best_h)
        t3 = min(time.time() + 80, dl - 20)
        rad = max(15, min(50, nn // 6))
        best_h, best_c = pair_srch(best_h, nn, t3,
            [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=rad)

    # Phase 4: Random restart attempts
    if time.time() < dl - 30:
        nn = len(best_h)
        for sigma in [0.02, 0.05, 0.01]:
            if time.time() > dl - 15:
                break
            noise = rng.randn(nn) * sigma
            hr = project_feasible(best_h + noise, nn)
            t4 = min(time.time() + 40, dl - 10)
            hr, cr = lbfgs_opt(hr, nn, t4, maxit=150)
            if cr < best_c:
                best_c = cr; best_h = hr

    # Phase 5: Final pair search + polish
    if time.time() < dl - 5:
        nn = len(best_h)
        t5 = min(time.time() + 30, dl - 5)
        best_h, best_c = pair_srch(best_h, nn, t5,
            [0.001, 0.0005, 0.0002, 0.0001], radius=30)
    if time.time() < dl - 2:
        nn = len(best_h)
        best_h, best_c = lbfgs_opt(best_h, nn, dl - 1, maxit=80)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
