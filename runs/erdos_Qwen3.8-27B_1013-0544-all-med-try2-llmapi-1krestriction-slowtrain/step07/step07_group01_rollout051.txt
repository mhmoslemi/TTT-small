    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

            h += (diff / tot) * room

    return val, c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / (alpha * dx)
    return float(val)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for pos in range(2 * n - 1):
        if w[pos] < 1e-14:
            continue
        k = pos - (n - 1)
        wm = w[pos] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _pen_obj(x, n, alpha, mu):
    v = smooth_max_obj_val(x, n, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def _pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    return smooth_max_obj_val(x, n, alpha)

def _slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(20, 1.0), (50, 5.0), (200, 10.0), (1000, 50.0),
               (5000, 200.0), (20000, 1000.0), (100000, 5000.0),
               (500000, 20000.0), (2000000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_descent(h, n, deadline, lr_init=0.005, alpha_init=200.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new; best_c = c_new; best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            alpha *= 1.3
        if lr < 1e-14:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return best_h, best_c

def grad_pair_search(h, n, deadline, step_size=0.002):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [100, 1000, 10000, 100000]:
        while time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if 0.0 <= h[i] + move <= 1.0 and 0.0 <= h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= move; h[j] += move
            if not improved:
                break
    return best_h, best_c

def upscale(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_new)
