import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_sa(h_init, n, deadline, max_swaps=400000, seed=99):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(seed)
    T_start = max(cur_c * 0.05, 0.005)
    T_end = 1e-10
    n_eff = max(1, min(max_swaps, 500000))
    cooling = (T_end / T_start) ** (1.0 / n_eff)
    for it in range(n_eff):
        if time.time() > deadline:
            break
        T = T_start * cooling ** it
        ri = rng.randint(len(zeros))
        rj = rng.randint(len(ones))
        i = zeros[ri]
        j = ones[rj]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += (1.0 - h[im[msk]])
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= (1.0 - h[jm[msk]])
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        delta[n - 1 + (j - i)] += 1.0
        delta[n - 1 + (i - j)] += 1.0
        nc = (corr + delta).max() * dx
        dE = nc - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-15)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros[ri] = j
            ones[rj] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
        h[i] = x_opt
        if abs(x_opt - h[i]) > 1e-15:
            corr += B * (x_opt - h[i])
            h[i] = x_opt
    return h, corr
