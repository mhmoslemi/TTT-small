def binary_local_search(h_init, n, deadline, max_sweeps=2000, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    current_h = h.copy()
    m_range = np.arange(-(n-1), n)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_swaps = rng.randint(2, 6)
            for _ in range(n_swaps):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = list(zeros)
            rng.shuffle(z_list)  # Randomize order for diversification
            for i in z_list:
                if time.time() > deadline:
                    break
                if i not in zeros:
                    continue
                o_list = list(ones)
                rng.shuffle(o_list)
                for j in o_list:
                    if j not in ones:
                        continue
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:

                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:

            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_exact(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return np.clip(h, 0, 1)
    if s > target:
        lo, hi = 0.0, max(h.max() + 0.5, 1.5)
    else:
        lo, hi = min(h.min() - 0.5, -1.5), 0.0
    for _ in range(80):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-7:
            continue
        k = idx - n + 1
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += w[idx] * dx * (1.0 - h[k:])
                grad[k:] -= w[idx] * dx * h[:n-k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += w[idx] * dx * (1.0 - h[:n-kk])
                grad[:n-kk] -= w[idx] * dx * h[kk:]
    return grad

def projected_gradient(h, n, alpha, deadline, max_iter=3000, step0=0.005):

                improved = True
                break

            step = step0 * 0.5

    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
