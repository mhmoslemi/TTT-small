def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_R(h, n):
    """Compute normalized autocorrelation R_k = (2/n) * sum_i h[i]*h[i+k]"""
    autocorr = np.correlate(h, h, mode='full')
    return (2.0 / n) * autocorr

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

            alpha = min(0.08, alpha * 1.02)
        elif rng.random() < np.exp(-max(delta, 0) / max(T, 1e-10)):
            h = h_new
            stuck += 1
        else:
            h = h_new  # still move, to explore
            stuck += 1
        
        if stuck > 100:
            alpha *= 0.8
            T *= 0.9
            if stuck > 500:
                # Restart from best
                h = best_h + rng.normal(0, 0.03, n)
                h = project_feasible(h, n)
                stuck = 0
                alpha = 0.04
                T = 0.05
        
        T = max(T, 0.005)
        alpha = max(alpha, 0.003)
    
    return best_h, best_c

def greedy_refine(h, n, deadline, rng):
    """Targeted greedy swaps based on autocorrelation"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        while time.time() < deadline:
            improved = False
            R = compute_autocorr(h, n)
            center = n - 1
            # Find the minimum lag
            min_k = np.argmin(R)
            # Also consider near-minimum lags
            k_list = np.argsort(R)[:5]
            
            for _ in range(100):
                k_idx = k_list[rng.randint(len(k_list))]
                k = abs(int(k_idx) - center)
                if k >= n:
                    continue
                valid = n - k
                
                # Score: positions that would most increase R_k
                score = np.zeros(n)
                score[:valid] += h[k:]
                score[k:] += h[:valid]
                
                # Pick i (high score, room to add) and j (low score, room to remove)
                i_cands = np.where((h < 1.0 - d/2))[0]
                j_cands = np.where((h > d/2))[0]
                if len(i_cands) == 0 or len(j_cands) == 0:
                    continue
                
                i = i_cands[np.argmax(score[i_cands] + rng.randn(len(i_cands))*0.1)]
                j = j_cands[np.argmin(score[j_cands] + rng.randn(len(j_cands))*0.1)]
                
                if i == j:
                    continue
                
                h[i] += d
                h[j] -= d
                c = compute_c5(h, n)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
            
            if not improved:
                break
    
    return best_h, best_c
