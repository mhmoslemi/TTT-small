def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    sh = alpha * (c - m)

    w = np.exp(sh - mx)
    w /= w.sum()
    return w

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def grad_c5(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(w)):
        if w[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:n])
    return grad

def obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    v = smooth_max_val(c, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def obj_pen_grad(x, n, alpha, mu):
    g = grad_c5(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    return float(smooth_max_val(c, alpha))

def eq_con(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, t_end, alpha=200000, maxiter=3000):

def nesterov_opt(h, n, t_end, alpha=500, lr0=0.05):
