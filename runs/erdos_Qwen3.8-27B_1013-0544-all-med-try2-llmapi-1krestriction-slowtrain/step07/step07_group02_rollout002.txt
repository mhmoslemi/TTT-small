def perturb_h(h,n,rng,sigma,kind):
    h2=h.copy()
    if kind==0:
        h2 += sigma*rng.randn(n)
    elif kind==1:
        h2 += sigma*(rng.rand(n)-0.5)
    elif kind==2:
        x=np.linspace(0.0,2.0,n)
        for r in range(3):
            f=int(rng.randint(1,7))
            ph=rng.uniform(0.0,2.0*np.pi)
            h2 += sigma*np.sin(float(f)*np.pi*x+ph)
    else:
        span=max(1,n//20)
        for r in range(6):
            a=int(rng.randint(0,n-span+1))
            l=int(rng.randint(1,span+1))
            h2[a:a+l] += sigma*(rng.rand(l)-0.5)

import numpy as np
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / float(n))

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if d > 0.0:
        lo, hi = 0.0, 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            if np.clip(h + mid, 0.0, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    elif d < 0.0:
        lo, hi = 0.0, 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            break
        h[free] += diff / float(free.size)
        h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if free.size:
            h[free[0]] = np.clip(h[free[0]] + diff, 0.0, 1.0)
        else:
            if diff > 0.0:
                inc = np.where(h < 1.0)[0]
                if inc.size:
                    h[inc[0]] = np.clip(h[inc[0]] + diff, 0.0, 1.0)
            else:
                dec = np.where(h > 0.0)[0]
                if dec.size:
                    h[dec[0]] = np.clip(h[dec[0]] + diff, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_old, n_new):
    xo = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    xn = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return np.interp(xn, xo, h_old, left=h_old[0], right=h_old[-1])

def perturb_h(h, n, rng, sigma, kind):
    h2 = h.copy()
    if kind == 0:
        h2 += sigma * rng.randn(n)
    elif kind == 1:
        h2 += sigma * (rng.rand(n) - 0.5)
    elif kind == 2:
        x = np.linspace(0.0, 2.0, n)
        for _ in range(3):
            f = int(rng.randint(1, 7))
            ph = rng.uniform(0.0, 2.0 * np.pi)
            h2 += sigma * np.sin(float(f) * np.pi * x + ph)
    else:
        span = max(1, n // 20)
        for _ in range(6):
            a = int(rng.randint(0, n - span + 1))
            l = int(rng.randint(1, span + 1))
            h2[a:a + l] += sigma * (rng.rand(l) - 0.5)
    return project_feasible(h2, n)

def local_search_pairs(h, n, deadline, rng, active_k=20, top_iters=12, stall=50000, max_iters=10000000):
    h = h.astype(float).copy()
    c = correlate_full(h)
    cur = float(c.max())
    if time.time() > deadline:
        return h, c, cur

    L = 2 * n - 1
    center = n - 1
    omh = 1.0 - h
    rev_omh = omh[::-1]
    bi = np.empty(L, dtype=float)
    bj = np.empty(L, dtype=float)
    p = np.empty(L, dtype=float)
    w = np.empty(L, dtype=float)

    A = min(active_k, L)
    if A > 0:
        active = np.argpartition(c, -A)[-A:] if A < L else np.arange(L)
    else:
        active = np.arange(0)

    it = 0
    no = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        phase = it % 160
        if phase < 50:
            i = it % n
            j = (i + 1) % n
        elif phase < 90:
            i = int(rng.randint(n))
            j = (i + 1 + int(rng.randint(0, max(2, n // 25)))) % n
        elif phase < 110:
            i = int(rng.randint(n))
            j = (i + n // 2 + int(rng.randint(0, max(2, n // 15)))) % n
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n

        if i == j:
            no += 1
            if no > stall:
                break
            continue

        a0 = float(h[i])
        b0 = float(h[j])
        Lb = max(-a0, b0 - 1.0)
        Ub = min(1.0 - a0, b0)
        if Ub - Lb <= 1e-13:
            no += 1
            if no > stall:
                break
            continue

        bi.fill(0.0)
        bi[i:i + n] += rev_omh
        bi[center - i:center - i + n] -= h
        bi[center] = 0.0

        bj.fill(0.0)
        bj[j:j + n] += rev_omh
        bj[center - j:center - j + n] -= h
        bj[center] = 0.0

        np.subtract(bi, bj, out=p)

        if A > 1:
            ac = c[active]
            ap = p[active]
            al, ar = Lb, Ub
            for _ in range(top_iters):
                m1 = al + (ar - al) / 3.0
                m2 = al + 2.0 * (ar - al) / 3.0
                f1 = float((ac + ap * m1).max())
                f2 = float((ac + ap * m2).max())
                if f1 < f2:
                    ar = m2
                else:
                    al = m1
            ttop = 0.5 * (al + ar)
            act_min = float((ac + ap * ttop).max())
            if act_min > cur + 1e-7:
                no += 1
                if no > stall:
                    break
                continue
        else:
            ttop = 0.5 * (Lb + Ub)

        span = Ub - Lb
        mid = 0.5 * (Lb + Ub)
        rr = Lb + rng.random() * span
        cross1 = center + j - i
        cross2 = center + i - j
        c0 = float(c[center])
        base_center = c0 - a0 * (1.0 - a0) - b0 * (1.0 - b0)

        best_val = cur
        best_t = 0.0
        for t in (0.0, Lb, Ub, mid, rr,
                  Lb + 0.25 * span, Lb + 0.75 * span,
                  ttop, ttop + 0.1 * span, ttop - 0.1 * span,
                  ttop + 0.03 * span, ttop - 0.03 * span,
                  0.5 * (mid + ttop), 0.5 * ttop):
            if t < Lb:
                t = Lb
            elif t > Ub:
                t = Ub

            np.multiply(p, t, out=w)
            w += c
            tt = t * t
            w[cross1] += tt
            w[cross2] += tt
            w[center] = base_center + (a0 + t) * (1.0 - a0 - t) + (b0 - t) * (1.0 - b0 + t)
            v = float(w.max())
            if v < best_val - 1e-10:
                best_val = v
                best_t = t

        if best_t != 0.0:
            np.multiply(p, best_t, out=w)
            c += w
            tt = best_t * best_t
            c[cross1] += tt
            c[cross2] += tt
            c[center] = base_center + (a0 + best_t) * (1.0 - a0 - best_t) + (b0 - best_t) * (1.0 - b0 + best_t)

            h[i] = a0 + best_t
            h[j] = b0 - best_t
            if h[i] < 1e-12:
                h[i] = 0.0
            elif h[i] > 1.0 - 1e-12:
                h[i] = 1.0
            if h[j] < 1e-12:
                h[j] = 0.0
            elif h[j] > 1.0 - 1e-12:
                h[j] = 1.0

            omh[i] = 1.0 - h[i]
            omh[j] = 1.0 - h[j]
            cur = best_val
            no = 0

            if no == 0 and it % 60 == 0 and A < L:
                active = np.argpartition(c, -A)[-A:]
        else:
            no += 1
            if no > 4000 and A < L:
                active = np.argpartition(c, -A)[-A:]
                no = 0
            if no > stall:
                break

        if it % 500 == 499:
            c2 = correlate_full(h)
            cc = float(c2.max())
            if abs(cc - cur) > 1e-8:
                c = c2
                cur = cc
                if A < L:
                    active = np.argpartition(c, -A)[-A:]

    return h, c, cur

    rng = np.random.RandomState(int(seed) % 2147483647)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).reshape(-1).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        vals = rng.rand(n0)
        order = np.argsort(vals)
        h0 = np.zeros(n0)
        h0[order[:n0 // 2]] = 1.0

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    t1 = min(t0 + 0.16 * total, dl - 5.0)
    if time.time() < t1:
        hh, _, _ = local_search_pairs(best_h, best_n, t1, rng)
        cc = compute_c5(hh, best_n)
        if cc < best_c - 1e-12:
            best_h = hh
            best_c = cc

    t2 = min(t0 + 0.55 * total, dl - 10.0)
    sigma = 0.08
    fail = 0
    trial = 0
    while time.time() < t2 - 3.0:
        trial += 1
        kind = trial % 4
        hp = perturb_h(best_h, best_n, rng, sigma, kind)
        rem = t2 - time.time()
        step = min(max(4.0, 0.22 * rem), 50.0)
        td = time.time() + min(step, max(2.0, rem - 1.0))
        hh, _, _ = local_search_pairs(hp, best_n, td, rng)
        cc = compute_c5(hh, best_n)
        if cc < best_c - 1e-12:
            best_h = hh
            best_c = cc
            fail = 0
            sigma = max(0.004, sigma * 0.75)
        else:
            fail += 1
            if fail % 3 == 0:
                sigma = min(0.25, sigma * 1.4)
            else:
                sigma = max(0.004, sigma * 0.92)

    t3 = min(t0 + 0.90 * total, dl - 15.0)
    for nh in (500, 600):
        if time.time() > t3 - 30.0:
            break
        if nh == best_n:
            continue
        rem = t3 - time.time()
        if rem < 20.0:
            break

        hx = interpolate_h(best_h, best_n, nh)
        hx = project_feasible(hx, nh)
        step = 0.55 * rem
        if step > 120.0:
            step = 120.0
        td = time.time() + min(step, rem - 5.0)
        hh, _, _ = local_search_pairs(hx, nh, td, rng)
        cc = compute_c5(hh, nh)
        if cc < best_c - 1e-12:
            best_h = hh
            best_n = nh
            best_c = cc

        rsigma = 0.03
        for _ in range(2):
            if time.time() > t3 - 4.0:
                break
            hp = perturb_h(best_h, best_n, rng, rsigma, 0)
            rem = t3 - time.time()
            step = min(25.0, max(4.0, 0.35 * rem))
            td = time.time() + min(step, max(2.0, rem - 1.0))
            hh, _, _ = local_search_pairs(hp, best_n, td, rng)
            cc = compute_c5(hh, best_n)
            if cc < best_c - 1e-12:
                best_h = hh
                best_c = cc
                rsigma = max(0.005, rsigma * 0.7)
            else:
                rsigma = min(0.12, rsigma * 1.35)

    if time.time() < dl - 4.0:
        hp = perturb_h(best_h, best_n, rng, 0.003, 0)
        hh, _, _ = local_search_pairs(hp, best_n, dl - 1.0, rng)
        cc = compute_c5(hh, best_n)
        if cc < best_c - 1e-12:
            best_h = hh
            best_c = cc

    if time.time() < dl - 1.0:
        hh, _, _ = local_search_pairs(best_h, best_n, dl - 0.5, rng)
        cc = compute_c5(hh, best_n)
        if cc < best_c - 1e-12:
            best_h = hh
            best_c = cc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
