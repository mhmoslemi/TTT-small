def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2 using bisection on threshold."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

def grad_at_lag(h, n, k):
    """Gradient of correlation at lag k with respect to h."""
    grad = np.zeros(n)
    # k > 0: corr_k = sum_j h[j+k]*(1-h[j]) for 0<=j<n-k
    # d/dh[i] = (1-h[i-k]) if 0<=i-k<n  [h[i] in first factor]
    #          - h[i+k] if 0<=i+k<n     [h[i] in second factor, with negative sign]
    if k > 0:
        grad[k:] += (1.0 - h[:-k])  # h[i] in first factor: i-k >= 0
        grad[:-k] -= h[k:]          # h[i] in second factor: i+k < n

        g = np.correlate(x, 1.0 - x, mode='full') * dx
        mx = g.max()
        shifted = alpha * (g - mx)
        sm = shifted.max()
        logsum = np.log(np.exp(shifted - sm).sum()) - np.log(m)
        return mx + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_delta(h, n, i, j, m_range):
    """Compute correlation change when setting h[i]=1, h[j]=0."""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def binary_best_improve(h_init, n, deadline, max_sweeps=200):
    """Best-improvement binary local search."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_delta_c = 0.0
        best_pair = None
        best_delta_arr = None
        
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = compute_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_delta_c:
                    best_delta_c = dc
                    best_pair = (i, j)
                    best_delta_arr = delta
        
        if best_pair is not None and best_delta_c < -1e-15:
            i, j = best_pair
            h[i], h[j] = 1.0, 0.0
            corr += best_delta_arr
            cur_c = (corr).max() * dx
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c

def binary_anneal(h_init, n, deadline, seed=42, n_iters=2000000):
    """Simulated annealing for binary improvement."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = 0.02
    cool = 0.99998
    scale = 5000.0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        
        delta = compute_delta(h, n, i, j, m_range)
        nc = (corr + delta).max() * dx
        d = nc - cur_c
        
        if d <= 0 or (T > 1e-12 and rng.random() < np.exp(-d * scale / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    
    return best_h, best_c

- Each pair requires O(n) = O(400) to compute delta
- Total per sweep: 40,000 × 400 = 16,000,000 operations

But wait, the `compute_delta` function uses NumPy operations that create temporary arrays. Let me optimize it.

Actually, looking at the delta computation more carefully:

For the swap h[i]→1, h[j]→0:
- The change in correlation at lag k is:
  - Position i contributes: when h[i] changes from 0 to 1
  - Position j contributes: when h[j] changes from 1 to 0

The correlation at lag k (k≥0) is: c[k] = sum_{t=0}^{n-1-k} h[t+k]*(1-h[t])

Change from h[i]: 0→1
- If i is in the "h[t+k]" slot: t+k=i, t=i-k. Contribution: (1-h[i-k]) if valid t
- If i is in the "(1-h[t])" slot: t=i. Contribution: -h[i+k] if valid

Change from h[j]: 1→0
- If j is in the "h[t+k]" slot: t+k=j, t=j-k. Contribution: -(1-h[j-k]) → -1+h[j-k] if valid
- If j is in the "(1-h[t])" slot: t=j. Contribution: +h[j+k] if valid

This is what the `compute_delta` function does, but using vectorized NumPy operations. The vectorized version is actually faster than a loop for n=400.

Let me also think about whether we can vectorize the "try all pairs" loop. For a fixed i (zero position), we want to try all j (one positions). The delta for pair (i,j) can be decomposed as:
delta(i,j) = delta_i_add + delta_j_remove + cross_term(i,j)

Where:
- delta_i_add = change from setting h[i]=0→1 (independent of j)
- delta_j_remove = change from setting h[j]=1→0 (independent of i)
- cross_term = the interaction term

Let me derive this:

The full delta for swapping i (0→1) and j (1→0):
- Set h[i]=1: delta_i[pos] for all pos
- Set h[j]=0: delta_j[pos] for all pos
- But there's a double-counting issue at positions where both affect the same correlation

Actually, the cross term is:
- For correlation at lag k: h[i] appears in terms where i = t+k or i = t
- h[j] appears in terms where j = t+k or j = t
- The cross term is when i is in one role and j is in the other role simultaneously
- This happens when: i = t+k AND j = t (so i-j = k) or j = t+k AND i = t (so j-i = k)

So the cross term is non-zero only at lags k = i-j and k = j-i!

This means:
delta(i,j) = delta_i + delta_j + cross(i,j)

where cross(i,j) is non-zero only at 2 specific lags. This allows us to:
1. Precompute delta_i for all zeros i
2. Precompute delta_j for all ones j
3. For each pair (i,j), just add them and fix up 2 positions

This reduces the per-pair cost from O(n) to O(1) (after O(n^2) precomputation).

Total cost: O(n^2) precomputation + O(n^2) for trying all pairs (since there are O(n^2) pairs and each takes O(1)).
For n=400: O(160,000) per sweep. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        ps = np.clip(h - mid, 0.0, 1.0).sum()
        if ps > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_smooth(h, n, alpha, maxiter=5000, ftol=1e-18):
    dx = 2.0 / n
    m = 2 * n - 1
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        mx = g.max()
        shifted = alpha * (g - mx)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 0)
        sm = shifted.max()
        logsum = np.log(np.exp(shifted - sm).sum()) - np.log(m)
        return mx + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
