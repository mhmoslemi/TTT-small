                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_opt(h, n, deadline, top_m=6, max_rounds=15):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 5:
                break
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
            try:
                res = minimize(_active_smooth_obj, h_curr, args=(n, top_idx, alpha),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_curr = h_new
            except Exception:
                pass
    return best_h, best_c

def nesterov_subgrad(h, n, deadline, top_k=5):
    h_curr = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.85
    lr = 0.005
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    patience = 0
    max_patience = 80
    while time.time() < deadline - 2:
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_k:]
        active = c[top_idx]
        m = active.max()
        shifted = 2000.0 * (active - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lag = k - (n - 1)
            grad += w * _lag_grad(h_curr, n, lag)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        v = beta * v - lr * grad / (gn + 1e-10)
        h_curr = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_curr, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_curr.copy()
            patience = 0
            lr = min(lr * 1.02, 0.05)
        else:
            patience += 1
            if patience > max_patience:
                lr *= 0.5
                patience = 0
                v *= 0.3
                if lr < 1e-6:
                    break
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for delta in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            if lag > 0:
                src, dst = 0, m
            else:
                src, dst = m, 0
            n_pair = n - m
            for i in range(n_pair):
                if time.time() > deadline:
                    break
                a = i + src
                b = i + dst
                if a >= n or b >= n:
                    continue
                if h[a] > h[b] + 0.005:
                    d = min(delta, (h[a] - h[b]) * 0.4, h[a], 1.0 - h[b])
                    if d < 1e-9:
                        continue
                    h[a] -= d
                    h[b] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[a] += d
                        h[b] -= d
                elif h[b] > h[a] + 0.005:
                    d = min(delta, (h[b] - h[a]) * 0.4, h[b], 1.0 - h[a])
                    if d < 1e-9:
                        continue
                    h[b] -= d
                    h[a] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[b] += d
                        h[a] -= d
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP
    if time.time() < deadline - 30:
        for alpha in [10000, 100000]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_opt(best_h, n, alpha, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 2: Nesterov subgradient
    if time.time() < deadline - 20:
        alloc = min(80, (deadline - time.time()) * 0.3)
        h_new, c_new = nesterov_subgrad(best_h, n, time.time() + alloc)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 15:
        alloc = min(80, (deadline - time.time()) * 0.35)
        h_new, c_new = active_set_opt(best_h, n, time.time() + alloc)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 4: Final SLSQP polish
    if time.time() < deadline - 10:
        h_new = slsqp_opt(best_h, n, 500000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 5: Local search
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(150, (deadline - t0) * 0.2)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    # Try other n values
    n_candidates = [200, 250, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.22)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Multi-start perturbations on best
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 40) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 + 0.01 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 1000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
