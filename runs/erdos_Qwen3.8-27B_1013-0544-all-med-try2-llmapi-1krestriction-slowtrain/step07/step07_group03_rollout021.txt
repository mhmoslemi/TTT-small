```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_max_obj(x, n, alpha):
    """Log-sum-exp approximation of max of correlations"""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    return max_c + np.log(np.exp(shifted).sum()) / alpha

def smooth_max_grad(x, n, alpha):
    """Gradient of log-sum-exp approximation of max of correlations"""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    p = np.exp(shifted)
    p /= p.sum()
    q = 1.0 - 2.0 * x
    grad = dx * np.correlate(p, q, mode='valid')
    return grad

def projected_gradient_descent(h, n, deadline, alpha_init=50, alpha_final=100000, 
                                 lr_init=0.01, n_steps=5000):
    """Projected gradient descent with smooth max"""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Adaptive alpha
        progress = step / n_steps
        alpha = alpha_init * (alpha_final / alpha_init) ** progress
        # Adaptive lr (decrease over time)
        lr = lr_init * (1.0 / (1.0 + 0.01 * step))
        
        grad = smooth_max_grad(h, n, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c
```

Actually, I realize there's a subtlety with the projected gradient. The projection to {sum=h_target, 0<=h<=1} is not trivial - it's a convex projection problem. The iterative method I wrote (alternating between sum adjustment and clipping) converges but might be slow.

A better approach: use the "water-filling" algorithm. To project onto {0<=h<=1, sum(h)=S}:
1. Clip to [0,1]
2. If sum > S: find threshold t such that sum(clip(h-t, 0, 1-t)) + min(t, ...)... 

Actually, the projection onto {0<=h<=1, sum(h)=S} can be done as:
- If all h in [0,1] and sum = S, done
- Otherwise, we need to find t such that sum(clip(h - t, 0, 1)) = S
  (for the case sum(h) > S, we decrease; for sum(h) < S, we increase)

Actually the standard result is: the projection of x onto {h: 0<=h<=1, sum(h)=S} is:
h* = clip(x - lambda, 0, 1) where lambda is chosen so sum(h*) = S

This is because the constraint set is the intersection of the box [0,1]^n and the hyperplane sum(h)=S. The projection can be found by bisection on lambda.

Let me implement this properly:

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    
    # Bisection on lambda
    # We want clip(h - lambda, 0, 1) to sum to target
    # f(lambda) = sum(clip(h - lambda, 0, 1)) is continuous and non-increasing in lambda
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune to fix any residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        # Adjust the interior points
        interior = (h_new > 1e-8) & (h_new < 1.0 - 1e-8)
        if interior.sum() > 0:
            h_new[interior] += diff / interior.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def swap_delta(h, i, j, n, m_range):
    d = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    d[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    d[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    d[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    d[msk] += h[jmn[msk]]
    return d

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = c.max()
    sh = alpha * (c - mc)
    return mc + np.log(np.exp(sh).sum()) / alpha

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = c.max()
    sh = alpha * (c - mc)
    p = np.exp(sh)
    p /= p.sum()
    q = 1.0 - 2.0 * x
    return dx * np.correlate(p, q, mode='valid')
```
