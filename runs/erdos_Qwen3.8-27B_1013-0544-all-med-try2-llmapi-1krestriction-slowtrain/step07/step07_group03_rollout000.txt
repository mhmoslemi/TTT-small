```
def project_feasible(y, n):
    target=0.5*n
    h=np.clip(np.asarray(y,dtype=float),0.0,1.0).copy()
    for _ in range(50):
       diff=target-h.sum()
       if abs(diff)<1e-13: break
       if diff>0:
          room=1.0-h; s=room.sum()
          if s<1e-15: break
          h += diff*room/s
       else:
          s=h.sum()
          if s<1e-15: break
          h += diff*h/s
    h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    if abs(diff)>1e-15:
       if diff>0:
          # add to values with largest room
          idx=np.argsort(1.0-h)[::-1]
          for a in idx:
             add=min(1.0-h[a], diff)
             h[a]+=add; diff-=add
             if abs(diff)<1e-15: break
       else:
          idx=np.argsort(h)
          for a in idx:
             sub=min(h[a], -diff)
             h[a]-=sub; diff+=sub
             if abs(diff)<1e-15: break
    # final exact-ish
    diff=target-h.sum()
    if abs(diff)>0:
       # choose a coordinate with room
       if diff>0:
          idx=int(np.argmax(1.0-h)); h[idx]=min(1.0, h[idx]+diff)
       else:
          idx=int(np.argmin(h)); h[idx]=max(0.0, h[idx]+diff)
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def corr_dx(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def compute_c5(h, n):
    return float(np.max(corr_dx(h, n)))


def project_feasible(y, n):
    h = np.array(y, dtype=float, copy=True)
    if h.size != int(n):
        h = np.resize(h, int(n))
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for a in order:
                if diff <= 1e-15:
                    break
                add = min(1.0 - h[a], diff)
                h[a] += add
                diff -= add
        else:
            order = np.argsort(h)
            for a in order:
                if -diff <= 1e-15:
                    break
                sub = min(h[a], -diff)
                h[a] -= sub
                diff += sub
    return h


def make_even(n):
    n = int(n)
    if n < 20:
        n = 20
    if n % 2:
        n += 1
    return n


def resample_to(h, n_new):
    n_old = int(len(h))
    n_new = make_even(n_new)
    if n_old == n_new:
        return project_feasible(h, n_new).copy()
    if n_old < 2:
        return project_feasible(np.full(n_new, float(np.mean(h))), n_new)
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    y = np.interp(xn, xo, h)
    return project_feasible(y, n_new)


def binarize(h, n):
    order = np.argsort(h)[::-1]
    b = np.zeros(int(n), dtype=float)
    b[order[:int(n) // 2]] = 1.0
    return b


def _accumulate_gradient(grad, h, k, n, dx, weight):
    l0 = max(0, -k)
    l1 = min(n - 1, n - 1 - k)
    if l1 >= l0:
        grad[l0:l1 + 1] += weight * (1.0 - h[l0 + k:l1 + k + 1])
        grad[l0 + k:l1 + k + 1] -= weight * h[l0:l1 + 1]


_SM_N = 400
_SM_ALPHA = 1.0


def _smo_cons(x):
    return float(x[:_SM_N].sum() - 0.5 * _SM_N)


def _smo_cons_jac(x):
    return np.ones(_SM_N, dtype=float)


def _smo_obj(x):
    n = _SM_N
    h = x[:n]
    c = corr_dx(h, n)
    m = float(c.max())
    a = _SM_ALPHA
    z = a * (c - m)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return m
    return m + float(np.log(s)) / a


def _smo_jac(x):
    n = _SM_N
    h = x[:n]
    c = corr_dx(h, n)
    m = float(c.max())
    a = _SM_ALPHA
    z = a * (c - m)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    grad = np.zeros(n, dtype=float)
    dx = 2.0 / n
    active = np.nonzero(w > 1e-12)[0]
    for q in active:
        k = int(q - (n - 1))
        _accumulate_gradient(grad, h, k, n, dx, float(w[q]))
    return grad


def smooth_slsqp(h0, n, alpha, maxiter):
    global _SM_N, _SM_ALPHA
    _SM_N = int(n)
    _SM_ALPHA = float(alpha)
    x0 = project_feasible(h0, n).copy()
    if len(x0) != n:
        x0 = project_feasible(np.resize(x0, n), n)
    bounds = [(0.0, 1.0)] * n
    cons = [
        {"type": "eq", "fun": _smo_cons, "jac": _smo_cons_jac}
    ]
    try:
        res = minimize(
            _smo_obj, x0, jac=_smo_jac, method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False}
        )
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(x0, n)


_ACT_N = 400
_ACT_Q = np.empty(0, dtype=int)


def _act_obj(x):
    return float(x[_ACT_N])


def _act_obj_jac(x):
    g = np.zeros(_ACT_N + 1, dtype=float)
    g[-1] = 1.0
    return g


def _act_cons(x):
    n = _ACT_N
    h = x[:n]
    t = float(x[n])
    c = corr_dx(h, n)
    return t - c[_ACT_Q]


def _act_jac(x):
    n = _ACT_N
    h = x[:n]
    K = int(len(_ACT_Q))
    J = np.zeros((K, n + 1), dtype=float)
    dx = 2.0 / n
    for a, q in enumerate(_ACT_Q):
        k = int(q - (n - 1))
        l0 = max(0, -k)
        l1 = min(n - 1, n - 1 - k)
        if l1 >= l0:
            J[a, l0:l1 + 1] += -dx * (1.0 - h[l0 + k:l1 + k + 1])
            J[a, l0 + k:l1 + k + 1] += dx * h[l0:l1 + 1]
        J[a, n] = 1.0
    return J


def _act_eq(x):
    return float(x[:_ACT_N].sum() - 0.5 * _ACT_N)


def _act_eq_jac(x):
    n = _ACT_N
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


def active_slsqp(h0, n, deadline, max_active=50, maxiter=40, repeats=1):
    global _ACT_N, _ACT_Q
    n = int(n)
    best = project_feasible(h0, n).copy()
    cur = best.copy()
    cc = corr_dx(cur, n)
    best_c = float(np.max(cc))
    cur_c = float(np.max(cc))
    if not np.isfinite(best_c):
        return best, best_c

    M = 2 * n - 1
    Kseq = []
    for kk in (20, 35, int(max(1, max_active))):
        kk = int(min(M, max(1, kk)))
        if kk not in Kseq:
            Kseq.append(kk)

    for maxK in Kseq:
        for _ in range(int(repeats)):
            if time.time() > deadline:
                break
            if maxK <= 0:
                continue
            order = np.argsort(cc)[::-1]
            active = order[:maxK].astype(int, copy=False)
            _ACT_N = n
            _ACT_Q = active
            t0 = max(best_c, float(np.max(cc[active]))) + 1e-9
            x0 = np.empty(n + 1, dtype=float)
            x0[:n] = cur
            x0[n] = min(2.5, max(0.0, t0))
            bounds = [(0.0, 1.0)] * n + [(0.0, 2.5)]
            cons = [
                {"type": "ineq", "fun": _act_cons, "jac": _act_jac},
                {"type": "eq", "fun": _act_eq, "jac": _act_eq_jac}
            ]
            try:
                res = minimize(
                    _act_obj, x0, jac=_act_obj_jac, method="SLSQP",
                    bounds=bounds, constraints=cons,
                    options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False}
                )
            except Exception:
                continue
            hnew = project_feasible(res.x[:n], n)
            cnew = corr_dx(hnew, n)
            cmaxnew = float(np.max(cnew))
            if not np.isfinite(cmaxnew):
                continue
            if cmaxnew < best_c - 1e-12:
                best = hnew.copy()
                best_c = cmaxnew
            if cmaxnew <= best_c + 1e-10:
                cur = hnew.copy()
                cc = cnew
                cur_c = cmaxnew
    return best, best_c


def gradient_matrix(h, n):
    n = int(n)
    M = 2 * n - 1
    G = np.zeros((M, n), dtype=float)
    dx = 2.0 / n
    for q in range(M):
        k = q - (n - 1)
        l0 = max(0, -k)
        l1 = min(n - 1, n - 1 - k)
        if l1 >= l0:
            G[q, l0:l1 + 1] += 1.0 - h[l0 + k:l1 + k + 1]
            G[q, l0 + k:l1 + k + 1] -= h[l0:l1 + 1]
    return G * dx


def lp_polish(h0, n, deadline, r=0.03, passes=2):
    if linprog is None:
        return project_feasible(h0, n), compute_c5(h0, n)
    n = int(n)
    best = project_feasible(h0, n).copy()
    bc = compute_c5(best, n)
    M = 2 * n - 1
    for _ in range(int(passes)):
        if time.time() > deadline:
            break
        c = corr_dx(best, n)
        G = gradient_matrix(best, n)
        lb = np.maximum(-float(r), -best)
        ub = np.minimum(float(r), 1.0 - best)
        A_ub = np.empty((M, n + 1), dtype=float)
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -c
        A_eq = np.zeros((1, n + 1), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1, dtype=float)
        obj = np.zeros(n + 1, dtype=float)
        obj[n] = 1.0
        bounds = list(zip(lb, ub)) + [(0.0, 2.5)]
        res = None
        try:
            res = linprog(
                obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method="highs", options={"presolve": True}
            )
        except Exception:
            try:
                res = linprog(
                    obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method="revised simplex"
                )
            except Exception:
                break
        if res is None or not getattr(res, "success", False):
            break
        y = np.array(res.x[:n], dtype=float, copy=True)
        if not np.all(np.isfinite(y)):
            break
        y -= y.mean()
        improved = False
        for step in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            hn = project_feasible(best + step * y, n)
            cn = float(chew(cm) if False else np.max(corr_dx(hn, n)))
            if np.isfinite(cn) and cn < bc - 1e-12:
                best = hn
                bc = cn
                improved = True
                break
        if not improved:
            break
    return best, bc


def polish_candidate(h0, n, deadline, heavy=False):
    n = make_even(n)
    best = project_feasible(h0, n).copy()
    bc = compute_c5(best, n)
    if not np.isfinite(bc):
        return best, bc
    scale = 1.0
    if n > 450:
        scale = max(0.35, 450.0 / float(n))

    if time.time() < deadline - 15:
        h2, c2 = active_slsqp(
            best, n, deadline - 10,
            max_active=55 if heavy else 35,
            maxiter=int((45 if heavy else 25) * scale),
            repeats=1
        )
        if c2 < bc:
            best, bc = h2, c2

    if heavy:
        alphas = [(1000, 40), (10000, 30), (100000, 25)]
    else:
        alphas = [(5000, 24), (50000, 18)]
    for alpha, itx in alphas:
        if time.time() > deadline - 10:
            break
        it_eff = max(8, int(float(itx) * scale))
        try:
            h2 = smooth_slsqp(best, n, alpha, it_eff)
            c2 = compute_c5(h2, n)
        except Exception:
            continue
        if np.isfinite(c2) and c2 < bc:
            best, bc = h2, c2

    if time.time() < deadline - 12:
        h2, c2 = active_slsqp(
            best, n, deadline - 8,
            max_active=65 if heavy else 45,
            maxiter=int((50 if heavy else 28) * scale),
            repeats=2 if heavy else 1
        )
        if c2 < bc:
            best, bc = h2, c2

    if time.time() < deadline - 8:
        h2, c2 = lp_polish(
            best, n, deadline,
            r=0.045 if heavy else 0.03,
            passes=2 if heavy else 1
        )
        if c2 < bc:
            best, bc = h2, c2

    if heavy and time.time() < deadline - 12:
        h2, c2 = active_slsqp(
            best, n, deadline - 5,
            max_active=45,
            maxiter=int(32 * scale),
            repeats=1
        )
        if c2 < bc:
            best, bc = h2, c2

    return best, bc


def candidate_ns(n0):
    raw = [
        int(n0),
        make_even(int(0.85 * n0)),
        make_even(int(0.95 * n0)),
        make_even(int(1.05 * n0)),
        make_even(int(1.15 * n0)),
        300,
        480,
        600
    ]
    out = []
    for v in raw:
        v = make_even(v)
        if 120 <= v <= 700 and v not in out:
            out.append(v)
    out.sort()
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 10.0)
    deadline = t0 + total

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = int(h0.size)
    except Exception:
        n0 = 400
        h0 = np.zeros(n0, dtype=float)
        h0[0:n0:2] = 1.0

    n0 = make_even(max(120, n0))
    if n0 > 700:
        n0 = 500
    if len(h0) != n0:
        h0 = resample_to(h0, n0)

    besth = project_feasible(h0, n0).copy()
    bestn = n0
    bestc = compute_c5(besth, bestn)

    hl = int(0.42 * total)
    besth, bestc = polish_candidate(besth, bestn, t0 + hl, heavy=True)
    bestn = int(len(besth))
    bestc = compute_c5(besth, bestn)

    seen = {bestn}
    for nh in candidate_ns(bestn):
        if nh in seen:
            continue
        if time.time() > deadline - 80:
            break
        seen.add(nh)
        hh = resample_to(besth, nh)
        cc = compute_c5(hh, nh)
        if np.isfinite(cc) and cc < bestc:
            besth, bestc, bestn = hh, cc, nh
        remain = deadline - 60
        if time.time() > max(t0 + 60.0, remain):
            continue
        quick = cc > bestc + 0.03
        cand_end = min(
            max(t0 + 60.0, remain),
            time.time() + (16.0 if quick else 38.0)
        )
        if cand_end <= time.time() + 3.0:
            continue
        h2, c2 = polish_candidate(hh, nh, cand_end, heavy=False)
        if c2 < bestc:
            besth, bestc, bestn = h2, c2, nh

    rng = np.random.RandomState(seed)
    for _ in range(2):
        if time.time() > deadline - 45:
            break
        noisy = project_feasible(besth + 0.008 * rng.randn(bestn), bestn)
        cno = compute_c5(noisy, bestn)
        if not np.isfinite(cno) or cno > bestc + 0.01:
            continue
        end = min(deadline - 35, time.time() + 28.0)
        if end <= time.time() + 3.0:
            continue
        h2, c2 = polish_candidate(noisy, bestn, end, heavy=False)
        if c2 < bestc:
            besth, bestc = h2, c2

    for shift in (bestn // 4, -bestn // 4, bestn // 8, -bestn // 8):
        if time.time() > deadline - 35:
            break
        rolled = np.roll(besth, int(shift))
        cr = compute_c5(rolled, bestn)
        if not np.isfinite(cr) or cr > bestc + 0.004:
            continue
        end = min(deadline - 25, time.time() + 22.0)
        if end <= time.time() + 3.0:
            continue
        h2, c2 = polish_candidate(rolled, bestn, end, heavy=False)
        if c2 < bestc:
            besth, bestc = h2, c2

    if time.time() < deadline - 35:
        h2, c2 = polish_candidate(besth, bestn, deadline - 5, heavy=True)
        if c2 < bestc:
            besth, bestc = h2, c2

    if bestn % 2 == 0:
        hb = binarize(besth, bestn)
        cb = compute_c5(hb, bestn)
        if cb < bestc:
            besth, bestc = hb, cb
        hc = 1.0 - hb
        cc = compute_c5(hc, bestn)
        if cc < bestc:
            besth, bestc = hc, cc

    besth = project_feasible(besth, bestn)
    bestc = compute_c5(besth, bestn)
    return besth, bestc, int(bestn)
```
