from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0 - h[:n-m]
    else:
        grad[:n-m] += 1.0 - h[m:]
        grad[m:] -= h[:n-m]
    return grad

def subgrad_descent(h, n, deadline, max_iters=500):
    """Subgradient method with momentum for the max-of-lags problem."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    momentum = np.zeros(n)
    lr = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        # Find top lags
        top_m = min(10, len(c))
        top_idx = np.argsort(c)[-top_m:]
        top_vals = c[top_idx]
        # Soft max weights
        alpha_w = 50.0
        w = np.exp(alpha_w * (top_vals - top_vals.max()))
        w /= w.sum()
        # Weighted gradient
        g = np.zeros(n)
        for i, idx in enumerate(top_idx):
            lag = idx - (n - 1)
            g += w[i] * lag_grad(h_curr, n, lag)
        # Momentum

        h_new = project_feasible(h_curr + momentum, n)
        c_new = compute_c5(h_new, n)

            lr = min(lr * 1.05, 0.05)

            lr *= 0.8

        n = len(best_h)
        h_new = slsqp_refine(best_h, n, alpha=500000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            h_new, c_new = flow_local_search(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    return best_h, best_c, len(best_h)
