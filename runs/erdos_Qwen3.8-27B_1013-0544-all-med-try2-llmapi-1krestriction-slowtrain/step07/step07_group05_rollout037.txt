import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sa_optimize(h_start, n, deadline, rng, t0_init=0.01, cooling=0.999, min_t=1e-8):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    T = t0_init
    steps = 0
    no_improve = 0
    
    while time.time() < deadline:
        # Adaptive perturbation size
        scale = max(0.005, 0.1 * T / t0_init)
        
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        
        delta = rng.uniform(-scale, scale)
        
        if h[i] + delta > 1.0 or h[j] - delta < 0.0:
            continue
        
        h[i] += delta
        h[j] -= delta
        
        new_c = compute_c5(h, n)
        
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] -= delta
            h[j] += delta
            no_improve += 1
        
        steps += 1
        T *= cooling
        
        # Reheat if stuck
        if no_improve > 50000 and T > min_t * 10:
            T = max(T, t0_init * 0.1)
            no_improve = 0
        
        # Cool faster if very good
        if steps % 100000 == 0:
            # Check time
            pass
    
    return best_h, best_c

            if max_d < 1e-12:
                T *= cooling
                continue
            delta = rng.uniform(-max_d, max_d) if delta > 0 else -rng.uniform(0, max_d)

        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h, n)
        total_count += 1

        if new_c <= cur_c or (T > min_t and rng.random() < math.exp(-(new_c - cur_c) / T)):
            cur_c = new_c
            accept_count += 1
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] -= delta
            h[j] += delta
            no_improve += 1

        steps += 1
        T *= cooling
        scale = max(min_scale, scale_init * max(0.01, T / t_init) ** 0.5)

        # Adaptive temperature adjustment
        if steps % 20000 == 0:
            rate = accept_count / max(1, total_count)
            if rate > 0.7:
                T *= 0.9
            elif rate < 0.05 and T > min_t * 10:
                T *= 1.5
                scale = min(scale_init, scale * 1.2)
            accept_count = 0
            total_count = 0

        # Reheat and reset if stuck
        if no_improve > 150000 and T > min_t * 50:
            h = best_h.copy()
            cur_c = best_c
            T = t_init * 0.2
            scale = scale_init * 0.2
            no_improve = 0

    return best_h, best_c

def local_pair_search(h, n, deadline, rng, n_trials_per_delta=20000):
    """Greedy local search with random pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for d in deltas:
        if time.time() > deadline - 2:
            break
        improved_any = False
        for _ in range(n_trials_per_delta):
            if time.time() > deadline - 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for sign in [1, -1]:
                nd = sign * d
                if h[i] + nd > 1.0 or h[j] - nd < 0.0:
                    continue
                h[i] += nd
                h[j] -= nd
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved_any = True
                else:
                    h[i] -= nd
                    h[j] += nd
        if not improved_any and d < 0.005:
            break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)

    # Phase 1: SLSQP from initial (45s)
    p1_end = min(t0 + 45, deadline - 800)
    for alpha in [2000, 10000, 50000]:
        if time.time() > p1_end:
            break
        ht = slsqp(h_init, n_init, alpha, maxiter=2000)
        c = compute_c5(ht, n_init)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 2: SA exploration (350s)
    p2_end = min(time.time() + 350, deadline - 400)
    if p2_end > time.time() + 30:
        h_sa, c_sa = sa_with_gradient(best_h, n_init, p2_end, rng,
                                      t_init=0.004, cooling=0.9996,
                                      scale_init=0.12, min_scale=0.001)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 3: SLSQP polish (80s)
    p3_end = min(time.time() + 80, deadline - 300)
    if p3_end > time.time() + 10:
        for alpha in [100000, 300000, 800000]:
            if time.time() > p3_end:
                break
            ht = slsqp(best_h, n_init, alpha, maxiter=2500)
            c = compute_c5(ht, n_init)
            if c < best_c:
                best_c = c
                best_h = ht

    # Phase 4: SA exploitation - lower temp (200s)
    p4_end = min(time.time() + 200, deadline - 180)
    if p4_end > time.time() + 20:
        h_sa, c_sa = sa_with_gradient(best_h, n_init, p4_end, rng,
                                      t_init=0.0015, cooling=0.9998,
                                      scale_init=0.04, min_scale=0.0003)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 5: Local pair search (120s)
    p5_end = min(time.time() + 120, deadline - 60)
    if p5_end > time.time() + 10:
        best_h, best_c = local_pair_search(best_h, n_init, p5_end, rng)

    # Phase 6: Final SLSQP + local search
    if time.time() < deadline - 20:
        ht = slsqp(best_h, n_init, 1500000, maxiter=3000)
        c = compute_c5(ht, n_init)
        if c < best_c:
            best_c = c
            best_h = ht

    if time.time() < deadline - 5:
        best_h, best_c = local_pair_search(best_h, n_init, deadline - 2, rng,
                                            n_trials_per_delta=5000)

    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
