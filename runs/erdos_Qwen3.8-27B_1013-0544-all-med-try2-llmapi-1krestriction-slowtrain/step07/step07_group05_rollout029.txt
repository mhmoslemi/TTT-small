    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    # O(k) = s_k * h[i] + b_k
    # s_k = dx * [(1-h[i+k]) if valid - h[i-k] if valid]  ... wait
    # Actually, let me just compute O(k) at h[i]=0 and h[i]=1, then interpolate
    
    h[i] = 0.0
    c_0 = np.correlate(h, 1 - h, mode='full') * dx
    h[i] = 1.0
    c_1 = np.correlate(h, 1 - h, mode='full') * dx
    
    # O(k) is linear in h[i], so:
    # O(k) = c_0[k] + h[i] * (c_1[k] - c_0[k])
    slope = (c_1 - c_0)  # slope for each k
    
    # We want to minimize max_k (c_0[k] + x * slope[k]) for x in [0,1]
    # This is a 1D convex piecewise linear minimization
    
    # The minimum is at x where the max transitions
    # Use bisection on the "active slope"
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c_0 + mid * slope
        max_val = vals.max()
        active = np.where(vals >= max_val - 1e-15)[0]
        avg_slope = slope[active].mean()
        if avg_slope > 1e-15:
            hi = mid  # decreasing, go left
        elif avg_slope < -1e-15:
            lo = mid  # increasing, go right
        else:
            break
    
    h[i] = (lo + hi) / 2
    return compute_c5(h, n)

Wait, this is elegant but I need to be careful. Let me verify the linearity claim.

O(k) = Σ_j h[j] * (1 - h[j+k]) * dx (with valid index restrictions)

When I vary h[i]:
- Term j=i: h[i] * (1 - h[i+k]) * dx → this is h[i] * (1 - h[i+k]) * dx
- Term j where j+k=i: h[i-k] * (1 - h[i]) * dx → this is h[i-k] * (1 - h[i]) * dx

So the part depending on h[i] is:
h[i]*(1-h[i+k])*dx + h[i-k]*(1-h[i])*dx

= dx * [h[i] - h[i]*h[i+k] + h[i-k] - h[i]*h[i-k]]

Wait, h[i]*h[i+k] is NOT linear in h[i] if h[i+k] is also variable! But in coordinate descent, we're fixing all other coordinates. So h[i+k] is a constant (not equal to h[i] unless k=0).

For k=0: term is h[i]*(1-h[i])*dx which IS quadratic in h[i]. But wait, for k=0, the correlation term is Σ_j h[j]*(1-h[j])*dx. The term for j=i is h[i]*(1-h[i])*dx = (h[i] - h[i]²)*dx. This is quadratic!

So for k=0, the function is not linear in h[i]. For k≠0, it is linear (assuming i+k ≠ i, i.e., k ≠ 0).

So the overall C₅ as a function of h[i] is:
max( h[i] - h[i]², max_{k≠0} (linear in h[i]) ) * dx

The first term is a concave parabola (opens downward), and the others are linear. The max of these is not necessarily convex!

Hmm, this complicates things. Let me just use the evaluation approach: compute C₅ at h[i]=0 and h[i]=1, and note that for k=0, the contribution is h[i](1-h[i])*dx which is bounded by dx/4. For k≠0, it's linear.

Here's a key optimization: I can precompute which lags are "active" (near the maximum) and only evaluate those. If only, say, 20 lags are active, I can compute those 20 correlation values in O(20*n) instead of O(n²).

```python
def compute_max_active(h, n, active_lags):
    """Compute only the active correlation values."""
    dx = 2.0 / n
    max_val = 0.0
    for k in active_lags:
        s = k - (n - 1)  # actual lag
        # Compute sum h[i] * (1-h[i+s]) for valid i
        if s >= 0:
            valid = slice(0, n - s)
            val = np.sum(h[valid] * (1 - h[valid + s])) * dx
        else:
            valid = slice(-s, n)
            val = np.sum(h[valid] * (1 - h[valid + s])) * dx
        max_val = max(max_val, val)
    return max_val
```

This is O(|A| * n) instead of O(n²). For |A|=20 and n=250, that's 5000 operations vs 62500. About 10x faster.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coordinate_optimize(h, n, i, n_coarse=100, n_fine=200):
    """Optimize coordinate i via 1D search."""
    best_c = compute_c5(h, n)
    orig = h[i]
    coarse = np.linspace(0.0, 1.0, n_coarse)
    best_val = best_c
    best_x = orig
    for x in coarse:
        h[i] = x
        c = compute_c5(h, n)
        if c < best_val:
            best_val = c
            best_x = x
    if best_x == orig:
        h[i] = orig
        return best_c
    fine = np.linspace(max(0.0, best_x - 0.02), min(1.0, best_x + 0.02), n_fine)
    for x in fine:
        h[i] = x
        c = compute_c5(h, n)
        if c < best_val:
            best_val = c
            best_x = x
    h[i] = best_x
    return best_val

def coordinate_descent(h, n, deadline, max_passes=200):
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            c = coordinate_optimize(h, n, i)
            if c < best_c - 1e-14:
                best_c = c
                improved = True
        if not improved:
            break
    return h, best_c

def subgradient_step(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    active = np.where(c >= max_c - 1e-6)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += (1 - h[s:]) * dx
            grad[s:] -= h[:n-s] * dx
        else:
            grad[-s:] += (1 - h[:n+s]) * dx
            grad[:n+s] -= h[-s:] * dx
    grad /= max(1, len(active))
    return grad, max_c

def subgradient_optimize(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current = h.copy()
    momentum = np.zeros(n)
    beta = 0.7
    step = 0.1
    patience = 0
    while time.time() < deadline:
        grad, max_c = subgradient_step(current, n)
        if max_c < best_c - 1e-10:
            best_c = max_c
            best_h = current.copy()
            patience = 0
        else:
            patience += 1
        if patience > 2000:
            current = best_h + rng.normal(0, 0.04, n)
            current = project_feasible(current, n)
            momentum = np.zeros(n)
            step = 0.1
            patience = 0
            continue
        momentum = beta * momentum - step * grad
        current = project_feasible(current + momentum, n)
        step = max(0.005, step * 0.9995)
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_bin = (h > 0.5).astype(float)
    deficit = n / 2.0 - h_bin.sum()
    if deficit > 0:
        non_ones = np.where(h_bin < 1.0)[0]
        idx = np.argsort(h[non_ones])[::-1][:int(np.ceil(deficit))]
        h_bin[non_ones[idx]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin > 0)[0]
        idx = np.argsort(h[ones])[:int(np.ceil(-deficit))]
        h_bin[ones[idx]] = 0.0
    c = compute_c5(h_bin, n)
    if c < best_c:
        best_c = c
        best_h = h_bin.copy()
    current = h_bin.copy()
    for _sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(current > 0.5)[0])
        zeros = list(np.where(current < 0.5)[0])
        rng_local = np.random.RandomState(42 + _sweep)
        order = list(range(len(ones)))
        rng_local.shuffle(order)
        for oi in order:
            if time.time() > deadline:
                return best_h, best_c
            i = ones[oi]
            for j in zeros:
                current[i] = 0.0
                current[j] = 1.0
                c = compute_c5(current, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = current.copy()
                    improved = True
                else:
                    current[i] = 1.0
                    current[j] = 0.0
        if not improved:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

        if c1b < best_c:
            best_c, best_h = c1b, h1b

    # Phase 2: Coordinate descent
    if time.time() < deadline - 200:
        alloc2 = min(180, (deadline - time.time()) * 0.25)
        h2, c2 = coordinate_descent(best_h.copy(), n, time.time() + alloc2)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Binary local search
    if time.time() < deadline - 180:
        alloc3 = min(100, (deadline - time.time()) * 0.15)
        h3, c3 = binary_local_search(best_h, n, time.time() + alloc3)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Subgradient from best
    if time.time() < deadline - 120:
        alloc4 = min(150, (deadline - time.time()) * 0.2)
        h4, c4 = subgradient_optimize(best_h.copy(), n, time.time() + alloc4, rng)
        if c4 < best_c:
            best_c, best_h = c4, h4

    if time.time() < deadline - 80:
        h5 = slsqp(best_h, n, 100000, maxiter=2000)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Coordinate descent again
    if time.time() < deadline - 60:
        h6, c6 = coordinate_descent(best_h.copy(), n, time.time() + min(60, (deadline - time.time()) * 0.3))
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Fine pair search
    if time.time() < deadline - 5:
        best_h, best_c = pair_search(best_h, n, deadline - 2,
            [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
