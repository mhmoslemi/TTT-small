def active_gradient(h, n, lags, weights, dx):
    g = np.zeros(n, dtype=float)
    for idx in range(len(lags)):
        wt = weights[idx]
        if wt <= 0.0: continue
        lag = lags[idx] - (n - 1)
        if lag >= 0:
            nv = n - lag
            g[:nv] += wt * dx * (1.0 - h[lag:])
            g[lag:] -= wt * dx * h[:nv]
        else:
            k = -lag
            nv = n - k
            g[k:] += wt * dx * (1.0 - h[:nv])
            g[:nv] -= wt * dx * h[k:]
    return g
```

import numpy as np
import time

GOLD = 0.6180339887498949
FRACS = (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.005)


def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * n
    y = np.clip(y, 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s < target:
            hi = mid
        elif s > target:
            lo = mid
        else:
            return np.clip(y - mid, 0.0, 1.0)
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)


def c5_full(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def c_vec_our(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full")[::-1] * dx


def top_weights(c, k, alpha):
    L = len(c)
    k = int(min(k, L))
    if k <= 0:
        return np.array([], dtype=int), np.array([])
    if k == L:
        top = np.arange(L)
    else:
        top = np.argpartition(c, -k)[-k:]
    vals = c[top]
    w = np.exp(np.clip(alpha * (vals - vals.max()), -50.0, 0.0))
    s = w.sum()
    if s <= 0.0:
        w = np.ones(k) * (1.0 / k)
    else:
        w /= s
    return top, w


def active_gradient(h, n, lags, weights, dx):
    g = np.zeros(n, dtype=float)
    for itr, idx in enumerate(lags):
        wt = float(weights[itr])
        if wt <= 0.0:
            continue
        m = int(idx) - (n - 1)
        if m >= 0:
            nv = n - m
            g[:nv] += wt * dx * (1.0 - h[m:])
            g[m:] -= wt * dx * h[:nv]
        else:
            k = -m
            nv = n - k
            g[k:] += wt * dx * (1.0 - h[:nv])
            g[:nv] -= wt * dx * h[k:]
    return g


def move_delta(h, n, i, j, dx):
    L = 2 * n - 1
    D = np.zeros(L, dtype=float)
    one = 1.0 - h
    rev = h[::-1]
    D[n - 1 - i : 2 * n - 1 - i] += dx * one
    D[i : n + i] -= dx * rev
    D[n - 1 - j : 2 * n - 1 - j] -= dx * one
    D[j : n + j] += dx * rev
    return D, n - 1 + (j - i), n - 1 + (i - j)


def c5_after_move(c, D, t, i1, i2, zidx, dx):
    if t == 0.0:
        return float(c.max())
    arr = c + t * D
    add = t * t * dx
    arr[i1] += add
    arr[i2] += add
    arr[zidx] -= 2.0 * add
    return float(arr.max())


def search_move_t(c, D, dx, dmax, i1, i2, zidx, cmax, iters=20):
    if dmax <= 1e-15:
        return 0.0, cmax
    a = 0.0
    b = dmax
    c1 = b - GOLD * (b - a)
    c2 = a + GOLD * (b - a)
    f1 = c5_after_move(c, D, c1, i1, i2, zidx, dx)
    f2 = c5_after_move(c, D, c2, i1, i2, zidx, dx)
    for _ in range(iters):
        if b - a < 1e-15:
            break
        if f1 <= f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - GOLD * (b - a)
            f1 = c5_after_move(c, D, c1, i1, i2, zidx, dx)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + GOLD * (b - a)
            f2 = c5_after_move(c, D, c2, i1, i2, zidx, dx)
    t = 0.5 * (a + b)
    f = c5_after_move(c, D, t, i1, i2, zidx, dx)
    if f > cmax:
        return 0.0, cmax
    if t > dmax:
        t = dmax
    return t, f


def pair_descent(h, n, deadline, rng, active_lags=22, cand_lags=16, eval_cands=12, golden_iters=20):
    h = project_feasible(h, n)
    c = c_vec_our(h, n)
    cmax = float(c.max())
    dx = 2.0 / float(n)
    zidx = n - 1
    alpha = 150.0
    no_imp = 0
    moves = 0

    while time.time() < deadline:
        max_lags = np.flatnonzero(c >= cmax - 1e-12)
        if max_lags.size > 128:
            sub = c[max_lags]
            max_lags = max_lags[np.argpartition(sub, -128)[-128:]]

        top, w = top_weights(c, active_lags, alpha)
        g = active_gradient(h, n, top, w, dx)
        order = np.argsort(g)
        lows = order[:cand_lags]
        highs = order[-cand_lags:][::-1]

        cands = []
        for ii in lows:
            ri = 1.0 - h[ii]
            if ri <= 1e-14:
                continue
            for jj in highs:
                if ii == jj:
                    continue
                rj = h[jj]
                if rj <= 1e-14:
                    continue
                dmax = ri if ri < rj else rj
                if dmax <= 1e-14:
                    continue
                score = (g[jj] - g[ii]) * dmax
                if score > 0.0:
                    cands.append((score, int(ii), int(jj), float(dmax)))

        if not cands:
            no_imp += 1
            if no_imp >= 4:
                break
            continue

        cands.sort(reverse=True)
        best_val = cmax
        best_tD = None
        best_t = 0.0
        best_i = -1
        best_j = -1
        bi1 = -1
        bi2 = -1
        ev = 0

        for score, ii, jj, _ in cands:
            if ev >= eval_cands:
                break
            ev += 1
            ri = 1.0 - h[ii]
            rj = h[jj]
            dmax = ri if ri < rj else rj
            if dmax <= 1e-14:
                continue
            D, i1, i2 = move_delta(h, n, ii, jj, dx)
            if max_lags.size:
                md = float(np.max(D[max_lags]))
                if md > 1e-13:
                    continue
            t, val = search_move_t(c, D, dx, dmax, i1, i2, zidx, cmax, golden_iters)
            if val < best_val:
                best_val = val
                best_tD = D
                best_t = t
                best_i = ii
                best_j = jj
                bi1 = i1
                bi2 = i2

        if best_tD is None or best_t <= 1e-15 or best_val >= cmax - 1e-14:
            no_imp += 1
            if no_imp >= 4:
                break
            continue

        h[best_i] += best_t
        h[best_j] -= best_t
        c += best_t * best_tD
        add = best_t * best_t * dx
        c[bi1] += add
        c[bi2] += add
        c[zidx] -= 2.0 * add
        cmax = float(c.max())
        no_imp = 0
        moves += 1
        if moves % 20 == 0:
            c = c_vec_our(h, n)
            cmax = float(c.max())

    return h, cmax


def random_pair_search(h, n, deadline, rng, max_no=5000):
    h = project_feasible(h, n)
    c = c_vec_our(h, n)
    cmax = float(c.max())
    dx = 2.0 / float(n)
    zidx = n - 1
    moves = 0
    no = 0

    while time.time() < deadline:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue

        if h[i] < h[j]:
            inc = i
            dec = j
            dm = min(1.0 - h[i], h[j])
        elif h[j] < h[i]:
            inc = j
            dec = i
            dm = min(1.0 - h[j], h[i])
        else:
            inc = i
            dec = j
            dm = min(1.0 - h[i], h[j])

        if dm <= 1e-14:
            continue

        D, i1, i2 = move_delta(h, n, inc, dec, dx)
        max_lags = np.flatnonzero(c >= cmax - 1e-12)
        if max_lags.size > 128:
            sub = c[max_lags]
            max_lags = max_lags[np.argpartition(sub, -128)[-128:]]
        if max_lags.size and float(np.max(D[max_lags])) > 1e-13:
            no += 1
            if no > max_no:
                break
            continue

        f = FRACS[int(rng.randint(len(FRACS)))]
        t = f * dm
        val = c5_after_move(c, D, t, i1, i2, zidx, dx)
        if val < cmax - 1e-14:
            h[inc] += t
            h[dec] -= t
            c += t * D
            add = t * t * dx
            c[i1] += add
            c[i2] += add
            c[zidx] -= 2.0 * add
            cmax = float(c.max())
            no = 0
            moves += 1
            if moves % 50 == 0:
                c = c_vec_our(h, n)
                cmax = float(c.max())
        else:
            no += 1
            if no > max_no:
                break

    return h, cmax


def gradient_flow(h, n, deadline, rng, active_lags=22, max_iters=120):
    h = project_feasible(h, n)
    cmax = c5_full(h, n)
    dx = 2.0 / float(n)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        c = c_vec_our(h, n)
        top, w = top_weights(c, active_lags, 150.0)
        g = active_gradient(h, n, top, w, dx)
        d = -g
        d -= d.mean()
        dn = float(np.max(np.abs(d)))
        if dn <= 1e-15:
            break
        d /= dn
        ok = False
        for s in (0.05, 0.02, 0.008, 0.003, 0.001):
            h_new = project_feasible(h + s * d, n)
            val = c5_full(h_new, n)
            if val < cmax - 1e-15:
                h = h_new
                cmax = val
                ok = True
                break
        if not ok:
            break
    return h, cmax


def binary_threshold(h, n):
    bb = np.zeros(n, dtype=float)
    idx = np.argsort(-h)
    if n % 2 == 0:
        bb[idx[:n // 2]] = 1.0
    else:
        bb[idx[:n // 2]] = 1.0
        bb[idx[n // 2]] = 0.5
    return project_feasible(bb, n)


def random_binary_start(n, rng):
    bb = np.zeros(n, dtype=float)
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    bb[idx] = 1.0
    if n % 2:
        bb[rng.randint(n)] += 0.5
    return project_feasible(bb, n)


def random_smooth_start(n, rng, mode):
    x = np.arange(n, dtype=float)
    if mode == 0:
        h = np.zeros(n, dtype=float)
        freqs = (1, 2, 3, 4, 6, 8, 12)
        for m in freqs:
            amp = rng.uniform(0.02, 0.25)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            h += amp * np.cos(2.0 * np.pi * (m * x) / float(n) + ph)
        h += 0.5
    else:
        freqs = (1, 2, 3, 4, 5, 6, 7, 8, 12, 16)
        m = float(int(rng.choice(freqs)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * m * x / float(n) + ph) + 0.05 * rng.randn(n)
    return project_feasible(h, n)


def interp_to_n(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)


def consider(h, n, best_h, best_c, best_n):
    h = project_feasible(h, n)
    c = c5_full(h, n)
    if c < best_c - 1e-15:
        return h.copy(), c, n
    return best_h, best_c, best_n
