from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient(h, n):
    """Compute subgradient of max correlation w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    # Get active lags
    tol = 1e-10
    active = np.where(np.abs(corr - c_max) <= tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)

            term2 = np.zeros(n)
            term1[lag:] = 1 - h[:n-lag]
            term2[:n-lag] = h[lag:]
        else:

            term1 = np.zeros(n)
            term2 = np.zeros(n)
            term1[:n-m_] = 1 - h[m_:]
            term2[m_:] = h[:m_]
        grad += dx * (term1 - term2)
    grad /= len(active)
    return grad, c_max

        pos_idx = np.where(grad > 0.01)[0]  # decrease these

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    tol = 1e-10 * max(1.0, abs(c_max))
    active = np.where(np.abs(corr - c_max) <= tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if lag >= 0:
            term1[lag:] = 1.0 - h[:n - lag]
            term2[:n - lag] = h[lag:]
        else:
            m_ = -lag
            term1[:n - m_] = 1.0 - h[m_:]
            term2[m_:] = h[:m_]
        grad += dx * (term1 - term2)
    grad /= len(active)
    return grad

def gradient_pair_search(h, n, deadline, max_rounds=500, temperature=0.0):
    best_c = compute_c5(h, n)
    round_idx = 0
    while time.time() < deadline and round_idx < max_rounds:
        round_idx += 1
        grad = subgradient(h, n)
        pos_idx = np.where(grad > 0.005)[0]
        neg_idx = np.where(grad < -0.005)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            # Fallback: try all pairs with small steps
            pos_idx = np.where(h < 0.99)[0]
            neg_idx = np.where(h > 0.01)[0]
            if len(pos_idx) == 0 or len(neg_idx) == 0:
                break
        improved = False
        # Prioritize pairs by gradient difference
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        for t in deltas:
            for i in pos_idx[:80]:
                for j in neg_idx[:80]:
                    if time.time() > deadline:
                        return h, best_c
                    if h[i] - t < 0 or h[j] + t > 1:
                        continue
                    h[i] -= t
                    h[j] += t
                    c = compute_c5(h, n)
                    accept = c < best_c - 1e-14
                    if not accept and temperature > 0:
                        accept = (c - best_c) < temperature
                    if accept:
                        best_c = min(best_c, c)
                        improved = True
                    else:
                        h[i] += t
                        h[j] -= t
        temperature *= 0.85
        if not improved and temperature < 1e-6:
            break
    return h, best_c

def mode_search(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 4, 5, 6, 8, 12]:

    return best_h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # SLSQP schedule: low alpha first, then increasing
    alphas = [200, 1000, 5000, 20000, 100000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp(h, n, a, maxiter=1000)

            best_c = c
            h = ht
    # Gradient pair search

    # Final SLSQP polish

        if c < best_c:
            best_c = c
            h = ht
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
