```
def project(x,target):
    x=np.array(x,dtype=float,copy=True)
    n=x.size
    if target<=0: return np.zeros(n)
    if target>=n: return np.ones(n)
    lo,hi=-1.0,2.0
    for _ in range(70):
        m=(lo+hi)*0.5
        s=np.sum(np.clip(x-m,0.0,1.0))
        if s>target: lo=m
        else: hi=m
    y=np.clip(x-(lo+hi)*0.5,0.0,1.0)
```

```
def smooth_fun(x,n,alpha):
    c=np.correlate(x,1.0-x,'full')
    T=float(c.max())
    s=alpha*(c-T)
    mx=float(s.max())
    e=np.exp(s-mx)
    se=float(e.sum())
    if se<=0: return T, np.zeros_like(x)
    f=T+(mx+np.log(se))/alpha
    return f, grad_weights(x,n,e/se)
```

```
def slsqp(h,n,alpha,maxit):
    h0=project(h,n/2.0)
    bnd=[(0.0,1.0) for _ in range(n)]
    cons=[{'type':'eq','fun':sum_fun,'jac':sum_jac,'args':(n,)}]
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def project_simplex(x, target):
    x = np.array(x, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(x - lam, 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-14:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.sum() > 0:
            y[free] += diff / float(free.sum())
        else:
            y += diff / float(n)
        y = np.clip(y, 0.0, 1.0)
    return y


def c5_value(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def grad_weights(h, n, w):
    g = np.zeros(n)
    off = n - 1
    nz = np.flatnonzero(w)
    for p in nz:
        wc = float(w[p])
        if wc <= 1e-16:
            continue
        k = int(p) - off
        if k >= 0:
            L, R = 0, n - 1 - k
        else:
            L, R = -k, n - 1
        g[L:R + 1] += wc
        g[L:R + 1] -= wc * h[L + k:R + k + 1]
        g[L + k:R + k + 1] -= wc * h[L:R + 1]
    return g


def sum_eq(x, n):
    return float(x.sum()) - 0.5 * n


def sum_eq_jac(x, n):
    return np.ones(n)


def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    s = alpha * (c - m)
    sm = float(s.max())
    e = np.exp(s - sm)
    se = float(e.sum())
    if se <= 0.0:
        return m, np.zeros_like(x)
    val = m + (sm + np.log(se)) / alpha
    return val, grad_weights(x, n, e / se)


def slsqp_refine(h, n, alpha, maxiter):
    h0 = project_simplex(h, 0.5 * n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac, "args": (n,)}]
    try:
        res = minimize(smooth_obj, h0, args=(n, alpha), method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-13})
        return project_simplex(res.x, 0.5 * n)
    except Exception:
        return h0


def subgradient(h, c, n, k_active):
    g = np.zeros(n)
    cmdx = c.size
    k_active = min(k_active, cmdx)
    idx = np.argsort(c)[-k_active:]
    if idx.size == 0:
        return g
    w = 1.0 / float(idx.size)
    off = n - 1
    for p in idx:
        k = int(p) - off
        if k >= 0:
            L, R = 0, n - 1 - k
        else:
            L, R = -k, n - 1
        g[L:R + 1] += w
        g[L:R + 1] -= w * h[L + k:R + k + 1]
        g[L + k:R + k + 1] -= w * h[L:R + 1]
    return g


def pair_eval(d, base, acoef, bcoef, unaffected_max):
    if base.size:
        m = float(np.max(base + acoef * d + bcoef * d * d))
    else:
        m = -1e100
    if m > unaffected_max:
        return m
    return unaffected_max


def try_pair_step(h, c, n, inc, dec, tol=1e-10):
    dmax = min(1.0 - float(h[inc]), float(h[dec]))
    if dmax <= 1e-12:
        return None
    base = []
    acoef = []
    bcoef = []
    unaffected_max = -1e100
    off = n - 1
    hi = int(h[inc])
    hj = int(h[dec])
    for p in range(2 * n - 1):

        if k >= 0:

        else:

        a = 0.0
        if L <= hi <= R:
            a += 1.0 - h[hi + k]
        if L + k <= hi <= R + k:
            a -= h[hi - k]
        if L <= hj <= R:
            a += -1.0 + h[hj + k]
        if L + k <= hj <= R + k:
            a += h[hj - k]
        b = 0.0
        if hj == hi + k or hi == hj + k:
            b = 1.0
        cp = float(c[p])
        if a != 0.0 or b != 0.0:
            base.append(cp)
            acoef.append(a)
            bcoef.append(b)
        elif cp > unaffected_max:
            unaffected_max = cp
    base = np.asarray(base, dtype=float)
    acoef = np.asarray(acoef, dtype=float)
    bcoef = np.asarray(bcoef, dtype=float)
    best = float(c.max())

    a0, b0 = 0.0, dmax
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b0 - gr * (b0 - a0)
    d1 = a0 + gr * (b0 - a0)
    f1 = pair_eval(c1, base, acoef, bcoef, unaffected_max)

def pair_polish(h, n, deadline, rng=None, k_active=24, max_trials=120):

def pipeline(h, n, deadline, rng=None, use_slsqp=False):
    h = project_simplex(h, 0.5 * n)
    best = c5_value(h, n)

    if use_slsqp and n <= 230 and time.time() < deadline - 20:
        if n <= 100:
            mt = 55
        elif n <= 160:
            mt = 42
        else:
            mt = 32
        for alpha in (500, 4000):

            ht = slsqp_refine(h, n, alpha, mt)
            ct = c5_value(ht, n)
            if ct < best - 1e-14:

    h, best2 = pair_polish(h, n, deadline, rng, k_active=32, max_trials=160)
    if best2 < best:
        h, best = h, best2
    return h, best


def random_restarts(h, n, best_c, deadline, rng, tries=3):
    for r in range(tries):
        if time.time() > deadline - 40:
            break
        scale = 0.025 * (1.0 + 0.45 * r)

        if ct < best_c - 1e-14:
            h, best_c = ht, ct
    return h, best_c

            h0 = make_start(parent, n)
            alloc = min(300.0, max(15.0, deadline - time.time() - 0.5 * reserve))
            slsqp = (n <= 220)
        else:
            h0 = make_start(parent, n)
            alloc = min(90.0, max(12.0, deadline - time.time() - 15.0))
            slsqp = (n <= 180)
        if alloc < 10:
            break
        ht, ct = pipeline(h0, n, time.time() + alloc, rng, use_slsqp=slsqp)
        if ct < best_c:
            best_c, best_h = ct, ht

    if best_h is None:
        n = int(n_parent)
        best_h = project_simplex(np.full(n, 0.5), 0.5 * n)
        best_c = c5_value(best_h, n)

    nbest = len(best_h)

    if time.time() < deadline - 80 and nbest <= 320:
        if nbest <= 120:
            mt = 60
        elif nbest <= 200:
            mt = 45
        else:
            mt = 30
        ht = slsqp_refine(best_h, nbest, 10000, mt)
        local_deadline = min(deadline - 20, time.time() + 60)
        ht, ct = pair_polish(ht, nbest, local_deadline, rng, k_active=40, max_trials=200)
        if ct < best_c:
            best_h, best_c = ht, ct
            nbest = len(best_h)

    if time.time() < deadline - 70:
        best_h, best_c = random_restarts(best_h, nbest, best_c, deadline - 35, rng, tries=3)
        nbest = len(best_h)

    if time.time() < deadline - 3:
        best_h, best_c = pair_polish(best_h, nbest, deadline - 1, rng,
                                     k_active=64, max_trials=450)

    best_h = project_simplex(best_h, 0.5 * len(best_h))
    best_c = c5_value(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
