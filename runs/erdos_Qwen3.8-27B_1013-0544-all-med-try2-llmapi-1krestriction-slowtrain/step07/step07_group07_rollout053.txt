        else:

            if tot < 1e-15:

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _compute_lag_grad(h, n, lag):
    """Gradient of c(lag) w.r.t. h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    return grad

def nesterov_subgrad(h, n, deadline, n_iter=5000):
    """Nesterov-accelerated subgradient method."""
    h_curr = h.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for t in range(1, n_iter):
        if time.time() > deadline:
            break
        # Nesterov momentum
        beta = (t - 1) / (t + 2) if t > 1 else 0.0
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute lags and find active set
        c = compute_all_lags(y, n)
        # Use top few lags with softmax weighting
        top_k = 5
        sorted_idx = np.argsort(c)[-top_k:]
        weights = np.zeros(2*n-1)
        c_top = c[sorted_idx]
        w = np.exp(50 * (c_top - c_top.max()))
        w /= w.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for idx, wi in zip(sorted_idx, w):
            lag = idx - (n - 1)
            grad += wi * _compute_lag_grad(y, n, lag)
        
        # Step size: Polyak-like
        step = best_c / (np.dot(grad, grad) + 1e-10)
        step = min(step, 0.05)  # cap
        
        h_new = y - step * grad
        h_new = project_barycentric(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
    
    return best_h, best_c

def outer_approx(h, n, deadline, max_iters=50):
    """Outer approximation with trust region."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
                cp.abs(h_var - h_curr) <= trust]
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            grad = _compute_lag_grad(h_curr, n, lag)
            cons.append(c[k] + 2.0/n * (grad @ (h_var - h_curr)) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return best_h, best_c
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_barycentric(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.7)
    
    return best_h, best_c

            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            # Weighted gradient over top lags
            c_top = c[top_idx]
            w = np.exp(100.0 * (c_top - c_top.max()))
            w /= w.sum()
            grad = np.zeros(n)
            for idx, wi in zip(top_idx, w):
                lag = int(idx) - (n - 1)
                grad += wi * _lag_grad(h, n, lag)
            idx_arr = np.arange(n)
            idx_arr = idx_arr[rng.permutation(n)][:300]
            best_gain = 1e-10
            best_a, best_b = -1, -1
            for i in range(300):
                a = idx_arr[i]
                if h[a] >= 1.0 - d * 1.5:
                    continue
                for j in range(i + 1, 300):
                    b = idx_arr[j]
                    diff = grad[a] - grad[b]
                    if diff < -best_gain and h[b] > d * 1.5:
                        best_gain = -diff
                        best_a, best_b = a, b
            if best_a < 0:
                break
            h[best_a] += d
            h[best_b] -= d
            h = np.clip(h, 0.0, 1.0)
            h = project_barycentric(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    h = project_barycentric(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Nesterov (50s)
    if time.time() < deadline - 200:
        h1, c1 = nesterov_subgrad(h, n, min(time.time() + 50, deadline - 200))
        if c1 < best_c:
            best_c, best_h = c1, h1
    # Phase 2: Outer approx (130s)
    if time.time() < deadline - 140:
        h2, c2 = outer_approx(best_h, n, min(time.time() + 130, deadline - 140))
        if c2 < best_c:
            best_c, best_h = c2, h2
    # Phase 3: SLSQP polish (70s)
    if time.time() < deadline - 80:
        for alpha in [30000, 100000, 500000]:
            if time.time() > deadline - 80:
                break
            h3 = slsqp_polish(best_h, n, alpha, maxiter=3000, ftol=1e-16)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c, best_h = c3, h3
    # Phase 4: Multi-lag local search (60s)
    if time.time() < deadline - 50:
        h4, c4 = multi_lag_local_search(best_h, n, min(time.time() + 60, deadline - 50), top_m=3)
        if c4 < best_c:
            best_c, best_h = c4, h4
    # Phase 5: Basin hopping (remaining - 30s)
    remaining = deadline - time.time() - 30
    if remaining > 30:
        n_restarts = max(1, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_barycentric(hp, n)
            alloc = min(35, deadline - time.time() - 25)
            if alloc < 15:
                break
            hr, cr = nesterov_subgrad(hp, n, time.time() + alloc * 0.4, n_iter=2000)
            if time.time() < deadline - 20:
                hr, cr = slsqp_polish(hr, n, 200000, maxiter=2000)
                cr = compute_c5(hr, n)
            if cr < best_c:
                best_c, best_h = cr, hr
    # Final polish
    if time.time() < deadline - 5:
        hf = slsqp_polish(best_h, n, 1000000, maxiter=3000, ftol=1e-16)
        cf = compute_c5(hf, n)
        if cf < best_c:
            best_c, best_h = cf, hf
        best_h = project_barycentric(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, n
