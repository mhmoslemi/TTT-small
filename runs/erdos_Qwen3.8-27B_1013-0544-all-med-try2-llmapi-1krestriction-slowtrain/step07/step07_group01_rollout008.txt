def marginal(h, i, n, m0):
    v = np.zeros(2*n-1, dtype=float)
    v[m0] = 1.0 - 2.0*h[i]
    # positive lag q=1..n-1
    if n - i - 1 > 0:
        v[m0+1:m0+n-i] += 1.0 - h[i+1:n]
    if i > 0:
        v[m0+1:m0+1+i] -= h[i-1::-1]
    # negative lag q=1..n-1
    if i > 0:
        v[m0-i:m0] += 1.0 - h[0:i]
    jlen = n - i - 1
    if jlen > 0:
        v[m0-jlen:m0] -= h[i+1:n][::-1]
    return v

import time
import numpy as np
from scipy.optimize import minimize

D_FRACS = np.array([1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
                    0.004, 0.001, 0.0003, 0.0001], dtype=float)


def compute_c5_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += diff * room / tot
        else:
            tot = s
            if tot < 1e-15:
                break
            h += diff * h / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def smooth_gradient_from_c(h, c, n, alpha):
    m = float(c.max())
    shift = np.maximum(alpha * (c - m), -745.0)
    w = np.exp(shift)
    sw = float(w.sum())
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= sw

    dx = 2.0 / float(n)
    m0 = n - 1
    g = np.zeros(n, dtype=float)
    for idx in np.nonzero(w > 1e-12)[0]:
        k = int(idx) - m0
        wm = float(w[idx]) * dx
        if k == 0:
            g += wm * (1.0 - 2.0 * h)
        elif k > 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - h[k:n])
            g[k:n] -= wm * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - h[0:nv])
            g[0:nv] -= wm * h[kk:n]
    return g


def smooth_gradient(h, n, alpha):
    c = compute_c5_vec(h, n)
    return smooth_gradient_from_c(h, c, n, alpha)


def smooth_obj_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = float(c.max())
    shift = np.maximum(alpha * (c - m), -745.0)
    mx = float(shift.max())
    return float(m + (mx + np.log(np.sum(np.exp(shift - mx)))) / alpha)


def pen_obj(x, n, alpha, mu):
    return smooth_obj_val(x, n, alpha) + mu * (float(x.sum()) - 0.5 * float(n)) ** 2


def pen_grad(x, n, alpha, mu):
    return smooth_gradient(x, n, alpha) + 2.0 * mu * (float(x.sum()) - 0.5 * float(n))


def lbfgs_phase(h, n, deadline, maxiter=200):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = c
    bounds = [(0.0, 1.0)] * n
    cfgs = ((1000.0, 1.0), (5000.0, 10.0), (50000.0, 100.0), (200000.0, 1000.0))
    for alpha, mu in cfgs:
        if time.time() > deadline - 4.0:
            break
        if deadline - time.time() < 5.0:
            break
        x0 = best_h.copy()
        try:
            res = minimize(pen_obj, x0, jac=pen_grad, args=(n, alpha, mu),
                           method="L-BFGS-B", bounds=bounds,
                           options={"maxiter": maxiter, "ftol": 1e-18,
                                    "gtol": 1e-15, "maxfun": maxiter * 6,
                                    "maxls": 40})
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2
        except Exception:
            pass
    return best_h, float(best_c)


def marginal(h, i, n, m0):
    v = np.zeros(2 * n - 1, dtype=float)
    v[m0] = 1.0 - 2.0 * h[i]
    if n - i - 1 > 0:
        v[m0 + 1:m0 + 1 + (n - i - 1)] += 1.0 - h[i + 1:n]
    if i > 0:
        v[m0 + 1:m0 + 1 + i] -= h[i - 1::-1]
    if i > 0:
        v[m0 - i:m0] += 1.0 - h[0:i]
    jlen = n - i - 1
    if jlen > 0:
        v[m0 - jlen:m0] -= h[i + 1:n][::-1]
    return v


def eval_pair_dvals(c, a, bidx, bb, dvals):
    dvals = np.asarray(dvals, dtype=float)
    if dvals.size <= 1:
        return 0.0, float(c.max())
    dmat = dvals[:, None]
    mat = c[None, :] + a[None, :] * dmat
    bmat = c[bidx][None, :] + a[bidx][None, :] * dmat
    mat[:, bidx] = -np.inf
    base = mat.max(axis=1)
    vals = np.maximum(base, (bmat + bb[None, :] * (dmat * dmat)).max(axis=1))
    k = int(np.argmin(vals))
    return float(dvals[k]), float(vals[k])


def make_d_vals(h, i, j):
    dn = min(float(h[i]), 1.0 - float(h[j]))
    dp = min(1.0 - float(h[i]), float(h[j]))
    if dn < 0.0:
        dn = 0.0
    if dp < 0.0:
        dp = 0.0
    if dn <= 1e-15 and dp <= 1e-15:
        return np.array([0.0], dtype=float)
    out = [0.0]
    for f in D_FRACS:
        if dn * f > 1e-15:
            out.append(-float(dn * f))
        if dp * f > 1e-15:
            out.append(float(dp * f))
    arr = np.array(out, dtype=float)
    arr = arr[(arr >= -dn - 1e-12) & (arr <= dp + 1e-12)]
    return np.unique(arr)


def candidate_pairs(g, n, k, pmax, rng):
    k = min(k, max(2, n - 1))
    pmax = min(pmax, n * (n - 1) // 2)
    order = np.argsort(g)
    lows = order[:k]
    highs = order[-k:][::-1]
    pairs = []
    seen = set()

    r = max(4, int(np.sqrt(max(1, pmax))))
    r = min(r, k)
    for a in highs[:r]:
        for b in lows[:r]:
            if a != b:
                ia, ib = int(a), int(b)
                key = (ia, ib) if ia < ib else (ib, ia)
                if key not in seen:
                    seen.add(key)
                    pairs.append((ia, ib))

    hl = highs[:min(3 * k, n)]
    ll = lows[:min(3 * k, n)]
    for _ in range(pmax - len(pairs)):
        if not len(hl) or not len(ll):
            break
        ii = int(hl[int(rng.randint(len(hl)))])
        jj = int(ll[int(rng.randint(len(ll)))])
        if ii == jj:
            continue
        key = (ii, jj) if ii < jj else (jj, ii)
        if key not in seen:
            seen.add(key)
            pairs.append((ii, jj))

    cnt = 0
    while len(pairs) < pmax and cnt < 20000:
        ii = int(rng.randint(n))
        jj = int(rng.randint(n))
        if ii != jj:
            key = (ii, jj) if ii < jj else (jj, ii)
            if key not in seen:
                seen.add(key)
                pairs.append((ii, jj))
        cnt += 1
    return pairs


def pair_local_search(h, deadline, k=36, pmax=600,
                      alpha_seq=(30.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0)):
    n = len(h)
    h = project_feasible(h, n)
    cvec = compute_c5_vec(h, n)
    c = float(cvec.max())
    best_h = h.copy()
    best_c = c
    dx = 2.0 / float(n)
    m0 = n - 1
    rng = np.random.RandomState(987654321)
    ai = 0
    no_impr = 0
    kcur = k
    pcur = pmax

    while time.time() < deadline:
        cvec = compute_c5_vec(h, n)
        c = float(cvec.max())
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()

        alpha = float(alpha_seq[ai % len(alpha_seq)])
        g = smooth_gradient_from_c(h, cvec, n, alpha)
        pairs = candidate_pairs(g, n, kcur, pcur, rng)
        improved = False

        for i, j in pairs:
            if time.time() > deadline:
                break
            dvals = make_d_vals(h, i, j)
            if dvals.size <= 1:
                continue

            a = marginal(h, i, n, m0)
            a -= marginal(h, j, n, m0)
            a *= dx

            dl = abs(i - j)
            if dl:
                bidx = np.array([m0, m0 + dl, m0 - dl], dtype=int)
                bb = np.array([-2.0, 1.0, 1.0], dtype=float) * dx
            else:
                bidx = np.array([m0], dtype=int)
                bb = np.array([-2.0], dtype=float) * dx

            best_d, best_val = eval_pair_dvals(cvec, a, bidx, bb, dvals)
            if best_d == 0.0 or best_val >= c - 1e-12:
                continue

            old_c = c
            cvec += a * best_d
            cvec[bidx] += bb * best_d * best_d
            h[i] += best_d
            h[j] -= best_d
            new_c = float(cvec.max())

            if new_c < old_c - 1e-14:
                c = new_c
                improved = True
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i] -= best_d
                h[j] += best_d
                cvec[bidx] -= bb * best_d * best_d
                cvec -= a * best_d
                c = old_c

        if improved:
            no_impr = 0
        else:
            no_impr += 1
            ai += 1
            if no_impr >= 3:
                kcur = min(80, kcur + 10)
                pcur = min(1600, int(pcur * 1.5))
            if no_impr >= 12:
                break

    cvec = compute_c5_vec(best_h, n)
    cbest = float(cvec.max())
    if cbest < best_c:
        best_c = cbest
    return best_h.copy(), float(best_c)


def interp_h(h_old, n_old, n_new):
