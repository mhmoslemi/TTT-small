def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    # Hmm, let me just use the definition directly.
    
    # Actually, the simplest way: use scipy.signal or just do it with matrix multiplication.
    # For n=400, a 400x400 matrix times a 400-vector is fast.
    
    # Build the Toeplitz matrix
    # T1: T1[i,j] = y[n-1+j-i]
    # T2: T2[i,j] = y[n-1+i-j]
    
    # T1 is upper-Hessenberg-like: T1[i,j] depends on j-i
    # T2 is the transpose of T1: T2[i,j] = y[n-1+i-j] = T1[j,i]
    
    # So T1 is a symmetric-ish Toeplitz (not quite, since y might not be symmetric)
    # Actually T1[i,j] = y[n-1+(j-i)] and T2[i,j] = y[n-1+(i-j)]
    # T2[i,j] = T1[j,i] only if y is symmetric, which it's not in general.
    # But T2 = T1.T (transpose)!
    # T1[i,j] = y[n-1+j-i], T1[j,i] = y[n-1+i-j] = T2[i,j]. Yes!
    
    # So: T1 @ (1-h) gives the first term
    # T1.T @ h gives the second term (which is T2 @ h)
    
    # Build T1:
    # T1[i,j] = y[n-1+j-i] for i,j in 0..n-1
    idx1, idx2 = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    T_idx = n - 1 + idx2 - idx1  # This gives the index into y for T1[i,j]
    T1 = y[T_idx]  # This creates an n x n matrix
    
    return dx * (T1 @ (1 - h) - T1.T @ h)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]

    h_b[idx[:n // 2]] = 1.0

                h[i] = x_opt
                improved = True

    return h, corr

def game_qp_solve(h, n, y, dx, maxiter=200):
    """Solve min_h sum_k y[k]*c[k]*dx using SLSQP with analytical gradient."""
    n_corr = 2 * n - 1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    
    def obj(x):
        return np.dot(y, np.correlate(x, 1 - x, mode='full')) * dx
    
    # Precompute the Toeplitz index
    idx1, idx2 = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    T_idx = n - 1 + idx2 - idx1
    
    def obj_grad(x):
        T1 = y[T_idx]
        return dx * (T1 @ (1.0 - x) - T1.T @ x)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    n_corr = 2 * n - 1
    y = np.ones(n_corr) / n_corr
    best_h = h.copy()

        h = h_new
