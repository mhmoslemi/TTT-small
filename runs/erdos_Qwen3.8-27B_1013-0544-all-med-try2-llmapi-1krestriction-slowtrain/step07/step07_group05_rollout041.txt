import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    """Compute smooth approximation of C5 and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    m = corr.max()
    shifted = alpha * (corr - m)
    # For numerical stability
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    val = m + log_sum / alpha
    w = exp_vals / np.sum(exp_vals)
    # Gradient: dx * correlate(w, 1-2h, 'valid')
    grad = dx * np.correlate(w, 1 - 2*h, mode='valid')
    return val, grad

def proj_grad_descent(h, n, deadline, rng):
    """Projected gradient descent with smooth max objective."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 200.0
    lr = 0.05
    no_improve = 0
    
    while time.time() < deadline:
        val, grad = smooth_grad(h, n, alpha)
        
        # Scale gradient to reasonable magnitude
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        
        # Adaptive step size
        step = lr / (gnorm + 1e-8)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        elif c_new < best_c + 1e-10:
            # Accept marginal improvement or no change (for exploration)
            h = h_new
            no_improve += 1
        else:
            no_improve += 1
            # Reduce step
            step = step * 0.5
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                no_improve = 0
        
        # Increase alpha over time (sharpen the max approximation)
        alpha *= 1.005
        if alpha > 50000:
            alpha = 50000
        
        lr *= 0.9995
        if lr < 1e-4:
            lr = 1e-4
        
        if no_improve > 200:
            # Reset with smaller alpha for exploration
            alpha = max(100, alpha * 0.5)
            lr = max(0.01, lr * 2)
            no_improve = 0
    
    return best_h, best_c

def pair_search_fast(h, n, deadline, deltas):
    """Fast pair search using only nearby pairs."""
    best_c = compute_c5(h, n)
    radius = min(n, max(20, n // 5))
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try both directions
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0, 1)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    if sm > 500:
        shifted = shifted - sm
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    val = m + np.log(s) / alpha
    w = exp_v / s
    grad = dx * np.correlate(w, 1 - 2 * h, mode='valid')
    return val, grad
