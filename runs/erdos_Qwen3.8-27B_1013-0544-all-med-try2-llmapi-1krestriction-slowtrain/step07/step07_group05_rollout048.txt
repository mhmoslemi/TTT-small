import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_fn(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """SA with random pair transfer moves."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    T = T0
    no_improve = 0
    
    while time.time() < deadline:
        # Check for reheat
        if no_improve > 5000 and T < T0 * 0.1:
            T = T0 * 0.3
            no_improve = 0
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        direction = 1.0 if rng.rand() < 0.5 else -1.0
        # h[i] += direction * delta, h[j] -= direction * delta
        max_d = min(h[i] if direction > 0 else h[j],
                     1.0 - (h[i] if direction > 0 else h[j]))
        
        if max_d < 1e-12:
            continue
        
        # Scale delta
        base_scale = 0.005 + 0.02 * (T / T0)
        delta = rng.uniform(0, min(max_d, base_scale))
        if delta < 1e-12:
            continue
        
        h[i] += direction * delta
        h[j] -= direction * delta
        
        new_c = compute_c5(h, n)
        dc = new_c - current_c
        
        if dc <= 0 or rng.rand() < np.exp(-dc / T):
            current_c = new_c
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] -= direction * delta
            h[j] += direction * delta
            no_improve += 1
        
        T *= cooling
    
    return best_h, best_c
