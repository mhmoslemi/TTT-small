            for s in starts:
                if time.time() > deadline - 120:
                    break
                alloc = min(25, (deadline - time.time() - 80) * 0.45)
                if alloc < 10:
                    break
                h_s, c_s = lbfgs_opt(s, n_coarse, time.time() + alloc,
                                      alphas=[500, 5000, 50000, 200000], topk=8)
                h_up = np.interp(x_fine, x_coarse, h_s)
                h_up = project_feasible(h_up, n)
                c_up = compute_c5(h_up, n)
                if c_up < best_c + 0.001:
                    alloc2 = min(35, deadline - time.time() - 60)
                    if alloc2 > 15:
                        h_up, c_up2 = lbfgs_opt(h_up, n, time.time() + alloc2,
                                                  alphas=[50000, 200000, 500000], topk=8)
                        if c_up2 < best_c:
                            best_c, best_h = c_up2, h_up

    # Phase 4: Random restarts with perturbation (120s)
    if time.time() < deadline - 80:
        for _ in range(10):
            if time.time() > deadline - 80:
                break
            scale = 0.02 + rng.uniform(0, 0.03)
            h_r = best_h + rng.randn(n) * scale
            h_r = project_feasible(h_r, n)
            alloc = min(12, (deadline - time.time() - 60) * 0.4)
            if alloc < 6:
                break
            h_r, c_r = lbfgs_opt(h_r, n, time.time() + alloc,
                                  alphas=[5000, 50000, 200000], topk=8)
            if c_r < best_c:
                best_c, best_h = c_r, h_r

    # Phase 5: Final SLSQP polish (60s)
    if time.time() < deadline - 35:
        h5, c5 = slsqp_opt(best_h, n, min(time.time() + 55, deadline - 15), alpha=1000000, topk=8)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Fine coordinate polish (remaining)
    if time.time() < deadline - 3:
        best_h, best_c = fine_polish(best_h, n, deadline - 1, radius=15)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
