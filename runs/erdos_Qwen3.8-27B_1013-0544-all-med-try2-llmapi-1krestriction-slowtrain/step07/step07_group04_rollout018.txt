import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    m_range = 2 * n - 1
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_binary(h_init, n, deadline, T_start, T_end, seed=42, n_rounds=3, perturb_prob=0.05):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    corr = np.correlate(hb, 1 - hb, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_max = cur_max
    best_hb = hb.copy()
    zero_list = list(np.where(hb == 0)[0])
    one_list = list(np.where(hb == 1)[0])
    n_lags = 2 * n - 1
    steps_per_round = max(int((deadline - time.time()) / n_rounds * 80000), 10000)
    for rnd in range(n_rounds):
        T = T_start * (0.3 ** rnd)
        T_cool = max(T_end, T * (0.01) ** (1.0 / steps_per_round))
        for step in range(steps_per_round):
            if time.time() > deadline:
                return best_hb, best_max * dx
            T = max(T * T_cool / T_start, T_end)
            a = zero_list[rng.randint(len(zero_list))]
            b = one_list[rng.randint(len(one_list))]
            if a == b:
                continue
            k = np.arange(-(n - 1), n, dtype=np.int32)
            ak = a - k
            ak2 = a + k
            bk = b - k
            bk2 = b + k
            delta = np.zeros(n_lags)
            m1 = (ak >= 0) & (ak < n)
            delta[m1] += 1.0 - hb[ak[m1]]
            m2 = (ak2 >= 0) & (ak2 < n)
            delta[m2] -= hb[ak2[m2]]
            m3 = (bk >= 0) & (bk < n)
            delta[m3] -= 1.0 - hb[bk[m3]]
            m4 = (bk2 >= 0) & (bk2 < n)
            delta[m4] += hb[bk2[m4]]
            k_ab = a - b
            if -(n - 1) <= k_ab <= (n - 1):
                delta[n - 1 + k_ab] += 2.0
            new_corr = corr + delta
            new_max = float(new_corr.max())
            if new_max < cur_max or (T > T_end * 1.01 and rng.rand() < np.exp(-(new_max - cur_max) / T)):
                hb[a], hb[b] = 1.0, 0.0
                corr = new_corr
                cur_max = new_max
                zero_list[rng.randint(len(zero_list))] = b
                one_list[rng.randint(len(one_list))] = a
                zero_list = [x for x in zero_list if x != b]
                zero_list.append(a)
                one_list = [x for x in one_list if x != a]
                one_list.append(b)
                if cur_max < best_max:
                    best_max = cur_max
                    best_hb = hb.copy()
            if rnd > 0 and step % 5000 == 0 and rng.rand() < perturb_prob:
                for _ in range(3):
                    a2 = zero_list[rng.randint(len(zero_list))]
                    b2 = one_list[rng.randint(len(one_list))]
                    hb[a2], hb[b2] = 1.0, 0.0
                    zero_list.remove(b2)
                    zero_list.append(a2)
                    one_list.remove(a2)
                    one_list.append(b2)
                    corr = np.correlate(hb, 1 - hb, mode='full').astype(np.float64)
                    cur_max = float(corr.max())
    return best_hb, best_max * dx
