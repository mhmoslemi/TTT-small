def compute_S(h):

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    xc = np.clip(x, 0.0, 1.0)
    if abs(float(xc.sum()) - target) < 1e-13:
        return xc
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_value(S, alpha):
    m = float(S.max())
    sh = alpha * (S - m)
    mx = float(sh.max())
    return m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha


def softmax_weights(S, alpha):
    m = S.max()
    sh = alpha * (S - m)
    mx = sh.max()
    e = np.exp(sh - mx)
    return e / e.sum()


def smooth_grad(S, h, p):
    n = len(h)
    g = np.zeros(n)
    idx0 = n - 1
    if p[idx0] > 1e-18:
        g += p[idx0] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = p[idx0 + d]
        wn = p[idx0 - d]
        if abs(wp) + abs(wn) < 1e-18:
            continue
        m = n - d
        if wp > 1e-18:
            g[:m] -= wp * h[d:]
            g[d:] += wp * (1.0 - h[:m])
        if wn > 1e-18:
            g[:m] += wn * (1.0 - h[d:])
            g[d:] -= wn * h[:m]
    return g

def try_pair(h, S, M, i, j, cand):

    idx0 = n - 1
    a = float(h[i])
    b = float(h[j])

    base = min(max(7, cand // 2 + 1), 15)

    nj = b - us

def pair_search(h, M, deadline, max_offsets=14, top_pairs=55, cand=21, max_sweeps=3000, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    n = len(h)
    best_h = h.copy()
    best_M = float(M)
    S = compute_S(h)
    M = float(S.max())
    no_improve = 0
    idx0 = n - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep and sweep % 8 == 0:
            h = project_box_sum(h, 0.5 * n)
            S = compute_S(h)
            M = float(S.max())
            if M < best_M:
                best_M = M
                best_h = h.copy()
        improved = False
        vals = np.maximum(S[idx0 + 1:], S[idx0 - 1::-1])
        noff = n - 1
        mo = min(max_offsets, noff)
        if mo > 0:
            order = np.argsort(vals)[::-1][:mo]
            for opi in order:
                if time.time() > deadline:
                    break
                d = int(opi) + 1
                seg = n - d
                if seg <= 0:
                    continue
                hi = h[:seg]
                lo = h[d:]
                score = np.maximum(hi * (1.0 - lo), lo * (1.0 - hi))
                if float(score.max()) < 1e-10:
                    continue
                tp = min(top_pairs, seg)
                if tp < seg:
                    part = np.argpartition(score, seg - tp)[seg - tp:]
                else:
                    part = np.arange(seg)
                part = part[np.argsort(score[part])[::-1]]
                for idx in part:
                    S, M, acc = try_pair(h, S, M, int(idx), int(idx) + d, cand)
                    if acc:
                        improved = True
                        if M < best_M:
                            best_M = M
                            best_h = h.copy()
                if time.time() > deadline:
                    break

                j = int(rng.randint(n))
                if i == j:

                if acc:
                    improved = True
                    if M < best_M:
                        best_M = M
                        best_h = h.copy()
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best_h, best_M


def grid_centers(x):

    return np.interp(x1, x0, h)

        w = np.ones(7) / 7.0
        base = np.convolve(base, w, mode="same")
        h = 0.5 + 0.85 * (base - 0.5)
    return project_box_sum(np.clip(h, 0.0, 1.0), 0.5 * n)


def optimize_at_n(h0, n, subdeadline, rng):
    h = project_box_sum(np.clip(h0, 0.0, 1.0), 0.5 * n)
    S = compute_S(h)
    M = float(S.max())
    t = time.time()
    if subdeadline - t > 55 and n >= 80:
        pg_end = t + min(110.0, (subdeadline - t) * 0.40)
        h, M = proj_grad(h, M, pg_end)
    if subdeadline - time.time() > 12:
        h, M = pair_search(h, M, subdeadline - 8, 14, 55, 21, 3000, rng)
    h = project_box_sum(h, 0.5 * n)
    S = compute_S(h)
    M = float(S.max())
    return h, M


def tighten_sum(h):
    target = 0.5 * len(h)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) > 1e-12:
        h = project_box_sum(h, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s) - 18.0)
    rng = np.random.RandomState(seed)
    best_h = None
    best_M = None

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float).copy().ravel()
        if len(parent) >= 20:
            parent = project_box_sum(parent, 0.5 * len(parent))
    except Exception:
        parent = None

    if parent is not None and deadline - time.time() > 80:
        n = len(parent)
        sub = time.time() + min(340.0, max(80.0, (deadline - time.time()) * 0.38))
        h, M = optimize_at_n(parent, n, min(sub, deadline - 20), rng)
        best_h = h
        best_M = float(M)

    n_raw = [80, 120, 160, 200, 240, 300, 360, 450]
    if best_h is not None:
        n_raw.append(len(best_h))
    if parent is not None:
        n_raw.append(len(parent))
    seen = set()
    ns = []
    for v in n_raw:
        v = int(v)
        if v % 2:
            v -= 1
        if 50 <= v <= 600 and v not in seen:
            seen.add(v)
            ns.append(v)
    ns.sort()

    for n_try in ns:
        if time.time() > deadline - 100:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 50:
            break
        sub = time.time() + min(170.0, remaining * 0.37)
        if sub - time.time() < 40:
            break
        if best_h is not None:
            st = interpolate_h(best_h, n_try)
        elif parent is not None:
            st = interpolate_h(parent, n_try)
        else:
            st = make_start(n_try, 2, rng)
        h, M = optimize_at_n(st, n_try, min(sub, deadline - 25), rng)
        if best_h is None or M < best_M - 1e-12:
            best_h = h
            best_M = float(M)

    if best_h is not None and time.time() < deadline - 160:
        n = len(best_h)
        for kind in range(6):
            if time.time() > deadline - 80:
                break
            st = make_start(n, kind, rng)
            st = 0.45 * st + 0.55 * interpolate_h(best_h, n)
            st = project_box_sum(st, 0.5 * n)
            rem = deadline - time.time() - 30
            sub = time.time() + min(95.0, rem * 0.33)
            if sub - time.time() < 35:
                break
            h, M = optimize_at_n(st, n, sub, rng)
            if M < best_M - 1e-12:
                best_h = h
                best_M = float(M)

    if best_h is None:
        best_h = project_box_sum(np.full(300, 0.5), 150.0)
        best_M = c5_raw(best_h) / (2.0 / 300.0)

    if time.time() < deadline - 35:
        rem = deadline - time.time()
        n = len(best_h)
        if rem > 60:
            pg_end = time.time() + min(35.0, rem * 0.20)
            best_h, best_M = proj_grad(best_h, best_M, pg_end)
        if time.time() < deadline - 12:
            best_h, best_M = pair_search(
                best_h, best_M, deadline - 6,
                20, 95, 25, 20000, rng
            )

    best_h = tighten_sum(np.clip(best_h, 0.0, 1.0))
    S = compute_S(best_h)
    n = len(best_h)
    c5_bound = float(S.max()) * (2.0 / n)
    return best_h, c5_bound, n
