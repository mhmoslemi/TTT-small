    best_h = h.copy()

    lr = lr_init

        v = momentum * v + (1 - momentum) * g
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)

            h = h_new

A better approach: instead of trying all pairs, use the gradient to identify the most promising pairs. The gradient tells us which coordinates would benefit most from being increased/decreased. So we can:
1. Find the coordinate with the most negative gradient (should increase)
2. Find the coordinate with the most positive gradient (should decrease)
3. Try moving weight from #2 to #1
4. This is O(n) per iteration instead of O(n²)

    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        # Find best donor (positive grad, should decrease) and receiver (negative grad, should increase)
        i = np.argmin(g)  # most negative gradient -> increase h[i]
        j = np.argmax(g)  # most positive gradient -> decrease h[j]
        if i == j:
            break
        # Try several step sizes
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.1]:
            if h[i] + s <= 1.0 and h[j] - s >= 0.0:
                h[i] += s; h[j] -= s

                    best_c = c; improved = True
                    break
                else:
                    h[i] -= s; h[j] += s
        if not improved:
            step *= 0.8
            if step < 1e-5:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def constraint_fn(x, n):
    return float(x.sum() - n / 2.0)

def constraint_jac(x, n):
    return np.ones(n)

def nesterov_gd(h, n, deadline, alpha_init=100.0, lr_init=0.005, momentum=0.9):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    alpha = alpha_init
    lr = lr_init
    it = 0
    while time.time() < deadline:
        it += 1
        alpha = min(alpha_init * (1.0 + it * 0.005), 2e6)
        g = grad_smooth_max(h, n, alpha)
        v = momentum * v + (1.0 - momentum) * g
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            improved = False
            for factor in [0.5, 0.25, 0.125]:
                h_try = project_feasible(h - lr * factor * v, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
            if not improved:
                lr *= 0.95
        if it % 150 == 0:
            h_bin = binary_project(h, n)
            h_blend = 0.7 * h + 0.3 * h_bin
            h_blend = project_feasible(h_blend, n)
            c_blend = compute_c5(h_blend, n)
            if c_blend < best_c:
                best_c = c_blend
                best_h = h_blend.copy()
                h = h_blend
            v *= 0.5
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fn, 'jac': constraint_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons, jac=smooth_max_grad,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def smart_pair_search(h, n, deadline, alpha=50000, step_init=0.01):
    best_c = compute_c5(h, n)
    step = step_init
    stall_count = 0
    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        i = np.argmin(g)
        j = np.argmax(g)
        if i == j:
            break
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.125]:
            if h[i] + s <= 1.0 and h[j] - s >= 0.0:
                h[i] += s
                h[j] -= s
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    h[i] -= s
                    h[j] += s
        if improved:
            stall_count = 0
        else:
            stall_count += 1
            step *= 0.85
            if step < 1e-6 or stall_count > 50:
                break
        # Also try top-5 receivers and donors
        for _ in range(3):
            if time.time() > deadline:
                break
            idx_recv = np.argsort(g)[:5]
            idx_send = np.argsort(g)[::-1][:5]
            for ri in idx_recv:
                for si in idx_send:
                    if ri == si:
                        continue
                    for s in [step, step * 0.5]:
                        if h[ri] + s <= 1.0 and h[si] - s >= 0.0:
                            h[ri] += s
                            h[si] -= s
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                break
                            else:
                                h[ri] -= s
                                h[si] += s
    return h, best_c

def exhaustive_pair(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(i + 1, hi):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        h[j] -= d
                        c = compute_c5(h, n)

                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:

    return h, best_c
