def slsqp_topsum(h, n, top_m, maxiter=5000):
    """SLSQP minimizing sum of top_m lags (smooth, no alpha tuning)"""
    c = compute_all_lags(h, n)
    top = np.argsort(c)[-top_m:]
    dx = 2.0 / n
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        return float(np.sum(cc[top]))
    
    def jac(x):
        g = np.zeros(n)
        for k in top:
            g += lag_grad(x, n, k - (n - 1))
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,

        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def benders(h_start, n, deadline, max_iters=50):
    """Cutting-plane method"""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h, n)
        thr = c.max() - 0.005
        active = np.where(c >= thr)[0]
        if len(active) < 10:
            active = np.argsort(c)[-25:]
        
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1]
        for k in active:
            g = lag_grad(h, n, k - (n - 1))
            cons.append(c[k] + g @ (hv - h) <= tv)
        cons.append(cp.sum(cp.abs(hv - h)) <= n * 0.08)
        
        prob = cp.Problem(cp.Minimize(tv), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        h_new = project_feasible(hv.value, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-11:
            best_c = c_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h)) < 1e-9:
            break
        h = h_new
    return best_h, best_c
