        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                        else:

            if not improved:
                break
    return h, best_c

def subgradient_method(h, n, deadline, rng):
    """Projected subgradient ascent on min of max corr"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    step = 0.05
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full') * dx
        k_star = int(np.argmax(c))
        
        s = k_star - (n - 1)
        grad = np.zeros(n)
        lo1, hi1 = max(0, s), min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += (1 - h[lo1-s:hi1-s])
        lo2, hi2 = max(0, -s), min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+s:hi2+s]
        
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - step * momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(0.1, step * 1.1)
        else:
            step *= 0.95
        
        step = max(0.001, step)
    
    return best_h, best_c

        h, best_c = momentum_subgrad(h, n, sg_end, rng)
        h = project_feasible(h, n)
        best_c = compute_c5(h, n)
    # Local search
    if time.time() < deadline - 4:
        if n <= 130:
            h, best_c = pair_search_sa(h, n, deadline - 4,
                                       [0.01, 0.005, 0.002, 0.001, 0.0005], temperature=0.0)
        else:
            h, best_c = near_pair_search(h, n, deadline - 4,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005], radius=45)
    # Final SLSQP
    if time.time() < deadline - 2:
        ht = slsqp_optimize(h, n, 300000, maxiter=1200)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c; h = ht
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    best_h = None
    best_c = float('inf')
    # Phase 1: optimize initial
    h_opt, c_opt = optimize_single(h_init, n_init, min(t0 + 280, deadline - 120), rng)
    best_h, best_c = h_opt, c_opt
    # Phase 2: other n values
    x_grid = np.linspace(0, 2, n_init)
    for n_try in [100, 150, 200, 300]:
        if time.time() > deadline - 80:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_grid, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(140, (deadline - time.time()) * 0.28)
        if alloc < 35:
            break
        h_opt, c_opt = optimize_single(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: multi-restart
    if best_h is not None and time.time() < deadline - 45:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.015 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 12) * 0.4)
            if alloc < 18:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 4: extended local search with very small steps
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 130:
            best_h, best_c = pair_search_sa(best_h, n, deadline - 4,
                                            [0.002, 0.001, 0.0005, 0.0002, 0.0001], temperature=0.0)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 4,
                                              [0.002, 0.001, 0.0005, 0.0002], radius=55)
    # Phase 5: final polish
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        ht = slsqp_optimize(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            if n <= 130:
                best_h, best_c = pair_search_sa(best_h, n, deadline - 1,
                                                [0.0005, 0.0002, 0.0001], temperature=0.0)
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.0005, 0.0002, 0.0001], radius=35)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
