import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

        return smooth_max_grad(x, n, alpha)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius or n))
                hi = min(n, i + (radius or n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:

                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:

            if not improved:
                break
    return h, best_c

            maxiter = int(min(400, max(30, time_per * 60)))
            try:
                res = minimize(obj_pen, h, jac=grad_pen, args=(n, alpha, mu),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-14,
                                        'gtol': 1e-11, 'maxfun': maxiter * 5,
                                        'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def optimize_slsqp(h, n, deadline, alpha=500000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_con, h, jac=grad_con, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-13})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def interp_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def coord_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    offsets = [0.05, 0.02, 0.01, 0.005, 0.002]
    for _ in range(25):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old = h[i]
            best_t = old
            best_ci = best_c
            for off in offsets:
                for t in [max(0.0, old - off), min(1.0, old + off)]:
                    if abs(t - old) < 1e-12:
                        continue
                    h[i] = t
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_ci - 1e-14:
                        best_ci = c
                        best_t = t
                h[i] = old
                h = project_feasible(h, n)
            for t in [0.0, 1.0]:
                if abs(t - old) < 1e-12:
                    continue
                h[i] = t
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_ci - 1e-14:
                    best_ci = c
                    best_t = t
                h[i] = old
                h = project_feasible(h, n)
            if best_t != old:
                h[i] = best_t
                h = project_feasible(h, n)
                if best_ci < best_c - 1e-14:
                    best_c = best_ci
                    improved = True
        if not improved:
            break
    return h, best_c

def pair_polish(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: Optimize at initial resolution
    t1 = min(t0 + 100, deadline - 500)
    h1, c1 = optimize_lbfgs(h, n, t1,
                             alphas=[500, 5000, 50000, 500000],
                             mus=[0.1, 1.0, 10.0])
    if c1 < best_c:
        best_c, best_h = c1, h1

    if time.time() < deadline - 440:
        hd = min(60, deadline - time.time() - 430)
        h2, c2 = optimize_slsqp(best_h, best_n, time.time() + hd, alpha=200000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 2: Upscale to higher resolutions
    for target_n in [400, 650, 900]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        h_up = interp_h(best_h, best_n, target_n)
        h_up = project_feasible(h_up, target_n)
        alloc1 = min(remaining * 0.35, 180)
        t2 = time.time() + alloc1
        h_up, c_up = optimize_lbfgs(h_up, target_n, t2,
                                     alphas=[10000, 50000, 200000, 1000000],
                                     mus=[0.1, 1.0])
        if time.time() < deadline - 80:
            alloc2 = min(remaining * 0.2, 70)
            t3 = time.time() + alloc2
            h_up, c_up = optimize_slsqp(h_up, target_n, t3, alpha=500000)
        if c_up < best_c - 1e-6:
            best_c, best_h, best_n = c_up, h_up, target_n

    # Phase 3: Multi-start perturbations
    if time.time() < deadline - 90:
        for attempt in range(5):
            if time.time() > deadline - 35:
                break
            scale = rng.uniform(0.01, 0.04)
            h_p = best_h + rng.uniform(-scale, scale, best_n)
            h_p = project_feasible(h_p, best_n)
            alloc = min(30, deadline - time.time() - 20)
            if alloc < 8:
                break
            h_p, c_p = optimize_lbfgs(h_p, best_n, time.time() + alloc * 0.7,
                                       alphas=[50000, 200000, 1000000],
                                       mus=[1.0, 10.0])
            if c_p < best_c:
                best_c, best_h = c_p, h_p

    # Phase 4: Coordinate descent
    if time.time() < deadline - 25:
        best_h, best_c = coord_descent(best_h, best_n, deadline - 12)

    # Phase 5: Pair polish
    if time.time() < deadline - 3:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1, radius=40)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
