def _eval_fg(h, n, alpha):
    c = compute_c5_arr(h,n)
    cm = float(np.max(c))
    with np.errstate(under='ignore', over='ignore'):
        z = float(alpha) * (c - cm)

        ez = np.exp(z - zm)

        w = np.zeros(len(c)); imax=int(np.argmax(c)); w[imax]=1.0
        logz = 0.0

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_arr(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(compute_c5_arr(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    if n <= 0:
        return h
    if not np.all(np.isfinite(h)):
        return np.full(n, target / float(n))

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    if not (lo < hi):
        return np.full(n, target / float(n))

    for _ in range(55):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if float(p.sum()) > target:
            lo = mid
        else:
            hi = mid

    p = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(60):
        res = target - float(p.sum())
        if abs(res) < 1e-13:
            break
        if res > 0.0:
            mask = p < 1.0 - 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            p[mask] += res / float(cnt)
        else:
            mask = p > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            p[mask] += res / float(cnt)
        p = np.clip(p, 0.0, 1.0)

    res = target - float(p.sum())
    if abs(res) > 0.0:
        if res > 0.0:
            idx = int(np.argmax(1.0 - p))
            p[idx] = min(1.0, p[idx] + res)
        else:
            idx = int(np.argmax(p))
            p[idx] = max(0.0, p[idx] + res)
    return np.clip(p, 0.0, 1.0)


def _eval_fg(h, n, alpha):
    c = compute_c5_arr(h, n)
    cm = float(np.max(c))
    alpha = float(alpha)

    with np.errstate(under="ignore", over="ignore"):
        z = alpha * (c - cm)
        zm = float(np.max(z))
        ez = np.exp(z - zm)
        zsum = float(ez.sum())

    if zsum <= 0.0 or not np.isfinite(zsum):
        w = np.zeros(len(c), dtype=float)
        w[int(np.argmax(c))] = 1.0
        logz = 0.0
    else:
        w = ez / zsum
        logz = zm + float(np.log(zsum))

    f = cm + logz / alpha
    center = n - 1

    pos_cum = np.cumsum(w[center:])
    neg_cum = np.zeros(n, dtype=float)
    if n > 1:
        neg_cum[1:] = np.cumsum(w[center - 1::-1])

    A = pos_cum + neg_cum[::-1]
    B = np.convolve(h, w, mode="full")[center:center + n]
    C = np.convolve(h, w[::-1], mode="full")[center:center + n]

    dx = 2.0 / float(n)
    g = dx * (A - B - C)
    scale = 0.5 * float(n)
    return f * scale, g * scale, cm


def objective_scaled(h, n, alpha):
    f, g, _ = _eval_fg(h, n, alpha)
    return f


def gradient_scaled(h, n, alpha):
    f, g, _ = _eval_fg(h, n, alpha)
    return g


def con_fun(h, n):
    return float(h.sum() - 0.5 * float(n))


def con_jac(h, n):
    return np.ones(n, dtype=float)


def slsqp_polish(h, n, alpha, maxiter):
    h0 = project_feasible(h, n)
    if maxiter <= 0:
        return h0
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": con_fun, "jac": con_jac, "args": (n,)}]
    try:
        res = minimize(
            objective_scaled,
            h0,
            args=(n, alpha),
            method="SLSQP",
            jac=gradient_scaled,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-11, "disp": False},
        )
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def derivative_matrix(h, n):
    h = np.asarray(h, dtype=float)
    L = 2 * n - 1
    center = n - 1
    dx = 2.0 / float(n)
    D = np.zeros((n, L), dtype=float)

    for q in range(n):
        p = center + q
        D[q:, p] = 1.0 - h[:n - q]
        D[:n - q, p] -= h[q:]

    for r in range(1, n):
        p = center - r
        D[:n - r, p] = 1.0 - h[r:]
        D[r:, p] -= h[:n - r]

    return D * dx


def exact_pair_search(h, n, deadline, deltas=(0.02, 0.006, 0.002, 0.0008)):
    h = project_feasible(h, n)
    c = compute_c5_arr(h, n)
    cm = float(np.max(c))
    best_h = h.copy()
    best_c = cm
    if n < 2:
        return best_h, best_c

    center = n - 1
    dx = 2.0 / float(n)
    D = derivative_matrix(h, n)
    moves = 0
    max_moves = 120

    while time.time() < deadline and moves < max_moves:
        best_score = cm - 1e-13
        best_move = None
        stop = False

        for i in range(n - 1):
            if time.time() > deadline or stop:
                break
            ai = D[i]
            hi = float(h[i])
            for j in range(i + 1, n):
                hj = float(h[j])
                aj = D[j]
                a = ai - aj
                dij = j - i
                p1 = center - dij
                p2 = center + dij

                for d in deltas:
                    b = dx * d * d

                    if hi + d <= 1.0 - 1e-12 and hj - d >= 1e-12:
                        m = float(np.max(c + d * a))
                        v1 = c[p1] + d * a[p1] + b
                        v2 = c[p2] + d * a[p2] + b
                        exact = v1 if v1 > v2 else v2
                        if m > exact:
                            exact = m
                        if exact < best_score:
                            best_score = exact
                            best_move = (i, j, d, 1.0)

                    if hi - d >= 1e-12 and hj + d <= 1.0 - 1e-12:
                        m = float(np.max(c - d * a))
                        v1 = c[p1] - d * a[p1] + b
                        v2 = c[p2] - d * a[p2] + b
                        exact = v1 if v1 > v2 else v2
                        if m > exact:
                            exact = m
                        if exact < best_score:
                            best_score = exact
                            best_move = (i, j, d, -1.0)

            if time.time() > deadline - 2.0:
                stop = True

        if best_move is None:
            break

        i, j, d, sgn = best_move
        h_prev = h.copy()
        h[i] += sgn * d
        h[j] -= sgn * d
        h = project_feasible(h, n)
        c_new = compute_c5_arr(h, n)
        cm_new = float(np.max(c_new))

        if cm_new < best_c - 1e-15:
            c = c_new
            cm = cm_new
            best_h = h.copy()
            best_c = cm
            moves += 1
            D = derivative_matrix(h, n)
        else:
            h = h_prev
            break

    out_h = project_feasible(best_h, n)
    out_c = compute_c5(out_h, n)
    return out_h, out_c


def projected_gradient(h, n, deadline, alphas=(100, 500, 2000, 10000)):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    muls = (1e-3, 1e-2, 3e-2, 1e-1, 3e-1, 1.0, 3.0, 10.0)

    for alpha in alphas:
        for _ in range(140):
            if time.time() >= deadline:
                break
            f, g, cm = _eval_fg(h, n, alpha)
            if cm < best_c - 1e-13:
                best_c = cm
                best_h = h.copy()

            norm = float(np.linalg.norm(g))
            if norm < 1e-12:
                break

            base = 1.0 / (norm + 1e-12)
            best_cv = cm
            best_v = None

            for mult in muls:
                t = base * mult
                if t > 12.0:
                    t = 12.0
                v = project_feasible(h - t * g, n)
                cv = compute_c5(v, n)
                if cv < best_cv - 1e-13:
                    best_cv = cv
                    best_v = v

            if best_v is None:
                break

            h = best_v
            if best_cv < best_c - 1e-13:
                best_c = best_cv
                best_h = h.copy()

    return best_h, best_c


def active_polish(h, n, deadline, topk=10):
    h = project_feasible(h, n)
    c = compute_c5_arr(h, n)
    cm = float(np.max(c))
    best_h = h.copy()
    best_c = cm
    scale = 0.5 * float(n)
    muls = (1e-3, 1e-2, 3e-2, 1e-1, 3e-1, 1.0, 3.0)
    no_improve = 0

    while time.time() < deadline:
        D = derivative_matrix(h, n)
        cm = float(np.max(c))
        if cm < best_c - 1e-13:
            best_c = cm
            best_h = h.copy()

        k = min(topk, c.size)
        if k <= 0:
            break
        idx = np.argsort(-c)[:k]

        dirs = []
        gavg = np.sum(D[:, idx], axis=1) * scale
        if float(np.linalg.norm(gavg)) > 1e-12:
            dirs.append(gavg)
        for p in idx:
            dirs.append(D[:, p] * scale)

        improved = False
        for dirvec in dirs:
            if time.time() >= deadline:
                break
            norm = float(np.linalg.norm(dirvec))
            if norm < 1e-12:
                continue
            base = 1.0 / (norm + 1e-12)
            for mult in muls:
                t = base * mult
                if t > 12.0:
                    t = 12.0
                v = project_feasible(h - t * dirvec, n)
                cv = compute_c5(v, n)
                if cv < cm - 1e-13:
                    h = v
                    c = compute_c5_arr(h, n)
                    cm = float(np.max(c))
                    if cm < best_c - 1e-13:
                        best_c = cm
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve >= 6:
                break
        else:
            no_improve = 0

    out_h = project_feasible(best_h, n)
    out_c = compute_c5(out_h, n)
    return out_h, out_c


def symmetrize(h):
    n = len(h)
    return project_feasible(0.5 * (h + h[::-1]), n)


def interp_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    y = np.interp(x_new, np.linspace(0.0, 2.0, n_old, endpoint=False), h)
    return project_feasible(y, n_new)


def make_alternatives(h, n, rng):
    starts = []
    starts.append(h.copy())
    starts.append(symmetrize(h))

    ksize = max(1, min(7, n))
    kernel = np.ones(ksize, dtype=float) / float(ksize)
    starts.append(np.convolve(h, kernel, mode="same"))

    x = np.arange(n, dtype=float) / float(n)
    for m in (1, 2, 3):
        starts.append(0.5 + 0.45 * np.sin(2.0 * np.pi * m * x))

    starts.append(0.5 * h + 0.5 * (0.5 + 0.45 * np.sin(2.0 * np.pi * x)))
    starts.append(0.5 * (h - h[::-1] + 1.0))
    starts.append(h + 0.04 * rng.randn(n))
    starts.append((h > 0.5).astype(float))

    out = []
    for s in starts:
        out.append(project_feasible(s, n))
    return out
