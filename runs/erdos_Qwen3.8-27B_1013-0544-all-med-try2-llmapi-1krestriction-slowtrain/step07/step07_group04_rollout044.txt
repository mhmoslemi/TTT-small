def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, alpha, epsilon=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # Active set: correlations within epsilon of max
    active = np.where(corr >= max_val - epsilon)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for m in active:
        # m corresponds to shift (m - (n-1))
        shift = m - (n - 1)
        for i in range(n):
            # h[i] contributes to corr[m] through h[i] * (1-h[i-shift])
            # and through (1-h[i]) * h[i+shift]
            idx1 = i - shift  # for the (1-h) factor
            idx2 = i + shift  # for the h factor in (1-h[i])
            if 0 <= idx1 < n:
                g[i] += (1 - h[idx1])
            if 0 <= idx2 < n:
                g[i] -= h[idx2]
    
    g *= dx
    # Projected gradient descent
    h_new = h - alpha * g
    h_new = project_feasible(h_new, n)
    return h_new

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):

def fast_binary_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    current_h = h.copy()

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(3, 8)):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = list(zeros)
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = list(ones)
                for j in o_list:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx

                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c
