    return float(np.max(c) * dx)

def project_feasible(h, n, iters=200):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    g = c * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    logsum = np.log(np.exp(shifted - sm).sum())
    val = m + (sm + logsum) / alpha
    w = np.exp(shifted - sm)
    w /= w.sum()
    s1 = np.zeros(n)
    s2 = np.zeros(n)
    for k in range(2 * n - 1):
        lo = max(0, n - 1 - k)
        hi = min(n - 1 + k, n - 1) + 1
        if lo < hi:
            s1[lo:hi] += w[k] * h[lo + k - (n - 1):hi + k - (n - 1)]
        lo2 = max(0, k - (n - 1))
        hi2 = min(k, n - 1) + 1
        if lo2 < hi2:
            s2[lo2:hi2] += w[k] * h[lo2 - k + (n - 1):hi2 - k + (n - 1)]
    grad = dx * (1.0 - s1 - s2)
    return val, grad

def slsqp_grad(h0, n, alpha, maxiter=3000, ftol=1e-16):
    def fun(x):
        v, _ = smooth_obj_grad(x, n, alpha)
        return v
    def jac(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def coord_descent_fast(h, n, max_sweeps=100, deadline=None):
    h = h.astype(float).copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            c0 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)
            h[i] = 1.0
            c1 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)
            h[i] = old
            if c0 < c1 - 1e-14 and old > 0:
                h[i] = 0.0
                improved = True
            elif c1 < c0 - 1e-14 and old < 1:
                h[i] = 1.0
                improved = True
        if not improved:
            break
    return h

def binary_coord_swap(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float) * (2.0 / n)
    ones_list = list(np.where(h > 0.999)[0])
    zeros_list = list(np.where(h < 0.001)[0])
    mixed_list = [i for i in range(n) if 0.001 <= h[i] <= 0.999]
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if deadline is not None and time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max()
                if nc < corr.max() - 1e-14:
                    corr += delta
                    h[i], h[j] = 1.0, 0.0
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, corr.max()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    cur_n = n

    # Phase 1: Binary swap to get a good binary baseline
    p1 = t0 + 0.05 * total
    hb, cb = binary_coord_swap(h, n, max_sweeps=300, deadline=p1)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: SLSQP with gradient - increasing alpha
    p2 = t0 + 0.45 * total
    alphas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000,
              3000000, 10000000, 30000000, 100000000]
    hh = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        hh = slsqp_grad(hh, n, alpha, maxiter=5000, ftol=1e-16)
        c = compute_c5(hh, n)
        if c < best_c:
            best_c, best_h = c, hh.copy()

    # Phase 3: Coordinate descent polish
    p3 = t0 + 0.55 * total
    hc = coord_descent_fast(best_h, n, max_sweeps=50, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 4: More SLSQP from polished result
    p4 = t0 + 0.72 * total
    for alpha in [30000000, 100000000, 300000000, 1000000000]:
        if time.time() > p4 - 5:
            break
        hh = slsqp_grad(best_h, n, alpha, maxiter=5000, ftol=1e-18)
        c = compute_c5(hh, n)
        if c < best_c:
            best_c, best_h = c, hh.copy()

    # Phase 5: Binary swap again on current best
    p5 = t0 + 0.78 * total
    hb2, cb2 = binary_coord_swap(best_h, n, max_sweeps=200, deadline=p5)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()

    # Phase 6: Try different resolutions
    p6 = t0 + 0.92 * total
    for nh in [200, 300, 500, 600, 800]:
        if time.time() > p6 - 20:
            break
        xo = np.linspace(0, 2, cur_n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = slsqp_grad(hh, nh, 10000, maxiter=2000, ftol=1e-14)
        hh = slsqp_grad(hh, nh, 100000, maxiter=3000, ftol=1e-16)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, cur_n = ch, hh.copy(), nh

    # Phase 7: Final perturbation restarts
    if time.time() < dl - 5:
        for trial in range(6):
            if time.time() > dl - 2:
                break
            noise_scale = 0.005 * (1 + trial * 0.5)
            hh = best_h + noise_scale * rng.randn(cur_n)
            hh = project_feasible(hh, cur_n)
            for alpha in [10000000, 100000000]:
                if time.time() > dl - 1:
                    break
                hh = slsqp_grad(hh, cur_n, alpha, maxiter=3000, ftol=1e-16)
                c = compute_c5(hh, cur_n)
                if c < best_c:
                    best_c, best_h = c, hh.copy()

    best_h = project_feasible(best_h, cur_n)
    best_c = compute_c5(best_h, cur_n)
    return best_h, best_c, cur_n
