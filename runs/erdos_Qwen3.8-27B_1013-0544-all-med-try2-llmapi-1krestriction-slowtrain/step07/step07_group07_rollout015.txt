def push_binary(h, n, threshold=0.5):
    """Push values toward 0 or 1 while maintaining sum constraint"""
    # Count how many are above and below threshold
    above = h > threshold
    below = ~above
    n_above = above.sum()
    n_below = below.sum()
    target_ones = n // 2  # We want exactly n/2 ones
    
    # If we have too many "above", flip some to 0
    # If we have too many "below", flip some to 1
    h_new = h.copy()
    if n_above > target_ones:
        # Flip the weakest "above" values to 0
        above_indices = np.where(above)[0]
        above_indices = above_indices[np.argsort(h[above_indices])]
        n_to_flip = n_above - target_ones
        h_new[above_indices[:n_to_flip]] = 0.0
    elif n_above < target_ones:
        below_indices = np.where(below)[0]
        below_indices = below_indices[np.argsort(-h[below_indices])]
        n_to_flip = target_ones - n_above
        h_new[below_indices[:n_to_flip]] = 1.0
    
    return project_feasible(h_new, n)

    h_curr = h.copy()
    best_h = h_curr.copy()

    step = step_init
    dx = 2.0 / n

            if tot < 1e-15:
                break
            h += (diff / tot) * h

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def slsqp_polish(h, n, alpha=200000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': slsqp_sum_con, 'jac': slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fine_local_search(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h_curr, n)
            top_k = int(np.argmax(c))
            lag = top_k - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            if lag > 0:
                for i in range(n - m):
                    j = i + m
                    if h_curr[i] > h_curr[j]:
                        delta = min(d, (h_curr[i] - h_curr[j]) * 0.5, h_curr[i], 1.0 - h_curr[j])
                        if delta < 1e-10:
                            continue
                        h_curr[i] -= delta
                        h_curr[j] += delta
                        c_new = compute_c5(h_curr, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h_curr[i] += delta
                            h_curr[j] -= delta
                    elif h_curr[j] > h_curr[i]:
                        delta = min(d, (h_curr[j] - h_curr[i]) * 0.5, h_curr[j], 1.0 - h_curr[i])
                        if delta < 1e-10:
                            continue
                        h_curr[i] += delta
                        h_curr[j] -= delta
                        c_new = compute_c5(h_curr, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h_curr[i] -= delta
                            h_curr[j] += delta
            else:
                for i in range(n - m):
                    j = i + m
                    if h_curr[j] > h_curr[i]:
                        delta = min(d, (h_curr[j] - h_curr[i]) * 0.5, h_curr[j], 1.0 - h_curr[i])
                        if delta < 1e-10:
                            continue
                        h_curr[j] -= delta
                        h_curr[i] += delta
                        c_new = compute_c5(h_curr, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h_curr[j] += delta
                            h_curr[i] -= delta
                    elif h_curr[i] > h_curr[j]:
                        delta = min(d, (h_curr[i] - h_curr[j]) * 0.5, h_curr[i], 1.0 - h_curr[j])
                        if delta < 1e-10:
                            continue
                        h_curr[j] += delta
                        h_curr[i] -= delta
                        c_new = compute_c5(h_curr, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h_curr[j] -= delta
                            h_curr[i] += delta
    return h_curr, best_c
