import numpy as np
from scipy.optimize import minimize
from math import gcd
import time


def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(x, n):
    x = np.asarray(x, dtype=float)
    target = 0.5 * float(n)
    lo = -float(x.max()) - 1e-12
    hi = 1.0 - float(x.min()) + 1e-12
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[np.nonzero(mask)[0][0]] += diff
        else:
            y += diff / float(n)
        y = np.clip(y, 0.0, 1.0)
    return y


def resample_scores(x, n):
    x = np.asarray(x, dtype=float)
    m = int(x.size)
    if m == n:
        return x.copy()
    if m == 1:
        return np.full(n, float(x[0]))
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return np.interp(xn, xo, x)


def scores_to_binary(scores, n):
    k = n // 2
    order = np.argsort(scores, kind='stable')[::-1]
    h = np.zeros(n, dtype=np.int8)
    h[order[:k]] = 1
    return h


def repair_binary(h, n):
    h = np.asarray(h, dtype=np.int8).copy()
    k = n // 2
    s = int(h.sum())
    if s > k:
        idx = np.flatnonzero(h == 1)
        if idx.size >= s - k:
            h[idx[:s - k]] = 0
    elif s < k:
        idx = np.flatnonzero(h == 0)
        if idx.size >= k - s:
            h[idx[:k - s]] = 1
    return h


def correlate_binary(h, n):
    a = h.astype(np.int32)
    b = (1 - a).astype(np.int32)
    return np.correlate(a, b, mode='full').astype(np.int32)


def build_binary_ab(h, n):
    L = 2 * n - 1
    off = n - 1
    p = np.flatnonzero(h == 1)
    q = np.flatnonzero(h == 0)
    A = np.zeros((q.size, L), dtype=np.int32)
    B = np.zeros((p.size, L), dtype=np.int32)
    for r, i in enumerate(q):
        i = int(i)
        A[r] = -np.bincount(p - i + off, minlength=L).astype(np.int32)
        A[r] += np.bincount(i - q + off, minlength=L).astype(np.int32)
        A[r, off] -= 1
    for cidx, j in enumerate(p):
        j = int(j)
        B[cidx] = -np.bincount(j - q + off, minlength=L).astype(np.int32)
        B[cidx] += np.bincount(p - j + off, minlength=L).astype(np.int32)
        B[cidx, off] -= 1
    return p, q, A, B


def binary_local_search(h, n, deadline, max_sweeps=300):
    h = repair_binary(h, n)
    c = correlate_binary(h, n)
    best_h = h.copy()
    best_max = int(c.max())
    off = n - 1

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        p, q, A, B = build_binary_ab(h, n)
        if p.size == 0 or q.size == 0:
            break
        row_idx = np.arange(p.size)
        prev_max = int(c.max())
        best_val = prev_max
        bi, bj = -1, -1

        for r in range(q.size):
            Ai = A[r]
            vals = (c[None, :] + Ai[None, :] + B).max(axis=1)
            qr = int(q[r])
            l1 = off + p - qr
            l2 = off + qr - p
            s1 = c[l1] + Ai[l1] + B[row_idx, l1] + 1
            s2 = c[l2] + Ai[l2] + B[row_idx, l2] + 1
            np.maximum(vals, s1, out=vals)
            np.maximum(vals, s2, out=vals)
            j = int(np.argmin(vals))
            val = int(vals[j])
            if val < best_val:
                best_val = val
                bi, bj = r, j

        if bi < 0:
            break

        i = int(q[bi])
        j = int(p[bj])
        h[i] = 1
        h[j] = 0
        c = correlate_binary(h, n)
        cur = int(c.max())
        if cur < prev_max and cur < best_max:
            best_max = cur
            best_h = h.copy()
        elif cur >= prev_max:
            break

    cb = correlate_binary(best_h, n)
    return best_h, int(cb.max())


def bit_reverse_scores(n):
    bits = int(np.round(np.log2(float(n))))
    x = np.arange(n, dtype=np.uint32)
    r = np.zeros(n, dtype=np.uint32)
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r.astype(float)


def thue_h(n):
    if (n & (n - 1)) != 0:
        return np.zeros(n, dtype=np.int8)
    bits = int(np.round(np.log2(float(n))))
    t = np.array([0], dtype=np.int8)
    for _ in range(bits):
        t = np.concatenate((t, (1 - t).astype(np.int8)))
    h = np.zeros(n, dtype=np.int8)
    h[t == 1] = 1
    return h


def golden_scores(n):
    i = np.arange(n, dtype=float)
    return np.remainder(0.6180339887498949 * i + 0.25, 1.0)


def mul_scores(n):
    m = 3
    while m < n and gcd(m, n) != 1:
        m += 2
    if m >= n:
        m = 3
    i = np.arange(n, dtype=np.int64)
    return ((i * m) % n).astype(float)


def quad_scores(n):
    i = np.arange(n, dtype=np.int64)
    return (((i * (i + 1)) // 2) % n).astype(float)


def perturb_binary(h, rng, n, nswaps):
    h = h.copy()
    for _ in range(nswaps):
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        a = int(rng.choice(ones))
        b = int(rng.choice(zeros))
        h[a] = 0
        h[b] = 1
    return h


def smooth_value_grad(h, n, alpha):
    h = np.asarray(h, dtype=float)
    alpha = float(alpha)
    corr = np.correlate(h, 1.0 - h, mode='full')
    d = corr * (2.0 / float(n))
    a = alpha * d
    m = float(a.max())
    e = np.exp(a - m)
    s = float(e.sum())
    if s <= 0.0:
        p = np.full(d.size, 1.0 / d.size)
        val = float(d.max())
    else:
        p = e / s
        val = (m + float(np.log(s))) / alpha

    grad = np.zeros(n, dtype=float)
    if d.size <= 400:
        active = np.arange(d.size)
    else:
        order = np.argsort(p)[::-1]
        cum = np.cumsum(p[order])
        k = int(np.searchsorted(cum, 0.999)) + 1
        if k > 160:
            k = 160
        if k < cum.size and cum[k - 1] < 0.8:
            active = np.nonzero(p > 1e-7)[0]
            if active.size > 400:
                active = order[:400]
        else:
            active = order[:k]

    off = n - 1
    for t in active:
        w = float(p[t])
        if w <= 0.0:
            continue
        lag = int(t - off)
        if lag >= 0:
            length = n - lag
            if length <= 0:
                continue
            i = np.arange(length)
            j = i + lag
            grad[j] += w * (1.0 - h[i])
            grad[i] -= w * h[j]
        else:
            ell = -lag
            i = np.arange(ell, n)
            j = i - ell
            grad[j] += w * (1.0 - h[i])
            grad[i] -= w * h[j]
    return val, grad


def projected_gradient_polish(h, n, alpha, deadline, maxit=200, step0=0.5):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    val, grad = smooth_value_grad(h, n, alpha)
    step = float(step0)

    for _ in range(maxit):
        if time.time() > deadline:
            break
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        use = step
        if use * gnorm > 1.0:
            use = 1.0 / gnorm
        accepted = False
        for _ in range(25):
            hn = project_feasible(h - use * grad, n)
            vn, gn = smooth_value_grad(hn, n, alpha)
            if vn < val - 1e-13:
                h = hn
                val = vn
                grad = gn
                ce = compute_c5(h, n)
                if ce < best_c:
                    best_c = ce
                    best_h = h.copy()
                step = min(2.0, use * 1.2)
                accepted = True
                break
            use *= 0.5
            if use < 1e-9:
                break
        if not accepted:
            break

    return best_h


def sum_eq(x, n):
    return x.sum() - 0.5 * float(n)


def sum_eq_jac(x, n):
    return np.ones_like(x)


def smooth_obj(x, n, alpha):
    return smooth_value_grad(x, n, alpha)[0]


def smooth_jac(x, n, alpha):
    return smooth_value_grad(x, n, alpha)[1]


def slsqp_polish(h, n, alpha, maxiter):
    h = project_feasible(h, n)
    try:
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_jac, bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': 1e-11, 'iprint': 0})
        if hasattr(res, 'x') and res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

            h2, c2 = continuous_polish(best_binary_h.astype(float), best_binary_n, deadline - 5.0, rng)
            if c2 < best_c:
                best_c = c2
                best_n = best_binary_n
                best_h = h2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
