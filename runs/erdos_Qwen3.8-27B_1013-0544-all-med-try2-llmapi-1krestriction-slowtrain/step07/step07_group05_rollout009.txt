import numpy as np

from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        add = diff / float(n)
        if diff > 0 and np.all(h + add <= 1.0):
            h += add
        elif diff < 0 and np.all(h + add >= 0.0):
            h += add
    return h


def raw_corr_b(h, b):
    return np.correlate(h, b, mode='full')


def eval_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def apply_transfer(h, b, c, i, j, d, n):
    if d == 0.0 or i == j:
        return
    brev = b[::-1]
    c[i:i + n] += d * brev
    s = n - 1 - i
    c[s:s + n] -= d * h
    c[j:j + n] -= d * brev
    s = n - 1 - j
    c[s:s + n] += d * h
    c[n - 1] -= 2.0 * d * d
    c[n - 1 + i - j] += d * d
    c[n - 1 + j - i] += d * d
    h[i] += d
    h[j] -= d
    b[i] -= d
    b[j] += d


def grad_one(h, p, n):
    l = p - (n - 1)
    g = np.zeros(n, dtype=float)
    s1 = max(0, l)
    e1 = min(n, l + n)
    if e1 > s1:
        g[s1:e1] += 1.0
        g[s1:e1] -= h[s1 - l:e1 - l]
    s2 = max(0, -l)
    e2 = min(n, n - l)
    if e2 > s2:
        g[s2:e2] -= h[s2 + l:e2 + l]
    return g


def gradient_weighted(h, c, n, tau):
    m = len(c)
    cmax = np.amax(c)
    w = np.exp((c - cmax) / tau)
    sw = w.sum()
    if sw <= 0.0:
        w = np.zeros(m, dtype=float)
        w[c >= cmax - 1e-11] = 1.0
        sw = max(1.0, float(w.sum()))
    w /= sw
    cum = np.concatenate(([0.0], np.cumsum(w)))
    A = cum[n:n + n] - cum[:n]
    S = np.convolve(h, w, mode='full')[n - 1:2 * n - 1]
    R = np.correlate(w, h, mode='full')[n - 1:2 * n - 1][::-1]
    return A - S - R


def make_gradient_pairs(g, h, k, max_pairs):
    n = len(g)
    if n < 2:
        return []
    k = min(k, n - 1)
    if k <= 0:
        return []
    low = g.copy()
    low[h >= 1.0 - 1e-10] = 1e100
    high = g.copy()
    high[h <= 1e-10] = -1e100
    if k < n:
        lo_idx = np.argpartition(low, k - 1)[:k]
        hi_idx = np.argpartition(-high, k - 1)[:k]
    else:
        lo_idx = np.argsort(low)[:k]
        hi_idx = np.argsort(-high)[:k]
    lo_idx = lo_idx[np.argsort(low[lo_idx])]
    hi_idx = hi_idx[np.argsort(-high[hi_idx])]
    pairs = []
    seen = set()
    for i in lo_idx:
        li = low[i]
        if li > 1e99:
            continue
        for j in hi_idx:
            hj = high[j]
            if hj < -1e99 or i == j:
                continue
            key = i * n + j if i < j else j * n + i
            if key in seen:
                continue
            seen.add(key)
            pairs.append((li - hj, i, j))
    pairs.sort()
    return [(i, j) for _, i, j in pairs[:max_pairs]]


def trial_oriented(h, b, c, i, j, d, base):
    maxd = min(1.0 - h[i], h[j])
    if maxd <= 1e-12:
        return base, 0.0, i, j
    if d > maxd:
        d = maxd
    if d <= 1e-12:
        return base, 0.0, i, j
    n = len(h)
    apply_transfer(h, b, c, i, j, d, n)
    val = float(c.max())
    apply_transfer(h, b, c, i, j, -d, n)
    return val, d, i, j


def try_pair_trial(h, b, c, i, j, fracs, base):
    n = len(h)
    best = base
    best_i = -1
    best_j = -1
    best_d = 0.0

    def add_trial(ii, jj, vals):
        nonlocal best, best_i, best_j, best_d
        for d in vals:
            if d <= 1e-12:
                continue
            val, dd, ai, aj = trial_oriented(h, b, c, ii, jj, d, base)
            if val < best - 1e-12:
                best = val
                best_i = ai
                best_j = aj
                best_d = dd

    if i != j:
        lim = min(1.0 - h[i], h[j])
        vals = []
        for f in fracs:
            d = f * lim
            if d > 1e-10:
                if not any(abs(d - v) < 1e-11 for v in vals):

                    vals.append(ad)
        if 0.98 * lim > 1e-10:
            if not any(abs(0.98 * lim - v) < 1e-11 for v in vals):
                vals.append(0.98 * lim)
        add_trial(i, j, vals)

        lim = min(1.0 - h[j], h[i])
        vals = []
        for f in fracs:
            d = f * lim
            if d > 1e-10:
                if not any(abs(d - v) < 1e-11 for v in vals):

def local_search(h_start, n, deadline, seed=0):
    if n < 4:
        return project_feasible(h_start, n), eval_c5(project_feasible(h_start, n), n)

    rng = np.random.RandomState(seed)
    h = project_feasible(h_start, n)
    b = 1.0 - h
    c = raw_corr_b(h, b)
    cur = float(c.max())
    best_h = h.copy()
    best_raw = cur
    eval_ct = 0
    no_move = 0
    best_stall = 0
    T = 0.25
    eps = 1e-11
    fracs_full = (0.01, 0.05, 0.15, 0.45, 0.90)

    while time.time() < deadline:
        if eval_ct > 1500:
            b = 1.0 - h
            c = raw_corr_b(h, b)
            cur = float(c.max())
            eval_ct = 0

        if cur < best_raw - 1e-12:
            best_raw = cur
            best_h = h.copy()

        if h.min() < -1e-9 or h.max() > 1.0 + 1e-9 or abs(h.sum() - 0.5 * n) > 1e-7:
            h = project_feasible(h, n)
            b = 1.0 - h
            c = raw_corr_b(h, b)
            cur = float(c.max())

        accepted = False

        moves = []
        seen = set()
        tau = 1.0 if n <= 400 else 1.5
        g = gradient_weighted(h, c, n, tau)
        for pair in make_gradient_pairs(g, h, 4, 6):
            ii, jj = pair
            key = ii * n + jj if ii < jj else jj * n + ii
            if key not in seen:
                seen.add(key)
                moves.append(pair)

        topk = min(8, len(c))
        if topk > 0:
            order = np.argpartition(c, -topk)[-topk:]
            for _ in range(2):
                p = int(rng.choice(order))
                g2 = grad_one(h, p, n)
                for pair in make_gradient_pairs(g2, h, 3, 3):
                    ii, jj = pair
                    key = ii * n + jj if ii < jj else jj * n + ii
                    if key not in seen:
                        seen.add(key)
                        moves.append(pair)

        best_val = cur
        best_i = best_j = -1
        best_d = 0.0
        tried = 0
        for pair in moves:
            if time.time() > deadline:
                break
            ii, jj = pair
            val, ai, aj, dd = try_pair_trial(h, b, c, ii, jj, fracs_full, cur)
            eval_ct += 12
            tried += 1
            if val < best_val - eps:
                best_val = val
                best_i = ai
                best_j = aj
                best_d = dd

        if best_i >= 0:
            apply_transfer(h, b, c, best_i, best_j, best_d, n)
            cur = float(c.max())
            accepted = True

        if not accepted:
            B = 22 if n <= 350 else 14
            best_mov = None
            best_mov_val = 1e100
            for _ in range(B):
                if time.time() > deadline:
                    break
                a = int(rng.randint(n))
                bb = int(rng.randint(n))
                if a == bb:
                    continue
                if rng.rand() < 0.5:
                    aa, cc = a, bb
                else:
                    aa, cc = bb, a
                maxd = min(1.0 - h[aa], h[cc])
                if maxd <= 1e-9:
                    continue
                d = maxd * rng.uniform(0.02, 0.95)
                val, dd, ia, jc = trial_oriented(h, b, c, aa, cc, d, cur)
                eval_ct += 1
                if val < best_mov_val:
                    best_mov_val = val
                    best_mov = (ia, jc, dd)

            if best_mov is not None:
                val, ia, jc, dd = best_mov
                if val < cur - eps:
                    apply_transfer(h, b, c, ia, jc, dd, n)
                    cur = float(c.max())
                    accepted = True
                else:
                    delta = (cur - val) / T if T > 1e-7 else 60.0
                    if delta < 0.0 and delta > -60.0 and rng.rand() < math.exp(delta):
                        apply_transfer(h, b, c, ia, jc, dd, n)
                        cur = float(c.max())
                        accepted = True
            T = max(1e-5, T * 0.975)

        if accepted:
            no_move = 0
            if cur < best_raw - 1e-10:
                best_raw = cur
                best_h = h.copy()
                best_stall = 0
            else:
                best_stall += 1
            T = max(1e-5, T * 0.995)
            if cur > best_raw + max(1.0, 10.0 * T):
                h = best_h.copy()
                b = 1.0 - h
                c = raw_corr_b(h, b)
                cur = float(c.max())
                no_move = 0
                best_stall = 0
                T = max(T, 0.10)
        else:
            no_move += 1
            best_stall += 1
            if no_move > 220 or (best_stall > 260 and T < 1e-3):
                break

    best_h = project_feasible(best_h, n)
    best_raw = eval_c5(best_h, n)
    return best_h, best_raw


def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if alpha < 1e-6:
        return m
    sh = alpha * (c - m)
    sh = sh - sh.max()
    return m + (sh.max() + np.log(np.sum(np.exp(sh)))) / alpha


def _sum_con(x, n):
    return float(x.sum() - 0.5 * n)


def _sum_jac(x, n):
    return np.ones(len(x), dtype=float)


def slsqp_polish(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = eval_c5(h, n)
    maxit = 180 if n <= 320 else 80
    for alpha in (2.0, 6.0, 25.0):
        if time.time() > deadline - 8:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
        try:
            res = minimize(
                smooth_obj, h, args=(n, alpha), method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxit, 'ftol': 1e-13}
            )
            hp = project_feasible(res.x, n)
            cp = eval_c5(hp, n)
            if cp < best_c:
                best_c = cp
                best_h = hp
        except Exception:
            pass
    return best_h, best_c


def resample_h(h, n0, n1, method='interp'):
    if n0 == n1:
        return h.copy()
    if method == 'repeat' and n1 % n0 == 0:
        return np.repeat(h, n1 // n0).copy()
    x0 = np.arange(n0) * (2.0 / n0)
    x1 = np.arange(n1) * (2.0 / n1)
    return np.interp(x1, x0, h)


def perturb_h(h, scale, rng):
    n = len(h)
    x = np.arange(n) * (2.0 / n)
    p = h + scale * rng.normal(0.0, 1.0, n)
    for _ in range(2):
        f = int(rng.randint(1, 5))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = scale * rng.uniform(0.25, 1.0)
        p += amp * np.sin(np.pi * f * x + ph)
    return project_feasible(p, n)


def binary_start(h):
    return project_feasible((h > 0.5).astype(float), len(h))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(15.0, float(budget_s))
    deadline = t0 + max(5.0, budget - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        n0 = 250
        x0 = np.arange(n0) * (2.0 / n0)
        h0 = 0.5 + 0.15 * np.sin(np.pi * x0) + 0.06 * np.sin(2.0 * np.pi * x0 + 0.7)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = eval_c5(best_h, n0)
    best_n = n0

    rem = max(1.0, deadline - t0)
    parent_dt = min(rem, max(5.0, 0.35 * budget))
    parent_deadline = t0 + parent_dt
    if parent_deadline > deadline - 12:
        parent_deadline = deadline - 12
    if parent_deadline < t0 + 3:
        parent_deadline = deadline

    hloc, rloc = local_search(h0, n0, parent_deadline, seed)
    cloc = eval_c5(hloc, n0)
    if cloc < best_c:
        best_h, best_c, best_n = hloc, cloc, n0

    up_n = best_n * 2 if best_n * 2 <= 800 else best_n
    if up_n != best_n and time.time() < deadline - 160:
        h_rep = resample_h(best_h, best_n, up_n, 'repeat')
        h_int = resample_h(best_h, best_n, up_n, 'interp')
        hr = project_feasible(h_rep, up_n)
        hi = project_feasible(h_int, up_n)
        cr = eval_c5(hr, up_n)
        ci = eval_c5(hi, up_n)
        h_up = hr if cr <= ci else hi
        remaining = deadline - time.time()
        dt = max(20.0, min(0.38 * remaining, 320.0))
        up_deadline = time.time() + dt
        if up_deadline <= deadline - 60 and up_deadline > time.time() + 20:
            h_u, r_u = local_search(h_up, up_n, up_deadline, seed + 101)
            c_u = eval_c5(h_u, up_n)
            if c_u < best_c:
                best_h, best_c, best_n = h_u, c_u, up_n

    scales = [0.01, 0.02, 0.05, 0.10, 0.20, 0.38]
    it = 0
    while time.time() < deadline - 30:
        remaining = deadline - time.time()
        if best_n > 500:
            alloc = min(55.0, 0.55 * remaining)
        elif best_n > 320:
            alloc = min(75.0, 0.60 * remaining)
        else:
            alloc = min(110.0, 0.65 * remaining)
        if alloc < 30.0:
            break

        if it % 4 == 3:
            h_start = binary_start(best_h)
        elif it % 7 == 5:
            h_start = project_feasible(rng.random(best_n), best_n)
        else:
            h_start = perturb_h(best_h, scales[it % len(scales)], rng)

        hloc, rloc = local_search(h_start, best_n, time.time() + alloc, seed + 1000 + it)
        cloc = eval_c5(hloc, best_n)
        if cloc < best_c:
            best_h, best_c = hloc, cloc
        it += 1

    if best_n <= 400 and time.time() < deadline - 35:
        hp, cp = slsqp_polish(best_h, best_n, deadline - 12)
        if cp < best_c:
            best_h, best_c = hp, cp

    if time.time() < deadline - 4:
        hfin, rfin = local_search(best_h, best_n, deadline, seed + 9999)
        cf = eval_c5(hfin, best_n)
        if cf < best_c:
            best_h, best_c = hfin, cf

    best_h = project_feasible(best_h, len(best_h))
    c_final = eval_c5(best_h, len(best_h))
    return best_h, c_final, len(best_h)
