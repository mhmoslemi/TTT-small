def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    """Compute smooth max objective and its gradient."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # For numerical stability
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    log_sum = mx + np.log(exp_shifted.sum())
    obj = m + log_sum / alpha
    # Softmax weights
    weights = exp_shifted / exp_shifted.sum()
    # Gradient: weighted sum of individual lag gradients
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        w = weights[k]
        lag = k - (n - 1)
        m_abs = abs(lag)
        if m_abs == 0:
            grad += w * (1.0 - 2.0 * x) * dx
        elif lag > 0:
            grad[:n - m_abs] += w * (-x[m_abs:]) * dx
            grad[m_abs:] += w * (1.0 - x[:n - m_abs]) * dx
        else:
            grad[:n - m_abs] += w * (1.0 - x[m_abs:]) * dx
            grad[m_abs:] += w * (-x[:n - m_abs]) * dx
    return obj, grad

Actually, let me reconsider the coordinate descent approach. The key insight is:

For coordinate i, each overlap_ℓ is affine in h[i] (except ℓ=0 which is quadratic/concave). So the problem "minimize max_ℓ f_ℓ(t) subject to t ∈ [0,1]" where f_ℓ is affine (or concave quadratic for ℓ=0) can be solved efficiently.

For the affine case, the min of the max of affine functions is at a breakpoint (intersection of two lines) or at the boundary. We can find it by:
1. Computing all slopes a_ℓ and intercepts b_ℓ
2. The max function is the upper envelope of lines
3. The minimum of the upper envelope is where two lines cross at the bottom of a "valley"

A simpler approach: bisection on T.
- Feasibility check: is there t ∈ [0,1] s.t. for all ℓ, f_ℓ(t) ≤ T?
- For affine f_ℓ(t) = a_ℓ*t + b_ℓ:
  - If a_ℓ > 0: t ≤ (T - b_ℓ) / a_ℓ
  - If a_ℓ < 0: t ≥ (T - b_ℓ) / a_ℓ
  - If a_ℓ = 0: need b_ℓ ≤ T
- For concave quadratic f_0(t) = -c*t^2 + d*t + e (with c > 0):
  - Need -c*t^2 + d*t + e ≤ T, i.e., c*t^2 - d*t + (T-e) ≥ 0
  - This is satisfied outside the roots of the quadratic

The intersection of all these half-intervals (and the complement of the quadratic's interior) with [0,1] gives the feasible set. If non-empty, T is achievable.

Bisection on T converges in O(log(1/eps)) steps, each O(n). Total: O(n * log(1/eps)) per coordinate update.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final projection to ensure exact sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    log_sum = mx + np.log(exp_s.sum())
    obj = m + log_sum / alpha
    weights = exp_s / exp_s.sum()
    # Only compute gradient for active lags (weights > threshold)
    active = weights > 1e-8
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        w = weights[k]
        lag = k - (n - 1)
        m_abs = abs(lag)
        if m_abs == 0:
            grad += w * (1.0 - 2.0 * x) * dx
        elif lag > 0:
            grad[:n - m_abs] += w * (-x[m_abs:]) * dx
            grad[m_abs:] += w * (1.0 - x[:n - m_abs]) * dx
        else:
            grad[:n - m_abs] += w * (1.0 - x[m_abs:]) * dx
            grad[m_abs:] += w * (-x[:n - m_abs]) * dx
    return obj, grad

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_and_grad, h, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      jac=True, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_method(h, n, deadline, max_iter=50000, lb=0.3):
    """Subgradient method with Polyak step size."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        cur_val = c.max()
        if cur_val < best_c - 1e-15:
            best_c = cur_val
            best_h = h.copy()
        # Find active lag
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m_abs = abs(lag)
        # Compute subgradient
        g = np.zeros(n)
        if m_abs == 0:
            g[:] = (1.0 - 2.0 * h) * dx
        elif lag > 0:
            g[:n - m_abs] -= h[m_abs:] * dx
            g[m_abs:] += (1.0 - h[:n - m_abs]) * dx
        else:
            g[:n - m_abs] += (1.0 - h[m_abs:]) * dx
            g[m_abs:] -= h[:n - m_abs] * dx
        # Polyak step size: (f(x) - f*) / ||g||^2
        gap = cur_val - lb
        g_norm_sq = g @ g
        if g_norm_sq < 1e-20:
            break
        step = gap / g_norm_sq
        step = min(step, 0.5)  # cap the step
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < cur_val:
            h = h_new
        else:
            # Halve the step
            h_new = h - step * 0.5 * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_val:
                h = h_new
            else:
                break  # stuck
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_sweeps=5000):
    """Coordinate descent handling the quadratic ℓ=0 case."""
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Compute coefficients a_k and b_k for each lag as function of t = h[i]
            c_old = compute_all_lags(h, n)
            h_i_old = h[i]
            
            # For each lag, compute the slope (coefficient of t) and intercept
            # overlap_k(t) = a_k * t + b_k (for k != 0)
            # overlap_0(t) = -dx*t^2 + dx*t + b_0 (quadratic)
            
            # Compute slopes
            slopes = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - (n - 1)
                m_abs = abs(lag)
                if m_abs == 0:
                    # Special: quadratic
                    slopes[k] = 0  # mark as quadratic
                elif lag > 0:
                    s = 0.0
                    if i + m_abs < n:
                        s += (1.0 - h[i + m_abs])
                    if i - m_abs >= 0:
                        s -= h[i - m_abs]
                    slopes[k] = s * dx
                else:
                    s = 0.0
                    if i + m_abs < n:
                        s += (1.0 - h[i - m_abs])
                    if i - m_abs >= 0:
                        s -= h[i + m_abs]
                    slopes[k] = s * dx
            
            # For the quadratic (lag 0): overlap_0(t) = dx*(t - t^2) + sum_{j≠i} h[j]*(1-h[j])*dx
            # = -dx*t^2 + dx*t + (overlap_0 - dx*h_i_old*(1-h_i_old))
            b_quad = c_old[n-1] - dx * h_i_old * (1 - h_i_old)
            
            # For linear lags: intercept b_k = c_old[k] - slopes[k] * h_i_old
            b_lin = c_old - slopes * h_i_old
            
            # Now solve: min over t ∈ [0,1] of max(quad(t), max_k (a_k*t + b_k))
            # Use bisection on T
            # Feasibility: exists t ∈ [0,1] s.t. quad(t) ≤ T AND for all k≠0, a_k*t + b_k ≤ T
            
            lo, hi = 0.0, 1.0
            # Find the current max as upper bound
            T_hi = max(c_old.max(), c_old[n-1])  # start with current value + a bit
            T_lo = max(0.0, T_hi - 0.5)  # will be refined
            
            # Bisection
            T_lo = 0.0
            T_hi = T_hi + 0.1
            for _ in range(50):
                T_mid = (T_lo + T_hi) / 2
                # Check feasibility at T = T_mid
                # For linear constraints: a_k * t + b_k ≤ T_mid
                t_min = 0.0
                t_max = 1.0
                feasible = True
                for k in range(2 * n - 1):
                    if k == n - 1:
                        continue  # skip quadratic
                    a = slopes[k]
                    b = b_lin[k]
                    if abs(a) < 1e-15:
                        if b > T_mid + 1e-15:
                            feasible = False
                            break
                    elif a > 0:
                        t_max = min(t_max, (T_mid - b) / a)
                    else:
                        t_min = max(t_min, (T_mid - b) / a)
                if not feasible or t_min > t_max + 1e-12:
                    T_hi = T_mid
                    continue
                # Check quadratic: -dx*t^2 + dx*t + b_quad ≤ T_mid
                # => dx*t^2 - dx*t + (T_mid - b_quad) ≥ 0
                # Roots: t = (dx ± sqrt(dx^2 - 4*dx*(T_mid - b_quad))) / (2*dx)
                #       = (1 ± sqrt(1 - 4*(T_mid - b_quad)/dx)) / 2
                disc = 1.0 - 4.0 * (T_mid - b_quad) / dx
                if disc < 0:
                    # Quadratic is always positive, so constraint is always satisfied
                    pass
                else:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    # Need t ≤ r1 OR t ≥ r2
                    # So feasible t: [t_min, r1] ∪ [r2, t_max] (intersected with [0,1])
                    if t_min <= r1 + 1e-12 or t_max >= r2 - 1e-12:
                        # Feasible
                        T_hi = T_mid
                    else:
                        T_lo = T_mid
                if T_hi - T_lo < 1e-10:
                    break
                if disc < 0:
                    T_hi = T_mid
                    continue
            
            # T_hi is our answer (the minimum achievable max value)
            # Now find the actual t: use T_hi and solve for t
            T = T_hi + 1e-10  # slight padding
            t_min = 0.0
            t_max = 1.0
            for k in range(2 * n - 1):
                if k == n - 1:
                    continue
                a = slopes[k]
                b = b_lin[k]
                if abs(a) < 1e-15:
                    continue
                elif a > 0:
                    t_max = min(t_max, (T - b) / a)
                else:
                    t_min = max(t_min, (T - b) / a)
            # Handle quadratic
            disc = 1.0 - 4.0 * (T - b_quad) / dx
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                # Pick t in [t_min, r1] or [r2, t_max]
                # Prefer the one that minimizes the quadratic (closer to 0.5)
                if t_min <= r1:
                    t_candidate = min(t_max, max(t_min, (t_min + r1) / 2))
                elif t_max >= r2:
                    t_candidate = min(t_max, max(t_min, (r2 + t_max) / 2))
                else:
                    t_candidate = (t_min + t_max) / 2
            else:
                t_candidate = (t_min + t_max) / 2
            
            t_candidate = np.clip(t_candidate, 0.0, 1.0)
            
            # Update
            h[i] = t_candidate
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                improved_any = True
            elif abs(c_new - c_old.max()) > 1e-10:
                # Revert if it got worse
                h[i] = h_i_old
        
        if not improved_any and sweep > 2:
            # Try projection
            h = project_feasible(h, n)
            c_check = compute_c5(h, n)
            if c_check < best_c:
                best_c = c_check
                best_h = h.copy()
            if sweep > 20:
                break
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    Z = exp_s.sum()
    obj = m + (mx + np.log(Z)) / alpha
    w = exp_s / Z
    # Gradient via active lags only
    grad = np.zeros(n)
    idx = np.where(w > 1e-10)[0]
    for k in idx:
        lag = k - (n - 1)
        ma = abs(lag)
        wk = w[k]
        if ma == 0:
            grad += wk * (1.0 - 2.0 * x) * dx
        elif lag > 0:
            grad[:n-ma] -= wk * x[ma:] * dx
            grad[ma:] += wk * (1.0 - x[:n-ma]) * dx
        else:
            grad[:n-ma] += wk * (1.0 - x[ma:]) * dx
            grad[ma:] -= wk * x[:n-ma] * dx
    return obj, grad

def _sum_eq(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_and_grad, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons, jac=True,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

            g[ma:] += (1.0 - h[:n-ma]) * dx

5. If infeasible, T is too small

def outer_loop(h, n, deadline, max_iter=100):
    import cvxpy as cp
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1]
        for k in range(2*n-1):
            lag = k - (n-1)
            ma = abs(lag)
            # Compute ∇overlap_k at h
            g = np.zeros(n)
            if ma == 0:
                g[:] = (1.0 - 2.0*h) * dx
            elif lag > 0:
                g[:n-ma] -= h[ma:] * dx
                g[ma:] += (1.0 - h[:n-ma]) * dx
            else:
                g[:n-ma] += (1.0 - h[ma:]) * dx
                g[ma:] -= h[:n-ma] * dx
            # Linearized: c[k] + g·(h_var - h) ≤ t_var
            # = c[k] - g·h + g·h_var ≤ t_var
            rhs = c[k] - g @ h
            cons.append(g @ h_var <= t_var - rhs)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h))
        h = h_new
        if step < 1e-10:
            break
    return best_h, best_c
