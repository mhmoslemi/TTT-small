```
import numpy as np

def corr_raw(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(corr_raw(h, n)) * (2.0 / n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
import math
import time
from scipy.optimize import minimize


def corr_raw(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, n):
    return float(np.max(corr_raw(h, n)) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        res = target - h.sum()
        if abs(res) < 1e-15:
            break
        for k in range(n):
            v = h[k] + res
            if 0.0 <= v <= 1.0:
                h[k] = v
                break
    return h


def sum_eq(x, n):
    return float(x.sum() - n / 2.0)


def sum_jac(x, n):
    return np.ones(n)


def soft_obj(x, n, alpha):
    raw = corr_raw(x, n)
    c = raw * (2.0 / n)
    cmax = float(np.max(c))
    z = alpha * (c - cmax)
    s = float(np.exp(z).sum())
    if s <= 0.0:
        return cmax
    return cmax + math.log(s) / alpha


def soft_jac(x, n, alpha):
    raw = corr_raw(x, n)
    dx = 2.0 / n
    c = raw * dx
    cmax = float(np.max(c))
    z = alpha * (c - cmax)
    e = np.exp(z)
    s = float(e.sum())
    g = np.zeros(n)
    if s <= 0.0:
        return g
    w = e / s
    q0 = n - 1
    for idx in np.nonzero(w > 1e-11)[0]:
        sidx = int(idx) - q0
        ws = float(w[idx])
        if sidx >= 0:
            m = n - sidx
            if m > 0:
                g[sidx:] += ws * (1.0 - x[:m])
                g[:m] -= ws * x[sidx:]
        else:
            d = -sidx
            m = n - d
            if m > 0:
                g[:m] += ws * (1.0 - x[d:])
                g[d:] -= ws * x[:m]
    return dx * g


def slsqp_smooth(h, n, alpha, maxiter=200):
    h = project_feasible(np.asarray(h, dtype=float).copy(), n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (n,)}]
        res = minimize(
            soft_obj, h, args=(n, alpha), jac=soft_jac,
            bounds=bounds, constraints=cons, method='SLSQP',
            options={'maxiter': int(maxiter), 'ftol': 1e-11, 'disp': False}
        )
        if res is not None and getattr(res, 'x', None) is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def append_quad_roots(pts, A, B, C, lo, hi):
    if abs(A) < 1e-14:
        if abs(B) > 1e-14:
            r = -C / B
            if lo - 1e-12 <= r <= hi + 1e-12:
                pts.append(r)
        return
    disc = B * B - 4.0 * A * C
    if disc < 0.0:
        if disc > -1e-9:
            disc = 0.0
        else:
            return
    r = math.sqrt(disc)
    x1 = (-B - r) / (2.0 * A)
    x2 = (-B + r) / (2.0 * A)
    if lo - 1e-12 <= x1 <= hi + 1e-12:
        pts.append(x1)
    if abs(x2 - x1) > 1e-14 and lo - 1e-12 <= x2 <= hi + 1e-12:
        pts.append(x2)


def best_pair_alpha(lo, hi, b0, g0, b1, g1, b2, g2, base):
    pts = [lo, hi]
    if lo <= 0.0 and 0.0 <= hi:
        pts.append(0.0)
    t = 0.25 * g0
    if lo - 1e-12 <= t <= hi + 1e-12:
        pts.append(t)
    t = -0.5 * g1
    if lo - 1e-12 <= t <= hi + 1e-12:
        pts.append(t)
    t = -0.5 * g2
    if lo - 1e-12 <= t <= hi + 1e-12:
        pts.append(t)
    append_quad_roots(pts, -3.0, g0 - g1, b0 - b1, lo, hi)
    append_quad_roots(pts, -3.0, g0 - g2, b0 - b2, lo, hi)
    db = g1 - g2
    if abs(db) > 1e-14:
        r = -(b1 - b2) / db
        if lo - 1e-12 <= r <= hi + 1e-12:
            pts.append(r)
    if base > -1e90:
        append_quad_roots(pts, -2.0, g0, b0 - base, lo, hi)
        append_quad_roots(pts, 1.0, g1, b1 - base, lo, hi)
        append_quad_roots(pts, 1.0, g2, b2 - base, lo, hi)
    span = hi - lo
    if span > 1e-13:
        for k in range(1, 9):
            pts.append(lo + span * (k / 8.0))
    best_v = 1e100
    best_a = 0.0
    for x in pts:
        if x < lo:
            x = lo
        elif x > hi:
            x = hi
        v = base
        t = b0 + x * (g0 - 2.0 * x)
        if t > v:
            v = t
        t = b1 + x * (g1 + x)
        if t > v:
            v = t
        t = b2 + x * (g2 + x)
        if t > v:
            v = t
        if v < best_v:
            best_v = v
            best_a = x
    return best_a, best_v


def pair_refine(h, deadline, max_passes=20, min_improve=1e-9):
    n = int(len(h))
    hh = project_feasible(np.asarray(h, dtype=float).copy(), n).tolist()
    L = 2 * n - 1
    q0 = n - 1
    TOP = 12 if L > 12 else L
    best_raw = float(np.max(corr_raw(np.asarray(hh), n)))
    pass_idx = 0
    while time.time() < deadline and pass_idx < max_passes:
        hh = project_feasible(np.asarray(hh, dtype=float), n).tolist()
        cvals = corr_raw(np.asarray(hh), n)
        cur = float(np.max(cvals))
        if cur < best_raw:
            best_raw = cur
        order = np.argsort(-cvals)[:TOP]
        top_list = list(zip(order.tolist(), cvals[order].tolist()))
        improved = False
        if pass_idx % 2 == 0:
            i_range = range(0, n - 1)
        else:
            i_range = range(n - 2, -1, -1)
        for i in i_range:
            hi = hh[i]
            for j in range(i + 1, n):
                hj = hh[j]
                lo = -hj
                t = hi - 1.0
                if t > lo:
                    lo = t
                hiu = 1.0 - hi
                if hj < hiu:
                    hiu = hj
                if hiu <= lo + 1e-14:
                    continue
                d = j - i
                q1 = q0 + d
                q2 = q0 - d
                base = -1e100
                for ti, tv in top_list:
                    if ti != q0 and ti != q1 and ti != q2:
                        base = tv
                        break
                g0 = 2.0 * (hj - hi)
                b0 = float(cvals[q0])
                if i >= d:
                    gi = 1.0 - hh[i - d]
                else:
                    gi = 0.0
                if i + d < n:
                    gi -= hh[j]
                gj = 1.0 - hh[j - d]
                if j + d < n:
                    gj -= hh[j + d]
                g1 = gi - gj
                b1 = float(cvals[q1])
                lim = n - 1 - d
                if i <= lim:
                    gi = 1.0 - hh[j]
                else:
                    gi = 0.0
                if i - d >= 0:
                    gi -= hh[i - d]
                if j <= lim:
                    gj = 1.0 - hh[j + d]
                else:
                    gj = 0.0
                if j - d >= 0:
                    gj -= hh[i]
                g2 = gi - gj
                b2 = float(cvals[q2])
                alpha, val = best_pair_alpha(lo, hiu, b0, g0, b1, g1, b2, g2, base)
                if val < cur - min_improve:
                    hh[i] += alpha
                    hh[j] -= alpha
                    cvals[q0] = b0 + alpha * g0 - 2.0 * alpha * alpha
                    cvals[q1] = b1 + alpha * g1 + alpha * alpha
                    cvals[q2] = b2 + alpha * g2 + alpha * alpha
                    cur = val
                    if cur < best_raw:
                        best_raw = cur
                    improved = True
                    order = np.argsort(-cvals)[:TOP]
                    top_list = list(zip(order.tolist(), cvals[order].tolist()))
        pass_idx += 1
        if not improved:
            break
    h_arr = project_feasible(np.asarray(hh, dtype=float), n)
    return h_arr, compute_c5(h_arr, n)


def interp_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)


def make_default_start(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.22 * np.sin(math.pi * x) + 0.10 * np.sin(2.0 * math.pi * x)
    return project_feasible(h, n)


def choose_better(bh, bc, bn, h, c, n):
    if c < bc - 1e-12:
        return h, c, n
    if abs(c - bc) <= 1e-9 and n < bn:
        return h, c, n
    return bh, bc, bn


def optimize_candidate(h_start, n, t_start, t_end, pair_passes=15):
    h = project_feasible(np.asarray(h_start, dtype=float).copy(), n)
    c = compute_c5(h, n)
    t_end = float(t_end)
    rem = t_end - time.time()
    if rem <= 12.0:
        return h, c
    if n <= 120:
        mit = 280
    elif n <= 200:
        mit = 180
    else:
        mit = 110
    slsqp_end = time.time() + min(110.0, max(18.0, rem * 0.24))
    if slsqp_end > t_end - 60.0:
        slsqp_end = t_end - 60.0
    for a in (500, 3000, 30000, 150000):
        if time.time() > slsqp_end or time.time() > t_end - 80.0:
            break
        ht = slsqp_smooth(h, n, a, maxiter=mit)
        c2 = compute_c5(ht, n)
        if c2 < c - 1e-14:
            h, c = ht, c2
    if time.time() < t_end - 10.0:
        pair_end = time.time() + max(18.0, (t_end - time.time()) * 0.45)
        if pair_end > t_end - 18.0:
            pair_end = t_end - 18.0
        h, c = pair_refine(h, pair_end, max_passes=pair_passes)
    if time.time() < t_end - 25.0:
        ht = slsqp_smooth(h, n, 200000, maxiter=70)
        c2 = compute_c5(ht, n)
        if c2 < c - 1e-14:
            h, c = ht, c2
    if time.time() < t_end - 3.0:
        h, c = pair_refine(h, t_end - 2.0, max_passes=max(3, pair_passes // 2))
    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        B = float(budget_s)
    except Exception:
        B = 1100.0
    deadline = t0 + B - 6.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
    except Exception:
        n0 = 200
        h0 = make_default_start(n0)
    if n0 < 40:
        n0 = 80
        h0 = make_default_start(n0)
    h0 = project_feasible(h0, n0)
    best_h, best_c, best_n = h0, compute_c5(h0, n0), n0

    if deadline - t0 > 220.0:
        phase_end = t0 + min(420.0, max(120.0, (deadline - t0) * 0.34))
        h1, c1 = optimize_candidate(h0, n0, t0, phase_end, 24)
        best_h, best_c, best_n = choose_better(best_h, best_c, best_n, h1, c1, n0)

    ns = [best_n]
    for m in (200, 150, 120, 90, 300):
        if 40 <= m <= 400 and m not in ns:
            ns.append(m)
    ns = ns[:5]

    for idx in range(1, len(ns)):
        n = ns[idx]
        if time.time() > deadline - 70.0:
            break
        left = len(ns) - idx
        alloc = (deadline - time.time()) / float(left)
        alloc = min(210.0, max(45.0, alloc))
        dend = time.time() + alloc
        if dend > deadline - 38.0:
            dend = deadline - 38.0
        if dend <= time.time() + 10.0:
            break
        hstart = interp_h(best_h, best_n, n)
        hstart = hstart + rng.normal(0.0, 0.015, n)
        hstart = project_feasible(hstart, n)
        hn, cn = optimize_candidate(hstart, n, time.time(), dend, 12)
        best_h, best_c, best_n = choose_better(best_h, best_c, best_n, hn, cn, n)

    if time.time() < deadline - 90.0:
        nb = int(best_n)
        restarts = int(max(2, min(5, (deadline - time.time()) / 55.0)))
        for r in range(restarts):
            if time.time() > deadline - 28.0:
                break
            amp = 0.02 + 0.07 * (float(r) / float(max(1, restarts - 1)))
            if r == 1 and restarts > 1:
                hp = (best_h > 0.5).astype(float)
                hp = project_feasible(hp, nb)
            else:
                hp = best_h + rng.normal(0.0, amp, nb)
                hp = project_feasible(hp, nb)
            dend = time.time() + min(55.0, max(20.0, (deadline - time.time() - 18.0) * 0.5))
            if dend > deadline - 10.0:
                dend = deadline - 10.0
            if dend <= time.time() + 8.0:
                break
            hn, cn = optimize_candidate(hp, nb, time.time(), dend, 10)
            best_h, best_c, best_n = choose_better(best_h, best_c, best_n, hn, cn, nb)

    if time.time() < deadline - 4.0:
        best_h, c = pair_refine(best_h, deadline - 1.0, max_passes=3)
        best_c = c

    best_h = project_feasible(best_h, int(best_n))
    best_c = compute_c5(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
```
