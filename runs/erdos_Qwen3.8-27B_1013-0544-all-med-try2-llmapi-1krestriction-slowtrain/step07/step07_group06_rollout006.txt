def compute_effect(h,n):
    L=2*n-1; z=n-1; eff=np.zeros((n,L),float)
    eff[:,z] = 1.0 - 2.0*h
    for d in range(1,n):
        m=n-d
        pos=z+d; neg=z-d
        d1h_d = 1.0 - h[d:d+m]
        m1h = -h[0:m]
        # positive
        eff[0:m,pos] += d1h_d
        eff[d:d+m,pos] += m1h
        # negative
        eff[d:d+m,neg] += 1.0 - h[0:m]
        eff[0:m,neg] += -h[d:d+m]

def compute_effect(h,n):
    L=2*n-1; z=n-1
    eff=np.zeros((n,L),float)
    eff[:,z] = 1.0 - 2.0*h
    for d in range(1,n):
        m=n-d; pos=z+d; neg=z-d
        hd = h[d:d+m]
        hdd = h[0:m]
        one_hd = 1.0 - hd
        one_hdd = 1.0 - hdd
        eff[0:m, pos] += one_hd
        eff[d:d+m, pos] += -hdd
        eff[d:d+m, neg] += one_hdd
        eff[0:m, neg] += -hd
    return eff
```

    h=np.array(h,dtype=float,copy=True)
    np.clip(h,0,1,out=h)
    target=n*0.5
    for _ in range(100):

        if abs(diff)<1e-14: break

import numpy as np
import time

SCAN_N = 24
SCAN_FRACS = np.linspace(1.0 / 64.0, 1.0, SCAN_N)


def c5_counts(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(h, n):
    return float(c5_counts(h, n).max()) * (2.0 / n)


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(100):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = (1.0 - h)[mask]
            rs = float(room.sum())
            if rs <= 0.0:
                break
            h[mask] += (diff / rs) * room
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            vals = h[mask]
            vs = float(vals.sum())
            if vs <= 0.0:
                break
            h[mask] += (diff / vs) * vals
    return np.clip(h, 0.0, 1.0)


def lag_weights(c, gamma):
    cmax = float(c.max())
    diff = cmax - c
    w = np.exp(-gamma * diff)
    wm = float(w.max())
    if wm > 0.0:
        w /= wm
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    return w


def compute_effect(h, n):
    L = 2 * n - 1
    z = n - 1
    eff = np.zeros((n, L), dtype=float)
    eff[:, z] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        pos = z + d
        neg = z - d
        hd = h[d:d + m]
        hdd = h[0:m]
        eff[0:m, pos] += 1.0 - hd
        eff[d:d + m, pos] += -hdd
        eff[d:d + m, neg] += 1.0 - hdd
        eff[0:m, neg] += -hd
    return eff


def pair_value_scan(c, B, dmax, diff, center, work):
    best_val = 1e100
    best_d = 0.0
    for f in SCAN_FRACS:
        d = f * dmax
        np.multiply(B, d, work)
        work += c
        d2 = d * d
        work[center] -= 2.0 * d2
        work[center + diff] += d2
        work[center - diff] += d2
        v = float(work.max())
        if v < best_val:
            best_val = v
            best_d = d
    return best_val, best_d


def pair_value_bin(c, B, center, diff, work):
    np.copyto(work, B)
    work += c
    work[center] -= 2.0
    work[center + diff] += 1.0
    work[center - diff] += 1.0
    return float(work.max())


def random_continuous_perturb(h, n, rng, sigma):
    h2 = h.copy()
    k = min(20, n)
    if k > 0:
        idx = rng.choice(n, k, replace=False)
        h2[idx] += rng.normal(0.0, sigma, k)
    return project_feasible(h2, n)


def balance_binary(a, n, rng):
    target = n // 2 if n % 2 == 0 else int(round(0.5 * n))
    a = (np.asarray(a, dtype=float) > 0.5).astype(float)
    cnt = int(in(a.sum()) + 0.5)
    diff = int(target) - cnt
    if diff > 0:
        idx = np.where(a < 0.5)[0]
        if diff <= len(idx):
            a[rng.permutation(idx)[:diff]] = 1.0
    elif diff < 0:
        idx = np.where(a > 0.5)[0]
        if -diff <= len(idx):
            a[rng.permutation(idx)[: -diff]] = 0.0
    return a


def random_binary(n, target, rng):
    a = np.zeros(n, dtype=float)
    if 0 < target < n:
        a[rng.choice(n, target, replace=False)] = 1.0
    return a


def binary_from_samples(h, target, rng, noise):
    n = len(h)
    if noise > 0.0:
        vals = h + rng.normal(0.0, noise, n)
    else:
        vals = h
    order = np.argsort(vals)[::-1]
    a = np.zeros(n, dtype=float)
    if target > 0:
        a[order[:int(target)]] = 1.0
    return balance_binary(a, n, rng)


def low_disc_binary(n, target, alpha, phase, rng):
    x = np.arange(n, dtype=float)
    f = np.mod(alpha * x + phase, 1.0)
    a = (f < 0.5).astype(float)
    return balance_binary(a, n, rng)


def mutate_binary(a, n, rng, swaps):
    a = a.copy()
    for _ in range(int(swaps)):
        ones = np.where(a > 0.5)[0]
        zeros = np.where(a < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[int(rng.integers(len(ones)))]
        q = zeros[int(rng.integers(len(zeros)))]
        a[p] = 0.0
        a[q] = 1.0
    return a


def binary_local_search(a, n, deadline, rng):
    target = n // 2 if n % 2 == 0 else int(round(0.5 * n))
    a = balance_binary(a, n, rng)
    c = c5_counts(a, n)
    best_val = float(c.max())
    best_a = a.copy()
    center = n - 1
    L = 2 * n - 1
    work = np.empty(L, dtype=float)
    stall = 0
    gamma = 1.0
    it = 0

    while time.time() < deadline:
        it += 1
        cur_best = float(c.max())
        eff = compute_effect(a, n)
        eff_score = eff.copy()
        eff_score[:, center] = 0.0
        w = lag_weights(c, gamma)
        Es = eff_score.dot(w)
        dec = np.full(n, np.inf, dtype=float)
        inc = np.full(n, np.inf, dtype=float)
        ones = a > 0.5
        zeros = a < 0.5
        dec[ones] = -Es[ones]
        inc[zeros] = Es[zeros]
        K = min(45, n)
        ps = np.argsort(dec)[:K]
        qs = np.argsort(inc)[:K]

        found_val = cur_best
        found = None
        for p in ps:
            if a[p] < 0.5:
                continue
            ep = eff[p]
            for q in qs:
                if a[q] > 0.5 or p == q:
                    continue
                B = eff[q] - ep
                v = pair_value_bin(c, B, center, p - q, work)
                if v < found_val - 1e-10:
                    found_val = v
                    found = (int(p), int(q))

        improved = False
        if found is not None:
            p, q = found
            a[q] = 1.0
            a[p] = 0.0
            c = c5_counts(a, n)
            val = float(c.max())
            if val > cur_best + 1e-8:
                a[q] = 0.0
                a[p] = 1.0
                c = c5_counts(a, n)
            else:
                improved = True
                if val < best_val - 1e-12:
                    best_val = val
                    best_a = a.copy()
                gamma = min(12.0, gamma + 0.2)

        if not improved:
            improved = False
            ones_idx = np.where(a > 0.5)[0]
            zeros_idx = np.where(a < 0.5)[0]
            if len(ones_idx) and len(zeros_idx):
                rv_val = cur_best
                rv_pair = None
                for _ in range(80):
                    p = ones_idx[int(rng.integers(len(ones_idx)))]
                    q = zeros_idx[int(rng.integers(len(zeros_idx)))]
                    if p == q:
                        continue
                    B = eff[q] - eff[p]
                    v = pair_value_bin(c, B, center, p - q, work)
                    if v < rv_val - 1e-10:
                        rv_val = v
                        rv_pair = (int(p), int(q))
                if rv_pair is not None:
                    p, q = rv_pair
                    a[q] = 1.0
                    a[p] = 0.0
                    c = c5_counts(a, n)
                    val = float(c.max())
                    if val > cur_best + 1e-8:
                        a[q] = 0.0

                        c = c5_counts(a, n)
                    else:
                        improved = True
                        if val < best_val - 1e-12:
                            best_val = val
                            best_a = a.copy()

        if not improved:
            stall += 1
            gamma = 0.6 + 0.7 * min(stall, 10)
            if stall >= 6:
                a = mutate_binary(a, n, rng, 5 + int(min(stall, 20)))
                c = c5_counts(a, n)
                stall = 0
                gamma = 1.0

    return best_a, best_val


def adam_phase(h, n, deadline, rng, gamma=3.0):
    h = project_feasible(h.copy(), n)
    c = c5_counts(h, n)
    best_val = float(c.max())
    best_h = h.copy()
    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    cur_val = best_val

    while time.time() < deadline and t < 5000:
        t += 1
        w = lag_weights(c, gamma + 0.05 * t)
        eff = compute_effect(h, n)
        g = w.dot(eff)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        mhat = m / (1.0 - beta1 ** t)
        vhat = v / (1.0 - beta2 ** t)
        lr = 0.04 / (np.sqrt(t) + 1.0)
        hn = project_feasible(h - lr * mhat / (np.sqrt(vhat) + eps), n)
        cn = c5_counts(hn, n)
        vn = float(cn.max())
        if vn < cur_val + 0.05:
            h = hn
            c = cn
            cur_val = vn
        if vn < best_val - 1e-12:
            best_val = vn
            best_h = h.copy()

    return best_h, best_val


def continuous_pair_phase(h, n, deadline, rng):

    center = n - 1

    stall = 0
    worse_streak = 0
    it = 0
