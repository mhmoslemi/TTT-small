def coord_descent_fast(h, n, max_sweeps, deadline=None):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max(x, alpha):
    m = x.max()
    sh = alpha * (x - m)
    sh = sh - sh.max()
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def grad_c5_smooth(h, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    val = soft_max(corr, alpha) + lam * (h.sum() - n / 2.0) ** 2
    m = corr.max()
    sh = alpha * (corr - m)
    sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(-(n - 1), n):
        wv = w[k + n - 1]
        if abs(wv) < 1e-16:
            continue
        if k >= 0:
            A[:n - k] += wv * h[k:]
            B[k:] += wv * h[:n - k]
        else:
            ss = -k
            A[ss:] += wv * h[:n - ss]
            B[:n - ss] += wv * h[ss:]
    grad = dx * (1.0 - A - B) + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def coord_descent(h, n, max_sweeps=300, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            cur = np.max(corr)
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            if val0 > cur + 1e-14 and val1 > cur + 1e-14:
                continue
            if val0 <= val1:
                lo, hi = 0.0, h[i]
                for _ in range(42):
                    mid = (lo + hi) / 2.0
                    if np.max(corr - B * mid) <= cur + 1e-14:
                        hi = mid
                    else:
                        lo = mid
                    if hi - lo < 1e-14: break
                nv = hi
            else:
                lo, hi = h[i], 1.0
                for _ in range(42):
                    mid = (lo + hi) / 2.0
                    if np.max(corr + B * mid) <= cur + 1e-14:
                        lo = mid
                    else:
                        hi = mid
                    if hi - lo < 1e-14: break
                nv = lo
            if abs(nv - h[i]) > 1e-14:
                corr += B * (nv - h[i])
                h[i] = nv
                improved = True
        if not improved:
            break
    return h

def lbfqsb_phase(h, n, alphas, lam_seq, maxiter, deadline):
    h = h.copy()
    for i, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        lam = lam_seq[min(i, len(lam_seq) - 1)]
        def obj(x):
            return grad_c5_smooth(x, n, alpha, lam)[0]
        def jac(x):
            return grad_c5_smooth(x, n, alpha, lam)[1]
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 80})
            h = project_feasible(res.x, n)
        except Exception:
            h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n).copy()
    best_c = compute_c5(best_h, n)
    best_n = n

    # Phase 1: Coordinate descent from initial
    p1 = t0 + 0.06 * total
    hc = coord_descent(best_h, n, 50, p1)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 2: L-BFGS-B with increasing alpha
    p2 = t0 + 0.40 * total
    alphas2 = [50, 200, 1000, 5000, 50000, 500000, 5000000, 50000000]
    h = lbfqsb_phase(best_h, n, alphas2, [1000.0, 1000.0, 100.0, 10.0, 1.0, 0.1, 0.01, 0.001], 3000, p2)
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 3: Coordinate descent refinement
    p3 = t0 + 0.55 * total
    hc = coord_descent(best_h, n, 300, p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 4: High-alpha L-BFGS-B
    p4 = t0 + 0.72 * total
    alphas4 = [100000000, 500000000, 2000000000]
    h = lbfqsb_phase(best_h, n, alphas4, [0.001, 0.0005, 0.0001], 5000, p4)
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Final coordinate descent
    p5 = t0 + 0.82 * total
    hc = coord_descent(best_h, n, 400, p5)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 6: Try different resolutions via interpolation
    p6 = t0 + 0.92 * total
    for nh in [300, 500, 600]:
        if time.time() > p6 - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = coord_descent(hh, nh, 100, p6 - 15)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), nh
            h = lbfqsb_phase(hh, nh, [5000000, 50000000, 500000000], [0.01, 0.001, 0.0001], 2000, p6 - 10)
            ch = compute_c5(h, nh)
            if ch < best_c:
                best_c, best_h, best_n = ch, h.copy(), nh
            hh2 = coord_descent(h, nh, 200, p6 - 8)
            hh2 = project_feasible(hh2, nh)
            ch2 = compute_c5(hh2, nh)
            if ch2 < best_c:
                best_c, best_h, best_n = ch2, hh2.copy(), nh

    # Phase 7: Perturbation restarts at best resolution
    n = best_n
    if time.time() < dl - 10:
        for trial in range(6):
            if time.time() > dl - 2:
                break
            scale = 0.001 + 0.003 * trial * 0.5
            h = best_h + scale * rng.randn(n)
            h = project_feasible(h, n)
            h = lbfqsb_phase(h, n, [50000000, 500000000], [0.001, 0.0001], 2000, dl - 2)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
            if time.time() < dl - 3:
                hc = coord_descent(h, n, 100, dl - 2)
                hc = project_feasible(hc, n)
                cc = compute_c5(hc, n)
                if cc < best_c:
                    best_c, best_h = cc, hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
