import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx < 500:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    else:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx < 500:
        w = np.exp(sh)
    else:
        w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _eq_con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _eq_con_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    return _sm_obj(x, n, alpha, 0.0)

def _slsqp_grad(x, n, alpha):
    return _sm_grad(x, n, alpha, 0.0)

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (2000, 20.0), (10000, 100.0),
               (50000, 500.0), (200000, 2000.0), (1000000, 10000.0),
               (5000000, 50000.0), (20000000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_sm_obj, h, jac=_sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=200000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgrad_descent(h, n, deadline, lr_init=0.003, top_k=3):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    dx = 2.0 / n
    for it in range(10000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                g[0:nv] += dx * (1.0 - h[k:n])
                g[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += dx * (1.0 - h[0:nv])
                g[0:nv] += dx * (-h[kk:n])
        g /= top_k
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g /= g_norm
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
        if lr < 1e-14:
            break
    return best_h, best_c

def focused_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            crit_idx = np.argsort(c)[-15:]
            for idx in crit_idx:
                if time.time() > deadline:
                    return h, best_c
                k = idx - (n - 1)
                g = np.zeros(n)
                if k >= 0:
                    nv = n - k
                    g[0:nv] += dx * (1.0 - h[k:n])
                    g[k:n] += dx * (-h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += dx * (1.0 - h[0:nv])
                    g[0:nv] += dx * (-h[kk:n])
                pos_idx = np.where(g > 1e-10)[0]
                neg_idx = np.where(g < -1e-10)[0]
                if len(pos_idx) == 0 or len(neg_idx) == 0:
                    continue
                i_star = pos_idx[np.argmax(g[pos_idx])]
                j_star = neg_idx[np.argmin(g[neg_idx])]
                if h[i_star] - d >= 0 and h[j_star] + d <= 1:
                    h[i_star] -= d
                    h[j_star] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        improved_any = True
                    else:
                        h[i_star] += d
                        h[j_star] -= d
            if not improved:
                break
        if not improved_any and d < 0.0005:
            break
    return h, best_c

def local_pair_search(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                            improved = True
                        else:

def upscale(h_coarse, n_coarse, n_fine, rng):
