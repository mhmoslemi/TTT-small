```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:int(abs(diff)) + 1]
            h[idx] = np.minimum(1.0, h[idx] + diff / len(idx))
        else:
            idx = np.argsort(h)[:int(abs(diff)) + 1]
            h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
```

```python
def compute_A_B(h, n, p, dx):
    s = n - 1
    w = p * dx  # length 2n-1
    one_minus_h = 1.0 - h
    
    # Build the Toeplitz matrix for A: T_A[i,j] = w[s + j - i]
    idx = np.arange(n)
    T_A = w[s + (idx[None, :] - idx[:, None])]  # broadcasts to n x n
    A = T_A @ one_minus_h - w[s] * one_minus_h  # subtract diagonal
    
    # Build Toeplitz for B: T_B[i,j] = w[s + i - j]
    T_B = w[s + (idx[:, None] - idx[None, :])]  # transpose of T_A
    B = T_B @ h - w[s] * h  # subtract diagonal
    
    return A, B
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def minmax_coord_descent(h, n, deadline, alpha_start=5.0, alpha_end=500.0, max_sweeps=50000):
    """Min-max algorithm with coordinate descent."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    s = n - 1
    idx = np.arange(n)
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if sweep % 500 == 0 and time.time() > deadline:
            break
        
        frac = min(sweep / max_sweeps, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Compute correlations
        c = np.correlate(h, 1 - h, mode='full') * dx
        
        # Dual weights
        c_shift = alpha * (c - c.max())
        p = np.exp(c_shift - c_shift.max())
        p /= p.sum()
        
        w = p * dx  # length 2n-1
        w_center = w[s]  # = p[s]*dx, the k=0 weight
        
        if w_center < 1e-15:
            # Fallback: just use gradient step
            g = grad_smooth_max(h, n, alpha)
            h_new = h - 0.01 * g
            h = project_feasible(h_new, n)
        else:
            # Compute A and B using Toeplitz
            # A[i] = sum_j w[s+j-i]*(1-h[j]) - w[s]*(1-h[i])
            T_A = w[s + (idx[None, :] - idx[:, None])]
            A = T_A @ (1.0 - h) - w_center * (1.0 - h)
            
            # B[i] = sum_j w[s+i-j]*h[j] - w[s]*h[i]
            T_B = w[s + (idx[:, None] - idx[None, :])]
            B = T_B @ h - w_center * h
            
            # Optimal h[i] = (A[i] + w_center - B[i]) / (2 * w_center)
            opt = (A + w_center - B) / (2.0 * w_center)
            h_new = np.clip(opt, 0.0, 1.0)
            h = project_feasible(h_new, n)
        
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
    
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(5000, 10.0), (50000, 50.0), (500000, 500.0), (5000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False

                    return h, best_c

                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def minmax_coord_descent(h, n, deadline, alpha_start=5.0, alpha_end=500.0, max_sweeps=50000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    s = n - 1
    idx = np.arange(n)
    no_improve = 0
    stall_limit = 3000

    for sweep in range(max_sweeps):
        if sweep % 1000 == 0 and time.time() > deadline:
            break
        frac = min(sweep / max_sweeps, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        c = np.correlate(h, 1 - h, mode='full') * dx
        c_shift = alpha * (c - c.max())
        p = np.exp(c_shift - c_shift.max())
        p /= p.sum()
        w = p * dx
        wc = w[s]

        if wc < 1e-12:
            g = grad_smooth_max(h, n, alpha)
            h = project_feasible(h - 0.01 * g, n)
        else:
            T_A = w[s + (idx[None, :] - idx[:, None])]
            A = T_A @ (1.0 - h) - wc * (1.0 - h)
            T_B = w[s + (idx[:, None] - idx[None, :])]
            B = T_B @ h - wc * h
            num = A + wc - B
            # Solve for lambda so sum(clip((num+lambda)/(2*wc),0,1)) = n/2
            target = n / 2.0
            lam_lo, lam_hi = -50.0, 50.0
            for _ in range(60):
                mid = (lam_lo + lam_hi) * 0.5
                vals = np.clip((num + mid) / (2.0 * wc), 0.0, 1.0)
                if vals.sum() > target:
                    lam_hi = mid
                else:
                    lam_lo = mid
            lam = (lam_lo + lam_hi) * 0.5
            h = np.clip((num + lam) / (2.0 * wc), 0.0, 1.0)

        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > stall_limit:
            break
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(2000, 5.0), (20000, 20.0), (200000, 200.0), (2000000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    try:

    if time.time() < deadline - 200:

        if c3 < best_c:

    # Phase 4: Additional L-BFGS-B at highest alpha
    if time.time() < deadline - 120:

        h4, c4 = lbfgs_phase(best_h, n, t4, max_iters=300)
        if c4 < best_c:
            best_c, best_h = c4, h4

    if time.time() < deadline - 80:
        t6 = deadline - 40

        if remaining > 30:
            n_r = 3
            per_r = remaining / n_r

                if time.time() > t6:

                h_pert = project_feasible(h_pert, n)
                h_r, c_r = minmax_coord_descent(h_pert, n, min(time.time() + per_r, t6),
                                                alpha_start=20.0, alpha_end=1000.0, max_sweeps=10000)
                if c_r < best_c:
                    best_c, best_h = c_r, h_r

    if time.time() < deadline - 5:

                             [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=30)
        if c7 < best_c:
            best_c, best_h = c7, h7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
