import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        sh = sh - sh.max()
        return m + (sh.max() + np.log(np.sum(np.exp(sh)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    mx = c.max()
    sh = alpha * (c - mx)
    if sh.max() > 500:
        sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:k + nv])
                grad[k:k + nv] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk + nv] += wm * (1.0 - h[:nv])
                grad[:nv] -= wm * h[kk:kk + nv]
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _lbfgs_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu = 10.0
    for alpha in [200, 2000, 20000, 100000, 500000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(_lbfgs_obj, h, args=(n, alpha, mu),
                          jac=_lbfgs_grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def adam_optimize(h, n, deadline, alpha_init=200.0, alpha_final=1000000.0):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        alpha = alpha_init * (alpha_final / alpha_init) ** (min(t, 3000) / 3000.0)
        lr = 0.02 * (1.0 + 0.1 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    return best_h, best_c

def coordinate_descent(h, n, deadline, max_passes=500):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    c_vals = compute_c5_vec(h, n)
    k_range = np.arange(-(n - 1), n)
    for p in range(max_passes):

        improved = False

            if time.time() > deadline:
                return h, best_c
            idx1 = i + k_range
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - k_range
            valid2 = (idx2 >= 0) & (idx2 < n)
            g = np.zeros(2 * n - 1)
            g[valid1] = 1.0 - h[idx1[valid1]]
            g[valid2] -= h[idx2[valid2]]
            lo, hi = -h[i], 1.0 - h[i]
            if hi - lo < 1e-10:

            a, b = lo, hi
            for _ in range(18):

                c_new = c_vals + mid * g * dx
                sgrad = g[c_new.argmax()] * dx
                if sgrad < -1e-8:
                    b = mid
                elif sgrad > 1e-8:
                    a = mid
                else:
                    break
            bd = (a + b) * 0.5
            c_new = c_vals + bd * g * dx
            nm = c_new.max()
            if nm < best_c - 1e-13:

                improved = True
        if p % 50 == 49:
            c_vals = compute_c5_vec(h, n)
            best_c = c_vals.max()
        if not improved:
            break
    return h, best_c
