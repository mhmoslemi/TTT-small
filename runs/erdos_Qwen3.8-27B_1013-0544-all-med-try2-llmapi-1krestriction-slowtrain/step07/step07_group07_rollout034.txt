import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(x, n, alpha):
    """Log-sum-exp approximation of max of lag values, with gradient."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Numerical stability
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(exp_shifted.sum())
    obj = m + log_sum / alpha
    # Gradient: p_k = exp(alpha*(c_k - m)) / sum(exp(alpha*(c_k - m)))
    p = exp_shifted / exp_shifted.sum()
    # Sum of p_k * grad_k
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        lag = k - (n - 1)
        g = _lag_grad(x, n, lag)
        grad += p[k] * g
    return obj, grad

def _lag_grad(x, n, lag):
    """Gradient of the lag value w.r.t. x."""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * x)
        return grad
    if lag > 0:
        # c = dx * sum_{i=0}^{n-1-lag} x[i+lag]*(1-x[i])
        # d/dx[j]: if j >= lag and j <= n-1: contribute (1-x[j-lag])
        #          if j >= 0 and j <= n-1-lag: contribute -x[j+lag]
        idx1 = np.arange(lag, n)  # j where x[j] appears as x[i+lag]
        grad[idx1] += dx * (1.0 - x[idx1 - lag])
        idx2 = np.arange(0, n - lag)  # j where x[j] appears as x[i]
        grad[idx2] -= dx * x[idx2 + lag]
    else:
        m = -lag
        # c = dx * sum_{n=m}^{n-1} x[n-m]*(1-x[n])
        # d/dx[j]: if 0 <= j <= n-1-m: contribute (1-x[j+m])
        #          if m <= j <= n-1: contribute -x[j-m]
        idx1 = np.arange(0, n - m)
        grad[idx1] += dx * (1.0 - x[idx1 + m])
        idx2 = np.arange(m, n)
        grad[idx2] -= dx * x[idx2 - m]
    return grad

    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        if p[idx] < 1e-13:
            continue
        lag = top_idx[idx] - (n - 1)
        grad += p[idx] * _lag_grad_vec(x, n, lag)
    return obj, grad

def _slsqp(h, n, alpha, obj_fn, maxiter=4000, ftol=1e-17):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for ss in step_sizes:
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                i = int(np.argsort(np.abs(h - 0.5))[0])
                if h[i] < 0.5:
                    j = int(np.argmin(h))
                else:
                    j = int(np.argmax(h))
                if i == j:
                    break
                d = min(ss, abs(h[i] - 0.5) * 0.4)
                if h[i] < 0.5:
                    h[i] += d; h[j] -= d
                else:
                    h[i] -= d; h[j] += d
            else:
                grad = _lag_grad_vec(h, n, lag)
                i = int(np.argmax(grad))
                j = int(np.argmin(grad))
                if i == j or grad[i] - grad[j] < 1e-12:
                    break
                d = min(ss, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                if d < 1e-10:
                    break
                h[i] -= d; h[j] += d
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new >= best_c - 1e-16:
                h = project_feasible(h, n)
                break
            best_c = c_new
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    n = n_init
    best_h = project_feasible(h_init.copy(), n) if h_init is not None else project_feasible(np.full(300, 0.5), 300)
    if h_init is None:
        n = 300
    best_c = compute_c5(best_h, n)
    alphas_phase1 = [500, 2000, 10000, 50000]
    for alpha in alphas_phase1:
        if time.time() > deadline - 80:
            break
        h_new = _slsqp(best_h, n, alpha, _full_obj_grad, maxiter=5000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 50:
        for top_m in [3, 5, 7]:
            if time.time() > deadline - 30:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_m:]
            h_new = _slsqp(best_h, n, 200000, _active_obj_grad, maxiter=4000, ftol=1e-17)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 40:
        for alpha in [200000, 500000]:
            if time.time() > deadline - 20:
                break
            h_new = _slsqp(best_h, n, alpha, _full_obj_grad, maxiter=6000, ftol=1e-18)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 15:
        for trial in range(3):
            if time.time() > deadline - 10:
                break
            scale = 0.008 * (1.0 + 0.5 * trial)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h_new = _slsqp(hp, n, 50000, _full_obj_grad, maxiter=3000, ftol=1e-17)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, deadline - 0.5)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
