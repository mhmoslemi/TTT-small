```python
    """Compute gradient matrix G[i,s] = (1-h[i+s]) - h[i-s]"""
    n = len(h)
    h_ext = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])  # h extended with zeros
    # G[i, s] for shift s = -(n-1), ..., (n-1), stored as index s+(n-1)

    # h[i-s] = h_ext[i-s + (n-1)]

    # For each i, we need h[i+s] for all s, which is h_ext[i + shifts + (n-1)]
    # This is a 2D indexing: rows i, columns shifts

    s_idx = shifts[None, :]  # (1, 2n-1)
    # h[i+s] = h_ext[i+s+(n-1)]

    # h[i-s] = h_ext[i-s+(n-1)]
    idx_hm = i_idx - s_idx + (n-1)  # (n, 2n-1)
    h_plus = h_ext[idx_hp]  # (n, 2n-1)
    h_minus = h_ext[idx_hm]  # (n, 2n-1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5_max(h):
    return float(np.max(c5_vec(h)))

def project(h):
    n = len(h)
    tgt = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-13:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-13:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def make_G(h):
    n = len(h)
    he = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    sh = np.arange(-(n - 1), n)
    ii = np.arange(n)[:, None]
    si = sh[None, :]
    return (1.0 - he[ii + si + n - 1]) - he[ii - si + n - 1]

def G_row(h, i):
    n = len(h)
    he = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    sh = np.arange(-(n - 1), n)
    return (1.0 - he[i + sh + n - 1]) - he[i - sh + n - 1]

def sobj(x, n, alpha, mu):
    c = c5_vec(x)
    cs = alpha * (c - c.max())
    m = cs.max()
    val = c.max() + (m + np.log(np.sum(np.exp(cs - m)))) / alpha
    val += mu * (x.sum() - n * 0.5) ** 2
    return float(val)

def sgrad(x, n, alpha, mu):
    c = c5_vec(x)
    cs = alpha * (c - c.max())
    m = cs.max()
    w = np.exp(cs - m)
    w /= w.sum()
    G = make_G(x)
    g = (2.0 / n) * (G @ w)
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def lbfgs_refine(h, deadline, alphas=None, mu=10.0):

        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    n = len(h)
    best_c = c5_max(h)
    bnds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline - 5:
            break
        if deadline - time.time() - 5 < 8:
            break
        try:
            r = minimize(sobj, h, jac=sgrad, args=(n, a, mu),
                        method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': 400, 'ftol': 1e-18,
                                 'gtol': 1e-16, 'maxfun': 12000})
            hn = project(r.x)
            cn = c5_max(hn)
            if cn < best_c:
                best_c = cn
                h = hn
        except Exception:
            pass
    return h, best_c

def full_sweep(h, c, G, delta, deadline):
```
