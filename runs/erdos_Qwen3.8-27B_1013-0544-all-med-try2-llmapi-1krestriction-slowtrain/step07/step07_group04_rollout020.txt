def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_smooth(h, n, alpha, penalty, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        viol = x.sum() - n / 2.0
        return val + penalty * viol * viol

                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})

        return project_feasible(res.x, n)

    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    sz = 2 * n - 1
    rng = np.random.RandomState(seed)
    cool = (T_end / T_start) ** (1.0 / max(n_iter, 1))
    T = T_start
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cool
        i = rng.randint(n)
        while h[i] != 0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h[j] != 1:
            j = rng.randint(n)
        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range
        delta = np.zeros(sz)
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < sz:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < sz:
            delta[k2] += 1.0
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        de = new_c - cur_c
        if de < 0 or rng.random() < np.exp(-de / max(T, 1e-12)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=5000, seed=42):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    sz = 2 * n - 1
    rng = np.random.RandomState(seed)
    for restart in range(5):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(max(2, n // 50)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for i in zeros:
                if time.time() > deadline:
                    break
                best_j = -1
                best_nc = cur_c
                best_delta = None
                for j in ones:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(sz)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < sz:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < sz:
                        delta[k2] += 1.0
                    nc = (corr + delta).max() * dx
                    if nc < best_nc - 1e-15:
                        best_nc = nc
                        best_j = j
                        best_delta = delta
                if best_j >= 0:
                    h[i] = 1.0
                    h[best_j] = 0.0
                    corr = corr + best_delta
                    cur_c = best_nc
                    improved = True
                    break
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            if not improved:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B (12%)
    p1_end = t0 + 0.12 * total
    h = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > p1_end - 15:
            break
        h = lbfgs_smooth(h, n, alpha, 50.0 * alpha, 2000, p1_end - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: SA (45%)
    p2_end = t0 + 0.55 * total
    sa_configs = [
        (0.06, 0.00005, 500000, 0),
        (0.04, 0.0001, 400000, 1000),
        (0.08, 0.00003, 400000, 2000),
        (0.03, 0.0002, 300000, 3000),
        (0.05, 0.00005, 400000, 4000),
        (0.07, 0.00002, 300000, 5000),
        (0.02, 0.0001, 300000, 6000),
    ]
    for T_s, T_e, n_it, s_off in sa_configs:
        if time.time() > p2_end - 30:
            break
        h_sa, c_sa = sa_run(best_h, n, p2_end - 20,
                           seed=seed + s_off, n_iter=n_it, T_start=T_s, T_end=T_e)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
    # Phase 3: Greedy binary (25%)
    p3_end = t0 + 0.80 * total
    h_g, c_g = greedy_binary(best_h, n, p3_end, max_sweeps=8000, seed=seed + 7000)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g.copy()
    # Phase 4: L-BFGS polish (8%)
    p4_end = t0 + 0.88 * total
    if time.time() < p4_end - 10:
        h_pol = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [100000, 1000000, 5000000]:
            if time.time() > p4_end - 5:
                break
            h_pol = lbfgs_smooth(h_pol, n, alpha, 100.0 * alpha, 1500, p4_end - 5)
            c = compute_c5(h_pol, n)
            if c < best_c:
                best_c = c
                best_h = h_pol.copy()
    # Phase 5: Final SA + greedy (5%)
    if time.time() < dl - 25:
        h_sa2, c_sa2 = sa_run(best_h, n, dl - 15, seed=seed + 9000,
                             n_iter=200000, T_start=0.015, T_end=0.00003)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2.copy()
    if time.time() < dl - 10:
        h_g2, c_g2 = greedy_binary(best_h, n, dl - 5, max_sweeps=2000, seed=seed + 10000)
        if c_g2 < best_c:
            best_c = c_g2
            best_h = h_g2.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
