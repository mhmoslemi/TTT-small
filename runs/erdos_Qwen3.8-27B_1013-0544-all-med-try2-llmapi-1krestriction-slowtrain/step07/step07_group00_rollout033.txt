def compute_c5(h, n):

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n):

    c = compute_c5_vec(h, n)

def pair_polish(h, n, deadline, radius=60):

        while any_improved and time.time() < deadline:

            for i in range(n):

                    return h, best_c

                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue

                        if c < best_c - 1e-15:

    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full'))) * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _wgrad(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def _lsm_grad(h, n, a):
    c = compute_c5_vec(h, n)
    cs = a * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    return _wgrad(h, n, w)

def _lsm_obj(x, n, a, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    cs = a * (c - m)
    mx = cs.max()
    v = m + (mx + np.log(np.sum(np.exp(cs - mx)))) / a
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def _lsm_jac(x, n, a, mu):
    g = _lsm_grad(x, n, a)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _ps_obj(x, n, p, mu):
    c = np.maximum(compute_c5_vec(x, n), 1e-15)
    v = (1.0 / p) * np.sum(c ** p) ** (1.0 / p)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def _ps_jac(x, n, p, mu):
    c = np.maximum(compute_c5_vec(x, n), 1e-15)
    w = c ** (p - 1)
    w /= w.sum()
    S = np.sum(c ** p)
    scale = (1.0 / p) * S ** (1.0 / p - 1.0)
    g = scale * _wgrad(x, n, w)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def opt_lbfgs(h, n, alpha, mu, deadline, maxit=300):
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(_lsm_obj, h, jac=_lsm_jac, args=(n, alpha, mu),
                     method='L-BFGS-B', bounds=b,
                     options={'maxiter': maxit, 'ftol': 1e-17, 'gtol': 1e-14,
                              'maxfun': maxit * 40, 'maxls': 50})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def opt_psum(h, n, p, mu, deadline, maxit=200):
    b = [(0.0, 1.0)] * n
    try:

                     method='L-BFGS-B', bounds=b,

                              'maxfun': maxit * 40, 'maxls': 50})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def opt_slsqp(h, n, deadline, alpha=500000, maxit=3000):
    b = [(0.0, 1.0)] * n
    def obj(x, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        cs = a * (c - m)
        mx = cs.max()
        return float(m + (mx + np.log(np.sum(np.exp(cs - mx)))) / a)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _wgrad(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def _lsm_grad(h, n, a):
    c = compute_c5_vec(h, n)
    cs = a * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    return _wgrad(h, n, w)

def _lsm_obj(x, n, a, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    cs = a * (c - m)
    mx = cs.max()
    v = m + (mx + np.log(np.sum(np.exp(cs - mx)))) / a
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def _lsm_jac(x, n, a, mu):
    g = _lsm_grad(x, n, a)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _ps_obj(x, n, p, mu):
    c = np.maximum(compute_c5_vec(x, n), 1e-15)
    v = (1.0 / p) * np.sum(c ** p) ** (1.0 / p)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def _ps_jac(x, n, p, mu):
    c = np.maximum(compute_c5_vec(x, n), 1e-15)
    w = c ** (p - 1)
    w /= w.sum()
    S = np.sum(c ** p)
    scale = (1.0 / p) * S ** (1.0 / p - 1.0)
    g = scale * _wgrad(x, n, w)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, a):
    c = compute_c5_vec(x, n)
    m = c.max()
    cs = a * (c - m)
    mx = cs.max()
    return float(m + (mx + np.log(np.sum(np.exp(cs - mx)))) / a)

def _slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def opt_lbfgs(h, n, alpha, mu, deadline, maxit=250):
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(_lsm_obj, h, jac=_lsm_jac, args=(n, alpha, mu),
                     method='L-BFGS-B', bounds=b,
                     options={'maxiter': maxit, 'ftol': 1e-17, 'gtol': 1e-14,
                              'maxfun': maxit * 40, 'maxls': 50})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def opt_psum(h, n, p, mu, deadline, maxit=200):
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(_ps_obj, h, jac=_ps_jac, args=(n, p, mu),
                     method='L-BFGS-B', bounds=b,
                     options={'maxiter': maxit, 'ftol': 1e-17, 'gtol': 1e-14,
                              'maxfun': maxit * 40, 'maxls': 50})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def opt_slsqp(h, n, deadline, alpha=500000, maxit=3000):
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'args': (n,)}]
    try:

                     bounds=b, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgrad_opt(h, n, deadline, n_act=15):
    best, bc = h.copy(), compute_c5(h, n)
    st = 0.003
    for _ in range(30000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        mx = c.max()
        thr = mx - max(1e-6, mx * 0.005)
        act = c >= thr
        if act.sum() == 0:
            act[c.argmax()] = True
        w = np.zeros_like(c)
        w[act] = 1.0 / act.sum()
        g = _wgrad(h, n, w)
        h_new = project_feasible(h - st * g, n)
        cn = compute_c5(h_new, n)

def pair_pol(h, n, deadline, rad=50):
    bc = compute_c5(h, n)
    for d in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]:
        imp = True
        while imp and time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, bc
                lo, hi = max(0, i - rad), min(n, i + rad + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1 and h[j] - d >= 0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15:

                    if h[i] - d >= 0 and h[j] + d <= 1:

                        if c < bc - 1e-15:

                        else:
