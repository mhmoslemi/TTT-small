    dx = 2.0/n
    h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full') * dx
        m = corr.max()
        # Subgradient: from the maximizing lag
        max_lags = np.where(np.isclose(corr, m, atol=1e-10))[0]
        # Average subgradient
        grad = np.zeros(n)
        for k in max_lags:
            lag = k - (n-1)
            # grad[i] += (1-h[i+lag]) - h[i-lag] for valid indices
            for i in range(n):
                g1 = i + lag
                g2 = i - lag
                if 0 <= g1 < n:
                    grad[i] += (1 - h[g1])
                if 0 <= g2 < n:
                    grad[i] -= h[g2]
        grad /= len(max_lags)
        h_new = h - step_size * grad
        h = project_feasible(h_new, n)
    return h

For the projected subgradient:

The challenge is that the subgradient can be noisy (multiple maximizing lags), and the step size needs to be carefully chosen.

- f(h) = max_k c_k + (1/alpha) * log(sum_k exp(alpha * (c_k - max_k)))
- This is smooth, and we can use L-BFGS or even plain gradient descent with line search.

L-BFGS might be much faster than SLSQP for this problem because:
- It doesn't need to solve a QP subproblem at each iteration
- It approximates the Hessian with only O(n) storage
- For a smooth objective, it converges faster

def compute_S(h, p):

    n = len(h)
    r = np.arange(-(n-1), n)

    # h[p-r]: valid where 0 <= p-r < n
    mask2 = (p - r >= 0) & (p - r < n)
    S[mask2] += h[p - r[mask2]]
    return S

    p2 = t0 + 0.70 * (dl - t0)
    h_cur = best_h.copy()
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > p2 - 8:
            break
        h_cur = lbfgs_smooth(h_cur, n, alpha, maxiter=4000, pgtol=1e-10)
        c = compute_c5(h_cur, n)
        if c < best_c:
            best_c, best_h = c, h_cur.copy()

    # Phase 3: Multi-resolution (15% time)
    p3 = t0 + 0.85 * (dl - t0)
    hr, cr = multi_resolution_optimize(best_h, n, p3, seed=seed)
    if cr < best_c:
        best_c, best_h = cr, hr.copy()

    # Phase 4: Final high-alpha polish + restarts (15% time)
    p4 = dl
    h_cur = best_h.copy()
    for alpha in [2000000, 10000000]:
        if time.time() > p4 - 3:
            break
        h_cur = lbfgs_smooth(h_cur, n, alpha, maxiter=5000, pgtol=1e-11)
        c = compute_c5(h_cur, n)
        if c < best_c:
            best_c, best_h = c, h_cur.copy()

    # Noisy restarts
    for trial in range(4):
        if time.time() > p4 - 1:
            break
        h_r = best_h + 0.008 * rng.randn(n)
        h_r = project_feasible(h_r, n)
        h_r = lbfgs_smooth(h_r, n, 500000, maxiter=2000, pgtol=1e-10)
        h_r = lbfgs_smooth(h_r, n, 5000000, maxiter=2000, pgtol=1e-11)
        c_r = compute_c5(h_r, n)
        if c_r < best_c:
            best_c, best_h = c_r, h_r.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
