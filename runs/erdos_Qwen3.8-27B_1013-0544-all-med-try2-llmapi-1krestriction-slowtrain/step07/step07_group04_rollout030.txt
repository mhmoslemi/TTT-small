import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    # w has length 2n-1, indexed by m = 0..2n-2
    # c[n-1+l] corresponds to lag l
    # gradient of c[n-1+l] w.r.t. h[i] = (1-h[i+l]) - h[i-l] (when valid)
    # Total grad[i] = sum_l w[n-1+l] * ((1-h[i+l]) - h[i-l])

        if wl < 1e-15:
            continue

            # i+l in [0,n-1] means i in [0, n-1-l]

            grad[:n-ll] -= wl * h[ll:]

def binary_search_v2(h, n, deadline, max_sweeps=10000, seed=42):
    target = n // 2
    dx = 2.0 / n
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_nc = cur_c
        best_i = -1
        best_j = -1
        
        for i in zeros:
            im = i + m_range
            imn = i - m_range
            msk_i = (im >= 0) & (im < n)
            msk_in = (imn >= 0) & (imn < n)
            delta_i = np.zeros(2*n-1)
            delta_i[msk_i] += 1.0 - h[im[msk_i]]
            delta_i[msk_in] -= h[imn[msk_in]]
            
            for j in ones:
                jm = j + m_range
                jmn = j - m_range
                msk_j = (jm >= 0) & (jm < n)
                msk_jn = (jmn >= 0) & (jmn < n)
                
                # Quick upper bound check
                # The max of (corr + delta_i + delta_j + corrections) 
                # is at most max(corr + delta_i) + max(|delta_j|) + 2
                # But this is too loose. Let's just compute.
                
                nc_approx = (corr + delta_i).max() * dx  # Upper bound (ignoring delta_j negative parts)
                if nc_approx < best_nc - 1e-10:
                    continue  # Can't improve (this is actually a lower bound on new max... no)
                # Actually (corr + delta_i).max() is NOT an upper bound on (corr + delta_i + delta_j + corr).max()
                # because delta_j could be positive somewhere.
                # Let me skip this optimization.
                
                delta_j = np.zeros(2*n-1)
                delta_j[msk_j] -= (1.0 - h[jm[msk_j]])
                delta_j[msk_jn] += h[jmn[msk_jn]]
                
                new_val = corr + delta_i + delta_j
                new_val[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                new_val[k1] += 1.0
                k2 = n - 1 + (i - j)
                new_val[k2] += 1.0
                
                nc = new_val.max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_i = i
                    best_j = j

            h[best_j] = 0.0
            corr = np.correlate(h, 1-h, mode='full').astype(float)

                best_h = h.copy()

            if rng.random() < 0.3:
                a = rng.choice(zeros)
                b = rng.choice(ones)
                h[a], h[b] = 1.0, 0.0
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur_c = corr.max() * dx
            else:
                break
    
    return best_h, best_c

        if nz == 0 or no == 0:

        iz = zeros[:, None].astype(float)
        lags = m_range[None, :].astype(float)

        delta_i[msk2] -= h[im_[msk2]]

        jp = (ones[:, None] + m_range[None, :])
        jm_ = (ones[:, None] - m_range[None, :])

        msk3 = (jp >= 0) & (jp < n)
        delta_j[msk3] -= (1.0 - h[jp[msk3]])
        msk4 = (jm_ >= 0) & (jm_ < n)
        delta_j[msk4] += h[jm_[msk4]]
        
        # Full delta: delta[i,j] = delta_i[i] + delta_j[j] + correction
        # Base (without corrections): corr + delta_i[i] + delta_j[j]
        # Shape: (nz, no, 2n-1)
        base = corr[None, None, :] + delta_i[:, None, :] + delta_j[None, :, :]
        
        # Apply corrections
        base[:, :, n-1] -= 2.0
        d_mat = (ones[None, :] - zeros[:, None])  # (nz, no)
        pos1 = (n - 1 + d_mat)  # (nz, no), values in [0, 2n-2]
        pos2 = (n - 1 - d_mat)  # (nz, no), values in [0, 2n-2]
        
        # Scatter add: base[i, j, pos1[i,j]] += 1
        # Use np.add.at (slow but correct)
        rows = np.repeat(np.arange(nz), no)
        cols = np.tile(np.arange(no), nz)
        flat = base.reshape(nz*no, 2*n-1)
        np.add.at(flat, (rows, pos1.reshape(nz*no)), 1.0)
        np.add.at(flat, (rows, pos2.reshape(nz*no)), 1.0)
        base = flat.reshape(nz, no, 2*n-1)
        
        final_max = base.max(axis=2)  # (nz, no)
        flat_idx = np.unravel_index(np.argmin(final_max), final_max.shape)
        nc = final_max[flat_idx] * dx
        
        if nc < cur_c - 1e-14:
            h[zeros[flat_idx[0]]] = 1.0
            h[ones[flat_idx[1]]] = 0.0
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # No improvement - try random kick
            if rng.random() < 0.2:
                a = rng.randint(nz)
                b = rng.randint(no)
                h[zeros[a]] = 1.0
                h[ones[b]] = 0.0
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur_c = corr.max() * dx
            else:
                break
    
    return best_h, best_c

```python
def smooth_optimize(h, n, alpha, beta, deadline, maxiter=500):
    def fun_and_jac(x):
        dx = 2.0 / n
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        s = x.sum() - n/2.0
        val += beta * s * s
        
        w = np.exp(sh - sm)
        w /= w.sum()
        grad = np.zeros(n)
        for l in range(-(n-1), n):
            wl = w[n-1+l]
            if wl < 1e-12: continue
            if l >= 0:
                if l < n:
                    grad[:n-l] += wl * (1.0 - x[l:])
                if l > 0:
                    grad[l:] -= wl * x[:n-l]
            else:
                ll = -l
                grad[ll:] += wl * (1.0 - x[:n-ll])
                grad[:n-ll] -= wl * x[ll:]
        grad += 2.0 * beta * s
        return val, grad
    
    x0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_and_jac, x0, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except:
        return h
```

The gradient computation is O(n²) per call (loop over 2n-1 lags, each O(n)). For n=250, that's 250*499 ≈ 125K ops per gradient eval. L-BFGS-B typically needs ~100-500 iterations, so total is ~12.5M-62.5M ops. Should take < 1s.

Actually wait, the gradient loop is in Python (looping over 499 lags). Each iteration does a few numpy ops on arrays of size ~250. So it's 499 * 3 * ~5μs ≈ 7.5ms per gradient eval. With 500 iterations, that's ~4s. Acceptable.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[n-1+l]
        if wl < 1e-12: continue
        if l >= 0:
            if l < n:
                grad[:n-l] += wl * (1.0 - x[l:])
            if l > 0:
                grad[l:] -= wl * x[:n-l]
        else:
            ll = -l
            grad[ll:] += wl * (1.0 - x[:n-ll])
            grad[:n-ll] -= wl * x[ll:]
    return grad

def smooth_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def lbfgs_smooth(h, n, alpha, beta, deadline, maxiter=500):
    def fun_jac(x):
        v = smooth_val(x, n, alpha)
        s = x.sum() - n / 2.0
        v += beta * s * s
        g = smooth_grad(x, n, alpha)
        g = g + 2.0 * beta * s
        return v, g
    x0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_jac, x0, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h
