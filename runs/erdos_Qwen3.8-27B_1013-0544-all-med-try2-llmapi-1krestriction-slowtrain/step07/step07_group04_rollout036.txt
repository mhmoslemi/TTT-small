Actually, let me be more careful with the indexing.

c[s] = Σ_n h[n+s] * (1-h[n]) for s in [-(n-1), n-1]

∂c[s]/∂h[i] = (1-h[i-s]) - h[i+s] (with boundary conditions)

So ∇f[i] = Σ_s w[s] * ((1-h[i-s]) - h[i+s])

The first term: Σ_s w[s] * (1-h[i-s]) = Σ_s w[s] - Σ_s w[s] * h[i-s] = 1 - (h ⊛ w)[i]
where (h ⊛ w)[i] = Σ_s h[i-s] * w[s] is the cross-correlation.

The second term: Σ_s w[s] * h[i+s] = Σ_s w[-s] * h[i-(-s)] = (h ⊛ w_flipped)[i]
where w_flipped[s] = w[-s].

So ∇f[i] = 1 - (h ⊛ w)[i] - (h ⊛ w_flipped)[i]

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def compute_A_all(h, indices):
    """For each index i in indices, compute A_i[s] = h[i-s]+h[i+s] for all s."""
    n = len(h)
    A = np.zeros((len(indices), 2*n-1))
    s = np.arange(-(n-1), n)
    for p, i in enumerate(indices):
        idx1 = i - s
        idx2 = i + s
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        A[p, valid1] += h[idx1[valid1]]
        A[p, valid2] += h[idx2[valid2]]
    return A

def binary_local_search(h, n, deadline, max_sweeps=5000, cooling=0.95, T_start=0.5, T_min=0.01, rng=None):
    """Efficient binary local search with simulated annealing."""
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    target = n // 2
    h = to_binary(h, n)
    
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    # Precompute A for zeros and B for ones
    A_zero = compute_A_all(h, zeros_list)  # (z, 2n-1)
    B_one = compute_A_all(h, ones_list)   # (o, 2n-1)
    
    T = T_start
    sweeps_without_improvement = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if not zeros_list or not ones_list:
            break
        
        z = len(zeros_list)
        o = len(ones_list)
        
        # Find best swap
        best_new_max = cur_max
        best_pi = -1  # index in zeros_list
        best_pj = -1  # index in ones_list
        
        # Vectorized: for each i in zeros, compute new max for all j in ones
        # new_c[i,j] = c + B_one[j] - A_zero[i]
        # We want argmin_{i,j} max_s new_c[i,j,s]
        
        # Process in chunks to save memory
        chunk_size = max(1, z)  # process all at once if z is small
        for i_start in range(0, z, chunk_size):
            i_end = min(i_start + chunk_size, z)
            A_chunk = A_zero[i_start:i_end]  # (chunk, 2n-1)
            # new_max[i_local, j] = max_s (c - A_chunk[i_local] + B_one[j])
            # = max_s (B_one[j] + (c - A_chunk[i_local]))
            base = c[None, :] - A_chunk  # (chunk, 2n-1)
            new_maxs = (B_one[None, :, :] + base[:, None, :]).max(axis=2)  # (chunk, o)
            # Find best in this chunk
            min_val, flat_idx = new_maxs.reshape(-1).min(), new_maxs.reshape(-1).argmin()
            if min_val < best_new_max:
                best_new_max = min_val
                best_pi = i_start + flat_idx // o
                best_pj = flat_idx % o
        
        # Accept or reject
        delta = best_new_max - cur_max
        if delta < -1e-12 or (T > T_min and rng.random() < np.exp(-max(0, -delta) / T)):
            # Perform swap
            if best_pi >= 0:
                i = zeros_list[best_pi]
                j = ones_list[best_pj]
                h[i], h[j] = 1.0, 0.0
                c = c + B_one[best_pj] - A_zero[best_pi]
                cur_max = c.max()
                
                # Update lists
                zeros_list.pop(best_pi)
                ones_list.pop(best_pj)
                zeros_list.append(j)
                ones_list.append(i)
                
                # Update A_zero and B_one
                # A_zero: remove row best_pi, add new row for j
                if best_pi < len(A_zero) - 1:
                    A_zero = np.delete(A_zero, best_pi, axis=0)
                else:
                    A_zero = A_zero[:-1]
                A_zero = np.vstack([A_zero, compute_A_all(h, [j])])
                
                # B_one: remove row best_pj, add new row for i
                if best_pj < len(B_one) - 1:
                    B_one = np.delete(B_one, best_pj, axis=0)
                else:
                    B_one = B_one[:-1]
                B_one = np.vstack([B_one, compute_A_all(h, [i])])
                
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
                    sweeps_without_improvement = 0
                else:
                    sweeps_without_improvement += 1
            else:
                # Random swap for exploration
                pi = rng.randint(z)
                pj = rng.randint(o)
                i = zeros_list[pi]
                j = ones_list[pj]
                h[i], h[j] = 1.0, 0.0
                c = c + B_one[pj] - A_zero[pi]
                cur_max = c.max()
                zeros_list.pop(pi)
                ones_list.pop(pj)
                zeros_list.append(j)
                ones_list.append(i)
                if pi < len(A_zero) - 1:
                    A_zero = np.delete(A_zero, pi, axis=0)
                else:
                    A_zero = A_zero[:-1]
                A_zero = np.vstack([A_zero, compute_A_all(h, [j])])
                if pj < len(B_one) - 1:
                    B_one = np.delete(B_one, pj, axis=0)
                else:
                    B_one = B_one[:-1]
                B_one = np.vstack([B_one, compute_A_all(h, [i])])
                sweeps_without_improvement += 1
        else:

        # Cool down

            T = max(T_min, T * cooling)
            sweeps_without_improvement = 0
        
        if T <= T_min and sweeps_without_improvement > 50:
            break
    
    return best_h, best_c * dx

def fft_correlate(a, b):
    """Cross-correlation using FFT. a ⊛ b = ifft(fft(a) * conj(fft(b)))"""
    n = len(a)
    size = 1
    while size < 2 * n:

    fa = np.fft.fft(a, size)
    fb = np.fft.fft(b, size)
    return np.real(np.fft.ifft(fa * np.conj(fb)))[:n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def compute_A_all(h, indices):
    n = len(h)
    A = np.zeros((len(indices), 2 * n - 1))
    s = np.arange(-(n - 1), n)
    for p, i in enumerate(indices):
        idx1 = i - s
        idx2 = i + s
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        A[p, v1] += h[idx1[v1]]
        A[p, v2] += h[idx2[v2]]
    return A

def binary_local_search(h, n, deadline, max_sweeps=20000, T_start=0.3, T_min=0.005, cooling=0.97, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    h = to_binary(h, n)
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    A_zero = compute_A_all(h, zeros_list)
    B_one = compute_A_all(h, ones_list)
    T = T_start
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        z = len(zeros_list)
        o = len(ones_list)
        if z == 0 or o == 0:
            break
        best_new_max = cur_max + 1.0
        best_pi = -1
        best_pj = -1
        A_chunk = A_zero
        base = c[None, :] - A_chunk
        new_maxs = (B_one[None, :, :] + base[:, None, :]).max(axis=2)
        flat_min_idx = new_maxs.argmin()
        best_new_max = new_maxs.reshape(-1)[flat_min_idx]
        best_pi = flat_min_idx // o
        best_pj = flat_min_idx % o
        delta = best_new_max - cur_max
        accept = delta < -1e-12
        if not accept and T > T_min:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if not accept:
            pi = rng.randint(z)
            pj = rng.randint(o)
            best_pi, best_pj = pi, pj
            best_new_max = (c + B_one[pj] - A_zero[pi]).max()
        i = zeros_list[best_pi]
        j = ones_list[best_pj]
        h[i], h[j] = 1.0, 0.0
        c = c + B_one[best_pj] - A_zero[best_pi]
        cur_max = c.max()
        zeros_list.pop(best_pi)
        ones_list.pop(best_pj)
        zeros_list.append(j)
        ones_list.append(i)
        if best_pi < len(A_zero) - 1:
            A_zero = np.delete(A_zero, best_pi, axis=0)
        else:
            A_zero = A_zero[:-1]
        A_zero = np.vstack([A_zero, compute_A_all(h, [j])])
        if best_pj < len(B_one) - 1:
            B_one = np.delete(B_one, best_pj, axis=0)
        else:
            B_one = B_one[:-1]
        B_one = np.vstack([B_one, compute_A_all(h, [i])])
        if cur_max < best_c - 1e-14:
            best_c = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            T = max(T_min, T * cooling)
            stall = 0
        if T <= T_min and stall > 200:
            break
    return best_h, best_c * dx

def continuous_grad_opt(h, n, deadline, alpha=2000, n_steps=5000):
    dx = 2.0 / n
    h = project_feasible(h, n)
    s = np.arange(-(n - 1), n)
    offset = 3 * n
    h_pad = np.zeros(4 * n)
    i_range = np.arange(n)
    pos1 = (offset + i_range[:, None] - s[None, :]).astype(int)
    pos2 = (offset + i_range[:, None] + s[None, :]).astype(int)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr0 = 0.5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        h_pad[offset:offset + n] = h
        c = np.correlate(h, 1 - h, mode='full')
        shifted = alpha * (c - c.max())
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        g1 = (w[None, :] * h_pad[pos1]).sum(axis=1)
        g2 = (w[None, :] * h_pad[pos2]).sum(axis=1)
        grad = 1.0 - g1 - g2
        lr = lr0 / (1 + step * 0.02)
        h = project_feasible(h - lr * grad, n)
        c_val = c.max() * dx
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        if cb2 < best_c:

    # Phase 3: Aggressive binary search with annealing

            if cb < best_c:

                best_h = hb.copy()
    best_h = project_feasible(best_h, n)

    return best_h, best_c, n
