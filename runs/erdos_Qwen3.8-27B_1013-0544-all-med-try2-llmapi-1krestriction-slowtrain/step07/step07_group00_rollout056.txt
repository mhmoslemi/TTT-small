def overlap_gradient_at_shift(h, n, k):
    """Gradient of overlap at shift k (where k = m - (n-1) for correlate index m)."""

    if k >= 0:
        # j-k >= 0 means j >= k, j-k < n means j < n+k (always true for j<n)
        j_range = range(k, n)
        grad[j_range] += (1.0 - h[:n-k])
        # Second term: -h[j+k] for j where 0<=j+k<n
        # j < n-k
        j_range2 = range(0, n-k)
        grad[j_range2] -= h[k:]
    else:

        # j-k < n means j < n+k = n-kk
        j_range = range(0, n-kk)

        # Second term: -h[j+k] = -h[j-kk] for j where 0<=j-kk<n
        # j >= kk
        j_range2 = range(kk, n)
        grad[j_range2] -= h[:n-kk]
    return grad

def focused_gradient_descent(h, n, deadline, max_step=0.1):

    best_h = h.copy()

        overlaps = compute_overlaps(h, n)
        c_curr = overlaps.max()
        
        # Identify active set
        threshold = c_curr - 0.01  # within 0.01 of max
        active = np.where(overlaps >= threshold)[0]
        
        # Compute weighted gradient
        weights = np.exp((overlaps[active] - c_curr) * 50.0)

            grad += weights[idx] * overlap_gradient_at_shift(h, n, k)
        
        # Project gradient to feasible direction (orthogonal to constraint normal)
        # The constraint is sum(h) = n/2, so the normal is (1,1,...,1)

        for _ in range(20):

            c_new = compute_c5(h_new, n)
            if c_new < c_curr - 1e-15:

                c_curr = c_new
                break
            alpha *= 0.5
        
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
        
        # Check convergence
        if alpha < 1e-10:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        m = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if m.any():
            hp[m] += diff / m.sum()
    return np.clip(hp, 0.0, 1.0)

def wgradient(h, n, weights):
    """Weighted sum of overlap gradients. weights is array of length 2n-1."""
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            kk = -k
            grad[:n - kk] += w * (1.0 - h[kk:])
            grad[kk:] -= w * h[:n - kk]
    return grad

def active_weights(overlaps, c_max, temp=0.005):
    """Compute softmax weights over overlaps."""
    shifted = (overlaps - c_max) / temp
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    s = w.sum()
    if s > 1e-30:
        w /= s
    else:
        idx = np.argmax(overlaps)
        w = np.zeros_like(overlaps)
        w[idx] = 1.0
    return w

def focused_descent(h, n, deadline, temp_init=0.01, temp_min=0.001):
    """Active-set focused gradient descent with adaptive temperature."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.05
    temp = temp_init
    
    while time.time() < deadline:
        overlaps = compute_overlaps(h, n)
        c = overlaps.max()
        w = active_weights(overlaps, c, temp)
        grad = wgradient(h, n, w)
        grad -= grad.mean()  # project onto sum=0 subspace
        
        # Line search
        improved = False
        a = alpha
        for _ in range(30):
            h_new = project_feasible(h - a * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c - 1e-16:
                h = h_new
                c = c_new
                improved = True
                break
            a *= 0.7
        if not improved:
            alpha *= 0.5
            temp = max(temp_min, temp * 0.95)
            if alpha < 1e-12:
                break
        else:
            alpha = min(alpha * 1.2, 0.2)
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def lbfgs_run(h, n, deadline, alpha_smooth=100000):
    """L-BFGS-B with smooth max objective."""
    def obj(x):
        c = compute_overlaps(x, n)
        m = c.max()
        sh = alpha_smooth * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha_smooth
    
    def grad(x):
        c = compute_overlaps(x, n)
        m = c.max()
        sh = alpha_smooth * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        g = wgradient(x, n, w)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 500, 'ftol': 1e-18, 'maxfun': 15000})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def combinatorial_moves(h, n, deadline, radius=30):
    """Try pair moves to improve C5."""
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0001]
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        h_i_new = h[i] + sign * d
                        h_j_new = h[j] - sign * d
                        if 0 <= h_i_new <= 1 and 0 <= h_j_new <= 1:
                            h[i] = h_i_new
                            h[j] = h_j_new
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                any_imp = True
                            else:
                                h[i] = h[i] - sign * d
                                h[j] = h[j] + sign * d
    return h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    # Remove duplicates
    positions = np.unique(positions)
    # If we lost some, add more
    if len(positions) < half:
        used = set(positions)
        for i in range(n):
            if i not in used and len(positions) < half:
                positions = np.append(positions, i)
                used.add(i)
    h[positions[:half]] = 1.0
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        m = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if m.any():
            hp[m] += diff / m.sum()
    return np.clip(hp, 0.0, 1.0)

def wgradient(h, n, weights):
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            kk = -k
            if kk < n:
                grad[:n - kk] += w * (1.0 - h[kk:])
                grad[kk:] -= w * h[:n - kk]
    return grad

def active_weights(overlaps, c_max, temp):
    shifted = np.clip((overlaps - c_max) / temp, -40.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        idx = int(np.argmax(overlaps))
        w = np.zeros_like(overlaps)
        w[idx] = 1.0
    else:
        w /= s
    return w

def focused_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.08
    temp = 0.008
    stall = 0
    while time.time() < deadline:
        overlaps = compute_overlaps(h, n)
        c = float(overlaps.max())
        w = active_weights(overlaps, c, temp)
        grad = wgradient(h, n, w)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        improved = False
        a = alpha
        for _ in range(25):
            h_new = project_feasible(h - a * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c - 1e-17:
                h = h_new
                c = c_new
                improved = True
                break
            a *= 0.6
        if improved:
            alpha = min(alpha * 1.15, 0.25)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            alpha *= 0.6
            temp = max(0.0005, temp * 0.92)
            if stall > 30 or alpha < 1e-10:
                break
    return best_h, best_c

def lbfgs_run(h, n, deadline, asmooth=80000):
    def obj(x):
        c = compute_overlaps(x, n)
        m = float(c.max())
        sh = asmooth * (c - m)
        mx = float(sh.max())
        return m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / asmooth

    def grad(x):
        c = compute_overlaps(x, n)
        m = float(c.max())
        sh = asmooth * (c - m)
        mx = float(sh.max())
        w = np.exp(sh - mx)
        s = w.sum()
        if s > 1e-30:
            w /= s
        return wgradient(x, n, w)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 400, 'ftol': 1e-18, 'maxfun': 12000})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def pair_moves(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    for d in [0.004, 0.002, 0.001, 0.0004]:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    positions = np.unique(positions)
    if len(positions) < half:
        used = set(positions.astype(int))
        extra = []
        for i in range(n):
            if i not in used:
                extra.append(i)
                if len(extra) == half - len(positions):
                    break
        positions = np.concatenate([positions, np.array(extra)])
    h[positions[:half]] = 1.0
    return project_feasible(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:

        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: L-BFGS + focused descent on initial
    t1 = min(t0 + 100, deadline - 500)
    if t1 > t0 + 15:
        h1, c1 = lbfgs_run(h, n, t1, 50000)
        t1b = min(t0 + 160, deadline - 450)
        if t1b > time.time() + 10:
            h1, c1 = focused_descent(h1, n, t1b)
        if c1 < best_c:
            best_c, best_h, best_n = c1, h1, n

    # Phase 2: Higher resolution
    for n_try in [800, 1000, 1200]:
        if time.time() > deadline - 150:
            break
        h_interp = interpolate_to(best_h, best_n, n_try)
        alloc = min(220, max(30, (deadline - time.time() - 50) * 0.5))
        t_end = time.time() + alloc
        t_lb = time.time() + alloc * 0.45
        h_t, c_t = lbfgs_run(h_interp, n_try, t_lb, 120000)
        if time.time() < t_end:
            t_fd = min(time.time() + (t_end - time.time()), t_end)
            h_t, c_t = focused_descent(h_t, n_try, t_fd)
        if c_t < best_c:
            best_c, best_h, best_n = c_t, h_t, n_try

    # Phase 3: Diverse restarts
    starts = [golden_start(best_n)]
    h_p1 = best_h + rng.randn(best_n) * 0.04
    starts.append(project_feasible(h_p1, best_n))
    h_p2 = np.clip(best_h * 0.5 + 0.25, 0.02, 0.98)
    starts.append(project_feasible(h_p2, best_n))

    for s in starts:
        if time.time() > deadline - 50:
            break
        alloc = min(50, max(12, (deadline - time.time() - 15) * 0.4))
        t_a = time.time() + alloc * 0.5
        h_s, c_s = lbfgs_run(s, best_n, t_a, 60000)
        t_b = min(time.time() + alloc, t_a + alloc * 0.5)
        if t_b > time.time() + 5:
            h_s, c_s = focused_descent(h_s, best_n, t_b)
        if c_s < best_c:
            best_c, best_h, best_n = c_s, h_s, best_n

    # Phase 4: Final focused descent
    t4 = min(time.time() + 80, deadline - 8)
    if t4 > time.time() + 5:
        best_h, best_c = focused_descent(best_h, best_n, t4)

    # Phase 5: Pair moves
    if time.time() < deadline - 1:
        best_h, best_c = pair_moves(best_h, best_n, deadline - 1, radius=35)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
