    lo = min(np.minimum(intercepts, intercepts + slopes))

            #   (1-h[j+lag])*dx * I[0<=j+lag<n] - h[j-lag]*dx * I[0<=j-lag<n]
            #
            # intercept: c[m] - v*slope[m]
            
            # Vectorized computation:
            lags = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
            # j+lag in [0, n):
            jp = j + lags  # shape (2n-1,)
            valid1 = (jp >= 0) & (jp < n)
            # j-lag in [0, n):
            jm = j - lags
            valid2 = (jm >= 0) & (jm < n)
            
            slope = np.zeros(2*n-1)
            slope[valid1] = (1.0 - h[jp[valid1]]) * dx
            slope[valid2] -= h[jm[valid2]] * dx
            
            intercept = c - v * slope
            
            # Solve min_t max_k (slope[k]*t + intercept[k]) for t in [0,1]
            t_opt = solve_1d_minimax_vec(slope, intercept)
            
            if abs(t_opt - v) > 1e-15:
                h[j] = t_opt
                any_improved = True
        
        passes += 1
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        if not any_improved and passes > 1:
            break
    
    return h, best_c

def solve_1d_minimax_vec(slopes, intercepts):
    """Vectorized 1D minimax solver."""
    # Binary search on V
    # f(t) = max_k(slopes[k]*t + intercepts[k])
    # At t=0: max(intercepts)
    # At t=1: max(slopes + intercepts)
    f0 = intercepts.max()
    f1 = (slopes + intercepts).max()
    
    # Lower bound on optimal V:
    # For each k, the line goes from intercept[k] at t=0 to (slopes[k]+intercepts[k]) at t=1
    # The minimum of the max must be >= max_k min(intercept[k], slopes[k]+intercept[k])
    lo = np.minimum(intercepts, slopes + intercepts).max()
    hi = max(f0, f1)
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        # Check feasibility: exists t in [0,1] with slopes[k]*t + intercepts[k] <= mid for all k
        pos = slopes > 1e-15
        neg = slopes < -1e-15
        zero = ~(pos | neg)
        
        if zero.any() and intercepts[zero].max() > mid:
            lo = mid
            continue
        
        lower = 0.0
        upper = 1.0
        if pos.any():
            ub = (mid - intercepts[pos]) / slopes[pos]
            upper = min(upper, ub.min())
        if neg.any():
            lb = (mid - intercepts[neg]) / slopes[neg]
            lower = max(lower, lb.max())
        
        if lower <= upper:
            hi = mid
        else:
            lo = mid
    
    # Recover t*
    V = hi
    pos = slopes > 1e-15
    neg = slopes < -1e-15
    lower = 0.0
    upper = 1.0
    if pos.any():
        upper = min(upper, ((V - intercepts[pos]) / slopes[pos]).min())
    if neg.any():
        lower = max(lower, ((V - intercepts[neg]) / slopes[neg]).max())
    
    if lower > upper + 1e-12:
        t = (lower + upper) / 2.0
    else:
        t = (lower + upper) / 2.0
    return float(np.clip(t, 0.0, 1.0))
