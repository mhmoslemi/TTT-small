            if tot < 1e-15:

    dx = 2.0 / n

    shifted = c - m

    if max_sh > 500:
        lse = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    else:
        lse = np.log(np.sum(np.exp(shifted)))
    return float(m + lse / alpha)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = c - m
    mx = shifted.max()
    if mx > 500:
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        lse = np.log(np.sum(np.exp(shifted)))
    return float(m + lse / alpha)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    best = h.copy()
    best_c = compute_c5(h, n)
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        c = compute_c5(res.x, n)
        if c < best_c:
            best = res.x
            best_c = c
    except Exception:
        pass
    return project_feasible(best, n)

        j = rng.randint(n)

            continue
        if cur_h[i] >= cur_h[j]:

        else:

            direction = 1
        if max_d < 1e-6:
            T = max(T * cooling, min_T)
            continue
        d = rng.uniform(0.001, min(move_scale, max_d))
        old_i, old_j = cur_h[i], cur_h[j]

            cur_h[j] += d
        else:
