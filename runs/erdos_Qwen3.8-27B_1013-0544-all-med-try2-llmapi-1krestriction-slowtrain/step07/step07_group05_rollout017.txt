import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    s = exp_sh.sum()
    return m + (mx + np.log(s)) / alpha, exp_sh / s

def grad_objective(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    _, weights = smooth_max(c, alpha)
    grad_h = np.zeros(n)
    for s_full in range(2 * n - 1):
        s = s_full - (n - 1)
        ws = weights[s_full] * dx
        if abs(ws) < 1e-16:
            continue
        j_lo = max(0, s)
        j_hi = min(n - 1, s + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad_h[idx] += ws * (1.0 - h[idx - s])
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad_h[idx2] += ws * (-h[idx2 + s])
    return c, grad_h

def adam_optimize(h, n, deadline, lr=0.005, beta1=0.9, beta2=0.999,
                  alpha_start=100, alpha_end=50000, max_iter=3000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = it / max(1, max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        c, g = grad_objective(h, n, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def pair_search_2p(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    nn = n if radius is None else radius
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    jlo, jhi = 0, n
                else:
                    jlo = max(0, i - nn)
                    jhi = min(n, i + nn + 1)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def slsqp_sls(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def f(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        val, _ = smooth_max(c, alpha)
        return val
    def fgrad(x):
        _, g = grad_objective(x, n, alpha)
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(f, h, jac=fgrad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or compute_c5(res.x, n) < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def multi_resolution(h_coarse, n_coarse, n_fine, deadline):
    x_c = np.linspace(0, 2, n_coarse)
    x_f = np.linspace(0, 2, n_fine)
    h_f = np.interp(x_f, x_c, h_coarse)
    h_f = project_feasible(h_f, n_fine)
    return h_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)

    # Phase 1: Adam optimization at initial resolution
    if time.time() < deadline - 60:
        h1, c1 = adam_optimize(h_init, n_init, deadline - 180,
                               lr=0.003, alpha_start=200, alpha_end=80000,
                               max_iter=2000)
        if c1 < best_c:
            best_c, best_h = c1, h1

    # Phase 2: SLSQP polish
    if time.time() < deadline - 120:
        for a in [5000, 20000, 100000]:
            if time.time() > deadline - 120:
                break
            hs = slsqp_sls(best_h, n_init, a, maxiter=1500)
            cs = compute_c5(hs, n_init)
            if cs < best_c:
                best_c, best_h = cs, hs

    # Phase 3: Pair search
    if time.time() < deadline - 60:
        best_h, best_c = pair_search_2p(best_h, n_init, deadline - 60,
                                        [0.01, 0.005, 0.002, 0.001, 0.0005],
                                        radius=50)

    # Phase 4: Multi-resolution - try different n
    for n_try in [150, 100, 200, 300]:
        if time.time() > deadline - 80:
            break
        if n_try == n_init:
            continue
        alloc = min(60, (deadline - time.time()) * 0.2)
        if alloc < 20:
            break
        h_interp = multi_resolution(best_h, n_init, n_try, time.time())
        h_opt, c_opt = adam_optimize(h_interp, n_try, time.time() + alloc,
                                     lr=0.003, alpha_start=200, alpha_end=50000,
                                     max_iter=1500)
        if time.time() < deadline - 40:
            h_opt = slsqp_sls(h_opt, n_try, 50000, maxiter=1000)
            c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    # Phase 5: Restart with perturbation
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 30) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = adam_optimize(hp, n, time.time() + alloc,
                                         lr=0.003, alpha_start=200,
                                         alpha_end=50000, max_iter=1200)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    # Phase 6: Final pair search + polish
    n = len(best_h)
    if time.time() < deadline - 10:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        if n <= 150:
            best_h, best_c = pair_search_2p(best_h, n, deadline - 5, deltas)
        else:
            best_h, best_c = pair_search_2p(best_h, n, deadline - 5,
                                            deltas[:3], radius=40)

    if time.time() < deadline - 3:
        hs = slsqp_sls(best_h, n, 200000, maxiter=2000)
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_c, best_h = cs, hs

    if time.time() < deadline - 1:
        if n <= 150:
            best_h, best_c = pair_search_2p(best_h, n, deadline - 0.5,
                                            [0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = pair_search_2p(best_h, n, deadline - 0.5,
                                            [0.0002, 0.0001], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
