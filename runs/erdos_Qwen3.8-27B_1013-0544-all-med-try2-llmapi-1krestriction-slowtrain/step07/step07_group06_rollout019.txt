import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_full(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_c5(h, n, alpha):
    c = compute_c5_full(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(100, 1.0), (1000, 5.0), (10000, 20.0),
                      (100000, 100.0), (1000000, 500.0)]:
        if time.time() > deadline - 3:
            break
        def obj(x, a=alpha, mu=mu):
            return smooth_max_val(x, n, a) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x, a=alpha, mu=mu):
            g = grad_c5(x, n, a)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def momentum_descent(h, n, deadline, max_iters=8000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    mu = 0.8
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        alpha = 200.0 * (2000.0 ** frac)
        lr = 0.03 * (1.0 + 0.02 * it) ** (-0.6)
        g = grad_c5(h, n, alpha)
        v = mu * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            v *= 0.3
    return best_h, best_c

def grad_topk(h, n, top_k=5, alpha_top=500.0):
    c = compute_c5_full(h, n)
    top = np.argsort(c)[-top_k:]
    top_vals = c[top]
    sh = alpha_top * (top_vals - top_vals.max())
    ws = np.exp(sh)
    ws /= ws.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, m_idx in zip(ws, top):
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wi * (1.0 - h[k:n])
                grad[k:n] += wi * (-h[0:nv])

            nv = n - kk
            if nv > 0:
                grad[kk:n] += wi * (1.0 - h[0:nv])
                grad[0:nv] += wi * (-h[kk:n])
    return grad * dx

                      (5000, 100.0), (50000, 500.0), (500000, 2000.0)]:

            break

        def gr(x):
            return grad_c5(x, n, a) + 2.0 * mu_ * (x.sum() - n / 2.0)
