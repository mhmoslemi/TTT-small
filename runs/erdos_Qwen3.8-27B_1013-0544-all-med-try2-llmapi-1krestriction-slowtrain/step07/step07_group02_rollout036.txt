from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_indices(n):
    J = np.arange(n)[:, None]
    M = np.arange(n)[None, :]
    idx1 = M - J + (n - 1)
    idx2 = J - M + (n - 1)
    return idx1, idx2

def lse_value(x, n, alpha, idx1, idx2):

    c = np.correlate(x, 1.0 - x, mode='full') * dx

    shifted = np.clip(alpha * (c - m), -700, 700)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def lse_gradient(x, n, alpha, idx1, idx2):
