from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=500):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 1.0), (200, 5.0), (1000, 20.0),
               (5000, 100.0), (20000, 500.0), (100000, 2000.0),
               (500000, 10000.0), (2000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-13:

            if tot < 1e-15:

            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha

    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1

            continue

            grad[k:n] += wm * (-h[0:nv])

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - nm1
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _lbfgs_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 1.0), (200, 5.0), (1000, 20.0),
               (5000, 100.0), (20000, 500.0), (100000, 2000.0),
               (500000, 10000.0), (2000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(partial(_lbfgs_obj, n=n, alpha=alpha, mu=mu), h,
                          jac=partial(_lbfgs_grad, n=n, alpha=alpha, mu=mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, n_golden=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    phi = (5.0 ** 0.5 - 1.0) / 2.0
    target = n / 2.0
    stall = 0
    while time.time() < deadline:
        prev_best = best_c
        for i in range(n):
            if time.time() > deadline:
                break
            a, b = 0.0, 1.0
            c1 = b - phi * (b - a)
            c2 = a + phi * (b - a)
            h[i] = c1
            f1 = compute_c5(h, n)
            h[i] = c2
            f2 = compute_c5(h, n)
            for _ in range(n_golden):
                if f1 < f2:
                    b = c2; c2 = c1; f2 = f1
                    c1 = b - phi * (b - a)
                    h[i] = c1
                    f1 = compute_c5(h, n)
                else:
                    a = c1; c1 = c2; f1 = f2
                    c2 = a + phi * (b - a)
                    h[i] = c2
                    f2 = compute_c5(h, n)
            h[i] = (a + b) * 0.5
            s = h.sum()
            diff = target - s
            if diff > 1e-10:
                room = 1.0 - h
                tot = room.sum()
                if tot > 1e-15:
                    h += (diff / tot) * room
            elif diff < -1e-10:
                tot = h.sum()
                if tot > 1e-15:
                    h += (diff / tot) * h
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        if best_c >= prev_best - 1e-13:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    h = project_feasible(best_h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h
    return best_h, best_c

def pair_refine(h, n, deadline, radius=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    while time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                return h, best_c
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
                    if h[i] <= 1.0 - d and h[j] >= d:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
        if not improved:
            break
    return h, best_c

def multi_start_refine(h_best, n, deadline, n_starts=4):
    rng = np.random.RandomState(123)
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    per_start = max(10.0, (deadline - time.time()) / n_starts)
    scales = [0.05, 0.03, 0.015, 0.01]
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        scale = scales[s % len(scales)]
        h_pert = h_best + rng.uniform(-scale, scale, n)
        h_pert = project_feasible(h_pert, n)
        t_end = min(time.time() + per_start * 0.75, deadline - 3)
        h_new, c_new = coord_descent(h_pert, n, t_end, n_golden=45)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c
