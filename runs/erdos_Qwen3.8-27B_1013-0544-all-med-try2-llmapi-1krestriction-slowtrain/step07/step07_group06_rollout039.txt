            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(50, deadline - 40 - time.time())
            if alloc < 20:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, min(time.time() + alloc * 0.4, deadline - 35),
                                    alphas=[100, 1000, 10000], max_iters=100)
            if time.time() < deadline - 20:
                h_t, c_t = adam_phase(h_t, n_try, min(time.time() + alloc * 0.3, deadline - 18),
                                       alpha_init=100, alpha_end=2000, lr_init=0.008, n_steps=800)
            if time.time() < deadline - 10:
                h_t, c_t = targeted_pair_search(h_t, n_try, min(time.time() + 10, deadline - 8),
                                                 [0.005, 0.001, 0.0001], top_lags=8, radius=50)
            if c_t < best_c:
                best_c, best_h = c_t, h_t
    # Phase 6: Fine pair search
    if time.time() < deadline - 10:
        best_h, best_c = targeted_pair_search(best_h, len(best_h), deadline - 3,
                                               [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                               top_lags=20, radius=80)
    # Final: one more L-BFGS polish
    if time.time() < deadline - 2:
        best_h, best_c = lbfgs_phase(best_h, len(best_h), deadline - 2,
                                      alphas=[50000, 200000], max_iters=50)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
