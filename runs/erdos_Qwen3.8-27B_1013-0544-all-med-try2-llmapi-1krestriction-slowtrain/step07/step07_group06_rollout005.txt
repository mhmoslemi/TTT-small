def project_feasible(h,n):
    h=np.clip(np.asarray(h,dtype=float),0.0,1.0)
    target=0.5*n
    if abs(float(h.sum())-target)<1e-12: return h
    lo=float(h.min())-1.0; hi=float(h.max())
    for _ in range(60):
        mid=0.5*(lo+hi)
        s=float(np.sum(np.clip(h-mid,0.0,1.0)))
        if s>target: hi=mid
        else: lo=mid
    h2=np.clip(h-0.5*(lo+hi),0.0,1.0)
    for _ in range(30):
        diff=target-float(h2.sum())
        if abs(diff)<1e-13: break
        mask=h2<1.0 if diff>0 else h2>0.0
        cnt=int(mask.sum())
        if cnt==0: break
        h2[mask]=np.clip(h2[mask]+diff/cnt,0.0,1.0)
    return h2

def compute_C(h,n):
    h=np.asarray(h,dtype=float)
    return np.correlate(h,1.0-h,mode='full')

def c5_from_C(C,n):
    return float(np.max(C))*(2.0/n)

    n=len(h); center=n-1
    m=float(np.max(C)); z=alpha*(C-m); mx=float(z.max())
    w=np.exp(z-mx); s=float(w.sum())

    else: w=w/s
    g=np.zeros(n,dtype=float)
    for idx in np.nonzero(w>1e-12)[0]:

        if idx==center:

import time
import numpy as np

D_FRACTIONS = (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001)
G_FRACTIONS = (1.0, 0.5, 0.25, 0.1)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max())
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    h2 = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - float(h2.sum())
        if abs(diff) < 1e-13:
            break
        mask = h2 < 1.0 if diff > 0.0 else h2 > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h2[mask] = np.clip(h2[mask] + diff / cnt, 0.0, 1.0)
    return h2

def compute_C(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full')

def c5_from_C(C, n):
    return float(np.max(C)) * (2.0 / n)

def grad_smooth(h, C, alpha):
    n = len(h)
    center = n - 1
    m = float(np.max(C))
    z = alpha * (C - m)
    mx = float(z.max())
    w = np.exp(z - mx)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
    else:
        w = w / s
    g = np.zeros(n, dtype=float)
    for idx in np.nonzero(w > 1e-12)[0]:
        wi = float(w[idx])
        if idx == center:
            g += wi * (1.0 - 2.0 * h)
        elif idx > center:
            r = idx - center
            mm = n - r
            g[r:n] += wi * (1.0 - h[0:mm])
            g[0:mm] -= wi * h[r:n]
        else:
            r = center - idx
            mm = n - r
            g[0:mm] += wi * (1.0 - h[r:n])
            g[r:n] -= wi * h[0:mm]
    return g

def fallback_initial(n, seed):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2.0 * np.pi * x)
    return project_feasible(h, n)

def resample_h(h, n_old, n_new):
    if int(n_old) == int(n_new):
        return np.asarray(h, dtype=float).copy()
    x0 = np.linspace(0.0, 2.0, n_old)
    x1 = np.linspace(0.0, 2.0, n_new)
    h1 = np.interp(x1, x0, np.asarray(h, dtype=float), left=float(h[0]), right=float(h[-1]))
    return project_feasible(h1, n_new)

def pair_linear(h, a, b, n, r, center):
    L = np.zeros(2 * n - 1, dtype=float)
    L[center] = 2.0 * (h[b] - h[a])
    if n > 1:
        Aa = r < (n - a)
        Ab = r <= a
        Ba = r < (n - b)
        Bb = r <= b
        LA = np.zeros(n - 1, dtype=float)
        LB = np.zeros(n - 1, dtype=float)
        P = np.zeros(n - 1, dtype=float)
        LA[Aa] += 1.0
        LA[Ba] -= 1.0
        LB[Ab] += 1.0
        LB[Bb] -= 1.0
        P[Aa] += h[a + r[Aa]]
        P[Ab] += h[a - r[Ab]]
        P[Ba] -= h[b + r[Ba]]
        P[Bb] -= h[b - r[Bb]]
        LA -= P
        LB -= P
        L[center - 1::-1] = LA
        L[center + 1:] = LB
    return L, abs(a - b)

def try_pair_move(h, C, cmax, a, b, max_step, tol, r, center):
    n = len(h)
    if a == b or n < 2:
        return None
    D = min(1.0 - h[a], h[b])
    if D <= 1e-12:
        return None
    Dcap = D if D < max_step else max_step
    L, dist = pair_linear(h, a, b, n, r, center)
    best_m = cmax
    best_d = 0.0
    best_C = None
    for fr in D_FRACTIONS:
        d = Dcap * fr
        if d <= 1e-9:
            continue
        tmp = C + d * L
        d2 = d * d
        tmp[center] -= 2.0 * d2
        if dist > 0:
            ip = center + dist
            im = center - dist
            if 0 <= ip < len(C):
                tmp[ip] += d2
            if 0 <= im < len(C):
                tmp[im] += d2
        m = float(tmp.max())
        if m < best_m:
            best_m = m
            best_d = d
            best_C = tmp
    if best_d > 0.0 and best_m < cmax - tol:
        h[a] = min(1.0, h[a] + best_d)
        h[b] = max(0.0, h[b] - best_d)
        return best_C, best_m
    return None

def try_gradient_step(h, C, cmax, g, max_step, tol, n):
    best_m = cmax
    best_h = None
    best_C = None
    for fr in G_FRACTIONS:
        h2 = project_feasible(h - max_step * fr * g, n)
        C2 = compute_C(h2, n)
        m2 = float(np.max(C2))
        if m2 < best_m:
            best_m = m2
            best_h = h2
            best_C = C2
    if best_h is not None and best_m < cmax - tol:
        return best_h, best_C, best_m
    return None

def gen_candidates(g, n, rng, top_k, rand_k):
    top_k = max(1, min(top_k, n))
    order = np.argsort(g)
    low = order[:top_k]
    high = order[n - top_k:]
    cands = []
    scores = []
    for a in low:
        ga = float(g[a])
        for b in high:
            if a != b and float(g[b]) > ga:
                cands.append((int(a), int(b)))
                scores.append(float(g[b]) - ga)
    pool = min(n, max(50, 2 * top_k))
    lowp = order[:pool]
    highp = order[n - pool:]
    for _ in range(rand_k):
        a = int(lowp[rng.integers(lowp.size)])
        b = int(highp[rng.integers(highp.size)])
        if a != b and float(g[b]) > float(g[a]):
            cands.append((a, b))
            scores.append(float(g[b]) - float(g[a]))
    for _ in range(max(8, rand_k // 3)):
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a == b:
            continue
        if float(g[a]) >= float(g[b]):
            a, b = b, a
        cands.append((a, b))
        scores.append(float(g[b]) - float(g[a]))
    seen = set()
    out = []
    outs = []
    for i, key in enumerate(cands):
        if key not in seen:
            seen.add(key)
            out.append(key)
            outs.append(scores[i])
    if not out:
        return out
    idx = np.argsort(-np.array(outs, dtype=float))
    res = []
    for i in idx:
        res.append(out[i])
    return res
