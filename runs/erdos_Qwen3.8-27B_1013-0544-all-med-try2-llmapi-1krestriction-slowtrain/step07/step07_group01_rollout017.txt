def precompute_all_grads(h, n):
    """Precompute gradient of each correlation w.r.t. h."""
    # grad[k_shift, j] = ∂c[k_shift]/∂h[j]
    all_grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        all_grads[m] = corr_grad_single(h, k, n)
    return all_grads

def eval_move(c, all_grads, i, j, d, dx):
    """Evaluate the change in correlations when h[i]+=d, h[j]-=d."""
    # Δc[m] = d * (all_grads[m, i] - all_grads[m, j]) * dx
    delta_c = d * (all_grads[:, i] - all_grads[:, j]) * dx
    new_c = c + delta_c
    return new_c.max()

        h, best_c = fast_pair_search(best_h, n, t2, deltas1, radius=80, max_sweeps=200)
        best_h = h.copy()

    # Phase 3: Greedy pair search targeting top correlations
    t3 = min(time.time() + 200, deadline - 200)
    if time.time() < t3:
        h3, c3 = greedy_pair_search(best_h, n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Very fine pair search
    t4 = min(time.time() + 200, deadline - 60)
    deltas2 = [0.0005, 0.0002, 0.0001, 0.00005]
    if time.time() < t4:
        h, best_c = fast_pair_search(best_h, n, t4, deltas2, radius=50, max_sweeps=300)
        best_h = h.copy()

    # Phase 5: Try alternate grid size
    if time.time() < deadline - 100:
        for n_try in [200, 400]:
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            c_int = compute_c5(h_interp, n_try)
            alloc = min(60, deadline - time.time() - 40)
            if alloc < 20:
                break
            # Quick L-BFGS on new grid
            bounds_t = [(0.0, 1.0)] * n_try
            try:
                res = minimize(smooth_max_obj_fn, h_interp, jac=smooth_max_grad_fn,
                              args=(n_try, 10000, 100.0), method='L-BFGS-B',
                              bounds=bounds_t,
                              options={'maxiter': 100, 'ftol': 1e-15})
                h_try = project_feasible(res.x, n_try)
                c_try = compute_c5(h_try, n_try)
            except Exception:
                h_try, c_try = h_interp, c_int
            # Quick pair search on new grid
            if time.time() < deadline - 30:
                h_try, c_try = fast_pair_search(h_try, n_try, min(time.time() + 30, deadline - 20),
                                                [0.005, 0.002, 0.001], radius=40, max_sweeps=100)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                n = n_try

    # Phase 6: Final polish
    nf = len(best_h)
    if time.time() < deadline - 10:
        deltas_final = [0.0001, 0.00005, 0.00002, 0.00001]
        h, c = fast_pair_search(best_h, nf, deadline - 5, deltas_final, radius=30, max_sweeps=200)
        if c < best_c:
            best_c = c
            best_h = h

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
