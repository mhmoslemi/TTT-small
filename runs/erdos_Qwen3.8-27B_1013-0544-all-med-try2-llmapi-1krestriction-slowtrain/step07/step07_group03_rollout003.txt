def try_direction(h, f, c, dx, d):
    d = np.asarray(d, dtype=float).copy()
    if d.size != len(h) or not np.all(np.isfinite(d)):

    d -= d.mean()
    m = float(np.max(np.abs(d)))
    if m <= 1e-12:

    d /= m
    tmax = 1.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        r = (1.0 - h[pos]) / d[pos]
        tmax = min(tmax, float(np.min(r)))
    if np.any(neg):
        r = h[neg] / (-d[neg])
        tmax = min(tmax, float(np.min(r)))

def weighted_gradient(h,f,alpha,info,K=None):
    n=len(h); M=len(f)
    if K is not None and K < M:
       idx=np.argpartition(f, -K)[-K:]
    else:
       idx=np.arange(M)
    fm=float(f[idx].max())
    z=alpha*(f[idx]-fm)
    z=z - z.max() # max 0
    e=np.exp(z); s=e.sum()
    if s <= 0: return np.zeros(n)
    w=e/s
    G=np.zeros(n)
    for ii in range(len(idx)):
       wk=float(w[ii])
       if wk <= 1e-16: continue
       kk=int(idx[ii])
       p1,t1,p2,t2=info[kk]
       G[p1] += wk*(1.0 - h[t1])
       G[p2] -= wk*h[t2]
    return G - G.mean()

import numpy as np
import time
from scipy.optimize import minimize, linprog


def project_simplex_box(y, n, tgt):
    y = np.asarray(y, dtype=float)
    if y.size != n:
        y = np.resize(y, n)
    if not np.all(np.isfinite(y)):
        return np.full(n, tgt / float(max(1, n)))
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - hi, 0.0, 1.0)
    for _ in range(24):
        d = tgt - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0.0:
            r = 1.0 - x
            r[r < 1e-15] = 0.0
            if r.sum() > 1e-15:
                x += (d / r.sum()) * r
        else:
            r = x.copy()
            r[r < 1e-15] = 0.0
            if r.sum() > 1e-15:
                x += (d / r.sum()) * r
        x = np.clip(x, 0.0, 1.0)
    return x


def c5_and_corr(h, n):
    dx = 2.0 / float(n)
    f = np.correlate(h, 1.0 - h, mode='full')
    return float(f.max() * dx), f


def make_shift_info(n):
    info = []
    for k in range(2 * n - 1):
        l = n - 1 - k
        if l >= 0:
            p1 = np.arange(n - l)
            t1 = p1 + l
            p2 = np.arange(l, n)
            t2 = p2 - l
        else:
            p1 = np.arange(-l, n)
            t1 = p1 + l
            p2 = np.arange(n + l)
            t2 = p2 - l
        info.append((p1, t1, p2, t2))
    return info


def weighted_gradient(h, f, alpha, info):
    w = np.exp(alpha * (f - float(f.max())))
    w /= w.sum()
    G = np.zeros(len(h), dtype=float)
    for k, wk in enumerate(w):
        if wk > 1e-15:
            p1, t1, p2, t2 = info[k]
            G[p1] += wk * (1.0 - h[t1])
            G[p2] -= wk * h[t2]
    return G - G.mean()


def smooth_func(x, alpha, P, n, info):
    f = np.correlate(x, 1.0 - x, mode='full')
    fm = float(f.max())
    lse = fm + float(np.log(np.exp(alpha * (f - fm)).sum())) / alpha
    err = x.sum() - 0.5 * n
    return float(lse + P * err * err)


def smooth_jac(x, alpha, P, n, info):
    f = np.correlate(x, 1.0 - x, mode='full')
    G = weighted_gradient(x, f, alpha, info)
    G += 2.0 * P * (x.sum() - 0.5 * n)
    return G


def run_smooth(x, n, info, alpha, P, maxiter):
    x = np.asarray(x, dtype=float)
    bounds = [(0.0, 1.0)] * n
    opts = {
        'maxiter': int(maxiter),
        'ftol': 1e-15,
        'gtol': 1e-9,
        'maxfun': max(200, 2 * int(maxiter)),
        'maxls': 20
    }
    try:
        res = minimize(smooth_func, x, args=(alpha, P, n, info),
                       method='L-BFGS-B', jac=smooth_jac,
                       bounds=bounds, options=opts)
    except Exception:
        return x
    x2 = getattr(res, 'x', None)
    if x2 is None or x2.size != n or not np.all(np.isfinite(x2)):
        return x
    return project_simplex_box(x2, n, 0.5 * n)


def try_direction(h, f, c, dx, d):
    d = np.asarray(d, dtype=float).copy()
    if d.size != len(h) or not np.all(np.isfinite(d)):
        return None
    d -= d.mean()
    mval = float(np.max(np.abs(d)))
    if mval <= 1e-12:
        return None
    d /= mval
    tmax = 1.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        tmax = min(tmax, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        tmax = min(tmax, float(np.min(h[neg] / (-d[neg]))))
    if not np.isfinite(tmax) or tmax <= 1e-12:
        return None
    tmax = min(1.0, tmax)
    Ac = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Bc = -np.correlate(d, d, mode='full')
    lin = np.linspace(0.0, tmax, 65)
    base = max(1e-15, tmax * 1e-7)
    geo = np.geomspace(base, tmax, 30) if base < tmax else np.array([tmax])
    ts = np.unique(np.concatenate((lin, geo, np.array([0.0, tmax]))))
    vals = f[None, :] + ts[:, None] * Ac[None, :] + (ts[:, None] ** 2) * Bc[None, :]
    cts = dx * vals.max(axis=1)
    i = int(np.argmin(cts))
    if cts[i] < c - 1e-14:
        h2 = h + float(ts[i]) * d
        h2 = project_simplex_box(h2, len(h), 0.5 * len(h))
        f2 = np.correlate(h2, 1.0 - h2, mode='full')
        c2 = float(f2.max() * dx)
        if c2 < c - 1e-15:
            return h2, f2, c2
    return None


def active_lp_polish(h, n, info, deadline, max_iters=80, m_active=256):
    h = np.asarray(h, dtype=float).copy()
    f = np.correlate(h, 1.0 - h, mode='full')
    c = float(f.max() * (2.0 / n))
    dx = 2.0 / float(n)
    M = len(f)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        m = min(M, max(32, m_active))
        idx = np.argpartition(f, -m)[-m:]
        rows = np.empty((m, n), dtype=float)
        for rpos, kpos in enumerate(idx):
            kk = int(kpos)
            p1, t1, p2, t2 = info[kk]
            row = rows[rpos]
            row.fill(0.0)
            row[p1] += 1.0 - h[t1]
            row[p2] -= h[t2]
        A_ub = np.empty((m, n + 1), dtype=float)
        A_ub[:, :n] = rows
        A_ub[:, -1] = -1.0
        cvec = np.zeros(n + 1)
        cvec[-1] = 1.0
        Aeq = np.ones(n + 1)
        Aeq[-1] = 0.0
        bounds = [(-1.0, 1.0)] * n + [(None, None)]
        d = None
        try:
            res = linprog(cvec, A_ub=A_ub, b_ub=np.zeros(m),
                          A_eq=Aeq.ravel(), b_eq=0.0, bounds=bounds,
                          method='highs')
            if res.success and res.x is not None:
                d = res.x[:n]
        except Exception:
            d = None
        if d is not None:
            out = try_direction(h, f, c, dx, d)
            if out is not None:
                h, f, c = out
                stall = 0
                continue
        G = weighted_gradient(h, f, 1000.0, info)
        out = try_direction(h, f, c, dx, -G)
        if out is not None:
            h, f, c = out
            stall = 0
            continue
        stall += 1
        if stall >= 6:
            break
    return h, c


def to_binary(h, n):
    order = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    b[order[:n // 2]] = 1.0
    return b


def interp_h(h, n, newn):
    if newn == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(newn) + 0.5) * (2.0 / newn)
    return project_simplex_box(np.interp(xn, xo, h), newn, 0.5 * newn)
