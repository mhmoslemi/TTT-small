class TimeStop(Exception):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(220):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (d / tot) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-15:

        if d > 0.0:

        else:

            if s < 1e-14:
                break
            h += (d / s) * h
    np.clip(h, 0.0, 1.0, out=h)
    return h


def make_tent(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return project_feasible(h, n)


def c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def c_max(h, n):
    return float(np.max(c_vec(h, n)))


def weighted_gradient_raw(h, n, w):
    w0 = float(w[n - 1])
    if n == 1:
        return w0 * (1.0 - 2.0 * h)
    pos = w[n:]
    neg = w[n - 2::-1]
    pref_pos = np.concatenate(([0.0], np.cumsum(pos)))
    pref_neg = np.concatenate(([0.0], np.cumsum(neg)))
    idx = np.arange(n)
    pidx = np.maximum(n - 1 - idx, 0)
    L = w0 + pref_pos[pidx] + pref_neg[idx]
    Pvec = np.zeros(2 * n - 1, dtype=float)
    P = pos + neg
    Pvec[n:] = P
    Pvec[:n - 1][::-1] = P
    Pvec[n - 1] = 2.0 * w0
    A = np.convolve(h, Pvec, mode="full")[n - 1:2 * n - 1]
    return L - A


def smooth_value(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    return float(m + (mx + np.log(e.sum())) / alpha)


def smooth_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return m, w


def lbfgs_obj(x, n, alpha, mu):
    c = c_vec(x, n)
    val = smooth_value(c, alpha)
    return float(val + mu * (x.sum() - 0.5 * n) ** 2)


def lbfgs_grad(x, n, alpha, mu):
    c = c_vec(x, n)
    _, w = smooth_weights(c, alpha)
    g = (2.0 / float(n)) * weighted_gradient_raw(x, n, w)
    s = x.sum() - 0.5 * n
    if abs(s) > 1e-18:
        g = g + (2.0 * mu * s) * np.ones(n, dtype=float)
    return g


def slsqp_obj(x, n, alpha):
    return smooth_value(c_vec(x, n), alpha)


def slsqp_grad(x, n, alpha):
    c = c_vec(x, n)
    _, w = smooth_weights(c, alpha)
    return (2.0 / float(n)) * weighted_gradient_raw(x, n, w)

    return float(x.sum() - 0.5 * n)

        best_c = cs
        best_h = hs
    return best_h, best_c


def curve_full(n, vals):
    vals = np.asarray(vals, dtype=float)
    m = len(vals)
    xu = np.linspace(0.0, 1.0, m)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.empty(n, dtype=float)
    left = x <= 1.0
    h[left] = np.interp(x[left], xu, vals)
    if np.any(~left):
        h[~left] = np.interp(2.0 - x[~left], xu, vals)
    return project_feasible(h, n)


def curve_obj(vals, n_eval):
    h = curve_full(int(n_eval), vals)
    return c_max(h, int(n_eval))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s)) - 3.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = int(len(h_init))
    except Exception:
        n_init = 400
        h_init = make_tent(n_init)

    if n_init < 64:
        n_init = 300
        h_init = make_tent(n_init)

    best_n = int(n_init)
    best_h = project_feasible(h_init, best_n)
    best_c = c_max(best_h, best_n)
    bounds = [(0.0, 1.0)] * best_n

    hs = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
    cs = c_max(hs, best_n)
    if cs < best_c:
        best_c = cs
        best_h = hs

    t1 = min(deadline - 430, t0 + 430)
    if t1 > time.time() + 15:
        best_h, best_c = local_lbfgs(best_h, best_n, t1, bounds)

    if time.time() < deadline - 260 and best_n <= 560:
        t2 = min(deadline - 230, time.time() + 100)
        best_h, best_c = slsqp_polish(best_h, best_n, t2, bounds, alpha=80000, maxiter=90)

    if time.time() < deadline - 170:
        t3 = min(deadline - 150, time.time() + 45)
        best_h, best_c = subgrad_momentum(best_h, best_n, t3)

    if time.time() < deadline - 110:
        t4 = min(deadline - 95, time.time() + 30)
        best_h, best_c = directed_pair(best_h, best_n, t4)

    if time.time() < deadline - 100:
        n_eval = 240
        for m in (6, 8):
            if time.time() > deadline - 45:
                break
            cbounds = [(0.0, 1.0)] * m
            try:
                res = differential_evolution(
                    curve_obj,
                    cbounds,
                    args=(n_eval,),
                    seed=seed,
                    maxiter=35,
                    popsize=9,
                    tol=1e-6,
                    polish=False,
                    init="latinhypercube",
                )
                vals = np.asarray(res.x, dtype=float)
            except Exception:
                vals = np.linspace(0.0, 1.0, m)
            hc = curve_full(n_eval, vals)
            cc = c_max(hc, n_eval)
            if cc < best_c + 5e-4:
                allo = min(45.0, max(12.0, 0.20 * (deadline - time.time())))
                if allo > 8.0:
                    hb, cb = local_lbfgs(hc, n_eval, time.time() + allo, [(0.0, 1.0)] * n_eval)
                    if cb < cc:
                        cc, hc = cb, hb
            if cc < best_c:
                best_c = cc
                best_h = hc
                best_n = n_eval
                bounds = [(0.0, 1.0)] * best_n

    if time.time() < deadline - 90:
        for n_try in (240, 320, 420, 600, 800):
            if time.time() > deadline - 50:
                break
            if n_try > 900 or n_try < 64:
                continue
            if abs(n_try - best_n) < max(20, 0.05 * best_n):
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_t = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            c_t = c_max(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
                bounds = [(0.0, 1.0)] * best_n
            allo = min(40.0, max(10.0, 0.16 * (deadline - time.time())))
            if allo > 6.0:
                h_t, c_t = local_lbfgs(h_t, n_try, time.time() + allo, [(0.0, 1.0)] * n_try)
                if c_t < best_c:
                    best_c = c_t
                    best_h = h_t
                    best_n = n_try
                    bounds = [(0.0, 1.0)] * best_n

    if time.time() < deadline - 80:
        h_p = best_h + rng.normal(0.0, 0.035, size=best_n)
        h_p = project_feasible(h_p, best_n)
        t5 = min(deadline - 60, time.time() + 45)
        h_p, c_p = local_lbfgs(h_p, best_n, t5, bounds)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
            best_n = len(h_p)
            bounds = [(0.0, 1.0)] * best_n

    if time.time() < deadline - 50:
        t6 = min(deadline - 40, time.time() + 30)
        best_h, best_c = subgrad_momentum(best_h, best_n, t6)

    if time.time() < deadline - 25:
        t7 = min(deadline - 18, time.time() + 18)
        best_h, best_c = directed_pair(best_h, best_n, t7)

    if time.time() < deadline - 20 and best_n <= 520:
        t8 = min(deadline - 10, time.time() + 15)
        best_h, best_c = slsqp_polish(best_h, best_n, t8, bounds, alpha=250000, maxiter=40)

    best_h = project_feasible(best_h, best_n)
    best_c = c_max(best_h, best_n)
    return best_h, best_c, best_n
