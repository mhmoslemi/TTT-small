def coordinate_descent_pass(h, n, max_iter=10):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    best_c = np.max(c)
    
    for iteration in range(max_iter):
        improved = False
        for i in range(n):
            # Compute affected range
            m_start = i
            m_end = min(i + n - 1, 2*n - 2)
            js = np.arange(m_end - m_start + 1)
            m_idx = js + i
            slopes = 1.0 - 2.0 * h[js]
            old_val = h[i]
            
            # Unaffected max
            c_unaff = best_c  # upper bound (might not be tight)
            if m_start > 0:
                c_unaff = min(c_unaff, np.max(c[:m_start]))
            if m_end < 2*n - 2:
                c_unaff = min(c_unaff, np.max(c[m_end+1:]))
            
            # Find optimal a
            a_best = find_optimal_a(c, m_idx, slopes, old_val, c_unaff)
            
            if abs(a_best - old_val) > 1e-10:
                # Update c
                c[m_idx] += (a_best - old_val) * slopes
                h[i] = a_best
                new_c = np.max(c)
                if new_c < best_c:
                    best_c = new_c
                improved = True
        
        if not improved:
            break
    
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Heavy coordinate descent
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    hat = h.copy()
    mu = 0.05
    while time.time() < deadline - 280:
        hat, c = cd_round(hat, c, n, mu)
        hat_p = project_feasible(hat, n)
        cp = compute_c5(hat_p, n)
        if cp < best_c:
            best_c = cp
            best_h = hat_p.copy()

    # Phase 2: L-BFGS-B refinement
    if time.time() < deadline - 130:
        h2, c2 = lbfgs_phase(best_h, n, deadline - 110, max_iter=250)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: More CD from L-BFGS result
    if time.time() < deadline - 80:
        h3 = best_h.copy()
        c3 = np.correlate(h3, 1 - h3, mode='full') * (2.0 / n)
        while time.time() < deadline - 70:
            h3, c3 = cd_round(h3, c3, n, 0.05)
            h3p = project_feasible(h3, n)
            cp = compute_c5(h3p, n)
            if cp < best_c:
                best_c = cp
                best_h = h3p.copy()

    # Phase 4: Try alternative resolution
    if time.time() < deadline - 60:
        for n_try in [250, 400]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(25, deadline - time.time() - 30)
            if alloc < 10:
                break
            c_t = np.correlate(h_interp, 1 - h_interp, mode='full') * (2.0 / n_try)
            h_t = h_interp.copy()
            while time.time() < t0 + budget_s - 35 and time.time() < time.time() + alloc - 5:
                h_t, c_t = cd_round(h_t, c_t, n_try, 0.05)
            h_t = project_feasible(h_t, n_try)
            ct = compute_c5(h_t, n_try)
            if ct < best_c:
                best_c = ct
                best_h = h_t.copy()
                n = n_try

    # Phase 5: Pair search polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 2,
                                     [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                     radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
