    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick L-BFGS refine on original grid
    if time.time() < deadline - 200:
        t1 = min(time.time() + 40, deadline - 200)
        h_init, c_init = run_lbfgs(best_h, n_init, t1, max_iter=200)
        if c_init < best_c:
            best_c = c_init
            best_h = h_init

    # Phase 2: Upscale to 600 and optimize
    n_fine = 600
    x_old = np.linspace(0, 2, n_init)
    x_new = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_new, x_old, best_h)
    h_fine = project_feasible(h_fine, n_fine)
    c_fine = compute_c5(h_fine, n_fine)
    if c_fine < best_c:
        best_c = c_fine
        best_h = h_fine

    # Phase 3: L-BFGS on 600-grid
    if time.time() < deadline - 100:
        t3 = min(time.time() + 180, deadline - 100)
        h_fine, c_fine = run_lbfgs(h_fine, n_fine, t3, max_iter=200)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine

    # Phase 4: SLSQP on 600-grid with increasing alphas
    if time.time() < deadline - 80:
        for alpha in [20000, 100000, 500000, 2000000]:
            if time.time() > deadline - 40:
                break
            t4 = min(time.time() + 70, deadline - 40)
            h_fine, c_fine = run_slsqp(h_fine, n_fine, alpha, 800, t4)
            if c_fine < best_c:
                best_c = c_fine
                best_h = h_fine

    # Phase 5: Gradient pair search on 600-grid
    if time.time() < deadline - 60:
        t5 = min(time.time() + 120, deadline - 60)
        h_fine, c_fine = grad_pair_search(h_fine, n_fine, t5, 0.003)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine

    # Phase 6: Pair search on 600-grid
    if time.time() < deadline - 15:
        t6 = deadline - 8
        h_fine, c_fine = pair_search(h_fine, n_fine, t6,
                                      [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                      radius=100)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine

    # Phase 7: Try 400-grid as alternative
    if time.time() < deadline - 20:
        n_alt = 400
        x_alt = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_alt, x_new, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        t7 = min(time.time() + 50, deadline - 10)
        h_alt, c_alt = run_lbfgs(h_alt, n_alt, t7, max_iter=150)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
