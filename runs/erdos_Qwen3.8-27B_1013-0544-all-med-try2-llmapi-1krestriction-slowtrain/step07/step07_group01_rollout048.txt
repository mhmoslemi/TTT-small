                break
    return best_h, best_c

def pair_search_fast(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] = max(0.0, h[i] - d)
                        h[j] = min(1.0, h[j] + d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
        h = best_h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS with many alpha levels (main optimization)
    t1 = t0 + budget_s * 0.40
    best_h, best_c = optimize_lbfgs(best_h, best_n, t1, max_iters=300)

    # Phase 2: SLSQP with multiple alphas
    t2 = min(t0 + budget_s * 0.55, deadline - budget_s * 0.3)
    for a in [50000, 200000, 1000000]:
        if time.time() > t2:
            break
        h2, c2 = optimize_slsqp(best_h, best_n, min(time.time() + 80, t2), alpha=a)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Coordinate descent with gradient guidance
    if time.time() < deadline - budget_s * 0.15:
        t3 = min(time.time() + budget_s * 0.10, deadline - budget_s * 0.15)
        best_h, best_c = coord_descent(best_h, best_n, t3, alpha=5000)

    # Phase 4: Fine pair search
    if time.time() < deadline - 30:
        t4 = min(time.time() + budget_s * 0.12, deadline - 15)
        best_h, best_c = pair_search_fast(best_h, best_n, t4,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    # Phase 5: Try other grid sizes
    remaining = deadline - time.time() - 20
    if remaining > 60:
        for n_try in [200, 400]:
            if time.time() > deadline - 30:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(40, remaining * 0.5)
            if alloc < 20:
                break
            h_t, c_t = optimize_lbfgs(h_interp, n_try, time.time() + alloc * 0.6, max_iters=200)
            if time.time() < deadline - 25:
                h_t, c_t = optimize_slsqp(h_t, n_try, time.time() + 15, alpha=500000)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 6: Final squeeze with very small steps
    if time.time() < deadline - 3:
        best_h, best_c = pair_search_fast(best_h, best_n, deadline - 1,
                                          [0.0005, 0.0002, 0.0001])

    # Final L-BFGS squeeze
    if time.time() < deadline - 2:
        best_h, best_c = optimize_lbfgs(best_h, best_n, deadline - 1, max_iters=50)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
