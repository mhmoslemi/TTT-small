    h=np.clip(h,0.0,1.0)

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5_correlate(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def counts_array(h, n):
    q = 1.0 - h
    c = np.empty(2 * n - 1, dtype=float)
    off = n - 1
    for l in range(n):
        m = n - l
        c[off + l] = float(np.dot(h[:m], q[l:]))
        if l:
            c[off - l] = float(np.dot(h[l:], q[:m]))
    return c


def smooth_value_weights(c, alpha):
    mx = float(c.max())
    sh = alpha * (c - mx)
    ev = np.exp(sh)
    sm = float(ev.sum())
    if sm <= 0.0 or not np.isfinite(sm):
        w = np.full_like(c, 1.0 / c.size, dtype=float)
        return mx, w
    val = mx + np.log(sm) / alpha
    w = ev / sm
    return val, w


def grad_fft_weights(h, n, w):
    q = 1.0 - h
    off = n - 1
    need = 3 * n - 2
    M = 1
    while M < need:
        M <<= 1
    fq = np.fft.fft(q, M)
    fh = np.fft.fft(h, M)
    fw = np.fft.fft(w, M)
    fwr = np.fft.fft(w[::-1], M)
    a = np.fft.ifft(fq * fwr).real
    b = np.fft.ifft(fh * fw).real
    return a[off:off + n] - b[off:off + n]


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(2000):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    np.clip(h, 0.0, 1.0, out=h)
    return h


def continuous_objective(x, n, alpha, mu):
    c = counts_array(x, n)
    val, _ = smooth_value_weights(c, alpha)
    if mu:
        d = float(x.sum()) - 0.5 * n
        val += mu * d * d
    return float(val)


def continuous_gradient(x, n, alpha, mu):
    c = counts_array(x, n)
    _, w = smooth_value_weights(c, alpha)
    g = grad_fft_weights(x, n, w)
    if mu:
        d = float(x.sum()) - 0.5 * n
        g = g + 2.0 * mu * d
    return g


def eq_fun(x, n):
    return float(x.sum() - 0.5 * n)


def eq_jac(x, n):
    return np.ones_like(x, dtype=float)

def select_top(vals, n):
    vals = np.asarray(vals, dtype=float)
    target = n // 2
    order = np.argsort(-vals, kind="stable")
    b = np.zeros(n, dtype=np.int8)
    b[order[:target]] = 1
    return b


def resample_h(h, n):
    h = np.asarray(h, dtype=float)
    m = len(h)
    if m == n:
        return h.copy()
    if m > 0 and np.all(np.abs(h - np.rint(h)) <= 1e-12):
        f = n // m
        if f > 1 and m * f == n:
            return np.repeat(h, f).copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, h)


def thue_morse_values(n, rng):
    seq = np.array([0], dtype=np.int8)
    while seq.size < n:
        seq = np.concatenate((seq, 1 - seq))
    seq = seq[:n].astype(float)
    seq += rng.rand(n) * 1e-3
    return seq


def make_binary_starts(h_cur, n, rng, count):
    base = resample_h(h_cur, n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    freq = 1.0 + 3.0 * rng.rand()
    phase = 2.0 * np.pi * rng.rand()
    vals_list = [
        base,
        base + rng.normal(0.0, 0.05, n),
        base + rng.normal(0.0, 0.20, n),
        base + rng.normal(0.0, 0.70, n),
        thue_morse_values(n, rng),
        rng.rand(n),
        np.sin(2.0 * np.pi * freq * x + phase) + rng.normal(0.0, 0.3, n),
        np.sin(2.0 * np.pi * 1.0 * x) + 0.8 * np.sin(2.0 * np.pi * 2.0 * x + phase) + rng.normal(0.0, 0.2, n),
    ]
    starts = []
    seen = set()
    for vals in vals_list:
        b = select_top(vals, n)
        key = b.tobytes()
        if key not in seen:
            seen.add(key)
            starts.append(b)
            if len(starts) >= count:
                break
    return starts


def perturb_binary(b, n, rng, k=5):
    b = np.asarray(b, dtype=np.int8).copy()
    for _ in range(k):
        ones = np.flatnonzero(b)
        zeros = np.flatnonzero(1 - b)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(ones[rng.randint(ones.size)])
        j = int(zeros[rng.randint(zeros.size)])
        if i != j:
            b[i] = 0
            b[j] = 1
    if int(b.sum()) != n // 2:
        b = select_top(b.astype(float), n)
    return b


def compute_binary_arrays(b, n):
    b = np.asarray(b, dtype=np.int8)
    q = (1 - b).astype(np.int16)
    pos = np.zeros(n, dtype=np.int64)
    neg = np.zeros(n, dtype=np.int64)
    epos = np.zeros((n, n), dtype=np.int16)
    eneg = np.zeros((n, n), dtype=np.int16)
    for l in range(1, n):
        left = b[:n - l]
        right = b[l:]
        pos[l] = int(np.count_nonzero(left & q[l:]))
        neg[l] = int(np.count_nonzero(right & q[:n - l]))
        epos[:n - l, l] = q[l:]
        epos[l:, l] -= left
        eneg[l:, l] = q[:n - l]
        eneg[:n - l, l] -= right
    return pos, neg, epos, eneg


def binary_local_search(b0, n, deadline, max_plateau=80):
    b = np.asarray(b0, dtype=np.int8).copy()
    if int(b.sum()) != n // 2:
        b = select_top(b.astype(float) + 1e-6, n)

    pos, neg, epos, eneg = compute_binary_arrays(b, n)
    cur_max = int(max(pos.max(), neg.max()))
    cur_sq = int((pos.astype(np.int64) ** 2).sum() + (neg.astype(np.int64) ** 2).sum())
    best_b = b.copy()
    best_count = cur_max
    best_sq = cur_sq
    plateau = 0

    while time.time() < deadline:
        ones = np.flatnonzero(b)
        zeros = np.flatnonzero(1 - b)
        if ones.size == 0 or zeros.size == 0:
            break

        pos16 = pos.astype(np.int16)
        neg16 = neg.astype(np.int16)
        zepos = epos[zeros]
        zeneg = eneg[zeros]
        rows = np.arange(zeros.size)

        best_o = -1
        best_z = -1
        best_mx = cur_max
        best_cand_sq = cur_sq

        for o in ones:
            base_pos = zepos - epos[o]
            base_neg = zeneg - eneg[o]
            cross = np.abs(zeros - o).astype(np.intp)
            base_pos[rows, cross] += 1
            base_neg[rows, cross] += 1

            new_pos = base_pos + pos16
            new_neg = base_neg + neg16

            row_max = np.maximum(new_pos.max(axis=1), new_neg.max(axis=1))
            min_r = int(row_max.min())
            if min_r > best_mx:
                continue

            if min_r < best_mx:
                target = min_r
                mask = row_max == target
            else:
                target = best_mx
                mask = row_max == target

            if target > cur_max:
                continue

            idxs = np.where(mask)[0]
            if idxs.size == 0:
                continue

            cp = new_pos[idxs].astype(np.int64)
            cn = new_neg[idxs].astype(np.int64)
            sq = (cp * cp + cn * cn).sum(axis=1)
            loc = int(np.argmin(sq))
            new_sq = int(sq[loc])
            if target < best_mx or (target == best_mx and new_sq < best_cand_sq - 1):
                best_mx = target
                best_cand_sq = new_sq
                best_o = int(o)
                best_z = int(zeros[idxs[loc]])

        if best_o < 0:
            break

        b[best_o] = 0
        b[best_z] = 1

        if best_mx < cur_max:
            plateau = 0
        else:
            plateau += 1
            if plateau > max_plateau:
                break

        pos, neg, epos, eneg = compute_binary_arrays(b, n)
        cur_max = int(max(pos.max(), neg.max()))
        cur_sq = int((pos.astype(np.int64) ** 2).sum() + (neg.astype(np.int64) ** 2).sum())
        if cur_max < best_count or (cur_max == best_count and cur_sq < best_sq - 1):
            best_count = cur_max
            best_sq = cur_sq
            best_b = b.copy()

    return best_b, float(best_count * (2.0 / n))
