    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - hp
            sr = room.sum()
            if sr > 1e-14:
                hp = hp + (diff / sr) * room
        else:
            sh = hp.sum()
            if sh > 1e-14:
                hp = hp * (1.0 + diff / sh)
    return np.clip(hp, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    """Return (val, grad) for smooth max of correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh = np.where(sh < -700, -700, sh)
    es = np.exp(sh)
    val = cmax + np.log(es.sum()) / alpha
    p = es / es.sum()
    
    # Gradient computation (O(n^2) but n is small)
    # grad[j] = dx * sum_s p[s] * [(1-h[j-s])*1{valid} - h[j+s]*1{valid}]
    # Using matrix approach
    n2 = p.shape[0]  # 2n-1
    rows = np.arange(n)
    cols = np.arange(n)
    # A[j,i] = p[j-i+n-1] (always valid since j-i in [-(n-1), n-1])
    idx_A = rows[:, None] - cols[None, :] + (n - 1)
    idx_B = cols[None, :] - rows[:, None] + (n - 1)
    A = p[idx_A]  # (n, n)
    Bm = p[idx_B]  # (n, n)
    grad = dx * (A @ (1.0 - h) - Bm @ h)
    return val, grad, cmax

def projected_grad_descent(h, n, alpha, max_iter, deadline, lr_scale=1.0):

    h = h.astype(float).copy()

        lr = lr_scale / (alpha * dx * 2.0)

        # Use a conservative estimate

            best_val = val_h

def coord_descent(h, n, max_sweeps=200, deadline=None):
    """Exact coordinate descent on the max correlation."""
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            # Bilinear: c + B*(x - h[i]) where x is new value
            # Actually: c_new[m] = c[m] + B[m] * (x - h[i])
            # We want min_x max_m (c[m] + B[m]*(x - h[i]))
            # = min_x max_m (c[m] - B[m]*h[i] + B[m]*x)
            # Let a[m] = c[m] - B[m]*h[i], then min_x max_m (a[m] + B[m]*x)
            a = corr - B * h[i]
            # Binary search on threshold T: exists x in [0,1] with a[m] + B[m]*x <= T for all m
            # For B[m] > 0: x <= (T - a[m]) / B[m]
            # For B[m] < 0: x >= (T - a[m]) / B[m]
            # For B[m] = 0: need a[m] <= T
            
            pos = B > 1e-14
            neg = B < -1e-14
            zero = ~(pos | neg)
            zero_max = a[zero].max() if zero.any() else -np.inf
            
            lo_v = min(a.min(), zero_max) - 1e-9
            hi_v = max(a.max(), zero_max) + 1e-9
            
            for _ in range(50):
                mid = 0.5 * (lo_v + hi_v)
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb = 0.0
                if neg.any():
                    lb = max(lb, ((mid - a[neg]) / B[neg]).max())
                ub = 1.0
                if pos.any():
                    ub = min(ub, ((mid - a[pos]) / B[pos]).min())
                if lb <= ub + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            # Find optimal x at threshold hi_v
            lb = 0.0
            if neg.any():
                lb = max(lb, ((hi_v - a[neg]) / B[neg]).max())
            ub = 1.0
            if pos.any():
                ub = min(ub, ((hi_v - a[pos]) / B[pos]).min())
            x_opt = np.clip(0.5 * (lb + ub), 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -700, 700)
        sm = np.log(np.exp(shifted).sum())
        return m + sm / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

            if best_j >= 0 and best_nc < cur_c - 1e-15:

                if neg.any():
                    lb = max(lb, ((mid - a[neg]) / B[neg]).max())
                ub = 1.0
                if pos.any():
                    ub = min(ub, ((mid - a[pos]) / B[pos]).min())
                if lb <= ub + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb = 0.0
            if neg.any():
                lb = max(lb, ((hi_v - a[neg]) / B[neg]).max())
            ub = 1.0
            if pos.any():
                ub = min(ub, ((hi_v - a[pos]) / B[pos]).min())
            x_opt = np.clip(0.5 * (lb + ub), 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = np.clip(alpha * (g - m), -700, 700)
        return m + np.log(np.exp(sh).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j, best_nc = -1, cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc, best_j = nc, j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                corr += delta
                h[i], h[j] = 1.0, 0.0
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h, n = h0.copy(), n0
    best_c = compute_c5(best_h, n)
    # Phase 1: Binary greedy
    hb, cb = greedy_binary(h0, n, t0 + 0.05 * total, 150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: Coord descent on result
    hc, _ = coord_descent(best_h, n, 400, t0 + 0.15 * total)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    # Phase 3: SLSQP progressive alpha (main)
    p3_dl = t0 + 0.52 * total
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000,
              2000000, 10000000, 50000000, 200000000, 1000000000,
              5000000000, 20000000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p3_dl - 5:
            break
        h = slsqp_smooth(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 4: Coord descent polish
    hc2, _ = coord_descent(best_h, n, 600, t0 + 0.65 * total)
    hc2 = project_feasible(hc2, n)
    cc2 = compute_c5(hc2, n)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()
    # Phase 5: More SLSQP at very high alpha
    p5_dl = t0 + 0.78 * total
    h = best_h.copy()
    for alpha in [100000000, 1000000000, 10000000000, 100000000000]:
        if time.time() > p5_dl - 3:
            break
        h = slsqp_smooth(h, n, alpha, 4000, 1e-17)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: Resolution exploration
    p6_dl = t0 + 0.90 * total
    for nh in [200, 300, 500, 600]:
        if time.time() > p6_dl - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, 200, p6_dl - 6)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
        for alpha in [100000, 1000000, 50000000]:
            if time.time() > p6_dl - 3:
                break
            hh = slsqp_smooth(hh, nh, alpha, 1500, 1e-15)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
    # Phase 7: Diverse noise restarts
    if time.time() < dl - 5:
        for trial in range(8):
            if time.time() > dl - 2:
                break
            scale = 0.001 * (1.0 + 0.5 * trial)
            h = best_h + scale * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [50000000, 500000000, 5000000000]:
                if time.time() > dl - 1:
                    break
                h = slsqp_smooth(h, n, alpha, 2000, 1e-17)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
    # Final polish
    if time.time() < dl - 1:
        hc3, _ = coord_descent(best_h, n, 100, dl - 0.5)
        hc3 = project_feasible(hc3, n)
        cc3 = compute_c5(hc3, n)
        if cc3 < best_c:
            best_c, best_h = cc3, hc3.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
