import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(len(x))

def slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=100):
    """Greedy local search with restricted neighborhood."""
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

    """Simulated annealing with periodic SLSQP refinement."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    max_step = 0.15

    window_count = 0

            if rate > 0.65:

            accept_count = 0

            j = (i + offset) % n
        else:

            j = rng.randint(n)

    return float(np.max(corr) * dx)

    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(len(x))

def slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def greedy_refine(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def sa_optimize(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    e_cur = best_c
    T = 0.004
    T_min = 0.00003
    cooling = 0.9999
    step_size = 0.05
    min_step = 0.003
    max_step = 0.15
    accept_count = 0
    window = 200
    wcount = 0
    stagnant = 0
    stag_limit = 3000
    slsqp_count = 0

    while time.time() < deadline - 15:
        if wcount >= window:
            rate = accept_count / wcount
            if rate > 0.6:
                step_size = min(max_step, step_size * 1.12)
            elif rate < 0.2:
                step_size = max(min_step, step_size * 0.88)
            accept_count = 0
            wcount = 0

        i = rng.randint(n)
        if rng.random() < 0.55:
            j = (i + rng.randint(-20, 21)) % n
        else:
            j = rng.randint(n)
        if i == j:
            continue

        delta = rng.uniform(-step_size, step_size)
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
            wcount += 1
            continue

        h[i] = hi
        h[j] = hj
        e_new = compute_c5(h, n)
        wcount += 1

        if e_new < e_cur:
            e_cur = e_new
            accept_count += 1
            stagnant = 0
            if e_new < best_c - 1e-15:
                best_c = e_new
                best_h = h.copy()
        else:
            if T > T_min and rng.random() < np.exp(-(e_new - e_cur) / T):
                e_cur = e_new
                accept_count += 1
            else:
                h[i] -= delta
                h[j] += delta
            stagnant += 1

        T = max(T_min, T * cooling)

        if stagnant > stag_limit and time.time() < deadline - 30:
            if slsqp_count < 10:
                alpha = 30000 + 20000 * slsqp_count
                ht = slsqp(h, n, alpha, maxiter=600)
                c = compute_c5(ht, n)
                if c < best_c:
                    best_c = c
                    best_h = ht.copy()
                    h = ht.copy()
                    e_cur = c
                else:
                    hp = best_h + rng.normal(0, 0.025, n)
                    hp = project_feasible(hp, n)
                    h = hp
                    e_cur = compute_c5(h, n)
                slsqp_count += 1
            stagnant = 0
            stag_limit = max(1500, int(stag_limit * 1.4))
            T = max(T, 0.002)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Main SA
    alloc1 = min(550, (deadline - t0) * 0.5)
    if alloc1 > 50:
        h_sa, c_sa = sa_optimize(h, n, t0 + alloc1, rng)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 2: Greedy refinement
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        rad = 50 if n_cur <= 300 else 40
        dl2 = min(time.time() + 150, deadline - 60)
        best_h, best_c = greedy_refine(best_h, n_cur, dl2,
                                       [0.02, 0.01, 0.005, 0.002, 0.001], radius=rad)

    # Phase 3: SLSQP polish
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        ht = slsqp(best_h, n_cur, 150000, maxiter=2000)
        c = compute_c5(ht, n_cur)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 4: Fine greedy
    if time.time() < deadline - 12:
        n_cur = len(best_h)
        rad = 40 if n_cur <= 250 else 30
        best_h, best_c = greedy_refine(best_h, n_cur, deadline - 5,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=rad)

    # Phase 5: Final SLSQP
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        ht = slsqp(best_h, n_cur, 300000, maxiter=2000)
        c = compute_c5(ht, n_cur)
        if c < best_c:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
