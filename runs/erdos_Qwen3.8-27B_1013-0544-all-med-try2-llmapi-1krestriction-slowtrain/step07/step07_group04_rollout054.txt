```python
def fast_swap_search(h, n, deadline):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    cur_max = c.max()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        improved = False
        best_delta = cur_max
        best_ij = None
        
        for i in zeros:
            for j in ones:
                # Compute Δc for swap (i: 0→1, j: 1→0)
                # Δc[m] = (1-h[i+m])*I + h[j-m]*I - (1-h[j+m])*I - h[i-m])*I
                # where m ranges from -(n-1) to n-1, corresponding to index m+(n-1) in the full corr
                
                delta = np.zeros(2*n-1)
                # (1-h[i+m]) for valid i+m
                lo = max(0, -(i))  # i+m >= 0 => m >= -i
                hi = min(n-1-i, n-1)  # i+m <= n-1 => m <= n-1-i
                m_range = np.arange(lo, hi+1)
                delta[m_range + (n-1)] += (1 - h[i + m_range])
                
                # h[j-m] for valid j-m
                lo2 = max(-(n-1), -(j))  # j-m <= n-1 => m >= j-(n-1)
                hi2 = min(j, n-1)  # j-m >= 0 => m <= j
                m_range2 = np.arange(lo2, hi2+1)
                delta[m_range2 + (n-1)] += h[j - m_range2]
                
                # -(1-h[j+m]) for valid j+m
                lo3 = max(-(n-1), -(j))  # j+m >= 0 => m >= -j
                hi3 = min(n-1-j, n-1)  # j+m <= n-1 => m <= n-1-j
                m_range3 = np.arange(lo3, hi3+1)
                delta[m_range3 + (n-1)] -= (1 - h[j + m_range3])
                
                # -h[i-m] for valid i-m
                lo4 = max(-(n-1), i-(n-1))  # i-m <= n-1 => m >= i-(n-1)
                hi4 = min(i, n-1)  # i-m >= 0 => m <= i
                m_range4 = np.arange(lo4, hi4+1)
                delta[m_range4 + (n-1)] -= h[i - m_range4]
                
                new_max = (c + delta).max()
                if new_max < best_delta - 1e-15:
                    best_delta = new_max
                    best_ij = (i, j)
        
        if best_ij is not None:
            i, j = best_ij
            h[i], h[j] = 1.0, 0.0
            c += delta  # need to recompute delta for the actual swap
            cur_max = best_delta
            improved = True
            # Update ones and zeros
            ...
        
        if not improved:
            break
    
    return h, cur_max * dx
```

```python
def simulated_annealing(h, n, deadline, seed=42):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    cur_max = c.max() * dx
    best_c = cur_max
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    rng = np.random.RandomState(seed)
    
    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    while time.time() < deadline and T > T_min:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        delta = compute_delta_swap(h, n, i, j)
        new_max = (c + delta).max() * dx
        
        if new_max < cur_max:
            # Accept
            h[i], h[j] = 1.0, 0.0
            c = c + delta
            cur_max = new_max
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()
        else:
            # Accept with probability exp(-Δ/T)
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                h[i], h[j] = 1.0, 0.0
                c = c + delta
                cur_max = new_max
                ones.remove(j); ones.append(i)
                zeros.remove(i); zeros.append(j)
        
        T *= cooling
    
    return best_h, best_c
```

The gradient of f_α is the weighted average of the individual gradients, with weights proportional to exp(α*c[k]) / sum exp(α*c[k]).

This is a smooth upper bound on max_k c[k], and as α→∞ it converges to the max.

```python
    m_offset = n - 1
    c = np.correlate(h, 1 - h, mode='full')
    
    # Softmax weights
    c_shifted = alpha * (c - c.max())
    weights = np.exp(c_shifted)
    weights /= weights.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for m_idx in range(len(weights)):
        if weights[m_idx] < 1e-10:
            continue
        m = m_idx - m_offset
        g = np.ones(n)
        if m >= 0:
            g[:n-m] -= (h[m:] + h[:n-m])  # h[j+m] + h[j-m] where j-m is valid
        else:
            g[-m:] -= (h[:n+m] + h[-m:])
        grad += weights[m_idx] * g
    
    h_new = h - lr * grad * dx  # scale by dx since objective includes dx
    h_new = project_feasible(h_new, n)
    return h_new, c.max() * dx
```
