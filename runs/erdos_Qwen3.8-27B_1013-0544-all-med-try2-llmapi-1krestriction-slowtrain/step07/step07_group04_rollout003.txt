import time
import numpy as np
from scipy.optimize import minimize


def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    target = 0.5 * n
    s = z.sum()
    if np.all(z >= -1e-12) and np.all(z <= 1.0 + 1e-12) and abs(s - target) < 1e-9:
        z = np.clip(z, 0.0, 1.0)
        z = z + (target - z.sum()) / n
        return np.clip(z, 0.0, 1.0)
    lo = -1.0 - float(np.max(z))
    hi = 1.0 - float(np.min(z))
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)


def compute_q_natural(h):
    n = len(h)
    if n == 0:
        return np.array([0.0])
    c = n - 1
    sh = float(h.sum())
    ac = np.correlate(h, h, mode="full")[c:c + n]
    q = np.empty(2 * n - 1, dtype=float)
    q[c] = sh - ac[0]
    if n > 1:
        cum = np.concatenate(([0.0], np.cumsum(h)))
        u = np.arange(1, n)
        q[c + u] = cum[n - u] - ac[u]
        q[c - u] = sh - cum[u] - ac[u]
    return q


def smooth_values(h, beta):
    q = compute_q_natural(h)
    qmax = float(q.max())
    beta = float(beta)
    if beta <= 1e-12:
        f = qmax
        w = np.zeros_like(q)
        w[np.argmax(q)] = 1.0
        return f, w
    s = beta * (q - qmax)
    e = np.exp(s)
    se = float(e.sum())
    f = qmax + np.log(se) / beta
    return f, e / se


def gradient_from_weights(h, w):
    n = len(h)
    if n == 1:
        return w[0] * (1.0 - 2.0 * h)
    c = n - 1
    wp = w[c + 1:]
    wn = w[:c][::-1]
    ws = wp + wn
    wsum = np.zeros(n, dtype=float)
    wsum[1:] = ws
    A = np.convolve(h, wsum, mode="full")[:n]
    B = np.convolve(h, wsum[::-1], mode="full")[c:c + n]
    pwn = np.concatenate(([0.0], np.cumsum(wn)))
    pwp = np.concatenate(([0.0], np.cumsum(wp)))
    swp = pwp[-1] - pwp
    return w[c] * (1.0 - 2.0 * h) + (pwn - A) + (swp - B)


def smooth_f(h, beta):
    f, _ = smooth_values(h, beta)
    return f


def smooth_full(h, beta):
    f, w = smooth_values(h, beta)
    return f, gradient_from_weights(h, w)


def objective_and_grad(x, beta, target):
    n = x.size + 1
    h = np.empty(n, dtype=float)
    h[:-1] = x
    last = target - float(x.sum())
    h[-1] = last
    f, w = smooth_values(h, beta)
    B = 1.0e5
    add = 0.0
    if last < 0.0:
        f += 0.5 * B * last * last
        add = -B * last
    elif last > 1.0:
        e = last - 1.0
        f += 0.5 * B * e * e
        add = -B * e
    g = gradient_from_weights(h, w)
    gx = g[:-1] - g[-1]
    if add != 0.0:
        gx = gx + add
    return f, gx


def exact_c5(h, n):
    if n <= 0:
        return float("inf")
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def projected_smooth(h, beta, maxit, deadline):

    if maxit <= 0 or n < 2:
        return h
    n = len(h)
    f, g = smooth_full(h, beta)
    v = np.zeros(n, dtype=float)

    mu = 0.85
    for _ in range(int(maxit)):
        if time.time() > deadline:

        a = alpha
        ok = False
        for _ in range(32):

            fp = smooth_f(hp, beta)

                h = hp
                f = fp
                ok = True
                alpha = max(1.1 * a, 1e-10)

            a *= 0.5
        if not ok:
            a = 0.1 * max(alpha, 1e-8)
            for _ in range(12):

                fp = smooth_f(hp, beta)

                    h = hp
                    f = fp
                    ok = True
                    alpha = max(a, 1e-10)

    return h


def continuous_polish(h, n, deadline, beta_list=None, maxfun=350, do_pg=True):
    h = project_feasible(h, n)
    if n <= 1:
        return h, exact_c5(h, n)
    target = 0.5 * n
    best = h.copy()
    bc = exact_c5(h, n)
    if beta_list is None:
        beta_list = [20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    bounds = [(0.0, 1.0)] * (n - 1)
    mf = int(max(60, maxfun))
    if n < 120:
        mf *= 2
    for beta in beta_list:
        if time.time() > deadline - 3.0:
            break
        x0 = h[:-1].copy()
        try:
            res = minimize(
                objective_and_grad,
                x0,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                args=(float(beta), target),
                options={"maxiter": max(40, mf // 2), "maxfun": mf,
                         "ftol": 1e-12, "maxls": 40},
            )
            h_new = np.empty(n, dtype=float)
            h_new[:-1] = res.x
            h_new[-1] = target - float(res.x.sum())
            h_new = project_feasible(h_new, n)
        except Exception:
            h_new = h
        if not np.all(np.isfinite(h_new)):
            h_new = h
        c = exact_c5(h_new, n)

            bc = c
            best = h_new.copy()
        h = h_new

            c2 = exact_c5(h2, n)
            if np.isfinite(c2) and c2 < bc:
                bc = c2
                best = h2.copy()
            h = h2

def exact_active_descent(h, deadline, maxit=500):
    h = project_feasible(h)
    n = len(h)
    if n <= 1:
        return h, compute_q_natural(h).max()
    T = float(compute_q_natural(h).max())
    bestT = T
    alpha = 0.01
    for _ in range(int(maxit)):
        if time.time() > deadline:
            break
        q = compute_q_natural(h)
        T = float(q.max())
        if T < bestT:
            bestT = T
        s = (q - T) / 1.0e-3
        e = np.exp(s)
        w = e / e.sum()
        g = gradient_from_weights(h, w)
        g = g - g.mean()
        if float(np.linalg.norm(g)) < 1e-10:
            break
        a = alpha
        ok = False
        for _ in range(24):
            hp = project_feasible(h - a * g)
            T2 = float(compute_q_natural(hp).max())
            if T2 < T - 1e-11:
                h = hp
                T = T2
                if T < bestT:
                    bestT = T
                ok = True
                alpha = max(1.05 * a, 1e-12)
                break
            a *= 0.5
        if not ok:
            alpha *= 0.5
            if alpha < 1e-10:
                break
    return h, bestT


def binary_projection(h, n):
    if n % 2:
        return np.full(n, 0.5)
    target = n // 2
    out = np.zeros(n, dtype=float)
    idx = np.argsort(-h, kind="stable")
    out[idx[:target]] = 1.0
    return out


def build_D(h, n):
    m = 2 * n - 1
    c = n - 1
    D = np.zeros((n, m), dtype=float)
    for u in range(1, n):
        N = n - u
        a = h[:N]
        b = h[u:u + N]
        ip = c + u
        im = c - u
        D[:N, ip] += 1.0 - b
        D[u:u + N, ip] -= a
        D[u:u + N, im] += 1.0 - a
        D[:N, im] -= b
    return D


def binary_find_swap(q, h, D, n, deadline):
    c = n - 1
    T = float(q.max())
    zeros = np.nonzero(h < 0.5)[0]
    ones = np.nonzero(h > 0.5)[0]
    if zeros.size == 0 or ones.size == 0:
        return None
    bestM = T
    best = None
    for i in zeros:
        if time.time() > deadline:
            break
        Di = D[i]
        Dones = D[ones]
        bmax = np.max(q[None, :] + Di[None, :] - Dones, axis=1)
        cand = np.nonzero(bmax <= T + 1e-8)[0]
        if cand.size:
            if cand.size > 25:
                cand = cand[np.argpartition(bmax, 24)[:25]]
            for idx in cand:
                j = ones[idx]
                row = q + Di - D[j]
                d = int(j - i)
                row[c + d] += 1.0
                row[c - d] -= 1.0
                M = float(row.max())
                if M < bestM - 1e-9:
                    bestM = M
                    best = (M, int(i), int(j))
                    if bestM <= T - 1.0 + 1e-7:
                        return best
    return best


def binary_d_search(h, n, deadline, max_moves=60, max_restarts=8):
    if n % 2:
        return project_feasible(h, n), exact_c5(h, n)
    h = binary_projection(h, n)
    q = compute_q_natural(h)

    cur = h.copy()

        if mv is None:
            if restarts >= max_restarts:

            restarts += 1
            cur = cur.copy()
            for _ in range(int(rng.randint(2, 6))):
                z = np.nonzero(cur < 0.5)[0]

        T = float(q.max())
        if T < bestT:
            bestT = T
            besth = h.copy()
        cur = h.copy()
        moves += 1
    return besth, bestT * (2.0 / n)
