   ```python
   import numpy as np
   from scipy.optimize import minimize
   import time

   def compute_c5_vec(h, n):
       dx = 2.0 / n
       return np.correlate(h, 1.0 - h, mode='full') * dx

   def compute_c5(h, n):
       return float(np.max(compute_c5_vec(h, n)))

   def project_feasible(h, n):
       target = 0.5 * n
       h = np.clip(h.copy(), 0.0, 1.0)
       for _ in range(2000):
           d = target - h.sum()
           if abs(d) < 1e-16:
               break
           if d > 0:
               r = 1.0 - h
               s = r.sum()
               if s < 1e-16:
                   break
               h += (d / s) * r
           else:
               s = h.sum()
               if s < 1e-16:
                   break
               h += (d / s) * h
       return np.clip(h, 0.0, 1.0)

   def grad_cmax(h, n, alpha):
       """Gradient of smooth max of c_k(h) with temperature 1/alpha."""
       c = compute_c5_vec(h, n)
       mx = c.max()
       sh = alpha * (c - mx)
       smx = sh.max()
       w = np.exp(sh - smx)
       w /= w.sum()
       dx = 2.0 / n
       g = np.zeros(2 * n - 1)  # not needed, just for clarity
       grad = np.zeros(n)
       for m in range(2 * n - 1):
           if w[m] < 1e-14:
               continue
           k = m - (n - 1)
           wm = w[m] * dx
           if k >= 0:
               nv = n - k
               grad[:nv] += wm * (1.0 - h[k:])
               grad[k:] += wm * (-h[:nv])
           else:
               kk = -k
               nv = n - kk
               grad[kk:] += wm * (1.0 - h[:nv])
               grad[:nv] += wm * (-h[kk:])
       return grad

   def penalty_obj(x, n, alpha, mu):
       c = compute_c5_vec(x, n)
       m = c.max()
       sh = alpha * (c - m)
       smx = sh.max()
       val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
       val += mu * (x.sum() - 0.5 * n) ** 2
       return float(val)

   def penalty_grad(x, n, alpha, mu):
       g = grad_cmax(x, n, alpha)
       g += 2.0 * mu * (x.sum() - 0.5 * n)
       return g

   def lbfgs_optimize(h, n, deadline, alphas=None, time_per=15):
       if alphas is None:
           alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
       best_c = compute_c5(h, n)
       bounds = [(0.0, 1.0)] * n
       for alpha in alphas:
           t_end = min(time.time() + time_per, deadline)
           if time.time() >= t_end:
               break
           mu = 1.0 + alpha * 1e-5
           try:
               res = minimize(penalty_obj, h, jac=penalty_grad,
                             args=(n, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 500, 'ftol': 1e-18,
                                      'gtol': 1e-14, 'maxls': 50})
               h = project_feasible(res.x, n)
               c = compute_c5(h, n)
               if c < best_c:
                   best_c = c
           except Exception:
               pass
       return h, best_c

   def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
       def obj(x):
           c = compute_c5_vec(x, n)
           m = c.max()
           sh = alpha * (c - m)
           smx = sh.max()
           return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
       
       def obj_grad(x):
           return grad_cmax(x, n, alpha)
       
       def eq_c(x):

       def eq_j(x):
           return np.ones(n)

       except Exception:

   def directed_pair_optimize(h, n, deadline, alpha=200000):
       """O(n) per iteration pair search using smooth max gradient."""
   ```
