def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    target=0.5*n
    for _ in range(5):
        s=float(np.sum(h))
        d=target-s
        if abs(d)<1e-13: break
        if d>0:
            denom=n-s
            if denom<=0: break
            h += (d/denom)*(1.0-h)
        else:
            if s<=0: break
            h += (d/s)*h
    return np.clip(h,0.0,1.0)

def project_feasible(h,n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    for _ in range(5):
        s = float(np.sum(h))
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0.0:

        else:

def refresh_state(h,n,dx):
    v = 1.0 - h

import numpy as np
import time
import math

COARSE_N = 21
GOLDEN_ITERS = 26
GOLDEN_INV_PHI = 0.6180339887498949


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    for _ in range(5):
        s = float(np.sum(h))
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            denom = n - s
            if denom <= 0.0:
                break
            h += (d / denom) * (1.0 - h)
        else:
            if s <= 0.0:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)


def c5_of(h, n):
    if n <= 0:
        return float("inf")
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def refresh_state(h, n, dx):
    v = 1.0 - h
    q = np.correlate(h, v, mode="full") * dx
    return q, v, float(np.max(q))


def eval_pair_value(base_q, lin_dx, delta, idx0, ida, idb, dx, vals, tlin):
    np.multiply(lin_dx, delta, out=tlin)
    vals[:] = base_q
    vals += tlin
    d2 = dx * delta * delta
    vals[idx0] -= 2.0 * d2
    vals[ida] += d2
    vals[idb] += d2
    return float(np.max(vals))


def apply_pair_delta(base_q, lin_dx, delta, idx0, ida, idb, dx, tlin):
    if delta == 0.0:
        return
    np.multiply(lin_dx, delta, out=tlin)
    base_q += tlin
    d2 = dx * delta * delta
    base_q[idx0] -= 2.0 * d2
    base_q[ida] += d2
    base_q[idb] += d2


def optimize_pair(h, v, base_q, n, dx, i, j, lin_buf, vals_buf, tlin_buf):
    if i == j:
        return False
    ai = h[i]
    aj = h[j]
    s = ai + aj
    if s < 0.0:
        s = 0.0
    elif s > 2.0:
        s = 2.0

    lo_a = max(0.0, s - 1.0)
    hi_a = min(1.0, s)
    if hi_a - lo_a < 1e-14:
        return False
    if ai < lo_a - 1e-10 or ai > hi_a + 1e-10:
        return False

    lo_d = lo_a - ai
    hi_d = hi_a - ai
    span = hi_d - lo_d
    if span < 1e-14:
        return False

    cur_c = float(np.max(base_q))

    lin = lin_buf
    lin.fill(0.0)
    vrev = v[::-1]
    lin[i:i + n] += vrev
    lin[j:j + n] -= vrev
    s1 = n - 1 - i
    s2 = n - 1 - j
    lin[s1:s1 + n] -= h
    lin[s2:s2 + n] += h
    lin *= dx

    idx0 = n - 1
    ida = idx0 + (i - j)
    idb = idx0 + (j - i)

    best_delta = 0.0
    best_val = cur_c

    for gg in range(COARSE_N):
        dd = lo_d + span * (float(gg) / float(COARSE_N - 1))
        if abs(dd) < 1e-16:
            continue
        val = eval_pair_value(base_q, lin, dd, idx0, ida, idb, dx, vals_buf, tlin_buf)
        if val < best_val - 1e-13:
            best_val = val
            best_delta = dd

    if best_val >= cur_c - 1e-12:
        return False

    width = max(1e-7, 0.09 * span)
    glo = max(lo_d, best_delta - width)
    ghi = min(hi_d, best_delta + width)

    if ghi - glo > 1e-14:
        c1 = ghi - GOLDEN_INV_PHI * (ghi - glo)
        c2 = glo + GOLDEN_INV_PHI * (ghi - glo)
        f1 = eval_pair_value(base_q, lin, c1, idx0, ida, idb, dx, vals_buf, tlin_buf)
        f2 = eval_pair_value(base_q, lin, c2, idx0, ida, idb, dx, vals_buf, tlin_buf)
        if f1 < best_val:
            best_val, best_delta = f1, c1
        if f2 < best_val:
            best_val, best_delta = f2, c2

        for _ in range(GOLDEN_ITERS):
            if ghi - glo < 1e-12:
                break
            if f1 < f2:
                ghi = c2
                c2 = c1
                f2 = f1
                c1 = ghi - GOLDEN_INV_PHI * (ghi - glo)
                f1 = eval_pair_value(base_q, lin, c1, idx0, ida, idb, dx, vals_buf, tlin_buf)
            else:
                glo = c1
                c1 = c2
                f1 = f2
                c2 = glo + GOLDEN_INV_PHI * (ghi - glo)
                f2 = eval_pair_value(base_q, lin, c2, idx0, ida, idb, dx, vals_buf, tlin_buf)
            if f1 < best_val:
                best_val, best_delta = f1, c1
            if f2 < best_val:
                best_val, best_delta = f2, c2

    for dd in (lo_d, hi_d, best_delta):
        if abs(dd) < 1e-16:
            continue
        val = eval_pair_value(base_q, lin, dd, idx0, ida, idb, dx, vals_buf, tlin_buf)
        if val < best_val - 1e-14:
            best_val = val
            best_delta = dd

    if abs(best_delta) < 1e-15:
        return False
    if best_delta < lo_d:
        best_delta = lo_d
    elif best_delta > hi_d:
        best_delta = hi_d

    val = eval_pair_value(base_q, lin, best_delta, idx0, ida, idb, dx, vals_buf, tlin_buf)
    if val >= cur_c - 1e-13:
        return False

    apply_pair_delta(base_q, lin, best_delta, idx0, ida, idb, dx, tlin_buf)
    a_new = ai + best_delta
    b_new = s - a_new
    h[i] = a_new
    h[j] = b_new
    v[i] = 1.0 - a_new
    v[j] = 1.0 - b_new
    return True


def coordinate_pass(h, v, base_q, n, dx, deadline, rng, lin_buf, vals_buf, tlin_buf):
    p = int(np.argmax(np.minimum(h, 1.0 - h)))
    order = rng.permutation(n)
    improved = False
    for ii in order:
        if time.time() > deadline:
            break
        i = int(ii)
        if i == p:
            continue
        if min(float(h[p]), 1.0 - float(h[p])) < 1e-8:
            p = int(np.argmax(np.minimum(h, 1.0 - h)))
            if i == p:
                continue
        if optimize_pair(h, v, base_q, n, dx, i, p, lin_buf, vals_buf, tlin_buf):
            improved = True
    return improved


def adjacent_pass(h, v, base_q, n, dx, deadline, rng, lin_buf, vals_buf, tlin_buf):
    order = rng.permutation(max(0, n - 1))
    improved = False
    for ii in order:
        if time.time() > deadline:
            break
        i = int(ii)
        j = i + 1
        if j >= n:
            continue
        if optimize_pair(h, v, base_q, n, dx, i, j, lin_buf, vals_buf, tlin_buf):
            improved = True
    return improved


def active_pass(h, v, base_q, n, dx, deadline, rng, lin_buf, vals_buf, tlin_buf):
    top_lags = np.argsort(base_q, kind="mergesort")[-10:][::-1]
    improved = False
    max_pairs = 80
    for k in top_lags:
        if time.time() > deadline:
            break
        L = abs(int(k) - (n - 1))
        if L == 0:
            continue
        m = int(n) - L
        if m <= 0:
            continue
        if m <= max_pairs:
            pos_iter = range(m)
        else:
            diff = np.abs(h[:m] - h[L:L + m])
            pos_iter = np.argsort(diff, kind="mergesort")[::-1][:max_pairs]
        for pos in pos_iter:
            if time.time() > deadline:
                break
            i = int(pos)
            j = i + L
            if abs(float(h[i]) - float(h[j])) < 1e-10:
                continue
            if optimize_pair(h, v, base_q, n, dx, i, j, lin_buf, vals_buf, tlin_buf):
                improved = True
    return improved


def random_pass(h, v, base_q, n, dx, deadline, rng, lin_buf, vals_buf, tlin_buf):
    improved = False
    for _ in range(60):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        if optimize_pair(h, v, base_q, n, dx, i, j, lin_buf, vals_buf, tlin_buf):
            improved = True
    return improved


def local_search(h, n, deadline, rng):
    dx = 2.0 / n
    h = project_feasible(h, n)
    base_q, v, cur = refresh_state(h, n, dx)
    best_h = h.copy()
    best_c = cur

    m = 2 * n - 1
    lin_buf = np.empty(m)
    vals_buf = np.empty(m)
    tlin_buf = np.empty(m)

    no_improve = 0
    pass_no = 0

    while time.time() < deadline:
        pass_no += 1
        improved = False

        if pass_no % 4 == 0:
            improved = active_pass(h, v, base_q, n, dx, deadline, rng,
                                   lin_buf, vals_buf, tlin_buf)
        elif pass_no % 4 == 1:
            improved = coordinate_pass(h, v, base_q, n, dx, deadline, rng,
                                       lin_buf, vals_buf, tlin_buf)
        elif pass_no % 4 == 2:
            improved = adjacent_pass(h, v, base_q, n, dx, deadline, rng,
                                     lin_buf, vals_buf, tlin_buf)
        else:
            improved = random_pass(h, v, base_q, n, dx, deadline, rng,
                                   lin_buf, vals_buf, tlin_buf)

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if improved or pass_no % 5 == 0:
            if abs(float(np.sum(h)) - 0.5 * n) > 1e-11:
                h = project_feasible(h, n)
            base_q, v, cur = refresh_state(h, n, dx)
            if cur < best_c - 1e-13:
                best_c = cur
                best_h = h.copy()

        if no_improve >= 7:
            if time.time() > deadline - 100:
                break
            scale = 0.003 + 0.002 * float(min(max(no_improve - 7, 0), 25))
            hp = best_h + rng.normal(0.0, scale, size=n)
            h = project_feasible(hp, n)
            base_q, v, cur = refresh_state(h, n, dx)
            if cur < best_c - 1e-13:
                best_c = cur
                best_h = h.copy()
            no_improve = 0
            pass_no = 0
            continue

        if pass_no > 20000:
            break

    base_q, v, cur = refresh_state(h, n, dx)
    if cur < best_c - 1e-13:
        best_c = cur
        best_h = h.copy()

    hp = project_feasible(best_h, n)
    cp = c5_of(hp, n)
    if cp <= best_c + 1e-11:
        best_h, best_c = hp, cp
    else:
        best_h, best_c = project_feasible(best_h, n), best_c
        bp = c5_of(best_h, n)
        if bp < best_c:
            best_c = bp
    return best_h, float(best_c)


def structure_start(n, rng):
    best = np.full(n, 0.5)
    best_c = c5_of(best, n)
    if n > 2:
