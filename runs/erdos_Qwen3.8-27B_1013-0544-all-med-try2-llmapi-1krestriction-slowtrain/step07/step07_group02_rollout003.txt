def smooth_obj_grad(h,beta,n):
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    dx = 2.0 / n
    shift = float(np.max(c) * dx)
    z = beta * (c * dx - shift)
    p = np.exp(z)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 0.0:
        p = np.zeros_like(c); p[int(np.argmax(c))] = 1.0
    else:
        p /= s
    a = np.zeros(n); b = np.zeros(n)
    comp = 1.0 - h
    off = n - 1
    threshold = 1e-12
    for idx in range(len(p)):
        wt = p[idx]
        if wt <= threshold: continue
        lag = idx - off
        if lag >= 0:
            sl = n - lag
            a[lag:] += wt * comp[:sl]
            b[:sl] += wt * h[lag:]
        else:
            r = -lag
            sl = n - r
            a[:sl] += wt * comp[r:]
            b[r:] += wt * h[:sl]
    grad = (a - b) * dx
    val = shift + math.log(s if s>0 else 1.0) / beta
    return val, grad, shift

def exact_sum(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    t = n / 2.0
    for _ in range(24):
        d = t - h.sum()
        if abs(d) < 1e-14:

        idx = np.flatnonzero(h < 1.0) if d > 0 else np.flatnonzero(h > 0.0)
        if len(idx) == 0:
            break
        h[idx] += d / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return exact_sum(h, n)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def smooth_obj_grad(h, beta, n):
    c = np.asarray(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
    dx = 2.0 / n
    shift = float(np.max(c) * dx)
    z = beta * (c * dx - shift)
    p = np.exp(z)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 0.0:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        p /= s
    a = np.zeros(n)
    b = np.zeros(n)
    comp = 1.0 - h
    off = n - 1
    for idx, wt in enumerate(p):
        if wt <= 1e-12:
            continue
        lag = idx - off
        if lag >= 0:
            sl = n - lag
            a[lag:] += wt * comp[:sl]
            b[:sl] += wt * h[lag:]
        else:
            r = -lag
            sl = n - r
            a[:sl] += wt * comp[r:]
            b[r:] += wt * h[:sl]
    val = shift + math.log(s) / beta
    return val, (a - b) * dx, shift


def slq_fgf(x):
    v, g, _ = smooth_obj_grad(x, SLQ['beta'], SLQ['n'])
    return v, g


def slq_cf(x):
    return x.sum() - SLQ['target']


def slq_cg(x):
    return np.ones_like(x)


def slsqp_smooth(h, n, beta, maxit, deadline, ftol=1e-13):
    if time.time() > deadline:
        return h
    SLQ['beta'] = int(beta)
    SLQ['n'] = int(n)
    SLQ['target'] = n / 2.0
    try:
        res = minimize(slq_fgf, h, jac=True, bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': slq_cf, 'jac': slq_cg}],
                       method='SLSQP', options={'maxiter': maxit, 'ftol': ftol, 'disp': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def proj_grad_smooth(h, n, beta, maxit, deadline, eta=0.03):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    val, grad, _ = smooth_obj_grad(h, beta, n)
    el = eta
    stall = 0
    for _ in range(maxit):
        if time.time() > deadline:
            break
        mg = float(np.max(np.abs(grad)))
        if mg < 1e-12:
            break
        u = grad / mg
        ok = False
        for fac in (1.0, 0.5, 0.25, 0.1, 0.05):
            y = project_feasible(h - u * el * fac * mg, n)
            if np.max(np.abs(y - h)) < 1e-14:
                break
            v2, g2, m2 = smooth_obj_grad(y, beta, n)
            if v2 < val - 1e-13:
                h = y
                val = v2
                grad = g2
                el = min(0.15, el * 1.05)
                if m2 < best_c:
                    best_c = m2
                    best_h = h.copy()
                ok = True
                break
        if ok:
            stall = 0
        else:
            el *= 0.5
            stall += 1
            if stall > 5:
                break
    return best_h, best_c


def to_binary(h, n):
    hb = np.zeros(n)
    idx = np.argsort(np.asarray(h, dtype=float))[::-1]
    hb[idx[:n // 2]] = 1.0
    return hb


def binary_search(h0, n, deadline, seed):
    h = to_binary(h0, n).astype(np.int8)
    cur = int(np.correlate(h, 1 - h, mode='full').max())
    best = h.copy()
    best_count = cur
    rng = np.random.RandomState(seed)
    temp = max(1.0, cur * 0.04)
    no = 0
    low = high = np.array([], dtype=int)
    for it in range(300000):
        if time.time() > deadline:
            break
        if it % 25 == 0:
            _, grad, _ = smooth_obj_grad(h.astype(float), 5, n)
            low = np.argsort(grad)
            high = np.argsort(-grad)
        pairs = []
        for i in low[:8]:
            if h[i] != 0:
                continue
            for j in high[:8]:
                if h[j] == 1 and i != j:
                    pairs.append((i, j))
                if len(pairs) >= 12:
                    break
            if len(pairs) >= 12:
                break
        zis = np.flatnonzero(h == 0)
        ois = np.flatnonzero(h == 1)
        for _ in range(3):
            if len(zis) and len(ois):
                pairs.append((zis[rng.randint(len(zis))], ois[rng.randint(len(ois))]))
        acc = False
        for i, j in pairs:
            if h[i] != 0 or h[j] != 1:
                continue
            h[i] = 1
            h[j] = 0
            cc = np.correlate(h, 1 - h, mode='full')
            nc = int(cc.max())
            accept = False
            if nc < cur:
                accept = True
            elif nc == cur and rng.rand() < 0.2:
                accept = True
            elif nc - cur <= temp:
                if math.exp(-(nc - cur) / max(temp, 1e-9)) > rng.rand():
                    accept = True
            if accept:
                cur = nc
                acc = True
                if nc < best_count:
                    best_count = nc
                    best = h.copy()
                temp = max(0.2, temp * 0.995)
                break
            h[i] = 0
            h[j] = 1
        if not acc:
            no += 1
            temp = max(0.2, temp * 0.998)
            if no > 800:
                break
    return best, best_count * (2.0 / n)


def solve_milp(n, Tlim, tlim):
    if n < 10 or n % 2 or tlim < 8 or Tlim is None or Tlim <= 0:
        return None
    try:
        P = n * (n - 1) // 2
        h = cp.Variable(n, boolean=True)
        z = cp.Variable(P, boolean=True)
        t = cp.Variable(nonneg=True)
        row = np.empty(P, dtype=int)
        col = np.empty(P, dtype=int)
        p = 0
        for i in range(n - 1):
            l = n - 1 - i
            row[p:p + l] = i
            col[p:p + l] = i + 1 + np.arange(l)
            p += l
        cons = [cp.sum(h) == n // 2, t >= 0,
                z <= h[row], z <= h[col], z >= h[row] + h[col] - 1]
        for d in range(1, n):
            ii = np.arange(n - d)
            idx = (ii * (2 * n - ii - 1) // 2) + (d - 1)
            sz = cp.sum(z[idx])
            cons.append(t >= cp.sum(h[d:]) - sz)
            cons.append(t >= cp.sum(h[:-d]) - sz)
        cons.append(t <= float(Tlim))
        prob = cp.Problem(cp.Minimize(t), cons)
        for sname in MSOLS:
            sol = getattr(cp, sname, None)
            if sol is None:
                continue
            ok = False
            for kw in ({'time_limit': int(tlim)}, {'timeLimit': int(tlim)}, {'seconds': int(int(tlim))}):
                try:
                    prob.solve(solver=sol, verbose=False, **kw)
                    ok = True
                    break
                except Exception:
                    pass
            if not ok:
                if tlim <= 8:
                    try:
                        prob.solve(solver=sol, verbose=False)
                    except Exception:
                        continue
                else:
                    continue
            hv = h.value
            if hv is None:
                continue
            hv = np.asarray(hv, dtype=float)
            if not np.all(np.isfinite(hv)):
                continue
            hb = (hv > 0.5).astype(float)
            diff = n // 2 - int(hb.sum())
            if diff:
                if diff > 0:
                    hb[np.argsort(hv)[::-1][:diff]] = 1.0
                else:
                    hb[np.argsort(hv)[:abs(diff)]] = 0.0
            c5 = compute_c5(hb, n)
            if c5 <= (Tlim + 1e-7) * (2.0 / n):
                return hb
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10.0
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 20:
        n = 400
        h = np.full(n, 0.5)

    h = exact_sum(project_feasible(h, n), n)
    best_h = h.copy()
    best_n = n
    best_c = compute_c5(h, n)

    for nh in (100, 120, 150, 160, 200, 300):
        if nh == n or nh < 50:
            continue
        xo = np.linspace(0.0, 2.0, n)
        xn = np.linspace(0.0, 2.0, nh)
        hh = exact_sum(project_feasible(np.interp(xn, xo, h), nh), nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c = ch
            best_h = hh
            best_n = nh

    if total > 220:
        mb = min(total * 0.38, 320.0)
        spent = 0.0
        for nh, fr in ((100, 0.22), (120, 0.24), (160, 0.30), (200, 0.24)):
            rem = deadline - time.time() - 220.0
            if rem < 25.0 or mb - spent < 25.0:
                break
            talloc = min(mb * fr + 8.0, rem * 0.7)
            Tl = int(math.floor(TARGET * nh / 2.0 - 1e-9))
            if Tl <= 0:
                continue
            tm = time.time()
            hm = solve_milp(nh, Tl, talloc)
            spent += time.time() - tm
            if hm is not None:
                ch = compute_c5(hm, nh)
                if ch < best_c:
                    best_c = ch
                    best_h = hm
                    best_n = nh

    if time.time() < deadline - 20:
        h = best_h.copy()
        n = best_n

        for beta in (30, 120, 400):
            if time.time() > deadline - 45:
                break
            h2, c2 = proj_grad_smooth(h, n, beta, 70, deadline - 45, 0.03)
            if c2 < best_c:
                best_c = c2
                best_h = h2
            h = h2

        maxit = 18 if n >= 350 else 35
        for beta in (100, 400, 1200):
            if time.time() > deadline - 70:
                break
            h2 = slsqp_smooth(h, n, beta, maxit, deadline - 70)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
            h = h2

        for trial in range(3):
            if time.time() > deadline - 70:
                break
            amp = 0.002 + 0.002 * trial
            hp = project_feasible(h + amp * rng.randn(n), n)
            hp, cp2 = proj_grad_smooth(hp, n, 80, 30, deadline - 70, 0.02)
            if cp2 < best_c:
                best_c = cp2
                best_h = hp
            h3 = slsqp_smooth(hp, n, 600, 18, deadline - 70)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c = c3
                best_h = h3
                h = h3

        if n % 2 == 0 and time.time() < deadline - 40:
            hb, cb = binary_search(h, n, deadline - 40, seed + 1)
            if cb < best_c:
                best_c = cb
                best_h = hb

    best_h = exact_sum(project_feasible(best_h, best_n), best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
