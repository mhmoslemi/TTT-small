import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_terms(h, i, n):
    ks = np.arange(-(n - 1), n)
    j1 = i - ks
    mask1 = (j1 >= 0) & (j1 < n)
    j2 = i + ks
    mask2 = (j2 >= 0) & (j2 < n)
    terms = np.zeros(2 * n - 1)
    terms[mask1] += (1.0 - h[j1[mask1]])
    terms[mask2] -= h[j2[mask2]]
    return terms

def coord_descent(h, n, n_cand=300, max_passes=500, deadline=None):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = c_raw.max() * dx
    best_h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n - 1):
            if deadline is not None and time.time() > deadline:
                break
            ts = compute_terms(h, n - 1, n)
            ti = compute_terms(h, i, n)
            base = ti - ts
            d_min = max(-h[i], h[n - 1] - 1.0)
            d_max = min(1.0 - h[i], h[n - 1])
            if d_max - d_min < 1e-12:
                continue
            deltas = np.linspace(d_min, d_max, n_cand)
            c_new = c_raw[None, :] + deltas[:, None] * base[None, :]
            c_new[:, n - 1] -= 2.0 * deltas[:, None] ** 2
            obj = (c_new * dx).max(axis=1)
            bi = int(np.argmin(obj))
            if obj[bi] < c_raw.max() * dx - 1e-15:
                d = deltas[bi]
                c_raw += d * base
                c_raw[n - 1] -= 2.0 * d * d
                h[i] += d
                h[n - 1] -= d
                improved = True
                cv = c_raw.max() * dx
                if cv < best_c - 1e-15:
                    best_c = cv
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-12:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _obj_pen(x, n, alpha, mu):
    v = smooth_max_obj(x, n, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return v

def _grad_pen(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_run(h, n, deadline, max_iters=300):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 50000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_obj_pen, h, jac=_grad_pen, args=(n, alpha, 10.0),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

        pass
