import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def adam_optimize(h, n, deadline, alpha_init=10000.0, alpha_final=500000.0, lr_init=0.01):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 5000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.3 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def lbfgs_phase(h, n, deadline, alphas=None, max_iters=300):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:

        mu = mus[min(i, len(mus)-1)]
        def obj(x):
            return smooth_max_obj(x, n, alpha, mu)
        def gr(x):
            return smooth_max_grad(x, n, alpha, mu)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None, grad_guided=False, alpha=10000):
    h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            if grad_guided:
                g = grad_smooth_max(h, n, alpha)
                order = np.argsort(-np.abs(g))
            else:
                order = np.arange(n)
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline):
    h = h.copy()

    for _ in range(200):
        if time.time() > deadline:

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def adam_optimize(h, n, deadline, alpha_init=10000.0, alpha_final=500000.0, lr_init=0.01):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 5000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.3 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def lbfgs_phase(h, n, deadline, alphas=None, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    mus = [1.0, 5.0, 20.0, 100.0, 500.0, 5000.0, 50000.0]
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 2:
            break
        mu = mus[min(i, len(mus) - 1)]
        _GP['n'] = n
        _GP['alpha'] = alpha
        _GP['mu'] = mu
        try:
            res = minimize(_smooth_obj, h, jac=_smooth_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None, grad_guided=False, alpha=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            if grad_guided:
                g = grad_smooth_max(h, n, alpha)
                order = np.argsort(-np.abs(g))
            else:
                order = np.arange(n)
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(300):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if 0.01 < h[i] < 0.99:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-10:
                continue
            mask = (h > abs(delta)) & (h < 1.0 - abs(delta))
            mask[i] = False
            if not mask.any():
                mask = (h > 1e-6) & (h < 1.0 - 1e-6)
                mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if time.time() < deadline - 250:

            h_t, c_t = adam_optimize(h_interp, n_try, time.time() + alloc * 0.5,

                                       max_iters=100)
            if c_t < best_c:

    if time.time() < deadline - 120:

        best_h, best_c = adam_optimize(best_h, n_best, t4,

    # Phase 5: Pair search (120s)
    if time.time() < deadline - 50:
        t5 = min(time.time() + 120, deadline - 50)
        best_h, best_c = pair_search(best_h, len(best_h), t5,
                                     deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                     radius=80, grad_guided=True, alpha=100000)

                                     radius=50, grad_guided=True, alpha=200000)

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)

Now let me think about potential issues:
1. The `_GP` global dict approach should work for scipy's minimize.
2. The pair search with radius=80 on n=300 means each pair (i,j) with |i-j|<=80 is tried, which is about 300*160 = 48000 pairs per delta. With 7 deltas, that's 336K evaluations per pass. Each evaluation is O(n) for the correlate (actually O(n) since we use np.correlate which is O(n log n) or O(n²) depending on implementation). For n=300, np.correlate should be fast.

Actually, np.correlate with mode='full' for n=300 elements produces 599 output values. The computation is O(n * 2n) = O(n²) in the worst case, but numpy uses FFT for large arrays. For n=300, it's probably direct computation, so about 90K multiply-adds. That's very fast.

The bottleneck in pair search is the Python loop. For n=300 with radius=80, we have about 300 * (2*80+1) / 2 ≈ 24000 pairs per pass (each pair tried in both directions). That's manageable.

Let me also think about whether I should try a fundamentally different approach. 

One idea: **Subgradient method with momentum**. Since the objective is the max of linear functions, we can use the cutting plane method or subgradient descent. The advantage is that it naturally handles the max structure without smoothing.

But the gradient of the max is only defined at the non-differentiable points (where multiple shifts achieve the max). The smooth max approach effectively handles this by assigning weights to all shifts.

Another idea: **Alternating minimization**. Fix the "active" set of shifts, then optimize h to minimize the maximum over those shifts (which is a convex problem - minimize t subject to g_k(h) <= t for active k). Then update the active set.

This is essentially the bisection approach:
- Binary search on t
- For each t, check if there exists h with g_k(h) <= t for all k (feasibility check)
- This is a linear feasibility problem (since g_k is linear in h for fixed k... wait, is it?)

Actually, g_k(h) = ∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - (h*h)(k)

The term (h*h)(k) is quadratic in h! So g_k(h) = 1 - (h*h)(k) is concave in h (since (h*h)(k) = h^T A_k h is a quadratic form, and A_k has a specific structure).

Wait, let me reconsider. (h*h)(k) = Σ_i h[i] h[i+k] (discretized). This is NOT linear in h - it's quadratic. So the problem is NOT convex.

This means we can't use simple convex feasibility. The problem is non-convex, which is why we need the optimization approaches we're using.

Hmm, but wait. Let me re-examine:
g_k(h) = Σ_i h[i](1-h[i+k]) * dx = Σ_i h[i]*dx - Σ_i h[i]*h[i+k]*dx

The first term is linear (and equals 1 by the constraint). The second term is the autocorrelation, which is quadratic.

So g_k(h) = 1 - autcorr_k(h) where autcorr_k is quadratic.

Minimizing max_k (1 - autcorr_k(h)) is equivalent to maximizing min_k autcorr_k(h).

The autocorrelation at lag k is: autcorr_k(h) = Σ_i h[i]h[i+k] * dx

This is a quadratic form in h. Maximizing the minimum of quadratic forms is a non-convex problem in general.

OK so the problem is definitely non-convex, and our approach of using smooth max + gradient methods is appropriate.

Let me also consider: at the optimum, what does the solution look like? Given the constraint ∫h = 1 and h ∈ [0,1], the solution should be a function that's concentrated in some region but with some structure to minimize the maximum overlap.

For a binary solution (h ∈ {0,1}), the problem becomes: find a subset S of [0,2] with measure 1 that maximizes min_k |S ∩ (S+k)|.

This is related to the "differently shifted" problem in additive combinatorics. The optimal set for this might have a fractal-like structure.

Anyway, let me finalize my code. I think the key improvements over the previous version are:

2. More aggressive L-BFGS-B with higher alpha values
3. Better time allocation for higher resolution attempts
4. Higher alpha in the final pair search phases

Let me also add one more idea: **restart from a perturbation**. If we're stuck at a local optimum, we can add noise and re-optimize. This is cheap to try.
