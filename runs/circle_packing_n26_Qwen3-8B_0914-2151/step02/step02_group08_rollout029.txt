import numpy as np
import random
import math
from deap import base, creator, tools
import copy

def run_packing():
    num_circles = 26
    POP_SIZE = 100
    GENERATIONS = 300
    MUTATION_RATE = 0.15
    CROSSOVER_RATE = 0.9
    TOURNAMENT_SIZE = 3

    # Define a fitness function that maximizes the sum of radii,
    # while minimizing overlaps
    def evaluate(individual):
        # Extract centers and radii
        centers = individual[:num_circles * 2].reshape(num_circles, 2)
        radii = individual[num_circles * 2:].reshape(num_circles)

        # Check if all circles are inside the unit square
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 0.0, 0.0  # Invalid chromosome

        # Check for overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 0.0, 0.0  # Invalid chromosome

        # Compute the sum of radii as the objective function
        total_radius = np.sum(radii)
        return total_radius, total_radius  # Maximize sum of radii

    # Define the genetic algorithm
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))  # Maximize sum of radii
    creator.create("Individual", list, fitness=creator.FitnessMax)

    # Initialize the toolbox
    toolbox = base.Toolbox()

    # Define the chromosome (we use flat representation)
    def generate_individual():
        centers = np.random.rand(num_circles, 2) * 0.8 + 0.1
        radii = np.random.rand(num_circles) * 0.2 + 0.05
        return np.concatenate([centers.reshape(-1), radii])

    toolbox.register("individual", tools.initSingleValued, base.types.Float, n=num_circles * 3)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    # Define the fitness evaluation
    toolbox.register("evaluate", evaluate)

    # Genetic operators
    def mut_gaussian(ind, mu, sigma, indpb):
        for i in range(len(ind)):
            if random.random() < indpb:
                ind[i] = np.random.normal(mu, sigma, 1)[0]
        return ind,

    def cx_two_point(ind1, ind2):
        size = len(ind1)
        # Two-point crossover
        # Choose crossover points
        cx1 = random.randint(1, size - 1)
        cx2 = random.randint(cx1, size - 1)
        # Swap the subarrays
        ind1[cx1:cx2], ind2[cx1:cx2] = ind2[cx1:cx2].copy(), ind1[cx1:cx2].copy()
        return ind1, ind2

    toolbox.register("mate", cx_two_point)
    toolbox.register("mutate", mut_gaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=TOURNAMENT_SIZE)

    # Initialize population
    population = toolbox.population(n=POP_SIZE)

    # Run the algorithm
    for gen in range(GENERATIONS):
        offspring = list()
        for child in range(POP_SIZE):
            ind = toolbox.clone(random.choice(population))
            if random.random() < CROSSOVER_RATE:
                # Crossover
                ind1 = random.choice(population)
                ind2 = random.choice(population)
                child1, child2 = toolbox.mate(ind1, ind2)
                offspring.append(child1)
                offspring.append(child2)
            else:
                # Mutation
                ind = toolbox.mutate(ind)
                offspring.append(ind)

        # Evaluate and select
        fits = list()
