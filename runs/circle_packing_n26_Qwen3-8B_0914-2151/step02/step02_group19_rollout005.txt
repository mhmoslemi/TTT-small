import scipy.optimize as opt

def run_packing():

    radii = np.random.rand(num_circles) * 0.1  # initial guess for radii

    def objective(x):

        x_flat = x

        radii_flat = x_flat[num_circles * 2:].reshape((num_circles,))

        total_energy = 0.0

        # Distance to boundaries
        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            total_energy += max(0, (x - r)) + max(0, (1 - x - r)) + max(0, (y - r)) + max(0, (1 - y - r))

        # Circle-circle overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)

    bounds = []

    initial_guess = np.concatenate([centers.reshape(-1), radii])

    # Optimization with SLSQP method, better convergence
    result = opt.minimize(
