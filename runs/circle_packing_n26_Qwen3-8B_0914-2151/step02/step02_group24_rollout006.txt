    # With two layers: some bottom-heavy, others top-heavy
    # Adjusted to allow more compact packing

    r0 = 0.08  # Initial guess for radius at core
    dist = 1.3 * r0  # Spacing

        x = i * dist
        y = 0.2

        radii[i] = r0

        x = i * dist - dist / 2
        y = 0.6
        centers[13 + i] = [x, y]
        radii[13 + i] = r0

    # Third layer (upper third): hexagonal right to left
    for i in range(13):
        x = (12 - i) * dist
        y = 0.8

        radii[26 - i - 1] = r0

    def objective(x, centers, radii, num_circles):

        centers_flat = x_flat[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x_flat[num_circles * 2:].reshape((num_circles,))

        total_energy = 0.0

        for i in range(num_circles):
