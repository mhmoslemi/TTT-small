import cvxpy as cp

def run_packing():

    centers = []

        else:
            x = col + 0.3
        y = row + 0.1
        centers.append([x, y])

    # Define variables
    radii = cp.Variable(num_circles)
    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
