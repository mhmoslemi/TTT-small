import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define center for the central circle
    center_central = np.array([0.5, 0.5])

    # Calculate positions for surrounding circles in a hexagonal grid
    # Number of surrounding circles = 25
    # Arrange in hexagonal pattern
    angles = np.linspace(0, 2 * np.pi, 25, endpoint=False)
    radii = np.full(25, 0.1)  # Initial estimate

    centers = []
    for i in range(25):
        r = 0.1
        angle = angles[i]
        x = center_central[0] + r * np.cos(angle)
        y = center_central[1] + r * np.sin(angle)
        centers.append([x, y])

    centers = np.array(centers)
    radii = np.full(25, 0.1)

    # Add a central circle
    centers = np.vstack((centers, center_central))
    radii = np.append(radii, 0.1)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for maximization

    # Constraints
    def constraint_dist(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j] + 1e-8
        return 0

    # Bounds: radii >= 0
    bounds = [(0, 1) for _ in range(26)]

    # Initial guess for radii
    initial_guess = np.full(26, 0.1)

    # Optimization
    result = opt.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": constraint_dist}]
    )

    centers_result = centers
    radii_result = result.x
    sum_radii = np.sum(radii_result)

    return centers_result, radii_result, sum_radii
