from scipy.optimize import minimize, Bounds
import matplotlib.pyplot as plt

def initialize_grid(n, m):  # n rows, m columns
    spacing = 1.0 / (m - 1) if m > 1 else 1.0
    spacing_h = 1.0 / (n - 1) if n > 1 else 1.0
    centers = []
    for i in range(n):
        for j in range(m):
            x = j * spacing
            y = i * spacing_h
            centers.append((x, y))
    return np.array(centers)

def run_packing():

    n_rows, n_cols = 6, 4

    n = len(centers)

    initial_radii = np.full(n, 0.05)

    # Optimization variables
    # x = [x1, y1, r1, x2, y2, r2, ...]

    x0 = np.concatenate([centers.flatten(), initial_radii])

    def objective(x):

        return -np.sum(radii)  # We minimize to maximize

        return [

    bounds = bounds_func(x0)

        radii = x[idx+2*n:]
        # Boundary constraints
