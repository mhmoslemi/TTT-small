import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Define the layout of centers in a hybrid packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    row_count = 7  # rows
    col_count = 8  # columns per row
    grid_rows = [7, 8, 7, 8, 7, 8, 7]  # for total 26 circles

    # Create a hexagonal grid
    for i, row_len in enumerate(grid_rows):
        row_y = i * (1 / (row_count - 1))  # y-position of row
        for col in range(row_len):
            col_x = col * (1 / (col_count - 1)) if i % 2 == 0 else (col - 0.5) * (1 / (col_count - 1))
            centers[i * row_len + col] = [col_x, row_y]

    # Initial uniform radius based on nearest neighbor distance
    r0 = 0.04
    radii = np.full(num_circles, r0)

    # Bounds and constraints
    bounds_centers = [(0, 1) for _ in range(2 * num_circles)]
    bounds_radii = [(0, 1) for _ in range(num_circles)]

    # Objective function
    def objective(x):
        return -np.sum(x[num_circles:])  # Maximize sum of radii

    # Constraints: each circle within the square
    def in_square(x):
        centers_flat = x[:2 * num_circles]
        radii_flat = x[2 * num_circles:]
        for i in range(num_circles):
            x_i, y_i = centers_flat[2 * i], centers_flat[2 * i + 1]
            r_i = radii_flat[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return False
        return True

    # Constraints: distance between center to center >= sum of radii
    def no_overlap(x):
        centers_flat = x[:2 * num_circles]
        radii_flat = x[2 * num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                x_i, y_i = centers_flat[2 * i], centers_flat[2 * i + 1]
                x_j, y_j = centers_flat[2 * j], centers_flat[2 * j + 1]
                r_i, r_j = radii_flat[i], radii_flat[j]
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                if dist_sq < (r_i + r_j)**2 - 1e-12:
                    return False
        return True

    # Box constraints
    def box_constraints(x):
        return all(0 <= x[i] <= 1 for i in range(2 * num_circles)) and all(0 <= x[i] <= 1 for i in range(2 * num_circles, 2 * num_circles + num_circles))

    # Combine all constraints into a single function
    def constraints(x):
        return in_square(x) and no_overlap(x) and box_constraints(x)

    # Optimization
    initial_guess = np.concatenate([centers.flatten(), radii])

    result = minimize(
        objective,
        initial_guess,
        bounds=bounds_centers + bounds_radii,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-12, "maxiter": 1000}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    centers_opt = result.x[:2 * num_circles].reshape(-1, 2)
    radii_opt = result.x[2 * num_circles:]

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
