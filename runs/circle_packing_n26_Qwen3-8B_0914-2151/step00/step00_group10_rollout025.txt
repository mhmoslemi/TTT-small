import math

    # So place 2 rows of 13, 1 row of 12, then 1 row of 10

    rows = 3

    radii = []

            x = col * col_spacing
            y = row_idx * row_spacing
            row_centers.append([x, y])
            row_radii.append(initial_r)
        centers.extend(row_centers)
        radii.extend(row_radii)

    centers = np.array(centers)
    radii = np.array(radii)
