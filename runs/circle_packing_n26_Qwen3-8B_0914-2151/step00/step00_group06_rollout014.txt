from scipy.optimize import minimize

def run_packing():

    rows = 10
    cols_in_row = 3  # Max number of circles in a row in 1x1 square
    r = 0.1  # Initial radius (will be optimized)

    # Function to create initial hexagonal grid
    def hexagonal_grid(rows, cols_in_row):

        for row in range(rows):

                x = col + 0.5 * (row % 2)

                centers.append((x, y))

    def constraint_center(x, idx):

        y_pos = centers[idx][1]
        return x_pos - r >= -1e-12 and x_pos + r <= 1 + 1e-12 and y_pos - r >= -1e-12 and y_pos + r <= 1 + 1e-12

    def constraint_overlap(x, i, j):
