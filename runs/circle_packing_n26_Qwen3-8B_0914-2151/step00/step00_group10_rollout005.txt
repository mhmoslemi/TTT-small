import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total sum

    # Define a grid-based starting point
    def initialize_grid(n=26, grid_rows=5, grid_cols=5):
        centers = []
        for i in range(n):
            row = i // grid_cols
            col = i % grid_cols
            x = col / (grid_cols - 1)
            y = row / (grid_rows - 1)
            centers.append([x, y])
        return np.array(centers)

    centers = initialize_grid()

    # Define constraints
    def constraints(x):
        # Non-negativity of radii
        if any(r < 1e-8 for r in x):
            return np.inf
        # Check boundary constraints
        for i in range(len(x)):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf
        # Check overlap constraints
        for i in range(len(x)):
            for j in range(i+1, len(x)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-12:
                    return np.inf
        return 0

    # Initial guess for radii
    radii_initial = np.full(26, 0.05)

    # Run optimization
    result = minimize(
        objective,
        radii_initial,
        bounds=[(0, 1) for _ in range(26)],
        constraints={
            'type': 'ineq',
            'fun': constraints
        }
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, np.sum(result.x)
