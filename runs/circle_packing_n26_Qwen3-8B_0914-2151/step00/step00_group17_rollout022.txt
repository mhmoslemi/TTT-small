import scipy.optimize as opt

def run_packing():

        x = np.linspace(0, 1, n)
        y = np.linspace(0, max_y, n)
        centers = np.column_stack((x, y))
        return centers

    lattice_centers = generate_lattice(5)
    centers = lattice_centers.reshape(-1, 2)
    radii = np.full(centers.shape[0], 0.1)  # First guess

    # Since the square is constrained to [0, 1]^2, radii must be adjusted to fit
    # Let's wait for optimization to do the job

    # We need to maximize sum(radii) under:
    # 1. Non-overlapping between circles
    # 2. Each circle's edge does not cross the unit square
    # 3. All circles are within the square

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_non_overlap(i, j, radii):
        dist = np.linalg.norm(centers[i] - centers[j])
        return dist - (radii[i] + radii[j])
    
    # Constraints: non-overlapping
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({
                'type': 'ineq',
                'fun': lambda radii, i=i, j=j: constraint_non_overlap(i, j, radii)
            })

    # Boundary constraint: each circle must be within the unit square
    for i in range(len(centers)):
        # x - r >= 0
        def x_min_constraint(radii, i=i):
            return centers[i, 0] - radii[i]
        constraints.append({

            return 1 - centers[i, 0] - radii[i]
        constraints.append({

            return centers[i, 1] - radii[i]
