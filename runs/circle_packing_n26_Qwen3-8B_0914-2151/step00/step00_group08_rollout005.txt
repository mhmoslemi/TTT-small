import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():
    # Define grid of centers in hexagonal lattice within a square
    def generate_hexagonal_grid(num_circles):
        grid = []
        for i in range(num_circles):
            row = i // 5
            col = i % 5
            x = col + 0.5 * (row % 2)
            y = row
            grid.append((x, y))
        return np.array(grid)

    # Initial guess for circle centers using hexagonal lattice
    initial_centers = generate_hexagonal_grid(26)
    initial_radii = np.ones(26) * 0.1  # Initial guess for radii

    # Create an optimization problem
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def bounds(radii):
        return [(0, 1) for _ in range(26)]  # Radius must be non-negative and small enough

    # Nonlinear constraints for circle packing
    def circle_constraints(radii, centers):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        return constraints

    # Constraints that circles must be within the square
    def square_constraints(radii, centers):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r >= -1e-8)
            constraints.append(x + r <= 1 + 1e-8)
            constraints.append(y - r >= -1e-8)
            constraints.append(y + r <= 1 + 1e8)
        return constraints

    # Combine all constraints
    def all_constraints(radii, centers):
        all_cons = []
        all_cons.extend(circle_constraints(radii, centers))
        all_cons.extend(square_constraints(radii, centers))
        return all_cons

    # Create bounds and nonlinear constraints

        return all_constraints(radii, centers)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
