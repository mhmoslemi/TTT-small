from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    algorithm to maximize the total sum of radii while respecting constraints.
    """
    n = 26
    max_iter = 100  # Number of iterations for optimization

    # Initial placement: staggered grid
    rows = 6
    cols = 5
    spacing = 1.0 / (cols - 1)  # Col spacing
    stagger = spacing / 2  # Stagger offset

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    for row in range(rows):
        for col in range(cols):
            x = col * spacing
            y = row * spacing

                x += stagger  # Staggered layout
            centers[idx, 0] = x

            radii[idx] = 0.05  # Initial estimate

        return -np.sum(x[26:])  # x[0:26] is centers, x[26:] is radii

    # Constraints:
    def boundary(x):
        # Ensure all circles are within the square [0,1]×[0,1]
        # Extract centers and radii from x
        centers = x[:26].reshape(n, 2)
        radii = x[26:]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if (x - r < -1e-8 or x + r > 1 + 1e-8

        return 0

    def non_overlap(x):

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:

        return 0

    # Combine constraints into a single vectorized form

            return -1

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
