import numpy as np
import scipy.optimize as optimize

def run_packing():
    # We'll start with an initial hexagonal packing with 26 circles
    # This provides a good starting configuration to optimize from

    # Define initial positions for circles using a hexagonal packing
    num_circles = 26
    positions = []
    radii = []

    # Starting radius for the hexagonal packing
    r0 = 0.1

    # Build the hexagonal grid
    for i in range(num_circles):
        x = i % 7
        y = i // 7
        x_pos = x + 0.5 * (1 - r0)  # Shift to center the grid in [0,1]
        y_pos = y + 0.5 * (1 - r0)
        positions.append([x_pos, y_pos])
        radii.append(r0)

    # Convert to numpy arrays
    centers = np.array(positions, dtype=np.float64)
    radii = np.array(radii, dtype=np.float64)

    # Objective function: maximize sum of radii
    def objective(x):
        # x is a flattened array of [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
        # Reshape to centers and radii
        flat_centers = x[:2 * num_circles].reshape((num_circles, 2))
        flat_radii = x[2 * num_circles:].reshape((num_circles,))

        # Objective: maximize sum of radii
        return -np.sum(flat_radii)  # minimize negative sum (so we maximize)

    # Constraints:
    # 1. All circles are within the unit square
    # 2. No two circles overlap
    # We'll build the constraints dynamically

    # Bounds for all variables (x, y, r)
    bounds = []
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)])  # x, y, r for each circle

    result = optimize.minimize(

    optimal_centers = result.x[:2 * num_circles].reshape((num_circles, 2))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
