import numpy as np
from scipy.optimize import minimize
from scipy.optimize import NonlinearConstraint
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess: Hexagonal packing with spacing between centers as 0.2
    # This creates a grid of 5 rows with 5, 6, 5, 6, 5 circles respectively
    # Rows are spaced 0.25 apart vertically for better packing density

    centers = []
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col * 0.2
            y = row * 0.25
            centers.append([x, y])
    centers = np.array(centers)

    # Initial radii: All equal for simplicity
    initial_radii = np.full(n, 0.03)

    # Objective function: Maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Bounds for each circle's radius
    bounds = Bounds(0, 1.0)

    # Constraint: Non-overlapping between any pair of circles
    def non_overlap(x):
        # x is a 2n-dimensional array: first n elements are centers, next n are radii
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]

        # Check if any circle is outside the unit square
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf

        # Check if any pairs of circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                r_i = radii[i]
                r_j = radii[j]
                if dist_sq < (r_i + r_j) ** 2 - 1e-12:
                    return np.inf

        return 0

    # Constraint function for non-overlapping
    def constraint(x):
        return non_overlap(x)

    # Define constraints
    nonlinear_constraint = NonlinearConstraint(
        constraint,
        0,
        0,
        jac="2-point"
    )

    # Full optimization variable: flatten centers and radii
    x0 = np.concatenate((centers.flatten(), initial_radii))

    # Run the optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[nonlinear_constraint],
        method="SLSQP"
    )

    # Extract results
    centers = result.x[:n * 2].reshape(n, 2)
    radii = result.x[n * 2:]
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
