import math

def generate_hexagonal_grid(n_circles):

    radii = []

    for i in range(n_circles):

        angle = 2 * math.pi * i / n_circles
        r = 0.5 * (1 + 0.5 * i)  # Adjust radius to fit within the square
        x = r * np.cos(angle)
        y = r * np.sin(angle)
        # Map to unit square with some random jitter to avoid overlap
        x = x + np.random.uniform(-0.1, 0.1)
        y = y + np.random.uniform(-0.1, 0.1)

        x = np.clip(x, 0.0, 1.0)

        centers.append([x, y])
        radii.append(radius * (1 - 0.1 * i / n_circles))  # Adjust radii based on distance to boundary

    return np.array(centers), np.array(radii)

def run_packing():
