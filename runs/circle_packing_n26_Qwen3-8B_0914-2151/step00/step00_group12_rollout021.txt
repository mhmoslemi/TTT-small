import scipy.optimize as opt

def run_packing():

    num_circles = 26
    grid_width = 5
    grid_height = 6

    # Initial grid layout (simple hexagonal grid with some variation)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate initial hexagonal grid
    row = 0
    col = 0

        row = i // grid_width
        col = i % grid_width
        # Offsets for hexagonal grid
        x = col + 0.5 * (row % 2)
        y = row
        centers[i] = [x / (grid_width - 1), y / (grid_height - 1)]
        # Initialize radii with a small value
        radii[i] = 0.05

    # Scale and offset to fit within [0, 1]^2
    centers /= max(centers.max(), 1 - centers.min())
    radii *= 2  # Tune the radii for initial estimate

    # Define the objective function to minimize
    def objective(x, centers, radii, num_circles):
        # x is a flattened array of centers and radii
        # Format x: [x0, y0, r0, x1, y1, r1, ..., xn, yn, rn]
        x_flat = x

        radii_flat = x_flat[num_circles * 2:].reshape((num_circles,))

        total_energy = 0.0

            y = centers_flat[i, 1]

    result = opt.minimize(

        x0=initial_guess,
        bounds=bounds,
        method="L-BFGS-B"
