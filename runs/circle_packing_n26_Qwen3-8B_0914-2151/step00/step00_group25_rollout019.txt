import cvxpy as cp

def run_packing():

    # Number of rows and columns for a hexagonal grid

    cols = 5
    grid = np.zeros((rows * cols, 2))
    spacing = 1.0 / (cols - 1)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            grid[i * cols + j] = [x, y]
    # Shuffle the grid to create random placement
    np.random.shuffle(grid)
    
    # Add circles near corners for better radii
    extra_circles = [
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
    ]
    centers = np.vstack((grid, extra_circles))
    n = centers.shape[0]
    
    # Define variables
    radii = cp.Variable(n)
    
    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []

    for i in range(n):

        constraints.append(radii[i] >= 0)
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(radii[i] + radii[j] <= dist + 1e-12)
    
    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)

    centers = centers.astype(np.float64)

    sum_radii = np.sum(radii)
