from scipy.optimize import minimize
from typing import Tuple

def initialize_hexagonal_layout(n_circles: int) -> np.ndarray:

    Initialize a hexagonal grid arrangement of circles in the unit square.
    """

    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    
    # Initialize centers
    centers = np.zeros((n_circles, 2))

        row = i // cols
        col = i % cols

        x /= cols - 1 if cols > 1 else 1
        y /= rows - 1 if rows > 1 else 1
        centers[i] = [x, y]
    return centers

def total_radius_sum(radii: np.ndarray) -> float:
    return np.sum(radii)
