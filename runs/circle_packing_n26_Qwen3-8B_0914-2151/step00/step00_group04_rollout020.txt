import numpy as np
import cvxpy as cp

def run_packing():
    # Initial grid-based placement for 26 circles
    # Use a 5x5 grid with one extra circle in the middle

    # Grid points for 5x5 grid
    grid_points = np.array([
        [i / 4, j / 4] for i in range(5) for j in range(5)
    ])
    # Add one more circle at the center
    center_point = np.array([0.5, 0.5])
    centers = np.vstack((grid_points, center_point))

    # Define variables
    n = centers.shape[0]
    radii = cp.Variable(n, nonneg=True)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be within the unit square
    for i in range(n):
        x, y = centers[i]
        # Ensure that the circle is within the square [0,1]^2
        constraint = (x - radii[i] >= 0) & (x + radii[i] <= 1) & (y - radii[i] >= 0) & (y + radii[i] <= 1)
        constraints.append(constraint)

    # No two circles may overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraint = dist >= radii[i] + radii[j]
            constraints.append(constraint)

    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve()

    # Extract results
    centers_result = centers
    radii_result = np.array([radii[i].value for i in range(n)])
    sum_radii = np.sum(radii_result)

    return centers_result, radii_result, sum_radii
