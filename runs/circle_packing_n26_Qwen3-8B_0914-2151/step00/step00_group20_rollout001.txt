import numpy as np
import cvxpy as cp

def run_packing():
    # Define the grid
    rows = 6
    cols = 4
    grid = []

    # Hexagonal grid of circles in rows
    for i in range(rows):
        for j in range(cols):
            x = j + (0.5 if i % 2 == 1 else 0)
            y = i * np.sqrt(3)/2
            grid.append((x, y))
    
    # Add two extra circles: one at bottom-left, one at top-right
    grid.append((0.1, 0.1))
    grid.append((0.9, 0.9))

    # Extract positions
    num_circles = len(grid)
    centers_var = cp.Variable((num_circles, 2))
    radii_var = cp.Variable(num_circles)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii_var))

    # Constraints
    constraints = []

    # Each circle must be inside square [0,1] × [0,1]
    for i in range(num_circles):
        x, y = centers_var[i]
        r = radii_var[i]
        constraints.append(x - r <= 1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r <= 1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # No overlap between circles
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers_var[i, 0] - centers_var[j, 0]
            dy = centers_var[i, 1] - centers_var[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            constraints.append(dist >= radii_var[i] + radii_var[j] - 1e-12)

    # Define and solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)

    # Extract results
    centers = centers_var.value
    radii = radii_var.value
    total_sum = np.sum(radii)

    return (centers, radii, total_sum)
