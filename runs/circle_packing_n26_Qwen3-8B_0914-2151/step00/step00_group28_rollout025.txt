import numpy as np
from scipy.optimize import lbfgs
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers
    centers = np.random.rand(26, 2) * 1.0
    radii = np.random.rand(26) * 0.1

    def objective_func(x):
        # x is concatenated vector of centers and radii
        flat_centers = x[:52].reshape((26, 2))
        flat_radii = x[52:]

        sum_radii = np.sum(flat_radii)
        return -sum_radii  # Minimize negative to maximize

    def bounds_func(x):
        # Bounds for the centers and radii
        center_bounds = Bounds([0.0, 0.0] * 26, [1.0, 1.0] * 26)
        radius_bounds = Bounds([0.0] * 26, [1.0] * 26)
        return center_bounds, radius_bounds

    # Initial guess for centers and radii
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Optimization
    result = lbfgs(objective_func, x0=initial_guess, bounds=bounds_func)

    # Extract results
    x = result.x
    flat_centers = x[:52].reshape((26, 2))
    flat_radii = x[52:]

    # Validate packing
    valid, message = validate_packing(flat_centers, flat_radii)
    if not valid:
        raise ValueError(message)

    return flat_centers, flat_radii, np.sum(flat_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
