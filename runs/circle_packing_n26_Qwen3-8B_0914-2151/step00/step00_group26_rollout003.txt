import numpy as np
from scipy.optimize import minimize

def run_packing():
    # 5x5 grid with one extra circle
    num_circles = 26
    grid_rows = 5
    grid_cols = 5

    # Generate initial grid centers
    centers = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j / (grid_cols - 1)
            y = i / (grid_rows - 1)
            centers.append((x, y))
    # One extra circle in the center
    centers.append((0.5, 0.5))
    centers = np.array(centers)

    # Radius calculation: ensure circles are not overlapping and stay within bounds
    # We will use a heuristic: spacing = 2 * r
    # Solve for r such that the largest circle fits in the square

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        # Ensure circles stay within the square
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1.0 + 1e-12:
                return -1
            if y - radii[i] < -1e-12 or y + radii[i] > 1.0 + 1e-12:
                return -1
        return 0

    def constraint2(radii):
        # Ensure no overlaps
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Initial guess for radii
    r = 1.0 / (grid_rows + grid_cols)  # Initial estimate
    radii = np.full(num_circles, r)

    constraints = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
    ]

    result = minimize(objective, radii, constraints=constraints, method='SLSQP')

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
