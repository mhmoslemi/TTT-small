import scipy.optimize as optimize

        return -np.sum(radii)  # Minimize negative sum = maximize sum

        return True, "Valid constraints"

    bounds = [(0.0, 1.0) for _ in range(26)]  # All radii between 0 and 1
    result = optimize.minimize(

        x0=np.zeros(26),
