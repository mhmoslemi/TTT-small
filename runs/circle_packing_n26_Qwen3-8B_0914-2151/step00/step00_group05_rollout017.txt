```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
import random

def run_packing():
    # Initial configuration: place circles in a hexagonal grid pattern
    # Starting with 1 circle at the center
    centers = np.array([[0.5, 0.5]])
    radii = np.array([0.1])
    
    # Add 25 more circles in hexagonal pattern
    for _ in range(25):
        # Generate random direction and distance
        theta = np.random.uniform(0, 2 * np.pi)
        r = np.random.uniform(0.1, 0.2)
        new_x = centers[-1, 0] + r * np.cos(theta)
        new_y = centers[-1, 1] + r * np.sin(theta)
        centers = np.vstack((centers, [new_x, new_y]))
        radii = np.append(radii, r)
    
    # Optimize the configuration
    def objective(x):
        # x has shape (52,) where first 26*2 are centers, next 26 are radii
        centers = x[:52].reshape(26, 2)
        radii = x[52:]
        
        # Ensure non-negative radii
        if np.any(radii < 0):
            return float('inf')
        
        # Compute total sum of radii
        total_sum = np.sum(radii)
        
        # Check boundary constraints
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return float('inf')
        
        # Check pairwise circle overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return float('inf')
        
        return -total_sum  # Minimize negative total sum to maximize total sum
    
    # Initial guess
    x0 = np.concatenate((centers.reshape(-1), radii))
    
    # Bounds for centers: [0, 1] for both x and y
    center_bounds = [(0, 1) for _ in range(52)]
    # Bounds for radii: [0, 0.5]
    radius_bounds = [(0, 0.5) for _ in range(26)]
    
    bounds = center_bounds + radius_bounds
    
    # Run optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    # Extract final centers and radii
    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]
    
    # Ensure valid packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
