import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
from scipy.spatial import ConvexHull
from typing import Tuple
import matplotlib.pyplot as plt

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii using an evolutionary algorithm with a regular grid start.
    """
    # Initial grid-based packing for starting points
    num_circles = 26
    grid_size = 5
    grid = np.linspace(0, 1, grid_size)
    grid_points = np.array([[x, y] for x in grid for y in grid])
    centers = grid_points[:num_circles]
    radii = np.full(num_circles, 0.05)
    
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Define constraints: circle must be within the unit square
    def constraint_boundary(radii):
        return np.sum((centers[i][0] - radii[i] < 0 or centers[i][1] - radii[i] < 0 or 
                      centers[i][0] + radii[i] > 1 or centers[i][1] + radii[i] > 1) for i in range(num_circles))
    
    # Define constraint: circles do not overlap
    def constraint_overlap(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-10:
                    return False
        return True
    
    # Combine constraints into a single function
    def constraints(radii):
        if any(r < 0 for r in radii):
            return False
        if not constraint_boundary(radii):
            return False
        if not constraint_overlap(radii):
            return False
        return True
    
    # Use scipy.optimize to solve the problem
    result = minimize(objective, radii, method='SLSQP', bounds=[(0, 1)] * num_circles, constraints=[{'type': 'ineq', 'fun': constraints}])
    
    # Extract the result
    optimal_radii = result.x
    sum_radii = np.sum(optimal_radii)
    return centers, optimal_radii, sum_radii
