import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initialize centers and radii
    centers = np.empty((26, 2))
    radii = np.zeros(26)
    
    # First, create a hexagonal grid in the unit square
    side_length = 0.16
    num_rows = 4
    num_cols = 7
    vertical_offset = (np.sqrt(3) / 2) * side_length

    # Generate initial hexagonal grid
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * side_length + 0.05
            y = i * vertical_offset + 0.05
            centers[i * num_cols + j] = [x, y]
            radii[i * num_cols + j] = side_length * 0.8  # 80% of the side length

    # Adjust for boundary constraints
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            dx = r - x
            centers[i, 0] = x + dx
        elif x + r > 1:
            dx = (1 - x) - r
            centers[i, 0] = x + dx
        if y - r < 0:
            dy = r - y
            centers[i, 1] = y + dy
        elif y + r > 1:
            dy = (1 - y) - r
            centers[i, 1] = y + dy

    # Define objective function to maximize total radius
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints: non-overlapping circles and within bounds
    def constraint_non_overlap(radii):
        dists = squareform(pdist(centers))
        return np.sum(dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-8)

    def constraint_boundary(radii):
        return np.sum((centers[:, 0] - radii) >= 0) + np.sum((1 - centers[:, 0] - radii) >= 0) + \
               np.sum((centers[:, 1] - radii) >= 0) + np.sum((1 - centers[:, 1] - radii) >= 0)

    # Optimization
    result = minimize(
        fun=objective,
        x0=radii,
        constraints=[
            {'type': 'ineq', 'fun': constraint_non_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    radii = result.x
    centers = np.array(centers)

    # Final validation
    if validate_packing(centers, radii)[0]:
        return (centers, radii, np.sum(radii))
    else:
        raise ValueError("Invalid packing: constraints not met.")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
