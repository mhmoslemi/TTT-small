    N = 26

    num_cols = 5

    grid_centers = []

            x = c * (1 / (num_cols - 1)) + 0.5 * (1 - (num_cols - 1) / (num_cols - 1))
            y = r * (1 / (num_rows - 1)) + 0.5 * (1 - (num_rows - 1) / (num_rows - 1))
            grid_centers.append([x, y])
    grid_centers = np.array(grid_centers)

    # Initial radii: same for all circles in the grid
    initial_radii = np.full(N, 0.1)

    # Optimization variables: flatten centers and radii
    def objective(x: np.ndarray) -> float:
        # x is flattened array of [x1, y1, r1, x2, y2, r2, ..., xN, yN, rN]
        centers = x[:2 * N].reshape(N, 2)

        return -np.sum(radii)  # Maximize sum of radii by minimizing -sum

    def constraints(x: np.ndarray) -> np.ndarray:
        centers = x[:2 * N].reshape(N, 2)
        radii = x[2 * N:].reshape(N)

        # Boundary constraints
        boundary_constraints = []
        for i in range(N):
            x_i, y_i = centers[i]
            r_i = radii[i]
            boundary_constraints.extend([
                x_i - r_i >= -1e-12,
                x_i + r_i <= 1 + 1e-12,
                y_i - r_i >= -1e-12,
                y_i + r_i <= 1 + 1e-12
            ])

                dist = np.sqrt(dx**2 + dy**2)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
