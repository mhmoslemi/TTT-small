import numpy as np
import scipy.optimize as opt

def run_packing():

    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    initial_centers = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j / (grid_cols - 1)
            y = i / (grid_rows - 1)
            initial_centers.append([x, y])
    initial_centers = np.array(initial_centers)

    # Get the initial uniform radius
    max_radius = 0.5 / np.sqrt(grid_rows**2 + grid_cols**2)
    initial_radii = np.full(num_circles, max_radius)

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Negative for minimization

    # Constraint: non-overlapping circles
    def non_overlapping_constraints(r, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                def constraint(r):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    return dist - (r[i] + r[j])
                constraints.append(opt.LinearConstraint(r, -np.inf, constraint))
        return constraints

    # Boundary constraint: radius must not exceed distance to boundary
    def boundary_constraints(r, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            def constraint_r(r):
                x, y = centers[i]
                dist_left = x
                dist_right = 1 - x
                dist_bottom = y
                dist_top = 1 - y
                return min(dist_left, dist_right, dist_bottom, dist_top) - r[i]
            constraints.append(opt.LinearConstraint(r, -np.inf, constraint_r))
        return constraints

    # Equality constraints not needed here

    # Optimization
    result = opt.minimize(

        initial_radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, np.inf) for _ in range(num_circles)],
