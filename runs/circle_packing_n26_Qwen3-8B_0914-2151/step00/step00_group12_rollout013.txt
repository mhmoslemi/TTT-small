import numpy as np
import scipy.optimize as opt

def run_packing():

    col_count = 6  # Number of columns: 5 rows*2 + 1 = 11, but we will clip

    # Generate initial positions in a hexagonal grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    radius = 0.05  # Initial estimate
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col * (1 + radius)
        y = row * (np.sqrt(3) * radius)
        centers[i] = [x, y]
        radii[i] = radius

    max_y = np.max(centers[:, 1])

    centers /= np.max(centers, axis=0)

    # Flatten the geometry to form the initial guess
    initial_guess = np.hstack((centers.ravel(), radii))

    def objective(x):

        radii_flat = x[n * 2:].reshape(n)
        return -np.sum(radii_flat)  # Minimize the negative of sum for maximization

    # Define constraints for non-overlapping and boundary conditions
    def constraint(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)

                idx += 1
        return violations

    bounds = []

        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 1))  # For radii
    bounds = bounds * 3  # x, y, and radii for all circles

    # Run Nelder-Mead optimization
    result = opt.minimize(

        x0=initial_guess,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint},
        method="Nelder-Mead"

    return (centers, radii, sum_radii)
