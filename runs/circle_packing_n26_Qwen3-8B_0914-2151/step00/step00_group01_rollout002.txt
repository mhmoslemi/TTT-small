import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid coordinates (5x5 arrangement with some extra)
    grid = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    
    # Initial radius (small enough to not overlap)
    initial_radius = 0.05
    radii = np.full(26, initial_radius)
    
    # Variables: for each circle, x, y, and r
    n = 26
    x = grid[:, 0]
    y = grid[:, 1]
    r = radii
    
    # Define the objective function to maximize the total radius sum
    def objective(vars):
        # vars is flattened: x1, y1, r1, x2, y2, r2, ... x26, y26, r26
        # Extract them
        r = vars[2::3]
        x = vars[::3]
        y = vars[1::3]
        total = np.sum(r)
        return -total  # minimize negative to maximize

    # Define constraints: non-overlapping and within square
    def constraint(vars):
        r = vars[2::3]
        x = vars[::3]
        y = vars[1::3]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                if dist < r[i] + r[j] - 1e-12:
                    return dist - (r[i] + r[j] - 1e-12)
        return 0  # No overlap

    # Optimization bounds: x, y in [0,1], r ≥ 0
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1)])  # x, y, r

    constraints = {'type': 'ineq', 'fun': constraint}

    initial_guess = []

    # Run optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract final centers and radii
    x_opt = result.x[::3]
    y_opt = result.x[1::3]
    r_opt = result.x[2::3]

    centers = np.column_stack((x_opt, y_opt))
    sum_radii = np.sum(r_opt)

    return (centers, r_opt, sum_radii)
