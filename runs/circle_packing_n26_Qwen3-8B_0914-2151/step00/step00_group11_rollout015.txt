```python
import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define the number of circles and initialize centers
    n = 26
    centers = np.random.rand(n, 2) * 0.9  # Initial configuration
    radii = np.random.rand(n) * 0.05  # Initial radii used as starting point

    # Define bounds for optimization: x, y ∈ [0, 1], radii ≥ 0
    bounds = [
        (0, 1) for _ in range(n * 2)
    ] + [
        (0, 1) for _ in range(n)
    ]
    
    # Objective: maximize sum of radii → minimize -sum(radii)
    def objective(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        return -np.sum(radii)

    # Constraints:
    # 1. Boundary constraints for each circle: center_x - radius ≥ 0
    #                    center_x + radius ≤ 1
    #                    center_y - radius ≥ 0
    #                    center_y + radius ≤ 1
    # 2. Non-overlap constraints: distance between centers ≥ sum of radii

    # Flatten x into a single array: [x0, y0, x1, y1, ..., xn, yn, r0, r1, ..., rn-1]
    def constraint_boundary(x, i):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        cx, cy = centers[i]
        r = radii[i]
        return cx - r

    def constraint_overlap(x, i, j):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        cx, cy = centers[i]
        dx, dy = centers[j]
        r_i, r_j = radii[i], radii[j]
        return np.sqrt((cx - dx)**2 + (cy - dy)**2) - r_i - r_j

    # Define all constraints
    cons = []
    for i in range(n):
        # Boundary constraint for circle i
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: constraint_boundary(x, i)
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1 - constraint_boundary(x, i)
        })
    
    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlap constraint between i and j
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)
            })
    
    # Run the optimization
    x0 = np.concatenate([centers.reshape(-1), radii])
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )
    
    # Extract the optimized result
    optimized_x = result.x
    centers_opt = optimized_x[:n*2].reshape(n, 2)
    radii_opt = optimized_x[n*2:]
    
    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)
    
    total_radius = np.sum(radii_opt)
    return centers_opt, radii_opt, total_radius
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
