import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    row_count = 5
    col_count = 5
    grid = np.zeros((row_count, col_count, 2))

        for j in range(col_count):
            x = j / (col_count - 1)
            y = i / (row_count - 1)
            grid[i, j] = [x, y]

    centers = grid.reshape(-1, 2)

    def objective(radii):
        return -np.sum(radii)  # We minimize the negative of the sum

        return np.sum(radii)  # We want to maximize this, but as an objective

                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]) >= 0)
        return np.array(constraints)

    # Bounds for radii: non-negative
    bounds = [(0, 1)] * num_circles

    initial_guess = np.ones(num_circles)

    result = optimize.minimize(
