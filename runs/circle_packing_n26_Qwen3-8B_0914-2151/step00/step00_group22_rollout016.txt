def create_initial_layout(n_circles):
    """Create an initial layout for circle centers in a unit square."""
    # Start with a grid layout
    num_rows = int(np.sqrt(n_circles))
    num_cols = num_rows if n_circles % num_rows == 0 else num_rows + 1
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j / (num_cols - 1)
            y = i / (num_rows - 1)
            centers.append([x, y])
    return np.array(centers)
