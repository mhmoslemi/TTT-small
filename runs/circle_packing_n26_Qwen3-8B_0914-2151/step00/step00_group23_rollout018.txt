import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define number of circles and grid dimensions
    n = 26
    grid_shape = (5, 5)  # 5x5 grid = 25 circles, then place 1 extra in center

    # Initial grid-based positions for hexagonal packing
    centers = []
    radii = []
    grid_size = 0.2  # step size between grid points
    for i in range(grid_shape[0]):
        for j in range(grid_shape[1]):
            # Adjusted to fit within [0,1]^2
            x = i * grid_size
            y = j * grid_size
            centers.append([x, y])
    
    # Place one extra circle in center
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Initial guess for radii
    radii = np.full(n, 0.05)

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Constraints: circle must fit in [0,1]^2, and must not overlap
    def constraint1(x):
        return np.sum(x)  # Must be >= 0

    cons = []
    cons.append({'type': 'ineq', 'fun': constraint1})

    # Constraint: circle must fit within square
    def boundary_constraint(x, i):
        idx = i
        r = x[idx]
        x_pos, y_pos = centers[i]
        if (x_pos - r < 0 or x_pos + r > 1 or y_pos - r < 0 or y_pos + r > 1):
            return -1e-6  # Violation of boundary
        return 0

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: boundary_constraint(x, i)})

    # Constraint: circles must not overlap
    def overlap_constraint(x, i, j):
        r_i = x[i]
        r_j = x[j]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        if dist < r_i + r_j - 1e-8:
            return -1e-6  # Overlap
        return 0

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: overlap_constraint(x, i, j)})

    # Optimize
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(n)],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    # Extract optimal radii and centers
    optimal_radii = result.x
    optimal_centers = centers

    return (optimal_centers, optimal_radii, np.sum(optimal_radii))
