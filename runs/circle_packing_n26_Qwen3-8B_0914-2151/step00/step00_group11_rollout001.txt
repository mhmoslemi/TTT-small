import scipy.optimize as opt

def run_packing():

    grid_spacing = 1.0 / (num_rows - 1)  # Ensure circles are placed densely
    grid_centers = np.array([
        [i * grid_spacing, j * grid_spacing]
        for i in range(num_rows)
        for j in range(num_cols)
    ])

    initial_radius = 0.15
    radii = np.full(num_rows * num_cols, initial_radius)

        sum_radii = np.sum(radii)

        for i in range(len(positions)):
