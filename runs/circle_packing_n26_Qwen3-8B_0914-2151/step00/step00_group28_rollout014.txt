import numpy as np
import cvxpy as cp

def run_packing():
    # Define grid layout for 26 circles
    rows = 5
    cols = 5
    gaps = 0.2  # spacing between circles for initial layout
    
    # Generate initial grid positions
    x = np.linspace(gaps, 1 - gaps, cols)
    y = np.linspace(gaps, 1 - gaps, rows)
    centers = np.array([[xi, yi] for yi in y for xi in x])
    n = len(centers)

    # Initial radii
    radii = np.full(n, 0.01)

    # Define variable for radii
    r = cp.Variable(n)

    # Objective: Maximize the total sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(n):
        x_i, y_i = centers[i]
        constraints.append(x_i - r[i] >= -1e-12)
        constraints.append(x_i + r[i] <= 1 + 1e-12)
        constraints.append(y_i - r[i] >= -1e-12)
        constraints.append(y_i + r[i] <= 1 + 1e-12)

    # Circle-neighbor constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append(dist >= r[i] + r[j] - 1e-12)

    # Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-5)

    # Extract solution
    radii_opt = r.value
    centers_opt = centers

    # Compute total radius sum
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
