from scipy.optimize import minimize

def run_packing():

        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_non_overlap(centers, radii):

        for i in range(n):
            for j in range(i + 1, n):
                def constraint(params):
                    return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j])
                constraints.append({'type': 'ineq', 'fun': constraint})
        return constraints

    def constraint_boundary(centers, radii):

        constraints = []

                x, y = centers[i]

                return x - r

                return 1.0 - x - r

                return y - r

                return 1.0 - y - r
            constraints.append({'type': 'ineq', 'fun': constraint_top})
