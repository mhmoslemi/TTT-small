import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    # 5x5 grid (25 circles) + 1 circle at the center

    grid_points.append([0.5, 0.5])

    # Optimization
    def objective(x):
        # x contains all parameters: first half are centers, second half are radii
        centers_opt = x[:2*26].reshape(26, 2)
        radii_opt = x[2*26:]

        # Objective: maximize sum of radii
        sum_radii = np.sum(radii_opt)

        for i in range(26):
            x, y = centers_opt[i]

            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        # Pairwise non-overlap check

    # Initial guess: centers and radii
    centers_initial = centers
    radii_initial = radii

    # Optimization settings
    bounds = []

        # Boundary for x, y: [0, 1]
        bounds.append((0, 1))
        bounds.append((0, 1))

    # Optimization with constraints
