import scipy.optimize as opt

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    grid_rows = 2
    grid_cols = 13

    dx = 1 / (grid_cols - 1)
    dy = 1 / (grid_rows - 1)

    centers = np.zeros((num_circles, 2))

            idx = i * grid_cols + j
            centers[idx, 0] = j * dx
            centers[idx, 1] = i * dy

    def total_radius_sum(radii):
        return np.sum(radii)

        return -total_radius_sum(x)

        y_center = centers[i, 1]
        return (x_center - r >= -1e-12) and (x_center + r <= 1 + 1e-12) and (y_center - r >= -1e-12) and (y_center + r <= 1 + 1e-12)
