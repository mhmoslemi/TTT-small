import numpy as np
import math

def run_packing():
    # Define the grid configuration: 5 rows with 5, 6, 6, 6, 5 circles respectively
    num_rows = 5
    cols = [5, 6, 6, 6, 5]
    total_circles = sum(cols)
    assert total_circles == 26, "Total circles must be 26"

    # Define spacing and radius parameters
    spacing = 0.17  # Spacing between circles
    horizontal_padding = 0.1
    vertical_padding = 0.1
    max_radius = 0.15

    centers = []
    radii = []

    # Set up the hexagonal grid
    row_height = spacing * math.sqrt(3)
    offset = spacing / 2

    for row_idx in range(num_rows):
        row = cols[row_idx]
        if row == 5:
            col_spacing = 0.25
        elif row == 6:
            col_spacing = 0.2
        else:
            col_spacing = 0.25

        x_start = horizontal_padding
        y_start = vertical_padding + row_idx * row_height

        for col in range(row):
            x = x_start + col * col_spacing
            y = y_start

            # For even rows, shift circles to the right by offset
            if row_idx % 2 == 1:
                x += offset

            # Place the circle
            centers.append([x, y])
            radii.append(min(max_radius, (spacing - 2 * 1e-12) / 2))

    centers = np.array(centers)
    radii = np.array(radii)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
