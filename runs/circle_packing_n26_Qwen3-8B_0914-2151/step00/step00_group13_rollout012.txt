import scipy.optimize as opt

def run_packing():
    # Define initial centers and radii as a guess
    # Start with 26 circles near the center but slightly spread out
    n = 26
    centers = np.array([[0.5, 0.5] for _ in range(n)])
    radii = np.random.uniform(0.01, 0.1, n)  # Initial random radii

    # Penalty coefficient for constraints
    penalty = 1e6

    # Define the objective function
    def objective(x):
        # x is a flat vector: first 2n elements are centers (x1, y1, x2, y2, ...), next n elements are radii
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        # Compute the sum of radii (to be maximized)
        sum_radii = np.sum(radii_flat)

        # Compute constraints
        # 1. Boundary constraints
        # 2. Non-overlap constraints

        # Boundary constraint: centers[i] - radii[i] >= 0 and centers[i] + radii[i] <= 1
        boundary_penalty = 0
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1:
                boundary_penalty += penalty * (x_i - r_i) ** 2
            if y_i - r_i < 0 or y_i + r_i > 1:
                boundary_penalty += penalty * (y_i - r_i) ** 2

        # Non-overlap constraint: distance between centers >= r_i + r_j
        overlap_penalty = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    overlap_penalty += penalty * (dist - (radii_flat[i] + radii_flat[j])) ** 2

        return -sum_radii + boundary_penalty + overlap_penalty  # Minimize negative sum to maximize sum

    # Flatten the initial guess
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds: x in [0,1], y in [0,1], radii >= 0

    for _ in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    # Optimization
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
