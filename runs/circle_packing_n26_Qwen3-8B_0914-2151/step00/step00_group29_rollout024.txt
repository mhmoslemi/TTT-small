from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint
import matplotlib.pyplot as plt

class CirclePacking:
    def __init__(self, n_circles):
        self.n_circles = n_circles
        self.centers = np.zeros((n_circles, 2))
        self.radii = np.zeros(n_circles)

    def place_initial_hexagonal(self, spacing=0.5):

            col = i % 7
            x = col * spacing

            radii = x[self.n_circles:]
            dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))

            radii = x[self.n_circles:]
            return np.min(centers - radii, axis=1), np.min(1 - centers - radii, axis=1)

        return NonlinearConstraint(non_overlap, -1e-6, 0), NonlinearConstraint(boundary, -1e-6, 0)

            return -np.sum(x[self.n_circles:])  # We minimize the negative of the sum
