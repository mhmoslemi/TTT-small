import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Step 1: Initial grid-based placement
    n = 26
    rows = 5
    cols = 6

    # Create a grid-based initial placement
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        centers[i] = [x, y]

    # Step 2: Define an objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Step 3: Optimization: We will optimize the positions and radii
    def optimize_packing(centers, radii):
        # Optimization variables: positions and radii
        # Flatten centers into a 1D vector
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0))  # x
            bounds.append((0.0, 1.0))  # y
            bounds.append((0.0, 1.0))  # radius

        # Initial guess: centers as before, radii as 0.01
        initial_guess = centers.reshape(-1)
        initial_guess = np.concatenate((initial_guess, np.full(n, 0.01)))

        # Optimization problem: maximize sum of radii with non-overlap constraints
        def constraint_func(x):
            # Extract centers and radii
            pos = x[:2*n].reshape(n, 2)
            r = x[2*n:]

            # Enforce boundary constraints
            for i in range(n):
                x_i, y_i = pos[i]
                r_i = r[i]
                if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                    return np.inf  # Violation
                if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                    return np.inf  # Violation

            # Enforce non-overlap between circles
            for i in range(n):
                for j in range(i + 1, n):
                    dx = pos[i, 0] - pos[j, 0]
                    dy = pos[i, 1] - pos[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < r[i] + r[j] - 1e-12:
                        return np.inf  # Violation

            return 0  # No violation

        # Run optimization
        result = optimize.minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints={"type": "ineq", "fun": constraint_func},
            method="SLSQP"

        # Extract final positions and radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
