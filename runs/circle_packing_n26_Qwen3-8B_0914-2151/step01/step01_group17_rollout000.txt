import scipy.optimize as optimize

                row.append([x, y])
            centers.extend(row)
        
        # Convert to NumPy
        centers = np.array(centers)
        return centers

    # Generate initial centers
    centers = initial_grid()
    n = len(centers)

    def objective(radii):
