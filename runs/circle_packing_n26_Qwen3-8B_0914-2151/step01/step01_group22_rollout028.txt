from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define a grid-based initial guess with some overlap and adjust via optimization

    # Initial guess: start with 4 circles at the corners
    initial_centers = np.array([
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.9], [0.9, 0.1],
        [0.1, 0.5], [0.5, 0.1], [0.5, 0.9], [0.9, 0.5],
        [0.5, 0.5], [0.3, 0.3], [0.7, 0.3], [0.3, 0.7], [0.7, 0.7],
        [0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75],
        [0.4, 0.4], [0.6, 0.4], [0.4, 0.6], [0.6, 0.6]
    ])

    n = len(initial_centers)

    # Initialize radii using the minimal possible radius that fits within boundaries
    def initial_radii(centers):
        return np.array([0.05 for _ in centers])

    # Define the cost function to minimize
    def cost_function(params: np.ndarray, centers: np.ndarray, n: int):
        # params = [r0, r1, ..., r25]
        radii = params
        # Compute pairwise distances
        distances = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                distances[i, j] = np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
                distances[j, i] = distances[i, j]

        # Compute the cost based on circle–circle overlap
        overlap_cost = np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                if distances[i, j] < radii[i] + radii[j]:
                    overlap_cost[i] += (radii[i] + radii[j] - distances[i, j]) ** 2
                    overlap_cost[j] += (radii[i] + radii[j] - distances[i, j]) ** 2

        # Ensure all circles are within the unit square
        boundary_cost = 0
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                boundary_cost += 1000000

        total_cost = np.sum(overlap_cost) + boundary_cost
        return total_cost

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
