import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List, Optional
import random

def validate_packing(centers: np.ndarray, radii: np.ndarray) -> Tuple[bool, str]:
    """
    Check if the circle packing is valid.
    """
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_initial_config(rows: int = 5, cols: int = 5) -> np.ndarray:
    """
    Generate an initial concentric pattern of circles with staggered rows.
    """
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            angle = 2 * np.pi * i / rows
            radius = 0.1 * (rows - i)
            x = 0.5 + radius * np.cos(angle)
            y = 0.5 + radius * np.sin(angle)
            centers.append([x, y])
            radii.append(radius)
    return np.array(centers), np.array(radii)

def objective(params: np.ndarray, centers: np.ndarray, radii: np.ndarray) -> float:
    """
    Objective function to maximize total sum of radii.
    """
    idx = 0
    new_centers = []
    new_radii = []
    for i in range(centers.shape[0]):
        new_centers.append(centers[i] + params[idx:idx+2])
        new_radii.append(radii[i] + params[idx+2:idx+4][0])
        idx += 4
    return -np.sum(new_radii)  # Minimize negative to maximize sum

def constraint_no_overlap(params: np.ndarray, centers: np.ndarray, radii: np.ndarray) -> float:
    """
    Constraint to ensure no overlaps between circles.
    """

    new_centers = []

            if dist[i, j] < new_radii[i] + new_radii[j] - 1e-12:
                return 1.0
    return 0.0

def constraint_bound(params: np.ndarray, centers: np.ndarray, radii: np.ndarray) -> float:
    """
    Constraint to ensure circles are fully within the unit square.
