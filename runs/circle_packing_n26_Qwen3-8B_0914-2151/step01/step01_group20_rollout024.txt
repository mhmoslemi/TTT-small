import scipy.optimize as opt
import cvxpy as cp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial guess for centers in a grid
    num_circles = 26
    rows = 6
    cols = 5

    radii = np.zeros(num_circles)

    for i in range(num_circles):
        col = i % cols
        row = i // cols
        centers[i, 0] = col / (cols - 1)
        centers[i, 1] = row / (rows - 1)
        radii[i] = 0.04  # Initial guess for radius

    # Define the optimization variables
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles)

    # Constraints
    constraints = []

    # Boundaries
    for i in range(num_circles):
        x_i, y_i = x[i]
        r_i = r[i]
        constraints += [x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1]

    # Non-overlap constraint
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = x[i, 0] - x[j, 0]
            dy = x[i, 1] - x[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints += [dist >= r[i] + r[j]]

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    result = problem.solve()

    centers = np.array([x[i].value for i in range(num_circles)])
