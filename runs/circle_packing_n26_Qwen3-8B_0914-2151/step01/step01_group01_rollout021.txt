import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define number of circles
    n = 26
    
    # Randomly generate initial positions near boundaries and corners
    # Use 40% near corners, 30% near edges, and 30% in center
    corners = np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.9], [0.9, 0.1]])
    edges = np.array([[0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]])
    center = np.array([0.5, 0.5])
    
    # Generate random positions near corners and edges
    positions = []
    for _ in range(n):
        if np.random.rand() < 0.4:
            # Place near corners
            idx = np.random.randint(4)
            pos = corners[idx] + np.random.uniform(-0.1, 0.1, 2)
        elif np.random.rand() < 0.3:
            # Place near edges
            idx = np.random.randint(4)
            pos = edges[idx] + np.random.uniform(-0.1, 0.1, 2)
        else:
            # Place near center
            pos = center + np.random.uniform(-0.2, 0.2, 2)
        positions.append(pos)
    centers = np.array(positions)
    
    # Initial radii (small random values)
    initial_radii = np.random.uniform(0.01, 0.1, n)
    
    # Objective function: maximize sum of radii (minimize -sum of radii)
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints:
    # 1. Each circle must not exceed the unit square
    # 2. No overlapping between circles
    
    bounds = [(0, 1) for _ in range(n)]
    
    # Create constraints for each pair of circles
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            def constraint_overlap(r, i=i, j=j):
                return np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (r[i] + r[j])
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap
            })
    
    # Constraints for not exceeding square boundaries
    for i in range(n):
        def constraint_boundary(r, i=i):
            return 1 - np.max([centers[i, 0] + r[i], 1 - centers[i, 0] + r[i], centers[i, 1] + r[i], 1 - centers[i, 1] + r[i]])
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary
        })
    
    # Optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    # Extract final solution
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
