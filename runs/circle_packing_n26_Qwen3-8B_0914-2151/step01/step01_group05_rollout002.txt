import scipy.optimize as opt

def run_packing():

    # This layout has the first row with one circle, second row with three circles, etc.
    # The layout is created by adding horizontal and vertical spacing in a staggered pattern
    centers = []
    for i in range(num_circles):
        if i % 2 == 0:
            row = i // 2
            col = i % 2
        else:
            row = (i - 1) // 2
            col = i % 2
        x = col + 0.5 * (row % 2)
        y = row
        centers.append([x, y])
    
    # Scale and normalize centers to [0, 1]^2
    centers = np.array(centers)
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, 0, 1)

    # Set initial radii as small values
    radii = np.full(num_circles, 0.1)

    # Define the objective function to maximize the sum of radii under constraints
    def objective(x):
        # x: [x0, y0, r0, x1, y1, r1, ...]
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x[num_circles * 2:].reshape((num_circles,))

        # Compute the sum of radii
        sum_radii = np.sum(radii_flat)

        # Enforce boundary constraints: circle must be fully within square
        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if (x - r < -1e-8 or x + r > 1 + 1e-8 or

                return -np.inf  # Penalize out-of-bound circles

        # Enforce non-overlap constraint: distance between centers >= sum of radii

                dx = centers_flat[i, 0] - centers_flat[j, 0]

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -np.inf  # Penalize overlapping circles

        return sum_radii

    # Define bounds: x, y in [0,1], each radius ≥ 0 (and we limit max radius for feasibility)
