import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define the number of circles
    num_circles = 26

    # Define the unit square boundaries
    square_left = 0.0
    square_right = 1.0
    square_bottom = 0.0
    square_top = 1.0

    # Initial setup: Place circles in a grid-like pattern
    grid_rows = 4
    grid_cols = 6
    spacing = 0.2
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    circle_index = 0

    # Place grid of circles in the square
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = i * spacing
            y = j * spacing
            # Ensure the circle is within the square
            if x < square_right - spacing and y < square_top - spacing:
                centers[circle_index, 0] = x
                centers[circle_index, 1] = y
                # Radius = minimum of (distance to left, right, bottom, top)
                radius = min(
                    square_right - x - spacing,
                    x - square_left - spacing,
                    square_top - y - spacing,
                    y - square_bottom - spacing
                )
                radii[circle_index] = max(0.01, radius)  # Minimum radius of 0.01
                circle_index += 1
            if circle_index >= num_circles:
                break

    # Fill up remaining circles near corners
    corners = [
        (square_left + 0.1, square_bottom + 0.1),
        (square_right - 0.1, square_bottom + 0.1),
        (square_left + 0.1, square_top - 0.1),
        (square_right - 0.1, square_top - 0.1)
    ]

    for corner in corners:
        # Add a circle of radius 0.15 at each corner
        if circle_index < num_circles:
            centers[circle_index, 0] = corner[0]
            centers[circle_index, 1] = corner[1]
            radii[circle_index] = 0.15
            circle_index += 1

    # Final check if we need more circles
    while circle_index < num_circles:
        # If we need more circles, add a few small ones for fill
        if circle_index < num_circles - 2:
            # Add small circles in the remaining space
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            # Radius of 0.05 as default
            radii[circle_index] = 0.05
            centers[circle_index, 0] = x
            centers[circle_index, 1] = y
            circle_index += 1

    # Initial guess for radii (based on fixed placement)
    initial_radii = radii.copy()

    # Define bounds for radii (non-negative)
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    # Define the constraints
    constraints = []

    # Boundary constraints: ensure circles stay within the square
    for i in range(num_circles):
        def constraint_boundary(radii, i=i):
            x, y = centers[i]
            return (1 - x - radii[i]) - 1e-8  # Ensure circle is inside the square
        constraints.append({'type': 'ineq', 'fun': constraint_boundary})

    # Non-overlapping constraints: distance between circle centers >= sum of radii
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def constraint_overlap(radii, i=i, j=j):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (radii[i] + radii[j]) - 1e-8
            constraints.append({'type': 'ineq', 'fun': constraint_overlap})

    # Define objective function
    def objective(radii):
        return -np.sum(radii)

    # Run the optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    # Extract final results
    centers = result.x[:num_circles]
    radii = result.x[:num_circles]
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
