from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import matplotlib.pyplot as plt

def run_packing():

    num_circles = 26
    grid_size = 5

    # Generate initial grid of centers (5x5)
    centers = np.zeros((num_circles, 2))
    for i in range(grid_size):
        for j in range(grid_size):
            centers[i * grid_size + j] = [j / (grid_size - 1), i / (grid_size - 1)]

    # Add 26th circle at the center if not already there
    if grid_size == 5:
        centers[-1] = [0.5, 0.5]

    # Initial radii (try to make appropriate initial guesses based on spacing)
    initial_radii = np.zeros(num_circles)
    spacing = 1 / (grid_size - 1)
    for i in range(num_circles):
        # Assign radii proportional to spacing
        if i < num_circles - 1:
            initial_radii[i] = spacing * 0.4  # smaller circles for grid
        else:
            # The 26th (central) circle can be slightly larger
            initial_radii[i] = spacing * 0.6

    # Define the objective function: minimize -sum of radii (equivalent to maximizing it)
    def objective(x):
        # x is a flat array: [r0, r1, ..., r25]
        r = x.reshape((num_circles,))
        return -np.sum(r)  # Minimize negative of sum radii (i.e., maximize sum)

    # Define constraints: no circle overlaps, no circle outside [0,1]^2
    def constraint_overlap(x):

        distances = cdist(centers, centers)
        for i in range(num_circles):

                if distances[i, j] < r[i] + r[j] - 1e-8:
                    return distances[i, j] - r[i] - r[j] + 1e-8
        return 0  # No overlap detected

        for i in range(num_circles):

            if (x_i - r[i] < -1e-8 or x_i + r[i] > 1 + 1e-8

                return x_i - r[i] + 1e-8  # Violation in x direction

    # Define constraints for the optimizer
    cons = [

        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    x0 = initial_radii.flatten()

    result = minimize(

        x0=x0,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(num_circles)],

        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}

    # Extract final result
    final_radii = result.x

    sum_radii = np.sum(final_radii)

    final_centers = final_centers.astype(np.float64)
    final_radii = final_radii.astype(np.float64)

    return final_centers, final_radii, sum_radii
