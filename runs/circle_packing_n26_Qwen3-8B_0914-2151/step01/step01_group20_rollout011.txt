import scipy.optimize as optimize

            centers.append([grid_x[i], grid_y[j]])

        return -np.sum(radii_flat)  # Negative for minimization

    # Constraints: circle must be within [0,1]^2
    def constraint_center(x, i):
        r = x[n + i]
        x_center = x[i]
        y_center = x[n + i + 1]
        return x_center - r >= 0

        x_center = x[i]
        y_center = x[n + i + 1]
        return x_center + r <= 1
