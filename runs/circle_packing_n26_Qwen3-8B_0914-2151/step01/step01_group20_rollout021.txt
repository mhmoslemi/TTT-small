import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial approximate hexagonal packing of 26 circles with equal radii
    # First, set up a grid for centers
    rows = 5
    cols = 6
    num_circles = rows * cols

    # Start with equal radii
    initial_radius = 0.05
    radii = np.full(num_circles, initial_radius)

    # Generate the initial grid for the centers
    centers = np.zeros((num_circles, 2))
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5
            centers[i * cols + j] = [x, y]

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint functions
    def constraint_non_overlap(radii):
        dist_matrix = cdist(centers, centers)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_out_of_bounds(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Combine constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_non_overlap},
        {'type': 'ineq', 'fun': constraint_out_of_bounds}
    ]

    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(num_circles)]
    )

    # Extract results
    optimal_radii = result.x
    optimal_centers = centers
    total_sum = np.sum(optimal_radii)

    return (optimal_centers, optimal_radii, total_sum)
