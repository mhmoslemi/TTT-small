import cvxpy as cp

    centers = np.array(centers)
    n = len(centers)

    # Variable to store radii
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: Each circle fits in the unit square
    constraints = []

    # Solve the problem
    problem = cp.Problem(objective, constraints)
