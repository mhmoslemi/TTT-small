import scipy.optimize as opt
import random

def run_packing():

    centers = np.random.rand(num_circles, 2)
    radii = np.full(num_circles, 0.05)

    def objective(x):

        # Format x: [x0, y0, r0, x1, y1, r1, ..., xn, yn, rn]
        x_flat = x

        radii_flat = x_flat[num_circles * 2:].reshape((num_circles,))

        total_energy = 0.0

                    total_energy += 100.0 * (dist - (radii_flat[i] + radii_flat[j]))**2
        return total_energy

    result = opt.minimize(
