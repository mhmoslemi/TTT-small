from scipy.optimize import minimize

def run_packing():

        return -np.sum(radii)

    def generate_hexagonal_grid(n=26):

                y = (i // 2 % 3) / 3
            else:
                x = (i // 2 + 0.5) / (n // 2)
                y = (i // 2 % 3) / 3
            centers[i, 0] = x
            centers[i, 1] = y
        return centers

    centers = generate_hexagonal_grid(26)

    bounds = [(0, 1) for _ in range(26)]

    constraints = []

                return np.sqrt(dx * dx + dy * dy) - (radii[i] + radii[j])
            constraints.append({

                'fun': constraint_overlap
            })

    for i in range(26):

    result = minimize(
