import numpy as np
import scipy.optimize as opt

def run_packing():
    # Set up a grid of circle centers
    rows = 5
    cols = 5
    spacing = 0.25  # Can be adjusted for better packing
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            centers.append([x, y])
    centers = np.array(centers)

    # Initial radii guess
    radii = np.full(len(centers), spacing / 2)

    # Define constraints and objective
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_boundary(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -np.inf
        return np.sum(radii)

    def constraint_overlap(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    # Optimization setup
    bounds = [(0, 1) for _ in range(len(radii))]
    constraints = [
        {'type': 'inequality', 'fun': constraint_boundary},
        {'type': 'inequality', 'fun': constraint_overlap}
    ]

    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    optimal_radii = result.x
    optimal_centers = centers

    sum_radii = np.sum(optimal_radii)
    return optimal_centers, optimal_radii, sum_radii
