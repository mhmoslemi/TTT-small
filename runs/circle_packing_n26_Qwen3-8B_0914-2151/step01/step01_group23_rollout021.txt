from scipy.optimize import minimize

    cols = 5
    spacing = 0.2  # A bit more than 1/5 to allow for expansion

    # Generate initial centers in a grid layout
    centers = []

        for j in range(cols):
            x = j * spacing
            y = i * spacing
            centers.append([x, y])

    radii = np.full(len(centers), 0.05)  # Start with small radii

        return -np.sum(radii)  # Minimize negative sum to maximize radii

    # Optimization
    result = minimize(
