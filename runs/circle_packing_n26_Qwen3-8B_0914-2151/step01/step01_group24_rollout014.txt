import scipy.optimize as opt

def run_packing():

        r = 0.05  # initial small radius for spacing

            x = col * (1 + r) / (5 + 1)
            y = row * (1 + r) / (5 + 1)
            centers[i] = [x, y]
        return centers

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total

    def constraint_non_overlap(centers, radii):
        return np.array([np.sqrt(np.sum((centers[i] - centers[j])**2)) - radii[i] - radii[j] for i in range(len(radii)) for j in range(i+1, len(radii))])

        return np.array([centers[i, 0] - radii[i], centers[i, 0] + radii[i], centers[i, 1] - radii[i], centers[i, 1] + radii[i] for i in range(len(radii))])

    def constraint_radii(radii):
        return np.array(radii)

    def run_packing():
