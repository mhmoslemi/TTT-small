import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import cdist


def run_packing():
    # Initial layout: 5x5 grid with one circle near a corner
    num_circles = 26
    grid_spacing = 1.0 / (5 - 1)  # Spacing between grid points

    # Initial centers
    centers = np.zeros((num_circles, 2))
    for i in range(5):
        for j in range(5):
            centers[5 * i + j] = [i * grid_spacing, j * grid_spacing]

    # Move the last circle to a corner
    centers[-1] = [0, 0]  # Move to the bottom-left corner

    # Initial radii: equal for simplicity
    radii = np.ones(num_circles) * 0.05

    # Define a function that computes the sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum by minimizing -sum

    # Define a function that computes constraints
    def constraints(radii, centers):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraint = dist - (radii[i] + radii[j]) + 1e-8
                constraints.append(constraint)
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return False
        return True

    # Optimization
    result = opt.minimize(
        fun=lambda x: objective(x, centers),
        x0=radii,
        bounds=[(0, 1) for _ in range(num_circles)],
        constraints=[{"type": "ineq", "fun": lambda x: constraints(x, centers)}],
        method="SLSQP"
    )

    # Extract results
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(message)

    return centers, radii, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
