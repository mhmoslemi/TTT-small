from scipy.optimize import minimize
import warnings

    n_circles = 26

            y = j / 4.0 + 0.05
            centers.append([x, y])

    radii = np.full(n_circles, 0.05)

        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: distance between circles >= sum of radii
    def constraint_dist(c, r):

                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx**2 + dy**2)

                    return False
        return True

    def constraint_bound(c, r):
        for i in range(n_circles):
            x, y = c[i]
            r_i = r[i]
            if x - r_i < -1e-8 or x + r_i > 1 + 1e-8 or y - r_i < -1e-8 or y + r_i > 1 + 1e-8:
