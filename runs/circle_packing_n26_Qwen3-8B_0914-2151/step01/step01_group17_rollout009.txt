import numpy as np
from scipy.optimize import minimize
from scipy.optimize.constraints import LinearConstraint, NonlinearConstraint

def run_packing():
    # Step 1: Initial grid configuration with 26 circles
    num_circles = 26
    centers = []
    for i in range(5):
        for j in range(5):
            x = j / 4.0
            y = i / 4.0
            centers.append([x, y])
    # Add one more circle in the center
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Step 2: Define optimization variables
    # We will use the radii as variables and estimate the positions via optimization

    # Step 3: Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize sum

    # Step 4: Constraints:
    # Constraint 1: All circles are within the unit square
    def unit_square_constraint(radii):
        # Implementing the constraint that for all circles, x - r >= 0, x + r <= 1, etc.
        # This is nonlinear so we use a NonlinearConstraint
        return np.array([
            centers[i, 0] - radii[i] >= -1e-12 for i in range(num_circles)
        ])

    # Constraint 2: All circles non-overlapping
    def non_overlapping_constraint(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([1.0])  # Constraint violated
        return np.array([0.0])

    # Define optimization problem
    # Radii are the variables (initial guess: same as the square of distance to neighbors)
    radii_initial = np.array([0.1 for _ in range(num_circles)])
    # Add a small perturbation to avoid local minima
    radii_initial += np.random.uniform(-0.01, 0.01, size=num_circles)

    result = minimize(
        objective,
        radii_initial,
        constraints=[
            NonlinearConstraint(unit_square_constraint, -np.inf, 0.0),
            NonlinearConstraint(non_overlapping_constraint, 0.0, 0.0)
        ],
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(num_circles)]
    )

    # Extract results
    radii = result.x
    sum_radii = np.sum(radii)
    centers = np.array(centers)

    return centers, radii, sum_radii
