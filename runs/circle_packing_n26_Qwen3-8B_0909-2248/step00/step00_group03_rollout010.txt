def generate_hexagonal_grid(n, square_size=1.0):
    """
    Generate a hexagonal grid of circles with n circles.
    """
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows
    radius = 0.0
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5 * i) * (np.sqrt(3) * radius)
            y = (i + 0.5) * (2 * radius)
            if x > square_size or y > square_size:
                continue
            centers.append([x, y])
            if len(centers) == n:
                break
        if len(centers) == n:
            break
    return np.array(centers), np.array([radius] * n)
