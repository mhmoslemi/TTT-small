import cvxpy as cp

        # For example: 3 rows with decreasing numbers of elements (as per 26 circles)

        # First, define rows for hexagonal packing
        row_sizes = [6, 5, 5, 4, 4, 3, 3, 2, 2, 1, 1]
        x_centers = [0.1, 0.3, 0.5, 0.7, 0.9, 0.1, 0.3, 0.5, 0.7, 0.9, 0.1]
        y_centers = [0.45, 0.6, 0.75, 0.45, 0.6, 0.75, 0.45, 0.6, 0.75, 0.45, 0.6]

        for i, (size, x, y) in enumerate(zip(row_sizes, x_centers, y_centers)):

                y_coord = y + (i % 2) * (1.0 / (row_sizes[i] - 1)) if size > 1 else y
                positions.append((x_coord, y_coord))

        return np.array(positions)

    initial_centers = generate_hexagonal_grid(n)
    initial_radii = np.full(n, 0.05)  # Initial guess for radii

    # CVXPY variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, non_negative=True)

    # Constraints
    constraints = [

        (centers[i, 0] - radii[i] >= 0, centers[i, 0] + radii[i] <= 1) for i in range(n)

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)

    centers.value = centers_initial
    radii.value = radii_initial

    # Solve the optimization problem
    prob.solve(solver=cp.SCS, eps=1e-5)

    # Extract the result
    centers_result = centers.value
    radii_result = radii.value

    # Compute the total sum of radii
    total_sum = np.sum(radii_result)

    return (centers_result, radii_result, total_sum)
