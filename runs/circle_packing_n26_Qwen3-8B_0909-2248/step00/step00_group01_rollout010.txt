import scipy.optimize as optimize

def run_packing():

    dx = 1.0 / 5

    # Place circles in a grid pattern

            row = i // 5
            col = i % 5
            centers[i, 0] = col * dx
            centers[i, 1] = row * dy
        else:
            # Place an extra circle in a corner (top-right)
            centers[i, 0] = 0.8
            centers[i, 1] = 0.8

    initial_radius = 0.05

    def objective(x):

        centers_flat = x[:n*2].reshape((n, 2))
        radii_flat = x[n*2:].reshape(n)
        return -np.sum(radii_flat)  # We minimize the negative sum to maximize the sum

    # Define constraints: non-overlapping and within square
    def constraint(x):
        centers_flat = x[:n*2].reshape((n, 2))
        radii_flat = x[n*2:].reshape(n)

        # Check distances between all circle pairs
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]

                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1
        return 0

    # Set up optimization problem

    # Perform optimization
    result = optimize.minimize(
