                x = j + 0.5 * i
                y = i + 0.5 * i
                grid.append([x, y])
        return np.array(grid)
    
    centers = hex_grid(24)

    def sum_radii(radii):

    # Optimization parameters
    bounds = [((0, 1), (0, 1)) for _ in range(26 * 2)]  # bounds for each center coordinate
    bounds += [(0, 0.1) for _ in range(26)]  # bounds for each radius
    constraints = []

    def add_distance_constraints(i, j):

            # x contains all variables: centers[0], centers[1], ... centers[25], radii[0], ..., radii[25]
            center_idx = 0
            for k in range(26):
                center_idx += 2
                if k == i:
                    x_i = x[center_idx: center_idx + 2]

            r_j = x[center_idx + 2 + j]
            return dist - (r_i + r_j) + 1e-12
        return constraint

    # Add constraints between every pair of circles
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({
