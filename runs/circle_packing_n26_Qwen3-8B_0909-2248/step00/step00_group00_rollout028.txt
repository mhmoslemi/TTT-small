import numpy as np
import scipy.optimize as optimize

def run_packing():

        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(centers, radii):
        n = len(radii)
        constraints = []

            # Boundary constraint: circle lies within the square [0,1]x[0,1]
            x, y = centers[i]
            r = radii[i]
            lb = np.array([x - r, y - r])
            ub = np.array([x + r, y + r])
            constraints.append(('bounds', lb, ub))
        
        # Non-overlap constraint: distance between centers >= sum of radii
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])

        return constraints

    result = optimize.minimize(

        x0=radii,
        bounds=[(0.0, 1.0) for _ in range(26)],
        method='SLSQP',

    # If optimization fails, return all-zero radii to avoid invalid packing
