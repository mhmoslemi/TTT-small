import numpy as np
import matplotlib.pyplot as plt

from scipy.optimize import minimize
import math

def pack_circles(n, max_radius=0.1):
    """Generates a rough hexagonal packing of n circles."""
    circles = []
    if n == 0:
        return np.array([]), np.array([])
    # Create a grid of circles in a hexagonal arrangement
    cols = int(math.sqrt(n))
    rows = (n + cols - 1) // cols
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j + 0.5 * i
            else:
                x = j + 0.5 * (i + 1)
            y = i
            r = max_radius
            circles.append((x, y, r))
    return np.array(circles)

def simulate_packing(config):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
