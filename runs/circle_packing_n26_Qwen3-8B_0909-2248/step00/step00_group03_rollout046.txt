    # Convert to numpy for vectorization
    centers = np.array([xs, ys]).T
    radii = np.full(centers.shape[0], 0.05)  # Initial small radii

    # Define the optimization objective: maximize sum of radii
    # Subject to constraints on non-overlapping and boundary constraints

    def objective(x):
        # x is a flattened array of [x0, y0, r0, x1, y1, r1, ...]
        # Reconstruct centers and radii
        flat_x = x[:len(centers) * 2]
        flat_r = x[len(centers) * 2:]
        centers_opt = np.reshape(flat_x, (len(centers), 2))
        radii_opt = flat_r

        # Ensure all radii are non-negative
        for r in radii_opt:
            if r < 0:
                return -np.inf

        # Calculate total radius sum
        total = np.sum(radii_opt)

        for i in range(len(centers_opt)):
