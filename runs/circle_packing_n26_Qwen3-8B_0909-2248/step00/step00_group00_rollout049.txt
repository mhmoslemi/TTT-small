import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initialize an approximate hexagonal grid of 26 circles in the unit square
    # The grid is sorted by rows, with each row shifted horizontally to form a hexagonal pattern

    # The number of circles in each row
    row_sizes = [3, 3, 3, 3, 3, 2, 2, 2, 2, 2, 1]  # Total of 26 circles
    num_rows = len(row_sizes)

    # Place the rows on a vertical scale from 0 to 1, spacing the rows vertically
    row_height = 1 / (num_rows + 1)
    max_radius = 0.1

    centers = []
    radii = []

    for i in range(num_rows):
        num_circles = row_sizes[i]
        row_y = i * row_height + row_height / 2
        spacing = 2 * max_radius  # Initial rough spacing between centers

        # For even rows, center the row horizontally
        if i % 2 == 0:
            x_start = 0.5 - (num_circles - 1) * spacing / 2
        else:
            x_start = 0.5 - (num_circles - 1) * spacing / 2 + spacing / 2

        for j in range(num_circles):
            x = x_start + j * spacing
            y = row_y
            centers.append([x, y])
            radii.append(max_radius)

    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Constraints:
    # 1. Each circle must be inside the unit square
    # 2. Each circle must be at least 1e-8 away from the boundary
    # 3. Each pair of circles must be at least 1e-8 apart from each other

    def constraint_boundary(x):
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = x[i]
            if (x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8 or
                y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8):
                return -1
        return 0

    # Define bounds for radii
    bounds = [(0, 1)] * len(radii)

    # Optimization settings
    constraints = (
        {'type': 'ineq', 'fun': constraint_boundary},
    )

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    radii_opt = result.x
    sum_radii = np.sum(radii_opt)

    return centers, radii_opt, sum_radii
