import scipy.optimize as opt

def run_packing():

    # Initial guess for radii
    radii = np.full_like(centers, 0.05)  # Small initial radius

    # Objective function to maximize sum of radii (minimize negative sum)
    def objective(radii_flat):
        return -np.sum(radii_flat)

    # Constraints for boundaries and no overlaps
    def constraint_boundary(radii_flat):
        # Ensure each circle is inside the square [0,1]×[0,1]
        # We reshape the flat radii and check each circle
        radii = radii_flat.reshape(centers.shape[0])
        violation = np.zeros_like(radii)
        for i in range(centers.shape[0]):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                violation[i] = 1
        return violation

    def constraint_overlap(radii_flat):
        # Ensure all circles are mutually non-overlapping
        radii = radii_flat.reshape(centers.shape[0])
        violation = np.zeros_like(radii)
        for i in range(centers.shape[0]):

                dy = centers[i, 1] - centers[j, 1]

                    violation[i] = 1
