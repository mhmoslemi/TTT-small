import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial grid configuration
    num_circles = 26
    n_rows = 5
    n_cols = 5
    padding = 0.05  # Small padding to avoid edge overlaps

    # Initial grid of centers
    x = np.linspace(0 + padding, 1 - padding, n_cols)
    y = np.linspace(0 + padding, 1 - padding, n_rows)
    centers = np.array([[x[i], y[j]] for j in range(n_rows) for i in range(n_cols)])
    radii = np.full(num_circles, 0.02)

    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Constraints: circles must stay inside square and not overlap
    def constraint_func(radii):
        dist_matrix = cdist(centers, centers)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(num_circles):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

    # Optimize
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1.0) for _ in range(num_circles)],
        constraints={'type': 'constraint', 'fun': constraint_func},
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract the optimal centers and radii
    if not result.success:

        raise ValueError("Final validation failed")

    total_sum = np.sum(optimal_radii)
    print(f"Final sum of radii: {total_sum}")

    return optimal_centers, optimal_radii, total_sum
