import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Define grid points inside the unit square
    # Using a hexagonal grid for initial placement
    centers = []
    for i in range(5):
        for j in range(5):
            x = (i + 0.5) / 5.0
            y = (j + 0.5) / 5.0
            centers.append([x, y])
    centers = np.array(centers)
    
    # Initialize radii as variables for optimization
    radii = cp.Variable(n)

    # Define the objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Define constraints
    constraints = []

    # Constraint: Each circle stays within the unit square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-8)
        constraints.append(x + radii[i] <= 1 + 1e-8)
        constraints.append(y - radii[i] >= -1e-8)
        constraints.append(y + radii[i] <= 1 + 1e-8)

    # Constraint: No overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Define and solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, max_iter=1000)
