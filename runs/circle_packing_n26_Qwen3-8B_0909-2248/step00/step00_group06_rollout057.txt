import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    def get_hexagonal_grid(n, max_x, max_y, padding):
        """Generate a hexagonal grid with n circles over [0, max_x] x [0, max_y]"""
        row_count = int(np.sqrt(n))
        col_count = int(np.ceil(n / row_count))
        offset = 0.5
        grid = []
        for i in range(row_count):
            for j in range(col_count):
                if i % 2 == 0:
                    x = j + offset
                else:
                    x = j + offset - 0.5
                y = i
                if x <= max_x - padding and y <= max_y - padding:
                    grid.append([x, y])
        return np.array(grid)

    def get_corner_circles(n, max_x, max_y, padding):

        circles = []
        for i in range(n):
            if i % 4 == 0:
                x, y = 0 + padding, 0 + padding
            elif i % 4 == 1:
                x, y = max_x - padding, 0 + padding
            elif i % 4 == 2:
                x, y = max_x - padding, max_y - padding
            else:
                x, y = 0 + padding, max_y - padding
            circles.append([x, y])
        return np.array(circles)

    def get_boundaries_circles(n, max_x, max_y, padding):
        """Generate boundary circles to maximize radius along edges"""
        circles = []
        for i in range(n):
            if i % 4 == 0:
                x, y = 0 + padding, np.random.uniform(padding, max_y - padding)
            elif i % 4 == 1:
                x, y = np.random.uniform(padding, max_x - padding), 0 + padding

                x, y = max_x - padding, np.random.uniform(padding, max_y - padding)

                x, y = np.random.uniform(padding, max_x - padding), max_y - padding
            circles.append([x, y])
        return np.array(circles)

    def pack_circles(max_x, max_y, n):
        """Packs n circles in [0,1]^2 to maximize the sum of their radii"""
        padding = 0.01
        corner_circles = get_corner_circles(n // 4, max_x, max_y, padding)
        boundary_circles = get_boundaries_circles(n - len(corner_circles), max_x, max_y, padding)

        centers = np.vstack([corner_circles, boundary_circles, hex_grid])

            return -np.sum(radii)

        def constraint_overlap(radii):
            dist_matrix = distance_matrix(centers, centers)
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    if dist_matrix[i, j] < radii[i] + radii[j] - 1e-6:
                        return -1
            return 0

            for i in range(len(centers)):
                x, y = centers[i]
                if x - radii[i] < -1e-6 or x + radii[i] > 1 + 1e-6 or y - radii[i] < -1e-6 or y + radii[i] > 1 + 1e-6:
                    return -1
            return 0

        bounds = [(0, 1) for _ in range(len(centers))]
        cons = [
            {"type": "ineq", "fun": constraint_overlap},
            {"type": "ineq", "fun": constraint_boundary}
        ]

        result = minimize(objective, x0=np.ones(len(centers)), bounds=bounds, constraints=cons)
        radii = result.x
        return centers, radii, np.sum(radii)

    centers, radii, total = pack_circles(1, 1, 26)
    return centers, radii, total
