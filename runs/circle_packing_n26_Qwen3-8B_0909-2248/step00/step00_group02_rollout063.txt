import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define the number of circles
    n = 26

    # Step 1: Initial layout - grid with some adjustment
    # Use a grid with 6 rows and 5 columns for a rough start
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 6)
    centers = np.array([ [x, y] for y in grid_y for x in grid_x ])

    # Step 2: Add a hexagonal pattern in the center
    hex_centers = []
    for i in range(3):
        for j in range(4):
            x = 0.5 + (j - 1.5) * 0.25
            y = 0.5 + (i - 1.0) * 0.25 + (i % 2) * 0.1
            if 0 <= x <= 1 and 0 <= y <= 1:
                hex_centers.append([x, y])
    centers = np.vstack((centers, hex_centers))
    centers = centers[:n]

    # Step 3: Define objective function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Negative for minimization

    # Step 4: Define constraints
    def constraint_overlap(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx**2 + dy**2
                if dist_sq < (radii[i] + radii[j])**2 - 1e-12:
                    return -1
        return 0

    def constraint_boundary(radii, centers):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return -1
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -1
        return 0

    # Step 5: Setup the optimization problem
    bounds = [(0, 1) for _ in range(n)]  # All radii are positive and within [0, 1]
    cons = [
        {'type': 'ineq', 'fun': lambda r: -constraint_overlap(r, centers)},
        {'type': 'ineq', 'fun': lambda r: -constraint_boundary(r, centers)}
    ]

    # Initial guess for radii
    initial_guess = np.full(n, 0.05)

    # Run the optimization
    result = opt.minimize(
        objective,
        initial_guess,
        args=(centers,),
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 200}
    )

    # Extract the final centers and radii
    radii = result.x
    centers = np.array(centers)

    # Return the result
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
