import cvxpy as cp

def run_packing():

        centers[i, 1] = row / 5 + np.random.rand() * 0.1

    # Variables
    radii = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    for i in range(n):
        # Each circle is within the unit square
        constraints.append(x[i, 0] - radii[i] >= -1e-12)
        constraints.append(x[i, 0] + radii[i] <= 1 + 1e-12)
        constraints.append(x[i, 1] - radii[i] >= -1e-12)

    for i in range(n):

    # Create problem and solve
    problem = cp.Problem(objective, constraints)
