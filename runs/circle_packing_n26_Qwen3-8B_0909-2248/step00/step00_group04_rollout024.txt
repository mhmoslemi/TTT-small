import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import Bounds
import warnings

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(x, n_circles):
    # x is a vector of [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    centers = x.reshape((n_circles, 2))
    radii = x[n_circles*2 : ]  # Extract radii from the end of the vector

    # Initial objective: maximize sum of radii (we use negative for minimization)
    obj = -np.sum(radii)

    # Add penalty for being outside the square
    for i in range(n_circles):
        x_i, y_i = centers[i]
        r_i = radii[i]
        if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
            obj -= 1e3 * r_i
        if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
            obj -= 1e3 * r_i

    # Add penalty for overlapping circles
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                obj -= 1e3 * radii[i] + 1e3 * radii[j]

    return obj

def run_packing():
    n_circles = 26
    n_params = n_circles * 2 + n_circles  # x, y, r for each circle

    # Initial configuration: 3 layers of circles
    # First layer (centered)
    centers = []

        angle = 2 * np.pi * i / n_circles
        r = 0.1  # Initial small radius
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Flatten parameters for optimization
    x0 = np.zeros(n_params)
    x0[:n_circles*2] = centers.flatten()
    x0[n_circles*2:] = radii

    # Define bounds for x and y coordinates, and radii
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(-1, 1), (-1, 1), (0, 1)])  # x, y, r

    # Optimization
    result = minimize(
        objective_function,
        x0,
        args=(n_circles,),
        method='Nelder-Mead',
        bounds=bounds,
        options={'xatol': 1e-6, 'fatol': 1e-6, 'disp': False}

    # Extract final centers and radii

    success, message = validate_packing(centers_final, radii_final)
    if not success:

    return centers_final, radii_final, np.sum(radii_final)
