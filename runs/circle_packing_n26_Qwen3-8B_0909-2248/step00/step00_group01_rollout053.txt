import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial positions: hybrid grid arrangement with some rows of 7 and some of 6
    # Row 1: 7 circles horizontally
    # Row 2: 6 circles horizontally
    # Row 3: 5 circles horizontally
    # Row 4: 4 circles horizontally
    # Row 5: 3 circles horizontally
    # Row 6: 2 circles horizontally
    # Row 7: 1 circle horizontally (corner)
    # This brings total = 7+6+5+4+3+2+1 = 28. We'll remove 2 circles for spacing / row distances

    # Starting from top-left (0,0)
    positions = []
    for i in range(7):
        x = i / 6.0
        y = 0.3
        positions.append([x, y])

    for i in range(6):
        x = i / 5.0
        y = 0.5
        positions.append([x, y])

    for i in range(5):
        x = i / 4.0
        y = 0.7
        positions.append([x, y])

    for i in range(4):
        x = i / 3.0
        y = 0.9
        positions.append([x, y])

    # Remove two circles for spacing
    positions.pop(1)
    positions.pop(2)

    centers = np.array(positions)
    radii = np.full(len(centers), 0.05)

    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    def constraint_radii(radii):
        return np.sum(radii)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(len(radii))]

    # Constraint to ensure circle positions and radii are within square
    def constraint_boundary(radii):
        return np.sum((centers - radii[:, np.newaxis])**2) - 1.0

    # Nonlinear constraints for circle non-overlap
    def constraint_overlap(radii):
        N = len(radii)
        total = 0
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    total += (radii[i] + radii[j] - dist) - 1e-12
        return total

    # Initial guess for radii
    initial_guess = np.full(len(radii), 0.05)

    # Set up optimization
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints
    )

    # Extract result
    radii = result.x
    centers = centers.copy()
    total_sum = np.sum(radii)
    return centers, radii, total_sum
