import scipy.optimize as optimize
import random

def run_packing():
    def generate_grid(n):

        grid = []
        for i in range(5):
            for j in range(5):
                grid.append([i / 4, j / 4])
        grid = np.array(grid)
        grid = np.unique(grid, axis=0)
        needed = n - len(grid)
        for _ in range(needed):
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            grid = np.vstack((grid, [x, y]))
        grid = np.unique(grid, axis=0)
        centers = grid[:n]
        return centers

    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)  # Minimize the negative sum to maximize the total

    def constraint_func(radii, centers):
        """Define constraints for the optimization problem."""
        n = len(radii)
        constraints = []
        # Boundary constraints: each circle must be within the unit square
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Generate initial layouts and optimize
    max_attempts = 5
    best_sum = 0
    best_centers = np.array([])
    best_radii = np.array([])

        bounds = [(0, 1) for _ in range(26)]

            objective,
            initial_radii,
            bounds=bounds,

            method='SLSQP',

                'disp': False
