from scipy.optimize import minimize

def run_packing():

        """Generate a hexagonal grid of n circles inside the unit square."""
        max_row = int(np.sqrt(n))
        grid = []
        for i in range(max_row):

                x = (i + 0.5) * spacing
                y = j * spacing
                grid.append([x, y])

            grid.pop()

        return -np.sum(radii)

        """Compute pairwise non-overlap constraints."""
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j]
        return 0.0

    def constraint_boundary(radii, centers):
        """Check if circles are within the unit square."""
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return x - radii[i]
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
