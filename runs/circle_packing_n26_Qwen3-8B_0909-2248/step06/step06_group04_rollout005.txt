import scipy.optimize as optimize
from scipy.optimize import minimize

def run_packing():

        """
        Generate an initial layout with 4 corner circles and the rest in a hexagonal grid.
        """
        # Define the corners
        corner_points = np.array([
            [0.1, 0.1],
            [0.9, 0.1],
            [0.9, 0.9],
            [0.1, 0.9]
        ])

        # Generate hexagonal layout for the remaining circles
        n_hex = n - 4
        hex_centers = []
        for i in range(n_hex):
            x = 0.5 + 0.1 * np.cos(2 * np.pi * i / n_hex)
            y = 0.5 + 0.1 * np.sin(2 * np.pi * i / n_hex)
            hex_centers.append([x, y])

        # Combine corner and hexagonal points
        all_centers = np.vstack((corner_points, hex_centers))
        return all_centers[:n]

    def objective(radii):
        """Maximize the sum of radii."""
        return -np.sum(radii)

    def constraint_fun(radii, centers):
        """Compute constraints for the optimization problem."""
        n = len(radii)
        constraints = []
        # Boundary constraints: circles must be inside the unit square
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    centers = generate_corner_hexagonal_layout(26)

    initial_radii = np.zeros(26)

        initial_radii[i] = min_dist * 0.7  # use 70% of the minimum edge distance as initial radius

    # Optimization bounds
    bounds = [(0, 1) for _ in range(26)]

    result = minimize(

        constraints={
