import scipy.optimize as optimize

            points.append([x, 0.1])
            points.append([x, 0.9])

            points.append([0.9, y])

        points = np.array(list(set(map(tuple, points))))
        return points[:n]

        """Maximize the sum of the radii."""
        return -np.sum(radii)

        """Non-overlapping constraints between all pairs of circles."""
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):

                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def boundary_constraints(radii, centers):

            constraints.append(y - radii[i])
            constraints.append(y + radii[i])

    centers = generate_initial_layout(26)
