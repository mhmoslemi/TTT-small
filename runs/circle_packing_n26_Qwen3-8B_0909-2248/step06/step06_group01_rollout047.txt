import scipy.optimize as optimize
from scipy.optimize import minimize

def generate_layout(n):
    """Generate structured and adaptive layout with extra edges/corners focus."""
    boundary_points = []
    # Grid points
    for x in [0.1, 0.3, 0.5, 0.7, 0.9]:

            boundary_points.append([x, y])

        boundary_points.append([0.95, y])

    boundary_points.append([0.05, 0.05])
    boundary_points.append([0.95, 0.05])
    boundary_points.append([0.05, 0.95])
    boundary_points.append([0.95, 0.95])
    # Add center
    boundary_points.append([0.5, 0.5])
    # Convert to numpy and remove duplicates
    boundary_points = np.array(boundary_points)
    boundary_points = np.unique(boundary_points, axis=0)
    # Add more points if needed
    if n > len(boundary_points):
        for i in range(n - len(boundary_points)):

                x = np.random.uniform(0.2, 0.8)
                y = np.random.uniform(0.2, 0.8)

                x = np.random.uniform(0.1, 0.9)
                y = np.random.uniform(0.1, 0.9)
                boundary_points = np.vstack((boundary_points, [x, y]))
    return boundary_points[:n]

def objective(radii):
    """Maximize the sum of the radii."""
    return -np.sum(radii)

def constraints(radii, centers):
    """Return constraints for the optimization."""
    n = len(radii)
    constraints = []
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i])
        constraints.append(x + radii[i])
        constraints.append(y - radii[i])
        constraints.append(y + radii[i])
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            constraints.append(dist - (radii[i] + radii[j]))
    return constraints

def run_packing():
