import numpy as np
import scipy.optimize as optimize

def run_packing():

        points = []
        # Corners and boundary points
        points.append([0.05, 0.05])
        points.append([0.95, 0.05])
        points.append([0.95, 0.95])
        points.append([0.05, 0.95])
        points.append([0.05, 0.5])
        points.append([0.95, 0.5])
        points.append([0.5, 0.05])
        points.append([0.5, 0.95])
        
        # Add edge points
        for x in np.linspace(0.05, 0.95, 10):
            points.append([x, 0.5])
            points.append([x, 0.05])
            points.append([x, 0.95])
            
        for y in np.linspace(0.05, 0.95, 10):
            points.append([0.05, y])
            points.append([0.5, y])
            points.append([0.95, y])
        
        # Ensure we have enough unique points
        unique_points = np.array(list(set(map(tuple, points))))
        if len(unique_points) < n:
            # Add more points if needed
            additional_points = []
            for i in range(n - len(unique_points)):
                # Random points within the square
                x = np.random.uniform(0.1, 0.9)
                y = np.random.uniform(0.1, 0.9)
                additional_points.append([x, y])
            unique_points = np.vstack((unique_points, additional_points))
        elif len(unique_points) > n:
            unique_points = unique_points[:n]
        return unique_points

    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)

    def constraints(radii, centers):
        """Non-overlapping and boundary constraints."""
        n = len(radii)
        constraints = []
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Generate initial layout for 26 circles
    centers = generate_optimal_layout(26)
    initial_radii = np.full(26, 0.05)

    bounds = [(0, 1) for _ in range(26)]

    # Set up the optimization with SLSQP
    result = optimize.minimize(

        initial_radii,
        bounds=bounds,
        constraints={

            'fun': lambda r: constraints(r, centers)
        },
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 2000, 'disp': False}
    )

    if not result.success:
        raise ValueError("Optimization failed with message: %s" % result.message)

    radii = result.x
    total_sum = np.sum(radii)

    return centers, radii, total_sum
