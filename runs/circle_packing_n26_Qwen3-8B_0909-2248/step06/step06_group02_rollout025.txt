import scipy.optimize as optimize

def run_packing():

        - Use hexagonal packing for inner circles
        - Use a probabilistic approach for boundary circles
        - Add some random points to fill gaps
        """

            for j in range(10):
                grid_points.append([i / 9, j / 9])

        kmeans = KMeans(n_clusters=n, random_state=0).fit(grid_points)
        centers = kmeans.cluster_centers_

        return -np.sum(radii)  # Minimize negative sum to maximize total

        # Boundary constraints: circles must be inside the unit square
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
