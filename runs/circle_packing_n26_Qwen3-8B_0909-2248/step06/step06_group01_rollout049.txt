import numpy as np
import scipy.optimize as optimize

def generate_initial_layout(n):
    """Generate an initial layout with a combination of corner, edge, and central points."""
    points = []

    # Add corners
    points.append([0, 0])
    points.append([1, 0])
    points.append([0, 1])
    points.append([1, 1])

    # Add midpoints of edges
    points.append([0.5, 0])
    points.append([1, 0.5])
    points.append([0.5, 1])
    points.append([0, 0.5])

    # Add center
    points.append([0.5, 0.5])

    # Add additional points spaced evenly in a hexagonal grid around the core
    for angle in np.linspace(0, 2 * np.pi, 10):
        x, y = 0.3 + 0.2 * np.cos(angle), 0.3 + 0.2 * np.sin(angle)
        points.append([x, y])

    # Convert to array and remove duplicates
    points = np.array(points)
    points = np.unique(points, axis=0)

    # If n exceeds the number of generated points, add more points via grid
    for i in range(n - len(points)):
        x = np.random.uniform(0.2, 0.8)
        y = np.random.uniform(0.2, 0.8)
        points = np.vstack((points, [x, y]))

    return points[:n]

def objective(radii, centers):
    """Objective function: maximize the sum of radii."""
    return -np.sum(radii)

def constraints(radii, centers):
    """Generate constraints for each circle to stay within square and not overlap with others."""
    n = len(radii)
    constraints = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i])
        constraints.append(x + radii[i])
        constraints.append(y - radii[i])
        constraints.append(y + radii[i])

    # Pairwise overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            constraints.append(dist - (radii[i] + radii[j]))

    return constraints

def run_packing():
    # Generate initial layout (with edge points, corners, and central cluster)
    centers = generate_initial_layout(26)

    # Initial radii guess: small, non-zero
    initial_radii = np.full(26, 0.05)

    # Bounds for each radius
    bounds = [(0, 1) for _ in range(26)]

    # Optimization settings
    # Use a more efficient approach with dynamic bounds and multiple iterations
    result = optimize.minimize(
        fun=lambda r: objective(r, centers),
        x0=initial_radii,
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': lambda r: constraints(r, centers)
        },
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 500}
    )

    if not result.success:
        raise ValueError("Optimization failed with message: %s" % result.message)

    radii = result.x
    total_sum = np.sum(radii)

    return centers, radii, total_sum
