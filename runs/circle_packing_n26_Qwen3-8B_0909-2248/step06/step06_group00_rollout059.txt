import numpy as np
import scipy.optimize as optimize
import random

def run_packing():
    def generate_initial_layout(n):
        """Generate an initial layout with strategic placement: hexagonal inside, edge/corner points."""
        if n == 0:
            return np.array([]), np.array([])

        # Generate a hexagonal grid inside the square
        hex_centers = []
        for i in range(5):
            for j in range(5):
                x = 0.2 + (i) * 0.2
                y = 0.2 + (j) * 0.2
                if i + j <= 5:  # ensure hexagonal patterns
                    hex_centers.append([x, y])
        hex_centers = np.array(hex_centers)

        # Add edge and corner points
        edge_centers = []
        for x in [0.1, 0.3, 0.5, 0.7, 0.9]:
            edge_centers.append([x, 0.1])
            edge_centers.append([x, 0.9])
        for y in [0.1, 0.3, 0.5, 0.7, 0.9]:
            edge_centers.append([0.1, y])
            edge_centers.append([0.9, y])
        edge_centers = np.array(edge_centers)

        # Combine and make unique
        all_centers = np.vstack((hex_centers, edge_centers))
        all_centers = np.unique(all_centers, axis=0)
        needed = n - len(all_centers)
        for _ in range(needed):
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            all_centers = np.vstack((all_centers, [x, y]))
        all_centers = np.unique(all_centers, axis=0)

        # Trim to n
        centers = all_centers[:n]
        return centers, np.full(n, 0.1)

    def objective(radii):
        """Maximize the total sum of the radii."""
        return -np.sum(radii)

    def constraint_func(radii, centers):
        """Define constraints for the optimization problem."""
        constraints = []

        # Boundary constraints
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append(x - radii[i])  # x - r >= 0
            constraints.append(x + radii[i])  # x + r <= 1
            constraints.append(y - radii[i])  # y - r >= 0
            constraints.append(y + radii[i])  # y + r <= 1

        # Pairwise non-overlap constraints
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))  # dist >= r_i + r_j

        return constraints

    # Generate initial layouts and optimize
    max_attempts = 5
    best_sum = 0
    best_centers = np.array([])
    best_radii = np.array([])

    for _ in range(max_attempts):
        centers, initial_radii = generate_initial_layout(26)
        bounds = [(0, 1) for _ in range(26)]
        result = optimize.minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints={
                'type': 'ineq',
                'fun': lambda r: constraint_func(r, centers)
            },
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 3000, 'eps': 1e-8}

        if result.success:

                best_radii = result.x
