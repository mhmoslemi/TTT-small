import numpy as np
import scipy.optimize as optimize

def run_packing():
    def generate_hexagonal_layout(n):
        """Generate a hexagonal grid of points with more strategic placement."""
        if n == 0:
            return np.array([]), np.array([])
        
        points = []
        radii = []
        for i in range(n):
            x = i % 6 / 6
            y = i // 6
            points.append([x, y])
            radii.append(0.05)
        return np.array(points), np.array(radii)

    def objective(radii):
        """Maximize the total sum of the radii."""
        return -np.sum(radii)  # Minimize the negative sum to maximize total
    
    def constraint_func(radii, centers):
        """Define constraints for the optimization problem."""
        constraints = []
        n = len(radii)
        
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Generate initial hexagonal layout
    centers, _ = generate_hexagonal_layout(26)
    initial_radii = np.full(26, 0.05)
