from scipy.optimize import minimize

def generate_refined_hexagonal_grid(n):
    """Generate a refined and boundary-optimized hexagonal grid layout."""
    points = []
    # Predefine points near edges and corners to optimize boundary utilization
    edge_points = [
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9],
        [0.1, 0.5], [0.9, 0.5], [0.5, 0.1], [0.5, 0.9], 
        [0.1, 0.2], [0.1, 0.8], [0.9, 0.2], [0.9, 0.8],
        [0.2, 0.1], [0.8, 0.1], [0.2, 0.9], [0.8, 0.9],
    ]
    # Generate hexagonal grid points in the central area
    for i in range(5):
        for j in range(5):
            x = 0.2 + i * 0.1 + j * 0.1 * np.sin(np.pi * i / 5)
            y = 0.2 + j * 0.1 + i * 0.1 * np.cos(np.pi * j / 5)
            points.append([x, y])
    # Convert points to NumPy
    points = np.array(points)
    # Add edge and corner points
    points = np.vstack((points, np.array(edge_points)))
    # Remove duplicates
    points = np.unique(points, axis=0)
    # Ensure we have at most n points
    return points[:n]

def run_packing():
    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)  # Minimize negative sum to maximize total

    def non_overlap_constraints(radii, centers):
        """Non-overlapping constraints between all pairs of circles."""
        constraints = []
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def boundary_constraints(radii, centers):
        """Boundary constraints for all circles to stay within [0,1]^2."""
        constraints = []
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Define the layout with preoptimized boundary points
    centers = generate_refined_hexagonal_grid(26)

    initial_radii = np.full(26, 0.05)

    bounds = [(0, 1) for _ in range(26)]

    radii = result.x
