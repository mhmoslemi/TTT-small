import scipy.optimize as optimize
from scipy.optimize import minimize

def generate_hexagonal_grid(n):

    boundary_points = []
    
    # Corners
    boundary_points.append([0.1, 0.1])
    boundary_points.append([0.9, 0.1])
    boundary_points.append([0.9, 0.9])
    boundary_points.append([0.1, 0.9])

    for x in np.linspace(0.1, 0.9, 5):
        boundary_points.append([x, 0.5])
    for y in np.linspace(0.1, 0.9, 5):
        boundary_points.append([0.5, y])
    
    # Interior
    interior_x = np.linspace(0.2, 0.8, 5)
    interior_y = np.linspace(0.2, 0.8, 5)
    for x in interior_x:
        for y in interior_y:
            boundary_points.append([x, y])
    
    # Reduce duplicate points
    boundary_points = np.array(boundary_points)
    boundary_points = np.unique(boundary_points, axis=0)
    
    # Ensure we have exactly n points, filling up with more interior points if needed
    if len(boundary_points) < n:
        for _ in range(n - len(boundary_points)):
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            boundary_points = np.vstack((boundary_points, [x, y]))
    
    return boundary_points[:n]

def run_packing():

    def non_overlap_constraints(radii):
