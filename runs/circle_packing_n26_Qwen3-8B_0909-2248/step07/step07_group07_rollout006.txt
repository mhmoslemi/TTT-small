import scipy.optimize as optimize

def run_packing():

            return np.array([]), np.array([])

        scale = 0.25
        points = []
        for row in range(int(np.sqrt(n)) + 1):
            for col in range(int(np.sqrt(n)) + 1):
                # Offset for even rows (hexagonal packing)
                x = col * (1 + 1/np.sqrt(3)) * scale
                y = row * (1 + 1/np.sqrt(3)) * scale
                # Ensure within [0, 1]
                x = np.clip(x, 0, 1)
                y = np.clip(y, 0, 1)
                points.append([x, y])

            x = np.random.uniform(0.2, 0.8)
            y = np.random.uniform(0.2, 0.8)
            points.append([x, y])
        
        points = np.unique(np.array(points), axis=0)
        return points[:n], np.array([0.05 for _ in range(n)])

        """Maximize the total sum of the radii."""
        return -np.sum(radii)  # Minimize the negative sum to maximize total

        constraints = []
        n = len(radii)
        
        # Boundary constraints: each circle must be within the unit square
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
