        all_points = np.unique(all_points, axis=0)

        # Keep only the first n points
        if len(all_points) >= n:
            return all_points[:n], np.array([0.05 for _ in range(n)])

        for _ in range(n - len(all_points)):

        n = len(radii)

        for i in range(n):
            x, y = centers[i]
