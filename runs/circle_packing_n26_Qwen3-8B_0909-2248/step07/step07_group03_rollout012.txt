import scipy.optimize as optimize
import random

def run_packing():
    def initialize_centers(n):
        """Initialize centers with strategic placement: corners and hexagonal grid."""
        corners = np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]])
        hex_centers = []
        for x in np.linspace(0.2, 0.8, 5):
            for y in np.linspace(0.2, 0.8, 5):
                hex_centers.append([x, y])
        hex_centers = np.array(hex_centers)
        initial_centers = np.vstack((corners, hex_centers))
        initial_centers = np.unique(initial_centers, axis=0)
        if len(initial_centers) >= n:
            return initial_centers[:n]
        remaining = n - len(initial_centers)
        for _ in range(remaining):
            x = np.random.uniform(0.2, 0.8)
            y = np.random.uniform(0.2, 0.8)
            initial_centers = np.vstack((initial_centers, [x, y]))
        return initial_centers[:n]

    def objective(radii):
        """Maximize the total sum of the radii."""
        return -np.sum(radii)  # Minimize the negative sum to maximize total
    
    def constraint_func(radii, centers):
        """Define constraints for the optimization problem."""
        constraints = []
        n = len(radii)
        # Boundary constraints: each circle must be within the unit square
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    centers = initialize_centers(26)

    result = optimize.minimize(

        },
        method='COBYLA',
        options={'ftol': 1e-10, 'maxiter': 2000, 'disp': False}
    )
