import scipy.optimize as optimize
import random

def run_packing():

        for i in range(n):
            # Generate a hexagonal lattice with spacing of 0.2
            x = i % 5 * 0.2
            y = i // 5 * (np.sqrt(3) * 0.2)
            # Adjust to fit within the unit square
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            points.append([x, y])
        # Extend with corner and edge points if needed
        if len(points) < n:

                points.append([x, y])
        return np.array(points), np.array([0.01 for _ in range(n)])
    
    def objective(radii):

        return -np.sum(radii)  # Minimize the negative sum to maximize total
    
    def constraint_func(radii, centers):
        """Define constraints for the optimization problem."""
        constraints = []
        n = len(radii)
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints
    
    # Generate initial layout
