import scipy.optimize as optimize
import random

def run_packing():

        """Generate a hexagonal packing of n points in the center region."""
        max_radius = 0.15

            r = max_radius * (1 - np.sqrt(i) / n)
            x = 0.5 + r * np.cos(angle)
            y = 0.5 + r * np.sin(angle)
            centers.append([x, y])
        return centers

        unique_edges = np.unique(edges, axis=0)

            y = np.random.uniform(0.2, 0.8)

        return clusters[:n]

        """Maximize the total sum of the radii."""
        return -np.sum(radii)

        """Define constraints for the optimization problem."""
        constraints = []
        n = len(radii)
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

        """Optimize radii for given centers."""
        initial_radii = np.full(len(centers), 0.05)
        bounds = [(0, 1) for _ in range(len(centers))]
        result = optimize.minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints={
                'type': 'ineq',
                'fun': lambda r: constraint_func(r, centers)
            },
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 300, 'eps': 1e-8}
        )
        if not result.success:
            raise ValueError("Optimization failed with message: %s" % result.message)
