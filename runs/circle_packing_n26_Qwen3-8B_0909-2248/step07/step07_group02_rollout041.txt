                    x = i / (grid_size - 1)
                    y = j / (grid_size - 1)
                    grid.append([x, y])
        return np.array(grid)

        n = len(radii)
        dists = np.zeros((n, n))

                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx*dx + dy*dy)
                dists[j, i] = dists[i, j]
        return dists
