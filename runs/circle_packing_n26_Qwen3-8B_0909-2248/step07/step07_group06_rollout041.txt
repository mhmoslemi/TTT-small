import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    def generate_hexagonal_initial(n):
        """Generate a hexagonal-packed layout for the initial positions."""
        if n == 0:
            return np.array([]), np.array([])

        max_row = int(np.sqrt(n))
        if max_row * (max_row + 1) < n:
            max_row += 1

        # Space between centers: 1.5x radius
        spacing = 1.2
        centers = []
        
        for row in range(max_row):
            for col in range(max_row + 1):
                x = col * spacing
                y = row * spacing * math.sqrt(3) / 2
                if x < 1.0 and y < 1.0 and x > 0.0 and y > 0.0:
                    centers.append([x, y])
        
        # If more circles needed, we add additional rows or adjust
        row = max_row
        col = 0
        while len(centers) < n:
            x = col * spacing
            y = row * spacing * math.sqrt(3) / 2
            if x < 1.0 and y < 1.0 and x > 0.0 and y > 0.0:
                centers.append([x, y])
            col += 1
            row += 1
        
        centers = np.array(centers)
        # Cut off any circles that go beyond square
        centers = [c for c in centers if 0.0 <= c[0] <= 1.0 and 0.0 <= c[1] <= 1.0]
        centers = np.array(centers)
        return centers, np.array([0.05 for _ in range(len(centers))])

    def objective(radii):
        """Maximize total sum of the radii."""
        return -np.sum(radii)

    def constraint_func(radii, centers):
        """Constraints for non-overlap and boundary adherence."""
        constraints = []
        n = len(radii)
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def optimize_layout(centers):
        """Run the optimization with SLSQP."""
        initial_radii = np.full(len(centers), 0.05)
        bounds = [(0, 1) for _ in range(len(centers))]
        result = optimize.minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints={
                'type': 'ineq',
                'fun': constraint_func,
                'args': (centers,)
            },
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 2000, 'eps': 1e-8}
        )

            raise ValueError("Optimization failed with message: %s" % result.message)

                best_centers = centers
                best_radii = radii
        except Exception as e:
            print("Optimization failed:", e)
    
    return best_centers, best_radii, best_sum
