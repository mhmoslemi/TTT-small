import numpy as np
import scipy.optimize as optimize

def run_packing():

        # Generate a hexagonal grid layout with rows and columns
        rows = int(np.ceil(n ** 0.5))
        cols = int(np.ceil(n / rows))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n)])
        return centers

    def compute_overlap_mask(centers):
        n = len(centers)
        overlap = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                overlap[i, j] = dist
        return overlap

    def objective(radii):
        return -np.sum(radii)

    def dynamic_overlap_constraints(radii, centers, overlap):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(overlap[i, j] - (radii[i] + radii[j]))
        return constraints

    def boundary_constraints(radii, centers):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Generate hexagonal layout and initial radii

    # Optimization
    result = optimize.minimize(
