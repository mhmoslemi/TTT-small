import numpy as np
import scipy.optimize as optimize

def run_packing():
    def generate_hexagonal_layer(n):
        # Create a hexagonal grid with circular spacing.
        # Each circle is spaced at distance sqrt(3)/2 (approx 0.866)
        # to maintain hexagonal packing.
        row = int(np.ceil(n ** 0.5))
        col = int(np.ceil(n / row))
        x = np.linspace(0.1, 0.9, col)
        y = np.linspace(0.1, 0.9, row)
        centers = np.array([(x[i % col], y[i // col]) for i in range(n)])
        return centers

    def compute_overlap_mask(centers):
        n = len(centers)
        overlap = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx**2 + dy**2)
                overlap[i, j] = distance
        return overlap

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii, centers, overlap_mask):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = overlap_mask[i, j]
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(radii, centers):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Generate a hybrid hexagonal layout with two layers
    lower_layer = generate_hexagonal_layer(13)
    upper_layer = generate_hexagonal_layer(13)
    centers = np.vstack([lower_layer, upper_layer])

    # Optimization bounds
    bounds = [(0, 1) for _ in range(26)]

    # Initial guess: small radius for all
    initial_radii = np.full(26, 0.1)

    # Compute the overlap matrix
    overlap_mask = compute_overlap_mask(centers)

    # Define optimization problem
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers, overlap_mask)},
            {'type': 'ineq', 'fun': lambda r: constraint_boundary(r, centers)}
        ],
        method='SLSQP',
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
