import numpy as np
import scipy.optimize as optimize

def run_packing():

        rows = int(np.ceil(n ** 0.5))
        cols = int(np.ceil(n / rows))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n)])
        return centers

    def compute_dist_matrix(centers):
        n = len(centers)
        dist_matrix = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_matrix[i, j] = np.sqrt(dx*dx + dy*dy)
        return dist_matrix

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii, centers, dist_matrix):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = dist_matrix[i, j]
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(radii, centers):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Generate the hexagonal packing layout
    centers = generate_hexagonal_packing(26)

    # Run the SLSQP optimization
    result = optimize.minimize(

        bounds=bounds,
        constraints=[
