import numpy as np
import scipy.optimize as optimize

def run_packing():
    def generate_hexagonal_layout(n_circles):
        # Hexagonal packing with variable rows and columns
        rows = int(np.ceil(np.sqrt(n_circles)))
        cols = int(np.ceil(n_circles / rows))
        radius_guess = 0.05
        spacing = 2 * radius_guess
        x = []
        y = []
        for i in range(n_circles):
            row = i // cols
            col = i % cols
            x_pos = col * spacing + radius_guess
            y_pos = row * (spacing * np.sqrt(3)) + radius_guess
            x.append(x_pos)
            y.append(y_pos)
        centers = np.array([x, y]).T
        return centers

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(radii):
        constraints = []
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Generate initial layout using hexagonal packing
    centers = generate_hexagonal_layout(26)
    n = len(centers)

    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n)]

    cons = []
    cons.append({'type': 'ineq', 'fun': constraint_overlap})
    cons.append({'type': 'ineq', 'fun': constraint_boundary})

    initial_radii = np.full(n, 0.05)

    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='COBYLA',
        tol=1e-10,
