import scipy.optimize as optimize

def run_packing():

        centers = []

            # Randomly place circle in remaining space
            x = np.random.uniform(0.05, 0.95)
            y = np.random.uniform(0.05, 0.95)
            centers.append((x, y))
        return np.array(centers)

    def objective(radii):
        return -np.sum(radii)

    def non_overlap_constraints(radii, centers):
        constraints = []
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def boundary_constraints(radii, centers):
        constraints = []
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Generate greedy initial layout
    centers = generate_greedy_layout(26)
    initial_radii = np.full(26, 0.1)

    # Run optimization
    result = optimize.minimize(
