        col = int(np.ceil(n / row))
        x = np.linspace(0.1, 0.9, col)
        y = np.linspace(0.1, 0.9, row)
        centers = np.array([(x[i % col], y[i // col]) for i in range(n)])

        centers = []
        for i in range(layer):
            for j in range(i + 1):
                centers.append((0.2 + i * 0.15, 0.2 + j * 0.15))
            for j in range(i, 0, -1):
                centers.append((0.2 + j * 0.15, 0.7 - i * 0.15))
        return np.array(centers)

    def compute_overlap_mask(centers):
        n = len(centers)
        overlap = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                overlap[i, j] = dist
        return overlap

    def objective(radii):
        return -np.sum(radii)

        n = len(radii)
        constraints = []
