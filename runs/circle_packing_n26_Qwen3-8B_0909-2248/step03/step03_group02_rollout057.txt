import scipy.optimize as optimize

def run_packing():
    def generate_initial_centers(n):
        # Initial configuration with hexagonal packing strategy
        row_size = int(np.ceil(np.sqrt(n)))
        col_size = int(np.ceil(n / row_size))
        centers = []
        for i in range(n):
            col_idx = i % col_size
            row_idx = i // col_size
            x = 0.05 + col_idx * (1 - 0.1) / (col_size - 1)
            y = 0.05 + row_idx * (1 - 0.1) / (row_size - 1)
            if col_idx % 2 == 1:
                y += 0.05
            centers.append([x, y])
        return np.array(centers)

    def objective(radii):
        return -np.sum(radii)

                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_bounds(radii):

        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])

    initial_radii = np.full(n, 0.05)

        method='SLSQP',

    final_centers = centers
