import scipy.optimize as optimize

def run_packing():

        col = int(np.ceil(n / row))

            c = i // col
            x = 0.1 + r * (0.8 / (col - 1))
            y = 0.1 + c * (0.8 / (row - 1))
            if r > col // 2:
                x += 0.4 * (col - r - 1) / (col - 1)
            centers.append([x, y])
        return np.array(centers)

    centers = generate_hexagonal_layout(n)

        return -np.sum(radii)

            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
