import numpy as np
import scipy.optimize as optimize
from scipy.stats import qmc

def run_packing():
    n_circles = 26

    # Generate initial centers with a halton sequence for better spread
    samples = qmc.Halton(dtype=np.float64, n=n_circles, scramble=True)
    centers = samples[:, :2]
    # Scale to [0.1, 0.9] to ensure some space for radii
    centers = 0.1 + 0.8 * centers

    def objective(radii):
        return -np.sum(radii)

    # Constraint: circle-circle non-overlapping
    def constraint_func(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    # Constraint: circle must be within the unit square
    def boundary_func(radii):
        constraints = []
        for i in range(n_circles):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Initial guess for radii (uniform, slightly higher to allow for better expansion)
    initial_radii = np.full(n_circles, 0.06)

    bounds = [(0, 1.0) for _ in range(n_circles)]

    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_func},
        {'type': 'ineq', 'fun': boundary_func},
    ]

    # Use SLSQP with tighter tolerances, more iterations, and better settings
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={
            'ftol': 1e-10,          # Tighter tolerance for precision
            'maxiter': 1000,        # More iterations for convergence
            'eps': 1e-8             # Smaller step size for optimization
        }
    )

    radii = result.x
    total_sum = np.sum(radii)

    return centers, radii, total_sum
