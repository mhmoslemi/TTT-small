import scipy.optimize as optimize

def run_packing():

            (0.1, 0.1), (0.9, 0.1), (0.9, 0.9), (0.1, 0.9),
            (0.1, 0.5), (0.5, 0.1), (0.5, 0.9), (0.9, 0.5)
        ]
        # Place the remaining circles in a hexagonal packing

    def objective(radii):

        return constraints

            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])

    result = optimize.minimize(objective, initial_radii, bounds=bounds, constraints=cons)
