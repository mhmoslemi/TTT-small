import numpy as np
import cvxpy as cp
import matplotlib.pyplot as plt

def run_packing():
    # Initial layout: divide the unit square into 26 evenly spaced grid cells
    # to spread the circles across the square
    n = 26
    grid = np.linspace(0.1, 0.9, n)
    centers = np.array([[x, y] for x in grid for y in grid])
    radii = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: non-overlapping and within the unit square
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j])
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-8)
        constraints.append(x + radii[i] <= 1 + 1e-8)
        constraints.append(y - radii[i] >= -1e-8)
        constraints.append(y + radii[i] <= 1 + 1e-8)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6, max_iter=5000, verbose=False)

    # Extract results
    centers = np.array([[x, y] for x in grid for y in grid])
    radii = np.array([radii[i].value for i in range(n)])

    return centers, radii, total_sum
