import scipy.optimize as optimize

def run_packing():

        cols = int(np.ceil(n_circles / rows))

        x, y = np.meshgrid(np.linspace(spacing, 1 - spacing, cols), np.linspace(spacing, 1 - spacing, rows))
        centers = np.column_stack((x.ravel(), y.ravel()))
        return centers

    def generate_corner_pockets(n_circles):
        corners = np.array([
            [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
        ])
        pockets = np.array([
            [0.2, 0.2], [0.2, 0.8], [0.8, 0.2], [0.8, 0.8]
        ])
        edge_pockets = np.array([
            [0.2, 0.5], [0.8, 0.5], [0.5, 0.2], [0.5, 0.8]
        ])
        centers = np.vstack([
            corners[:n_circles // 4],
            pockets[:n_circles // 4],
            edge_pockets[:n_circles // 4]
        ])
        return centers

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def constraint_overlap(radii):

        constraints = []
        for i in range(n):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(radii):

            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    initial_radii = np.full(n, 0.05)

    result = optimize.minimize(objective, initial_radii, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x
