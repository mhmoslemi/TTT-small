import scipy.optimize as optimize

def run_packing():

        rows = int(np.ceil(n ** 0.5))
        cols = int(np.ceil(n / rows))

        x = np.linspace(0.15, 0.85, cols)
        y = np.linspace(0.15, 0.85, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n)])
        return centers

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []

                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def constraint_boundary(radii):
        constraints = []
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Create initial layout
    centers = initial_layout(26)

    n = len(centers)

    cons = []

    cons.append({'type': 'ineq', 'fun': constraint_boundary})

    # Run the optimization
    result = optimize.minimize(
