import scipy.optimize as optimize

def run_packing():

    # Generate a more refined initial configuration using a hexagonal packing
    def generate_hexagonal_packing(n):
        angles = np.linspace(0, 2*np.pi, n, endpoint=False)
        radii = np.sqrt(0.5 * (1 - np.cos(2*np.pi / n)))  # Hexagon-based radius
        x = np.cos(angles) * radii
        y = np.sin(angles) * radii
        # Shift to fit within the unit square
        shift = 0.5 * (1 - np.max(np.abs(x)) - np.max(np.abs(y)))
        x += shift
        y += shift
        centers = np.column_stack((x, y))

        centers = np.clip(centers, 0, 1)
        return centers

    centers = generate_hexagonal_packing(n_circles)

    def objective(radii):
        return -np.sum(radii)

    # Constraint for circle-circle spacing
    def constraint_circle(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def constraint_boundary(radii):
        constraints = []
        for i in range(n_circles):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i] - 1)
            constraints.append(y - radii[i])
            constraints.append(y + radii[i] - 1)

    result = optimize.minimize(

        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-12, 'maxiter': 1000}
