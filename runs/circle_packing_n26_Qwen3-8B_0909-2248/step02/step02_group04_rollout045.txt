def run_packing():

        rows = max_rows

        centers = []
        for i in range(rows):
            for j in range(cols):

                    break

        return centers

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []
        for i in range(n):

    result = optimize.minimize(
