import scipy.optimize as optimize

def run_packing():

        cols = int(np.ceil(n_circles / rows))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n_circles)])
        return centers

    def constraint_overlap(i, j, centers, radii):
        dist = np.linalg.norm(centers[i] - centers[j])
        return dist - (radii[i] + radii[j])

    def get_constraints(centers):

        constraints = []

                    return constraint_overlap(i, j, centers, radii)
                constraints.append({'type': 'ineq', 'fun': c})

        for i in range(n):
            def x_low(radii):
                return centers[i][0] - radii[i]

            def x_high(radii):
                return 1 - centers[i][0] - radii[i]

            def y_low(radii):
                return centers[i][1] - radii[i]

            def y_high(radii):
                return 1 - centers[i][1] - radii[i]

            constraints.append({'type': 'ineq', 'fun': x_low})
            constraints.append({'type': 'ineq', 'fun': x_high})
            constraints.append({'type': 'ineq', 'fun': y_low})
            constraints.append({'type': 'ineq', 'fun': y_high})

        return constraints
