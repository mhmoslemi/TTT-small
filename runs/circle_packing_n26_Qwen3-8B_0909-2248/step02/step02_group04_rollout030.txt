import scipy.optimize as optimize

def run_packing():

        # Generate a hexagonal grid with adaptive spacing
        rows = int(np.ceil((n_circles ** 0.5)))
        cols = int(np.ceil(n_circles / rows))
        spacing = 0.2
        x = np.linspace(spacing, 1 - spacing, cols)
        y = np.linspace(spacing, 1 - spacing, rows)
        centers = []
        for row in range(rows):
            for col in range(cols):
                if row % 2 == 0:
                    x_pos = x[col]
                else:
                    x_pos = x[col] + spacing / 2
                y_pos = y[row]
                centers.append([x_pos, y_pos])
        return np.array(centers)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):

                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(radii):
        constraints = []
        for i in range(len(radii)):

            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Generate initial layout using hexagonal grid
    centers = generate_hexagonal_layout(26)

    result = optimize.minimize(

        method='COBYLA',
