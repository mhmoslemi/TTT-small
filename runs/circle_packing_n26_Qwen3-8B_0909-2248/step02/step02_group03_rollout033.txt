import scipy.optimize as optimize

def run_packing():

    def generate_hexagonal_packing(n):
        cols = int(np.ceil(n / 2))
        rows = int(np.ceil(n / cols))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = []
        for i in range(rows):
            for j in range(cols):
                if i % 2 == 0:
                    x_pos = x[j]
                else:
                    x_pos = x[j] + (x[1] - x[0]) / 2
                    if x_pos > 0.9:
                        x_pos = 0.9 - (x[1] - x[0]) / 2
                centers.append([x_pos, y[i]])
        return np.array(centers)

    # Generate the initial layout
    centers = generate_hexagonal_packing(n_circles)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint for circle-circle distances to be at least radii[i] + radii[j]
    def constraint_circle_circle(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    # Constraint for each circle to fit within [0,1]×[0,1]

            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i] - 1)
            constraints.append(y - radii[i])
            constraints.append(y + radii[i] - 1)
        return np.array(constraints)

    # Initial guess
    initial_radii = np.full(n_circles, 0.05)

    # Bounds for radii: 0 to 1
    bounds = [(0, 1) for _ in range(n_circles)]

    cons = []
    cons.append({'type': 'ineq', 'fun': constraint_circle_circle})

    # Run optimization with advanced settings
    result = optimize.minimize(
