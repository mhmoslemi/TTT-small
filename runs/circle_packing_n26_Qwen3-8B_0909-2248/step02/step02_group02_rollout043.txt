import scipy.optimize as optimize

def run_packing():

        # Generate a hexagonal grid layout with uneven rows and columns
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))

        y = np.linspace(0.1, 0.9, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n)])
        return centers

    def objective(radii):
        # Objective is to maximize the sum of radii, so minimize negative sum
        return -np.sum(radii)

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def compute_boundary_constraints(centers, radii):
        constraints = []
        n = len(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(x + r)
            constraints.append(y - r)
            constraints.append(y + r)
        return constraints
