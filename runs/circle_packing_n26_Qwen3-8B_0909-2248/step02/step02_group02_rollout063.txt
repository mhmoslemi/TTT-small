import numpy as np
import random
import matplotlib.pyplot as plt
from deap import base, creator, tools, algorithms

def run_packing():
    # Define parameters
    n_circles = 26
    max_iterations = 1000
    population_size = 100
    offspring_size = 100
    crossover_prob = 0.9
    mutation_prob = 0.2
    mutation_strength = 0.1
    tournament_size = 3

    # Genetic algorithm setup
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    # Initialize toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_pos", random.uniform, 0, 1)
    toolbox.register("attr_radius", random.uniform, 0, 0.2)
    toolbox.register("individual", tools.initIterate, creator.Individual, 
                    [toolbox.attr_pos(), toolbox.attr_pos(), toolbox.attr_radius() for _ in range(n_circles)])
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    # Fitness function
    def eval_packing(individual):
        centers = np.array([individual[i*3+0:i*3+2] for i in range(n_circles)])
        radii = np.array([individual[i*3+2] for i in range(n_circles)])
        
        # Check for boundary violations
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return 0.0,  # Penalize boundary violations

        # Check for overlapping circles
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-8:
                    return 0.0,  # Penalize overlaps

        # Return total radius sum as fitness
        return np.sum(radii), 

    toolbox.register("evaluate", eval_packing)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=mutation_strength, 
                    indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=tournament_size)

    # Create initial population
    pop = toolbox.population(n=population_size)

    # Run genetic algorithm
    for gen in range(max_iterations):
        offspring = list(map(toolbox.clone, pop))
        offspring = algorithms.varAnd(offspring, toolbox, 
                                    cxpb=crossover_prob, mutpb=mutation_prob)
        fits = list(toolbox.evaluate(ind) for ind in offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(offspring + pop, population_size)

    # Extract best individual
    best = max(pop, key=lambda ind: ind.fitness.values[0])
    centers = np.array([best[i*3+0:i*3+2] for i in range(n_circles)])
    radii = np.array([best[i*3+2] for i in range(n_circles)])
    total_sum = np.sum(radii)

    return centers, radii, total_sum
