import numpy as np
import scipy.optimize as optimize

def run_packing():
    n_circles = 26

    # Define staggered layout with staggered rows
    def generate_staggered_layout(n):
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n)])
        return centers

    centers = generate_staggered_layout(n_circles)

    def objective(radii):
        return -np.sum(radii)

    # Function to compute pairwise circle-circle distances
    def constraint_func(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    # Function to compute boundary constraints
    def boundary_func(radii):
        constraints = []
        for i in range(n_circles):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Initial guess for radii (uniform)
    initial_radii = np.full(n_circles, 0.05)

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(n_circles)]

    # Define constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': constraint_func})
    cons.append({'type': 'ineq', 'fun': boundary_func})

    # Fine-tune the layout for better corner and edge utilization
    # Manually adjust positions of a few circles to place them in corners
    # This will help maximize radii
    corner_indices = [0, 1, 2, 3, 4, 5]

        centers[idx] = [0.1, 0.1] if idx == 0 else [0.9, 0.1] if idx == 1 else [0.9, 0.9] if idx == 2 else [0.1, 0.9] if idx == 3 else [0.1, 0.5] if idx == 4 else [0.9, 0.5]

    # Use SLSQP optimization with tighter tolerances and higher max iterations
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    radii = result.x
    total_sum = np.sum(radii)
