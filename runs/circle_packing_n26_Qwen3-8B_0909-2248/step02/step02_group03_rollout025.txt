import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Create a hexagonal-like base layout
    def generate_hexagonal_layout(n):
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = []
        for i in range(n):
            row = i // cols
            col = i % cols
            # Hexagonal coordinate system
            x_center = x[col] + (0.1 if row % 2 else 0.0)
            y_center = y[row]
            centers.append((x_center, y_center))
        return np.array(centers)

    centers = generate_hexagonal_layout(n_circles)

    def objective(radii):
        return -np.sum(radii)

        for i in range(n_circles):

                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def boundary_func(radii):

        for i in range(n_circles):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Initial guess for radii: better estimate based on minimal spacing

    result = optimize.minimize(
