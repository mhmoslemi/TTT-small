import scipy.optimize as optimize

def run_packing():

        # Start with a basic hexagonal layout
        rows = int(np.ceil(n_circles / 2))
        cols = int(np.ceil(n_circles / rows)) if rows != 0 else n_circles
        centers = []
        for i in range(rows):
            for j in range(cols):
                r = 0.8
                if i % 2 == 0:
                    x = j * r
                    y = i * r
                else:
                    x = (j + 0.5) * r
                    y = i * r
                if x < 1 - 1e-6 and x > 1e-6 and y < 1 - 1e-6 and y > 1e-6:
                    centers.append((x, y))
        return np.array(centers)

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(centers, radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(centers, radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(x + r)
            constraints.append(y - r)
            constraints.append(y + r)
        return constraints

    def repulsion_func(centers, radii):

        n = len(radii)
        for i in range(n):
