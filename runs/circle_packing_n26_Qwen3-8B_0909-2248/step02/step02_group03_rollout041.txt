import numpy as np
import scipy.optimize as optimize

def run_packing():
    n_circles = 26

    def generate_hexagonal_packing(n):
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = np.array([(x[i % cols], y[i // cols]) for i in range(n)])
        return centers

    centers = generate_hexagonal_packing(n_circles)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - radii[i] - radii[j])
        return np.array(constraints)

    def boundary_func(radii):
        constraints = []
        for i in range(n_circles):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    result = optimize.minimize(

    )
