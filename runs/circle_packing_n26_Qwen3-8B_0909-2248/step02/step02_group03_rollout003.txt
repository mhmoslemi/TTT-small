import numpy as np
import scipy.optimize as optimize

def run_packing():

    def generate_hexagonal_packing(n):
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        x = np.linspace(0.1, 0.9, cols)
        y = np.linspace(0.1, 0.9, rows)
        centers = []
        for i in range(rows):
            for j in range(cols):
                # Hexagonal lattice: offset even rows
                if i % 2 == 0:
                    x_pos = x[j]
                else:
                    x_pos = x[j] + 0.4  # Shift for offset rows
                y_pos = y[i]
                if x_pos < 0.1 or x_pos > 0.9 or y_pos < 0.1 or y_pos > 0.9:
                    continue
                centers.append((x_pos, y_pos))
        return np.array(centers)

    centers = generate_hexagonal_packing(n_circles)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def boundary_func(radii):
        constraints = []
        for i in range(n_circles):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Initial guess for radii: uniform with small offset based on distance
    initial_distances = np.linalg.norm(centers[0] - centers[1]) * np.ones(n_circles)

    cons = []

    cons.append({'type': 'ineq', 'fun': boundary_func})

    # Use SLSQP with tighter tolerances and maximum iterations
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
