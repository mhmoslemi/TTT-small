import scipy.optimize as optimize
import random
from itertools import combinations

def generate_hexagonal_layout(n):
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = 0.5 + (j - (cols - 1)/2) * (0.5 / (cols - 1))
            y = 0.5 + i * (0.5 / (rows - 1))
            if i % 2 == 1:
                x += 0.25 / (cols - 1)
            centers.append((x, y))
    centers = np.array(centers)[:n]
    return centers

def perturb_centers(centers, eps=0.1):
    """Perturb centers slightly to avoid symmetry and encourage better packing."""
    perturb = np.random.normal(0, eps, centers.shape)
    return np.clip(centers + perturb, 0, 1)

def run_packing():
