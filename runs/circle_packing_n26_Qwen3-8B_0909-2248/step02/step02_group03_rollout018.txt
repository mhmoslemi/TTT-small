import numpy as np
import scipy.optimize as optimize

def run_packing():

            (0.1, 0.1), (0.9, 0.1), (0.9, 0.9), (0.1, 0.9),
            (0.1, 0.5), (0.5, 0.1), (0.5, 0.9), (0.9, 0.5),
            (0.5, 0.5),
        ]
        if n <= len(border_circles):
            return np.array(border_circles[:n])
        else:
            # Fill the remaining with evenly spaced circles
            cols = int(np.ceil(np.sqrt(n - len(border_circles))))
            rows = int(np.ceil((n - len(border_circles)) / cols))
            x = np.linspace(0.2, 0.8, cols)
            y = np.linspace(0.2, 0.8, rows)
            remaining_centers = np.array([(x[i % cols], y[i // rows]) for i in range(n - len(border_circles))])
            return np.vstack((np.array(border_circles), remaining_centers))

    centers = generate_packing(n_circles)

    def objective(radii):
        return -np.sum(radii)

    # Constraints: circle-circle and circle-boundary
    def constraint_func(radii):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Initial guess for radii (uniform)
    initial_radii = np.full(n_circles, 0.05)

    bounds = [(0, 1) for _ in range(n_circles)]
