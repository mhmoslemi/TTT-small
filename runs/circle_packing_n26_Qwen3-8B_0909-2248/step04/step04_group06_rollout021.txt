import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import random

def run_packing():

        """Generate a hexagonal packing of n_circles in a more evenly distributed manner."""
        if n_circles == 0:
            return np.array([])

        # Create a grid of points in hexagonal lattice for better spread
        spacing = 0.25
        grid = []
        for row in range(int(np.sqrt(n_circles)) + 1):
            for col in range(int(np.sqrt(n_circles)) + 1):
                x = col * spacing
                y = row * spacing * np.sqrt(3) / 2
                if np.sqrt(x**2 + y**2) <= 1:
                    grid.append((x, y))
        # Add extra points if needed to reach n_circles

            x = random.uniform(0, 1)
            y = random.uniform(0, 1)
            grid.append((x, y))
        return np.array(grid)

    def objective(radii):
        """Objective: maximize sum of radii."""
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):
        """Constraint: distance between centers must be at least sum of radii."""
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])

    def constraint_boundary(i, center, radius):
        """Constraint: circle within the unit square."""
        x, y = center
        return np.array([
            x - radius,
            x + radius,
            y - radius,
            y + radius
        ])

    def optimize_layout(centers):

        initial_radii = np.full(n, 0.05)

        result = minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
