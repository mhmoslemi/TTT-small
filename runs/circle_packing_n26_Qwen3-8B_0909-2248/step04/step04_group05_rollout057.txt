import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize

def run_packing():

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def constraint_boundary(radii):

        for i in range(n):

            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Bounds for radii: non-negative and not exceeding 1.0
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Constrains for overlap and boundary
    cons = [{'type': 'ineq', 'fun': constraint_overlap},

    initial_radii = np.full(n, 0.05)

    # Optimization setup
    result = minimize(

        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
