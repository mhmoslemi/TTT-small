import numpy as np
import scipy.optimize as optimize

def run_packing():
    def lens_hexagonal(n_lenses):
        """Generate a lens-like hexagonal grid with n_lenses."""
        if n_lenses == 0:
            return np.array([])

        centers = []
        for i in range(n_lenses):
            row = i // 4
            col = i % 4
            x = 0.3 + col * 0.25
            y = 0.3 + row * 0.25 * np.sqrt(3) / 2
            centers.append((x, y))
        return np.array(centers)

    def objective(radii):
        """Objective: maximize the sum of radii."""
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):
        """Ensure distance between centers is at least the sum of radii."""
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])

    def constraint_boundary(i, center, radius):
        """Ensure circles are within the unit square."""
        x, y = center
        return np.array([
            x - radius,
            x + radius,
            y - radius,
            y + radius
        ])

    def optimize_layout(centers):
        n = centers.shape[0]
        # Initial radii guess: 0.1
        initial_radii = np.full(n, 0.1)

        bounds = [(0, 1) for _ in range(n)]  # Radii must be between 0 and 1

        # Constraints: for each pair of circles, distance >= sum of radii
        def constraint_func(radii):
            constraints = []
            for i in range(n):
                for j in range(i + 1, n):
                    constraints.append(constraint_overlap(i, j, centers, radii))
            return np.array(constraints)

        # Boundary constraints per circle
        def boundary_constraints(radii):
            constraints = []
            for i in range(n):
                x, y = centers[i]
                constraints.extend([
                    x - radii[i],
                    x + radii[i],
                    y - radii[i],
                    y + radii[i]
                ])
            return np.array(constraints)

        # Use SLSQP with bounds and constraints
        cons = []
        cons.append({'type': 'ineq', 'fun': constraint_func})
        cons.append({'type': 'ineq', 'fun': boundary_constraints})

        result = optimize.minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        return result.x

    # Optimize
    radii = optimize_layout(centers)
    total_sum = np.sum(radii)
