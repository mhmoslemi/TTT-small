import scipy.optimize as optimize

        # Then add corner and edge circles to maximize boundary use

        centers = []
        for i in range(n - 4):
            x = 0.1 + (i % col) * (1 - 0.2) / (col - 1)
            y = 0.1 + (i // col) * (1 - 0.2) / (row - 1)

            centers.append([x, y])

        centers.append([0.1, 0.9])

        return centers
