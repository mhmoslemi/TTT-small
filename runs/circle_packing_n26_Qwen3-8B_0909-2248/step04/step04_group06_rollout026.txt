import scipy.optimize as optimize

def refine_hexagonal(n_circles):
    """Generate a hexagonal packing of n_circles."""
    if n_circles == 0:
        return np.array([])

    spacing = 0.2

        row = i // 5  # Distribute across 5 rows
        col = i % 5
        x = 0.1 + col * spacing
        y = 0.1 + row * spacing * np.sqrt(3) / 2
        centers.append((x, y))
    centers.sort(key=lambda p: (p[1], p[0]))
    return np.array(centers)
