import numpy as np
import scipy.optimize as optimize

def run_packing():
    def custom_initial_layout(n):
        grid_size = int(np.ceil(np.sqrt(n)))
        centers = []
        for i in range(n):
            row = i // grid_size
            col = i % grid_size
            x = 0.1 + col * (1 - 0.2) / (grid_size - 1)
            y = 0.1 + row * (1 - 0.2) / (grid_size - 1)
            if col % 2 == 1:
                y += 0.05
            centers.append([x, y])
        return np.array(centers)

    def objective_func(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_func(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def boundary_constraints(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    n = 26
    initial_centers = custom_initial_layout(n)
    initial_radii = np.full(n, 0.05)

    bounds = [(0, 1) for _ in range(n)]

    cons = [
        {'type': 'ineq', 'fun': constraint_func},
        {'type': 'ineq', 'fun': boundary_constraints}
    ]

    result = optimize.minimize(
        objective_func,
        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='Nelder-Mead',
        options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 2000}
    )

    centers = initial_centers
    radii = result.x
    total_sum = np.sum(radii)

    return centers, radii, total_sum
