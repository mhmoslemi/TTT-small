import numpy as np
import scipy.optimize as optimize

def run_packing():
    def uniform_grid(n_circles):
        """Generate a uniform grid of n_circles"""
        if n_circles == 0:
            return np.array([])
        centers = []
        for i in range(n_circles):
            row = i // 5
            col = i % 5
            x = col * 0.25
            y = row * 0.25
            centers.append((x, y))
        return np.array(centers)

    def objective(radii):
        return -np.sum(radii)

    def distance_constraint(i, j, centers, radii):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])

    def boundary_constraints(i, center, radius):
        x, y = center
        return np.array([
            x - radius,
            x + radius,
            y - radius,
            y + radius
        ])

    def optimize_layout(centers):
        n = centers.shape[0]
        initial_radii = np.full(n, 0.05)

        bounds = [(0, 1) for _ in range(n)]

        def constraint_func(radii):
            constraints = []
            for i in range(n):
                for j in range(i + 1, n):
                    constraints.append(distance_constraint(i, j, centers, radii))
            return np.array(constraints)

        def boundary_func(radii):
            constraints = []
            for i in range(n):
                x, y = centers[i]
                constraints.extend([
                    x - radii[i],
                    x + radii[i],
                    y - radii[i],
                    y + radii[i]
                ])
            return np.array(constraints)

        cons = []
        cons.append({'type': 'ineq', 'fun': constraint_func})
        cons.append({'type': 'ineq', 'fun': boundary_func})

        result = optimize.minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 10000}
        )
        return result.x

    def add_corner_circles(centers, n_corners):
        corners = np.array([
            (0, 0),
            (1, 0),
            (1, 1),
            (0, 1)
        ])
        added_centers = []
        for i in range(n_corners):
            x, y = corners[i]
            added_centers.append((x, y))
        return np.vstack((centers, added_centers))

    centers = uniform_grid(24)
    centers = add_corner_circles(centers, 4)
    radii = optimize_layout(centers)
    total_sum = np.sum(radii)

    return centers, radii, total_sum
