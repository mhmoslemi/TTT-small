import scipy.optimize as optimize

def run_packing():

        row = int(np.ceil(np.sqrt(n)))

        spacing = 1.0 / (col - 1) if col > 1 else 1.0

            x = 0.1 + (i % col) * spacing
            y = 0.1 + (i // col) * spacing
