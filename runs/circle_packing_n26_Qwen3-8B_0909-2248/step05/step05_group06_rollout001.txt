import numpy as np
import scipy.optimize as optimize

def run_packing():
    def generate_grid(n):
        """Generate an optimized grid with better density and corner coverage."""
        # Create a grid with horizontal and vertical spacing
        points = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col * 0.2 + np.random.uniform(-0.05, 0.05)
            y = row * 0.2 + np.random.uniform(-0.05, 0.05)
            # Ensure circles stay within the unit square
            x = np.clip(x, 0.1, 0.9)
            y = np.clip(y, 0.1, 0.9)
            points.append([x, y])
        return np.array(points)

    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)

    def constraint_func(radii, centers):
        """Ensure circles do not overlap."""
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def boundary_constraints(radii, centers):
        """Ensure circles fit within the unit square."""
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return np.array(constraints)

    # Generate initial grid of centers
    centers = generate_grid(26)

    # Set initial radii
    initial_radii = np.full(26, 0.05)

    # Define bounds for radii (0 to 1)
    bounds = [(0, 1) for _ in range(26)]

    # Set up the optimization problem
    cons = []
    cons.append({'type': 'ineq', 'fun': constraint_func, 'args': (centers, )})
    cons.append({'type': 'ineq', 'fun': boundary_constraints, 'args': (centers, )})

    # Run the optimizer with BFGS for better convergence
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=cons,
        method='BFGS',
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if not result.success:
        raise ValueError("Optimization failed with message: %s" % result.message)

    radii = result.x

    return centers, radii, total_sum
