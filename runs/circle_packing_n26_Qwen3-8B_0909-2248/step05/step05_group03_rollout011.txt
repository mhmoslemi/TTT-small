def run_packing():

        points.append([0.9, 0.1])

        points.append([0.9, 0.9])

            points.append([x, y])

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False

        return True
