import numpy as np
import scipy.optimize as optimize

def run_packing():
    def generate_initial_layout(n):
        """Generate a hybrid layout combining boundary points, corner points, and central points."""
        # Boundary points including corners
        boundary_points = []
        for x in [0, 0.1, 0.3, 0.5, 0.7, 0.9, 1]:
            for y in [0, 0.1, 0.3, 0.5, 0.7, 0.9, 1]:
                boundary_points.append([x, y])
        # Remove duplicates
        boundary_points = np.unique(boundary_points, axis=0)
        # Add central points in case of overpopulation
        if n > len(boundary_points):
            for i in range(n - len(boundary_points)):
                x = np.random.uniform(0.2, 0.8)
                y = np.random.uniform(0.2, 0.8)
                boundary_points = np.vstack((boundary_points, [x, y]))
        return boundary_points[:n]

    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)

    def constraints(radii, centers):
        """Compute constraints for the optimization problem."""
        n = len(radii)
        constraints = []

        # Boundary constraints: circles must be inside the unit square
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])

        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Generate initial layout for 26 circles
    centers = generate_initial_layout(26)
    # Initial radii guess based on distance to edges
    initial_radii = np.zeros(26)
    for i in range(26):
        x, y = centers[i]
        # Assign an initial radius based on distance to edges
        min_dist = np.min([x, 1 - x, y, 1 - y])
        initial_radii[i] = min_dist * 0.7  # Adjusted to 70% of the minimum distance as initial radius

    # Optimization bounds
    bounds = [(0, 1) for _ in range(26)]

    # Optimization settings
    options = {
        'ftol': 1e-10,
        'maxiter': 1500,
        'eps': 1e-8
    }

    # Run optimization
    result = optimize.minimize(

        bounds=bounds,
