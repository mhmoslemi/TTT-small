import scipy.optimize as optimize

def run_packing():

        """Generates a more compact hexagonal grid layout for circle centers."""
        centers = []

        # Corner and boundary circles
        corners = [(0.1, 0.1), (0.1, 0.9), (0.9, 0.9), (0.9, 0.1)]
        for x, y in corners:
            centers.append([x, y])
        
        # Midpoints of edges
        edges = [
            (0.5, 0.1), (0.9, 0.5), (0.5, 0.9), (0.1, 0.5),
            (0.5, 0.1), (0.1, 0.5), (0.5, 0.9), (0.9, 0.5)
        ]
        for x, y in edges:
            centers.append([x, y])

        # Interior circles
        for i in range(1, 8):
            for j in range(1, 8):
                x = 0.25 + (i - 1) * 0.25
                y = 0.25 + (j - 1) * 0.25
                if 0.1 <= x <= 0.9 and 0.1 <= y <= 0.9:
                    centers.append([x, y])

        # Trim to n circles
        return np.array(centers[:n])

    def calc_dist(c1, c2):
        """Calculate Euclidean distance between two centers."""
        return np.linalg.norm(np.array(c1) - np.array(c2))

    def objective(radii):
        """Objective function to maximize the sum of radii."""
        return -np.sum(radii)  # Minimize negative sum to maximize total

    def constraint_overlap(radii, centers):
        """Ensure all circles do not overlap."""
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = calc_dist(centers[i], centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    def constraint_boundary(radii, centers):
        """Ensure all circles lie within the unit square."""
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    # Initial hexagonal grid with 26 points
    centers = grid_hexagonal(26)

    # Initial radii (uniform, small starting value)
    initial_radii = np.full(26, 0.05)

    # Define bounds for radii (0 to 1)
    bounds = [(0, 1) for _ in range(26)]

    # Run optimization
    result = optimize.minimize(
