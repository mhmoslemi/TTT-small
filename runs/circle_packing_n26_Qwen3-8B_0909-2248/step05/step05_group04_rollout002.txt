import scipy.optimize as optimize
from scipy.optimize import minimize

        boundary_points = []

        boundary_points.append([0.0, 0.0])
        boundary_points.append([1.0, 0.0])
        boundary_points.append([1.0, 1.0])
        boundary_points.append([0.0, 1.0])
        # Edges
        boundary_points.append([0.0, 0.5])
        boundary_points.append([0.5, 0.0])
        boundary_points.append([1.0, 0.5])
        boundary_points.append([0.5, 1.0])

        for x in [0.2, 0.4, 0.6, 0.8]:
            boundary_points.append([x, 0.0])
            boundary_points.append([x, 1.0])
        for y in [0.2, 0.4, 0.6, 0.8]:
            boundary_points.append([0.0, y])
            boundary_points.append([1.0, y])
        boundary_points = np.array(boundary_points)
        boundary_points = np.unique(boundary_points, axis=0)
        return boundary_points

    def generate_hexagonal_interior_points(n_interior):
        """Generate interior points using a hexagonal packing."""
        points = []
        for i in range(n_interior):
            row = i // 5
            col = i % 5
            x = 0.2 + 0.2 * col
            y = 0.2 + 0.2 * row
            points.append([x, y])
        return np.array(points)

    # Generate initial layout: boundary and interior points
    n_boundary = 12  # Corners and edges
    n_interior = 14  # Remaining points
    if n_boundary + n_interior != 26:
        raise ValueError("Total number of points must be 26.")

    boundary_points = generate_boundary_points()
    interior_points = generate_hexagonal_interior_points(n_interior)
    centers = np.vstack((boundary_points, interior_points))

    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)

    def non_negative_constraint(radii):
        """Ensure all radii are non-negative."""
        return np.all(radii >= 0)

    def boundary_constraints(radii):
        """Ensure all circles stay within the unit square."""
        constraints = []
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    def non_overlapping_constraints(radii):
        """Ensure non-overlapping between all pairs of circles."""
        constraints = []
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    bounds = [(0, 1) for _ in range(26)]

    constraints = []
    constraints.append({'type': 'ineq', 'fun': non_negative_constraint})

    # Use SLSQP for better gradient handling
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}

    if not result.success:
        raise ValueError("Optimization failed with message: %s" % result.message)

    radii = result.x
    total_sum = np.sum(radii)

    return centers, radii, total_sum
