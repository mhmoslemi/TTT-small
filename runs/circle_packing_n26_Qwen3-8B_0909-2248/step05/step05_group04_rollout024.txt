import scipy.optimize as optimize

        rows = int((n)**0.5)
        cols = n // rows + 1
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * (i % 2)
                y = i
                layout.append([x, y])
        # Add boundary points near corners
        boundary_points = []
        for x in [0.1, 0.3, 0.5, 0.7, 0.9]:

        # Trim to n points

        """Maximize the sum of the radii."""
        return -np.sum(radii)

        n = len(radii)
        constraints = []

            constraints.append(x - radii[i])
