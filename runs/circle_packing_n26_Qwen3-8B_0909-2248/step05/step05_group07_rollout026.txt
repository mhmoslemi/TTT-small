import scipy.optimize as optimize

        points = []
        spacing = 0.15
        for i in range(n):
            row = i // 3
            col = i % 3
            x = 0.5 + (col - 1.5) * spacing
            y = 0.5 + (row - 1.5) * spacing * np.sqrt(3)/2
            points.append([x, y])
        return np.array(points)

    def objective(radii):
        """Maximize the sum of the radii."""
        return -np.sum(radii)  # Minimize the negative sum to maximize the total

        """Ensure circles are within the unit square."""
        constraints = []
        for i in range(centers.shape[0]):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])
        return constraints

    def inequality_constraints(centers, radii):
        """Ensure circles do not overlap."""
        constraints = []
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints
