import numpy as np
import cvxpy as cp

def run_packing():
    # Step 1: Create an initial good layout of circle centers
    # We use a combination of corner placement and spaced columns/rows
    centers = []
    # Top and bottom rows
    for i in range(5):
        # Top row
        centers.append([0.1 + i * 0.2, 0.1])
        centers.append([0.1 + i * 0.2, 0.9])
        # Middle rows
        centers.append([0.1 + i * 0.2, 0.5])
    
    # Add more centers in the middle
    for x in np.linspace(0.1, 0.9, 7):
        for y in np.linspace(0.1, 0.9, 7):
            centers.append([x, y])
    
    # Now we use 26 points from this generated list
    centers = np.array(centers)[:26]

    # Define variables
    radii = cp.Variable(26)

    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Constraint 1: No circle extends outside [0,1]x[0,1]
    for i in range(26):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-8)
        constraints.append(x + radii[i] <= 1 + 1e-8)
        constraints.append(y - radii[i] >= -1e-8)
        constraints.append(y + radii[i] <= 1 + 1e-8)

    # Constraint 2: No circle overlaps with any other
    for i in range(26):
        for j in range(i+1, 26):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-8, max_iter=1000, verbose=False)

    # Return the results
    if not problem.status in ['optimal', 'optimal_inaccurate']:
        raise RuntimeError(f"Optimization failed with status: {problem.status}.")
    
    return centers, radii.value, np.sum(radii.value)
