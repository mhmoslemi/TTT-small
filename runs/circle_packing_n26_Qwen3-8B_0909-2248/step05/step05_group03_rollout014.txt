import cvxpy as cp

import warnings

    # We generate a hexagonal grid of circles within the square

    # Estimate the largest circle radius (assuming all circles have same radius for initial grid)
    # For 26 circles, a rough hexagonal arrangement that fits in square
    # This is not the final packing, but gives a starting guess

    initial_centers = []

        col = i % 5
        x = 0.1 + (col + 0.5) * (0.8 / 5)

        initial_centers.append([x, y])

    initial_centers = np.array(initial_centers)

    # Define radius variable
    radii = cp.Variable(n)

    # Define constraints
    constraints = []

    # Boundary constraints: each circle must be inside the square
    for i in range(n):
        x, y = initial_centers[i]
        constraints.append(x - radii[i] >= -1e-12)

        constraints.append(y - radii[i] >= -1e-12)

    for i in range(n):

            dist = cp.norm(initial_centers[i] - initial_centers[j])
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Optimization objective: maximize total sum of radii
    objective = cp.Maximize(cp.sum(radii))
