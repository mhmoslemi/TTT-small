import scipy.optimize as optimize

def run_packing():

        grid = []

        for x in np.linspace(0.2, 0.8, 5):
            grid.append([x, 0.1])
            grid.append([x, 0.9])

        # Points along left and right boundaries
        for y in np.linspace(0.2, 0.8, 5):
            grid.append([0.1, y])
            grid.append([0.9, y])

        # Inner dense hexagonal layer
        for x in np.linspace(0.3, 0.7, 5):
            for y in np.linspace(0.3, 0.7, 5):
                grid.append([x, y])

        # Trim and keep unique points
        grid = np.array(grid)
        grid = np.unique(grid, axis=0)
        return grid[:n]

    def objective(radii):
        """Objective to maximize the sum of radii."""
        return -np.sum(radii)

    def constraint(radii, centers):
        """Constraints for optimization: packing and boundary."""
        n = len(radii)
        constraints = []

        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i])
            constraints.append(x + radii[i])
            constraints.append(y - radii[i])
            constraints.append(y + radii[i])

        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    bounds = [(0, 1) for _ in range(26)]
