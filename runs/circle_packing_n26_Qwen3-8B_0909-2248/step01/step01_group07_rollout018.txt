import numpy as np
import scipy.optimize as optimize

def run_packing():

    # We'll use a 5-row setup: 5, 5, 5, 5, 6 circles in each row
    # Row height = sqrt(3)/2 * diameter => we'll use interactive optimization to refine

    # Initial grid setup:
    # 2 rows of 5 circles (offset)
    # 3 rows of 5 circles (no offset)
    # 1 row of 6 circles (no offset)
    # Total: 5 + 5 + 5 + 5 + 6 = 26 circles
    num_rows = 5
    num_cols = [5, 5, 5, 5, 6]

    def initial_positions(num_cols):
        positions = []

        for i in range(num_rows):
            col = num_cols[i]
            row_centers = np.linspace(0.5, 0.5, col)
            row_centers = np.sort(row_centers)
            for x in row_centers:
                positions.append([x, 0.5 * (i + 0.5)])
        return np.array(positions)

    # Initialize guess
    centers = initial_positions(num_cols)
    n = len(centers)

    # Initial radii guess: diameter = min distance between adjacent circles
    radii = np.ones(n) * 0.05

    def obj_func(radii):
        return -np.sum(radii)

    def ineq_constraints(radii):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (radii[i] + radii[j]) >= 1e-8)

    opts = {

        'maxiter': 1000
    }

    bounds = [(0, 0.5) for _ in range(n)]

    result = optimize.minimize(

        x0=radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options=opts
