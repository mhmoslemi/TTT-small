        y = np.linspace(0, 1, rows)

                x_offset = (j + (i % 2)) * r_hex

                centers.append([x_offset, y_offset])

        return centers

            r = radii[i]
            # Enforce boundary constraints
            new_x = x
            new_y = y
            if x - r < 0:
                new_x = r

                new_x = 1 - r

                new_y = r
