    for i in range(26):

        dist_to_left = x

        dist_to_bottom = y
        dist_to_top = 1 - y
        radii[i] = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
    return radii

def cost_function(radii, centers):

    return -np.sum(radii)  # minimize negative sum => maximize sum

def constraints(centers, radii, args):

        # Non-negativity constraint
