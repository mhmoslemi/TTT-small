import cvxpy as cp
import matplotlib.pyplot as plt

def run_packing():

    n = 26

    # We'll fit as many rows as possible in the unit square
    # Let's choose 4 rows (more rows would be too close to the top)

    # Initial grid setup
    centers = []
    radii = []

    # First row
    row = np.linspace(0.1, 0.9, 5)

        centers.append([row[i], 0.1])
        radii.append(0.1)

        radii.append(0.1)

    centers = np.array(centers)
    radii = np.array(radii)

    # Define optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize the sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)

    x_initial = centers
    r_initial = radii

    # For non-convex problems like circle packing, we might need a more complex optimization
    # For the sake of this approach, use CVXPY with the SCS solver

    prob = cp.Problem(obj, constraints)
    prob.solve(warm_start=(x_initial, r_initial), solver='SCS')

    # Return the optimal centers and radii
    return (x.value, r.value, np.sum(r.value))
