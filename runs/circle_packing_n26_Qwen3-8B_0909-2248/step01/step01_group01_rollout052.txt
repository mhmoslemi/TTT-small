import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial arrangement: 5 columns of 5 circles (total 25 circles) + 1 circle at the bottom-right
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Step 1: Initial placement of 25 circles in 5 columns
    for i in range(5):
        for j in range(5):
            # Center column at x = (i + 0.5)/5, and row at y = (j + 0.5)/5
            x = (i + 0.5) / 5.0
            y = (j + 0.5) / 5.0
            centers[i * 5 + j] = [x, y]

    # Step 2: Add one circle near the bottom-right corner
    centers[-1] = [0.9, 0.1]
    radii[:25] = 0.1
    radii[-1] = 0.1

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(radii):
        # Make sure all radii are non-negative
        if np.any(radii < 0):
            return -np.inf

        # Check for overlap between all pairs
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return np.sum(radii)

    # Bounds for radii: non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Optimization step
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_overlap}]
    )

    # Final check to ensure all constraints were met
    # If optimization failed, return initial (small) radii
    if not result.success:
        return np.array([[0.5, 0.5]] * n), np.zeros(n), 0.0

    final_radii = result.x
    final_sum = np.sum(final_radii)
    return centers, final_radii, final_sum
