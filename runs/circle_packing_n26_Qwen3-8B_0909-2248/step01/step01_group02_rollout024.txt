import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Initial radii for all circles: same as minimal spacing (approx 0.1)
    radius_guess = 0.1
    n = 26
    centers = []
    for i in range(n):
        row = i // 3
        col = i % 3
        x = col * (1 - 2 * radius_guess)
        y = row * (1 - 2 * radius_guess)
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radius guess (same for all circles)
    radii = np.full(n, radius_guess)

    # Define objective: maximize total radius sum
    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Define constraints
    def constraint_boundary(x):
        bounds = np.array([[0, 1]] * 2)
        return np.sum(np.abs(x - bounds), axis=1)

    def constraint_nonoverlap(x):
        # Convert x to (centers, radii) format
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]
        # Compute pairwise distances between centers
        dists = np.sqrt(((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2).sum(axis=2))
        # Check pairwise non-overlapping condition: distance >= sum of radii
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1  # Violation
        return 0

    # Flatten x for optimization: first n*2 values are centers, next n values are radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Optimization bounds: centers must be in [0, 1]^2, radii ≥ 0
    bounds = []
    for _ in range(n * 2):
        bounds.append((0, 1))
    for _ in range(n):
        bounds.append((0, 1))

    # Optimization constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_nonoverlap},
    ]

    # Perform the optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 100, 'disp': False}
    )
