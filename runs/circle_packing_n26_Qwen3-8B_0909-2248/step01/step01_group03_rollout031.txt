import numpy as np
import scipy.optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial grid-based packing
    grid = np.linspace(0.1, 0.9, 5)
    grid_x, grid_y = np.meshgrid(grid, grid)
    centers = np.column_stack((grid_x.ravel(), grid_y.ravel()))
    radii = np.full(25, 0.1)

    # Add one more circle to optimize
    centers = np.vstack((centers, [0.5, 0.5]))
    radii = np.append(radii, 0.1)

    def objective(radii):
        # Ensure non-negative radii
        radii = np.clip(radii, 0, 1)
        return -np.sum(radii)  # Minimize negative sum to maximize

    def constraint1(radii):
        # Ensure all circles fit in the unit square
        for i in range(26):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return -1
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return -1
        return 0

    def constraint2(radii):
        # Ensure no circles overlap
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Optimize with Nelder-Mead
    result = scipy.optimize.minimize(

        constraints=[

        ],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
