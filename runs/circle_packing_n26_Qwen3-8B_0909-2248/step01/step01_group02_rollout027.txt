import numpy as np
from scipy.optimize import minimize
import warnings

def run_packing():

    n_circles = 26

    # Start with a hexagonal grid layout (approximate)
    # Simulate a hexagonal grid, and scale to fit in unit square
    # The layout is initially defined as rows with varying numbers of circles
    # Inspired by the hexagonal packing
    # Start with 5 rows, with row lengths as [5, 6, 7, 6, 5]

    def hexagonal_grid(n_rows, shift, radius):
        """Generate a hexagonal grid of circles based on rows and row shifts"""
        centers = []
        for row in range(n_rows):
            x = np.arange(0, n_rows, 1) * radius * np.sqrt(3)
            x += radius * np.sqrt(3) * row * shift
            y = row * 2 * radius
            centers.append(np.column_stack((x, y)))
        centers = np.vstack(centers)
        return centers

    # First, make a base hexagonal grid inside the square
    # We'll later optimize the radii and positions to fit better

    # Suggested initial grid with 5 rows
    n_rows = 5
    shift = 0.5
    radius_base = 0.1

    base_centers = hexagonal_grid(n_rows, shift, radius_base)

    # Now, define a function to compute the sum of radii and constraints
    # We will optimize the radii and positions of the circles

    def objective(x):
        """Objective function: minimize negative sum of radii (i.e., maximize sum)"""
        return -np.sum(x)

    def constraints(x):
        """Compute the constraints for the optimization problem"""
        # x contains all the circle parameters: [x1, y1, r1, x2, y2, r2, ...]
        n = len(x) // 3
        centers = x.reshape((n, 3))[:, :2]
        radii = x.reshape((n, 3))[:, 2]

        # Check that all radii are non-negative
        if np.any(radii < 0 - 1e-12):
            return -np.inf

        # Check that all circles are within the unit square
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -np.inf

        # Check that no two circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf

        return 0

    # Initial guess: place all circles with same radius and in a grid layout

        y = base_centers[i, 1]
        initial_centers.append([x, y])

    initial_x = np.array([x for x in np.vstack(initial_centers).flatten()])
    initial_x = np.concatenate((initial_x, np.array(initial_radii)))

    # We will minimize -sum(radii) to maximize sum(radii)

    centers = res.x[:2 * n_circles].reshape((n_circles, 2))

    return (centers, radii, sum_radii)
