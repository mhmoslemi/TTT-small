import scipy.optimize as opt

def run_packing():

    num_rows = 6
    num_cols = 5
    total_circles = num_rows * num_cols

    # Initial grid positions (hexagonal packing)
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.ones(total_circles) * 0.05  # Initial guess

    # Optimization bounds for x and y
    bounds = [(0, 1) for _ in range(2 * total_circles)]

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    def constraints_func(x):

        x_vals = x[:2 * total_circles]

        radii = x_vals[2 * total_circles:]

        # Boundary constraints: x and y must be within [0, 1]
        constraints = []
        for i in range(total_circles):
            x = centers[i, 0]
            y = centers[i, 1]
            r = radii[i]
            constraints.append((x - r >= -1e-12) * 1)
            constraints.append((x + r <= 1 + 1e-12) * 1)
            constraints.append((y - r >= -1e-12) * 1)

                dist = np.sqrt(dx ** 2 + dy ** 2)
                constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    x0 = np.concatenate([centers.flatten(), radii])

    cons = {'type': 'ineq', 'fun': constraints_func}

    centers_opt = result.x[:2 * total_circles].reshape((total_circles, 2))
