def run_packing():
    # Initial guess: Hexagonal packing
    num_circles = 26
    circles_per_row = [7, 8, 6, 6, 5, 4]  # row counts
    total_rows = len(circles_per_row)
    row_height = 1.0 / total_rows
    row_positions = np.zeros((num_circles, 2))
    idx = 0
    for i in range(total_rows):
        row_y = i * row_height
        if i % 2 == 0:
            # Even rows: left-aligned
            row_x = np.linspace(0, 1, circles_per_row[i])
            row_x = row_x * (1.0 - 2 * row_x)  # Slight adjustment to avoid corners

            # Odd rows: right-aligned
            row_x = np.linspace(0.5, 1.5, circles_per_row[i]) - 0.5
        for j in range(circles_per_row[i]):
            row_x_j = row_x[j]
            row_y_i = row_y
            row_positions[idx, 0] = row_x_j
            row_positions[idx, 1] = row_y_i
            idx += 1

    initial_radius = 0.1
    radii = np.full(num_circles, initial_radius)

        # x[0:num_circles] are the radii
        return -np.sum(x[0:num_circles])  # Minimize negative sum

    # Bounds: radii >= 0
    bounds = Bounds(0, np.inf)
    constraints = []

    def center_bounds(x):
        centers = x[num_circles:]
        for i in range(num_circles):
            x_i = centers[i*2]
            y_i = centers[i*2 + 1]
            if not (0 - 1e-12 <= x_i <= 1 + 1e-12 and 0 - 1e-12 <= y_i <= 1 + 1e-12):
                return False
        return True

    def in_square(x):
        # x[:num_circles] are radii
        # x[num_circles:] are centers
        centers = x[num_circles:]
        for i in range(num_circles):
            x_i = centers[i*2]
            y_i = centers[i*2 + 1]
            r_i = x[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or

                return False

    def no_overlap(x):

        radii = x[:num_circles]

                dx = centers[i*2] - centers[j*2]
                dy = centers[i*2 + 1] - centers[j*2 + 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    constraints.append(LinearConstraint(lambda x: np.array([0.0]*num_circles), -np.inf, np.inf))

    result = minimize(

        bounds=bounds,
        constraints=constraints,

        raise RuntimeError(f"Optimization failed: {result.message}")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
