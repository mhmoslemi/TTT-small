import numpy as np
from scipy.optimize import minimize
from scipy.optimize.constraints import LinearConstraint
import warnings

def run_packing():

    grid_size = 5  # Number of rows and columns in the grid
    num_circles = 26
    if num_circles > grid_size ** 2:

    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            centers.append([x, y])
    centers = np.array(centers)

    # Initial guess for radii
    radii = np.full(len(centers), 0.05)

    def objective(radii):
        return -np.sum(radii)

        return np.array([radii[i] + radii[j] - dist_matrix[i] for i in range(len(centers)) for j in range(i+1, len(centers))])

    # Convert constraints into optimization format
    constraints = []
