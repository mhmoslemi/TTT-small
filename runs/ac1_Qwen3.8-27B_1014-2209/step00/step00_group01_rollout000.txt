import numpy as np
import time
from scipy import optimize

def propose_candidate(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 15  # leave 15s buffer
    
    # Start with provided sequence
    best_sequence = list(np.maximum(0, height_sequence_1))
    best_score = evaluate_sequence(best_sequence)
    
    curr = np.array(best_sequence, dtype=float)
    
    while time.time() < deadline:
        # Normalize
        s_sum = np.sum(curr)
        if s_sum < 1e-10:
            break
        s = curr / s_sum
        
        # Compute convolution
        c = np.convolve(s, s)
        
        # Find top constraints
        # We want to minimize max(c), so focus on the largest entries
        top_idx = np.argsort(c)[-min(len(c), 200):]
        top_val = c[top_idx]
        M = c[top_idx[0]]  # the max
        
        # LP: maximize sum(g) s.t. conv_matrix[top_idx] @ g <= top_val, g >= 0
        # But conv_matrix is sparse... let's build it for just the top constraints
        n = len(s)
        # For constraint k: sum_i s_i * g_{k-i} <= top_val[j]
        # where 0 <= k-i < n
        
        c_vec = -np.ones(n)
        A_ub_rows = []
        b_ub_vals = []
        
        for k in top_idx.astype(int):
            row = np.zeros(n)
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    row[i] = s[j]  # s_{k-i}
            A_ub_rows.append(row)
            b_ub_vals.append(M)  # use the max as the bound for simplicity
        
        # Add nonnegativity: -g <= 0
        A_ub = np.vstack([np.array(A_ub_rows), -np.eye(n)])
        b_ub = np.concatenate([np.array(b_ub_vals), np.zeros(n)])
        
        result = optimize.linprog(c_vec, A_ub=A_ub, b_ub=b_ub, method='highs',
                                   options={'time_limit': 5, 'disp': False})
        
        if result.success:
            g = result.x
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                g = g / g_sum  # normalize g to have sum 1
                # Update
                t = 0.1  # step size
                curr = (1 - t) * s + t * g
                curr = np.maximum(0, curr)
                # Scale back
                curr = curr * s_sum  # keep the same scale
        else:
            # Fallback: random perturbation
            idx = np.random.randint(len(curr))
            curr[idx] = max(0, curr[idx] + np.random.randn() * 0.1 * s_sum / n)
        
        score = evaluate_sequence(list(curr))
        if score < best_score:
            best_score = score
            best_sequence = list(curr)
