def convolve_autocorr(a):
    """Compute full linear autocorrelation conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.irfft(A * A, fft_size)
    return B[:size]

    n = len(a)
