```python
def solve_reduced_LP(f, M, n, max_iter=50):
    """Solve LP with constraint generation."""
    # Active constraints start with just the center
    active_k = [n - 1]
    g = np.ones(n) / n  # initial feasible point
    for _ in range(max_iter):
        # Solve LP: max sum(g) s.t. (f*g)_k <= M for k in active_k, g >= 0
        c = -np.ones(n)
        A_ub = []
        b_ub = []
        for k in active_k:
            row = np.zeros(n)
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                row[j] = f[k - j]
            A_ub.append(row)
            b_ub.append(M)
        # g >= 0
        A_ub.append(-np.eye(n))
        b_ub.append(np.zeros(n))
        
        A_ub = np.vstack(A_ub)
        b_ub = np.hstack(b_ub)
        
        res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'disp': False, 'time_limit': 5})
        if not res.success:
            return None
        g = res.x
        
        # Check all constraints
        conv_fg = np.convolve(f, g)
        violated = np.where(conv_fg > M * (1 + 1e-9))[0]
        if len(violated) == 0:
            return g
        # Add the most violated constraint
        worst = violated[np.argmax(conv_fg[violated])]
        if worst not in active_k:
            active_k.append(worst)
    
    return g
```

```python
import numpy as np
import time
from scipy import optimize

def compute_conv(f):
    return np.convolve(f, f)

def solve_lp_direction(f, M, n, max_constraints=200, time_limit_per_lp=2.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0, using constraint generation."""
    active_ks = [n - 1]  # Start with center constraint
    g = None
    for _ in range(max_constraints):
        # Build constraint matrix
        A_rows = []
        b_vals = []
        for k in active_ks:
            row = np.zeros(n)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                row[j] = f[k - j]
            A_rows.append(row)
            b_vals.append(M)
        
        A_ub = np.vstack(A_rows + [-np.eye(n)])
        b_ub = np.concatenate([b_vals, np.zeros(n)])
        c = -np.ones(n)
        
        res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                               options={'disp': False, 'time_limit': time_limit_per_lp})
        if not res.success:
            return g  # Return previous solution if available
        g = res.x
        
        # Check all constraints
        fg = np.convolve(f, g)
        violations = np.where(fg > M * (1 + 1e-6))[0]
        if len(violations) == 0:
            return g
        
        # Add most violated constraint
        worst_k = violations[np.argmax(fg[violations])]
        if worst_k not in active_ks:
            active_ks.append(worst_k)
        else:
            # Already in active set but violated - infeasible or numerical issue
            return g
```
