def compute_score(seq):
    """Compute the AC1 score without calling evaluate_sequence (for speed during search)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = b.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / sum_a**2)

def solve_direction_lp(f, M, time_limit=8):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Use sparse matrix."""
    n = len(f)
    c = -np.ones(n)
    # Build sparse constraint matrix for conv(f, b) <= M
    # row k: sum_i f[i] * b[k-i] for valid i
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    
    A_conv = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    A_neg = csr_matrix(-np.eye(n))
    A = csr_matrix.vstack([A_conv, A_neg])
    b_vec = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
    
    result = optimize.linprog(c, A_ub=A, b_ub=b_vec, method='highs',
                              options={'time_limit': time_limit, 'disp': False})
    if result.success:
        return result.x
    return None

from scipy.sparse import csr_matrix

def compute_score_fast(seq):
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def solve_direction(sp_f, M, n, time_limit=10):
    c = -np.ones(n)
    rows, cols, vals = [], [], []
    f = sp_f
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(f[i])
    A_conv = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

                           options={'time_limit': time_limit, 'disp': False})

    return None

    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def build_conv_matrix(f, n):
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
