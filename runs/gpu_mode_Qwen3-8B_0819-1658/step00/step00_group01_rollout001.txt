import triton
import triton.language as tl
import torch
import math
import numpy as np

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Determine output tensor
    output = torch.zeros(bs, seq_len, seq_len, dim, device=input_tensor.device)

    # Define kernel
    @triton.jit
    def trit_mul_kernel(
        x_ptr, mask_ptr, left_proj_w_ptr, right_proj_w_ptr, left_gate_w_ptr,
        right_gate_w_ptr, out_gate_w_ptr, to_out_norm_w_ptr, to_out_norm_b_ptr,
        to_out_w_ptr,
        x_shape, mask_shape, seq_len, bs, dim, hidden_dim,
        num_warps: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_C: tl.constexpr
    ):
        # Thread ID
        pid = tl.program_id(axis=0)
        # Thread block along sequence
        block_seq = pid
        # Current sequence index
        i = block_seq * BLOCK_N + tl.arange(0, BLOCK_N)
        # Index into the batch
        b = tl.arange(0, bs)
        # Initialize output
        out = tl.zeros((BLOCK_N, hidden_dim), dtype=tl.float32)
        # Iterate over all sequences j
        for j in range(seq_len):
            # Load input and mask
            x_load = tl.load(x_ptr + b[:, None] * seq_len * seq_len * dim + i[:, None] * seq_len * dim + j * dim, 
                             mask=tl.load(mask_ptr + b[:, None] * seq_len * seq_len + i[:, None] * seq_len + j, 
                                          dtype=tl.float32), 
                             dtype=tl.float32)
            # Apply LayerNorm
            x_norm = tl.dot(x_load, norm_weight.T) + norm_bias
            # Project left and right
            left_proj = tl.dot(x_norm, left_proj_weight.T)
            right_proj = tl.dot(x_norm, right_proj_weight.T)
            # Apply gates
            left_gate = tl.dot(x_norm, left_gate_weight.T).sigmoid()
            right_gate = tl.dot(x_norm, right_gate_weight.T).sigmoid()
            out_gate = tl.dot(x_norm, out_gate_weight.T).sigmoid()
            # Apply gates
            left = left_proj * left_gate
            right = right_proj * right_gate
            # Multiply and accumulate
            # Einsum for each output dimension
            for c in range(hidden_dim):
                tmp = tl.zeros((BLOCK_N, 1), dtype=tl.float32)
                for k in range(seq_len):
                    left_k = tl.load(left_ptr + b * seq_len * seq_len * hidden_dim + i * seq_len * hidden_dim + k * hidden_dim + c)
                    right_k = tl.load(right_ptr + b * seq_len * seq_len * hidden_dim + j * seq_len * hidden_dim + k * hidden_dim + c)
                    tmp += left_k * right_k
                out[:, c] = tmp
        # Apply LayerNorm
        out_norm = tl.dot(out, to_out_norm_weight.T) + to_out_norm_bias
        # Apply gate
        out_gate = tl.dot(x_norm, out_gate_weight.T).sigmoid()
        out = out_norm * out_gate
        # Output
        tl.store(output_ptr + b[:, None] * seq_len * seq_len * dim + i[:, None] * seq_len * dim, out)

    # Launch kernel with multiple blocks to cover the entire sequence matrix
    num_blocks = (seq_len + BLOCK_N - 1) // BLOCK_N
    for block in range(num_blocks):
        trit_mul_kernel(
            x_ptr=input_tensor,
            mask_ptr=mask,
            left_proj_w_ptr=left_proj_weight,
            right_proj_w_ptr=right_proj_weight,
            left_gate_w_ptr=left_gate_weight,
            right_gate_w_ptr=right_gate_weight,
            out_gate_w_ptr=out_gate_weight,
            to_out_norm_w_ptr=to_out_norm_weight,
            to_out_norm_b_ptr=to_out_norm_bias,
            to_out_w_ptr=to_out_weight,
            x_shape=(bs, seq_len, seq_len, dim),
            mask_shape=(bs, seq_len, seq_len),
            seq_len=seq_len,
            bs=bs,
            dim=dim,
            hidden_dim=hidden_dim,
            num_warps=4,
            BLOCK_N=128,
            BLOCK_C=128
        )
    
    return output
