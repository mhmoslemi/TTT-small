    The kernel optimizes the forward pass over large sequence dimensions using Triton's capabilities.
    This kernel handles both mask and no-mask scenarios with high performance for large-seqlen inputs.
    """

    dim, hidden_dim = config["dim"], config["hidden_dim"]

    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()

    right_proj_weight = weights["right_proj.weight"].contiguous()

    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # Output tensor of shape [bs, seq_len, seq_len, dim]
    output = torch.zeros((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=input_tensor.dtype)

    # Helper function to perform the TriMul operation on a single batch
    @triton.jit
    def tritondot_kernel(
        input_a_ptr, input_b_ptr, mask_ptr,
        output_ptr,
        seq_len, dim,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    ):
        # Compute the thread indices
        pid = tl.program_id(axis=0)
        num_pid_n = tl.cdiv(seq_len, BLOCK_N)
        pid_n = pid % num_pid_n
        pid_m = pid // num_pid_n

        # Compute offsets
        offsets_m = pid_m * BLOCK_N + tl.arange(0, BLOCK_N)
        offsets_k = tl.arange(0, BLOCK_K)
        offsets_d = tl.arange(0, BLOCK_D)

        # Load masks
        mask = tl.load(mask_ptr + (offsets_m[:, None] * seq_len + offsets_k[None, :]))
        mask = mask > 0.5  # Assuming mask is 0/1 or 0.0/1.0, but needs to be adapted according to actual mask format

        # Load input_a
        input_a = tl.load(input_a_ptr + (offsets_m[:, None] * seq_len + offsets_k[None, :]))
        input_a = input_a * tl.reshape(mask, (BLOCK_N, 1, 1))  # Apply mask

        # Load input_b
        input_b = tl.load(input_b_ptr + (offsets_k[:, None] * seq_len + offsets_m[None, :]))
        input_b = input_b * tl.reshape(mask, (1, 1, BLOCK_N))  # Apply mask

        # Compute tensor product
        out = tl.einsum('ijk,ikl->ijl', input_a, input_b)
        out = out * tl.reshape(mask, (1, 1, 1, BLOCK_N))  # Apply mask again

        # Store result
        tl.store(output_ptr + (offsets_m[:, None] * seq_len + offsets_d[None, :]), out)

    # Launch Triton kernel
    grid = (triton.cdiv(seq_len, 128), batch_size)
    tritondot_kernel[grid](
        input_tensor, input_tensor, mask,
        output,
        seq_len, dim,
        BLOCK_N=128, BLOCK_K=128, BLOCK_D=dim
    )

    return output
