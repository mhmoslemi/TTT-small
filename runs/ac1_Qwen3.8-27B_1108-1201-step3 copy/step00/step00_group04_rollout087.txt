"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Strategy: (1) Grid search over polynomial ramps i^p, (2) Gradient-based
local search exploiting the scale invariance, (3) Multi-start refinement.
Key insight: asymmetric sequences (ramps) beat symmetric ones since
b[n-1]=sum(a_i^2)>=S^2/n forces obj>=2 for symmetric a.
"""

import numpy as np
import time


def fast_objective(a):
    """Compute objective without full security checks (for internal use)."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    M = b.max()
    return 2.0 * n * M / (S * S)


def make_ramp(n, p, offset=0):
    """Polynomial ramp: (i+offset+1)^p for i=0..n-1."""
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    return i ** p


def make_exp_ramp(n, beta):
    """Exponential ramp: exp(beta * i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = i / (n - 1)
    else:
        t = np.zeros(1)
    return np.exp(beta * t)


def compute_peak_gradient(a, top_k=3):
    """Compute gradient contribution from top-k active lags."""
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    S = a.sum()

    # Find top-k lags
    thresh = M * (1.0 - 1e-9)
    active_all = np.where(b >= thresh)[0]
    if len(active_all) > top_k:
        vals = b[active_all]
        top_idx = np.argpartition(vals, -top_k)[-top_k:]
        active = active_all[top_idx]
    else:
        active = active_all

    # Compute gradient of M w.r.t. a
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        contrib = 2.0 * a[k - js]
        # Fix diagonal (where 2*j == k)
        if k % 2 == 0:
            jd = k // 2
            if j_start <= jd <= j_end:
                contrib[jd - j_start] = a[jd]
        grad[j_start:j_end + 1] += contrib

    # d(objective)/da ∝ S * grad_M - M (dropping constant 2n/S^3 > 0)
    grad_obj = S * grad - M
    return grad_obj, M, S, b


def gradient_descent(a, budget_s, initial_eta=None, verbose=True):
    """Gradient descent with adaptive step size."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, 1000.0)
    S = a.sum()
    if S < 0.01:
        return a, np.inf

    if initial_eta is None:
        initial_eta = 0.5 * n / S  # scale-appropriate step

    best_a = a.copy()
    best_score = fast_objective(best_a)
    current_score = best_score
    eta = initial_eta
    patience = 0
    max_patience = 500
    start = time.monotonic()
    iteration = 0

    while time.monotonic() - start < budget_s:
        grad_obj, M, S_cur, b = compute_peak_gradient(a, top_k=3)

        a_new = a - eta * grad_obj / (S_cur * S_cur)
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Ensure minimum sum
        if a_new.sum() < 0.01:
            a_new = a_new + 0.001 * np.ones(n)

        new_score = fast_objective(a_new)

        if np.isfinite(new_score) and new_score < current_score:
            a = a_new
            current_score = new_score
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and best_score < 1.5:
                    print(f"    GD iter {iteration}: score={best_score:.8f}, n={n}")
        else:
            a = a_new  # still move, but don't count as improvement
            patience += 1
            eta *= 0.95

        if patience > max_patience:
            # Restart from best with larger step
            a = best_a.copy()
            eta = initial_eta * (1 + 0.1 * np.random.randn())
            eta = max(eta, initial_eta * 0.1)
            current_score = best_score
            patience = 0

        iteration += 1

    return best_a, best_score


def block_perturb(a, rng):
    """Random block multiplicative perturbation."""
    n = len(a)
    a_new = a.copy()
    n_blocks = rng.integers(1, max(2, n // 500) + 1)
    for _ in range(n_blocks):
        width = max(2, int(rng.integers(n // 100, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)
        factor = np.exp(rng.normal(0, 0.3))
        a_new[start:end] *= factor
    return np.clip(a_new, 0.0, 1000.0)
