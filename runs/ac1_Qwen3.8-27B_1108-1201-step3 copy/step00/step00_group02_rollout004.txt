def solve_lp(f, M, n):
    """Solve the LP: max sum(b) s.t. f*b <= M (element-wise), b >= 0."""
    # Build constraint matrix A where A[k, j] = f[k-j] if valid
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0

    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    if res.success:
        return res.x
    else:
        return None
