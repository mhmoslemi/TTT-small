Gradient-based optimization with smooth max (log-sum-exp) approximation

Uses FFT for O(n log n) per-iteration cost. Combines parametric initialization,

import numpy as np
import time


def _fft_convolve(a, N_fft):
    """Compute full self-convolution via FFT."""
    F = np.fft.rfft(a, N_fft)
    return np.fft.irfft(F * F, N_fft)


def _smooth_max_grad(a, tau, N_fft):
    """
    Compute the smooth gradient of 2n * L_tau(conv(a,a)) / S^2.
    Returns (gradient, L_tau_value, max_b).
    """
    b = _fft_convolve(a, N_fft)
    b_max = np.max(b)
    shifted = (b - b_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    L = b_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    S = np.sum(a)
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    g_full = np.fft.irfft(np.conj(P) * A, N_fft)
    g = g_full[:len(a)]
    
    grad = (2.0 * len(a) / S**2) * g - (4.0 * len(a) * L / S**3)
    return grad, L, b_max

_NFFT = 0


def _compute_conv_and_grad(a, tau):
    """
    Compute self-convolution b, smooth max L_tau, and gradient.
    Returns (grad, L, b_max).
    """
    n = len(a)
    N = _NFFT
    
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)

"""
Annealed gradient descent with smooth max (log-sum-exp) approximation
to minimize 2n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses FFT-based O(n log n) gradient computation with additive updates
and projection onto non-negativity. Combines parametric initialization,
multi-scale annealing, and random restarts.
"""

import numpy as np
import time


def _get_fft_len(n):
    """Next power of 2 >= 3n-2 (needed for cross-correlation)."""
    needed = 3 * n - 2
    size = 1
    while size < needed:
        size <<= 1
    return size


def _obj_val(a, N):
    """Compute 2n*max(conv(a,a))/S^2 using FFT."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)
    return float(2.0 * n * np.max(b) / (S * S))


def _grad_smooth(a, tau, N):
    """
    Compute gradient of 2n*L_tau(conv(a,a))/S^2.
    Uses FFT for both convolution and cross-correlation.
    Returns (grad, L_tau, b_max).
    """
    n = len(a)
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)
    b = b[:2 * n - 1]
    
    b_max = np.max(b)
    shifted = (b - b_max) / tau
    exps = np.exp(shifted)
    s = np.sum(exps)
    L = b_max + tau * np.log(s)
    p = exps / s
    
    S = np.sum(a)
    
    # Cross-correlation c[j] = sum_k p[k]*a[k-j] via FFT
    P = np.fft.rfft(p, N)  # Note: p has length 2n-1, zero-padded to N
    c = np.fft.irfft(np.conj(P) * A, N)[:n]
    
    # Gradient: 2n/S^2 * c - 4nL/S^3
    grad = (2.0 * n / (S * S)) * c - (4.0 * n * L) / (S * S * S)
    return grad, L, b_max


def _gd_step(a, grad, step_size):
    """
    Additive gradient step with projection and renormalization.
    Returns new a.
    """
    n = len(a)
    alpha = step_size / (np.max(np.abs(grad)) + 1e-30)
    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0.0
    Sn = np.sum(a_new)
    if Sn < 1e-15:
        return a.copy()
    return a_new / Sn


def _parametric_inits(n, rng):
    """Generate diverse parametric initializations."""
    cands = []
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Uniform
    cands.append(np.ones(n))
    
    # Gaussians
    for sf in [0.05, 0.08, 0.1, 0.12, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5]:
        sigma = sf * n
        cands.append(np.exp(-0.5 * ((x - center) / sigma) ** 2))
    
    # Bimodal
    for sf in [0.05, 0.08, 0.1, 0.15, 0.2]:
        sigma = sf * n
        for f1 in [0.2, 0.25, 0.3]:
            c1, c2 = n * f1, n * (1 - f1)
            cands.append(0.5 * (np.exp(-0.5 * ((x - c1) / sigma) ** 2) +
                                np.exp(-0.5 * ((x - c2) / sigma) ** 2)))
    
    # Raised cosine
    cands.append(0.5 * (1 + np.cos(2 * np.pi * (x - center) / n)))
    cands.append(0.5 * (1 + np.cos(np.pi * (x - center) / (n / 2))))
    
    # Super-Gaussian
    for p in [3, 4, 6, 8]:
        for sf in [0.15, 0.2, 0.25, 0.3]:
            sigma = sf * n
            cands.append(np.exp(-np.abs(x - center) / sigma) ** p)
    
    # Triangular
    cands.append(np.maximum(0, 1 - np.abs(x - center) / (n / 2)))
    
    # Random
    for _ in range(5):
        cands.append(np.abs(rng.normal(0, 1, n)))
    
    # Smooth random (low-pass filtered)
    for _ in range(5):
        r = np.abs(rng.normal(0, 1, n))
        # Apply a Gaussian smoothing
        kernel_size = max(3, n // 50)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4.0))
        kernel /= kernel.sum()
        cands.append(np.convolve(r, kernel, mode='same'))
    
    return cands


def _run_gd(a, tau, step_size, n_iters, N, deadline, check_every=100):
    """
    Run gradient descent with given tau and step_size for n_iters iterations.
    Returns (best_a, best_score).
    """
    n = len(a)
    best = a.copy()
    best_score = _obj_val(a, N)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        grad, L, b_max = _grad_smooth(a, tau, N)
        
        if not np.all(np.isfinite(grad)):
            break
        
        a_new = _gd_step(a, grad, step_size)
        a = a_new
        
        if (it + 1) % check_every == 0:
            sc = _obj_val(a, N)
            if sc < best_score:
                best = a.copy()
                best_score = sc
    
    # Final check
    sc = _obj_val(a, N)
    if sc < best_score:
        best = a.copy()
        best_score = sc
    
    return best, best_score
