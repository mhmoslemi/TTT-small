LP-based local search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the SOTA direction-finding LP with adaptive step size and multiple restarts.
"""

import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix, csr_matrix


def _build_conv_matrix_indices(n):
    """Precompute column indices and indptr for the convolution matrix."""
    idx = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)[None, :]).flatten()
    indptr = (np.arange(n + 1, dtype=np.int64) * n)
    return idx, indptr


def _lp_direction(a, M, idx_template, indptr):
    """Solve the direction-finding LP. Returns g vector or None."""
    n = len(a)
    data = np.tile(a, n)
    A = csc_matrix((data, idx_template, indptr), shape=(2 * n - 1, n))
    b = M * np.ones(2 * n - 1)
    c = -np.ones(n)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',

def _build_conv_structure(n):

    idx = (np.arange(n, dtype=np.int64)[:, None] +

    indptr = (np.arange(n + 1, dtype=np.int64) * n)

def _score(a):

    n = len(a)
    s = a.sum()

        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max() / (s * s))


def _solve_direction_lp(a, M, idx_template, indptr):
    """Solve the direction-finding LP: max sum(g) s.t. conv(a,g)<=M, g>=0."""
    n = len(a)
    data = np.tile(a, n)
    A = csc_matrix((data, idx_template, indptr), shape=(2 * n - 1, n))
    b = M * np.ones(2 * n - 1)

                      method='highs', options={'presolve': True})

    except Exception:
        pass
    return None

        best_score = final_score

    print(f"\nFinal: score={best_score:.6f}, n={n}", flush=True)
    return [float(x) for x in best]
