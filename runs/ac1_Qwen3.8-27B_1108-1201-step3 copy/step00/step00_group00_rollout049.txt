```python
def conv_gradient(a, k):
    n = len(a)
    grad = np.zeros(n)
    # Indices j where 0 <= k-j < n, i.e., k-n+1 <= j <= k, and 0 <= j < n
    j_start = max(0, k - n + 1)
    j_end = min(n-1, k)
    if j_start <= j_end:
        indices = j_start + np.arange(j_end - j_start + 1)
        grad[indices] = 2 * a[k - indices]
        if k % 2 == 0 and j_start <= k//2 <= j_end:
            grad[k//2] = a[k//2]
    return grad
```

```python
import numpy as np
import time

def convolve(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def compute_objective(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    b = np.convolve(a, a, mode='full')
    return 2 * len(a) * np.max(b)

def subgradient(a, k):
    """Compute subgradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        j_indices = np.arange(j_start, j_end + 1)
        g[j_indices] = 2.0 * a[k - j_indices]
        if k % 2 == 0:
            jm = k // 2
            if j_start <= jm <= j_end:
                g[jm] = a[jm]
    return g

def project(a, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    # Binary search on the threshold
    a_shifted = a.copy()
    lo = -np.max(a_shifted) - target_sum
    hi = np.max(a_shifted)
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.maximum(0, a_shifted - mid)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(0, a_shifted - (lo + hi) / 2)
    s = np.sum(projected)
    if s > 0:
        projected *= target_sum / s
    return projected

def optimize_sequence(a, time_budget):
    """Optimize a sequence using subgradient descent."""
    a = a.astype(np.float64).copy()
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones_like(a)
        s = 1.0
    a /= s  # normalize to sum=1
    
    n = len(a)
    deadline = time.time() + time_budget
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    momentum = np.zeros_like(a)
    lr = 0.01
    momentum_coeff = 0.9
    
    while time.time() < deadline:
        b = np.convolve(a, a, mode='full')
        k_max = np.argmax(b)
        
        # Also consider top-k for smoother gradient
        top_k = np.argsort(b)[-5:]
        g = np.zeros_like(a)
        for k in top_k:
            g += subgradient(a, k) / len(top_k)
        
        # Heavy ball momentum
        momentum = momentum_coeff * momentum - lr * g
        a_new = a + momentum
        
        # Project
        a_new = project(a_new, 1.0)
        
        # Evaluate
        obj_new = compute_objective(a_new)
        
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        
        if obj_new < best_obj + 0.01:  # accept if not too bad
            a = a_new
        else:
            # Rejected step, reduce lr
            lr *= 0.9
            momentum = 0.9 * momentum  # dampen momentum
        
        if lr < 1e-10:
            break
            
    return best_a, best_obj
```

```python
import time

def _conv(a):
    return np.convolve(a, a, mode='full')

def _objective(a):
    """Objective for normalized (sum=1) non-negative sequence."""
    b = _conv(a)
    return 2.0 * len(a) * np.max(b)

def _subgrad_at(a, k):
    """Subgradient of (a*a)[k] wrt a."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k - idx]
        if k % 2 == 0:
            jm = k // 2
            if js <= jm <= je:
                g[jm] = a[jm]
    return g

def _project_nn_sum(a, s=1.0):
    """Project a onto {x >= 0, sum(x) = s}."""
    if np.max(a) <= 0:
        return np.full_like(a, s / len(a))
    lo = 0.0
    hi = np.max(a) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        v = np.maximum(0.0, a - mid)
        if np.sum(v) > s:
            lo = mid
        else:
            hi = mid
    v = np.maximum(0.0, a - 0.5 * (lo + hi))
    sv = np.sum(v)
    if sv > 0:
        v *= s / sv
    return v

def _subgrad_descent(a0, time_budget, top_k=3):
    a = a0.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a /= s
    
    n = len(a)
    deadline = time.monotonic() + time_budget
    best_a = a.copy()
    best_obj = _objective(a)
    
    mu = 0.7  # momentum
    v = np.zeros_like(a)
    lr = 1.0 / np.sqrt(n)
    
    iteration = 0
    while time.monotonic() < deadline:
        b = _conv(a)
        # Top-k indices
        k_vals = np.argpartition(b, -top_k)[-top_k:]
        g = np.zeros_like(a)
        for k in k_vals:
            g += _subgrad_at(a, k)
        g /= top_k
        g -= np.mean(g)  # center to not affect sum
        
        v = mu * v - lr * g
        a_new = a + v
        a_new = _project_nn_sum(a_new, 1.0)
        
        obj = _objective(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        
        # Adaptive lr
        if obj < best_obj + 1e-4:
            a = a_new
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.95
            v *= 0.5
        
        iteration += 1
        
    return best_a, best_obj

    a = a0.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:

    else:
        a /= s

    best_obj = _objective(a)

    temp = 0.01
    cool_rate = 0.999

        if method < 0.4:
            # Block perturbation
            width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            a_new = a.copy()
            a_new[start:stop] *= np.exp(rng.normal(0, 0.15))
        elif method < 0.7:

        elif method < 0.85:

            if abs(s1 - s2) < width:

            a_new = a.copy()
            a_new[s1:s1+width] = a[s2:s2+width]

        else:

            g -= np.mean(g)

            a_new = a - 0.01 * g / (np.max(np.abs(g)) + 1e-10)

        sn = np.sum(a_new)

        else:
```

```python
def _fast_conv(a):
    """Compute a*a (full linear convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _obj_norm(a):
    """Objective for sum=1 normalized sequence."""
    b = _fast_conv(a)
    return 2.0 * len(a) * np.max(b)

def _sg_at(a, k):
    """Subgradient of (a*a)[k] wrt a."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k - idx]
        if k % 2 == 0:
            jm = k // 2
            if js <= jm <= je:
                g[jm] = a[jm]
    return g

def _proj(a, s=1.0):
    """Project onto {x >= 0, sum = s}."""
    if np.max(a) <= 0:
        return np.full(a.shape, s / len(a))
    lo, hi = 0.0, np.max(a)
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        v = np.maximum(0.0, a - mid)
        if v.sum() > s:
            lo = mid
        else:
            hi = mid
    v = np.maximum(0.0, a - 0.5 * (lo + hi))
    sv = v.sum()
    if sv > 0:
        v *= s / sv
    return v

        std_b = b.std()

        peak_indices = np.where(b > threshold)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(b)])
        elif len(peak_indices) > n_peaks:
            # Keep top n_peaks
            peak_indices = peak_indices[np.argsort(b[peak_indices])[-n_peaks:]]

        g = np.zeros(n)
        for k in peak_indices:
            g += _sg_at(a, k)
        g /= len(peak_indices)
        g -= g.mean()
        
        # Line search
        best_lr = lr
        best_local_obj = _obj_norm(a)
        for trial_lr in [lr, lr*0.5, lr*2, lr*0.1]:

            obj_trial = _obj_norm(a_trial)

                best_lr = trial_lr

            if best_cand_obj < best_obj:
                best_a = a.copy()
                best_obj = best_cand_obj
                temp = max(temp * 0.98, min_temp)
```
