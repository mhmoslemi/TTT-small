"""
Minimize 2n*max(f*f)/(sum(f))^2 using smooth gradient descent with multiple initializations.
Uses Adam optimizer with soft-max approximation to the max function, and variance minimization
as an alternative objective. Tries multiple sequence lengths and initializations.
"""

import numpy as np
import time

def next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def smooth_max_val(b, tau):
    """Soft max of array b."""
    b_max = np.max(b)
    return b_max + tau * np.log(np.sum(np.exp((b - b_max) / tau)))

def smooth_max_weights(b, tau):
    """Softmax weights (probabilities) for the gradient."""
    b_max = np.max(b)
    e = np.exp((b - b_max) / tau)
    return e / np.sum(e)

def objective_value(f):
    """Compute the exact objective."""
    n = len(f)
    if n == 0 or np.sum(f) < 0.01:
        return np.inf
    b = np.convolve(f, f, mode='full')
    return 2.0 * n * np.max(b) / (np.sum(f))**2

def cross_corr_fft(a, b, n_out):
    """Compute cross-correlation of a and b using FFT, return first n_out elements.
    result[i] = sum_k a[k] * b[k-i] (with b out of range = 0)
    """
    L = next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    result = np.fft.irfft(A * B.conj(), n=L)
    return result[:n_out]

def adam_smooth_max(f, tau, n_steps, lr, beta1=0.8, beta2=0.999, eps=1e-8):
    """Adam descent on smooth max objective. Returns best (f, score)."""
    f = f.astype(np.float64).copy()
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    best_f = f.copy()
    best_score = objective_value(f)
    
    for t in range(n_steps):
        b = np.convolve(f, f, mode='full')
        W = np.sum(f)
        S = smooth_max_val(b, tau)
        p = smooth_max_weights(b, tau)
        
        # Gradient
        g = cross_corr_fft(p, f, n)
        grad = (4.0 * n / W**2) * g - (4.0 * n * S / W**3)
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = np.maximum(f, 1e-10)
        f = f / np.sum(f)
        
        score = objective_value(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def adam_variance(f, n_steps, lr, beta1=0.8, beta2=0.999, eps=1e-8):
    """Adam descent on variance objective (minimize ||f*f||^2). Returns best (f, score)."""
    f = f.astype(np.float64).copy()
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    best_f = f.copy()
    best_score = objective_value(f)
    
    for t in range(n_steps):
        b = np.convolve(f, f, mode='full')
        W = np.sum(f)
        
        # Gradient of ||b||^2: 4 * (b cross-correlated with f)
        g = cross_corr_fft(b, f, n)
        grad = 4.0 * g / W**2  # account for normalization
        
        # Actually, if we're minimizing sum(b_k^2) / (sum f)^4 (normalized),
        # the gradient is more complex. Let's just minimize sum(b_k^2) 
        # and handle normalization separately.
        # d(sum b_k^2)/d f_i = 4 * sum_k b_k * f_{k-i}
        grad = 4.0 * g
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = np.maximum(f, 1e-10)
        f = f / np.sum(f)
        
        score = objective_value(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def make_mseq_init(n, rng):
    """Create a non-negative initialization from an m-sequence."""
    # Find the largest m-1 power of 2 that's <= n
    m = int(np.log2(n))
    length = (1 << m) - 1
    # Generate m-sequence using LFSR
    # Use a primitive polynomial for GF(2^m)
    # For simplicity, use a known primitive polynomial
    # Actually, let's just use a random binary sequence as a proxy
    # Or use the Zadoff-Chu sequence (simpler)
    
    # Zadoff-Chu: z_n = exp(-i * pi * n^2 / (N+1)) for N even, or exp(-i * pi * n*(n+1) / N) for N odd
    # Take real part, shift to be non-negative
    N = length  # or n
    if N % 2 == 0:
        z = np.exp(-1j * np.pi * np.arange(N)**2 / N)
    else:
        z = np.exp(-1j * np.pi * np.arange(N) * (np.arange(N) + 1) / N)
    
    # Take real part and shift to non-negative
    real_part = np.real(z)
    f = (real_part - np.min(real_part)) + 0.1  # shift to be positive
    f = f / np.sum(f)
    
    if len(f) != n:
        # Pad or truncate
        if len(f) < n:
            f = np.pad(f, (0, n - len(f)))
        else:
            f = f[:n]
        f = f / np.sum(f)
    
    return f

def make_structured_init(n):
    """Create a structured initialization based on the binomial coefficient pattern."""
    # f[j] proportional to C(2j, j) / 4^j for j = 0, ..., n-1
    # This is related to the flat autocorrelation condition
    j = np.arange(n)
    # Use log for numerical stability
    # log(C(2j,j)/4^j) = log(C(2j,j)) - j*log(4)
    # log(C(2j,j)) = sum_{k=1}^{j} log(2j-k+1) - sum_{k=1}^{j} log(k)
    # = sum_{k=j+1}^{2j} log(k) - log(j!)
    # For large j, C(2j,j)/4^j ≈ 1/sqrt(pi*j)
    
    # Simple approximation: f[j] ≈ 1/sqrt(j+1)
    f = 1.0 / np.sqrt(j.astype(np.float64) + 1.0)
    f = f / np.sum(f)
    return f

def make_symmetric_structured_init(n):
    """Create a symmetric structured initialization."""
    half = n // 2
    j = np.arange(half + 1)
    f_half = 1.0 / np.sqrt(j.astype(np.float64) + 1.0)
    if n % 2 == 0:
        f = np.concatenate([f_half, f_half[-2::-1]])
    else:
        f = np.concatenate([f_half, f_half[-1::-1]])
    f = f / np.sum(f)
    return f

def block_local_search(f, rng, n_iterations, score_func=objective_value):
    """Simple block perturbation local search."""
    best = f.copy()
    best_score = score_func(best)
    n = len(best)
    
    for _ in range(n_iterations):
        # Random block perturbation
        width = max(1, n // 100)
        width = int(rng.integers(width // 2, width * 2))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)
        
        candidate = best.copy()
        candidate[start:end] *= np.exp(rng.normal(0, 0.1, size=end-start))
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        
        score = score_func(candidate)
        if score < best_score:
            best = candidate
            best_score = score
    
    return best, best_score
