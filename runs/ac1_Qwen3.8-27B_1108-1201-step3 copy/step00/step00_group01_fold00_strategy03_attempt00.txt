def make_block_perturbation(base, rng):
    # Check if base is a list
    if not isinstance(base, list) or any(x < 0 for x in base):
        base = [float(x) for x in base]
    candidate = np.array(base, dtype=np.float64).copy()
    n = len(candidate)
    # Choose a random window length
    window_low, window_high = 10, min(99, n-10)  # Ensure at least 10 elements not on edge
    window_length = rng.integers(window_low, window_high + 1)
    # Choose a start index
    start_id = rng.integers(0, n - window_length)
    # Select the window
    window = rng.uniform(0.0, 1.2)  # Small factor change
    candidate[start_id:start_id + window_length] *= window
    # Optional mirror swap
    if hasattr(rng, 'float') and rng.random() < 0.3:
