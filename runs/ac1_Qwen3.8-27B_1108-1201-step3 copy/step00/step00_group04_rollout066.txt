"""
Minimizes  2*n*max(a*a)/(sum a)^2  over a>=0.

Key insight baked into this program: for SYMMETRIC a the convolution peak is the
energy and the uniform sequence is locally optimal at value 2.0, so to beat 2.0 we
must use NON-SYMMETRIC sequences (mass pushed toward one end / gaps of zeros that
break the central overlap).  We therefore (i) seed many non-symmetric starting
points, (ii) repeatedly apply the best-response LP from the state-of-the-art
(maximize sum(g) subject to g>=0 and (f*g)<=M), blending with an adaptive step,
and (iii) sprinkle in fast FFT-based local perturbations to escape flat spots.
All objective evaluations inside the loop use a fast rfft autocorrelation so we
can afford many iterations in 1800 s.
"""

import time
import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix


def fast_objective(a):
    """Fast evaluation via rfft; matches evaluate_sequence for a>=0 finite."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    size = 2 * n - 1
    F = np.fft.rfft(a, size=size)
    conv = np.real(np.fft.irfft(F * F, size=size))
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def build_conv_matrix(f):
    """(2n-1 x n) sparse matrix A with (A g)[k] = (f * g)[k] (full)."""
    n = f.size
    ks = np.arange(2 * n - 1)
    j0 = np.maximum(0, ks - n + 1)
    j1 = np.minimum(n - 1, ks)
    counts = j1 - j0 + 1
    row_idx = np.repeat(ks, counts)
    j0_rep = np.repeat(j0, counts)
    offs = np.concatenate([np.arange(c) for c in counts])
    col_idx = j0_rep + offs
    vals = f[col_idx]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_lp_best_response(f, M):
    """Maximize sum(g) s.t. g>=0 and (f*g)[k] <= M for all k."""
    n = f.size
    try:
        A = build_conv_matrix(f)
        g = cp.Variable(n, nonneg=True)
        gv = cp.gvec(g, n)
        cons = [A @ gv <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(gv)), cons)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            prob.solve(solver=cp.ECOS)
        if g.value is None:
            return None
        out = np.asarray(g.value, dtype=np.float64).ravel()
        if not np.all(np.isfinite(out)):
            return None
        return np.clip(out, 0.0, None)
    except Exception:
        return None


def block_perturb(a, rng):
    """Cheap non-negative local perturbation (multiplicative block + small mass swap)."""
    c = a.copy()
    n = c.size
    w = int(rng.integers(max(2, n // 120), max(3, n // 12)))
    s = int(rng.integers(0, max(1, n - w)))
    c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.18)))
    if rng.random() < 0.5 and n > 2 * w:
        w2 = int(rng.integers(max(2, n // 200), max(3, w)))
        s2 = int(rng.integers(0, max(1, n - w2)))
        if s2 != s:
            c[s2:s2 + w2] *= float(np.exp(rng.normal(0.0, 0.10)))
    return np.clip(c, 0.0, 1000.0)


def make_starts(n, parent):
    """Return a list of diverse (mostly non-symmetric) initial sequences."""
    idx = np.arange(n, dtype=np.float64)
    starts = []

    def push(x):
        x = np.clip(np.asarray(x, dtype=np.float64), 0.0, 1000.0)
        if x.size == n and np.all(np.isfinite(x)) and np.sum(x) > 0.01:
            starts.append(x.copy())

    if parent is not None and np.asarray(parent).size == n:
        push(parent)
    push(np.ones(n))                                   # symmetric baseline (=2.0)
    push(1.0 - idx / max(1, n - 1))                    # decreasing ramp
    push((idx / max(1, n - 1)) ** 2)                   # increasing ramp^2
    push(np.exp(-idx * (3.0 / max(1, n - 1))))         # exponential decay
    push(np.where(np.arange(n) % 2 == 0, 1.0, 0.15))   # asymmetric comb
    push(1.0 + 0.0 * idx - np.exp(-(idx / n) ** 2 * 8.0) * 0.0)  # placeholder keep
    c = np.ones(n)
    c[int(0.6 * n):] = 0.25                            # front-loaded block w/ notch
    push(c)
    return starts


def refine_one(start, deadline, rng):
    """Run best-response LP + local perturb on a single start within deadline."""
    f = np.clip(np.asarray(start, dtype=np.float64), 0.0, 1000.0)
    f = f / max(1e-9, f.sum()) * float(np.sqrt(2.0 * f.size))  # stable normalization
    best = f.copy()
    best_score = fast_objective(best)
    stalls = 0
    last_progress = time.monotonic()
    while time.monotonic() < deadline:
        M = fast_objective_peak(best)
        g = solve_lp_best_response(best, M)
        step_accepted = False
        if g is not None and np.sum(g) > 1e-9:
            g = g * (best.sum() / np.sum(g))
            for t in (0.5, 0.25, 0.1, 0.35):
                cand = np.clip((1.0 - t) * best + t * g, 0.0, 1000.0)
                sc = fast_objective(cand)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = cand, sc
                    step_accepted = True
                    break
        if step_accepted:
            stalls = 0
            last_progress = time.monotonic()
        else:
            # local escape moves
            moved = False
            for _ in range(4):
                cand = block_perturb(best, rng)
                sc = fast_objective(cand)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = cand, sc
                    moved = True
            if not moved:
                stalls += 1
            else:
                last_progress = time.monotonic()
        # cut the start early if long-stalled
        if stalls > 25 or (time.monotonic() - last_progress) > 60.0 and stalls > 6:
            break
    return best, best_score


def fast_objective_peak(a):
    n = a.size
    size = 2 * n - 1
    F = np.fft.rfft(a, size=size)
    return float(np.max(np.real(np.fft.irfft(F * F, size=size))))

            out = [1.0] * n
            official = float(evaluate_sequence(out))
            print(f"[final-fallback] uniform score={official:.6f}")
    except Exception:
        pass

    return out
