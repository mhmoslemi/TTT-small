import numpy as np
import time

def smooth_max(values, T):
    """Compute smooth approximation of max using log-sum-exp."""
    shifted = T * (values - np.max(values))
    log_sum = np.log(np.sum(np.exp(shifted))) + T * np.max(values)
    return log_sum / T

def compute_smooth_objective(p, T):
    """Compute smooth approximation of max(convolve(p,p))."""
    c = np.convolve(p, p)
    return smooth_max(c, T)

def compute_gradient(p, T):
    """Compute gradient of smooth objective w.r.t. p."""
    c = np.convolve(p, p)
    # Compute softmax of T*c
    Tc = T * c
    Tc_shifted = Tc - np.max(Tc)
    exp_Tc = np.exp(Tc_shifted)
    s = exp_Tc / np.sum(exp_Tc)
    
    # Compute h = correlate(s, p, 'full')[n-1:2*n-1]
    n = len(p)
    h = np.correlate(s, p, mode='full')[n-1:2*n-1]
    
    return 2 * h

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0][-1] + 1  # 1-indexed
    theta = cssv[k-1] / k
    return np.maximum(v - theta, 0)

def optimize_length(n, T_init=10, T_final=100, n_steps=5000, lr_init=0.1, seed=42):
    """Optimize for a given length n."""
    rng = np.random.default_rng(seed)
    w = rng.standard_normal(n)  # Initial weights
    p = np.exp(w) / np.sum(np.exp(w))
    
    best_p = p.copy()
    best_obj = np.max(np.convolve(p, p))
    
    for step in range(n_steps):
        # Anneal T
        T = T_init + (T_final - T_init) * step / max(1, n_steps-1)
        lr = lr_init / (1 + step / 100)
        
        # Compute gradient w.r.t. p
        grad_p = compute_gradient(p, T)
        
        # Convert to gradient w.r.t. w using softmax chain rule
        grad_w = p * (grad_p - np.sum(p * grad_p))
        
        # Update

        p = np.exp(w - np.max(w)) / np.sum(np.exp(w - np.max(w)))
        
        # Evaluate true objective
        obj = np.max(np.convolve(p, p))
        if obj < best_obj:
            best_obj = obj
            best_p = p.copy()
    
    return best_p, 2 * n * best_obj

import numpy as np
import time

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using smooth gradient descent
with log-sum-exp approximation of the max, followed by local refinement.
Uses softmax parameterization to maintain the simplex constraint.
"""

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0).astype(np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    k = k[-1] + 1
    theta = cssv[k - 1] / k
    result = np.maximum(v - theta, 0.0)
    s = np.sum(result)
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def optimize_length(n, steps, T_start, T_end, lr_start, seed, init_type='random'):
    """Run smooth gradient descent for a given sequence length."""
    rng = np.random.default_rng(seed)

    if init_type == 'random':
        w = rng.standard_normal(n)
    elif init_type == 'uniform':
        w = np.zeros(n)
    elif init_type == 'gaussian':

        w = -0.5 * (2.0 / n) * x ** 2
    elif init_type == 'sine':
        w = np.log(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)) ** 2 + 1e-15)
    elif init_type == 'u_shape':
        x = np.arange(n) / max(1, n - 1)
        w = np.log(0.3 + 0.7 * np.cos(np.pi * (x - 0.5)) ** 2 + 1e-15)
    else:
        w = rng.standard_normal(n)
    
    p = np.exp(w - np.max(w))
    p /= np.sum(p)
    
    best_p = p.copy()
    best_true_max = np.max(np.convolve(p, p))
    
    eval_interval = max(1, steps // 20)
    
    for step in range(steps):
        # Anneal temperature
        frac = step / max(1, steps - 1)
        T = T_start + (T_end - T_start) * frac
        lr = lr_start / (1.0 + step / 50.0)

        c = np.convolve(p, p)

        max_c = np.max(c)
        Tc_shifted = T * (c - max_c)
        exp_Tc = np.exp(Tc_shifted)
        sum_exp = np.sum(exp_Tc)
        
        # Softmax weights for gradient
        s = exp_Tc / sum_exp
        
        # Gradient: 2 * correlate(s, p)[n-1 : 2n-1]
        h = np.correlate(s, p, mode='full')[n - 1:2 * n - 1]
        grad = 2.0 * h

        p = project_simplex(p)

            true_max = np.max(c)

def local_refine(p, n_steps=100):
    """Local refinement: find argmax of convolution and try to reduce it."""
    n = len(p)
    best_p = p.copy()
    best_max = np.max(np.convolve(p, p))
    
    for step in range(n_steps):
        c = np.convolve(best_p, best_p)
        k_max = np.argmax(c)
        current_max = c[k_max]
        
        # Find which positions contribute most to c[k_max]
        # c[k_max] = sum_j p[j] * p[k_max - j]
        contributions = {}
        for j in range(n):
            k_j = k_max - j
            if 0 <= k_j < n:
                contributions[j] = best_p[j] * best_p[k_j]
        
        if not contributions:
            break
        
        # Get top contributors
        sorted_items = sorted(contributions.items(), key=lambda x: -x[1])
        
        # Try reducing the largest contributor
        improved = False
        for j, val in sorted_items[:5]:
            k_j = k_max - j
            if j == k_j:
                continue
            # Try moving a small amount from position j to a different position
            delta = best_p[j] * 0.01
            new_p = best_p.copy()
            new_p[j] -= delta
            # Distribute delta to a random position
            target = np.random.randint(n)
            new_p[target] += delta
            new_p = np.maximum(new_p, 0)
            s = np.sum(new_p)
            if s < 1e-15:
                continue
            new_p /= s
            
            new_c = np.convolve(new_p, new_p)
            new_max = np.max(new_c)
            if new_max < best_max:
                best_max = new_max
                best_p = new_p
                improved = True
                break
        
        if not improved:
            break
    
    return best_p, best_max
