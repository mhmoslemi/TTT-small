Let me also think about the projection onto the probability simplex. Given a vector v, the projection onto {x >= 0, sum x = 1} is:
x_i = max(0, v_i - tau) where tau is chosen so that sum x_i = 1.

Finding tau requires a bisection or sorting-based method. The standard algorithm (from Duchi et al. 2008) is:
1. Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find rho = max{j: v_(j) - (sum_{i=1}^j v_(i)) / j > 0}
3. tau = (sum_{i=1}^rho v_(i)) / rho
4. x_i = max(0, v_i - tau)

This is O(n log n) due to sorting. For n = 6962, this is fast.

Alternatively, I can use scipy.optimize or a simple bisection.

import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Use the sorting-based algorithm
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def smooth_max(c, tau):
    """Compute smooth max of array c using log-sum-exp."""
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(tau * (c - c_max)))) / tau

def compute_convolution(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2*n-1]
    return c

"""
Gradient-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2
using log-sum-exp smoothing of the max, FFT-based convolution, and
projection onto the probability simplex. Multiple initializations and
sequence lengths are explored to escape local minima.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_mask = u * np.arange(1, n + 1) > (cv - 1)
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.max(np.where(rho_mask)[0])
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def fft_conv_square(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2 * n - 1]
    return c


def next_pow2(x):
    """Next power of 2 >= x."""
    L = 1
    while L < x:
        L *= 2
    return L


def compute_score_fft(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 using FFT."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_conv_square(a)
    return float(2.0 * n * np.max(c) / (s * s))


def gradient_step(a, L, L2, tau):
    """
    Compute one gradient step for the smooth objective.
    Returns: (gradient, smooth_value, weights)
    """
    n = len(a)
    # Compute convolution
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2 * n - 1]

    # Compute softmax weights (smoothed max weights)
    c_max = np.max(c)
    exp_c = np.exp(tau * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    smooth_val = c_max + np.log(np.sum(exp_c)) / tau

    # Compute gradient using FFT cross-correlation
    W = np.fft.rfft(weights, L2)
    A = np.fft.rfft(a, L2)
    corr = np.fft.irfft(W * np.conj(A), L2)
    grad = 2.0 * corr[:n]

    return grad, smooth_val, weights, c, c_max, exp_c


def line_search(a, grad, L, initial_step, max_backtracks=10):
    """Backtracking line search."""
    n = len(a)
    fa = np.fft.rfft(a, L)
    c_old = np.fft.irfft(fa * fa, L)[:2 * n - 1]

    step = initial_step
    for _ in range(max_backtracks):
        a_new = project_simplex(a - step * grad)
        fa_new = np.fft.rfft(a_new, L)
        c_new = np.fft.irfft(fa_new * fa_new, L)[:2 * n - 1]
        # Use max as a simple objective (no smoothing in line search for speed)
        if np.max(c_new) < np.max(c_old) - 1e-15:
            return a_new, step
        step *= 0.5
    return a, step


def optimize_from_init(a_init, deadline, tau_start=0.5, tau_end=100.0,
                       step_start=0.5, label=""):
    """
    Run gradient descent from initialization a_init.
    Uses increasing tau and adaptive step size.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64).copy())
    L = next_pow2(2 * n - 1)
    L2 = next_pow2(3 * n - 2)

    best_score = compute_score_fft(a)
    best_a = a.copy()

    n_total = max(1, int((deadline - time.monotonic()) / 0.003))
    n_total = min(n_total, 500000)  # cap iterations
    if n_total < 10:
        n_total = 10

    tau = tau_start
    step = step_start
    smooth_tol = 1e-10

    it = 0
    last_improve_score = best_score
    no_improve_count = 0

    while time.monotonic() < deadline - 2.0:
        # Update tau (geometric increase)
        progress = it / max(1, n_total)
        tau = tau_start * (tau_end / tau_start) ** progress
        tau = min(tau, tau_end)

        # Compute gradient
        grad, smooth_val, weights, c, c_max, exp_c = gradient_step(a, L, L2, tau)

        # Adaptive step size using gradient norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            # Try a random perturbation to escape
            rng = np.random.default_rng(it)
            a = project_simplex(a + 0.01 * rng.standard_normal(n))
            continue

        # Scale step by gradient magnitude
        effective_step = step / (grad_norm + 1e-8) * np.max(a) + 1e-8
        effective_step = min(effective_step, 5.0)
        effective_step = max(effective_step, 1e-8)

        # Line search
        a_new, used_step = line_search(a, grad, L, effective_step, max_backtracks=8)

        if np.max(a_new) != np.max(a):  # If improvement was found
            a = a_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            # If stuck, try larger perturbation
            if no_improve_count > 50:
                rng = np.random.default_rng(it)
                a = project_simplex(a * (1 + 0.05 * rng.standard_normal(n)))
                no_improve_count = 0
                step *= 0.5

        # Periodic evaluation of true score
        if it % 50 == 0:
            score = compute_score_fft(a)
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()
                last_improve_score = score

        # Decay step size slowly
        if it % 200 == 0 and it > 0:
            step *= 0.95

        it += 1

    # Final evaluation
    final_score = compute_score_fft(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if label:
        print(f"  {label}: best_score={best_score:.6f}, iters={it}, n={n}")

    return best_a, best_score


def random_local_search(a_init, deadline, max_evals=200):
    """Simple random local search starting from a_init."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score_fft(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for _ in range(max_evals):
        if time.monotonic() > deadline - 5.0:
            break

        # Random block perturbation
        width = max(1, int(rng.integers(n // 100, n // 5)))
        start = int(rng.integers(0, n - width + 1))
        a_candidate = a.copy()
        a_candidate[start:start + width] *= np.exp(rng.normal(0, 0.3, width))
        a_candidate = np.clip(a_candidate, 0, 1000)
        a_candidate = project_simplex(a_candidate)

        score = compute_score_fft(a_candidate)
        if score < best_score:
            best_score = score
            best_a = a_candidate
            a = a_candidate

        # Occasionally try full random perturbation
        if rng.random() < 0.05:
            a_candidate = a * np.exp(rng.normal(0, 0.5, n))
            a_candidate = project_simplex(a_candidate)
            score = compute_score_fft(a_candidate)
            if score < best_score:
                best_score = score
                best_a = a_candidate
                a = a_candidate

    return best_a, best_score
