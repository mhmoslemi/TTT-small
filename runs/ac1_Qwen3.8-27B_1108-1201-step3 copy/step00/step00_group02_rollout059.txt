            num_try = min(20, zero_count)
            zero_positions = np.where(zero_mask)[0]
            trial_idx = rng.choice(len(zero_positions), size=num_try, replace=False)
            for p in zero_positions[trial_idx]:
                new_max = try_flip_0to1(conv, a, p, n)
                new_J = 2.0 * n * new_max / (new_k_0to1 * new_k_0to1)
                if new_J < best_new_J:
                    best_new_J = new_J
                    best_pos = p
                    best_type = 1

        # Apply best flip
        if best_pos >= 0 and best_new_J < J:
            p = best_pos
            if best_type == 0:
                conv[p:p+n] -= a
                a[p] = 0.0
                k -= 1
                total_flips_1to0 += 1
            else:
                conv[p:p+n] += a
                a[p] = 1.0
                k += 1
                total_flips_0to1 += 1
            J = best_new_J
            max_b = float(conv.max())
            no_improve = 0

            if J < best_tracker[0]:
                best_tracker[0] = J
                best_tracker[1] = a.copy()
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Try a perturbation: flip 5-20 random bits
                num_perturb = rng.integers(5, 20)
                positions = rng.choice(n, size=num_perturb, replace=False)
                for p in positions:
                    if a[p] > 0:
                        conv[p:p+n] -= a
                        a[p] = 0.0
                        k -= 1
                    else:
                        conv[p:p+n] += a
                        a[p] = 1.0
                        k += 1
                max_b = float(conv.max())
                if k > 10:
                    J = 2.0 * n * max_b / (k * k)
                break

    return a, conv, k, J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using binary local search.

    Key insight: A sparse binary sequence (~50% ones) has a convolution that
    can be made nearly flat through local optimization. The theoretical minimum
    of J is 2n/(2n-1) ≈ 1 for large n. Random binary sequences already achieve
    J ≈ 1.1-1.3, and greedy local search with restarts improves further.

    Algorithm:
    1. Initialize with random binary sequence (k ≈ n/2 ones).
    2. Maintain convolution incrementally (O(n) per bit flip evaluation).
    3. Greedily flip bits (1→0 at peak contributors, 0→1 at random zeros)
       to minimize J.
    4. Use perturbation + restarts to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Determine sequence length from parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = len(parent)
    except (NameError, Exception):
        n = 6962

    # Global best tracker: [best_J, best_array]
    best_tracker = [np.inf, None]

    # Number of restarts based on time budget
    # Each restart gets a portion of the time
    num_restarts = 8
    restart_time = min(180.0, (budget_s - 20.0) / num_restarts)

    for restart in range(num_restarts):
        if time.monotonic() > deadline - 10.0:
            break

        restart_deadline = time.monotonic() + restart_time

        # Initialize random binary sequence
        k = n // 2
        a = np.zeros(n, dtype=np.float64)
        ones_idx = rng.choice(n, size=k, replace=False)
        a[ones_idx] = 1.0

        # Compute initial convolution
        conv = compute_initial_conv(a)
        max_b = float(conv.max())
        J = 2.0 * n * max_b / (k * k)

        print(f"Restart {restart}: n={n}, k={k}, max_b={max_b:.0f}, J={J:.6f}")

        if J < best_tracker[0]:
            best_tracker[0] = J
            best_tracker[1] = a.copy()

        # Run local search
        a, conv, k, J = local_search_binary(
            a, conv, n, k, J, restart_deadline, rng, best_tracker
        )

        print(f"Restart {restart} done: J={J:.6f}, k={k}, "
              f"global_best={best_tracker[0]:.6f}")

    # Final validation with authoritative evaluator
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
    else:
        # Fallback: ones sequence (gives J=2)
        final_seq = np.ones(n, dtype=np.float64)

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"Final evaluated score: {final_score:.6f}")

    if not np.isfinite(final_score):
        # Should not happen, but fallback
        final_seq = np.ones(256, dtype=np.float64)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"Fallback score: {final_score:.6f}")

    return [float(x) for x in final_seq]
