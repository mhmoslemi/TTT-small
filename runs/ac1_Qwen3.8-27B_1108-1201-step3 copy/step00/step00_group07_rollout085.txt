import numpy as np
import time

from scipy.optimize import minimize, Bounds, linprog
from scipy.sparse import diags


"""
Search for a non-negative step function minimizing

    2*n*max(conv(a,a)) / sum(a)**2

The search works in mean-one density coordinates d (sum(d)=n, a=d/n), so the
score is exactly 2*max(conv(d,d))/n.  The main optimizer is a smoothed-max
log-softmax gradient method (L-BFGS-B) over log-softmax densities.  Several
non-symmetric constructions are tried, small patterns are upscaled to larger
lengths, and the best candidate is polished with a limited mirror-descent pass,
an optional LP-based Frank-Wolfe-like direction, another log-softmax refinement,
and a short local stochastic perturbation pass.
"""


def next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def fft_convolve(d, N=None):
    n = len(d)
    if N is None:
        N = next_pow2(2 * n - 1)
    N = int(N)
    if N < 2 * n - 1:
        N = next_pow2(2 * n - 1)
    D = np.fft.rfft(d, N)
    c = np.fft.irfft(D * np.conj(D), N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def score_d(d, N=None):
    n = len(d)
    if n == 0:
        return float("inf")
    c = fft_convolve(d, N)
    return float(2.0 * float(c.max()) / float(n))


def d_from_x(x, n):
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        y = np.zeros(n, dtype=float)
        m = min(n, len(x))
        y[:m] = x[:m]
        x = y
    if not np.all(np.isfinite(x)):
        return np.ones(n, dtype=float)
    z = x - np.max(x)
    p = np.exp(z)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=float)
    d = (n * p) / s
    d = np.clip(d, 1e-12, None)
    ss = float(d.sum())
    if ss <= 0.0 or not np.isfinite(ss):
        return np.ones(n, dtype=float)
    d *= n / ss
    return d


def make_start_d(n, seed, style=0):
    rng = np.random.default_rng(int(seed) * 1000003 + int(style) * 7919)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    d = np.ones(n, dtype=float)

    if style % 2 == 0:
        # Mostly anti-symmetric about the midpoint (sine-only).  These break the
        # uniform fixed point and tend to lower the middle of the convolution.
        mags = (1, 2, 3, 4, 5, 6, 7, 8)
        base = (0.45, 0.24, 0.13, 0.07, 0.04, 0.025, 0.016, 0.010)
        for m, b in zip(mags, base):
            amp = b * float(rng.uniform(0.4, 1.25))
            if rng.random() < 0.5:
                amp = -amp
            d += amp * np.sin(2.0 * np.pi * m * x)
    else:
        # Less structured, low-frequency random start.
        mags = (1, 2, 3, 4, 5)
        base = (0.35, 0.20, 0.11, 0.06, 0.03)
        for m, b in zip(mags, base):
            amp = b * float(rng.uniform(0.3, 1.2))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            d += amp * np.sin(2.0 * np.pi * m * x + phase)

    d += 0.010 * rng.standard_normal(n)
    d = np.clip(d, 1e-6, None)
    s = float(d.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=float)
    d *= n / s
    return d


def resample_d(d, n2):
    d = np.asarray(d, dtype=float).ravel()
    n1 = len(d)
    if n1 == 0:
        return np.ones(int(n2), dtype=float)
    if n1 == int(n2):
        out = d.copy()
    else:
        x1 = np.linspace(0.0, 1.0, n1, endpoint=False)
        x2 = np.linspace(0.0, 1.0, int(n2), endpoint=False)
        out = np.interp(x2, x1, d)
    out = np.clip(out, 1e-12, None)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(int(n2), dtype=float)
    out *= int(n2) / s
    return out


class Objective:
    """Smoothed self-convolution maximum objective on log-softmax parameters."""

    def __init__(self, beta, n, N):
        self.beta = float(beta)
        self.n = int(n)
        self.N = int(N)

    def __call__(self, x):
        n = self.n
        N = self.N
        x = np.asarray(x, dtype=float).ravel()
        if x.size != n:
            y = np.zeros(n, dtype=float)
            m = min(n, len(x))
            y[:m] = x[:m]
            x = y
        if not np.all(np.isfinite(x)):
            return 1e9, np.zeros(n, dtype=float)

        z = x - np.max(x)
        p = np.exp(z)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            return 1e9, np.zeros(n, dtype=float)

        q = p / s
        d = n * q

        D = np.fft.rfft(d, N)
        c = np.fft.irfft(D * np.conj(D), N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        h = c / float(n)
        hmax = float(h.max())

        zc = self.beta * (h - hmax)
        w = np.exp(zc)
        wsum = float(w.sum())
        if not np.isfinite(wsum) or wsum <= 0.0:
            return 1e9, np.zeros(n, dtype=float)

        val = hmax + float(np.log(wsum)) / self.beta
        w /= wsum

        W = np.fft.rfft(w[::-1], N)
        hconv = np.fft.irfft(D * W, N)
        r = hconv[n - 1:2 * n - 1][::-1].copy()

        # Gradient of the smooth max wrt d:
        #   G_i = (2/n) * sum_k w_k d_{k-i}
        G = (2.0 / float(n)) * r
        S = float(np.dot(G, d))
        grad = d * G - q * S
        grad -= float(grad.mean())

        # Tiny gauge regularizer to keep log-parameters near zero mean.
        reg = 1e-12
        val += reg * float(np.dot(x, x))
        grad += 2.0 * reg * x

        scale = 100.0
        if not np.isfinite(val):
            val = 1e9
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return scale * float(val), scale * grad


def lbfgs_refine_d(d, deadline):
    d = np.asarray(d, dtype=float).ravel()
    n = len(d)
    best_d = d.copy()
    best_s = score_d(d)
    if not np.isfinite(best_s):
        best_d = np.ones(n, dtype=float)
        best_s = 2.0

    x = np.log(np.clip(d, 1e-30, None))
    x -= float(x.mean())
    x = np.clip(x, -30.0, 30.0)

    N = next_pow2(3 * n - 2)
    bets = (10.0, 20.0, 40.0, 80.0)

    if n <= 256:
        maxfuns = (80, 100, 120, 120)
    elif n <= 512:
        maxfuns = (100, 120, 150, 150)
    elif n <= 1024:
        maxfuns = (120, 150, 180, 180)
    elif n <= 2048:
        maxfuns = (90, 110, 130, 130)
    else:
        maxfuns = (70, 85, 100, 100)

    bounds = Bounds(-30.0, 30.0)

    for beta, maxf in zip(bets, maxfuns):
        if time.monotonic() > deadline - 5.0:
            break
        if maxf <= 1:
            continue
        obj = Objective(beta, n, N)
        try:
            res = minimize(
                obj,
                x,
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxfun": int(maxf),
                    "maxiter": int(maxf),
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "disp": False,
                },
            )
        except Exception:
            break

        if hasattr(res, "x") and res.x is not None:
            x2 = np.asarray(res.x, dtype=float).ravel()
            if x2.size != n:
                x2 = d_from_x(x2, n)  # fallback: discard bad x below if nonfinite
            if x2.size == n and np.all(np.isfinite(x2)):
                x2 = np.clip(x2, -30.0, 30.0)
                x2 -= float(x2.mean())
                d2 = d_from_x(x2, n)
                s2 = score_d(d2)
                if np.isfinite(s2) and s2 < best_s:
                    best_d = d2
                    best_s = s2
                x = x2

    return best_d, float(best_s)


def mirror_polish(d, deadline, max_steps=50, eta=0.08, beta=40.0):
    d = np.asarray(d, dtype=float).ravel()
    n = len(d)
    best_d = d.copy()
    best_s = score_d(d)
    if not np.isfinite(best_s):
        return np.ones(n, dtype=float), 2.0

    Nbig = next_pow2(3 * n - 2)
    Nsmall = next_pow2(2 * n - 1)
    eta_cur = float(eta)

    for _ in range(int(max_steps)):
        if time.monotonic() > deadline - 5.0:
            break
        D = np.fft.rfft(d, Nbig)
        c = np.fft.irfft(D * np.conj(D), Nbig)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        s = float(2.0 * float(c.max()) / float(n))
        if s < best_s:
            best_d = d.copy()
            best_s = s

        h = c / float(n)
        hmax = float(h.max())
        z = beta * (h - hmax)
        w = np.exp(z)
        wsum = float(w.sum())
        if not np.isfinite(wsum) or wsum <= 0.0:
            break
        w /= wsum

        W = np.fft.rfft(w[::-1], Nbig)
        hconv = np.fft.irfft(D * W, Nbig)
        r = hconv[n - 1:2 * n - 1][::-1].copy()
        r -= float(r.mean())

        improved = False
        for et in (eta_cur, 0.5 * eta_cur, 0.25 * eta_cur):
            if et <= 0.0:
                break
            cand = d * np.exp(-et * r)
            cand = np.clip(cand, 1e-12, None)
            ss = float(cand.sum())
            if ss <= 0.0 or not np.isfinite(ss):
                continue
            cand *= n / ss
            sc = score_d(cand, Nsmall)
            if np.isfinite(sc) and sc < best_s - 1e-14:
                best_d = cand
                best_s = sc
                d = cand
                eta_cur = min(1.0, eta_cur * 1.05)
                improved = True
                break

        if not improved:
            eta_cur *= 0.5
            if eta_cur < 0.01:
                break

    return best_d, float(best_s)


def lp_polish(d, deadline, max_iter=6, per_lp_seconds=15):
    d = np.asarray(d, dtype=float).ravel()
    n = len(d)
    if n < 32 or n > 1600:
        return d, score_d(d)

    best_d = d.copy()
    best_s = score_d(best_d)
    if not np.isfinite(best_s):
        return d, 2.0

    Nsmall = next_pow2(2 * n - 1)
    j = np.arange(n)
    q = np.sin(2.0 * np.pi * j / float(n))
    qmax = float(np.max(np.abs(q)))
    if qmax > 0.0:
        q /= qmax
    eps = 1e-5

    for _ in range(int(max_iter)):
        if time.monotonic() > deadline - 30.0:
            break

        dscale = float(best_d.max())
        if dscale <= 1e-300 or not np.isfinite(dscale):
            break
        dn = best_d / dscale
        if not np.all(np.isfinite(dn)):
            break

        try:
            diag_list = [np.full(n - r, dn[r], dtype=float) for r in range(n)]
            A = diags(diag_list, offsets=list(range(n)), shape=(2 * n - 1, n), format="csc")
            c_obj = -np.ones(n, dtype=float) - eps * q
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=np.ones(2 * n - 1, dtype=float),
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": int(per_lp_seconds), "presolve": True},
            )
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None or not np.all(np.isfinite(res.x)):
            break

        g = np.asarray(res.x, dtype=float)
        C = float(g.sum())
        if C <= 1e-12:
            break

        gmean = g * (float(n) / C)
        gmax = float(gmean.max())
        if not np.isfinite(gmax) or gmax > 25.0:
            # If the LP extreme point is too spiky, blend with the current
            # mean-one direction.  Line search below still protects quality.
            gmean = 0.7 * gmean + 0.3 * np.ones(n, dtype=float)
            gmean *= float(n) / float(gmean.sum())

        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            cand = (1.0 - t) * best_d + t * gmean
            cand = np.clip(cand, 1e-12, None)
            ss = float(cand.sum())
            if ss <= 0.0 or not np.isfinite(ss):
                continue
            cand *= n / ss
            sc = score_d(cand, Nsmall)
            if np.isfinite(sc) and sc < best_s - 1e-13:
                best_d = cand
                best_s = sc
                improved = True
                print("LP polish score:", best_s, flush=True)
                break

        if not improved:
            break

    return best_d, float(best_s)


def stochastic_polish(d, deadline, rng):
    d = np.asarray(d, dtype=float).ravel()
    n = len(d)
    if n < 2:
        return d, score_d(d)

    best_d = d.copy()
    best_s = score_d(d)
    if not np.isfinite(best_s):
        return np.ones(n, dtype=float), 2.0

    Nsmall = next_pow2(2 * n - 1)
    wmin = max(2, n // 2000)
    wmax = max(wmin + 1, n // 40)
    sigma = 0.12
    stagnant = 0

    while time.monotonic() < deadline - 5.0:
        cand = best_d.copy()
        width = int(rng.integers(wmin, wmax + 1))
        start = int(rng.integers(0, n - width + 1))
        cand[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 1e-12, None)
        ss = float(cand.sum())
        if ss <= 0.0 or not np.isfinite(ss):
            continue
        cand *= n / ss

        sc = score_d(cand, Nsmall)
        if np.isfinite(sc) and sc < best_s - 1e-14:
            best_d = cand
            best_s = sc
            stagnant = 0
            sigma = min(0.25, sigma * 1.03)
        else:
            stagnant += 1
            if stagnant > 250:
                sigma = max(0.03, sigma * 0.75)
                stagnant = 0

    return best_d, float(best_s)

            except Exception:
                pass

        for i in range(count):
            if len(starts) >= count + 1:
                break
            starts.append(make_start_d(n, int(seed) + 17 * n + 131 * i, i))

        if best_d is not None:
            try:
                starts.append(resample_d(best_d, n))
            except Exception:
                pass

        for idx, d0 in enumerate(starts):
            if time.monotonic() > max(lbfgs_deadline, t0 + 60.0):
                break
            if time.monotonic() > deadline - 80.0:
                break

            try:
                d0 = np.asarray(d0, dtype=float).ravel()
                d0 = np.clip(d0, 1e-12, None)
                s0 = float(d0.sum())
                if s0 <= 0.0 or not np.isfinite(s0):
                    continue
                d0 *= n / s0
            except Exception:
                continue

            s0 = score_d(d0)
            if np.isfinite(s0) and s0 < best_s:
                best_s = s0
                best_d = d0.copy()
                print(f"n={n} start={idx} initial score={best_s:.10f}", flush=True)

            try:
                dref, sref = lbfgs_refine_d(d0, deadline - 60.0)
            except Exception:
                dref, sref = d0, s0

            if np.isfinite(sref) and sref < best_s:
                best_s = sref
                best_d = dref
                print(f"n={n} start={idx} L-BFGS score={best_s:.10f}", flush=True)

            # Quick mirror polish on promising candidates.
            if np.isfinite(sref) and sref < min(best_s + 0.05, 2.0) and time.monotonic() < deadline - 120.0:
                try:
                    dp, sp = mirror_polish(dref, deadline - 90.0, max_steps=35, eta=0.07, beta=40.0)
                    if np.isfinite(sp) and sp < best_s:
                        best_s = sp
                        best_d = dp
                        print(f"n={n} start={idx} mirror score={best_s:.10f}", flush=True)
                except Exception:
                    pass

        print(f"Finished n={n}, best score so far={best_s:.10f}", flush=True)

    # If the parent is a long-ish useful construction, try one direct refinement.
    if (
        parent_clean is not None
        and 1024 <= parent_clean.size <= 8192
        and time.monotonic() < deadline - 260.0
    ):
        try:
            d0 = parent_clean.copy()
            std = float(np.std(d0))
            if std < 0.02:
                x = np.linspace(0.0, 1.0, d0.size, endpoint=False)
                d0 = 0.6 * d0 + 0.4 * (1.0 + 0.55 * np.sin(2.0 * np.pi * x))
                d0 = np.clip(d0, 1e-8, None)
            s0 = float(d0.sum())
            if s0 > 0.0 and np.isfinite(s0):
                d0 *= d0.size / s0
            dref, sref = lbfgs_refine_d(d0, deadline - 120.0)
            if np.isfinite(sref) and sref < best_s:
                best_s = sref
                best_d = dref
                print("Parent refinement score:", best_s, flush=True)
        except Exception:
            pass

    # LP polish the best candidate when its length is moderate.
    if best_d is not None and time.monotonic() < deadline - 140.0:
        try:
            d_lp, s_lp = lp_polish(best_d, deadline - 110.0, max_iter=6, per_lp_seconds=15)
            if np.isfinite(s_lp) and s_lp < best_s:
                best_s = s_lp
                best_d = d_lp
        except Exception:
            pass

    # Another L-BFGS refinement on the current best.
    if best_d is not None and time.monotonic() < deadline - 90.0:
        try:
            d2, s2 = lbfgs_refine_d(best_d, deadline - 50.0)
            if np.isfinite(s2) and s2 < best_s:
                best_s = s2
                best_d = d2
        except Exception:
            pass

    # Final stochastic local polish.
    if best_d is not None and time.monotonic() < deadline - 20.0:
        try:
            d_st, s_st = stochastic_polish(best_d, deadline - 10.0, rng)
            if np.isfinite(s_st) and s_st < best_s:
                best_s = s_st
                best_d = d_st
        except Exception:
            pass

    if best_d is None:
        best_d = np.ones(256, dtype=float)
        best_s = 2.0

    best_d = np.asarray(best_d, dtype=float).ravel()
    n = len(best_d)
    if n == 0:
        best_d = np.ones(256, dtype=float)
        n = 256

    best_d = np.where(np.isfinite(best_d), best_d, 0.0)
    best_d = np.clip(best_d, 0.0, None)
    ssum = float(best_d.sum())
    if ssum <= 0.0 or not np.isfinite(ssum):
        best_d = np.ones(256, dtype=float)
        n = 256
        ssum = float(n)
    best_d *= n / ssum

    # Returning a*d/n has sum 1 and values in [0,1], safely inside evaluator bounds.
    out = best_d / float(n)

    try:
        exact = evaluate_sequence([float(v) for v in out])
        print("Final internal score:", best_s, flush=True)
        print("Final evaluate_sequence score:", exact, flush=True)
    except Exception:
        print("Final internal score:", best_s, flush=True)

    return [float(v) for v in out]
