Minimize  2*n*max(convolve(f,f)) / (sum f)^2  for a non-negative sequence f.

The objective is scale invariant, so we optimize on the simplex (sum f = 1)
and only need to minimize  n * max(f*f).  We use a log-sum-exp ("smooth max")
surrogate for the max of the auto-convolution and run projected/steepest
descent with adaptive backtracking line search and monotone annealing of the
temperature.  The key gradient identity used is: for the full auto-convolution
c[k] = sum_i f[i] f[k-i] (zero padded), dc[k]/df[j] = 2*f[k-j], so the
surrogate gradient is a simple cross-correlation computed with the FFT.  The
true (max-based) objective tracks the best iterate throughout.
"""

from scipy.fft import fft, ifft, next_fast_len


def _as_normalized_vector(parent):
    """Return a finite, non-negative, sum-to-one numpy vector from `parent`."""
    if isinstance(parent, (list, tuple, np.ndarray)):
        v = np.asarray(parent, dtype=np.float64).ravel()
    else:
        v = np.array([], dtype=np.float64)
    if v.size == 0:
        v = np.ones(256, dtype=np.float64)
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if not np.isfinite(s) or s <= 1e-12:
        v = np.ones(np.max(v.size, 256), dtype=np.float64)
        s = v.sum()
    else:
        v = v / s
    # keep a sane length and a positive sum after normalization
    if v.size < 2:
        v = np.ones(256, dtype=np.float64) / 256.0
    return v


def _convolution_full(v):
    """Full auto-convolution of a 1-D vector (length 2n-1)."""
    return np.convolve(v, v, mode="full")


def _project_simplex(v):
    """Clip at zero and renormalize to sum 1 (cheap simplex projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 1e-15:
        n = v.size
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / s


def _smooth_max_value(c, eps):
    """Value and softmax weights of the log-sum-exp surrogate of max(c)."""
    m = c.max()
    e = np.exp(eps * (c - m))
    z = e.sum()
    if z <= 0.0 or not np.isfinite(z):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return m, p
    val = m + np.log(z) / eps
    p = e / z
    return val, p


def _surrogate_gradient(p, f, N):
    """g[j] = 2 * sum_k p[k] * f[k-j]  for j in 0..n-1  (f zero padded)."""

    P = fft(p, N)
    r = np.real(ifft(P * np.conj(F)))
    n = f.size
    return 2.0 * r[:n]
