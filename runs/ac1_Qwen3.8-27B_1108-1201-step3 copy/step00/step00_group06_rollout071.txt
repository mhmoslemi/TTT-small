Minimizes the evaluation score 2*n*max(conv(a,a))/(sum(a))^2 using 
FFT-based gradient descent on a smoothed peak objective, with periodic 
random restarts for exploration.

import numpy as np
import time


def compute_gradient_fft(a, h):
    """Compute g[i] = 2 * sum_k h[k] * a[k-i] using FFT."""
    n = len(a)
    # g[i] = 2 * conv(h, a[::-1])[i + n - 1]
    N = 1
    while N < len(h) + n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    A = np.fft.rfft(a[::-1], N)
    result = np.fft.irfft(H * A, N)
    g = 2.0 * result[n-1 : n-1+n]
    return g

    if s < 0.01:
        return float('inf')
    return float(2 * n * c.max() / (s * s))
