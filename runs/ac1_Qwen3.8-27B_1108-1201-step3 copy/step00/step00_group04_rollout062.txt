```python
def lp_step(a, M):
    """Given fixed a and bound M, find b that maximizes sum(b) subject to convolve(a,b) <= M and b >= 0."""
    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: For each k = 0, ..., 2n-2:
    #   sum_{j} A[k,j] * b[j] <= M
    # where A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve LP
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if not result.success:
        return a.copy()
    
    b = result.x
    return b
```

```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def solve_lp(a, M):
    """Solve: max sum(b) s.t. convolve(a, b) <= M, b >= 0"""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(b)
    
    # Build constraint matrix sparsely
    # Constraint k: sum_j A[k,j] * b[j] <= M
    # A[k,j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if not result.success:
        return None
    return result.x
```

```python
import numpy as np
import time

def compute_objective_and_grad(a):
    """Compute objective value and gradient."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, None
    
    b = np.convolve(a, a)
    max_b = np.max(b)
    k_star = np.argmax(b)
    
    obj = 2 * n * max_b / (S * S)
    
    # Gradient
    # g = b[k_star], h = S^2
    # f = 2n * g / h
    # df/da = 2n * (dg/da * h - g * dh/da) / h^2
    # dg/da = 2 * a_reversed_shifted (vector where [m] = 2*a[k_star - m] if valid)
    # dh/da = 2 * S * ones
    
    g = max_b
    h = S * S
    
    # dg/da[m] = 2 * a[k_star - m] if 0 <= k_star - m < n, else 0
    dg = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            dg[m] = 2 * a[j]
    
    dh = 2 * S * np.ones(n)
    
    grad = 2 * n * (dg * h - g * dh) / (h * h)
    
    return obj, grad

def gradient_descent(a, deadline, rng):
```
