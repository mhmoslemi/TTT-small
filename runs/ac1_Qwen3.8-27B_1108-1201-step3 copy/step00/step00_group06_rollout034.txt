```python
import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    # c = -1 for all variables (we want to maximize sum)
    c = -np.ones(n)
    
    # A: shape (2n-1, n), A_eq @ g <= M
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # For constraint k: sum_j f[j] * g[k-j] <= M
    # g index = k - j, so j = k - g_index
    # A[k, g_index] = f[k - g_index] if 0 <= k - g_index < n
    
    # Build A efficiently
    # A is a banded matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is like a Toeplitz matrix
    
    # For large n, building A explicitly uses O(n^2) memory. For n=5000, that's 25M floats = 200MB. Might be too much.
    
    # Alternative: use the 'highs' method which can handle sparse matrices
    from scipy.sparse import csr_matrix
    
    # Build sparse A
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)
```

```python
    n = len(f)
```

```python
    for _ in range(n_passes):
        order = rng.permutation(n)
        for j in order:
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            length = k_end - k_start

            delta = max(0.0, delta_max)
            if delta > 1e-15:
                g[j] += delta
                c[k_start:k_end] += delta * f_slice
    
    return g
```
