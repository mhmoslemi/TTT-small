import time

def project_simplex(v, S):
    """Project v onto the simplex {x >= 0, sum(x) = S}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (S * S)

Hmm, let me think about this more carefully. The issue with the projected gradient method is that the projection can significantly change the direction, causing oscillations.

A better approach might be **Frank-Wolfe (Conditional Gradient)** method:
- At each step, find the vertex of the simplex that minimizes the linearization of the objective.
- Move towards that vertex.

The Frank-Wolfe step:
1. Compute gradient g of the smooth surrogate.
2. Find s = argmin_{x in simplex} g^T x = e_j where j = argmin(g) (the simplest vertex with the smallest gradient component).
3. Move: a_new = (1-t)*a + t*s, where t is chosen by line search or t = 2/(k+2).

But the simplex has n vertices, and the step is always towards a single vertex. This might be slow.

Alternative: **Mirror Descent with entropy regularizer**:
- Use the KL divergence as the Bregman divergence.
- The update is: `a_new_i ∝ a_i * exp(-lr * g_i)`
- Then normalize: `a_new = a_new / sum(a_new) * S`

This is the **multiplicative weights** update! It's simple and effective for simplex-structured problems.

import numpy as np
import time


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = _next_pow2(n)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _project_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not cond.any():
        return np.full(n, S / n)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _obj(a):

    S = a.sum()
    if S < 0.01:
        return np.inf
    b = _fft_convolve(a, a)

with adaptive temperature, multiple inits, and random restarts.

import numpy as np
import time


def _npo2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _conv(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = _npo2(n)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _proj(v, S):
    """Project onto {x >= 0, sum x = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    c = u * np.arange(1, n + 1) > (cs - S)
    if not c.any():
        return np.full(n, S / n)
    r = np.max(np.nonzero(c)[0])
    th = (cs[r] - S) / (r + 1.0)
    return np.maximum(v - th, 0.0)


def _objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = _conv(a, a)
    return 2.0 * n * b.max() / (S * S)

    S = a.sum()

    w = np.exp(np.clip(T * (b - bm), -100, 0))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        w[n - 1] = 1.0
    else:
        w /= ws
    ar = a[::-1]
    gc = _conv(w, ar)

Mirror-descent (multiplicative-weights) on the softmax-smoothed
max-autocorrelation objective, with adaptive temperature,
block-perturbation restarts, and multiple diverse initialisations.
"""

import numpy as np
import time


def _npo2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _conv(a, b):
    total = len(a) + len(b) - 1
    sz = _npo2(total)
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)[:total]


def _proj_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    c = u * np.arange(1, n + 1) > (cs - S)
    if not c.any():
        return np.full(n, S / n)
    r = int(np.max(np.nonzero(c)[0]))
    th = (cs[r] - S) / (r + 1.0)
    return np.maximum(v - th, 0.0)


def _obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * _conv(a, a).max() / (S * S)


def _compute_grad(a, T):
    """Return (softmax-weighted) gradient of max-convolution at a."""
    n = len(a)
    b = _conv(a, a)
    bm = b.max()
    w = np.exp(np.clip(T * (b - bm), -200.0, 0.0))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        w[n - 1] = 1.0
    else:
        w /= ws
    ar = a[::-1]
    gc = _conv(w, ar)
    return 2.0 * gc[n - 1:2 * n - 1]


def _md_update(a, grad, lr):
    """Multiplicative-weights (mirror) update on the simplex."""
    n = len(a)
    S = a.sum()
    a_new = a * np.exp(-lr * grad)
    s = a_new.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    return a_new / s * S


def _pg_update(a, grad, lr):
    """Projected gradient update on the simplex."""
    n = len(a)
    S = a.sum()
    return _proj_simplex(a - lr * grad, S)
