import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import scipy.sparse as sp

def compute_score(a):
    """Compute the score for a sequence a."""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * np.max(b) / (sum_a ** 2)

def solve_lp_for_direction(a, M):
    """
    Solve the LP: maximize sum(g) subject to g >= 0, (a*g)_k <= M for all k.
    Returns the optimal g.
    """
    n = len(a)
    # Build the sparse constraint matrix A where A @ g <= M (for all k)
    # (a*g)_k = sum_j A[k,j] * g[j] where A[k,j] = a[k-j] if valid
    
    # The matrix is (2n-1) x n, sparse
    # Row k has non-zeros at columns j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # And 0 <= j < n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # In scipy form: min -sum(g) s.t. A @ g <= M, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.zeros(n)

5. Normalize: a = a / sum(a)

```python
def fast_peak_descent(a, n, n_iters=100000, lr=0.01, k_top=1):
    """Fast peak descent using FFT-based convolution."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a)) / (a.sum()**2)
    
    for it in range(n_iters):
        # Compute convolution using FFT
        b = np.fft.irfft(np.fft.rfft(a, 2*n) ** 2, 2*n-1)
        # Or just use np.convolve for simplicity (O(n^2) but fast in C)
        b = np.convolve(a, a)
        
        M = b.max()
        
        # Find top-K peak positions
        if k_top == 1:
            k_star = b.argmax()
            # Gradient: grad_i = a[k*-i] for valid i
            grad = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            for j in range(j_start, j_end + 1):
                grad[j] = a[k_star - j]
        else:
            # Use top-K positions
            top_k = np.argsort(b)[-k_top:]
            grad = np.zeros(n)
            for k_s in top_k:
                j_start = max(0, k_s - n + 1)
                j_end = min(n - 1, k_s)
                for j in range(j_start, j_end + 1):
                    grad[j] += a[k_s - j] / k_top
        
        # Update
        a = a - lr * grad / M
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a_sum
        
        # Check score
        b_new = np.convolve(a, a)
        score = 2.0 * n * b_new.max() / (a.sum()**2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
```
