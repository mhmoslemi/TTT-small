    """Compute gradient of smooth max objective."""
    A_hat = np.fft.rfft(a, N_fft)
    P_hat = np.fft.rfft(p, N_fft)

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using multi-restart smooth gradient descent
with log-sum-exp approximation of max, FFT-based gradients, momentum, annealing,
and diverse structured initializations including asymmetric windows.
"""

import numpy as np
import time


def compute_grad_fft(a, p, N_fft):
    """Compute gradient of smooth max(conv(a,a)) objective using FFT cross-correlation."""
    A_hat = np.fft.rfft(a, N_fft)
    P_hat = np.fft.rfft(p, N_fft)
    corr = np.fft.irfft(np.conj(A_hat) * P_hat, N_fft)[:len(a)]
    return 2.0 * corr


def fast_score(a):
    """Fast computation of objective (assumes sum(a)=1)."""
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max()


defsmooth_step(a, tau, N_fft):
    """One smooth gradient step. Returns (new_a, grad)."""
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    p_sum = exp_c.sum()
    if p_sum <= 0 or not np.isfinite(p_sum):
        return a, np.zeros_like(a)
    p = exp_c / p_sum
    grad = compute_grad_fft(a, p, N_fft)
    return grad


def generate_inits(n, rng):

def optimize_from_init(a, n, N_fft, deadline, rng, tau_init=0.05, lr_init=0.5, label=""):
    """Run smooth gradient descent from a given initialization."""
    a = a.copy()
    # Normalize
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = n
    a = a / S

    best_a = a.copy()
    best_score = fast_score(a)

    tau = tau_init
    lr = lr_init
    mu = np.zeros(n)  # momentum
    beta_m = 0.9
    iteration = 0
    last_check = time.monotonic()
    no_improve_count = 0

    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        c_max = c.max()
        exp_c = np.exp((c - c_max) / tau)
        p_sum = exp_c.sum()
        if p_sum <= 0 or not np.isfinite(p_sum):
            break
        p = exp_c / p_sum

        grad = compute_grad_fft(a, p, N_fft)

        # Momentum step
        mu = beta_m * mu + (1.0 - beta_m) * grad
        a = a - lr * mu
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-15:
            a = best_a.copy()
        else:
            a = a / S

        # Anneal
        tau = max(tau * 0.9995, 0.005)
        lr = max(lr * 0.9999, 0.01)

        iteration += 1

        if time.monotonic() - last_check > 2.0:
            score = fast_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
                if no_improve_count > 8:
                    # Perturb to escape local min
                    a = a + 0.01 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    S2 = a.sum()
                    if S2 > 0:
                        a = a / S2
                    mu = np.zeros(n)
                    tau = max(tau * 2, 0.05)
                    lr = max(lr * 2, 0.1)
                    no_improve_count = 0
            last_check = time.monotonic()

    return best_a, best_score
