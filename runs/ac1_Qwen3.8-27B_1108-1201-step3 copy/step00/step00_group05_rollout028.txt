import time

def compute_obj(a):
    """Compute J = 2n * max(convolve(a,a)) / sum(a)^2"""
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2 * n * np.max(c) / s**2

def obj_val(a):
    """Just the value to minimize (scale-invariant, use sum=1)."""
    n = len(a)
    c = np.convolve(a, a)
    return np.max(c)

def fft_convolve(a):
    """Fast convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]

def optimize_sequence(a, max_time, initial_rate=0.1, rng=None):

        rng = np.random.default_rng()
    
    n = len(a)
    a = a / np.sum(a)  # Normalize to sum = 1
    best_a = a.copy()
    best_val = obj_val(a)
    start = time.monotonic()
    
    eta = initial_rate
    patience = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve(a)
        m_star = np.argmax(c)
        peak = c[m_star]
        
        if peak < best_val:
            best_val = peak
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:

                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                patience = 0
                continue

        # In practice, the "direction" to decrease a_j is proportional to a[m_star - j]
