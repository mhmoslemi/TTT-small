**Projected Subgradient Method:**

Since the problem is non-convex, we can use a subgradient method with projection:
1. Compute the subgradient of $M(f) = \max_k (f*f)_k$.
2. Take a step in the negative subgradient direction.
3. Project onto the constraint set $\{f \geq 0, \sum f = S\}$.

The projection onto $\{f \geq 0, \sum f = S\}$ (the simplex) is:
$f_i^* = \max(0, f_i - \tau)$ where $\tau$ is chosen so that $\sum f_i^* = S$.
This is the "water-filling" algorithm.

def _np2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length() if n > 1 else 1

def _conv(f):
    """Linear convolution f*f via FFT."""
    n = len(f)
    N = _np2(2 * n - 1)
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)[:2*n-1]

def _grad(f, w, N):
    """Compute element-wise gradient: sum_k w_k * f_{k-i} for all i."""
    n = len(f)
    wr = w[::-1]
    W = np.fft.rfft(wr, N)
    F = np.fft.rfft(f, N)
    c = np.fft.irfft(W * F, N)
    return c[n-1:2*n-1][::-1]

def _proj(v, S):
    """Project v onto {x>=0, sum(x)=S} (simplex projection)."""
    n = len(v)
    if np.max(v) <= 0:
        return np.full(n, S / n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (css - S)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, S / n)
    r = idx[-1] + 1
    theta = (css[r - 1] - S) / r
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out *= S / s
    return out

def _obj(f):
    """Objective: 2n * max(f*f) / (sum(f))^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-10:
        return np.inf
    c = _conv(f)
    return 2.0 * n * c.max() / (S * S)
