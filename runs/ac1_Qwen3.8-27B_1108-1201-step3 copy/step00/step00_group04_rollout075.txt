by working on a probability simplex representation p (sum p = 1, p >= 0), where the
scale-invariant objective is  F(p) = 2*n*max(convolve(p,p)).

For symmetric p the convolution peak equals sum p^2 >= 1/n (Cauchy-Schwarz), which forces
F >= 2 (the uniform-sequence value, the previous SOTA).  To beat 2 we must break symmetry.
We therefore (1) build several *asymmetric* structured starts (sine-phased, ramp, edge-weighted,
random sparse), and (2) run a projected subgradient method on the simplex using a

Multiple lengths and restarts are explored under a hard wall-clock budget, and the best
sequence found is returned (the evaluator is consulted as the authoritative score).

import time

# ----------------------------------------------------------------------
# Core objective + smoothed gradient (operates on p with sum(p)==1, p>=0)
# ----------------------------------------------------------------------
def exact_obj(p):
    """True (scale-invariant) objective 2*n*max(convolve(p,p))."""
    n = len(p)
    return 2.0 * n * float(np.max(np.convolve(p, p)))


def lsmax_grad(p, tau):
    """Return (smoothed objective, gradient) for 2*n*tau*logsumexp(conv(p,p)/tau)."""
    n = len(p)
    c = np.convolve(p, p)
    z = c / tau
    z = z - z.max()
    ez = np.exp(z)
    w = ez / ez.sum()
    cs = np.concatenate(([0.0], np.cumsum(w)))
    Wsum = cs[n:] - cs[:-n]                      # Wsum[m] = sum_{k=m}^{m+n-1} w[k]
    A = np.correlate(w, p, 'full')[n - 1:n - 1 + n]  # A[m] = sum_j p[j]*w[m+j]
    grad = 2.0 * n * (A + p * Wsum)
    F = 2.0 * n * tau * (np.log(ez.sum()) + (c / tau).max() - (c / tau).max())  # placeholder
    F = 2.0 * n * tau * (np.log(ez.sum()) + (c.max() / tau))   # = 2n*tau*logsumexp(c/tau)
    return F, grad

def _normalise_nonneg(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s <= 0:
        x = np.ones_like(x)
        s = 1.0
    return x / s

    i = np.arange(n)
    f = 1.0 / (2.0 * max(1, n - 1))
    x = 1.0 + eps * np.cos(2.0 * np.pi * f * i)
    if rng is not None:
        x = x + rng.random() * 1e-6          # break perfect structure
    return _normalise_nonneg(x)


def init_ramp(n, rng=None):
    x = np.linspace(0.05, 1.0, n)
    return _normalise_nonneg(x)


def init_edge(n, rng):
    """Mass pushed towards the two ends (strongly asymmetric)."""
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i).astype(float)
    x = np.exp(-d / max(1.0, n * 0.12))
    x = x + rng.random() * 1e-6
    return _normalise_nonneg(x)


def init_random(n, rng):
    x = rng.random(n) + 1e-3
    return _normalise_nonneg(x)


def init_sparse_two(n, rng):
    """Two uneven spikes: a classic asymmetric start."""
    x = np.zeros(n)
    h = rng.uniform(0.2, 0.7)
    x[0] = h
    x[n - 1] = 1.0 - h
    x += rng.random(n) * 1e-4
    return _normalise_nonneg(x)


# ----------------------------------------------------------------------
# Projected subgradient method on the simplex (momentum + backtracking)
# ----------------------------------------------------------------------
def projected_gradient(p0, tau0, tau_end, iters, deadline, rng):
    n = len(p0)
    p = _normalise_nonneg(p0.copy())
    best = p.copy()
    best_exact = exact_obj(p)
    u = np.zeros(n)
    lr = 1.0 / (2.0 * n)          # initial 1/L-ish step (L ~ 2n * max|a| scale)
    mom = 0.85
    for it in range(iters):
        if time.monotonic() > deadline:
            break
        # anneal temperature
        frac = it / max(1, iters)
        tau = tau0 * (tau_end / tau0) ** frac
        F, g = lsmax_grad(p, tau)
        g_tan = g - g.mean()                  # tangent to the {sum==1} hyperplane
        u = mom * u - lr * g_tan
        p_new = p + u
        p_new = np.clip(p_new, 0.0, None)
        s = p_new.sum()
        if s <= 0:
            p_new = p / p.sum()
        else:
            p_new = p_new / s
        # backtracking line search on the exact objective
        cand = p_new
        cand_val = exact_obj(cand)
        step_ok = cand_val <= best_exact + 1e-12
        t = 0.5
        tries = 0
        while (not step_ok) and tries < 8:
            cand = p + t * u
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0:
                cand = cand / s
            cand_val = exact_obj(cand)
            step_ok = cand_val < exact_obj(p) + 1e-12
            t *= 0.5
            tries += 1
        if step_ok:
            p = cand
        else:
            lr *= 0.7
        ev = exact_obj(p)
        if ev < best_exact - 1e-12:
            best_exact = ev
            best = p.copy()
        # occasional small random jitter to escape saddles
        if rng is not None and (it % 40 == 0) and frac < 0.9:
            p = p + rng.normal(0.0, 1e-4, n)
            p = _normalise_nonneg(p)
    return best, best_exact


# ----------------------------------------------------------------------
# Lightweight coordinate polish (incremental convolution update)
# ----------------------------------------------------------------------
def coord_polish(a, deadline, rng):
    """Refine raw non-negative a by 1-D trials using an O(n) convolve update per trial."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = float(a.sum())
    if S <= 0:
        return a
    c = np.convolve(a, a)
    cur = 2.0 * n * float(np.max(c)) / (S * S)
    best = a.copy()
    best_cur = cur
    order = list(range(n))
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        for m in order:
            old = a[m]
            if old <= 0:
                candidates = [1e-4, 1e-3, 1e-2]
            else:
                candidates = [0.0,
                              0.5 * old, 0.75 * old, 1.25 * old, 2.0 * old,
                              old * 1.1, old * 0.9]
            base = a[m]
            oldS = S
            oldcmax = float(np.max(c))
            applied = None
            bestv = None
            bestval = cur
            for v in candidates:
                v = float(v)
                dv = v - base
                if dv == 0.0:
                    continue
                # update c in place (O(n))
                km = m
                # cross terms: c[k] += 2*dv*a[k-m] for k where 0<=k-m<n, k!=2m
                lo = max(0, m)
                hi = min(len(c), m + n)
                kidx = np.arange(lo, hi)
                midx = kidx - m                       # in [0,n)
                mask = (kidx != 2 * m)
                c[kidx[mask]] += 2.0 * dv * a[midx[mask]]
                # diagonal term c[2m] += dv*(v+base)
                c[2 * m] += dv * (v + base)
                # S update
                newS = S + dv
                if newS <= 0:
                    # revert
                    kidx2 = np.arange(lo, hi)
                    c[kidx2[mask]] -= 2.0 * dv * a[midx[mask]]
                    c[2 * m] -= dv * (v + base)
                    continue
                newval = 2.0 * n * float(np.max(c)) / (newS * newS)
                # revert c fully for next trial
                c[kidx[mask]] -= 2.0 * dv * a[midx[mask]]
                c[2 * m] -= dv * (v + base)
                if newval < bestval:
                    bestval = newval
                    bestv = v
            if bestv is not None:
                # commit bestv with one clean update
                dv = bestv - base
                kidx = np.arange(max(0, m), min(len(c), m + n))
                midx = kidx - m
                mask = (kidx != 2 * m)
                c[kidx[mask]] += 2.0 * dv * a[midx[mask]]
                c[2 * m] += dv * (bestv + base)
                S += dv
                a[m] = bestv
                cur = bestval
                if cur < best_cur - 1e-12:
                    best_cur = cur
                    best = a.copy()
    if best_cur < cur:
        return best
    return a


# ----------------------------------------------------------------------
# Top-level search
# ----------------------------------------------------------------------
def propose_candidate(seed=123, budget_s=1800, **kwargs):
    """Search for a low-objective non-negative step sequence within budget_s seconds."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    total = max(3.0, float(budget_s))
    hard_deadline = t0 + total - 2.0

    def remaining():
        return max(0.0, hard_deadline - time.monotonic())

    best_seq = None
    best_score = np.inf

    # Authoritative check helper
    def record(seq):
        nonlocal best_seq, best_score
        try:
            sc = float(evaluate_sequence([float(v) for v in seq]))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(seq, dtype=np.float64).copy()
            print("  new best score = %.6f  (n=%d, t=%.1fs)" %
                  (best_score, len(best_seq), time.monotonic() - t0))

    # Seed from the previously found construction if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0:
            record(parent)
    except Exception:
        parent = None

    # Try a bank of (length, factory, hyperparams).  Allocate time greedily.
    lengths = [600, 1024, 2048, 3809, 6000]
    if parent is not None:
        # keep parent length available too
        if parent.size not in lengths:
            lengths.append(parent.size)

    inits = [
        ("uniform", lambda n, r: init_uniform(n, r)),
        ("sine",    lambda n, r: init_sine(n, 0.9, r)),
        ("sine2",   lambda n, r: init_sine(n, 1.0, r)),
        ("ramp",    lambda n, r: init_ramp(n, r)),
        ("edge",    lambda n, r: init_edge(n, r)),
        ("rand",    lambda n, r: init_random(n, r)),
        ("sparse2", lambda n, r: init_sparse_two(n, r)),
    ]

    # Order work items; cheaper (smaller n) first so we always make progress.
    items = []
    for n in sorted(lengths):
        for name, fn in inits:
            items.append((n, name, fn))

    # Rough per-item time share, smaller n gets fewer seconds (faster per iter anyway)
    it = 0
    n_items = len(items)
    while it < n_items and remaining() > 5.0:
        n, name, fn = items[it]
        share = remaining() / (n_items - it)          # even split of the rest
        # cap share for very large n, give small n a floor
        share = min(share, max(8.0, total / 12.0))
        item_deadline = time.monotonic() + share
        p0 = fn(n, rng)
        # choose tau schedule based on n (LS approx -> 0)
        tau0 = max(1e-3, 1.0 / n)
        tau_end = max(1e-6, 1e-4 / n)
        try:
            p_opt, e_opt = projected_gradient(
                p0, tau0, tau_end,
                iters=int(max(50, share * max(30.0, 4000.0 / n))),
                deadline=item_deadline, rng=rng)
            # polish in raw space
            raw = p_opt.astype(np.float64)
            if remaining() > 3.0:
                raw = coord_polish(raw, min(time.monotonic() + remaining() * 0.5, hard_deadline), rng)
            record(raw)
        except Exception as ex:        # never let one item kill the run
            print("  item (%d,%s) failed: %s" % (n, name, ex))
        it += 1

    # Final polish of the overall best if it exists and we still have time
    if best_seq is not None and remaining() > 3.0:
        try:
            polished = coord_polish(best_seq.copy(), hard_deadline, rng)
            record(polished)
        except Exception as ex:
            print("  final polish failed: %s" % ex)

    # Guarantee a valid return
    if best_seq is None:
        n = 512
        best_seq = init_uniform(n, rng)
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    print("DONE. best_score=%.6f n=%d time=%.1fs" %
          (best_score, len(best_seq), time.monotonic() - t0))
    return [float(v) for v in best_seq]
