import numpy as np
import time

try:
    from scipy.optimize import minimize

except Exception:
    minimize = None

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_convolve_same(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=np.float64)
    if la != lb:
        L = max(la, lb)
        aa = np.zeros(L, dtype=np.float64)
        bb = np.zeros(L, dtype=np.float64)
        aa[:la] = a
        bb[:lb] = b
        a = aa
        b = bb
        la = lb = L
    else:
        L = la
    N = _next_pow2(2 * L - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    out = np.fft.irfft(fa * fb, N)
    return out[:2 * L - 1]


def normalize_nonnegative(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if len(v) == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return np.full(len(v), 1.0 / float(len(v)), dtype=np.float64)
    return v / s


def f_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    L = len(z)
    if L == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    zm = float(np.max(z)) if L else 0.0
    if not np.isfinite(zm):
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    f = np.exp(z - zm)
    s = float(f.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    return f / s


def upsample_vec(v, n):
    v = normalize_nonnegative(v)
    m = len(v)
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if m == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if m == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if m == n:
        return v.copy()
    xo = np.linspace(0.0, 1.0, m)
    xn = np.linspace(0.0, 1.0, n)
    out = np.interp(xn, xo, v)
    return normalize_nonnegative(out)


def downsample_vec(v, m):
    v = normalize_nonnegative(v)
    n = len(v)
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if n == 0:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    if n == m:
        return v.copy()
    xo = np.linspace(0.0, 1.0, n)
    xn = np.linspace(0.0, 1.0, m)
    out = np.interp(xn, xo, v)
    return normalize_nonnegative(out)


def hard_score_f(f):
    f = normalize_nonnegative(f)
    L = len(f)
    if L == 0:
        return float("inf")
    c = fft_convolve_same(f, f)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return float("inf")
    return float(2.0 * L * float(np.max(c)))


def cross_corr_p_f(p, f):
    L = len(f)
    if L == 0:
        return np.array([], dtype=np.float64)
    if L == 1:
        return np.array([float(p[0] * f[0])], dtype=np.float64)
    N = _next_pow2(len(p) + L - 1)
    P = np.fft.rfft(p, N)
    Fr = np.fft.rfft(f[::-1], N)
    conv = np.fft.irfft(P * Fr, N)
    return conv[L - 1:2 * L - 1]


def smooth_score_grad_z(z, L, tau):
    f = f_from_z(z)
    c = fft_convolve_same(f, f)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return 1e100, np.zeros(L), 1e100
    cmax = float(np.max(c))
    hard = float(2.0 * L * cmax)
    if not np.isfinite(cmax):
        return 1e100, np.zeros(L), 1e100

    if (not np.isfinite(tau)) or tau <= 0.0:
        tol = max(1e-15, 1e-12 * cmax)
        idx = np.flatnonzero(c >= cmax - tol)
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        p = np.zeros_like(c)
        p[idx] = 1.0 / float(idx.size)
        smooth = cmax
    else:
        scaled = (c - cmax) / tau
        e = np.exp(scaled)
        se = float(e.sum())
        if (not np.isfinite(se)) or se <= 0.0:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
            smooth = cmax
        else:
            p = e / se
            smooth = float(cmax + tau * np.log(se))

    g = cross_corr_p_f(p, f)
    grad_f = 4.0 * float(L) * g
    mean = float(np.dot(f, grad_f))
    grad_z = f * (grad_f - mean)
    if not np.all(np.isfinite(grad_z)):
        return 1e100, np.zeros(L), hard
    return float(2.0 * L * smooth), grad_z, hard


def build_design(L):
    x = (np.arange(L) + 0.5) / float(L)
    cols = []
    x2 = x * x
    x3 = x2 * x
    x4 = x2 * x2
    x5 = x4 * x
    cols.append(x)
    cols.append(x2)
    cols.append(x3)
    cols.append(x4)
    cols.append(x5)
    cols.append(np.sin(np.pi * x))
    cols.append(np.cos(np.pi * x))
    for m in (2, 4, 6):
        ang = 2.0 * np.pi * m * x
        cols.append(np.sin(ang))
        cols.append(np.cos(ang))
    B = np.column_stack(cols)

    wedge = np.sin(np.pi * x) ** 0.5
    wedge = np.maximum(wedge, 1e-12)
    log_wedge = np.log(wedge)
    return B, log_wedge


def param_f_from_theta(theta, L, B, log_wedge):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    if B is None or log_wedge is None or L <= 0:
        return np.full(L, 1.0 / float(max(1, L)), dtype=np.float64)
    P = B.shape[1]
    if theta.size != P:
        theta = np.zeros(P, dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    z = log_wedge + B.dot(theta)
    return f_from_z(z)


def param_hard_score(theta, L, B, log_wedge):
    try:
        f = param_f_from_theta(theta, L, B, log_wedge)
        return hard_score_f(f)
    except Exception:
        return 1e100


def param_score_grad(theta, L, B, log_wedge, tau):
    try:
        f = param_f_from_theta(theta, L, B, log_wedge)
    except Exception:
        return 1e100, np.zeros(B.shape[1])

    c = fft_convolve_same(f, f)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return 1e100, np.zeros(B.shape[1])
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return 1e100, np.zeros(B.shape[1])

    if (not np.isfinite(tau)) or tau <= 0.0:
        tol = max(1e-15, 1e-12 * cmax)
        idx = np.flatnonzero(c >= cmax - tol)
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        p = np.zeros_like(c)
        p[idx] = 1.0 / float(idx.size)
        smooth = cmax
    else:
        scaled = (c - cmax) / tau
        e = np.exp(scaled)
        se = float(e.sum())
        if (not np.isfinite(se)) or se <= 0.0:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
            smooth = cmax
        else:
            p = e / se
            smooth = float(cmax + tau * np.log(se))

    g = cross_corr_p_f(p, f)
    grad_f = 4.0 * float(L) * g
    mean = float(np.dot(f, grad_f))
    grad_z = f * (grad_f - mean)
    grad_theta = B.T.dot(grad_z)
    if not np.all(np.isfinite(grad_theta)):
        return 1e100, np.zeros(B.shape[1])
    return float(2.0 * L * smooth), grad_theta


def fit_theta_to_f(f, B, log_wedge):
    try:
        f = normalize_nonnegative(f)
        target = np.log(np.maximum(f, 1e-300)) - log_wedge
        target = target - float(np.mean(target))
        Bc = B - float(np.mean(B, axis=0))
        sol = np.linalg.lstsq(Bc, target, rcond=None)[0]
        sol = np.where(np.isfinite(sol), sol, 0.0)
        return np.clip(sol, -5.0, 5.0)
    except Exception:
        return np.zeros(B.shape[1])

    deadline = start + max_time
    P = B.shape[1]
    bounds = [(-4.5, 4.5) for _ in range(P)]
    rng = np.random.default_rng(555)

    best_f = None
    best_hard = float("inf")
    starts = []

    if base_f is not None:
        starts.append(fit_theta_to_f(downsample_vec(base_f, L), B, log_wedge))
    starts.append(np.zeros(P, dtype=np.float64))

    for _ in range(26):
        starts.append(rng.normal(0.0, 1.6, P))

    if differential_evolution is not None and (deadline - start) > 35.0:
        try:
            res = differential_evolution(
                param_hard_score,
                bounds,
                args=(L, B, log_wedge),
                maxiter=35,
                popsize=12,
                tol=1e-5,
                seed=12345,
            )
            if res is not None and np.all(np.isfinite(res.x)):
                starts.append(np.clip(res.x, -4.5, 4.5))
        except Exception:
            pass

    if differential_evolution is not None and (deadline - start) > 35.0:
        try:
            res2 = differential_evolution(
                param_hard_score,
                bounds,
                args=(L, B, log_wedge),
                maxiter=25,
                popsize=10,
                tol=1e-5,
                seed=777,
            )
            if res2 is not None and np.all(np.isfinite(res2.x)):
                starts.append(np.clip(res2.x, -4.5, 4.5))
        except Exception:
            pass

    stages = (3.0, 1.0, 0.4, 0.15, 0.06, 0.02)
    opts = {"maxiter": 60, "ftol": 1e-12, "gtol": 1e-7, "maxls": 40}

    for th0 in starts:
        if time.monotonic() > deadline:
            break
        th = np.asarray(th0, dtype=np.float64).ravel()
        if th.size != P:
            th = np.zeros(P, dtype=np.float64)
        th = np.where(np.isfinite(th), th, 0.0)
        th = np.clip(th, -5.0, 5.0)

        for st in stages:
            if time.monotonic() > deadline:
                break
            tau = st / float(L)
            if minimize is not None:
                try:
                    res = minimize(
                        param_score_grad,
                        th,
                        method="L-BFGS-B",
                        args=(L, B, log_wedge, tau),
                        jac=True,
                        bounds=bounds,
                        options=opts,
                    )
                    if np.all(np.isfinite(res.x)):
                        th = res.x
                except Exception:
                    pass

            f = param_f_from_theta(th, L, B, log_wedge)
            h = hard_score_f(f)
            if np.isfinite(h) and h < best_hard:
                best_hard = h
                best_f = f

    return best_f


def run_adam_from_f(f0, max_time, rng=None):
    max_time = max(1.0, float(max_time))
    t0 = time.monotonic()
    f = normalize_nonnegative(f0)
    L = len(f)
    if L == 0:
        return f, float("inf")

    z = np.log(np.maximum(f, 1e-300))
    z = z - float(np.max(z))
    if not np.all(np.isfinite(z)):
        z = np.zeros(L, dtype=np.float64)

    m = np.zeros(L, dtype=np.float64)
    v = np.zeros(L, dtype=np.float64)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    best_f = f.copy()
    best_z = z.copy()
    best_hard = hard_score_f(f)
    no_improve = 0

    schedule_iters = 8000.0
    while time.monotonic() - t0 < max_time:
        t += 1
        frac = min(1.0, t / schedule_iters)
        tau = (4.0 / float(L)) * ((0.03 / 4.0) ** frac)
        lr = 0.7 * ((0.03 / 0.7) ** frac)

        score, grad, hard = smooth_score_grad_z(z, L, tau)
        if not np.isfinite(score) or not np.all(np.isfinite(grad)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        if np.isfinite(hard) and hard < best_hard:
            best_hard = float(hard)
            best_f = f_from_z(z).copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        max_step = 1.2 * (1.0 - frac) + 0.25
        step = np.clip(step, -max_step, max_step)
        z = np.clip(z - step, -35.0, 35.0)

        if no_improve > 1500 and rng is not None:
            ftest = best_f.copy()
            lo = max(1, L // 800)
            hi = max(lo + 1, L // 40)
            if hi <= L:
                width = int(rng.integers(lo, hi))
                width = min(width, L)
                if width >= 1:
                    start_i = int(rng.integers(0, max(1, L - width + 1)))
                    stop_i = min(L, start_i + width)
                    ftest[start_i:stop_i] *= np.exp(rng.normal(0.0, 0.35, stop_i - start_i))
                    ftest = normalize_nonnegative(ftest)
                    htest = hard_score_f(ftest)
                    if np.isfinite(htest) and htest < best_hard:
                        best_hard = float(htest)
                        best_f = ftest.copy()
                        best_z = np.log(np.maximum(best_f, 1e-300))
                        best_z -= float(np.max(best_z))
                    z = best_z + rng.normal(0.0, 0.05, L)
                    z = np.clip(z, -35.0, 35.0)
                    m.fill(0.0)
                    v.fill(0.0)
                    t = 0
                    no_improve = 0

    return best_f, best_hard


def pool_add(pool, f):
    f = normalize_nonnegative(f)
    if len(f) == 0:
        return
    h = hard_score_f(f)
    if np.isfinite(h):
        pool.append([f, float(h)])


def pool_key(item):
    return item[1]


def add_power_seeds(pool, L):
    x = (np.arange(L) + 0.5) / float(L)
    x = np.maximum(x, 1e-12)
    y = np.maximum(1.0 - x, 1e-12)
    edge = np.maximum(np.sin(np.pi * x), 1e-12) ** 0.5

    vals = (0.5, 0.8, 1.0, 1.4, 1.9, 2.6, 3.6, 5.0)
    for a in vals:
        for b in vals:
            f = (x ** a) * (y ** b)
            pool_add(pool, f)
            f2 = f * (1.0 + 0.25 * np.cos(2.0 * np.pi * x))
            f2 = np.maximum(f2, 1e-12)
            pool_add(pool, f2)

    for q in (2.0, 3.0):
        for c in (0.5, 1.0, 1.5, 2.0):
            f = edge * (1.0 + 0.6 * np.cos(2.0 * np.pi * c * np.power(x, q)))
            f = np.maximum(f, 1e-12)
            pool_add(pool, f)


def generate_structured_seed(L, rng, base, B, logw):
    x = (np.arange(L) + 0.5) / float(L)
    edge = np.maximum(np.sin(np.pi * x), 1e-12) ** 0.5
    kind = int(rng.integers(0, 10))
    f = None

    if kind == 0:
        p = B.shape[1]
        th = rng.normal(0.0, 1.6, p)
        f = param_f_from_theta(th, L, B, logw)
    elif kind == 1:
        f = edge.copy()
        for _ in range(int(rng.integers(1, 4))):
            freq = float(rng.uniform(0.1, 3.5))
            amp = float(rng.uniform(0.2, 1.1))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            f = f * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        f = np.maximum(f, 1e-8)
    elif kind == 2:
        f = edge * 1e-3
        K = int(rng.integers(1, 4))
        for _ in range(K):
            amp = float(np.exp(rng.normal(0.0, 0.7)))
            c = float(rng.uniform(0.05, 0.95))
            w = float(rng.uniform(0.03, 0.3))
            f = f + amp * np.exp(-0.5 * ((x - c) / w) ** 2)
        f = np.maximum(f, 1e-12)
    elif kind == 3:
        q = 2.0 if rng.random() < 0.5 else 3.0
        c = float(rng.uniform(0.2, 2.0))
        amp = float(rng.uniform(0.3, 1.2))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        f = edge * (1.0 + amp * np.cos(2.0 * np.pi * c * np.power(x, q) + phase))
        f = np.maximum(f, 1e-8)
    elif kind == 4:
        a = rng.normal(0.0, 1.2, 6)
        yv = a[0] + a[1] * x + a[2] * x * x + a[3] * (x ** 3) + a[4] * (x ** 4) + a[5] * (x ** 5)
        f = edge * np.exp(yv - float(np.max(yv)))
    elif kind == 5:
        if base is None or len(base) == 0:
            f = np.ones(L, dtype=np.float64)
        else:
            f = upsample_vec(base, L)
        f = f * np.exp(rng.normal(0.0, 0.5, L))
        f = np.maximum(f, 1e-12)
    elif kind == 6:
        aa = float(rng.uniform(0.2, 3.0))
        bb = float(rng.uniform(0.2, 3.0))
        f = edge * (x ** aa) * ((1.0 - x) ** bb)
        if rng.random() < 0.7:
            amp = float(rng.uniform(0.2, 0.8))
            freq = float(rng.uniform(0.5, 2.5))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            f = f * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            f = np.maximum(f, 1e-8)
    elif kind == 7:
        yv = np.zeros(L, dtype=np.float64)
        yv = yv + float(rng.uniform(-1.0, 1.0)) * x
        yv = yv + float(rng.uniform(-1.0, 1.0)) * x * x
        for mm in range(1, 5):
            wc = float(rng.uniform(-1.2, 1.2))
            ws = float(rng.uniform(-1.2, 1.2))
            ang = 2.0 * np.pi * mm * x
            yv = yv + wc * np.cos(ang) + ws * np.sin(ang)
        f = edge * np.exp(yv - float(np.max(yv)))
    elif kind == 8:
        f = edge.copy()
        f1_c = float(rng.uniform(0.2, 1.8))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        f2_c = float(rng.uniform(0.2, 1.8))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        f = f * (1.0 + 0.7 * np.cos(2.0 * np.pi * f1_c * x + p1))
        f = f * (1.0 + 0.5 * np.cos(2.0 * np.pi * f2_c * np.power(x, 2.0) + p2))
        f = np.maximum(f, 1e-8)
    else:
        f = edge * np.exp(rng.normal(0.0, 0.8, L))

    return normalize_nonnegative(f)


def safe_exact_seq(seq):
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing
    2*n*max(np.convolve(f,f))/(sum(f))**2.

    Strategy:
      1. Build a broad seed pool from deterministic skewed beta/power shapes,
         chirps, and random structured Fourier/Gaussian/PwL shapes.
      2. Run a low-dimensional parametric global refinement at medium resolution
         using differential evolution + L-BFGS-B.
      3. Run full-length Adam/mirror-descent refinement on the best candidates,
         with periodic block perturbations to escape local minima.
      4. Optionally repeat on a slightly longer upsampled sequence if time remains.
    """
    start = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    total = max(60.0, total)
    reserve = 90.0
    deadline = start + max(120.0, total - reserve)

    base = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if arr.size > 0 and np.isfinite(s) and s > 0.0:
            base = arr / s
    except Exception:
        base = None

    if base is None or len(base) == 0:
        n = 4096
        base = np.ones(n, dtype=np.float64) / float(n)
    else:
        L0 = int(len(base))
        if L0 < 1024:
            n = 2048
        elif L0 > 12000:
            n = 10000
        else:
            n = L0
        if n != L0:
            base = upsample_vec(base, n)
    n = int(len(base))

    B_full, logw_full = build_design(n)
    rng = np.random.default_rng(int(seed) + 1)

    best_f = base.copy()
    best_internal = hard_score_f(best_f)
    best_exact = safe_exact_seq(best_f.tolist())
    if not np.isfinite(best_exact):
        best_exact = 1e100

    print(f"n={n} initial_internal={best_internal:.8f} initial_exact={best_exact:.8f}")

    pool = []
    pool_add(pool, base)

    # Deterministic skewed beta/power seeds are valuable and cheap.
    add_power_seeds(pool, n)

    # Random structured seeds.
    for i in range(64):
        if time.monotonic() > deadline - 20:
            break
        f = generate_structured_seed(n, rng, base, B_full, logw_full)
        pool_add(pool, f)

    pool.sort(key=pool_key)
    if len(pool) > 0:
        print(f"seed_pool_best_internal={pool[0][1]:.8f}")
        for i in range(min(4, len(pool))):
            f = pool[i][0]
            val = safe_exact_seq(f.tolist())
            if np.isfinite(val) and val < best_exact:
                best_exact = val
                best_f = f.copy()
                best_internal = hard_score_f(f)

    # Medium-resolution parametric global search.
    param_start = time.monotonic()
    remaining_for_param = deadline - param_start
    param_time = max(90.0, min(700.0, 0.42 * remaining_for_param if remaining_for_param > 0 else 120.0))
    M_param = 512
    if n < 512:
        M_param = max(64, min(512, n))
    B_m, logw_m = build_design(M_param)
    param_f = run_param_global(M_param, B_m, logw_m, param_time, base)
    if param_f is not None:
        f_up = upsample_vec(param_f, n)
        pool_add(pool, f_up)
        h_up = hard_score_f(f_up)
        if h_up < best_internal:
            best_internal = h_up
            best_f = f_up.copy()
        val = safe_exact_seq(f_up.tolist())
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_f = f_up.copy()
    print(f"param_internal_best={best_internal:.8f} param_exact={best_exact:.8f} elapsed={time.monotonic()-start:.1f}s")

    pool.sort(key=pool_key)
    active = []
    for i in range(min(10, len(pool))):
        f = pool[i][0]
        s = pool[i][1]
        active.append([f, float(s), 0])

    full_deadline = deadline - 30.0
    last_exact_check = 0.0

    while time.monotonic() < full_deadline and active:
        bi = 0
        bs = active[0][1]
        for i in range(1, len(active)):
            if active[i][1] < bs:
                bs = active[i][1]
                bi = i

        remaining = full_deadline - time.monotonic()
        if remaining < 12.0:
            break

        if len(active) > 5:
            budget = min(120.0, remaining * 0.20)
        elif len(active) == 5:
            budget = min(180.0, remaining * 0.25)
        elif len(active) == 4:
            budget = min(260.0, remaining * 0.32)
        elif len(active) == 3:
            budget = min(380.0, remaining * 0.40)
        elif len(active) == 2:
            budget = min(600.0, remaining * 0.50)
        else:
            budget = min(remaining - 10.0, 1000.0)
        budget = max(5.0, budget)

        f0 = active[bi][0]
        f_new, h_new = run_adam_from_f(f0, budget, rng)

        if np.isfinite(h_new) and h_new < active[bi][1] - 1e-9:
            active[bi][0] = f_new
            active[bi][1] = float(h_new)
            active[bi][2] = 0
            if h_new < best_internal:
                best_internal = float(h_new)
                best_f = f_new.copy()
                now = time.monotonic()
                if h_new < best_exact - 1e-6 or now - last_exact_check > 30.0:
                    val = safe_exact_seq(f_new.tolist())
                    last_exact_check = now
                    if np.isfinite(val) and val < best_exact:
                        best_exact = float(val)
                        best_f = f_new.copy()
                print(f"full_refine_internal={best_internal:.8f} exact={best_exact:.8f} elapsed={now-start:.1f}s")
        else:
            fails = active[bi][2] + 1
            if fails >= 2:
                del active[bi]
            else:
                active[bi][2] = fails

    # Final exact validations.
    for entry in active:
        val = safe_exact_seq(entry[0].tolist())
        if np.isfinite(val) and val < best_exact:
            best_exact = float(val)
            best_f = entry[0].copy()

    val = safe_exact_seq(best_f.tolist())
    if np.isfinite(val) and val < best_exact:
        best_exact = float(val)

    # If time remains, try a modest length increase, which can reduce discretization error.
    now = time.monotonic()
    if now < deadline - 35.0 and n >= 1500:
        n2 = min(12000, max(n + 512, int(n * 1.25)))
        if n2 > n:
            f2 = upsample_vec(best_f, n2)
            h2 = hard_score_f(f2)
            if h2 < best_internal + 1e-7:
                f2, h2 = run_adam_from_f(f2, max(5.0, deadline - now - 25.0), rng)
                val2 = safe_exact_seq(f2.tolist())
                if np.isfinite(val2) and val2 < best_exact:
                    best_exact = float(val2)
                    best_f = f2.copy()
                    best_internal = h2
                    print(f"length_expansion_exact={best_exact:.8f}")

    final = normalize_nonnegative(best_f)
    final = np.clip(final, 0.0, 1000.0)
    print(f"final_internal={hard_score_f(final):.8f} final_exact={best_exact:.8f}")
    return [float(x) for x in final]
