def spectral_flatten(a, alpha=0.5, iterations=100):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    for _ in range(iterations):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Flatten: new_mag = mag^alpha
        new_mag = mag ** alpha
        # Remove DC (or keep it for normalization)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, N)[:n]

        s = np.sum(a_new)
        if s < 1e-10:

        a_new = a_new / s
