"""
Coordinate descent with incremental convolution updates and random restarts
to minimize 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses O(n) per-candidate evaluation via in-place convolution tracking.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def convolve_self(f):
    """Compute f * f using numpy."""
    return np.convolve(f, f)


def score_from_conv(conv, total, n):
    """Compute the objective from convolution and total sum."""
    if total < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (total * total)


def cd_pass(f, n, order, conv, total):
    """
    One pass of coordinate descent.
    f: current array (modified in place)
    n: length
    order: permutation of indices
    conv: current convolution (modified in place)
    total: current sum of f
    Returns updated total.
    """
    fmax = f.max()
    for i in order:
        old = f[i]
        # Build candidate list
        if old > 1e-12:
            cands = [0.0,
                     old * 0.1, old * 0.25, old * 0.5, old * 0.75,
                     old * 1.25, old * 1.5, old * 2.0, old * 4.0]
        else:
            cands = [0.0, 0.001, 0.01, 0.1, 1.0]
        # Add some absolute-scale candidates
        if fmax > 0:
            cands.append(fmax * 0.5)
            cands.append(fmax)

        best_v = old
        best_sc = score_from_conv(conv, total, n)

        k_start = i
        k_end = min(i + n, 2 * n - 1)
        seg_len = k_end - k_start

        # Precompute unchanged max (positions outside [k_start, k_end))
        left_max = conv[:k_start].max() if k_start > 0 else 0.0
        right_max = conv[k_end:].max() if k_end < (2 * n - 1) else 0.0
        unch_max = max(left_max, right_max)

        base_vals = conv[k_start:k_end].copy()

        for v in cands:
            if v < 0:
                v = 0.0
            delta = v - old
            if abs(delta) < 1e-15:
                continue
            new_total = total + delta
            if new_total < 1e-15:
                continue

            # Compute changed values
            changes = 2.0 * delta * f[:seg_len]
            # Fix the k=2i position (index 2i - k_start = i in the segment)
            if i < seg_len:
                changes[i] = 2.0 * old * delta + delta * delta

            new_vals = base_vals + changes
            changed_max = new_vals.max()
            overall_max = changed_max if changed_max > unch_max else unch_max
            sc = 2.0 * n * overall_max / (new_total * new_total)

            if sc < best_sc:
                best_sc = sc
                best_v = v

        # Apply best
        if best_v != old:
            delta = best_v - old
            f[i] = best_v
            total += delta
            changes = 2.0 * delta * f[:seg_len]
            if i < seg_len:
                changes[i] = 2.0 * old * delta + delta * delta
            conv[k_start:k_end] += changes

    return total


def lp_update(f, n, M, rows, cols):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    vals = np.tile(f, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_neg = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_neg, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True,
                                               'time_limit': 60})
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None
