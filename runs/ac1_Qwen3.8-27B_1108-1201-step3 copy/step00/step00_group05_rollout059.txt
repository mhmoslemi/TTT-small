Hybrid subgradient + simulated annealing optimizer for minimizing

def fast_full_conv(x):
    """Full linear convolution via FFT for speed."""
    n = len(x)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    result = np.fft.irfft(X * X, fft_size)
    return result[:size]


def compute_score(x):
    """Compute 2*n*max(conv(x,x))/sum(x)^2 directly."""
    n = len(x)
    s = x.sum()
    if s < 0.01:
        return np.inf
    b = fast_full_conv(x)
    return 2.0 * n * float(b.max()) / (s * s)

def subgradient_direction(seq, b, n, top_k=8):
    """Compute weighted subgradient from top-k lags of the convolution."""
    g = np.zeros(n)
    current_max = b.max()
    if current_max <= 0:
        return g
    # Get top lags
    top_indices = np.argpartition(-b, min(top_k, 2 * n - 2))[:top_k]
    top_indices = top_indices[np.argsort(-b[top_indices])]
    for m in top_indices:
        w = (b[m] / current_max) ** 25.0
        if w < 0.01:
            continue
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] += w * seq[m - idx]
    return g
