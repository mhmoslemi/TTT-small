Combines spectral flattening (frequency-domain), LP-guided direction finding,
and adaptive subgradient descent for peak reduction.

def score_fn(a):
    """Compute the objective directly (fast, no Python overhead)."""
    n = a.size
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def spectral_flatten(a, num_iters=50, gamma=0.3):
    """Iteratively flatten the magnitude spectrum, projecting to non-negative."""
    n = a.size
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S
    for _ in range(num_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        # Target: constant magnitude equal to DC/1 (flat)
        # Use a fraction of DC as target to avoid being too aggressive
        target = mag[0]  # DC component = sum = 1
        # Soft update on non-DC bins
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - gamma) * mag[1:] + gamma * target
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = A[0]  # Preserve DC
        a_new = np.fft.irfft(A_new, n=n)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-15:
            break
        a_new = a_new / S_new
        a = a_new
    return a

def lp_get_direction(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns the optimal b or None if LP fails.
    """
    n = a.size
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    # A[k, i] = a[k-i] if 0 <= k-i < n
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])

    c_obj = -np.ones(n)  # Minimize -sum(b) = maximize sum(b)
    b_ub = np.full(num_constraints, M)
    
    res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success:
        return res.x
    return None

def subgradient_step(a, peak_fraction=0.85):
    """
    One step of subgradient descent targeting near-peak positions of a*a.
    Returns updated a (normalized to sum 1).
    """
    n = a.size
    c = np.convolve(a, a)
    M = c.max()
    if M <= 0:
        return a
    
    # Find near-peak positions
    threshold = peak_fraction * M
    peak_indices = np.where(c >= threshold)[0]
    
    # Compute combined subgradient: g[i] = 2 * sum_{k in peaks} a[k-i]
    g = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # g[lo:hi+1] += a[k-hi : k-lo+1][::-1]
        seg = a[k - hi : k - lo + 1][::-1]
        g[lo:hi + 1] += seg
    g *= 2.0
    
    # Center to preserve sum
    g -= g.mean()
    
    # Compute max safe step
    pos = g > 1e-15
    if pos.any():
        alpha_max = (a[pos] / g[pos]).min()
    else:
        alpha_max = 1.0
    alpha_max = max(alpha_max, 0.0)
    
    # Take a fraction of max step
    alpha = 0.5 * alpha_max
    a_new = a - alpha * g
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new < 1e-15:
        return a
    a_new = a_new / S_new
    return a_new

Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines spectral flattening (frequency-domain projection), subgradient
descent on convolution peaks, and adaptive random search.

import numpy as np

def score_fn(a):
    """Compute the objective directly."""
    n = a.size
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def spectral_flatten(a, num_iters=50, gamma=0.3):
    """Iteratively flatten magnitude spectrum, project to non-negative, renormalize."""
    n = a.size
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S
    for _ in range(num_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        target = mag[0]
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - gamma) * mag[1:] + gamma * target
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = A[0]
        a_new = np.fft.irfft(A_new, n=n)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-15:
            break
        a = a_new / S_new
    return a


def subgradient_step(a, peak_fraction=0.85):

    if M <= 0:

    threshold = peak_fraction * M

    g = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        seg = a[k - hi:k - lo + 1][::-1]
        g[lo:hi + 1] += seg
    g *= 2.0
    g -= g.mean()
    pos = g > 1e-15

    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new < 1e-15:
        return a.copy()
    return a_new / S_new

    """Normalize to sum 1, non-negative."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:

    return x / s
