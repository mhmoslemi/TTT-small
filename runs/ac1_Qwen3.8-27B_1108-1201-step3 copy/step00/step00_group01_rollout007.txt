```python
def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # A_ub has shape (2n-1, n), A_ub[k, j] = a[k-j] if 0 <= k-j < n
    # b_ub = M * ones(2n-1)
    # c = -ones(n) (minimize -sum(g))
    
    # For memory efficiency, use a sparse matrix
    from scipy.sparse import coo_matrix
    
    # A_ub[k, j] = a[k-j] for 0 <= k-j < n
    # Non-zeros: for k = 0,...,2n-2, j = max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A_ub = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    b_ub = M * np.ones(2*n-1)
    c = -np.ones(n)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```
