def compute_convolution(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    A = fft(a, size)
    C = ifft(A * A)
    return C.real[:2*n-1]

def compute_subgradient(c, a, M_indices):

    n = len(a)
    size = next_fast_len(2 * n - 1)
    # Subgradient at position k is sum of a[j-k] for j in M_indices

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def fast_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    A = fft(a, size)
    C = ifft(A * A).real[:2*n-1]
    M = C.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def subgradient_descent(a_init, rng, deadline, max_iters=100000):
    """Run subgradient descent to minimize J."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    a = a / S  # Normalize to sum 1
    
    best_a = a.copy()
    best_score = fast_score(a)
    
    # Adaptive step size
    eta = 0.01 / n  # Initial step size (scaled by n)
    patience = 0
    max_patience = 500
    
    size = next_fast_len(2 * n - 1)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        A = fft(a, size)
        C = ifft(A * A).real[:2*n-1]
        M = C.max()
        S = a.sum()
        
        # Find max indices
        M_indices = np.where(C >= M * (1 - 1e-10))[0]
        
        # Compute subgradient
        # g_k = sum_{j in M_indices} a[(j-k) % ... ] 
        # Actually g_k = sum_{j in M_indices} a[j-k] where a[m]=0 for m<0 or m>=n
        # This is the cross-correlation of indicator(M_indices) with a
        
        # Efficient computation:
        # g = ifft(fft(conj(fft(indicator))) * fft(a)) ... no
        # Actually: g_k = sum_j indicator[j] * a[j-k]
        # This is (indicator * a_flipped)[k + n - 1] in convolution
        # Or equivalently, it's the cross-correlation
        
        # Let's use: g = ifft(fft(indicator, size) * conj(fft(a, size))).real[:n]
        # Wait, that gives the cross-correlation which is what we want.
        
        indicator = np.zeros(size)
        indicator[M_indices] = 1.0
        # Cross-correlation: (indicator ⋆ a)[k] = sum_j indicator[j] * conj(a[j-k])
        # For real a: = sum_j indicator[j] * a[j-k]
        # = sum_j indicator[j] * a[j-k] which is what we want!
        
        FA = fft(a, size)
        FI = fft(indicator, size)
        G = ifft(FI * conj(FA)).real[:n]
        
        # Gradient of J = 2n*M/S^2 with respect to a_k:
        # dJ/da_k = 2n * G_k / S^2 - 2n*M * 2*a_k / S^3
        #         = (2n/S^2) * (G_k - 2*M*a_k/S)
        
        grad = (2.0 * n / (S * S)) * (G - 2.0 * M * a / S)
        
        # Subgradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize (maintain sum = 1)
        S_new = a_new.sum()
        if S_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S_new
        
        # Check score
        score = fast_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        # Adaptive step size
        if patience > max_patience:
            eta *= 0.5
            patience = 0
        
        # Occasionally increase step size to escape local minima
        if it % 10000 == 5000:
            eta *= 1.2
        
        # Add small noise to escape local minima
        if it % 5000 == 2500:
            a = a + rng.normal(0, 1e-4/n, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score

def fast_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 using FFT."""

    size = next_fast_len(2 * n - 1)
    A = fft(a, size)
    C = ifft(A * A).real[:2*n-1]
    M = C.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def small_n_optimize(n, rng, time_budget):

    def objective(a):

        s = a.sum()

        A = fft(a, sff)
        C = ifft(A*A).real[:2*n-1]
        return 2.0 * n * C.max() / 1.0  # S=1 after normalization
    
    best = np.inf
    best_a = None
    
    # Multiple random starts
    n_starts = min(50, max(10, int(time_budget * 10)))
    for i in range(n_starts):
        a0 = rng.random(n) + 0.1

            break
