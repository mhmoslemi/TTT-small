    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

   a. Just increase (let the sum grow), or
   b. Also decrease some coordinates.

If we just increase, the sum grows, and since the objective is scale-invariant, we can renormalize. The key is that the "shape" changes in a way that reduces the peak.

    n = len(f)

def multi_peak_subgradient(f, K=5):
    """Compute a weighted subgradient from the top K peaks."""
    conv = np.convolve(f, f)
    top_K = np.argsort(conv)[::-1][:K]
    weights = np.exp(conv[top_K] / 0.1)  # Softmax weights
    weights /= np.sum(weights)
    
    g = np.zeros(len(f))
    n = len(f)
    for w, k in zip(weights, top_K):
        js = np.arange(n)
        valid = (k - js >= 0) & (k - js < n)
        g[valid] += w * 2.0 * f[k - js[valid]]
    return g, conv
