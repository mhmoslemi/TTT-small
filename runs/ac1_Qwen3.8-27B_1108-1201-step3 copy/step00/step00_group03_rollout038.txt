import time
from numpy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a, b=None):
    """Compute linear convolution of a with b (or a with itself if b is None)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = fft(a, size)
    B = fft(b, size)
    C = ifft(A * B).real
    return C[:n+m-1]

def solve_lp(f, M, n):
    """Solve: max sum(b) subject to f*b <= M (all lags), b >= 0."""
    # Variables: b[0], ..., b[n-1]
    # Objective: max sum(b) => min -sum(b)
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_j f[j] * b[k-j]
    # For a given k, b[i] appears with coefficient f[k-i] (if 0 <= k-i <= n-1)
    # So the constraint matrix A_ub has rows for k and columns for i:
    # A_ub[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0
    
    # Build the constraint matrix
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    return None

2n * max(convolve(a,a)) / (sum(a))^2 for non-negative sequences.

Strategy:

import numpy as np
import time
from numpy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Compute linear convolution a*b (or a*a) using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1 << (na + nb - 2).bit_length()
    if size < na + nb - 1:
        size <<= 1
    A = fft(a, size)
    B = fft(b, size)
    return ifft(A * B).real[:na + nb - 1]


def build_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        rows_list.extend([k] * count)
        cols_list.extend(range(i_start, i_end + 1))
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_maxmin(a, M, n, rows, cols):
    """Solve: max t s.t. b[i] >= t, (a*b)[k] <= M for all k, b >= 0, t >= 0."""
    # Variables: [b_0, ..., b_{n-1}, t]
    nvar = n + 1
    # Objective: minimize -t => c = [0,...,0, -1]
    c_obj = np.zeros(nvar)
    c_obj[n] = -1.0

    # Constraints:
    # 1) (a*b)[k] <= M for k=0..2n-2
    vals = a[rows - cols]

    A_rows = rows
    A_cols = cols
    A_vals = vals
    # We need to add the t column (all zeros for these constraints)

    all_rows = np.concatenate([A_rows, np.full(n, n, dtype=np.int32)])
    all_cols = np.concatenate([A_cols, np.arange(n, dtype=np.int32)])
    all_vals = np.concatenate([A_vals, np.full(n, -1.0)])

    n_cons = (2 * n - 1) + n  # convolution constraints + b[i] >= t constraints
    A_sparse = csr_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1 + n, nvar))

    b_ub = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': 5, 'presolve': True})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None


def gradient_step(a, r, lr, n):
    """
    One subgradient step for minimizing max r[k].
    Returns updated a.
    """
    max_r = np.max(r)
    # Active lags: within 0.1% of max
    thresh = max_r * 0.999
    active = np.where(r >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(r))])

    # Compute average subgradient over active lags
    # For active lag k: grad_k[i] = 2*a[k-i] if 0 <= k-i <= n-1
    g = np.zeros(n)
    for k in active:
        if k < n:
            g[:k + 1] += 2.0 * a[k:-1:-1]
        else:
            g[k - n + 1:] += 2.0 * a[n - 1:k - n:-1]
    g /= len(active)

    # Update
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a
    return a_new / s


def optimize_lp_phase(a, rows, cols, n, time_limit, rng):
    """Run LP-based iterative optimization."""
    t_step = 0.4
    M = np.max(fast_convolve(a))
    best_M = M
    best_a = a.copy()
    stall_count = 0

    for it in range(20000):
        if time.monotonic() > time_limit:
            break

        b, t_val = solve_lp_maxmin(a, M, n, rows, cols)
        if b is None:
            break

        sum_b = np.sum(b)
        if sum_b < 1e-15:
            break

        # Normalize b to match sum of a
        b = b * (np.sum(a) / sum_b)

        # Update
        a_new = (1 - t_step) * a + t_step * b
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s

        # Evaluate
        r = fast_convolve(a_new)
        M_new = np.max(r)

        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Accept or adapt
        if M_new <= M * 1.005:
            a = a_new
            M = M_new
        else:
            t_step = max(0.05, t_step * 0.8)

        # If stalled, add perturbation
        if stall_count > 200:
            noise = rng.normal(0, 0.05, n)
            a = a + noise * np.std(a)
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)
            M = np.max(fast_convolve(a))
            stall_count = 0
            t_step = 0.3

        if it % 500 == 0:
            print(f"    LP it={it}: M={M:.8f}, best_M={best_M:.8f}, t={t_step:.3f}")

    return best_a, best_M


def optimize_gradient_phase(a, n, time_limit, rng):
    """Run gradient descent refinement."""
    best_M = np.max(fast_convolve(a))
    best_a = a.copy()
    lr = 0.4
    momentum = 0.85
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    t_nest = 1.0
    consecutive_bad = 0

    for it in range(500000):
        if time.monotonic() > time_limit:
            break

        # Nesterov step: compute gradient at y
        r = fast_convolve(y)
        max_r = np.max(r)

        if max_r < best_M:
            best_M = max_r
            best_a = y.copy()
            consecutive_bad = 0

        thresh = max_r * 0.999
        active = np.where(r >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(r))])

        g = np.zeros(n)
        for k in active:
            if k < n:
                g[:k + 1] += 2.0 * y[k:-1:-1]
            else:
                g[k - n + 1:] += 2.0 * y[n - 1:k - n:-1]
        g /= len(active)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new

        v = momentum * v + (1 - momentum) * g
        y = y - lr * v / t_new
        y = np.maximum(y, 0.0)
        sy = np.sum(y)
        if sy < 1e-15:
            y = a.copy()
            break
        y = y / sy

        a = y + beta * (y - a)
        a = np.maximum(a, 0.0)
        sa = np.sum(a)
        if sa < 1e-15:
            a = y
        else:
            a = a / sa
        t_nest = t_new

        # Adaptive learning rate
        consecutive_bad += 1
        if consecutive_bad > 5000:
            lr = max(0.01, lr * 0.9)
            consecutive_bad = 0
        elif consecutive_bad == 0:
            lr = min(1.0, lr * 1.01)

        # Periodic perturbation to escape local minima
        if it > 0 and it % 20000 == 0:
            noise = rng.normal(0, 0.02, n)
            a = a + noise * np.std(a)
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0

        if it % 10000 == 0 and it > 0:
            print(f"    Grad it={it}: M={max_r:.8f}, best_M={best_M:.8f}, lr={lr:.4f}")

    return best_a, best_M
