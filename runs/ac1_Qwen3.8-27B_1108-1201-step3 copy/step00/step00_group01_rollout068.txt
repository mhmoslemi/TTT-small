"""
Step-function search that minimizes  n-based peak/self-energy ratio
    f(a) = 2*n*max(conv(a,a)) / (sum(a))^2 .
Because f is scale-invariant we optimize on the probability simplex
(sum a = 1, a >= 0) where the objective is simply 2*n*max(conv(a,a)).

Key idea (the real improvement over the previous random-block search):
a *smooth* maximizer (log-sum-exp) of the convolution is used so that the
gradient is NON-ZERO even at the all-ones stationary point (the hard max has
zero subgradient there and is therefore stuck at score 2).  The soft gradient
  d softmax(b)/da[j] = 2 * sum_i a[i]*p[i+j] ,  p ~ softmax(tau*b)
pushes mass out of the flat top of the triangular autocorrelation, letting us
escape score 2 and approach the ~1.5 target.  A projected gradient step onto
the simplex is used with annealed temperature/step, several structured restarts,
and a final distributed local (block) polish.  The authoritative evaluator is
called periodically to keep track of the true best.
"""

import time

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _proj_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = np.nonzero(u * idx > (css - 1.0))[0]
    if rho.size == 0:
        rho = np.array([n - 1])
    r = int(rho[-1])
    theta = (css[r] - 1.0) / (r + 1.0)
    w = v - theta
    out = w.copy()
    out[out < 0.0] = 0.0
    return out


def _softmax_weights(b, tau):
    m = b.max()
    e = np.exp(tau * (b - m))
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(b, 1.0 / b.size)
    return e / s


def _soft_grad(a, p, N):
    """Return g[j] = 2*sum_i a[i]*p[i+j]  for j=0..n-1  (FFT cross-corr)."""
    n = a.shape[0]
    lp = p.size
    f = p[::-1]
    fa = np.fft.rfft(a, N)
    ff = np.fft.rfft(f, N)
    c = np.fft.irfft(fa * ff, N)
    c = np.ascontiguousarray(c[: (n + lp - 1)])
    hi = lp - 1          # == 2n-2
    lo = lp - 1 - (n - 1)  # == n-2  (first index of corr, before reverse)
    g = c[lo: hi + 1][::-1]
    return 2.0 * g


def _local_polish(parent, rng, deadline, eval_every, score_of):
    """Distributed block multiplicative perturbations from parent."""
    best = parent.copy()
    best_score = score_of(best)
    n = best.shape[0]
    it = 0
    while time.monotonic() < deadline:
        cand = best.copy()
        width = int(rng.integers(max(1, n // 400), max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.35)))
        if rng.random() < 0.4:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand / s
        it += 1
        if it % eval_every == 0:
            sc = score_of(cand)
        else:
            b = np.convolve(cand, cand)
            sc = 2.0 * n * float(b.max()) / 1.0  # sum==1
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


def _run_soft_descent(start, n, budget_s, rng, eval_every, score_of):
    """Annealed projected soft-gradient descent on the simplex."""
    deadline = time.monotonic() + budget_s
    a = start.copy()
    s0 = a.sum()
    if s0 <= 0.0:
        a[:] = 1.0 / n
        s0 = a.sum()
    a = a / s0
    N = _next_pow2(3 * n - 3)

    tau_lo, tau_hi = 4.0, 50.0
    lr_lo, lr_hi = 1.0e-5, 4.0e-4
    noise_lo, noise_hi = 0.0, 2.0e-4

    best = a.copy()
    best_score = score_of(a)
    n_iter = 0
    stall = 0
    last_log = 0.0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        maxb = float(b.max())
        cur = 2.0 * n * maxb  # sum(a)==1

        # anneal temperature and step with iteration count
        t = min(1.0, n_iter / 6000.0)
        tau = tau_lo + (tau_hi - tau_lo) * (t ** 1.5)
        lr = lr_hi + (lr_lo - lr_hi) * t
        noise = noise_hi * (1.0 - t)

        p = _softmax_weights(b, tau)
        g = _soft_grad(a, p, N)

        a_new = _proj_simplex(a - lr * g)
        # temperature-driven random redistribution (helps escape flat tops)
        if noise > 0.0 and rng.random() < 0.6:
            a_new = a_new + noise * (rng.standard_normal(n) + 1.0)
            a_new = np.clip(a_new, 0.0, None)
            ss = a_new.sum()
            if ss <= 0.0:
                a_new = a
            else:
                a_new = a_new / ss
        a = a_new
        n_iter += 1

        if n_iter % eval_every == 0:
            sc = score_of(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()
                stall = 0
            else:
                stall += 1
            # occasional reheat to escape a local valley
            if stall > 40 and noise > 0.0:
                a = 0.5 * a + 0.5 * (rng.random(n) + 1e-6)
                ss = a.sum()
                a = np.clip(a, 0.0, None)
                a = a / max(ss, 1e-9)
                stall = 0
            now = time.monotonic()
            if now - last_log > 30.0:
                print("  soft-descent it=%d score=%.6f best=%.6f tau=%.1f lr=%.1e"
                      % (n_iter, cur, best_score, tau, lr), flush=True)
                last_log = now

    return best, best_score, n_iter

    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s < 0.01:
        out = np.ones(n)
        s = out.sum()
    out = out / s * math.sqrt(2.0 * n)  # nice scale, still any value is valid
    print("FINAL best_score=%.6f  elapsed=%.1fs"
          % (best_score, time.monotonic() - t0), flush=True)

    return [float(x) for x in out]
