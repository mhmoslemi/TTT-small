            a += rng.normal(0, 0.1, n)

            c = np.convolve(a, a)

            weights = np.exp(c[top_k] / epsilon)
            weights /= weights.sum()

    best_score = float(evaluate_sequence(best.tolist()))
    
    def normalize(a):
        s = a.sum()
        if s < 1e-15:
            return np.ones(n)
        return a / s

        """Compute full linear convolution using FFT."""
        A = fft(a, L)
        C = ifft(A * A)
        return C.real[:2*n-1]
    
    def compute_objective(a):
        s = a.sum()
        if s < 1e-15:
            return np.inf
        c = fast_convolve(a)
        return 2.0 * n * c.max() / (s * s)
    
    def smooth_max_gradient(a, c, epsilon, k_top=20):
        """Compute the gradient of the smooth max of c."""
        # Find top k_top indices
        k_indices = np.argsort(c)[-k_top:]
        # Compute softmax weights
        vals = c[k_indices] / epsilon
        weights = np.exp(vals - vals.max())
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k, w in zip(k_indices, weights):
            # grad[i] += 2 * w * a[k-i] for 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # a[k-lo], a[k-lo-1], ..., a[k-hi] -> indices k-hi to k-lo in a
                # We need a[k-i] for i from lo to hi
                # a[k-lo], a[k-lo-1], ..., a[k-hi]
                # This is a[k-hi : k-lo+1] reversed
                segment = a[k-hi:k-lo+1]
                grad[lo:hi+1] += 2.0 * w * segment[::-1]
        
        return grad
    
    # Try different initializations
    inits = [
        "parent",  # start from the given sequence
        "uniform",  # all ones
        "parabolic",  # U-shaped
