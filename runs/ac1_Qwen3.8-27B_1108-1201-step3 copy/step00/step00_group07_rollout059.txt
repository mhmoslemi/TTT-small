import time

def fast_self_convolve(f):
    """Compute f*f (full convolution) using FFT."""
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)
    return conv[:2*n-1]

def compute_score_fast(f):
    """Compute score using FFT-based convolution."""
    n = len(f)
    h = fast_self_convolve(f)
    max_h = np.max(h)
    sum_f = np.sum(f)
    if sum_f < 1e-10:
        return np.inf
    return float(2 * n * max_h / (sum_f ** 2))

def mirror_descent_steps(f, eta_schedule, n_total_steps):
    """Perform mirror descent steps."""
    f = f.copy()
    f = np.clip(f, 0, None)
    if f.sum() < 1e-15:
        return f
    f /= f.sum()
    n = len(f)
    
    for step in range(n_total_steps):
        h = fast_self_convolve(f)
        k_star = int(np.argmax(h))
        
        # Subgradient
        s = np.zeros(n)
        i_lo = max(0, k_star - (n-1))
        i_hi = min(n-1, k_star)
        s[i_lo:i_hi+1] = f[k_star - i_lo : k_star - i_hi - 1 : -1]
        
        # Scale subgradient
        max_s = np.max(s)
        if max_s > 0:
            eta = eta_schedule(step)
            f = f * np.exp(-eta * s / max_s)
        
        f = np.clip(f, 1e-30, None)
        f /= f.sum()
    
    return f

But for practical purposes, using just one maximum point should still give a descent direction (since it's a subgradient of the convex function max(f*f)... wait, max(f*f) is not convex in f!

Actually, f*f is a quadratic form in f, and max of quadratic forms is generally non-convex. So the "subgradient" approach is not theoretically justified, but it might still work heuristically.

Let me think of this differently. The function max(f*f) is the maximum of 2n-1 quadratic functions, each of the form (f*f)[k] = sum_{i,j: i+j=k} f[i]*f[j]. Each of these is a convex function of f (since it's a quadratic form with a PSD matrix... wait, is it?).

(f*f)[k] = sum_i f[i] * f[k-i] (where the sum is over valid i). This can be written as f^T A_k f where A_k is a symmetric matrix with A_k[i,j] = 1/2 if i+j=k (and i≠j) and A_k[i,i] = 1 if 2i=k.

Is A_k PSD? Let's see: f^T A_k f = sum_i f[i]f[k-i] = (f shifted) · f. This is the inner product of f and a shifted version of f, which is the cross-correlation at a specific lag. This is NOT necessarily non-negative! For example, if f has both positive and negative entries, it could be negative. But since we constrained f ≥ 0, it IS non-negative.

So for f ≥ 0, each (f*f)[k] ≥ 0, and the maximum is well-defined. But the problem of minimizing max(f*f) over f ≥ 0, sum(f) = 1 is non-convex.

The mirror descent approach I described is a heuristic that uses one "active" constraint at a time. It's similar to the " Cutting Plane" or "Subgradient" method for non-convex problems, which can work well in practice.

Let me also add a "smoothing" step: instead of using just the argmax, I can use a smooth approximation. The "tropical" or "log-sum-exp" function is a smooth approximation to the max:
max(x_1, ..., x_m) ≈ (1/τ) * log(sum exp(τ * x_i))

For the subgradient of this smooth approximation:
∂/∂f[i] [ (1/τ) * log(sum_k exp(τ * (f*f)[k])) ]
= sum_k (exp(τ*(f*f)[k]) / sum_j exp(τ*(f*f)[j])) * ∂(f*f)[k]/∂f[i]
= sum_k w_k * f[k-i] (where w_k = softmax(τ*(f*f)))

This is a weighted average of the "subgradients" from each constraint, with weights proportional to how close each constraint is to being active. This is much smoother and should give better convergence.

def mw_lp_solve(f, M, n_iters=1000, eta=0.1):
    """
    Use multiplicative weights to approximate the LP:
    max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    
    Returns (best_p, best_max_violation) where p is a probability vector.
    The LP optimum value is approximately M / best_max_violation.
    """
    n = len(f)
    n_k = 2 * n - 1
    
    # Q: distribution over constraints (k indices)
    Q = np.ones(n_k) / n_k
    
    best_max_stats = np.inf
    best_p = None
    
    for it in range(n_iters):
        # Compute prefix sums of Q for efficient (A^T Q) computation
        Q_prefix = np.cumsum(Q)
        Q_prefix = np.concatenate([[0], Q_prefix])
        
        # (A^T Q)[i] = f[i] * sum_{k=i}^{i+n-1} Q[k]
        # = f[i] * (Q_prefix[min(i+n, n_k)] - Q_prefix[i])
        k_hi = np.minimum(np.arange(n) + n, n_k)
        k_lo = np.arange(n)
        Q_sum = Q_prefix[k_hi] - Q_prefix[k_lo]
        ATQ = f * Q_sum
        
        # Best response: one-hot at argmax
        i_star = int(np.argmax(ATQ))
        p = np.zeros(n)
        p[i_star] = 1.0
        
        # Compute (A p) = f shifted by i_star
        Ap = np.zeros(n_k)
        Ap[i_star:i_star+n] = f
        
        # Track best
        max_ap = np.max(Ap)
        if max_ap < best_max_stats:
            best_max_stats = max_ap
            best_p = p.copy()

        # For numerical stability:
        log_ratios = eta * Ap / M
        log_ratios -= np.max(log_ratios)
        Q_new = Q * np.exp(log_ratios)
        Q_new /= Q_new.sum()
        Q = Q_new
    
    return best_p, best_max_stats

        # i* = argmax_i f[i] * sum_{k=i}^{i+n-1} Q[k]
        Q_prefix = np.concatenate([[0], np.cumsum(Q)])

        i_star = int(np.argmax(ATQ))

        Pf = np.convolve(P, f)  # full convolution, length 2n-1

        payoff_sum += payoff
        
        # Accumulate
        P_avg[i_star] += 1.0 / n_iters
        
        # Update P
        # P[i] ∝ P[i] * exp(-eta * f[i] * Q_sum[i])
        log_update = -eta * f * Q_sum
        log_update -= np.max(log_update)
        P_new = P * np.exp(log_update)
        P_new /= P_new.sum()

        # Q[k] ∝ Q[k] * exp(eta * Pf[k])
        log_update_q = eta * Pf
        log_update_q -= np.max(log_update_q)
        Q_new = Q * np.exp(log_update_q)
        Q_new /= Q_new.sum()
        Q = Q_new

def fast_self_convolve(f):
    """Compute full self-convolution using FFT."""
    n = len(f)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    conv = np.fft.irfft(F * F, size)
    return conv[:2*n-1]

def fast_convolve(a, b):
    """Compute full convolution of a and b using FFT."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)
    return conv[:n_a + n_b - 1]

def fast_self_convolve(f):

        g = mw_game_solve(f, n_iters=1500)

        f_new = np.clip(f_new, 1e-30, None)
        f_new /= f_new.sum()

        score = compute_score(f_new)
        if score < best_score - 1e-9:
            best_f = f_new.copy()
            best_score = score
            stall_count = 0
            if stall_count == 0 or int(stall_count) % 10 == 0:
                print(f"LP: score={best_score:.6f}, t_lr={t_lr:.4f}")
        else:
            stall_count += 1

        f = f_new

        if stall_count > 80:
            t_lr = max(0.01, t_lr * 0.7)
            stall_count = 0
            # Restart from best
            if rng.random() < 0.3:
                f = best_f * np.exp(rng.normal(0, 0.1, n))
                f = np.clip(f, 1e-15, None)
                f /= f.sum()

    print(f"After Phase 1: score={best_score:.6f}, time={time.monotonic()-t_start:.1f}s")

    # Phase 2: Smooth mirror descent refinement
    phase2_deadline = t_start + budget_s * 0.75
    f = best_f.copy()

    tau_schedule = [10.0, 30.0, 100.0, 300.0]
    mi = 0
    while time.monotonic() < phase2_deadline:
        tau = tau_schedule[mi % len(tau_schedule)]
        eta0 = 2.0
        f = smooth_mirror_descent(f, 300, tau, eta0, 0.995)
        score = compute_score(f)
        if score < best_score - 1e-9:
            best_f = f.copy()
            best_score = score
            print(f"MD tau={tau:.0f}: score={best_score:.6f}")
        mi += 1

        # Occasionally perturb
        if mi % 5 == 0:
            f = best_f.copy()
            width = rng.integers(max(1, n // 100), max(2, n // 15))
            start = int(rng.integers(0, n - width))
            f[start:start + width] *= np.exp(rng.normal(0, 0.25, width))
            f = np.clip(f, 1e-15, None)
            f /= f.sum()

    print(f"After Phase 2: score={best_score:.6f}, time={time.monotonic()-t_start:.1f}s")

    # Phase 3: Fine-grained local search
    while time.monotonic() < deadline:
        f = best_f.copy()
        # Small random perturbation
        n_pert = rng.integers(1, max(2, n // 50))
        for _ in range(n_pert):
            idx = int(rng.integers(0, n))
            f[idx] *= np.exp(rng.normal(0, 0.1))
        f = np.clip(f, 1e-15, None)
        f /= f.sum()

        f = smooth_mirror_descent(f, 100, 50.0, 1.0, 0.98)
        score = compute_score(f)
        if score < best_score - 1e-9:
            best_f = f.copy()
            best_score = score
            print(f"Fine: score={best_score:.6f}")

    # Final verification with the actual evaluator
    final_score = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    print(f"Final: score={best_score:.6f}, time={time.monotonic()-t_start:.1f}s")

    return [float(v) for v in best_f]
