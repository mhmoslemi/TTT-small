import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def project_simplex(v):

    n = len(v)

def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp(fixed, solver=None):
    """Given fixed sequence, find x minimizing max conv(fixed, x) s.t. x>=0, sum(x)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(fixed)
    x_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed)
    constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    x_val = np.asarray(x_var.value, dtype=np.float64)
    x_val = np.maximum(x_val, 0)
    s = x_val.sum()
    if s < 1e-15:
        return None, None
    x_val = x_val / s
    return x_val, float(M_var.value)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation)
    L = 1 << (2 * len(w) + n).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    corr = irfft(W * np.conj(A), L)[:n]
    return 2.0 * corr


def mirror_descent_smooth(a0, time_limit, T_start=3.0, T_end=400.0,
                          stages=10, steps_per_stage=1500, eta_init=0.1):
    """Mirror descent (multiplicative weights) on log-sum-exp smoothed objective."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 1e-300))
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, stages)

    for stage in range(stages):
        T = temps[stage]
        eta = eta_init / (1.0 + stage * 0.5)
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            grad = grad_smooth_max(x, T)
            # Multiplicative weights update
            log_update = -eta * grad
            log_update -= log_update.max()  # numerical stability
            x_new = x * np.exp(log_update)
            s = x_new.sum()
            if s < 1e-30:
                break
            x_new = x_new / s
            x_new = np.maximum(x_new, 0)
            s2 = x_new.sum()
            if s2 < 1e-30:
                break
            x = x_new / s2
            if step % 200 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=300.0,
                lr_init=0.06, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * (step_count + 1) * (step_count + 1))) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0 if step_count > 0 else 1.0
            beta = (t_prev - 1.0) / t_curr
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def lp_nesterov_refine(a0, time_limit, solver=None, n_max_iters=50000,
                       blend_base=0.5, adapt=True):
    """LP-based best response with adaptive Nesterov dual averaging."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    blend = blend_base
    b_history = []
    b_history_len = 50

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp(a, solver)
        if b_new is None:
            break

        # Adaptive blending: use median of recent b's for stability
        b_history.append(b_new.copy())
        if len(b_history) > b_history_len:
            b_history.pop(0)
        if len(b_history) >= 5:
            b_med = np.median(np.stack(b_history), axis=0)
            b_med = np.maximum(b_med, 0)
            s_med = b_med.sum()
            if s_med > 1e-15:
                b_med = b_med / s_med
            else:
                b_med = b_new

        # Adaptive step size
        if adapt:
            alpha = 2.0 / (it + 2)
            # Reduce step if oscillating
            if stall_count > 2:
                alpha *= 0.5
        else:
            alpha = 2.0 / (it + 2)

        y = (1 - alpha) * y + alpha * b_med
        y = project_simplex(y)

        # Nesterov-combined iterate
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        # Evaluate both iterates
        sc_a = score_normalized(a)
        sc_y = score_normalized(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0
            a = y.copy()  # adopt y if better

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                # Perturb y to escape local trap
                rng = np.random.default_rng(it * 7 + 3)
                pers_size = max(1, n // 10)
                idx = rng.choice(n, size=pers_size, replace=False)
                y[idx] += 0.03 * rng.standard_normal(pers_size)
                y = project_simplex(y)
                a = 0.7 * a + 0.3 * y
                a = project_simplex(a)
                stall_count = 0
                b_history = []

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def double_lp_alternate(a0, time_limit, solver=None, n_iters=100, blend=0.6):
    """Double-LP: alternately fix a (solve for b) and fix b (solve for a)."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Fix a, solve for b
        b, M_b = solve_lp(a, solver)
        if b is None:
            break
        # Fix b, solve for a_new
        a_new, M_a = solve_lp(b, solver)
        if a_new is None:
            break
        # Blend to avoid oscillation
        a_blended = (1 - blend) * a + blend * a_new
        a_blended = project_simplex(a_blended)

        sc_blend = score_normalized(a_blended)
        sc_new = score_normalized(a_new)
        if sc_blend <= sc_new:
            a = a_blended
            sc = sc_blend
        else:
            a = a_new
            sc = sc_new

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (it + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def active_set_coord_descent(a0, time_limit, n_steps=30000):
    """Coordinate descent focused on active (near-peak) constraints."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        # Focus on interior peaks (exclude boundaries)
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf

        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        # Identify near-peak positions
        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        # Compute contribution of each coordinate to near-peak values
        contrib_score = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            # Weight by how close this peak is to the max
            weight = 1.0 / (1.0 + 10.0 * (peak_val - b[k]) / (peak_val + 1e-30))
            contrib_score[idx] += weight * contribs

        # Try reducing top contributors
        top_coords = np.argsort(contrib_score)[::-1][:15]
        improved = False

        for i in top_coords:
            if a[i] < 1e-12:
                continue
            for factor in [0.96, 0.93, 0.89, 0.85, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
                else:
                    no_improve += 1

            if improved:
                break

        if not improved:
            # Try pair moves on the peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-12 or a[j] < 1e-12:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                        else:
                            no_improve += 1
                    if improved:
                        break
                if improved:
                    break

            if not improved:
                # Try triple reduction
                if len(top_contrib) >= 3:
                    a_trial = a.copy()
                    for ci in top_contrib[:3]:
                        i2 = idx[ci]
                        if a_trial[i2] > 1e-12:
                            a_trial[i2] *= 0.95
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 2

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_diverse_init(n, init_type, rng):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if init_type == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    elif init_type == 1:
        a = 1.0 + 0.7 * (2 * x - 1)
    elif init_type == 2:
        a = 1.0 + 0.4 * np.sin(4 * np.pi * x + 0.3)
    elif init_type == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a / a.max() + 0.2
    elif init_type == 4:
        a = 1.0 + 0.3 * np.sin(6 * np.pi * x)
    elif init_type == 5:
        a = 1.0 + 0.6 * np.cos(2 * np.pi * x * 1.5)
    elif init_type == 6:
        # Sparse-like: alternating high/low
        a = np.ones(n)
        a[::3] = 1.5
        a[1::3] = 0.7
    elif init_type == 7:
        # Random with smoothing
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with mirror descent, LP refinement,
    double-LP, and active-set coordinate descent.
    """
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_diverse_init(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Nesterov PG (warm-up) ===
    print(f"\n=== Phase 1: Nesterov PG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 360.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")

                              lr_init=0.05, n_stages=10, steps_per_stage=2000)
        try_update(a_ng, "ng_main")

    # === Phase 2: Mirror descent (multiplicative weights) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")

                                        stages=8, steps_per_stage=1000, eta_init=0.08)
        try_update(a_md, "mirror")

    # === Phase 3: LP-based Nesterov refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 480.0)

        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_nesterov_refine(best_seq, t_alloc, solver=solver,
                                     n_max_iters=60000, blend_base=0.5, adapt=True)
        try_update(a_lp, "lp_main")

    # === Phase 4: Double-LP alternating ===
    print(f"\n=== Phase 4: Double-LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 240.0)
    if t_alloc > 20:

    # === Phase 5: Second LP round from improved solution ===

        try_update(a_lp2, "lp_2")

    # === Phase 6: Active-set coordinate descent ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.6f}) ===")

    alt_lengths = sorted(set(alt_lengths))

        rem = deadline - time.monotonic()
        if rem < 50:

        if t_total < 25:

        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)

                                      n_max_iters=15000, blend_base=0.5, adapt=True)
        try_update(a_alt, f"alt_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 10:

            try_update(a_alt2, f"alt_n{alt_n}_cd")

        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
