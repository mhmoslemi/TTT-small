def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def get_lp_solver():

    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob.solve(solver=solver_name)

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = a_fixed
    first_row = np.zeros(n)
    first_row[0] = a_fixed[0]
    return toeplitz(first_col, first_row)


def solve_lp_best_response(a_fixed, solver=None, T_cache=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if T_cache is None:
        T = build_conv_matrix(a_fixed)
    else:
        T = T_cache

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[b_val < 1e-15] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def random_phase_init(n, n_trials=80, seed=42):
    rng = np.random.default_rng(seed)
    best_p = None
    best_score = float('inf')

    L = 1 << (2 * n - 1).bit_length()

    for trial in range(n_trials):
        phases = rng.random(L) * 2 * np.pi
        spec = np.exp(1j * phases)
        spec[0] = 1.0
        for k in range(1, L // 2 + 1):
            idx = L - k
            spec[idx] = np.conj(spec[k])
        p = np.real(ifft(spec))[:n]
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-10:
            continue
        p = p / s
        b = fast_convolve(p)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_p = p.copy()

    if best_p is None:
        best_p = np.ones(n) / n
    return best_p, best_score

def nesterov_fw(a0, time_limit, solver=None, n_max_iters=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    T_cache = build_conv_matrix(a)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M_init = best_score / (2.0 * n)
        b_new, M_b = solve_lp_best_response(y, solver=solver, T_cache=None)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)

        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y = y_new / sy

        a_new = a + alpha * (b_new - a)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

        b_a = fast_convolve(a)

        b_y = fast_convolve(y)
        score_y = 2.0 * n * b_y.max()

            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 200 == 0:

def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.1, maxiter=20000):

    t0 = time.monotonic()

    def lse_func(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        T_val = _current_T[0]
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        T_val = _current_T[0]
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    global _current_T
    _current_T = [T_start]

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        _current_T[0] = stage_T
        try:
            res = minimize(
                fun=lse_func,
                x0=z,
                jac=lse_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    # Phase 0: Better initialization via random phase
    print(f"\n=== Phase 0: Initialization ===")
    rem = deadline - time.monotonic()

    if parent is None:
        # No parent, generate good init
        n_init = 960
        t_init = min(rem * 0.05, 60.0)
        p_rp, sc_rp = random_phase_init(n_init, n_trials=int(max(20, t_init * 20)), seed=seed)
        sc_rp_eval = evaluate_safe(p_rp)
        print(f"  Random phase init: n={n_init}, score={sc_rp_eval:.6f}")
        best_seq = p_rp
        best_score = sc_rp_eval
        n_best = n_init
    else:
        # Have parent, try to improve init slightly
        n_curr = len(best_seq)
        t_init = min(rem * 0.03, 30.0)
        p_rp, sc_rp = random_phase_init(n_curr, n_trials=int(max(10, t_init * 10)), seed=seed + 1)
        sc_rp_eval = evaluate_safe(p_rp)
        if sc_rp_eval < best_score:
            best_seq = p_rp
            best_score = sc_rp_eval
            print(f"  New init from random phase: score={sc_rp_eval:.6f}")
        else:
            print(f"  Kept parent init: score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    # Phase 1: Main Nesterov Frank-Wolfe
    print(f"\n=== Phase 1: Nesterov FW (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_fw1 = min(rem * 0.35, 500.0)
    if t_fw1 > 30:
        print(f"  Budget: {t_fw1:.0f}s")
        a_fw, _ = nesterov_fw(best_seq, t_fw1, solver=solver, n_max_iters=30000)
        try_update(a_fw, "fw_main")

    # Phase 2: L-BFGS polish
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_lb1 = min(rem * 0.15, 200.0)
    if t_lb1 > 20:
        print(f"  Budget: {t_lb1:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lb1, T_start=50.0, T_end=0.1, maxiter=15000)
        try_update(a_lb, "lbgfs_1")

    # Phase 3: Second FW round (finer)
    print(f"\n=== Phase 3: FW Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_fw2 = min(rem * 0.25, 350.0)
    if t_fw2 > 20:
        print(f"  Budget: {t_fw2:.0f}s")
        a_fw2, _ = nesterov_fw(best_seq, t_fw2, solver=solver, n_max_iters=40000)
        try_update(a_fw2, "fw_2")

    # Phase 4: L-BFGS with tighter T
    print(f"\n=== Phase 4: L-BFGS Tight (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_lb2 = min(rem * 0.10, 150.0)
    if t_lb2 > 15:
        a_lb2 = lbgfs_refine(best_seq, t_lb2, T_start=20.0, T_end=0.05, maxiter=10000)
        try_update(a_lb2, "lbgfs_2")

    # Phase 5: Local swap
    print(f"\n=== Phase 5: Local Swap (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_ls = min(rem * 0.10, 120.0)
    if t_ls > 10:
        a_ls, _ = local_swap_search(best_seq, t_ls, n_steps=80000)
        try_update(a_ls, "swap")

    # Phase 6: Try alternative length
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    n_curr = len(best_seq)

    alt_candidates = []
    for target in [1024, 1152, 1280, 850]:
        if target != n_curr and 500 <= target <= 2000:
            alt_candidates.append(target)

    for alt_n in alt_candidates[:2]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.30, 200.0)
        if t_total < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)

        # Also try random phase for this length
        t_rp = 15.0
        p_rp_alt, _ = random_phase_init(alt_n, n_trials=int(t_rp * 3), seed=seed + alt_n)
        sc_rp_alt = evaluate_safe(p_rp_alt)
        if sc_rp_alt < sc_alt:
            init_alt = p_rp_alt
            sc_alt = sc_rp_alt

        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"    Alt init n={alt_n}: {sc_alt:.7f}")

        t_fw_alt = t_total * 0.65
        a_alt, _ = nesterov_fw(init_alt, t_fw_alt, solver=solver, n_max_iters=20000)
        try_update(a_alt, f"fw_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None:
            t_lb_alt = min(rem * 0.25, 60.0)
            if t_lb_alt > 10:
                a_lb_alt = lbgfs_refine(best_seq, t_lb_alt, T_start=30.0, T_end=0.1, maxiter=8000)
                try_update(a_lb_alt, f"lbgfs_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.05, maxiter=8000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ls2 = min(rem * 0.25, 50.0)
        a_ls2, _ = local_swap_search(best_seq, t_ls2, n_steps=40000)
        try_update(a_ls2, "final_swap")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_fw_fin = min(rem * 0.40, 60.0)
        a_fw_fin, _ = nesterov_fw(best_seq, t_fw_fin, solver=solver, n_max_iters=20000)
        try_update(a_fw_fin, "final_fw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
