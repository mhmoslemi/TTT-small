import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_best_response(a, solver):
    """Solve the LP: min M s.t. conv(a,b) <= M, sum b = 1, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix
    # T @ b = conv(a, b)
    # We need T @ b <= M for all 2n-1 positions
    # Instead of building the full matrix, use the structure
    
    # Actually, for cvxpy, we can express conv(a,b) directly
    # But it's easier to build the matrix for medium n
    
    # For n=1024, the matrix is 2047 x 1024, which is fine
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization (primary driver)
2. Subgradient descent with multi-peak weighting and simplex projection
3. Adaptive coordinate descent with incremental convolution updates
4. Random perturbation and restart for escaping local minima
5. Fine-grained polishing for last-mile improvements

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.ECOS_BB, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def subgradient_descent(a0, time_limit, lr_init=0.02, top_k=20, verbose=True):
    """
    Subgradient descent with multi-peak weighting and simplex projection.
    Uses Euclidean (not mirror) projection for better local behavior.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        # Compute multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        # Softmax weighting over peaks
        temp = 0.003 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient of max_k [sum_i a[i]*a[k-i]] w.r.t. a[j]
        # = sum_k w[k] * 2 * a[k-j] (for valid k-j)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Gradient step + projection
        a_new = a - lr * sg
        a_new = simplex_project(a_new)

        # Track running average
        it_f = float(it)
        a = a_new
        conv = fast_convolve(a)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

                unaff_left = conv[:k_start].max()
            else:

            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes

def lp_alternating(a0, time_limit, solver, verbose=True):
    """
    LP-based best-response alternating minimization.
    a -> b*(a) -> a*(b*) -> ...
    Uses full iterate initially, then damped updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: best response b* to current a
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Evaluate b_new
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Decide update strategy
        if it <= 20:
            # Full iterate
            a = b_new.copy()
        elif it <= 50:
            # Aggressive damping
            alpha = 0.7
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            # Conservative damping
            alpha = max(0.3, 10.0 / (it + 10.0))
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        # Evaluate updated a
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check stagnation
        if score_a > best_score * (1 + 1e-8):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        # If stuck, try a random perturbation
        if stagnation > 30:
            noise_scale = 0.01 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.001:
                a = a_pert
            stagnation = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):

        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Two humps
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Periodic
        period = int(rng.integers(3, 10))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # W-shape
        t = np.arange(n)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    else:
        # Smooth random
        freqs = rng.integers(0, 20, size=n)
        phases = rng.uniform(0, 2 * np.pi, size=n)
        a = 0.5 + 0.4 * np.cos(freqs * np.pi * 2 * np.arange(n) / n + phases)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def peak_shaving(a0, time_limit, verbose=True):
    """
    Focus specifically on reducing the top peaks.
    For each peak, identify contributing coordinate pairs and
    redistribute mass to reduce the peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top peaks
        K = 5
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.95:
                continue

            # Find the most contributing coordinate pairs
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Contributions: a[i] * a[k_star - i]
            pairs = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                pairs.append((i, j, a[i] * a[j]))

            pairs.sort(key=lambda x: x[2], reverse=True)

            # Try reducing the top contributing pair
            for (i, j, contrib) in pairs[:3]:
                if contrib < 1e-12:
                    continue
                # Try reducing the larger of a[i], a[j]
                if a[i] >= a[j]:
                    target = i
                    other = j
                else:
                    target = j
                    other = i

                # Try reducing target by various amounts
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = -a[target] * frac
                    if delta == 0:
                        continue

                    # Move mass to 'other' coordinate (keeps sum same approximately)
                    # Actually, let's just try reducing and see
                    new_a = a.copy()
                    new_a[target] = max(new_a[target] + delta, 0)

                    # Update convolution incrementally
                    k_s = target
                    k_e = min(target + n - 1, 2 * n - 2)
                    new_conv = conv.copy()
                    new_conv[k_s:k_e + 1] += 2.0 * delta * a[k_s - target:k_e - target + 1]
                    if 2 * target >= k_s and 2 * target <= k_e:
                        idx = 2 * target - k_s
                        new_conv[idx] = conv[2 * target] + 2.0 * delta * a[target] + delta * delta

                    new_M = float(new_conv.max())
                    if new_M < M - 1e-12:
                        a = new_a
                        conv = new_conv
                        M = new_M
                        improved = True
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        break

        if not improved and rounds > 50:
            break

        if verbose and rounds % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. LP best-response alternating (main workhorse, ~55% time)
    2. Subgradient descent with simplex projection (~15% time)
    3. Adaptive coordinate descent (~20% time)
    4. Peak shaving + fine polish (~10% time)
    With diverse initializations and random restarts throughout.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(777)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    # Find LP solver
    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.55

    # Run LP alternating from parent
    t_alloc = min(t1_end - time.monotonic(), 400.0)
    print(f"  LP alt from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Run LP alternating from diverse inits
    inits = [
        diverse_init(n, rng, 0),  # Triangle
        diverse_init(n, rng, 1),  # U-shape
        diverse_init(n, rng, 2),  # Cosine
        diverse_init(n, rng, 3),  # Two humps
    ]

    for idx, init in enumerate(inits):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(remaining * 0.4, 120.0)
        print(f"  LP alt init {idx+1}/{len(inits)}, budget={t_alloc:.0f}s")
        a_tmp, s_tmp = lp_alternating(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_init{idx+1}")

    # Second round of LP on best so far
    if time.monotonic() < t1_end - 60:
        t_alloc = min(t1_end - time.monotonic(), 200.0)
        print(f"  LP alt round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_round2")

    # === Phase 2: Subgradient Descent ===
    print(f"\n=== Phase 2: Subgradient Descent (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    t_alloc = min(t2_end - time.monotonic(), 150.0)
    if t_alloc > 30:
        print(f"  SGD budget={t_alloc:.0f}s")
        a_sgd, s_sgd = subgradient_descent(best_seq, t_alloc, lr_init=0.01, top_k=30, verbose=True)
        try_update(a_sgd, "sgd")

        # Also try from LP result
        if time.monotonic() < t2_end - 30:
            a_sgd2, s_sgd2 = subgradient_descent(a_lp, min(t2_end - time.monotonic(), 80.0),
                                                  lr_init=0.005, top_k=50, verbose=False)
            try_update(a_sgd2, "sgd_lp")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t3_end - time.monotonic(), 250.0)
    if t_alloc > 30:
        print(f"  CD budget={t_alloc:.0f}s")
        a_cd, s_cd = adaptive_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

        # CD from LP result
        if time.monotonic() < t3_end - 60:
            t_alloc2 = min(t3_end - time.monotonic(), 120.0)
            a_cd2, s_cd2 = adaptive_coord_descent(a_lp, t_alloc2, verbose=True, label="lp")
            try_update(a_cd2, "cd_lp")

        # CD from SGD result
        if time.monotonic() < t3_end - 30:
            t_alloc3 = min(t3_end - time.monotonic(), 60.0)
            a_cd3, s_cd3 = adaptive_coord_descent(a_sgd, t_alloc3, verbose=True, label="sgd")
            try_update(a_cd3, "cd_sgd")

    # === Phase 4: Peak Shaving + Fine Polish ===
    print(f"\n=== Phase 4: Peak Shaving + Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 20

    # Peak shaving
    if time.monotonic() < t4_end - 60:
        t_ps = min(t4_end - time.monotonic() - 60, 80.0)
        if t_ps > 20:
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shaving(best_seq, t_ps, verbose=True)
            try_update(a_ps, "peak_shave")

    # Fine polish
    if time.monotonic() < t4_end - 10:
        t_fp = min(t4_end - time.monotonic() - 10, 100.0)
        if t_fp > 15:
            print(f"  FinePolish budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine")

    # Final CD pass
    if time.monotonic() < t4_end - 5:
        t_final = min(t4_end - time.monotonic() - 5, 40.0)
        if t_final > 10:
            a_fin, s_fin = adaptive_coord_descent(best_seq, t_final, verbose=False, label="final")
            try_update(a_fin, "final_cd")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
