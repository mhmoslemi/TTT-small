    y = a0.copy()
    a = a0.copy()
    t = 1
    best_score = compute_score_normalized(a0)
    best_a = a0.copy()

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using log-sum-exp (LSE) gradient 
optimization with Adam, Nesterov LP alternating, multi-length exploration, 
and aggressive local peak-shaving refinement.
Key improvements: numerically stable LSE gradient as smooth surrogate for max,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Compute 2*n*max(c) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Evaluate sequence using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lse_score_and_grad_z(z, beta):
    """Compute LSE objective value and gradient w.r.t. log-domain z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    beta_c = beta * c
    bc_max = beta_c.max()
    exp_bc = np.exp(beta_c - bc_max)
    p = exp_bc / exp_bc.sum()
    lse_val = 2.0 * n * (bc_max + np.log(np.sum(exp_bc))) / beta
    g_a = 4.0 * n * np.correlate(p, a, mode='valid')
    grad_z = a * (g_a - np.dot(g_a, a))
    return lse_val, grad_z, a


def pnorm_score_and_grad_z(z, p):
    """Compute p-norm objective value and gradient w.r.t. log-domain z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    cp_vals = c_pos ** p
    S = np.sum(cp_vals)
    if S < 1e-300:
        val = 0.0
        grad_z = np.zeros_like(z)
    else:
        val = 2.0 * n * S ** (1.0 / p)
        factor = S ** (1.0 / p - 1.0)
        w = c_pos ** (p - 1.0)
        g_a = 4.0 * n * factor * np.correlate(w, a, mode='valid')
        grad_z = a * (g_a - np.dot(g_a, a))
    return val, grad_z, a


def lse_adam_optimize(a0, time_limit, beta_schedule=None):
    """Optimize using LSE objective with Adam in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()

    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 400, 800]

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_global = 0

    best_score = compute_score_normalized(a0)
    best_z = z.copy()
    t0 = time.monotonic()

    for stage_beta in beta_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        stage_start = time.monotonic()
        stage_budget = remaining * 0.85
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        best_stage_score = best_score
        best_stage_z = best_z.copy()
        no_improve = 0

        while time.monotonic() - stage_start < stage_budget:
            t += 1
            t_global += 1

            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a = exp_z / exp_z.sum()

            c = fast_convolve(a)
            beta_c = stage_beta * c
            bc_max = beta_c.max()
            exp_bc = np.exp(beta_c - bc_max)
            p = exp_bc / exp_bc.sum()

            g_a = 4.0 * n * np.correlate(p, a, mode='valid')
            grad = a * (g_a - np.dot(g_a, a))

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.5 / (1.0 + 0.1 * np.sqrt(t))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.max()  # Renormalize for stability

            # Periodic evaluation
            if t % 20 == 0 or t == 1:
                z_max = z.max()
                exp_z = np.exp(z - z_max)
                a_check = exp_z / exp_z.sum()
                score = compute_score_normalized(a_check)
                if score < best_stage_score - 1e-10:
                    best_stage_score = score
                    best_stage_z = z.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 50:
                    z = best_stage_z.copy()
                    break

        # Update global best
        z_max = best_stage_z.max()
        exp_z = np.exp(z_max if False else best_stage_z - best_stage_z.max())
        a_stage = exp_z / exp_z.sum()
        score_stage = compute_score_normalized(a_stage)
        if score_stage < best_score:
            best_score = score_stage
            best_z = best_stage_z.copy()
        z = best_z.copy()

        elapsed = time.monotonic() - t0
        print(f"      LSE-Adam beta={stage_beta:.0f} ({elapsed:.0f}s): best_score={best_score:.6f}")

    z_max = best_z.max()
    exp_z = np.exp(best_z - z_max)
    return exp_z / exp_z.sum()


def pnorm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    best_score = compute_score_normalized(a0)
    best_z = z.copy()

    for stage_p in temps:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters = max(50, int(remaining * 0.4))

        def func(z_val):
            val, _, _ = pnorm_score_and_grad_z(z_val, stage_p)
            return val

        def jac(z_val):
            _, grad, _ = pnorm_score_and_grad_z(z_val, stage_p)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': iters * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = compute_score_normalized(a_cur)
        if score < best_score:
            best_score = score
            best_z = z.copy()

        elapsed = time.monotonic() - t0
        print(f"      p-norm LBFGS p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = best_z.max()
    exp_z = np.exp(best_z - z_max)
    return exp_z / exp_z.sum()


def build_conv_matrix(a):
    """Build convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score_normalized(a)
        score_y = compute_score_normalized(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 300 == 0:
            if best_score < prev_stall_score - 1e-8:

                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:

def local_peak_shaving(a0, time_limit, n_steps=50000):
    """Aggressive local peak shaving: reduce values contributing to peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 200:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf

        peak_val = b_work.max()
        near_thresh = peak_val * 0.92
        near_peaks = np.where(b_work >= near_thresh)[0][:10]

        improved = False

        # Compute contribution scores
        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contribs = a[idx] * a[k - idx]
                contrib[idx] += contribs

        # Try reducing top contributors
        top_coords = np.argsort(contrib)[::-1][:20]

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.75]:
                diff = a[i] * (1 - factor)
                a_trial = a.copy()
                a_trial[i] = max(0, a[i] - diff)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                score = compute_score_normalized(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pair adjustments
        if not improved:
            peak_k = int(near_peaks[0]) if len(near_peaks) > 0 else n // 2
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_pairs = np.argsort(contribs)[::-1][:8]
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.88, 0.80]:
                    for f2 in [0.93, 0.88, 0.80]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        score = compute_score_normalized(a_trial)
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Random perturbation
        if not improved and step % 50 == 0:
            rng = np.random.default_rng(step * 7 + 3)
            n_pert = int(rng.integers(2, 8))
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.08, 0.08))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = compute_score_normalized(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
        else:
            stall = 0

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2.0
    else:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 5.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update(best_state, a, label):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LSE-Adam, p-norm L-BFGS, Nesterov LP,
    local peak shaving, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=3)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: LSE-Adam (primary optimizer) ===
    print(f"\n=== Phase 1: LSE-Adam n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, betas: 5->800")
        a_lse = lse_adam_optimize(best_state['seq'], t_alloc,
                                  beta_schedule=[5, 10, 20, 40, 80, 150, 300, 600, 1000])
        try_update(best_state, a_lse, "lse_adam_1")

    # === Phase 2: p-norm L-BFGS (refinement) ===
    print(f"\n=== Phase 2: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, p: 50->2000")
        a_pn = pnorm_lbfgs_refine(best_state['seq'], t_alloc,
                                   p_start=50.0, p_end=2000.0, n_stages=7)
        try_update(best_state, a_pn, "pnorm_lbfgs_1")

    # === Phase 3: Nesterov LP ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver,
                                      n_max_iters=200000)
        try_update(best_state, a_lp, "nesterov_lp_1")

    # === Phase 4: Local peak shaving ===
    print(f"\n=== Phase 4: Local Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_lr, _ = local_peak_shaving(best_state['seq'], t_alloc, n_steps=80000)
        try_update(best_state, a_lr, "local_shave_1")

    # === Phase 5: LSE-Adam round 2 (higher betas) ===
    print(f"\n=== Phase 5: LSE-Adam r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, betas: 100->3000")
        a_lse2 = lse_adam_optimize(best_state['seq'], t_alloc,
                                    beta_schedule=[100, 200, 400, 800, 1500, 3000])
        try_update(best_state, a_lse2, "lse_adam_2")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-128, -96, -64, -32, 32, 64, 96, 128, 192]:
        alt_n = base + delta
        if 512 <= alt_n <= 2560:
            candidate_lengths.add(alt_n)
    for rn in [640, 768, 896, 1024, 1152, 1280, 1408, 1536]:
        if 512 <= rn <= 2560 and abs(rn - base) > 32:
            candidate_lengths.add(rn)

    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 120.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # LSE-Adam on new length
        t_sub = t_total * 0.7
        a_alt = lse_adam_optimize(init_alt, t_sub,
                                   beta_schedule=[20, 50, 100, 200, 500])
        improved = try_update(best_state, a_alt, f"lse_n{alt_n}")

        if not improved and t_total * 0.2 > 15:
            # Try p-norm LBFGS too
            a_alt2 = pnorm_lbfgs_refine(init_alt, t_total * 0.3,
                                         p_start=30.0, p_end=300.0, n_stages=4)
            try_update(best_state, a_alt2, f"pnorm_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.25, 50.0)
            a_pert = lse_adam_optimize(perturbed, t_sub,
                                        beta_schedule=[50, 100, 300, 600])
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 60.0)
        a_fin = lse_adam_optimize(best_state['seq'], t_fin,
                                   beta_schedule=[200, 500, 1000, 2000])
        try_update(best_state, a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.30, 40.0)
        a_lr2, _ = local_peak_shaving(best_state['seq'], t_lr, n_steps=30000)
        try_update(best_state, a_lr2, "final_shave")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.40, 40.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state['seq'], t_lp_fin, solver=solver,
                                          n_max_iters=30000)
        try_update(best_state, a_lp_fin, "final_lp")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")

    return [float(x) for x in best_state['seq']]
