def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_of_max_conv(a, k_star):
    """Compute subgradient of max_k conv(a,a) w.r.t. a at peak k_star."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for i in range(js, je + 1):
        g[i] = 2.0 * a[k_star - i]
    return g


def subgradient_top_k(a, k_list):
    """Average subgradient over top-k peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            g[i] += 2.0 * a[k - i]
    g /= len(k_list)
    return g

        a = _softmax(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp
    return func


def _lbgfs_jac_factory(n, T_val, t0, time_limit):
    def jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = _softmax(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    return jac


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def nesterov_direct_subgradient(a0, time_limit, n_top=3, max_iters=200000):
    """Nesterov-accelerated projected subgradient on the TRUE objective."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_sequence(a)
    t0 = time.monotonic()
    eta0 = 2.0 / np.sqrt(n)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        top_k = np.argsort(b)[::-1][:n_top]
        
        g = np.zeros(n)
        for k in top_k:
            g += get_self_subgradient(y, int(k))
        g /= n_top
        
        eta = eta0 / np.sqrt(it + 1)
        theta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        
        y_new = project_simplex(y - eta * g)
        a_new = y_new + ((theta - 1.0) / theta) * (y_new - a)
        a_new = project_simplex(a_new)
        
        y = a_new
        a = a_new
        
        score = score_sequence(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def lip_dfine(a0, time_limit, m=50, solver=None):
    """
    Active-set LP: iteratively solve smaller LPs with growing constraint set.
    Fix a, solve for b with only active constraints.
    """
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_sequence(a)
    t0 = time.monotonic()
    
    if solver is None:
        solver = find_lp_solver()
    
    active_set = set()
    # Initialize active set with top peaks
    b_init = fast_convolve(a)
    active_set = set(np.argsort(b_init)[::-1][:m].tolist())
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min t s.t. (T_a @ b)[k] <= t for k in active_set, b>=0, sum b=1
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(b_var) == 1]
        
        for k in active_set:
            coeff = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                coeff[j] = a[k - j]
            constraints.append(coeff @ b_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            continue
        b_val = b_val / s
