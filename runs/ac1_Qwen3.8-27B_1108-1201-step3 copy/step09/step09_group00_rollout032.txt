def fast_convolve(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    """Compute c_i = sum_m w[i+m] * a[m] for i=0,...,n-1 using FFT."""
    n = len(a)
    w_len = len(w)  # should be 2n-1

        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.3
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternation, FFT-accelerated L-BFGS LogSumExp,
    peak-shaving, subgradient, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nlp_1")

    # === Phase 2: L-BFGS LogSumExp (FFT-accelerated) ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.1, 0.05, 0.03]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=10000)
        try_update(a_lb, "lbgfs_1")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_ps, _ = peak_shaving(best_seq, t_alloc, n_steps=100000)
        try_update(a_ps, "peak_1")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=60000)
        try_update(a_sg, "subgrad_1")

    # === Phase 5: Second L-BFGS pass at very low temperature ===
    print(f"\n=== Phase 5: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        T_stages2 = [3.0, 1.5, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02]
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages2, maxiter_per_stage=8000)
        try_update(a_lb2, "lbgfs_2")

    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alt Length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        alt_ns = [int(n_best * 1.3), int(n_best * 0.8)]
        for alt_n in alt_ns:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            if alt_n <= 100 or alt_n > 5000:
                continue
            t_alt = min(rem * 0.4, 200.0)
            if t_alt < 30:
                break
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.55, solver=solver, n_max_iters=50000)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 30:
                T_stages3 = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1]
                a_alt2 = lbgfs_refine(best_seq, min(rem * 0.4, 100.0), T_stages=T_stages3, maxiter_per_stage=5000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 7: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 7: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.25, 120.0)
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        for freq in [1, 2, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbation = 0.015 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
            perturbed = np.maximum(best_seq + perturbation, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            t_seg = min(t_pert / 4.0, rem * 0.4)
            if t_seg < 10:
                break
            a_pert, _ = nesterov_lp_refine(perturbed, t_seg, solver=solver, n_max_iters=30000)
            try_update(a_pert, f"pert_f{freq}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin,
                                 T_stages=[2.0, 1.0, 0.5, 0.3, 0.15, 0.1, 0.05, 0.03, 0.02],
                                 maxiter_per_stage=6000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.4, 30.0)
        if t_ps > 8:
            a_ps2, _ = peak_shaving(best_seq, t_ps, n_steps=30000)
            try_update(a_ps2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_sg = min(rem * 0.5, 20.0)
        if t_sg > 5:
            a_sg2, _ = subgradient_refine(best_seq, t_sg, n_steps=15000)
            try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
