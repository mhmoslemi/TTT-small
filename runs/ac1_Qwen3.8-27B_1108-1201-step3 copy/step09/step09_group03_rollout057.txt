def simplex_project(v, target_sum=1.0):

def compute_subgradient(a, conv, n_peaks=50, temperature=0.003):
    """
    Compute the multi-peak subgradient of max(conv(a,a)).
    Returns gradient vector g where g_i = 2 * sum_k w_k * a[k-i].

    n = len(a)
    conv_len = 2 * n - 1
    M = conv.max()
    
    # Select active peaks
    k_act = min(n_peaks, conv_len)
    if k_act == conv_len:
        peaks = np.arange(conv_len)
    else:
        peaks = np.argpartition(conv, -k_act)[-k_act:]
    
    pv = conv[peaks]
    temp = temperature * max(M, 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(len(peaks)) / len(peaks)

    # The affected part:
    a_idx = k_start - i  # = 0
    a_end = k_end - i    # = min(n-1, 2n-2-i)
    a_shifted = a[a_idx:a_end + 1]  # a[k-i] for k in [k_start, k_end]
    
    base = conv[k_start:k_end + 1].copy()
    quadratic_at = 2 * i - k_start  # index where k = 2i (if in range)
    has_quadratic = (2 * i >= k_start) and (2 * i <= k_end)
    
    def g_of_delta(delta):
        """Compute max_k f_k(delta)."""
        vals = base + 2.0 * delta * a_shifted
        if has_quadratic:
            vals = vals.copy()
            vals[quadratic_at] += delta * delta
        return max(unaff_max, float(vals.max()))
    
    # Find optimal delta via ternary/bisection search
    # g(delta) is convex in delta (max of convex functions)
    # Domain: delta in [-a[i], upper_bound]
    lower = -a[i]
    # Upper bound: where the quadratic term makes things worse
    upper = a[i] * 3.0
    # Also limit by where linear terms get too large
    max_slope = 2.0 * a_shifted.max() if len(a_shifted) > 0 else 1.0
    if max_slope > 1e-10:
        upper = min(upper, (best_score_estimate(unaff_max, base) - unaff_max) / max_slope)
    upper = max(upper, a[i] * 0.5)
    
    # If unaff_max dominates, we only need to reduce the affected max below unaff_max
    # Use golden section search on the convex function g(delta)
    gr = (np.sqrt(5) + 1) / 2
    a_g = lower
    b_g = upper
    
    # Quick check: is delta=0 already optimal?
    g0 = g_of_delta(0.0)
    
    # Sample a few points to get a rough idea
    best_delta = 0.0
    best_g = g0
    
    # Golden section search
    for _ in range(50):
        c = b_g - (b_g - a_g) / gr
        d = a_g + (b_g - a_g) / gr
        g_c = g_of_delta(c)
        g_d = g_of_delta(d)
        if g_c < g_d:
            b_g = d
        else:
            a_g = c
        
        mid = (a_g + b_g) / 2
        g_mid = g_of_delta(mid)
        if g_mid < best_g:
            best_g = g_mid
            best_delta = mid
        
        if b_g - a_g < 1e-12 * max(1.0, abs(best_delta)):
            break
    
    # Also check boundary points
    for test_d in [lower, 0.0, -a[i] * 0.5, -a[i] * 0.9, a[i] * 0.1, a[i] * 0.3]:
        if test_d < lower - 1e-15 or test_d > upper + 1e-15:
            continue
        g_t = g_of_delta(test_d)
        if g_t < best_g:
            best_g = g_t
            best_delta = test_d
    
    return best_delta, best_g


def coord_descent_exact(a0, time_budget, n_peaks_grad=50, verbose=True):
    """
    Coordinate descent with exact 1D optimization per coordinate.
    Uses sensitivity-based ordering: process coordinates most likely to improve first.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_budget:
        sweep += 1
        improved = False
        sweep_improvement = 0.0
        
        # Compute sensitivity for each coordinate
        # Sensitivity of coordinate i: how much can we reduce M by changing a[i]?
        # Approximate by: max_k (2 * a[k-i]) over active k's minus the "average"
        # Simpler: just iterate in reverse order (center coordinates affect more peaks)
        
        if sweep % 4 == 0:
            # Occasionally reorder by sensitivity
            M_curr = conv.max()
            active_peaks = np.where(conv > 0.95 * M_curr)[0]
            sensitivity = np.zeros(n)
            for k in active_peaks[:100]:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                sensitivity[i_start:i_end + 1] += a[max(0, k - i_end):k - i_start + 1][::-1]
            order = np.argsort(-sensitivity)
        else:
            order = np.arange(n)[::-1]  # center-out
        
        for i in order:
            if a[i] < 1e-14:
                continue
            if time.monotonic() - t0 > time_budget:
                break
            
            delta, new_M = exact_1d_optimize(a, conv, i, n_factors=30)
            
            if abs(delta) > 1e-14 and new_M < M - 1e-12:
                # Apply update
                a[i] = max(a[i] + delta, 0.0)
                
                # Update convolution incrementally
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                a_idx = k_start - i
                a_end = k_end - i
                a_shifted_old = a[a_idx:a_end + 1].copy()
                # Note: a[i] has already been updated, so a_shifted includes the new value
                # We need to use the OLD a[i] for the linear term
                # Actually, the update formula uses the old a (before change)
                # conv_new[k] = conv_old[k] + 2*delta*a_old[k-i] + delta^2*[k==2i]
                # But a[i] is already updated. Let's use delta properly:
                # The change is: 2*delta*a_old[k-i] + delta^2*[k==2i]
                # a_old[k-i] = a_new[k-i] - delta*[k-i==i] = a_new[k-i] - delta*[k==2i]
                # So: 2*delta*(a_new[k-i] - delta*[k==2i]) + delta^2*[k==2i]
                #     = 2*delta*a_new[k-i] - 2*delta^2*[k==2i] + delta^2*[k==2i]
                #     = 2*delta*a_new[k-i] - delta^2*[k==2i]
                
                a_shifted_new = a[a_idx:a_end + 1]
                changes = 2.0 * delta * a_shifted_new - delta * delta * np.zeros(len(a_shifted_new))
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] -= delta * delta
                    
                    # More carefully:
                    # old_conv[k] + 2*delta*a_old[k-i] + delta^2*[k=2i]
                    # For k=2i: a_old[i] = a[i]_old = a[i]_new - delta
                    # So: old_conv[2i] + 2*delta*(a_new[i] - delta) + delta^2
                    #     = old_conv[2i] + 2*delta*a_new[i] - 2*delta^2 + delta^2
                    #     = old_conv[2i] + 2*delta*a_new[i] - delta^2
                    # For k != 2i: a_old[k-i] = a_new[k-i] (since k-i != i)
                    # So: old_conv[k] + 2*delta*a_new[k-i]
                    
                    # Let me just recompute:
                    changes = 2.0 * delta * a_shifted_new
                    changes[idx] -= delta * delta
                else:
                    changes = 2.0 * delta * a_shifted_new
                
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max_recompute(conv, k_start, k_end), float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
                sweep_improvement += (M_old - new_M) if 'M_old' in dir() else 0
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")
    
    return best_a, best_score


def unaff_max_recompute(conv, k_start, k_end):
    """Compute max of conv outside [k_start, k_end]."""
    total_len = len(conv)
    if k_start > 0:
        left_max = conv[:k_start].max()
    else:
        left_max = -1e30
    if k_end + 1 < total_len:
        right_max = conv[k_end + 1:].max()
    else:
        right_max = -1e30
    return max(left_max, right_max)


def best_score_estimate(unaff_max, base):
    """Estimate a reasonable bound."""
    return max(unaff_max, float(base.max()))


def spectral_flatten(a0, time_budget, damping=0.2, alpha=0.4, verbose=True):
    """
    Iteratively flatten the power spectrum of a to reduce convolution peaks.
    Uses soft amplitude modification in the frequency domain.
    """
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        power = np.abs(A) ** 2
        
        # Soft flattening: reduce peaks, boost valleys
        mean_p = power[power > 0].mean() if (power > 0).any() else 1e-10
        # Ratio to apply
        safe_power = np.maximum(power, 1e-10)
        ratio = mean_p / safe_power
        # Damp the effect
        ratio_damped = ratio ** damping
        # Also cap the ratio to avoid extreme changes
        ratio_damped = np.clip(ratio_damped, 0.5, 2.0)
        
        A_new = A * ratio_damped
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Project to simplex
        a_new = simplex_project(a_new, 1.0)
        
        # Blend
        a_blended = (1.0 - alpha) * a + alpha * a_new
        a_blended = simplex_project(a_blended, 1.0)
        
        conv_new = fast_convolve(a_blended)
        s_new = 2.0 * n * float(conv_new.max())
        
        if s_new < best_score - 1e-10:
            best_score = s_new
            best_a = a_blended.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_blended
        
        if no_improve > 100:
            # Reduce damping
            damping *= 0.7
            alpha *= 0.8
            no_improve = 0
            if damping < 0.01:
                break
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpecFlatten it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def fine_polish(a0, time_budget, verbose=True):
    """
    Fine coordinate descent with very small steps for final polishing.
    Uses the exact 1D optimizer but with tighter tolerances.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        sweep += 1
        improved = False
        
        for i in range(n // 2, -1, -1):  # Center out
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-12:
                continue
            
            delta, new_M = exact_1d_optimize(a, conv, i)
            
            if abs(delta) > 1e-14 and new_M < M - 1e-13:
                a[i] = max(a[i] + delta, 0.0)
                
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                a_shifted = a[k_start - i:k_end - i + 1]
                changes = 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] -= delta * delta
                
                conv[k_start:k_end + 1] += changes
                unaff = unaff_max_recompute(conv, k_start, k_end)
                M = max(unaff, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweep} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def try_multi_start(a0, time_budget, n_starts=5):
    """
    Try multiple perturbed versions of a0 and pick the best.
    """
    n = len(a0)
    rng = np.random.default_rng(123)
    best_a = a0.copy()
    best_score = score_of(a0)
    
    per_start = time_budget / n_starts
    t0 = time.monotonic()
    
    for s in range(n_starts):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Generate perturbation
        if s == 0:
            candidate = a0.copy()
        elif s == 1:
            # Smooth perturbation
            noise = rng.standard_normal(n)
            # Smooth it
            from scipy.ndimage import gaussian_filter1d
            noise = gaussian_filter1d(noise, sigma=n * 0.05)
            candidate = a0 + noise * 0.01
            candidate = simplex_project(candidate, 1.0)
        elif s == 2:
            # Periodic perturbation
            period = rng.integers(3, 20)
            candidate = a0 * (1.0 + 0.05 * np.sin(2 * np.pi * np.arange(n) / period))
            candidate = simplex_project(candidate, 1.0)
        elif s == 3:
            # Random flip of a few coordinates
            candidate = a0.copy()
            flip_idx = rng.choice(n, size=n // 50, replace=False)
            candidate[flip_idx] *= rng.uniform(0.5, 2.0, size=len(flip_idx))
            candidate = simplex_project(candidate, 1.0)
        else:
            # Larger perturbation
            candidate = a0 + rng.standard_normal(n) * 0.02
            candidate = simplex_project(candidate, 1.0)
        
        # Quick refinement
        a_ref, s_ref = nesterov_subgradient_descent(candidate, per_start * 0.6, n_peaks=30, temperature=0.002)
        s_eval = score_of(a_ref)
        
        if s_eval < best_score:
            best_score = s_eval
            best_a = a_ref.copy()
    
    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
    1. Spectral flattening (FFT-based power spectrum equalization)
    2. Nesterov accelerated subgradient descent
    3. Exact coordinate descent
    4. Multi-start perturbation
    5. Fine polishing
    """
    deadline = time.monotonic() + 1785.0
    
    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        if parent.sum() < 0.01:
            raise ValueError
    except Exception:
        parent = np.ones(1024, dtype=np.float64)
    
    parent = parent / parent.sum()
    n = len(parent)
    
    best_a = parent.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial: n={n}, score={best_score:.6f}")
    
    # === Phase 1: Spectral Flattening ===
    remaining = deadline - time.monotonic() - 800
    t1 = min(remaining * 0.15, 120.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Spectral Flattening (budget={t1:.0f}s) ===")
        a1, s1 = spectral_flatten(best_a, t1, damping=0.15, alpha=0.3, verbose=True)
        s1_eval = float(evaluate_sequence(a1.tolist()))
        if s1_eval < best_score:
            best_score = s1_eval
            best_a = a1.copy()
            print(f"  Improved: {best_score:.6f}")
    
    # === Phase 2: Nesterov Subgradient Descent ===
    remaining = deadline - time.monotonic() - 600
    t2 = min(remaining * 0.3, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Nesterov Subgradient (budget={t2:.0f}s) ===")
        a2, s2 = nesterov_subgradient_descent(best_a, t2, n_peaks=60, temperature=0.003)
        s2_eval = float(evaluate_sequence(a2.tolist()))
        if s2_eval < best_score:
            best_score = s2_eval
            best_a = a2.copy()
            print(f"  Improved: {best_score:.6f}")
    
    # === Phase 3: Exact Coordinate Descent ===
    remaining = deadline - time.monotonic() - 500
    t3 = min(remaining * 0.4, 400.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Exact Coord Descent (budget={t3:.0f}s) ===")
        a3, s3 = coord_descent_exact(best_a, t3, verbose=True)
        s3_eval = float(evaluate_sequence(a3.tolist()))
        if s3_eval < best_score:
            best_score = s3_eval
            best_a = a3.copy()
            print(f"  Improved: {best_score:.6f}")
    
    # === Phase 4: Multi-start with short refinement ===
    remaining = deadline - time.monotonic() - 200
    t4 = min(remaining * 0.4, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: Multi-Start (budget={t4:.0f}s) ===")
        a4, s4 = try_multi_start(best_a, t4, n_starts=4)
        s4_eval = float(evaluate_sequence(a4.tolist()))
        if s4_eval < best_score:
            best_score = s4_eval
            best_a = a4.copy()
            print(f"  Improved: {best_score:.6f}")
    
    # === Phase 5: Fine Polishing ===
    remaining = deadline - time.monotonic() - 20
    t5 = min(remaining * 0.7, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Fine Polishing (budget={t5:.0f}s) ===")
        a5, s5 = fine_polish(best_a, t5, verbose=True)
        s5_eval = float(evaluate_sequence(a5.tolist()))
        if s5_eval < best_score:
            best_score = s5_eval
            best_a = a5.copy()
            print(f"  Improved: {best_score:.6f}")
    
    # === Phase 6: Final coordinate descent pass ===
    remaining = deadline - time.monotonic() - 5
    t6 = min(remaining * 0.8, 60.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final CD (budget={t6:.0f}s) ===")
        a6, s6 = coord_descent_exact(best_a, t6, verbose=False)
        s6_eval = float(evaluate_sequence(a6.tolist()))
        if s6_eval < best_score:
            best_score = s6_eval
            best_a = a6.copy()
            print(f"  Improved: {best_score:.6f}")
    
    # Final verification
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}")
    return [float(x) for x in best_a]
