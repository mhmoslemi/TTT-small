import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def score_norm(a):
    """Compute score for a sequence assumed to have sum=1."""
    n = len(a)
    M = fast_convolve(a).max()
    return 2.0 * n * M

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_quadratic_residue(n, rng):
    """Initialize using quadratic residue structure for flat autocorrelation."""
    # Find a prime near n/2 for the QR construction
    target = n // 2
    p = target
    while p > target - 100:
        if _is_prime(p):
            break
        p -= 1
    if p < 10:
        return np.ones(n) / n
    
    # Create QR sequence of length p
    qr = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            qr[i] = 1.0
    qr[0] = 1.0  # Include 0
    
    # Repeat to fill length n with some randomization
    a = np.zeros(n)
    n_repeats = n // p + 1
    for rep in range(n_repeats):
        start = rep * p
        end = min(start + p, n)
        if start >= n:
            break
        length = end - start
        # Add small random shift for diversity
        shift = rng.integers(0, p)
        for i in range(length):
            a[start + i] = qr[(i + shift) % p]
    
    # Add small noise and ensure non-negativity
    a += rng.random(n) * 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True


def init_bimodal(n, rng):
    """Initialize with bimodal distribution."""
    center1 = n * rng.uniform(0.15, 0.35)
    center2 = n * rng.uniform(0.65, 0.85)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_symmetric_cosh(n, rng):
    """Symmetric initialization with cosh profile (eminimizes boundary effects)."""
    t = np.linspace(-1, 1, n)
    alpha = rng.uniform(0.5, 2.0)
    a = np.cosh(alpha * t) ** 0.5
    # Add ripple
    freq = rng.uniform(2, 8)
    amp = rng.uniform(0.05, 0.2)
    a += amp * np.cos(2 * np.pi * freq * t)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def exact_coord_descent(a, conv, time_limit, max_sweeps=200, verbose=False, label=""):
    """
    Exact coordinate descent: for each coordinate, find the optimal delta
    via bisection to minimize the max convolution value.
    """
    n = len(a)
    a = a.copy()
    conv = conv.copy()
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Randomize coordinate order to avoid systematic bias
        order = rng.permutation(n)

        for i in order:
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Compute unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, unaff_right)

            current_affected_max = float(conv[k_start:k_end + 1].max())

            # If current max is not in affected region, no single-coordinate improvement
            if current_affected_max <= unaff_max + 1e-15:
                continue

            # Bisection for optimal delta (most negative delta keeping affected <= unaff_max)
            # Feasible at delta=0? If unaff_max >= current_affected_max, yes (skip)
            # We want the largest delta (closest to 0) such that max_affected <= unaff_max
            
            # Quick check: if unaff_max is already >= M, no improvement from this coord
            if unaff_max >= M - 1e-15:
                # The peak is not in the affected region after all, or tied
                # Try to see if we can still reduce
                pass

            lo = -a[i]  # most negative feasible
            hi = 0.0    # current (infeasible target: max_affected > unaff_max)

            # Check if lo is feasible
            affected_lo = conv[k_start:k_end + 1].copy()
            b_vec = a[max(0, k_start - i):min(n, k_end + 1 - i)]
            affected_lo = affected_lo + 2.0 * lo * b_vec
            if 2 * i >= k_start and 2 * i <= k_end:
                idx_2i = 2 * i - k_start
                affected_lo[idx_2i] += lo * lo
            if affected_lo.max() > unaff_max + 1e-12:
                # Can't even reach unaff_max by zeroing this coordinate
                continue

            # Bisection
            for _ in range(40):
                mid = (lo + hi) / 2.0
                affected_mid = conv[k_start:k_end + 1] + 2.0 * mid * b_vec
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    # Make a copy to avoid modifying original
                    affected_mid = affected_mid.copy() if not affected_mid.flags.owndata else affected_mid.copy()
                    affected_mid[idx_2i] += mid * mid
                if float(affected_mid.max()) > unaff_max:
                    hi = mid
                else:
                    lo = mid

            delta = lo
            if delta >= -1e-15:
                continue

            # Apply the update
            a[i] += delta
            a[i] = max(a[i], 0.0)

            # Update conv
            changes = 2.0 * delta * b_vec
            conv[k_start:k_end + 1] += changes
            if 2 * i >= k_start and 2 * i <= k_end:
                idx_2i = 2 * i - k_start
                conv[2 * i] += delta * delta

            new_M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
            if new_M < M - 1e-15:
                M = new_M
                improved = True
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.10f}")

    return best_a, best_score, sweeps


def mirrored_coord_descent(a, conv, time_limit, lr=0.1, verbose=False, label=""):
    """
    Coordinate descent with mirror descent step: instead of exact delta,
    use a multiplicative update based on the local gradient.
    """
    n = len(a)
    a = a.copy()
    conv = conv.copy()
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Identify peak positions
        peak_threshold = M * 0.9
        peak_mask = conv >= peak_threshold
        peak_count = peak_mask.sum()

        if peak_count == 0:
            break

        # Compute pressure on each coordinate from peaks
        pressure = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for k in peak_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                w = conv[k] / M  # weight by peak height relative to M
                pressure[i] += w * a[j]
                if j != i:
                    pressure[j] += 0.5 * w * a[i]

        # Normalize pressure
        p_max = pressure.max()
        if p_max > 1e-20:
            pressure = pressure / p_max

        # Multiplicative update
        a_new = a * np.exp(-lr * pressure)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        if M_new < M - 1e-14:
            a = a_new
            conv = conv_new
            M = M_new
            improved = True
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

            # Decrease lr as we converge
            if sweeps % 50 == 0:
                lr *= 0.9
        else:
            # Revert and decrease learning rate
            lr *= 0.8
            no_improve += 1
            if no_improve > 20 or lr < 1e-6:
                break
        if improved:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            print(f"    MCD {label} sweep={sweeps}: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def nesterov_smooth_descent(a0, time_limit, lr_init=0.01, beta_init=50.0, verbose=False, label=""):
    """
    Nesterov-accelerated mirror descent with log-sum-exp smoothing of the max.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    y = a.copy()  # Nesterov momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * a_prev
        y_proj = simplex_project(y)

        # Compute smooth gradient
        conv_y = fast_convolve(y_proj)
        max_cv = float(conv_y.max())
        scaled = beta * (conv_y - max_cv)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights /= Z

        # Gradient: grad[i] = 2 * sum_k weights[k] * y_proj[k-i]
        grad = np.zeros(n)
        # Vectorized: grad[i] = 2 * (weights * y_proj) convolved appropriately
        # grad[i] = 2 * sum_k w_k * y_proj[k-i] for valid k-i
        # This is 2 * convolve(weights, y_proj) but only the valid part
        # Actually: grad[i] = 2 * sum_{k: 0<=k-i<n} w_k * y_proj[k-i]
        # = 2 * (w * y_proj)[i] where (w*y_proj) is cross-correlation
        # = 2 * np.convolve(weights, y_proj)[n-1-i] ... let me just do it directly
        
        # Efficient computation via FFT
        # grad[i] = 2 * sum_k w_k * y_proj[k-i] * [0 <= k-i < n]
        # Let's define: for each k, y_proj[k-i] is nonzero when 0 <= k-i < n, i.e., i in [k-n+1, k]
        # grad = 2 * sum_k w_k * y_proj[k-i] (with bounds)
        # This is: grad = 2 * ifft(fft(conj(w)) * fft(y_proj)) ... 
        # Actually it's a cross-correlation: grad[i] = sum_k w_k * y_proj[k-i]
        # = cross_correlation(w, y_proj)[i + n - 1] (with appropriate indexing)
        
        # Simpler: use the fact that it's a convolution
        # grad[i] = 2 * sum_k w_k * y_proj[k-i]
        # Let v = flip(y_proj), then grad[i] = 2 * (w * v)[n-1-i] ... 
        # Even simpler: grad = 2 * np.convolve(weights, y_proj[::-1])[n-1:2*n-1][::-1]
        
        # Or just: for the auto-conv case, grad[i] = 2 * sum_k w_k * a[k-i]
        # = 2 * (the i-th coefficient of the "correlation" of w with y_proj)
        # Using FFT:
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(weights, fft_size)
        Y = rfft(y_proj, fft_size)
        # We want: grad[i] = sum_k w_k * y_proj[k-i]
        # This is the cross-correlation, which is IFFT(FFT(w) * conj(FFT(y)))
        # But we need to be careful with the indexing
        corr = irfft(W * np.conj(Y), fft_size)
        # corr[k] = sum_j w[j] * y_proj[j-k] ... 
        # Actually: IFFT(FFT(w) * conj(FFT(y)))[k] = sum_j w[j] * y[(j-k) mod N]
        # For our purpose (valid range), grad[i] = 2 * corr[n-1-i] or similar
        # Let me just compute it directly for correctness (n is at most ~2048)
        
        # Direct computation (O(n^2) but with the peak weights being sparse-ish)
        active_peaks = np.where(weights > 1e-10)[0]
        for k in active_peaks:
            w_k = weights[k]
            if w_k < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end + 1] += 2.0 * w_k * y_proj[k - i_start:i_end + 1]

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_prev = a.copy()
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        lr = lr_init / (1.0 + 0.1 * gamma)

        if no_improve > 200:
            beta *= 1.5
            no_improve = 0
            if beta > 10000:
                break

        if verbose and it % 20 == 0:
            print(f"    NSD {label} it={it}: score={best_score:.6f} beta={beta:.0f}")

    return best_a, best_score


def peak_targeted_refine(a, conv, time_limit, verbose=False, label=""):
    """
    Focus specifically on reducing the top peaks.
    Identify the K largest conv positions and do targeted optimization.
    """
    n = len(a)
    a = a.copy()
    conv = conv.copy()
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top K peaks
        K = min(5, 2 * n - 1)
        peak_vals, peak_idx = np.sort(conv)[::-1][:K], np.argsort(conv)[::-1][:K]

        # For each peak, compute which coordinates contribute most
        pressure = np.zeros(n)
        for k in peak_idx:
            if conv[k] < M * 0.8:
                break
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                contrib = a[i] * a[j]
                pressure[i] += contrib * (conv[k] / M)
                if j != i:
                    pressure[j] += 0.5 * contrib * (conv[k] / M)

        p_max = pressure.max()
        if p_max < 1e-20:
            break
        pressure = pressure / p_max

        # Try multiple step sizes
        best_delta = 0
        best_M = M
        for eta in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            a_try = a * np.exp(-eta * pressure)
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            M_try = fast_convolve(a_try).max()
            if M_try < best_M:
                best_M = M_try
                best_delta = eta

        if best_delta > 0:
            a = a * np.exp(-best_delta * pressure)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            M = float(conv.max())
            if M < best_score / (2.0 * n):
                best_score = 2.0 * n * M
                best_a = a.copy()
        else:
            break

        if verbose and it % 5 == 0:
            print(f"    PTR {label} it={it}: score={best_score:.6f}")

    return best_a, best_score


def pair_optimization(a, conv, time_limit, verbose=False, label=""):
    """
    Alternating optimization of pairs of coordinates.
    For each promising pair (i,j), optimize jointly by fixing one and optimizing the other.
    """
    n = len(a)
    a = a.copy()
    conv = conv.copy()
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(99)

    # Find coordinates that are near peaks
    peak_positions = np.where(conv >= M * 0.85)[0]

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Generate candidate pairs from peak-contributing coordinates
        pairs = []
        for k in peak_positions[:10]:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # Get top contributing pairs
            contribs = []
            for i in range(i_start, i_end + 1):
                j = k - i
                if i != j and a[i] > 1e-10 and a[j] > 1e-10:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)
            for _, i, j in contribs[:3]:
                pairs.append((i, j))

        # Deduplicate and limit
        seen = set()
        unique_pairs = []
        for i, j in pairs:
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                unique_pairs.append(key)
        unique_pairs = unique_pairs[:50]

        rng.shuffle(unique_pairs)

        for i, j in unique_pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue

            # Alternate: fix j, optimize i; then fix i, optimize j
            for _ in range(3):
                # Optimize i with j fixed
                delta_i = _optimal_single_delta(i, a, conv)
                if abs(delta_i) > 1e-14:
                    _apply_delta(i, delta_i, a, conv)
                    M = float(conv.max())
                    if M < best_score / (2.0 * n):
                        best_score = 2.0 * n * M
                        best_a = a.copy()
                        improved = True

                # Optimize j with i fixed
                delta_j = _optimal_single_delta(j, a, conv)
                if abs(delta_j) > 1e-14:
                    _apply_delta(j, delta_j, a, conv)
                    M = float(conv.max())
                    if M < best_score / (2.0 * n):
                        best_score = 2.0 * n * M
                        best_a = a.copy()
                        improved = True

        if not improved and it > 5:
            break

    return best_a, best_score


def _optimal_single_delta(i, a, conv):
    """Find optimal delta for coordinate i using bisection."""
    n = len(a)
    if a[i] < 1e-15:
        return 0.0

    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)

    # Unaffected max
    unaff_max = -1e30
    if k_start > 0:
        unaff_max = float(conv[:k_start].max())
    if k_end < 2 * n - 2:
        unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

    current_affected_max = float(conv[k_start:k_end + 1].max())
    if current_affected_max <= unaff_max + 1e-14:
        return 0.0

    b_vec = a[max(0, k_start - i):min(n, k_end + 1 - i)]
    has_diag = (2 * i >= k_start and 2 * i <= k_end)

    # Check if zeroing gives improvement
    affected_zero = conv[k_start:k_end + 1] - 2.0 * a[i] * b_vec
    if has_diag:
        idx = 2 * i - k_start
        affected_zero[idx] += a[i] * a[i]
    if float(affected_zero.max()) > unaff_max + 1e-12:
        return 0.0

    # Bisection
    lo = -a[i]
    hi = 0.0
    for _ in range(35):
        mid = (lo + hi) / 2.0
        aff = conv[k_start:k_end + 1] + 2.0 * mid * b_vec
        if has_diag:
            aff = aff.copy()
            aff[2 * i - k_start] += mid * mid
        if float(aff.max()) > unaff_max:
            hi = mid
        else:
            lo = mid
    return lo


def _apply_delta(i, delta, a, conv):
    """Apply delta to coordinate i and update conv in-place."""
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    b_vec = a[max(0, k_start - i):min(n, k_end + 1 - i)]

    a[i] = max(a[i] + delta, 0.0)
    conv[k_start:k_end + 1] += 2.0 * delta * b_vec
    if 2 * i >= k_start and 2 * i <= k_end:
        conv[2 * i] += delta * delta


def simulated_annealing(a0, time_limit, T_init=0.3, T_min=0.01, verbose=False, label=""):
    """
    Simulated annealing with structured block moves.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n
    a = a / s
    current_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    T = T_init
    rng = np.random.default_rng(77)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Choose a block size
        block_size = int(rng.integers(5, min(50, n // 4)))
        start = int(rng.integers(0, n - block_size))
        block = slice(start, start + block_size)

        # Generate perturbation
        a_try = a.copy()
        # Random perturbation on the block
        a_try[block] = a_try[block] * np.exp(rng.standard_normal(block_size) * 0.3)
        # Also try zeroing part of the block
        if rng.random() < 0.3:
            zero_frac = rng.uniform(0.2, 0.6)
            n_zero = max(1, int(block_size * zero_frac))
            zero_idx = start + rng.choice(block_size, n_zero, replace=False)
            a_try[zero_idx] = 0

        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try

        try_score = 2.0 * n * fast_convolve(a_try).max()

        # Acceptance criterion
        if try_score < current_score or rng.random() < np.exp(-(try_score - current_score) / T):
            a = a_try
            current_score = try_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

        # Cool down
        T = max(T_min, T * 0.995)

        # After cooling, do a quick CD pass
        if it % 50 == 0:
            a, current_score, _ = exact_coord_descent(a, fast_convolve(a), 2.0)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

        if verbose and it % 100 == 0:
            print(f"    SA {label} it={it} T={T:.4f}: best={best_score:.6f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining exact CD, mirror descent,
    peak targeting, pair optimization, and simulated annealing.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = 2.0
        print("No parent, using uniform init")

    n_parent = len(parent)

    # === Phase 1: Multi-length exploration with diverse inits ===
    print("\n=== Phase 1: Multi-length Diverse Init ===")
    t1_start = time.monotonic()
    n_values = [n_parent]
    for alt in [n_parent - 128, n_parent + 128, 768, 896, 1024, 1152, 1280]:
        if alt >= 256 and alt <= 2048 and alt not in n_values:
            n_values.append(alt)
    n_values = n_values[:6]

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > 120:
            break

        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_seq(parent, n)))
        inits.append(("qr", init_quadratic_residue(n, rng)))
        inits.append(("bimodal", init_bimodal(n, rng)))
        inits.append(("cosh", init_symmetric_cosh(n, rng)))
        inits.append(("uniform", np.ones(n) / n))

        for name, init in inits:
            t_left = 120.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            # Quick CD to get a reasonable score
            a_init = init.copy()
            conv_init = fast_convolve(a_init)
            a_c, s_c, _ = exact_coord_descent(a_init, conv_init, 8.0, max_sweeps=30, label=f"p1_{name}")
            candidates.append((s_c, a_c, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top-5: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:5]]))

    for s_c, a_c, n_c, name_c in candidates[:10]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{name_c}_n{n_c}")

    # === Phase 2: Intensive exact CD ===
    print(f"\n=== Phase 2: Intensive Exact CD (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    n_best = len(best_seq)
    current = best_seq.copy()

    for rnd in range(8):
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 100.0)
        conv_cur = fast_convolve(current)
        a_opt, s_opt, sw = exact_coord_descent(current, conv_cur, t_alloc, max_sweeps=500,
                                                verbose=True, label=f"p2_r{rnd}")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"cd_r{rnd}")
        current = best_seq.copy()

        # Small perturbation to escape
        if rnd < 5:
            noise = rng.standard_normal(n_best) * 0.02
            current = current * (1.0 + noise)
            current = np.maximum(current, 0)
            s = current.sum()
            if s > 1e-15:
                current = current / s
            else:
                current = best_seq.copy()

    # === Phase 3: Nesterov smooth descent ===
    print(f"\n=== Phase 3: Nesterov Smooth Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_md = min(rem * 0.35, 200.0)
        a_md, s_md = nesterov_smooth_descent(best_seq, t_md, lr_init=0.02, beta_init=30.0,
                                              verbose=True, label="nsd1")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "nsd1")

        # Second round with higher beta
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            a_md2, s_md2 = nesterov_smooth_descent(best_seq, min(rem * 0.3, 80.0),
                                                    lr_init=0.01, beta_init=100.0, label="nsd2")
            best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "nsd2")

    # === Phase 4: Peak-targeted refinement ===
    print(f"\n=== Phase 4: Peak-Targeted Refine (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_ptr = min(rem * 0.4, 120.0)
        conv_best = fast_convolve(best_seq)
        a_ptr, s_ptr = peak_targeted_refine(best_seq, conv_best, t_ptr, verbose=True, label="ptr")
        best_seq, best_score = check_and_update(a_ptr, best_seq, best_score, "ptr")

    # === Phase 5: Pair optimization ===
    print(f"\n=== Phase 5: Pair Optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_pair = min(rem * 0.4, 100.0)
        conv_best = fast_convolve(best_seq)
        a_pair, s_pair = pair_optimization(best_seq, conv_best, t_pair, verbose=True, label="pair")
        best_seq, best_score = check_and_update(a_pair, best_seq, best_score, "pair")

    # === Phase 6: Simulated annealing ===
    print(f"\n=== Phase 6: Simulated Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 60:
        t_sa = min(rem * 0.5, 150.0)
        a_sa, s_sa = simulated_annealing(best_seq, t_sa, T_init=0.2, T_min=0.005,
                                          verbose=True, label="sa")
        best_seq, best_score = check_and_update(a_sa, best_seq, best_score, "sa")

    # === Phase 7: More CD variants on the best ===
    print(f"\n=== Phase 7: Additional CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        for lr in [0.05, 0.1, 0.2]:
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_mcd = min(rem * 0.3, 50.0)
            conv_b = fast_convolve(best_seq)
            a_mcd, s_mcd = mirrored_coord_descent(best_seq, conv_b, t_mcd, lr=lr, label=f"mcd_{lr}")
            best_seq, best_score = check_and_update(a_mcd, best_seq, best_score, f"mcd_{lr}")

    # === Phase 8: Final intensive polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        # Multiple short CD rounds
        for i in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            t_polish = min(rem * 0.4, 30.0)
            conv_b = fast_convolve(best_seq)
            a_pol, s_pol, _ = exact_coord_descent(best_seq, conv_b, t_polish, max_sweeps=200,
                                                   label=f"polish{i}")
            best_seq, best_score = check_and_update(a_pol, best_seq, best_score, f"polish{i}")

    # === Phase 9: Try slightly different lengths ===
    print(f"\n=== Phase 9: Length Variation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 15
    if rem > 30:
        n_cur = len(best_seq)
        for delta_n in [64, -64, 128, -128]:
            alt_n = n_cur + delta_n
            if alt_n < 256 or alt_n > 2500:
                continue
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            init_alt = interpolate_seq(best_seq, alt_n)
            t_alt = min(rem * 0.5, 20.0)
            conv_alt = fast_convolve(init_alt)
            a_alt, s_alt, _ = exact_coord_descent(init_alt, conv_alt, t_alt, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
