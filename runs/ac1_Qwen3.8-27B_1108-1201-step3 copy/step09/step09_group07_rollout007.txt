2. LP best-response with Nesterov dual averaging for fine refinement

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_lse(a, T):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def grad_topk(a, T, k=5):
    """Gradient focusing on top-k peaks using weighted softmax."""
    n = len(a)
    b = fast_convolve(a)
    # Identify top-k positions
    topk_idx = np.argsort(b)[::-1][:k]
    # Create a sharpened weight distribution over all 2n-1 positions
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # Boost weights on top-k positions
    boost = np.ones(len(w))
    boost[topk_idx] = 10.0
    w = w * boost
    w = w / w.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]

def mirror_descent(a0, time_limit, T_start=5.0, T_end=200.0, n_stages=8, steps_per_stage=1500):
    """Mirror descent (multiplicative weights) with entropy regularization."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            # Adaptive learning rate
            eta = 0.3 / (T * (1.0 + 0.01 * global_step))

            # Multiplicative update: a_i <- a_i * exp(-eta * grad_i)

def nesterov_pg(a0, time_limit, T_start=2.0, T_end=300.0, lr_init=0.08,
                n_stages=10, steps_per_stage=2500, use_topk=False, topk=5):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)

            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

                grad = grad_topk(y, T, topk)
            else:
                grad = grad_lse(y, T)

            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 50 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)

    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:

            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine(a0, time_limit, solver=None, n_max_iters=100000,
              restart_interval=500, restart_perturb=0.03):
    """LP-based Nesterov dual averaging with periodic randomization."""
    n = len(a0)
    y = project_simplex(np.maximum(a0, 0))
    a_avg = y.copy()
    best_a = y.copy()
    best_score = score_normalized(y)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a_avg = ((it + 1) * a_avg + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a_avg = project_simplex(a_avg)

        # Periodic randomization to escape local optima
        if (it + 1) % restart_interval == 0:
            rng = np.random.default_rng(it * 7919)
            y = y + restart_perturb * rng.standard_normal(n)
            y = project_simplex(y)

        score_a = score_normalized(a_avg)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a_avg.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M={M_b:.6f}")

    return best_a, best_score

    stuck = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Get top-k peaks (excluding boundaries which are naturally lower)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        topk_pos = np.argsort(b_work)[::-1][:top_k]

        improved = False
        # For each top peak, find the largest contributors and reduce them
        for pk in topk_pos:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Find top contributors
            top_contrib = np.argsort(contribs)[::-1][:8]

            for ci in top_contrib:
                i = idx[ci]
                j = pk - i
                if i >= n or j >= n:
                    continue
                if a[i] < 1e-12:
                    continue
                # Try reducing the larger of the pair
                if i == j:
                    target = i
                    factors = [0.95, 0.90, 0.85, 0.80]
                else:
                    if a[i] >= a[j]:
                        target = i
                    else:
                        target = j
                    factors = [0.97, 0.94, 0.90, 0.85]

                for f in factors:
                    a_trial = a.copy()
                    a_trial[target] *= f
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top contributors across multiple peaks
            targets = set()
            for pk in topk_pos[:3]:
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top_c = np.argsort(contribs)[::-1][:3]
                for ci in top_c:
                    targets.add(idx[ci])

            if targets:
                a_trial = a.copy()
                for t in list(targets)[:5]:
                    if a[t] > 1e-12:
                        a_trial[t] *= 0.96
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            stuck += 1
            if stuck > 100:
                break
        else:
            stuck = 0

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      PS step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init(n, mode, rng):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Sine wave
        a = 1.0 + 0.6 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        # Edge-concentrated (high at edges, low in middle)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * x)
    elif mode == 2:
        # Two peaks
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        a = 0.5 + np.exp(-20 * (x - c1) ** 2) + np.exp(-20 * (x - c2) ** 2)
    elif mode == 3:
        # Anti-symmetric (high where mirror is low)
        a = 1.0 + 0.7 * np.sin(4 * np.pi * x)
    elif mode == 4:
        # Random smooth
        n_coarse = max(4, n // 64)
        coarse = rng.uniform(0.5, 1.5, n_coarse)
        a = np.interp(x, np.linspace(0, 1, n_coarse), coarse)
    elif mode == 5:
        # Cosine with phase
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x + phase) + 0.3 * np.cos(4 * np.pi * x + phase)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining mirror descent, Nesterov PG,
    LP refinement, and multi-peak shaving with randomized restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_init(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov PG on parent ===
    print(f"\n=== Phase 1: Nesterov PG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg(best_seq, t_alloc, T_start=3.0, T_end=400.0,
                              lr_init=0.1, n_stages=10, steps_per_stage=3000)
        try_update(a_ng, "ng_main")

    # === Phase 2: Mirror Descent (different geometry) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, T_start=5.0, T_end=300.0,
                                 n_stages=8, steps_per_stage=1200)
        try_update(a_md, "mirror_desc")

    # === Phase 3: LP Refinement (main workhorse) ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, solver=solver,
                            n_max_iters=200000, restart_interval=400,
                            restart_perturb=0.02)
        try_update(a_lp, "lp_main")

    # === Phase 4: Multi-Peak Shaving ===
    print(f"\n=== Phase 4: Multi-Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_ps, _ = multi_peak_shave(best_seq, t_alloc, top_k=5, n_steps=80000)
        try_update(a_ps, "multi_peak")

    # === Phase 5: Diverse restarts at same length ===
    print(f"\n=== Phase 5: Diverse Restarts (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        modes = [1, 3, 4, 5]  # Edge-concentrated, anti-sym, random smooth, multi-cos
        for mode in modes:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.20, 100.0)
            if t_sub < 30:
                break
            a_init = make_init(n_best, mode, rng)
            # Blend with current best for warm start
            a_init = 0.3 * a_init + 0.7 * best_seq
            a_init = project_simplex(a_init)
            print(f"  mode={mode}, budget={t_sub:.0f}s")
            # Quick PG then LP
            a_sub, _ = nesterov_pg(a_init, t_sub * 0.5, T_start=2.0, T_end=200.0,
                                   lr_init=0.08, n_stages=6, steps_per_stage=1500)
            try_update(a_sub, f"restart_m{mode}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = lp_refine(a_sub, min(rem * 0.3, 40.0), solver=solver,
                                      n_max_iters=10000, restart_interval=300,
                                      restart_perturb=0.03)
                try_update(a_sub2, f"restart_m{mode}_lp")

    # === Phase 6: Different lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for delta in [-100, -50, -20, -10, 10, 20, 32, 48, 64, 96, 128, 160]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 3072:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.15, 100.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt, _ = nesterov_pg(init_alt, t_total * 0.5, T_start=2.0, T_end=200.0,
                               lr_init=0.08, n_stages=6, steps_per_stage=1000)
        try_update(a_alt, f"ng_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 35.0), solver=solver,
                                  n_max_iters=8000, restart_interval=400,
                                  restart_perturb=0.02)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.25, 120.0)
        a_fin, _ = lp_refine(best_seq, t_fin, solver=solver,
                             n_max_iters=150000, restart_interval=600,
                             restart_perturb=0.01)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final multi-peak shave ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.15, 60.0)
        a_ps2, _ = multi_peak_shave(best_seq, t_ps, top_k=3, n_steps=30000)
        try_update(a_ps2, "final_shave")

    # === Phase 9: Very short LP micro-refinement ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_micro = min(rem * 0.20, 30.0)
        a_micro, _ = lp_refine(best_seq, t_micro, solver=solver,
                               n_max_iters=50000, restart_interval=1000,
                               restart_perturb=0.005)
        try_update(a_micro, "micro_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
