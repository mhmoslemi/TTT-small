def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name

    _lp_solver_cached = cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):

def compute_ideal_spectrum(n, L):
    """
    Compute the ideal power spectrum shape that would produce a flat
    auto-convolution. The ideal c[k] = 1/(2n-1) for k=0..2n-2 corresponds
    to P = |DFT(box)|^2 where box is 1/(2n-1) on [0, 2n-2] and 0 elsewhere.
    """
    box = np.zeros(L)
    box[:2 * n - 1] = 1.0 / (2 * n - 1)
    ideal = np.abs(fft(box)) ** 2
    total = ideal.sum()
    if total > 0:
        ideal = ideal / total
    return ideal


def freq_domain_flatten(a, n_iters=200, alpha_schedule=None, rng=None, time_limit=None, t_start=None):
    """

        return a
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)

    ideal_spec = compute_ideal_spectrum(n, L)

    if alpha_schedule is None:
        alpha_schedule = np.linspace(0.99, 0.7, n_iters)

    if t_start is None:
        t_start = time.monotonic()

    for it in range(n_iters):
        if time_limit and (time.monotonic() - t_start) > time_limit:
            break

        # Zero-pad and FFT
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = fft(a_pad)

        # Current power spectrum
        P = np.abs(A) ** 2
        P_total = P.sum()
        if P_total < 1e-30:
            break
        P_norm = P / P_total

        # Blend toward ideal spectrum
        alpha = alpha_schedule[min(it, len(alpha_schedule) - 1)]
        P_new = alpha * P_norm + (1.0 - alpha) * ideal_spec
        # Ensure non-negative

        P_new_total = P_new.sum()
        if P_new_total < 1e-30:
            break
        P_new = P_new / P_new_total * P_total

        # New magnitude, keep phase

        A_new = mag_new * np.exp(1j * phase)

        a = np.maximum(a_recovered[:n], 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            continue
        a = a / s

        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a


def freq_domain_compression(a, n_iters=100, gamma=0.5, time_limit=None, t_start=None):
    """
    Compress the dynamic range of the power spectrum using power-law:
    P_new = P^gamma (for gamma < 1, this compresses peaks and raises valleys).
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)

    if t_start is None:
        t_start = time.monotonic()

    for it in range(n_iters):
        if time_limit and (time.monotonic() - t_start) > time_limit:
            break

        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break
        P_norm = P / P_max

        # Power-law compression
        g = gamma * (1.0 - it / max(n_iters, 1)) + 0.3  # Start with gamma, end at 0.3
        P_new = P_norm ** g
        P_new = P_new / P_new.sum() * P.sum()

        mag_new = np.sqrt(np.maximum(P_new, 0))
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)

        a_recovered = np.real(ifft(A_new))
        a = np.maximum(a_recovered[:n], 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            continue
        a = a / s

        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a

def active_set_descent(a0, time_limit, n_steps=50000):
    """
    Active-set-aware coordinate descent. At each step, identify the peak lags,
    compute the subgradient, and perform a projected gradient step with
    adaptive step size and backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    consecutive_fail = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find active set: lags within 0.1% of max
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute average subgradient over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = evaluate_local(a_trial)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            consecutive_fail = 0
            # Slightly increase lr if we've been improving
            if no_improve == 0 and lr < 0.05:
                lr *= 1.01
        else:
            no_improve += 1
            consecutive_fail += 1
            if no_improve >= 10:
                lr *= 0.85
                no_improve = 0
            if consecutive_fail >= 100:
                # Try a random perturbation to escape
                rng_esc = np.random.default_rng(step)
                a_pert = a + rng_esc.standard_normal(n) * 0.005
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                sc_pert = evaluate_local(a_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_a = a_pert.copy()
                    a = a_pert
                    consecutive_fail = 0
                else:
                    # Revert to best and try again
                    a = best_a.copy()
                    consecutive_fail = 0

        lr = max(1e-7, min(lr, 0.1))

    return best_a, best_score

            contribs = a[idx] * a[peak_k - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.998, 0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    # Redistribute proportionally to non-peak-contributing elements
                    mask = np.ones(n, dtype=bool)
                    mask[top_positions] = False
                    others = a_trial[mask]
                    if others.sum() < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

    return best_a, best_score


def structured_init(n, seed=42):
    """Create structured initial sequences with various shapes."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)

    if choice == 0:
        # Asymmetric exponential decay
        a = np.exp(-3.0 * x) + 0.1
    elif choice == 1:
        # Asymmetric ramp with oscillation
        a = (1 - x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 2:
        # Bimodal (mass at edges)
        a = np.exp(-5 * x) + np.exp(-5 * (1 - x)) + 0.05
    elif choice == 3:
        # Slepian-like (prolate)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 4:
        # Cosine taper (asymmetric)
        a = np.cos(np.pi * x * 0.7 / 2) ** 2
    elif choice == 5:
        # Power law
        a = (x + 0.01) ** 0.5
    elif choice == 6:
        # Gaussian off-center
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.3
    else:
        # Linear with noise
        a = 1 - 0.5 * x + 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_refine(a0, time_limit):
    """
    L-BFGS with LogSumExp smoothing at multiple temperature scales.
    Works in the log-domain for better handling of small values.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
    best_score = evaluate_local(a)
    best_z = z.copy()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = min(3000, max(50, int(remaining * 3)))

        def func(z_inner):
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            max_val = b_inner.max()
            exp_vals = np.exp(b_inner - max_val)
            log_sum_exp = max_val + np.log(exp_vals.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_inner):
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            max_val = b_inner.max()
            exp_vals = np.exp(b_inner - max_val)
            w = exp_vals / exp_vals.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_inner)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
            S_val = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S_val)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = evaluate_local(a_check)
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
        except Exception:
            pass

    return softmax_from_z(best_z)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def multi_freq_refine(a0, time_limit):
    """
    Multiple rounds of frequency-domain flattening with different strategies,
    interleaved with time-domain optimization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    n_rounds = max(3, int(time_limit / 30))

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        t_sub = remaining / (n_rounds - rnd) * 0.8

        # Strategy 1: Blend toward ideal spectrum
        if rnd % 3 == 0:
            alpha_sched = np.linspace(0.995, 0.85, 50)
            a_new = freq_domain_flatten(a, n_iters=50, alpha_schedule=alpha_sched,
                                        time_limit=t_sub, t_start=t0)
        # Strategy 2: Power-law compression
        elif rnd % 3 == 1:
            a_new = freq_domain_compression(a, n_iters=50, gamma=0.6,
                                            time_limit=t_sub, t_start=t0)
        # Strategy 3: Active-set descent
        else:
            a_new, _ = active_set_descent(a, t_sub, n_steps=5000)

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Mix: take a step toward a_new from best_a
            mix = 0.3
            a_mixed = (1 - mix) * best_a + mix * a_new
            a_mixed = np.maximum(a_mixed, 0)
            sm = a_mixed.sum()
            if sm > 1e-15:
                a_mixed = a_mixed / sm
                sc_m = evaluate_local(a_mixed)
                if sc_m < best_score:
                    best_score = sc_m
                    best_a = a_mixed.copy()
                    a = a_mixed.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining frequency-domain spectrum shaping,
    Nesterov LP, LogSumExp L-BFGS, and active-set descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Frequency-domain flattening (NEW - main innovation) ===
    print(f"\n=== Phase 1: Freq-Domain Flattening n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_freq, _ = multi_freq_refine(best_seq, t_alloc)
        try_update(a_freq, "freq_flatten_1")

    # === Phase 2: Nesterov LP ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 3: Frequency-domain again (refined) ===
    print(f"\n=== Phase 3: Freq-Domain Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_freq2, _ = multi_freq_refine(best_seq, t_alloc)
        try_update(a_freq2, "freq_flatten_2")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = logsumexp_refine(best_seq, t_alloc)
        try_update(a_lb, "logsumexp")

    # === Phase 5: Active-set descent (precise) ===
    print(f"\n=== Phase 5: Active-Set Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_as, _ = active_set_descent(best_seq, t_alloc, n_steps=100000)
        try_update(a_as, "active_set")

    # === Phase 6: Targeted peak reduction ===
    print(f"\n=== Phase 6: Peak Reduction (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_pk, _ = targeted_peak_reduction(best_seq, t_alloc, n_trials=50000)
        try_update(a_pk, "peak_reduce")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Short frequency domain pass
        t_sub = min(rem * 0.3, 60.0)
        if t_sub > 10:
            a_fin1, _ = multi_freq_refine(best_seq, t_sub)
            try_update(a_fin1, "final_freq")

        # Short active-set pass
        rem = deadline - time.monotonic()
        t_sub = min(rem * 0.3, 40.0)
        if t_sub > 10:
            a_fin2, _ = active_set_descent(best_seq, t_sub, n_steps=50000)
            try_update(a_fin2, "final_as")

        # Short LogSumExp pass
        rem = deadline - time.monotonic()
        t_sub = min(rem * 0.3, 30.0)
        if t_sub > 5:
            a_fin3 = logsumexp_refine(best_seq, t_sub)
            try_update(a_fin3, "final_ls")

    # === Phase 8: Try slightly different length ===
    print(f"\n=== Phase 8: Alt Length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        alt_n = n_best + 100
        t_alt = min(rem * 0.3, 150.0)
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.5, solver=solver, n_max_iters=30000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_alt2, _ = multi_freq_refine(best_seq, min(rem * 0.4, 60.0))
            try_update(a_alt2, f"freq_n{alt_n}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
