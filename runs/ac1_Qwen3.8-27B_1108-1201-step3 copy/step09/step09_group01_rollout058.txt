"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0]**2]) if n == 1 else np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project onto probability simplex (non-negative, sums to 1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def smooth_max_gradient(a, beta):
    """
    Compute gradient of LSE(beta * conv(a,a)) w.r.t. a.
    Uses FFT trick: grad[j] = 2 * sum_l a[l] * p[j+l]
    where p = softmax(beta * conv).
    Returns (gradient, current_max, lse_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    shifted = np.clip(shifted, -700, 0)  # numerical stability
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        # Degenerate case: uniform weights on max positions
        p = np.zeros_like(conv)
        p[conv == max_conv] = 1.0 / (conv == max_conv).sum()
    else:
        p = p / p_sum
    
    # Compute grad[j] = 2 * sum_{l=0}^{n-1} a[l] * p[j+l]
    # Using FFT: cross-correlation via reversed a
    N_fft = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A_rev = rfft(a_rev, N_fft)
    P = rfft(p, N_fft)
    C = irfft(A_rev * P, N_fft)
    # C[n-1+j] = sum_m a[m] * p[j+m] for j = 0, ..., n-1
    grad = 2.0 * C[n-1 : 2*n-1]
    
    lse_val = max_conv + np.log(p_sum) / beta if p_sum > 0 else max_conv
    return grad, max_conv, lse_val

            d = 2.0 * a[0:n]  # d[k-j] = 2*a[k-j], so d is just 2*a
            
            # Current affected max
            aff_max = affected.max()
            cur_max = max(unaff_max, aff_max)
            
            # Find optimal delta
            # conv_new[k] = conv[k] + delta * d[k-j] for k != 2j
            # conv_new[2j] = conv[2j] + 2*delta*a[j] + delta^2 (if 2j is in range)
            
            # The objective is max(unaff_max, max of affected_new)
            # We want to find delta that minimizes this.
            
            # Strategy: try a set of candidate deltas
            # 1. Delta that would bring the current max down to unaff_max
            # 2. Delta that equalizes top peaks
            # 3. Small perturbations around 0
            
            candidates = [0.0]
            
            # Try to reduce the peak
            peak_in_affected = np.argmax(affected)
            peak_val = affected[peak_in_affected]
            peak_d = d[peak_in_affected]
            
            if peak_d != 0 and peak_val > unaff_max:
                # delta that makes affected[peak] = unaff_max
                delta_eq = (unaff_max - peak_val) / peak_d
                candidates.append(delta_eq)
            
            # Try to equalize top 2 peaks in affected
            sorted_idx = np.argsort(affected)[::-1]
            for i2 in range(min(3, len(sorted_idx))):
                for i3 in range(i2+1, min(i2+3, len(sorted_idx))):
                    k2, k3 = sorted_idx[i2], sorted_idx[i3]
                    d2, d3 = d[k2], d[k3]
                    if d2 != d3 and (d2 != 0 or d3 != 0):
                        # affected[k2] + delta*d2 = affected[k3] + delta*d3
                        delta_eq2 = (affected[k3] - affected[k2]) / (d2 - d3)
                        candidates.append(delta_eq2)
            
            # Add fractional reductions of a[j]
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5, 0.9]:
                candidates.append(-frac * a[j])
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                candidates.append(frac * a[j])
            
            # Evaluate candidates
            best_delta = 0.0
            best_new_max = cur_max
            
            for delta in candidates:
                new_val = a[j] + delta
                if new_val < -1e-15:
                    continue
                if new_val > 1.0:
                    continue
                
                # Compute new affected
                new_affected = affected + delta * d
                # Handle quadratic term at k = 2j
                if 2*j >= k_start and 2*j <= k_end:
                    idx_2j = 2*j - k_start
                    new_affected[idx_2j] = conv[2*j] + 2*delta*a[j] + delta*delta
                
                new_max = max(unaff_max, new_affected.max())
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-14:
                # Apply update
                a[j] = max(a[j] + best_delta, 0.0)
                
                # Update conv
                changes = best_delta * d
                if 2*j >= k_start and 2*j <= k_end:
                    idx_2j = 2*j - k_start
                    changes = changes.copy()
                    changes[idx_2j] = 2*best_delta*(a[j] - best_delta) + best_delta*best_delta
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, conv[k_start:k_end+1].max())
                
                sum_a = a.sum()
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

            stalling += 1
            if stalling > 50:
                break

            stalling = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")
    
    return best_a, best_score, sweeps

def init_from_parent(parent, n, rng):
    """Interpolate parent to length n with some randomization."""
    old_n = len(parent)
    if old_n == n:
        return parent.copy()
    s = parent.sum()
    if s < 1e-15:
        return np.ones(n) / n
    f = parent / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_gaussian_mixture(n, rng, n_components=3):
    """Initialize with a Gaussian mixture (non-symmetric)."""
    a = np.zeros(n)
    for _ in range(n_components):
        center = rng.uniform(0.1, 0.9) * n
        width = rng.uniform(0.02, 0.15) * n
        amp = rng.uniform(0.5, 1.5)
        t = np.arange(n)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_linear_ramp(n, rng):
    """Initialize with a linear ramp (strongly non-symmetric)."""
    t = np.arange(n) / n
    direction = rng.choice([1, -1])
    if direction == 1:
        a = t + 0.1 * rng.random(n)
    else:
        a = (1 - t) + 0.1 * rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_quadratic(n, rng):
    """Initialize with a quadratic profile."""
    t = np.arange(n) / n
    c = rng.uniform(0.3, 0.7)
    a = 1.0 - ((t - c) / rng.uniform(0.3, 0.5))**2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth max (log-sum-exp) gradient with mirror descent on simplex
2. Exact coordinate descent with analytic candidate deltas
3. Multi-length exploration (768 to 4096)
4. Adaptive beta scheduling and GD+CD alternation
5. Fine polishing with sub-gradient steps

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0] ** 2]) if n == 1 else np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng_arr = np.arange(1, n + 1)
    cond = u * rng_arr > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def smooth_max_grad_fft(a, beta):
    """
    Compute gradient of LSE(beta * conv(a,a)) w.r.t. a using FFT.
    grad[j] = 2 * sum_{l=0}^{n-1} a[l] * p[j+l]
    where p = softmax(beta * conv).
    Returns (grad, max_conv, lse_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = float(conv.max())
    shifted = beta * (conv - max_conv)
    shifted = np.clip(shifted, -700, 0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        nmax = int((conv == max_conv).sum())
        p[conv == max_conv] = 1.0 / nmax
    else:
        p = p / p_sum

    # FFT cross-correlation trick
    N_fft = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A_rev = rfft(a_rev, N_fft)
    P = rfft(p, N_fft)
    C = irfft(A_rev * P, N_fft)
    grad = 2.0 * C[n - 1:2 * n - 1]

    lse_val = max_conv + np.log(p_sum) / beta if p_sum > 0 else max_conv
    return grad, max_conv, lse_val

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, max_conv, lse = smooth_max_grad_fft(a, beta)

        # Scale gradient for stability
        grad_scale = max_conv + 1e-10
        grad_norm = grad / grad_scale

        # Mirror descent: a_new = a * exp(-lr * grad)
        log_a = np.log(a + 1e-30) - lr * grad_norm
        a_new = np.exp(log_a - log_a.max())
        a_new = simplex_project(a_new)

        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 300:
            beta *= 1.5
            lr *= 0.85
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 200 == 0:
            beta *= beta_growth
            lr *= lr_decay

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GMD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.1f} lr={lr:.6f}")

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with analytic candidate delta generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    sum_a = 1.0
    t0 = time.monotonic()
    sweeps = 0
    stalling = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for j in range(n):
            if a[j] < 1e-18:
                continue

            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            d = 2.0 * a[0:n]  # d[k-j] = 2*a[k-j]

            cur_max = max(unaff_max, float(affected.max()))

            # Generate candidate deltas
            candidates = [0.0]

            # Peak reduction: delta that brings top affected value to unaff_max
            aff_max_idx = int(affected.argmax())
            aff_max_val = float(affected[aff_max_idx])
            peak_d = d[aff_max_idx]
            if peak_d != 0 and aff_max_val > unaff_max:
                candidates.append((unaff_max - aff_max_val) / peak_d)

            # Equalize top 3 peaks
            sorted_idx = np.argsort(affected)[::-1][:4]
            for ii in range(len(sorted_idx)):
                for jj in range(ii + 1, len(sorted_idx)):
                    d2, d3 = d[sorted_idx[ii]], d[sorted_idx[jj]]
                    if abs(d2 - d3) > 1e-20:
                        candidates.append((float(affected[sorted_idx[jj]]) - float(affected[sorted_idx[ii]])) / (d2 - d3))

            # Fractional moves
            for frac in [0.005, 0.02, 0.05, 0.1, 0.2, 0.4, 0.8]:
                candidates.append(-frac * a[j])
            for frac in [0.005, 0.02, 0.05, 0.1, 0.3]:
                candidates.append(frac * a[j])

            best_delta = 0.0
            best_new_max = cur_max

            for delta in candidates:
                new_val = a[j] + delta
                if new_val < -1e-15 or new_val > 1.0:
                    continue

                new_affected = affected + delta * d
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx_2j = 2 * j - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2j] = conv[2 * j] + 2.0 * delta * a[j] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[j] = max(a[j] + best_delta, 0.0)
                sum_a += best_delta

                changes = best_delta * d
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx_2j = 2 * j - k_start
                    changes = changes.copy()
                    changes[idx_2j] = 2.0 * best_delta * (a[j] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            stalling += 1
            if stalling > 100:
                break
        else:
            stalling = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score, sweeps


def fine_polish(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent with tiny multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    sum_a = 1.0
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for j in range(n):
            if a[j] < 1e-18:
                continue

            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            d = 2.0 * a[0:n]
            cur_max = max(unaff_max, float(affected.max()))

            best_delta = 0.0
            best_new_max = cur_max

            # Very fine multiplicative steps
            for frac in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98,
                         1.0005, 1.001, 1.002, 1.005, 1.01]:
                delta = a[j] * (frac - 1.0)
                if a[j] + delta < -1e-15:
                    continue

                new_affected = affected + delta * d
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx_2j = 2 * j - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2j] = conv[2 * j] + 2.0 * delta * a[j] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[j] = max(a[j] + best_delta, 0.0)
                sum_a += best_delta

                changes = best_delta * d
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx_2j = 2 * j - k_start
                    changes = changes.copy()
                    changes[idx_2j] = 2.0 * best_delta * (a[j] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score, sweeps


def init_interpolate(src, n):
    """Interpolate a probability vector to length n."""
    old_n = len(src)
    if old_n == n:
        return src.copy()
    s = src.sum()
    if s < 1e-15:
        return np.ones(n) / n
    f = src / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def init_gaussian_mixture(n, rng, n_comp=3):
    """Non-symmetric Gaussian mixture initialization."""
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_comp):
        center = rng.uniform(0.05, 0.95) * n
        width = rng.uniform(0.03, 0.2) * n
        amp = rng.uniform(0.3, 1.5)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_ramp(n, rng):
    """Linear ramp (strongly non-symmetric)."""
    t = np.arange(n, dtype=np.float64) / n
    if rng.random() < 0.5:
        a = t
    else:
        a = 1.0 - t
    a += 0.05 * rng.random(n)  # small noise
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def init_filtered(n, rng):
    """Initialize with a filtered/structured profile."""
    t = np.arange(n, dtype=np.float64)
    n_freq = int(rng.integers(2, 6))
    a = np.ones(n)
    for _ in range(n_freq):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Multi-phase optimization combining FFT-accelerated gradient descent,
    exact coordinate descent, and fine polishing across multiple sequence lengths.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(777)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    def do_update(a, lbl):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{lbl}]: {score:.8f} (n={len(a_norm)})")

    # === Phase 1: Multi-length gradient descent ===
    print("\n=== Phase 1: Multi-length GD ===")
    t1 = time.monotonic()

    # Try a spread of lengths
    n_list = [1024, 1280, 1536, 2048, 2560, 3072, 3584, 4096]
    if parent is not None:
        n_par = len(parent)
        if n_par not in n_list:
            n_list.insert(0, n_par)

    n_list = sorted(set(n_list))
    n_list = [n for n in n_list if 512 <= n <= 4096]

    # Quick pass: for each n, try 1-2 inits with short GD
    for n in n_list:
        rem = deadline - time.monotonic() - 1000
        if rem < 30:
            break
        t_alloc = max(5.0, min(60.0, rem * 0.15))

        # Parent interpolation
        if parent is not None:
            init_p = init_interpolate(parent, n)
            a_opt, s_opt = gradient_mirror_descent(init_p, t_alloc, beta_init=40, lr_init=0.03,
                                                   label=f"p1_par_{n}")
            do_update(a_opt, f"p1_par_{n}")

        rem = deadline - time.monotonic() - 900
        if rem < 20:
            continue
        t_alloc2 = max(5.0, min(40.0, rem * 0.15))
        init_g = init_gaussian_mixture(n, rng, n_comp=4)
        a_opt2, s_opt2 = gradient_mirror_descent(init_g, t_alloc2, beta_init=40, lr_init=0.03,
                                                 label=f"p1_gaus_{n}")
        do_update(a_opt2, f"p1_gaus_{n}")

    print(f"Phase 1 done ({time.monotonic() - t1:.0f}s), best={best_score:.8f}, n={len(best_seq) if best_seq is not None else 0}")

    # === Phase 2: Intensive GD at best length with varied betas ===
    print(f"\n=== Phase 2: Intensive GD (best={best_score:.8f}) ===")
    t2 = time.monotonic()

    if best_seq is not None:
        for beta_try in [30, 60, 120, 250, 500]:
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            t_alloc = min(rem * 0.15, 100.0)
            a_opt, s_opt = gradient_mirror_descent(best_seq, t_alloc, beta_init=beta_try,
                                                   lr_init=0.01, beta_growth=1.005,
                                                   label=f"p2_b{beta_try}")
            do_update(a_opt, f"p2_b{beta_try}")

    print(f"Phase 2 done ({time.monotonic() - t2:.0f}s), best={best_score:.8f}")

    # === Phase 3: Exact coordinate descent ===
    print(f"\n=== Phase 3: Exact CD (best={best_score:.8f}) ===")
    t3 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.4, 300.0)
            a_opt, s_opt, sw = exact_coord_descent(best_seq, t_alloc, label="p3")
            do_update(a_opt, f"p3_sw{sw}")

    print(f"Phase 3 done ({time.monotonic() - t3:.0f}s), best={best_score:.8f}")

    # === Phase 4: GD + CD alternation ===
    print(f"\n=== Phase 4: GD+CD alternating (best={best_score:.8f}) ===")
    t4 = time.monotonic()

    if best_seq is not None:
        for rnd in range(6):
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break

            # GD step
            t_gd = min(rem * 0.2, 80.0)
            beta_cur = 80 + rnd * 80
            a_gd, _ = gradient_mirror_descent(best_seq, t_gd, beta_init=beta_cur,
                                              lr_init=0.008, label=f"p4g{rnd}")
            do_update(a_gd, f"p4g{rnd}")

            # CD step
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_cd = min(rem * 0.35, 120.0)
            a_cd, _, sw = exact_coord_descent(best_seq, t_cd, label=f"p4c{rnd}")
            do_update(a_cd, f"p4c{rnd}")

    print(f"Phase 4 done ({time.monotonic() - t4:.0f}s), best={best_score:.8f}")

    # === Phase 5: Fine polishing ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.8f}) ===")
    t5 = time.monotonic()

    if best_seq is not None:
        for pr in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 100.0)
            a_fp, s_fp, _ = fine_polish(best_seq, t_alloc, label=f"p5r{pr}")
            do_update(a_fp, f"p5r{pr}")

    print(f"Phase 5 done ({time.monotonic() - t5:.0f}s), best={best_score:.8f}")

    # === Phase 6: Length perturbation ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.8f}) ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        deltas = [128, 256, 512, -128, -256]
        for dn in deltas:
            alt_n = n_best + dn
            if alt_n < 512 or alt_n > 5000:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            init_alt = init_interpolate(best_seq, alt_n)
            t_gd = min(rem * 0.2, 60.0)
            a_o, _ = gradient_mirror_descent(init_alt, t_gd, beta_init=50, lr_init=0.01,
                                             label=f"p6g_{alt_n}")
            do_update(a_o, f"p6g_{alt_n}")
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                continue
            t_cd = min(rem * 0.3, 60.0)
            a_c, _, _ = exact_coord_descent(a_o, t_cd, label=f"p6c_{alt_n}")
            do_update(a_c, f"p6c_{alt_n}")

    print(f"Phase 6 done ({time.monotonic() - t6:.0f}s), best={best_score:.8f}")

    # === Phase 7: Final intensive CD + fine polish ===
    print(f"\n=== Phase 7: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            t_cd = min(rem * 0.4, 120.0)
            a_final_cd, _, _ = exact_coord_descent(best_seq, t_cd, label="p7cd")
            do_update(a_final_cd, "p7cd")

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_final_fp, _, _ = fine_polish(best_seq, min(rem * 0.5, 60.0), label="p7fp")
            do_update(a_final_fp, "p7fp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
