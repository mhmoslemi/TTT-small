"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with very slow multi-stage log-sum-exp cooling (30 stages)
2. Targeted peak shaving with smart mass redistribution and pair reduction
3. Multi-length exploration with interpolation
4. Adaptive time budgeting focused on gradient methods (no LP for speed)
Key differences: pure gradient optimization with much more careful cooling,
targeted coordinate descent that considers pair interactions, and better
time allocation to maximize number of effective optimization steps.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_seq(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_of(a):
    """Compute score 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    b = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2.0 * n * b.max() / (s * s))


def lse_func_and_grad(a, T):
    """Log-sum-exp smoothed objective and gradient (a-space)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    w = exp_Tb / exp_Tb.sum()
    func_val = (2.0 * n / T) * lse
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w[k] * a[k - idx]
    return func_val, grad


def a_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_stage(z, T, iters, t0, time_limit):
    """Single L-BFGS stage in log-domain."""
    def func_inner(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = a_from_z(z_in)
        val, _ = lse_func_and_grad(a, T)
        return val

    def jac_inner(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = a_from_z(z_in)
        _, grad_a = lse_func_and_grad(a, T)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=func_inner, x0=z, jac=jac_inner,
            method='L-BFGS-B',
            options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': iters * 500, 'disp': False}
        )
        return res.x
    except Exception:
        return z


def lbgfs_full(a0, time_limit, T_start=150.0, T_end=0.03, n_stages=30, iters_per_stage=80):
    """Full L-BFGS with geometric cooling schedule."""
    t0 = time.monotonic()
    z = z_from_a(a0).copy()
    temps = np.geomspace(T_start, T_end, n_stages)

    for i in range(len(temps)):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        T = temps[i]
        # Use more iterations at lower temperatures
        if T < 1.0:
            it = iters_per_stage * 2
        elif T < 10.0:
            it = iters_per_stage * 1.5
        else:
            it = iters_per_stage
        z = lbgfs_stage(z, T, it, t0, time_limit)
        a_curr = a_from_z(z)
        sc = score_of(a_curr)
        if i % 5 == 0 or i == len(temps) - 1:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS stage {i}/{len(temps)-1} T={T:.4f}: score={sc:.8f} ({elapsed:.0f}s)")

    return a_from_z(z)
