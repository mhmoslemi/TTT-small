"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score directly."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def ande_p_norm_grad_log(z, p):
    """
    Objective and gradient in log-domain (softmax parameterization).
    Objective: 2n * (sum max(c_k,0)^p)^(1/p) where a = softmax(z), sum(a)=1.
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = np.sum(c_p)
    if S < 1e-300:
        val = 2.0 * n * 1e-300
        return val, np.zeros_like(z)
    val = 2.0 * n * S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    grad = a * (g_a - np.dot(g_a, a))
    return val, grad


def pnorm_lbfgs(a0, time_limit, p_schedule):
    """L-BFGS in log-domain with p-norm smoothing schedule."""
    n = len(a0)
    a_init = np.maximum(a0, 1e-300)
    z = np.log(a_init / a_init.sum())
    z = z - z.max()
    t0 = time.monotonic()
    total_iters_per_stage = 0
    n_stages = len(p_schedule)
    for i, p_val in enumerate(p_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        iters = max(50, int(4000 * (remaining / max(1, time_limit)) * (n_stages - i) / n_stages))
        iters = max(iters, 50)

        def func_x(z_x):
            v, _ = ande_p_norm_grad_log(z_x, p_val)
            return v

        def jac_x(z_x):
            _, g = ande_p_norm_grad_log(z_x, p_val)
            return g

        try:
            res = minimize(
                fun=func_x,
                x0=z,
                jac=jac_x,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at p={p_val}: {e}")
            break

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    p={p_val:.0f} ({elapsed:.0f}s): score={sc:.8f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def subgradient_refine(a0, time_limit, n_max_iters=200000):
    """
    Direct subgradient method for minimizing max(conv(a,a)) with sum(a)=1.
    Uses adaptive step size based on best score found so far.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Lower bound for Polyak step (we know score >= 2 theoretically, but use a tighter bound)
    lb = best_score * 0.999  # Use current best as practical lower bound
    step_scale = 1.0
    patience = 0
    best_patience_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        n_peaks = 10
        peak_threshold = c_max * 0.95
        active = np.where(c >= peak_threshold)[0]

        # Subgradient: sum of d(c_k)/d(a_i) over active peaks
        grad = np.zeros(n)
        for k in active:
            # d(c_k)/d(a_i) = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            a_vals = a[idx]
            # For c_k = sum_j a[j]*a[k-j]: d/d(a_i) = a[k-i] + a[i-k]
            # a[k-i] is defined when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            # a[i-k] is defined when 0 <= i-k < n, i.e., k <= i <= k+n-1
            # In range [js, je], i goes from js to je
            # a[k-i]: k-i ranges from k-je to k-js, all in [0, n-1] by construction
            # a[i-k]: i-k ranges from js-k to je-k
            gi = a[k - idx]  # a[k-i] for i in idx
            # For a[i-k]: i-k ranges from js-k to je-k
            # js-k can be negative, je-k can be >= 0
            lo = max(0, k)  # i-k >= 0 means i >= k
            hi = min(n - 1, k + n - 1)  # i-k < n means i < k+n
            # But i is already in [js, je]
            lo2 = max(js, k)
            hi2 = min(je, k + n - 1)
            if lo2 <= hi2:
                idx2 = np.arange(lo2, hi2 + 1)
                gi2 = a[idx2 - k]  # a[i-k]
                grad[idx2] += gi2
            grad[idx] += gi

        # Normalize gradient (project onto simplex)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-300:
            break

        # Adaptive step size
        gap = c_max - lb
        if gap < 1e-10:
            step = step_scale * 0.01 / (1.0 + it * 0.001)
        else:
            step = step_scale * 1.0 / (np.sqrt(it + 1) * g_norm)
            step = min(step, 0.05)

        # Projected gradient step on simplex
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            patience = 0
            step_scale = min(step_scale * 1.2, 5.0)
        else:
            patience += 1
            if patience % 500 == 0:
                step_scale = max(step_scale * 0.9, 0.1)
                if patience > 5000:
                    # Perturb to escape
                    rng = np.random.default_rng(it)
                    a_new = a + 0.01 * rng.standard_normal(n)
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    sc = compute_score(a_new)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new
                    patience = 0

        a = a_new

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a

def spectral_reshape(a0, n_iters=500, alpha=0.5):
    """
    Iterative spectral flattening: modify the amplitude spectrum to be flatter,
    then reconstruct and clip to maintain non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a0
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Flatten: new_mag = mag^(1-alpha) * mean_mag^alpha
        mag_safe = np.maximum(mag, 1e-15)
        mean_mag = mag_safe.mean()
        new_mag = mag_safe ** (1.0 - alpha) * mean_mag ** alpha
        # Preserve DC
        new_mag[0] = mag[0]
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_state['score']:.8f}, n={len(best_state['seq'])}, time={elapsed:.0f}s")

    if best_state['seq'] is None:
        # Fallback: uniform
        a_fb = np.ones(960) / 960
        return [float(x) for x in a_fb]

    return [float(x) for x in best_state['seq']]
