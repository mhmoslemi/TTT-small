"""

1. Multi-length exploration (1024-2048) with diverse initializations

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def quick_score(a):
    """Fast score computation for normalized (sum=1) non-negative a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())


def interp_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f_n)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss > 1e-15:
        out /= ss
    else:
        out = np.ones(new_n) / new_n
    return out


def update_conv_after_delta(conv, k_start, k_end, delta, a_vals, quad_idx, ai, n):
    """Update convolution slice after changing a[i] by delta."""
    changes = 2.0 * delta * a_vals
    if 0 <= quad_idx < len(a_vals):
        changes = changes.copy()
        changes[quad_idx] = 2.0 * delta * ai + delta * delta
    conv[k_start:k_end + 1] += changes


def exact_cd_sweep(a, conv, M, n_full_iters=35):
    """
    One full sweep of exact coordinate descent.
    For each coordinate, solve the 1D convex subproblem via bisection on the subgradient.
    Returns (a, conv, M, n_improvements).
    """
    n = len(a)
    n_improvements = 0

    for i in range(n):
        ai = a[i]
        if ai < 1e-15:
            continue

        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        length_aff = k_end - k_start + 1

        # Unaffected max
        if k_start > 0:
            unaff_left = float(conv[:k_start].max())
        else:
            unaff_left = -1e30
        if k_end < 2 * n - 2:
            unaff_right = float(conv[k_end + 1:].max())
        else:
            unaff_right = -1e30
        U = max(unaff_left, unaff_right)

        # Affected data
        affected = conv[k_start:k_end + 1]
        a_vals = a[k_start - i:k_end - i + 1]
        quad_idx = 2 * i - k_start
        has_quad = (0 <= quad_idx < length_aff)

        # Bisection to find optimal delta
        # f(delta) = max(U, affected + 2*delta*a_vals + [quad]*delta^2)
        # f is convex in delta, minimum where subgradient contains 0

        lo = -ai
        hi = ai * 3.0 + np.sqrt(max(M, 1e-20))

        # Quick check: is the current value already optimal?
        # Evaluate slope at delta=0

        if has_quad:
            pass  # quad term is 0 at delta=0
        M_0 = max(U, float(np.max(affected)))

        else:
            active_0 = affected >= M_0 - 1e-18
            if active_0.any():
                slopes = 2.0 * a_vals[active_0]

                    # slope at quad position is 2*a[i] + 2*0 = 2*ai

                    pass

            else:
                slope_0 = 0.0

        if abs(slope_0) < 1e-14:
            # Already at a stationary point
            continue

            lo_b = 0.0
            hi_b = hi
        else:
            # Optimum is at delta < 0
            lo_b = lo
            hi_b = 0.0

                    slope = 0.0

                lo_b = mid
            else:
                hi_b = mid

        best_d = (lo_b + hi_b) / 2.0

        # Evaluate improvement

            vals_new[quad_idx] += best_d * best_d
        new_M = max(U, float(np.max(vals_new)))

def smooth_grad_step(a, conv, beta, lr):
    """
    One step of mirror descent using smooth gradient (log-sum-exp).
    Returns updated a.
    """
    n = len(a)
    max_c = float(conv.max())
    scaled = beta * (conv - max_c)
    scaled = np.clip(scaled, -500.0, 0.0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i]
    # Compute via FFT (cross-correlation)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    h = irfft(W * np.conj(A), fft_size)[:n]
    grad = 2.0 * h

    # Mirror descent: a_new = a * exp(-lr * grad), then project
    log_a = np.log(a + 1e-30)
    log_a -= lr * grad
    a_new = np.exp(log_a - log_a.max())
    a_new = simplex_project(a_new)
    return a_new


def nesterov_smooth_descent(a_init, time_limit, beta_init=20.0, lr_init=0.01):
    """
    Nesterov-accelerated smooth gradient descent.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()

    y = a.copy()  # Nesterov momentum
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov: y = (1+g)/g * a - 1/g * y_prev (simplified: y = a + (g-1)*(a-y_prev)/g)
        y_new = (1.0 + gamma) * a - gamma * y
        y = y_new
        y_proj = simplex_project(y)

        # Compute conv of y_proj for gradient
        conv_y = fast_convolve(y_proj)

        # Smooth gradient
        max_c = float(conv_y.max())
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -500.0, 0.0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            w = np.zeros_like(conv_y)
            w[idx] = 1.0
        else:
            w /= w_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(y_proj, fft_size)
        h = irfft(W * np.conj(A), fft_size)[:n]
        grad = 2.0 * h

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        # Evaluate
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-9:
                break
        elif it % 30 == 0:
            beta *= 1.05

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def sa_perturb(a, rng, temperature):
    """Simulated annealing style perturbation."""
    n = len(a)
    a_new = a.copy()
    # Gaussian noise scaled by temperature
    noise = rng.standard_normal(n) * temperature
    a_new += noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 7))

    for j in range(n_comp):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.05, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile[:n//2], profile[:n//2][::-1]])

        a_work = best_seq.copy()
        temperature = 0.02
        cycle = 0

        while time.monotonic() - t4_start < t4_budget:
            cycle += 1
            # CD phase
            conv = fast_convolve(a_work)
            M = float(conv.max())
            t_cd_start = time.monotonic()
            cd_sweeps = 0
            while time.monotonic() - t_cd_start < min(30.0, t4_budget - (time.monotonic() - t4_start)):
                a_work, conv, M, nimp = exact_cd_sweep(a_work, conv, M, n_full_iters=30)
                cd_sweeps += 1
                if nimp == 0:
                    break

            score = 2.0 * n_best * M
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a_work.copy()
                print(f"  SA cycle {cycle}: NEW BEST {score:.8f} (cd_sweeps={cd_sweeps}, T={temperature:.4f})")

            # Perturbation
            a_work = sa_perturb(best_seq, rng, temperature)
            temperature *= 0.95
            if temperature < 0.001:
                temperature = 0.01  # Reheat

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 30:
            for alt_n in [n_best + 128, n_best + 256, n_best - 128, 2048, 1536]:
                if alt_n < 512 or alt_n > 2560:
                    continue
                if alt_n == n_best:
                    continue
                rem = deadline - time.monotonic()
                if rem < 15:
                    break
                a_alt = interp_seq(best_seq, alt_n)
                conv = fast_convolve(a_alt)
                M = float(conv.max())
                t_start = time.monotonic()
                while time.monotonic() - t_start < min(rem * 0.3, 40.0):
                    a_alt, conv, M, nimp = exact_cd_sweep(a_alt, conv, M, n_full_iters=30)
                    if nimp == 0:
                        break
                score = 2.0 * alt_n * M
                actual_sc = check_score(a_alt, f"alt_n{alt_n}")
                if np.isfinite(actual_sc) and actual_sc < best_score:
                    best_score = actual_sc
                    best_seq = a_alt.copy()
                    print(f"  Alt n={alt_n}: NEW BEST {best_score:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            n_best = len(best_seq)
            a_work = best_seq.copy()
            conv = fast_convolve(a_work)
            M = float(conv.max())
            t_start = time.monotonic()
            while time.monotonic() - t_start < rem - 5:
                a_work, conv, M, nimp = exact_cd_sweep(a_work, conv, M, n_full_iters=40)
                score = 2.0 * n_best * M
                if score < best_score - 1e-13:
                    best_score = score
                    best_seq = a_work.copy()
                if nimp == 0:
                    break
            print(f"  Final polish done: score={2.0 * n_best * M:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = check_score(best_seq, "fallback")

    # Final verification
    fs = check_score(best_seq, "final")
    if np.isfinite(fs):
        best_score = fs
    else:
        # Should not happen
        best_seq = np.ones(1024) / 1024
        best_score = check_score(best_seq, "fallback2")

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
