def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def build_conv_matrix(a):

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

    return T

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    return b_val, float(M_var.value), None
        except Exception:
            continue
    return None, None, None


def slp_iteration(a, step_size, solver, active_frac=0.9995):
    """
    One SLP iteration: linearize auto-convolution around a, solve LP for delta.
    The linearization is: c[k](a+delta) ≈ c[k](a) + 2*(T@delta)[k]
    where T = build_conv_matrix(a).
    Returns (delta, new_M_est) or (None, None).
    """
    n = len(a)
    c = fast_convolve(a)
    M = c.max()

    # Active set: peaks within active_frac of max
    threshold = M * active_frac
    active_idx = np.where(c >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c))])

    T = build_conv_matrix(a)
    T_active = T[active_idx, :]
    c_active = c[active_idx]

    delta_var = cp.Variable(n)
    M_new = cp.Variable()

    # Constraints:
    # 1) c[k] + 2*(T@delta)[k] <= M_new for active k
    # 2) sum(delta) = 0
    # 3) a + delta >= 0  =>  delta >= -a
    # 4) |delta| <= step_size

    constraints = [
        2.0 * T_active @ delta_var - M_new <= -c_active,
        cp.sum(delta_var) == 0,
        delta_var >= -a * 0.95,
        delta_var <= step_size,
    ]

    prob = cp.Problem(cp.Minimize(M_new), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            delta = np.asarray(delta_var.value, dtype=np.float64)
            delta = np.clip(delta, -0.95 * a, step_size)
            return delta, float(M_new.value)
    except Exception:
        pass

    # Fallback: try alternative solver
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                delta = np.asarray(delta_var.value, dtype=np.float64)
                delta = np.clip(delta, -0.95 * a, step_size)
                return delta, float(M_new.value)
        except Exception:
            continue
    return None, None


def slp_refine(a0, time_limit, solver, max_iters=5000):
    """
    SLP refinement: iteratively linearize and solve LP to reduce the peak.
    Uses adaptive step size and verifies actual improvement.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    step_size = 0.05
    stall_count = 0
    min_step = 1e-5

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        delta, _ = slp_iteration(a, step_size, solver, active_frac=0.9995)
        if delta is None:
            step_size *= 0.8
            if step_size < min_step:
                break
            continue

        # Take the step and verify improvement
        a_trial = a + delta
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()

        if score < best_score - 1e-12:
            a = a_trial
            best_score = score
            best_a = a.copy()
            stall_count = 0
            # Slightly increase step size on success (up to initial)
            step_size = min(step_size * 1.1, 0.05)
        else:
            # No improvement: try half step
            a_half = a + 0.5 * delta
            a_half = np.maximum(a_half, 0)
            sh = a_half.sum()
            if sh > 1e-15:
                a_half = a_half / sh
                score_half = 2.0 * n * fast_convolve(a_half).max()
                if score_half < best_score - 1e-12:
                    a = a_half
                    best_score = score_half
                    best_a = a.copy()
                    stall_count = 0
                else:
                    stall_count += 1

            step_size *= 0.85
            if step_size < min_step:
                break

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} step={step_size:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 10:
                noise_scale = 0.02 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0

            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def compute_logsumexp_obj(z, n, T_temp):
    """Compute smoothed objective in log-domain."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n) * log_sum_exp


def compute_logsumexp_grad(z, n, T_temp):
    """Compute gradient of smoothed objective in log-domain."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (2.0 * T_temp * grad_a - 2.0 * T_temp * S)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing at decreasing temperatures."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.03, 0.02]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            try:
                return compute_logsumexp_obj(z, n, stage_T)
            except Exception:
                return 1e15

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            try:
                return compute_logsumexp_grad(z, n, stage_T)
            except Exception:
                return np.zeros_like(z)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a


def peak_directed_descent(a0, time_limit, n_steps=30000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find all significant peaks
        threshold = max_b * 0.998
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:3]:  # Focus on top peaks
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.997, 0.995, 0.99, 0.985]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    if others.sum() < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 300:
                break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def exact_subgradient_refine(a0, time_limit, n_steps=30000):
    """Exact subgradient method targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute average subgradient over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
