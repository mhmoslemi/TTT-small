"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based O(n log n) gradient computation (key speedup over LP)
2. FISTA (Nesterov-accelerated proximal gradient) on log-sum-exp smoothed objective
3. Multi-stage geometric temperature annealing
4. Spectral (frequency-domain) perturbations for escaping local minima
5. Direct peak-shaving coordinate descent for final refinement
6. Multi-length exploration with warm-start interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_arr = np.clip(a_arr, 0.0, 1000.0)
        s = a_arr.sum()
        if s < 0.01:
            return float('inf')
        b = fast_convolve(a_arr)
        return float(2 * len(a_arr) * b.max() / (s * s))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_conv_and_weights(a, T):
    """Compute convolution and log-sum-exp weights."""
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    log_sum_exp = max_Tc + np.log(exp_Tc.sum())
    return c, w, log_sum_exp


def compute_gradient_z(a, w, T, n):
    """Compute gradient of smoothed objective w.r.t. z using FFT.
    
    grad_a[i] = 4n * sum_k w[k] * a[k-i] = 4n * (w cross-correlate a)[i]
    grad_z = a * (grad_a - dot(a, grad_a))
    """
    # Cross-correlation: (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 4.0 * n * corr
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def objective_smoothed(a, T):
    """Compute smoothed objective value."""
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    log_sum_exp = max_Tc + np.log(exp_Tc.sum())
    return (2.0 * len(a) / T) * log_sum_exp


def true_objective(a):
    """Compute true objective value (with sum=1)."""
    return 2.0 * len(a) * fast_convolve(a).max()


def fista_smoothed(a0, time_limit, T_values, max_iters_per_stage=None):
    """FISTA with log-sum-exp smoothing, multi-stage temperature.
    
    Args:
        a0: initial probability vector
        time_limit: total time budget in seconds
        T_values: list of temperatures (decreasing)
        max_iters_per_stage: max iterations per stage (None = time-based)
    
    Returns:
        best_a: best probability vector found
        best_score: corresponding true objective value
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = a0.copy()
    best_score = true_objective(a0)
    t0 = time.monotonic()

    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # FISTA parameters
        # Conservative Lipschitz estimate
        L = 8.0 * n * max(T, 1.0)
        step = 1.0 / L
        # Cap step size for stability
        step = min(step, 1.0 / (4.0 * n))

        z_curr = z.copy()
        z_prev = z.copy()
        z_nes = z.copy()
        k_counter = 1.0

        stage_start = time.monotonic()
        max_stage_time = min(remaining, time_limit * 0.25)

        while time.monotonic() - stage_start < max_stage_time:
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z_nes)
            c, w, lse = compute_conv_and_weights(a, T)
            grad_z = compute_gradient_z(a, w, T, n)

            # Backtracking: check if step is too large
            # (simplified: just use fixed step, FISTA is robust to conservative L)

            # FISTA update
            z_prev = z_curr.copy()
            z_curr = z_curr - step * grad_z

            # Nesterov momentum
            k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_counter * k_counter))
            beta = (k_counter - 1.0) / k_new
            z_nes = z_curr + beta * (z_curr - z_prev)
            k_counter = k_new

            # Periodic check
            if int(k_counter) % 100 == 0:
                a_check = softmax_from_z(z_nes)
                score = true_objective(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        z = z_nes.copy()
        # Update best at end of stage
        a_final = softmax_from_z(z)
        score_final = true_objective(a_final)
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()

    return best_a, best_score


def fista_with_restarts(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=12, n_restarts=3):
    """FISTA with periodic spectral restarts to escape local minima."""
    n = len(a0)
    best_a = a0.copy()
    best_score = true_objective(a0)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    time_per_repair = time_limit / n_restarts

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        stage_budget = min(rem, time_per_repair * (1.0 + 0.3 * (n_restarts - 1 - restart)))

        # On restarts after the first, add spectral perturbation
        if restart > 0:
            # Perturb in frequency domain
            rng_r = np.random.default_rng(42 + restart * 1000)
            a_work = best_a.copy()
            # Get FFT
            fft_size = 1 << (2 * n - 1).bit_length()
            A_fft = rfft(a_work, fft_size)
            # Add noise to mid/high frequencies
            noise = (rng_r.standard_normal(len(A_fft)) + 1j * rng_r.standard_normal(len(A_fft))) * 0.05
            A_fft[1:] += noise[1:]
            A_fft[-1] = np.conj(A_fft[1])  # maintain real symmetry
            a_pert = norm_sequence(irfft(A_fft, fft_size)[:n])
        else:
            a_pert = best_a.copy()

        a_new, score_new = fista_smoothed(a_pert, stage_budget, temps)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

    return best_a, best_score


def peak_shave_fast(a0, time_limit, n_steps=50000):
    """Fast peak shaving using FFT-based gradient at the peak."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = true_objective(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Ignore endpoints (they're always small)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-1] = -np.inf
            b_work[-2] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        if peak_val <= b.max() * 0.9999:
            break

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            # Try reducing a[i] and redistributing
            for factor in [0.90, 0.80, 0.70, 0.50]:
                a_trial = a.copy()
                a_trial[i] *= factor
                diff = a[i] * (1.0 - factor)
                if diff < 1e-16:
                    continue
                # Redistribute to positions that barely contribute to the peak
                # Positions far from the peak's support
                dist_to_peak = np.abs(np.arange(n) - (peak_k - i))
                # Prefer positions with low a values and far from peak
                low_score = a.copy() + 0.1 * dist_to_peak
                low_score[i] = np.inf
                # Add to the 25% lowest scoring positions
                n_low = max(1, n // 4)
                low_positions = np.argsort(low_score)[:n_low]
                low_positions = low_positions[low_positions != i]
                if len(low_positions) > 0:
                    add_amount = diff / len(low_positions)
                    a_trial[low_positions] += add_amount
                a_trial = norm_sequence(a_trial)
                score = true_objective(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=20000):
    """Direct coordinate descent with line search on individual coordinates."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = true_objective(a)
    t0 = time.monotonic()

    # Precompute which coordinates to try (cycle through)
    order = np.random.default_rng(7).permutation(n)
    pos = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        i = order[pos % n]
        pos += 1

        if a[i] < 1e-12:
            continue

        # Binary search for best value of a[i]
        # We move mass from i to a random other position j
        j = order[(pos + n // 2) % n]
        if i == j:
            continue

        best_local = a[i]
        best_local_score = best_score

        # Try several fractions
        for frac in [0.7, 0.5, 0.3, 0.1]:
            a_trial = a.copy()
            a_trial[i] *= frac
            a_trial[j] += a[i] * (1.0 - frac)
            a_trial = norm_sequence(a_trial)
            sc = true_objective(a_trial)
            if sc < best_local_score:
                best_local_score = sc
                best_local = a_trial[i]
                a = a_trial
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        temps2 = np.geomspace(30.0, 0.1, 10)
        a_r, s_r = fista_with_restarts(best_seq, t_alloc, T_start=30.0, T_end=0.1,
                                        n_stages=10, n_restarts=4)
        try_update(a_r, "fista_restart")

    # === Phase 3: Low-temperature FISTA polish ===
    print(f"\n=== Phase 3: FISTA Low-T Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 2->0.02")
        temps3 = np.geomspace(2.0, 0.02, 8)
        a_p, s_p = fista_smoothed(best_seq, t_alloc, temps3)
        try_update(a_p, "fista_lowT")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shave_fast(best_seq, t_alloc)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 1.33, 0.75]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [768, 896, 1024, 1152, 1280]:
        if 384 <= fixed_n <= 3000 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.18, 120.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        temps_alt = np.geomspace(50.0, 0.2, 10)
        a_alt, s_alt = fista_smoothed(init_alt, t_total * 0.7, temps_alt)
        try_update(a_alt, f"fista_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            t_ps = min(rem * 0.3, 30.0)
            if t_ps > 5:
                a_alt_ps, _ = peak_shave_fast(best_seq, t_ps)
                try_update(a_alt_ps, f"ps_n{alt_n}")

    # === Phase 6: Final coordinate descent + peak shave ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd = min(rem * 0.25, 60.0)
        if t_cd > 10:
            a_cd, _ = coordinate_descent_refine(best_seq, t_cd)
            try_update(a_cd, "coord_descent")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.3, 40.0)
        if t_ps > 5:
            a_ps2, _ = peak_shave_fast(best_seq, t_ps)
            try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        # One more FISTA round with very low T
        temps_fin = [1.0, 0.3, 0.1, 0.03]
        t_fin = min(rem * 0.4, 30.0)
        if t_fin > 5:
            a_fin, s_fin = fista_smoothed(best_seq, t_fin, temps_fin)
            try_update(a_fin, "final_fista")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
