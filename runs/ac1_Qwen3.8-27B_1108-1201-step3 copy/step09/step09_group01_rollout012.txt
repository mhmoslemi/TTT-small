"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax gradient descent with simulated annealing (primary driver)
2. Convolution variance matching as intermediate objective
3. FFT-based efficient gradient computation
4. Peak-targeted multiplicative updates
5. Fine coordinate descent for final polishing
6. LP fictitious play with Nesterov averaging for global improvement
7. Multi-length exploration with adaptive time allocation

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_gradient(a, p):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a[i].
    Uses FFT for the cross-correlation part.
    """
    n = len(a)
    # r[i] = sum_k a[k] * p[k+i] = (a cross-corr p)[i]
    size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, size)
    P = rfft(p, size)
    r = irfft(np.conj(A) * P, size)[:n]

    # W[i] = sum_{k=i}^{i+n-1} p[k] (sliding window sum)
    p_pad = np.zeros(2 * n)
    p_pad[:len(p)] = p
    p_cumsum = np.cumsum(p_pad)
    W = np.zeros(n)
    W[0] = p_cumsum[n - 1]
    for i in range(1, n):
        W[i] = p_cumsum[i + n - 1] - p_cumsum[i - 1]

    grad = r + a * W

    # Extra term for k=2i (the diagonal where a[i]*a[i] appears)
    diag_mask = 2 * np.arange(n) < len(p)
    grad[diag_mask] += a[diag_mask] * p[2 * np.arange(n)[diag_mask]]

    return grad


def softmax_grad_descent(a0, time_limit, beta_init=20.0, eta_init=0.05,
                         t_init=0.01, verbose=False, label=""):
    """
    Softmax gradient descent with simulated annealing.
    Uses multiplicative updates on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    current_score = best_score

    beta = beta_init
    eta = eta_init
    temperature = t_init
    t_cool = max(1.0, time_limit / 200)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 500

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax weights
        max_conv = conv.max()
        scaled = beta * (conv - max_conv)
        scaled = np.clip(scaled, -60, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
        p = weights / Z

        # Compute gradient
        grad = compute_gradient(a, p)

        # Multiplicative update: a_new = a * exp(-eta * grad)
        log_a = np.log(a + 1e-30)
        log_a -= eta * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        # Evaluate
        conv_new = fast_convolve(a_new)
        new_score = 2.0 * n * conv_new.max()

        # Simulated annealing acceptance
        if new_score <= current_score:
            accept = True
        else:
            delta = new_score - current_score
            if temperature < 1e-10:
                accept = False
            else:
                accept = np.random.random() < np.exp(-delta / temperature)

        if accept:
            a = a_new
            conv = conv_new
            current_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive parameters
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit
        temperature = t_init * np.exp(-3.0 * frac)

        if no_improve > patience:
            eta *= 0.9
            beta *= 1.1
            no_improve = 0
            if eta < 1e-10:
                break

        # Gradually increase beta (sharper max approximation)
        if it % 100 == 0:
            beta *= 1.05

        if verbose and it % 200 == 0:
            print(f"    SMD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.1f} eta={eta:.6f}")

    return best_a, best_score


def conv_variance_objective(a):
    """Compute variance of convolution (smooth proxy for max)."""
    conv = fast_convolve(a)

def variance_matching_step(a, p_target, eta):
    """One step of gradient descent on convolution variance."""
    n = len(a)
    conv = fast_convolve(a)
    mean_c = conv.mean()

    # Gradient of sum_k (c[k] - mean)^2 w.r.t. a[i]
    # = 2 * sum_k (c[k]-mean) * dc[k]/da[i]
    # Use same structure as compute_gradient but with (c[k]-mean) as weights
    weights = conv - mean_c
    grad = compute_gradient(a, np.abs(weights) + 1e-30)  # approximate
    # Actually let's just use the softmax approach for stability
    # Better: gradient of sum c[k]^2
    grad = compute_gradient(a, conv)
    # The true gradient of sum c[k]^2 is 2 * sum_k c[k] * dc[k]/da[i]
    # compute_gradient gives sum_k p[k] * dc[k]/da[i]
    # So if p = c (unnormalized), we need to be careful
    # Let's just use: grad_i = 2 * sum_k c[k] * dc[k]/da[i]

    # For simplicity, let's just use the normalized version
    c_norm = conv / conv.sum()
    grad = compute_gradient(a, c_norm)
    return grad


def peak_shave(a, n_peaks=5, strength=0.02):
    """
    Identify coordinates that most contribute to the top peaks
    and reduce them proportionally.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.9
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions[:20]:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            w = a[i] * a[j]
            pressure[i] += w
            if j != i:
                pressure[j] += w * 0.5

    p_max = pressure.max()
    if p_max < 1e-20:
        return a.copy()
    pressure = pressure / p_max

    a_new = a * (1.0 - strength * pressure)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fast_coords_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent using vectorized operations for the update step.
    More efficient for larger n.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    factors = [0.99, 0.98, 0.97, 0.995, 1.005, 1.01, 1.02, 0.95, 1.03, 0.9, 1.05]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Vectorized: for each factor, compute new_max
            deltas = np.array([a[i] * (f - 1.0) for f in factors])
            new_vals = a[i] + deltas
            valid = (new_vals >= 0) & (new_vals <= 1.0)
            if not valid.any():
                continue

            # new_affected[k] = conv[k] + 2*delta*a[k-i] (+ delta^2 if k=2i)
            # For vectorization over deltas:
            a_portion = a[0:n]  # This is a[k-i] for k in [i, i+n-1]
            # Shape: (n_factors, n_affected)
            # conv[k_start:k_end+1] has length n
            affected = conv[k_start:k_end + 1]

            # changes[d, :] = 2*deltas[d] * a_portion[:]
            changes = 2.0 * deltas[:, None] * a_portion[None, :]
            # Add diagonal correction
            if 2 * i >= k_start and 2 * i <= k_end:
                idx_2i = 2 * i - k_start
                changes[:, idx_2i] += deltas * deltas

            new_affected = affected[None, :] + changes
            new_maxes = np.maximum(unaff_max, new_affected.max(axis=1))

            valid = valid & (new_maxes < M - 1e-16)
            if not valid.any():
                continue

            # Pick the best (minimum new_max)
            best_idx = np.argmin(new_maxes[valid])
            valid_indices = np.where(valid)[0]
            best_d_idx = valid_indices[best_idx]
            best_delta = deltas[best_d_idx]
            best_new_max = new_maxes[best_d_idx]

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update conv
                old_ai = a[i] - best_delta
                for k in range(k_start, k_end + 1):
                    r = k - i
                    if 0 <= r < n:
                        if r == i:
                            conv[k] += 2.0 * best_delta * old_ai + best_delta * best_delta
                        else:
                            conv[k] += best_delta * (a[r] + old_ai)

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                sum_a = a.sum()
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """LP fictitious play with Nesterov-style averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s

    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma_it = 2.0 / (iters + 2.0)
        a_avg = (1.0 - gamma_it) * a_avg + gamma_it * a
        b_avg = (1.0 - gamma_it) * b_avg + gamma_it * b

        candidates = [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score:
                best_score = score
                best_a = cand_n.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 7))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)

    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Softmax gradient descent with simulated annealing
    2. Peak shaving heuristics
    3. Fast coordinate descent
    4. LP fictitious play
    5. Multi-length exploration
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(123)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        score_p = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(score_p):
            best_seq = parent.copy()
            best_score = score_p
            print(f"Parent: n={len(parent)}, score={score_p:.8f}")

    # Get LP solver
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # === Phase 1: Multi-length softmax gradient descent ===
    print("\n=== Phase 1: Multi-length SMD ===")
    t1_start = time.monotonic()
    p1_budget = 400.0

    lengths = [1024, 1280, 1536]
    if parent is not None and len(parent) not in lengths:
        lengths.insert(0, len(parent))

    phase1_results = []
    for n in lengths:
        elapsed = time.monotonic() - t1_start
        if elapsed > p1_budget:
            break
        t_left = p1_budget - elapsed
        t_alloc = max(30.0, min(t_left / 2.0, 150.0))

        # Interpolate parent or create init
        if parent is not None:
            init = interpolate_seq(parent, n)
        else:
            init = init_symmetric(n, rng)

        print(f"  SMD n={n}, budget={t_alloc:.0f}s")
        a_smd, score_smd = softmax_grad_descent(
            init, t_alloc, beta_init=15.0, eta_init=0.03, t_init=0.005,
            verbose=False, label=f"smd_n{n}"
        )
        phase1_results.append((score_smd, a_smd, n, "smd"))

        # Peak shave and re-optimize
        a_shaved = peak_shave(a_smd, n_peaks=5, strength=0.03)
        t_left = p1_budget - (time.monotonic() - t1_start)
        if t_left > 20:
            a_ps, score_ps = softmax_grad_descent(
                a_shaved, min(30.0, t_left * 0.3), beta_init=30.0, eta_init=0.02,
                t_init=0.003, label=f"ps_n{n}"
            )
            phase1_results.append((score_ps, a_ps, n, "peak_shave"))

    phase1_results.sort(key=lambda x: x[0])
    for sc, a_r, n_r, lbl in phase1_results[:5]:
        best_seq, best_score = check_and_update(a_r, best_seq, best_score, f"p1_{lbl}_n{n_r}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Intensive coordinate descent ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    p2_budget = 400.0

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(4):
            rem = p2_budget - (time.monotonic() - t2_start)
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 120.0)
            print(f"  CD round {rnd+1}/4, n={n_best}, budget={t_alloc:.0f}s")
            a_cd, score_cd = fast_coords_descent(
                current, t_alloc, verbose=False, label=f"cd2_r{rnd}"
            )
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"p2_cd_r{rnd}")

            # Perturb for next round
            if rnd < 2:
                if rnd == 0:
                    current = peak_shave(best_seq, strength=0.02)
                else:
                    # Random restructure
                    n_curr = len(best_seq)
                    keep = rng.random(n_curr) > 0.1
                    n_keep = keep.sum()
                    if n_keep < n_curr // 5:
                        keep = np.ones(n_curr, dtype=bool)
                    a_r = best_seq.copy()
                    a_r[~keep] = 0
                    a_r[keep] = a_r[keep] * (1.0 + rng.standard_normal(keep.sum()) * 0.15)
                    a_r = np.maximum(a_r, 0)
                    s_r = a_r.sum()
                    if s_r > 1e-15:
                        current = a_r / s_r
                    else:
                        current = best_seq.copy()
            else:
                current = best_seq.copy()

    # === Phase 3: Softmax GD with different params ===
    print(f"\n=== Phase 3: SMD variants (best={best_score:.8f}) ===")
    t3_start = time.monotonic()
    p3_budget = 200.0

    if best_seq is not None:
        for beta_try, eta_try in [(50, 0.01), (100, 0.005), (30, 0.05)]:
            rem = p3_budget - (time.monotonic() - t3_start)
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  SMD beta={beta_try}, eta={eta_try}, budget={t_alloc:.0f}s")
            a_smd, score_smd = softmax_grad_descent(
                best_seq, t_alloc, beta_init=beta_try, eta_init=eta_try,
                t_init=0.002, label=f"smd_b{beta_try}"
            )
            best_seq, best_score = check_and_update(a_smd, best_seq, best_score, f"p3_smd_b{beta_try}")

    # === Phase 4: LP Fictitious Play ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.8f}) ===")
    t4_start = time.monotonic()
    p4_budget = 300.0

    if best_seq is not None:
        n_best = len(best_seq)
        for n_fp in [n_best, min(n_best + 256, 2048)]:
            rem = p4_budget - (time.monotonic() - t4_start)
            if rem < 60:
                break
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  LPFP n={n_fp}, budget={t_alloc:.0f}s")
            initial = interpolate_seq(best_seq, n_fp)
            a_fp, score_fp = lp_fictitious_play(
                n_fp, t_alloc, initial=initial, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"p4_lpf_n{n_fp}")

            # Polish with CD
            rem = p4_budget - (time.monotonic() - t4_start)
            if rem > 30:
                a_pol, score_pol = fast_coords_descent(
                    a_fp, min(40.0, rem * 0.3), label=f"pol_n{n_fp}"
                )
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, f"p4_pol_n{n_fp}")

    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 30:
        # Multiple rounds of CD with peak shaving
        a_cur = best_seq.copy()
        for i in range(3):
            rem = deadline - time.monotonic() - 5
            if rem < 20:
                break
            t_cd = min(rem * 0.3, 80.0)
            a_f, score_f = fast_coords_descent(a_cur, t_cd, label=f"final_cd{i}")
            best_seq, best_score = check_and_update(a_f, best_seq, best_score, f"final_cd{i}")
            if i < 2:
                a_cur = peak_shave(best_seq, strength=0.01)

        # One more SMD with very fine params
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_m, score_m = softmax_grad_descent(
                best_seq, min(rem * 0.5, 50.0),
                beta_init=200.0, eta_init=0.003, t_init=0.001,
                label="final_smd"
            )
            best_seq, best_score = check_and_update(a_m, best_seq, best_score, "final_smd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
