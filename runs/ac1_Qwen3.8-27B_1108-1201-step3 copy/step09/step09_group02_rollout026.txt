def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=50):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            coef = 2.0 * a_partners
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            
            c_quad = c_vals[2 * i - k_lo] if has_quad else 0.0
            qi = 2 * i - k_lo if has_quad else -1
            
            upper = max(5.0 * v, 5.0 / n)
            best_x = golden_section_search(_cd_eval, 0.0, upper, max_iter=30,
                                           c_vals=c_vals, coef=coef, unaff_max=unaff_max,
                                           has_quad=has_quad, quad_idx=qi, c_quad=c_quad, n=n)
            
            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[qi] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max())
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _compute_subgradient(a, conv, n_peaks=10):
    """Compute a subgradient of f(a) = 2n*max(conv(a,a)) at a."""
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    temp = 0.01 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return np.zeros(n)
    w = w / ws
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]
    
    return grad

def subgradient_descent(a0, time_limit, n_peaks=15):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.005
    it = 0
    no_improve = 0
    a_avg = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = _compute_subgradient(a, conv, n_peaks)
        a_new = simplex_project(a - lr * grad)
        a = a_new
        conv = fast_convolve(a)
        sc = 2.0 * n * float(conv.max())
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
        
        a_avg = a_avg + (a - a_avg) / (it + 1)
        sc_avg = 2.0 * n * float(fast_convolve(a_avg).max())
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
    
    return best_a, best_score

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    iters = 0
    lp_fail_count = 0
    
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        iters += 1
        
        # Solve LP
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            lp_fail_count += 1
            if lp_fail_count > 20:
                break
            continue
        lp_fail_count = 0
        
        # Precompute quadratic coefficients for line search
        A_k = fast_convolve(a)  # (a*a)
        B_k = fast_cross_convolve(a, b)  # (a*b)
        C_k = fast_convolve(b)  # (b*b)
        # (1-t)^2 A + 2t(1-t) B + t^2 C = A + t(-2A+2B) + t^2(A-2B+C)
        q0 = A_k
        q1 = 2.0 * (B_k - A_k)
        q2 = A_k - 2.0 * B_k + C_k
        
        # Golden section search for optimal t
        best_t = golden_section_search(_ls_eval, 0.0, 1.0, max_iter=40,
                                       q0=q0, q1=q1, q2=q2, n=n)
        
        # Update
        a_new = (1.0 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            continue
        a_new = a_new / cs
        
        sc = 2.0 * n * float(fast_convolve(a_new).max())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic CD
        if iters % cd_every == 0:
            rem2 = time_limit - (time.monotonic() - t0)
            if rem2 > 10:
                t_cd = min(cd_time, rem2 * 0.1)
                a_cd, sc_cd = exact_coord_descent(a, t_cd)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_a = a_cd.copy()
                a = a_cd  # Continue from CD result
        
        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score, iters
