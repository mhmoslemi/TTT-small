import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of(a):
    """Compute the score for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    c = fast_convolve(a_norm)
    return 2.0 * n * c.max()

    a = a / s

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming a is normalized (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * float(c.max())

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using pair-move coordinate descent
with multi-peak awareness, combined with mirror descent and multi-start.

Key innovation: pair-move operations that transfer mass between coordinate pairs,
preserving the sum constraint and directly targeting convolution peaks.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for a normalized (sum=1) sequence."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * float(c.max())


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def pair_move_cd(a0, time_limit, top_k=5, verbose=True, label=""):
    n = len(a0)
    a = normalize(a0.astype(np.float64)).copy()
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    sweep = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        sweep_improvements = 0

        # Get active peaks
        k_act = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_act)[-k_act:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        for i in range(n):
            if a[i] < 1e-13:
                continue

            # Target the main peak
            k_main = int(peak_idx[0])

            # Find best j to move to
            j_lo = max(0, k_main - n + 1)
            j_hi = min(n - 1, k_main)
            if j_lo >= j_hi:
                continue

            # Minimize a[k_main - j] over valid j
            # a[k_main - j] for j in [j_lo, j_hi]
            # = a[k_main - j_hi : k_main - j_lo + 1][::-1]
            candidates = a[k_main - j_hi : k_main - j_lo + 1][::-1]
            j_best = j_lo + int(np.argmin(candidates))

            if j_best == i:
                # Try second best
                candidates_excl = candidates.copy()
                candidates_excl[i - j_lo] = np.inf
                if np.isfinite(candidates_excl.min()):
                    j_best = j_lo + int(np.argmin(candidates_excl))
                else:
                    continue

            # Compute slope: slope[k] = 2*(a[k-j] - a[k-i]) for k in affected range
            k_start = min(i, j_best)
            k_end = min(max(i + n - 1, j_best + n - 1), 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Build slope
            length = k_end - k_start + 1
            slope = np.zeros(length)

            # -2*a[k-i] for k where 0 <= k-i <= n-1
            ki_valid_lo = max(k_start, i)  # k >= i means k-i >= 0
            ki_valid_hi = min(k_end, i + n - 1)  # k-i <= n-1 means k <= i+n-1
            if ki_valid_lo <= ki_valid_hi:
                # k ranges from ki_valid_lo to ki_valid_hi
                # k-i ranges from ki_valid_lo-i to ki_valid_hi-i
                slope[ki_valid_lo - k_start : ki_valid_hi - k_start + 1] -= 2.0 * a[ki_valid_lo - i : ki_valid_hi - i + 1]

            # +2*a[k-j] for k where 0 <= k-j <= n-1
            kj_valid_lo = max(k_start, j_best)
            kj_valid_hi = min(k_end, j_best + n - 1)
            if kj_valid_lo <= kj_valid_hi:
                slope[kj_valid_lo - k_start : kj_valid_hi - k_start + 1] += 2.0 * a[kj_valid_lo - j_best : kj_valid_hi - j_best + 1]

            # Try different step sizes
            max_delta = a[i]
            best_delta = 0.0
            best_new_M = M

            for frac in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
                delta = max_delta * frac
                if delta < 1e-15:
                    continue
                new_vals = conv[k_start:k_end + 1] + delta * slope
                new_M = max(unaff_max, float(new_vals.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta > 0:
                # Apply move
                a[i] -= best_delta
                a[j_best] += best_delta
                if a[i] < 0:
                    a[i] = 0.0
                conv[k_start:k_end + 1] += best_delta * slope

                # Second-order correction
                delta_sq = best_delta * best_delta
                # At k = i + j_best: subtract 2*delta^2
                k_over = i + j_best
                if 0 <= k_over < 2 * n - 1:
                    conv[k_over] -= 2.0 * delta_sq
                # At k = 2*i: add delta^2
                k_2i = 2 * i
                if 0 <= k_2i < 2 * n - 1:
                    conv[k_2i] += delta_sq
                # At k = 2*j_best: add delta^2
                k_2j = 2 * j_best
                if 0 <= k_2j < 2 * n - 1:
                    conv[k_2j] += delta_sq

                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                improved = True
                sweep_improvements += 1

        # Periodic recomputation to fix floating point drift
        if sweep % 100 == 0:
            conv_recomputed = fast_convolve(a)
            # Use the recomputed version (more accurate)
            conv = conv_recomputed
            M = float(conv.max())

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PMCD {label} sweep={sweep} ({elapsed:.0f}s): best_score={2.0*n*best_M:.8f} M={best_M:.10f}")

    return best_a, 2.0 * n * best_M


def mirror_descent_peaks(a0, time_limit, lr_init=0.05, top_k=20, verbose=True, label=""):
    """Mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = normalize(a0.astype(np.float64)).copy()
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * (pv.max() + 1e-30)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += 2.0 * w[idx] * a[k - i_end : k - i_start + 1][::-1]

        # Mirror descent (multiplicative)
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng, method="mixed"):
    """Generate structured initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "cosine":
        t = np.linspace(-1, 1, n)
        power = rng.uniform(1, 4)
        a = (0.5 * (1 + np.cos(np.pi * t))) ** power
    elif method == "two_peak":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif method == "periodic":
        period = int(rng.integers(3, 20))
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.3 + 0.7 * abs(np.cos(2 * np.pi * phase))
    elif method == "sparse":
        a = np.zeros(n)
        n_nonzero = int(n * rng.uniform(0.3, 0.7))
        indices = rng.choice(n, size=n_nonzero, replace=False)
        a[indices] = 1.0
    elif method == "ramp":
        t = np.arange(n, dtype=np.float64) / n
        a = t ** rng.uniform(0.5, 3.0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

                if 0 <= k2j < 2 * n - 1:
                    conv[k2j] += d2
                kij = i + j_best
                if 0 <= kij < 2 * n - 1:
                    conv[kij] -= 2.0 * d2

                M = max(unaff_max, float(conv[k_start : k_end + 1].max()))
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                improved = True

        if sweep % 20 == 0:
            conv = fast_convolve(a).copy()
            M = float(conv.max())

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePMCD {label} sw={sweep} ({elapsed:.0f}s): score={2.0*n*best_M:.8f}")

    return best_a, 2.0 * n * best_M


def mirror_descent_peaks(a0, time_limit, lr_init=0.05, top_k=20, verbose=True, label=""):
    n = len(a0)
    a = normalize(a0.astype(np.float64)).copy()
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * (pv.max() + 1e-30)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_s = max(0, k - n + 1)
            i_e = min(n - 1, k)
            sg[i_s:i_e + 1] += 2.0 * w[idx] * a[k - i_e : k - i_s + 1][::-1]

        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng, method="mixed"):
    if method == "uniform":
        return np.ones(n) / n
    elif method == "cosine":
        t = np.linspace(-1, 1, n)
        power = rng.uniform(1, 4)
        a = (0.5 * (1 + np.cos(np.pi * t))) ** power
    elif method == "two_peak":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif method == "periodic":
        period = int(rng.integers(3, 20))
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.3 + 0.7 * abs(np.cos(2 * np.pi * phase))
    elif method == "sparse":
        a = np.zeros(n)
        n_nonzero = int(n * rng.uniform(0.3, 0.7))
        indices = rng.choice(n, size=n_nonzero, replace=False)
        a[indices] = 1.0
    elif method == "ramp":
        t = np.arange(n, dtype=np.float64) / n
        a = t ** rng.uniform(0.5, 3.0)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score - 1e-12:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():

            p_score = float(evaluate_sequence(parent.tolist()))

        pass

        best_seq = parent.copy()

    else:

    for m in methods:

    # Also try noisy version
    noisy = best_seq + rng.standard_normal(n_best) * 0.03
    noisy = np.maximum(noisy, 0)
    initializations.append(noisy / noisy.sum())

    for idx, init in enumerate(initializations):
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:

        t_alloc = min(remaining * 0.2, 40.0)

    print(f"  After Phase 1: best={best_score:.8f}")

            x_old = np.linspace(0, 1, len(parent))

    else:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
