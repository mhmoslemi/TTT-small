"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive mirror descent with Polyak step and subgradient at active peaks
2. Nesterov-accelerated LP alternating with dual warm-starts and adaptive step
3. Multi-stage L-BFGS on LogSumExp with aggressive polynomial cooling
4. Power-mean smoothing as alternative final polish
5. Directed local search focusing on peak-equalization moves
Key insight: the problem is a zero-sum game; LP alternating is fictitious play
with Nesterov acceleration for O(1/k^2) convergence.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, tol=1e-8):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, **{'eps_abs': tol, 'eps_rel': tol})
    except Exception:
        try:
            prob.solve(solver=solver)
        except Exception:
            pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, n_active_frac=0.08):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def compute_subgradient_topk(a, k=10):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    top_k = np.argsort(b)[::-1][:k]
    g = np.zeros(n)
    for kk in top_k:
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx = np.arange(js, je + 1)
        g[idx] += a[kk - idx]
    g /= len(top_k)
    return g, max_b


def mirror_descent_adaptive(a0, time_limit, n_active_frac=0.06):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    use_topk = False

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 20000 == 10000:
            use_topk = not use_topk

        if use_topk:
            g, max_b = compute_subgradient_topk(a, k=8)
        else:
            g, max_b = compute_subgradient(a, n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-12:
            break

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.75
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    last_noise_time = 0.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Adaptive Nesterov step: use gap to lower bound
        gap = best_score - f_lb
        if gap > 1e-10:
            # Standard Nesterov: 2/(k+2)
            alpha = 2.0 / (it + 2)
            # But cap it based on how much improvement the LP response gives
            # Smaller step when we're close to optimal
            if gap < 0.01:
                alpha *= min(1.0, 0.01 / gap)
                alpha = max(alpha, 2.0 / (it + 2))
        else:
            alpha = 0.01

        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score

            if stall_count >= 6 and elapsed - last_noise_time > 30:
                rng = np.random.default_rng(it * 31 + 7)
                noise_scale = 0.01 * (stall_count / 6.0)
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                last_noise_time = elapsed
                stall_count = 0

            if (it + 1) % 2000 == 0:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} gap={best_score-f_lb:.6f}")

    return best_a, best_score


def lbgfs_logsumexp_refine(a0, time_limit, T_start=500.0, T_end=0.05, n_stages=12):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_func(z_in, T):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        val = (2.0 * n / T) * lse
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-16:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S2 = np.dot(grad_a, a)
        grad_z = a * (grad_a - S2)
        return val, grad_z

    def make_func(T):
        def func(z_in):
            v, _ = lse_func(z_in, T)
            return v
        return func

    def make_jac(T):
        def jac(z_in):
            _, j = lse_func(z_in, T)
            return j
        return jac

    # Polynomial cooling: T_k = T_start * (T_end/T_start)^(k/(n_stages-1))
    # But use more aggressive: T_k = T_start * (1 - k/n_stages)^3
    temps = []
    for i in range(n_stages):
        frac = i / max(1, n_stages - 1)
        T = T_start * (1.0 - 0.999 * frac) ** 4
        T = max(T, T_end)
        temps.append(T)

    total_iter_budget = 20000
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        maxiter_stage = min(total_iter_budget // n_stages, 3000)
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': maxiter_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter_stage * 500, 'maxls': 50, 'disp': False}
            )
            z = res.x
            if time.monotonic() - t0 > time_limit:
                break
        except Exception:
            pass

    return softmax_from_z(z)


def power_mean_refine(a0, time_limit, p_start=20, p_end=2000, n_stages=6):
    """Use power mean (sum b_k^p)^(1/p) as smooth approximation to max."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def pm_func(z_in, p):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = np.maximum(fast_convolve(a), 0)
        bp = b ** p
        sum_bp = bp.sum()
        if sum_bp < 1e-300:
            return 0, np.zeros_like(z_in)
        val = 2.0 * n * sum_bp ** (1.0 / p)
        # Gradient: d/da val = 2n * (1/p) * sum_bp^(1/p - 1) * p * b^(p-1) * db/da
        #          = 2n * sum_bp^(1/p - 1) * sum_k b_k^(p-1) * (db_k/da)
        coeff = 2.0 * n * sum_bp ** (1.0 / p - 1.0)
        w = bp / sum_bp  # normalized weights b^p / sum(b^p)
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-16:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S2 = np.dot(grad_a, a)
        grad_z = a * (grad_a - S2)
        return val, grad_z

    def make_func(T):
        def func(z_in):
            v, _ = pm_func(z_in, T)
            return v
        return func

    def make_jac(T):
        def jac(z_in):
            _, j = pm_func(z_in, T)
            return j
        return jac

    # Geometric spacing of p values
    ps = np.geomspace(p_start, p_end, n_stages)
    for p_val in ps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        maxiter_stage = 2000
        try:
            res = minimize(
                fun=make_func(p_val),
                x0=z,
                jac=make_jac(p_val),
                method='L-BFGS-B',
                options={'maxiter': maxiter_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def local_peak_equalization(a0, time_limit, n_steps=50000):
    """Targeted local search: identify peak positions and try to equalize them."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        threshold = peak_val * 0.999
        peaks = np.where(b >= threshold)[0]

        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        # Find positions in 'a' that contribute most to the peak
        contrib = np.zeros(n)
        for k in peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Try moving mass from high-contrib positions to low-contrib positions
        # that don't increase other peaks
        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        best_trial_score = best_score

        for si in top_src:
            if a[si] < 1e-12:
                continue
            for di in low_dst:
                if si == di:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12]:
                    amount = a[si] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    if a_trial[si] < 0 or a_trial[di] < 0:
                        continue
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_trial_score - 1e-14:
                        best_trial_score = score
                        a_trial_best = a_trial.copy()

        if best_trial_score < best_score - 1e-14:
            a = a_trial_best
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 500 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    amount = a[i] * 0.02
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * x / 2) ** 4
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        # Bimodal
        a = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    elif choice == 4:
        # U-shaped (edge concentration)
        a = (x - 0.5) ** 2 + 0.01
    elif choice == 5:
        # Piecewise: high edges, low middle
        a = np.where(x < 0.2, 2.0, np.where(x > 0.8, 2.0, 0.5))
    elif choice == 6:
        # Wasli-like: smooth step
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        # Try multiple initializations, keep best
        best_init = None
        best_init_score = float('inf')
        for trial in range(6):
            init = structured_init(n_init, rng)
            sc = evaluate_safe(init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = best_init_score
        n_best = n_init
        print(f"Best initial: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Mirror descent for rapid initial convergence
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, sc_md = mirror_descent_adaptive(best_seq, t_alloc, n_active_frac=0.06)
        try_update(a_md, "MD_1")

    # Phase 2: Nesterov LP - main workhorse
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "NLP_1")

    # Phase 3: L-BFGS LogSumExp with aggressive cooling
    print(f"\n=== Phase 3: L-BFGS LSE (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 500->0.05")
        a_lb = lbgfs_logsumexp_refine(best_seq, t_alloc, T_start=500.0, T_end=0.05, n_stages=12)
        try_update(a_lb, "LBFGS_LSE_1")

    # Phase 4: Power mean refinement
    print(f"\n=== Phase 4: Power Mean (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pm = power_mean_refine(best_seq, t_alloc, p_start=20, p_end=2000, n_stages=6)
        try_update(a_pm, "PowerMean_1")

    # Phase 5: Local peak equalization
    print(f"\n=== Phase 5: Local PE (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_pe, _ = local_peak_equalization(best_seq, t_alloc, n_steps=100000)
        try_update(a_pe, "LocalPE_1")

    # Phase 6: Second Nesterov LP round (from improved point)
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp2, "NLP_2")

    # Phase 7: Final L-BFGS polish at very high T
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 50->0.02")
        a_fin = lbgfs_logsumexp_refine(best_seq, t_alloc, T_start=50.0, T_end=0.02, n_stages=10)
        try_update(a_fin, "FinalLBFGS")

    # Phase 8: Final local search
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 80.0)
    if t_alloc > 10:
        a_ls, _ = local_peak_equalization(best_seq, t_alloc, n_steps=50000)
        try_update(a_ls, "FinalLocal")

    # Phase 9: Alternative lengths (if time remains)
    rem = deadline - time.monotonic()
    if rem > 120:
        print(f"\n=== Phase 9: Alt Lengths (best={best_score:.6f}, rem={rem:.0f}s) ===")
        alt_lengths = []
        for mult in [1.25, 1.5, 0.75]:
            alt_n = int(n_best * mult)
            if 384 <= alt_n <= 2500 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))[:1]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_total = min(rem * 0.4, 150.0)
            print(f"  n={alt_n}, budget={t_total:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            sc_alt = evaluate_safe(init_alt)
            cur = init_alt
            if sc_alt < best_score:
                best_score = sc_alt
                best_seq = cur.copy()

            t_lp = t_total * 0.7
            a_alt, _ = nesterov_lp_refine(cur, t_lp, solver=solver, n_max_iters=100000)
            try_update(a_alt, f"LP_n{alt_n}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
