def fast_convolve(a):

    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

                y = y / s
            else:
                y = np.ones(n) / n
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step_count % 1000 == 0:
                        print(f"    NG step={step_count}, T={T:.1f}, score={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    if n > 5000:
        return None, None
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        sc_a = score_normalized(a)
        sc_y = score_normalized(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if it % 500 == 0:
            if best_score >= prev_best - 1e-9:
                stall += 1
            else:
                stall = 0
            prev_best = best_score
            if stall >= 5:
                rng = np.random.default_rng(it)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                stall = 0

        if (it + 1) % 2000 == 0:
            print(f"    LP iter={it+1}, score={best_score:.8f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_inner = b[1:-1] if len(b) > 2 else b
        if len(b_inner) == 0:
            break
        peak_k = int(np.argmax(b_inner)) + (1 if len(b) > 2 else 0)
        threshold = b[peak_k] * 0.9
        near = np.where(b >= threshold)[0]
        near = near[(near > 0) & (near < 2 * n - 2)]

        contrib = np.zeros(n)
        for k in near[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            c = a[idx] * a[k - idx]
            top = np.argsort(c)[::-1][:6]
            for ci in top:
                contrib[idx[ci]] += c[ci]

        top_coords = np.argsort(contrib)[::-1][:15]
        improved = False

        for i in top_coords:
            if a[i] < 1e-15:
                continue
            for f in [0.98, 0.95, 0.90, 0.85, 0.80]:
                a_t = a.copy()
                a_t[i] *= f
                a_t = project_simplex(a_t)
                sc = score_normalized(a_t)
                if sc < best_score - 1e-12:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for i in top_coords[:5]:
                if a[i] < 1e-15:
                    continue
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                c = a[idx] * a[peak_k - idx]
                top_c = np.argsort(c)[::-1][:3]
                for ci in top_c:
                    j = peak_k - idx[ci]
                    if j == i or j < 0 or j >= n:
                        continue
                    if a[j] < 1e-15:
                        continue
                    for f1 in [0.95, 0.90]:
                        for f2 in [0.95, 0.90]:
                            a_t = a.copy()
                            a_t[i] *= f1
                            a_t[j] *= f2
                            a_t = project_simplex(a_t)
                            sc = score_normalized(a_t)
                            if sc < best_score - 1e-12:
                                a = a_t
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_t = a.copy()
            for i in top_coords[:3]:
                if a[i] > 1e-15:
                    a_t[i] *= 0.97
            a_t = project_simplex(a_t)
            sc = score_normalized(a_t)
            if sc < best_score - 1e-12:
                a = a_t
                best_score = sc
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def make_init(n, rng, style=0):
    x = np.linspace(0, 1, n)
    if style == 0:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x)
    elif style == 1:
        a = 1.0 + 0.5 * (2 * x - 1)
    elif style == 2:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2) + 0.1
    elif style == 3:
        a = 1.0 + 0.4 * np.cos(4 * np.pi * x)
    elif style == 4:
        a = 1.0 + 0.6 * np.sin(6 * np.pi * x + 0.3)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, eval={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform n={n_init}: score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Phase 1: Quick exploration of short sequences
    print("\n=== Phase 1: Short sequence exploration ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        for n_try in [2, 3, 4, 5, 6, 8, 10, 12, 16]:
            if time.monotonic() > deadline:
                break
            a = np.ones(n_try) / n_try
            try_update(a, f"uniform_n{n_try}")
        # Try some non-uniform short sequences
        for n_try in [4, 6, 8, 10, 12, 16, 20, 30, 50]:
            if time.monotonic() > deadline:
                break
            for _ in range(20):
                a = rng.dirichlet(np.ones(n_try))
                try_update(a, f"rand_n{n_try}")

    # Phase 2: Nesterov PG on main sequence
    print(f"\n=== Phase 2: Nesterov PG (n={n_best}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg(best_seq, t_alloc, T_start=5.0, T_end=400.0,
                              lr_init=0.06, n_stages=10, steps_per_stage=3000)
        try_update(a_ng, "ng_main")

    # Phase 3: LP refinement
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30 and n_best <= 5000:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp, "lp_1")

    # Phase 4: Peak shaving
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")

    # Phase 5: Second round of optimization with different params
    print(f"\n=== Phase 5: Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_ng2, _ = nesterov_pg(best_seq, t_alloc, T_start=20.0, T_end=800.0,
                               lr_init=0.02, n_stages=8, steps_per_stage=2000)

        rem = deadline - time.monotonic()
        if rem > 20 and n_best <= 3000:
            a_lp2, _ = lp_refine(best_seq, min(rem * 0.3, 60.0), solver=solver, n_max_iters=20000)
            try_update(a_lp2, "lp_r2")

    # Phase 6: Multi-start with different initializations

    rem = deadline - time.monotonic()

        styles = [0, 1, 2, 3, 4, 5]

                                   lr_init=0.08, n_stages=8, steps_per_stage=1500)
            try_update(a_sub, f"multi_s{style}")

                try_update(a_sub2, f"multi_s{style}_lp")

    # Phase 7: Try different lengths

                a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 30.0), solver=solver, n_max_iters=10000)

    # Phase 8: Final LP polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")

    else:

        best_score = evaluate_safe(best_seq)
        print(f"\nFALLBACK: {best_score:.8f}")

    return [float(x) for x in best_seq]
