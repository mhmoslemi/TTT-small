Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with adaptive log-sum-exp temperature
2. Fast vectorized coordinate descent with exact line search
3. Nesterov-accelerated subgradient method with Cesàro averaging

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_gradient_smooth(a, weights):
    """Compute gradient of smooth objective using FFT-based Hankel matrix product."""
    n = len(a)
    # grad[i] = 2 * sum_j weights[i+j] * a[j] for j in [0, n-1]
    # Using: grad = 2 * conv(flip(weights), a)[n-1 : 2n-1][::-1]
    weights_flipped = weights[::-1]
    conv_wa = convolve_ab(weights_flipped, a)
    return 2.0 * conv_wa[n-1:2*n-1][::-1]

def smooth_grad_descent(a0, time_limit, beta_init=5.0, beta_end=300.0, lr_init=0.05, verbose=False):
    """Smooth gradient descent with adaptive temperature schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    beta = beta_init
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Softmax weights
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -60, 0)
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
        else:
            weights = weights / wsum
        
        # Gradient via FFT
        grad = compute_gradient_smooth(a, weights)
        
        # Multiplicative update (exponentiated gradient)
        log_a = np.log(a + 1e-30) - lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)
        
        # Evaluate
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature and learning rate
        # Increase beta gradually (sharper approximation)
        if it < 100:
            beta = min(beta * 1.02, beta_end)
        else:
            beta = min(beta * 1.005, beta_end)
        
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-6:
                break
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SGD it={it} ({elapsed:.0f}s): beta={beta:.1f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score

    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov step size
        t_k = 2.0 / (it + 1.0)  # or (1 + sqrt(1 + 4*(it-1)^2)) / 2

        conv_y = fast_convolve(y)

    x = a.copy()

                #                     = 2 * delta * a[i] + delta^2 for k_idx = quad_idx
                
                # But wait, after the update, a[i] has changed. The formula uses the OLD a[k-i].
                # Since k-i = k_idx (because k_start = i), and a[k_idx] for k_idx != i is unchanged.
                # For k_idx = i: a[k_idx] = a[i] (old value), and the change is 2*delta*a[i]_old + delta^2
                # But the formula 2*delta*a[k_idx] gives 2*delta*a[i]_old, which is correct for the linear part.
                # The delta^2 is the extra term.
                
                conv[k_start:k_end + 1] += 2.0 * best_delta * a[0:n_affected]
                if has_quadratic:
                    conv[k_start + quad_idx] += best_delta * best_delta
                
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                    no_improve = 0
                improved = True
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FCD3 {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.10f}")
    
    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            affected = conv[k_start:k_end + 1]
            n_affected = len(affected)
            has_quadratic = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quadratic else -1
            coeffs = 2.0 * a[0:n_affected]
            
            best_delta = 0.0
            best_new_max = M
            
            # Very fine steps
            for factor in [0.999, 0.998, 0.997, 0.996, 0.995, 1.001, 1.002, 1.003, 1.004, 1.005]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * coeffs
                if has_quadratic:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                conv[k_start:k_end + 1] += 2.0 * best_delta * a[0:n_affected]
                if has_quadratic:
                    conv[k_start + quad_idx] += best_delta * best_delta
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_score

def smooth_grad_cd_alternating(a0, time_limit, verbose=False):
    """
    Alternate between smooth gradient descent and coordinate descent.
    This combines the global exploration of SGD with the precise local optimization of CD.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    phase = 0
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        if phase % 3 == 0:
            # Smooth gradient descent
            t_alloc = min(rem * 0.3, 60.0)
            a_sgd, s_sgd = smooth_grad_descent(a, t_alloc, beta_init=10.0, beta_end=200.0, lr_init=0.03)
            if s_sgd < best_score:
                best_score = s_sgd
                best_a = a_sgd.copy()
            a = a_sgd
            if verbose:
                print(f"  Phase {phase}: SGD -> {s_sgd:.6f}")
        
        elif phase % 3 == 1:
            # Coordinate descent
            t_alloc = min(rem * 0.3, 60.0)
            a_cd, s_cd, _ = fast_coord_descent_v3(a, t_alloc, label=f"ph{phase}")
            if s_cd < best_score:
                best_score = s_cd
                best_a = a_cd.copy()
            a = a_cd
            if verbose:
                print(f"  Phase {phase}: CD -> {s_cd:.6f}")
        
        else:
            # Fine coordinate descent
            t_alloc = min(rem * 0.2, 30.0)
            a_fine, s_fine = fine_coord_descent(a, t_alloc)
            if s_fine < best_score:
                best_score = s_fine
                best_a = a_fine.copy()
            a = a_fine
            if verbose:
                print(f"  Phase {phase}: FineCD -> {s_fine:.6f}")
        
        phase += 1
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_symmetric(n, rng):
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_bimodal(n, rng):
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_uniform(n):
    return np.ones(n) / n

def init_gaussian(n, rng):
    center = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.1, 0.3)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    return best_seq, best_score
