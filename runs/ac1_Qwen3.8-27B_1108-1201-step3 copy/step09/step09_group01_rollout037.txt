"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam mirror descent with log-sum-exp smoothing and adaptive temperature
2. Convex 1D line search per coordinate (ternary search on max of linear functions)
3. FFT-accelerated gradient and Hessian-vector products
4. Multi-resolution strategy (coarse to fine)
5. Population-based exploration with diverse initializations
6. Spectral reweighting for flat-spectrum initialization

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(c, a, beta):
    """
    Compute gradient of logsumexp(beta*c) wrt a.
    c = conv(a,a) of length 2n-1, a of length n.
    grad[i] = 2 * sum_j a[j] * w[i+j]
    where w = softmax(beta * c)
    Computed via: conv(reverse(a), w)[n-1 : 2n-1] * 2
    """
    n = len(a)
    # Softmax
    shifted = beta * (c - c.max())
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        s = 1.0
    w = w / s
    
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * conv(reverse(a), w)[n-1+i]
    rev_a = a[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = rfft(rev_a, fft_size)
    W = rfft(w, fft_size)
    conv_rw = irfft(A * W, fft_size)[:total_len]
    grad = 2.0 * conv_rw[n-1 : n-1+n]
    return grad

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - 1))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_mirror_descent(a0, time_limit, beta_init=20.0, beta_final=500.0, 
                        lr_init=0.01, verbose=False):
    """
    Adam-style mirror descent for minimizing max(conv(a,a)) on the simplex.
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    # Adam state
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    
    while time.monotonic() - t0 < total_time:
        t += 1
        
        # Adaptive beta: increase over time
        progress = min(1.0, t / max(1, int(total_time / 0.02)))
        beta = beta_init * (beta_final / beta_init) ** progress
        
        # Compute smooth gradient
        c = fast_convolve(a)
        grad = compute_smooth_grad(c, a, beta)
        # Normalize gradient by beta (since we're using logsumexp(beta*c), 
        # the gradient of (1/beta)*logsumexp(beta*c) is (1/beta)*grad_logsumexp)
        grad = grad / beta
        
        # Adam update in log-space (mirror descent)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Mirror descent step: a_new = a * exp(-lr * m_hat / (sqrt(v_hat) + eps))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a = np.log(a + 1e-30)
        log_a -= step
        a_new = np.exp(log_a - log_a.max())
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-10:
                break
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AMD t={t} ({elapsed:.0f}s): beta={beta:.1f}, lr={lr:.6f}, score={best_score:.8f}")
    
    return best_a, best_score

def coord_descent_ternary(a0, time_limit, tol=1e-10, max_sweeps=10000, verbose=False):
    """
    Coordinate descent with ternary search for the optimal step in each coordinate.
    Since the 1D subproblem is convex (min of max of linear functions),
    ternary search works well.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    c = fast_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            # k ranges from i to i+n-1 (clipped to [0, 2n-2])
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            if k_lo > 0:
                unaff_left = float(c[:k_lo].max())
            else:
                unaff_left = -np.inf
            if k_hi < 2 * n - 2:
                unaff_right = float(c[k_hi+1:].max())
            else:
                unaff_right = -np.inf
            unaff_max = max(unaff_left, unaff_right)
            
            # Affected part: c[k] + 2*delta*a[k-i] + delta^2 (if k == 2i)
            affected_k = np.arange(k_lo, k_hi + 1)
            affected_c = c[k_lo:k_hi+1].copy()
            # Weights: a[k-i] for k in [k_lo, k_hi]
            idx = affected_k - i  # This gives [k_lo-i, ..., k_hi-i] = [0, ..., min(n-1, k_hi-i)]
            weights = a[idx]
            
            # Check if 2i is in [k_lo, k_hi]
            has_diagonal = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            def obj_1d(delta):
                """Objective value for a given delta (change in a[i])."""
                if a[i] + delta < -1e-15:
                    return np.inf
                new_val = a[i] + delta
                if new_val < 0:
                    return np.inf
                new_affected = affected_c + 2.0 * delta * weights
                if has_diagonal:
                    diag_idx = 2 * i - k_lo
                    new_affected[diag_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                return new_max
            
            # Ternary search for optimal delta
            # Range: delta in [-a[i], a[i] * 2] (we allow increasing too)
            lo = -a[i] * 0.999
            hi = a[i] * 2.0
            
            # Quick check: is the current point already good?
            f0 = float(max(unaff_max, float((affected_c).max())))
            
            # Try a few quick evaluations first
            best_delta = 0.0
            best_f = f0
            
            # Try some heuristic steps
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5, -0.01, -0.05, -0.1, -0.2, -0.5]:
                d = frac * a[i]
                if a[i] + d < 0:
                    continue
                f = obj_1d(d)
                if f < best_f - 1e-15:
                    best_f = f
                    best_delta = d
            
            # If heuristic found improvement, do ternary search around it
            if best_delta != 0.0:
                # Narrow down with ternary search
                # The function is convex, so ternary search works
                if best_delta < 0:
                    lo2 = -a[i] * 0.999
                    hi2 = 0.0
                else:
                    lo2 = 0.0
                    hi2 = a[i] * 2.0
                
                for _ in range(20):
                    m1 = lo2 + (hi2 - lo2) / 3.0
                    m2 = hi2 - (hi2 - lo2) / 3.0
                    f1 = obj_1d(m1)
                    f2 = obj_1d(m2)
                    if f1 < f2:
                        hi2 = m2
                    else:
                        lo2 = m1
                
                best_delta = (lo2 + hi2) / 2.0
                best_f = obj_1d(best_delta)
            
            if best_f < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update c
                delta = best_delta
                c[k_lo:k_hi+1] += 2.0 * delta * weights
                if has_diagonal:
                    c[2*i] += delta * delta
                M = max(unaff_max, float(c[k_lo:k_hi+1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved and sweep > 20:
            break
        elif not improved and sweep > 5:
            # No improvement in this sweep; check if we've been stuck
            pass
    
    return best_a, best_score, sweep

def spectral_init(n, rng, n_components=5):
    """
    Initialize using spectral method: try to create a sequence with flat power spectrum.
    This gives a good starting point since flat spectrum ≈ flat convolution.
    """
    # Start with random phase in frequency domain
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random phase, flat magnitude
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Create a "target" magnitude spectrum that's approximately flat
    # but has the right energy
    target_mag = np.ones(fft_size // 2 + 1)
    target_mag[0] = 0  # DC component handled separately
    
    A = target_mag * np.exp(1j * phase)
    # Make it Hermitian-symmetric for real output
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size//2+1] = A
    for k in range(1, fft_size // 2 + 1):
        if k <= fft_size - k:
            A_full[fft_size - k] = np.conj(A[k])
    
    a = np.fft.ifft(A_full).real[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def flat_conv_init(n, rng):
    """
    Initialize by solving for a sequence whose convolution is approximately flat.
    Uses iterative refinement.
    """
    # Start with a good initial guess
    a = np.ones(n) / n
    
    # Iteratively flatten the convolution
    for _ in range(100):
        c = fast_convolve(a)
        target = 1.0 / (2 * n - 1)
        # "Pressure" on each coordinate
        # grad[i] = sum_k (c[k] - target) * 2*a[k-i] / ((2n-1)*target)
        # Simplified: just do gradient descent on sum((c[k]-target)^2)
        diff = c - target
        # Gradient of sum(diff^2) wrt a[i] = 2 * sum_k diff[k] * 2*a[k-i]
        rev_a = a[::-1]
        fft_size = 1 << (3 * n).bit_length()
        A = rfft(rev_a, fft_size)
        D = rfft(diff, fft_size)
        conv_ad = irfft(A * D, fft_size)[:n + len(diff) - 1]
        grad = 2.0 * 2.0 * conv_ad[n-1:n-1+n]
        
        # Gradient step
        lr = 0.01
        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / max(s, 1e-15)
    
    return a

def diverse_init(n, rng, method="mixed"):
    """Generate diverse initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "symmetric":
        half = (n + 1) // 2
        t = np.linspace(0, 1, half)
        ncomp = int(rng.integers(2, 7))
        profile = np.ones(half)
        for _ in range(ncomp):
            freq = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.5)
            phase = rng.uniform(0, np.pi)
            profile += amp * np.cos(np.pi * freq * t + phase)
        profile = np.maximum(profile, 0.001)
        if n % 2 == 0:
            a = np.concatenate([profile, profile[::-1][:-1]])
        else:
            a = np.concatenate([profile, profile[::-1][:-1]])
        # Ensure correct length
        if len(a) != n:
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a)), a)
        a = np.maximum(a, 0)
        s = a.sum()
        return a / max(s, 1e-15)
    elif method == "bimodal":
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        t = np.arange(n)
        a = np.exp(-0.5*((t-c1)/w1)**2) + np.exp(-0.5*((t-c2)/w2)**2)
        return a / a.sum()
    elif method == "spectral":
        return spectral_init(n, rng)
    elif method == "flatconv":
        return flat_conv_init(n, rng)
    else:
        methods = ["uniform", "symmetric", "bimodal", "spectral", "flatconv"]
        m = methods[int(rng.integers(len(methods)))]
        return diverse_init(n, rng, method=m)

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)


def compute_smooth_grad(c, a, beta):
    """
    Compute gradient of logsumexp(beta*c) wrt a.
    c = conv(a,a) of length 2n-1, a of length n.
    grad[i] = 2 * sum_j a[j] * w[i+j] where w = softmax(beta*c)
    Computed via conv(reverse(a), w)[n-1 : 2n-1] * 2
    """
    n = len(a)
    shifted = beta * (c - c.max())
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        s = 1.0
    w = w / s

    rev_a = a[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = rfft(rev_a, fft_size)
    W = rfft(w, fft_size)
    conv_rw = irfft(A * W, fft_size)[:total_len]
    grad = 2.0 * conv_rw[n - 1:n - 1 + n]
    return grad


def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_mirror_descent(a0, time_limit, beta_init=20.0, beta_max=500.0,
                        lr_init=0.01, verbose=False):
    """Adam-style mirror descent minimizing max(conv(a,a)) on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    total_steps = max(1, int(time_limit / 0.02))

    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / total_steps)
        beta = beta_init * (beta_max / beta_init) ** progress

        c = fast_convolve(a)
        grad = compute_smooth_grad(c, a, beta)
        grad = grad / beta

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        log_a -= step
        a_new = np.exp(log_a - log_a.max())
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum
        a_new = np.maximum(a_new, 0)
        a_new = a_new / max(a_new.sum(), 1e-15)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

            no_improve = 0
            if lr < 1e-10:

        if verbose and t % 500 == 0:

def coord_descent_ternary(a0, time_limit, max_sweeps=5000, verbose=False, label=""):

    else:
        a = a / s

    c = fast_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0
    no_impr_sweeps = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            if k_lo > 0:
                unaff_left = float(c[:k_lo].max())
            else:
                unaff_left = -np.inf
            if k_hi < 2 * n - 2:
                unaff_right = float(c[k_hi + 1:].max())
            else:
                unaff_right = -np.inf
            unaff_max = max(unaff_left, unaff_right)

            n_affected = k_hi - k_lo + 1
            affected_c = c[k_lo:k_hi + 1]
            idx = np.arange(k_lo, k_hi + 1) - i
            weights = a[idx]
            has_diag = (2 * i >= k_lo) and (2 * i <= k_hi)
            diag_pos = 2 * i - k_lo if has_diag else -1

            cur_max_affected = float(affected_c.max())
            current_f = max(unaff_max, cur_max_affected)

            # Heuristic quick checks
            best_delta = 0.0
            best_f = current_f
            for frac in [0.01, 0.05, 0.1, 0.2, 0.4, -0.01, -0.05, -0.1, -0.2, -0.4]:

                nv = ai + d
                if nv < 0:

                    na[diag_pos] += d * d

                if fm < best_f - 1e-15:

                    best_delta = d

                    na1 = affected_c + 2.0 * m1 * weights

                    na2 = affected_c + 2.0 * m2 * weights
                    if has_diag:
                        na2 = na2.copy()
                        na2[diag_pos] += m2 * m2
                    f2 = max(unaff_max, float(na2.max()))
                    if f1 < f2:
                        hi2 = m2
                    else:
                        lo2 = m1

                best_delta = (lo2 + hi2) / 2.0
                nv = ai + best_delta
                if nv >= 0:
                    na = affected_c + 2.0 * best_delta * weights
                    if has_diag:
                        na = na.copy()
                        na[diag_pos] += best_delta * best_delta

        else:

        if verbose and sweep % 50 == 0:

def spectral_init(n, rng):
    """Initialize via random-phase flat-spectrum in frequency domain."""
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(50):
        phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        mag = np.ones(fft_size // 2 + 1)
        A = mag * np.exp(1j * phase)
        A_full = np.zeros(fft_size, dtype=complex)
        A_full[:fft_size // 2 + 1] = A
        for k in range(1, fft_size // 2 + 1):
            if fft_size - k < len(A_full):
                A_full[fft_size - k] = np.conj(A[k])
        a = np.fft.ifft(A_full).real[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            score = 2.0 * n * fast_convolve(a).max()
            return a
    return np.ones(n) / n


def flat_conv_init(n, rng):
    """Initialize by iteratively flattening the convolution."""
    a = np.ones(n) / n
    for it in range(200):
        c = fast_convolve(a)
        target = 1.0 / (2 * n - 1)
        diff = c - target
        rev_a = a[::-1]
        total_len = n + len(diff) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = rfft(rev_a, fft_size)
        D = rfft(diff, fft_size)
        conv_ad = irfft(A * D, fft_size)[:total_len]
        grad = 4.0 * conv_ad[n - 1:n - 1 + n]
        lr = 0.005
        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def symmetric_init(n, rng):
    """Create symmetric initialization with cosine superposition."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    ncomp = int(rng.integers(2, 7))
    profile = np.ones(half) * 0.5
    for _ in range(ncomp):
        freq = rng.uniform(0.3, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.001)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1][:-1]])
    else:
        a = np.concatenate([profile, profile[::-1][:-1]])
    if len(a) != n:
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a)), a)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / max(s, 1e-15)

    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.maximum(np.interp(new_x, old_x, f_norm), 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result


def peak_pressure_perturb(a, rng, strength=0.05):
    """Perturb by reducing peak contributors in convolution."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    if M < 1e-15:
        return a.copy()
    threshold = 0.9 * M
    peak_pos = np.where(c >= threshold)[0]
    pressure = np.zeros(n)
    for k in peak_pos[:100]:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]
    pmax = pressure.max()
    if pmax > 1e-20:
        pressure /= pmax
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_restructure(a, rng, frac=0.15):
    """Mildly restructure by zeroing some coordinates and adding noise."""
    n = len(a)
    a_new = a.copy()
    mask = rng.random(n) > frac
    n_keep = mask.sum()
    if n_keep < max(10, n // 10):
        mask = np.ones(n, dtype=bool)
    a_new[~mask] *= 0.05
    a_new[mask] *= (1.0 + 0.2 * rng.standard_normal(mask.sum()))
    # Add small mass to some zeroed positions
    zero_mask = a_new < 1e-10
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.05))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a_new[add_idx] = rng.random(n_add) * a.max() * 0.3
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-length diverse initialization + quick CD
    Phase 2: Alternating Adam MD + coordinate descent with perturbations
    Phase 3: Fine polishing
    Phase 4: Length exploration
    Phase 5: Final optimization
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(777)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    # === Phase 1: Diverse multi-length search ===
    print("\n=== Phase 1: Diverse Multi-Length Search ===")
    t1_start = time.monotonic()
    n_values = [512, 768, 1024, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    phase1_cands = []
    for n in n_values:
        if time.monotonic() > deadline - 1100:
            break
        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_seq(parent, n)))
        inits.append(("symm", symmetric_init(n, rng)))
        inits.append(("bi", bimodal_init(n, rng)))
        inits.append(("spec", spectral_init(n, rng)))
        inits.append(("flat", flat_conv_init(n, rng)))

        for name, init in inits:
            if time.monotonic() > deadline - 1000:
                break
            t_left = 1000 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(12.0, t_left / 3.0))
            a_opt, s_opt, _ = coord_descent_ternary(init, t_alloc, max_sweeps=300)
            phase1_cands.append((s_opt, a_opt, n, name))

    phase1_cands.sort(key=lambda x: x[0])
    print(f"  Phase 1 done ({time.monotonic()-t1_start:.0f}s): {len(phase1_cands)} cands, top-3: " +
          ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in phase1_cands[:3]]))

    for s_c, a_c, n_c, name_c in phase1_cands[:12]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{name_c}_n{n_c}")

    # === Phase 2: Intensive alternating optimization ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.7f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        current = best_seq.copy()
        n_cur = len(current)
        n_rounds = 10

        for rnd in range(n_rounds):
            rem = deadline - time.monotonic() - 350
            if rem < 30:
                break

            # Adam mirror descent
            t_amd = min(rem * 0.35, 100.0)
            if t_amd > 10:
                beta_i = 10 + 8 * rnd
                beta_f = 150 + 100 * rnd
                lr_i = 0.008 / (1.0 + 0.3 * rnd)
                aamd, samd = adam_mirror_descent(current, t_amd,
                                                  beta_init=beta_i, beta_max=beta_f,
                                                  lr_init=lr_i)
                best_seq, best_score = check_and_update(aamd, best_seq, best_score, f"amd_r{rnd}")
                if samd < 2.0 * n_cur * fast_convolve(current).max():
                    current = aamd.copy()

            # Coordinate descent
            rem = deadline - time.monotonic() - 300
            if rem < 15:
                break
            t_cd = min(rem * 0.35, 80.0)
            if t_cd > 10:
                acd, scd, sw = coord_descent_ternary(current, t_cd, max_sweeps=5000, label=f"r{rnd}")
                best_seq, best_score = check_and_update(acd, best_seq, best_score, f"cd_r{rnd}")
                current = acd.copy()

            # Perturbation
            if rnd < n_rounds - 2:
                if rnd % 3 == 0:
                    current = peak_pressure_perturb(best_seq, rng, strength=0.04 + 0.01 * rnd)
                elif rnd % 3 == 1:
                    current = random_restructure(best_seq, rng, frac=0.1 + 0.05 * rnd)
                else:
                    # Blend with a fresh init
                    fresh = symmetric_init(n_cur, rng)
                    current = 0.7 * best_seq + 0.3 * fresh
                    current = np.maximum(current, 0)
                    current = current / current.sum()

        print(f"  Phase 2 done ({time.monotonic()-t2_start:.0f}s): best={best_score:.7f}")

    # === Phase 3: Fine polishing ===
    print(f"\n=== Phase 3: Fine Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_fine = min(rem * 0.6, 120.0)
            a_fine, s_fine, _ = coord_descent_ternary(best_seq, t_fine, max_sweeps=20000, label="fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_polish")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        rem = deadline - time.monotonic() - 60
        for dn in [128, -128, 256, -64, 64]:
            new_n = n_b + dn
            if new_n < 256 or new_n > 3000 or rem < 15:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            t_alt = min(rem * 0.5, 50.0)
            init_alt = interpolate_seq(best_seq, new_n)
            a_alt, s_alt, _ = coord_descent_ternary(init_alt, t_alt, max_sweeps=3000, label=f"n{new_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{new_n}")
            # Quick AMD on the new length
            rem = deadline - time.monotonic() - 20
            if rem > 10 and s_alt < 1.6:
                a_amd, s_amd = adam_mirror_descent(a_alt, min(rem * 0.5, 30.0),
                                                    beta_init=50, beta_max=500, lr_init=0.003)
                best_seq, best_score = check_and_update(a_amd, best_seq, best_score, f"amd_n{new_n}")

    # === Phase 5: Final optimization ===
    print(f"\n=== Phase 5: Final (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            # Final AMD
            t_fin = min(rem * 0.4, 60.0)
            a_fin, s_fin = adam_mirror_descent(best_seq, t_fin,
                                                beta_init=200, beta_max=2000, lr_init=0.001)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_amd")

            # Final CD
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                a_cd, s_cd, _ = coord_descent_ternary(best_seq, rem * 0.8, max_sweeps=10000, label="final")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
