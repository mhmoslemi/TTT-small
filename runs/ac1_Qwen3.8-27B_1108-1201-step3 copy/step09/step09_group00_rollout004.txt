def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_logsumexp_obj(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def compute_logsumexp_grad(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def spectral_shaping_init(n, n_iters=50, gamma=0.5, rng=None):
    """Create initial sequence with flat |FFT|^2 using iterative spectral shaping."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.ones(n) / n
    L = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = fft(a, L)
        A[0] = 1.0  # Fix DC component (sum = 1)
        
        # Target: flat magnitude for f = 1, ..., L//2
        mags = np.abs(A[1:L//2+1])
        target = np.sqrt(np.mean(mags**2))
        
        # Interpolate magnitudes toward target
        new_mags = (1 - gamma) * target + gamma * mags
        phases = np.angle(A[1:L//2+1])
        
        A[1:L//2+1] = new_mags * np.exp(1j * phases)
        A[L - (L//2):0:-1] = np.conj(A[1:L//2+1][::-1])
        
        # Transform back
        a_new = np.real(ifft(A, L)[:n])
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a = a_new
    
    return a


def spectral_perturb(a, rng, scale=0.02):
    """Perturb a sequence in the frequency domain for stall-breaking."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    A = fft(a, L)
    
    # Add random phase perturbation to non-DC bins
    perturbation = (rng.standard_normal(L) + 1j * rng.standard_normal(L)) * scale
    perturbation[0] = 0  # Don't perturb DC
    # Maintain conjugate symmetry
    for f in range(1, L//2 + 1):
        perturbation[L - f] = np.conj(perturbation[f])
    
    A_pert = A + perturbation
    # Enforce conjugate symmetry
    A_pert[1:L//2+1] = A_pert[1:L//2+1]
    A_pert[L - (L//2):0:-1] = np.conj(A_pert[1:L//2+1][::-1])
    
    a_pert = np.real(ifft(A_pert, L)[:n])
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def multiplicative_weight_refine(a0, time_limit, n_steps=50000):
    """
    Multiplicative weight descent for peak shaving.
    Uses the subgradient structure: grad_a max(b) at peak k* is a shifted version of a.
    Update: a_j <- a_j * exp(-eta * grad_j), then renormalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.1
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks (within 1% of max)
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * g
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()

            eta *= 0.95
            no_improve = 0

            elapsed = time.monotonic() - t0
            print(f"    MultiWeight {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):

    return float(2.0 * n * b.max() / (s * s))
