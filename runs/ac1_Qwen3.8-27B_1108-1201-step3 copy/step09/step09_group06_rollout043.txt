def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_S1_direct(a, w):
    n = len(a)
    S1 = np.zeros(n)
    for j in range(n):
        S1[j] = np.dot(w[j:j + n], a)
    return S1


def compute_S1_fft(a, w):
    n = len(a)
    fft_size = 1 << (3 * n - 3).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    S1_full = irfft(A * np.conj(W), fft_size)
    return S1_full[:n]

def compute_subgradient(a, frac=0.08):
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()

        active = np.array([int(np.argmax(c))])
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    return w, max_c

    b_val = np.asarray(b_var.value, dtype=np.float64)

def spectral_perturb(a, rng, strength=0.1):
    n = len(a)
    a_c = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    mask = np.abs(freqs) > 0.05
    noise = rng.standard_normal(len(a_c)) * strength * np.abs(a_c).mean()
    a_c[mask] += noise
    a_pert = np.fft.irfft(a_c, n)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x ** 0.7) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        a = np.ones(n)
        a[:n // 2] = 0.5
        a = a + 0.1 * np.sin(4 * np.pi * np.linspace(0, 1, n))
    else:
        a = 1.0 / (1.0 + np.exp(-5 * (2 * np.linspace(0, 1, n) - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Phase 1: L-BFGS LogSumExp with increasing T (main optimization)
    print(f"\n=== Phase 1: L-BFGS LSE (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 420.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        T_sched = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
        a_lse, z_lse = lbgfs_lse_optimize(best_seq, t_alloc, T_sched, iters_per_stage=3000)
        try_update(a_lse, "lse_p1")

    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "lp_p2")
