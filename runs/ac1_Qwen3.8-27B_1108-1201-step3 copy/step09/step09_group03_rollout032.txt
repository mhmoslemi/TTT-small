import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    """Build the convolution matrix T where T[k,j] = a[k-j] if 0<=k-j<=n-1."""
    n = len(a)
    # Use the Toeplitz-like structure
    # Each column j is a shifted version of a
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.tile(a, n)
    # rows should be: for column j, the non-zero rows are j, j+1, ..., j+n-1
    # So row = j + m for m in 0..n-1
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    T = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T

def solve_lp(a, solver=None):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0. Returns b*, M*."""
    n = len(a)
    T = build_T(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [
        T @ b <= M,
        cp.sum(b) == 1
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if solver is None:
        solver = _find_solver()
    prob.solve(solver=solver, **_solver_params(solver))
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return np.ones(n)/n, fast_convolve(a).max()
    b_val = b_val / s
    M_val = float(M.value) if M.value is not None else fast_convolve(a).max()
    return b_val, M_val

_cached_solver = None
def _find_solver():
    global _cached_solver
    if _cached_solver is not None:
        return _cached_solver
    for name in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = name
                return name
        except Exception:
            continue
    _cached_solver = cp.SCS
    return cp.SCS

def _solver_params(solver):
    if solver == cp.SCS:
        return {'eps': 1e-7, 'max_iters': 100000}
    elif solver == cp.ECOS:
        return {'abstol': 1e-8, 'reltol': 1e-8, 'maxit': 100000}
    else:
        return {}

    """
    LP-based alternating minimization (fictitious play).
    Returns best sequence and score found.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    avg_b = a.copy()  # Running average of b responses
    t0 = time.monotonic()
    it = 0
    
    if verbose:
        print(f"    LP alternating: n={n}, initial score={best_score:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_star, M_star = solve_lp(a, solver)
        
        # Update a using harmonic step
        alpha = 2.0 / (it + 2.0)
        a = (1.0 - alpha) * a + alpha * b_star
        
        # Update running average
        avg_b = (it * avg_b + b_star) / (it + 1)
        
        a = np.maximum(a, 0)
        a = a / a.sum()
        avg_b = np.maximum(avg_b, 0)
        avg_b = avg_b / avg_b.sum()
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_avg = 2.0 * n * fast_convolve(avg_b).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_b.copy()
        
        # Also evaluate b_star directly
        score_b = 2.0 * n * fast_convolve(b_star).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_star.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): score_a={score_a:.6f}, score_avg={score_avg:.6f}, best={best_score:.6f}, M*={M_star:.8f}")
    
    return best_a, best_score

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2`. The key insight is that this can be framed as a symmetric zero-sum game where the LP oracle gives exact best responses, and the running average of iterates converges to the minimax equilibrium.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with running averages (main optimizer)
2. Multi-start with diverse initializations and lengths
3. Fast coordinate descent for local polishing
4. Adaptive perturbation and restart
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_dense(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_T_sparse(a):
    n = len(a)
    j_arr = np.arange(n)
    m_arr = np.arange(n)
    rows = np.repeat(j_arr, n) + np.tile(m_arr, n)
    cols = np.tile(j_arr, n)
    data = np.tile(a, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


_cached_solver = None


def get_solver():
    global _cached_solver
    if _cached_solver is not None:
        return _cached_solver
    for name in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = name
                return name
        except Exception:
            continue
    _cached_solver = cp.SCS
    return cp.SCS


def solver_kwargs(solver):
    if solver == cp.SCS:
        return {'eps': 1e-7, 'max_iters': 200000, 'verbose': False}
    elif solver == cp.ECOS:
        return {'abstol': 1e-8, 'reltol': 1e-8, 'maxit': 200000}
    elif solver == cp.CLARABEL:
        return {}
    return {}


def solve_best_response(a, solver):
    n = len(a)
    T = build_T_sparse(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [
        T @ b <= M,
        cp.sum(b) == 1
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver, **solver_kwargs(solver))
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return np.ones(n) / n, float(fast_convolve(a).max())
    b_val = b_val / s
    M_val = float(M.value) if M.value is not None else float(fast_convolve(a).max())
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return np.ones(n) / n, float(fast_convolve(a).max())
    return b_val, M_val


def lp_fictitious_play(a_init, time_budget, solver, verbose=True, label=""):
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * fast_convolve(a).max())
    avg_b = a.copy().astype(np.float64)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    if verbose:
        print(f"  LP-FP [{label}] n={n}, init score={best_score:.6f}")

    while time.monotonic() - t0 < time_budget:
        t_lp_start = time.monotonic()
        try:
            b_star, M_star = solve_best_response(a, solver)
        except Exception:
            break
        t_lp = time.monotonic() - t_lp_start
        it += 1

        alpha = 2.0 / (it + 2.0)
        a = (1.0 - alpha) * a + alpha * b_star
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / ss

        avg_b = (it * avg_b + b_star) / (it + 1.0)
        avg_b = np.maximum(avg_b, 0.0)
        ss2 = avg_b.sum()
        if ss2 < 1e-15:
            avg_b = a.copy()
        else:
            avg_b = avg_b / ss2

        score_a = float(2.0 * n * fast_convolve(a).max())
        score_avg = float(2.0 * n * fast_convolve(avg_b).max())
        score_b = float(2.0 * n * fast_convolve(b_star).max())

        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_b.copy()
            improved = True
        if score_b < best_score:
            best_score = score_b
            best_a = b_star.copy()
            improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP [{label}] it={it} ({elapsed:.0f}s, {t_lp*1000:.0f}ms/LP): "
                  f"a={score_a:.6f} avg={score_avg:.6f} best={best_score:.6f} M*={M_star:.8f}")

        if no_improve > 2000:
            break

    return best_a, best_score


def subgradient_mirror_descent(a_init, time_budget, lr_init=0.02, top_k=50, verbose=True, label=""):
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = float(2.0 * n * conv.max())
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            break

        k_act = min(top_k, 2 * n - 1)
        if k_act <= (2 * n - 1):
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)

        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a = a * np.exp(-lr_init * sg)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        conv = fast_convolve(a)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD [{label}] it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coord_descent(a_init, time_budget, verbose=True, label=""):
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected_len = k_end - k_start + 1
            a_shifted = a[0:affected_len]

            best_delta = 0.0
            best_new_M = M
            base = a[i]

            for factor in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5]:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_aff = conv[k_start:k_end + 1].copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                a[i] = max(a[i] + best_delta, 0.0)
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD [{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def cosine_window_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.3, 2.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def periodic_init(n, rng):
    period = int(rng.integers(3, 10))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.45 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def upsample_sequence(a, factor=2):
    n = len(a)
    a_new = np.zeros(n * factor)
    for i in range(n):
        a_new[i * factor:(i + 1) * factor] = a[i] / factor
    a_new = a_new / a_new.sum()
    return a_new


def evaluate_safe(a):
    try:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        return score
    except Exception:
        return float('inf')
