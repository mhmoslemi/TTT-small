def fft_convolve(a, b=None):
    """Compute a*b (linear convolution) using FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    B = rfft(np.asarray(b, dtype=np.float64), N)
    return irfft(A * B, N)[:size]

def fft_cross_corr(w, a):
    """Compute cross-correlation of w and a: cross[i] = sum_j w[i+j]*a[j] for i=0..n-1"""
    n = len(a)
    len_w = len(w)
    N = 1 << (n + len_w - 1 - 1).bit_length()  # need N >= n + len_w - 1
    W = rfft(np.asarray(w, dtype=np.float64), N)
    A = rfft(np.asarray(a, dtype=np.float64), N)
    cross = irfft(W * np.conj(A), N)[:n]
    return cross

def logsumexp(T, b):
    """Compute log(sum(exp(T*b))) in a numerically stable way."""
    max_Tb = T * b.max()
    return max_Tb + np.log(np.exp(T * b - max_Tb).sum())

def compute_fista_grad(a, T_val):
    """Compute gradient of (2n/T) * logsumexp(T * conv(a,a)) with respect to a."""

    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    cross = fft_cross_corr(w, a)
    return (4.0 * n / T_val) * cross

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Constraint-generation LP alternating with multiplicative weights fallback
2. FISTA on LogSumExp-smoothed objective with temperature annealing
3. Directed peak-reduction local search with sensitivity analysis

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b=None):
    """Linear convolution a*b via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    B = rfft(np.asarray(b, dtype=np.float64), N)
    return irfft(A * B, N)[:size]


def fft_cross_corr(w, a):
    """cross[i] = sum_j w[i+j]*a[j] for i=0..n-1, via FFT."""
    n = len(a)
    len_w = len(w)
    N = 1 << (n + len_w - 2).bit_length()
    W = rfft(np.asarray(w, dtype=np.float64), N)
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(W * np.conj(A), N)[:n]


def logsumexp_val(T_val, b):
    """Stable log(sum(exp(T*b)))."""
    m = T_val * b.max()
    return m + np.log(np.exp(T_val * b - m).sum())


def compute_smoothed_grad(a, T_val):
    """Gradient of (2n/T)*logsumexp(T*conv(a,a))."""
    n = len(a)
    b = fft_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    e = np.exp(Tb - max_Tb)
    w = e / e.sum()
    cross = fft_cross_corr(w, a)
    return (4.0 * n / T_val) * cross


def smoothed_obj(a, T_val):
    """(2n/T)*logsumexp(T*conv(a,a))."""
    n = len(a)
    b = fft_convolve(a)
    return (2.0 * n / T_val) * logsumexp_val(T_val, b)


def fista_smoothed(a0, time_limit, T_start=80.0, T_end=0.5, n_stages=12):
    """FISTA on LogSumExp-smoothed objective with annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        y = a.copy()
        t_k = 1.0
        L = (4.0 * n / stage_T) * 2.0 * fft_convolve(a).max() + 1.0
        max_iters = max(200, int(remaining / 0.05))
        impr = 0

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad_y = compute_smoothed_grad(y, stage_T)
            alpha = 1.0 / max(L, 1e-10)

            for _ in range(40):
                x_new = y - alpha * grad_y
                x_new = np.maximum(x_new, 0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y.copy()
                    break
                x_new = x_new / sx
                f_new = smoothed_obj(x_new, stage_T)
                f_y = smoothed_obj(y, stage_T)
                bound = f_y + np.dot(grad_y, x_new - y) + 0.5 / alpha * np.dot(x_new - y, x_new - y)
                if f_new <= bound + 1e-9:
                    break
                alpha *= 0.5
            L = max(1.0 / alpha, L * 1.2)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = x_new
            t_k = t_new

            score = 2.0 * n * fft_convolve(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                impr = 0
            else:
                impr += 1
            if impr > 500 and it > 500:
                break

        a = best_a.copy()
        elapsed = time.monotonic() - t0
        print(f"    FISTA stage {stage_idx} T={stage_T:.1f} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix_rows(a_fixed, indices):
    """Build rows of convolution matrix at given indices. T[k,j] = a[k-j] if valid."""
    n = len(a_fixed)
    rows = []
    for k in indices:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            row[js:je + 1] = a_fixed[k - je:k - js + 1]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    return np.zeros((0, n))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_lp_active_set(a_fixed, active_idx, solver, max_iters=200):
    """Constraint-generation LP: solve with active constraints, check violations, add."""
    n = len(a_fixed)
    K_max = min(len(active_idx), 200)
    active = list(active_idx[:K_max])

    for outer in range(max_iters):
        if len(active) == 0:
            active = [int(np.argmax(fft_convolve(a_fixed)))]

        T_sub = build_conv_matrix_rows(a_fixed, active)
        if T_sub.shape[0] == 0:
            break

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solve_kwargs = {'eps': 1e-9, 'max_iter': 200000}
        if solver == cp.SCS:
            solve_kwargs = {'EPS': 1e-9, 'MAX_ITERS': 200000}
        elif solver == cp.ECOS:
            solve_kwargs = {}
        try:
            prob.solve(solver=solver, **solve_kwargs)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None, None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = float(M_var.value)

        # Check for violated constraints
        conv_ab = fft_convolve(a_fixed, b_val)
        violations = np.where(conv_ab > M_val * (1 + 1e-7) + 1e-12)[0]
        new_viol = [int(v) for v in violations if v not in active]

        if len(new_viol) == 0:
            return b_val, M_val

        # Add top violators (up to 50 per outer iteration)
        viol_vals = conv_ab[violations]
        order = np.argsort(viol_vals)[::-1]
        to_add = [int(violations[idx]) for idx in order[:50] if int(violations[idx]) not in active]
        active.extend(to_add)
        active = active[:K_max]

    return b_val, M_val


def lp_alternating(a0, b0, time_limit, solver, n_max_outer=50000):
    """Alternating: fix a, solve LP for b; fix b, mirror descent for a."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    b = np.maximum(np.asarray(b0, dtype=np.float64), 0)
    sb = b.sum()
    if sb < 1e-15:
        b = np.ones(n) / n
        sb = 1.0
    b = b / sb
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    active_idx = list(np.argsort(fft_convolve(a, b))[::-1][:30])
    stall = 0

    for it in range(n_max_outer):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for b given a
        b_new, M_lp = solve_lp_active_set(a, active_idx, solver, max_iters=50)
        if b_new is not None:
            # Update b with step
            alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
            b = (1 - alpha) * b + alpha * b_new
            b = np.maximum(b, 0)
            sb = b.sum()
            if sb > 1e-15:
                b = b / sb

            # Update a using Nesterov-style
            a = (1 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
            else:
                a = b.copy()

        # Compute score
        score_a = 2.0 * n * fft_convolve(a).max()
        score_b = 2.0 * n * fft_convolve(b).max()
        cur_best = min(score_a, score_b)
        if cur_best < best_score - 1e-12:
            best_score = cur_best
            best_a = a.copy() if score_a < score_b else b.copy()
            stall = 0
        else:
            stall += 1

        # Update active set periodically
        if it % 20 == 0:
            conv_ab = fft_convolve(a, b)
            active_idx = list(np.argsort(conv_ab)[::-1][:50])

        if stall > 200:
            # Perturb
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.random(n) * 0.01
            b = b + noise
            b = np.maximum(b, 0)
            b = b / b.sum()
            a = a + noise * 0.5
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_lp={M_lp:.6f}")

    return best_a, best_score


def mwu_peak_reduction(a0, time_limit, eta=0.1, n_iters=200000):
    """Multiplicative weights update targeting the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_ab = fft_convolve(a, b)
        peak_val = conv_ab.max()
        threshold = peak_val - 0.15 * peak_val
        peaks = np.where(conv_ab >= threshold)[0]

        # Compute sensitivity vector
        sens = np.zeros(n)
        for k in peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sens[idx] += a[k - idx]

        # MWU update: decrease b where sensitivity is high
        eta_eff = eta * (peak_val + 1e-10)
        b_new = b * np.exp(-eta_eff * sens / (sens.max() + 1e-10))
        b_new = np.maximum(b_new, 1e-15)
        b_new = b_new / b_new.sum()

        # Interpolate a towards b
        alpha = 2.0 / (np.sqrt(it + 1) + 2.0)
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a = a / a.sum()
        b = b_new

        score = 2.0 * n * fft_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def directed_local_search(a0, time_limit, n_steps=100000):
    """Targeted local search: move mass from high-sensitivity to low-sensitivity positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fft_convolve(a)
        peak_val = b.max()
        threshold = peak_val - 0.08 * peak_val
        peaks = np.where(b >= threshold)[0]

        # Compute contribution to peaks from each position
        contrib = np.zeros(n)
        for k in peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Also compute "potential" (low contribution to peaks but could help elsewhere)
        total_contrib = contrib.sum() + 1e-15

        improved = False
        # Try moving from high-contrib to low-contrib
        top_src = np.argsort(contrib)[::-1][:10]
        low_dst = np.argsort(contrib)[:10]

        for si in top_src:
            if improved:
                break
            if a[si] < 1e-14:
                continue
            for di in low_dst:
                if si == di:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    score = 2.0 * n * fft_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break

        if not improved:
            # Random perturbation
            if step > 100:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-14:
                    amount = a[i] * rng.uniform(0.01, 0.08)
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fft_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

        if no_improve > 3000:
            # Larger perturbation
            noise = rng.random(n) * 0.05
            a_trial = a + noise
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            a = a_trial
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n, endpoint=False)
    if choice == 0:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = (0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)) ** 2
    elif choice == 3:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / 2)
        a = a ** 2
    elif choice == 5:
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif choice == 6:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 11 * x)
    else:
        a = (0.3 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.3 * np.cos(2 * np.pi * 4 * x + 0.5)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: combines multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n=960, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)}) (improved by {prev - sc:.6f})")

    # Phase 1: FISTA smoothed (annealing from high to low T)
    print(f"\n=== Phase 1: FISTA Smoothed (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_f, _ = fista_smoothed(best_seq, t_alloc, T_start=60.0, T_end=0.5, n_stages=10)
        try_update(a_f, "fista_1")

    # Phase 2: LP alternating (main workhorse)
    print(f"\n=== Phase 2: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, best_seq.copy(), t_alloc, solver)
        try_update(a_lp, "lp_alt_1")

    # Phase 3: MWU peak reduction
    print(f"\n=== Phase 3: MWU Peak Reduction (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mwu, _ = mwu_peak_reduction(best_seq, t_alloc, eta=0.08)
        try_update(a_mwu, "mwu_1")

    # Phase 4: Second FISTA round (lower T)
    print(f"\n=== Phase 4: FISTA r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_f2, _ = fista_smoothed(best_seq, t_alloc, T_start=20.0, T_end=0.2, n_stages=8)
        try_update(a_f2, "fista_2")

    # Phase 5: Directed local search
    print(f"\n=== Phase 5: Local Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_ls, _ = directed_local_search(best_seq, t_alloc)
        try_update(a_ls, "local_1")

    # Phase 6: Try alternative lengths
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.2, 1.5, 0.85, 1.1, 0.7]:
        alt_n = int(n_best * mult)
        if 480 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.30, 180.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolated n={alt_n} improved: {sc_alt:.6f}")

        # Quick LP on new length
        t_lp = t_total * 0.5
        a_alt, _ = lp_alternating(init_alt, init_alt.copy(), t_lp, solver)
        try_update(a_alt, f"lp_n{alt_n}")

        # Quick FISTA
        rem = deadline - time.monotonic()
        if rem > 20 and len(best_seq) == alt_n:
            t_f = min(rem * 0.3, 60.0)
            a_alt2, _ = fista_smoothed(best_seq, t_f, T_start=30.0, T_end=0.3, n_stages=6)
            try_update(a_alt2, f"fista_n{alt_n}")

    # Phase 7: Final LP polish on best
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.4, 100.0)
        a_fin, _ = lp_alternating(best_seq, best_seq.copy(), t_fin, solver)
        try_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ls = min(rem * 0.3, 60.0)
        a_ls2, _ = directed_local_search(best_seq, t_ls)
        try_update(a_ls2, "final_ls")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_f = min(rem * 0.4, 40.0)
        a_f3, _ = fista_smoothed(best_seq, t_f, T_start=10.0, T_end=0.1, n_stages=5)
        try_update(a_f3, "final_fista")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
