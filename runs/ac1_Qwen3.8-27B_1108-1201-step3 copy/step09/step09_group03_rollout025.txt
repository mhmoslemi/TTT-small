def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_gradient(a, conv, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv_k)) with respect to a.
    grad[i] = 2 * sum_j softmax_w[i+j] * a[j]
    """
    n = len(a)
    # Compute softmax weights
    shifted = conv - conv.max()
    w = np.exp(tau * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:

    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(np.conj(W) * A, N)[:n]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with diverse initializations (cosine, sinc, periodic, spectral)
2. Smooth gradient descent with log-sum-exp approximation and Adam optimizer + temperature annealing
3. Coordinate descent with incremental convolution updates focusing on active peak set
4. Fine-grained adaptive polishing
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

_rng = np.random.default_rng(2024)


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, conv, tau):
    """
    Gradient of (1/tau)*log(sum_k exp(tau*conv_k)) w.r.t. a.
    grad[i] = 2 * sum_j w[i+j] * a[j]  (cross-correlation)
    """
    n = len(a)
    shifted = conv - conv.max()
    w = np.exp(tau * shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= ws

    # Cross-correlation: g[i] = sum_j w[i+j] * a[j]
    # Using FFT: IFFT(conj(FFT(w,N)) * FFT(a,N))[:n]
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(np.conj(W) * A, N)[:n]
    return 2.0 * corr


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            return p / p.sum()
    except Exception:
        pass
    return None


def score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def init_cosine(n, power=1.5):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / max(center, 1)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_sinc(n, bw=0.4):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) * bw
    a = np.sinc(x / np.pi) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_periodic(n, period):
    t = np.arange(n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t / period)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_random_smooth(n):
    ncomp = max(4, n // 8)
    freqs = _rng.uniform(0.5, 4.0, size=ncomp)
    phases = _rng.uniform(0, 2 * np.pi, size=ncomp)
    amps = _rng.uniform(0.3, 1.0, size=ncomp)
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * t / n + p)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_spectral(parent, n):
    """Create a spectral-smoothed version of parent at length n."""
    if parent is None:
        return init_cosine(n)
    # Interpolate parent to length n
    if len(parent) == n:
        a = parent.copy()
    else:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, parent)
    # Apply lowpass filter in frequency domain
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    # Keep only low frequencies
    cutoff = int(fft_size * 0.3)
    A[cutoff:] = 0
    a = irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def adam_smooth_gd(a_init, time_limit, tau_init=5.0, lr_init=0.05,
                   verbose=True, label=""):
    """Smooth gradient descent with Adam and temperature annealing."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # Temperature schedule parameters
    tau_start = tau_init
    tau_end = max(2.0 * n, 100.0)  # Final temperature (large = sharp max)
    
    no_improve = 0
    patience = 3000
    current_lr = lr_init

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Temperature: exponential annealing
        frac = min(1.0, t / (time_limit * 30))
        log_tau = np.log(tau_start) + frac * (np.log(tau_end) - np.log(tau_start))
        tau = np.exp(log_tau)

        # Compute gradient
        grad = compute_smooth_grad(a, conv, tau)

        # Gradient clipping
        gn = np.linalg.norm(grad)
        if gn > 5.0:
            grad *= 5.0 / gn

        # LR schedule
        lr = current_lr / (1.0 + 0.0005 * t)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a /= s

        conv = fast_convolve(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart logic
        if no_improve > patience:
            noise_scale = 0.005 + 0.01 * (no_improve / patience)
            a = a + _rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            s = a.sum()
            a /= s if s > 1e-15 else 1.0
            conv = fast_convolve(a)
            no_improve = 0
            patience = max(1000, int(patience * 0.7))
            current_lr *= 0.6

        if verbose and (t % 200 == 0 or t <= 3):
            elapsed = time.monotonic() - t0
            print(f"    Adam {label}: t={t} ({elapsed:.0f}s) tau={tau:.0f} best={best_score:.6f}")

    return best_a, best_score


def coord_desc(a_init, time_limit, top_k=30, verbose=True, label=""):
    """
    Coordinate descent with incremental convolution tracking.
    Focuses updates on coordinates contributing to top peak positions.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    # Precompute for fast unaffected-max queries
    # We'll maintain prefix and suffix max arrays, updated after each change
    prefix_max = np.zeros(2 * n - 1)
    prefix_max[0] = conv[0]
    for i in range(1, 2 * n - 1):
        prefix_max[i] = max(prefix_max[i - 1], conv[i])
    suffix_max = np.zeros(2 * n - 1)
    suffix_max[2 * n - 2] = conv[2 * n - 2]
    for i in range(2 * n - 3, -1, -1):
        suffix_max[i] = max(suffix_max[i + 1], conv[i])

    def get_unaffected_max(k_start, k_end):
        """Max of conv outside [k_start, k_end]."""
        lo = -1e30
        hi = -1e30
        if k_start > 0:
            lo = prefix_max[k_start - 1]
        if k_end < 2 * n - 2:
            hi = suffix_max[k_end + 1]
        return max(lo, hi)

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        # Determine coordinate order: prioritize coords near peak positions
        if sweep % 10 == 1:
            peak_idx = np.argpartition(conv, -(top_k + 1))[-(top_k + 1):]
            # Build priority: coords that appear in many peak convolutions get higher priority
            priority = np.zeros(n)
            for k in peak_idx:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                priority[js:je + 1] += 1
            order = np.argsort(-priority, kind='stable')
        else:
            order = np.arange(n)

        for i in order:
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = get_unaffected_max(k_start, k_end)

            # a_shifted[j] = a[k_start - i + j] = a[j] for j in [0, k_end-k_start]
            a_seg = a[0:k_end - k_start + 1]

            best_delta = 0.0
            best_new_M = M

            # Try several step sizes (both directions)
            base = a[i]
            for factor in [-0.7, -0.4, -0.2, -0.1, 0.1, 0.2, 0.4, 0.7]:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # New affected conv values
                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_seg
                # Fix the quadratic term at position 2i
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i <= k_end - k_start:
                    new_aff = new_aff.copy()
                    new_aff[pos_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                old_val = a[i]
                a[i] = old_val + best_delta

                # Update convolution incrementally
                changes = 2.0 * best_delta * a_seg
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i <= k_end - k_start:
                    changes = changes.copy()
                    changes[pos_2i] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                # Update prefix/suffix max (approximate - rebuild periodically)
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Rebuild prefix/suffix every 20 sweeps to control error accumulation
        if sweep % 20 == 0:
            prefix_max[0] = conv[0]
            for i in range(1, 2 * n - 1):
                prefix_max[i] = max(prefix_max[i - 1], conv[i])
            suffix_max[2 * n - 2] = conv[2 * n - 2]
            for i in range(2 * n - 3, -1, -1):
                suffix_max[i] = max(suffix_max[i + 1], conv[i])

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label}: sweep={sweep} ({elapsed:.0f}s) best={best_score:.6f}")

    return best_a, best_score


def fine_polish(a_init, time_limit, verbose=True, label=""):
    """Very fine coordinate descent for last-mile improvements."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            a_seg = a[0:k_end - k_start + 1]

            best_delta = 0.0
            best_new_M = M

            # Fine steps around current value
            for factor in [0.995, 0.99, 0.985, 0.98, 1.02, 1.015, 1.01, 1.005]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_seg
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i <= k_end - k_start:
                    new_aff = new_aff.copy()
                    new_aff[pos_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                old_val = a[i]
                a[i] = old_val + best_delta
                changes = 2.0 * best_delta * a_seg
                pos_2i = 2 * i - k_start
                if 0 <= pos_2i <= k_end - k_start:
                    changes = changes.copy()
                    changes[pos_2i] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish {label}: sweep={sweep} ({elapsed:.0f}s) best={best_score:.6f}")

    return best_a, best_score


def evaluate_and_track(a, best_seq, best_score, label=""):
    """Evaluate via the official function and track best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}) ***")
    return best_seq, best_score
