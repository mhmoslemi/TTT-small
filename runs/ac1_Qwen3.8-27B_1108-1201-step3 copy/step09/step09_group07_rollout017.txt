import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def score_normalized(a):
    """Score assuming sum(a)=1 (faster, no sum computation needed)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_start=5.0, T_end=500.0,
                 lr_init=0.05, n_stages=8, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1)**2) / 2.0
            beta = (t_prev - 1.0) / t_curr
            y = project_simplex(x + beta * (x - x_prev))
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def coord_descent_sweep(a, n_sweeps=50, time_limit=None, t0=None, verbose=True):
    """
    Coordinate descent exploiting the fact that conv(a,a)[k] is linear (or 
    quadratic for one k) in each individual coordinate a[i].
    
    For coordinate i, we precompute:
    - base_conv: convolution with a[i] = 0
    - d[k]: coefficient of t in conv[k] (2*a[k-i] for valid k, 0 otherwise)
    
    Then conv_t[k] = base_conv[k] + d[k]*t for k != 2i
    and conv_t[2i] = base_conv[2i] + t^2
    
    We use golden section search to find optimal t for each coordinate.
    """
    n = len(a)
    a = a.copy()
    best_score = score(a)
    
    if t0 is None:
        t0 = time.monotonic()
    
    golden = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
    n_gs_evals = 40  # golden section iterations
    mean_val = a.sum() / n
    
    for sweep in range(n_sweeps):
        if time_limit is not None:
            if time.monotonic() - t0 > time_limit:
                break
        
        improved = False
        # Randomize order for better mixing
        order = np.random.permutation(n)
        
        for i in order:
            if time_limit is not None:
                if time.monotonic() - t0 > time_limit:
                    break
            
            t_old = a[i]
            if t_old < 1e-15 and np.random.random() > 0.1:
                continue  # skip mostly-zero coords (sometimes re-activate)
            
            # Precompute base convolution (a with a[i]=0)
            a_zero = a.copy()
            a_zero[i] = 0.0
            base_conv = fast_convolve(a_zero)
            S_base = a.sum() - t_old
            
            # Influence: d[k] = 2*a[k-i] for k in [i, i+n-1], k != 2i
            # Actually: for k where k-i is in [0, n-1] and k != 2i
            d = np.zeros(2 * n - 1)
            # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            # d[k] = 2*a[k-i] for k in [k_start, k_end], k != 2i
            valid_k = np.arange(k_start, k_end + 1)
            valid_d = 2.0 * a[valid_k - i]
            # Exclude k = 2i
            mask = valid_k != 2 * i
            d[valid_k[mask]] = valid_d[mask]
            
            # For k = 2i: conv[2i] = base_conv[2i] + t^2
            
            # Objective function for golden section
            def obj(t):
                if t < 0:
                    t = 0.0
                S_t = S_base + t
                if S_t < 0.01:
                    return float('inf')
                # Compute max conv
                conv_vals = base_conv + d * t
                if 2 * i < len(conv_vals):
                    conv_vals[2 * i] = base_conv[2 * i] + t * t
                max_c = conv_vals.max()
                return 2.0 * n * max_c / (S_t * S_t)
            
            # Golden section search
            # Determine range: [0, upper]
            upper = max(3.0 * t_old, 3.0 * mean_val, 0.05)
            # Try expanding if needed
            f_upper = obj(upper)
            while f_upper > best_score and upper < 10.0:
                upper *= 1.5
                f_upper = obj(upper)
            
            # Check if current is already optimal
            f_curr = obj(t_old)
            if f_curr >= best_score - 1e-12:
                continue
            
            lo = 0.0
            hi = upper
            a1 = lo + (1.0 - golden) * (hi - lo)
            a2 = lo + golden * (hi - lo)
            f1 = obj(a1)
            f2 = obj(a2)
            
            for _ in range(n_gs_evals):
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = lo + (1.0 - golden) * (hi - lo)
                    f1 = obj(a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + golden * (hi - lo)
                    f2 = obj(a2)
            
            t_opt = (lo + hi) / 2.0
            f_opt = obj(t_opt)
            
            if f_opt < best_score - 1e-14:
                a[i] = t_opt
                best_score = f_opt
                improved = True
        
        if verbose and (sweep + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD sweep {sweep+1} ({elapsed:.0f}s): score={best_score:.8f}")
        
        if not improved and sweep > 2:
            break
    
    return a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                return 2.0 * n * max_c / (S_t * S_t)

            # Bracket
            upper = max(3.0 * v, 3.0 * mean_val, 0.05)
            f_u = eval_t(upper)
            f_0 = eval_t(0.0)
            f_v = eval_t(v)

            # Determine if range needs expanding
            best_local = min(f_u, f_0, f_v)
            if best_local >= best_score - 1e-14:
                continue

            if f_u < best_local:
                upper *= 2.0
            lo = 0.0
            hi = upper

            a1 = lo + (1.0 - golden) * (hi - lo)
            a2 = lo + golden * (hi - lo)
            f1 = eval_t(a1)
            f2 = eval_t(a2)

            for _ in range(n_gs):
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = lo + (1.0 - golden) * (hi - lo)
                    f1 = eval_t(a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + golden * (hi - lo)
                    f2 = eval_t(a2)

            t_opt = (lo + hi) / 2.0
            f_opt = eval_t(t_opt)

            if f_opt < best_score - 1e-14:
                # Update convolution incrementally
                conv_full += 2.0 * (t_opt - v) * infl
                conv_full[2 * i] = c2i + t_opt * t_opt
                S = S_base + t_opt
                a[i] = t_opt
                best_score = f_opt
                improved = True

        if verbose and (sweep + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD sweep {sweep+1} ({elapsed:.0f}s): score={best_score:.8f}")

        if not improved:
            break

    return a, best_score


def lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        Cultur = a_fixed[:size - j]
        C[j:j + len(Cultur), j] = Cultur
    constraints = [C @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in ['ECOS', 'SCS', 'CLARABEL', 'PDLP']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init(n, seed, style=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if style == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x + rng.uniform(0, 6.28))
    elif style == 1:
        a = 1.0 + 0.7 * np.cos(4 * np.pi * x + rng.uniform(0, 6.28))
    elif style == 2:
        mu = rng.uniform(0.3, 0.7)
        a = np.exp(-4.0 * (x - mu) ** 2) + 0.3 * np.exp(-4.0 * (x - (1-mu)) ** 2) + 0.2
    elif style == 3:
        a = 1.0 + 0.3 * np.sin(6 * np.pi * x) + 0.2 * np.sin(10 * np.pi * x + 1.0)
    elif style == 4:
        a = 1.0 + 0.8 * (2 * x - 1) * np.cos(3 * np.pi * x)
    else:
        a = rng.chisquare(df=3, size=n) + 0.5
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov PG, fast coordinate descent,
    LP refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    if best_seq is None:
        best_seq = make_init(960, 42, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Nesterov PG polish ===
    print(f"\n=== Phase 1: NG Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 150.0)
    if t1 > 30:
        a1, _ = nesterov_pg(best_seq, t1, T_start=5.0, T_end=500.0,
                            lr_init=0.04, n_stages=8, steps_per_stage=3000)
        try_update(a1, "ng1")

    # === Phase 2: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 500.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a2, _ = coord_descent(best_seq, n_sweeps=500, time_limit_s=t2, verbose=True)
        try_update(a2, "cd1")

    # === Phase 3: NG again on improved point ===
    print(f"\n=== Phase 3: NG Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 20:
        a3, _ = nesterov_pg(best_seq, t3, T_start=10.0, T_end=800.0,
                            lr_init=0.02, n_stages=6, steps_per_stage=2000)
        try_update(a3, "ng2")

    # === Phase 4: More coordinate descent ===
    print(f"\n=== Phase 4: CD Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, 350.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a4, _ = coord_descent(best_seq, n_sweeps=500, time_limit_s=t4, verbose=True)
        try_update(a4, "cd2")

    # === Phase 5: LP-based refinement ===
    print(f"\n=== Phase 5: LP Refinement (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.15, 200.0)
        if t5 > 30:
            print(f"  Budget: {t5:.0f}s")
            a_cur = best_seq.copy()
            t0_lp = time.monotonic()
            for lp_it in range(200):
                if time.monotonic() - t0_lp > t5:
                    break
                b_br, M_br = lp_best_response(a_cur, solver)
                if b_br is None:
                    break
                alpha = 2.0 / (lp_it + 2)
                a_cur = project_simplex((1 - alpha) * a_cur + alpha * b_br)
                sc = evaluate_safe(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_cur.copy()
                    print(f"    LP iter {lp_it}: {sc:.8f}")
                if lp_it % 20 == 0:
                    print(f"    LP iter {lp_it}: cur_score={evaluate_safe(a_cur):.8f}")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        alt_ns = []
        for delta in [-40, -20, 0, 20, 40, 60, 80, 120, 200]:
            alt_n = n_best + delta
            if 384 <= alt_n <= 3000:
                alt_ns.append(alt_n)
        alt_ns = sorted(set(alt_ns))[:5]

        for alt_n in alt_ns:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.20, 120.0)
            if t_sub < 30:
                break
            print(f"  n={alt_n}, budget={t_sub:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick NG
            t_ng = t_sub * 0.4
            a_alt, _ = nesterov_pg(init_alt, t_ng, T_start=5.0, T_end=300.0,
                                   lr_init=0.05, n_stages=5, steps_per_stage=2000)
            try_update(a_alt, f"ng_n{alt_n}")
            # CD
            rem = deadline - time.monotonic()
            t_cd = min(rem * 0.4, 60.0)
            if t_cd > 10:
                a_alt2, _ = coord_descent(a_alt, n_sweeps=200, time_limit_s=t_cd)
                try_update(a_alt2, f"cd_n{alt_n}")

    # === Phase 7: Multi-start with different inits ===
    print(f"\n=== Phase 7: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for style in [1, 2, 3, 4, 5]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_sub = min(rem * 0.20, 80.0)
            if t_sub < 20:
                break
            a_init = make_init(n_best, seed + style * 100, style)
            t_ng = t_sub * 0.5
            a_ms, _ = nesterov_pg(a_init, t_ng, T_start=3.0, T_end=200.0,
                                  lr_init=0.08, n_stages=5, steps_per_stage=2000)
            try_update(a_ms, f"ms_s{style}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_ms2, _ = coord_descent(a_ms, n_sweeps=100, time_limit_s=min(rem * 0.3, 40.0))
                try_update(a_ms2, f"ms_s{style}_cd")

    # === Phase 8: Final CD polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.7, 120.0)
        a_final, _ = coord_descent(best_seq, n_sweeps=300, time_limit_s=t8, verbose=True)
        try_update(a_final, "final_cd")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, sum={best_seq.sum():.4f}")

    return [float(max(0.0, x)) for x in best_seq]
