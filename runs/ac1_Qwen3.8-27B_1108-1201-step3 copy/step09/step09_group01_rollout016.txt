Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Peak-structured perturbations for escaping local minima
3. Multi-resolution (coarse-to-fine) optimization
4. LP-based global refinement

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS, cp.GLOP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)

def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""

        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    n = len(a)

    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):

            if has_2i:
                idx = 2 * i - k_start
                new_aff = new_aff.copy()
                new_aff[idx] = conv[2 * i] + 2.0 * delta * ai + delta * delta

            new_M = max(unaff_max, float(new_aff.max()))
            new_score = 2.0 * n * new_M / (new_s * new_s)

            if new_score < best_score - 1e-14:
                best_score = new_score
                best_delta = delta

        if best_delta != 0.0:
            # Apply update
            a[i] = max(a[i] + best_delta, 0.0)
            s += best_delta

            # Update convolution
            changes = 2.0 * best_delta * a[0:n].copy()
            # Re-index: slope_vec was a[k_start-i : k_end+1-i]
            # changes for position k is 2*delta*a[k-i]
            conv[k_start:k_end + 1] += 2.0 * best_delta * a[k_start - i:k_end + 1 - i]
            if has_2i:
                idx = 2 * i - k_start
                conv[2 * i] = conv[2 * i] + 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
            M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
            improved = True

    return a, conv, M, s, improved


def coord_descent_optimize(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent optimization with periodic perturbations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_auto_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Multiple passes per "sweep" for faster convergence
        for _ in range(3):
            a, conv, M, s, imp = coord_descent_sweep(a, conv, M, s, n)
            improved = improved or imp

        cur_score = 2.0 * n * M / (s * s)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            # Perturbation to escape local minimum
            rng = np.random.default_rng(sweeps)
            # Peak-aware perturbation
            conv_cur = fast_auto_conv(a)
            M_cur = conv_cur.max()
            peak_thresh = M_cur * 0.9
            peak_pos = np.where(conv_cur >= peak_thresh)[0]

            pressure = np.zeros(n)
            for k in peak_pos:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    prod = a[i] * a[j]
                    pressure[i] += prod
                    if j != i:
                        pressure[j] += prod * 0.5

            pmax = pressure.max()
            if pmax > 1e-20:
                pressure = pressure / pmax

            strength = 0.02 + 0.01 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n)
            a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-10:
                a = a_new / sn
                conv = fast_auto_conv(a)
                M = float(conv.max())
                s = 1.0
                no_improve = 0
            else:
                a = best_a.copy()
                conv = fast_auto_conv(a)
                M = float(conv.max())
                s = 1.0
                no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_auto_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            has_2i = (2 * i >= k_start and 2 * i <= k_end)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            aff = conv[k_start:k_end + 1]
            slope_vec = a[k_start - i:k_end + 1 - i]
            cur_score = 2.0 * n * M / (s * s)

            # Very fine steps
            for factor in [0.999, 0.998, 0.997, 0.996, 0.9995, 1.001, 1.002, 1.003]:
                delta = ai * (factor - 1.0)
                new_ai = ai + delta
                if new_ai < -1e-15:
                    continue

                new_s = s + delta
                if new_s < 1e-10:
                    continue

                new_aff = aff + 2.0 * delta * slope_vec
                if has_2i:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * ai + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                new_score = 2.0 * n * new_M / (new_s * new_s)

                if new_score < cur_score - 1e-15:
                    # Accept
                    a[i] = new_ai
                    s = new_s
                    conv[k_start:k_end + 1] += 2.0 * delta * a[k_start - i:k_end + 1 - i]
                    if has_2i:
                        conv[2 * i] = conv[2 * i] + 2.0 * delta * (new_ai - delta) + delta * delta
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cur_score = new_score
                    improved = True
                    break  # Only one update per coordinate per pass

        if not improved and sweeps > 20:
            break

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, 2.0 * n * fast_auto_conv(best_a).max()


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Fictitious play with Nesterov averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_seq = a.copy()
    b_seq = b.copy()

    best_a = a.copy()
    best_score = compute_score(a)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma_it = 2.0 / (iters + 2.0)
        a_seq = (1.0 - gamma_it) * a_seq + gamma_it * a
        b_seq = (1.0 - gamma_it) * b_seq + gamma_it * b

        for cand in [a, b, a_seq, b_seq, 0.5 * (a_seq + b_seq)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_auto_conv(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def damped_lp_refine(a0, time_limit, solver=None, verbose=False):
    """Damped LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    if n > 2048:
        return a0, compute_score(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_auto_conv(a).max()
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_auto_conv(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DLP it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = compute_score(a_norm)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-length diverse initialization + fast CD
    Phase 2: Intensive CD with peak-aware perturbations from parent
    Phase 3: LP fictitious play with Nesterov averaging
    Phase 4: Damped LP polish
    Phase 5: Fine polishing
    Phase 6: Final intensive optimization
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = compute_score(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-length diverse initialization ===
    print("\n=== Phase 1: Diverse Init + Fast CD ===")
    t1_start = time.monotonic()

    n_values = [768, 896, 1024, 1152, 1280, 1408]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 200:
            break

        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interpolate_seq(parent, n)))
        init_cands.append(("symmetric", init_symmetric(n, rng)))
        init_cands.append(("bimodal", init_bimodal(n, rng)))
        init_cands.append(("gaussian", init_gaussian(n, rng)))
        init_cands.append(("tent", init_tent(n, rng)))
        init_cands.append(("uniform", np.ones(n) / n))

        for name, cand in init_cands:
            t_left = 200.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(3.0, min(8.0, t_left / 4.0))
            a_opt, s_opt, _ = coord_descent_optimize(cand, t_alloc, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, top-5: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:5]]))

    for s_c, a_c, n_c, it_c in candidates[:10]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Intensive CD from best with perturbations ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(8):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(rem * 0.2, 100.0)
            print(f"  CD round {rnd+1}/8, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")
            a_opt, s_opt, sweeps = coord_descent_optimize(
                current, t_alloc, verbose=True, label=f"ph2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_fcd_r{rnd}")

            # Structured perturbation to escape local optimum
            if rnd < 6:
                conv_cur = fast_auto_conv(best_seq)
                M_cur = conv_cur.max()
                # Peak positions
                peak_thresh = M_cur * 0.85
                peak_pos = np.where(conv_cur >= peak_thresh)[0]

                if len(peak_pos) > 0:
                    # Identify "hot" coordinates
                    hot = np.zeros(n_best)
                    for k in peak_pos:
                        i_start = max(0, k - n_best + 1)
                        i_end = min(n_best - 1, k)
                        for i in range(i_start, i_end + 1):
                            j = k - i
                            hot[i] += best_seq[i] * best_seq[j]
                            if j != i:
                                hot[j] += best_seq[i] * best_seq[j] * 0.3

                    hmax = hot.max()
                    if hmax > 1e-20:
                        hot = hot / hmax

                    # Perturb: reduce hot, increase cold
                    strength = 0.03 + 0.015 * rnd
                    noise = rng.standard_normal(n_best)
                    current = best_seq * (1.0 - strength * hot + 0.4 * strength * noise)
                    current = np.maximum(current, 0)
                    sc = current.sum()
                    if sc > 1e-10:
                        current = current / sc
                else:
                    current = best_seq * (1.0 + 0.05 * rng.standard_normal(n_best))
                    current = np.maximum(current, 0)
                    sc = current.sum()
                    if sc > 1e-10:
                        current = current / sc
            else:
                current = best_seq.copy()

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: NFP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 896, 1152, 768]:
            if alt_n != n_best and 512 <= alt_n <= 1536:
                fp_lengths.append(alt_n)
        fp_lengths = fp_lengths[:3]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  NFP n={n_fp}, budget={t_alloc:.0f}s")

            if best_seq is not None:
                initial = interpolate_seq(best_seq, n_fp)
            else:
                initial = None

            a_fp, score_fp, iters = nesterov_fictitious_play(
                n_fp, t_alloc, initial=initial, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"nfp_n{n_fp}")

            # Damped LP polish
            rem = deadline - time.monotonic() - 150
            if rem > 30:
                t_dl = min(rem * 0.3, 80.0)
                a_dl, _ = damped_lp_refine(a_fp, t_dl, solver=solver, verbose=False)
                best_seq, best_score = check_and_update(a_dl, best_seq, best_score, f"dlp_n{n_fp}")

    # === Phase 4: More CD rounds from the best ===
    print(f"\n=== Phase 4: More CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        for rnd in range(4):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 80.0)
            a_opt, s_opt, _ = coord_descent_optimize(
                best_seq, t_alloc, verbose=False, label=f"ph4r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph4_r{rnd}")

            # Light perturbation
            if rnd < 3:
                noise = rng.standard_normal(len(best_seq))
                perturbed = best_seq * (1.0 + 0.02 * noise)
                perturbed = np.maximum(perturbed, 0)
                sc = perturbed.sum()
                if sc > 1e-10:
                    best_seq_for_cd = perturbed / sc
                else:
                    best_seq_for_cd = best_seq.copy()

    # === Phase 5: Fine polishing ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_fine = min(rem * 0.5, 120.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(
                best_seq, t_fine, verbose=True, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd_1")

        # Another fine pass
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_f2, s_f2 = fine_coord_descent(
                best_seq, min(rem * 0.6, 40.0), label="fine2")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_cd_2")

    # === Phase 6: Try alternative lengths from best ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            for alt_n in [n_best + 128, n_best + 256, n_best - 128]:
                if alt_n <= 512 or alt_n > 2048:
                    continue
                if rem < 15:
                    break
                t_alt = min(rem * 0.35, 40.0)
                rem = deadline - time.monotonic() - 10
                if rem < 10:
                    break
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt, _ = coord_descent_optimize(
                    init_alt, t_alt, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = compute_score(best_seq)

    # Final verification
    fs = compute_score(best_seq)
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
