import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

                # c_k changes by 2*d*a[k-i] for k in [k_start, k_end], k != 2i
                # c_{2i} changes by d^2 (quadratic term)
                a_shifted = a[k_start - i:k_end - i + 1]  # a[k-i] for k in range

                # The change to c_k (before renorm):
                # For k != 2i: +2*d*a[k-i]
                # For k = 2i: +2*v*d + d^2
                delta_c = 2.0 * d * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    delta_c = delta_c.copy()
                    delta_c[idx] = 2.0 * v * d + d * d

                # New c values (before renorm)
                new_c_affected = c[k_start:k_end + 1] + delta_c

                # After renormalization: divide by new_sum^2
                new_max_affected = float(new_c_affected.max()) / (new_sum * new_sum)
                new_M_unaff = M_unaff / (new_sum * new_sum)
                new_M = max(new_max_affected, new_M_unaff)
                new_score = 2.0 * n * new_M

                if new_score < best_new_score:
                    best_new_score = new_score
                    best_delta = d

            if best_delta != 0.0 and best_new_score < best_score - 1e-14:
                # Apply the update
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a = np.ones(n) / n
                    c = fast_convolve(a)
                    M = float(c.max())
                else:
                    a = a / s_new
                    c = fast_convolve(a)
                    M = float(c.max())

                if 2.0 * n * M < best_score:
                    best_score = 2.0 * n * M
                    best = a.copy()
                    sweep_improvement += (best_score - 2.0 * n * M) * 0 + (2.0 * n * M - best_score)  # just tracking
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                # Perturb to escape local minimum
                mask = a > 1e-10
                if mask.sum() > 10:
                    a[mask] *= np.exp(rng.global_rng.standard_normal(mask.sum()) * 0.02)
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        c = fast_convolve(a)
                        M = float(c.max())
                    no_improve_sweeps = 0
                else:
                    break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best, best_score


def pair_cd(a0, time_limit, verbose=True):
    """
    Pair coordinate descent: try changing pairs of coordinates simultaneously.
    Focuses on pairs within the "active set" (contributing to max).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_convolve(a)
    M = float(c.max())
    best_score = 2.0 * n * M
    best = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = float(c.max())

        if 2.0 * n * M < best_score:
            best_score = 2.0 * n * M
            best = a.copy()

        # Find active set: positions within 10% of max
        threshold = M * 0.95
        active_k = np.where(c >= threshold)[0]

        if len(active_k) == 0:
            break

        # For each active k, the contributing indices are [max(0,k-n+1), min(n-1,k)]
        # Gather all active indices
        active_indices = set()
        for k in active_k:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                if a[i] > 1e-8:
                    active_indices.add(i)

        active_indices = sorted(active_indices)
        if len(active_indices) < 2:
            break

        # Try random pairs from active set
        n_pairs_per_iter = min(200, len(active_indices) * 2)
        rng_local = np.random.default_rng(it)
        for _ in range(n_pairs_per_iter):
            i = active_indices[rng_local.integers(len(active_indices))]
            j = active_indices[rng_local.integers(len(active_indices))]
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue

            # Try transferring mass from j to i (and vice versa)
            # Change a[i] += d, a[j] -= d (keeps sum constant)
            # Best d range: (-a[j], a[i])
            d_max = min(a[i], a[j]) * 0.5
            for d_frac in [-0.3, -0.1, -0.03, 0.03, 0.1, 0.3]:
                d = d_max * d_frac
                a_trial = a.copy()
                a_trial[i] += d
                a_trial[j] -= d
                if a_trial[i] < 0 or a_trial[j] < 0:
                    continue
                c_trial = fast_convolve(a_trial)
                M_trial = float(c_trial.max())
                score_trial = 2.0 * n * M_trial
                if score_trial < best_score - 1e-13:
                    a = a_trial
                    c = c_trial
                    M = M_trial
                    best_score = score_trial
                    best = a.copy()
                    break

    return best, best_score


def fw_alternating(a0, time_limit, verbose=True):
    """
    Frank-Wolfe alternating minimization for global optimization.
    Optimizes b given a (via FW on simplex), then a given b.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    b_conv = convolve_ab(a, b)
    best = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1

        # --- Optimize b given a ---
        n_fw = 100
        for step in range(n_fw):
            k_star = int(np.argmax(b_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))

            t = 2.0 / (step + 2.0)

            # Away step
            if step >= 10:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = b * grad_full
                i_away = int(np.argmax(away_scores))
                if b[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                    t_away = min(t * 0.5, b[i_away] * 0.8)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)
                    b_conv[ks_s:ks_e + 1] -= t_away * a
                    b[i_away] = max(b[i_away] - t_away, 0.0)

            # Standard FW step
            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)
            b_conv = (1.0 - t) * b_conv
            b_conv[ks_s:ks_e + 1] += t * a
            b = (1.0 - t) * b
            b[j_star] += t

        # Evaluate b
        b_score = 2.0 * n * fast_convolve(b).max()
        if b_score < best_score:
            best_score = b_score
            best = b.copy()

        # --- Optimize a given b ---
        a_conv = convolve_ab(b, a)
        for step in range(n_fw):
            k_star = int(np.argmax(a_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))

            t = 2.0 / (step + 2.0)

            if step >= 10:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = a * grad_full
                i_away = int(np.argmax(away_scores))
                if a[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                    t_away = min(t * 0.5, a[i_away] * 0.8)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)
                    a_conv[ks_s:ks_e + 1] -= t_away * b
                    a[i_away] = max(a[i_away] - t_away, 0.0)

            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)
            a_conv = (1.0 - t) * a_conv
            a_conv[ks_s:ks_e + 1] += t * b
            a = (1.0 - t) * a
            a[j_star] += t

        # Evaluate a
        a_score = 2.0 * n * fast_convolve(a).max()
        if a_score < best_score:
            best_score = a_score
            best = a.copy()

        # Also try average
        avg = 0.5 * (a + b)
        avg = avg / avg.sum()
        avg_score = 2.0 * n * fast_convolve(avg).max()
        if avg_score < best_score:
            best_score = avg_score
            best = avg.copy()

        # Sync periodically
        if round_num % 5 == 0:
            a = best.copy()
            b = best.copy()
            b_conv = convolve_ab(a, b)

        if verbose and round_num % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW round={round_num} ({elapsed:.0f}s): best={best_score:.6f}")

    return best, best_score


def lp_refine(a0, time_limit, verbose=True):
    """LP-based refinement using cvxpy."""
    try:
        import cvxpy as cp
    except ImportError:
        print("    cvxpy not available, skipping LP")
        return a0.copy(), 2.0 * len(a0) * fast_convolve(a0).max()

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s

    best = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    # Find solver
    solver = None
    for sname in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sname)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = sname
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    print(f"    LP solver: {solver}")

    # Build T matrix: T @ b = conv(a, b), shape (2n-1, n)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b

        score_b = 2.0 * n * fast_convolve(b_val).max()
        if score_b < best_score:
            best_score = score_b
            best = b_val.copy()

        # Damped update
        alpha = 3.0 / (it + 2.0)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best = a_new.copy()

        a = a_new

        # Rebuild T
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best, best_score


def fine_cd_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_convolve(a)
    M = float(c.max())
    best_score = 2.0 * n * M
    best = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Very fine factors
    factors = [0.995, 0.998, 0.999, 1.001, 1.002, 1.005]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            v = a[i]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = c[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = c[k_end + 1:].max()
            else:
                unaff_right = -1e30
            M_unaff = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_score = 2.0 * n * M

            for factor in factors:
                d = v * (factor - 1.0)
                new_sum = 1.0 + d
                if new_sum < 1e-10:
                    continue

                delta_c = 2.0 * d * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    delta_c = delta_c.copy()
                    delta_c[idx] = 2.0 * v * d + d * d

                new_c_affected = c[k_start:k_end + 1] + delta_c
                new_max_affected = float(new_c_affected.max()) / (new_sum * new_sum)
                new_M_unaff = M_unaff / (new_sum * new_sum)
                new_M = max(new_max_affected, new_M_unaff)
                new_score = 2.0 * n * new_M

                if new_score < best_new_score - 1e-15:
                    best_new_score = new_score
                    best_delta = d

            if best_delta != 0.0 and best_new_score < best_score - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                s_new = a.sum()
                if s_new < 1e-15:
                    break
                a = a / s_new
                c = fast_convolve(a)
                M = float(c.max())
                if 2.0 * n * M < best_score:
                    best_score = 2.0 * n * M
                    best = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best, best_score


def periodic_init(n, rng):
    """Initialize with a periodic pattern."""
    period = int(rng.integers(2, 9))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_window_init(n, rng):
    """Initialize with a cosine window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.5, 3.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-strategy optimization:
    1. FW alternating minimization (global)
    2. Smooth Riemannian optimization (adaptive log-sum-exp)
    3. Coordinate descent with renormalization (medium steps)
    4. Pair CD (breaking local minima)
    5. LP refinement (active-set tightening)
    6. Fine CD polishing (last-mile)
    7. Multi-length exploration
    """
    deadline = time.monotonic() + 1790.0
    local_rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))

    n_best = len(best_seq)
    print(f"Starting: best={best_score:.6f}, n={n_best}")

    # === Phase 1: FW Global Optimization on parent ===
    remaining = deadline - time.monotonic() - 600
    if remaining > 100:
        t1 = min(remaining * 0.3, 200.0)
        print(f"\n=== Phase 1: FW (budget={t1:.0f}s) ===")
        a_fw, s_fw = fw_alternating(best_seq, t1, verbose=True)
        best_seq, best_score = check_and_update(a_fw, best_seq, best_score, "fw_parent")

        # Also try diverse inits
        inits = [
            cosine_window_init(n_best, local_rng),
            periodic_init(n_best, local_rng),
        ]
        for idx, init in enumerate(inits):
            remaining = deadline - time.monotonic() - 500
            if remaining < 60:
                break
            t_alloc = min(remaining * 0.2, 100.0)
            a_fwi, s_fwi = fw_alternating(init, t_alloc, verbose=False)
            best_seq, best_score = check_and_update(a_fwi, best_seq, best_score, f"fw_init{idx}")

    # === Phase 2: Smooth Riemannian Optimization ===
    remaining = deadline - time.monotonic() - 400
    if remaining > 60:
        t2 = min(remaining * 0.4, 250.0)
        print(f"\n=== Phase 2: Smooth Riemannian (budget={t2:.0f}s) ===")
        a_sm, s_sm = smooth_riemannian_optimize(best_seq, t2, verbose=True)
        best_seq, best_score = check_and_update(a_sm, best_seq, best_score, "smooth")

    # === Phase 3: Coordinate Descent (medium steps) ===
    remaining = deadline - time.monotonic() - 300
    if remaining > 60:
        t3 = min(remaining * 0.45, 200.0)
        print(f"\n=== Phase 3: CD medium (budget={t3:.0f}s) ===")
        a_cd, s_cd = coord_descent_renorm(best_seq, t3, verbose=True)
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_medium")

    # === Phase 4: Pair CD ===
    remaining = deadline - time.monotonic() - 200
    if remaining > 60:
        t4 = min(remaining * 0.4, 120.0)
        print(f"\n=== Phase 4: Pair CD (budget={t4:.0f}s) ===")
        a_pair, s_pair = pair_cd(best_seq, t4, verbose=True)
        best_seq, best_score = check_and_update(a_pair, best_seq, best_score, "pair_cd")

    # === Phase 5: LP Refinement ===
    remaining = deadline - time.monotonic() - 150
    if remaining > 60:
        t5 = min(remaining * 0.5, 120.0)
        print(f"\n=== Phase 5: LP (budget={t5:.0f}s) ===")
        try:
            a_lp, s_lp = lp_refine(best_seq, t5, verbose=True)
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "lp")
        except Exception as e:
            print(f"  LP failed: {e}")

    # === Phase 6: Fine CD Polishing ===
    remaining = deadline - time.monotonic() - 80
    if remaining > 30:
        t6 = min(remaining * 0.5, 80.0)
        print(f"\n=== Phase 6: Fine CD (budget={t6:.0f}s) ===")
        a_fine, s_fine = fine_cd_polish(best_seq, t6, verbose=True)
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

    # === Phase 7: Try different lengths ===
    remaining = deadline - time.monotonic() - 30
    if remaining > 60:
        print(f"\n=== Phase 7: Multi-length (budget={remaining:.0f}s) ===")
        for n_try in [512, 2048]:
            remaining = deadline - time.monotonic() - 30
            if remaining < 40:
                break
            t7 = min(remaining * 0.4, 80.0)
            print(f"  Trying n={n_try} (budget={t7:.0f}s)")

            # Initialize by resampling parent
            if parent is not None and len(parent) != n_try:
                # Linear interpolation
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n_try)
                init_try = np.interp(x_new, x_old, parent)
            else:
                init_try = np.ones(n_try) / n_try
            init_try = np.maximum(init_try, 0)
            init_try = init_try / init_try.sum()

            # Quick smooth optimization
            a_try, s_try = smooth_riemannian_optimize(init_try, t7 * 0.5, verbose=False)
            # Quick CD
            a_try2, s_try2 = coord_descent_renorm(a_try, t7 * 0.4, verbose=False)
            best_seq, best_score = check_and_update(a_try2, best_seq, best_score, f"n{n_try}")

    # === Phase 8: Final fine polish ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 15:
        t8 = min(remaining * 0.7, 30.0)
        print(f"\n=== Phase 8: Final polish (budget={t8:.0f}s) ===")
        a_fin, s_fin = fine_cd_polish(best_seq, t8, verbose=False)
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
