def smoothed_mirror_descent(a0, time_limit, tau_init=100.0, tau_min=10.0, lr_init=0.1):
    """
    Smoothed subgradient method using log-sum-exp smoothing.
    Uses Nesterov averaging for better convergence.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score(a)
    
    # Nesterov average
    avg_a = a.copy()
    total_weight = 0.0
    current_weight = 1.0
    
    t0 = time.monotonic()
    it = 0
    tau = tau_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Compute smooth gradient using log-sum-exp
        # weights[k] = exp(tau * conv[k]) / sum(exp(tau * conv[.]))
        # For numerical stability, subtract max
        shifted = tau * (conv - M)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / exp_shifted.sum()
        
        # Gradient: grad[j] = sum_k weights[k] * a[k-j] (if valid)
        grad = np.zeros(n)
        # Only iterate over significant weights
        threshold = 1e-8
        significant = np.where(weights > threshold)[0]
        for k in significant:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end+1] += weights[k] * a[k-i_end:k-i_start+1][::-1]
        
        # Mirror descent step (multiplicative weights on the simplex)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Update Nesterov average
        avg_a = (total_weight * avg_a + current_weight * a_new) / (total_weight + current_weight)
        total_weight += current_weight
        
        a = a_new

        if cur_score < best_score - 1e-12:

            best_a = a.copy()
            no_improve = 0
