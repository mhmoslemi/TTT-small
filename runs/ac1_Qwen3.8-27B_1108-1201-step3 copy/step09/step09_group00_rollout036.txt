Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP-alternating with Nesterov
acceleration, aggressive LogSumExp L-BFGS annealing, peak shaving, and
multi-length exploration. Key improvements over previous: uses scipy HiGHS for

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]
    subject to sum(b)=1, b>=0. Returns (b, M).
    Uses scipy linprog with HiGHS.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build T matrix
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    # LP: minimize M
    # s.t. T @ b - M <= 0  (size constraints)
    #      sum(b) = 1      (1 constraint)
    #      b >= 0
    # Variables: [b_0, ..., b_{n-1}, M]
    
    A_ub = np.column_stack([T, -np.ones(size)])
    b_ub = np.zeros(size)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                           'primal_feasibility_tolerance': 1e-10})
    
    if res.status != 0:
        return None, None
    
    b_val = np.maximum(res.x[:n], 0)
    M_val = float(res.x[n])
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val

        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)

            y = a.copy()
        else:
            y = y / sy

        a = np.maximum(a, 0)

            a = a / sa

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP-alternating with Nesterov
acceleration (HiGHS solver), aggressive LogSumExp L-BFGS with FFT-accelerated
gradients, direct peak shaving, parametric initialization, and multi-length
exploration. Key improvements: scipy HiGHS for LP (much faster than CVXPY),
FFT-based gradient computation, better annealing schedule, and targeted
peak-directional coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


# Global variables for L-BFGS callback workaround
_LBFGS_N = 0
_LBFGS_T = 1.0


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]
    subject to sum(b)=1, b>=0. Uses scipy linprog with HiGHS.
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1

    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    A_ub = np.column_stack([T, -np.ones(size)])
    b_ub = np.zeros(size)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, None

    if res.status != 0:
        return None, None

    b_val = np.maximum(res.x[:n], 0)
    M_val = float(res.x[n])
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val

    """Nesterov dual averaging LP refinement with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)
    n_fail = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y)
        if b_new is None:
            n_fail += 1
            if n_fail > 20:
                break
            continue
        n_fail = 0

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                noise_scale = 0.02 / np.sqrt(stall_count)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def compute_lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp smoothed objective and gradient (FFT-accelerated)."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj = (2.0 * n / T_temp) * (max_Tb + np.log(exp_Tb.sum()))

    # Gradient via FFT: grad_a[j] = 2 * cross_corr(w, a)[j]
    # cross_corr(w, a)[j] = (flip(a) * w)[n-1+j]
    flip_a = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    FA = rfft(flip_a, fft_size)
    FW = rfft(w, fft_size)
    cross = irfft(FA * FW, fft_size)
    grad_a = 2.0 * cross[n - 1:n - 1 + n]

    inner = np.dot(grad_a, a)
    grad_z = a * (grad_a - inner) * (2.0 * n / T_temp)

    return obj, grad_z


def _lse_func(z):
    obj, _ = compute_lse_obj_grad(z, _LBFGS_N, _LBFGS_T)
    return obj


def _lse_grad(z):
    _, g = compute_lse_obj_grad(z, _LBFGS_N, _LBFGS_T)
    return g


def lse_lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and annealed temperature."""
    global _LBFGS_N, _LBFGS_T
    n = len(a0)
    _LBFGS_N = n
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        _LBFGS_T = stage_T
        iters = max(50, min(maxiter_per_stage, int(remaining * 1.5)))

        try:
            res = minimize(
                fun=_lse_func, x0=z, jac=_lse_grad,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_shaving(a0, time_limit, n_steps=200000):
    """Direct subgradient method targeting the exact argmax of convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g *= 2.0

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-8, lr)

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-18:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-16:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def parametric_init(n, rng):
    """Try parametric window shapes and return the best one."""
    x = np.linspace(-1, 1, n)
    candidates = []

    for p in [0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = (1.0 - x * x) ** p
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0]:
        a = np.exp(-p * x * x)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    for p in [1.0, 2.0, 3.0, 4.0, 6.0]:
        a = np.cos(np.pi * x / 2.0) ** p
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    for p in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]:
        a = 1.0 / (1.0 + p * x * x)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    # Hamming-like
    a = 0.54 + 0.46 * np.cos(np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        candidates.append(a / s)

    # Blackman-like
    a = 0.42 + 0.5 * np.cos(np.pi * x) + 0.08 * np.cos(2.0 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        candidates.append(a / s)

    best_a = None
    best_score = float('inf')
    for a in candidates:
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_score = 2.0
    return best_a, best_score
