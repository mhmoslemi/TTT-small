Let me think about one more thing: the **efficient coordinate descent**.

For coordinate i, changing a[i] by delta affects conv[k] for k in [i, i+n-1] (clipped to [0, 2n-2]). The change is:
- conv[k] += 2*delta*a[k-i] for k != 2i
- conv[2i] += 2*delta*a[i] + delta^2 (quadratic term)

The new max is max(unaffected_max, max of affected range after update).

To find the optimal delta, we need to find the largest decrease in M. This is a 1D optimization problem:
- f(delta) = max(unaffected_max, max_{k in affected} (conv[k] + 2*delta*a_shifted[k] + [quadratic term at 2i]))
- We want to minimize f(delta) over delta such that a[i] + delta >= 0.

The function f(delta) is piecewise linear (except for the quadratic term at one point). The minimum occurs at a breakpoint or where the derivative changes sign.

For efficiency, I can:
1. Compute the "breakpoints" where the max switches between different conv values.
2. Check f(delta) at each breakpoint and at delta=0.

But this is O(n) per coordinate, which is the same as the naive approach. The "unaffected max" trick just avoids recomputing the full max.

A simpler approach: try a few candidate deltas (e.g., the values that would make each affected conv[k] equal to M, or equal to unaffected_max). This gives O(n) candidates, and we evaluate f at each.

Actually, the previous code's approach of trying a few multiplicative factors is simpler and probably good enough.

Let me also think about **pairwise coordinate descent**: optimizing a[i] and a[j] together. This is a 2D problem and can escape 1D local minima. But it's O(n^2) per sweep, which might be too slow for n=1024.

I'll stick with single-coordinate descent but with more aggressive step sizes.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization (primary engine)
2. Fast coordinate descent with subgradient awareness
3. Smoothed log-sum-exp gradient descent for local escape
4. Multi-scale exploration with diverse restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """
    Solve LP: max sum(b) s.t. T@b <= M, b >= 0.
    Returns normalized b or None on failure.
    """
    n = len(a)
    T = build_T(a)
    
    b_var = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(b_var))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_alternating(a_init, time_limit, solver=None, verbose=True):
    """
    LP-based alternating minimization.
    Fixed a -> solve LP for best b -> update a via damped step.
    """
    if solver is None:
        solver = _find_solver()
    
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = lp_best_response(a, M, solver)
        if b is None:
            break
        
        b_conv = fast_convolve(b)
        b_score = 2.0 * n * float(b_conv.max())
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()
        
        # Damped update with adaptive step
        alpha = min(0.5, 2.0 / (it + 3.0))
        a_new = (1 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a_conv = fast_convolve(a_new)
        a_new_score = 2.0 * n * float(a_conv.max())
        
        if a_new_score < best_score:
            best_score = a_new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = a_conv
        M = float(conv.max())
        
        if no_improve > 200:
            # Perturb to escape
            rng = np.random.default_rng(it)
            a = a + rng.standard_normal(n) * 0.02
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            conv = fast_convolve(a)
            M = float(conv.max())
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP alt it={it} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")
    
    return best_a, best_score

def coord_descent(a_init, time_limit, verbose=True, label=""):
    """
    Fast coordinate descent with subgradient-based step selection.
    Uses the "unaffected max" trick for O(n) per coordinate.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0
    
    # Precompute prefix and suffix max for fast unaffected max queries
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Randomize order for diversity
        order = np.random.default_rng(sweeps * 7 + 13).permutation(n)
        
        for i in order:
            if a[i] < 1e-18:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            k_len = k_end - k_start + 1
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            # a_shifted[k] = a[k - i] for k in [k_start, k_end]
            a_shifted = a[0:k_len]
            
            # Find optimal delta by trying key candidates
            # The affected conv values after delta: conv[k_start:k_end+1] + 2*delta*a_shifted
            # (plus quadratic correction at k=2i)
            current_affected = conv[k_start:k_end + 1].copy()
            
            # Key delta values: those that make some affected[k] = M or = unaff_max
            # delta_k = (target - affected[k]) / (2*a_shifted[k]) for a_shifted[k] > 0
            candidates = set()
            candidates.add(0.0)
            
            # Try to reduce the current max
            max_idx_in_affected = int(np.argmax(current_affected))
            if a_shifted[max_idx_in_affected] > 1e-18:
                # Delta that would reduce this peak to unaff_max
                target = min(M, unaff_max) if unaff_max > -1e29 else M
                d = (target - current_affected[max_idx_in_affected]) / (2.0 * a_shifted[max_idx_in_affected])
                candidates.add(d)
            
            # Also try proportional steps
            for factor in [0.5, 0.3, 0.1, -0.1, -0.3, -0.5]:
                candidates.add(a[i] * factor)
            
            # Also try to set a[i] to 0
            candidates.add(-a[i])
            
            best_delta = 0.0
            best_new_M = M
            
            for delta in candidates:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = current_affected + 2.0 * delta * a_shifted
                # Quadratic correction at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0:
                # Re-verify with finer search around best_delta
                for frac in [0.5, 0.7, 0.9, 1.0, 1.1, 1.3]:
                    d_try = best_delta * frac
                    new_val = a[i] + d_try
                    if new_val < -1e-15:
                        continue
                    new_affected = current_affected + 2.0 * d_try * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_affected[idx] = conv[2 * i] + 2.0 * d_try * a[i] + d_try * d_try
                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < best_new_M - 1e-15:
                        best_new_M = new_M
                        best_delta = d_try
                
                if best_delta != 0.0 and best_new_M < M - 1e-14:
                    a[i] = max(a[i] + best_delta, 0.0)
                    # Update conv incrementally
                    changes = 2.0 * best_delta * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        changes = changes.copy()
                        changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                    conv[k_start:k_end + 1] += changes
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")
    
    return best_a, best_score


def smoothed_descent(a_init, time_limit, verbose=True):
    """
    Gradient descent on the smoothed objective:
    f(a) = (1/tau) * log(sum exp(tau * (a*a)[k]))
    with normalization constraint handled by re-normalizing after each step.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    tau = 5.0 / max(float(conv.max()), 1e-10)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.7
            tau *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Compute softmax weights over conv values
        # Use top-k for efficiency
        k = min(50, 2 * n - 1)
        top_k_idx = np.argpartition(conv, -k)[-k:]
        top_k_vals = conv[top_k_idx]
        
        scaled = tau * top_k_vals
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_k_idx)) / len(top_k_idx)
        else:
            w = w / w_sum
        
        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for idx in range(len(top_k_idx)):
            if w[idx] < 1e-14:
                continue
            k = int(top_k_idx[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Multiplicative (mirror) descent step
        a_new = a * np.exp(-lr * grad / max(grad.max(), 1e-10))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def diverse_init(n, seed):
    """Generate a diverse initial sequence."""
    rng = np.random.default_rng(seed)
    strategy = rng.integers(0, 5)
    
    if strategy == 0:
        # Cosine window
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(0.5, 3.0)
        a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    elif strategy == 1:
        # Two humps
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.1, 0.25)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif strategy == 2:
        # Periodic
        period = int(rng.integers(3, 15))
        phases = rng.uniform(0, 1, period)
        a = np.array([0.3 + 0.7 * phases[i % period] for i in range(n)])
    elif strategy == 3:
        # Linear ramp with noise
        t = np.linspace(0, 1, n)
        a = t + 0.3
    else:
        # Random log-normal
        a = np.exp(rng.standard_normal(n) * 0.5)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}")
    return [float(x) for x in best_a]
