def peak_spread_move(a, rng, n_peaks=5, spread_frac=0.3):
    """
    Identify the top peaks in the convolution and spread mass away from them.
    For each peak, reduce the contributing coordinates and add mass to random cold positions.
    """
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    M = conv.max()
    
    # Get top peaks
    peak_threshold = M * (1.0 - 0.1 * (n_peaks / 10))
    peak_positions = np.where(conv >= peak_threshold)[0]
    
    if len(peak_positions) == 0:
        return a
    
    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_positions[:n_peaks]:
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                product = a[i] * a[j]
                pressure[i] += product
                if j != i:
                    pressure[j] += 0.5 * product
    
    # Normalize pressure
    p_max = pressure.max()
    if p_max < 1e-20:
        return a
    pressure = pressure / p_max
    
    # Cold positions: low pressure
    cold_mask = pressure < 0.3
    if cold_mask.sum() < 10:
        cold_mask = pressure < np.percentile(pressure, 30)
    
    # Reduce high-pressure coords, add to cold coords
    reduction = 0.1 * spread_frac
    a_hot = a * pressure * reduction
    total_reduced = a_hot.sum()
    
    a = a - a_hot
    # Distribute to cold positions
    cold_weights = np.zeros(n)
    cold_weights[cold_mask] = np.ones(cold_mask.sum())
    a += total_reduced * cold_weights / cold_weights.sum()
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def annealing_perturb(a, rng, temperature):
    """Multiplicative perturbation with given temperature."""
    n = len(a)
    noise = rng.standard_normal(n) * temperature
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_restructure(a, rng, frac=0.2):
    """Randomly zero out and redistribute."""
    n = len(a)
    a = a.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = np.maximum(a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3), 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.2))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def block_coord_descent(a0, time_limit, block_size=4, verbose=False, label=""):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with adaptive bidirectional step search
2. Simulated annealing with convolution-structure-informed perturbations
3. Peak-targeted mass redistribution
4. Multi-length exploration
5. Progressive fine polishing

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with adaptive step sizes.
    Normalizes to sum=1, minimizes 2n*max(conv).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Adaptive step factors: start coarse, get finer over sweeps
        if sweeps <= 5:
            factors = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01,
                       -0.3, -0.2, -0.15, -0.1, -0.08, -0.05, -0.03, -0.02, -0.01]
        elif sweeps <= 15:
            factors = [0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01, 0.005,
                       -0.15, -0.1, -0.08, -0.05, -0.03, -0.02, -0.01, -0.005]
        else:
            factors = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002,
                       -0.05, -0.03, -0.02, -0.01, -0.005, -0.003, -0.002]

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Compute unaffected max
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end + 1 < 2 * n - 1:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # If unaffected max already exceeds current M, no point in changing
            # (but we might still reduce M if the affected part was the max)
            affected = conv[k_start:k_end + 1]
            old_ai = a[i]

            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = factor * old_ai
                new_val = old_ai + delta
                if new_val < -1e-14:
                    continue
                if new_val > 5.0:
                    continue

                # Compute new affected conv values
                # For k in [k_start, k_end]: partner = k - i
                # Change = 2*delta*a[partner] if partner != i
                #          2*delta*old_ai + delta^2 if partner == i
                partner = np.arange(k_start, k_end + 1) - i
                # partner is always in [0, n-1] for valid k range
                new_affected = affected + 2.0 * delta * a[partner]
                # Fix diagonal (k = 2i)
                diag_k = 2 * i
                if k_start <= diag_k <= k_end:
                    idx_diag = diag_k - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_diag] = affected[idx_diag] + 2.0 * delta * old_ai + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                # Apply update
                old_ai = a[i]
                a[i] = max(old_ai + best_delta, 0.0)
                s += best_delta

                # Update conv in-place
                partner = np.arange(k_start, k_end + 1) - i
                change_vals = 2.0 * best_delta * a[partner]
                # a[partner] is current a (unchanged for partner != i)
                # But a[i] just changed, so if partner == i, a[partner] = new a[i
                # We need old a[i] for the formula
                diag_k = 2 * i
                if k_start <= diag_k <= k_end:
                    idx_diag = diag_k - k_start
                    change_vals = change_vals.copy()
                    change_vals[idx_diag] = 2.0 * best_delta * old_ai + best_delta * best_delta
                else:
                    # Check if any partner equals i (shouldn't happen for k in [i, i+n-1] with k != 2i)
                    # Actually partner = k-i, and partner = i when k = 2i, which we handled above
                    pass

                conv[k_start:k_end + 1] += change_vals
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.2e}")

    return best_a, best_score

def peak_spread(a, rng, n_top=10):
    """
    Identify top convolution peaks and redistribute mass from contributing
    coordinates to cold (low-convolution) positions.
    """
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    M = conv.max()

    # Get top peak positions
    top_k = min(n_top, len(conv))
    peak_idx = np.argsort(conv)[-top_k:]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            prod = a[i] * a[j]
            pressure[i] += prod
            if j != i:
                pressure[j] += 0.5 * prod

    p_max = pressure.max()
    if p_max < 1e-20:
        return a
    pressure = pressure / p_max

    # Reduce hot coordinates
    reduction_rate = 0.15
    mass_to_redistribute = (a * pressure * reduction_rate).sum()

    a_reduced = a - a * pressure * reduction_rate
    a_reduced = np.maximum(a_reduced, 0)

    # Add mass to cold coordinates (those with low pressure)
    cold_threshold = 0.2
    cold_mask = pressure < cold_threshold
    if cold_mask.sum() < 5:
        cold_mask = pressure < np.percentile(pressure, 30)

    cold_weights = np.zeros(n)
    cold_weights[cold_mask] = 1.0
    total_cold_w = cold_weights.sum()
    if total_cold_w > 0:
        a_reduced += mass_to_redistribute * cold_weights / total_cold_w

    a_reduced = np.maximum(a_reduced, 0)
    s = a_reduced.sum()
    if s < 1e-15:
        return a.copy()
    return a_reduced / s


def anneal_perturb(a, rng, temp):
    """Multiplicative annealing perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * temp
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def asymmetric_init(n, rng, center1=0.25, width1=0.1, center2=0.7, width2=0.2):
    """Create an asymmetric bimodal initialization."""
    t = np.arange(n)
    a = (np.exp(-0.5 * ((t - n * center1) / (n * width1)) ** 2) +
         0.6 * np.exp(-0.5 * ((t - n * center2) / (n * width2)) ** 2) +
         0.1 * np.ones(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
