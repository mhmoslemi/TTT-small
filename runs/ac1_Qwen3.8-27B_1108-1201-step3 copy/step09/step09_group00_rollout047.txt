def fast_best_response(a_fixed, n_inner=200, lr_init=0.1):
    n = len(a_fixed)
    b = np.ones(n) / n
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for i in range(n_inner):
        # Compute convolution
        A = rfft(a_fixed, fft_size)
        B = rfft(b, fft_size)
        conv = irfft(A * B, fft_size)[:size]
        
        # Find max position
        k_star = int(np.argmax(conv))
        
        # Compute gradient
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = a_fixed[k_star - j]
        
        # Learning rate decay
        lr = lr_init / (1 + i * 0.01)
        
        # Projected gradient step
        b_new = b - lr * grad
        b_new = np.maximum(b_new, 0)
        s = b_new.sum()
        if s > 1e-15:
            b_new = b_new / s
        else:
            b_new = b.copy()
        b = b_new
    
    return b

This is much faster than solving an LP, and for the Nesterov outer loop, an approximate best response is sufficient.

But wait, the Nesterov method requires the exact best response to converge. If we use an approximate one, the convergence might be slower or it might not converge at all.

However, in practice, a few iterations of the inner subgradient method should give a good approximate best response, and the outer Nesterov loop can handle some noise.

Let me think about this more carefully. The Nesterov method for the problem:
min_b f(b) = max_k (a*b)_k s.t. b in simplex

The update is:
y = (1-alpha) * y_prev + alpha * b_best_response(y_prev)
b = ((it+1)*b_prev + alpha*(it+2)*b_best_response) / (it+1 + alpha*(it+2))

If b_best_response is approximate, the method still makes progress but might oscillate more.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast Nesterov with approximate best-response (subgradient inner loop)
2. Exact LP refinement
3. LogSumExp L-BFGS polishing
4. Peak-directed coordinate descent and perturbation escape
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlate(a, p):
    """Compute cross-correlation: result[j] = sum_m a[m] * p[m+j] for valid indices.
    Returns an array of length n = len(a)."""
    n = len(a)
    len_p = len(p)
    # We need result[j] = sum_{m=0}^{n-1} a[m] * p[m+j] for j=0,...,n-1
    # where p[m+j] is 0 if m+j >= len_p
    # This is equivalent to the cross-correlation of a and p
    # Using FFT: reverse a, then convolve with p
    N = n + len_p - 1
    fft_size = 1 << (N - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, fft_size)
    P = rfft(p, fft_size)
    conv = irfft(A * P, fft_size)
    # conv[k] = sum_m a_rev[m] * p[k-m] = sum_m a[n-1-m] * p[k-m]
    # Let m' = n-1-m, then m = n-1-m', and:
    # conv[k] = sum_{m'} a[m'] * p[k-n+1+m']
    # So conv[n-1+j] = sum_{m'} a[m'] * p[j+m'] = result[j]
    return conv[n-1:n-1+n]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


# LP solver cache
_lp_solver_cached = None

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fast_best_response(a_fixed, n_inner=100, lr_init=0.5):
    """Fast approximate best response using subgradient method."""
    n = len(a_fixed)
    b = np.ones(n) / n
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A_fft = rfft(a_fixed, fft_size)
    
    for i in range(n_inner):
        # Compute convolution
        B_fft = rfft(b, fft_size)
        conv = irfft(A_fft * B_fft, fft_size)[:size]
        
        # Find max position
        k_star = int(np.argmax(conv))
        
        # Compute gradient: grad[j] = a_fixed[k_star - j] for valid j
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad[idx] = a_fixed[k_star - idx]
        
        # Also consider near-max positions for stability
        threshold = conv.max() * 0.9
        near_max = np.where(conv >= threshold)[0]
        if len(near_max) > 1:
            grad_avg = np.zeros(n)
            for k in near_max:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_avg[idx] += a_fixed[k - idx]
            grad_avg /= len(near_max)
            # Blend
            grad = 0.5 * grad + 0.5 * grad_avg
        
        # Learning rate with decay
        lr = lr_init / (1 + 0.1 * i)
        
        # Projected gradient step
        b_new = b - lr * grad
        b_new = np.maximum(b_new, 0)
        s = b_new.sum()
        if s > 1e-15:
            b_new = b_new / s
        else:
            b_new = b.copy()
        b = b_new
    
    return b


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_lse_obj(z, n, T_val):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_val)
    lse = max_b + T_val * np.log(exp_b.sum())
    return (2.0 * n) * lse


def compute_lse_grad(z, n, T_val):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_val)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    # Gradient in z-space: a * (grad_a - S)
    return a * (grad_a - S) * 2.0 * n


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return compute_lse_obj(z, n, stage_T)
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return compute_lse_grad(z, n, stage_T)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def nesterov_fast(a0, time_limit, n_inner_bp=100, n_max_iters=500000):
    """Nesterov method with fast (subgradient) best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Precompute FFT of y for fast convolution
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Fast best response for y
        b_new = fast_best_response(y, n_inner=n_inner_bp, lr_init=0.5)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        if (it + 1) % 10 == 0:
            score_a = 2.0 * n * fast_convolve(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            score_y = 2.0 * n * fast_convolve(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovFast iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    # Final evaluation
    score_a = 2.0 * n * fast_convolve(a).max()
    if score_a < best_score:
        best_score = score_a
        best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov method with exact LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(42)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        if (it + 1) % 50 == 0:

            score_y = 2.0 * n * fast_convolve(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            
            if stall_rounds >= 10:
                noise_scale = 0.02 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_rounds = 0
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            sc = 2.0 * n * fast_convolve(a).max()
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} (current={sc:.8f})")

def peak_descent(a0, time_limit, n_steps=30000):
    """Coordinate descent targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 300:
                break
        else:
            stall = 0
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method targeting active peak positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-6, lr)
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def structured_init(n, rng, variant=0):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 1:
        a = np.sin(np.pi * x) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif variant == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif variant == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif variant == 6:
        # Hamming-like window
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    elif variant == 7:
        # Uniform with slight edge boost
        a = np.ones(n)
        a[0] *= 1.5
        a[-1] *= 1.5
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
