import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def l2_smoother_grad(a):
    """Compute gradient of sum_k (a*a)_k^2 w.r.t. a using FFT.
    Returns (gradient, c) where c = a*a."""
    n = len(a)
    c = fast_convolve(a)
    h = a[::-1]
    size = n + len(c) - 1
    fft_size = 1 << (size - 1).bit_length()
    H = rfft(h, fft_size)
    C = rfft(c, fft_size)
    conv = irfft(H * C, fft_size)
    g = conv[n - 1:n - 1 + n]
    return g, c


def l2_objective_and_grad(a):
    """Objective: sum_k c_k^2, gradient computed via FFT."""
    n = len(a)
    grad, c = l2_smoother_grad(a)
    obj = np.dot(c, c)
    return obj, grad


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve: min M s.t. T@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.CLARABEL, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        s = 1.0
    a = a / s
    y = a.copy()

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0

    for it in range(n_max_iters):

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:

        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))

        if sa < 1e-15:

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

def l2_lbgfs_refine(a0, time_limit, maxiter=10000):
    """Minimize L2 norm of autocorrelation using L-BFGS with FFT gradient."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def func(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        obj, _ = l2_objective_and_grad(a)
        return obj

    def jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        grad_a, _ = l2_objective_and_grad(a)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=func, x0=z, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 200, 'disp': False}
        )
        z = res.x
    except Exception:
        pass
    return softmax_from_z(z)


def lsg_lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.1, maxiter=10000):
    """LogSumExp smoothed max via L-BFGS with geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(z_in and a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=make_func(stage_T), x0=z, jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        if stage_idx % 3 == 2:
            elapsed = time.monotonic() - t0
            a_cur = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_cur).max()
            print(f"    LSG stage {stage_idx+1}/{n_stages} T={stage_T:.3f} ({elapsed:.0f}s): score={sc:.8f}")

    return softmax_from_z(z)


def mirror_descent_adaptive(a0, time_limit, n_active_frac=0.05):
    """Mirror descent with adaptive step and varying active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-16:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.7
                no_improve = 0
            lr = max(lr, 0.02)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.4f}")

    return best_a, best_score


def local_swap_search(a0, time_limit, n_steps=30000):
    """Targeted local search using peak contribution analysis."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        for si in top_src:
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-16:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * 0.02
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-16:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and step > 8000:
            break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(2 * x) + 0.15 * np.cos(4 * x) + 0.1 * np.cos(7 * x)
    else:
        a = np.ones(n)
        a = a * (1 + 0.1 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s_val = a.sum()
    if s_val < 1e-15:
        return np.ones(n) / n
    return a / s_val


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s_val = f.sum()
    if s_val < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s_val
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s_val = a.sum()
        if s_val < 0.01:
            return
        a = a / s_val
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP Nesterov (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.42, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: L2-norm variance polishing (NEW) ===
    print(f"\n=== Phase 2: L2 Variance Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_l2 = l2_lbgfs_refine(best_seq, t_alloc, maxiter=20000)
        try_update(a_l2, "l2_polish_1")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp LBFSGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 50->0.05")
        a_lsg = lsg_lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=0.05, maxiter=12000)
        try_update(a_lsg, "lsg_1")

    # === Phase 4: Second LP round (from improved point) ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Mirror descent with tight active set ===
    print(f"\n=== Phase 5: Mirror Desc Tight (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 20:
        a_md, _ = mirror_descent_adaptive(best_seq, t_alloc, n_active_frac=0.02)
        try_update(a_md, "md_tight")

    # === Phase 6: L2 polish again ===
    print(f"\n=== Phase 6: L2 Polish r2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 70.0)
    if t_alloc > 15:
        a_l2r = l2_lbgfs_refine(best_seq, t_alloc, maxiter=10000)
        try_update(a_l2r, "l2_polish_2")

    # === Phase 7: LogSumExp with very low T ===
    print(f"\n=== Phase 7: LSG low-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 70.0)
    if t_alloc > 15:
        a_lsg2 = lsg_lbgfs_refine(best_seq, t_alloc, T_start=5.0, T_end=0.02, maxiter=8000)
        try_update(a_lsg2, "lsg_lowT")

    # === Phase 8: Local swap search ===
    print(f"\n=== Phase 8: Local Swap (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_ls, _ = local_swap_search(best_seq, t_alloc, n_steps=40000)
        try_update(a_ls, "local_swap")

    # === Phase 9: Alternative lengths ===
    print(f"\n=== Phase 9: Alt Lengths (best={best_score:.8f}) ===")
    alt_lengths = []
    for mult in [1.1, 1.25, 0.9, 1.5]:
        alt_n = int(n_best * mult)
        if 400 <= alt_n <= 2500 and abs(alt_n - n_best) > 50:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.4, 120.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  *** Alt init improved: {sc_alt:.8f}")

        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_refine(cur_best, t_lp, solver=solver, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_l2 = min(rem * 0.3, 40.0)
            if t_l2 > 10:
                a_alt2 = l2_lbgfs_refine(best_seq, t_l2, maxiter=5000)
                try_update(a_alt2, f"l2_n{alt_n}")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.3, 50.0)
        a_fin = lsg_lbgfs_refine(best_seq, t_fin, T_start=3.0, T_end=0.02, maxiter=5000)
        try_update(a_fin, "final_lsg")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.3, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=10000)
        try_update(a_lp_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ls = min(rem * 0.3, 20.0)
        a_ls2, _ = local_swap_search(best_seq, t_ls, n_steps=10000)
        try_update(a_ls2, "final_swap")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
