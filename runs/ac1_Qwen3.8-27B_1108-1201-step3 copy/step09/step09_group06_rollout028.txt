def fast_convolve(a):

def softmax_from_z(z):

    s = e.sum()

    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    exp_Tb = np.exp(Tb - m)
    w = exp_Tb / exp_Tb.sum()
    
    if K < len(w):
        top_idx = np.argsort(w)[::-1][:K]
        top_w = w[top_idx].copy()
        top_w /= top_w.sum()
    else:
        top_idx = np.arange(len(w))
        top_w = w
    
    s_vec = np.zeros(n)
    wm = 0.0
    for ki in range(len(top_idx)):
        k = int(top_idx[ki])
        wk = top_w[ki]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        s_vec[idx] += wk * a[k - idx]
        wm += wk * b[k]
    
    return (4.0 * n / T) * a * (s_vec - wm)
