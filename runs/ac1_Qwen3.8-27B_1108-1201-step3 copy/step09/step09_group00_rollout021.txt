"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast peak shaving,
subgradient descent, spectral reshaping, and LP-based refinement.
Key improvements: O(n log n) peak shaving via FFT, multi-length search,

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Compute objective value locally (sum=1 normalization)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_max_autocorr(a):
    """
    Compute subgradient of max_k conv(a,a)[k] with respect to a.
    Uses the average gradient over near-maximal positions.
    """
    n = len(a)

    max_c = c.max()

    threshold = max_c * 0.995

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Fast swap-based peak shaving
- Simulated annealing for global exploration
- LP-based Nesterov refinement for precision
- Multi-length search
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, np.float64), fs)
    return irfft(A * A, fs)[:size]

def eval_obj(a):
    """Compute 2*n*max(conv)/sum² locally."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    n = len(a)
    return 2.0 * n * fast_conv(a).max() / (s * s)

def norm1(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T s.t. T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_br(a_fixed, solver):
    """LP best response: minimize M s.t. conv(a,b)<=M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_conv_matrix(a_fixed)
    cons = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            bv = np.asarray(b.value, np.float64)
            bv = np.maximum(bv, 0)
            s = bv.sum()
            if s > 1e-15:
                return bv / s, float(M.value)
    except:
        pass
    return None, None

def nesterov_lp(a0, time_limit, solver, n_max=200000):
    """Nesterov dual averaging with LP best response."""
    n = len(a0)
    a = norm1(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_sc = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    prev_best = best_sc
    stall = 0
    rng = np.random.default_rng(99)
    
    for it in range(n_max):
        if time.monotonic() - t0 > time_limit:
            break
        br, _ = solve_lp_br(y, solver)
        if br is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * br
        y = norm1(y)
        a = ((it+1)*a + alpha*(it+2)*br) / (it+1 + alpha*(it+2))
        a = norm1(a)
        
        sc_a = 2.0 * n * fast_conv(a).max()
        sc_y = 2.0 * n * fast_conv(y).max()
        if sc_a < best_sc:
            best_sc = sc_a; best_a = a.copy()
        if sc_y < best_sc:
            best_sc = sc_y; best_a = y.copy()
        
        if (it+1) % 1000 == 0:
            if best_sc < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_sc
            if stall >= 10:
                y = y + rng.standard_normal(n) * 0.02
                y = norm1(y)
                stall = 0
            el = time.monotonic() - t0
            print(f"    NL it={it+1} ({el:.0f}s): F={best_sc:.10f}")
    
    return best_a, best_sc

def swap_shave(a0, time_limit, n_max=2000000, verbose=True):

    n = len(a)
    c = fast_conv(a)
    cur = c.max()
    best_a = a.copy()
    best = cur
    t0 = time.monotonic()

        i_s = sr[np.argmax(al)]

            if nm < cur - 1e-13:

def sa_search(a0, time_limit, T0=0.1, T1=0.001, n_max=500000, verbose=True):
    """Simulated annealing on the sequence."""
    a = norm1(a0.copy())
    n = len(a)
    cur_sc = 2.0 * n * fast_conv(a).max()
    best_a = a.copy()
    best_sc = cur_sc

    for it in range(n_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule
        frac = min(1.0, it / n_max)
        T = T0 * (T1 / T0) ** frac

        i = rng.integers(n)
        factor = np.exp(rng.standard_normal() * 0.3)
        an = a.copy()
        an[i] *= factor
        # Redistribute
        diff = a[i] - an[i]
        if abs(diff) < 1e-15:
            continue
        mask = np.ones(n, bool)
        mask[i
