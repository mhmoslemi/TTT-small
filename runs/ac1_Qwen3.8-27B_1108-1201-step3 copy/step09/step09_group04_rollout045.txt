        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if this is an improvement
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < score:
            a = a_new
            # Decrease learning rate slightly
            lr = lr * 0.999
        else:
            # Reject and reduce learning rate
            lr = lr * 0.95
            v = v * momentum  # damp momentum
        
        if it % 1000 == 0:
            print(f"  ElasticNet iter={it}: score={best_score:.6f}, lr={lr:.6f}")
    
    return best_a, best_score
```

This is a stochastic sub-gradient-like method. The key difference from the p-norm approach is:
1. It focuses on the top peaks (not all peaks)
2. It uses momentum for faster convergence
3. It has a simpler gradient computation

However, I'm not sure this will be better than the p-norm L-BFGS approach, which uses second-order information.

Let me think of another angle: **What if we use the LP approach but with a better outer-loop update?**

The previous Nesterov approach uses the optimal step size alpha = 2/(t+2). We could try:
- Polyak step size (using the known gap)
- Backtracking line search
- Adaptive step size based on observed progress

Or, we could use a **bisection on M** approach:
1. For a given M, check if there exists a >= 0, sum(a) = 1, with convolve(a,a)[k] <= M for all k
2. This is a feasibility problem (QP with quadratic constraints)
3. Use bisection to find the minimum feasible M

The feasibility problem is:
- min 0 (or sum of violations)
- s.t. a >= 0, sum(a) = 1, convolve(a,a)[k] <= M for all k

This is a QP (quadratic objective is 0, but constraints are quadratic). It's non-convex in general.

However, we can use the SDP relaxation:
- Let X = a * a^T (rank-1 constraint)
- The constraint sum_j X[j, k-j] <= M is linear in X
- a >= 0 is implied by X (diagonal of X is a^2)
- sum(a) = 1 is harder to express in X

This gets complicated. Let me stick with the Nesterov approach but improve it.

Let me also think about whether there's a closed-form or semi-analytical solution.

The problem is: min_a 2n * max_k sum_j a[j]a[k-j] s.t. sum(a) = 1, a >= 0.

By homogeneity, we can fix sum(a) = 1. The problem is: min_
 max_k (a^T C_k a) where C_k is the matrix with (C_k)_{j,l} = 1 if j+l = k (for valid j, l).

This is a min-max quadratic programming problem. In general, it's NP-hard, but for our structured C_k matrices, there might be efficient algorithms.

For the specific case where the maximum is at a single position k*, the problem becomes:
min_a a^T C_{k*} a s.t. sum(a) = 1, a >= 0.

This is a quadratic program with a single quadratic objective. The solution can be found by KKT conditions. The Hessian is 2*C_{k*}, which is symmetric. The matrix C_{k*} has a specific structure: it's the "anti-diagonal" matrix for the k-th row of the convolution.

Actually, C_k has (C_k)_{j,l} = 1 if j + l = k and 0 <= j, l < n. So it's a matrix with 1s along the anti-diagonal j + l = k. This matrix has rank min(k+1, n, 2n-1-k).

The QP min a^T C_k a s.t. 1^T a = 1, a >= 0 can be solved efficiently.

But the actual problem has the max over ALL k, not just one. So we need to solve the full min-max problem.

```python
"""
Multi-strategy optimization: elastic-net gradient with momentum, 
improved Nesterov LP with active-set tracking, block coordinate descent,
and aggressive multi-length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fn(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def elastic_net_optimize(a_init, time_limit, threshold_frac=0.90, initial_lr=0.005, momentum=0.95):
    """Optimize using elastic-net penalty on near-peak values with momentum."""
    n = len(a_init)
    a = normalize(a_init)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    lr = initial_lr
    stall = 0
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        threshold = max_b * threshold_frac
        excess = np.maximum(b - threshold, 0)
        if excess.max() < 1e-15:
            break
        
        # Gradient: 2 * correlate(excess, a, 'valid')
        g = 2.0 * np.correlate(excess, a, mode='valid')
        
        # Add small regularization for smoothness
        if n > 2:
            g[1:-1] += 0.001 * (2*a[1:-1] - a[:-2] - a[2:])
        
        # Momentum update with adaptive lr
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < cur_score:
            a = a_new
            lr = min(lr * 1.01, 0.05)  # slowly increase lr
        else:
            lr = lr * 0.9
            v = v * 0.5
        
        if stall > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = a + 0.02 * rng.standard_normal(n)
            a = normalize(a)
            v = np.zeros(n)
            lr = initial_lr
            stall = 0
        
        if it % 2000 == 0:
            print(f"    EN iter={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = name
                return _solver
        except:
            continue
    _solver = cp.ECOS
    return _solver


def lp_best_response(a_fixed):
    """Find b >= 0, sum(b)=1 minimizing max convolve(b, a_fixed)."""
    n = len(a_fixed)
    solver = get_solver()
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a_init, time_limit, n_iters=50000):
    """Nesterov dual averaging for the min-max QP."""
    n = len(a_init)
    a = normalize(a_init)
    y = a.copy()
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star = lp_best_response(y)
        if b_star is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_star) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        
        sc_a = score_fn(a)
        sc_y = score_fn(y)
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            
            if stall >= 3:
                rng = np.random.default_rng(it * 7 + 3)
                y = y + 0.03 * rng.standard_normal(n)
                y = normalize(y)
                stall = 0
                prev_best = best_score
            
            print(f"    NLP iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def block_coordinate_descent(a_init, time_limit, block_size=16):
    """Optimize blocks of consecutive elements using scipy minimize."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    for round_idx in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            idx = slice(start, end)
            
            def block_obj(x, idx=idx):
                a_trial = a.copy()
                a_trial[idx] = np.maximum(x, 0)
                a_trial = normalize(a_trial)
                return score_fn(a_trial)
            
            x0 = a[idx].copy()
            try:
                res = minimize(block_obj, x0, method='Nelder-Mead',
                              options={'maxiter': 50, 'xatol': 1e-10, 'fatol': 1e-12, 'disp': False})
                a_trial = a.copy()
                a_trial[idx] = np.maximum(res.x, 0)
                a_trial = normalize(a_trial)
                sc = score_fn(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_a = a.copy()
                    best_score = sc
            except:
                pass
    
    return best_a, best_score


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = (np.cos(np.pi * x) ** 2 + 0.1)
    elif variant == 4:
        xc = (x - 0.5) * 2
        a = np.exp(-xc ** 2 * 3)
    elif variant == 5:
        # Tukey window with r=0.3
        r = 0.3
        a = np.ones(n)
        m = int(r * n / 2)
        if m > 0:
            for i in range(m):
                t = 0.5 + 0.5 * np.cos(np.pi * i / m)
                a[i] = t
                a[n-1-i] = t
    elif variant == 6:
        # Flat-top
        a = (1 - 0.2125440 * np.cos(2*np.pi*x) + 0.2300088 * np.cos(4*np.pi*x)
             - 0.0820117 * np.cos(6*np.pi*x) + 0.0058657 * np.cos(8*np.pi*x)
             - 0.0003512 * np.cos(10*np.pi*x))
        a = np.abs(a)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def p_norm_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=7):
    """L-BFGS with p-norm smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            val = (np.sum(c_pos ** stage_p)) ** (1.0 / stage_p)
            return 2.0 * n * val
        
        def jac(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            p_val = c_pos ** stage_p
            S = np.sum(p_val)
            if S < 1e-300:
                return np.zeros_like(z_val)
            factor = S ** (1.0 / stage_p - 1.0)
            w = c_pos ** (stage_p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            g_a = 4.0 * n * factor * corr
            return a * (g_a - np.dot(g_a, a))
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': 40000, 'disp': False})
            z = res.x
        except:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        print(f"      p-norm p={stage_p:.1f} ({time.monotonic()-t0:.0f}s): score={sc:.6f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()
