import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)

def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b * (1 - n_active_frac)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_refine(a0, time_limit, T_start=150.0, T_end=0.3, maxiter=10000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=make_func(stage_T), x0=z, jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def swap_eval(b, a, i, j, delta, n):
    """Compute max of new convolution after transferring delta from i to j. O(n)."""
    # new_b_k = b_k + 2*delta*(a_{k-j} - a_{k-i}) + delta^2*D_k
    # D_k = 1 if k==2j, -2 if k==j+i, 1 if k==2i, 0 otherwise
    size = 2 * n - 1
    new_b = b.copy()
    # Linear terms
    # a_{k-j} for k=0..size-1: a_m where m=k-j, valid if 0<=m<n
    if j < n:
        shift_j = np.zeros(size)
        shift_j[j:j+n] = a
        new_b += 2.0 * delta * shift_j
    if i < n:
        shift_i = np.zeros(size)
        shift_i[i:i+n] = a
        new_b -= 2.0 * delta * shift_i
    # Quadratic terms
    if 2 * j < size:
        new_b[2 * j] += delta * delta
    if j + i < size:
        new_b[j + i] -= 2.0 * delta * delta
    if 2 * i < size:
        new_b[2 * i] += delta * delta
    return new_b.max()


def swap_search_intensive(a0, time_limit, n_premium_pairs=20, n_random_pairs=10):
    """Intensive swap-based local search with O(n) evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5000
    anneal_temp = 0.001

    for step in range(20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        # Recompute b periodically
        if step % 100 == 0:
            b = fast_convolve(a)

        max_b = b.max()
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute blame for each coordinate
        blame = np.zeros(n)
        for k in active[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            blame[idx] += a[idx] * a[k - idx]

        # Find donors (high blame) and receivers (low blame)
        sorted_idx = np.argsort(blame)[::-1]
        donors = sorted_idx[:n_premium_pairs]
        receivers = sorted_idx[-n_premium_pairs:]

        # Also add random pairs
        rand_d = rng.integers(n, size=n_random_pairs)
        rand_r = rng.integers(n, size=n_random_pairs)

        improved = False
        tested = 0

        for di in list(donors) + list(rand_d):
            if a[di] < 1e-12:
                continue
            for rj in list(receivers) + list(rand_r):
                if di == rj:
                    continue
                if tested >= 60:
                    break
                tested += 1

                # Try several delta values
                base_delta = a[di]
                deltas = [base_delta * 0.1, base_delta * 0.25, base_delta * 0.5]
                best_d = 0.0
                best_v = 2.0 * n * max_b  # current score

                for d in deltas:

                        best_v = v
                        best_d = d

                # Refine with golden section
                if best_d > 0:
                    lo = 0.0
                    hi = a[di]
                    gr = (np.sqrt(5) - 1) / 2
                    for _ in range(15):
                        m1 = hi - gr * (hi - lo)
                        m2 = lo + gr * (hi - lo)
                        v1 = 2.0 * n * swap_eval(b, a, di, rj, m1, n)
                        v2 = 2.0 * n * swap_eval(b, a, di, rj, m2, n)
                        if v1 < v2:
                            hi = m2
                        else:
                            lo = m1
                        if v1 < best_v:
                            best_v = v1
                            best_d = m1
                        if v2 < best_v:
                            best_v = v2
                            best_d = m2

                if best_v < best_score * (1 - 1e-10) or (best_v < best_score + anneal_temp and rng.random() < 0.01):
                    a[di] -= best_d
                    a[rj] += best_d
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    if best_v < best_score:
                        best_score = best_v
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    improved = True
                    break
            if improved:
                break

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * np.linspace(0, 1, n))
        a = np.maximum(a, 0) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(5 * x)
        a = np.maximum(a, 0)
    elif choice == 5:
        a = rng.random(n)
        a = a + 0.3 * np.exp(-np.linspace(0, 1, n) ** 2 * 5)
    elif choice == 6:
        # Two-hump
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    else:
        # Asymmetric ramp
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.3
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: LP alternating (warm start)
    print(f"\n=== Phase 1: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 220.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp, "lp_p1")

    # Phase 2: Intensive swap search (main workhorse)
    print(f"\n=== Phase 2: Swap Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_swap, _ = swap_search_intensive(best_seq, t_alloc, n_premium_pairs=15, n_random_pairs=8)
        try_update(a_swap, "swap_p2")

    # Phase 3: L-BFGS polish
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine_v2(best_seq, t_alloc, T_start=80.0, T_end=0.1, maxiter=15000)
        try_update(a_lb, "lbgfs_p3")

    # Phase 4: Another swap search round
    print(f"\n=== Phase 4: Swap Search r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_swap2, _ = swap_search_intensive(best_seq, t_alloc, n_premium_pairs=20, n_random_pairs=5)
        try_update(a_swap2, "swap_p4")

    # Phase 5: Alternative lengths
    print(f"\n=== Phase 5: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.1, 1.25, 0.85, 1.5, 0.75]:
        alt_n = int(n_best * mult)
        if 400 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        print(f"  n={alt_n}, remaining={rem:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolated n={alt_n}: {sc_alt:.6f}")

        # Quick LP + swap on this length
        t_sub = min(rem * 0.4, 200.0)
        if t_sub > 20:
            t_lp = t_sub * 0.4
            t_sw = t_sub * 0.6
            a_alt_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=10000)
            try_update(a_alt_lp, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 10:
                a_alt_sw, _ = swap_search_intensive(best_seq, min(rem * 0.5, t_sw),
                                                    n_premium_pairs=12, n_random_pairs=5)
                try_update(a_alt_sw, f"swap_n{alt_n}")

    # Phase 6: Final polish
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 100.0)
        a_fin = lbgfs_refine_v2(best_seq, t_fin, T_start=20.0, T_end=0.05, maxiter=10000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ls = min(rem * 0.4, 80.0)
        a_ls2, _ = swap_search_intensive(best_seq, t_ls, n_premium_pairs=25, n_random_pairs=5)
        try_update(a_ls2, "final_swap")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_md = min(rem * 0.4, 40.0)
        a_md_fin, _ = mirror_descent_adaptive(best_seq, t_md, n_active_frac=0.04)
        try_update(a_md_fin, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.asarray(best_seq, dtype=np.float64)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
