   **Now let me also think about spectral methods:**

   The peak of the autocorrelation is related to the flatness of the power spectrum. Specifically:
   `conv(a,a)[k] = sum_j a[j]*a[k-j]`
   
   The DFT of the autocorrelation is `|A(f)|^2` where `A(f)` is the DFT of `a`.
   
   The peak of the autocorrelation is at `k=0` (or wherever it is). If the power spectrum `|A(f)|^2` were perfectly flat, the autocorrelation would be a delta function (peak only at k=0, zero elsewhere). In that case, `max conv` would be `conv[0] = sum a[j]^2`.
   
   So the ideal is to make `|A(f)|^2` as flat as possible. This is the "flat top" problem.
   
   **Spectral shaping algorithm:**
   1. Compute `A = FFT(a)`
   2. Compute power spectrum `P = |A|^2`
   3. Shape `P`: e.g., `P_new = P^alpha` for `alpha < 1` (compresses dynamic range)
   4. Set `|A_new| = sqrt(P_new)`, keeping the phase of `A`
   5. `a_new = IFFT(A_new)`
   6. Project to simplex: `a_new = max(a_new, 0) / sum(max(a_new, 0))`
   7. Repeat
   
   The parameter `alpha` controls the aggressiveness. `alpha = 1` is no change, `alpha -> 0` makes the spectrum perfectly flat (but the time-domain signal becomes more "random").
   
   This is a well-known technique in signal processing (e.g., for designing flat-top sampling windows).

"""

1. Exact coordinate descent with O(n) incremental convolution updates
2. Nesterov LP best-response with dual tracking and adaptive steps
3. Spectral shaping (iterative frequency-domain flat-top design)
4. L-BFGS with LogSumExp smoothing for final precision polish
Key innovation: coordinate descent exploits the quadratic structure of
conv(a,a) in each coordinate to solve exact 1D maximin subproblems,
with O(n) convolution updates via rank-2 modifications.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def incremental_conv_update(conv, a, i, old_x, new_x):
    """Update conv in-place after changing a[i] from old_x to new_x. O(n)."""
    n = len(a)
    delta = new_x - old_x
    if abs(delta) < 1e-16:
        return
    # conv(a+delta*e_i, a+delta*e_i) = conv(a,a) + 2*delta*conv(e_i,a) + delta^2*delta_{2i}
    # conv(e_i, a)[k] = a[k-i] for valid k-i
    js = max(0, 0 - i)  # k-i >= 0 => k >= i
    # k ranges from i to min(i+n-1, 2n-2)
    k_start = i
    k_end = min(i + n, 2 * n - 1)
    # Get a[0..n-1] shifted: a[k-i] for k in [k_start, k_end)
    idx = np.arange(0, k_end - k_start)
    conv[k_start:k_end] += 2.0 * delta * a[idx]
    # delta^2 term at position 2*i
    if 0 <= 2 * i < len(conv):
        conv[2 * i] += delta * delta


def solve_1d_maximin(base, d, e, x_min=0.0, x_max=None):
    """Find x >= x_min minimizing max_k (base[k] + d[k]*x + e[k]*x^2).
    Uses bisection on the subgradient. O(60 * len(base))."""
    if x_max is None:
        # Estimate upper bound
        x_max = x_min + 1.0
        # Expand if needed
        for _ in range(20):
            vals = base + d * x_max + e * x_max * x_max
            subgrads = d + 2.0 * e * x_max
            max_idx = np.argmax(vals)
            if subgrads[max_idx] < 0:
                x_max *= 2.0
            else:
                break
    lo = x_min
    hi = x_max
    if hi < x_min:
        return x_min
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = base + d * mid + e * mid * mid
        max_val = vals.max()
        active = vals >= max_val - 1e-14 * max(1.0, abs(max_val))
        if not active.any():
            active = vals >= max_val - 1e-10 * max(1.0, abs(max_val))
        subgrads = d[active] + 2.0 * e[active] * mid
        if subgrads.min() > 1e-12:
            hi = mid
        elif subgrads.max() < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coordinate_descent(a0, time_budget, verbose=True):

            peak_k = int(np.argmax(conv))
            # Coordinates that contribute most to the peak
            js = max(0, peak_k - n + 1)

            contribs = np.zeros(n)
            if je >= js:
                idx = np.arange(js, je + 1)
                contribs[idx] = a[idx] * a[peak_k - idx]
            order = np.argsort(-contribs)
        else:
            order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            x_old = a[i]
            # Compute base, d, e

            # e[2i] = 1 if 0 <= 2i < 2n-1

            k_end = min(i + n, 2 * n - 1)
            if k_end > k_start:

            # Zero out the position where e=1 (k=2i) since there the term is x^2 not 2*a[i]*x
            if 0 <= 2 * i < 2 * n - 1:
                d[2 * i] = 0.0

            e = np.zeros(2 * n - 1)
            if 0 <= 2 * i < 2 * n - 1:
                e[2 * i] = 1.0

            base = conv - d * x_old - e * x_old * x_old
            # Ensure base is consistent (numerical)
            # Solve 1D problem
            x_new = solve_1d_maximin(base, d, e, x_min=0.0)

                incremental_conv_update(conv, a, i, x_old, x_new)
                a[i] = x_new

        # Renormalize periodically to maintain scale
        sa = a.sum()
        if sa < 1e-10:
            a = np.ones(n) / n
            conv = fast_convolve(a)
            sa = 1.0
        elif sa > 10.0 or sa < 0.1:
            a = a / sa
            conv = fast_convolve(a)
            sa = 1.0

        cur_max = conv.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            no_improve_sweeps = 0
        else:

                    a = a_pert
                    conv = conv_pert
                else:

            no_improve_sweeps = 0

def spectral_shaping(a0, time_budget, alpha_start=2.0, alpha_end=0.3, n_iters=None):
    """Iterative spectral shaping: flatten |FFT(a)|^2 to reduce autocorrelation peak."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if n_iters is None:
        n_iters = 50000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        # Adaptive alpha: start large (aggressive), decrease over time
        frac = min(1.0, it / (n_iters * 0.8))
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Compute FFT
        A = rfft(a)
        P = np.abs(A) ** 2
        # Shape: P_new = P^alpha (compresses if alpha < 1, expands if alpha > 1)
        # We want to flatten, so use alpha < 1 for the power
        P_shaped = np.power(P, alpha)
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_shaped) * np.exp(1j * phase)
        # IFFT
        a_new = irfft(A_new, n)
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): score={best_score:.8f} alpha={alpha:.3f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def compute_logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp smoothed objective and gradient in log-domain.
    Objective: (2n/T) * log(sum_k exp(T * conv_k / (2n)))  ... simplified.
    Actually: f(z) = (2n/T) * LSE(T * conv(softmax(z), softmax(z)) / (2n))
    which approximates 2n * max(conv) as T -> inf."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    scaled = T_temp * conv / (2.0 * n)
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    logsumexp = max_s + np.log(exp_s.sum())
    obj = logsumexp  # This approximates max(conv) up to factor

    # Gradient
    w = exp_s / exp_s.sum()
    grad_conv = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_conv[idx] += w[k] * a[k - idx]
    # d(conv_k)/d(a_i) = 2 * a[k-i] (for valid k-i), so:
    # grad_a = 2 * sum_k w[k] * a shifted
    # Actually grad_conv above is sum_k w[k] * a[k - idx], which is half the gradient
    # Full gradient of conv_k w.r.t. a is: 2 * flip_conv(a, k)
    # Let me redo: conv_k = sum_j a[j]*a[k-j]
    # d(conv_k)/d(a[i]) = a[k-i] + a[i-k] (if valid) = 2*a[k-i] (if k!=2i and valid)
    # So grad_a[i] = sum_k w[k] * 2*a[k-i] (for valid k-i)
    grad_a = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] = 2.0 * w[k] * a[k - idx] + grad_a[idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_budget, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS on LogSumExp-smoothed objective in log-space."""
    n = len(a0)
    z = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300))

        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break

            if time.monotonic() - t0 > time_budget:
                return 1e15
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = fast_convolve(a)
            scaled = stage_T * conv
            max_s = scaled.max()
            return max_s + np.log(np.exp(scaled - max_s).sum())

        def jac(z):
            if time.monotonic() - t0 > time_budget:
                return np.zeros_like(z)
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = fast_convolve(a)
            scaled = stage_T * conv
            max_s = scaled.max()
            exp_s = np.exp(scaled - max_s)

            grad_a = np.zeros(n)
            for k in range(len(conv)):

                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': maxiter_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Try a few structured inits
        best_init = None
        best_init_score = float('inf')
        for n_try in [960, 1100, 1200, 1400]:
            x = np.linspace(0, 1, n_try)
            for func in [
                lambda x: np.cos(np.pi * x / 2) ** 2,
                lambda x: np.sin(np.pi * x) ** 2,
                lambda x: np.exp(-3 * x * (1 - x)),
                lambda x: 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x),
            ]:
                a_try = np.maximum(func(x), 0)
                s = a_try.sum()
                if s < 0.01:
                    continue
                a_try = a_try / s
                sc = evaluate_local(a_try)
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = a_try.copy()
        if best_init is None:
            best_init = np.ones(1000) / 1000
            best_init_score = evaluate_local(best_init)
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP (45% of budget) ===
    print(f"\n=== Phase 1: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=400000)
        try_update(a1, "nlp")

    # === Phase 2: Exact Coordinate Descent (30% of budget) ===
    print(f"\n=== Phase 2: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a2, _ = coordinate_descent(best_seq, t_alloc)
        try_update(a2, "coord_desc")

    # === Phase 3: Spectral Shaping (15% of budget) ===
    print(f"\n=== Phase 3: Spectral Shaping (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        # Try different alpha ranges
        for a_start, a_end in [(3.0, 0.5), (1.5, 0.3), (5.0, 1.0)]:
            rem = deadline - time.monotonic()
            t_sub = min(rem * 0.33, t_alloc / 3.0 + 5)
            if t_sub < 5:
                break
            a3, _ = spectral_shaping(best_seq, t_sub, alpha_start=a_start, alpha_end=a_end)
            try_update(a3, f"spectral_{a_start}_{a_end}")

    # === Phase 4: Second Coordinate Descent pass (10% of budget) ===
    print(f"\n=== Phase 4: CD Pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a4, _ = coordinate_descent(best_seq, t_alloc)
        try_update(a4, "cd2")

    # === Phase 5: L-BFGS LogSumExp final polish (remaining time) ===
    print(f"\n=== Phase 5: L-BFGS Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.5, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a5 = lbgfs_logsumexp(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=4000)
        try_update(a5, "lbgfs")

    # === Phase 6: Final CD + L-BFGS micro-polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.4, 60.0)
        if t_cd > 5:
            a6, _ = coordinate_descent(best_seq, t_cd)
            try_update(a6, "cd_final")
        rem = deadline - time.monotonic()
        if rem > 15:
            T_final = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
            a7 = lbgfs_logsumexp(best_seq, rem * 0.6, T_stages=T_final, maxiter_per_stage=3000)
            try_update(a7, "lbgfs_final")

    # === Phase 7: Try slightly different length ===
    rem = deadline - time.monotonic()
    if rem > 100 and best_seq is not None:
        print(f"\n=== Phase 7: Alt Length (score={best_score:.8f}) ===")
        alt_n = int(n_best * 1.15)
        t_alt = min(rem * 0.4, 150.0)
        if t_alt > 30:
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = coordinate_descent(init_alt, t_alt * 0.5)
            try_update(a_alt, f"cd_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_alt2 = lbgfs_logsumexp(a_alt, rem * 0.5,
                                         T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
                                         maxiter_per_stage=2000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # Final
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
