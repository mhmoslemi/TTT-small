import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def compute_grad_fft(a, w, n):
    """Compute gradient g_i = 2 * sum_{j=0}^{n-1} a[j] * w[j+i] using FFT."""
    fft_len = 1 << (2 * n).bit_length()
    A_ft = rfft(a, fft_len)
    W_ft = rfft(w, fft_len)
    return 2.0 * irfft(np.conj(A_ft) * W_ft)[:n]


def lse_adam_descent(a0, time_limit, T0=0.005, T_min=0.0005,

    Minimizes 2n * max_k (a*a)[k] on the simplex.
    Uses adaptive temperature and restarts.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())

    T = T0
    lr = lr0
    beta1 = 0.85
    beta2 = 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    epsilon = 1e-8

    t0 = time.monotonic()
    no_improve = 0
    restart_count = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit:
        # Compute softmax weights over convolution peaks
        c = conv
        c_max = float(c.max())
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / exp_c.sum()

        # Compute gradient via FFT
        grad = compute_grad_fft(a, w, n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        t += 1

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    factors = [0.995, 0.99, 0.985, 0.97, 0.95, 1.05, 1.03, 1.015, 1.01, 1.005]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            best_delta = 0.0
            cur_local_max = max(unaff_max, float(conv[k_lo:k_hi + 1].max()))

            for factor in factors:
                delta = v * (factor - 1.0)
                x = v + delta
                if x < -1e-15:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = max(unaff_max, float(nv.max()))
                if local_max < cur_local_max - 1e-16:
                    cur_local_max = local_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def basin_hopping(a0, time_limit, local_opt, rng, n_perturbs=10, verbose=False):
    """Basin hopping with the given local optimizer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 40.0)

        # Perturb
        sigma = 0.015 + 0.025 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Peak-breaking perturbation
        conv_p = fast_convolve(a)
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 5)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 + rng.standard_normal() * 0.04)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = local_opt(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    BH p={p} ({elapsed:.0f}s): improved to {best_score:.8f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining log-sum-exp gradient descent,
    exact coordinate descent, and basin hopping.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Quick initialization and evaluation ===
    print("\n=== Phase 1: Init + Quick ECD ===")
    t1_start = time.monotonic()

    n_values = [1024, 768, 512]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values[:4]:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 120:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('cos', init_cosine(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))

        for itype, cand in inits[:4]:
            t_left = 120.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(10.0, t_left / 6.0))
            a_opt, s_opt = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: best={candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:3]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Log-sum-exp Adam gradient descent (main workhorse) ===
    print(f"\n=== Phase 2: LSE-Adam Gradient Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_start = time.monotonic()

        # Run LSE-Adam with different T values
        for T_val in [0.008, 0.005, 0.003]:
            rem = deadline - time.monotonic() - 600
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 180.0)
            print(f"  LSE-Adam T={T_val}, budget={t_alloc:.0f}s")
            a_lse, s_lse = lse_adam_descent(best_seq, t_alloc, T0=T_val,
                                            verbose=True, label=f"T{T_val}")
            best_seq, best_score = check_and_update(a_lse, best_seq, best_score, f"lse_T{T_val}")

            # Follow up with coordinate descent
            rem = deadline - time.monotonic() - 500
            if rem > 20:
                t_cd = min(rem * 0.25, 60.0)
                a_cd, s_cd = exact_coord_descent(a_lse, t_cd, label="post_lse")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_lse")

        # Nesterov variant
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_nc = min(rem * 0.3, 100.0)
            print(f"  Nesterov LSE, budget={t_nc:.0f}s")
            a_nc, s_nc = nesterov_lse_descent(best_seq, t_nc,
                                              verbose=True, label="nesterov")
            best_seq, best_score = check_and_update(a_nc, best_seq, best_score, "nesterov")

            rem = deadline - time.monotonic() - 350
            if rem > 20:
                t_cd2 = min(rem * 0.2, 40.0)
                a_cd2, s_cd2 = exact_coord_descent(a_nc, t_cd2, label="post_nc")
                best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_after_nc")

    # === Phase 3: Intensive coordinate descent ===
    print(f"\n=== Phase 3: Intensive ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t3_start = time.monotonic()

        current = best_seq.copy()
        for rnd in range(5):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 120.0)
            print(f"  ECD round {rnd + 1}, budget={t_alloc:.0f}s")
            a_opt, s_opt, _ = _ecd_wrapper(current, t_alloc, verbose=True, label=f"ph3_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph3_ecd_r{rnd}")

            if best_score > 1.5035:
                perturbed = a_opt + rng.standard_normal(n_best) * 0.02
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 4: Basin hopping ===
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 40:
            t_bh = min(rem * 0.7, 250.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")

            def local_opt_fn(a_init, t_alloc):
                """Combined local optimizer: LSE-Adam then ECD."""
                # Quick LSE
                t_lse = t_alloc * 0.4
                t_ecd = t_alloc * 0.6
                a1, s1 = lse_adam_descent(a_init, t_lse, T0=0.004, label="bh_lse")
                a2, s2 = exact_coord_descent(a1, t_ecd, label="bh_ecd")
                if s2 <= s1:
                    return a2, s2
                else:
                    return a1, s1

            a_bh, s_bh = basin_hopping(best_seq, t_bh, local_opt_fn, rng,
                                       n_perturbs=12, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            n_best = len(best_seq)
            alt_lengths = [l for l in [n_best - 128, n_best + 128, 1024, 1280]
                           if 256 <= l <= 2048 and l != n_best]
            for alt_n in alt_lengths[:3]:
                rem = deadline - time.monotonic() - 120
                if rem < 20:
                    break
                t_alt = min(rem * 0.3, 50.0)
                alt_init = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt = lse_adam_descent(alt_init, t_alt * 0.5, T0=0.005,
                                                label=f"alt_n{alt_n}")
                a_alt2, s_alt2 = exact_coord_descent(a_alt, t_alt * 0.5, label=f"ecd_n{alt_n}")
                best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_fine = min(rem * 0.4, 60.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine,
                                                verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd2 = min(rem * 0.5, 30.0)
            a_cd2, s_cd2 = exact_coord_descent(best_seq, t_cd2, label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # One more LSE pass with very low T
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_final, s_final = lse_adam_descent(best_seq, min(rem * 0.7, 20.0),
                                                T0=0.002, T_min=0.0005, label="final_lse")
            best_seq, best_score = check_and_update(a_final, best_seq, best_score, "last_lse")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def _ecd_wrapper(a0, time_limit, verbose=False, label=""):
    """Wrapper for exact_coord_descent that returns (a, score, sweeps)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())

    a_opt, s_opt, sweeps = exact_coord_descent(a0, time_limit, verbose=verbose, label=label)
    return a_opt, s_opt, sweeps
