One key insight I want to incorporate: **the Nesterov update can be improved by using a "dual" variable**.

In the standard Nesterov for the game:
- y_{t+1} = (1 - alpha_t) * y_t + alpha_t * b_{t+1} (dual variable)
- a_{t+1} = (t * a_t + alpha_t * (t+1) * b_{t+1}) / (t + alpha_t * (t+1)) (primal, averaged)

The key is that b_{t+1} is computed from y_t (the extrapolated dual), not from a_t. This gives the acceleration.

A variant that sometimes works better is to use a "heavy ball" momentum on the primal:
- a_{t+1} = a_t + beta_t * (b_{t+1} - a_t)
where beta_t is a momentum parameter.

Or the "polyak" step:
- a_{t+1} = a_t + gamma_t * (b_{t+1} - a_t)
where gamma_t = 2/(t+2) (same as Nesterov's alpha).

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)

    if s < 1e-15:

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with adaptive momentum and peak-directed restarts."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    no_improve_iters = 0
    rng = np.random.default_rng(777)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Adaptive Nesterov: use 2/(it+2) but with occasional larger steps
        if it < 10:
            alpha = 1.0  # Full step initially for fast convergence
        else:
            alpha = 2.0 / (it + 2)

        # Nesterov dual update
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal update (averaged)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # Stall detection with peak-directed perturbation
        if it % 200 == 0 and it > 0:
            if best_score < prev_score - 1e-6:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score

            if stall_count >= 4:
                # Peak-directed perturbation: push mass away from peak-contributing positions
                b_conv = fast_convolve(a)
                k_peak = int(np.argmax(b_conv))
                js = max(0, k_peak - n + 1)

                contribs = a[idx] * a[k_peak - idx]
                top_n = max(3, n // 10)
                top_positions = idx[np.argsort(contribs)[::-1][:top_n]]

                # Perturb: reduce top contributors, increase bottom
                bottom_n = max(3, n // 10)

def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000, n_stages=None):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def frank_wolfe_refine(a0, time_limit, n_steps=50000):
    """Frank-Wolfe conditional gradient with away steps and adaptive line search."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = a[k_star - i]

        i_toward = int(np.argmin(g))

        best_trial = None
        best_trial_score = best_score

        # Adaptive step sizes based on progress
        step_sizes = [0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.03]
        for t_val in step_sizes:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score_trial = 2.0 * n * fast_convolve(a_trial).max()
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()

        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.4, 0.3, 0.2, 0.1, 0.05]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=15000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                if contribs[ci] > 0:
                    contrib_count[idx[ci]] += contribs[ci]

        top_coords = np.argsort(contrib_count)[::-1][:15]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.89, 0.84, 0.78]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                low_coords = np.argsort(contrib_count)[:n // 4]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-10
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                else:
                    mask_others = np.ones(n, dtype=bool)
                    mask_others[i] = False
                    others = a_trial[mask_others]
                    if others.sum() > 0:
                        a_trial[mask_others] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        # Asymmetric: heavy on one side
        x = np.linspace(0, 1, n)
        a = np.exp(-x * 4) * (1 + 0.2 * np.sin(6 * np.pi * x))
    elif mode == 7:
        # Two-peaked
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) ** 2) * 40) + np.exp(-((x - 0.75) ** 2) * 40)
    elif mode == 8:
        # Power law decay
        x = np.linspace(0.01, 1, n)
        a = x ** (-0.5)
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def nesterov_lp_multimomentum(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov LP with multiple momentum schedules, switching when stalled."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    schedules = [
        lambda it: 1.0 if it < 5 else 2.0 / (it + 2),
        lambda it: 1.0 if it < 5 else 2.0 / (it + 1),
        lambda it: min(1.0, 0.1 * (it + 1)) if it < 10 else 2.0 / (it + 2),
        lambda it: 1.0 if it < 20 else 1.5 / (it - 18 + 2),
    ]

    for sched_idx in range(len(schedules)):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Reset iteration counter for each schedule
        it_offset = 0
        local_stall = 0
        local_best = best_score

        for it in range(n_max_iters // len(schedules)):
            if time.monotonic() - t0 > time_limit:
                break
            total_it = it_offset + it
            alpha = schedules[sched_idx](total_it)

            b_new, M_b = solve_lp_best_response(y, solver)
            if b_new is None:
                continue

            y = (1 - alpha) * y + alpha * b_new
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
                sy = 1.0
            y = y / sy

            a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
            else:
                a = a / sa

            score_a = 2.0 * n * fast_convolve(a).max()
            if score_a < best_score - 1e-10:
                best_score = score_a
                best_a = a.copy()
                local_stall = 0
            else:
                local_stall += 1

            if local_stall > 300:
                # Perturb and break to next schedule
                noise = rng.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                break

        it_offset += it

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining accelerated LP, L-BFGS smoothing,
    Frank-Wolfe, peak shaving, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return sc

    # === Phase 1: Multi-momentum Nesterov LP ===
    print(f"\n=== Phase 1: Multi-Momentum Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_multimomentum(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp, "nesterov_multi_1")

    # === Phase 2: Standard Nesterov LP (different schedule) ===
    print(f"\n=== Phase 2: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=60000)
        try_update(a_lp2, "nesterov_2")

    # === Phase 3: L-BFGS with multi-stage annealing ===
    print(f"\n=== Phase 3: L-BFGS Extended (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 200->0.05, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=200.0, T_end=0.05, maxiter=12000, n_stages=12)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Frank-Wolfe local refinement ===
    print(f"\n=== Phase 4: Frank-Wolfe (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 126.0)
    if t_alloc > 10:
        a_fw, _ = frank_wolfe_refine(best_seq, t_alloc, n_steps=50000)
        try_update(a_fw, "frank_wolfe")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 90.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=20000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 2.0, 0.75, 1.33, 1.1]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [512, 768, 1024, 1280, 1536, 2048, 3072]:
        if 256 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))[:5]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 45:
            break
        t_total = min(rem * 0.15, 120.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=20000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb = min(rem * 0.3, t_total * 0.35)
            if t_lb > 10:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=30.0, T_end=0.1, maxiter=3000, n_stages=6)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        # Peak-directed perturbation
        b_conv = fast_convolve(best_seq)
        k_peak = int(np.argmax(b_conv))
        js = max(0, k_peak - n_cur + 1)
        je = min(n_cur - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = best_seq[idx] * best_seq[k_peak - idx]
        top_positions = idx[np.argsort(contribs)[::-1][:n_cur // 8]]
        bottom_positions = idx[np.argsort(contribs)[:n_cur // 8]]

        for scale in [0.04, 0.08, 0.12]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq.copy()
            for p in top_positions:
                perturbed[p] *= (1.0 - scale)
            for p in bottom_positions:
                perturbed[p] *= (1.0 + scale)
            perturbed = norm_sequence(perturbed)

            t_sub = min(rem * 0.4, 60.0)
            a_pert, _ = nesterov_lp_refine(perturbed, t_sub, solver=solver, n_max_iters=15000)
            try_update(a_pert, f"pert_peak_{scale}")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_pert2 = lbgfs_refine(best_seq, min(rem * 0.3, 30.0), T_start=10.0, T_end=0.1, maxiter=3000, n_stages=6)
                try_update(a_pert2, f"pert_lbgfs_{scale}")

    # === Phase 8: Final polish cascade ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.25, 45.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.02, maxiter=5000, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_fw = min(rem * 0.2, 25.0)
        a_fw2, _ = frank_wolfe_refine(best_seq, t_fw, n_steps=15000)
        try_update(a_fw2, "final_fw")

    rem = deadline - time.monotonic()
    if rem > 8:
        t_ps = min(rem * 0.2, 20.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=8000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 8:
        t_lp_fin = min(rem * 0.3, 15.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
