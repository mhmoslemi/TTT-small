def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    prob = cp.Problem(cp.Minimize(M_var), constraints)

                continue

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):

def kkt_fixed_point(a0, time_limit, max_iters=1000):
    """
    KKT fixed-point iteration for precise optimization.
    
    At each iteration:
    1. Identify active set S (peak positions in conv(a,a))
    2. Set lambda = uniform on S
    3. Solve the linear system M @ a = c * 1, sum(a) = 1
    4. Update a, repeat
    
    The matrix M has M[j,m] = lambda[j+m] if j+m in S, else 0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set: positions where b[k] is within tolerance of max
        # Use a tight tolerance to keep the active set small
        tol = max_b * 1e-6 + 1e-15
        active = np.where(b >= max_b - max_b * 1e-4)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # If active set is too large, restrict to top K
        K = min(len(active), 15)
        sorted_active = active[np.argsort(b[active])[::-1][:K]]
        
        # Set lambda uniform on active set
        lam = np.zeros(2 * n - 1)
        lam[sorted_active] = 1.0 / K
        
        # Build the linear system: M @ a = c * 1, 1^T @ a = 1
        # M[j, m] = lam[j+m] if j+m is in S, else 0
        # System: [M  -1] [a]   [0]
        #         [1^T 0] [c] = [1]
        
        # Build M as a dense matrix (n x n) - for n=1200 this is 1.44M entries, manageable
        # M[j, m] = lam[j+m] for j+m in range(0, 2n-1)
        # Create index matrix
        jm = np.add.outer(np.arange(n), np.arange(n))  # n x n matrix of j+m values
        # M = lam[jm] but lam is only defined on [0, 2n-2]
        M = np.where(jm < 2 * n - 1, lam[jm], 0.0)
        
        # Build the augmented system
        # [M  -1] [a]   [0]
        # [1^T  0] [c] = [1]
        
        # Solve using numpy
        try:
            aug_M = np.vstack([np.hstack([M, -np.ones((n, 1))]),
                               np.hstack([np.ones((1, n)), [[0.0]])])])
            aug_b = np.concatenate([np.zeros(n), [1.0]])
            sol = np.linalg.solve(aug_M, aug_b)
            a_new = sol[:n]
            c_new = sol[n]
        except np.linalg.LinAlgError:
            # Fallback: use least squares
            try:
                aug_M = np.vstack([np.hstack([M, -np.ones((n, 1))]),
                                   np.hstack([np.ones((1, n)), [[0.0]])])])
                aug_b = np.concatenate([np.zeros(n), [1.0]])
                sol, _, _, _ = np.linalg.lstsq(aug_M, aug_b, rcond=None)
                a_new = sol[:n]
                c_new = sol[n]
            except Exception:
                continue
        
        # Check non-negativity
        if np.min(a_new) < -1e-10:
            # Clip and re-normalize
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
            a_new = a_new / sn
        else:
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
            a_new = a_new / sn
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            score_d = 2.0 * n * fast_convolve(a).max()
            if score_d < best_score:
                best_score = score_d
                best_a = a.copy()
        
        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={it+1} ({elapsed:.1f}s): score={best_score:.8f}, |S|={K}")
    
    return best_a, best_score

def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = _softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            log_sum_exp = max_b + np.log(exp_b.sum())
            return (2.0 * n / stage_T) * log_sum_exp
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = _softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return _softmax_from_z(z)


def _softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def evaluate_local(a):

def peak_shaving(a0, time_limit, n_steps=50000):
    """Peak shaving: iteratively reduce elements that contribute most to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find elements contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Reduce the top contributor
        top_pos = idx[np.argmax(contribs)]
        factor = 0.999
        a_trial = a.copy()
        a_trial[top_pos] *= factor
        diff = a[top_pos] - a_trial[top_pos]
        if diff < 1e-16:
            break
        # Redistribute
        mask = np.ones(n, dtype=bool)
        mask[top_pos] = False
        others = a_trial[mask]
        if others.sum() < 1e-15:
            break
        a_trial[mask] = others + diff * others / others.sum()
        a_trial = a_trial / a_trial.sum()
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            a = a_trial
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
