def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(n):

def build_conv_matrix_for(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_lp_solver():
    """Find a working LP solver globally."""
    global _cached_solver
    if _cached_solver is not None:
        return _cached_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver_name
                return _cached_solver
        except Exception:
            continue
    _cached_solver = cp.ECOS
    return _cached_solver


def solve_lp_best_response(a_fixed, solver=None, max_iters=300):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b, M) or (None, None) on failure."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build T efficiently
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iters, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    # Fallback solvers
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=50000,
                         stall_patience=12, print_interval=2000):
    """Nesterov dual-averaging LP alternation with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 20:
                break
            continue

        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal iterated average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection with perturbation
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.02 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            if print_interval and (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_LP={M_b:.8f}")

    return best_a, best_score


def logsumexp_smoothed_obj(z, n, T_stage, T_matrix):
    """Compute LogSumExp-smoothed objective in log-space."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    # Weighted LogSumExp focusing on top peaks
    sorted_b = np.sort(b)[::-1]
    top_k = min(50, len(b))
    # Use top-k aware temperature
    top_vals = sorted_b[:top_k]
    max_val = top_vals[0]
    # Smooth with emphasis on top peaks
    weights = np.ones(len(b))
    peak_idx = np.argsort(b)[::-1][:top_k]
    weights[peak_idx] = 5.0
    wT = T_stage * weights
    exp_wT = np.exp(wT * (b - max_val))
    log_sum_exp = max_val + np.log(exp_wT.sum()) / weights.sum()
    return (2.0 * n / T_stage) * log_sum_exp


def logsumexp_grad(z, n, T_stage, T_matrix):
    """Gradient of the LogSumExp-smoothed objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    # Standard LogSumExp gradient
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0
    S = np.dot(grad_a, a)
    return a * (grad_a / T_stage - S / T_stage)


def lbgfs_logsumexp(a0, time_limit, T_stages=None, maxiter_per_stage=4000):
    """L-BFGS with LogSumExp smoothing, multi-stage temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02]

    n_stages = len(T_stages)
    iters_per = max(20, maxiter_per_stage // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_loc(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse

        def jac_loc(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            grad_a *= 2.0 / T_stage
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_coord_descent(a0, time_limit, n_steps=50000, top_m=10):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        for peak_k in top_peaks[:4]:
            # Find elements contributing to this peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            # Contribution of each a[pos] to this peak: a[pos] * a[peak_k - pos]
            contribs = a[idx] * a[peak_k - idx]
            # Try reducing top contributors
            order = np.argsort(contribs)[::-1][:5]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.985, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Simultaneous small reduction of top 3 contributors to max peak
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 300:
                    break

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method targeting active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.015
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: positions within 0.5% of max
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Average gradient over active positions
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Gradient step with projection
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            lr *= 0.92
            no_improve = 0
        lr = max(1e-7, lr)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        # Slightly asymmetric bell
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    else:
        # Power-law decay
        a = (1 - x) ** 0.7 + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Primary Nesterov LP (longest) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 600.0)
    if t1 > 40:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t1, solver=solver,
                                        max_iters=200000, stall_patience=10)
        try_update(a_lp, "nlp_phase1")

    # === Phase 2: LogSumExp L-BFGS refinement ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 350.0)
    if t2 > 20:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02]
        print(f"  T_stages ({len(T_stages)} stages), budget={t2:.0f}s")
        a_lb = lbgfs_logsumexp(best_seq, t2, T_stages=T_stages, maxiter_per_stage=12000)
        try_update(a_lb, "lbgfs_phase2")

    # === Phase 3: Peak-directed coordinate descent ===
    print(f"\n=== Phase 3: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 15:
        a_cd, _ = peak_coord_descent(best_seq, t3, n_steps=80000, top_m=8)
        try_update(a_cd, "peak_cd")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 10:
        a_sg, _ = subgradient_refine(best_seq, t4, n_steps=60000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Second LP pass (refined) ===
    print(f"\n=== Phase 5: LP Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.14, 200.0)
    if t5 > 30:
        a_lp2, _ = nesterov_lp_optimize(best_seq, t5, solver=solver,
                                         max_iters=100000, stall_patience=15)
        try_update(a_lp2, "nlp_phase5")

    # === Phase 6: Multi-start diversity ===
    print(f"\n=== Phase 6: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        # Try perturbed versions with short LP
        t_each = min(rem / 5.0, 50.0)
        for pi in range(3):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            if pi == 0:
                # Cosine perturbation
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.03 * np.cos(2 * x) + 0.02 * np.cos(3 * x)
            elif pi == 1:
                # Random perturbation
                pert = 0.02 * rng.standard_normal(n_cur)
            else:
                # Different frequency
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.025 * np.sin(5 * x)
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            a_res, _ = nesterov_lp_optimize(a_pert, t_each, solver=solver,
                                             max_iters=20000, stall_patience=8)
            try_update(a_res, f"perturb_{pi}")

    # === Phase 7: Alternative length ===
    print(f"\n=== Phase 7: Alt length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        alt_n = int(n_cur * 1.2)
        t_alt = min(rem * 0.4, 150.0)
        if t_alt > 20:
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_optimize(init_alt, t_alt * 0.55, solver=solver,
                                             max_iters=30000, stall_patience=10)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                a_alt2 = lbgfs_logsumexp(best_seq, rem * 0.4,
                                          T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                          maxiter_per_stage=3000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.5, 50.0)
        if t_fin > 8:
            a_fin = lbgfs_logsumexp(best_seq, t_fin,
                                     T_stages=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01],
                                     maxiter_per_stage=5000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.5, 25.0),
                                       n_steps=30000, top_m=6)
        try_update(a_cd2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg2, _ = subgradient_refine(best_seq, min(rem * 0.6, 15.0), n_steps=20000)
        try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
