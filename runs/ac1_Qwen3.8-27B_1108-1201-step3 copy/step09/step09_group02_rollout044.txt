import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_conv(a, conv):
    """Compute score from precomputed convolution (assumes sum(a)=1)."""
    n = len(a)
    return 2.0 * n * float(conv.max())


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        r = np.nonzero(cond)[0][-1]
        theta = (cssv[r] - 1.0) / (r + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: minimize M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

        # Adaptive ordering: process coordinates near the argmax first
        k_max = int(np.argmax(conv))

        # Coordinates i that appear in conv[k_max] are most important
        # conv[k_max] = sum_i a[i]*a[k_max - i] (with valid indices)
        # So all i in [max(0,k_max-n+1), min(n-1,k_max)] affect the peak
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        priority = np.zeros(n)
        priority[i_start:i_end+1] = 1.0
        # Secondary: coordinates near other top peaks
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        for pk in peaks:
            ps = max(0, int(pk) - n + 1)
            pe = min(n - 1, int(pk))
            priority[ps:pe+1] += 0.3

        order = np.argsort(-priority)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i

                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            qm = unaff_max

            qi = quad_idx

def spectral_shaping_step(a, conv, n_modes=20, rng=None):
    """
    Modify the sequence in the frequency domain to reduce peak.
    Targeted at reducing the dominant spectral peaks.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)

    # Compute power spectrum
    P = np.abs(A) ** 2

    # The peak of autocorrelation relates to the flatness of P
    # If P is flat, autocorrelation is concentrated at lag 0
    # We want to make P as flat as possible while maintaining a >= 0

    # Strategy: slightly flatten the spectrum
    # Target: P_target = mean(P) * (1 + eps * max(0, P - mean(P))/mean(P))
    mean_P = P.mean()
    if mean_P < 1e-30:
        return a

    # Gentle flattening
    alpha = 0.1 + 0.3 * rng.random()
    P_new = (1 - alpha) * P + alpha * mean_P

    # Reconstruct with new amplitudes, keep phases
    phases = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phases)

    # Ensure A[0] is real and positive (DC)
    A_new[0] = np.abs(A_new[0])
    # Hermitian symmetry is maintained by irfft

    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new


def subgradient_peak_shaving(a, conv, lr, n_peaks=10, rng=None):
    """Subgradient step targeting the top peaks with soft weighting."""
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]

    temp = 0.003 * max(float(pv.max()), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    a_new = simplex_project(a - lr * grad)
    return a_new

            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = decay / (iters + decay)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)

            if cs < 1e-10:

            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 50.0)

        # Choose perturbation type
        pert_type = p % 3

        if pert_type == 0:
            # Gaussian noise
            sigma = 0.015 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Spectral perturbation
            perturbed = spectral_shaping_step(a, fast_convolve(a), n_modes=30, rng=rng)
            # Add small noise
            perturbed += rng.standard_normal(n) * 0.01
        else:
            # Peak-directed perturbation: reduce values that contribute to peak
            conv_p = fast_convolve(a)
            k_max = int(np.argmax(conv_p))
            i_start = max(0, k_max - n + 1)
            i_end = min(n - 1, k_max)
            perturbed = a.copy()
            # Reduce values in the peak region
            n_reduce = max(3, (i_end - i_start + 1) // 4)
            idx = np.random.default_rng(p * 1000 + 42).choice(
                np.arange(i_start, i_end + 1), size=n_reduce, replace=False)
            for ii in idx:
                perturbed[ii] *= (1.0 - 0.1 * rng.random())
            # Compensate by increasing others
            others = np.setdiff1d(np.arange(n), idx)
            n_incr = max(3, len(others) // 10)
            idx2 = rng.choice(others, size=min(n_incr, len(others)), replace=False)
            for ii in idx2:
                perturbed[ii] *= (1.0 + 0.1 * rng.random())

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} type={pert_type}: {best_score:.8f}")

    return best_a, best_score


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_zadoff_chu_based(n, rng):
    """Initialize based on Zadoff-Chu sequence properties."""
    root = int(rng.integers(1, max(2, n // 4)))
    m = int(rng.integers(1, max(2, n // 8)))
    t = np.arange(n, dtype=np.float64)
    # Zadoff-Chu: exp(-i*pi*m*k*(k+1)/n)
    # Take real part magnitude
    zc = np.exp(-1j * np.pi * m * t * (t + 1) / n)
    a = np.abs(zc)
    # Add smooth envelope
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.7)
    envelope = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = a * (0.5 + 0.5 * envelope)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_mixed(n, rng):
    """Mix of multiple harmonics."""
    n_harmonics = int(rng.integers(4, 12))
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.4)
    for j in range(n_harmonics):
        freq = int(rng.integers(1, max(2, n // 3)))
        amp = rng.random() * 0.3
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def ft_polish(a0, time_limit):
    """Final tightening: very fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    best_score = score_from_conv(a, conv)
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    factors = [0.998, 0.995, 0.99, 0.985, 0.975, 0.96, 1.04, 1.025, 1.015, 1.01, 1.005, 1.002]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_factor = 1.0
            best_local_max = float(conv[k_lo:k_hi + 1].max())

            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                if local_max < best_local_max - 1e-16:
                    best_local_max = local_max
                    best_factor = factor

            if best_factor != 1.0:
                delta = v * (best_factor - 1.0)
                a[i] = max(a[i] + delta, 0.0)

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = score_from_conv(a, conv)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

    return best_a, best_score
