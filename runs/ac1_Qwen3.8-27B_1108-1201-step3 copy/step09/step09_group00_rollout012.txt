"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using semismooth Newton on the active set,
LP-alternating initialization, and multi-length exploration.
Key innovation: Instead of subgradient/L-BFGS polishing, uses Newton's method on
the KKT conditions to achieve quadratic convergence to the local optimum.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            raise Exception("Not optimal")
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_alternating(a0, time_limit, solver=None, max_iters=50000):
    """LP alternating with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    prev_best = best_score
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall > 5:
                noise = rng.standard_normal(n) * 0.01
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall = 0

    return best_a


def find_active_set(a, tol=1e-8):
    """Find positions where convolution is within tol of the max."""
    b = fast_convolve(a)
    max_b = b.max()
    active = np.where(b >= max_b - tol)[0]
    return active, max_b


def newton_active_set(a0, active_set, max_iters=100, tol=1e-12):
    """
    Newton's method on the KKT system for the given active set.
    Solves: (a*a)[k] = M for k in active_set, sum(a) = 1, a >= 0
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    m = len(active_set)
    t0 = time.monotonic()

    for it in range(max_iters):
        # Compute current convolution
        b = fast_convolve(a)
        M = b.max()

        # Check convergence: all active peaks equal to max, inactive below
        active_vals = b[active_set]
        inactive_mask = np.ones(len(b), dtype=bool)
        inactive_mask[active_set] = False
        max_inactive = b[inactive_mask].max() if inactive_mask.any() else -1

        if (active_vals.max() - active_vals.min()) < tol * max(1, M):
            if max_inactive <= M + tol * max(1, M):
                break

        # Build the Newton system
        # Variables: a (n) + M (1) = n+1 variables
        # Equations: (a*a)[k_i] - M = 0 for each active k_i (m equations)
        #             sum(a) - 1 = 0 (1 equation)
        # Total: m+1 equations, n+1 variables (underdetermined if m < n)

        # Residual
        F = np.zeros(m + 1)
        for ii, k in enumerate(active_set):
            F[ii] = b[k] - M
        F[m] = a.sum() - 1.0

        # Jacobian (m+1) x (n+1)
        # Row ii (for active k_i): d(a*a)[k_i]/da - e_M
        # Row m: ones (for sum constraint)

        # For the underdetermined system, we solve the minimum norm least squares
        # J * dx = -F
        # where dx = (da, dM)

        # Build J efficiently
        # J[ii, j] = 2*a[k_ii - j] if 0 <= k_ii - j <= n-1 else 0
        # J[ii, n] = -1  (derivative wrt M)
        # J[m, j] = 1 for all j
        # J[m, n] = 0

        J = np.zeros((m + 1, n + 1))
        for ii, k in enumerate(active_set):
            for j in range(n):
                diff = k - j
                if 0 <= diff < n:
                    J[ii, j] = 2.0 * a[diff]
            J[ii, n] = -1.0
        J[m, :n] = 1.0
        J[m, n] = 0.0

        # Solve minimum norm least squares: J @ dx = -F
        # Use SVD for stability
        try:
            U, s, Vt = np.linalg.svd(J, full_matrices=False)
            # Minimum norm solution: dx = Vt^T @ (s^{-1} @ (U^T @ (-F)))
            # Pseudoinverse: dx = Vt^T @ diag(1/s) @ U^T @ (-F)
            r = U.T @ (-F)
            # Regularize small singular values
            s_clip = np.maximum(s, 1e-15)
            dx = Vt.T @ (r / s_clip)
        except Exception:
            # Fallback to lstsq
            try:
                dx, _, _, _ = np.linalg.lstsq(J, -F, rcond=None)
            except Exception:
                break

        da = dx[:n]
        dM = dx[n]

        # Damped Newton step
        step = 1.0
        for _ in range(20):
            a_trial = a + step * da
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            b_trial = fast_convolve(a_trial)
            # Check if the active set constraints are being satisfied
            active_max = b_trial[active_set].max()
            inactive_max = b_trial[inactive_mask].max() if inactive_mask.any() else -1
            new_M = max(active_max, inactive_max)
            # Objective should not increase
            old_obj = 2.0 * n * M
            new_obj = 2.0 * n * new_M
            if new_obj <= old_obj + 1e-12:
                a = a_trial
                break
            step *= 0.5

        else:
            break  # Line search failed

    return a


def newton_refine(a0, time_limit=30.0):
    """Iteratively find active set and apply Newton's method."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for round in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        # Find active set
        active, max_b = find_active_set(a, tol=max_b * 0.01 if max_b > 0 else 0.01)

        # Expand active set slightly to be safe
        # Include positions within 5% of max
        b = fast_convolve(a)
        expanded_active = np.where(b >= b.max() * 0.97)[0]

        if len(expanded_active) > 30:
            # Too many active, use top 15
            top_k = np.argsort(b)[::-1][:15]
            expanded_active = np.sort(top_k)

        # Apply Newton
        a_new = newton_active_set(a, expanded_active)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try with just the single peak
            peak_pos = np.argmax(b)
            a_new2 = newton_active_set(a, np.array([peak_pos]))
            score2 = 2.0 * n * fast_convolve(a_new2).max()
            if score2 < best_score:
                best_score = score2
                best_a = a_new2.copy()
                a = a_new2
            else:
                break

    return best_a


def lbgfs_smooth_refine(a0, time_limit=60.0):
    """L-BFGS refinement with LogSumExp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    from scipy.optimize import minimize

    T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Build the gradient computation
        def func(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            max_b = b.max()
            return (2.0 * n / T) * (max_b + np.log(np.sum(np.exp(b - max_b))))

        def jac(z_val):
            a_val = np.exp(z_val - z_val.max())
            s = a_val.sum()
            a_val = a_val / s
            b = fast_convolve(a_val)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            # Gradient of (2n/T) * logsumexp(b) wrt a
            # db[k]/da[j] = 2*a[k-j] if valid
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_val[k - idx]
            # Chain rule for softmax
            S = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S)
            return (2.0 * n / T) * grad_z

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 5000, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': 200000, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_result = np.exp(z - z.max())
    a_result = a_result / a_result.sum()
    return a_result


def evaluate_local(a):
    """Local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Sinc-like
        a = np.sinc(3 * (x - 0.5))
        a = np.maximum(a, 0)
    else:
        # Triangular
        a = 1 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent_refine(a0, time_limit=30.0):
    """Coordinate descent targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        top_positions = np.sort(top_positions)[::-1]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

    return best_a

**Approach: Water-filling on the convolution**

The idea: we want all b[k] to be equal to some value M. This means:
b[k] = M for all k = 0, ..., 2n-2

This gives 2n-1 equations in n unknowns (the a[i] values). This is overdetermined, so an exact solution doesn't exist. But we can find the a that minimizes the variance of b, or minimizes (max(b) - min(b))^2, or minimizes sum((b[k] - M)^2) for some M.

This is a nonlinear least squares problem! We can solve it with Levenberg-Marquardt or Gauss-Newton.

Let's set up: minimize sum_k (b[k] - M)^2 where M is also a variable (or M = mean(b) = sum(a)^2/(2n-1) which is fixed by normalization).

Wait, sum_k b[k] = (sum a)^2 = 1 (with normalization). So M = 1/(2n-1) is the average. We want to minimize sum_k (b[k] - 1/(2n-1))^2, which is equivalent to minimizing the variance of b.

This is a well-posed nonlinear least squares problem! The Jacobian is:
J[k,j] = db[k]/da[j] = 2*a[k-j] (if 0 <= k-j <= n-1)

So J is a (2n-1) x n matrix. We want to solve:
minimize ||J(a) * da - (target - b(a))||^2 + lambda*||da||^2

where target = ones(2n-1) / (2n-1) (the flat target).

This is a Gauss-Newton iteration! Let me implement this.

Actually, this is even simpler. Since sum(b) = 1 is fixed, minimizing the variance of b is the same as minimizing sum(b[k]^2) = ||b||^2. And ||b||^2 = a^T (C^T C) a where C is the convolution operator. This is a quadratic form!

Wait: ||conv(a,a)||^2 = sum_k (sum_i a[i]*a[k-i])^2

This is NOT a quadratic form in a; it's a quartic form. So we can't minimize it directly by solving a linear system.

But we can use Gauss-Newton: at each step, linearize b(a) around the current a, and solve the linear least squares problem.

def gauss_newton_flat(a0, time_limit=60.0):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(1000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Target: make b as flat as possible
        # Since sum(b) = 1, target = 1/(2n-1)
        target = np.ones(2*n-1) / (2*n-1)
        residual = b - target

        # Jacobian: J[k,j] = db[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        # Build J efficiently
        J = np.zeros((2*n-1, n))
        for j in range(n):
            valid_k = np.arange(j, j + n)  # k where k-j is in [0, n-1]
            # Actually k-j ranges from 0 to n-1, so k ranges from j to j+n-1
            # But k must be in [0, 2n-2]
            k_max = min(j + n - 1, 2*n - 2)
            valid_k = np.arange(j, k_max + 1)
            J[valid_k, j] = 2.0 * a[valid_k - j]

        # Gauss-Newton step: minimize ||J @ da + residual||^2
        # (J @ da = -residual)
        # But we also need to maintain sum(a) = 1 and a >= 0

        # Solve the normal equations: (J^T J + lambda*I) @ da = -J^T @ residual
        try:
            JtJ = J.T @ J
            JtR = J.T @ residual
            # Add regularization
            lam = 1e-8
            JtJ_reg = JtJ + lam * np.eye(n)
            da = np.linalg.solve(JtJ_reg, -JtR)
        except Exception:
            try:
                da, _, _, _ = np.linalg.lstsq(J, -residual, rcond=None)
            except Exception:
                break

        # Projected step: maintain sum=1 and a>=0
        a_trial = a + da
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_trial = a + factor * da
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    break

    return best_a

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]
