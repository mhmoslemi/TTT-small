import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    """Compute full cross-convolution of a with b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    """Convert log-space z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))

                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, fractional_top=0.08):
    """Compute subgradient of max(conv(a,a)) w.r.t. a using top peaks."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - fractional_top * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def nesterov_mirror_descent(a0, time_limit, frac_start=0.1, frac_end=0.03):
    """Nesterov-accelerated mirror descent with adaptive active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a).copy()
    z_prev = z.copy()
    a_prev = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    # Nesterov parameters
    t_k = 1.0
    lr = 1.0
    no_improve = 0
    max_no_improve = 50
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    
    total_steps = 0
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive fractional top
        progress = min(1.0, step / 50000.0)
        frac = frac_start + (frac_end - frac_start) * progress
        
        # Nesterov extrapolation
        t_prev = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_prev - 1.0) / t_k
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute subgradient at y
        g, max_y = compute_subgradient(y, frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-18:
            break
        
        # Polyak step size
        gap = best_score - f_lb
        if gap < 1e-12:
            break
        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)
        
        # Mirror descent update
        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a_prev = a.copy()
            a = a_new
            z_prev = z.copy()
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 3.0)
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Nesterov restart
                z = z_from_a(a)
                z_prev = z.copy()
                a_prev = a.copy()
                t_k = 1.0
                lr *= 0.6
                no_improve = 0

            print(f"      NMD {elapsed:.0f}s step={step}: score={best_score:.8f} lr={lr:.4f} frac={frac:.4f}")
    
    return best_a, best_score, total_steps
