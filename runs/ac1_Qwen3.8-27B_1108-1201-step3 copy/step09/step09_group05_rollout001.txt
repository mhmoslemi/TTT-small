        top_ci = int(np.argmax(contribs))
        i = idx[top_ci]
        j = peak_k - i
        
        improved = False
        
        # Try reducing a[i]
        for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
            a_trial = a.copy()
            reduction = a_trial[i] * (1 - factor)
            a_trial[i] *= factor
            # Redistribute to positions that don't contribute to the peak
            # Find positions that don't pair with anything at peak_k
            safe_mask = np.ones(n, dtype=bool)
            for ii in range(js, je + 1):
                partner = peak_k - ii
                if 0 <= partner < n:
                    safe_mask[ii] = False
                    safe_mask[partner] = False
            if safe_mask.sum() > 0:
                safe_vals = a_trial[safe_mask]
                weights = safe_vals + 1e-10
                a_trial[safe_mask] += reduction * weights / weights.sum()
            else:
                # Can't find safe positions, just scale down
                a_trial = a_trial / a_trial.sum()
            
            a_trial = normalize(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Try random perturbation
                rng = np.random.default_rng(step)
                a_trial = a + 0.02 * rng.standard_normal(n)
                a_trial = normalize(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0
    
    return best_a, best_score

def nesterov_lp_simple(a0, time_limit, n_max_iters=50000):
    """
    Simplified Nesterov LP alternating optimization.
    """
    n = len(a0)
    a = normalize(a0).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: minimize max_k conv(y, b) s.t. b >= 0, sum(b) = 1
        # Using subgradient method instead of full LP for speed
        b = a.copy()
        for inner in range(20):
            conv_yb = np.convolve(y, b)
            k_star = int(np.argmax(conv_yb))
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            g[idx] = y[k_star - idx]
            lr = 1.0 / (np.sqrt(inner + 1) * np.sqrt(np.dot(g, g)) + 1e-10)
            b = b - lr * g
            b = np.maximum(b, 0)
            b = b / b.sum()
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a_new = ((it + 1) * a + (it + 2) * b) / (2 * (it + 2))
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stall = 0
        else:
            a = a_new  # Still use for Nesterov momentum
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if it % 100 == 0 and sc >= best_score:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(42 + it)
                y = y + 0.05 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
    
    return best_a, best_score

def try_update(best_state, a, label):
    """Try to update the best state."""
    if a is None:
        return best_state
    a = normalize(a)
    sc = score_normalized(a)
    if sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: score={sc:.6f} (n={len(a)})")
    return best_state

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp = f_interp / s
    return f_interp

def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 4)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 2, n))
    else:
        a = np.ones(n) * (1 + 0.2 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
