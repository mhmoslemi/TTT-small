def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def score_of(a):
    n = len(a)
    b = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2.0 * n * b.max() / (s * s))


def evaluate_safe(a):

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_try = [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_try:
        if s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:

                return b_val, float(M_var.value)
        except Exception:

def lbgfs_smoothed_objective(z_in, T_val, n, t0, time_limit):
    if time.monotonic() - t0 > time_limit:
        return 1e15, None
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    val = (2.0 * n / T_val) * log_sum_exp
    return val, (a, b, exp_Tb)


def lbgfs_smoothed_grad(z_in, T_val, n, t0, time_limit):
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    size = len(b)
    for k in range(size):
        if w[k] < 1e-16:
            continue
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.03, maxiter=20000, n_stages=15):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            val, _ = lbgfs_smoothed_objective(z_in, stage_T, n, t0, time_limit)
            return val

        def jac(z_in):
            return lbgfs_smoothed_grad(z_in, stage_T, n, t0, time_limit)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                if t_lb_pert > 10:
                    a_pert2 = lbgfs_refine(best_seq, t_lb_pert, T_start=15.0, T_end=0.05, maxiter=5000, n_stages=8)
                    try_update(a_pert2, f"pert_lb_{scale}")
    print(f"  After Phase 8: best={best_score:.6f}, elapsed={time.monotonic()-t_start:.0f}s")

    # Phase 9: Final combined polish
    print(f"\n=== Phase 9: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.01, maxiter=8000, n_stages=10)
        try_update(a_fin, "final_lb")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.25, 40.0)
        a_ps, _ = peak_equalize(best_seq, t_ps, n_steps=20000)
        try_update(a_ps, "final_pe")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_fw = min(rem * 0.25, 30.0)
        a_fw2, _ = frank_wolfe_refine(best_seq, t_fw, n_steps=30000)
        try_update(a_fw2, "final_fw")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.30, 25.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, max_iters=10000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
        return [float(x) for x in best_seq]

    # Fallback
    fallback = np.ones(256) / 256
    fs = evaluate_safe(fallback)
    print(f"\nFallback score: {fs:.6f}")
    return [float(x) for x in fallback]
