    if n == 0:
        return v

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: (1) Nesterov projected gradient warmup, (2) extended LP-based
alternating minimization with Nesterov dual averaging, (3) peak shaving,
(4) perturbation-based multi-start with LP refinement, (5) nearby-length
exploration, (6) final polish. Key improvements over prior: longer LP
refinement, adaptive perturbation schedule, symmetric-aware peak reduction,
and multi-solver fallback for robustness.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cache = None


def get_lp_solver():
    global _lp_solver_cache
    if _lp_solver_cache is not None:
        return _lp_solver_cache
    for name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = name
                return name
        except Exception:
            continue
    _lp_solver_cache = cp.ECOS
    return _lp_solver_cache


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=300.0, lr_init=0.08,
                n_stages=10, steps_per_stage=3000):
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            lr = lr_init / (1.0 + 0.004 * step_count)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0
            if step_count > 0:
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 50 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000, alpha_coeff=2.0):
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score + 1e-6
    rng_local = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = alpha_coeff / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0 and it > 0:
            if best_score >= prev_best - 1e-8:
                stall_count += 1
            else:
                stall_count = 0
            prev_best = best_score
            if stall_count >= 5:
                noise_scale = 0.015 * max(1.0, 10.0 / (it + 1))
                y = y + noise_scale * rng_local.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def make_diverse_init(n, variant, rng):
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x)
    elif variant == 1:
        a = 1.0 + 0.6 * np.cos(2 * np.pi * (x + 0.1))
    elif variant == 2:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2) + 0.1
    elif variant == 3:
        a = 1.0 + 0.5 * np.sin(3 * np.pi * x + rng.uniform(0, np.pi))
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

                zero_idx = rng.choice(n_best, size=n_best // 40, replace=False)
                a_pert[zero_idx] = 0

            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            print(f"  Perturb {pidx}: score_init={score_normalized(a_pert):.6f}")
            a_sub, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=50000, alpha_coeff=2.0)
            try_update(a_sub, f"pert_{pidx}")

    # === Phase 5: Nearby lengths ===
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-40, -20, 20, 40, 60]:
        alt_n = n_best + delta
        if 400 <= alt_n <= 2048:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]
    print(f"\n=== Phase 5: Alt lengths {alt_lengths} (best={best_score:.6f}) ===")
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_sub = min(rem * 0.25, 150.0)
        if t_sub < 30:
            break
        print(f"  n={alt_n}, budget={t_sub:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_lp_refine(init_alt, t_sub, solver, n_max_iters=50000, alpha_coeff=2.0)
        try_update(a_alt, f"alt_n{alt_n}")

    # === Phase 6: Second extended LP from best ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 200.0)
    print(f"\n=== Phase 6: Final LP polish (best={best_score:.6f}, {t_alloc:.0f}s) ===")
    if t_alloc > 20:
        a_fin, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=100000, alpha_coeff=2.0)
        try_update(a_fin, "final_lp")

    # === Phase 7: Final peak shave ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 40.0)
    if t_alloc > 5:
        a_ps2, _ = peak_shave_local(best_seq, t_alloc, n_steps=10000)
        try_update(a_ps2, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
