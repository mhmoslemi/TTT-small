"""
Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2
using SCA (Successive Convex Approximation) with LP subproblems,
accelerated LP game, and gradient-based refinement.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_sca_matrix(a_t):
    """Build matrix for SCA: 2*T(a_t) @ a <= M + (a_t*a_t)"""
    return 2.0 * build_conv_matrix(a_t)


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_sca_lp(a_t, solver):
    """
    SCA-LP: min M s.t. 2*T(a_t)@a - (a_t*a_t) <= M, a >= 0, sum(a) = 1
    """
    n = len(a_t)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_sca = build_sca_matrix(a_t)  # 2 * T(a_t)
    conv_tt = fast_convolve(a_t)  # (a_t * a_t)[k]
    
    # Constraint: 2*T(a_t)@a - conv_tt <= M  =>  2*T(a_t)@a - M <= conv_tt
    constraints = [
        T_sca @ a_var - M_var <= conv_tt,
        cp.sum(a_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    return a_val, float(M_var.value)


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def sca_iteration(a0, time_limit, solver, n_max_iters=500, momentum=0.5):
    """
    SCA (Successive Convex Approximation) with LP subproblems.
    Uses momentum/averaging for stability.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    y = a.copy()  # momentum variable
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_new, M_val = solve_sca_lp(y, solver)
        if a_new is None:
            # Fallback: try with current a
            a_new, M_val = solve_sca_lp(a, solver)
            if a_new is None:
                continue
        
        # Blending for stability
        alpha = 0.7 if it < 20 else 0.9
        a_trial = alpha * a_new + (1 - alpha) * a
        a_trial = project_simplex(a_trial)
        
        score = score_normalized(a_trial)
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            # Update y with momentum
            y = 0.5 * y + 0.5 * a_trial
        else:
            # Try the raw SCA solution
            score_raw = score_normalized(a_new)
            if score_raw < best_score:
                best_score = score_raw
                best_a = a_new.copy()
                y = 0.5 * y + 0.5 * a_new
            else:
                # Decay y towards a
                y = 0.8 * y + 0.2 * best_a
        
        a = best_a  # Track best for next SCA linearization
        # Actually, for SCA we should use the current iterate, not the best
        # But using best helps convergence to a good solution
        
        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCA iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def nesterov_lp_game(a0, time_limit, solver, n_max_iters=10000):
    """Nesterov dual averaging for the LP best-response game."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        
        # Averaged iterate
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)
        
        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 200 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it * 7 + 3)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0
                prev_best = best_score
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=200.0,
                lr_init=0.05, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)
            
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0 if step_count > 0 else 1.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
        
        if step_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")
        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score


def multiplicative_weight_update(a0, time_limit, T=50.0, n_steps=5000):
    """Multiplicative weight (entropy mirror descent) on smoothed max."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 1e-15))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad = grad_smooth_max(a, T)
        # Multiplicative update
        eta = 1.0 / np.sqrt(step + 1)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()
            break
        
        if step % 50 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=20000):

            b_work[-1] = -np.inf

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_freq_init(n, rng):
    """Create initialization with flat power spectrum (low autocorrelation peak)."""
    x = np.linspace(0, 1, n)
    # Use a sum of sinusoids with random phases
    n_harm = rng.integers(3, 8)
    a = np.ones(n) / n
    for k in range(1, n_harm + 1):
        freq = rng.integers(2, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.1 / k
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. SCA (Successive Convex Approximation) with LP subproblems - new approach
2. Nesterov accelerated LP best-response game
3. Nesterov projected gradient on log-sum-exp smoothed objective
4. Peak shaving coordinate descent
5. Multi-start and multi-length exploration
Key innovation: SCA-LP directly linearizes the non-convex auto-convolution
at the current iterate, giving a convex (LP) subproblem that converges
faster than the game-based approach for this problem structure.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_sca_lp(a_t, solver):
    """
    SCA-LP: min M s.t. 2*T(a_t)@a - (a_t*a_t) <= M, a >= 0, sum(a) = 1
    This linearizes the auto-convolution at a_t via first-order Taylor expansion.
    """
    n = len(a_t)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_sca = 2.0 * build_conv_matrix(a_t)
    conv_tt = fast_convolve(a_t)

    constraints = [
        T_sca @ a_var - M_var <= conv_tt,
        cp.sum(a_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    return a_val, float(M_var.value)


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def sca_iteration(a0, time_limit, solver, n_max_iters=300, blend_rate=0.6):
    """
    SCA (Successive Convex Approximation) with LP subproblems.
    At each step, linearize the auto-convolution and solve the resulting LP.
    Uses adaptive blending for stability.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    y = a.copy()
    stalled = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_new, M_val = solve_sca_lp(y, solver)
        if a_new is None:
            a_new, M_val = solve_sca_lp(a, solver)
            if a_new is None:
                continue

        # Adaptive blending: more conservative early, more aggressive later
        if it < 10:
            alpha = 0.5
        elif it < 30:
            alpha = blend_rate
        else:
            alpha = min(0.9, blend_rate + 0.2 * it / n_max_iters)

        a_trial = alpha * a_new + (1 - alpha) * a
        a_trial = project_simplex(a_trial)

        score = score_normalized(a_trial)
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            stalled = 0
            y = 0.4 * y + 0.6 * a_trial
        else:
            stalled += 1
            if stalled >= 8:
                # Reset y to best to escape
                y = best_a.copy()
                stalled = 0
            else:
                y = 0.7 * y + 0.3 * best_a

        a = best_a

        if (it + 1) % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCA iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_lp={M_val:.6f}")

    return best_a, best_score


def nesterov_lp_game(a0, time_limit, solver, n_max_iters=10000):
    """Nesterov dual averaging for the LP best-response game."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 100 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 6:
                rng = np.random.default_rng(it * 13 + 7)
                y = y + 0.02 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0
                prev_best = best_score

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=200.0,
                lr_init=0.05, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.002 * step_count)

            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0) if step_count > 0 else 1.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 200 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

        if (step_count) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")
        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score

        je = min(n - 1, peak_k)

                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:

                        for f2 in [0.94, 0.88]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = project_simplex(a_trial)
                            score = score_normalized(a_trial)
                            if score < best_score - 1e-15:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for ci in top_contrib[:6]:
                a_trial[idx[ci]] *= 0.95
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                # Try adding mass at low-contributing positions
                center = peak_k // 2
                far_coords = np.argsort(np.abs(np.arange(n) - center))[::-1][:8]
                found = False
                for fc in far_coords:
                    a_trial = a.copy()
                    a_trial[fc] += 5e-5
                    a_trial = project_simplex(a_trial)
                    score = score_normalized(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        found = True
                        break
                if not found:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_asymmetric_init(n, rng):
    """Create asymmetric initial sequence with varied structures."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif choice == 1:
        a = 1.0 + 0.6 * (2 * x - 1)
    elif choice == 2:
        a = 1.0 + 0.4 * np.sin(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        c1 = 0.25 + 0.15 * rng.random()
        c2 = 0.65 + 0.1 * rng.random()
        a = np.exp(-6.0 * (x - c1) ** 2) + 0.4 * np.exp(-6.0 * (x - c2) ** 2)
    elif choice == 4:
        a = 1.0 + 0.3 * np.cos(6 * np.pi * x)
    elif choice == 5:
        a = 1.0 + 0.4 * np.sin(3 * 2 * np.pi * x + 1.0)
    else:
        n_harm = rng.integers(3, 7)
        a = np.ones(n) / n
        for k in range(n_harm):
            freq = rng.integers(2, n // 3)
            phase = rng.uniform(0, 2 * np.pi)
            a += (0.08 / (k + 1)) * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
