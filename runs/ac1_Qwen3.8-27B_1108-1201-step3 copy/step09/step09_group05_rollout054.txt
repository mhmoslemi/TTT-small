import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score(a):
    """Compute 2*n*max_conv/sum^2 for normalized a."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), prob


def peak_shave(a0, time_limit, n_steps=50000):
    """
    Fast coordinate descent peak shaving.
    At each step: identify the peak, find top contributors,
    reduce them and redistribute to low-impact positions.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        
        # Find the peak (ignore edges which are small)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -1
            b_work[-1] = -1
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]
        
        # Find contributing pairs for this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        pair_idx = peak_k - idx
        contribs = a[idx] * a[pair_idx]
        
        # Top contributors
        top_n = min(10, len(idx))
        top_order = np.argsort(contribs)[::-1][:top_n]
        
        # Find low-impact positions (positions that don't contribute much to ANY peak)
        # Approximate: positions with small a value
        low_positions = np.argsort(a)[:n // 5]
        
        improved = False
        
        # Try reducing top contributors and redistributing
        for ci in top_order:
            i = idx[ci]
            j = pair_idx[ci]
            if a[i] < 1e-14:
                continue
            
            # Try multiple reduction factors
            for factor in [0.95, 0.90, 0.85, 0.78, 0.70]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                
                a_trial = a.copy()
                a_trial[i] -= diff
                
                # Redistribute to low positions
                mask = np.zeros(n, dtype=bool)
                mask[low_positions] = True
                mask[i] = False
                if mask.sum() > 0:
                    targets = a_trial[mask] + 1e-12
                    a_trial[mask] = targets + diff * targets / targets.sum()
                    
                a_trial = norm_sequence(a_trial)
                s = score(a_trial)
                if s < best_score - 1e-12:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Also try pair reduction (reduce both i and j)
        if not improved:
            for ci in top_order[:5]:
                i = idx[ci]
                j = pair_idx[ci]
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                
                for f in [0.93, 0.87, 0.80]:
                    a_trial = a.copy()
                    diff_i = a[i] * (1 - f)
                    diff_j = a[j] * (1 - f)
                    a_trial[i] -= diff_i
                    a_trial[j] -= diff_j
                    
                    mask = np.zeros(n, dtype=bool)
                    mask[low_positions] = True
                    mask[i] = False
                    mask[j] = False
                    if mask.sum() > 0:
                        total_diff = diff_i + diff_j
                        targets = a_trial[mask] + 1e-12
                        a_trial[mask] = targets + total_diff * targets / targets.sum()
                    
                    a_trial = norm_sequence(a_trial)
                    s = score(a_trial)
                    if s < best_score - 1e-12:

def subgradient_project(a0, time_limit, n_steps=100000):
    """
    Subgradient method with projection onto simplex.
    Uses the subgradient of max_k (a*a)[k] which is determined by the argmax.
    """
    n = len(a0)
    a = norm_sequence(a0).copy()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    eta = 1.0 / (n ** 0.5)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Subgradient: grad[i] = a[peak_k - i] for valid i
        grad = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for i in range(js, je + 1):
            grad[i] = a[peak_k - i]
        
        # Adaptive step size
        alpha = eta / (step + 1) ** 0.5
        
        a_trial = a - alpha * grad
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s
        
        s_val = score(a_trial)
        if s_val < best_score:
            best_score = s_val
            best_a = a_trial.copy()
        a = a_trial
    
    return best_a, best_score


def lse_refine(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=8):
    """Log-sum-exp smoothing refinement with decreasing temperature."""
    from scipy.optimize import minimize
    
    n = len(a0)
    a = norm_sequence(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 2000 // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_convolve(a_in)
            Tb = stage_T * b
            max_Tb = Tb.max()
            return (2.0 * n / stage_T) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum()))
        
        def jac(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_convolve(a_in)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov-accelerated LP alternating optimization."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        if it % 200 == 0:
            if best_score < prev_score - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 5:
                rng_local = np.random.default_rng(777 + it)
                noise = rng_local.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)

    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max_conv/sum^2 for a sequence."""
    n = len(a)

    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

    return T

                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]

            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with warm-restart strategy:
    Alternates between LP (global) and local methods (LSE, peak shave),
    using warm restarts to escape plateaus.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sequence(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Initial LP ===
    print(f"\n=== Phase 1: Initial LP (budget 40%) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  LP time: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        sc = evaluate_safe(a_lp)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lp.copy()
            print(f"  LP result: {sc:.8f}")

    # === Phase 2: LSE refinement ===
    print(f"\n=== Phase 2: LSE-LBFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 15:
        a_lb = lse_refine(best_seq, t_alloc, T_start=60.0, T_end=0.3, n_stages=10)
        sc = evaluate_safe(a_lb)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lb.copy()
            print(f"  LSE result: {sc:.8f}")

    # === Phase 3: Multi-peak shaving ===
    print(f"\n=== Phase 3: Multi-Peak Shave (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 10:
        a_ps, _ = multi_peak_shave(best_seq, t_alloc, n_steps=60000)
        sc = evaluate_safe(a_ps)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_ps.copy()
            print(f"  Peak shave result: {sc:.8f}")

    # === Phase 4: Warm-restart LP (second round) ===
    print(f"\n=== Phase 4: Warm-Restart LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 220.0)
    if t_alloc > 15:
        # Perturb slightly before restart
        perturbed = best_seq + 0.02 * rng.standard_normal(len(best_seq))
        perturbed = norm_sequence(perturbed)
        a_lp2, _ = nesterov_lp(perturbed, t_alloc, solver=solver, n_max_iters=50000)
        sc = evaluate_safe(a_lp2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lp2.copy()
            print(f"  LP2 result: {sc:.8f}")

    # === Phase 5: LSE polish ===
    print(f"\n=== Phase 5: LSE Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_lb2 = lse_refine(best_seq, t_alloc, T_start=15.0, T_end=0.1, n_stages=8)
        sc = evaluate_safe(a_lb2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lb2.copy()
            print(f"  LSE2 result: {sc:.8f}")

    # === Phase 6: Peak shave final ===
    print(f"\n=== Phase 6: Final Peak Shave (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 8:
        a_ps2, _ = multi_peak_shave(best_seq, t_alloc, n_steps=30000)
        sc = evaluate_safe(a_ps2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_ps2.copy()
            print(f"  PS2 result: {sc:.8f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.8f}) ===")
    alt_lengths = [800, 1024, 1152, 1280, 640]
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        if alt_n == n_best:
            continue
        t_total = min(rem * 0.22, 100.0)
        if t_total < 25:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        t_lp = t_total * 0.55
        a_alt, _ = nesterov_lp(init_alt, t_lp, solver=solver, n_max_iters=20000)
        sc = evaluate_safe(a_alt)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_alt.copy()
            print(f"  NEW BEST at n={alt_n}: {sc:.8f}")

        rem = deadline - time.monotonic()
        if rem > 12 and best_seq is not None and len(best_seq) == alt_n:
            t_ps = min(rem * 0.35, 35.0)
            if t_ps > 8:
                a_alt2, _ = multi_peak_shave(best_seq, t_ps, n_steps=15000)
                sc2 = evaluate_safe(a_alt2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_alt2.copy()
                    print(f"  PS at n={alt_n}: {sc2:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 12:
        t_fin = min(rem * 0.35, 35.0)
        a_fin = lse_refine(best_seq, t_fin, T_start=5.0, T_end=0.05, n_stages=6)
        sc = evaluate_safe(a_fin)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  Final LSE: {sc:.8f}")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps3 = min(rem * 0.35, 25.0)
        if t_ps3 > 5:
            a_ps3, _ = multi_peak_shave(best_seq, t_ps3, n_steps=15000)
            sc = evaluate_safe(a_ps3)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_ps3.copy()
                print(f"  Final PS: {sc:.8f}")

    rem = deadline - time.monotonic()
    if rem > 8:
        t_lp_fin = min(rem * 0.5, 12.0)
        if t_lp_fin > 4:
            a_lp_fin, _ = nesterov_lp(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
            sc = evaluate_safe(a_lp_fin)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_lp_fin.copy()
                print(f"  Final LP: {sc:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
