    At each step, solve an LP to find the best response to the linearized objective.
    """

    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()

    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    active_k = []  # list of (k, linear_approx_coef) pairs
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        # Update active set: keep top peaks
        active_peaks = np.argpartition(conv, -max_active)[-max_active:]
        
        # Build LP: minimize M
        # s.t. 2 * sum_m a_LP[m] * a[k-m] - conv_k(a) <= M for each active k
        #      a_LP >= 0, sum(a_LP) = 1
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in active_peaks:
            k = int(k)
            # Linear approx: 2 * sum_m a_var[m] * a[k-m] - conv[k] <= M_var
            # The coefficient vector for a_var is: 2 * a[k-m] for valid m
            coef = np.zeros(n)

            i_end = min(n - 1, k)
            coef[i_start:i_end+1] = 2.0 * a[k - i_end:k - i_start + 1][::-1]
            constraints.append(coef @ a_var - conv[k] <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_lp = np.asarray(a_var.value, dtype=np.float64)
        a_lp = np.maximum(a_lp, 0)
        s_lp = a_lp.sum()
        if s_lp < 1e-15:
            break
        a_lp = a_lp / s_lp
        
        # Frank-Wolfe update
        t = 2.0 / (it + 2.0)
        a = (1 - t) * a + t * a_lp
        
        # Also try the LP solution directly
        lp_score = 2.0 * n * fast_convolve(a_lp).max()
        if lp_score < best_score:

    # Cosine window
    x = (t - (n - 1) / 2) / ((n - 1) / 2)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** 2 + 0.05
    inits.append(a / a.sum())

    # Periodic asymmetric
    freq = 1.5
    phase = 0.3
    a = 0.5 + 0.35 * np.cos(2 * np.pi * freq * t / n + phase) + 0.1 * t / n
    a = a + 0.05
    inits.append(a / a.sum())

    # Random smooth
    a = rng.uniform(0.3, 1.0, n)
    # Smooth it
    kernel = np.ones(10) / 10
    a = np.convolve(a, kernel, mode='same')
    inits.append(a / a.sum())

    return inits


def lp_frank_wolfe(a0, time_limit, solver, max_active=40, verbose=True, label=""):
    """
    Frank-Wolfe with linearized cutting planes (LP subproblem).
    Linearize max_k conv_k at current point, solve LP, take FW step.
    """
    import cvxpy as cp

    n = len(a0)
    a = normalize_simplex(a0.copy())

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it % 50 == 0 or it == 1:
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Get active peaks
        if it % 10 == 1 or it == 1:
            active_peaks = np.argpartition(conv, -max_active)[-max_active:]
        else:
            active_peaks = np.argpartition(conv, -max_active)[-max_active:]

        # Build LP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active_peaks:
            k = int(k)
            coef = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            coef[i_start:i_end + 1] = 2.0 * a[k - i_end:k - i_start + 1][::-1]
            constraints.append(coef @ a_var - conv[k] <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        a_lp = np.asarray(a_var.value, dtype=np.float64)
        a_lp = np.maximum(a_lp, 0)
        s_lp = a_lp.sum()
        if s_lp < 1e-15:
            break
        a_lp = a_lp / s_lp

        # Check LP solution directly
        lp_score = 2.0 * n * fast_convolve(a_lp).max()
        if lp_score < best_score - 1e-12:
            best_score = lp_score
            best_a = a_lp.copy()

        # FW update with step size
        t_step = 2.0 / (it + 2.0)
        a = (1.0 - t_step) * a + t_step * a_lp

        # Periodic check
        if it % 50 == 0:
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP-FW[{label}] it={it} ({elapsed:.0f}s): score={score:.6f} best={best_score:.6f} M={M_var.value:.8f}")

            if no_improve > 200:
                break

    return best_a, best_score


def mirror_descent(a0, time_limit, lr_init=0.05, top_k=15, momentum=0.3,
                   verbose=True, label=""):
    """
    Mirror descent (exponential weights) with multi-peak subgradient and momentum.
    """
    n = len(a0)
    a = normalize_simplex(a0.copy())

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    rng_local = np.random.default_rng(it + 42)

    while time.monotonic() - t0 < time_limit:
        it += 1

        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive lr
        if no_improve > 300:
            lr *= 0.75
            no_improve = 0
            if lr < 1e-10:
                break
            # Diversify
            if it > 200:
                noise = rng_local.standard_normal(n) * lr * 5
                a = normalize_simplex(a + noise)
                conv = fast_convolve(a)

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.008 * max(M, 1e-10)
        weights = np.exp((pv - pv.max()) / temp)
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if weights[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += 2.0 * weights[idx] * a[k - i_end:k - i_start + 1][::-1]

        # Momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Mirror descent step
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 0.0)
        sn = a.sum()
        if sn < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / sn

        conv = fast_convolve(a)

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD[{label}] it={it} ({elapsed:.0f}s): score={score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, label="", fine=False):
    """
    Coordinate descent with incremental convolution updates.
    Tries to reduce the max convolution value by adjusting one coordinate at a time.
    """
    n = len(a0)
    a = normalize_simplex(a0.copy())

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    if fine:
        factors = [0.97, 0.98, 0.99, 0.995, 1.005, 1.01, 1.02, 1.03]
    else:
        factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Affected slice
            a_shift = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in factors:
                new_val = a[i] * factor
                delta = new_val - a[i]
                if new_val < -1e-14:
                    continue

                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx2 = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx2] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx2 = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx2] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > (10 if fine else 20):
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweep} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def find_lp_solver():
    """Find a working LP solver for cvxpy."""
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK, cp.CVXOPT]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def propose_candidate():
    """
    Multi-phase optimization:
    1. Multi-start exploration with diverse inits + LP-FW
    2. Focused mirror descent refinement
    3. Coordinate descent polishing
    4. Final verification
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"Starting: n={n}, best_score={best_score:.6f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    # === Phase 1: Multi-start LP-FW exploration ===
    print(f"\n=== Phase 1: LP-FW Multi-Start ===")
    inits = generate_inits(n, rng, parent)
    # Shuffle but keep parent first
    if parent is not None:
        rest = inits[1:]
        rng.shuffle(rest)
        inits = [inits[0]] + rest

    t0 = time.monotonic()
    phase1_budget = deadline - t0 - 600  # reserve 600s for later phases
    per_init_time = max(30.0, phase1_budget / len(inits))

    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic() - 550
        if remaining < 30:
            break
        t_alloc = min(per_init_time, remaining * 0.6)
        if idx == 0:
            t_alloc = min(max(t_alloc, 100.0), remaining * 0.3)

        label = f"init{idx}"
        print(f"  LP-FW {label} (budget={t_alloc:.0f}s)")
        a_res, s_res = lp_frank_wolfe(init, t_alloc, solver, max_active=40,
                                       verbose=False, label=label)
        best_seq, best_score = update_best(a_res, best_seq, best_score, label)

    # === Phase 2: Focused Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 300
    if remaining > 60:
        t_md = min(remaining * 0.5, 250.0)
        print(f"  MD budget={t_md:.0f}s")

        # Run MD with different top_k values
        for top_k, lr_init in [(10, 0.03), (20, 0.02), (40, 0.015)]:
            remaining = deadline - time.monotonic() - 250
            if remaining < 30:
                break
            t_alloc = min(t_md / 3, remaining * 0.4)
            a_md, s_md = mirror_descent(best_seq, t_alloc, lr_init=lr_init,
                                        top_k=top_k, momentum=0.3,
                                        verbose=False, label=f"md_k{top_k}")
            best_seq, best_score = update_best(a_md, best_seq, best_score, f"md_k{top_k}")

    # === Phase 3: More LP-FW on best ===
    print(f"\n=== Phase 3: LP-FW Refinement (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 60:
        t_lp2 = min(remaining * 0.5, 150.0)
        print(f"  LP-FW2 budget={t_lp2:.0f}s")
        a_lp2, s_lp2 = lp_frank_wolfe(best_seq, t_lp2, solver, max_active=60,
                                       verbose=True, label="refine")
        best_seq, best_score = update_best(a_lp2, best_seq, best_score, "lp_refine")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 100
    if remaining > 30:
        t_cd = min(remaining * 0.5, 120.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_cd, verbose=True, label="main")
        best_seq, best_score = update_best(a_cd, best_seq, best_score, "cd_main")

    # === Phase 5: Fine CD Polish ===
    print(f"\n=== Phase 5: Fine CD Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 30
    if remaining > 20:
        t_fine = min(remaining * 0.6, 80.0)
        print(f"  FineCD budget={t_fine:.0f}s")
        a_fine, s_fine = coord_descent(best_seq, t_fine, verbose=False, label="fine", fine=True)
        best_seq, best_score = update_best(a_fine, best_seq, best_score, "fine_cd")

    # === Final MD polish ===
    remaining = deadline - time.monotonic() - 15
    if remaining > 20:
        t_final = min(remaining * 0.6, 60.0)
        a_fin, s_fin = mirror_descent(best_seq, t_final, lr_init=0.01, top_k=30,
                                      momentum=0.2, verbose=False, label="final")
        best_seq, best_score = update_best(a_fin, best_seq, best_score, "final_md")

    # === Final verification ===
    best_seq = np.maximum(best_seq, 0.0)
    s_final = best_seq.sum()
    if s_final > 0:
        best_seq = best_seq / s_final

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        print(f"\nFINAL VERIFIED SCORE: {final_score:.6f} (n={len(best_seq)})")
    else:
        print("\nWARNING: Final verification failed, using internal score")

    return [float(x) for x in best_seq]
