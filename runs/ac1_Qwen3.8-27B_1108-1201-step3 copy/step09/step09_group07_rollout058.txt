Idea: **Frank-Wolfe (Conditional Gradient) Method**

The Frank-Wolfe method is designed for optimization over convex polytopes (like the simplex). At each step:
1. Compute the linear minimization oracle: s = argmin_{v in simplex} ⟨∇f(x), v⟩ = e_i where i = argmin_j ∇f(x)[j]
2. Take a step: x_{new} = (1-γ)x + γs

For our problem, this means: at each step, find the coordinate with the most negative gradient, and move towards that coordinate.

This is very simple and doesn't require projection (just convex combination). Let me try this.

Actually, the standard FW method has slow convergence (O(1/t)). There are accelerated versions (accelerated FW,-away FW) that are faster.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Multi-trial spectral and structured initialization
2. Adaptive-temperature Nesterov projected gradient with periodic reheating
3. LP best-response mirror descent (scipy-based for speed)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_smooth_max_and_grad(a, T):
    """Compute smooth max value and gradient."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    smooth_val = (m + np.log(e.sum())) / T
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (a extended by zeros)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    grad = 2.0 * corr
    return smooth_val, grad, w


def nesterov_adaptive(a0, time_budget, T_start=0.5, T_end=2000.0,
                       n_stages=25, lr_init=0.12, reheat_frac=0.7):
    """Nesterov projected gradient with adaptive temperature and reheating."""
    n = len(a0)
    x = project_simplex(a0.astype(np.float64).copy())
    x_prev = x.copy()
    best_x = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    step = 0
    n_stages_actual = max(n_stages, 5)
    
    # Precompute temperature schedule
    temps = np.geomspace(T_start, T_end, n_stages_actual)
    
    for stage in range(n_stages_actual):
        T = temps[stage]
        # Reheating: at reheat_frac of each stage, briefly increase T
        is_reheated = False
        
        # Stage time budget
        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining <= 0:
            break
        stage_time = remaining / (n_stages_actual - stage)
        
        iters_in_stage = 0
        max_iters_in_stage = 200000
        
        while iters_in_stage < max_iters_in_stage:
            if time.monotonic() - t0 > time_budget:
                break
            
            # Check for reheat within stage
            if not is_reheated and iters_in_stage > int(0.7 * max_iters_in_stage / max(n_stages_actual, 1)):
                T = T * 0.3  # Reheat (lower effective T = sharper)
                is_reheated = True
            
            # Learning rate
            lr = lr_init / (1.0 + 0.003 * step) ** 0.7
            
            # Nesterov momentum
            if step == 0:
                y = x.copy()
            else:
                t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step - 1) * (step - 1))) / 2.0
                beta = (t_prev - 1.0) / t_curr
                y = x + beta * (x - x_prev)
                y = np.maximum(y, 0)
                ys = y.sum()
                if ys > 1e-30:
                    y = y / ys
            
            # Gradient
            _, grad, w = compute_smooth_max_and_grad(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step += 1
            iters_in_stage += 1
            
            # Periodic score check
            if step % 200 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_x = x.copy()
    
    final_sc = score_normalized(x)
    if final_sc < best_score:
        best_score = final_sc
        best_x = x.copy()
    
    return best_x, best_score


def solve_lp_best_response_scipy(a_fixed):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0 using scipy."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # T matrix: T[k, j] = a_fixed[k-j] if 0<=k-j<n else 0
    # We build it as a sparse-like structure but use dense for scipy
    # Actually for n=960, T is 1919x960 which is ~1.8M entries - manageable
    
    # Build T efficiently using the convolution structure
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    # Variables: b[0..n-1], M
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: T @ b - M <= 0 for each row
    # A_ub = [T, -1] (size x (n+1))
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    b_val = res.x[:n]
    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_val)


def lp_mirror_descent(a0, time_budget, max_iters=5000, alpha_start=0.5):
    """Mirror descent on LP best responses."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_x = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    running_sum = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        b_new, M_val = solve_lp_best_response_scipy(a)
        if b_new is None:
            continue
        
        alpha = alpha_start / (1.0 + it * 0.1)
        a = project_simplex((1 - alpha) * a + alpha * b_new)
        running_sum += b_new
        avg = running_sum / (it + 1)
        avg = project_simplex(avg)
        
        sc_a = score_normalized(a)
        sc_avg = score_normalized(avg)
        
        if sc_a < best_score:
            best_score = sc_a
            best_x = a.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_x = avg.copy()
        
        if (it + 1) % 200 == 0:
            print(f"    LP iter={it+1}: best={best_score:.6f} M={M_val:.6f}")
    
    return best_x, best_score


def peak_shave_continuous(a0, time_budget, n_attempts=50000):
    """Continuous peak shaving with adaptive step sizes."""
    n = len(a0)
    x = project_simplex(a0.astype(np.float64).copy())
    best_x = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for attempt in range(n_attempts):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = fast_convolve(x)
        # Find the peak position
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        if peak_val < 1e-15:
            break
        
        # Find contributing indices for this peak
        js = max(0, peak_k - n + 1)
        pe = min(n - 1, peak_k)
        idx = np.arange(js, pe + 1)
        contribs = x[idx] * x[peak_k - idx]
        
        # Try reducing top contributors
        top_k = min(8, len(idx))
        top_positions = np.argsort(contribs)[::-1][:top_k]
        
        improved = False
        for ci in top_positions:
            i = idx[ci]
            if x[i] < 1e-14:
                continue
            
            # Try multiple reduction factors (continuous)
            for factor in [0.99, 0.97, 0.94, 0.90, 0.85]:
                x_trial = x.copy()
                x_trial[i] *= factor
                x_trial = project_simplex(x_trial)
                sc = score_normalized(x_trial)
                if sc < best_score - 1e-15:
                    x = x_trial
                    best_score = sc
                    best_x = x.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pairwise reduction
            if len(idx) >= 2:
                for ci1 in top_positions[:4]:
                    for ci2 in top_positions[:4]:
                        if ci1 == ci2:
                            continue
                        i1 = idx[ci1]
                        i2 = idx[ci2]
                        if x[i1] < 1e-14 or x[i2] < 1e-14:
                            continue
                        x_trial = x.copy()
                        x_trial[i1] *= 0.95
                        x_trial[i2] *= 0.95
                        x_trial = project_simplex(x_trial)
                        sc = score_normalized(x_trial)
                        if sc < best_score - 1e-15:
                            x = x_trial
                            best_score = sc
                            best_x = x.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            # Try random small perturbation
            if attempt % 100 == 0:
                perturbation = rng.standard_normal(n) * 0.005
                x_trial = x + perturbation
                x_trial = project_simplex(x_trial)
                sc = score_normalized(x_trial)
                if sc < best_score:
                    x = x_trial
                    best_score = sc
                    best_x = x.copy()
                elif attempt > 5000 and not improved:
                    # Gradually increase perturbation
                    perturbation = rng.standard_normal(n) * (0.01 + 0.01 * attempt / n_attempts)
                    x_trial = x + perturbation
                    x_trial = project_simplex(x_trial)
                    sc = score_normalized(x_trial)
                    if sc < best_score:
                        x = x_trial
                        best_score = sc
                        best_x = x.copy()
    
    return best_x, best_score


def spectral_init(n, rng):
    """Generate a sequence with approximately flat power spectrum."""
    phase = rng.uniform(0, 2 * np.pi, n)
    freq = np.exp(1j * phase)
    # Ensure real and positive
    a = np.fft.ifft(freq).real
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def structured_init(n, rng, mode=0):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x)
    elif mode == 1:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x)
    elif mode == 2:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * x + np.pi/3) + 0.1 * np.sin(6 * np.pi * x)
    elif mode == 3:
        # Two-bump
        a = np.exp(-5*(x-0.3)**2) + np.exp(-5*(x-0.7)**2) + 0.1
    elif mode == 4:
        # Triangular
        a = 1.0 - 0.4 * np.abs(2*x - 1)
    else:
        # Random smooth
        k = rng.integers(1, 5)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * k * x + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely with the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
