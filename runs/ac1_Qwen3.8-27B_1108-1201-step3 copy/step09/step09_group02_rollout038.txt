"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed objective with adaptive temperature
2. L-BFGS with simplex projection (projected gradient method)
3. Efficient gradient computation exploiting convolution structure
4. Multi-scale optimization (small n -> large n)
5. LP fictitious play for global refinement
6. Basin hopping with structured perturbations
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_softmax_gradient(a, conv, tau, n_peaks=20):
    """
    Compute gradient of 2*n*tau*log(sum_k exp(c_k/tau)) with respect to a.
    
    Uses top-K trick: for small tau, only the top K peaks contribute significantly.
    
    grad_m = 2n * sum_k w_k * d(c_k)/d(a_m)
    where w_k = exp(c_k/tau) / sum_j exp(c_j/tau)
    and d(c_k)/d(a_m) = 2*a_{k-m} (if 0<=k-m<=n-1 and k!=2m) or a_m (if k=2m)
    """
    n = len(a)
    # Get top K peaks
    if n_peaks < 2*n-1:
        top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    else:
        top_idx = np.arange(2*n-1)
    
    c_top = conv[top_idx]
    # Compute softmax weights (shift by max for numerical stability)
    c_shifted = c_top - c_top.max()
    exp_vals = np.exp(c_shifted / tau)
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        if w[idx] < 1e-15:
            continue
        k = int(top_idx[idx])
        wi = 2.0 * w[idx]  # factor of 2 from chain rule
        
        # d(c_k)/d(a_m) = 2*a_{k-m} for m in [max(0,k-n+1), min(n-1,k)]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        m_range = np.arange(m_start, m_end + 1)
        partners = k - m_range  # these are in [0, n-1]
        
        # For each m, d(c_k)/d(a_m) = a_{k-m} + (a_{k-m} if k-m != m else 0)
        # = 2*a_{k-m} if k != 2m, or a_m if k == 2m
        contributions = 2.0 * a[partners]
        # Handle the self-term where k-m = m, i.e., m = k/2
        if k % 2 == 0:
            m_self = k // 2
            if m_start <= m_self <= m_end:
                # contributions[m_self - m_start] should be a[m_self] not 2*a[m_self]
                contributions[m_self - m_start] = a[m_self]
        
        grad[m_range] += wi * contributions
    
    return 2.0 * n * grad

def softmax_lbd(a0, time_limit, tau0=0.01, tau_min=1e-4, verbose=False, label=""):
    """
    L-BFGS-like optimization using soft-max smoothed objective.
    Uses projected gradient with momentum and backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    
    t0 = time.monotonic()
    tau = tau0
    lr = 0.1
    momentum = np.zeros(n)
    beta = 0.9
    it = 0
    no_improve = 0
    n_peaks = min(50, 2*n-1)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 50000:
            break
        
        # Compute gradient using soft-max
        grad = compute_softmax_gradient(a, conv, tau, n_peaks)
        
        # Momentum update
        momentum = beta * momentum + (1-beta) * grad
        
        # Line search with backtracking
        step = lr
        for retry in range(15):
            a_new = a - step * momentum
            a_new = simplex_project(a_new)
            
            conv_new = fast_convolve(a_new)
            score_new = 2.0 * n * float(conv_new.max())
            
            if score_new < best_score * (1 + 1e-10):
                break
            step *= 0.5
        
        if step < 1e-12:
            break
        
        a_new = simplex_project(a - step * momentum)
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
        else:
            a = a_new
            conv = conv_new
            no_improve += 1
        
        # Adapt learning rate and temperature
        if no_improve > 20:
            tau *= 0.7
            lr *= 0.8
            if tau < tau_min:
                break
            no_improve = 0
        elif it % 100 == 0:
            lr *= 0.995
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SLD {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}, tau={tau:.6f}")
    
    return best_a, best_score

                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]

            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n, _nl=n_aff):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 5.0 / n)
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = compute_score(c)
            fd = compute_score(d)
            for _ in range(30):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = compute_score(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = compute_score(d)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    amps[0] = 1.0
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_two_bump(n, rng):
    """Two-bump initialization to reduce center peak of convolution."""
    t = np.arange(n, dtype=np.float64)
    # Two bumps at 1/4 and 3/4
    center1 = n * rng.uniform(0.15, 0.35)
    center2 = n - center1
    width = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - center1) / width) ** 2) + np.exp(-0.5 * ((t - center2) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_asymmetric_exp(n, rng):
    """Exponential decay from one end."""
    alpha = rng.uniform(0.5, 3.0)
    t = np.arange(n, dtype=np.float64)
    if rng.random() < 0.5:
        a = np.exp(-alpha * t / n)
    else:
        a = np.exp(-alpha * (n - 1 - t) / n)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine_bell(n, rng):
    """Cosine bell (raised cosine window)."""
    t = np.linspace(-1, 1, n)
    beta = rng.uniform(1.0, 5.0)
    a = np.cos(np.pi * t / 2) ** (2 * beta)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_alternating(n, rng):
    """Alternating high-low pattern to reduce center convolution peak."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(1.0, 4.0)
    amp = rng.uniform(0.1, 0.4)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=10, verbose=False):
    """Basin hopping: perturb and re-optimize."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 60.0)

        # Random Gaussian perturbation
        sigma = 0.02 + 0.03 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Peak-breaking perturbation
        conv_p = fast_convolve(a)
        top_k = min(5, 2*n-1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 5)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 + rng.standard_normal() * 0.05)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()

    return best_a, best_score

            a_partners = a[partners]

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_new_max = float(conv[k_lo:k_hi+1].max())

            for factor in [0.99, 0.98, 0.95, 1.05, 1.03, 1.01, 1.005]:
                delta = v * (factor - 1.0)

                if x < -1e-15:

                if has_quad:

                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent with top-K peak exploitation (O(K*n) per step)
2. LP fictitious play with geometric (log-space) interpolation for faster convergence
3. Exact coordinate descent with golden section search for precise local optimization
4. Multi-scale strategy: optimize at smaller n, interpolate, refine at larger n
5. Basin hopping with structured peak-breaking perturbations
6. Adaptive temperature for soft-max to balance exploration and exploitation

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_gradient_toppk(a, conv, tau, K):
    """
    Compute gradient of 2*n*tau*log(sum_k exp(c_k/tau)) using top-K peaks.
    Returns gradient vector of length n.
    
    grad_m = 4*n * sum_k w_k * a_{k-m} for 0 <= k-m <= n-1
    where w_k = exp(c_k/tau) / sum_j exp(c_j/tau)
    """
    n = len(a)
    # Get top K peaks
    if K < 2 * n - 1:
        top_idx = np.argpartition(conv, -K)[-K:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    c_top = conv[top_idx]
    c_shifted = c_top - c_top.max()
    exp_vals = np.exp(c_shifted / tau)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        if w[idx] < 1e-15:
            continue
        k = int(top_idx[idx])
        wi = 4.0 * n * w[idx]
        
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        m_range = np.arange(m_start, m_end + 1)
        partners = k - m_range
        
        grad[m_range] += wi * a[partners]
    
    return grad

def softmax_gradient_descent(a0, time_limit, tau0=0.005, tau_min=1e-5, K=30, verbose=False, label=""):
    """
    Projected gradient descent on soft-max smoothed objective.
    Uses momentum, adaptive learning rate, and backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    
    t0 = time.monotonic()
    tau = tau0
    lr = 0.5
    momentum = np.zeros(n)
    beta = 0.85
    it = 0
    no_improve = 0
    K_current = K
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 100000:
            break
        
        # Compute gradient using top-K soft-max
        grad = compute_gradient_toppk(a, conv, tau, K_current)
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        
        # Backtracking line search
        step = lr
        found = False
        for retry in range(10):
            a_try = a - step * momentum
            a_try = simplex_project(a_try)
            
            conv_try = fast_convolve(a_try)
            score_try = 2.0 * n * float(conv_try.max())
            
            if score_try <= best_score * (1.0 + 1e-8) or score_try < 2.0 * n * float(conv.max()) * 0.999:
                found = True
                break
            step *= 0.5
        
        if not found or step < 1e-14:
            # Try a smaller fixed step
            step = max(lr * 0.01, 1e-8)
            a_try = a - step * momentum
            a_try = simplex_project(a_try)
            conv_try = fast_convolve(a_try)
            score_try = 2.0 * n * float(conv_try.max())
        
        a = a_try
        conv = conv_try
        score_new = 2.0 * n * float(conv.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 30:
            tau *= 0.6
            lr *= 0.75
            K_current = min(K_current + 5, min(80, 2*n-1))
            if tau < tau_min:
                tau = tau_min
                break
            no_improve = 0
            beta = min(0.95, beta * 1.02)
        elif it % 500 == 0:
            lr *= 0.98
            # Gradually tighten the focus
            K_current = max(10, K_current - 1)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}, tau={tau:.6f}, K={K_current}")
    
    return best_a, best_score

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]

            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Golden section search
            upper = max(3.0 * v, 5.0 / n)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            # Define score function inline
            def score_fn(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                         _hq=has_quad, _qi=quad_idx, _cs=current_sum, _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)
            
            fc = score_fn(c)
            fd = score_fn(d)
            for _ in range(28):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = score_fn(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = score_fn(d)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

                    best_delta = delta

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    amps[0] = 1.0
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_two_bump(n, rng):
    """Two-bump initialization to reduce center peak of convolution."""
    t = np.arange(n, dtype=np.float64)
    center1 = n * rng.uniform(0.15, 0.35)
    center2 = n * rng.uniform(0.65, 0.85)
    width = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - center1) / width) ** 2) + np.exp(-0.5 * ((t - center2) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_asymmetric_exp(n, rng):
    """Exponential decay from one end."""
    alpha = rng.uniform(0.5, 3.0)
    t = np.arange(n, dtype=np.float64)
    if rng.random() < 0.5:
        a = np.exp(-alpha * t / n)
    else:
        a = np.exp(-alpha * (n - 1 - t) / n)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine_bell(n, rng):
    """Raised cosine bell."""
    t = np.linspace(-1, 1, n)
    beta = rng.uniform(1.0, 5.0)
    a = np.cos(np.pi * t / 2.0) ** (2.0 * beta)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_alternating(n, rng):
    """Alternating pattern to reduce center convolution peak."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(1.0, 5.0)
    amp = rng.uniform(0.1, 0.4)
    phase = rng.uniform(0, 2.0 * np.pi)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_fourier_random(n, rng):
    """Random Fourier synthesis."""
    n_harmonics = rng.integers(5, 20)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_harmonics, n//2), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):
    """
    LP fictitious play with geometric (log-space) interpolation and dual averaging.
    Maintains an averaged iterate for better convergence properties.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()  # Dual averaging accumulator
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            break

        # Geometric interpolation: a_new ~ a^(1-alpha) * b^alpha
        # In log space: log(a_new) = (1-alpha)*log(a) + alpha*log(b)
        best_score_ls = best_score
        best_candidate = None
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            # Log-space interpolation (with epsilon floor for zeros)
            a_safe = np.maximum(a, 1e-15)
            b_safe = np.maximum(b, 1e-15)
            log_a = (1.0 - alpha) * np.log(a_safe) + alpha * np.log(b_safe)
            cand = np.exp(log_a)
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score_ls:
                best_score_ls = sc
                best_candidate = cand

        if best_candidate is not None:
            a = best_candidate
            if best_score_ls < best_score:
                best_score = best_score_ls
                best_a = a.copy()
        else:
            # Fallback: linear interpolation
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        # Update dual average
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        # Check averaged iterate
        cs = a_avg.sum()
        if cs > 1e-10:
            a_avg_norm = a_avg / cs
            sc_avg = 2.0 * n * float(fast_convolve(a_avg_norm).max())
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg_norm.copy()

        if verbose and iters % 5 == 0:
            print(f"    LP-FP it={iters}: best={best_score:.8f}")

    return best_a, best_score

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=8, verbose=False):
    """Basin hopping with structured perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 25:
            break
        t_alloc = min(rem * 0.5, 80.0)

        # Gaussian perturbation
        sigma = 0.015 + 0.025 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Peak-breaking: reduce values at positions contributing to top peaks
        conv_p = fast_convolve(a)
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 4)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 - abs(rng.standard_normal()) * 0.08)
        
        # Boost low-contribution positions
        low_positions = rng.choice(n, size=n // 10, replace=False)
        for ii in low_positions:
            perturbed[ii] *= (1.0 + abs(rng.standard_normal()) * 0.05)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score
