    e = np.exp(z_shift)
    s = e.sum()

    """L-BFGS with log-sum-exp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).astype(np.float64)
    z = z - z.mean()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax(z_in)
            b = fast_autocorr(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax(z_in)
            b = fast_autocorr(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a0.copy()
    best_score, _ = compute_score(best_a)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters = max(20, int(remaining * 8))
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
            a_new = softmax(z)
            sc, _ = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shave_optimize(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent: reduce peak contributors, redistribute."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score, _ = compute_score(best_a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        # Focus on interior peaks (exclude edges which are hard to reduce)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-1] = -np.inf
            b_work[-2] = -np.inf
        peak_k = int(np.argmax(b_work))

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_n]]

        improved = False
        for i in top_positions:
            if a[i] < 1e-12:
                continue
            for factor in [0.98, 0.96, 0.93, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                diff = a[i] - a_trial[i]
                if diff < 1e-16:
                    continue
                # Redistribute to low-contribution positions
                low_mask = np.ones(n, dtype=bool)
                low_mask[top_positions] = False
                low_coords = np.where(low_mask)[0]
                if len(low_coords) > 0:
                    # Distribute proportional to current values (or uniformly for zero entries)
                    weights = a_trial[low_coords] + 1e-10
                    a_trial[low_coords] += diff * weights / weights.sum()
                a_trial = normalize_nonneg(a_trial)
                score = compute_score(a_trial)[0]
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair modification
            found = False
            for ci in range(min(6, len(idx))):
                i = idx[np.argsort(contribs)[::-1][ci]]
                j = peak_k - i
                if i == j or i >= n or j >= n or a[i] < 1e-12 or a[j] < 1e-12:
                    continue
                for f1 in [0.94, 0.89, 0.84]:
                    for f2 in [0.94, 0.89, 0.84]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        # Redistribute
                        diff = a[i] - a_trial[i] + a[j] - a_trial[j]
                        if diff < 1e-16:
                            continue
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others = a_trial[mask]
                        if others.sum() > 1e-15:
                            a_trial[mask] += diff * (others + 1e-10) / (others.sum() + 1e-10 * mask.sum())
                        a_trial = normalize_nonneg(a_trial)
                        score = compute_score(a_trial)[0]
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break
            if not improved:
                no_improve += 1
                if no_improve > 100:
                    break
        else:
            no_improve = 0

    return best_a, best_score


def structured_init(n, mode=0, rng=None):
    """Create structured initial sequences with good autocorrelation properties."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 1:
        # Two-peak at edges
        a = 0.3 + 0.7 * np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 2:
        # Three-peak
        a = 0.2 + 0.4 * np.cos(2 * np.pi * x) ** 2 + 0.4 * np.cos(4 * np.pi * x) ** 2
    elif mode == 3:
        # Edge-concentrated (good for flat convolution)
        a = (x * (1 - x)) ** 0.5 + 0.1
        a = 1.0 / a  # Invert to concentrate at edges
    elif mode == 4:
        # Sinusoidal with high-frequency component
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    elif mode == 5:
        # Logarithmic edge concentration
        eps = 1e-6
        a = -np.log(np.maximum(x, eps) * np.maximum(1 - x, eps) + eps)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return normalize_nonneg(a_interp)


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent, compute initial score
    2. Nesterov subgradient (main, fast iterations)
    3. L-BFGS polish (smooth, multi-temperature)
    4. Peak shaving (coordinate descent)
    5. Repeat 2-4 with perturbations
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n_init = 960
        best_seq = structured_init(n_init, mode=0, rng=rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize_nonneg(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    n_cur = len(best_seq)
    f_lower = 2.0 * n_cur / (2.0 * n_cur - 1.0)

    # === Phase 1: Nesterov subgradient (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_cur}")
        a_nest, _ = nesterov_softmax_optimize(best_seq, t_alloc, eta0=1.0, f_lower=f_lower)
        try_update(a_nest, "nesterov_1")

    # === Phase 2: L-BFGS multi-temperature polish ===
    print(f"\n=== Phase 2: L-BFGS Polish T=50->0.05 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_smooth_optimize(best_seq, t_alloc, T_start=50.0, T_end=0.05, n_stages=12)
        try_update(a_lb, "lbgfs_1")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shave_optimize(best_seq, t_alloc, n_steps=100000)
        try_update(a_ps, "peak_shave_1")

    # === Phase 4: Second Nesterov round (finer) ===
    print(f"\n=== Phase 4: Nesterov Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_nest2, _ = nesterov_softmax_optimize(best_seq, t_alloc, eta0=0.3, f_lower=f_lower)
        try_update(a_nest2, "nesterov_2")

    # === Phase 5: Low-temperature L-BFGS ===
    print(f"\n=== Phase 5: L-BFGS Low-T Polish (best={ best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb2 = lbgfs_smooth_optimize(best_seq, t_alloc, T_start=10.0, T_end=0.02, n_stages=10)
        try_update(a_lb2, "lbgfs_lowT")

    # === Phase 6: Perturbation + re-optimize ===
    print(f"\n=== Phase 6: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = normalize_nonneg(perturbed)
            t_sub = min(rem * 0.3, 120.0)
            if t_sub < 15:
                break
            # Quick Nesterov on perturbed
            t_nest = t_sub * 0.6
            a_p, _ = nesterov_softmax_optimize(perturbed, t_nest, eta0=0.5, f_lower=f_lower)
            try_update(a_p, f"pert_{scale}_nest")
            # Quick L-BFGS
            rem = deadline - time.monotonic()
            if rem > 15 and best_seq is not None:
                t_lbp = min(rem * 0.3, t_sub * 0.35)
                if t_lbp > 10:
                    a_lp = lbgfs_smooth_optimize(best_seq, t_lbp, T_start=20.0, T_end=0.05, n_stages=8)
                    try_update(a_lp, f"pert_{scale}_lb")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    base_n = len(best_seq)
    for delta in [-60, -30, 30, 60, -96, 96, -160, 160]:
        alt_n = base_n + delta
        if 384 <= alt_n <= 3000:
            alt_lengths.append(alt_n)
    # Also try round numbers
    for fn in [768, 896, 1024, 1152]:
        if 384 <= fn <= 3000 and fn != base_n:
            alt_lengths.append(fn)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_total = min(rem * 0.2, 150.0)
        if t_total < 20:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_to_length(best_seq, alt_n)
        # Nesterov on new length
        t_nest_alt = t_total * 0.55
        a_alt, _ = nesterov_softmax_optimize(init_alt, t_nest_alt, eta0=0.8,
                                             f_lower=2.0 * alt_n / (2.0 * alt_n - 1.0))
        try_update(a_alt, f"nex_n{alt_n}")

        # L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_lb_alt = min(rem * 0.3, t_total * 0.4)
            if t_lb_alt > 10:
                a_alt2 = lbgfs_smooth_optimize(best_seq, t_lb_alt, T_start=20.0, T_end=0.03, n_stages=8)
                try_update(a_alt2, f"lb_n{alt_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    f_lower = 2.0 * n_cur / (2.0 * n_cur - 1.0)

    if rem > 30:
        t1 = min(rem * 0.3, 60.0)
        a_f1, _ = nesterov_softmax_optimize(best_seq, t1, eta0=0.2, f_lower=f_lower)
        try_update(a_f1, "final_nest")

    rem = deadline - time.monotonic()
    if rem > 20:
        t2 = min(rem * 0.3, 50.0)
        a_f2 = lbgfs_smooth_optimize(best_seq, t2, T_start=5.0, T_end=0.01, n_stages=8)
        try_update(a_f2, "final_lb")

    rem = deadline - time.monotonic()
    if rem > 15:
        t3 = min(rem * 0.3, 40.0)
        a_f3, _ = peak_shave_optimize(best_seq, t3, n_steps=50000)
        try_update(a_f3, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t4 = min(rem * 0.4, 30.0)
        a_f4, _ = nesterov_softmax_optimize(best_seq, t4, eta0=0.1, f_lower=f_lower)
        try_update(a_f4, "final_nest2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
