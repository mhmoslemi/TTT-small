def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def make_T(a_fixed):
    """Build (2n-1) x n matrix where (T @ b)_k = sum_j a_fixed[j]*b[k-j]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed[j]
