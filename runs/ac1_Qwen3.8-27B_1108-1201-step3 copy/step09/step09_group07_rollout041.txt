def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)

def subgrad_at(a, k):
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    grad = np.zeros(n)
    grad[lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return 2.0 * grad

    b = fast_convolve(a)

    # Weight by peak value for better direction
    weights = b[top]
    weights = weights / weights.sum()
    grad = np.zeros_like(a)
    for k, w in zip(top, weights):

def simulate_annealing(a0, time_limit, T_start=0.01, T_end=0.001):

        b = fast_convolve(a)

            contribs = a_trial[idx] * a_trial[k - idx]

            a_trial[top_i] *= (1 - 0.05 * np.random.exponential(1))

        # Accept

                best_a = a.copy()

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
