```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact LP-based alternating minimization
with adaptive steps, followed by mirror descent and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build T matrix where (T @ b) = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_best_response(a, solver=None):
    """Solve LP to find b minimizing max(conv(a, b)) subject to sum(b)=1, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    if solver is None:
        prob.solve(solver=cp.ECOS, max_iter=50000, abstol=1e-10, reltol=1e-10)
    else:
        prob.solve(solver=solver, max_iter=50000, abstol=1e-10, reltol=1e-10)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val /= s
    return b_val


def fw_best_response(a, n_steps=300):
    """Frank-Wolfe approximation of the best response (faster than LP for large n)."""
    n = len(a)
    b = np.ones(n) / n
    b_conv = convolve_ab(a, b)
    
    for step in range(n_steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad_slice))
        t = 2.0 / (step + 2.0)
        
        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t
    
    return b, b_conv.max()


def lp_alternating(a_init, time_limit, solver=None, verbose=True, use_exact_lp=True, n_fw=200):
    """LP-based alternating minimization."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        prev_score = best_score
        
        # Get best response
        if use_exact_lp:
            b = solve_best_response(a, solver)
            if b is None:
                # Fallback to FW
                b, _ = fw_best_response(a, n_fw)
        else:
            b, _ = fw_best_response(a, n_fw)
        
        # Evaluate b as candidate
        b_score = 2.0 * n * fast_convolve(b).max()
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()
        
        # Adaptive convex combination
        best_alpha = None
        best_alpha_score = best_score
        alphas = [0.5, 0.3, 0.4, 0.2, 0.6, 0.7, 0.1]
        for alpha in alphas:
            a_try = (1 - alpha) * a + alpha * b
            a_try = np.maximum(a_try, 0)
            sn = a_try.sum()
            if sn < 1e-15:
                continue
            a_try = a_try / sn
            s_try = 2.0 * n * fast_convolve(a_try).max()
            if s_try < best_alpha_score:
                best_alpha_score = s_try
                best_alpha = alpha
        
        if best_alpha is not None:
            a = (1 - best_alpha) * a + best_alpha * b
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            a = 0.95 * a + 0.05 * b
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        # Check for stalling
        if best_score < prev_score - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 30:
            # Perturb to escape local minimum
            a = a + 0.01 * np.random.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP alt it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def mirror_descent_refine(a0, time_limit, lr_init=0.05, top_k=50, verbose=True):
    """Mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-9:
                    break
        
        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act < 2 * n - 1:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)
        
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True):
    """Coordinate descent with sensitivity-based steps."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0
    
    # Precompute prefix and suffix max for fast "unaffected max"
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            span = k_end - k_start + 1
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())
            
            a_shifted = a[max(0, k_start - i):k_end - i + 1]
            # a_shifted[j] = a[k_start + j - i] for j=0..span-1
            
            best_delta = 0.0
            best_new_M = M
            
            base_step = a[i]
            for factor in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]:
                delta = base_step * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * (a[i] - delta) + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(old_val + best_delta, 0.0)
                
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (old_val - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def _find_solver():
    """Find a working LP solver."""
    global _cached_solver
    if '_cached_solver' not in globals() or _cached_solver is None:
        _cached_solver = None
        for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _cached_solver = solver_name
                    return solver_name
            except Exception:
                continue
        _cached_solver = cp.ECOS
    return _cached_solver


def cosine_init(n, power=1.0):
    """Cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def geometric_init(n, r=0.95):
    """Geometric decay initialization (symmetric)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = r ** np.abs(t - center)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact LP-based alternating minimization (cvxpy) with adaptive steps
2. FW away-step as fast fallback for best responses
3. Multi-peak mirror descent with adaptive learning rate
4. Sensitivity-based coordinate descent for local polishing
5. Multi-start with diverse initializations and perturbation restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build T matrix where T @ b = conv(a, b). Shape: (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_best_response_lp(a, solver=None):
    """Solve LP: min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    if solver is None:
        prob.solve(solver=cp.ECOS, max_iter=100000, abstol=1e-12, reltol=1e-12)
    else:
        try:
            prob.solve(solver=solver, max_iter=100000, abstol=1e-12, reltol=1e-12)
        except Exception:
            prob.solve(solver=cp.ECOS, max_iter=100000, abstol=1e-12, reltol=1e-12)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val /= s
    return b_val


def fw_best_response_away(a, n_steps=500):
    """FW with away-step for faster convergence to LP optimum."""
    n = len(a)
    b = np.ones(n) / n
    b_conv = convolve_ab(a, b)

    for step in range(n_steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad_slice))
        t = 2.0 / (step + 2.0)

        # Away step
        if step >= 20:
            grad_full = np.zeros(n)
            grad_full[i_start:i_end + 1] = grad_slice
            away_scores = b * grad_full
            i_away = int(np.argmax(away_scores))
            if b[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                t_away = min(t * 0.4, b[i_away] * 0.7)
                ks_s = i_away
                ks_e = min(ks_s + n - 1, 2 * n - 2)
                b_conv[ks_s:ks_e + 1] -= t_away * a
                b[i_away] -= t_away
                b[i_away] = max(b[i_away], 0.0)

        # Standard FW step
        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t

    return b


def lp_alternating(a_init, time_limit, solver=None, verbose=True, use_lp=True, n_fw=400):
    """Alternating minimization with exact LP or FW for inner problem."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        prev_best = best_score

        # Get best response
        if use_lp:
            b = solve_best_response_lp(a, solver)
            if b is None:
                b = fw_best_response_away(a, n_fw)
        else:
            b = fw_best_response_away(a, n_fw)

        # Evaluate b
        b_score = 2.0 * n * fast_convolve(b).max()
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()

        # Adaptive convex combination with line search
        best_alpha = None
        best_alpha_score = best_score
        alphas = [0.5, 0.35, 0.45, 0.25, 0.6, 0.4, 0.2, 0.7]
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * b
            a_try = np.maximum(a_try, 0.0)
            sn = a_try.sum()
            if sn < 1e-15:
                continue
            a_try = a_try / sn
            s_try = 2.0 * n * fast_convolve(a_try).max()
            if s_try < best_alpha_score:
                best_alpha_score = s_try
                best_alpha = alpha

        if best_alpha is not None:
            a = (1.0 - best_alpha) * a + best_alpha * b
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            a = 0.95 * a + 0.05 * b
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Stall detection and perturbation
        if best_score < prev_best - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 25:
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall_count = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-alt it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent_refine(a0, time_limit, lr_init=0.04, top_k=50, verbose=True):
    """Mirror descent (multiplicative weights) with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-10:
                    break

        # Multi-peak subgradient with softmax weighting
        k_act = min(top_k, 2 * n - 1)
        if k_act < 2 * n - 1:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)

        pv = conv[peaks]
        temp = 0.008 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, label=""):

        a = np.ones(n) / n
    else:
        a = a / s

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            span = k_end - k_start + 1

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # a_shifted[j] = a[k_start + j - i] = a[j] for j=0..span-1
            a_shifted = a[0:span]

            best_delta = 0.0
            best_new_M = M
            old_val = a[i]

            for factor in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]:
                delta = old_val * factor
                new_val = old_val + delta
                if new_val < -1e-16:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * old_val + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(old_val + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            span = k_end - k_start + 1

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[0:span]
            best_delta = 0.0
            best_new_M = M
            old_val = a[i]

            for factor in [0.995, 0.99, 0.985, 1.005, 1.01, 1.015]:
                delta = old_val * (factor - 1.0)
                new_val = old_val + delta
                if new_val < -1e-16:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * old_val + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(old_val + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _find_solver():
    """Find a working LP solver."""
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    _cached_solver = None
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver_name
                return solver_name
        except Exception:
            continue
    _cached_solver = cp.ECOS
    return _cached_solver


def cosine_init(n, power=1.0):
    """Cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def geometric_init(n, r=0.95):
    """Symmetric geometric decay."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = r ** np.abs(t - center)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def flat_top_init(n, beta=0.3):
    """Flat-top window (maximally flat in frequency)."""
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    a = (1.0 - beta) * np.cos(np.pi * x) + beta * np.cos(2.0 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _check_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Exact LP alternating minimization with adaptive steps
    2. Diverse initializations with short LP runs
    3. Multi-peak mirror descent
    4. Coordinate descent + fine polishing
    5. Final LP pass
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        n_best = len(best_seq)
    else:
        n_best = 1024
        best_seq = np.ones(n_best) / n_best
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"Starting best: {best_score:.6f} (n={n_best})")

    solver = _find_solver()
    print(f"LP solver: {solver}")

    # === Phase 1: LP Alternating on parent (main optimization) ===
    remaining = deadline - time.monotonic() - 500.0
    if remaining > 60:
        t_alloc = min(remaining * 0.45, 350.0)
        print(f"\n=== Phase 1: LP Alt on parent (budget={t_alloc:.0f}s) ===")
        a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver=solver, use_lp=True, verbose=True)
        best_seq, best_score = _check_update(a_lp, best_seq, best_score, "lp_parent")

    # === Phase 1b: Diverse inits with shorter LP runs ===
    remaining = deadline - time.monotonic() - 400.0
    if remaining > 60:
        inits = [
            cosine_init(n_best, power=1.5),
            cosine_init(n_best, power=2.5),
            geometric_init(n_best, r=0.97),
            flat_top_init(n_best, beta=0.35),
        ]
        # Noisy parent
        noisy = best_seq + 0.03 * rng.standard_normal(n_best)
        noisy = np.maximum(noisy, 0)
        noisy /= noisy.sum()
        inits.append(noisy)

        for idx, init in enumerate(inits):
            remaining = deadline - time.monotonic() - 400.0
            if remaining < 30:
                break
            t_alloc = min(remaining * 0.18, 60.0)
            print(f"\n=== Phase 1b-{idx}: LP Alt diverse (budget={t_alloc:.0f}s) ===")
            a_lp, s_lp = lp_alternating(init, t_alloc, solver=solver, use_lp=True, verbose=False)
            best_seq, best_score = _check_update(a_lp, best_seq, best_score, f"diverse_{idx}")

    # === Phase 2: Mirror Descent ===
    remaining = deadline - time.monotonic() - 300.0
    if remaining > 60:
        t_alloc = min(remaining * 0.4, 200.0)
        print(f"\n=== Phase 2: Mirror Descent (budget={t_alloc:.0f}s) ===")
        a_md, s_md = mirror_descent_refine(best_seq, t_alloc, lr_init=0.03, top_k=60, verbose=True)
        best_seq, best_score = _check_update(a_md, best_seq, best_score, "mirror_descent")

    # === Phase 3: Coordinate Descent ===
    remaining = deadline - time.monotonic() - 250.0
    if remaining > 60:
        t_alloc = min(remaining * 0.5, 180.0)
        print(f"\n=== Phase 3: Coord Descent (budget={t_alloc:.0f}s) ===")
        a_cd, s_cd = coord_descent(best_seq, t_alloc, verbose=True, label="main")
        best_seq, best_score = _check_update(a_cd, best_seq, best_score, "coord_descent")

    # === Phase 4: Second LP pass ===
    remaining = deadline - time.monotonic() - 150.0
    if remaining > 60:
        t_alloc = min(remaining * 0.5, 100.0)
        print(f"\n=== Phase 4: LP Pass 2 (budget={t_alloc:.0f}s) ===")
        a_lp2, s_lp2 = lp_alternating(best_seq, t_alloc, solver=solver, use_lp=True, verbose=True)
        best_seq, best_score = _check_update(a_lp2, best_seq, best_score, "lp_pass2")

    # === Phase 5: Fine Coord Descent ===
    remaining = deadline - time.monotonic() - 60.0
    if remaining > 30:
        t_alloc = min(remaining * 0.6, 80.0)
        print(f"\n=== Phase 5: Fine CD (budget={t_alloc:.0f}s) ===")
        a_fine, s_fine = fine_coord_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = _check_update(a_fine, best_seq, best_score, "fine_cd")

    # === Phase 6: Final short CD ===
    remaining = deadline - time.monotonic() - 20.0
    if remaining > 15:
        t_alloc = min(remaining * 0.7, 40.0)
        a_fin, s_fin = coord_descent(best_seq, t_alloc, verbose=False, label="final")
        best_seq, best_score = _check_update(a_fin, best_seq, best_score, "final")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 0.01:
            best_seq = best_seq / s
            final_score = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(final_score):
                best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
