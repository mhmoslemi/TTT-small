import numpy as np
import cvxpy as cp
import time
from scipy.sparse import diags, csr_matrix
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_rows(a, row_indices):
    """Build specific rows of the convolution matrix T_a.
    Returns a list of (sparse_row, c_k) for each row index in row_indices."""
    n = len(a)
    rows = []
    for k in row_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        cols = np.arange(js, je + 1)
        vals = a[cols]
        row = csr_matrix((vals, (np.zeros(len(cols), dtype=int), cols)), shape=(1, n))
        rows.append(row)
    return rows

def qp_trust_region(a, t0, K_size=30, eta_default=0.01):
    """Solve a QP trust-region step.
    Returns (a_new, improved, M_new)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Find active set: indices where c is close to max
    threshold = c_max - 0.1 * c_max  # top 10% of max
    active = np.where(c >= threshold)[0]
    # Also ensure we include the actual max
    active = np.unique(np.concatenate([active, [np.argmax(c)]]))
    active = active[:K_size]  # limit to K_size
    
    # Build constraints: (T_a)_k @ d <= M - c_k for k in active
    # Constraint matrix: rows are active_k, columns are d
    rows_data = []
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        cols = np.arange(js, je + 1)
        vals = a[cols]
        rows_data.append((vals, cols, k))
    
    if not rows_data:
        return a.copy(), False, c_max
    
    # Build sparse constraint matrix
    all_rows = []
    all_cols = []
    all_vals = []
    for idx, (vals, cols, k) in enumerate(rows_data):
        all_rows.extend([idx] * len(cols))
        all_cols.extend(cols)
        all_vals.extend(vals)
    
    m = len(rows_data)
    T_active = csr_matrix((all_vals, (all_rows, all_cols)), shape=(m, n))
    
    # QP: min M + (1/(2*eta)) * ||d||^2
    # s.t. T_active @ d <= M - c[active_valid]
    #      sum(d) = 0
    #      a + d >= 0
    
    d_var = cp.Variable(n)
    M_var = cp.Variable()
    
    c_active = np.array([c[k] for _, _, k in rows_data])

        T_active @ d_var <= M_var - c_active,
        cp.sum(d_var) == 0,
        d_var >= -a
    ]

def proj_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(a, n_active=20, eta=0.001):
    """One subgradient step with simplex projection."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Active set
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:n_active]
    
    # Subgradient: g_i = 2 * sum_{k in active} a_{k-i}
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    
    # Step
    a_new = a - eta * g
    a_new = proj_simplex(a_new, target_sum=1.0)
    
    c_new = fast_convolve(a_new)
    c_new_max = c_new.max()

    """Alternating LP: alternate between solving for a and b."""
    n = len(a)
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    
    for it in range(n_iters):
        # Solve for b given a: min M s.t. T_a @ b <= M, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        M1 = cp.Variable()
        
        # Build T_a (rows for all 2n-1 convolution positions)
        rows_data = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                rows_data.append((np.array([]), np.array([], dtype=int)))
            else:
                cols = np.arange(js, je + 1)
                vals = a[cols]
                rows_data.append((vals, cols))
        
        all_rows = []
        all_cols = []
        all_vals = []
        for idx, (vals, cols) in enumerate(rows_data):
            if len(cols) > 0:
                all_rows.extend([idx] * len(cols))
                all_cols.extend(cols)
                all_vals.extend(vals)
        
        T_a = csr_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n))
        
        cons_b = [T_a @ b_var <= M1, cp.sum(b_var) == 1]
        prob_b = cp.Problem(cp.Minimize(M1), cons_b)
        try:
            prob_b.solve(solver=solver)
        except Exception:
            break
        if prob_b.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        sb = b_val.sum()
        if sb < 1e-15:
            break
        b_val /= sb
        
        # Solve for a given b: min M s.t. T_b @ a <= M, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        M2 = cp.Variable()
        
        rows_data_a = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                rows_data_a.append((np.array([]), np.array([], dtype=int)))
            else:
                cols = np.arange(js, je + 1)
                vals = b_val[cols]
                rows_data_a.append((vals, cols))
        
        all_rows_a = []
        all_cols_a = []
        all_vals_a = []
        for idx, (vals, cols) in enumerate(rows_data_a):
            if len(cols) > 0:
                all_rows_a.extend([idx] * len(cols))
                all_cols_a.extend(cols)
                all_vals_a.extend(vals)
        
        T_b = csr_matrix((all_vals_a, (all_rows_a, all_cols_a)), shape=(2*n-1, n))
        
        cons_a = [T_b @ a_var <= M2, cp.sum(a_var) == 1]
        prob_a = cp.Problem(cp.Minimize(M2), cons_a)
        try:
            prob_a.solve(solver=solver)
        except Exception:
            break
        if prob_a.status not in ('optimal', 'optimal_inaccurate'):
            break
        a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0)
        sa = a_val.sum()
        if sa < 1e-15:
            break
        a_val /= sa
        
        # Evaluate
        c = fast_convolve(a_val)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a_val.copy()
        
        # Print progress
        if (it + 1) % 50 == 0:
            print(f"    LP alt iter {it+1}: score={2*n*best_score:.6f}")
    
    return best_a, 2 * n * best_score

def local_refine(a, time_limit, n_steps=5000):
    """Local refinement via coordinate descent on peaks."""
    n = len(a)
    best_a = a.copy()
    best_c_max = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Find the peak
        peak_k = np.argmax(c)
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_i = idx[np.argmax(contribs)]
        
        improved = False
        for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
            a_trial = a.copy()
            reduction = a_trial[top_i] * (1 - factor)
            a_trial[top_i] -= reduction
            # Redistribute
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others_sum = a_trial[mask].sum()
            if others_sum > 1e-15:
                a_trial[mask] += reduction * a_trial[mask] / others_sum
            c_trial = fast_convolve(a_trial)
            if c_trial.max() < best_c_max - 1e-15:
                a = a_trial
                best_c_max = c_trial.max()
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try pair reduction
            top2 = np.argsort(contribs)[::-1][:5]
            pair_improved = False
            for ci in top2:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                for f1 in [0.98, 0.95, 0.90]:
                    for f2 in [0.98, 0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial /= st
                        c_trial = fast_convolve(a_trial)
                        if c_trial.max() < best_c_max - 1e-15:
                            a = a_trial
                            best_c_max = c_trial.max()
                            best_a = a.copy()
                            pair_improved = True
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random perturbation
            if step % 200 == 0:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(2, 6)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial /= st
                    c_trial = fast_convolve(a_trial)
                    if c_trial.max() < best_c_max - 1e-15:
                        a = a_trial
                        best_c_max = c_trial.max()
                        best_a = a.copy()
                        improved = True
            else:
                break
    
    return best_a, 2 * n * best_c_max

import cvxpy as cp
import time
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute 2n * max(a*a) / (sum(a))^2"""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)

def proj_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgrad_max_conv(a):
    """Compute a subgradient of max_k (a*a)_k at point a.
    Uses all near-max indices with weights proportional to exp(c_k / tau)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Soft assignment with temperature
    tau = 0.01 * c_max
    weights = np.exp((c - c_max) / tau)
    weights /= weights.sum()
    
    # Gradient: g_i = 2 * sum_k w_k * a_{k-i}
    g = np.zeros(n)
    # Only consider significant weights
    sig_idx = np.where(weights > 1e-4)[0]
    for k in sig_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * weights[k] * a[k - idx]
    
    return g, c_max

def backtracking_subgrad(a, max_iter=5000, time_limit=60.0):
    """Projected subgradient with backtracking line search."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    eta = 0.1
    prev_a = a.copy()
    prev_c_max = fast_convolve(a).max()
    stall = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, c_max = subgrad_max_conv(a)
        
        # Try step
        a_trial = a - eta * g
        a_trial = proj_simplex(a_trial, 1.0)
        
        c_trial = fast_convolve(a_trial).max()

            prev_a = a.copy()
            prev_c_max = c_max
            a = a_trial
            eta = min(eta * 1.1, 1.0)  # increase step slightly
            stall = 0
        else:
            eta *= 0.5
            stall += 1
            if stall > 100:

                a = a + 0.05 * np.random.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 0.1
                stall = 0
        
        sc = score_of(a)
        if sc < best_score:

            best_a = a.copy()
            if (it + 1) % 1000 == 0:
                print(f"    Subgrad iter {it+1}: score={sc:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a hybrid approach combining
QP trust-region steps, p-norm smoothed L-BFGS, LP alternation, and aggressive
local refinement. Key innovations: QP-based second-order method with adaptive
active sets, multi-phase time allocation, and structured local search for final polish.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, line_search


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute score for normalized sequence (sum=1)."""
    return 2.0 * len(a) * fast_convolve(a).max()


def full_score(a):
    """Compute the full score 2*n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)


def proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x >= 0.1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build full convolution matrix T where T @ b = convolve(a, b). Shape: (2n-1, n)."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def qp_step(a, solver, K=20, eta=0.01):
    """QP trust-region step. Returns (a_new, c_max_new, improved)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Active set: top K peaks
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:K]
    
    # Build constraint rows for active peaks
    constr_vals = []
    constr_rows = []
    constr_cols = []
    c_active = []
    for ii, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            constr_vals.append(a[j])
            constr_rows.append(ii)
            constr_cols.append(j)
        c_active.append(c[k])
    
    if not constr_vals:
        return a.copy(), c_max, False
    
    T_act = csr_matrix((constr_vals, (constr_rows, constr_cols)), shape=(len(active), n))
    c_active = np.array(c_active)
    
    # QP: min M + (1/(2*eta)) * ||d||^2
    # s.t. T_act @ d - M <= -c_active
    #      sum(d) = 0
    d_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [
        T_act @ d_var - M_var <= -c_active,
        cp.sum(d_var) == 0
    ]
    
    objective = cp.Minimize(M_var + (1.0 / (2.0 * eta)) * cp.sum_squares(d_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        return a.copy(), c_max, False
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), c_max, False
    
    d_val = np
