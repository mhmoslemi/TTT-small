def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a, b):
    """Compute convolve(a, b) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            return objective_z(z_val, stage_p)

        def jac(z_val):
            return grad_z(z_val, stage_p)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        if stage_p <= p_start or stage_p >= p_end or np.abs(stage_p - temps[0]) < 1 or np.abs(stage_p - temps[-1]) < 1:
            print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

def gradient_descent_smooth(a0, time_limit, n_iters=100000):
    """
    Gradient descent with adaptive step size on the p-smoothed objective.
    Uses the simplex constraint (a >= 0, sum(a) = 1) via Euclidean projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Start with moderate p
    p_schedule = [10, 50, 200, 1000]
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        iters = max(100, 40000 // len(p_schedule))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            if remaining < 5:
                break
            
            # Compute gradient of p-norm smoothed max
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            p_val = c_pos ** p
            S = np.sum(p_val)
            if S < 1e-300:
                break
            factor = S ** (1.0 / p - 1.0)
            w = c_pos ** (p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            grad = 2.0 * factor * corr
            
            # Project gradient onto tangent space of simplex
            grad_tangent = grad - np.mean(grad)
            
            # Adaptive step size (backtracking)
            step_size = 1.0
            for _ in range(20):
                a_trial = a - step_size * grad_tangent
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    step_size *= 0.5
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                current_score = 2.0 * n * fast_convolve(a).max()
                if score_trial < current_score:
                    break
                step_size *= 0.5
            
            a = a - step_size * grad_tangent
            a = np.maximum(a, 0)
            st = a.sum()
            a = a / st
            
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        c = fast_convolve(a)

        active = np.where(c >= max_c - 1e-6)[0]

        grad = np.zeros(n)
        for k in active:

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
            if k % 2 == 0 and k // 2 < n:

        grad /= max(1, len(active))
        
        # Mirror descent update (exponential)
        # a_new_i = a_i * exp(-eta * grad_i) / sum
        log_a = np.log(np.maximum(a, 1e-300)) - eta * grad

3. Improved multi-peak local refinement
4. Gradient descent on smoothed objective with backtracking
5. Multi-length exploration with interpolation
6. Diverse restarts and perturbation strategies
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def objective_z(z, p):
    """2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def grad_z(z, p):
    """Gradient w.r.t. log-domain z of 2n * p-norm of convolve(a,a)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            return objective_z(z_val, stage_p)

        def jac(z_val):
            return grad_z(z_val, stage_p)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def mirror_descent(a0, time_limit, max_iters=100000):
    """
    Multiplicative weights (mirror descent on simplex) for minimizing max convolution.
    Uses direct subgradient of max with exponential updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.5
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        # Get active set (indices where c is within tolerance of max)
        threshold = max_c * (1 - 1e-6)
        active = np.where(c >= threshold)[0]

        # Compute subgradient: (1/|active|) * sum_{k in active} T^T e_k
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
            if k % 2 == 0:
                m_self = k // 2
                if m_self < n:
                    grad[m_self] += a[m_self]
        n_active = max(1, len(active))
        grad /= n_active

        # Exponential (mirror descent) update
        log_shift = -eta * grad
        log_shift -= log_shift.max()  # numerical stability
        a_new = a * np.exp(log_shift)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        # Check improvement periodically
        if it % 50 == 0:
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-10:
                best_score = score
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1

        a = a_new

        # Decay learning rate
        eta = max(0.05, 0.5 / (1.0 + it * 0.005))

        # If stalled, add small perturbation
        if stall > 20:
            rng = np.random.default_rng(it + 12345)
            a = a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MirrorDescent iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def grad_descent_smooth(a0, time_limit, max_iters=50000):
    """
    Gradient descent with backtracking on p-smoothed max over the simplex.
    Projects onto {a >= 0, sum(a) = 1} via Euclidean projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    p_schedule = [10, 50, 200, 1000, 3000]
    steps_per_p = max(200, max_iters // len(p_schedule))

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        step_size = 1.0
        for it in range(steps_per_p):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            p_val = c_pos ** p
            S = np.sum(p_val)
            if S < 1e-300:
                break
            factor = S ** (1.0 / p - 1.0)
            w = c_pos ** (p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            grad = 2.0 * factor * corr

            # Project to tangent space of simplex
            grad_t = grad - np.mean(grad)

            # Backtracking line search
            alpha = step_size
            for _ in range(15):
                a_trial = a - alpha * grad_t
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    alpha *= 0.5
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                score_cur = 2.0 * n * fast_convolve(a).max()
                if score_trial < score_cur - 1e-16:
                    break
                alpha *= 0.5

            a = a - alpha * grad_t
            a = np.maximum(a, 0)
            st = a.sum()
            a = a / st

            step_size = alpha  # Keep track of successful step size

            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def local_refinement_v2(a0, time_limit, n_steps=50000):
    """
    Improved local refinement with multi-peak analysis and diverse strategies.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        improved = False

        # Strategy 1: Reduce top contributor at peak, redistribute
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) > 0:
            contribs = a[idx] * a[peak_k - idx]
            top_i = int(idx[np.argmax(contribs)])

            for frac in [0.03, 0.07, 0.12, 0.20, 0.30]:
                removed = a[top_i] * frac
                if removed < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[top_i] -= removed
                if a_trial[top_i] < 0:
                    a_trial[top_i] = 0
                    removed = a[top_i]
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += removed * a_trial[mask] / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        # Strategy 2: Simultaneous dual-peak reduction
        if not improved and len(b) > 2:
            b_sorted = np.argsort(b)[::-1]
            p1, p2 = int(b_sorted[0]), int(b_sorted[1])
            if p1 != p2 and b[p2] > 0.8 * b[p1]:
                js1, je1 = max(0, p1-n+1), min(n-1, p1)
                js2, je2 = max(0, p2-n+1), min(n-1, p2)
                common = [m for m in range(max(js1, js2), min(je1, je2)+1)
                          if 0 <= p1-m < n and 0 <= p2-m < n]
                if len(common) > 0:
                    scores = np.array([a[m]*a[p1-m] + a[m]*a[p2-m] for m in common])
                    best_m = int(common[np.argmax(scores)])
                    for frac in [0.10, 0.20, 0.30]:
                        removed = a[best_m] * frac
                        if removed < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[best_m] -= removed
                        if a_trial[best_m] < 0:
                            a_trial[best_m] = 0
                            removed = a[best_m]
                        mask = np.ones(n, dtype=bool)
                        mask[best_m] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial[mask] += removed * a_trial[mask] / others_sum
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

        # Strategy 3: Move mass to "safe" positions (don't affect peak)
        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_from = int(idx[np.argmax(contribs)])

            safe = [pos for pos in range(n) if not (0 <= peak_k - pos < n)]
            if len(safe) > 0:
                safe_arr = np.array([a[pos] for pos in safe])
                best_to = int(safe[np.argmax(safe_arr)])
                for frac in [0.05, 0.10, 0.15]:
                    moved = a[top_from] * frac
                    if moved < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[top_from] -= moved
                    a_trial[best_to] += moved
                    if a_trial[top_from] < 0:
                        continue
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 4: Random perturbation
        if not improved and step % 300 == 0:
            rng = np.random.default_rng(step * 31 + 7)
            n_pert = int(rng.integers(2, 8))
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.10, 0.10))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        # Strategy 5: Larger random restart
        if not improved and step % 1000 == 0 and step > 0:
            rng = np.random.default_rng(step * 13 + 3)
            a_trial = a + 0.08 * rng.standard_normal(n)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 5000:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalV2 step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-10)
    elif variant == 6:
        # Super-Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-np.abs(x) ** 4 * 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state with a new candidate."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False

            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.30, 70.0)
            if t_sub < 15:
                break
            a_pert = p_norm_lbfgs_refine(perturbed, t_sub, p_start=20.0, p_end=500.0, n_stages=6)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final polish (very high p + local) ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_fin = min(rem * 0.35, 120.0)
        a_fin = p_norm_lbfgs_refine(best_state['seq'], t_fin, p_start=200.0, p_end=5000.0, n_stages=5)
        try_update(best_state, a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_lr = min(rem * 0.30, 80.0)
        a_lr2, _ = local_refinement_v2(best_state['seq'], t_lr, n_steps=50000)
        try_update(best_state, a_lr2, "final_local")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
