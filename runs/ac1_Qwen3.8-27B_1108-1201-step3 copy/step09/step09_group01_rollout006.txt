def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    # Standard algorithm
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) > 0:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def spectral_shaping(a, alpha=0.5, n_iter=100):
    """Iteratively flatten the power spectrum of a."""
    n = len(a)
    a = a.copy()
    fft_size = 1 << (2*n-1).bit_length()  # enough for linear conv

        # Flatten the rest

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) > 0:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def spectral_shaping(a, time_limit, alpha_init=0.95, alpha_end=0.999, n_sweeps=None):

    Each step: FFT -> compress power spectrum -> IFFT -> project to non-negative simplex.
    """

    if s < 1e-15:

    a = a / s

        P_new[1:] = P_new_non_dc

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.GLpk, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    # Running average
    a_avg = a.copy()
    b_avg = a.copy()

    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Best response: solve LP
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            gamma = 2.0 / (iters + 2.0)
            b_avg = (1.0 - gamma) * b_avg + gamma * b_new

        a_new, M_a = solve_lp_best_response(b_avg, solver)
        if a_new is not None:
            gamma = 2.0 / (iters + 2.0)
            a_avg = (1.0 - gamma) * a_avg + gamma * a_new
            a = a_new

        # Evaluate candidates
        for cand in [a, b_avg, a_avg, 0.5 * (a_avg + b_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

    return best_a, best_score, iters


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_chebyshev_like(n, rng):
    """Initialize with a Chebyshev-like window (known to have low sidelobes)."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    x = (t - center) / center
    # Dolph-Chebyshev-like: T_m(x) pattern
    m = rng.uniform(2, 6)
    a = np.cosh(m * np.arccosh(np.clip(1 / np.maximum(np.abs(x), 1e-10), 0, np.pi))) ** -1
    a = np.maximum(a, 1e-6)
    s = a.sum()
    return a / s


def init_two_slope(n, rng):
    """Initialize with a two-slope (trapezoidal) window."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    half_width = n * rng.uniform(0.3, 0.8)
    a = np.ones(n)
    edge = n * rng.uniform(0.05, 0.25)
    a[:int(edge)] = np.linspace(0.1, 1.0, int(edge))
    a[-int(edge):] = np.linspace(1.0, 0.1, int(edge))
    s = a.sum()
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

    print("\n=== Phase 1: Diverse Init + Fast CD ===")

            t_alloc = max(3.0, min(8.0, t_left / 4.0))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, top-5: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:5]]))

    for s_c, a_c, n_c, it_c in candidates[:10]:
