    b = fast_convolve(a)

    max_Tb = Tb.max()

        js = max(0, k - n + 1)

2. L-BFGS with log-sum-exp smoothing and multi-stage geometric annealing (local)
3. Peak-balancing subgradient method targeting top peaks directly (polish)
4. Frank-Wolfe with away steps (local moves)
5. Multi-length exploration with warm-start interpolation
6. Pair-mixing coordinate descent for final micro-optimization
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with stall detection and restarts."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    restart_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5 and restart_count < 5:
                # Restart with perturbation
                rng_local = np.random.default_rng(777 + restart_count * 131)
                noise_scale = 0.02 + 0.03 * restart_count
                noise = rng_local.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall_count = 0
                restart_count += 1

    return best_a, best_score


def lbgfs_obj(z, T_val, n):
    """L-BFGS objective: smoothed max of convolution in log-domain."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lbgfs_grad(z, T_val, n):
    """L-BFGS gradient in log-domain."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.05, maxiter=10000, n_stages=None):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        maxiter_stage = iters_per_stage
        # Use more iterations at lower temperatures (final stages)
        if si >= n_stages - 3:
            maxiter_stage = int(iters_per_stage * 2.0)

        try:
            res = minimize(
                fun=lbgfs_obj,
                x0=z,
                jac=lbgfs_grad,
                args=(stage_T, n),
                method='L-BFGS-B',
                options={'maxiter': maxiter_stage, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': maxiter_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_balance_refine(a0, time_limit, n_steps=30000, n_peaks=5):
    """Subgradient method targeting top peaks with adaptive step size."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        size = len(b)

        # Find top peaks (exclude edges which always have lower values)
        b_work = b.copy()
        if size > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-2] = -np.inf
            b_work[-1] = -np.inf

        top_k = np.argsort(b_work)[::-1][:n_peaks]

        # Compute combined subgradient for top peaks
        grad_a = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a[k - idx]

        # Project gradient to simplex tangent: subtract mean weighted by a
        S = np.dot(grad_a, a)
        tan_grad = a * (grad_a - S)

        # Adaptive step size (Barzilai-Borwein inspired)
        a_new = a - eta * tan_grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-15:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.1, 0.1)
        else:
            no_improve += 1
            eta *= 0.8
            if eta < 1e-6:
                break

        if no_improve > 200:
            break

    return best_a, best_score


def frank_wolfe_refine(a0, time_limit, n_steps=50000):
    """Frank-Wolfe with away steps."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = a[k_star - i]

        i_toward = int(np.argmin(g))
        best_trial = None
        best_trial_score = best_score

        for t_val in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score_trial = 2.0 * n * fast_convolve(a_trial).max()
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()

        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.3, 0.2, 0.1, 0.05]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        # Random non-negative
        a = rng.random(n)
    elif mode == 7:
        # Smoothed random
        a = rng.random(n)
        kernel = np.exp(-np.linspace(-3, 3, min(n, 31)) ** 2)
        a = np.convolve(a, kernel, mode='same')
    elif mode == 8:
        # Two-hump
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def pair_mix_refine(a0, time_limit, n_steps=20000):
    """Pair-mixing coordinate descent for final micro-optimization."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    rng_local = np.random.default_rng(42)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Identify contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Pick top contributor and a random low contributor
        top_local = np.argsort(contribs)[::-1]
        low_local = np.argsort(contribs)

        improved = False
        for ti in top_local[:5]:
            for li in low_local[:5]:
                if ti == li:
                    continue
                i = idx[ti]
                j = idx[li]
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue

                # Transfer small amount from i to j
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random pair
            i = rng_local.integers(0, n)
            j = rng_local.integers(0, n)
            if i == j or a[i] < 1e-15:
                no_improve += 1
            else:
                for frac in [0.03, 0.07, 0.12]:
                    delta = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 500:
                break
        else:
            no_improve = 0

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=15000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                if contribs[ci] > 0:
                    contrib_count[idx[ci]] += contribs[ci]

        top_coords = np.argsort(contrib_count)[::-1][:15]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                low_coords = np.argsort(contrib_count)[:n // 4]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-10
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                else:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def run_full_pipeline(a_init, time_limit, solver):
    """Run the full optimization pipeline on a given initial sequence."""
    n = len(a_init)
    a = norm_sequence(a_init)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    def try_update(cand, label):
        nonlocal best_a, best_score
        if cand is None:
            return
        cand = norm_sequence(cand)
        sc = 2.0 * n * fast_convolve(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"    [{label}] {sc:.8f}")

    # Phase A: Nesterov LP (global)
    rem = time_limit - (time.monotonic() - t0)
    t_lp = min(rem * 0.35, 400.0)
    if t_lp > 20:
        print(f"  Nesterov LP: budget={t_lp:.0f}s, n={n}")
        a_lp, _ = nesterov_lp_refine(a, t_lp, solver=solver, n_max_iters=100000)
        try_update(a_lp, "nesterov_lp")

    # Phase B: L-BFGS (local)
    rem = time_limit - (time.monotonic() - t0)
    t_lb = min(rem * 0.30, 350.0)
    if t_lb > 20:
        print(f"  L-BFGS: budget={t_lb:.0f}s, T=80->0.03")
        a_lb = lbgfs_refine(best_a, t_lb, T_start=80.0, T_end=0.03, maxiter=12000, n_stages=12)
        try_update(a_lb, "lbgfs_main")

    # Phase C: Peak balance (finer local)
    rem = time_limit - (time.monotonic() - t0)
    t_pb = min(rem * 0.15, 150.0)
    if t_pb > 10:
        print(f"  Peak Balance: budget={t_pb:.0f}s")
        a_pb, _ = peak_balance_refine(best_a, t_pb, n_steps=20000, n_peaks=5)
        try_update(a_pb, "peak_balance")

    # Phase D: L-BFGS low-T polish
    rem = time_limit - (time.monotonic() - t0)
    t_lb2 = min(rem * 0.15, 150.0)
    if t_lb2 > 10:
        print(f"  L-BFGS polish: budget={t_lb2:.0f}s, T=10->0.01")
        a_lb2 = lbgfs_refine(best_a, t_lb2, T_start=10.0, T_end=0.01, maxiter=8000, n_stages=10)
        try_update(a_lb2, "lbgfs_polish")

    # Phase E: Pair mixing (micro)
    rem = time_limit - (time.monotonic() - t0)
    t_pm = min(rem * 0.20, 120.0)
    if t_pm > 10:
        print(f"  Pair Mix: budget={t_pm:.0f}s")
        a_pm, _ = pair_mix_refine(best_a, t_pm, n_steps=30000)
        try_update(a_pm, "pair_mix")

    # Phase F: Final L-BFGS
    rem = time_limit - (time.monotonic() - t0)
    t_fin = min(rem * 0.15, 60.0)
    if t_fin > 5:
        a_fin = lbgfs_refine(best_a, t_fin, T_start=5.0, T_end=0.01, maxiter=5000, n_stages=8)
        try_update(a_fin, "final_lb")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based Nesterov, L-BFGS annealing,
    peak balancing, pair mixing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Main pipeline on the best initial sequence ===
    print(f"\n=== Main Pipeline n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.55, 900.0)
    if t_main > 60:
        a_main, sc_main = run_full_pipeline(best_seq, t_main, solver)
        try_update_global(a_main, "main_pipeline")

    # === Try alternative initializations ===
    print(f"\n=== Alternative Inits (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_modes = [1, 4, 6, 7, 8]
    for mode in alt_modes:
        if rem < 120:
            break
        t_alt = min(rem * 0.12, 200.0)
        if t_alt < 30:
            break
        print(f"  Mode {mode}: budget={t_alt:.0f}s")
        alt_seq = structured_init(n_best, rng, mode=mode)
        a_alt, _ = run_full_pipeline(alt_seq, t_alt, solver)
        try_update_global(a_alt, f"alt_mode{mode}")

    # === Try different lengths ===
    print(f"\n=== Multi-Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0, 0.625, 1.33]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 5120 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [512, 640, 768, 896, 1024, 1280, 1536, 2048]:
        if 256 <= fixed_n <= 5120 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    # Sort by proximity to n_best (prefer similar lengths)
    alt_lengths.sort(key=lambda x: abs(x - n_best))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 150:
            break
        t_total = min(rem * 0.15, 250.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = run_full_pipeline(init_alt, t_total, solver)
        try_update_global(a_alt, f"len_{alt_n}")

    # === Perturbation escape ===
    print(f"\n=== Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for scale in [0.05, 0.10, 0.20]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_pert = min(rem * 0.15, 180.0)
            if t_pert < 30:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)
            print(f"  Scale {scale}: budget={t_pert:.0f}s")
            a_pert, _ = run_full_pipeline(perturbed, t_pert, solver)
            try_update_global(a_pert, f"pert_{scale}")

    # === Final intensive polish on best ===
    print(f"\n=== Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_fin = min(rem * 0.4, 300.0)
        # L-BFGS with very tight temperature
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.005, maxiter=10000, n_stages=12)
        try_update_global(a_fin, "final_lb_tight")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_pm = min(rem * 0.3, 100.0)
        a_pm, _ = pair_mix_refine(best_seq, t_pm, n_steps=50000)
        try_update_global(a_pm, "final_pair")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_pb = min(rem * 0.3, 60.0)
        a_pb, _ = peak_balance_refine(best_seq, t_pb, n_steps=20000, n_peaks=8)
        try_update_global(a_pb, "final_peak")

    # === Ultra-final L-BFGS ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_uf = min(rem * 0.5, 40.0)
        a_uf = lbgfs_refine(best_seq, t_uf, T_start=3.0, T_end=0.003, maxiter=4000, n_stages=6)
        try_update_global(a_uf, "ultra_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
