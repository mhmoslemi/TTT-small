import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)

def hann_window(n, beta=1.0):
    """Hann window raised to power beta."""
    x = np.linspace(0, 1, n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x))
    w = w ** beta
    return norm_sequence(w)


def hamming_window(n, beta=1.0):
    """Hamming window raised to power beta."""
    x = np.linspace(0, 1, n)
    w = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    w = np.maximum(w, 0) ** beta
    return norm_sequence(w)


def blackman_window(n, beta=1.0):
    """Blackman window raised to power beta."""
    x = np.linspace(0, 1, n)
    w = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    w = np.maximum(w, 0) ** beta
    return norm_sequence(w)


def beta_dist_window(n, alpha=1.0):
    """Beta distribution shape: (x*(1-x))^alpha."""
    x = np.linspace(0, 1, n)
    w = (x * (1 - x)) ** alpha
    w[0] = 0
    w[-1] = 0
    return norm_sequence(w)


def kaiser_window(n, beta=2.0):
    """Kaiser window."""
    x = np.linspace(-1, 1, n)
    # Simplified Kaiser: use Bessel function approximation
    # Actually, let's use a simpler approximation
    w = np.exp(-beta * x**2)  # Gaussian approximation
    return norm_sequence(w)


def fourier_window(n, coeffs):
    """Fourier series window: sum of cos(2*pi*k*x) terms."""
    x = np.linspace(0, 1, n)
    w = np.zeros(n)
    for k, c in enumerate(coeffs, start=0):
        w += c * np.cos(2 * np.pi * k * x)
    w = np.maximum(w, 0)
    return norm_sequence(w)


def asymmetric_window(n, shift=0.0, width=1.0):
    """Gaussian-like window with optional shift."""
    x = np.linspace(0, 1, n)
    center = 0.5 + shift
    w = np.exp(-((x - center) / width)**2)
    return norm_sequence(w)


# === Exponentiated Gradient (Mirror Descent) ===

def exponentiated_gradient(a0, time_limit, n_steps=50000, eta_scale=1.0, top_k=5):
    """
    Mirror descent with exponentiated gradient step.
    Uses multi-peak subgradients for smoother optimization.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    # Adaptive step size
    eta = eta_scale / math.sqrt(n)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        
        # Find top peaks (excluding boundaries for stability)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        sorted_idx = np.argsort(b_work)[::-1]
        
        # Use top peaks with exponential weighting
        weights = np.exp(-np.arange(min(top_k, len(sorted_idx))) * 0.3)
        weights /= weights.sum()
        top_indices = sorted_idx[:min(top_k, len(sorted_idx))]
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_indices):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad[i] += w_k * 2.0 * a[k - i]
        
        # Exponentiated gradient update
        # a_new[j] proportional to a[j] * exp(-eta * grad[j])
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        s_new = score(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Reduce step size if stuck
            if no_improve % 100 == 0 and no_improve > 0:
                eta *= 0.5
        
        # Also try a blend with original
        if step % 10 == 0:
            for t in [0.3, 0.5, 0.7]:
                a_blend = (1 - t) * a + t * a_new
                a_blend = norm_sequence(a_blend)
                s_blend = score(a_blend)
                if s_blend < best_score:
                    best_score = s_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    break
    
    return best_a, best_score


# === L-BFGS with Log-Sum-Exp ===

def lbgfs_smooth(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=8):
    """
    L-BFGS optimization using log-sum-exp smoothing of the max function.
    Works in probability simplex via softmax reparametrization.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    def func_and_grad(z_in, T_val):
        a = np.exp(z_in - z_in.max())
        a = a / a.sum()
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        ls = max_Tb + np.log(exp_Tb.sum())
        val = (2.0 * n / T_val) * ls
        
        # Gradient w.r.t. z
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        return val, grad_z
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 5000 // n_stages)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def fun(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            v, _ = func_and_grad(z_in, T)
            return v
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            _, g = func_and_grad(z_in, T)
            return g
        
        try:
            res = minimize(
                fun=fun, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    return a_out / a_out.sum()


# === KKT Analysis ===

def kkt_analysis(a):
    """
    Analyze KKT conditions for the current solution.
    Returns the active set and a measure of optimality gap.
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    active = np.where(b >= M * 0.999)[0]
    
    # Check stationarity: for support of a, the gradient should be constant
    support = np.where(a > 1e-10)[0]
    if len(support) == 0 or len(active) == 0:
        return M, active, 0.0
    
    # Compute the weighted gradient
    # At optimum: sum_k lambda_k * 2*a[k-j] = mu for all j in support
    # This means: for j in support, the "convolution of active set with a" should be constant
    # Simplified check: look at the spread of the gradient over the support
    
    grad_vals = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_vals[idx] += 2.0 * a[k - idx]
    
    grad_on_support = grad_vals[support]
    spread = grad_on_support.std() / max(grad_on_support.mean(), 1e-15)
    
    return M, active, spread


# === Peak Shaving (improved) ===

def peak_shave(a0, time_limit, n_steps=20000):
    """Aggressive peak shaving with redistribution."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        
        # Find contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        # Sort by contribution (descending)
        order = np.argsort(contribs)[::-1]
        
        improved = False
        for ci in order[:20]:
            i = idx[ci]
            j = k_peak - i
            if i == j or i >= n or j >= n:
                continue
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            # Try reducing the larger of a[i], a[j]
            if a[i] >= a[j]:
                reduce_i, keep_i = i, j
            else:
                reduce_i, keep_i = j, i
            
            for factor in [0.95, 0.90, 0.85, 0.80, 0.75]:
                a_trial = a.copy()
                diff = a_trial[reduce_i] * (1 - factor)
                a_trial[reduce_i] -= diff
                
                # Redistribute to positions that don't contribute to this peak
                # Find positions where a[pos] * a[k_peak - pos] is small
                contribs_trial = a_trial[np.arange(js, je+1)] * a_trial[k_peak - np.arange(js, je+1)]
                # Add small contribution from reduce_i
                # Find low-contrib positions
                low_positions = np.where(contribs_trial < np.percentile(contribs_trial, 30))[0]
                if len(low_positions) == 0:
                    low_positions = np.arange(n)
                
                # Redistribute proportionally
                target = a_trial[low_positions] + 1e-12
                a_trial[low_positions] = target + diff * target / target.sum()
                a_trial = norm_sequence(a_trial)
                
                s = score(a_trial)
                if s < best_score - 1e-12:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score


# === Symmetric Optimization ===

def to_symmetric(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return norm_sequence(a_sym)


def symmetric_lbgfs(a0, time_limit, T_start=30.0, T_end=0.3, n_stages=8):
    """L-BFGS that enforces symmetry."""
    n = len(a0)
    a_sym = to_symmetric(a0)
    half_n = (n + 1) // 2
    z_half = np.log(np.maximum(a_sym[:half_n], 1e-300)).copy()
    t0 = time.monotonic()
    
    def expand(z_h):
        a_h = np.exp(z_h - z_h.max())
        a_h = a_h / a_h.sum()
        if n % 2 == 0:
            a_full = np.concatenate([a_h, a_h[::-1]])
        else:
            a_full = np.concatenate([a_h, a_h[-2::-1]])
        return norm_sequence(a_full)
    
    def fun(z_h):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = expand(z_h)
        return score(a)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 3000 // n_stages)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # For symmetric case, just use a wrapper
        def fun_lse(z_h):
            a = expand(z_h)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            ls = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T) * ls
        
        def jac_lse(z_h):
            a = expand(z_h)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            # Project to symmetric part
            grad_sym = 0.5 * (grad_a + grad_a[::-1])
            grad_h = grad_sym[:half_n]
            # Account for softmax
            S = np.dot(grad_h, np.exp(z_h - z_h.max()) / np.exp(z_h - z_h.max()).sum())
            a_h = np.exp(z_h - z_h.max()) / np.exp(z_h - z_h.max()).sum()
            return a_h * (grad_h - S)
        
        try:
            res = minimize(
                fun=fun_lse, x0=z_h, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-10, 'disp': False}
            )
            z_h = res.x
        except Exception:
            pass
    
    return expand(z_h)
