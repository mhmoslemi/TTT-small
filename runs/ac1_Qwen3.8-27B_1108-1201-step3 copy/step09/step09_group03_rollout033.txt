import time

def fast_conv(a):
    """Compute auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_ab(a, b):

    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    return 2 * len(a_norm) * fast_conv(a_norm).max()

One more consideration: the "biased averaging" from Nesterov's acceleration. The idea is:
- After $T$ iterations of accelerated gradient, the average $\bar{x}_T = \sum_{t=1}^T w_t x_t$ (with specific weights $w_t$) has a better bound than any single iterate.
- For running time, we can compute the average incrementally.

The specific scheme (from Nesterov 2018, "Lectures on Convex Optimization"):
- $x_0$ = initial point.
- For $t = 1, 2, \ldots, T$:
  - $\alpha_t = \frac{t+1}{2}$ (or similar)
  - $y_t = \frac{\sum_{s=1}^{t-1} \alpha_s x_s}{\sum_{s=1}^{t-1} \alpha_s}$ (biased average)
  - $g_t = \partial f(y_t)$ (subgradient)
  - $x_t = \Pi_{\mathcal{X}}(x_{t-1} - \frac{2}{L(t+1)} g_t)$ (for smooth case) or mirror step

For non-smooth convex functions on the simplex with mirror descent:
- $x_t = \text{softmax}(-\eta_t g_t + \log(x_{t-1}))$ (normalized)

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_ab(a, b):
    """Linear convolution a*b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv)/sum^2 efficiently."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    an = a / s
    return 2.0 * len(an) * fast_conv(an).max()


def multi_peak_subgradient(a, conv, top_k=30, temp_scale=0.01):
    """
    Compute weighted subgradient using top-k peaks with softmax weights.
    Returns gradient vector g of length n where g[i] represents the 
    subgradient component for coordinate i.
    """
    n = len(a)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n)

    k_act = min(top_k, 2 * n - 1)
    if k_act <= 1:
        peaks = np.array([int(np.argmax(conv))])
        w = np.array([1.0])
    else:
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = temp_scale * M
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

    # Compute subgradient: g[i] = sum_k w[k] * 2 * a[k-i] (if valid)
    g = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # g[i] += w[idx] * 2 * a[k-i]
        # a[k-i] for i in [i_start, i_end] is a[k-i_start : k-i_end+1] reversed
        g[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

    return g


def mirror_descent_accelerated(a0, time_limit, top_k=30, temp_scale=0.008,
                                lr_init=0.04, verbose=True, label=""):
    """
    Mirror descent with Nesterov-style acceleration and biased averaging.
    Uses multi-peak subgradient for better convergence.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    # Biased average tracking
    avg_a = a.copy().astype(np.float64)
    avg_weight = 1.0
    best_avg_score = best_score
    best_avg_a = a.copy()

    # Nesterov momentum
    a_prev = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break

        # Compute multi-peak subgradient
        g = multi_peak_subgradient(a, conv, top_k=top_k, temp_scale=temp_scale)

        # Mirror descent step (multiplicative weight update)
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Nesterov extrapolation
        alpha = 2.0 / (it + 1.5)
        a_momentum = a_new + alpha * (a_new - a_prev)
        a_momentum = np.maximum(a_momentum, 0.0)
        sm = a_momentum.sum()
        if sm > 1e-15:
            a_momentum = a_momentum / sm

        # Use the momentum version for next subgradient evaluation,
        # but track both for best-iterate selection
        a_prev = a
        a = a_momentum
        conv = fast_conv(a)

        # Update biased average
        w_t = it  # linear weights favor recent iterates
        avg_a = (avg_weight * avg_a + w_t * a) / (avg_weight + w_t)
        avg_weight += w_t

        # Check average periodically

    # Final comparison
    if best_avg_score < best_score:

        best_score = best_avg_score
