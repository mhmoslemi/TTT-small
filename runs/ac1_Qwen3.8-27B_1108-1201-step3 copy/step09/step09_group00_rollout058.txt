        try:
            x = cp.Variable(2, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return s
        except:
            pass
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_given_a(a, solver=None):
    if solver is None:
        solver = get_solver()
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_conv_matrix(a)
    cons = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            bv = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
            s = bv.sum()
            if s < 1e-15:
                return None, None
            return bv/s, float(M.value)
    except:
        pass
    # Try fallback solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            b2 = cp.Variable(n, nonneg=True)
            M2 = cp.Variable()
            prob2 = cp.Problem(cp.Minimize(M2), [T @ b2 <= M2, cp.sum(b2) == 1])
            prob2.solve(solver=alt, verbose=False)
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                bv = np.maximum(np.asarray(b2.value, dtype=np.float64), 0)
                s = bv.sum()
                if s > 1e-15:
                    return bv/s, float(M2.value)
        except:
            pass
    return None, None

def eval_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s*s))

def softmax_z(z):
    zs = z - z.max()
    e = np.exp(zs)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def z_of(a):
    return np.log(np.maximum(a, 1e-300))

def lse_obj_val(z, n, T):
    a = softmax_z(z)
    b = fast_convolve(a)
    mb = b.max()
    return (2.0*n/T) * (mb + np.log(np.sum(np.exp(T*(b - mb)))))

def lse_grad_val(z, n, T):
    a = softmax_z(z)
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(T*(b - mb))
    w = e / e.sum()
    # grad_a[m] = 4n * sum_k w[k] * a[k-m]
    # Compute via FFT: 4n * (w * a_rev)[n-1 : n-1+n]
    a_rev = a[::-1]
    conv_size = len(b) + n - 1
    fs = 1 << (conv_size - 1).bit_length()
    W = rfft(w, fs)
    AR = rfft(a_rev, fs)
    conv = irfft(W * AR, fs)
    grad_a = 4.0 * n * conv[n-1 : n-1+n]
    S = np.dot(grad_a, a)
    return a
