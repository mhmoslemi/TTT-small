The Frank-Wolfe method for minimizing a convex function f over a convex set C:
1. Compute gradient g = ∇f(a_t)
2. Find s = argmin_{x in C} g^T x (the farthest-point / linear minimization oracle)
3. Update: a_{t+1} = (1-γ_t) * a_t + γ_t * s

For our problem, C = {a >= 0, sum(a) = 1} (the simplex). The linear minimization oracle is: find the vertex of the simplex that minimizes g^T x, which is just e_j where j = argmin_i g_i.

So the Frank-Wolfe update is:
- Compute gradient g = ∇f(a_t)
- Find j = argmin_i g_i
- Update: a_{t+1} = (1-γ) * a_t + γ * e_j

This is very simple! Each iteration costs O(n) for the gradient (with FFT it's O(n log n)) and O(n) for the update. For n=960, each iteration is ~10 microseconds. We can do millions of iterations!

The issue with vanilla Frank-Wolfe is that it converges slowly (O(1/t)). But **accelerated Frank-Wolfe** (using momentum) converges at O(1/t^2). And **shrinking Frank-Wolfe** (SHRINKCW) can converge linearly under strong convexity.

Let me implement the accelerating Frank-Wolfe (also known as the "lazy" or "strongly separable" version):

```

    s = argmin_{x in C} g^T x  (for simplex: s = e_{argmin g})
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated Frank-Wolfe for fast initial convergence
2. LP Best Response with exact line search (main workhorse)
3. Multi-restart LP with perturbations
4. Coordinate descent peak shaving for final polish
Key innovation: line search in LP alternating method for monotone convergence.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def grad_smooth_max(a, T_param):
    """Gradient of smoothed max conv via log-sum-exp."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T_param * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def afw_optimize(a0, time_limit, T_start=5.0, T_end=200.0, n_stages=8):
    """Accelerated Frank-Wolfe with temperature annealing."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T_param = temps[stage]
        while time.monotonic() - t0 < time_limit:
            # Compute gradient
            g = grad_smooth_max(y, T_param)
            # LMO: pick coordinate with minimum gradient
            j = int(np.argmin(g))
            # Frank-Wolfe step
            gamma = 2.0 / (step_count + 2)
            a_new = (1 - gamma) * a + gamma * np.zeros(n)
            a_new[j] += gamma
            # Nesterov momentum for y
            if step_count > 0:
                beta = (step_count - 2) / (step_count + 1)
                y = a_new + beta * (a_new - a)
            else:
                y = a_new.copy()
            a = a_new
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                if step_count % 1000 == 0:
                    print(f"      AFW step={step_count}: score={best_score:.6f} T={T_param:.1f}")

    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [cp.sum(b_var) == 1]
    # Add convolution constraints
    constraints.append(T @ b_var <= M_var)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def line_search_factor(a, b_star, prev_alpha=0.5, n_trials=16):
    """Find best alpha in [0,1] to minimize score of (1-alpha)*a + alpha*b_star."""
    # Use golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo = 0.0
    hi = 1.0
    
    # Warm start: try near prev_alpha first
    if 0.05 < prev_alpha < 0.95:
        mid = prev_alpha
        f_mid = score_normalized((1 - mid) * a + mid * b_star)
        if f_mid < score_normalized(a):
            # Good move, do golden section in [max(0, mid-0.2), min(1, mid+0.2)]
            lo = max(0.0, mid - 0.3)
            hi = min(1.0, mid + 0.3)
    
    # Golden section search
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    f_c = score_normalized((1 - c) * a + c * b_star)
    f_d = score_normalized((1 - d) * a + d * b_star)
    
    for _ in range(n_trials):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            f_c = score_normalized((1 - c) * a + c * b_star)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + gr * (hi - lo)
            f_d = score_normalized((1 - d) * a + d * b_star)
    
    best_alpha = (lo + hi) / 2.0
    return best_alpha


def lp_line_search_optimize(a0, time_limit, solver=None, max_iters=50000):
    """LP best response with line search."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_alpha = 0.5
    it = 0
    stall_count = 0
    prev_best = best_score

    while it < max_iters and time.monotonic() - t0 < time_limit:
        b_star = solve_lp_best_response(a, solver)
        if b_star is None:
            it += 1
            continue
        
        alpha = line_search_factor(a, b_star, prev_alpha)
        a_new = (1 - alpha) * a + alpha * b_star
        a_new = project_simplex(np.maximum(a_new, 0))
        
        score_new = score_normalized(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
            prev_alpha = alpha
        else:
            stall_count += 1
            # Try smaller step
            for factor in [0.5, 0.3, 0.1]:
                a_try = (1 - alpha * factor) * a + alpha * factor * b_star
                a_try = project_simplex(np.maximum(a_try, 0))
                s_try = score_normalized(a_try)
                if s_try < best_score:
                    best_score = s_try
                    best_a = a_try.copy()
                    a_new = a_try
                    break
            else:
                a_new = a  # no improvement
        
        a = a_new
        it += 1
        
        if it % 100 != 0:
            continue
        
        if best_score < prev_best - 1e-9:
            stall_count = 0
        prev_best = best_score
        
        if stall_count >= 5:
            # Perturb to escape
            rng = np.random.default_rng(it)
            a = a + 0.01 * rng.standard_normal(n)
            a = project_simplex(np.maximum(a, 0))
            stall_count = 0
        
        elapsed = time.monotonic() - t0
        print(f"      LP-LS iter={it} ({elapsed:.0f}s): score={best_score:.6f} alpha={prev_alpha:.3f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=50000):
    """Coordinate descent peak shaving."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    step = 0

    while step < n_steps and time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find the peak location
        peak_k = int(np.argmax(b))
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.94, 0.90]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pairwise reduction
            for ci in top_idx[:3]:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.98, 0.95]:
                    for f2 in [0.98, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Try reducing all top coordinates simultaneously
            a_trial = a.copy()
            for ci in top_idx[:3]:
                i = idx[ci]
                if a[i] > 1e-14:
                    a_trial[i] *= 0.98
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                # Try a small random perturbation
                rng = np.random.default_rng(step)
                a_trial = a + 0.005 * rng.standard_normal(n)
                a_trial = project_simplex(np.maximum(a_trial, 0))
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break
        
        step += 1
        if step % 1000 == 0:
            print(f"      PS step={step}: score={best_score:.6f}")

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence with the actual evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def make_asymmetric_init(n, alpha, rng):
    """Create asymmetric initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-alpha * x) + 0.1
    elif choice == 1:
        a = (x ** alpha) + 0.1
    elif choice == 2:
        a = 1.0 + alpha * np.sin(2 * np.pi * x)
    elif choice == 3:
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.2 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 1.0 + alpha * (2 * x - 1)
    else:
        a = np.exp(-2.0 * (1 - x)) + 0.05
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def kkt_refine(a0, time_limit, solver=None):
    """KKT-based refinement: identify active set and verify optimality."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_val = b.max()
        # Find active set (within 0.1% of max)
        threshold = peak_val * 0.999
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Check KKT: at optimum, 2*(w * a)_i should be constant (for i where a_i > 0)
        # where w is uniform over active set
        w = np.zeros(2 * n - 1)
        for k in active:
            w[k] = 1.0 / len(active)
        
        # Compute 2 * (correlate a with w)
        fft_size = 1 << (n + len(w) - 1).bit_length()
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        corr = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        
        # For KKT, corr[i] should be constant for all i where a[i] > 0
        positive = a > 1e-10
        if positive.sum() < 2:
            break
        corr_pos = corr[positive]
        corr_mean = corr_pos.mean()
        corr_var = corr_pos.std()
        
        # If variance is small, we're close to KKT
        if corr_var / abs(corr_mean) < 1e-4:
            print(f"      KKT: near-optimal (rel var={corr_var/abs(corr_mean):.2e})")
            break
        
        # Use LP to get best response, then update
        b_star = solve_lp_best_response(a, solver)
        if b_star is None:
            break
        
        alpha = line_search_factor(a, b_star, n_trials=10)
        a_new = (1 - alpha) * a + alpha * b_star
        a_new = project_simplex(np.maximum(a_new, 0))
        score = score_normalized(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a, best_score
