"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Anderson-accelerated LP alternating
with adaptive smoothing, L-BFGS polish, and multi-restart exploration.
Key innovation: Anderson acceleration for superlinear convergence, LogSumExp
smoothing for the inner problem, and structured perturbation for global search.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_inner_lp(a_fixed, prev_b=None, solver=None):
    """Solve min_b max_k conv(a_fixed, b) s.t. sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(np.asarray(a_fixed, dtype=np.float64))
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    if prev_b is not None:
        b_var.value = prev_b.astype(np.float64)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
                return b_val, float(M_var.value)
    except Exception:
        pass
    
    for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val /= s
                    return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def lse_best_response(a_fixed, n_inner_steps=50, lr=0.1):
    """Approximate best response using LogSumExp smoothing and gradient descent."""
    n = len(a_fixed)
    b = np.ones(n) / n
    T = build_conv_matrix(a_fixed)
    
    for step in range(n_inner_steps):
        Tb = T @ b
        # LogSumExp gradient
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad = T.T @ w  # gradient of LSE(Tb) w.r.t. b
        b = b - lr * grad
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        else:
            b /= s
        lr *= 0.98  # decay
    
    return b


def project_simplex(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

The fixed-point map is: `F(a) = best_response(a)` where `best_response(a) = argmin_b max_k (a*b)[k]` s.t. `sum(b)=1, b>=0`.

Anderson acceleration:

- At step t+1:

  2. Form the KKT system from the history.
  3. Solve for `alpha`.
  4. Set `a_{t+1} = sum_{i=0}^{m} alpha_i * F(a_{t-i})`.
  5. Project `a_{t+1}` onto the simplex.

In Anderson acceleration for `x = F(x)`:
- We compute `y_{t+1} = sum_{i=0}^{m} alpha_i * F(x_{t-i})` where `alpha` minimizes `||G @ alpha||^2` s.t. `1^T alpha = 1`.
- Then we set `x_{t+1} = y_{t+1}` (or with a damped step: `x_{t+1} = (1-tau)*x_t + tau*y_{t+1}`).

The key is that `y_{t+1}` is a convex combination of the F-values, not the x-values.

Let me re-derive. We have `g(x) = x - F(x)`. We want `g(x) = 0`.

At step t, we want to find `x_{t+1}` such that `g(x_{t+1}) ≈ 0`.

We approximate `g` as affine: `g(x) ≈ g(x_t) + J_t (x - x_t)` where `J_t` is the Jacobian (which we don't have).

Anderson's trick: approximate `g(x_{t+1})` using the history:
`g(x_{t+1}) ≈ g(x_t) + sum_{i=1}^{m} gamma_i [g(x_{t-i+1}) - g(x_{t-i})]`

where `x_{t+1} = F(x_t) + sum_{i=1}^{m} gamma_i [F(x_{t-i+1}) - F(x_{t-i})]`

This is equivalent to: `x_{t+1} = sum_{i=0}^{m} alpha_i F(x_{t-i})` with `alpha_0 = 1 + gamma_1`, `alpha_i = gamma_{i+1} - gamma_i` for `1 <= i <= m-1`, `alpha_m = -gamma_m`, and `sum alpha_i = 1`.

The `gamma` (or equivalently `alpha`) is chosen to minimize `||g(x_{t+1})||^2` (approximately).

Using the approximation `g(x_{t+1}) ≈ G @ alpha` where `G` has columns `g(x_{t-i})`:
`min_alpha ||G @ alpha||^2` s.t. `1^T alpha = 1`

This gives the KKT system I described earlier.

```python
def anderson_step(history_x, history_Fx, m):
    """Compute Anderson acceleration step.
    history_x: list of x values [x_t, x_{t-1}, ..., x_{t-m}] (newest first)
    history_Fx: list of F(x) values corresponding to history_x
    m: memory parameter
    Returns: new x (convex combination of F(x) values)
    """
    # Use last m+1 points
    xs = history_x[:m+1]  # x_t, x_{t-1}, ..., x_{t-m}
    Fxs = history_Fx[:m+1]  # F(x_t), F(x_{t-1}), ..., F(x_{t-m})
    
    # Compute residuals
    gs = [x - Fx for x, Fx in zip(xs, Fxs)]
    
    # G: n x (m+1) matrix with columns g_i
    G = np.column_stack(gs)
    
    # KKT: [G^T G, G^T 1; 1^T G, 0] [alpha; lambda] = [0; 1]
    dim = m + 1
    GtG = G.T @ G  # (m+1) x (m+1)
    Gt1 = np.ones(dim)  # G^T @ 1_vector = sum of rows of G... 
    
    # Wait, G^T @ 1 where 1 is a vector of ones of length n
    # G^T @ ones_n = [sum(g_1), sum(g_2), ..., sum(g_{m+1})]
    # But actually, the constraint is 1^T alpha = 1, i.e., alpha_1 + ... + alpha_{m+1} = 1
    # And we minimize ||G alpha||^2
    # Lagrangian: alpha^T G^T G alpha - 2*lambda*(1^T alpha - 1)
    # Stationarity: 2*G^T G * alpha - 2*lambda*1 = 0
    # So: G^T G * alpha = lambda * 1
    # Constraint: 1^T alpha = 1
    
    # KKT system:
    # [G^T G, 1] [alpha]   [0]
    # [1^T, 0]   [lambda] = [1]
    
    KKT = np.zeros((dim + 1, dim + 1))
    KKT[:dim, :dim] = GtG
    KKT[:dim, dim] = 1.0
    KKT[dim, :dim] = 1.0
    KKT[dim, dim] = 0.0
    
    rhs = np.zeros(dim + 1)
    rhs[dim] = 1.0
    
    # Add regularization for numerical stability
    KKT_reg = KKT + 1e-10 * np.eye(dim + 1)
    KKT_reg[dim, dim] = 0.0  # Keep the (dim,dim) entry as 0
    
    try:
        sol = np.linalg.solve(KKT_reg, rhs)
        alpha = sol[:dim]
    except np.linalg.LinAlgError:
        alpha = np.ones(dim) / dim
    
    # x_{t+1} = sum alpha_i * F(x_{t-i})
    x_new = np.zeros(len(xs[0]))
    for i in range(dim):
        x_new += alpha[i] * Fxs[i]
    
    return x_new

Now, one issue with Anderson acceleration is that it can be unstable, especially with the projection onto the simplex. The damping factor helps, but it can slow down convergence.

Let me also add a "restart" mechanism: if the objective starts increasing, restart from the best point.

Let me also think about the LogSumExp inner problem more carefully. The idea is:
- Instead of solving `min_b max_k (T b)_k` (an LP), solve `min_b (1/T) log(sum_k exp(T (T b)_k))` (a smooth convex problem).
- For large T, this approximates the LP.
- The gradient is `T^T @ softmax(T * (T b))`.
- We can solve this by gradient descent with a decaying step size.

The advantage is that each gradient step is O(n^2) (matrix-vector multiply), which for n=1200 is about 1.4M operations - very fast.

With 50 inner steps per outer iteration, and 100 outer iterations, that's 5000 * 1.4M = 7 billion operations... that might be too slow.

Actually, `T @ b` where T is (2n-1) x n can be computed as a convolution in O(n log n) using FFT. And `T^T @ w` is also a convolution. So each gradient step is O(n log n).

For n=1200, n log n ≈ 1200 * 11 ≈ 13200. With 50 inner steps and 100 outer steps, that's 5000 * 13200 ≈ 66 million operations. That's fast!

Let me implement the LogSumExp inner problem using FFT:

```python
def lse_best_response_fast(a_fixed, n_inner_steps=100, T_param=50.0):
    """Fast LogSumExp best response using FFT for the gradient."""
    n = len(a_fixed)
    b = np.ones(n) / n
    
    for step in range(n_inner_steps):
        # Compute T @ b = conv(a, b)
        Tb = np.convolve(a_fixed, b)  # length 2n-1
        
        # Compute gradient: T^T @ softmax(T * Tb)
        # T^T @ w = cross-correlation(a, w)
        # Specifically, (T^T @ w)[i] = sum_k w[k] * a[k-i] (with bounds)
        # This is the same as conv(w, a_flipped)[n-1-i]... 
        # Actually, let me think about this more carefully.
        # (T @ b)[k] = sum_j a[j] * b[k-j] for valid k,j
        # d/d b[i] of (T @ b)[k] = a[k-i] if 0 <= k-i < n, else 0
        # So (T^T @ w)[i] = sum_k w[k] * a[k-i] for valid k,i
        # = sum_j w[i+j] * a[j] for valid j (substituting k = i+j)
