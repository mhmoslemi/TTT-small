def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.CBC, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(3.0 * v, 4.0 / n)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)

            def _score_at(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            fc = _score_at(c)
            fd = _score_at(d)
            for _ in range(35):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _score_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _score_at(d)

            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def alt_projection(a0, time_limit, n_iter_max=5000, verbose=False, label=""):
    """
    Alternating projection between:
    - Constant modulus in frequency domain (flat spectrum)
    - Non-negative, sum=1 in time domain
    This is the Gerchberg-Saxton / error reduction method adapted for our problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Forward FFT
        A = fft(a, fft_size)

        # Target: constant magnitude
        target_mag = np.full(fft_size, np.sqrt(fft_size / n))
        # Keep phase, replace magnitude
        phases = np.angle(A)
        # For real signal, enforce conjugate symmetry
        A_new = target_mag * np.exp(1j * phases)
        # Enforce conjugate symmetry
        for k in range(1, fft_size // 2):
            A_new[fft_size - k] = np.conj(A_new[k])
        A_new[0] = np.abs(A_new[0])

        # Inverse FFT
        a_new = np.real(ifft(A_new, fft_size)[:n])

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Relaxation: blend with old
        a = 0.5 * a + 0.5 * a_new

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"    AltProj {label} it={it}: score={best_score:.8f}")

    return best_a, best_score

def pair_adjust_descent(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent that adjusts PAIRS of coordinates simultaneously.
    For a peak at lag k, the contribution from pair (j, k-j) is a[j]*a[k-j].
    We adjust pairs to reduce peak contributions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    # Find top peaks
    n_peaks = min(20, 2 * n - 1)
    peak_indices = np.argsort(conv)[::-1][:n_peaks]

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        # Refresh peak list periodically
        if rng.random() < 0.1:
            peak_indices = np.argsort(conv)[::-1][:n_peaks]

        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            pairs = []
            for j in range(j_start, j_end + 1):
                other = k - j
                if j <= other:
                    pairs.append((j, other))

            if not pairs:
                continue

            # For each pair, try to adjust
            for (j, other) in pairs:
                if time.monotonic() - t0 > time_limit:
                    break
                aj = a[j]
                ao = a[other]
                if aj < 1e-14 and ao < 1e-14:
                    continue

                # Try redistributing mass between j and other
                # Keep aj + ao constant
                total = aj + ao
                if total < 1e-14:
                    continue

                # Try different splits
                best_frac = aj / total
                best_local = float(conv[k])

                for frac_try in [0.0, 0.25, 0.5, 0.75, 1.0,
                                 best_frac - 0.1, best_frac + 0.1,
                                 best_frac - 0.3, best_frac + 0.3]:
                    frac_try = max(0.0, min(1.0, frac_try))
                    new_aj = total * frac_try
                    new_ao = total * (1.0 - frac_try)
                    if abs(new_aj - aj) < 1e-14 and abs(new_ao - ao) < 1e-14:
                        continue

                    # Estimate contribution change at lag k
                    old_contrib = aj * ao
                    new_contrib = new_aj * new_ao
                    delta_k = new_contrib - old_contrib

                    # Also check effect on other lags (approximate)
                    # For lag k' involving j: contribution changes by (new_aj - aj) * a[k'-j]
                    # This is complex to track, so just check if it helps at k
                    if delta_k < -1e-14:
                        # Check if it hurts other peaks too much
                        # Quick check: just apply and verify
                        if delta_k < best_local * 0 - 1e-14:  # always true for negative
                            pass
                        best_local = float(conv[k]) + delta_k
                        best_frac = frac_try

                if best_frac != aj / total:
                    new_aj = total * best_frac
                    new_ao = total * (1.0 - best_frac)
                    a[j] = new_aj
                    a[other] = new_ao
                    # Update convolution approximately
                    conv = fast_convolve(a)
                    cur_score = 2.0 * n * float(conv.max())
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if not improved:
            # Perturb to escape
            idx = rng.integers(0, n)
            a[idx] *= (1.0 + rng.uniform(-0.1, 0.1))
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            conv = fast_convolve(a)

    return best_a, best_score

def accel_fictitious_play(a0, time_limit, solver=None, verbose=False):
    """
    Accelerated LP Fictitious Play with:
    - Time-average strategy tracking
    - Adaptive mixing rate
    - Periodic perturbation to escape saddle points
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    a_avg = np.zeros(n)
    iters = 0
    rng = np.random.default_rng(777)
    stuck_count = 0
    last_best = best_score

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        # Solve LP best response
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 10:
                break
            continue
        stuck_count = 0

        # Adaptive mixing rate
        alpha = 2.0 / (iters + 3.0)
        if iters > 50:
            alpha = max(0.05, alpha)

        # Update time average
        a_avg = ((iters - 1) / iters) * a_avg + (1.0 / iters) * b

        # Learn: move towards best response
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        # Check current, time-average, and best response
        for cand, name in [(a, 'cur'), (a_avg, 'avg'), (b, 'br')]:
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        # Periodic perturbation if stuck
        if iters % 30 == 0 and best_score >= last_best - 1e-10:
            stuck_count += 1
            if stuck_count > 3:
                # Add small noise
                pert = a + rng.standard_normal(n) * 0.02
                pert = np.maximum(pert, 0)
                ps = pert.sum()
                if ps > 1e-10:
                    a = pert / ps
                    a_avg = a_avg * 0.5 + a * 0.5  # reset average partially
                stuck_count = 0
                last_best = best_score

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AFWP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    return a / s

def init_fourier_mix(n, rng):
    n_harmonics = rng.integers(5, 20)
    amps = rng.random(n_harmonics) * 0.25
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_directed_perturb(a, conv, rng, strength=0.05):
    """Perturb coordinates that contribute most to the top peaks."""
    n = len(a)
    top_k = min(10, 2 * n - 1)
    peaks = np.argpartition(conv, -top_k)[-top_k:]

    # Compute responsibility of each coordinate
    responsibility = np.zeros(n)
    for k in peaks:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            responsibility[j] += a[j] * a[k - j]

    # Perturb more where responsibility is high
    max_resp = responsibility.max()
    if max_resp < 1e-15:
        return a
    weights = responsibility / max_resp
    perturbation = rng.standard_normal(n) * strength * (0.5 + weights)
    a_new = a * (1.0 + perturbation)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def basin_hopping_peak(a0, time_limit, cd_func, rng, n_rounds=12, verbose=False):
    """Basin hopping with peak-directed perturbations and adaptive strength."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    conv = fast_convolve(a)

    for p in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 80.0)

        # Adaptive strength: increase if stuck
        strength = 0.03 + 0.04 * (p / max(1, n_rounds))
        if best_score > 1.504:
            strength *= 1.5

        # Peak-directed perturbation
        conv_cur = fast_convolve(a)
        perturbed = peak_directed_perturb(a, conv_cur, rng, strength)

        # Also try a random perturbation
        if rng.random() < 0.4:
            pert2 = a + rng.standard_normal(n) * strength * 0.8
            pert2 = np.maximum(pert2, 0)
            s2 = pert2.sum()
            if s2 > 1e-10:
                pert2 = pert2 / s2
            sc2 = compute_score(pert2)
            sc1 = compute_score(perturbed)
            if sc2 < sc1:
                perturbed = pert2

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:
                print(f"    BHP p={p}: improved to {best_score:.8f}")
        else:
            # Keep old a but move slightly towards perturbed
            a = 0.8 * a + 0.2 * perturbed
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s

    return best_a, best_score

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_local_max = float(conv[k_lo:k_hi + 1].max())

            for factor in [0.99, 0.98, 0.96, 0.94, 0.90,
                           1.06, 1.04, 1.02, 1.01, 1.005]:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                global_max = max(unaff_max, local_max)
                ns = current_sum + (x - v)
                sc = 2.0 * n * global_max / (ns * ns)
                if sc < 2.0 * n * best_local_max / (current_sum * current_sum) - 1e-14:
                    best_local_max = global_max
                    best_delta = x - v

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick optimization ===
    print("\n=== Phase 1: Diverse Init + Quick ECD ===")
    t1_start = time.monotonic()

    n_values = [512, 768, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
        for delta in [-128, 128]:
            alt = np_ + delta
            if 256 <= alt <= 2048 and alt not in n_values:
                n_values.append(alt)

    candidates = []
    for n in n_values[:4]:
        if time.monotonic() - t1_start > 200:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat_spec', init_flat_spectrum(n, rng)))
        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))

        for itype, cand in inits[:5]:
            t_left = 200.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(15.0, t_left / 6.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Alternating projection ===
    print(f"\n=== Phase 2: Alt Projection (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 900
        if rem > 30:
            t_alt = min(rem * 0.3, 150.0)
            n_alt = len(best_seq)
            print(f"  AltProj n={n_alt}, budget={t_alt:.0f}s")
            a_ap, s_ap = alt_projection(best_seq, t_alt, verbose=True, label=f"n{n_alt}")
            best_seq, best_score = check_and_update(a_ap, best_seq, best_score, f"altproj_{n_alt}")

            # Also try on a fresh flat-spectrum init
            a_flat = init_flat_spectrum(n_alt, rng)
            a_ap2, s_ap2 = alt_projection(a_flat, t_alt * 0.5, label=f"n{n_alt}_flat")
            best_seq, best_score = check_and_update(a_ap2, best_seq, best_score, f"altproj_flat")

    # === Phase 3: Accelerated LP Fictitious Play ===
    print(f"\n=== Phase 3: Accelerated Fictitious Play (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = list(set([n_best, 1024, 768]))
        fp_lengths = [x for x in fp_lengths if 256 <= x <= 1536][:2]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 500
            if rem < 60:
                break
            t_alloc = min(rem * 0.4, 300.0)
            print(f"  AFWP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp)
            a_fp, score_fp = accel_fictitious_play(initial, t_alloc, solver=solver,
                                                    verbose=True, label=f"n{n_fp}")
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"afwp_{n_fp}")

            # Follow up with exact CD
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                t_cd = min(rem * 0.3, 100.0)
                a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label=f"post_afwp_{n_fp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"ecd_after_afwp_{n_fp}")

    # === Phase 4: Intensive exact coordinate descent ===
    print(f"\n=== Phase 4: Intensive ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(5):
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"ph4_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd}")

            if best_score > 1.5035:
                perturbed = a_opt + rng.standard_normal(n_best) * 0.04
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 5: Peak-directed basin hopping ===
    print(f"\n=== Phase 5: Peak-Directed Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_bh = min(rem * 0.6, 200.0)
            print(f"  BHP budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping_peak(best_seq, t_bh, exact_coord_descent, rng,
                                           n_rounds=10, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "peak_bh")

    # === Phase 6: Pair adjustment + fine polishing ===
    print(f"\n=== Phase 6: Pair Adjust + Fine Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n = len(best_seq)

        # Pair adjustment
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_pair = min(rem * 0.3, 80.0)
            print(f"  PairAdjust budget={t_pair:.0f}s")
            a_pair, s_pair = pair_adjust_descent(best_seq, t_pair, label=f"n{n}")
            best_seq, best_score = check_and_update(a_pair, best_seq, best_score, f"pair_{n}")

        # Fine coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_fine = min(rem * 0.5, 80.0)
            print(f"  FineCD budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd2 = min(rem * 0.6, 50.0)
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, t_cd2, label="polish2")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish2")

    # === Phase 7: Final multi-length check ===
    print(f"\n=== Phase 7: Multi-Length Check (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            n_cur = len(best_seq)
            for alt_n in [n_cur - 64, n_cur + 64, n_cur + 128]:
                if alt_n < 256 or alt_n > 2048:
                    continue
                if time.monotonic() > deadline - 5:
                    break
                a_interp = interpolate_seq(best_seq, alt_n)
                t_short = min(rem * 0.3, 10.0)
                rem = deadline - time.monotonic() - 3
                if rem < 5:
                    break
                a_opt, s_opt, _ = exact_coord_descent(a_interp, max(3.0, t_short),
                                                      label=f"ml_{alt_n}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ml_{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
