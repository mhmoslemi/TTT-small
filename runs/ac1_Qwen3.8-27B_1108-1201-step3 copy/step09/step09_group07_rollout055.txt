"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best response
iterations with Nesterov acceleration, analytical initialization,
and local refinement.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.CVXOPT]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.SCS


def make_init(n, mode='cosine'):

def nesterov_best_response(a0, time_limit, solver, n_max_iters=100000):
    """
    Nesterov-accelerated best response iteration:
    Alternate solving LP best responses, using Nesterov averaging.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    x = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response for current x
        b_new, M_val = solve_best_response(x, solver)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        x = (1 - alpha) * x + alpha * b_new
        x = project_simplex(x)
        
        # Nesterov average
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)
        
        # Check scores
        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_x = score_normalized(x)
        if score_x < best_score:
            best_score = score_x
            best_a = x.copy()
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"        BR iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    for j in range(n):
        T[j:j + n, j] = a
    return T
