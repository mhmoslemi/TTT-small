    n = len(arr)

    prefix[0] = -np.inf

def compute_suffix_max(arr):

    n = len(arr)

    suffix[n] = -np.inf

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse asymmetric initialization (ramp, exponential, sine, sparse patterns)
2. Fast coordinate descent with sensitivity-based ordering
3. Multi-length exploration (n = 1024, 2048, 4096)
4. Fine-grained polishing with adaptive step sizes
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_conv(a, conv):
    n = len(a)
    return 2.0 * n * conv.max()

def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def init_linear_ramp(n, rng):
    a = np.arange(1, n+1, dtype=np.float64)
    a = a / a.sum()
    return a

def init_exp_decay(n, rng, direction='right'):
    alpha = rng.uniform(0.5, 3.0)
    if direction == 'right':
        a = np.exp(-alpha * np.arange(n) / n)
    else:
        a = np.exp(-alpha * (n-1-np.arange(n)) / n)
    a = a / a.sum()
    return a

def init_sine(n, rng):
    phase = rng.uniform(0, 2*np.pi)
    freq = rng.uniform(0.3, 1.5)
    a = 0.5 + 0.5 * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)/n
    else:
        a = a / s
    return a

def init_two_bump(n, rng):
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5*((t-c1)/w1)**2) + np.exp(-0.5*((t-c2)/w2)**2)
    a = a / a.sum()
    return a

def init_sparse_pattern(n, rng, density=0.7):
    mask = rng.random(n) < density
    a = np.where(mask, 1.0, 0.0)
    s = a.sum()
    if s < 1:
        a = np.ones(n)/n
    else:
        a = a / s
    return a

def init_from_parent(parent, rng, noise_level=0.1):
    n = len(parent)
    a = parent + rng.standard_normal(n) * noise_level
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return parent.copy()
    return a / s

def coord_descent(a0, time_limit, verbose=False, label=""):
    """Fast coordinate descent for minimizing max(conv(a,a)) with sum(a)=1, a>=0."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    # Factors for step sizes
    coarse_factors = [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]
    fine_factors = [0.95, 0.98, 1.02, 1.05]
    
    order = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Use fine factors after some sweeps
        if sweeps > 100:
            factors = fine_factors
        else:
            factors = coarse_factors
        
        for i in order:
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            aff_len = k_end - k_start + 1
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                um = conv[:k_start].max()
                if um > unaff_max:
                    unaff_max = um
            if k_end < 2 * n - 2:
                um = conv[k_end+1:].max()
                if um > unaff_max:
                    unaff_max = um
            
            # a[k-i] for k in [k_start, k_end]
            a_shifted = a[k_start - i : k_end - i + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                # New affected values
                new_affected = conv[k_start:k_end+1] + delta * (a_shifted + a[i])
                # Fix quadratic term at k=2i
                if 2*i >= k_start and 2*i <= k_end:
                    idx = 2*i - k_start
                    new_affected[idx] += delta * delta
                
                nm = new_affected.max()
                if nm < unaff_max:
                    nm = unaff_max
                
                if nm < best_new_M:
                    best_new_M = nm
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                conv[k_start:k_end+1] += best_delta * (a_shifted + a[i])
                if 2*i >= k_start and 2*i <= k_end:
                    idx = 2*i - k_start
                    conv[2*i] += best_delta * best_delta
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if sweeps % 50 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.8f}")
        
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
        
        # Re-randomize order every 50 sweeps
        if sweeps % 50 == 0:
            np.random.shuffle(order)
    
    return best_a, best_score
