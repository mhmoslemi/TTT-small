import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def accelerated_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent with Nesterov momentum and adaptive multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Nesterov momentum
    y = a.copy()
    conv_y = fast_convolve(y)
    
    best_a = a.copy()
    best_score = 2.0 * n * conv_y.max()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.1
    mu = 0.9  # momentum parameter
    no_improve = 0
    temp = 0.01  # softmax temperature
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y
        M = conv_y.max()
        
        # Multi-peak subgradient
        k_act = min(50, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * y[k - i_end:k - i_start + 1][::-1]
        
        # Mirror descent step on y
        y_new = y * np.exp(-lr * sg)
        y_new = np.maximum(y_new, 0.0)
        sn = y_new.sum()
        if sn < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / sn
        
        # Nesterov momentum update
        beta = (1 + np.sqrt(1 + 4 * mu * mu * it * it)) / 2
        # Actually, standard Nesterov: t_{k+1} = (1+sqrt(1+4t_k^2))/2
        # y_{k+1} = a_{k+1} + (t_k - 1)/t_{k+1} * (a_{k+1} - a_k)
        # Let me use a simpler momentum formulation
        
        # Simple heavy ball momentum:
        a_new = (1 - mu) * y_new + mu * (y_new + (y_new - a))
        a_new = np.maximum(a_new, 0.0)
        sn2 = a_new.sum()
        if sn2 < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn2
        
        a = a_new
        y = y_new  # Use the MD step for the gradient evaluation point
        conv_y = fast_convolve(y)
        
        # Evaluate
        score = 2.0 * n * conv_y.max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr = max(lr * 0.85, 0.001)
            temp = max(temp * 0.7, 0.001)
            no_improve = 0
        elif no_improve > 100:
            lr = max(lr * 0.95, 0.001)
        
        # Periodic shake
        if it % 500 == 0 and no_improve > 50:
            shake = 0.02 * np.ones(n)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    best_seq, best_score = try_structural_tricks(best_seq, best_score, best_seq)

    # === Phase 4: Second round MD + CD (remaining time) ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 60:
        print(f"\n=== Phase 4: Second pass (best={best_score:.8f}) ===")
        t4a = remaining * 0.35
        a_md2, _ = mirror_descent_adaptive(best_seq, t4a, verbose=False)
        score_md2 = float(evaluate_sequence(a_md2.tolist()))
        if score_md2 < best_score:
            best_score = score_md2
            best_seq = a_md2.copy()
            print(f"  MD2: {best_score:.8f}")

        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 30:
            a_cd2, _ = coordinate_descent(best_seq, remaining2, verbose=False, fine=True, label="pass2")
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if score_cd2 < best_score:
                best_score = score_cd2
                best_seq = a_cd2.copy()
                print(f"  CD2: {best_score:.8f}")

    # === Final: Very fine polish with whatever time remains ===
    remaining = deadline - time.monotonic() - 5
    if remaining > 15:
        a_polish, _ = coordinate_descent(best_seq, remaining, verbose=False, fine=True, label="polish")
        score_polish = float(evaluate_sequence(a_polish.tolist()))
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()

    # Final structural tricks
    best_seq, best_score = try_structural_tricks(best_seq, best_score, best_seq)

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s = best_seq.sum()
    if s < 0.01:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
