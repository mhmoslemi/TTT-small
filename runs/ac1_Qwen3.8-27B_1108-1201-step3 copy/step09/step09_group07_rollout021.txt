def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def nesterov_projected_gradient(a0, time_limit, T_start=3.0, T_end=300.0,
                                 lr_init=0.06, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with smooth max."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)

            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev_calc = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
                beta = (t_prev_calc - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            # Gradient of smooth max
            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            e = np.exp(Tb - m)
            w = e / e.sum()
            fft_size = 1 << (n + len(w) - 1).bit_length()
            A = rfft(y, fft_size)
            W = rfft(w, fft_size)
            corr = irfft(np.conj(W) * A, fft_size)[:n]
            grad = 2.0 * corr

            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    final_sc = score_normalized(x)
    if final_sc < best_score:
        best_score = final_sc
        best_a = x.copy()

    return best_a, best_score


def lp_alternate_refine(a0, time_limit, solver=None, n_max_iters=200000,
                        peak_target_freq=200, peak_target_n=5, peak_reduction_factor=0.02):
    """
    LP alternating refinement with peak-targeted subproblems.
    Uses adaptive Nesterov step and dual-informed restarts.
    """

    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine if we should use peak-targeted LP
        use_peak_target = (it > 500 and (it % peak_target_freq == 0))

        if use_peak_target:
            # Find top peaks
            b_curr = fast_convolve(a)
            peak_vals, peak_lags_arr = np.sort(b_curr)[::-1], np.argsort(b_curr)[::-1]
            top_peaks = peak_lags_arr[:peak_target_n].tolist()
            # Peak reduction: small fraction of current peak
            peak_reduction = peak_reduction_factor * peak_vals[0]
            b_new, M_b, _ = solve_lp_best_response(y, solver,
                                                     peak_lags=top_peaks,
                                                     peak_reduction=peak_reduction)
        else:
            b_new, M_b, _ = solve_lp_best_response(y, solver)

        if b_new is None:
            stall_count += 1
            if stall_count >= 10:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                y = y + 0.01 * rng.standard_normal(n)
                y = project_simplex(y)
                a = a + 0.005 * rng.standard_normal(n)
                a = project_simplex(a)
                stall_count = 0
            continue

        # Adaptive step size
        if it < 100:
            alpha = 0.15
        elif it < 500:
            alpha = 0.08
        elif it < 2000:
            alpha = 0.04
        else:
            alpha = max(0.02, 2.0 / (it + 2))

        # Nesterov-type update
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        # Momentum-weighted primal update
        momentum = min(1.0, (it + 1) / 1000.0)
        a = (1 - momentum * alpha * 0.5) * a + momentum * alpha * 0.5 * b_new
        a = project_simplex(a)

        # Evaluate
        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        score_y = score_normalized(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Stagnation check and restart
        if it % 300 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                rng = np.random.default_rng(it)
                # Structured perturbation: add asymmetric noise
                x_grid = np.linspace(0, 1, n)
                noise = 0.02 * np.sin(2 * np.pi * rng.uniform(1, 8) * x_grid + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = project_simplex(y)
                a = a + 0.5 * noise
                a = project_simplex(a)
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_surgery(a0, time_limit, n_steps=50000):
    """
    Direct peak surgery: for the current peak lag, move mass from the smaller
    to the larger element in peak-contributing pairs to reduce the product.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_count > 5000:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        if peak_val <= 0:
            break

        # Find contributing pairs for peak lag
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        paired_idx = peak_k - idx
        contribs = a[idx] * a[paired_idx]

        # Sort by contribution (largest first)
        order = np.argsort(contribs)[::-1]

        improved = False
        for oi in order[:20]:
            i = idx[oi]
            j = paired_idx[oi]
            if i == j:
                continue
            if i >= n or j >= n or i < 0 or j < 0:
                continue

            ai = a[i]
            aj = a[j]
            if ai < 1e-14 or aj < 1e-14:
                continue

            # Move mass from smaller to larger to reduce product
            if ai > aj:
                # Move from j to i (increase the larger, decrease the smaller)
                source = j
                target = i
            else:
                source = i
                target = j

            # Try different amounts of transfer
            max_transfer = min(a[source], 0.3 * a[source])
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = frac * max_transfer
                if delta < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[source] -= delta
                a_trial[target] += delta
                if a_trial[source] < 0 or a_trial[target] < 0:
                    continue
                # Check sum is preserved (it should be by construction)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()

                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                no_improve_count = 0
                break

        if not improved:
            # Try reducing the peak element directly
            a_trial = a.copy()
            a_trial[peak_k % n] *= 0.95  # crude: reduce element at peak position
            # Better: reduce the top contributor
            if len(order) > 0:
                a_trial = a.copy()
                i_top = idx[order[0]]
                a_trial[i_top] *= 0.97
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try pair reduction at second peak
            b_sorted = np.sort(b)[::-1]
            if len(b_sorted) > 1 and b_sorted[1] > 0.5 * b_sorted[0]:
                peak_k2 = int(np.argsort(b)[::-1][1])
                js2 = max(0, peak_k2 - n + 1)
                je2 = min(n - 1, peak_k2)
                idx2 = np.arange(js2, je2 + 1)
                paired2 = peak_k2 - idx2
                contribs2 = a[idx2] * a[paired2]
                order2 = np.argsort(contribs2)[::-1]
                for oi in order2[:10]:
                    i2 = idx2[oi]
                    j2 = paired2[oi]
                    if i2 == j2 or i2 >= n or j2 >= n:
                        continue
                    if a[i2] < 1e-14 or a[j2] < 1e-14:
                        continue
                    if a[i2] > a[j2]:
                        source2, target2 = j2, i2
                    else:
                        source2, target2 = i2, j2
                    for frac in [0.05, 0.1, 0.15]:
                        delta2 = frac * a[source2]
                        a_trial = a.copy()
                        a_trial[source2] -= delta2
                        a_trial[target2] += delta2
                        if a_trial[source2] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            no_improve_count += 1

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakSurgery step={step+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def make_asymmetric_init(n, alpha, rng):
    """Create asymmetric initial sequence with strong asymmetry."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + alpha * np.sin(2 * np.pi * x)
    elif choice == 1:
        a = 1.0 + alpha * (2 * x - 1)
    elif choice == 2:
        a = 1.0 + alpha * np.sin(2 * np.pi * x + 0.3)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 1.0 + alpha * np.sin(4 * np.pi * x)
    elif choice == 5:
        # Strongly asymmetric: ramp
        a = 0.3 + 0.7 * x
    elif choice == 6:
        # Two-peak asymmetric
        a = np.exp(-6.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.8) ** 2)
    else:
        # Random smooth
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + alpha * 0.5 * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):

        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP alternating with peak targeting,
    Nesterov projected gradient, peak surgery, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_asymmetric_init(n_init, 0.7, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            n_best = len(a)
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov PG for quick improvement ===
    print(f"\n=== Phase 1: Nesterov PG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_projected_gradient(best_seq, t_alloc, T_start=3.0, T_end=400.0,
                                               lr_init=0.06, n_stages=8, steps_per_stage=2000)
        try_update(a_ng, "ng_main")

    # === Phase 2: Main LP alternating with peak targeting ===
    print(f"\n=== Phase 2: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternate_refine(best_seq, t_alloc, solver=solver,
                                       n_max_iters=100000, peak_target_freq=150,
                                       peak_target_n=5, peak_reduction_factor=0.03)
        try_update(a_lp, "lp_main_1")

    # === Phase 3: Peak surgery ===
    print(f"\n=== Phase 3: Peak Surgery (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        a_ps, _ = peak_surgery(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_surgery_1")

    # === Phase 4: Second LP round (different parameters) ===
    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        a_lp2, _ = lp_alternate_refine(best_seq, t_alloc, solver=solver,
                                        n_max_iters=80000, peak_target_freq=100,
                                        peak_target_n=8, peak_reduction_factor=0.05)
        try_update(a_lp2, "lp_round_2")

    # === Phase 5: Another peak surgery ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ps2, _ = peak_surgery(best_seq, t_alloc, n_steps=20000)
        try_update(a_ps2, "peak_surgery_2")

    # === Phase 6: Multi-start with different inits at same n ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for attempt in range(3):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 200.0)
            if t_sub < 30:
                break
            alpha = 0.5 + 0.3 * attempt
            a_init = make_asymmetric_init(n_best, alpha, rng)
            print(f"  MultiStart {attempt}: alpha={alpha:.1f}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_projected_gradient(a_init, t_sub * 0.5,
                                                    T_start=2.0, T_end=300.0,
                                                    lr_init=0.08, n_stages=6,
                                                    steps_per_stage=1500)
            try_update(a_sub, f"ms_{attempt}_ng")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = lp_alternate_refine(a_sub, min(rem * 0.3, 60.0),
                                                 solver=solver, n_max_iters=20000,
                                                 peak_target_freq=200)
                try_update(a_sub2, f"ms_{attempt}_lp")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-30, -20, -10, 10, 20, 30, 50, 70]:
        alt_n = n_best + delta
        if 200 <= alt_n <= 3000:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:5]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.20, 150.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_projected_gradient(init_alt, t_total * 0.5,
                                                T_start=3.0, T_end=300.0,
                                                lr_init=0.07, n_stages=5,
                                                steps_per_stage=1000)
        try_update(a_alt, f"alt_ng_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = lp_alternate_refine(a_alt, min(rem * 0.3, 40.0),
                                             solver=solver, n_max_iters=15000)
            try_update(a_alt2, f"alt_lp_n{alt_n}")

    # === Phase 8: Final LP polish + peak surgery ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        a_fin, _ = lp_alternate_refine(best_seq, t_fin, solver=solver,
                                        n_max_iters=30000, peak_target_freq=100,
                                        peak_target_n=3, peak_reduction_factor=0.02)
        try_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_ps_fin, _ = peak_surgery(best_seq, min(rem * 0.5, 30.0), n_steps=10000)
        try_update(a_ps_fin, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"  Verified final score: {best_score:.6f}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
