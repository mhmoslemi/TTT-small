        if np.isnan(x) or np.isinf(x):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def mirror_descent_multi_peak(a0, time_limit, lr_init=0.05, top_k=20, verbose=True):
    """
    Mirror descent (multiplicative weight update) using multi-peak subgradient.
    Key improvement: uses top-k peaks with softmax weighting to break symmetry.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break

        # Compute multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        # Get top-k peak indices
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient: sg[i] = sum_k w[k] * 2 * a[k-i] (if valid)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # sg[i] += w[idx] * 2 * a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] = a[k-i_start : k-i_end+1][::-1]
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Also track running average (explained by the convergence theory)
        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        conv = fast_convolve(a)

        # Periodically check the running average
        if it % 100 == 0:
            avg_score = 2.0 * n * fast_convolve(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, n_trials=10, verbose=True, label=""):
    """
    Fast coordinate descent with O(n) sensitivity analysis per coordinate.
    Uses adaptive step sizes based on local curvature estimation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute prefix/suffix maxima for fast "unaffected max" queries
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # The "affected" range: conv[k] changes when a[i] changes
            # conv[k] += 2*delta*a[k-i] (approximately, for k != 2i)
            a_shifted = a[k_start - i:k_end - i + 1]  # a[k-i] for k in [k_start, k_end]

            best_delta = 0.0
            best_new_M = M

            # Adaptive step sizes based on current value
            base_step = a[i]
            factors = [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]

            for factor in factors:
                delta = base_step * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # Compute new affected values
                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                # Fix the quadratic term at k = 2i if in range
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score

def periodic_init(n, rng):
    """Initialize with a periodic pattern (alternating high/low)."""
    period = int(rng.integers(2, 8))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_window_init(n, rng):
    """Initialize with a cosine raised-to-power window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.5, 3.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_hump_init(n, rng):
    """Initialize with two peaks (non-palindromic)."""
    t = np.arange(n, dtype=np.float64)
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width = n * rng.uniform(0.1, 0.25)
    a = np.exp(-0.5 * ((t - center1) / width) ** 2) + np.exp(-0.5 * ((t - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
