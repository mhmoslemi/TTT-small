```python
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = [1.0]
    
    # Inequalities: T@x - M <= 0  and  -x <= 0
    A_ub = np.zeros((m + n, n+1))
    A_ub[:m, :n] = T
    A_ub[:m, n] = -1  # -M
    b_ub = np.zeros(m)
    
    A_ub[m:, :n] = -np.eye(n)  # -x <= 0
    # b_ub[m:] = 0 (already set)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if res.success:
        x = res.x[:n]
        M = res.x[n]
        x = np.maximum(x, 0)
        s = x.sum()
        if s > 0:
            x = x / s
        return x, M
    return None, None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response alternating with Nesterov momentum (scipy HiGHS)
2. Symmetric game formulation: both players solve the same type of LP
3. Log-sum-exp gradient refinement with multi-stage annealing

5. Multi-length exploration and adaptive re-tuning

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_best_response(fixed_seq):
    """Solve min M s.t. T@x <= M, sum(x)=1, x>=0 using HiGHS LP."""
    n = len(fixed_seq)
    m = 2 * n - 1
    T = build_conv_matrix(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((m + n, n + 1))
    A_ub[:m, :n] = T
    A_ub[:m, n] = -1.0
    A_ub[m:, :n] = -np.eye(n)
    b_ub = np.zeros(m + n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            x = np.maximum(res.x[:n], 0.0)
            s = x.sum()
            if s > 1e-15:
                x = x / s
                M = max(fast_convolve(fixed_seq) @ x)
                return x, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate_nesterov(a0, time_limit, beta_start=0.5, beta_end=0.95):
    """LP alternating with Nesterov-style momentum."""
    n = len(a0)
    x = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = x.sum()
    x = x / s
    y = x.copy()
    x_prev = x.copy()
    y_prev = y.copy()
    best_a = x.copy()
    best_score = 2.0 * n * fast_convolve(x).max()
    t0 = time.monotonic()
    max_iters = 500000
    eval_count = 0

    for k in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Adaptive momentum: start low, increase over time
        progress = min(1.0, k / 2000.0)
        beta = beta_start + (beta_end - beta_start) * progress

        # Extrapolate
        x_hat = (1.0 - beta) * x + beta * (x - x_prev)
        y_hat = (1.0 - beta) * y + beta * (y - y_prev)
        x_hat = np.maximum(x_hat, 0.0)
        y_hat = np.maximum(y_hat, 0.0)
        sx = x_hat.sum()
        sy = y_hat.sum()
        if sx < 1e-15:
            x_hat = x.copy()
        else:
            x_hat = x_hat / sx
        if sy < 1e-15:
            y_hat = y.copy()
        else:
            y_hat = y_hat / sy

        # Best responses
        x_new = solve_best_response(y_hat)
        x_new = x_new[0] if x_new is not None else x.copy()
        y_new = solve_best_response(x_hat)
        y_new = y_new[0] if y_new is not None else y.copy()

        x_prev = x
        y_prev = y
        x = x_new
        y = y_new

        # Track best self-convolution
        if k % 5 == 0:
            score_x = 2.0 * n * fast_convolve(x).max()
            score_y = 2.0 * n * fast_convolve(y).max()
            score_avg = 2.0 * n * fast_convolve((x + y) / 2.0).max()
            if score_x < best_score:
                best_score = score_x
                best_a = x.copy()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            if score_avg < best_score:
                best_score = score_avg
                best_a = ((x + y) / 2.0).copy()
            eval_count += 1

        if k % 500 == 0 and k > 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt iter={k+1} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.3f}")

    return best_a, best_score


def lse_gradient_refine(a0, time_limit, T_stages=None):
    """Log-sum-exp smoothed gradient descent with multi-stage annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    if T_stages is None:
        T_stages = [200.0, 80.0, 30.0, 10.0, 3.0, 1.0, 0.3, 0.1]

    n_stages = len(T_stages)
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break
        T_val = T_stages[stage]
        max_iter = max(20, int(remaining * 8))

        def func_lse(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_convolve(a_in)
            Tb = T_val * b
            mTb = Tb.max()
            return (2.0 * n / T_val) * (mTb + np.log(np.exp(Tb - mTb).sum()))

        def jac_lse(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_convolve(a_in)
            Tb = T_val * b
            mTb = Tb.max()
            exp_Tb = np.exp(Tb - mTb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_iter * 200, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    LSE stage {stage} error: {e}")
            break

        a = np.exp(z - z.max())
        a = a / a.sum()
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    LSE T={T_val}: score={score:.6f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def subgradient_descent(a0, time_limit, n_active_frac=0.05):
    """Mirror descent with subgradient of max-conv, adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 1.0
    no_improve = 0
    f_lb = 2.0 * n / (2.0 * n - 1.0)

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-9:
            break

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        z_new = z_new - z_new.max()
        a_new = np.exp(z_new)
        a_new = a_new / a_new.sum()
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 5.0)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.7
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 100000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def coord_descent_refine(a0, time_limit, n_sweeps=500):
    """Coordinate descent: for each j, find optimal a[j] given others fixed.
    Uses the fact that conv[k] changes linearly in a[j] for affected k."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        sweep_start = time.monotonic()

        # Randomize order
        order = rng.permutation(n)

        improved = False
        for j in order:
            old_val = a[j]
            if old_val < 1e-15:
                continue

            conv = fast_convolve(a)
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            ks = np.arange(k_start, k_end + 1)
            d = a[ks - j]
            C = conv[ks] - old_val * d

            # Find t >= 0 minimizing max(C + t*d)
            # Since d >= 0, f(t) is non-decreasing, so min at t=0
            # But we need to maintain sum(a) = 1, so reduce others
            # Simpler: try t = old_val * factor for various factors
            best_t = old_val
            best_local_max = conv.max()

            # Try reducing a[j] and seeing if we can redistribute to lower max
            # Affected positions: conv[ks] decrease by old_val*d when a[j] -> 0
            # Unaffected: conv[ks_complement] unchanged
            conv_no_j = conv.copy()
            conv_no_j[ks] = C
            max_unaffected = conv_no_j.max()

            # If setting a[j]=0 and the max is max_unaffected, that's good
            # But then sum decreases by old_val, others need to increase
            # After renorm: others scale by 1/(1-old_val)
            # New max = max(max_unaffected/(s-old_val), ...)
            # This is complex; just try a few values
            for factor in [0.5, 0.7, 0.3, 0.0, 1.5, 2.0]:
                t = old_val * factor
                if t < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = t
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = fast_convolve(a_trial).max()
                if sc * 2.0 * n < best_score - 1e-12:
                    best_t = t
                    best_local_max = sc * 2.0 * n / (st * st)
                    a = a_trial
                    best_score = 2.0 * n * fast_convolve(a).max()
                    best_a = a.copy()
                    improved = True
                    break

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s

        if sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep} ({elapsed:.0f}s): score={best_score:.6f}")

        if not improved and sweep > 50:
            break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        a = rng.random(n)
        a = a ** 2
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 0.5 + 0.1 * np.sin(20 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_descent(best_seq, t_alloc, n_active_frac=0.04)
        try_update(a_sg, "subgrad_1")

    # Phase 3: LSE gradient refinement
    print(f"\n=== Phase 3: LSE Gradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, _ = lse_gradient_refine(best_seq, t_alloc,
                                       T_stages=[100.0, 40.0, 15.0, 5.0, 2.0, 0.5, 0.15, 0.05])
        try_update(a_lse, "lse_1")

    # Phase 4: LP alternating again (warm-started from improved point)
    print(f"\n=== Phase 4: LP Alternating r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternate_nesterov(best_seq, t_alloc, beta_start=0.7, beta_end=0.97)
        try_update(a_lp2, "lp_alt_2")

    # Phase 5: Coord descent fine-tune
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coord_descent_refine(best_seq, t_alloc, n_sweeps=200)
        try_update(a_cd, "coord_desc")

    # Phase 6: Explore different lengths
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.4, 250.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            n_best = alt_n

        # LP alternating on new length
        t_lp = t_total * 0.6
        a_alt, _ = lp_alternate_nesterov(cur_best, t_lp)
        try_update(a_alt, f"lp_n{alt_n}")

        # LSE on new length
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lse = min(rem * 0.3, 60.0)
            a_alt2, _ = lse_gradient_refine(best_seq, t_lse,
                                            T_stages=[50.0, 20.0, 7.0, 2.0, 0.5, 0.1])
            try_update(a_alt2, f"lse_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.4, 100.0)
        a_fin, _ = lse_gradient_refine(best_seq, t_fin,
                                       T_stages=[20.0, 7.0, 2.0, 0.5, 0.1, 0.03])
        try_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_sg = min(rem * 0.3, 60.0)
        a_sg2, _ = subgradient_descent(best_seq, t_sg, n_active_frac=0.02)
        try_update(a_sg2, "final_subgrad")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp3 = min(rem * 0.4, 50.0)
        a_lp3, _ = lp_alternate_nesterov(best_seq, t_lp3, beta_start=0.8, beta_end=0.98)
        try_update(a_lp3, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
