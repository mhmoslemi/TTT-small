import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence. Returns 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                print(f"  LP solver: {solver_name}")
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def multiplicative_weights_nesterov(a0, time_limit, eta_init=0.01, n_stages=8, steps_per_stage=5000):
    """Nesterov-accelerated multiplicative weights on the simplex."""
    n = len(a0)
    x = np.maximum(a0, 1e-15).copy()
    x = x / x.sum()
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    step_count = 0

    for stage in range(n_stages):
        eta = eta_init / (1.0 + 0.1 * stage)
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            # Nesterov momentum
            t_curr = 1.0 + step_count
            t_prev = 1.0 + (step_count - 1) if step_count > 0 else 1.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()

            # Compute subgradient at peak
            b = fast_convolve(y)
            peak_k = int(np.argmax(b))
            # grad[i] = 2*a[peak_k - i] if 0 <= peak_k - i < n else 0
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            grad = np.zeros(n)
            grad[js:je+1] = 2.0 * y[peak_k - je:peak_k - js][::-1]
            
            x_prev = x.copy()
            # Multiplicative update
            x = y * np.exp(-eta * grad / y)
            x = np.maximum(x, 1e-15)
            s = x.sum()
            x = x / s
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step_count % 2000 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      MW step={step_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lp_refine_aggressive(a0, time_limit, solver=None, n_max_iters=100000):
    """Aggressive LP best-response with Nesterov averaging and dual tracking."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    fail_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            fail_count += 1
            if fail_count > 20:
                # Fallback: use uniform
                b_new = np.ones(n) / n
            continue
        fail_count = 0
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Primal averaging
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()

        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        score_y = score_normalized(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it * 7 + 13)
                y = y + 0.03 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_b={M_b:.6f}")

def peak_equalize(a0, time_limit, k_peaks=5):
    """Try to equalize the top k peaks of the convolution."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find top k peaks (excluding endpoints which are always small)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_indices = np.argsort(b_work)[::-1][:k_peaks]
        peak_indices = sorted(peak_indices)
        peak_values = b[peak_indices]
        target = peak_values.mean()
        
        # For each peak, compute which coordinates contribute most
        # and try to reduce the max peak while maintaining others
        improved = False
        for pk in peak_indices:
            pk_val = b[pk]
            if pk_val <= target * 1.001:
                continue
            # Find top contributors to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_i = np.argsort(contribs)[::-1][:3]
            
            for ti in top_i:
                i = idx[ti]
                j = pk - i
                if i == j:
                    # Reduce a[i] slightly
                    for factor in [0.99, 0.98, 0.97]:
                        a_trial = a.copy()
                        a_trial[i] *= factor
                        # Compensate by increasing other non-peak-contributing elements
                        a_trial = np.maximum(a_trial, 0)
                        a_trial /= a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                else:
                    if j < 0 or j >= n:
                        continue
                    for f1 in [0.99, 0.98]:
                        for f2 in [1.0, 1.01]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial /= a_trial.sum()
                            sc = score_normalized(a_trial)
                            if sc < best_score - 1e-12:
                                a = a_trial.copy()
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try simultaneous small adjustment
            for factor in [0.995, 0.99, 0.985]:
                a_trial = a.copy()
                for pk in peak_indices:
                    js = max(0, pk - n + 1)
                    je = min(n - 1, pk)
                    idx = np.arange(js, je + 1)
                    contribs = a[idx] * a[pk - idx]
                    top_i = np.argsort(contribs)[::-1][:2]
                    for ti in top_i:
                        a_trial[idx[ti]] *= factor
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=50000):
    """Pure subgradient method on the max-convolution."""
    n = len(a0)
    x = np.maximum(a0, 0).copy()
    x = x / x.sum()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    ck = 1.0  # step size denominator
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(x)
        peak_k = int(np.argmax(b))
        
        # Subgradient
        grad = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        grad[js:je+1] = 2.0 * x[peak_k - je:peak_k - js][::-1]
        
        # Adaptive step size
        lr = 1.0 / (ck + step * 0.01)
        x_new = x - lr * grad
        x_new = project_simplex(x_new)
        
        x = x_new
        if step % 50 == 0:
            sc = score_normalized(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()

    return best_a, best_score


def make_init(n, style, rng):
    """Create various initialization styles."""
    x = np.linspace(0, 1, n)
    if style == 0:  # uniform
        a = np.ones(n)
    elif style == 1:  # sinusoidal
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    elif style == 2:  # asymmetric
        a = 1.0 + 0.5 * (2 * x - 1)
    elif style == 3:  # two bumps
        a = np.exp(-5.0 * (x - 0.3)**2) + 0.5 * np.exp(-5.0 * (x - 0.7)**2) + 0.1
    elif style == 4:  # random smooth
        coeff = rng.standard_normal(5)
        a = 1.0 + 0.3 * (coeff[0]*np.sin(2*np.pi*x) + coeff[1]*np.sin(4*np.pi*x) +
                         coeff[2]*np.cos(2*np.pi*x) + coeff[3]*np.sin(6*np.pi*x) +
                         coeff[4]*np.cos(4*np.pi*x))
    else:  # cosine window
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence. Returns 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def multiplicative_weights(a0, time_limit, eta_init=0.015, n_stages=10, steps_per_stage=8000):
    """Nesterov-accelerated multiplicative weights with smoothed gradient."""
    n = len(a0)

def peak_equalize(a0, time_limit, k_peaks=5, n_outer=500):

        max_peak_val = b[max_peak_idx]
        second_max = max(b[peak_indices[1:]] if len(peak_indices) > 1 else 0.0)

        if max_peak_val <= second_max * 1.0001:
            break  # Peaks are already balanced

        js = max(0, max_peak_idx - n + 1)
        je = min(n - 1, max_peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_peak_idx - idx]
        total_contrib = contribs.sum()

        # Try reducing top contributors

                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)

                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:

                        a_trial = np.maximum(a_trial, 0)

                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:

                            best_score = sc

    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def coordinate_descent_polish(a0, time_limit, n_iterations=200000):
    """Sequential coordinate descent on the max-convolution objective."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    indices = np.random.default_rng(123).permutation(n)

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        i = indices[it % n]
        if a[i] < 1e-15:
            continue

        # Try various multipliers for coordinate i
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Does coordinate i contribute to the peak?
        if not (0 <= peak_k - i < n):
            # Try increasing this coordinate (might reduce peak by spreading mass)
            for factor in [1.01, 1.02, 1.05]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial /= a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    break
        else:
            # Does coordinate i contribute? Try reducing
            for factor in [0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial /= a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    break

        if it % 10000 == 0:
            print(f"      CD iter={it} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: multiplicative weights + aggressive LP refinement
    + peak equalization + coordinate descent polish + multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_init(n_init, 1, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Quick Multiplicative Weights (10%) ===
    print(f"\n=== Phase 1: MW Init n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, _ = multiplicative_weights(best_seq, t_alloc, eta_init=0.02,
                                         n_stages=6, steps_per_stage=5000)
        try_update(a_mw, "mw_init")

    # === Phase 2: Main LP Refinement (45%) ===
    print(f"\n=== Phase 2: Main LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 810.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000,
                            perturb_interval=400, perturb_size=0.015)
        try_update(a_lp, "lp_main")

    # === Phase 3: Peak Equalization (8%) ===
    print(f"\n=== Phase 3: Peak Equalization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 144.0)
    if t_alloc > 10:
        a_pe, _ = peak_equalize(best_seq, t_alloc, k_peaks=5, n_outer=1000)
        try_update(a_pe, "peak_eq")

    # === Phase 4: LP Refinement Round 2 (15%) ===
    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 20:
        a_lp2, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000,
                             perturb_interval=300, perturb_size=0.01)
        try_update(a_lp2, "lp_2")

    # === Phase 5: Coordinate Descent Polish (7%) ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 126.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_polish(best_seq, t_alloc, n_iterations=300000)
        try_update(a_cd, "cd_polish")

    # === Phase 6: Multi-Start (10%) ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        styles = [0, 3, 5]
        for style in styles:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 25:
                break
            a_init = make_init(n_best, style, rng)
            # Quick MW

                                              n_stages=4, steps_per_stage=3000)
            try_update(a_sub, f"mw_s{style}")

                a_sub2, _ = lp_refine(a_sub, min(rem * 0.4, t_sub * 0.5), solver=solver,
                                      n_max_iters=20000, perturb_interval=200)
                try_update(a_sub2, f"lp_s{style}")

    # === Phase 7: Alternative Lengths (5%) ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = sorted(set([n_best - 80, n_best - 40, n_best + 40, n_best + 80]))
    alt_lengths = [x for x in alt_lengths if 300 <= x <= 3000]

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        t_total = min(rem * 0.25, 90.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = multiplicative_weights(init_alt, t_total * 0.4, eta_init=0.02,
                                          n_stages=4, steps_per_stage=3000)
        try_update(a_alt, f"mw_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = lp_refine(a_alt, min(rem * 0.4, 25.0), solver=solver,
                                  n_max_iters=10000, perturb_interval=150)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 8: Final LP Polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.50, 60.0)
        a_fin, _ = lp_refine(best_seq, t_fin, solver=solver, n_max_iters=50000,
                             perturb_interval=300, perturb_size=0.005)
        try_update(a_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
