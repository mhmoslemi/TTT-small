def simplex_project(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    return np.ones(n) / n

        a = np.ones(n) / n

        if score_b < best_score:

**Approach: Subgradient Method with Nesterov Acceleration**

The objective is f(a) = max_k (T_a @ a)_k where T_a depends on a. This is non-smooth (max) and non-convex (T_a depends on a).

But if we fix T (i.e., fix one copy of a), then g(b) = max_k (T @ b)_k is convex in b, and we can use subgradient methods.

The subgradient of g at b is: T @ e_{k*} where k* = argmax_k (T@b)_k.

With Nesterov acceleration for the simplex-constrained version:
- Maintain momentum
- Step size: η_t = 2/(t+2)
- Update: b_{t+1} = simplex_project(y_t - η_t * T @ e_{k*})
- y_{t+1} = b_{t+1} + (t/(t+2))*(b_{t+1} - b_t)

This gives O(1/t²) convergence for the convex problem, which is better than O(1/t) for plain FW.

def accelerated_fw(a_init, b_init, time_limit, n_steps=100000, verbose=True):
    """
    Accelerated FW (Nesterov) for min_b max_k (conv(a,b)[k]) s.t. b in simplex.
    This is the "inner" problem of the alternating scheme.
    """
    n = len(a_init)
    a = a_init.copy()
    b = b_init.copy() / b_init.sum()
    
    # Initialize
    conv_ab = convolve_ab(a, b)  # length 2n-1
    # Nesterov momentum
    y = b.copy()
    beta = 0.0  # momentum parameter
    
    for t in range(n_steps):
        # Find active constraint
        k_star = int(np.argmax(conv_ab))
        
        # Subgradient: T[:, k_star]... wait, no.
        # g(b) = max_k (T@b)_k where T[k,i] = a[k-i]
        # subgradient at b is the k*-th row of T: sg[i] = a[k_star - i] for valid i
        # Actually, g(b) = max_k sum_i T[k,i]*b[i], so subgrad = T[k_star, :]
        # T[k_star, i] = a[k_star - i] if 0 <= k_star - i < n, else 0
        
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        # sg[i] = a[k_star - i] for i in [i_start, i_end]
        # a[k_star - i] for i in [i_start, i_end] = a[k_star-i_start : k_star-i_end+1][::-1]
        sg = np.zeros(n)
        sg[i_start:i_end+1] = a[k_star-i_end:k_star-i_start+1][::-1]
        
        # Step size
        eta = 2.0 / (t + 2.0)
        
        # FW step: find vertex that minimizes sg @ v
        # min_v sg @ v s.t. v in simplex => v = e_{argmin(sg)}
        j_star = int(np.argmin(sg))
        
        # Standard FW: b_new = (1-eta)*b + eta*e_{j_star}
        # Accelerated: use momentum
        # b_candidate = (1-eta)*y + eta*e_{j_star}
        e_j = np.zeros(n)
        e_j[j_star] = 1.0
        b_new = (1.0 - eta) * y + eta * e_j
        
        # FISTA-style: beta_{t+1} = (1 - sqrt(1 + 4*beta_t²)) / 2
        beta_new = (1.0 - np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = b_new + beta_new * (b_new - b)
        y_new = simplex_project(y_new)  # Ensure feasibility
        
        # Update convolution
        conv_ab = (1.0 - eta) * conv_ab
        # Add eta * conv(a, e_{j_star}) = eta * shift(a, j_star)
        ks_s = j_star
        ks_e = min(j_star + n - 1, 2 * n - 2)
        conv_ab[ks_s:ks_e+1] += eta * a
        
        b = b_new
        beta = beta_new
    
    return b
