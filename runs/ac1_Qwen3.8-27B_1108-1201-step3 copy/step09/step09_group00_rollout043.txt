```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The program starts from the parent construction when available and applies
several different local/global heuristics:

1. Adaptive soft-weighted quadratic refinement.
   We alternately choose weights w_k on the convolution peaks (tempered by
   an adaptive number of active peaks), then minimize the smooth surrogate
   2n * sum_k w_k (a*a)_k in log-space with L-BFGS.

2. Full log-sum-exp L-BFGS refinement with high temperatures.
   This directly approximates the true max-convolution objective.

3. Sequential linearized LP polishing.
   Around the current iterate, active peak constraints are linearized and a
   small LP is solved to produce a descent direction.

4. Random multiplicative-walk polish and zero-threshold postprocessing.

The code is written to survive hard time limits and always returns the best
sequence found.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

_DEADLINE = 0.0


def time_left():
    if _DEADLINE <= 0.0:
        return 1.0e18
    return max(0.0, _DEADLINE - time.monotonic())


def next_pow2(x):
    x = int(max(2, x))
    return 1 << (x - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if floor > 0.0:
        a = np.maximum(a, floor)
    s = float(a.sum())
    if s <= 1.0e-15:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a / s


def make_positive(a, floor=1.0e-12):
    return normalize_positive(a, floor=floor)


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = rfft(a, N)
    c = irfft(A * np.conj(A), N)[:size]
    c = np.maximum(c, 0.0)
    return c


def fast_corr_w_a(w, a):
    """
    c[j] = sum_i a[i] w[j+i], for j=0..n-1, where w has length 2n-1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    L = w.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if L == 0:
        return np.zeros(n, dtype=np.float64)
    N = next_pow2(L + n - 1)
    aw = np.zeros(N, dtype=np.float64)
    aw[:n] = a
    ww = np.zeros(N, dtype=np.float64)
    ww[:L] = w[::-1]
    full = irfft(rfft(aw, N) * rfft(ww, N), N)
    c = full[n - 1:2 * n - 1][::-1]
    c = np.maximum(c, 0.0)
    return c


def local_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    b = fast_convolve(a)
    mx = float(b.max())
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * a.size * mx / (s * s))


def softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return z
    zs = z - float(np.max(z))
    e = np.exp(zs)
    s = float(e.sum())
    if s <= 1.0e-300:
        return np.ones_like(z, dtype=np.float64) / z.size
    return e / s


def logsafe(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    return np.log(np.maximum(a, 1.0e-300))


def _lbfgs_callback(k, fval, x):
    if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE - 10.0:
        return True
    return False


def run_lbfgs(z0, fun, jac, args, maxiter):
    z0 = np.asarray(z0, dtype=np.float64).ravel().copy()
    if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE - 2.0:
        return z0
    try:
        res = minimize(
            fun=fun,
            x0=z0,
            jac=jac,
            args=args,
            method="L-BFGS-B",
            options={
                "maxiter": int(max(1, maxiter)),
                "ftol": 1.0e-15,
                "gtol": 1.0e-8,
                "maxls": 30,
                "maxcor": 20,
                "disp": False,
            },
            callback=_lbfgs_callback,
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception as e:
        print("LBFGS failed:", e)
    return z0


def lse_val(z, T):
    if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE:
        return 1.0e100
    a = softmax(z)
    n = a.size
    b = fast_convolve(a)
    Tb = T * b
    m = float(Tb.max())
    e = np.exp(Tb - m)
    s = float(e.sum())
    if s <= 1.0e-300:
        return float(2.0 * n * b.max())
    return float((2.0 * n / T) * (m + np.log(s)))


def lse_jac(z, T):
    if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE:
        return np.zeros_like(z, dtype=np.float64)
    a = softmax(z)
    n = a.size
    b = fast_convolve(a)
    Tb = T * b
    m = float(Tb.max())
    e = np.exp(Tb - m)
    s = float(e.sum())
    if s <= 1.0e-300:
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / s
    c = fast_corr_w_a(w, a)
    G = 4.0 * n * c
    S = float(np.dot(G, a))
    return a * (G - S)


def prepare_weight_info(w, max_sparse=32):
    """
    For sparse weight vectors, return precomputed pieces to avoid FFTs.
    w must have length 2n-1.
    """
    w = np.asarray(w, dtype=np.float64).ravel()
    nz = np.nonzero(w)[0]
    if nz.size == 0:
        return None
    if nz.size > max_sparse:
        return None
    L = w.size
    n = (L + 1) // 2
    info = []
    for k in nz:
        k = int(k)
        wk = float(w[k])
        if wk <= 0.0:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        info.append((k, wk, js, je))
    if not info:
        return None
    return info


def weighted_val(z, w, info):
    if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE:
        return 1.0e100
    a = softmax(z)
    n = a.size
    if info is not None:
        val = 0.0
        for k, wk, js, je in info:
            pos = np.arange(js, je + 1)
            val += wk * float(np.dot(a[pos], a[k - pos]))
        return float(2.0 * n * val)
    b = fast_convolve(a)
    return float(2.0 * n * np.dot(w, b))


def weighted_jac(z, w, info):
    if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE:
        return np.zeros_like(z, dtype=np.float64)
    a = softmax(z)
    n = a.size
    if info is not None:
        c = np.zeros(n, dtype=np.float64)
        for k, wk, js, je in info:
            pos = np.arange(js, je + 1)
            c[pos] += wk * a[k - pos]
        G = 4.0 * n * c
    else:
        c = fast_corr_w_a(w, a)
        G = 4.0 * n * c
    S = float(np.dot(G, a))
    return a * (G - S)


def weights_from_T(b, T):
    b = np.maximum(np.asarray(b, dtype=np.float64).ravel(), 0.0)
    if b.size == 0:
        return np.zeros(0, dtype=np.float64)
    logw = T * np.log(np.maximum(b, 1.0e-300))
    m = float(logw.max())
    e = np.exp(logw - m)
    s = float(e.sum())
    if s <= 1.0e-300:
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return w
    return e / s


def choose_T_for_target(b, target_count):
    b = np.asarray(b, dtype=np.float64).ravel()
    maxb = float(b.max())
    if maxb <= 1.0e-15:
        return 1.0
    L = b.size
    if target_count <= 1 or L <= 1:
        return min(100.0 / maxb, 500.0 / maxb)
    order = np.argsort(b)[::-1]
    idx = min(max(1, int(target_count)), L) - 1
    r = float(b[order[idx]] / maxb)
    if r >= 0.999999:
        return min(100.0 / maxb, 500.0 / maxb)
    if r <= 1.0e-12:
        r = 1.0e-12
    T = float(np.log(10.0) / (-np.log(r)))
    cap = 500.0 / maxb
    if T < 1.0:
        T = 1.0
    if T > cap:
        T = cap
    return float(T)


def improve_or_reject(current, current_score, trial, tolerance=1.0e-11):
    """
    Try trial, then shrinked convex combinations.
    Returns (accepted, current, current_score, trial_score)
    """
    trial = normalize_positive(trial, floor=0.0)
    trial_score = local_score(trial)
    if trial_score < current_score - tolerance:
        return True, trial, trial_score, trial_score
    for alpha in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
        cand = normalize_positive((1.0 - alpha) * current + alpha * trial, floor=0.0)
        sc = local_score(cand)
        if sc < current_score - tolerance:
            return True, cand, sc, sc
    return False, current, current_score, trial_score


def adaptive_soft_fixed_refine(a0, budget):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    if a0.size == 0:
        return a0
    n = a0.size
    current = make_positive(a0, 1.0e-12)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    if budget <= 10.0:
        return best

    t0 = time.monotonic()
    rng = np.random.default_rng(987654)
    targets = [32, 16, 8, 4, 2, 1]
    target_idx = 0
    no_imp = 0
    stale = 0
    outer_since_best = 0
    outer = 0
    iter_scale = max(0.25, 1200.0 / float(n))
    tol = 1.0e-11

    while time.monotonic() - t0 < budget - 10.0:
        if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE - 25.0:
            break

        if current_score > best_score + 5.0e-4:
            current = best.copy()
            current_score = best_score
            target_idx = 0
            no_imp = 0
            stale = 0
            outer_since_best = 0
            continue

        target = targets[min(target_idx, len(targets) - 1)]
        b = fast_convolve(current)
        maxb = float(b.max())
        if maxb <= 1.0e-15:
            break
        T = choose_T_for_target(b, target)
        w = weights_from_T(b, T)
        info = prepare_weight_info(w, max_sparse=32)
        z = logsafe(current)
        maxiter = int(min(900, max(80, (170 + 25 * target_idx) * iter_scale)))
        z = run_lbfgs(z, weighted_val, weighted_jac, (w, info), maxiter)
        trial = normalize_positive(softmax(z), floor=0.0)

        ok, current, current_score, trial_score = improve_or_reject(
            current, current_score, trial, tol
        )

        if ok:
            no_imp = 0
            stale = 0
            if current_score < best_score - tol:
                best_score = current_score
                best = current.copy()
                outer_since_best = 0
            else:
                outer_since_best += 1
        else:
            no_imp += 1
            stale += 1
            outer_since_best += 1

            if stale >= 3:
                if target_idx < len(targets) - 1:
                    target_idx += 1
                elif target_idx > 0:
                    target_idx = max(0, target_idx - 2)
                stale = 0

            if no_imp >= 10 and no_imp % 7 == 0:
                sigma = 0.005 + 0.001 * min(no_imp, 30)
                pert = normalize_positive(
                    current * np.exp(sigma * rng.standard_normal(n)), floor=0.0
                )
                psc = local_score(pert)
                if psc < current_score + 5.0e-5:
                    current = pert
                    current_score = psc

        outer += 1
        if outer % 20 == 0:
            print(
                "  soft_fixed outer={:4d} target={:3d} best={:.10f} current={:.10f}".format(
                    outer, target, best_score, current_score
                )
            )

        if outer_since_best > 140:
            break
        if no_imp > 200:
            break

    return best


def lse_refine(a0, budget):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    if a0.size == 0:
        return a0
    n = a0.size
    current = make_positive(a0, 1.0e-12)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    if budget <= 15.0:
        return best

    t0 = time.monotonic()
    mults = [1, 2, 4, 8, 16, 32, 64, 128, 256]
    iter_scale = max(0.25, 1200.0 / float(n))
    stale = 0
    tol = 1.0e-11

    for i, mult in enumerate(mults):
        if time.monotonic() - t0 > budget - 10.0:
            break
        if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE - 25.0:
            break

        b = fast_convolve(current)
        maxb = float(b.max())
        if maxb <= 1.0e-15:
            break
        T = float(mult) * float(n)
        cap = 400.0 / maxb
        if T > cap:
            T = cap
        if T < 1.0:
            T = 1.0

        maxiter = int(min(1400, max(100, (220 + 70 * i) * iter_scale)))
        z = logsafe(current)
        z = run_lbfgs(z, lse_val, lse_jac, (T,), maxiter)
        trial = normalize_positive(softmax(z), floor=0.0)
        ok, current, current_score, trial_score = improve_or_reject(
            current, current_score, trial, tol
        )

        if ok:
            stale = 0
            if current_score < best_score - tol:
                best_score = current_score
                best = current.copy()
        else:
            # Allow very small surrogate-progress moves, but do not let current drift far.
            if trial_score < current_score + 1.0e-8 and trial_score <= best_score + 1.0e-5:
                current = trial
                current_score = trial_score
            else:
                stale += 1

        if (i + 1) % 3 == 0:
            print(
                "  lse stage T={:.3f} best={:.10f} current={:.10f}".format(
                    T, best_score, current_score
                )
            )

        if stale >= 4:
            break

    return best


def lp_linearized_refine(a0, budget, initial_active=10):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    if a0.size == 0:
        return a0
    n = a0.size
    current = make_positive(a0, 1.0e-12)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    if budget <= 20.0 or n == 0:
        return best

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]

    t0 = time.monotonic()
    rng = np.random.default_rng(246810)
    active_count = max(4, int(initial_active))
    no_imp = 0
    outer = 0
    tol = 1.0e-11

    while time.monotonic() - t0 < budget - 10.0:
        if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE - 25.0:
            break

        b = fast_convolve(current)
        maxb = float(b.max())
        if maxb <= 1.0e-15:
            break

        order = np.argsort(b)[::-1]
        close = np.where(b >= maxb - max(1.0e-12, 0.01 * maxb))[0]
        if close.size <= 2 * active_count:
            active = np.unique(np.concatenate((order[:active_count], close)))
        else:
            active = order[:active_count]
        if active.size > 32:
            active = order[:32]

        m = int(active.size)
        A_ub = np.zeros((m, n + 1), dtype=np.float64)
        b_ub = np.zeros(m, dtype=np.float64)

        for i, k in enumerate(active):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            pos = np.arange(js, je + 1)
            A_ub[i, pos] = 2.0 * current[k - pos]
            A_ub[i, -1] = -1.0
            b_ub[i] = b[k]

        try:
            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception as e:
            print("LP failed:", e)
            no_imp += 1
            if no_imp > 20:
                break
            continue

        if not res.success:
            no_imp += 1
            if no_imp > 20:
                break
            continue

        xv = np.asarray(res.x[:-1], dtype=np.float64)
        xv = np.maximum(xv, 0.0)
        s = float(xv.sum())
        if s <= 1.0e-15:
            no_imp += 1
            if no_imp > 20:
                break
            continue
        xv /= s

        ok, current, current_score, trial_score = improve_or_reject(
            current, current_score, xv, tol
        )

        if ok:
            no_imp = 0
            if current_score < best_score - tol:
                best_score = current_score
                best = current.copy()
            if active_count > 4:
                active_count = max(4, active_count - 1)
        else:
            no_imp += 1
            t_opt = float(res.x[-1])
            if t_opt >= maxb - 1.0e-9:
                active_count = min(32, active_count + 2)
            if no_imp >= 15 and no_imp % 7 == 0:
                pert = normalize_positive(
                    current * np.exp(0.008 * rng.standard_normal(n)), floor=0.0
                )
                psc = local_score(pert)
                if psc < current_score + 5.0e-5:
                    current = pert
                    current_score = psc

        outer += 1
        if outer % 10 == 0:
            print(
                "  LP outer={:4d} active={:3d} best={:.10f} current={:.10f}".format(
                    outer, active_count, best_score, current_score
                )
            )

        if no_imp >= 30:
            break

    return best


def random_walk_refine(a0, budget):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    if a0.size == 0:
        return a0
    n = a0.size
    current = make_positive(a0, 1.0e-12)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    if budget <= 10.0:
        return best

    t0 = time.monotonic()
    rng = np.random.default_rng(555123)
    step = 0.015
    no_imp = 0
    tol = 1.0e-12
    outer = 0

    while time.monotonic() - t0 < budget - 10.0:
        if _DEADLINE > 0.0 and time.monotonic() > _DEADLINE - 25.0:
            break
        trial = current * np.exp(step * rng.standard_normal(n))
        trial = normalize_positive(trial, floor=0.0)
        sc = local_score(trial)
        if sc < current_score - tol:
            current = trial
            current_score = sc
            no_imp = 0
            step = min(0.08, step * 1.12)
            if current_score < best_score - tol:
                best_score = current_score
                best = current.copy()
        else:
            no_imp += 1
            step = max(0.001, step * 0.92)
            if no_imp % 60 == 0:
                step = min(0.08, step * 1.8)

        outer += 1
        if outer % 500 == 0:
            print(
                "  randwalk outer={:5d} step={:.5f} best={:.10f} current={:.10f}".format(
                    outer, step, best_score, current_score
                )
            )
        if no_imp > 600:
            break

    return best


def zero_refine(a):
    a = normalize_positive(a, floor=0.0)
    if a.size == 0:
        return a
    best = a.copy()
    best_score = local_score(a)
    maxa = float(best.max())
    if maxa <= 0.0:
        return best

    thresholds = [1.0e-8, 1.0e-7, 1.0e-6, 1.0e-5, 1.0e-4, 5.0e-4, 1.0e-3]
    for th in thresholds:
        trial = best.copy()
        mask = trial < th * maxa
        trial[mask] = 0.0
        trial = normalize_positive(trial, floor=0.0)
        sc = local_score(trial)
        if sc < best_score - 1.0e-12:
            best_score = sc
            best = trial

    for cnt in (1, 2, 3, 5, 8, 12):
        trial = best.copy()
        trial[:cnt] = 0.0
        trial[-cnt:] = 0.0
        trial = normalize_positive(trial, floor=0.0)
        sc = local_score(trial)
        if sc < best_score - 1.0e-12:
            best_score = sc
            best = trial

    return best


def structured_init(n, rng):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    f1 = 1.0 + rng.random() * 4.0
    f2 = 5.0 + rng.random() * 9.0
    f3 = 13.0 + rng.random() * 11.0
    p1 = rng.random() * 2.0 * np.pi
    p2 = rng.random() * 2.0 * np.pi
    p3 = rng.random() * 2.0 * np.pi
    a = (
        1.0
        + 0.45 * np.cos(2.0 * np.pi * f1 * x + p1)
        + 0.35 * np.cos(2.0 * np.pi * f2 * x + p2)
        + 0.25 * np.cos(2.0 * np.pi * f3 * x + p3)
    )
    a = np.maximum(a, 0.0) + 1.0e-7
    return normalize_positive(a, floor=0.0)


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    new_n = max(2, int(new_n))
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(f.sum())
    if s <= 1.0e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, fn)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1.0e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return g / gs


def propose_candidate(seed=42, budget_s=1800.0):
    global _DEADLINE

    t_start = time.monotonic()
    reserve = 40.0 if float(budget_s) > 120.0 else 5.0
    _DEADLINE = t_start + max(30.0, float(budget_s) - reserve)

    rng = np.random.default_rng(seed)
    best_a = None
    best_score = float("inf")

    # Load parent if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size > 0 and float(p.sum()) > 1.0e-15:
            best_a = normalize_positive(p, floor=0.0)
            best_score = local_score(best_a)
            print(
                "Loaded parent: n={:d} score={:.10f}".format(int(p.size), best_score)
            )
        else:
            raise ValueError("invalid parent")
    except Exception:
        n0 = 1200
        best_a = structured_init(n0, rng)
        best_score = local_score(best_a)
        print(
            "Fallback initial: n={:d} score={:.10f}".format(int(best_a.size), best_score)
        )

    if best_a is None or best_a.size == 0 or not np.all(np.isfinite(best_a)):
        best_a = structured_init(1024, rng)
        best_score = local_score(best_a)

    # If the inherited sequence is enormous, downsample it to keep runtime sane.
    if best_a.size > 8000:
        best_a = interpolate_sequence(best_a, 3000)
        best_score = local_score(best_a)
        print("Downsampled parent to n={:d} score={:.10f}".format(int(best_a.size), best_score))

    def update_best(a):
        nonlocal best_a, best_score
        if a is None or a.size == 0:
            return
        a = normalize_positive(a, floor=0.0)
        sc = local_score(a)
        if np.isfinite(sc) and sc < best_score - 1.0e-12:
            best_a = a
            best_score = sc
            print("  new global best: {:10.10f} (n={:d})".format(sc, int(a.size)))
        elif best_a is None:
            best_a = a
            best_score = sc

    # Phase 1: main adaptive soft-weighted refinement.
    rem = time_left()
    if rem > 80.0:
        alloc = min(rem * 0.55, 920.0)
        print("\nPhase 1: adaptive soft-weighted refinement budget={:.0f}s".format(alloc))
        a_new = adaptive_soft_fixed_refine(best_a, alloc)
        update_best(a_new)

    # Phase 2: full log-sum-exp L-BFGS refinement.
    rem = time_left()
    if rem > 90.0:
        alloc = min(rem * 0.30, 470.0)
        print("\nPhase 2: LSE L-BFGS refinement budget={:.0f}s".format(alloc))
        a_new = lse_refine(best_a, alloc)
        update_best(a_new)

    # Phase 3: sequential linearized LP polishing.
    rem = time_left()
    if rem > 90.0:
        alloc = min(rem * 0.35, 330.0)
        print("\nPhase 3: linearized LP refinement budget={:.0f}s".format(alloc))
        a_new = lp_linearized_refine(best_a, alloc)
        update_best(a_new)

    # Phase 4: short soft-weighted polish.
    rem = time_left()
    if rem > 50.0:
        alloc = min(rem * 0.45, 180.0)
        print("\nPhase 4: short soft-weighted polish budget={:.0f}s".format(alloc))
        a_new = adaptive_soft_fixed_refine(best_a, alloc)
        update_best(a_new)

    # Phase 5: alternates if time remains.
    rem = time_left()
    if rem > 220.0 and best_a is not None:
        n = int(best_a.size)

        # 5a: high-frequency multiplicative perturb + local refine.
        x = np.linspace(0.0, 1.0, n)
        for freq in (5, 11):
            rem = time_left()
            if rem < 120.0:
                break
            alloc = min(rem * 0.25, 140.0)
            ph = rng.random() * 2.0 * np.pi
            pert = best_a + 0.012 * np.cos(2.0 * np.pi * freq * x + ph)
            pert = normalize_positive(pert, floor=0.0)
            print("\nPhase 5a: freq={:d} perturbed alternates budget={:.0f}s".format(freq, alloc))
            a_new = adaptive_soft_fixed_refine(pert, alloc)
            update_best(a_new)

        # 5b: length-increased interpolant + noise.
        rem = time_left()
        if rem > 160.0:
            new_n = min(4096, max(n + 128, int(1.2 * n)))
            alloc = min(rem * 0.5, 220.0)
            cand = interpolate_sequence(best_a, new_n)
            cand = normalize_positive(
                cand + 0.003 * np.random.default_rng(seed + 777).standard_normal(new_n),
                floor=0.0,
            )
            print("\nPhase 5b: length alternates n={:d} budget={:.0f}s".format(new_n, alloc))
            a_new = adaptive_soft_fixed_refine(cand, alloc)
            update_best(a_new)

    # Phase 6: random walk polish.
    rem = time_left()
    if rem > 45.0:
        alloc = min(rem * 0.45, 130.0)
        print("\nPhase 6: random walk polish budget={:.0f}s".format(alloc))
        a_new = random_walk_refine(best_a, alloc)
        update_best(a_new)

    # Final zero-threshold cleanup.
    if best_a is not None:
        best_a = zero_refine(best_a)
        best_score = local_score(best_a)

    # Final direct check and optional official evaluator call if available.
    if best_a is not None:
        try:
            b = np.convolve(best_a, best_a)
            s = float(best_a.sum())
            direct = float(2.0 * best_a.size * float(b.max()) / (s * s))
            print("Final FFT/local score: {:.12f}".format(best_score))
            print("Final direct np.convolve score: {:.12f}".format(direct))
            best_score = direct
        except Exception:
            pass

    out = []
    for x in best_a:
        y = float(x)
        if not np.isfinite(y) or y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0
        out.append(y)

    if time_left() > 12.0:
        try:
            ev = evaluate_sequence(out)
            print("evaluate_sequence(final):", ev)
        except Exception:
            pass

    return out
```
