Option C is cleanest. Let v be unconstrained, and a = softmax(v) * (some scale) ... but we need sum(a) = 1, so a = exp(v) / sum(exp(v)).

The gradient with respect to v is:
∂F/∂v_i = sum_j (∂F/∂a_j) * (∂a_j/∂v_i)
= sum_j grad_j * (a_j * δ_{j,i} - a_j * a_i)
= grad_i * a_i - a_i * sum_j grad_j * a_j
= a_i * (grad_i - sum_j grad_j * a_j)

So the gradient w.r.t. v is simply a_i * (grad_i - <grad, a>).

This is the standard "softmax gradient" formula.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based alternating minimization (exact best response)
2. Smooth (log-sum-exp) optimization with L-BFGS and continuation
3. Multi-peak mirror descent with adaptive learning rate
4. Fast coordinate descent with sensitivity analysis
5. Multi-scale and diverse initialization
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

# Globals for time tracking
_TOTAL_DEADLINE = time.monotonic() + 1780.0

def time_remaining():
    return _TOTAL_DEADLINE - time.monotonic()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax(v):
    """Numerically stable softmax."""
    v_shifted = v - v.max()
    e = np.exp(v_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def cross_corr_slidedot(a, p):
    """
    Compute r[i] = sum_{j=0}^{n-1} a[j] * p[j+i] for i = 0, ..., n-1
    where p has length 2n-1.
    Uses the identity: r[i] = conv(rev(a), p)[n-1+i]
    """
    n = len(a)
    a_rev = a[::-1]
    c = convolve_ab(a_rev, p)  # length (n-1) + (2n-1) - 1 = 3n-3
    # r[i] = c[n-1+i] for i = 0, ..., n-1
    return c[n-1:n-1+n]

def smooth_obj_grad(a, tau):
    """
    Compute the smooth objective 2n*logsumexp(conv(a,a)/tau) and its gradient.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Log-sum-exp for numerical stability
    max_c = c.max()
    exp_c = np.exp((c - max_c) / tau)
    log_sum_exp = max_c / tau + np.log(exp_c.sum())
    obj = 2.0 * n * log_sum_exp
    
    # Softmax weights
    p = exp_c / exp_c.sum()
    
    # Gradient: 4n/tau * (slidedot)(a, p)
    r = cross_corr_slidedot(a, p)
    grad = (4.0 * n / tau) * r
    
    return obj, grad

def softmax_obj_grad(v, tau):
    """Objective and gradient for L-BFGS in softmax parameterization."""
    n = len(v)
    a = softmax(v)
    obj, grad_a = smooth_obj_grad(a, tau)
    grad_v = a * (grad_a - np.dot(grad_a, a))
    return obj, grad_v

def run_lbfgs_continuation(v0, time_budget, verbose=True):
    """
    L-BFGS optimization with smooth continuation (decreasing tau).
    """
    n = len(v0)
    best_v = v0.copy()
    a = softmax(v0)
    c = fast_convolve(a)
    best_score = 2.0 * n * c.max()
    t0 = time.monotonic()
    
    # Tau schedule: start large, decrease geometrically
    # Start with tau such that the smooth approx is very flat
    c_range = c.max() - c.min()
    tau_start = max(c_range * 0.5, 1e-6)
    tau_end = max(c_range * 0.01, 1e-8)
    
    # Number of continuation steps
    n_steps = 8
    tau_schedule = np.geomspace(tau_start, tau_end, n_steps)
    
    time_per_step = time_budget / n_steps
    
    for step, tau in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_budget:
            break
        
        _current_tau = tau
        
        def fun(x):
            return softmax_obj_grad(x, _current_tau)
        
        try:
            result = minimize(fun, best_v, method='L-BFGS-B', jac=True,
                            options={'maxiter': 200, 'gtol': 1e-10, 'maxls': 50})
            
            if result.fun < float('inf'):
                a_new = softmax(result.x)
                c_new = fast_convolve(a_new)
                score_new = 2.0 * n * c_new.max()
                if score_new < best_score:
                    best_score = score_new
                    best_v = result.x.copy()
        except Exception as e:
            if verbose:
                print(f"    L-BFGS step {step} failed: {e}")
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS step={step} tau={tau:.6f} ({elapsed:.0f}s): best_score={best_score:.6f}")
    
    a_best = softmax(best_v)
    return a_best, best_score

def mirror_descent_v2(a0, time_budget, verbose=True):
    """
    Improved mirror descent with:
    - Adaptive number of active peaks
    - Better learning rate schedule
    - Periodic restart from running average
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.05
    no_improve = 0
    avg_a = a.copy().astype(np.float64)
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-9:
                break
        
        # Adaptive number of active peaks
        # Use peaks that are within 1% of max
        threshold = M * 0.999
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        if len(active) > 50:
            # Use top 50
            active = np.argpartition(conv, -50)[-50:]
        
        pv = conv[active]
        temp = 0.003 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(active)):
            if w[idx] < 1e-14:
                continue
            k = int(active[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        avg_a = (it * avg_a + a_new) / (it + 1)
        a = a_new
        conv = fast_convolve(a)
        
        # Periodically check the running average
        if it % 200 == 0:
            avg_score = 2.0 * n * fast_convolve(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD v2 it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def lp_alternating(a0, time_budget, verbose=True):

    LP-based alternating minimization.
    """

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    # Find a solver
    solver = None
    for s_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

        print(f"    LP solver: {solver}")
