    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def score_normalized(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def evaluate_safe(a):
    """Evaluate sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_fft_gradient(a, w):
    """Compute grad[i] = 2*sum_k w[k]*a[k-i] via FFT convolution trick.
    a: length n, w: length 2n-1. Returns: length n array.
    """
    n = len(a)
    a_flip = a[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a_flip, fft_size)
    W = rfft(w, fft_size)
    c = irfft(A * W, fft_size)[:conv_len]
    return 2.0 * c[n - 1:n - 1 + n]


def lbgfs_objective(z):
    """LSE-smoothed objective for L-BFGS (uses global _LBFGS_STATE)."""
    T = _LBFGS_STATE['T']
    n = _LBFGS_STATE['n']
    if time.monotonic() - _LBFGS_STATE['t0'] > _LBFGS_STATE['time_limit']:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * lse


def lbgfs_gradient(z):
    """Gradient of LSE-smoothed objective w.r.t. z (uses global _LBFGS_STATE)."""
    T = _LBFGS_STATE['T']
    n = _LBFGS_STATE['n']
    if time.monotonic() - _LBFGS_STATE['t0'] > _LBFGS_STATE['time_limit']:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    raw_grad = compute_fft_gradient(a, w)
    grad_a = n * raw_grad
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.001, n_stages=20):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    _LBFGS_STATE['n'] = n
    _LBFGS_STATE['t0'] = t0
    _LBFGS_STATE['time_limit'] = time_limit

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        _LBFGS_STATE['T'] = float(stage_T)

        maxiter = max(100, int(3000000 * remaining / max(1, n_stages - stage_idx)))
        maxiter = min(maxiter, 50000)

        try:
            res = minimize(
                fun=lbgfs_objective,
                x0=z,
                jac=lbgfs_gradient,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter * 300, 'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            # Always use the new z (even if not better in exact max, it's better in LSE)
            z = z_new
            current = score_normalized(softmax_from_z(z))
            print(f"    L-BFGS stage {stage_idx + 1}/{n_stages} T={stage_T:.4f} score={current:.8f} (best={best_score:.8f})")
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx + 1} error: {e}")

    a_final = softmax_from_z(z)
    sc_final = score_normalized(a_final)
    if sc_final < best_score:
        best_a = a_final
        best_score = sc_final
    return best_a, best_score


def fw_subgradient_refine(a0, time_limit, n_steps=50000, M_peaks=5):
    """Frank-Wolfe with away steps using multi-peak subgradient."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top M peaks (exclude edges which are naturally smaller)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_indices = np.argsort(b_work)[::-1][:M_peaks]

        # Compute average subgradient from top M peaks
        grad_a = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a[k - idx]
        grad_a /= M_peaks

        # FW toward step
        i_toward = int(np.argmin(grad_a))
        improved = False

        for t_val in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break

        # Away step
        if not improved:
            mask = a > 1e-10
            if mask.sum() > 2:
                g_masked = grad_a.copy()
                g_masked[~mask] = -np.inf
                i_away = int(np.argmax(g_masked))
                for gamma in [0.2, 0.1, 0.05, 0.02]:
                    a_trial = a.copy()
                    a_trial[i_away] *= (1.0 - gamma)
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            # Try pair moves on top contributing pairs of the main peak
            k_star = int(peak_indices[0])
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_star - idx]
            top_ci = np.argsort(contribs)[::-1][:4]
            pair_found = False
            for ci in top_ci:
                i = int(idx[ci])
                j = int(k_star - i)
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-12 or a[j] < 1e-12:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            pair_found = True
                            improved = True
                            break
                    if pair_found:
                        break
                if pair_found:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 500:
                break
        else:
            no_improve = 0

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=20000):
    """Direct coordinate descent targeting the main peak."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        k_star = int(np.argmax(b_work))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        top_coords = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_coords:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to low-contrib positions
                low_coords = np.argsort(contribs)[:n // 3]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-12
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 300:
                break
        else:
            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 3
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 10) * (1 + 0.2 * np.cos(6 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 5 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent or create structured init
    2. Aggressive L-BFGS (T: 50->0.5) for global structure
    3. Very aggressive L-BFGS (T: 0.5->0.001) for precision
    4. FW subgradient refinement
    5. Peak coordinate descent
    6. Multi-length exploration (n*1.1, n*1.25, n*1.5)
    7. Final polish on best
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Try multiple inits and pick best
        best_init = None
        best_init_score = float('inf')
        for n_try in [768, 960, 1024]:
            for mode in range(6):
                a_try = structured_init(n_try, rng, mode)
                sc = evaluate_safe(a_try)
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = a_try
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: L-BFGS high-to-mid temperature ===
    print(f"\n=== Phase 1: L-BFGS (T: 50->0.5) n={n_best} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_p1, _ = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=0.5, n_stages=15)
        try_update(a_p1, "lbgfs_p1")

    # === Phase 2: L-BFGS low temperature (precision) ===
    print(f"\n=== Phase 2: L-BFGS (T: 0.5->0.001) n={n_best} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, current best={best_score:.8f}")
        a_p2, _ = lbgfs_refine(best_seq, t_alloc, T_start=0.5, T_end=0.001, n_stages=18)
        try_update(a_p2, "lbgfs_p2")

    # === Phase 3: FW subgradient refinement ===
    print(f"\n=== Phase 3: FW Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_fw, _ = fw_subgradient_refine(best_seq, t_alloc, n_steps=50000, M_peaks=5)
        try_update(a_fw, "fw_subgrad")

    # === Phase 4: Peak coordinate descent ===
    print(f"\n=== Phase 4: Peak Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 80.0)
    if t_alloc > 10:
        a_cd, _ = peak_coordinate_descent(best_seq, t_alloc, n_steps=20000)
        try_update(a_cd, "peak_cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration (best={best_score:.8f}) ===")
    alt_lengths = []
    for mult in [1.1, 1.25, 1.5, 0.85, 1.33]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 5120 and abs(alt_n - n_best) > 16:
            alt_lengths.append(alt_n)
    for fixed_n in [1024, 1280, 1536, 2048]:
        if 256 <= fixed_n <= 5120 and abs(fixed_n - n_best) > 16:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.15, 200.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # L-BFGS on new length
        t_l1 = t_total * 0.6
        a_alt, _ = lbgfs_refine(init_alt, t_l1, T_start=10.0, T_end=0.01, n_stages=12)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_l2 = min(rem * 0.3, 60.0)
            if t_l2 > 10:
                a_alt2, _ = lbgfs_refine(best_seq, t_l2, T_start=1.0, T_end=0.005, n_stages=10)
                try_update(a_alt2, f"lbgfs2_n{alt_n}")

    # === Phase 6: Final aggressive L-BFGS polish on best ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.20, 200.0)
        if t_fin > 20:
            print(f"  Budget: {t_fin:.0f}s")
            a_fin, _ = lbgfs_refine(best_seq, t_fin, T_start=2.0, T_end=0.001, n_stages=15)
            try_update(a_fin, "final_lbgfs")

    # === Phase 7: Final FW + coordinate ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fw = min(rem * 0.15, 60.0)
        if t_fw > 10:
            a_fw2, _ = fw_subgradient_refine(best_seq, t_fw, n_steps=20000, M_peaks=3)
            try_update(a_fw2, "final_fw")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.15, 50.0)
        if t_cd > 10:
            a_cd2, _ = peak_coordinate_descent(best_seq, t_cd, n_steps=10000)
            try_update(a_cd2, "final_cd")

    # === Phase 8: One more L-BFGS at very low T ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_vl = min(rem * 0.25, 80.0)
        if t_vl > 10:
            a_vl, _ = lbgfs_refine(best_seq, t_vl, T_start=0.1, T_end=0.001, n_stages=10)
            try_update(a_vl, "very_low_T")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
