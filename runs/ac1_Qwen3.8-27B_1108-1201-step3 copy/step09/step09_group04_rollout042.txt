def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the score 2*n*max_conv/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)

def p_norm_obj_and_grad(a, p):
    """Compute p-norm of positive part of convolution and its gradient w.r.t. a."""

            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            return 2.0 * n * (np.sum(c_pos ** p)) ** (1.0 / p)
        
        def jac(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            c_p = c_pos ** p
            S = np.sum(c_p)
            if S < 1e-300:
                return np.zeros_like(z_val)
            factor = p * (2.0 * n) * S ** (1.0 / p - 1.0)
            w = c_pos ** (p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            g_a = factor * corr
            # Gradient w.r.t. z via softmax
            return a * (g_a - np.dot(g_a, a))
        
        try:
            res = minimize(func, z, jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                                   'maxfun': maxiter * 300, 'disp': False})
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        score = compute_score(a)
        print(f"    p={p}: score={score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def subgradient_nesterov(a0, time_limit, n_iters=5000):
    """Nesterov accelerated subgradient method for minimizing max conv."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()
    momentum = 1.0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient (subgradient) at y
        c = fast_convolve(y)
        peak_idx = np.argmax(c)
        n_local = len(y)
        # Gradient of conv[peak_idx] w.r.t. y
        js_start = max(0, peak_idx - n_local + 1)
        js_end = min(n_local - 1, peak_idx)
        idx = np.arange(js_start, js_end + 1)
        grad = np.zeros(n_local)
        grad[idx] = y[peak_idx - idx]
        # Also the symmetric part
        idx2 = peak_idx - idx
        valid = (idx2 >= 0) & (idx2 < n_local)
        if np.any(valid):
            grad[idx2[valid]] += y[idx[valid]]
        
        # Step size
        alpha = 2.0 / (it + 2.0)
        
        # Nesterov update
        new_y = (1 - alpha) * y + alpha * a  # This is actually the momentum term
        # Projected gradient step on a
        a_new = a - alpha * 0.1 * grad  # Small step
        # Actually, let's use a proper step size
        # ||grad||_2
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            continue
        step = alpha * 0.5 / grad_norm  # Normalized step
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y_new = a_new + (momentum / t_k) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y
        else:
            y_new = a_new
        
        momentum = t_k
        a = a_new
        y = y_new
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if (it + 1) % 1000 == 0:
            print(f"    SubGrad iter={it+1}: score={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse-like convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_best_response(a_fixed, solver=None):
    """LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternating(a0, time_limit, solver=None, n_iters=30000):
    """LP-based alternating optimization with Nesterov acceleration."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2.0)
        
        # Nesterov-type update
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        # Fenchel dual averaging for a
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = compute_score(a)
        score_y = compute_score(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 2000 == 0:
            print(f"    LP iter={it+1}: score={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score


def local_refine(a0, time_limit, n_steps=50000):
    """Aggressive local refinement with peak shaving."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find the peak
        peak_val = c.max()
        peak_idx = np.argmax(c)
        
        # Get active set (lags near the peak)
        threshold = peak_val * 0.95
        active = np.where(c >= threshold)[0]
        
        improved = False
        
        # For each active lag, find the top contributing pairs and reduce them
        for k in active[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top = np.argsort(contribs)[::-1][:5]
            
            for ci in top:
                i = idx[ci]
                j = k - i
                if i == j or j < 0 or j >= n or a[i] < 1e-12 or a[j] < 1e-12:
                    continue
                
                for f in [0.93, 0.90, 0.87, 0.83]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing the largest single element
            max_idx = np.argmax(a)
            if a[max_idx] < 1e-12:
                break
            for f in [0.95, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[max_idx] *= f
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 0.01:
                    continue
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Random perturbation every 500 steps
            if step % 500 == 0 and step > 0:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(3, 10)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.1, 0.1))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            
            if not improved and step % 2000 == 1999:
                break
        
        if step % 10000 == 0 and step > 0:
            print(f"    Local step={step}: score={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def frequency_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    # FFT
    A = np.fft.rfft(a)
    # Perturb the phase (but keep magnitude mostly the same)
    # Actually, for non-negative a, the phase is fixed by a.
    # Instead, perturb the magnitude at high frequencies
    noise = rng.standard_normal(len(A)) * strength
    # Only perturbe high frequencies
    cutoff = len(A) // 4
    noise[:cutoff] = 0
    A_pert = A + 1j * noise
    a_pert = np.fft.irfft(A_pert, n)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s > 0.01:
        a_pert = a_pert / s
    return a_pert


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence."""
    if rng is None:
        rng = np.random.default_rng(42)
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif variant == 2:
        # Bimodal at edges
        x = np.linspace(-1, 1, n)
        a = np.exp(-((np.abs(x) - 0.6) ** 2) * 10)
    elif variant == 3:
        # Asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) * 8) + 0.5 * np.exp(-((x - 0.7) ** 2) * 8)
    elif variant == 4:
        # Square-ish with rounded edges
        x = np.linspace(0, 1, n)
        a = np.minimum(1.0, np.minimum(5*x, 5*(1-x)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive multi-phase optimization:
(1) Very high p-norm L-BFGS (p up to 5000) approaching true max optimization,
(2) LP-based alternating with Nesterov dual averaging,

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def p_norm_smoothing_z(z, p, n):
    """Objective: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_gradient_z(z, p, n):
    """Gradient w.r.t. z of 2n * (sum c_k^p)^(1/p) with a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = np.sum(c_p)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = p * (2.0 * n) * S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_stages_func(z, p, n):
    return p_norm_smoothing_z(z, p, n)


def p_norm_stages_jac(z, p, n):
    return p_norm_gradient_z(z, p, n)


def p_norm_lbfgs_optimize(a0, time_limit, p_list):
    """L-BFGS with schedule of p values in log-domain (softmax parameterization)."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    for p in p_list:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        maxiter = max(50, int(min(remaining * 25, 2000)))

        try:
            res = minimize(
                fun=p_norm_stages_func,
                x0=z,
                args=(p, n),
                jac=p_norm_stages_jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': maxiter * 400, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        score = compute_score(a)
        print(f"    p={p}: score={score:.7f} ({time.monotonic()-t0:.1f}s)")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def build_conv_matrix(a):
    """Build convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

        except Exception:

def solve_best_response_lp(a_fixed, solver):
    """LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_nesterov_optimize(a0, time_limit, solver, n_iters=100000):
    """Nesterov dual-averaging LP alternating optimization."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    y = a.copy()
    stall = 0
    prev_best = best_score

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_best_response_lp(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        denom = (it + 1) + alpha * (it + 2)
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / denom
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = compute_score(a)
        sc_y = compute_score(y)

        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 5:
                rng = np.random.default_rng(it)
                y = y + 0.03 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
                prev_best = best_score
            print(f"    LP iter={it+1}: best={best_score:.7f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score

        threshold = peak_val * 0.97

                    a_trial = a.copy()
                    a_trial[i] *= factor
                    if i != j and 0 <= j < n and a_trial[j] > 0:
                        a_trial[j] *= 0.97
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Strategy 2: Reduce single largest element
        if not improved:
            max_i = np.argmax(a)
            if a[max_i] > 1e-12:
                for factor in [0.96, 0.93, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[max_i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 3: Spread from peak to trough
        if not improved:
            max_i = np.argmax(a)
            min_i = np.argmin(a)
            if a[max_i] > 1e-10 and a[min_i] < a[max_i] * 0.1:
                for frac in [0.02, 0.05, 0.1]:
                    transfer = a[max_i] * frac
                    a_trial = a.copy()
                    a_trial[max_i] -= transfer
                    a_trial[min_i] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
            # Periodic perturbation
            if no_improve % 500 == 0:
                rng = np.random.default_rng(step * 7 + 3)
                n_pert = rng.integers(5, 20)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_pert = a.copy()
                for i in indices:
                    a_pert[i] *= (1.0 + rng.uniform(-0.15, 0.15))
                a_pert = np.maximum(a_pert, 0)
                st = a_pert.sum()
                if st > 0.01:
                    a_pert = a_pert / st
                    sc = compute_score(a_pert)
                    if sc < best_score:
                        a = a_pert
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0

        if step % 20000 == 0 and step > 0:
            print(f"    Local step={step}: best={best_score:.7f} ({time.monotonic()-t0:.0f}s}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def freq_perturb(a, rng, strength):
    """Perturb in frequency domain (high-frequency noise)."""
    n = len(a)
    A = np.fft.rfft(a)
    noise = rng.standard_normal(len(A)) * strength
    cutoff = max(1, len(A) // 5)
    noise[:cutoff] *= 0.1
    A_pert = A + 1j * noise
    a_pert = np.fft.irfft(A_pert, n)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s > 0.01:
        a_pert = a_pert / s
    return a_pert


def structured_init(n, variant, rng):
    """Generate structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4.0)
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-((np.abs(x) - 0.7) ** 2) * 12.0)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) * 10.0) + 0.6 * np.exp(-((x - 0.7) ** 2) * 10.0)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = np.minimum(1.0, np.minimum(4.0 * x, 4.0 * (1.0 - x)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update_state(best_score, best_seq, a, label):
    """Try to update best state. Returns (new_best_score, new_best_seq)."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return best_score, best_seq
    a = a / s
    sc = compute_score(a)
    if sc < best_score - 1e-13:
        sc_real = evaluate_safe(a)
        if np.isfinite(sc_real) and sc_real < best_score:
            print(f"  NEW BEST [{label}]: {sc_real:.7f} (n={len(a)})")
            return sc_real, a.copy()
    return best_score, best_seq


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: aggressive p-norm L-BFGS (p up to 5000),
    LP Nesterov alternating, targeted peak-shaving, frequency perturbation,
    and systematic multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, 1, rng)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Generated init: n={n_init}, score={best_score:.7f}")

    n_cur = len(best_seq)

    # === Phase 1: Aggressive p-norm L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: p-norm L-BFGS (best={best_score:.7f}, n={n_cur}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        p_list = [10, 20, 35, 50, 75, 100, 150, 200, 300, 500, 750, 1000, 2000, 3000, 5000]
        a_pn = p_norm_lbfgs_optimize(best_seq, t_alloc, p_list)
        best_score, best_seq = try_update_state(best_score, best_seq, a_pn, "pn_main")

    # === Phase 2: LP Nesterov ===
    print(f"\n=== Phase 2: LP Nesterov (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 20:
        a_lp, _ = lp_nesterov_optimize(best_seq, t_alloc, solver, n_iters=80000)
        best_score, best_seq = try_update_state(best_score, best_seq, a_lp, "lp_1")

    # === Phase 3: Local peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 10:
        a_lr, _ = local_peak_shave(best_seq, t_alloc, n_steps=80000)
        best_score, best_seq = try_update_state(best_score, best_seq, a_lr, "local_1")

    # === Phase 4: p-norm round 2 from LP result ===
    print(f"\n=== Phase 4: p-norm r2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        p_list2 = [200, 500, 1000, 2000, 4000, 8000]
        a_pn2 = p_norm_lbfgs_optimize(best_seq, t_alloc, p_list2)
        best_score, best_seq = try_update_state(best_score, best_seq, a_pn2, "pn_r2")

    # === Phase 5: Frequency perturbation + re-optimize ===
    print(f"\n=== Phase 5: Freq Perturbation (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for strength in [0.03, 0.08, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = freq_perturb(best_seq, rng, strength)
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 15:
                break
            a_opt = p_norm_lbfgs_optimize(a_pert, t_sub, [50, 100, 300, 600, 1200])
            best_score, best_seq = try_update_state(best_score, best_seq, a_opt, f"freq_{strength}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.7f}) ===")
    n_cur = len(best_seq)
    candidate_ns = sorted(set(
        [n_cur + d for d in [-128, -96, -64, -48, -32, 32, 48, 64, 96, 128]
         if 300 <= n_cur + d <= 2000] +
        [512, 640, 768, 832, 896, 960, 1024, 1088, 1152, 1280, 1408, 1536]
    ), key=lambda x: abs(x - n_cur))

    n_explored = 0
    for alt_n in candidate_ns:
        rem = deadline - time.monotonic()
        if rem < 70 or n_explored >= 4:
            break
        if alt_n == n_cur or abs(alt_n - n_cur) < 16:
            continue
        n_explored += 1
        t_total = min(rem * 0.16, 160.0)
        if t_total < 35:
            break

        print(f"  n={alt_n} (budget={t_total:.0f}s)")
        a_interp = interpolate_seq(best_seq, alt_n)

        # Quick p-norm
        t_pn = t_total * 0.55
        a_opt = p_norm_lbfgs_optimize(a_interp, t_pn, [30, 60, 100, 200, 500, 1000])
        best_score, best_seq = try_update_state(best_score, best_seq, a_opt, f"len_{alt_n}_pn")

        # If improved, LP polish
        rem = deadline - time.monotonic()
        if rem > 25:
            t_lp = min(rem * 0.2, 80.0)
            a_lp2, _ = lp_nesterov_optimize(best_seq, t_lp, solver, n_iters=15000)
            best_score, best_seq = try_update_state(best_score, best_seq, a_lp2, f"len_{alt_n}_lp")

    # === Phase 7: Final aggressive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.3, 100.0)
        p_list_fin = [500, 1000, 2000, 5000, 10000]
        a_fin = p_norm_lbfgs_optimize(best_seq, t_fin, p_list_fin)
        best_score, best_seq = try_update_state(best_score, best_seq, a_fin, "final_pn")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_lp = min(rem * 0.3, 70.0)
        a_lp3, _ = lp_nesterov_optimize(best_seq, t_lp, solver, n_iters=30000)
        best_score, best_seq = try_update_state(best_score, best_seq, a_lp3, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.4, 50.0)
        a_lr2, _ = local_peak_shave(best_seq, t_lr, n_steps=50000)
        best_score, best_seq = try_update_state(best_score, best_seq, a_lr2, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    return [float(max(0.0, x)) for x in best_seq]
