def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def power_sum_lbfgs(a0, time_limit, p_start=50, p_end=500, n_stages=8):
    """L-BFGS with power-sum smoothing and increasing p."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    ps = np.geomspace(p_start, p_end, n_stages)
    
    for p in ps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            val, _ = power_sum_obj_and_grad(a, p)
            return val
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            _, grad_a = power_sum_obj_and_grad(a, p)
            # Transform gradient from a-space to z-space
            # a = softmax(z), so da_i/dz_j = a_i * (delta_{ij} - a_j)
            # d obj / d z_j = sum_i (d obj / d a_i) * da_i / d z_j
            # = sum_i grad_a_i * a_i * (delta_{ij} - a_j)
            # = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 500000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def subgradient_descent(a0, time_limit, n_steps=50000):
    """Subgradient method with line search for the max-convolution objective."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Subgradient at peak
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        g[idx] = a[peak_k - idx]
        
        # Polyak step size or diminishing
        alpha = 1.0 / (step + 10)
        
        # Try multiple step sizes
        improved = False
        for t_val in [alpha, alpha * 0.5, alpha * 2, 0.01, 0.005, 0.001]:
            a_trial = a - t_val * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score * (1 - 1e-10):
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try moving mass from high-gradient to low-gradient positions
            # This is a Frank-Wolfe-like step
            i_from = int(np.argmax(g * a))  # position with most "contribution" to peak
            i_to = int(np.argmin(g))  # position with least gradient
            if a[i_from] > 1e-10 and i_from != i_to:
                for frac in [0.1, 0.05, 0.02, 0.01]:
                    transfer = a[i_from] * frac
                    a_trial = a.copy()
                    a_trial[i_from] -= transfer
                    a_trial[i_to] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score * (1 - 1e-10):
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break
    
    return best_a, best_score

            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxfun': 1000000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def lattice_optimize(a0, time_limit, n_steps=30000):
    """
    Optimizes by adjusting pairs of coordinates to flatten the convolution.
    Uses a "lattice" structure: identify the top contributing pairs at the peak
    and adjust them to reduce the peak while maintaining the sum.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stagnation = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Get the pairs that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Sort by contribution
        order = np.argsort(contribs)[::-1]
        
        improved = False
        for oi in order[:8]:
            i = idx[oi]
            j = peak_k - i
            if i == j or i >= n or j >= n:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Try to reduce the product a[i]*a[j] while maintaining sum
            # Option 1: reduce both proportionally, redistribute to other positions
            for f in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                # Redistribute the saved mass to positions that don't contribute to peak
                saved = a[i] * (1-f) + a[j] * (1-f)
                # Find positions not in the peak window
                peak_mask = np.zeros(n, dtype=bool)
                peak_mask[peak_mask] = False  # reset
                # Positions that affect the peak: idx range
                peak_mask[js:je+1] = True
                non_peak = ~peak_mask
                if non_peak.sum() > 0:
                    # Distribute to non-peak positions proportional to current value
                    target = a_trial[non_peak] + 1e-10
                    a_trial[non_peak] = target + saved * target / target.sum()
                else:
                    # Distribute to low-contribution positions
                    low_contrib = (a * a) < np.median(a * a)
                    if low_contrib.sum() > 0:
                        target = a_trial[low_contrib] + 1e-10
                        a_trial[low_contrib] = target + saved * target / target.sum()
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score * (1 - 1e-12):
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try single coordinate moves
            g = np.zeros(n)
            g[js:je+1] = a[peak_k - idx]
            
            # Move from high-g to low-g
            i_from = int(np.argmax(g * a))
            i_to = int(np.argmin(g))
            
            if a[i_from] > 1e-10 and i_from != i_to:
                for t in [0.05, 0.03, 0.01, 0.005, 0.001]:
                    transfer = a[i_from] * t
                    a_trial = a.copy()
                    a_trial[i_from] -= transfer
                    a_trial[i_to] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score * (1 - 1e-12):
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 500:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, mode=0):
    """Create various structured initial sequences."""
    if mode == 0:
        # Cosine squared
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif mode == 2:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 3
    elif mode == 3:
        # Optimal for flat convolution (theoretical)
        # a_i proportional to sin(pi*(i+0.5)/n) for flat autocorrelation
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.abs(a) ** 0.5
    elif mode == 4:
        # Bivariate normal slice
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 4)
    elif mode == 5:
        # Optimal when only considering the center of convolution
        # We want sum a_i^2 minimized => uniform
        # But boundaries matter, so use a "tapered uniform"
        a = np.ones(n)
        taper = n // 10
        if taper > 0:
            a[:taper] = np.linspace(0.5, 1, taper)
            a[-taper:] = np.linspace(1, 0.5, taper)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

2. Power-sum smoothed L-BFGS for different gradient profiles

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, T):
    """Compute LSE-smoothed objective and gradient w.r.t. z (log-domain)."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj = (2.0 * n / T) * (max_Tb + np.log(exp_Tb.sum()))
    
    w = exp_Tb / exp_Tb.sum()
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def power_sum_obj_grad_z(z, p):
    """Compute power-sum smoothed objective and gradient w.r.t. z (log-domain)."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    if b_max < 1e-300:
        return 0.0, np.zeros(n)
    
    b_scaled = b / b_max
    bp = np.power(b_scaled, p)
    log_sum = np.log(bp.sum())
    obj = 2.0 * n * b_max * np.exp(log_sum / p)
    
    # Gradient w.r.t. a:
    # d/da_i of (sum b^p)^(1/p) = (sum b^p)^(1/p-1) * sum_k b_k^(p-1) * db_k/da_i
    # Using scaled: coeff * sum_k (b_k/b_max)^(p-1) * a[k-i]
    b_p_minus_1 = np.power(b_scaled, p - 1)
    coeff = bp.sum() ** (1.0 / p - 1.0)
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if b_p_minus_1[k] < 1e-30:

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += coeff * b_p_minus_1[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def power_sum_lbfgs(a0, time_limit, p_start=50, p_end=1000, n_stages=8, maxiter_per_stage=2000):
    """L-BFGS with power-sum smoothing and increasing exponent."""

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    ps = np.geomspace(p_start, p_end, n_stages)
    
    for p in ps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            b_max = b.max()
            if b_max < 1e-300:
                return 0.0
            b_scaled = b / b_max
            bp = np.power(b_scaled, p)
            return 2.0 * n * b_max * np.exp(np.log(bp.sum()) / p)
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            n_local = len(a)
            b = fast_convolve(a)
            b_max = b.max()
            if b_max < 1e-300:
                return np.zeros(n_local)
            b_scaled = b / b_max
            b_p_minus_1 = np.power(b_scaled, p - 1)
            bp = np.power(b_scaled, p)
            coeff = bp.sum() ** (1.0 / p - 1.0)
            grad_a = np.zeros(n_local)
            for k in range(len(b)):
                if b_p_minus_1[k] < 1e-30:
                    continue
                js = max(0, k - n_local + 1)
                je = min(n_local - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += coeff * b_p_minus_1[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def lattice_optimize(a0, time_limit, n_steps=50000):
    """
    Optimizes by adjusting pairs of coordinates to reduce the peak convolution.
    Targets the top-contributing pairs at the peak position.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stagnation = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        order = np.argsort(contribs)[::-1]
        
        improved = False
        max_oi = min(10, len(order))
        for oi in range(max_oi):
            i = idx[order[oi]]
            j = peak_k - i
            if i == j or i >= n or j >= n:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Reduce the pair and redistribute
            for f in [0.96, 0.93, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                saved = a[i] * (1 - f) + a[j] * (1 - f)
                
                # Redistribute to positions far from peak contribution
                # Positions that contribute to peak: [js, je]
                # Move mass to positions outside this range or to low-contribution positions
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0 and saved > 1e-15:
                    target = a_trial[non_peak] + 1e-10
                    a_trial[non_peak] += saved * target / target.sum()
                elif saved > 1e-15:
                    # Redistribute to lowest (a_i)^2 positions
                    a_sq = (a_trial * a_trial).copy()
                    a_sq[i] = np.inf
                    a_sq[j] = np.inf
                    low_positions = np.argsort(a_sq)[:n // 3]
                    target = a_trial[low_positions] + 1e-10
                    a_trial[low_positions] += saved * target / target.sum()
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try single coordinate moves: reduce high-grad, increase low-grad
            g = np.zeros(n)
            g[js:je + 1] = a[peak_k - idx]
            
            # Find best (i_from, i_to) pair
            i_from = int(np.argmax(g * a))
            i_to = int(np.argmin(g))
            
            if a[i_from] > 1e-10 and i_from != i_to:
                for t in [0.08, 0.05, 0.03, 0.01, 0.005, 0.002]:
                    transfer = a[i_from] * t
                    a_trial = a.copy()
                    a_trial[i_from] -= transfer
                    a_trial[i_to] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            
            if not improved:
                # Try triple move: reduce 2 peak contributors, increase 1 non-contributor
                top2 = np.argsort(g * a)[::-1][:2]
                bot1 = np.argsort(g)[:1]
                if len(top2) == 2 and a[top2[0]] > 1e-10 and a[top2[1]] > 1e-10:
                    i1, i2 = top2
                    j1 = int(bot1[0])
                    if i1 != j1 and i2 != j1:
                        for t in [0.04, 0.02, 0.01]:
                            a_trial = a.copy()
                            a_trial[i1] *= (1 - t)
                            a_trial[i2] *= (1 - t)
                            a_trial[j1] += a[i1] * t + a[i2] * t
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
        
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 800:
                break
    
    return best_a, best_score
