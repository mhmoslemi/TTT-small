def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def soft_max_weights(c, beta):
    """Compute softmax weights on convolution values."""
    max_c = c.max()
    scaled = beta * (c - max_c)
    scaled = np.clip(scaled, -100, 0)
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= s
    return w

def compute_gradient(a, w):
    """
    Compute grad[i] = 2 * sum_j a[j] * w[i+j]
    Using convolution: grad[i] = 2 * conv(a[::-1], w, 'full')[i + n - 1]
    """
    n = len(a)
    rev_a = a[::-1]
    conv_result = irfft(rfft(rev_a, 4*n) * rfft(w, 4*n), 4*n)[:3*n-2]
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad

def multiplicative_gradient_descent(a, time_limit, beta_init=20.0, lr_init=0.5, 
                                      symmetric=True, anneal=True, verbose=False):
    """
    Weiszfeld-style multiplicative gradient descent.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    noise_scale = 0.1 if anneal else 0.0
    rng = np.random.default_rng(123)
    no_improve = 0
    
    half_n = (n + 1) // 2 if symmetric else n
    use_sym = symmetric
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Get working vector (first half if symmetric)
        if use_sym:
            a_work = a[:half_n].copy()
        else:
            a_work = a.copy()
        
        # Compute convolution and weights
        c = fast_convolve(a)
        
        # Adaptive beta: increase over time
        beta = beta_init * (1.0 + 0.01 * it)
        beta = min(beta, 500.0)
        
        w = soft_max_weights(c, beta)
        
        # Compute gradient for working variables
        grad_full = compute_gradient(a, w)
        
        if use_sym:
            # Average gradient with its mirror
            grad = (grad_full[:half_n] + grad_full[n - half_n:][::-1]) / 2.0
        else:
            grad = grad_full
        
        # Multiplicative update
        # a_new[i] = a[i] * (mean_grad / grad[i])^lr
        mean_grad = grad.mean()
        if mean_grad < 1e-30:
            break
        
        ratio = mean_grad / (grad + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        
        a_work *= np.power(ratio, lr)
        
        # Add noise for exploration
        if anneal and noise_scale > 0.001:
            noise = rng.standard_normal(len(a_work))
            a_work *= np.exp(noise_scale * noise)
            noise_scale *= 0.995
        
        # Project to non-negative
        a_work = np.maximum(a_work, 0.0)
        
        # Enforce symmetry and normalize
        if use_sym:
            a = np.zeros(n)
            a[:half_n] = a_work
            if n % 2 == 0:
                a[half_n:] = a_work[::-1]
            else:
                a[half_n:] = a_work[-2::-1]
        else:
            a = a_work
        
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
            s = 1.0
        a = a / s

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.9
            if lr < 0.01:
                break
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MGD it={it} ({elapsed:.0f}s): score={best_score:.6f} beta={beta:.1f} lr={lr:.4f}")
    
    return best_a, best_score

    # Extract first half
    a_half = a[:half_n].copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    noise_scale = 0.05 if anneal else 0.0
    rng = np.random.default_rng(42)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Reconstruct full a from half
        a = np.concatenate([a_half, a_half[::-1]])
        
        # Compute convolution
        c = fast_convolve(a)
        
        # Adaptive beta
        beta = min(beta_init * (1.0 + 0.005 * it), 300.0)
        
        # Compute weights
        w = soft_max_weights(c, beta)
        
        # Compute gradient (full)
        grad_full = compute_conv_gradient(a, w)
        
        # Average with mirror for symmetric constraint
        grad_half = (grad_full[:half_n] + grad_full[-half_n:][::-1]) / 2.0
        
        # Multiplicative update
        mean_grad = grad_half.mean()
        if mean_grad < 1e-30:
            break
        
        ratio = mean_grad / (grad_half + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        
        a_half = a_half * np.power(ratio, lr)
        
        # Add noise
        if anneal and noise_scale > 0.0005:
            noise = rng.standard_normal(half_n)
            a_half *= np.exp(noise_scale * noise)
            noise_scale *= 0.998
        
        # Non-negativity
        a_half = np.maximum(a_half, 0.0)
        
        # Normalize (sum of full = 1, so sum of half = 0.5)
        s_half = a_half.sum()
        if s_half < 1e-30:
            a_half = np.ones(half_n) / half_n
            s_half = 0.5
        a_half = a_half * (0.5 / s_half)
        
        # Evaluate
        a_full = np.concatenate([a_half, a_half[::-1]])
        score = 2.0 * n * fast_convolve(a_full).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_full.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            if lr < 0.005:
                break
            no_improve = 0
        elif it % 200 == 0 and no_improve < 50:
            # Occasionally increase lr for exploration
            lr = min(lr * 1.1, lr_init)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MGD {label} it={it} ({elapsed:.0f}s): {best_score:.6f} beta={beta:.0f} lr={lr:.4f}")
    
    return best_a, best_score


def mgd_general(a_init, time_limit, beta_init=20.0, lr_init=0.3,
                anneal=True, verbose=False, label=""):
    """Multiplicative gradient descent without symmetry constraint."""
    n = len(a_init)
    
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    noise_scale = 0.05 if anneal else 0.0
    rng = np.random.default_rng(77)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        c = fast_convolve(a)
        beta = min(beta_init * (1.0 + 0.005 * it), 300.0)
        w = soft_max_weights(c, beta)
        grad = compute_conv_gradient(a, w)
        
        mean_grad = grad.mean()
        if mean_grad < 1e-30:
            break
        
        ratio = mean_grad / (grad + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        
        a = a * np.power(ratio, lr)
        
        if anneal and noise_scale > 0.0005:
            noise = rng.standard_normal(n)
            a *= np.exp(noise_scale * noise)
            noise_scale *= 0.998
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
        else:
            a /= s
        
        score = 2.0 * n * fast_convolve(a).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.9
            if lr < 0.005:
                break
            no_improve = 0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MGDgen {label} it={it} ({elapsed:.0f}s): {best_score:.6f}")
    
    return best_a, best_score


def coord_descent_polish(a_init, time_limit, verbose=False, label=""):
    """
    Peak-directed coordinate descent for final polishing.
    For each coordinate, finds the optimal 1D step to minimize the max convolution.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            # Affected convolution indices
            k_lo = i
            k_hi = min(i + n - 1, 2*n - 2)
            
            # Unaffected max
            if k_lo > 0:
                unaff_left = conv[:k_lo].max()
            else:
                unaff_left = -np.inf
            if k_hi < 2*n - 2:
                unaff_right = conv[k_hi+1:].max()
            else:
                unaff_right = -np.inf
            unaff_max = max(unaff_left, unaff_right)
            
            # Affected portion
            affected = conv[k_lo:k_hi+1].copy()
            
            # The change in conv when a[i] -> a[i] + d:
            # conv[k] += 2*d*a[k-i] + d^2 (if k=2i)
            # conv[k] += 2*d*a[k-i] (otherwise)
            
            # a[k-i] for k in [k_lo, k_hi] is a[0], a[1], ..., a[n-1]
            a_contrib = a[0:n]  # a[k-i] for k = i, i+1, ..., i+n-1
            
            # Find optimal d by bisection/ternary search on the 1D function
            # f(d) = max(unaff_max, max(affected + 2*d*a_contrib + d^2*delta_{2i}))
            
            # The affected + 2*d*a_contrib is linear in d (plus d^2 at one point)
            # So f(d) is the max of an affine function and a constant (and one quadratic)
            
            # For efficiency, try a set of candidate d values
            # The optimal d is where the "top" of the affected peak is just at level
            # max(unaff_max, current_max - something)
            
            # Simple approach: try to reduce the max
            # Current max in affected region
            aff_max = affected.max()
            
            if aff_max <= unaff_max + 1e-15:
                # This coordinate doesn't affect the global max, try to increase it
                continue
            
            # The peak position within affected
            peak_idx = affected.argmax()
            peak_k = k_lo + peak_idx
            
            # Try several step sizes
            best_d = 0.0
            best_new_max = M
            
            for frac in [0.5, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5]:
                d = -frac * a[i]
                new_val = a[i] + d
                if new_val < 0:
                    continue
                
                new_aff = affected + 2.0 * d * a_contrib
                if peak_k == 2*i:
                    new_aff[peak_idx] += d * d
                
                new_max = max(unaff_max, new_aff.max())
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_d = d
            
            # Also try positive steps (increase)
            for frac in [0.1, 0.3, 0.5]:
                d = frac * a[i]
                new_aff = affected + 2.0 * d * a_contrib
                if peak_k == 2*i:
                    new_aff[peak_idx] += d * d
                
                new_max = max(unaff_max, new_aff.max())
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_d = d
            
            if best_d != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_d, 0.0)
                s = a.sum()
                
                # Update convolution
                conv[k_lo:k_hi+1] += 2.0 * best_d * a_contrib
                if 2*i >= k_lo and 2*i <= k_hi:
                    conv[2*i] += best_d * best_d
                
                M = max(unaff_max, conv[k_lo:k_hi+1].max())
                score = 2.0 * n * M / (s * s)
                
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD {label} sweep={sweep} ({elapsed:.0f}s): {best_score:.6f}")
    
    return best_a, best_score


def init_hann(n):
    k = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * k / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_supergaussian(n, p=2.5):
    k = np.arange(n)
    t = 2.0 * k / (n - 1) - 1.0
    a = np.maximum(1.0 - np.abs(t)**p, 0.0) ** (1.0/p)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_flattop(n, taper=0.1):
    k = np.arange(n)
    a = np.ones(n)
    taper_n = int(taper * n)
    for i in range(taper_n):
        t = (i + 1) / taper_n
        val = 0.5 * (1.0 - np.cos(np.pi * t))
        a[i] = val
        a[n-1-i] = val
    return a / a.sum()


def init_cosine_comb(n, n_harmonics=3):
    """Superposition of cosines to create a shape with low autocorrelation sidelobes."""
    k = np.arange(n)
    t = k / n
    a = np.ones(n)
    # Add cosine components with carefully chosen frequencies
    freqs = [2, 3, 5]  # These create "gaps" in the spectrum
    amps = [0.3, 0.2, 0.1]
    for f, amp in zip(freqs[:n_harmonics], amps[:n_harmonics]):
        a += amp * np.cos(2.0 * np.pi * f * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_parent_interp(parent, n):
    if parent is None:
        return np.ones(n) / n
    old_n = len(parent)
    if old_n == n:
        return parent.copy()
    old_t = np.linspace(0, 1, old_n)
    new_t = np.linspace(0, 1, n)
    a = np.interp(new_t, old_t, parent)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_perturb_symmetric(a, rng, strength=0.1):
    """Perturb a symmetric sequence while maintaining symmetry."""
    n = len(a)
    half_n = n // 2
    a_half = a[:half_n].copy()
    noise = rng.standard_normal(half_n) * strength
    a_half = a_half + noise
    a_half = np.maximum(a_half, 0.0)
    s = a_half.sum()
    if s < 1e-15:
        return a.copy()
    a_half = a_half * (0.5 / s)
    return np.concatenate([a_half, a_half[::-1]])


def random_perturb_general(a, rng, strength=0.1):
    """Perturb a general sequence."""
    n = len(a)
    noise = rng.standard_normal(n) * strength * a.mean()
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def peak_directed_perturb(a, rng, strength=0.05):
    """Perturb coordinates that contribute most to the peak."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    
    # Find peak positions
    threshold = M * 0.9
    peak_pos = np.where(c >= threshold)[0]
    
    # Compute contribution of each coordinate to peaks
    contrib = np.zeros(n)
    for k in peak_pos:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            contrib[i] += a[i] * a[k - i]
    
    # Normalize
    c_max = contrib.max()
    if c_max > 1e-30:
        contrib /= c_max
    
    # Reduce high-contribution coords, increase low
    a_new = a.copy()
    reduce_mask = contrib > 0.7
    increase_mask = contrib < 0.3
    a_new[reduce_mask] *= (1.0 - strength)
    a_new[increase_mask] *= (1.0 + strength * 0.5)
    
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
