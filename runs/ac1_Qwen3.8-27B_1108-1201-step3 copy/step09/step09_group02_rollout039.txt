import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def init_polynomial_ramp(n, rng, degree=1):
    """Polynomial ramp initialization with random asymmetry."""
    t = np.linspace(0, 1, n)
    # Random polynomial of given degree
    coeffs = [1.0] + [rng.uniform(-0.3, 0.3) for _ in range(degree)]
    a = np.ones(n)
    for i, c in enumerate(coeffs[1:], 1):
        a += c * (t - 0.5) ** i
    # Ensure positivity
    a_min = a.min()
    if a_min < 0:
        a -= a_min + 0.01
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_quadratic_bowl(n, rng, invert=False):

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    return a / s

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()

    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section 1D search.
    For each coordinate, finds the value that minimizes the full score.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            # Adaptive search range: wider when value is small
            upper = max(3.0 * v, 4.0 / n, 0.5 * a.max())
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

                    new_sum = current_sum + di + dj
                    if new_sum < 0.01:
                        continue
                    new_score = 2.0 * n * max(unaff_max, float(nv.max())) / (new_sum * new_sum)

                    if new_score < best_local + unaff_max and new_score < 2.0 * n * max(unaff_max, float(nv.max())) / (new_sum * new_sum):
                        pass  # just track
                    # Actually just track the best
                    if new_score < best_score - 1e-14 or (best_di == 0 and best_dj == 0 and new_score < 2.0 * n * max(unaff_max, float(nv.max())) / (new_sum * new_sum) * 0.999):
                        # Simplified: just accept if better than current
                        pass

            # Simplified approach: just try single-coordinate updates via the main ECD
            # This function is a fallback; the main ECD is more efficient
            break

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_a, best_score, sweeps

def informed_basin_hopping(a0, time_limit, rng, n_perturbs=15, verbose=False):
    """
    Informed basin hopping: perturb coordinates that contribute to convolution peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 45.0)

        conv_p = fast_convolve(a)
        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        pv = conv_p[peaks]
        # Weight by how close to max
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        # Build perturbation targeting peak-contributing coordinates
        perturbed = a.copy()
        for idx in range(len(peaks)):
            if w[idx] < 0.01:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1)
            # Perturb the contributing coordinates
            n_perturb = max(1, len(idx_range) // 4)
            picks = rng.choice(idx_range, size=min(n_perturb, len(idx_range)), replace=False)
            for ii in picks:
                sigma = 0.02 + 0.05 * w[idx]
                perturbed[ii] *= (1.0 + rng.standard_normal() * sigma)

        # Also add global noise
        perturbed += rng.standard_normal(n) * 0.01
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"hb_p{p}")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    IBH p={p} ({elapsed:.0f}s): improved to {best_score:.8f}")

    return best_a, best_score

def spectral_perturb_optimize(a0, time_limit, rng, verbose=False):
    """
    Add small sinusoidal perturbation in frequency domain, then re-optimize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    n_attempts = max(3, int((time_limit - 5) / 30))
    n_attempts = min(n_attempts, 15)

    for att in range(n_attempts):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.6, 40.0)

        # Find a perturbation frequency that targets the peak
        conv_a = fast_convolve(a)
        peak_lag = int(np.argmax(conv_a))

        # Try perturbing at a frequency related to the peak
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.02 + 0.03 * rng.random()

        t = np.arange(n)
        perturbation = amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        perturbed = a + perturbation
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"sp_a{att}")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    SP att={att} freq={freq}: improved to {best_score:.8f}")

    return best_a, best_score

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_factor = 1.0
            for factor in [0.998, 0.995, 0.99, 0.985, 0.98, 0.97,
                           1.002, 1.005, 1.01, 1.015, 1.02, 1.03]:
                x = v * factor
                if x < 0:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                ns = current_sum + (x - v)
                if ns < 0.01:
                    continue
                sc = 2.0 * n * max(unaff_max, local_max) / (ns * ns)
                cur_sc = 2.0 * n * max(unaff_max, float(c_vals.max() + 2.0 * v * a_partners if not has_quad else c_vals.max())) / (current_sum * current_sum)
                # Simplified: just check if local max decreased
                if local_max < float((c_vals + 2.0 * v * a_partners).max()) - 1e-16:
                    if has_quad:
                        old_max = max(c_vals[quad_idx] + v * v,
                                      float((c_vals + 2.0 * v * a_partners).max()))
                        # Hmm this is getting complicated, just use the score
                    pass

            # Simpler approach: try each factor and keep the best score
            cur_max_aff = float(conv[k_lo:k_hi+1].max())
            best_new_score = 2.0 * n * max(unaff_max, cur_max_aff) / (current_sum * current_sum)
            best_x = v

            for factor in [0.995, 0.99, 0.985, 0.97, 0.95, 1.05, 1.03, 1.015, 1.01, 1.005]:
                x = v * factor
                if x < 0:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                ns = current_sum + (x - v)
                if ns < 0.01:
                    continue
                sc = 2.0 * n * max(unaff_max, float(nv.max())) / (ns * ns)
                if sc < best_new_score:
                    best_new_score = sc
                    best_x = x

            if best_x != v:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def get_lp_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
        return cp.ECOS
    except Exception:
        return None

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build convolution matrix T: T @ b = conv(a_fixed, b)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play_with_linesearch(a0, time_limit, solver, verbose=False):
    """LP fictitious play with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Line search
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs
            score_new = 2.0 * n * float(fast_convolve(a).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():

    Multi-phase optimization:
    Phase 1: Diverse initialization + quick exact coordinate descent
    Phase 2: Intensive exact coordinate descent on best candidates
    Phase 3: Informed basin hopping + spectral perturbation
    Phase 4: LP fictitious play (selective)
    Phase 5: Fine polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    if solver is not None:
        print(f"Using LP solver: {solver}")
    else:
        print("No LP solver available, skipping LP phases")

    parent = None
    try:

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick optimization ===
    print("\n=== Phase 1: Diverse Init + Quick ECD ===")
    t1_start = time.monotonic()

    n_values = [1024, 768, 512]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('asym', init_asymmetric(n, rng)))
        inits.append(('flat_spec2', init_flat_spectrum(n, rng)))

        for itype, cand in inits:
            t_left = 200.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(10.0, t_left / 6.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:8]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Intensive exact coordinate descent ===
    print(f"\n=== Phase 2: Intensive ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

            # If we're stuck, perturb
            if abs(s_opt - best_score) < 1e-6 and rnd > 0:
                perturbed = a_opt + rng.standard_normal(n_best) * 0.02
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

            # Follow up with exact CD on the LP result
            rem = deadline - time.monotonic() - 100
            if rem > 20:
                t_cd = min(rem * 0.5, 60.0)
                a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label="post_fp")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ecd_after_fp")

    # === Phase 5: Try different sequence lengths ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [512, 768, 1024, 1536]:
            if alt_n == n_best:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 40.0)
            interp = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = exact_coord_descent(interp, t_alloc, label=f"alt_n{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Multiple rounds of exact CD
        for polish_round in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 10:

            t_cd = min(rem * 0.4, 60.0)
            a_pol, s_pol, _ = exact_coord_descent(best_seq, t_cd, label=f"polish{polish_round}")
            best_seq, best_score = check_and_update(a_pol, best_seq, best_score, f"polish{polish_round}")

        # Fine coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_fine = min(rem * 0.6, 40.0)
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                    verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # One more ECD pass
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_fin, s_fin, _ = exact_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_ecd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
        best_seq = np.asarray(best_seq)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
