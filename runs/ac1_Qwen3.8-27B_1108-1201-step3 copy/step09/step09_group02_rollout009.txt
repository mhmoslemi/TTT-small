One more thought: the "pairwise coordinate descent" idea. Instead of optimizing one coordinate at a time, optimize two coordinates (i, j) simultaneously. This is a 2D optimization problem. The score as a function of (x_i, x_j) is:

For each lag k, the new value is:
- If k involves both i and j: c_k + 2*x_i*a[k-i] + 2*x_j*a[k-j] + corrections
- If k involves only i: c_k + 2*x_i*a[k-i]
- If k involves only j: c_k + 2*x_j*a[k-j]
- If k involves neither: unchanged

- Grid search on a 2D grid

        # Update with momentum-like behavior
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.9
            continue
        a_new = a_new / s
        conv_new = fast_convolve(a_new)
        sc = 2.0 * n * float(conv_new.max())

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                temp *= 0.95
                no_improve = 0
            # Keep old a if no improvement
            a_new = a
            conv_new = conv

        a = a_new
        conv = conv_new

        # Adapt temperature
        temp = max(temp * 0.999, 1e-6 * float(conv.max()))

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SS n={n} it={it} ({elapsed:.0f}s): score={sc:.8f}, lr={lr:.5f}, T={temp:.6f}")

    return best_a, best_score

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_antipodal(n, rng):
    """
    Initialize with antipodal structure: first half high, second half low.
    This reduces the center auto-conv value.
    """
    t = np.arange(n, dtype=np.float64)
    # Create a smooth profile that's high on left, low on right
    center = n * rng.uniform(0.3, 0.5)
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    # Add some structure to the right half
    center2 = n * rng.uniform(0.6, 0.85)
    width2 = n * rng.uniform(0.1, 0.25)
    a += 0.5 * np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_periodic_antipodal(n, rng):
    """
    Periodic antipodal: alternating high/low regions.
    """
    n_periods = rng.integers(2, 8)
    period = n / n_periods
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.3
    for p in range(n_periods):
        start = int(p * period)
        end = int((p + 1) * period)
        if p % 2 == 0:
            # High region in first half of period
            mid = (start + end) // 2
            w = (end - start) * rng.uniform(0.2, 0.4)
            c = start + (end - start) * 0.3
            a[start:end] += np.exp(-0.5 * ((t[start:end] - c) / w) ** 2)
        else:
            # High region in second half of period
            c = start + (end - start) * 0.7
            w = (end - start) * rng.uniform(0.2, 0.4)
            a[start:end] += np.exp(-0.5 * ((t[start:end] - c) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def differential_evolution(pop, scores, F, CR, rng, eval_func, time_limit):
    """
    One generation of differential evolution.
    pop: list of sequences, scores: list of scores (lower is better)
    Returns updated pop and scores.
    """
    pop_size = len(pop)
    n = len(pop[0])
    t0 = time.monotonic()

    for i in range(pop_size):
        if time.monotonic() - t0 > time_limit:
            break
        # Select 3 distinct random individuals
        candidates = [j for j in range(pop_size) if j != i]
        a_idx, b_idx, c_idx = rng.choice(candidates, 3, replace=False)

        # Mutation
        mutant = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])

        # Crossover (binomial)
        trial = pop[i].copy()
        r = rng.integers(n)
        for j in range(n):
            if rng.random() < CR or j == r:
                trial[j] = mutant[j]

        # Ensure validity
        trial = np.maximum(trial, 0)
        s = trial.sum()
        if s < 1e-15:
            continue
        trial = trial / s

        # Selection
        trial_score = eval_func(trial)
        if trial_score < scores[i]:
            pop[i] = trial
            scores[i] = trial_score

    return pop, scores

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with multiplicative perturbations for final polish."""
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            best_delta = 0.0
            cur_local_max = float(np.max(c_vals + 2.0 * v * a_partners))
            if has_quad:
                cur_local_max = max(cur_local_max, c_vals[quad_idx] + v * v)

            for factor in [0.999, 0.998, 0.995, 0.99, 0.985, 0.98, 1.02, 1.015, 1.01, 1.005, 1.002, 1.001]:
                x = v * factor
                if x < -1e-15:
                    continue
                nv_max = float(np.max(c_vals + 2.0 * x * a_partners))
                if has_quad:
                    qv = c_vals[quad_idx] + x * x
                    if qv > nv_max:
                        nv_max = qv
                if nv_max < cur_local_max - 1e-16:
                    cur_local_max = nv_max
                    best_delta = x - v

            if abs(best_delta) > 1e-15:
                delta = best_delta
                apply_coord_update(a, conv, i, v, c_vals, a_partners, has_quad, quad_idx, v + delta)
                current_sum += delta
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via the official evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + spectral subgradient + quick CD
    Phase 2: Intensive coordinate descent
    Phase 3: Differential evolution + CD refinement
    Phase 4: Spectral subgradient refinement
    Phase 5: Final fine polishing
    """
    deadline = _get_deadline()
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # ============ Phase 1: Diverse Init + Spectral Subgradient + Quick CD ============
    print("\n=== Phase 1: Diverse Init + Spectral SG + Quick CD ===")
    t1_start = time.monotonic()

    n_target = 1024
    if parent is not None:
        n_target = len(parent)

    candidates = []

    # Generate diverse initializations
    inits = []
    if parent is not None:
        inits.append(('parent', parent.copy() if len(parent) == n_target else interpolate_seq(parent, n_target)))
    inits.append(('flat_spec', init_flat_spectrum(n_target, rng)))
    inits.append(('flat_spec2', init_flat_spectrum(n_target, rng)))
    inits.append(('antipodal', init_antipodal(n_target, rng)))
    inits.append(('periodic', init_periodic_antipodal(n_target, rng)))
    inits.append(('periodic2', init_periodic_antipodal(n_target, rng)))

    for itype, cand in inits:
        elapsed = time.monotonic() - t1_start
        if elapsed > 400:
            break
        t_left = 450.0 - elapsed
        if t_left < 30:
            break

        # Spectral subgradient (global shape optimization)
        t_sg = min(60.0, t_left * 0.4)
        a_sg, s_sg = spectral_subgradient_descent(cand, t_sg, rng)
        candidates.append((s_sg, a_sg, f"sg_{itype}"))

        # Quick coordinate descent after SG
        t_cd = min(40.0, t_left * 0.3)
        a_cd, s_cd, _ = exact_coord_descent(a_sg, t_cd, rng)
        candidates.append((s_cd, a_cd, f"cd_{itype}"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best internal: {candidates[0][0]:.8f}")

    for s_c, a_c, label_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{label_c}")

    # ============ Phase 2: Intensive Coordinate Descent ============
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(5):
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem * 0.25, 150.0)
            print(f"  CD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc, rng, verbose=True, label=f"ph2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_cd_r{rnd}")

            # Check if we should perturb to escape
            if best_score > 1.5035 and rnd < 3:
                perturbed = a_opt + rng.standard_normal(n_best) * 0.03
                current = normalize_nonneg(perturbed)
            else:
                current = a_opt.copy()

    # ============ Phase 3: Differential Evolution ============
    print(f"\n=== Phase 3: Differential Evolution (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_de = min(rem * 0.5, 250.0)
            print(f"  DE budget={t_de:.0f}s, n={n_best}")

            # Build initial population
            pop_size = 12
            pop = [best_seq.copy()]
            scores = [best_score]

            # Add diverse members
            for _ in range(pop_size - 1):
                if rng.random() < 0.5:
                    member = best_seq + rng.standard_normal(n_best) * 0.05
                else:
                    t_phase = rng.uniform(0, 2 * np.pi)
                    pert = init_flat_spectrum(n_best, rng)
                    member = 0.5 * best_seq + 0.5 * pert
                member = normalize_nonneg(member)
                sc = compute_score(member)
                pop.append(member)
                scores.append(sc)

            # Run DE for budget
            F = 0.5
            CR = 0.7
            gen = 0
            t_de_start = time.monotonic()

            while time.monotonic() - t_de_start < t_de:
                gen += 1
                gen_time = min(30.0, (t_de - (time.monotonic() - t_de_start)) * 0.8)
                if gen_time < 5:
                    break
                pop, scores = differential_evolution(pop, scores, F, CR, rng, compute_score, gen_time)

                # Track best
                best_idx = np.argmin(scores)
                if scores[best_idx] < best_score - 1e-8:
                    best_seq, best_score = check_and_update(pop[best_idx], best_seq, best_score, f"de_g{gen}")

                # Periodic CD refinement of the best
                if gen % 5 == 0:
                    t_cd_ref = min(30.0, (t_de - (time.monotonic() - t_de_start)) * 0.3)
                    if t_cd_ref > 10:
                        a_ref, s_ref, _ = exact_coord_descent(pop[best_idx], t_cd_ref, rng)
                        if s_ref < scores[best_idx]:
                            pop[best_idx] = a_ref
                            scores[best_idx] = s_ref
                            best_seq, best_score = check_and_update(a_ref, best_seq, best_score, f"de_cd_g{gen}")

                # Adapt F and CR
                F = max(0.3, F * 0.98)
                CR = min(0.9, CR + 0.005)

                if gen % 10 == 0:
                    elapsed = time.monotonic() - t_de_start
                    print(f"    DE gen={gen} ({elapsed:.0f}s): best_pop={min(scores):.8f}, global_best={best_score:.8f}")

    # ============ Phase 4: Spectral Subgradient Refinement ============
    print(f"\n=== Phase 4: Spectral SG Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sg2 = min(rem * 0.6, 120.0)
            print(f"  SG budget={t_sg2:.0f}s")
            a_sg2, s_sg2 = spectral_subgradient_descent(best_seq, t_sg2, rng, verbose=True)
            best_seq, best_score = check_and_update(a_sg2, best_seq, best_score, "sg_refine")

            # CD after SG
            rem = deadline - time.monotonic() - 80
            if rem > 20:
                t_cd3 = min(rem * 0.5, 60.0)
                a_cd3, s_cd3, _ = exact_coord_descent(a_sg2, t_cd3, rng)
                best_seq, best_score = check_and_update(a_cd3, best_seq, best_score, "cd_after_sg")

    # ============ Phase 5: Final Polish ============
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_fine = min(rem * 0.5, 80.0)
            print(f"  FineCD budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine, verbose=True, label="final")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_final")

        # Another ECD pass
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd4 = min(rem * 0.6, 40.0)
            a_cd4, s_cd4, _ = exact_coord_descent(best_seq, t_cd4, rng)
            best_seq, best_score = check_and_update(a_cd4, best_seq, best_score, "ecd_final")

        # Last fine pass
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_last, s_last = fine_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
            best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
