import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_smoothed_grad(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) using FFT."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    p = np.exp(beta * (b - b_max))
    p /= p.sum()
    
    # Cross-correlation: grad_i = 2 * sum_k p_k * a_{k-i}
    N = 1 << (len(p) + n - 2).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    corr = irfft(P * np.conj(A), N)
    grad = 2.0 * corr[:n]
    return grad, p, b

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_max_b = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    # Adaptive step size
    step = 0.005
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, p, b = compute_smoothed_grad(y, beta)
        
        # FISTA update
        z = y - step * grad
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = z + ((t - 1.0) / t_new) * (z - a)
        a = simplex_project(z)
        y = simplex_project(y)
        t = t_new
        
        # Evaluate
        b_a = fast_convolve(a)
        max_b = b_a.max()
        
        if max_b < best_max_b - 1e-14:
            best_max_b = max_b
            best_a = a.copy()
            no_improve = 0
        else:

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def cross_corr_fft(p, a):
    """Compute cross-correlation (p ⋆ a)[i] = sum_k p[k+i] * a[k] for i=0..len(a)-1."""
    n = len(a)
    N = 1 << (len(p) + n - 2).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    return irfft(P * np.conj(A), N)[:n]

def compute_smoothed_grad(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    p = np.exp(beta * (b - b_max))
    p /= p.sum()
    grad = 2.0 * cross_corr_fft(p, a)
    return grad, b

def fista_smoothed(a0, time_limit, beta=20.0, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_max_b = float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0
    step = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, b = compute_smoothed_grad(y, beta)
        
        z = y - step * grad
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = z + ((t - 1.0) / t_new) * (z - a)
        a_new = simplex_project(z)
        y = simplex_project(y)
        a = a_new
        t = t_new
        
        b_a = fast_convolve(a)
        max_b = float(b_a.max())
        
        if max_b < best_max_b - 1e-14:
            best_max_b = max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            print(f"    FISTA {label} it={it} ({elapsed:.0f}s): max_b={best_max_b:.10f}")
    
    return best_a, 2.0 * n * best_max_b
