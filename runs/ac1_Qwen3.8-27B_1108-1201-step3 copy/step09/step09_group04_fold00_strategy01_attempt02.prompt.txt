def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_objective_value(a, p):
    """Compute (sum c_k^p)^(1/p) for c = convolve(a, a)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 0.0)
    return (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_gradient_a(a, p):
    """Gradient of (sum c_k^p)^(1/p) w.r.t. a, where c = convolve(a,a).
    Returns g_a such that g_a[i] = d/d(a_i) (sum c_k^p)^(1/p)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(a)
    # factor = (sum c_k^p)^(1/p - 1)
    factor = S ** (1.0 / p - 1.0)
    # w_k = c_k^(p-1)
    w = c_pos ** (p - 1.0)
    # g_a[i] = factor * sum_k w_k * d(c_k)/d(a_i) = factor * 2 * correlate(w, a)[i]
    # correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j]

    corr = np.correlate(w, a, mode='valid')
    # The 'valid' mode gives length len(w) - len(a) + 1 = (2n-1) - n + 1 = n
    g_a = 2.0 * factor * corr
    return g_a

def p_norm_gradient_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    # Gradient w.r.t. z: a * (g_a - dot(g_a, a))
    return a * (g_a - np.dot(g_a, a))

def build_conv_matrix_sparse(a):
    """Build the convolution matrix T where T @ b = convolve(a, b).
    Returns (size, n) matrix."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""

    try:
        prob.solve(solver=solver)

        pass

                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

        return None, None

def structured_init(n, rng, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        # Bimodal: large at ends
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        # Gaussian at center
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
