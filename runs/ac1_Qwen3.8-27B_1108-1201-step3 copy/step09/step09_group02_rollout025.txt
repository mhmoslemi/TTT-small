import numpy as np

_lp_solver_cached = None


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):

def exact_coord_descent(a0, time_limit, verbose=False, label=""):

    no_improve = 0

    gr = (np.sqrt(5.0) - 1.0) / 2.0

        improved = False

            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            # Golden section search with precomputed data
            _c = c_vals
            _ap = a_partners
            _qm = unaff_max
            _hq = has_quad
            _qi = quad_idx
            _cs = current_sum
            _n_local = n

            def score_at_x(x):
                ns = _cs + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n_local * nm / (ns * ns)

                else:

                    fc = fd

            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-12:

                a[i] = best_x

                changes = 2.0 * delta * a_partners

def nesterov_subgradient(a0, conv0, time_limit, n_peaks=12, verbose=False):
    """
    Nesterov-accelerated projected subgradient descent.
    Minimizes 2n * max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov momentum point
    conv = fast_convolve(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    beta = 0.0  # momentum

    # Initial learning rate estimate
    M = float(conv.max())
    L = 2.0 * n * 2.0 * M  # Lipschitz constant estimate
    lr0 = 1.0 / np.sqrt(L) if L > 0 else 0.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov momentum update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        alpha = (1.0 + beta) / (1.0 + beta_new)
        beta = beta_new

        y_new = a + alpha * (a - y)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / max(y_new.sum(), 1e-15)
        y = y_new

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_act = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        temp = 0.003 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(k_act) / k_act
        else:
            w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * y[k - j]

        # Step size (Nesterov: O(1/k))
        lr = lr0 / np.sqrt(it + 1.0)
        a_candidate = y - lr * 2.0 * n * grad
        a_candidate = simplex_project(a_candidate)

        # Evaluate
        conv_a = fast_convolve(a_candidate)
        sc = 2.0 * n * float(conv_a.max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_candidate.copy()

        a = a_candidate

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSGD it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            for f in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                d = v * (f - 1.0)
                if -v * 0.99 < d < 3.0 / n:
                    candidates.append(d)

            for d in candidates:
                x = v + d
                if x < 0 or x > 3.0 / n:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                ns = current_sum + d
                if ns < 0.01:
                    continue
                sc = 2.0 * n * nm / (ns * ns)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = d

            if best_delta != 0.0 and best_sc < best_score - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                current_sum += best_delta
                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PTC {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_asymmetric(n, rng):
    """Asymmetric initialization."""
    t = np.arange(n, dtype=np.float64)
    shift = rng.uniform(0.2, 0.8)
    center = n * shift
    width = n * rng.uniform(0.15, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(3, 15)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_harmonics, n // 2), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_barker_like(n, rng):
    """Initialize using a Barker-like sequence pattern, then smooth."""
    # Create a binary-like pattern
    a = np.zeros(n)
    # Use a random walk with mean 0.5
    steps = rng.choice([-1, 1], size=n) * 0.1
    a[0] = 0.5
    for i in range(1, n):
        a[i] = np.clip(a[i - 1] + steps[i], 0.01, 0.99)
    # Smooth
    kernel = np.ones(5) / 5.0
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a_smooth / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=6, verbose=False):
    """Basin hopping with structured perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 45.0)

        # Perturb
        sigma = 0.015 + 0.025 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Peak-breaking perturbation
        conv_p = fast_convolve(a)
        top_k = min(3, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 8)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 + rng.standard_normal() * 0.03)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Diverse initialization + fast analytic CD
    2. LP fictitious play with adaptive step
    3. Peak-targeted coordinate descent
    4. Nesterov subgradient
    5. Exact CD polishing
    6. Basin hopping for escape
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + fast CD ===
    print("\n=== Phase 1: Diverse Init + Fast CD (200s) ===")
    t1_start = time.monotonic()
    t1_budget = 200.0

    n_values = [768, 1024]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values[:3]:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > t1_budget * 0.7:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))
        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('asym', init_asymmetric(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('barker', init_barker_like(n, rng)))

        for itype, cand in inits:
            t_left = t1_budget - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(2.0, min(8.0, t_left / 6.0))
            a_opt, s_opt, _ = fast_coord_descent_analytic(cand, t_alloc, label=f"p1_{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{n_c}_{it_c}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    t2_budget = 300.0

    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 768]:
            if alt_n != n_best:
                fp_lengths.append(alt_n)

        for n_fp in fp_lengths[:2]:
            rem = deadline - time.monotonic() - 600
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, t2_budget * 0.6)
            print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

            if n_fp != n_best:
                initial = interpolate_seq(best_seq, n_fp)
            else:
                initial = best_seq.copy()

            a_fp, score_fp, iters = fictitious_play_with_linesearch(
                initial, t_alloc, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_{n_fp}")

            # Quick CD after FP
            rem = deadline - time.monotonic() - 500
            if rem > 15:
                t_cd = min(rem * 0.3, 40.0)
                a_cd, s_cd, _ = fast_coord_descent_analytic(a_fp, t_cd, label=f"post_fp_{n_fp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"ecd_postfp_{n_fp}")

    # === Phase 3: Peak-targeted CD ===
    print(f"\n=== Phase 3: Peak-Targeted CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:

                                               verbose=True, label="ph3")

            # Second round with more peaks
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                t_alloc2 = min(rem * 0.4, 100.0)
                a_ptc2, s_ptc2, _ = peak_targeted_cd(best_seq, None, t_alloc2, n_peaks=15,
                                                     verbose=True, label="ph3b")
                best_seq, best_score = check_and_update(a_ptc2, best_seq, best_score, "ptc_ph3b")

    # === Phase 4: Nesterov Subgradient ===
    print(f"\n=== Phase 4: Nesterov Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  NSGD budget={t_alloc:.0f}s")
            a_nsgd, s_nsgd = nesterov_subgradient(best_seq, None, t_alloc,
                                                  n_peaks=15, verbose=True)

    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.6, 150.0)
            print(f"  BH budget={t_alloc:.0f}s")
            a_bh, s_bh = basin_hopping(best_seq, t_alloc, fast_coord_descent_analytic,
                                       rng, n_perturbs=5, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 6: Final Exact CD Polishing ===
    print(f"\n=== Phase 6: Final Exact CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Round 1: fast CD
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  Fast CD polish, budget={t_alloc:.0f}s")

    # Fallback
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))

        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
