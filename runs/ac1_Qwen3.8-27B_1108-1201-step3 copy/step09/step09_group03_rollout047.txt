from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent with adaptive temperature
2. Coordinate descent with adaptive line search and active-peak awareness
3. Peak equalization via targeted coordinate adjustments
4. Multi-start with diverse structured initializations (Fourier, periodic, bursty)
5. Multiple length trials (1024, 2048) for broader exploration
6. Perturbation-based escape from local minima

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def smoothed_subgradient(a, tau):
    """
    Compute the smoothed subgradient of max(conv(a,a)) using log-sum-exp.
    Returns (gradient, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    # Numerically stable softmax
    c_max = c.max()
    scaled = tau * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -50, 50)
    exp_c = np.exp(scaled)
    Z = exp_c.sum()
    if Z < 1e-30:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_c / Z
    # Cross-correlation: grad[i] = 2*tau * sum_k p[k] * a[k-i]
    # = 2*tau * convolve(p, a[::-1])[n-1+i]
    b = a[::-1]
    full = convolve_ab(p, b)
    grad = 2.0 * tau * full[n - 1: n - 1 + n]
    return grad, c


def mirror_descent_smoothed(a0, time_limit, tau_start=20.0, tau_end=300.0,
                            lr_init=0.08, verbose=True, label=""):
    """
    Mirror descent (multiplicative update) with log-sum-exp smoothed gradient.
    Adaptively anneals temperature and learning rate.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal temperature
        progress = min(it / max(1, int(time_limit / 0.5)), 1.0)
        tau = tau_start + (tau_end - tau_start) * progress

        # Compute smoothed gradient
        grad, c = smoothed_subgradient(a, tau)

        # Track actual score
        actual_score = 2.0 * n * c.max()
        if actual_score < best_score - 1e-12:
            best_score = actual_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt learning rate
        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-9:
                break

        # Mirror descent step: a * exp(-lr * grad)
        # Scale gradient for stability
        grad_scale = np.abs(grad).max()
        if grad_scale > 1e-15:

        log_update = -lr * grad_normed
        log_update = np.clip(log_update, -10, 10)

def fw_alternating(a_init, time_limit, n_fw_steps=100, verbose=True, label=""):

        a = a / s

        round_num += 1

        # --- Optimize b given a ---
        for step in range(n_fw_steps):
            k_star = int(np.argmax(b_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            if step >= 20:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = b * grad_full
                i_away = int(np.argmax(away_scores))
                if b[i_away] > 1e-8 and grad_full[i_away] > 1e-8:
                    t_away = min(t * 0.3, b[i_away] * 0.5)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)

                    b[i_away] -= t_away
                    b[i_away] = max(b[i_away], 0.0)
                    # Renormalize
                    sb = b.sum()
                    if sb > 1e-15:
                        b = b / sb
                        b_conv = convolve_ab(a, b)

            # FW step
            b_conv = (1.0 - t) * b_conv
            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)

                i_away = int(np.argmax(away_scores))
                if a[i_away] > 1e-8 and grad_full[i_away] > 1e-8:
                    t_away = min(t * 0.3, a[i_away] * 0.5)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)
                    a_conv[ks_s:ks_e + 1] -= t_away * b
                    a[i_away] -= t_away
                    a[i_away] = max(a[i_away], 0.0)
                    sa = a.sum()
                    if sa > 1e-15:
                        a = a / sa
                        a_conv = convolve_ab(b, a)

            a_conv = (1.0 - t) * a_conv
            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)

        # coord i affects conv[k] for k in [i, i+n-1]
        # Score: sum of conv[k] for active k in that range
        importance = np.zeros(n)
        for k in active_peaks:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            importance[i_lo:i_hi + 1] += 1

        # Process coordinates in order of importance (descending)
        order = np.argsort(importance)[::-1]

        for idx in range(n):
            i = int(order[idx])
            if a[i] < 1e-12:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected = slice(k_start, k_end + 1)
            affected_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_score = best_score

            # Adaptive steps: relative to a[i] and absolute
            base = max(a[i], M / n * 0.1)
            factors = [-0.5, -0.3, -0.15, -0.05, -0.02, 0.02, 0.05, 0.15, 0.3, 0.5]
            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-12:
                    continue
                new_S = S + delta

                # New affected conv values
                new_affected = conv[affected].copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos = 2 * i - k_start
                    new_affected[pos] += delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < 0:
                    continue
                new_score = 2.0 * n * new_M / (new_S * new_S)
                if new_score < best_new_score:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < best_score - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                S += best_delta

                # Update conv incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos = 2 * i - k_start
                    changes = changes.copy()
                    changes[pos] += best_delta * best_delta
                conv[affected] += changes
                M = max(unaff_max, float(conv[affected].max()))

                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-Active {label} sweep={sweep} ({elapsed:.0f}s): "
                  f"best={best_score:.6f} M={M:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=True, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    S = float(a.sum())
    best_score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_score = best_score

            # Very fine steps
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 1.01, 1.02, 1.005, 1.002, 1.001]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                new_S = S + delta

                new_affected = conv[k_start:k_end + 1].copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos = 2 * i - k_start
                    new_affected[pos] += delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < 0:
                    continue
                new_score = 2.0 * n * new_M / (new_S * new_S)
                if new_score < best_new_score - 1e-15:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < best_score - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                S += best_delta
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos = 2 * i - k_start
                    changes = changes.copy()
                    changes[pos] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweep > 5:
            no_improve += 1
            if no_improve > 20:
                break
        elif not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and sweep % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweep} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, verbose=True):
    """
    Try to equalize the top peaks by targeted coordinate adjustments.
    For pairs of active peaks, find coordinates that differentially affect them.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    S = float(a.sum())
    best_score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:top_k]
        peak_values = conv[peak_indices]

        if len(peak_indices) < 2:
            break

        # For each pair of top peaks, try to reduce the difference
        max_pk = peak_values[0]
        improved = False

        for pi in range(1, min(5, len(peak_indices))):
            k1 = int(peak_indices[0])  # max peak
            k2 = int(peak_indices[pi])  # second peak
            v1 = conv[k1]
            v2 = conv[k2]

            if v1 - v2 < 1e-12:
                continue

            # Find coordinate i that affects k1 more than k2
            # coord i affects k1 if 0 <= k1-i <= n-1
            # coord i affects k2 if 0 <= k2-i <= n-1
            # We want: a[k1-i] > a[k2-i] (so reducing a[i] helps k1 more)
            # Or: a[k1-i] < a[k2-i] (so increasing a[i] hurts k1 less)

            i_lo = max(0, k1 - n + 1, k2 - n + 1)
            i_hi = min(n - 1, k1, k2)

            if i_hi < i_lo:
                continue

            # Try a few candidate coordinates
            for trial_i in range(int(i_lo), min(int(i_hi) + 1, int(i_lo) + 50)):
                if a[trial_i] < 1e-12:
                    continue

                contrib_k1 = a[k1 - trial_i] if 0 <= k1 - trial_i <= n - 1 else 0.0
                contrib_k2 = a[k2 - trial_i] if 0 <= k2 - trial_i <= n - 1 else 0.0

                if contrib_k1 <= contrib_k2:
                    continue  # Reducing a[trial_i] would hurt k2 more

                # Try small reduction of a[trial_i]
                for factor in [0.95, 0.9, 0.85, 0.8]:
                    delta = a[trial_i] * (factor - 1.0)
                    new_a = a.copy()
                    new_a[trial_i] = max(a[trial_i] + delta, 0.0)
                    new_conv = fast_convolve(new_a)
                    new_M = float(new_conv.max())
                    new_S = float(new_a.sum())
                    if new_S < 0.01:
                        continue
                    new_score = 2.0 * n * new_M / (new_S * new_S)
                    if new_score < best_score - 1e-14:
                        a = new_a
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved and it > 5:
            break

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def cosine_window_init(n, rng, power_range=(0.5, 4.0)):
    """Initialize with a cosine raised-to-power window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(*power_range)
    a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def fourier_init(n, rng):
    """Initialize via random Fourier phases with smooth power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    freqs = np.fft.rfftfreq(fft_size)
    # Smooth power spectrum: concentrated at low frequencies
    bandwidth = rng.uniform(0.05, 0.3)
    power = np.exp(-2.0 * (freqs / bandwidth) ** 2)
    phases = rng.uniform(0, 2 * np.pi, len(power))
    spectrum = power * np.exp(1j * phases)
    # Make it real and non-negative
    a = np.fft.irfft(spectrum * fft_size, n=fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def bursty_init(n, rng):
    """Initialize with bursty pattern: groups of high values separated by gaps."""
    a = np.zeros(n)
    n_bursts = int(rng.integers(3, 10))
    burst_width = int(n / (n_bursts * 2))
    burst_width = max(2, burst_width)
    for _ in range(n_bursts):
        start = int(rng.integers(0, max(1, n - burst_width)))
        width = int(rng.integers(max(2, burst_width // 2), burst_width + 1))
        width = min(width, n - start)
        if width < 1:
            continue
        height = rng.uniform(0.5, 1.5)
        # Tapered burst
        t = np.arange(width)
        taper = np.sin(np.pi * t / max(width - 1, 1)) ** rng.uniform(0.5, 2.0)
        a[start:start + width] += height * taper
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def periodic_modulated_init(n, rng):
    """Periodic pattern with random modulation."""
    period = int(rng.integers(3, 15))
    t = np.arange(n, dtype=np.float64)
    base = 0.5 + 0.4 * np.cos(2 * np.pi * t / period + rng.uniform(0, 2 * np.pi))
    # Add slow modulation
    mod_freq = rng.uniform(0.01, 0.1)
    modulation = 1.0 + rng.uniform(0.2, 0.5) * np.sin(2 * np.pi * mod_freq * t)
    a = base * modulation
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_hump_init(n, rng):
    """Two Gaussian humps at random positions."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def full_pipeline(a_init, time_limit, verbose=True):
    """Run the full optimization pipeline on a single initialization."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()

    # Phase A: Smoothed mirror descent
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        t_md = min(remaining * 0.35, 150.0)
        a_md, s_md = mirror_descent_smoothed(a, t_md, tau_start=20, tau_end=200,
                                             lr_init=0.08, verbose=verbose, label="A")
        if s_md < best_score:
            best_score = s_md
            best_a = a_md.copy()

    # Phase B: FW alternating
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        t_fw = min(remaining * 0.3, 120.0)
        a_fw, s_fw = fw_alternating(best_a, t_fw, n_fw_steps=80,
                                    verbose=verbose, label="B")
        if s_fw < best_score:
            best_score = s_fw
            best_a = a_fw.copy()

    # Phase C: Active coordinate descent
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        t_cd = min(remaining * 0.35, 200.0)
        a_cd, s_cd = coord_descent_active(best_a, t_cd, verbose=verbose, label="C")
        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()

    # Phase D: Peak equalization
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        t_pe = min(remaining * 0.25, 100.0)
        a_pe, s_pe = peak_equalization(best_a, t_pe, verbose=verbose)
        if s_pe < best_score:
            best_score = s_pe
            best_a = a_pe.copy()

    # Phase E: Fine coordinate descent
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 20:
        t_fine = min(remaining * 0.5, 150.0)
        a_fine, s_fine = fine_coord_descent(best_a, t_fine, verbose=verbose, label="E")
        if s_fine < best_score:
            best_score = s_fine
            best_a = a_fine.copy()

    return best_a, best_score

        noisy = np.maximum(noisy, 0)
        noisy = noisy / noisy.sum()
        inits[5] = ("noisy_parent", noisy)

            print(f"  Init '{name}', budget={t_alloc:.0f}s")

    print(f"\n=== Phase 3: Larger length n=2048 (best={best_score:.6f}) ===")

        f = interp1d(x_old, best_seq, kind='cubic', fill_value="extrapolate")

    print(f"\n=== Phase 4: Perturbation escape (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 100
    if remaining > 60:
        a = best_seq.copy()
        n = len(a)
        for pert_idx in range(3):
            remaining = deadline - time.monotonic() - 60
            if remaining < 30:
                break
            # Structured perturbation
            if pert_idx == 0:
                # Random block swap
                block_size = n // 8
                i = int(rng.integers(0, n - 2 * block_size))
                j = int(rng.integers(0, n - 2 * block_size))
                if j < i:
                    i, j = j, i
                if j - i < block_size:

                a_pert = a.copy()
                a_pert[i:i + block_size], a_pert[j:j + block_size] = \
                    a_pert[j:j + block_size].copy(), a_pert[i:i + block_size].copy()

                phases = np.angle(A)
                new_phases = phases.copy()
                new_phases[mid_mask] = rng.uniform(0, 2 * np.pi, mid_mask.sum())
                A_new = np.abs(A) * np.exp(1j * new_phases)
                a_pert = irfft(A_new, 2 * n)[:n]
                a_pert = np.maximum(a_pert, 0)

            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 0.01:
                continue
            a_pert = a_pert / sp

            t_pert = min(remaining * 0.3, 60.0)
            a_pr, s_pr = full_pipeline(a_pert, t_pert, verbose=False)
            best_seq, best_score = check_and_update(a_pr, best_seq, best_score, f"pert_{pert_idx}")

    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final polishing (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 20
    if remaining > 20:

        a = best_seq.copy()

        t_fine = min(remaining * 0.4, 60.0)
        a_f, s_f = fine_coord_descent(a, t_fine, verbose=False, label="final")
        best_seq, best_score = check_and_update(a_f, best_seq, best_score, "final_cd")

        # Active CD
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            t_ac = min(remaining * 0.6, 40.0)
            a_ac, s_ac = coord_descent_active(best_seq, t_ac, verbose=False, label="final_active")
            best_seq, best_score = check_and_update(a_ac, best_seq, best_score, "final_active")

    # === Final verification ===
    print(f"\n=== Final Verification ===")
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
