import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_conv_grad(a, w):
    """Compute grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k]) using FFT."""
    n = len(a)
    size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, size)
    A = rfft(a, size)
    term1 = irfft(W * np.conj(A), size)[:n]
    term2 = irfft(W * A, size)[:n]
    return term1 + term2

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_score(a):
    """Compute the score for a normalized sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 4.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def mwu_optimize(a0, time_limit, eta_init=500.0, verbose=False, label=""):
    """
    Multiplicative Weight Update optimization.
    Maintains dual weights y over convolution lags, updates primal via
    projected subgradient, updates dual via exponential weighting.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    y = np.ones(2 * n - 1) / (2 * n - 1)
    a_avg = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = 0.05
    eta = eta_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 100 == 0:
            eta *= 0.98
            lr *= 0.995

        # Compute weighted gradient using current dual weights
        grad_a = compute_conv_grad(a, y)

        # Projected gradient step on simplex
        a_new = simplex_project(a - lr * grad_a)

        # Compute convolution for score and dual update
        conv = fast_convolve(a)
        score = 2.0 * n * float(conv.max())

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Also check a_new and a_avg
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a_avg = (it * a_avg + a_new) / (it + 1)
        conv_avg = fast_convolve(a_avg)
        score_avg = 2.0 * n * float(conv_avg.max())
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()

        # Update dual weights (centered MWU)
        conv_centered = conv - conv.mean()
        y *= np.exp(eta * conv_centered)
        y_max = y.max()
        if y_max > 0:
            y /= y_max
        y_sum = y.sum()
        if y_sum > 0:
            y /= y_sum
        else:
            y = np.ones(2 * n - 1) / (2 * n - 1)

        a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.0f}")

    return best_a, best_score, it

def smoothed_lbfgs(a0, time_limit, T_init=2000.0, T_max=50000.0, verbose=False, label=""):
    """
    L-BFGS optimization with log-sum-exp smoothed objective.
    Uses log-parameterization for non-negativity.
    Anneals T (smoothness parameter) from small to large.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    # Log parameterization (add small epsilon to avoid log(0))
    x = np.log(a + 1e-15)
    x -= x.mean()  # center for numerical stability

    best_score = 2.0 * n * float(fast_convolve(a).max())
    best_a = a.copy()
    t0 = time.monotonic()

    def obj_and_grad(x, T):
        a_local = np.exp(x - x.max())
        S = a_local.sum()
        a_local = a_local / S

        conv = fast_convolve(a_local)
        T_conv = T * conv
        max_TC = T_conv.max()
        exp_Tc = np.exp(T_conv - max_TC)
        exp_sum = exp_Tc.sum()
        w = exp_Tc / exp_sum

        logsumexp_val = max_TC + np.log(exp_sum)
        g_val = logsumexp_val / T

        # Gradient of g w.r.t. a
        grad_g = compute_conv_grad(a_local, w)

        # F = 2n * g (since sum a = 1)
        F = 2.0 * n * g_val

        # Chain rule for log-parameterization
        dot = np.dot(a_local, grad_g)
        grad_x = 2.0 * n * a_local * (grad_g - dot)

        return F, grad_x

    # Anneal T
    T = T_init
    n_anneal = 8
    anneal_rates = np.linspace(0, 1, n_anneal + 1)
    T_targets = T_init * (T_max / T_init) ** anneal_rates

    for stage in range(n_anneal):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_stage = rem * (1.0 / (n_anneal - stage)) * 0.8
        T = T_targets[stage]

        # Run L-BFGS for this stage
        result = minimize(
            lambda x: obj_and_grad(x, T)[0],
            x,
            jac=lambda x: obj_and_grad(x, T)[1],
            method='L-BFGS-B',
            options={'maxiter': 300, 'maxfun': 500, 'gtol': 1e-8}
        )

        if result.success or result.fun < best_score:
            x = result.x
            a_stage = np.exp(x - x.max())
            a_stage = a_stage / a_stage.sum()
            a_stage = np.maximum(a_stage, 0)
            ss = a_stage.sum()
            if ss > 1e-15:
                a_stage /= ss
            sc = get_score(a_stage)
            if sc < best_score:
                best_score = sc
                best_a = a_stage.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LBFGS {label} stage={stage} T={T:.0f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_arc_sinh(n, rng):
    """Arcsinh-shaped initialization."""
    t = np.linspace(-1, 1, n)
    a = np.arcsinh(1.0 + 0.5 * t**2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_bimodal(n, rng):
    """Bimodal initialization (mass at two locations)."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.1, 0.3)
    w2 = n * rng.uniform(0.1, 0.3)
    frac = rng.uniform(0.3, 0.7)
    a = frac * np.exp(-0.5 * ((t - c1) / w1) ** 2) + (1 - frac) * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interp_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def basin_hopping(a0, time_limit, cd_time_per_perturb, rng, n_perturbs=10, verbose=False):
    """Basin hopping with varied perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < cd_time_per_perturb + 5:
            break

        # Choose perturbation strategy
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce values at positions contributing to top peaks
            conv_p = fast_convolve(a)
            top_k = min(5, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.1 * rng.random())
        elif strategy == 2:
            # Swap: exchange values between random positions
            perturbed = a.copy()
            n_swaps = rng.integers(5, 20)
            for _ in range(n_swaps):
                i, j = rng.integers(0, n, 2)
                perturbed[i], perturbed[j] = perturbed[j], perturbed[i]
        else:
            # Frequency-domain perturbation
            size = 1 << (2 * n - 1).bit_length()
            A = rfft(perturbed.astype(np.float64) if False else a, size)
            # Add random phase perturbation to non-DC components
            phase_shift = rng.standard_normal(len(A)) * 0.1
            A *= np.exp(1j * phase_shift)
            A[0] = np.abs(A[0])  # Keep DC positive
            perturbed = irfft(A, size)[:n]

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, cd_time_per_perturb)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} strat={strategy}: improved to {best_score:.8f}")

    return best_a, best_score

def fine_polish(a0, time_limit, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        if no_improve > 10:
            break
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_local = float(c_vals.max() + 2.0 * v * a_partners.max()) if not has_quad else float(c_vals.max())

            for factor in [0.995, 0.99, 0.98, 0.96, 0.94, 1.03, 1.06, 0.85]:
                delta = v * (factor - 1.0)
                x = v + delta
                if x < 0:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                if local_max < best_local - 1e-16:
                    best_local = local_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                current_sum += best_delta
                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
                no_improve = 0

        if not improved:
            break

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization combining:
    - Diverse initialization + exact coordinate descent
    - MWU (multiplicative weight update) for fast dual-based search
    - L-BFGS with smoothed objective for different basin exploration
    - Basin hopping for global search
    - Fine polishing for final improvement
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick coordinate descent ===
    print("\n=== Phase 1: Diverse Init + Quick ECD ===")
    t1_start = time.monotonic()

    n_values = [1024, 768, 512]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values[:4]:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 150:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interp_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))
        inits.append(('cos', init_cosine(n, rng)))
        inits.append(('arc', init_arc_sinh(n, rng)))
        inits.append(('bimodal', init_bimodal(n, rng)))

        for itype, cand in inits:
            t_left = 150.0 - (time.monotonic() - t1_start)
            if t_left < 4:
                break
            t_alloc = max(3.0, min(8.0, t_left / 8.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc)
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Intensive coordinate descent on best ===
    print(f"\n=== Phase 2: Intensive ECD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_start = time.monotonic()
        current = best_seq.copy()

        for rnd in range(5):
            rem = deadline - time.monotonic() - 600
            if rem < 20
