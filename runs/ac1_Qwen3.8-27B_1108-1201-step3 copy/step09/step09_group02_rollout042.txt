def fast_convolve(a):
   n=len(a)
   if n==0: return np.empty(0)
   size=2*n-1
   L=1 << (size-1).bit_length()
   A=np.fft.rfft(a, L)
   c=np.fft.irfft(A*A, L)[:size]

   return c

def simplex_project(v):

    if cond.any():

       theta=(cssv[rho]-1.0)/(rho+1.0)

       return np.ones(n)/n

"""
Search for nonnegative step-function heights minimizing evaluate_sequence.

The solver combines:
- fast FFT convolution-based local modeling,
- exact coordinate descent with golden-section 1D searches,
- LP fictitious play using scipy.optimize.linprog when available,
- subgradient descent with active peak weighting,
- random/structured restarts and length interpolation,
- final black-box verification via evaluate_sequence.
"""

import numpy as np
import time
import math
from operator import itemgetter


def normalize_sum1(a):
    """Clip to nonnegative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).flatten()
    if a.size == 0:
        return a
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def safe_eval(seq):
    """Safely call evaluate_sequence and return a finite float or inf."""
    try:
        out = []
        for x in seq:
            xf = float(x)
            if not np.isfinite(xf):
                return float("inf")
            out.append(xf)
        val = evaluate_sequence(out)
        val = float(val)
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float("inf")


def check_update(a, best_seq, best_score, label=""):
    """Evaluate a candidate by the real evaluator and update best if improved."""
    a = np.asarray(a, dtype=np.float64).flatten()
    if a.size == 0:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return best_seq, best_score
    a = a / s
    sc = safe_eval(a.tolist())
    if np.isfinite(sc) and sc < best_score:
        print(f"NEW BEST [{label}]: score={sc:.10f} n={a.size}")
        return a.copy(), sc
    return best_seq, best_score


def fast_convolve(a):
    """Linear auto-convolution by FFT, clipped to be nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:size]
    c = np.maximum(c, 0.0)
    return c


def local_score_norm(a):
    """Score for a normalized nonnegative sequence: 2n * max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float("inf")
    c = fast_convolve(a)
    return 2.0 * n * float(np.max(c))


def simplex_project(v):
    """Project onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    inds = np.arange(1, n + 1)
    cond = u * inds > (cssv - 1.0)
    if cond.any():
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n, dtype=np.float64) / n


def interpolate_seq(f, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    f = normalize_sum1(f)
    old_n = len(f)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    g = np.maximum(g, 0.0)
    return normalize_sum1(g)


def eval_coord_with_buf(x, base, two, c_quad, U, v, tmp):
    """
    Evaluate the 1D coordinate objective for a single coordinate update.

    The affected values are:
      - affine line values stored implicitly as base + x*two, except quad index,
      - one quadratic value c_quad + x*x.
    The unaffected maximum is U.
    """
    if x < 0.0:
        x = 0.0
    s = 1.0 + x - v
    if s <= 1e-12:
        return 1e100
    np.multiply(two, x, out=tmp)
    tmp += base
    line_max = float(tmp.max())
    mq = c_quad + x * x
    nm = U
    if line_max > nm:
        nm = line_max
    if mq > nm:
        nm = mq
    return nm / (s * s)


def golden_search_buf(lo, hi, base, two, c_quad, U, v, tmp, max_iter=12):
    """Golden-section search using the buffered coordinate evaluator."""
    if hi <= lo:
        x = 0.5 * (lo + hi)
        f = eval_coord_with_buf(x, base, two, c_quad, U, v, tmp)
        return x, f

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_coord_with_buf(c, base, two, c_quad, U, v, tmp)
    fd = eval_coord_with_buf(d, base, two, c_quad, U, v, tmp)

    for _ in range(max_iter):
        if hi - lo < 1e-13:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_coord_with_buf(c, base, two, c_quad, U, v, tmp)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_coord_with_buf(d, base, two, c_quad, U, v, tmp)

    x = 0.5 * (lo + hi)
    f = eval_coord_with_buf(x, base, two, c_quad, U, v, tmp)
    return x, f


def exact_cd(a0, time_limit, gs_iter=12, verbose=False, label=""):
    """
    Exact coordinate descent.

    For each coordinate i, the affected convolution slice is conv[i:i+n].
    The 1D objective is minimized by a short golden-section search plus a few
    extra endpoint candidates.
    """
    a = normalize_sum1(a0)
    n = len(a)
    if n == 0:
        return a, float("inf"), 0

    conv = fast_convolve(a)
    current_max = float(np.max(conv))
    best_score = 2.0 * n * current_max
    best_a = a.copy()

    t0 = time.monotonic()
    L = 2 * n - 1

    two_buf = np.empty(n, dtype=np.float64)
    base_buf = np.empty(n, dtype=np.float64)
    tmp1 = np.empty(n, dtype=np.float64)
    tmp2 = np.empty(n, dtype=np.float64)

    rng = np.random.default_rng(777)
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = int(order[idx])
            v = float(a[i])

            # Prepare affine bases for affected lags.
            np.multiply(a, 2.0, out=two_buf)
            base_buf[:] = conv[i:i + n]
            np.multiply(two_buf, v, out=tmp1)
            base_buf -= tmp1

            c_quad = float(conv[2 * i]) - v * v
            base_buf[i] = -np.inf  # quadratic handled separately.

            U = 0.0
            if i > 0:
                U = float(np.max(conv[:i]))
            end = i + n
            if end < L:
                um = float(np.max(conv[end:]))
                if um > U:
                    U = um

            hi = max(2.0 * max(v, 1.0 / float(n)), 20.0 / float(n), 0.08)
            if hi > 1.0:
                hi = 1.0
            if hi < v:
                hi = min(1.0, v + 1e-8)
            if hi <= 0.0:
                hi = 0.01

            xg, _ = golden_search_buf(
                0.0, hi, base_buf, two_buf, c_quad, U, v, tmp2, gs_iter
            )

            best_x = v
            best_f = eval_coord_with_buf(v, base_buf, two_buf, c_quad, U, v, tmp2)

            candidates = (0.0, xg, hi, 0.5 * v, 0.25 * v, 1.5 * v)
            for xval in candidates:
                if xval < 0.0 or xval > hi:
                    continue
                fval = eval_coord_with_buf(xval, base_buf, two_buf, c_quad, U, v, tmp2)
                if fval < best_f:
                    best_f = fval
                    best_x = xval

            if best_f < current_max - 1e-14:
                delta = best_x - v
                new_sum_preview = 1.0 + delta
                if new_sum_preview > 1e-8:
                    np.multiply(two_buf, delta, out=tmp1)
                    conv[i:i + n] += tmp1
                    # Quad term: affine update already added 2*delta*v, add delta^2.
                    conv[2 * i] += delta * delta

                    a[i] = best_x
                    scale = 1.0 / new_sum_preview
                    a *= scale
                    conv *= scale * scale

                    current_max = float(best_f)
                    score = 2.0 * n * current_max
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    improved = True

        if verbose and (sweep % 10 == 0 or sweep == 1):
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweep} ({elapsed:.0f}s): local={best_score:.10f}")

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

    best_a = normalize_sum1(best_a)
    conv_best = fast_convolve(best_a)
    final_score = 2.0 * n * float(np.max(conv_best))
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score, sweep


def subgradient_search(a0, time_limit, top_k=80, verbose=False, label=""):
    """
    Subgradient descent on max conv(a,a), using weights on the active peaks.
    The step is followed by simplex projection and local line search.
    """
    a = normalize_sum1(a0)
    n = len(a)
    if n == 0:
        return a, float("inf"), 0

    conv = fast_convolve(a)
    current_max = float(np.max(conv))
    best_score = 2.0 * n * current_max
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.01
    it = 0
    no_improve = 0
    m = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        cmax = float(np.max(conv))
        if cmax <= 1e-14:
            break

        if top_k >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(conv, -top_k)[-top_k:]

        temp = max(1e-15, 0.05 * cmax)
        w = np.exp((conv[idx] - cmax) / temp)
        ws = float(w.sum())
        if ws <= 0.0:
            break
        w = w / ws

        grad = np.zeros(n, dtype=np.float64)
        for pos in range(len(idx)):
            wk = float(w[pos])
            if wk < 1e-10:
                continue
            k = int(idx[pos])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1)
            grad[i_vals] += 2.0 * wk * a[k - i_vals]

        gmax = float(np.max(np.abs(grad)))
        if gmax <= 1e-15:
            break
        grad = grad / gmax

        best_sc = best_score
        best_cand = best_a.copy()

        for scale in (0.1, 0.25, 0.5, 1.0, 2.0, 4.0, 8.0):
            step = eta * scale
            cand = simplex_project(a - step * grad)
            sc = local_score_norm(cand)
            if sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_sc < best_score - 1e-12:
            best_score = best_sc
            best_a = best_cand.copy()
            a = best_a.copy()
            conv = fast_convolve(a)
            eta = min(eta * 1.15, 0.2)
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    SUBGRAD {label} it={it}: local={best_score:.10f}")
        else:
            no_improve += 1
            eta *= 0.8
            if no_improve >= 30 or eta < 1e-5:
                break

    return normalize_sum1(best_a), best_score, it


def lp_best_response(a):
    """
    Given a (sum=1), solve:
      min t
      s.t. conv(a,q)_k <= t for all k,
           q >= 0, sum q = 1.
    Returns (q, t) or (None, None).
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix, eye, hstack
    except Exception:
        return None, None

    a = normalize_sum1(a)
    n = len(a)
    m = 2 * n - 1

    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)

    pos = 0
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        if length <= 0:
            continue
        idx = np.arange(i_start, i_end + 1)
        rows[pos:pos + length] = k
        cols[pos:pos + length] = idx
        data[pos:pos + length] = a[k - idx]
        pos += length

    if pos != total:
        rows = rows[:pos]
        cols = cols[:pos]
        data = data[:pos]

    A = csr_matrix((data, (rows, cols)), shape=(m, n))
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_ub = hstack((A, -eye(m, format="csr")), format="csr")
    b_ub = np.zeros(m, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n + 1)]

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
        if res.status != 0:
            return None, None
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="interior-point",
            )
            if res.status != 0:
                return None, None
        except Exception:
            return None, None

    q = np.asarray(res.x[:n], dtype=np.float64)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-12:
        return None, None
    q = q / s
    t = float(res.x[-1])
    return q, t


def fictitious_play(a0, time_limit, max_iters=12, verbose=False, label=""):
    """LP fictitious play with line search on self convolution."""
    a = normalize_sum1(a0)
    if len(a) > 1800:
        return a, local_score_norm(a), 0

    best_a = a.copy()
    best_score = local_score_norm(a)
    a_avg = a.copy()
    t0 = time.monotonic()
    it = 0

    alphas = (0.1, 0.2, 0.3, 0.5, 0.7, 1.0)

    while time.monotonic() - t0 < time_limit:
        if it >= max_iters:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        q, _ = lp_best_response(a)
        if q is None:
            break

        it += 1
        best_cand = a.copy()
        best_mix_score = best_score

        for alpha in alphas:
            cand = (1.0 - alpha) * a + alpha * q
            cand = normalize_sum1(cand)
            sc = local_score_norm(cand)
            if sc < best_mix_score:
                best_mix_score = sc
                best_cand = cand

        sc_q = local_score_norm(q)
        if sc_q < best_mix_score:
            best_mix_score = sc_q
            best_cand = normalize_sum1(q)

        if best_mix_score < best_score:
            best_score = best_mix_score
            best_a = best_cand.copy()

        alpha = 2.0 / (it + 2.0)
        a = normalize_sum1((1.0 - alpha) * a + alpha * q)
        a_avg = (it * a_avg + a) / (it + 1)

        sc_avg = local_score_norm(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        if verbose and it % 2 == 0:
            print(f"    FP {label} it={it}: local={best_score:.10f}")

    return normalize_sum1(best_a), best_score, it


def make_initial_candidates(n, parent, rng, max_cands=260):
    """Generate diverse nonnegative initializations of length n."""
    cands = []

    if parent is not None:
        cands.append(interpolate_seq(parent, n))
    else:
        cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    t = np.arange(n, dtype=np.float64)
    d = np.minimum(t, n - 1.0 - t)

    betas = (0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.5)
    alphas = (0.0, 0.2, 0.4, 0.7)

    raw_periods = (n // 2, n // 3, n // 4, max(3, int(math.sqrt(n))), max(3, n // 7))
    periods = []
    for p0 in raw_periods:
        p = max(3, min(n - 1, p0))
        if p not in periods:
            periods.append(p)

    for beta in betas:
        base = np.exp(-beta * d)
        for alpha in alphas:
            for period in periods:
                if alpha == 0.0:
                    a = base.copy()
                else:
                    a = base * (1.0 + alpha * np.cos(2.0 * np.pi * t / float(period)))
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 1e-15:
                    cands.append(a / s)
                if len(cands) > max_cands:
                    return cands

    for _ in range(90):
        power = rng.uniform(0.6, 3.5)
        a = np.abs(rng.standard_normal(n)) ** power
        if rng.random() < 0.5:
            a = a[::-1]
        s = float(a.sum())
        if s > 1e-15:
            cands.append(a / s)
        if len(cands) > max_cands:
            return cands

    raw_len = (50, 150, 300, 600, n)
    lengths = []
    for m0 in raw_len:
        m = max(5, min(n - 1, m0))
        if m not in lengths:
            lengths.append(m)

    for m in lengths:
        for _ in range(8):
            if m <= 0:
                m = n
            idx = rng.choice(n, size=min(m, n), replace=False)
            a = np.zeros(n, dtype=np.float64)
            if rng.random() < 0.5:
                a[idx] = 1.0
            else:
                a[idx] = rng.lognormal(0.0, 0.6, size=len(idx))
            s = float(a.sum())
            if s > 1e-15:
                cands.append(a / s)
            if len(cands) > max_cands:
                return cands

    for _ in range(25):
        slope = rng.uniform(-1.0, 1.0)
        denom = max(1, n - 1)
        a = 1.0 + slope * t / denom
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s > 1e-15:
            cands.append(a / s)

    return cands


def perturb_sequence(a, rng, sigma=0.02, mode="add"):
    """Create a randomized perturbation of a normalized sequence."""
    a = normalize_sum1(a)
    n = len(a)
    if n == 0:
        return a

    if mode in (None, "add"):
        a = a + rng.standard_normal(n) * sigma
    elif mode in ("mul",):
        a = a * np.exp(rng.standard_normal(n) * sigma)
    elif mode in ("mix",):
        b = np.abs(rng.standard_normal(n)) ** rng.uniform(0.5, 2.0)
        b = normalize_sum1(b)
        a = 0.55 * a + 0.45 * b
    else:
        r = rng.random()
        if r < 0.4:
            a = a + rng.standard_normal(n) * sigma
        elif r < 0.7:
            a = a * np.exp(rng.standard_normal(n) * sigma)
        else:
            b = np.abs(rng.standard_normal(n)) ** rng.uniform(0.5, 2.0)
            b = normalize_sum1(b)
            a = 0.55 * a + 0.45 * b

    return normalize_sum1(a)


def basin_hopping(a0, time_limit, rng, verbose=False, label=""):
    """Perturb + short exact coordinate descent basin hopping."""
    a = normalize_sum1(a0)
    n = len(a)
    if n == 0:
        return a, float("inf"), 0

    best_a = a.copy()
    best_score = local_score_norm(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 12:
            break

        sigma = rng.uniform(0.008, 0.045)
        mode_int = int(rng.integers(0, 3))
        mode = ("add", "mul", "mix")[mode_int]
        cand = perturb_sequence(a, rng, sigma, mode)

        t_cd = min(28.0, max(2.0, rem * 0.25))
        cand, sc, _ = exact_cd(cand, t_cd, gs_iter=8, verbose=False, label=f"bh{it}")

        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand.copy()
            if verbose:
                print(f"    BASSIN {label} it={it}: local={best_score:.10f}")

    return normalize_sum1(best_a), best_score, it


def propose_candidate():
    """Main entry point: return the best sequence found within the time budget."""
    start = time.monotonic()
    deadline = start + 1785.0  # leave a small margin for return/serialization.
    rng = np.random.default_rng(20240517)

    best_seq = None
    best_score = float("inf")

    # Try to use the parent construction if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        if p.size > 0 and np.all(np.isfinite(p)):
            p_clip = np.maximum(p, 0.0)
            s = float(p_clip.sum())
            if s > 1e-8:
                parent = p_clip / s
                best_seq = parent.copy()
                best_score = safe_eval(parent.tolist())
                print(f"Parent: n={parent.size}, score={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        best_score = safe_eval(best_seq.tolist())
        if not np.isfinite(best_score):
            best_score = 2.0
        print(f"Initial uniform fallback: score={best_score:.10f}")

    # Choose a primary working length.
    n_base = int(best_seq.size)
    if n_base < 256 or n_base > 2500:
        n_base = 1024

    # Quick diverse candidate generation and local scoring.
    cands = make_initial_candidates(n_base, parent, rng, max_cands=240)
    cands.append(interpolate_seq(best_seq, n_base))

    scored = []
    for cand in cands:
        a = normalize_sum1(cand)
        if a.size != n_base:
            continue
        sc = local_score_norm(a)
        if np.isfinite(sc):
            scored.append((sc, a))

    if len(scored) > 0:
        scored.sort(key=itemgetter(0))
        parent_c = interpolate_seq(best_seq, n_base)
        parent_sc = local_score_norm(parent_c)

        best_candidate = None
        best_candidate_score = float("inf")
        for sc, a in scored[:12]:
            best_seq, best_score = check_update(a, best_seq, best_score, f"quick_n{n_base}")
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = a.copy()

        if parent_sc < best_candidate_score:
            best_candidate_score = parent_sc
            best_candidate = parent_c.copy()

        if best_candidate is None:
            best_candidate = scored[0][1].copy()
    else:
        best_candidate = interpolate_seq(best_seq, n_base)
        best_candidate_score = local_score_norm(best_candidate)

    # LP fictitious play on the primary length.
    if n_base <= 1500:
        rem = deadline - time.monotonic()
        if rem > 140:
            t_lp = min(380.0, rem * 0.25)
            a_fp, s_fp, _ = fictitious_play(best_candidate, t_lp, max_iters=12, verbose=True, label=f"n{n_base}")
            best_seq, best_score = check_update(a_fp, best_seq, best_score, f"fp_n{n_base}")
            if s_fp < best_candidate_score:
                best_candidate = a_fp.copy()
                best_candidate_score = s_fp

    # Intensive exact coordinate descent with occasional restarts.
    current = best_candidate.copy()
    cur_score = local_score_norm(current)
    round_budget = 70.0

    while deadline - time.monotonic() > 360:
        rem = deadline - time.monotonic()
        t_cd = min(round_budget, max(5.0, rem - 360))
        a_opt, s_opt, _ = exact_cd(current, t_cd, gs_iter=11, verbose=True, label=f"cd_n{n_base}")
        best_seq, best_score = check_update(a_opt, best_seq, best_score, f"cd_n{n_base}")

        if s_opt < cur_score - 1e-9:
            current = a_opt.copy()
            cur_score = s_opt
            best_candidate = current.copy()
            best_candidate_score = s_opt
            round_budget = min(round_budget * 1.0, 85.0)
        else:
            current = perturb_sequence(a_opt, rng, 0.018 + 0.02 * rng.random(), "add")
            cur_score = local_score_norm(current)
            if cur_score < best_candidate_score:
                best_candidate = current.copy()
                best_candidate_score = cur_score

    # Subgradient polishing.
    rem = deadline - time.monotonic()
    if rem > 160:
        t_sg = min(110.0, max(20.0, rem - 140))
        a_sg, s_sg, _ = subgradient_search(best_seq, t_sg, top_k=90, verbose=True, label=f"n{best_seq.size}")
        best_seq, best_score = check_update(a_sg, best_seq, best_score, f"sg_n{best_seq.size}")
        if s_sg < best_candidate_score:
            best_candidate = a_sg.copy()
            best_candidate_score = s_sg

    # Try larger lengths quickly, since longer support can improve the ratio.
    rem = deadline - time.monotonic()
    if rem > 180:
        tried = set()
        tried.add(int(best_seq.size))
        for new_n in (1536, 1800, 2048):
            if new_n in tried:
                continue
            tried.add(new_n)
            rem = deadline - time.monotonic()
            if rem < 90:
                break

            a0 = interpolate_seq(best_seq, new_n)
            a_sg, s_sg, _ = subgradient_search(a0, min(45.0, rem * 0.35), top_k=70, verbose=False, label=f"long{new_n}")
            t_cd = min(55.0, max(5.0, rem * 0.35))
            a_cd, s_cd, _ = exact_cd(a_sg, t_cd, gs_iter=9, verbose=False, label=f"longcd{new_n}")

            best_seq2 = a_cd.copy()
            best_seq2, best_score2 = check_update(best_seq2, best_seq, best_score, f"long{new_n}")

            # Local best from this branch.
            if s_cd < s_sg:
                branch_best = a_cd
                branch_sc = s_cd
            else:
                branch_best = a_sg
                branch_sc = s_sg

            if branch_sc < best_candidate_score:
                best_candidate = normalize_sum1(branch_best)
                best_candidate_score = branch_sc

        # If the larger sequence actually became the true best, continue with it.
        if best_seq.size != best_candidate.size:
            pass

    # Basin hopping on the current best true sequence.
    rem = deadline - time.monotonic()
    if rem > 120:
        t_bh = min(140.0, max(20.0, rem - 90))
        a_bh, s_bh, _ = basin_hopping(best_seq, t_bh, rng, verbose=True, label=f"bh_n{best_seq.size}")
        best_seq, best_score = check_update(a_bh, best_seq, best_score, f"bh_n{best_seq.size}")
        if s_bh < best_candidate_score:
            best_candidate = a_bh.copy()
            best_candidate_score = s_bh

    # Final polish on the best sequence observed by the true evaluator.
    rem = deadline - time.monotonic() - 4.0
    if rem > 10 and best_seq is not None:
        n_final = int(best_seq.size)
        if n_final > 2500:
            t_final = min(70.0, max(8.0, rem * 0.5))
            gs_final = 8
        else:
            t_final = min(160.0, max(10.0, rem))
            gs_final = 13
        a_pol, s_pol, _ = exact_cd(best_seq, t_final, gs_iter=gs_final, verbose=True, label="final")
        best_seq, best_score = check_update(a_pol, best_seq, best_score, "final_cd")

        a_pol2, s_pol2, _ = subgradient_search(best_seq, min(25.0, max(4.0, deadline - time.monotonic() - 3.0)), top_k=60, verbose=False, label="final_sg")
        best_seq, best_score = check_update(a_pol2, best_seq, best_score, "final_sg")

    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        best_score = safe_eval(best_seq.tolist())

    # Final normalization and an extremely small positive floor.
    best_seq = normalize_sum1(best_seq)
    best_seq = best_seq + 1e-12
    best_seq = best_seq / float(best_seq.sum())

    final_score = safe_eval(best_seq.tolist())
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL: n={best_seq.size}, score={best_score:.10f}")
    return [float(x) for x in best_seq]
