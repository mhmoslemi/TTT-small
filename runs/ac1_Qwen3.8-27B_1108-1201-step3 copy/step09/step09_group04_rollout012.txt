def p_norm_obj_and_grad_z(z, p):
    """Compute p-norm objective and gradient in log domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        val = 2.0 * n
        grad = np.zeros_like(z)
        return val, grad
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    grad = a * (g_a - np.dot(g_a, a))
    val = 2.0 * n * (S ** (1.0 / p))
    return val, grad

def coordinate_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent with adaptive step sizes."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find peak
        peak_k = b.argmax()
        peak_val = b[peak_k]
        
        # Contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the largest contributor
        top_i = idx[np.argmax(contribs)]
        reduced = False
        
        for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
            delta = a[top_i] * (1 - factor)
            if delta < 1e-16:
                delta = a[top_i]
                factor = 0
            a_trial = a.copy()
            a_trial[top_i] -= delta
            if a_trial[top_i] < 0:
                a_trial[top_i] = 0
            # Redistribute
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] = others + delta * others / others_sum
            sc = score_of(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                reduced = True
                break
        
        if not reduced:
            # Try adding to a non-contributing position
            # Find position with lowest contribution to peak
            if len(idx) < n:
                non_idx = np.setdiff1d(np.arange(n), idx)
                if len(non_idx) > 0:
                    # Add to the position farthest from peak
                    far_i = non_idx[np.argmax(np.abs(non_idx - (peak_k // 2)))]
                    for factor in [1.01, 1.02, 1.05]:
                        a_trial = a.copy()
                        a_trial[far_i] *= factor
                        a_trial = a_trial / a_trial.sum()
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            reduced = True
                            break
        
        if not reduced and step % 1000 == 0 and step > 0:
            break
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0

def structured_init(n, variant=0):
    """Create structured initial sequence of length n (normalized to sum=1)."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 4:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 5:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = 1 - x ** 2
    elif variant == 6:
        # Watermelon shape
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.3 * np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def enforce_symmetry(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym

For a non-symmetric sequence, or when multiple peaks are active, the structure is more complex.

When multiple peaks are active (say k1 and k2):
a[k1-i] + a[i-k1] = a[k2-i] + a[i-k2] = λ for all i in support.

This gives: a[k1-i] - a[k2-i] = a[i-k1] - a[i-k2]

This is a system of difference equations that constrains the structure of a.

For practical purposes, this suggests that the optimal sequence has a specific "frequency" structure related to the peak positions.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using multi-stage p-norm L-BFGS
in log domain, subgradient polishing, coordinate descent, and systematic
length exploration. Key improvements over previous: better p-schedule,
momentum-based restarts, symmetry exploitation, and tighter final polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute score = 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def p_norm_compute(z, p):
    """Compute p-norm objective value and gradient in log domain simultaneously."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return 2.0 * n, np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    grad = a * (g_a - np.dot(g_a, a))
    val = 2.0 * n * (S ** (1.0 / p))
    return val, grad


def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=8, restart_every=3):
    """Multi-stage p-norm L-BFGS in log domain with periodic restarts."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = score_of(a0)

    temps = np.geomspace(p_start, p_end, n_stages)

    for stage_i, stage_p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        frac = (stage_i + 1) / n_stages
        # More iterations early, fewer late
        iters = max(80, int(3000 * (1.8 - 0.8 * frac)))

        def func(z_val):
            val, _ = p_norm_compute(z_val, stage_p)
            return val

        def jac(z_val):
            _, grad = p_norm_compute(z_val, stage_p)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate and update best
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score_of(a_cur)
        if sc < best_score:
            best_score = sc
            best_z = z.copy()

        elapsed = time.monotonic() - t0
        print(f"      p-norm stage {stage_i+1}/{n_stages} p={stage_p:.1f} ({elapsed:.0f}s): score={sc:.6f}")

        # Periodic restart: go back to best if we got stuck
        if (stage_i + 1) % restart_every == 0 and stage_i < n_stages - 1:
            # Check if recent stages improved
            recent_score = sc
            if recent_score > best_score * (1 + 1e-4):
                z = best_z.copy()
                print(f"      Restart from best (score={best_score:.6f})")

    z = best_z
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def subgrad_polish(a0, time_limit, n_max_iters=30000):
    """
    Subgradient Nesterov method using exact max with soft-max weighting.
    Uses O(n log n) gradient via correlation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    prev_y = a.copy()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute soft-max weights over peaks
        b = fast_convolve(y)
        peak_val = b.max()
        # Temperature for soft-max: start wide, narrow over time
        progress = it / max(1, n_max_iters)
        temp = peak_val * (0.1 * (1 - progress) + 0.005 * progress)
        temp = max(temp, 1e-10)
        weights = np.exp((b - peak_val) / temp)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        weights = weights / w_sum

        # Compute gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        # Term 1: sum_k w[k] * a[k-i] = correlate(w, a, 'valid')[i]
        t1 = np.correlate(weights, a, mode='valid')
        # Term 2: sum_k w[k] * a[i-k]
        # This equals sum_j a[j] * w[i-j] for j <= i
        # = cross-correlation of a and w at lag i
        # Use FFT: it's the same as correlate of a with reversed w, shifted
        # Simpler: t2[i] = sum_{j=0}^{min(i,n-1)} w[i-j] * a[j]
        # This is the 'valid' mode of correlate(w, a) but with different alignment
        # Actually: let's compute via full correlation and extract
        # t2[i] = (a * w_flipped)[i + n - 1] where * is linear conv... 
        # Simplest: just compute directly (O(n^2) but n is at most ~2000)
        # For n up to 2000, O(n^2) = 4M ops per iter, at 100M ops/s that's 0.04s per iter
        # Too slow for 30000 iters. Let's use FFT.
        
        # t2[i] = sum_j a[j] * w[i-j] for 0<=j<n, 0<=i-j<2n-1
        # Pad a to length 2n-1, compute correlation with w
        a_padded = np.concatenate([a, np.zeros(n-1)])
        # t2[i] = sum_j a_padded[j] * w[i-j] = (a_padded * w_rev)[i + (2n-1) - 1 - ...]
        # Actually: convolution (a_padded * w_rev)[m] = sum_j a_padded[j] * w_rev[m-j]
        # w_rev[k] = w[2n-2-k]
        # (a_padded * w_rev)[m] = sum_j a_padded[j] * w[2n-2-(m-j)] = sum_j a_padded[j] * w[2n-2-m+j]
        # We want t2[i] = sum_j a[j] * w[i-j]
        # Set 2n-2-m+j = i-j => j = i - (2n-2-m) + j... this is getting circular.
        
        # Let me just use: t2[i] = sum_{j=0}^{n-1} a[j] * w[i-j] * 1{0 <= i-j < 2n-1}
        # = sum_{j=max(0,i-(2n-2))}^{min(i, n-1)} a[j] * w[i-j]
        # For i in 0..n-1: j ranges from max(0, i-2n+2) to min(i, n-1) = 0 to min(i, n-1)
        # Since i < n and 2n-2 >= n for n >= 2, the lower bound is always 0.
        # So t2[i] = sum_{j=0}^{i} a[j] * w[i-j] for i = 0..n-1
        # This is exactly the first n values of the linear convolution of a and w!
        # conv(a, w)[i] = sum_j a[j] * w[i-j] for j where both are defined
        # Since w has length 2n-1 and a has length n, conv has length n + 2n-1 - 1 = 3n-2
        # conv(a, w)[i] = sum_{j=0}^{min(i,n-1)} a[j] * w[i-j] for i where 0<=i-j<2n-1
        # For i = 0..n-1: j from 0 to i, and i-j from 0 to i < n < 2n-1, so all valid.
        # So t2[i] = conv(a, w)[i] for i = 0..n-1.
        
        # Compute conv(a, w) via FFT
        conv_size = n + (2 * n - 1) - 1  # = 3n-2
        fft_size = 1 << (conv_size - 1).bit_length()
        A = rfft(a, fft_size)
        W = rfft(weights, fft_size)
        conv_aw = irfft(A * W, fft_size)[:n]  # First n elements
        t2 = conv_aw

        grad = t1 + t2

        # Project onto tangent space of simplex (sum = 0)
        grad = grad - grad.mean()

        # Nesterov accelerated update
        alpha = 2.0 / (it + 2)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        beta = (it + 1.0 - 1.0) / t_k

        y_new = y - alpha * grad / peak_val
        y_new = (1.0 - beta) * y_new + beta * prev_y
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = 1.0
        y_new = y_new / s_y

        prev_y = y
        y = y_new

        # Evaluate periodically
        if (it + 1) % 200 == 0 or it < 10:
            sc = score_of(y)
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
                if (it + 1) % 2000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_steps=30000):
    """Aggressive coordinate descent with peak shaving and redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_count > 5000:
            break

        b = fast_convolve(a)
        peak_k = int(b.argmax())
        peak_val = b[peak_k]

        # Find all significant peaks (within 5% of max)
        threshold = peak_val * 0.95
        peak_indices = np.where(b >= threshold)[0]

        improved = False

        # Strategy: reduce top contributors to the main peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try single-coordinate reductions
        top_coords = idx[np.argsort(contribs)[::-1][:10]]
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.995, 0.99, 0.98, 0.96, 0.93]:
                delta = a[i] * (1.0 - factor)
                if delta < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = a[i] - delta
                if a_trial[i] < 0:
                    a_trial[i] = 0.0
                    delta = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += delta * a_trial[mask] / others_sum
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve_count = 0
                    improved = True
                    break
            if improved:
                break

        # Strategy: pair adjustment
        if not improved:
            top_pairs = idx[np.argsort(contribs)[::-1][:5]]
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i >= n or j >= n or i < 0 or j < 0 or i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.99, 0.97, 0.95]:
                    for f2 in [0.99, 0.97, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial /= st
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            no_improve_count = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve_count += 1

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDesc step={step+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence of length n, normalized to sum=1."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x * (1.0 - x))
    elif variant == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.5)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif variant == 5:
        x = np.linspace(-1, 1, n)
        a = (1.0 - x ** 2) * (1.0 + 0.2 * x ** 4)
    elif variant == 6:
        # Watermelon with ripple
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1.0 + 0.15 * np.cos(3.0 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 7:
        # Super-Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 4 * 2.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def enforce_symmetry(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    a_sym = 0.5 * (a + a[::-1].copy())
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym

            a_opt, _ = subgrad_polish(best_state['seq'], t_sub, n_max_iters=30000)
            try_update(a_opt, "final_subgrad")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_sub = min(rem * 0.5, 35.0)
        if t_sub > 10:
            a_opt, _ = coordinate_descent(best_state['seq'], t_sub, n_steps=10000)
            try_update(a_opt, "final_coord")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
