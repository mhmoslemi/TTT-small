def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    k = k[-1]
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0.0)

        if s < 1e-15:
            return None, current_active
        b_val = b_val / s
        M_val = float(M_var.value)

        full_conv = T_full @ b_val
        violated = full_conv > M_val + 1e-9
        if not violated.any():
            return b_val, current_active

        violated_idx = np.where(violated)[0]
        worst_order = violated_idx[np.argsort(full_conv[violated_idx])[::-1]]
        added = 0
        for idx in worst_order:
            if idx not in current_active:
                current_active = np.append(current_active, idx)
                added += 1
            if added >= 20:
                break
        if len(current_active) >= len(T_full):
            break

    return b_val, current_active


def nesterov_lp_active(a0, time_limit, solver, n_active_init=100, n_active_max=300):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()

    T_full = build_conv_matrix(a)

    b_init = np.ones(n) / n
    conv_init = T_full @ b_init
    active_indices = np.argsort(conv_init)[::-1][:min(n_active_init, 2 * n - 1)]

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    it = 0
    momentum_boost = 1.0

    while time.monotonic() - t0 < time_limit:
        if it % 20 == 0:
            T_full = build_conv_matrix(y)

        b_new, active_indices = solve_lp_active_set(y, T_full, active_indices, solver, max_inner=10)
        if b_new is None:
            b_new = a.copy()

        alpha_base = 2.0 / (it + 2)
        alpha = min(alpha_base * momentum_boost, 0.8)

        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 50 == 0:
            if best_score < prev_score - 1e-8:
                momentum_boost = min(momentum_boost * 1.02, 1.5)
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score

            if stall_count >= 10:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                active_indices = np.argsort(T_full @ a)[::-1][:min(n_active_init, 2 * n - 1)]
                momentum_boost = 1.0
                stall_count = 0

            if it % 500 == 0 and it > 0:
                elapsed = time.monotonic() - t0
                print(f"    NLAP iter={it} ({elapsed:.0f}s): score={best_score:.6f} nact={len(active_indices)}")

        it += 1

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.03, maxiter_total=20000):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter_total // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final


def coordinate_descent_polish(a0, time_limit, n_rounds=500):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    order = np.arange(n)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if rnd % 3 == 0:
            order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            b_curr = fast_convolve(a)
            peak_k = int(np.argmax(b_curr))

            # Try moving mass from i to a few other positions
            # Find positions where adding mass would help most
            # Contribution of a[i] to peak: a[i] * a[peak_k - i] (if valid)
            # We want to move to position j where a[peak_k - j] is small
            contrib_to_peak = a[np.clip(peak_k - i, 0, n - 1)] if 0 <= peak_k - i < n else 0

            # Try a few candidates
            for frac in [0.05, 0.15, 0.3]:
                amount = a[i] * frac
                if amount < 1e-15:
                    continue
                # Move to the position with least contribution to peak
                rest = np.delete(np.arange(n), i)
                contribs = np.array([a[np.clip(peak_k - j, 0, n - 1)] if 0 <= peak_k - j < n else 0 for j in rest])
                # Pick the one with smallest contribution
                best_j = rest[np.argmin(contribs)]
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[best_j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break

        if rnd % 100 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD round={rnd} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = np.maximum(a, 0) ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.exp(-2 * (x - 0.5) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(2 * np.pi * 2 * x) ** 2
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Active-set Nesterov LP (main workhorse)
    print(f"\n=== Phase 1: Active-Set Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.55
    print(f"  Budget: {t_alloc:.0f}s")
    a_lp, _ = nesterov_lp_active(best_seq, t_alloc, solver, n_active_init=120, n_active_max=400)
    try_update(a_lp, "nesterov_lp_1")

    # Phase 2: Second Nesterov LP round (tighter, longer)
    print(f"\n=== Phase 2: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.25
    print(f"  Budget: {t_alloc:.0f}s")
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_active(best_seq, t_alloc, solver, n_active_init=200, n_active_max=500)
        try_update(a_lp2, "nesterov_lp_2")

    # Phase 3: L-BFGS LogSumExp polishing
    print(f"\n=== Phase 3: L-BFGS LogSumExp (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.12
    print(f"  Budget: {t_alloc:.0f}s")
    if t_alloc > 20:
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=50.0, T_end=0.02, maxiter_total=15000)
        try_update(a_lb, "lbgfs_1")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.3, 1.5, 0.75, 2.0]:
        alt_n = int(n_best * mult)
        if 500 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.3, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_active(init_alt, t_lp, solver, n_active_init=100, n_active_max=300)
        try_update(a_alt, f"lp_n{alt_n}")

    # Phase 5: Coordinate descent polish
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.4, 80.0)
        a_cd, _ = coordinate_descent_polish(best_seq, t_cd, n_rounds=200)
        try_update(a_cd, "coord_desc")

    # Phase 6: Final L-BFGS
    print(f"\n=== Phase 6: Final L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.01, maxiter_total=8000)
        try_update(a_fin, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
