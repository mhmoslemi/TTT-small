def fast_convolve(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T @ b)[k] = sum_j a_fixed[j] * b[k-j]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def get_lp_solver():

def solve_lp_best_response(a_fixed, solver=None, verbose=False):
    """Solve LP: min M s.t. T@a_fixed @ b <= M, sum(b)=1, b>=0.
    Returns (b*, M*, dual_y) where dual_y are the dual variables for the inequality constraints."""
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a_fixed)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: (a_fixed * b)[k] <= M for all k
    # We use the direct formulation
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    # Get dual variables for inequality constraints
    dual_y = None
    try:
        # The dual variables correspond to the T @ b_var <= M_var constraint
        # In cvxpy, dual values are accessed via constraint.dual_value
        if hasattr(constraints[0], 'dual_value') and constraints[0].dual_value is not None:
            dual_y = np.asarray(constraints[0].dual_value, dtype=np.float64)
            dual_y = np.maximum(dual_y, 0)
    except Exception:
        dual_y = None
    
    M_val = float(M_var.value)
    return b_val, M_val, dual_y


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=500000):
    """Nesterov-accelerated LP best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    rng = np.random.default_rng(42)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_new, M_b, dual_y = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Check improvement
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # Stall detection and perturbation
        if it % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            
            if stall_count >= 10:
                # Perturb
                noise_scale = 0.05 * (stall_count - 9)
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def dual_guided_mirror_descent(a0, time_limit, n_active_frac=0.1):
    """Mirror descent using dual-guided subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Compute subgradient: weight near-max lags
        threshold = max_b * (1.0 - n_active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Weight active lags by (b[k] - threshold) for smoother gradient
        weights = b[active] - threshold
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(active))
            w_sum = len(active)
        weights = weights / w_sum
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]
        
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break
        
        gap = best_score - f_lb
        if gap < 1e-12:
            break
        
        polyak_lr = gap / (g_norm_inf ** 2 + 1e-30)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)
        
        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.4f}")
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000):
    """L-BFGS on LogSumExp approximation of the max."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                wk = w[k]
                if wk < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def local_perturbation_search(a0, time_limit, n_steps=50000):
    """Local search via targeted perturbations near peak lags."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_val = b.max()
        threshold = peak_val * 0.95
        near_peaks = np.where(b >= threshold)[0]
        
        # Find positions contributing most to peaks
        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]
        
        # Try reducing high-contribution positions and adding to low ones
        top_k = int(np.argmax(contrib))
        low_k = int(np.argmin(contrib))
        
        if a[top_k] < 1e-14 or top_k == low_k:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-14:
                continue
            amount = a[i] * 0.05
            a_trial = a.copy()
            a_trial[i] -= amount
            a_trial[j] += amount
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
        else:
            amount = a[top_k] * 0.1
            a_trial = a.copy()
            a_trial[top_k] -= amount
            a_trial[low_k] += amount
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            a = a_trial
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
    
    return best_a, best_score


def structured_init(n, rng):

        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(5 * x) + 0.1 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)

    else:
        a = np.ones(n) * (1.0 + 0.3 * np.sin(4 * np.pi * np.arange(n) / n))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T @ b)[k] = sum_j a_fixed[j]*b[k-j]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (clipped for numerical safety)."""
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None

def solve_lp_best_response(a_fixed, solver=None):

    try:

        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)

def dual_guided_mirror_descent(a0, time_limit, n_active_frac=0.05, use_dual_weights=True):
    """Mirror descent with dual-guided or threshold-based subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    
    for step in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Weight lags near the maximum
        threshold = max_b * (1.0 - n_active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Use quadratic weighting for smoother gradient
        weights = (b[active] - threshold) ** 2
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(active))

            weights = weights / w_sum
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)

            break
        
        gap = best_score - f_lb

def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.05, maxiter=10000):
    """L-BFGS on LogSumExp approximation with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                wk = w[k]
                if wk < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(15, maxiter // n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

            no_improve += 1
            # Occasional random kick
            if no_improve % 200 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-14:
                    amount = a[i] * 0.02
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
        
        if no_improve > 10000:
            break
    
    return best_a, best_score


def structured_init(n, rng):
    """Diverse initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(3 * x) + 0.15 * np.cos(5 * x) + 0.1 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 15 * (x - 0.5) ** 2)
    else:
        a = np.ones(n) * (1.0 + 0.2 * np.sin(6 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns the best sequence found within budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"No parent: {e}")
    
    n_best = len(best_seq) if best_seq is not None else 960
    
    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp, "lp_1")
    
    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = dual_guided_mirror_descent(best_seq, t_alloc, n_active_frac=0.04)
        try_update(a_md, "md_1")
    
    # === Phase 3: LP Round 2 ===
    print(f"\n=== Phase 3: LP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 280.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "lp_2")
    
    # === Phase 4: L-BFGS Polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=80.0, T_end=0.05, maxiter=8000)
        try_update(a_lb, "lbgfs_1")
    
    # === Phase 5: Multi-Length Exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.8f}) ===")
    # Try lengths around n_best
    alt_lengths = []
    for mult in [1.05, 1.1, 1.15, 1.2, 1.3, 1.5, 0.9, 0.85]:
        alt_n = int(round(n_best * mult))
        if 600 <= alt_n <= 2000 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)
    alt_lengths.sort()
    
    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.25, 150.0)
        if t_total < 25:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")
        
        init_alt = interpolate_sequence(best_seq, alt_n)
        
        t_lp = t_total * 0.75
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")
        
        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            t_md = min(rem * 0.3, 30.0)
            if t_md > 8:
                a_alt2, _ = dual_guided_mirror_descent(best_seq, t_md, n_active_frac=0.03)
                try_update(a_alt2, f"md_n{alt_n}")
    
    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.35, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=15.0, T_end=0.03, maxiter=6000)
        try_update(a_fin, "final_lbgfs")
    
    rem = deadline - time.monotonic()
    if rem > 25:
        t_lp_fin = min(rem * 0.35, 60.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=20000)
        try_update(a_lp_fin, "final_lp")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ls = min(rem * 0.4, 30.0)
        a_ls2, _ = local_perturbation_search(best_seq, t_ls, n_steps=15000)
        try_update(a_ls2, "final_local")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
