The LP is:
```
min M
s.t. (T @ b)_k <= M for all k
     1^T b = 1
     b >= 0
```

The dual has variables `lambda_k >= 0` for each lag constraint and `nu` (free) for the equality constraint. The KKT conditions give:
- `T^T @ lambda + nu * 1 - mu = 0` where `mu_i >= 0` is the dual for `b_i >= 0`
- Complementarity: `lambda_k * ((T @ b)_k - M) = 0` and `mu_i * b_i = 0`
- `1^T b = 1`

At the optimum, `b` is supported on coordinates where `mu_i = 0`, i.e., where `T^T @ lambda + nu = 0`.

This suggests an iterative method:
1. Start with uniform `b`
2. Compute the "tight" lags: `S = {k : (T @ b)_k = M}`
3. Solve the smaller system: find `b` supported on the "free" coordinates that equalizes `(T @ b)_k` over `k in S`
4. Update `S` and repeat

This is essentially a **water-filling** algorithm! And it can be much faster than a full LP solve.

def water_filling_best_response(a_fixed, max_iter=1000):
    """Find best response b using water-filling (iterative)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)  # (size, n)
    
    b = np.ones(n) / n
    for _ in range(max_iter):
        Tb = T @ b
        M = Tb.max()
        # Tight lags
        tol = 1e-10 * M
        tight = Tb >= M - tol
        n_tight = tight.sum()
        
        if n_tight < 2:
            break
        
        # For the water-filling step, we want to equalize Tb[k] for k in tight set
        # This means solving: T[tight, :] @ b = M * 1, 1^T b = 1, b >= 0
        # But this is an underdetermined system (n_tight + 1 equations, n unknowns)
        # We want the minimum-norm solution or the solution closest to current b
        
        # Actually, let's use a simpler approach: gradient descent on the 
        # "violation" of the equalization constraint
        grad = np.zeros(n)
        for k in np.where(tight)[0]:
            grad += T[k, :]
        # The gradient of (Tb[k] - M)^2/2 w.r.t. b for k in tight set
        # is T[tight, :].T @ (Tb[tight] - M)
        # But we want to REDUCE the max, so we push mass away from tight lags
        
        # Simpler: just do a subgradient step
        # The subgradient of max(Tb) w.r.t. b is the average of T[k,:] for k in argmax
        active = np.where(tight)[0][:n_tight]  # limit to avoid overflow
        subgrad = T[active, :].mean(axis=0)
        
        # Step: reduce b where subgrad is high, increase where low
        # Project onto simplex
        step_size = 0.01
        b_new = b - step_size * (subgrad - subgrad.mean())
        b_new = np.maximum(b_new, 0)
        b_new = b_new / b_new.sum()
        
        if np.max(np.abs(b_new - b)) < 1e-12:
            b = b_new
            break
        b = b_new
    
    return b

    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= cp.ones(size) * M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternate_nesterok_tk(a0, time_limit, solver=None, max_iters=50000,
                             perturb_every=200, perturb_scale=0.03):
    """LP alternating with standard Nesterov t_k momentum and periodic perturbations."""
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_k = 1.0
    rng = np.random.default_rng(int(time.time()) % 100000)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = 1.0 - t_k / t_new

        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        a_new = y_new + (t_k - 1.0) / t_new * (y_new - y)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        y = y_new
        a = a_new
        t_k = t_new

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

        if it % perturb_every == perturb_every - 1:
            noise = rng.random(n) * perturb_scale * y.max()
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            t_k = 1.0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t_k it={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_active_frac=0.05, max_iters=200000):
    """Subgradient method in log-space with Polyak step size."""
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    lr = 0.3
    no_improve = 0

    for step in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Gradient in z-space: a * (g - a . g)
        S = np.dot(g, a)
        grad_z = a * (g - S)
        g_norm = np.sqrt(np.dot(grad_z, grad_z))
        if g_norm < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-10:
            break

        polyak_lr = gap / (2.0 * best_score * g_norm + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * grad_z
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 3.0)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.4f}")

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000):
    """L-BFGS on log-sum-exp smooth approximation in log-space."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros(len(z_in))
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(len(z_in))
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - len(z_in))
                je = min(len(z_in) - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def coordinate_descent_polish(a0, time_limit, max_rounds=5000):
    """Coordinate descent: for each element, find optimal value given others fixed."""
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Random permutation of coordinates
        perm = rng.permutation(n)
        improved = False

        for i in perm:
            if a[i] < 1e-15:
                continue
            # Try moving mass from i to a random other coordinate
            j = rng.integers(n)
            if j == i:
                continue

            for frac in [0.01, 0.02, 0.05, 0.1]:
                amount = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and rnd > 500:
            break

        if rnd % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def peak_directed_search(a0, time_limit, max_rounds=3000):
    """Targeted search: identify peak lags, reduce contributing mass, redistribute."""
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(42)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find elements that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.zeros(n)
        idx = np.arange(js, je + 1)
        contrib[idx] = a[idx] * a[peak_k - idx]

        # Find "quiet" positions (low contribution to all peaks)
        all_contrib = np.zeros(n)
        top_lags = np.argsort(b)[::-1][:5]
        for k in top_lags:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            idx2 = np.arange(js2, je2 + 1)
            all_contrib[idx2] += a[idx2] * a[k - idx2]

        # Move mass from high-contrib src to low-contrib dst
        src_candidates = np.argsort(contrib)[::-1][:10]
        dst_candidates = np.argsort(all_contrib)[:10]

        improved = False
        found = False
        for si in src_candidates:
            if a[si] < 1e-12:
                continue
            for di in dst_candidates:
                if si == di:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not improved and rnd > 300:
            # Add small random perturbation
            noise = rng.random(n) * 0.01 * a.max()
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if rnd % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDir {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2.0
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.5
        a = a ** 0.5
    elif choice == 6:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    else:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) ** 1.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_at_length(a_init, time_budget, solver, rng):
    """Full optimization pipeline at a fixed length."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    t0 = time.monotonic()

    def remaining():
        return max(0.0, time_budget - (time.monotonic() - t0))

    best_a = a.copy()
    best_score = evaluate_safe(a)

    # Stage 1: LP alternating with Nesterov t_k
    t_alloc = min(remaining() * 0.40, time_budget * 0.40)
    if t_alloc > 15:
        a_lp, _ = lp_alternate_nesterok_tk(a, t_alloc, solver=solver)
        sc = evaluate_safe(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()

    # Stage 2: Subgradient polish
    t_alloc = min(remaining() * 0.20, 120.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_polish(best_a, t_alloc)
        sc = evaluate_safe(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()

    # Stage 3: L-BFGS LogSumExp
    t_alloc = min(remaining() * 0.20, 100.0)
    if t_alloc > 10:
        a_lb = lbgfs_logsumexp(best_a, t_alloc, T_start=50.0, T_end=0.1, maxiter=5000)
        sc = evaluate_safe(a_lb)
        if sc < best_score:
            best_score = sc
            best_a = a_lb.copy()

    # Stage 4: Peak-directed search
    t_alloc = min(remaining() * 0.15, 60.0)
    if t_alloc > 5:
        a_pd, _ = peak_directed_search(best_a, t_alloc)
        sc = evaluate_safe(a_pd)
        if sc < best_score:
            best_score = sc
            best_a = a_pd.copy()

    # Stage 5: Coordinate descent
    t_alloc = min(remaining() * 0.10, 40.0)
    if t_alloc > 5:
        a_cd, _ = coordinate_descent_polish(best_a, t_alloc, max_rounds=2000)
        sc = evaluate_safe(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 960
        best_seq = structured_init(n_default, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best: n={n_parent}, score={best_score:.8f}")
    print(f"Total budget: {budget_s:.0f}s")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Optimize at parent length ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Optimize at n={n_parent} (budget={t1:.0f}s) ===")
        a_opt, sc = optimize_at_length(best_seq, t1, solver, rng)
        if a_opt is not None:
            try_update(a_opt, f"main_n{n_parent}")

    # === Phase 2: Alternative lengths ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 2: Alternative lengths (remaining={rem:.0f}s) ===")

    alt_lengths = []
    for mult in [1.1, 1.2, 1.3, 1.5, 0.9, 0.8, 1.7, 2.0]:
        alt_n = int(n_parent * mult)
        if 400 <= alt_n <= 3500 and alt_n != n_parent:
            alt_lengths.append(alt_n)
    # Also try some fixed good lengths
    for fixed_n in [768, 1024, 1280, 1536]:
        if fixed_n != n_parent and 400 <= fixed_n <= 3500:
            alt_lengths.append(fixed_n)

    alt_lengths = sorted(set(alt_lengths))[:4]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.30, 250.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={alt_n} (budget={t_alloc:.0f}s)")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        a_opt, sc = optimize_at_length(cur_best, t_alloc, solver, rng)
        if a_opt is not None:
            try_update(a_opt, f"alt_n{alt_n}")

    # === Phase 3: Multi-start at best length ===
    rem = deadline - time.monotonic()
    if rem > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Multi-start at n={n_best} (remaining={rem:.0f}s) ===")

        for attempt in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.35, 200.0)
            if t_alloc < 20:
                break

            # Create a perturbed version of the best
            noise_scale = 0.05 * (attempt + 1)
            noise = rng.random(n_best) * noise_scale
            perturbed = best_seq + noise
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            print(f"  Attempt {attempt+1}, noise={noise_scale:.3f}, budget={t_alloc:.0f}s")
            a_opt, sc = optimize_at_length(perturbed, t_alloc, solver, rng)
            if a_opt is not None:
                try_update(a_opt, f"restart_{attempt+1}")

    # === Phase 4: Try a fresh structured init at the best length ===
    rem = deadline - time.monotonic()
    if rem > 50:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Fresh inits at n={n_best} (remaining={rem:.0f}s) ===")

        for init_i in range(2):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 180.0)
            if t_alloc < 20:
                break

            fresh = structured_init(n_best, rng)
            sc_fresh = evaluate_safe(fresh)
            print(f"  Fresh init {init_i}: score={sc_fresh:.6f}")

            a_opt, sc = optimize_at_length(fresh, t_alloc, solver, rng)
            if a_opt is not None:
                try_update(a_opt, f"fresh_{init_i}")

    # === Phase 5: Final intensive polish ===
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        print(f"\n=== Phase 5: Final polish (remaining={rem:.0f}s) ===")
        n_best = len(best_seq)
        t_polish = min(rem * 0.5, 120.0)

        # Subgradient with very tight active set
        a_sg, _ = subgradient_polish(best_seq, t_polish * 0.5, n_active_frac=0.02)
        try_update(a_sg, "final_subgrad")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.4, 60.0)
            a_cd, _ = peak_directed_search(best_seq, t_cd)
            try_update(a_cd, "final_peakdir")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_lb = min(rem * 0.5, 40.0)
            a_lb = lbgfs_logsumexp(best_seq, t_lb, T_start=5.0, T_end=0.05, maxiter=3000)
            try_update(a_lb, "final_lbgfs")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
