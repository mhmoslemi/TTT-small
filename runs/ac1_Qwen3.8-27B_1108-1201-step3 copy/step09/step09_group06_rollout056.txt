import numpy as np

def fast_convolve(a):
    """Linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(a, b):
    """Linear convolution of two sequences via FFT. Both length n."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def t_transpose_matvec(w, a):
    """
    Compute T_a^T @ w where T_a is the convolution matrix defined by a.
    (T_a^T w)_j = sum_{l=0}^{n-1} a_l * w_{l+j}
    Uses FFT for O(n log n) complexity.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    # c_j = (flip(a) * w)_{j+n-1}
    flip_a = a[::-1]
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(flip_a, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)[:size]
    return conv[n-1:n-1+n]


def hessian_matvec(v, a):
    """Compute (T_a^T T_a) @ v."""
    tv = convolve_pair(a, v)
    return t_transpose_matvec(tv, a)


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    i = np.max(np.where(cond)[0]) + 1
    top_i = u[:i]
    rho = (top_i.sum() - 1.0) / i
    return np.maximum(v - rho, 0.0)


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    n = len(a)

    return T


def solve_lp_best_response(a, solver=None):
    """Solve: min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0."""
    if solver is None:

    T = build_conv_matrix(a)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):

        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def estimate_lipschitz(a, n_power_iter=30):
    """Estimate 2*lambda_max(T_a^T T_a) via power iteration."""
    n = len(a)
    rng = np.random.default_rng(123)
    v = rng.standard_normal(n)
    v /= np.linalg.norm(v)
    rho = 0.0
    for _ in range(n_power_iter):
        Hv = hessian_matvec(v, a)
        norm = np.linalg.norm(Hv)
        if norm < 1e-15:
            break
        rho = float(v @ Hv)
        v = Hv / norm
    return max(2.0 * rho + 1e-10, 1e-6)


def fista_qp_for_a(a, max_iters=100):
    """
    Solve: min ||conv(a, b) - 1/(2n-1)||^2 s.t. sum(b)=1, b>=0
    using FISTA. Returns optimal b.
    """
    n = len(a)
    c = np.ones(2 * n - 1) / (2 * n - 1)
    
    L = estimate_lipschitz(a)
    
    def qp_grad(b):
        tb = convolve_pair(a, b)
        diff = tb - c
        return 2.0 * t_transpose_matvec(diff, a)
    
    b = np.ones(n) / n
    y = b.copy()
    t_k = 1.0
    
    for k in range(max_iters):
        g = qp_grad(y)
        b_new = project_simplex(y - g / L)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = b_new + (t_k - 1.0) / t_new * (b_new - b)
        b = b_new
        t_k = t_new
    
    return b
