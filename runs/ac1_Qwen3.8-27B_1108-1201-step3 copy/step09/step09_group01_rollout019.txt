def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def evaluate_safe(a):

def simulated_annealing(a0, time_limit, rng, verbose=False):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M
    
    T = M * 0.01  # Initial temperature
    T_min = M * 1e-6
    cooling = 0.999
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Choose a move
        move_type = rng.integers(0, 4)
        a_new = a.copy()
        
        if move_type == 0:
            # Single coordinate change
            i = rng.integers(0, n)
            frac = rng.uniform(0.01, 0.5)
            a_new[i] = a[i] * (1 + frac * rng.uniform(-1, 1))
        elif move_type == 1:
            # Swap mass between two coordinates
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.5)
            direction = rng.choice([-1, 1])
            a_new[i] = a[i] + direction * amt
            a_new[j] = a[j] - direction * amt
        elif move_type == 2:
            # Block perturbation
            center = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 20))
            lo = max(0, center - width)
            hi = min(n, center + width + 1)
            a_new[lo:hi] = a[lo:hi] * (1.0 + rng.standard_normal(hi - lo) * 0.1)
        else:
            # Gaussian perturbation
            a_new = a + rng.standard_normal(n) * (M / n) * 0.5
        
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        
        new_conv = fast_convolve(a_new)
        new_M = float(new_conv.max())
        new_score = 2.0 * n * new_M
        old_score = 2.0 * n * M
        
        delta = new_score - old_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-30)):
            a = a_new
            conv = new_conv
            M = new_M
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                best_score = new_score
        
        T *= cooling
        if T < T_min:
            T = T_min
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return _lp_solver_cached
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build conv matrix: T[k, j] such that (T @ b)[k] = conv(a,b)[k]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def vectorized_exact_cd(a0, time_limit, n_samples=80, verbose=False, label=""):
    """Vectorized exact coordinate descent with dense line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                t_candidates = np.array([1e-8, 1e-6, 1e-4, 1e-3, max(a.max() * 0.005, 1e-5)])
            else:
                t_min = -a[i] * 0.999
                t_max = min(1.0 - a[i], a[i] * 2.0)
                if t_max <= t_min:
                    t_max = t_min + a[i] * 0.5
                t_candidates = np.linspace(t_min, t_max, n_samples)

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected_conv = conv[k_start:k_end + 1]
            a_affected = a[0:affected_len]

            has_diag = (2 * i >= k_start) and (2 * i <= k_end)

            if has_diag:
                diag_idx = 2 * i - k_start
                mask = np.ones(affected_len, dtype=bool)
                mask[diag_idx] = False
                nd_conv = affected_conv[mask]
                nd_a = a_affected[mask]
                diag_val = affected_conv[diag_idx]
                diag_a = a_affected[diag_idx]

                # Vectorized: (n_samples, num_non_diag)
                nd_max = (nd_conv[np.newaxis, :] + t_candidates[:, np.newaxis] * (2.0 * nd_a[np.newaxis, :])).max(axis=1)
                diag_max = diag_val + 2.0 * t_candidates * diag_a + t_candidates ** 2
                new_affected_max = np.maximum(nd_max, diag_max)
            else:
                new_affected_max = (affected_conv[np.newaxis, :] + t_candidates[:, np.newaxis] * (2.0 * a_affected[np.newaxis, :])).max(axis=1)

            new_M = np.maximum(new_affected_max, unaff_max)
            best_idx = new_M.argmin()
            best_new_M = new_M[best_idx]
            best_t = t_candidates[best_idx]

            if best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_t, 0.0)
                changes = 2.0 * best_t * a_affected
                if has_diag:
                    changes = changes.copy()
                    changes[diag_idx] = 2.0 * best_t * (a[i] - best_t) + best_t * best_t
                conv[k_start:k_end + 1] += changes
                M = best_new_M
                improved = True

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  VEC_CD {label} sweep={sweeps} ({elapsed:.0f}s): M={M:.10f} score={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_cd(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent with tiny multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    factors = [0.9995, 0.999, 0.998, 0.995, 0.99, 0.985,
               1.0005, 1.001, 1.002, 1.005, 1.01, 1.015, 1.02]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_affected = a[0:affected_len]
            affected_conv = conv[k_start:k_end + 1]

            has_diag = (2 * i >= k_start) and (2 * i <= k_end)
            diag_idx = 2 * i - k_start if has_diag else -1

            best_t = 0.0
            best_new_M = M

            for f in factors:
                t = a[i] * (f - 1.0)
                new_vals = affected_conv + 2.0 * t * a_affected
                if has_diag:
                    new_vals = new_vals.copy()
                    new_vals[diag_idx] = conv[2 * i] + 2.0 * t * a[i] + t * t
                new_M = max(unaff_max, float(new_vals.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_t = t

            if best_t != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_t, 0.0)
                changes = 2.0 * best_t * a_affected
                if has_diag:
                    changes = changes.copy()
                    changes[diag_idx] = 2.0 * best_t * (a[i] - best_t) + best_t * best_t
                conv[k_start:k_end + 1] += changes
                M = best_new_M
                improved = True

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            break

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FineCD {label} sweep={sweeps} ({elapsed:.0f}s): M={M:.12f} score={best_score:.8f}")

    return best_a, best_score, sweeps


def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with Nesterov averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma = 2.0 / (it + 2.0)
        a_avg = (1 - gamma) * a_avg + gamma * a
        b_avg = (1 - gamma) * b_avg + gamma * b

        for cand in [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score:
                best_score = score
                best_a = cand_n.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FP it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def smooth_gradient_descent(a0, time_limit, beta_init=50.0, lr_init=0.01, verbose=False):
    """Gradient descent using log-sum-exp smooth approximation of max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        max_conv = conv.max()
        scaled = beta * (conv - max_conv)
        scaled = np.clip(scaled, -60, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient: grad[i] = sum_k weights[k] * d(conv[k])/da[i]
        grad = np.zeros(n)
        for k in range(len(weights)):
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if j == i:
                    grad[i] += w * a[j]
                else:
                    grad[i] += 2.0 * w * a[j]

        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SmoothGD it={it} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.6f} beta={beta:.1f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, rng, verbose=False):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M

    T_init = max(M * 0.005, 1e-5)
    T_min = max(M * 1e-7, 1e-10)
    T = T_init
    cooling = 0.9995

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        move_type = rng.integers(0, 5)
        a_new = a.copy()

        if move_type == 0:
            i = rng.integers(0, n)
            frac = rng.uniform(0.02, 0.5)
            a_new[i] = a[i] * (1.0 + frac * rng.uniform(-1, 1))
        elif move_type == 1:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.4)
            direction = 1.0 if rng.random() < 0.5 else -1.0
            a_new[i] = a[i] + direction * amt
            a_new[j] = a[j] - direction * amt
        elif move_type == 2:
            center = rng.integers(0, n)
            width = max(1, int(rng.integers(1, n // 15)))
            lo = max(0, center - width)
            hi = min(n, center + width + 1)
            a_new[lo:hi] = a[lo:hi] * (1.0 + rng.standard_normal(hi - lo) * 0.15)
        elif move_type == 3:
            # Swap-based move
            i, j = rng.integers(0, n, size=2)
            if i != j:
                diff = a[i] - a[j]
                a_new[i] = a[j]
                a_new[j] = a[i]
        else:
            # Global small perturbation
            a_new = a + rng.standard_normal(n) * (M / n) * 0.3
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
        
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        new_conv = fast_convolve(a_new)
        new_M = float(new_conv.max())
        new_score = 2.0 * n * new_M
        old_score = 2.0 * n * M

        delta = new_score - old_score
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            a = a_new
            conv = new_conv
            M = new_M
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                best_score = new_score

        T *= cooling
        if T < T_min:
            T = T_min

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SA it={it} ({elapsed:.0f}s): best={best_score:.6f} T={T:.2e}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
