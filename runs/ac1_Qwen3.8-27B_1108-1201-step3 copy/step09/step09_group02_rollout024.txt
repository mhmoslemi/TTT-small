    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

        if fc <= fd:

            c = hi - gr * (hi - lo); fc = f(c)

            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = f(d)

                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            
            if has_quad:
                quad_idx = 2 * i - k_lo
            
            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                          _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                          _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)
            
            upper = max(3.0 * v + 1.0/n, 5.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=40)
            
            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners.copy()
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_score, sweeps

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def mirror_descent(a, g, eta):
    """Mirror descent step on the simplex starting from a."""
    log_b = np.log(np.maximum(a, 1e-30)) - eta * g
    log_b -= log_b.max()
    b = np.exp(log_b)
    s = b.sum()

    return b / s

def golden_section_search(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo); fc = f(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = f(d)
    return (lo + hi) * 0.5

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1

    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
