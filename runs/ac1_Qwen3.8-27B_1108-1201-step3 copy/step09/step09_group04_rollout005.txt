import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def p_norm_objective(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val

def p_norm_gradient(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))

def p_norm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:

            z = res.x

            pass
    z_max = z.max()

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_obj(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_grad(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters = max(50, min(300, int(remaining * 5)))
        try:
            res = minimize(
                fun=lambda zv: p_norm_obj(zv, stage_p),
                x0=z,
                jac=lambda zv: p_norm_grad(zv, stage_p),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): {sc:.6f}")
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

def nesterov_lp(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov accelerated method for the bilinear minimax problem."""
    if solver is None:
        solver = get_solver()
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Build and solve LP: min M s.t. conv(y,b)_k <= M, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_vec = np.zeros(size)
        for j in range(n):
            T_vec[j:j+n] += y * 0  # placeholder, use direct conv
        # Build constraint vector: conv(y, b) <= M
        # conv(y,b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} y[i] * b[k-i]
        # Use the fact that this is a Toeplitz-like system
        # For efficiency, build the matrix sparsely
        # Actually, just use the vectorized form
        # conv(y, b) can be computed as: for each k, dot product of y segment with b segment
        
        # Build constraint matrix as dense (n is at most ~2000, size is ~4000)
        # T[k, i] = y[i] if 0 <= k-i < n, i.e., k-n+1 <= i <= k
        T_mat = np.zeros((size, n))
        for i in range(n):
            T_mat[i:i+n, i] = y
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            b_val = np.ones(n) / n
        else:
            b_val = b_val / s_b

        # Nesterov update with adaptive step
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_val
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        a_num = (it + 1) * a + (alpha * (it + 2)) * b_val
        a_den = it + 1 + alpha * (it + 2)
        a_new = a_num / a_den
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        y = y_new
        a = a_new

        # Track best
        c_a = fast_convolve(a)
        score_a = 2.0 * n * c_a.max()
        c_y = fast_convolve(y)
        score_y = 2.0 * n * c_y.max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # Stall detection with perturbation
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 5:
                rng = np.random.default_rng(it*7 + 3)
                noise = rng.standard_normal(n) * (0.02 + 0.01 * stall)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
                prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP iter={it+1} ({elapsed:.0f}s): {best_score:.6f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_steps=50000):
    """Local search that explicitly tries to equalize the top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Find top peaks (excluding boundaries which are structurally smaller)
        c_inner = c.copy()
        c_inner[0] = -1
        c_inner[-1] = -1
        peak_val = c_inner.max()
        # Get peaks within 5% of maximum
        threshold = peak_val * 0.95
        peak_indices = np.where(c_inner >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c_inner)])

        improved = False

        # Strategy: reduce contributions to the highest peaks
        # For each peak k, the contribution is sum_i a[i]*a[k-i]
        # Reduce the largest a[i]*a[k-i] pairs
        top_peak = peak_indices[0]
        lo = max(0, top_peak - n + 1)
        hi = min(n - 1, top_peak)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[top_peak - idx]
        
        # Try reducing the top contributor
        top_i_rel = np.argmax(contribs)
        top_i = idx[top_i_rel]
        top_j = top_peak - top_i
        
        if top_i != top_j:
            # Reduce a[top_i] and redistribute
            for frac in [0.98, 0.96, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[top_i] *= frac
                diff = a[top_i] - a_trial[top_i]
                # Redistribute to non-peak contributors
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others = a_trial[mask].sum()
                if others > 1e-15:
                    a_trial[mask] += diff * a_trial[mask] / others
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        else:
            # Top peak is at i=j (center), reduce it
            for frac in [0.98, 0.96, 0.94, 0.90]:
                a_trial = a.copy()
                a_trial[top_i] *= frac
                diff = a[top_i] - a_trial[top_i]
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others = a_trial[mask].sum()
                if others > 1e-15:
                    a_trial[mask] += diff * a_trial[mask] / others
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        # Strategy 2: Pair adjustment for second-highest peak
        if not improved and len(peak_indices) > 1:
            second_peak = peak_indices[1]
            lo2 = max(0, second_peak - n + 1)
            hi2 = min(n - 1, second_peak)
            idx2 = np.arange(lo2, hi2 + 1)
            contribs2 = a[idx2] * a[second_peak - idx2]
            top_i2_rel = np.argmax(contribs2)
            top_i2 = idx2[top_i2_rel]
            top_j2 = second_peak - top_i2
            if top_i2 != top_j2 and 0 <= top_j2 < n:
                for f1 in [0.97, 0.95, 0.92]:
                    for f2 in [0.97, 0.95, 0.92]:
                        a_trial = a.copy()
                        a_trial[top_i2] *= f1
                        a_trial[top_j2] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st > 0.01:
                            a_trial = a_trial / st
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                    if improved:
                        break

        # Strategy 3: Small Gaussian perturbation on random coordinates
        if not improved and step % 50 == 0:
            rng = np.random.default_rng(step * 13 + 7)
            n_pert = rng.integers(2, 8)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved and step % 200 == 0:
            # Larger random restart
            rng = np.random.default_rng(step * 31 + 1)
            a_trial = a + rng.standard_normal(n) * 0.01
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved and step > 1000:
            break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakEq step={step} ({elapsed:.0f}s): {best_score:.6f}")

    return best_a, best_score


def freq_domain_init(n, variant=0):
    """Generate initialization inspired by frequency domain design."""
    if variant == 0:
        # Flat spectrum with small rolloff
        freqs = np.fft.rfftfreq(n) * n
        band = np.sqrt(n) * 0.5
        target = np.exp(-0.5 * (freqs / band) ** 2)
    elif variant == 1:
        # Raised cosine spectrum
        freqs = np.fft.rfftfreq(n) * n
        band = np.sqrt(n) * 0.4
        t = np.abs(freqs) / band
        target = np.where(t <= 1, 1 - 0.5 * (1 - np.cos(np.pi * t)), 0)
    elif variant == 2:
        # Waterfall spectrum
        freqs = np.fft.rfftfreq(n) * n
        band = np.sqrt(n) * 0.6
        target = 1.0 / (1.0 + 0.5 * (np.abs(freqs) / band) ** 4)
    else:
        target = np.ones(len(np.fft.rfftfreq(n)))
    
    target = np.maximum(target, 1e-10)
    # IFFT of target spectrum (real, symmetric)
    full_spec = np.zeros(n, dtype=complex)
    full_spec[:len(target)] = target
    full_spec[-len(target)+1:] = np.conj(target[:-1][::-1])
    a = np.real(np.fft.ifft(full_spec))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        xx = np.linspace(-1, 1, n)
        a = np.exp(-xx ** 2 * 3)
    elif variant == 5:
        a = np.exp(-2 * x * (1 - x)) * np.exp(-2 * (1-x) * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])
    print(f"Starting: n={n_best}, score={best_state['score']:.6f}")

    def try_update(a_new, label):
        a_new = np.maximum(a_new, 0).astype(np.float64)
        s = a_new.sum()
        if s < 0.01:
            return False
        a_new = a_new / s
        sc = evaluate_safe(a_new)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a_new.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_new)})")
            return True
        return False

    # === Phase 1: Quick p-norm L-BFGS ===
    print(f"\n=== Phase 1: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn = p_norm_lbfgs(best_state['seq'], t_alloc, p_start=10.0, p_end=200.0, n_stages=7)
        try_update(a_pn, "pn1")

    # === Phase 2: Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_state['seq'], t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nlp1")

    # === Phase 3: Peak equalization ===
    print(f"\n=== Phase 3: Peak Equalization (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        a_pe, _ = peak_equalization(best_state['seq'], t_alloc, n_steps=80000)
        try_update(a_pe, "peak_eq")

    # === Phase 4: High-p L-BFGS ===
    print(f"\n=== Phase 4: High-p L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn2 = p_norm_lbfgs(best_state['seq'], t_alloc, p_start=100.0, p_end=1000.0, n_stages=6)
        try_update(a_pn2, "pn2")

    # === Phase 5: Nesterov LP round 2 ===
    print(f"\n=== Phase 5: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp(best_state['seq'], t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp2, "nlp2")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-64, -32, 32, 64, 96, 128, -128, 192]:
        alt_n = base + delta
        if 512 <= alt_n <= 2048:
            candidate_lengths.add(alt_n)
    for rn in [768, 832, 896, 1024, 1088, 1152, 1216]:
        if 512 <= rn <= 2048 and abs(rn - base) > 16:
            candidate_lengths.add(rn)
    
    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_seq(best_state['seq'], alt_n)
        # Quick optimization
        t_sub = t_total * 0.6
        a_alt = p_norm_lbfgs(init_alt, t_sub, p_start=30.0, p_end=300.0, n_stages=5)
        improved = try_update(a_alt, f"len{alt_n}")
        
        if improved:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp = min(rem * 0.2, 80.0)
                a_alt2, _ = nesterov_lp(best_state['seq'], t_lp, solver=solver, n_max_iters=50000)
                try_update(a_alt2, f"len{alt_n}_lp")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        # Try freq-domain initializations
        for vd in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_fd = freq_domain_init(n_cur, variant=vd)
            # Blend with current best
            blend = 0.5
            a_mix = blend * best_state['seq'] + (1 - blend) * a_fd
            a_mix = np.maximum(a_mix, 0)
            a_mix = a_mix / a_mix.sum()
            t_sub = min(rem * 0.25, 50.0)
            a_mix_opt = p_norm_lbfgs(a_mix, t_sub, p_start=20.0, p_end=200.0, n_stages=4)
            try_update(a_mix_opt, f"freq{vd}")

        # Noise restart
        if deadline - time.monotonic() > 30:
            for ns in [0.05, 0.10]:
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                perturbed = best_state['seq'] + ns * rng.standard_normal(n_cur)
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                t_sub = min(rem * 0.3, 40.0)
                a_pert = p_norm_lbfgs(perturbed, t_sub, p_start=30.0, p_end=200.0, n_stages=4)
                try_update(a_pert, f"noise{ns}")

    # === Phase 8: Final peak equalization + polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_pe = min(rem * 0.30, 60.0)
        a_pe2, _ = peak_equalization(best_state['seq'], t_pe, n_steps=30000)
        try_update(a_pe2, "final_pe")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.30, 40.0)
        a_fin = p_norm_lbfgs(best_state['seq'], t_fin, p_start=200.0, p_end=800.0, n_stages=4)
        try_update(a_fin, "final_pn")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.35, 40.0)
        a_lpf, _ = nesterov_lp(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=30000)
        try_update(a_lpf, "final_lp")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - t_start
    print(f"\nFINAL: score={best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
