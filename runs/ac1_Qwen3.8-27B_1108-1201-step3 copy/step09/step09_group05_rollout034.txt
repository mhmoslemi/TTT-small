Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Active-set peak equalization via targeted coordinate adjustments

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_a_from_w(a, w, n):
    """
    Compute gradient of smoothed objective w.r.t. a given softmax weights w over convolution.
    g_i = 4n * sum_k w_k * a_{k-i} (for valid indices)
    """
    g = np.zeros(n)
    for i in range(n):
        # sum_j a[j] * w[i+j] for j=0..n-1
        g[i] = np.dot(a, w[i:i + n])
    return 4.0 * n * g


def lbgfs_adaptive(a0, time_limit, T_stages=None):
    """
    L-BFGS with log-sum-exp smoothing and adaptive multi-stage temperature.
    T_stages: list of (T, max_iter) tuples. If None, uses default schedule.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    if T_stages is None:
        T_stages = [
            (50.0, 800),
            (25.0, 1000),
            (12.0, 1500),
            (6.0, 2000),
            (3.0, 3000),
            (1.5, 4000),
            (0.7, 5000),
            (0.3, 6000),
            (0.15, 6000),
            (0.05, 8000),
        ]

    for T, max_iter in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        t_start_stage = time.monotonic()

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = compute_grad_a_from_w(a, w, n)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': max_iter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_current = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_current).max()
        if score < best_score:
            best_score = score
            best_a = a_current.copy()
            print(f"    L-BFGS T={T}: score={score:.8f}, time={time.monotonic()-t0:.1f}s")

    return best_a

    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        sorted_idx = np.argsort(b_work)[::-1]
        top_peaks = sorted_idx[:5]
        peak_val = b_work[top_peaks[0]]

        improved = False
        for pk in top_peaks:
            if b_work[pk] < peak_val * 0.95:
                continue
            # Find top contributing pairs for this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            total = contribs.sum()
            if total < 1e-15:
                continue
            top_k = np.argsort(contribs)[::-1][:10]

            for ci in top_k:
                i = idx[ci]
                j = pk - i
                if i >= n or j >= n or i < 0 or j < 0:
                    continue

                # Try reducing a[i] and redistributing
                for factor in [0.95, 0.90, 0.85, 0.75]:
                    a_trial = a.copy()
                    reduction = a[i] * (1.0 - factor)
                    if reduction < 1e-18:
                        continue
                    a_trial[i] *= factor

                    # Redistribute to low-convolution positions
                    # Find positions with low contribution to all top peaks
                    low_score = np.zeros(n)
                    for tp in top_peaks[:3]:
                        js2 = max(0, tp - n + 1)
                        je2 = min(n - 1, tp)
                        for jj in range(js2, je2 + 1):
                            low_score[jj] += a[jj] * a[tp - jj]

                    low_coords = np.argsort(low_score)[:n // 3]
                    low_coords = low_coords[low_coords != i]
                    if len(low_coords) == 0:
                        continue

                    dist_weights = a[low_coords] + 1e-12
                    a_trial[low_coords] += reduction * dist_weights / dist_weights.sum()
                    a_trial = a_trial / a_trial.sum()

                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-16:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try pair adjustment on the worst peak
            pk = top_peaks[0]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_pairs = np.argsort(contribs)[::-1][:8]

            found = False
            for ci in top_pairs:
                i = idx[ci]
                j = pk - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.92, 0.85, 0.78]:
                    for f2 in [0.92, 0.85, 0.78]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-16:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break
            if not improved:
                no_improve += 1
        else:
            no_improve = 0

    return best_a


def coordinate_descent_log(a0, time_limit, n_steps=30000):
    """
    Coordinate descent in log-domain: for each coordinate, find optimal value
    via 1D line search (golden section).
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    order = np.arange(n)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 2000:
            break

        idx_i = order[step % n]
        z_save = z[idx_i]

        # Golden section search around current value
        best_z = z_save
        best_f = 1e30

        a_curr = softmax_from_z(z)
        b_curr = fast_convolve(a_curr)
        f_curr = b_curr.max()

        for scale in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            for direction in [1, -1]:
                z_trial = z.copy()
                z_trial[idx_i] = z_save + direction * scale
                a_trial = softmax_from_z(z_trial)
                b_trial = fast_convolve(a_trial)
                f_trial = b_trial.max()
                if f_trial < best_f:
                    best_f = f_trial
                    best_z = z_trial[idx_i]

        if best_z != z_save and best_f < f_curr - 1e-15:
            z[idx_i] = best_z
            no_improve = 0
        else:
            no_improve += 1

        if step % 500 == 0:
            a_check = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_check).max()
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
            # Periodically shuffle
            if step % 5000 == 0:
                rng.shuffle(order)

    a_final = softmax_from_z(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_polyak(a0, time_limit, n_steps=20000):
    """
    Subgradient method with Polyak step size on the max of convolution.
    Works in probability simplex.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    f_star = 1.0 / (2.0 * n - 1.0)  # lower bound on max(conv) for sum=1
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        f_val = b[k_star]

        # Subgradient: g_i = 2*a_{k_star - i} for valid i
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]

        # Polyak step size
        gap = f_val - f_star
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        step_size = gap / g_norm_sq
        step_size = min(step_size, 1.0)  # cap step size

        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a


def multi_start_lbgfs(a0, time_limit, n_starts=3):
    """
    Run L-BFGS from multiple perturbed starting points, keep the best.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    time_per_start = (time_limit - 10) / n_starts

    for s in range(n_starts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        if s == 0:
            start = a0.copy()
        else:
            # Perturb the current best
            scale = 0.05 * (s + 1)
            perturbed = a0 + scale * rng.standard_normal(n)
            start = norm_sequence(perturbed)

        short_stages = [
            (10.0, 500),
            (5.0, 800),
            (2.0, 1500),
            (1.0, 2000),
            (0.5, 2000),
            (0.2, 2000),
            (0.1, 2000),
        ]

        a_result = lbgfs_adaptive(start, time_per_start, T_stages=short_stages)
        score = 2.0 * n * fast_convolve(a_result).max()
        if score < best_score:
            best_score = score
            best_a = a_result.copy()
            print(f"    Multi-start {s}: score={score:.8f}")

    return best_a


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x * x) * (1 + 0.2 * np.cos(6 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Long adaptive L-BFGS on parent length
    Phase 2: Peak equalization
    Phase 3: Multi-length exploration
    Phase 4: Multi-start L-BFGS for diversification
    Phase 5: Final polish with lowest temperature
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Long adaptive L-BFGS (50% of time) ===
    print(f"\n=== Phase 1: Adaptive L-BFGS n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.50, 900.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        full_stages = [
            (30.0, 600),
            (15.0, 800),
            (8.0, 1200),
            (4.0, 2000),
            (2.0, 3000),
            (1.0, 4000),
            (0.5, 5000),
            (0.2, 6000),
            (0.1, 8000),
            (0.05, 8000),
            (0.02, 10000),
        ]
        a_lp = lbgfs_adaptive(best_seq, t_alloc, T_stages=full_stages)
        try_update(a_lp, "lbgfs_phase1")

    # === Phase 2: Peak equalization (10% of time) ===
    print(f"\n=== Phase 2: Peak Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pe = peak_equalize(best_seq, t_alloc, n_steps=80000)
        try_update(a_pe, "peak_eq_1")

        # Short L-BFGS after peak equalization
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sub = min(rem * 0.3, 60.0)
            short_stages = [(2.0, 1000), (1.0, 2000), (0.5, 2000), (0.2, 2000)]
            a_after = lbgfs_adaptive(best_seq, t_sub, T_stages=short_stages)
            try_update(a_after, "lbgfs_after_peak")

    # === Phase 3: Multi-length exploration (20% of time) ===
    print(f"\n=== Phase 3: Multi-Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_multi = min(rem * 0.20, 300.0)

    alt_lengths = []
    for mult in [1.25, 1.5, 2.0, 0.75, 1.33]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [512, 768, 1024, 1536, 2048]:
        if 256 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))[:4]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(t_multi / len(alt_lengths), rem * 0.8)
        if t_total < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # L-BFGS on new length
        medium_stages = [
            (20.0, 400),
            (10.0, 600),
            (5.0, 1000),
            (2.0, 1500),
            (1.0, 2000),
            (0.5, 2000),
            (0.2, 2000),
        ]
        a_alt = lbgfs_adaptive(init_alt, t_total * 0.7, T_stages=medium_stages)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        # Peak equalize on new length
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            a_alt_pe = peak_equalize(best_seq, t_total * 0.3, n_steps=20000)
            try_update(a_alt_pe, f"peak_n{alt_n}")

    # === Phase 4: Multi-start diversification (12% of time) ===
    print(f"\n=== Phase 4: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_multi = multi_start_lbgfs(best_seq, t_alloc, n_starts=3)
        try_update(a_multi, "multi_start")

    # === Phase 5: Subgradient + L-BFGS combo (5% of time) ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 15:
        a_sg = subgradient_polyak(best_seq, t_alloc, n_steps=30000)
        try_update(a_sg, "subgrad")

        rem = deadline - time.monotonic()
        if rem > 20:
            polish_stages = [(1.0, 2000), (0.5, 2000), (0.2, 2000)]
            a_after_sg = lbgfs_adaptive(best_seq, min(rem * 0.6, 40.0), T_stages=polish_stages)
            try_update(a_after_sg, "lbgfs_after_sg")

    # === Phase 6: Final extreme low-temperature polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_final = min(rem * 0.8, 120.0)
        print(f"  Final polish budget: {t_final:.0f}s")
        final_stages = [
            (1.0, 2000),
            (0.5, 3000),
            (0.2, 4000),
            (0.1, 5000),
            (0.05, 5000),
            (0.02, 6000),
            (0.01, 6000),
        ]
        a_final = lbgfs_adaptive(best_seq, t_final, T_stages=final_stages)
        try_update(a_final, "final_polish")

        # Peak equalize one more time
        rem = deadline - time.monotonic()
        if rem > 20:
            a_final_pe = peak_equalize(best_seq, min(rem * 0.5, 30.0), n_steps=20000)
            try_update(a_final_pe, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
