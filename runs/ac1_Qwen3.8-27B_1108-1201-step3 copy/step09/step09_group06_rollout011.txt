    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_full(a_fixed, solver, warm_start=None):
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_reduced(a_fixed, solver, k_constraints=100, current_b=None):
    """Solve LP with only top-k constraints for speed."""
    n = len(a_fixed)
    size = 2 * n - 1
    if current_b is None:
        current_b = np.ones(n) / n
    conv_vals = convolve_with(a_fixed, current_b)
    top_idx = np.argsort(conv_vals)[::-1][:k_constraints]
    T = build_conv_matrix(a_fixed)
    T_reduced = T[top_idx, :]
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_reduced @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    # Verify against full convolution
    full_conv = convolve_with(a_fixed, b_val)
    M_full = float(full_conv.max())
    if M_full > M_var.value * 1.01 + 1e-10:
        return None, None  # Reduced LP solution violates full constraints
    return b_val, M_full


def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def mirror_descent(a0, time_limit, n_active_frac=0.06):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b = compute_subgradient(a, n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - 2.0 * n / (2.0 * n - 1.0)
        if gap < 1e-10:
            break

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.4f}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, n_max_iters=200000, use_reduced=False, k_red=80):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_reduced and it > 50:
            b_new, M_b = solve_lp_reduced(y, solver, k_constraints=k_red, current_b=a)
            if b_new is None:
                b_new, M_b = solve_lp_full(y, solver)
        else:
            b_new, M_b = solve_lp_full(y, solver)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_score - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 6:
                rng = np.random.default_rng(it * 7 + 13)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_make_func(z_in, n, T_val, t0, time_limit):
    if time.monotonic() - t0 > time_limit:
        return 1e15
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lbgfs_make_jac(z_in, n, T_val, t0, time_limit):
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=12, maxiter_total=20000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    iters_per_stage = max(20, maxiter_total // n_stages)

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        T_val = T_start * (T_end / T_start) ** (stage / max(n_stages - 1, 1))

        def func_lbgfs(z_in):
            return lbgfs_make_func(z_in, n, T_val, t0, time_limit)

        def jac_lbgfs(z_in):
            return lbgfs_make_jac(z_in, n, T_val, t0, time_limit)

        try:
            res = minimize(
                fun=func_lbgfs,
                x0=z,
                jac=jac_lbgfs,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if stage % 4 == 3:
            elapsed = time.monotonic() - t0
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    L-BFGS stage {stage} ({elapsed:.0f}s) T={T_val:.2f}: score={sc:.8f}")

    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, n_passes=500):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = a[js:je + 1] * a[peak_k - (je):peak_k - js - 1:-1] if js <= je else np.array([])

        if len(contrib) == 0:
            continue

        # Find best pair to swap
        top_i = np.argmax(contrib) + js
        bottom_i = np.argmin(contrib) + js

        if a[top_i] < 1e-14:
            continue

        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            amount = a[top_i] * frac
            a_trial = a.copy()
            a_trial[top_i] -= amount
            a_trial[bottom_i] += amount
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved and p > 50:
            # Try random pairs
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    amount = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and p > 100:
            break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 2)
        a = np.maximum(a, 0)
    else:
        # Chirp magnitude with envelope
        x = np.linspace(0, 1, n)
        chirp = np.sin(2 * np.pi * x ** 2 * 2)
        a = np.abs(chirp) * np.sin(np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Phase 1: Mirror descent for quick improvement
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, n_active_frac=0.05)
        try_update(a_md, "mirror_desc_1")

    # Phase 2: Nesterov LP (main workhorse)
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t_alloc, solver=solver, n_max_iters=200000, use_reduced=False)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 3: L-BFGS LogSumExp with geometric annealing
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=200.0, T_end=0.1, n_stages=14, maxiter_total=25000)
        try_update(a_lb, "lbgfs_1")

    # Phase 4: Second Nesterov LP round
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        # Use reduced LP after initial iterations for speed
        a_lp2, _ = nesterov_lp(best_seq, t_alloc, solver=solver, n_max_iters=300000, use_reduced=True, k_red=100)
        try_update(a_lp2, "nesterov_lp_2")

    # Phase 5: Coordinate descent
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_passes=2000)
        try_update(a_cd, "coord_desc")

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0, 1.1]:
        alt_n = int(n_best * mult)
        if 400 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.25, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            n_best = alt_n

        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp(init_alt, t_lp, solver=solver, n_max_iters=15000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_md = min(rem * 0.25, 40.0)
            if t_md > 10:
                a_alt2, _ = mirror_descent(best_seq, t_md, n_active_frac=0.04)
                try_update(a_alt2, f"md_n{alt_n}")

    # Phase 7: Final L-BFGS polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.05, n_stages=10, maxiter_total=10000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ls = min(rem * 0.2, 50.0)
        a_ls2, _ = coordinate_descent(best_seq, t_ls, n_passes=3000)
        try_update(a_ls2, "final_coord")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.3, 30.0)
        if t_lp_fin > 5:
            a_lp_fin, _ = nesterov_lp(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
            try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
