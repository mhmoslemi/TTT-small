def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(n, a):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.PDLP


def solve_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(n, a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating with Nesterov
acceleration, multiple diverse initializations, multi-length exploration,
and subgradient polish. Optimized for 1800s budget.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(n, a):
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    T = np.where(mask, a[idx.clip(0, n - 1)], 0.0)
    return T


def get_lp_solver():
    for name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.PDLP


def solve_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(n, a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver] + [s for s in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL] if s != solver]
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                return b_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_nesterov(a0, time_limit, solver, max_iters=200000, verbose_every=500):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose_every > 0 and (it + 1) % verbose_every == 0:
            print(f"    LP iter {it+1}: score={best_score:.8f}, M={M_b:.8f}")

    return best_a, best_score

def subgradient_polish(a0, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.999)[0]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= float(len(active))
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 50:
                    lr *= 0.9
                    no_improve = 0
        lr = max(lr, 1e-10)
        if step % 10000 == 0:
            print(f"    SG step {step}: score={best_score:.8f}")

    return best_a, best_score
