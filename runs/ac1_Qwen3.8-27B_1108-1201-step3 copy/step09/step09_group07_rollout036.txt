            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max conv(a,b) subject to b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_nesterov_extrapolation(a0, time_limit, solver=None, max_iters=80000):
    """LP best response with Nesterov-style extrapolation and adaptive damping."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    a_old = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    gamma = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Nesterov-style: extrapolate from previous direction
        direction = b_new - a_old
        a_old = a.copy()

        # Try extrapolated step
        a_trial = a + gamma * direction
        a_trial = project_simplex(a_trial)
        score_trial = score_normalized(a_trial)

        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
            stall_count = 0
            gamma = min(gamma * 1.02, 2.5)
        elif score_trial <= score_normalized(a) + 0.5 * (best_score - score_normalized(a)) * 0.01:
            a = a_trial.copy()
            gamma = max(gamma * 0.95, 0.3)
        else:
            gamma = max(gamma * 0.85, 0.1)

        # Also track the averaged sequence (dual averaging)
        alpha_avg = 2.0 / (it + 2)
        if not hasattr(lp_nesterov_extrapolation, '_avg_a'):
            lp_nesterov_extrapolation._avg_a = b_new.copy()
        else:
            lp_nesterov_extrapolation._avg_a = (1 - alpha_avg) * lp_nesterov_extrapolation._avg_a + alpha_avg * b_new

        avg_a = project_simplex(lp_nesterov_extrapolation._avg_a)
        score_avg = score_normalized(avg_a)
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()
            stall_count = 0

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                rng = np.random.default_rng(it + 12345)
                noise = 0.01 * rng.standard_normal(n)
                a = project_simplex(a + noise)
                lp_nesterov_extrapolation._avg_a = project_simplex(lp_nesterov_extrapolation._avg_a + 0.005 * noise)
                gamma = 1.0
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-NE iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}, gamma={gamma:.3f}")

    # Reset attribute
    lp_nesterov_extrapolation._avg_a = None
    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, T_start=5.0, T_end=200.0, n_stages=8, steps_per_stage=3000):
    """Mirror descent (multiplicative weights) on smoothed max-convolution objective."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    total_steps = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_convolve(a)
            Tb = T * b
            m = Tb.max()
            p = np.exp(Tb - m)
            p = p / p.sum()

            # Gradient of smoothed objective
            # grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
            fft_size = 1 << (n + len(p) - 1).bit_length()
            A = rfft(a, fft_size)
            P = rfft(p, fft_size)
            corr = irfft(np.conj(P) * A, fft_size)[:n]
            grad = 2.0 * corr

            # Multiplicative weight update with adaptive step size
            eta = 1.0 / (np.sqrt(total_steps + 1) * np.sqrt(np.log(n + 1)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-15)
            s = a.sum()
            a = a / s
            total_steps += 1

            if total_steps % 100 == 0:
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if total_steps % 1000 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      MD step={total_steps} ({elapsed:.0f}s): score={best_score:.6f}, T={T:.1f}")

    return best_a, best_score


def importance_subgradient(a0, time_limit, n_samples=5, n_steps=20000):
    """Subgradient method with importance sampling of peak locations."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    T = 50.0
    lr = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        Tb = T * b
        m = Tb.max()
        p = np.exp(Tb - m)
        p = p / p.sum()

        # Sample k values with importance
        k_samples = np.random.choice(2 * n - 1, size=n_samples, replace=False, p=p)

        grad = np.zeros(n)
        for k in k_samples:
            weight = p[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad[i] += weight * 2.0 * a[k - i]

        # Projected gradient step with backtracking
        alpha = lr / (1.0 + 0.001 * step)
        a_trial = project_simplex(a - alpha * grad)
        sc = score_normalized(a_trial)

        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            alpha *= 0.5
            a_trial = project_simplex(a - alpha * grad)
            sc = score_normalized(a_trial)
            if sc < score_normalized(a):
                a = a_trial.copy()

        if step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      IS step={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def simulated_annealing_lp(a0, time_limit, solver=None, n_steps=5000):
    """Simulated annealing with LP-informed proposals."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    T_sa = score_normalized(a) * 0.005
    T_min = score_normalized(a) * 0.0001
    cooling = 0.9995
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        T_sa = max(T_sa * cooling, T_min)

        # Generate proposal: mix of LP best response and random perturbation
        choice = rng.random()
        if choice < 0.3:
            # LP best response
            b_new, _ = solve_lp_best_response(a, solver)
            if b_new is None:
                b_new = a
            a_trial = project_simplex(0.7 * a + 0.3 * b_new)
        elif choice < 0.6:
            # Gaussian perturbation
            noise = T_sa * 100 * rng.standard_normal(n)
            a_trial = project_simplex(a + noise)
        elif choice < 0.8:
            # Peak-targeting: reduce coordinates contributing to the peak
            b = fast_convolve(a)
            k_peak = int(np.argmax(b))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_peak - idx]
            top = idx[np.argsort(contribs)[::-1][:5]]
            a_trial = a.copy()
            for i in top:
                a_trial[i] *= (1.0 - 0.05 * rng.random())
            a_trial = project_simplex(a_trial)
        else:
            # Local shuffle: swap pairs
            a_trial = a.copy()
            i, j = rng.integers(0, n, size=2)
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
            a_trial = project_simplex(np.maximum(a_trial, 0))

        score_trial = score_normalized(a_trial)
        delta = score_trial - score_normalized(a)

        if delta < 0 or rng.random() < np.exp(-delta / max(T_sa, 1e-10)):
            a = a_trial.copy()
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()

        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA step={step} ({elapsed:.0f}s): best={best_score:.6f}, T_sa={T_sa:.8f}")

    return best_a, best_score


def nesterov_projected_gradient(a0, time_limit, T_start=2.0, T_end=300.0,
                                 lr_init=0.08, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.005 * step_count)

            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = project_simplex(x + beta * (x - x_prev))

            # Gradient via FFT
            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w = w / w.sum()

            fft_size = 1 << (n + len(w) - 1).bit_length()
            A = rfft(y, fft_size)
            W = rfft(w, fft_size)
            corr = irfft(np.conj(W) * A, fft_size)[:n]
            grad = 2.0 * corr

            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step_count % 2000 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def make_structured_init(n, rng):
    """Generate a structured non-uniform initialization."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x)
    elif choice == 1:
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x)
    elif choice == 2:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x + 1.0) + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 4:
        a = 1.0 + 0.5 * (2 * x - 1)
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(6 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x)
    elif choice == 6:
        a = np.abs(np.sin(2 * np.pi * x)) ** 0.5 + 0.3
    else:
        a = 1.0 + 0.4 * rng.standard_normal(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining LP extrapolation, mirror descent,
    importance-sampled subgradients, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Quick Nesterov PG to establish baseline ===
    print(f"\n=== Phase 1: NG Baseline n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_projected_gradient(best_seq, t_alloc, T_start=2.0, T_end=200.0,
                                               lr_init=0.08, n_stages=6, steps_per_stage=2000)
        try_update(a_ng, "ng_baseline")

    # === Phase 2: LP Nesterov Extrapolation (main workhorse) ===
    print(f"\n=== Phase 2: LP Nesterov Extrapolation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_nesterov_extrapolation(best_seq, t_alloc, solver=solver, max_iters=100000)
        try_update(a_lp, "lp_ne_1")

    # === Phase 3: Mirror Descent (different geometry) ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, talloc if 'talloc' in dir() else t_alloc,
                                           T_start=10.0, T_end=300.0, n_stages=6, steps_per_stage=2000)
        try_update(a_md, "mirror_desc")

    # === Phase 4: Importance-sampled subgradient ===
    print(f"\n=== Phase 4: Importance Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_is, _ = importance_subgradient(best_seq, t_alloc, n_samples=8, n_steps=30000)
        try_update(a_is, "importance_sg")

    # === Phase 5: LP refinement round 2 (from improved point) ===
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.6f}) ===")

        print(f"  Budget: {t_alloc:.0f}s")

        print(f"  n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_projected_gradient(init_alt, t_alt * 0.5, T_start=2.0, T_end=200.0,
                                                lr_init=0.08, n_stages=5, steps_per_stage=1000)
        try_update(a_alt, f"len_{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = lp_nesterov_extrapolation(a_alt, min(rem * 0.4, 25.0), solver=solver, max_iters=1500)
            try_update(a_alt2, f"len_{alt_n}_lp")

    # === Phase 9: Final LP polish ===

    rem = deadline - time.monotonic()
    if rem > 30:

        try_update(a_fin, "final_lp")

    # === Phase 10: Final NG polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_ng = min(rem * 0.15, 40.0)
        a_ngf, _ = nesterov_projected_gradient(best_seq, t_ng, T_start=50.0, T_end=500.0,
                                                lr_init=0.02, n_stages=5, steps_per_stage=1000)
        try_update(a_ngf, "final_ng")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
