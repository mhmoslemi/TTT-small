"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with adaptive beta (log-sum-exp smoothing of max) - O(n log n) per step
2. Momentum-based (Adam) optimization for faster convergence
3. Exact coordinate descent with golden section for polishing
4. Basin hopping with spectral perturbations for global exploration
5. LP-based fictitious play for structured improvements
6. Multi-phase time allocation with adaptive scheduling

import numpy as np
from scipy.fft import rfft, irfft
import time

_DEADLINE = time.monotonic() + 1785.0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def score_of_normalized(a):
    """Compute score 2*n*max(conv(a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def smooth_max_conv(a, beta):
    """Smoothed max of conv(a) using log-sum-exp."""
    conv = fast_convolve(a)
    m = float(conv.max())
    shifted = beta * (conv - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + log_sum / beta

def compute_conv_weights(conv, beta):
    """Compute softmax weights over convolution peaks."""
    m = float(conv.max())
    w = np.exp(beta * (conv - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        argmax = np.argmax(conv)
        w[argmax] = 1.0
    else:
        w = w / s
    return w

def smooth_gradient_conv(a, w):
    """
    Compute gradient of sum_k w_k * conv_k(a) w.r.t. a.
    grad_i = 2 * sum_m w_{m+i} * a_m  (cross-correlation of w with a)
    """
    n = len(a)
    # Need L >= (2n-1) + n - 1 = 3n-2 for linear cross-correlation
    L = 1 << (3 * n - 2).bit_length()
    A = rfft(a, L)
    W = rfft(w, L)
    # Cross-correlation: (w star a)[i] = sum_m w[m+i] * a[m]
    # = ifft(FFT(w) * conj(FFT(a)))
    corr = irfft(W * np.conj(A), L)[:n]
    return 2.0 * corr

def smooth_objective(a, beta):
    """Smoothed objective: 2*n*smooth_max(conv(a))."""
    n = len(a)
    return 2.0 * n * smooth_max_conv(a, beta)

def smooth_grad_full(a, beta):
    """Gradient of smoothed objective."""
    n = len(a)
    conv = fast_convolve(a)
    w = compute_conv_weights(conv, beta)
    corr = smooth_gradient_conv(a, w)
    return 4.0 * n * corr

def smooth_grad_and_value(a, beta):
    """Compute both smoothed objective value and gradient in one pass."""
    n = len(a)
    conv = fast_convolve(a)
    m = float(conv.max())
    shifted = beta * (conv - m)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        val = 2.0 * n * m
        return val, np.zeros(n)
    w = w / ws
    val = 2.0 * n * (m + np.log(ws) / beta)
    L = 1 << (3 * n - 2).bit_length()
    A = rfft(a, L)
    W = rfft(w, L)
    corr = irfft(W * np.conj(A), L)[:n]
    grad = 4.0 * n * corr
    return val, grad

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent: for each coordinate, golden section search
    to find the value minimizing the full score.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(2.0 * v, 4.0 / n)

            def obj(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            best_x = golden_section_search(obj, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def smooth_gradient_descent(a0, time_limit, beta_init=5.0, beta_final=None,
                            lr_init=1.0, momentum=0.9, verbose=False, label=""):
    """
    Smoothed gradient descent with adaptive beta and momentum.
    Minimizes the smooth approximation of the max-conv objective.
    """
    if beta_final is None:
        beta_final = 200.0

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_true_score = score_of_normalized(a)
    t0 = time.monotonic()

    # Adam optimizer state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Beta schedule: exponential from beta_init to beta_final
    total_budget = time_limit
    beta = beta_init
    lr = lr_init
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed_frac = (time.monotonic() - t0) / total_budget
        # Exponential beta schedule
        progress = min(1.0, elapsed_frac / 0.9)  # ramp up over first 90%
        beta = beta_init * (beta_final / beta_init) ** progress

        # Compute gradient
        conv = fast_convolve(a)
        m_conv = float(conv.max())
        shifted = beta * (conv - m_conv)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = w / ws

        L = 1 << (3 * n - 2).bit_length()
        A = rfft(a, L)
        W = rfft(w, L)
        corr = irfft(W * np.conj(A), L)[:n]
        grad = 4.0 * n * corr

        # Adam update
        t_adam += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad ** 2
        m_hat = m_buf / (1 - beta1 ** t_adam)
        v_hat = v_buf / (1 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = simplex_project(a_new)

        # Evaluate true score periodically
        if iteration % 50 == 0 or iteration == 1:
            true_score = score_of_normalized(a_new)
            if true_score < best_true_score - 1e-10:
                best_true_score = true_score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 20:
                # Reduce learning rate
                lr *= 0.7
                no_improve = 0

        a = a_new

        # Decay learning rate
        lr = lr_init * (0.999 ** iteration)

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            true_sc = score_of_normalized(a)
            print(f"    SGD {label} it={iteration} ({elapsed:.0f}s): "
                  f"beta={beta:.1f} lr={lr:.4f} true={true_sc:.8f} best={best_true_score:.8f}")

    return best_a, best_true_score, iteration

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_spectral_shape(n, rng, flatten_power=1.0):
    """
    Initialize by shaping the spectrum: start with random phases,
    then flatten the amplitude spectrum partially.
    """
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    # Start with random amplitudes
    amps = rng.uniform(0.5, 1.5, n_freq)
    # Partially flatten: amps = amps^alpha where alpha < 1 flattens
    alpha = rng.uniform(0.3, 0.8)
    amps_flat = amps ** alpha
    # Normalize
    amps_flat = amps_flat / amps_flat.mean()
    spectrum = amps_flat * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.maximum(a[:n], 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

                continue
            cand = cand / cs
            sc = score_of_normalized(cand)
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            a_new = a_new / cs
            score_new = score_of_normalized(a_new)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            a = a / cs

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def spectral_perturbation(a, rng, strength=0.1):
    """Perturb in the frequency domain."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    # Add random phase perturbation
    phase_noise = rng.uniform(-strength, strength, A.shape)
    A = A * np.exp(1j * phase_noise)
    a_new = irfft(A, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a / a.sum()
    return a_new / s

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization combining smoothed gradient descent,
    exact coordinate descent, basin hopping, and LP fictitious play.
    """
    global _DEADLINE
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = init_flat_spectrum(1024, rng)
        best_seq = parent.copy()
        best_score = score_of_normalized(parent)
        print(f"Generated parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Smoothed Gradient Descent (40% time) ===
    print(f"\n=== Phase 1: Smoothed GD (best={best_score:.8f}) ===")
    t1_start = time.monotonic()
    t1_budget = 700.0  # ~40% of time

    # Try multiple initializations and beta schedules
    n_values = [n_parent]
    for alt_n in [1024, 768, 1280, 512, 1536]:
        if alt_n not in n_values and 256 <= alt_n <= 2048:
            n_values.append(alt_n)

    candidates = []

    for n in n_values[:4]:
        elapsed = time.monotonic() - t1_start
        if elapsed > t1_budget - 30:
            break
        t_per_init = max(10.0, (t1_budget - elapsed - 30) / 6.0)

        inits = []
        inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('flat2', init_spectral_shape(n, rng)))
        inits.append(('cos', init_cosine(n, rng)))
        inits.append(('flat3', init_flat_spectrum(n, rng)))
        inits.append(('shape2', init_spectral_shape(n, rng)))

        for itype, init_a in inits:
            elapsed = time.monotonic() - t1_start
            t_left = t1_budget - elapsed
            if t_left < 15:
                break
            t_alloc = min(t_per_init, t_left)

            a_opt, s_opt, _ = smooth_gradient_descent(
                init_a, t_alloc, beta_init=10.0, beta_final=500.0,
                lr_init=2.0, label=f"sgd_n{n}_{itype}"
            )
            candidates.append((s_opt, a_opt, n, itype))

            # Also try different beta schedule
            if t_left > 30:
                a_opt2, s_opt2, _ = smooth_gradient_descent(
                    init_a, t_alloc * 0.7, beta_init=5.0, beta_final=200.0,
                    lr_init=1.0, label=f"sgd2_n{n}_{itype}"
                )
                candidates.append((s_opt2, a_opt2, n, itype + "b"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best local: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:8]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Basin Hopping with Smooth GD (25% time) ===
    print(f"\n=== Phase 2: Basin Hopping (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    t2_budget = 450.0

    n_best = len(best_seq)
    current_best = best_seq.copy()
    n_hops = 12

    for hop in range(n_hops):
        elapsed = time.monotonic() - t2_start
        rem = t2_budget - elapsed
        if rem < 30:
            break
        t_alloc = min(rem * 0.6, 60.0)

        # Different perturbation strategies
        strat = hop % 4
        if strat == 0:
            # Gaussian perturbation
            perturbed = current_best + rng.standard_normal(n_best) * 0.03
        elif strat == 1:
            # Spectral perturbation
            perturbed = spectral_perturbation(current_best, rng, strength=0.15)
        elif strat == 2:
            # Peak-breaking: reduce elements at high-conv peaks
            conv_b = fast_convolve(current_best)
            top_k = min(10, 2 * n_best - 1)
            peaks = np.argpartition(conv_b, -top_k)[-top_k:]
            perturbed = current_best.copy()
            for k in peaks:
                i_start = max(0, int(k) - n_best + 1)
                i_end = min(n_best - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                perturbed[picks] *= (1.0 + rng.standard_normal() * 0.08)
        else:
            # Mixed: some random re-initialization blended
            new_init = init_flat_spectrum(n_best, rng)
            blend = 0.3 + 0.3 * rng.random()
            perturbed = (1.0 - blend) * current_best + blend * new_init

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = smooth_gradient_descent(
            perturbed, t_alloc, beta_init=15.0, beta_final=300.0,
            lr_init=1.5, label=f"bh{hop}"
        )
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"bh{hop}")

        if a_opt is not None and s_opt < score_of_normalized(current_best):
            current_best = a_opt.copy()

    # === Phase 3: Exact Coordinate Descent (20% time) ===
    print(f"\n=== Phase 3: Exact CD (best={best_score:.8f}) ===")
    t3_start = time.monotonic()
    t3_budget = 360.0

    n_best = len(best_seq)
    current = best_seq.copy()

    for rnd in range(6):
        elapsed = time.monotonic() - t3_start
        rem = t3_budget - elapsed
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 80.0)
        print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")

        a_opt, s_opt, sweeps = exact_coord_descent(
            current, t_alloc, verbose=True, label=f"cd_r{rnd}"
        )
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd}")

        # Check if we're stuck
        if s_opt > 1.5040 and rnd < 4:
            # Perturb to escape
            perturbed = a_opt + rng.standard_normal(n_best) * 0.04
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            current = perturbed
        else:
            current = a_opt.copy()

    # === Phase 4: LP Fictitious Play (10% time) ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.8f}) ===")
    t4_start = time.monotonic()
    t4_budget = 180.0

    solver = get_lp_solver()
    if solver is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 768]:
            if alt_n != n_best and 256 <= alt_n <= 1536:
                fp_lengths.append(alt_n)

        for n_fp in fp_lengths[:2]:
            elapsed = time.monotonic() - t4_start
            rem = t4_budget - elapsed
            if rem < 60:
                break
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  FP n={n_fp}, budget={t_alloc:.0f}s, solver={solver}")

            initial = interpolate_seq(best_seq, n_fp)
            a_fp, score_fp, iters = fictitious_play(
                initial, t_alloc, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

            # Follow up with CD
            elapsed = time.monotonic() - t4_start
            rem = t4_budget - elapsed
            if rem > 20:
                t_cd = min(rem * 0.5, 40.0)
                a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label=f"post_fp_{n_fp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"ecd_fp_{n_fp}")

    # === Phase 5: Final Polishing (5% time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    t5_start = time.monotonic()
    t5_budget = 90.0

    if best_seq is not None:
        # Fine coordinate descent
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem > 15:
            t_fine = min(rem * 0.5, 40.0)
            a_fine, s_fine, _ = fine_coord_descent(
                best_seq, t_fine, verbose=True, label="final_fine"
            )
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another ECD round
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem > 10:
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, min(rem * 0.5, 30.0), label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Final fine pass
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem > 5:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 15.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
