import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def z_from_a(a):
    """Convert to log-domain."""
    return np.log(np.maximum(a, 1e-300))

def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        b = fast_convolve(a)
        return float(2 * len(a) * b.max() / (s * s))
    except Exception:
        return float('inf')

def compute_score_normalized(a):
    """Compute score assuming sum(a)=1."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def gnc_lbgfs(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15):
    """Graduated non-convexity L-BFGS optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Compute gradient of sum_k w_k * b_k w.r.t. a
            # b_k = sum_i a[i] * a[k-i]
            # db_k/da[j] = a[k-j] (if valid)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # More iterations for lower temperatures
        frac = (stage_i + 1) / n_stages
        iters = int(50 + 200 * frac ** 2)
        
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters * 500, 'disp': False, 'maxls': 50}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def structured_init(n, mode, rng):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.abs(2*x - 1)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1-x)) * (1 + 0.2*np.cos(4*np.pi*x))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        # Tukey window
        r = 0.3
        a = np.ones(n)
        m = int(n * r / 2)
        if m > 0:
            for i in range(m):
                a[i] = 0.5 * (1 + np.cos(np.pi * (2*i/n/r - 1)))
                a[n-1-i] = a[i]
    elif mode == 7:
        # Optimal for low autocorrelation: raised cosine squared
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def frank_wolfe_away(a0, time_limit, n_steps=20000):

    n = len(a0)

    t0 = time.monotonic()
    no_improve = 0

            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = a[k_star - idx]
        
        best_trial = None
        best_trial_score = best_score
        
        # Toward step
        i_toward = int(np.argmin(g))
        for t_val in [0.5, 0.3, 0.15, 0.08]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = norm_sequence(a_trial)
            score_trial = compute_score_normalized(a_trial)
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()
        
        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.3, 0.2, 0.1]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)

                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length diverse initialization (Blackman, Tukey, raised cosine, Gaussian)
2. Graduated non-convexity (GNC) L-BFGS with adaptive long low-temperature tail
3. Precise coordinate descent with peak-targeted single-coordinate moves
4. Mirror descent (multiplicative weight) with adaptive step size
5. Structured perturbation with re-optimization
6. Ultra-precise final polish with very low temperature
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def z_from_a(a):
    """Convert to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_score_normalized(a):
    """Compute score assuming sum(a)=1."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def evaluate_safe(a):
    """Evaluate sequence safely using the actual evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def gnc_lbgfs(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=20):
    """
    Graduated non-convexity L-BFGS.
    Starts with smooth log-sum-exp surrogate and progressively sharpens.
    Uses more iterations at lower temperatures for precise convergence.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # More iterations at lower temperatures (they're more important)
        frac = (stage_i + 1) / n_stages
        iters = int(80 + 400 * frac ** 3)

        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={
                    'maxiter': iters,
                    'ftol': 1e-22,
                    'gtol': 1e-16,
                    'maxfun': iters * 1000,
                    'maxls': 80,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def mirror_descent(a0, time_limit, n_steps=20000):
    """
    Mirror descent (multiplicative weight) with adaptive step size.
    Natural for simplex-constrained problems.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    eta = 0.1
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_peak = int(np.argmax(b))

        # Subgradient
        g = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        g[idx] = a[k_peak - idx]

        # Mirror descent step: a_new[i] ∝ a[i] * exp(-eta * g[i])
        log_update = -eta * g
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()

        score_new = compute_score_normalized(a_new)

        if score_new < best_score - 1e-18:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.05, 1.0)  # Increase step if improving
        else:
            no_improve += 1
            eta = max(eta * 0.9, 0.001)  # Decrease step if not improving
            # Try a smaller step
            a_small = a * np.exp(-eta * 0.1 * g)
            a_small = np.maximum(a_small, 1e-300)
            a_small = a_small / a_small.sum()
            score_small = compute_score_normalized(a_small)
            if score_small < best_score - 1e-18:
                a = a_small
                best_score = score_small
                best_a = a.copy()
                no_improve = 0

        if no_improve > 200:
            break

    return best_a, best_score


def structured_init(n, mode, rng):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        xm = (x - 0.5) * 2
        a = np.exp(-xm ** 2 * 4)
    elif mode == 2:
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        a = np.exp(-5 * x * (1 - x))
    elif mode == 4:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif mode == 5:
        # Tukey window with r=0.4
        r = 0.4
        a = np.ones(n)
        m = int(n * r / 2)
        if m > 0:
            for i in range(m):
                a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (n * r) - 1)))
                a[n - 1 - i] = a[i]
    elif mode == 6:
        # Raised cosine squared (Hann-like)
        a = np.cos(np.pi * x) ** 2
    elif mode == 7:
        # Optimal flat-top approximation
        a = 0.36 - 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x) - 0.04 * np.cos(6 * np.pi * x)
    elif mode == 8:
        # Gaussian with oscillation
        xm = (x - 0.5) * 2
        a = np.exp(-xm ** 2 * 3) * (1 + 0.15 * np.cos(6 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 4 * x))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Diverse initialization at multiple lengths and shapes
    2. GNC L-BFGS main optimization (45% of time)
    3. Precise coordinate descent (15%)
    4. Mirror descent (10%)
    5. Second GNC L-BFGS with lower temperature (15%)
    6. Final coordinate descent polish (5%)
    7. Multi-length exploration (10%)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 0: Load parent ===
    print("Phase 0: Loading parent...")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = structured_init(n_init, 6, rng)
        best_score = evaluate_safe(best_seq)
        print(f"  Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Diverse initialization exploration (5% time) ===
    print(f"\nPhase 1: Diverse init (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        lengths = sorted(set([
            n_best,
            max(256, int(n_best * 0.75)),
            max(256, int(n_best * 1.25)),
            max(256, int(n_best * 1.5)),
            max(256, int(n_best * 1.1)),
            max(256, int(n_best * 0.9)),
        ]))

        for alt_n in lengths:
            if time.monotonic() > deadline - t_alloc + 60:
                break
            if alt_n == n_best:
                continue
            for mode in [0, 2, 4, 5, 6, 7]:
                if time.monotonic() > deadline:
                    break
                a_init = structured_init(alt_n, mode, rng)
                sc = evaluate_safe(a_init)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_init.copy()
                    n_best = alt_n
                    print(f"  Better init: n={alt_n}, mode={mode}, score={sc:.6f}")

        # Try interpolating parent to nearby lengths
        for mult in [0.9, 1.1, 1.2, 1.33, 0.8]:
            alt_n = int(n_best * mult)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                a_interp = interpolate_sequence(best_seq, alt_n)
                sc = evaluate_safe(a_interp)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_interp.copy()
                    n_best = alt_n
                    print(f"  Better interp: n={alt_n}, score={sc:.6f}")

    # === Phase 2: GNC L-BFGS main (45% time) ===
    print(f"\nPhase 2: GNC L-BFGS main (best={best_score:.6f})")
    n_cur = len(best_seq)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 800.0)
    if t_alloc > 30:
        print(f"  n={n_cur}, T: 500->0.03, budget={t_alloc:.0f}s")
        a_gnc = gnc_lbgfs(best_seq, t_alloc, T_start=500.0, T_end=0.03, n_stages=22)
        try_update(a_gnc, "gnc_main")

    # === Phase 3: Precise coordinate descent (15% time) ===
    print(f"\nPhase 3: Coord Descent (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 15:
        print(f"  budget={t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_precise(best_seq, t_alloc, n_rounds=10000)
        try_update(a_cd, "coord_desc")

    # === Phase 4: Mirror descent (10% time) ===
    print(f"\nPhase 4: Mirror Descent (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        print(f"  budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, n_steps=50000)
        try_update(a_md, "mirror_desc")

    # === Phase 5: Second GNC L-BFGS pass - lower temperature (15% time) ===
    print(f"\nPhase 5: GNC L-BFGS pass 2 (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  T: 30->0.02, budget={t_alloc:.0f}s")
        a_gnc2 = gnc_lbgfs(best_seq, t_alloc, T_start=30.0, T_end=0.02, n_stages=15)
        try_update(a_gnc2, "gnc_pass2")

    # === Phase 6: Final coordinate descent polish (5% time) ===
    print(f"\nPhase 6: Final Coord Desc (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        a_cd2, _ = coordinate_descent_precise(best_seq, t_alloc, n_rounds=20000)
        try_update(a_cd2, "final_cd")

    # === Phase 7: Multi-length exploration (10% time) ===
    print(f"\nPhase 7: Multi-Length Exploration (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_cur = len(best_seq)
        alt_lengths = []
        for mult in [0.85, 0.9, 1.05, 1.1, 1.15, 1.25, 1.4, 1.6]:
            alt_n = int(n_cur * mult)
            if 256 <= alt_n <= 4096 and alt_n != n_cur:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_total = min(rem * 0.18, 200.0)
            if t_total < 40:
                break
            print(f"  n={alt_n}, budget={t_total:.0f}s")

            a_init = interpolate_sequence(best_seq, alt_n)

            # GNC on new length
            t_gnc = t_total * 0.65
            a_alt = gnc_lbgfs(a_init, t_gnc, T_start=50.0, T_end=0.05, n_stages=12)
            try_update(a_alt, f"gnc_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
                t_cd = min(rem * 0.3, 50.0)
                if t_cd > 15:
                    a_alt2, _ = coordinate_descent_precise(best_seq, t_cd, n_rounds=5000)
                    try_update(a_alt2, f"cd_n{alt_n}")

    # === Phase 8: Structured perturbation + re-optimization ===
    print(f"\nPhase 8: Perturbation Escape (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        t_total = min(rem * 0.12, 150.0)

        for pert_mode in range(4):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            x = np.linspace(0, 1, n_cur)
            if pert_mode == 0:
                noise = 0.04 * np.sin(8 * np.pi * x)
            elif pert_mode == 1:
                noise = 0.04 * rng.standard_normal(n_cur)
            elif pert_mode == 2:
                noise = 0.03 * np.cos(16 * np.pi * x)
            else:
                noise = 0.06 * (np.random.default_rng(100 + pert_mode).random(n_cur) - 0.5)

            a_pert = best_seq + noise
            a_pert = norm_sequence(a_pert)

            t_sub = min(t_total / 4, rem * 0.7)
            t_gnc_pert = t_sub * 0.7
            if t_gnc_pert > 10:
                a_pert_opt = gnc_lbgfs(a_pert, t_gnc_pert, T_start=20.0, T_end=0.05, n_stages=8)
                try_update(a_pert_opt, f"pert{pert_mode}")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_cd_pert = min(rem * 0.4, t_sub * 0.3)
                if t_cd_pert > 8:
                    a_pert_cd, _ = coordinate_descent_precise(best_seq, t_cd_pert, n_rounds=3000)
                    try_update(a_pert_cd, f"pert_cd{pert_mode}")

    # === Phase 9: Final ultra-low-temperature polish ===
    print(f"\nPhase 9: Final Ultra-Precise Polish (best={best_score:.6f})")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 150.0)
        print(f"  T: 8->0.01, budget={t_fin:.0f}s")
        a_fin = gnc_lbgfs(best_seq, t_fin, T_start=8.0, T_end=0.01, n_stages=12)
        try_update(a_fin, "final_gnc")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd_fin = min(rem * 0.5, 80.0)
        a_cd_fin, _ = coordinate_descent_precise(best_seq, t_cd_fin, n_rounds=30000)
        try_update(a_cd_fin, "final_cd2")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_md_fin, _ = mirror_descent(best_seq, rem, n_steps=20000)
        try_update(a_md_fin, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
