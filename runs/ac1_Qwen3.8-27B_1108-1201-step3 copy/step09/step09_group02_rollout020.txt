_lp_solver_cached = None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section 1D search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Randomize coordinate order for diversity
        order = np.arange(n)
        if sweeps % 3 == 0:
            rng_local = np.random.default_rng(sweeps)
            rng_local.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]

            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 4.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    if not isinstance(changes, list):
                        changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final renormalization
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv_f = fast_convolve(a)
    final_score = 2.0 * n * float(conv_f.max())
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score, sweeps

                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_local_max = float(conv[k_lo:k_hi+1].max())

            factors = [0.999, 0.998, 0.995, 0.99, 0.985, 0.98, 0.97,
                       0.95, 1.02, 1.03, 1.05, 1.07, 1.10, 1.15]
            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = max(unaff_max, float(nv.max()))
                ns = 1.0 + (x - v)
                if ns < 0.01:
                    continue
                sc = 2.0 * n * local_max / (ns * ns)
                if sc < best_score + (best_local_max - local_max) * 0.1:
                    if local_max < best_local_max - 1e-16:
                        best_local_max = local_max
                        best_delta = x - v

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_barker_like(n, rng):
    """Barker-like initialization using pseudo-random sign sequence."""
    # Create a pseudo-random +/-1 sequence
    signs = rng.choice([-1.0, 1.0], size=n)
    # Smooth it to make it non-negative
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(signs + 1.0, kernel, mode='same')
    a = np.maximum(smoothed[:n], 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_zadoff_chu(n, rng):
    """Zadoff-Chu inspired initialization."""
    m = rng.integers(1, n // 2)
    t = np.arange(n, dtype=np.float64)
    # Zadoff-Chu: exp(-j*pi*m*n*(n+1)/N)
    phase = np.pi * m * t * (t + 1) / n
    a = 1.0 + np.cos(phase)  # Always non-negative, between 0 and 2
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(3, 12)
    amps = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_harmonics, n//2), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def mirror_descent_step(a, conv, lr, n_peaks=15, temp_scale=0.01):
    """
    One mirror descent step in log domain.
    Uses active-set identification and softmax reparametrization.
    """
    n = len(a)
    # Identify active peaks with soft weighting
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    max_pv = pv.max()
    if max_pv < 1e-15:
        return a
    temp = temp_scale * max_pv
    w = np.exp((pv - max_pv) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    # Compute subgradient of 2n*conv[k]/(sum a)^2 for each active peak
    # With sum(a)=1: grad_a[i] = 2n * sum_k w_k * (a[i+k_k] + a[i-k_k] - 2*a[i]*conv[k_k])
    # Simplified: grad_a[i] = 2n * sum_k w_k * (a[i+k_k] + a[i-k_k])
    # (The -2*a[i]*conv[k] term is constant per i for fixed k, and after
    #  summing over k with weights summing to 1, it becomes -2*a[i]*max_conv)
    
    grad_a = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wk = w[idx]
        # d(conv[k])/d(a[i]) for k >= 0:
        # = a[i+k] if i+k < n, plus a[i-k] if i-k >= 0
        if k == 0:
            grad_a += 2.0 * wk * a
        else:
            # a[i+k] contribution: for i in [0, n-1-k], grad[i] += a[i+k]
            grad_a[:n-k] += wk * a[k:]
            # a[i-k] contribution: for i in [k, n-1], grad[i] += a[i-k]
            grad_a[k:] += wk * a[:n-k]

    # Convert to log-domain gradient (mirror gradient)
    # In log domain with a = exp(w)/sum(exp(w)):
    # grad_log[m] = a[m] * (grad_a[m] - dot(grad_a, a))
    mean_grad = np.dot(grad_a, a)
    grad_log = a * (grad_a - mean_grad)

    # Update in log domain
    w_a = np.log(np.maximum(a, 1e-300))
    w_a = w_a - lr * grad_log
    # Renormalize via softmax
    w_a = w_a - w_a.max()
    a_new = np.exp(w_a)
    a_new = a_new / a_new.sum()

    return a_new

def mirror_descent_optimize(a0, time_limit, verbose=False, label=""):
    """
    Mirror descent optimization in log domain with adaptive learning rate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    lr = 0.005
    temp_scale = 0.01
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive learning rate
        lr = 0.005 / (1.0 + 0.01 * it)
        temp_scale = max(0.003, 0.01 * (1.0 - 0.1 * it / 100.0))

        a = mirror_descent_step(a, conv, lr, n_peaks=15, temp_scale=temp_scale)
        conv = fast_convolve(a)
        sc = 2.0 * n * float(conv.max())

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 50:
                break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, it

def active_set_optimize(a0, time_limit, verbose=False, label=""):
    """
    Active-set optimization: identify active lags, then use projected
    gradient specifically targeting those constraints.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Identify active set: lags where conv is within 5% of max
        max_conv = conv.max()
        active_lags = np.where(conv >= 0.95 * max_conv)[0]
        
        # Weight by how close to max
        weights = np.zeros(2 * n - 1)
        for k in active_lags:
            weights[k] = (conv[k] / max_conv) ** 10  # Sharp weighting
        
        ws = weights.sum()
        if ws < 1e-15:
            break
        weights = weights / ws

        # Compute combined subgradient
        grad = np.zeros(n)
        for k in active_lags:
            if weights[int(k)] < 1e-10:
                continue
            kk = int(k)
            wk = weights[kk]
            if kk == 0:
                grad += 2.0 * wk * a
            else:
                grad[:n-kk] += wk * a[kk:]
                grad[kk:] += wk * a[:n-kk]

        # Projected gradient step
        lr = 0.003 / (1.0 + 0.005 * it)
        a_new = simplex_project(a - lr * grad)
        conv_new = fast_convolve(a_new)
        sc = 2.0 * n * float(conv_new.max())

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
        else:
            # Try smaller step
            lr_half = lr * 0.5
            a_new2 = simplex_project(a - lr_half * grad)
            conv_new2 = fast_convolve(a_new2)
            sc2 = 2.0 * n * float(conv_new2.max())
            if sc2 < best_score - 1e-12:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
                conv = conv_new2
            else:
                # No improvement; break after many failures
                if it % 200 == 0 and it > 500:
                    break

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, it

def spectral_shaping_init(n, rng, parent=None):
    """
    Initialize by trying to flatten the auto-convolution spectrum.
    Uses iterative spectral shaping.
    """
    if parent is not None:
        a = interpolate_seq(parent, n).copy()
    else:
        a = init_flat_spectrum(n, rng)

    # Iterative shaping: repeatedly reduce the peak
    for _ in range(50):
        conv = fast_convolve(a)
        peak_val = conv.max()
        mean_val = conv.mean()
        
        # For each peak, try to reduce it by adjusting the contributing elements
        for _inner in range(3):
            k_peak = int(np.argmax(conv))
            # Reduce elements that contribute to this peak
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx = np.arange(i_start, i_end + 1)
            
            if len(idx) == 0:
                break
            
            # Reduce the largest contributors
            contributions = np.minimum(a[idx], a[idx + k_peak if k_peak >= 0 else 0]) if k_peak != 0 else a[idx]
            if k_peak == 0:
                contributions = a[idx]
            else:
                j_start = 0
                j_end = n - 1 - k_peak
                contributions = np.minimum(a[idx], a[idx + k_peak])
            
            # Reduce top contributors slightly
            top_frac = max(1, len(idx) // 4)
            top_idx = idx[np.argsort(contributions)[-top_frac:]]
            a[top_idx] *= 0.95
            
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                return np.ones(n) / n
            a = a / s
            conv = fast_convolve(a)

    return a

def basin_hopping_advanced(a0, time_limit, cd_func, rng, verbose=False):
    """
    Advanced basin hopping with multiple perturbation strategies.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    
    perturbation_strategies = []
    
    def perturb_gaussian(sigma):
        p = a + rng.standard_normal(n) * sigma
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        return p

    def perturb_peak_break():
        p = a.copy()
        conv_p = fast_convolve(a)
        top_k = min(10, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            kk = int(k)
            i_start = max(0, kk - n + 1)
            i_end = min(n - 1, kk)
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 4)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                p[ii] *= (1.0 + rng.standard_normal() * 0.1)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        return p

    def perturb_swap():
        p = a.copy()
        n_swaps = rng.integers(5, 20)
        for _ in range(n_swaps):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            delta = rng.standard_normal() * 0.02 * max(p[i], p[j], 1.0/n)
            p[i] = max(0, p[i] - delta)
            p[j] = max(0, p[j] + delta)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        return p

    def perturb_block():
        p = a.copy()
        block_size = rng.integers(n // 16, n // 4)
        start = rng.integers(0, n - block_size)
        # Swap or modify a block
        shift = rng.integers(-block_size, block_size + 1)
        if shift != 0:
            segment = p[start:start+block_size].copy()
            new_start = (start + shift) % n
            if new_start + block_size <= n:
                p[start:start+block_size] = 0
                p[new_start:new_start+block_size] = segment
        # Add noise
        p += rng.standard_normal(n) * 0.01
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        return p

    perturbation_strategies = [
        lambda: perturb_gaussian(0.02),
        lambda: perturb_gaussian(0.05),
        lambda: perturb_gaussian(0.1),
        lambda: perturb_peak_break(),
        lambda: perturb_swap(),
        lambda: perturb_block(),
    ]

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 40.0)

        # Choose perturbation strategy
        strat = rng.integers(0, len(perturbation_strategies))
        perturbed = perturbation_strategies[strat]()

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:
                print(f"    BH-adv p={iteration}: improved to {best_score:.8f}")

    return best_a, best_score

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k]."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix efficiently
    # conv(a, b)[k] = sum_j a[j] * b[j+k] (for valid j)
    # Express as T @ b <= M where T[k, j] = a[j+k] or a[j-k]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Line search
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

        inits.append(('zc', init_zadoff_chu(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))

        for itype, cand in inits:
            t_left = phase1_budget - (time.monotonic() - t1_start)
            if t_left < 4:
                break
            t_alloc = max(2.0, min(10.0, t_left / 6.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Intensive ECD on best (200s) ===
    print(f"\n=== Phase 2: Intensive ECD (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    phase2_budget = 200.0

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()
        for rnd in range(4):
            rem = phase2_budget - (time.monotonic() - t2_start)
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 80.0)
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"ph2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_ecd_r{rnd}")

            if best_score > 1.5038:
                perturbed = a_opt + rng.standard_normal(n_best) * 0.04
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: Mirror descent (log-domain) (150s) ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.8f}) ===")
    t3_start = time.monotonic()
    phase3_budget = 150.0

    if best_seq is not None:
        n_best = len(best_seq)
        # Run mirror descent from current best
        rem = min(phase3_budget, deadline - time.monotonic() - 400)
        if rem > 20:
            t_md = min(rem * 0.6, 100.0)
            a_md, s_md, it_md = mirror_descent_optimize(best_seq, t_md,
                                                        verbose=True, label="md_main")
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror_descent")

            # Follow up with CD
            rem = deadline - time.monotonic() - 350
            if rem > 15:
                a_cd, s_cd, _ = exact_coord_descent(a_md, min(rem*0.5, 50.0), label="cd_after_md")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_after_md")

        # Also try mirror descent from a different initialization
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            a_init = init_flat_spectrum(n_best, rng)
            t_md2 = min(rem * 0.5, 60.0)
            a_md2, s_md2, _ = mirror_descent_optimize(a_init, t_md2, label="md_alt")
            best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "md_alt")
            # Polish
            rem = deadline - time.monotonic() - 300
            if rem > 10:
                a_pol, s_pol, _ = exact_coord_descent(a_md2, min(rem*0.5, 30.0), label="pol_md_alt")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "pol_md_alt")

    # === Phase 4: Active-set optimization (100s) ===
    print(f"\n=== Phase 4: Active-Set (best={best_score:.8f}) ===")
    t4_start = time.monotonic()
    phase4_budget = 100.0

    if best_seq is not None:
        rem = min(phase4_budget, deadline - time.monotonic() - 300)
        if rem > 15:
            t_as = min(rem * 0.7, 80.0)
            a_as, s_as, it_as = active_set_optimize(best_seq, t_as,
                                                    verbose=True, label="as_main")
            best_seq, best_score = check_and_update(a_as, best_seq, best_score, "active_set")
            # Polish
            rem = deadline - time.monotonic() - 250
            if rem > 10:
                a_pol, s_pol, _ = exact_coord_descent(a_as, min(rem*0.5, 30.0), label="pol_as")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "pol_as")

    # === Phase 5: LP Fictitious Play (150s) ===
    print(f"\n=== Phase 5: LP Fictitious Play (best={best_score:.8f}) ===")
    t5_start = time.monotonic()
    phase5_budget = 150.0

    if best_seq is not None and solver is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 768]:
            if alt_n != n_best and 256 <= alt_n <= 1536:
                fp_lengths.append(alt_n)
        fp_lengths = list(set(fp_lengths))[:2]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_fp = min(phase5_budget * 0.5, rem * 0.5, 80.0)
            initial = interpolate_seq(best_seq, n_fp)
            a_fp, score_fp, iters = fictitious_play(initial, t_fp, solver=solver, verbose=True)
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

            rem = deadline - time.monotonic() - 200
            if rem > 10:
                a_cd, s_cd, _ = exact_coord_descent(a_fp, min(rem*0.4, 40.0), label=f"cd_fp_{n_fp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_fp_{n_fp}")

    # === Phase 6: Advanced basin hopping (200s) ===
    print(f"\n=== Phase 6: Advanced Basin Hopping (best={best_score:.8f}) ===")
    t6_start = time.monotonic()
    phase6_budget = 200.0

    if best_seq is not None:
        rem = min(phase6_budget, deadline - time.monotonic() - 200)
        if rem > 30:
            t_bh = min(rem * 0.7, 150.0)
            a_bh, s_bh = basin_hopping_advanced(best_seq, t_bh, exact_coord_descent,
                                                rng, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 7: Final intensive polishing (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_fine = min(rem * 0.5, 80.0)
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another exact CD
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, min(rem * 0.6, 50.0), label="ecd_polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Mirror descent final
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_md_f, s_md_f, _ = mirror_descent_optimize(best_seq, min(rem * 0.5, 30.0), label="md_final")
            best_seq, best_score = check_and_update(a_md_f, best_seq, best_score, "md_final")

        # Final fine
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_fine")

        # Very final ECD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_vf, s_vf, _ = exact_coord_descent(best_seq, min(rem * 0.8, 15.0), label="vf")
            best_seq, best_score = check_and_update(a_vf, best_seq, best_score, "very_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
