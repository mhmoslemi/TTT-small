import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_and_grad_p(a, p):
    """Compute p-norm of conv(a,a) and its gradient w.r.t. a."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    cp = c_pos ** p
    S = np.sum(cp)
    if S < 1e-300:
        return 0.0, np.zeros_like(a)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    n = len(a)
    corr = np.correlate(w, a, mode='valid')
    val = S ** (1.0 / p)
    grad = 2.0 * factor * corr
    return val, grad


def objective_lse(a, beta):
    """LogSumExp of conv(a,a) scaled by beta."""
    c = fast_convolve(a)
    c_max = c.max()
    vals = np.exp(beta * (c - c_max))
    return c_max + np.log(np.sum(vals)) / beta

            if time.monotonic() - t0 > time_limit:
                break

            val_y, grad_y = objective_and_grad_p(y, p)
            # Project gradient: subtract component along (1,1,...,1) for simplex
            grad_proj = grad_y - np.mean(grad_y)
            grad_proj[grad_proj < 0] = np.where(grad_proj < 0, grad_proj, 0)  # keep as is for now

            # Backtracking line search (simplified: fixed lr with occasional adjustment)
            a_new = project_simplex(y - lr * grad_y)

            val_a, _ = objective_and_grad_p(a_new, p)

            # Update Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count)) / 2.0
            y_new = a_new + ((t_count - 1.0) / t_new) * (a_new - a)
            y_new = np.maximum(y_new, 0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new / sy
            else:
                y_new = a_new.copy()

            a = a_new
            y = y_new
            t_count = t_new

            # Slowly decay learning rate
            lr = lr_base * n / np.sqrt(it + 1.0) * 0.5

            # Track best
            actual_max = fast_convolve(a).max()
            if actual_max < best_val * (1.0 - 1e-10):
                best_val = actual_max
                best_a = a.copy()

            if it % 10000 == 0 and it > 0:
                elapsed = time.monotonic() - t0
                print(f"        Nesterov p={p} iter={it} ({elapsed:.0f}s): max_b={best_val:.8f}")

    return best_a


def subgradient_max_nesterov(a0, time_limit, top_k=10, max_iters=100000):
    """Subgradient method on max(conv(a,a)) with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    t_count = 1.0
    lr = 0.1 / n

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(y)
        # Get top-k peaks
        if top_k < len(c):
            top_idx = np.argsort(c)[::-1][:top_k]
        else:
            top_idx = np.argsort(c)[::-1]

        # Subgradient: average of gradients at top peaks
        # d(c_k)/d(a_i) = 2 * a[k-i] if 0 <= k-i < n else 0 (for the valid part)
        # Actually: c_k = sum_j a[j]*a[k-j], d/d(a_i) = sum_j: (delta_{ij}*a[k-j] + a[j]*delta_{k-j,i})
        # = a[k-i] (if valid) + a[i] (if k-i valid, i.e., the other term)
        # More precisely: d(c_k)/d(a_i) = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
        # Wait: c_k = sum_j a[j] * a[k-j]. d/d(a_i): for term j=i, gives a[k-i].
        # For term k-j=i, i.e., j=k-i, gives a[j] = a[k-i].
        # So d(c_k)/d(a_i) = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
        # But we need to be careful: if i == k-i (i.e., 2i=k), then it's a[i]^2 and
        # d/d(a_i) = 2*a[i], which is the same formula.

        grad = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * a[k - j]
                # Also the symmetric contribution
                other = k - j
                if 0 <= other < n:
                    grad[other] += 2.0 * a[j]

        # Wait, this double counts. Let me be more careful.
        # c_k = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # d(c_k)/d(a_i) = sum_j [delta_{j,i}*a[k-j] + a[j]*delta_{k-j,i}]
        # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, from j=k-i)
        # = 2*a[k-i] if 0<=k-i<n
        grad = np.zeros(n)
        for k in top_idx:
            idx = k - np.arange(n)  # k - j for j=0,...,n-1
            valid = (idx >= 0) & (idx < n)
            if np.any(valid):
                grad += 2.0 * a[idx[valid]] * 0  # placeholder
                # Actually grad[j] gets 2*a[k-j] for valid k-j
                target_j = np.arange(n)[valid]  # these are j values
                # No wait: d(c_k)/d(a_j) = 2*a[k-j] if 0<=k-j<n
                # So for each j in 0..n-1, if 0<=k-j<n, then grad[j] += 2*a[k-j]
                for j in range(max(0, k - n + 1), min(n, k + 1)):
                    grad[j] += 2.0 * a[k - j]

        # Normalize subgradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-30:
            grad = grad / grad_norm

        # Step
        a_new = project_simplex(y - lr * grad * n)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count)) / 2.0
        y_new = a_new + ((t_count - 1.0) / t_new) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = a_new.copy()

        a = a_new
        y = y_new
        t_count = t_new
        lr = 2.0 / (it + 3.0) * 0.5 / n

        # Track best
        actual_max = fast_convolve(a).max()
        if actual_max < best_val * (1.0 - 1e-12):
            best_val = actual_max
            best_a = a.copy()

        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        SubGradMax iter={it+1} ({elapsed:.0f}s): max_b={best_val:.8f}")

    return best_a


def active_peak_shaving(a0, time_limit, n_iters=50000):
    """Directly reduce the top peaks by adjusting contributing coefficients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (it + 1) % 5000 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 2:
                break

        c = fast_convolve(a)
        # Find the single biggest peak
        peak_k = np.argmax(c)
        peak_val = c[peak_k]

        # Get all (i, j) pairs contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        partners = peak_k - idx
        contribs = a[idx] * a[partners]
        total_contrib = np.sum(contribs)

        if total_contrib < 1e-15:
            break

        # Find the top contributors
        top_contrib_idx = np.argsort(contribs)[::-1][:10]

        improved = False
        for ti in top_contrib_idx:
            j = idx[ti]
            k = partners[ti]
            if a[j] < 1e-14 or a[k] < 1e-14:
                continue

            # Try to reduce the product a[j]*a[k]
            # Strategy: reduce the larger one, redistribute
            if a[j] >= a[k]:
                # Reduce a[j]
                for frac in [0.7, 0.5, 0.3, 0.5]:
                    reduction = a[j] * (1.0 - frac)
                    if reduction < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[j] -= reduction
                    if a_trial[j] < 0:
                        a_trial[j] = 0.0
                        reduction = a[j]
                    # Redistribute to non-contributing positions
                    mask = np.ones(n, dtype=bool)
                    mask[js:je+1] = False  # Don't add to positions that also contribute
                    mask[j] = False
                    if mask.sum() < 1:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += reduction * others / others_sum
                    new_max = fast_convolve(a_trial).max()
                    if new_max < best_val * (1.0 - 1e-12):
                        a = a_trial
                        best_val = new_max
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Reduce a[k]
                for frac in [0.7, 0.5, 0.3]:
                    reduction = a[k] * (1.0 - frac)
                    if reduction < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[k] -= reduction
                    if a_trial[k] < 0:
                        a_trial[k] = 0.0
                        reduction = a[k]
                    mask = np.ones(n, dtype=bool)
                    mask[js:je+1] = False
                    mask[k] = False
                    if mask.sum() < 1:
                        mask = np.ones(n, dtype=bool)
                        mask[k] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += reduction * others / others_sum
                    new_max = fast_convolve(a_trial).max()
                    if new_max < best_val * (1.0 - 1e-12):
                        a = a_trial
                        best_val = new_max
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            # Try pair reduction at multiple peaks
            if it % 1000 == 0:
                top_peaks = np.argsort(c)[::-1][:3]
                for pk in top_peaks:
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    idx2 = np.arange(js2, je2 + 1)
                    partners2 = pk - idx2
                    contribs2 = a[idx2] * a[partners2]
                    top2 = np.argsort(contribs2)[::-1][:5]
                    for ti in top2:
                        j = idx2[ti]
                        k2 = partners2[ti]
                        for f in [0.8, 0.6, 0.4]:
                            a_trial = a.copy()
                            a_trial[j] *= f
                            a_trial[k2] *= f
                            st = a_trial.sum()
                            if st < 1e-15:
                                continue
                            a_trial /= st
                            new_max = fast_convolve(a_trial).max()
                            if new_max < best_val * (1.0 - 1e-12):
                                a = a_trial
                                best_val = new_max
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break

        if not improved and it > 5000:
            break

    return best_a


def lbfgs_pnorm_z(z0, time_limit, p_stages=None, max_iter_per_stage=2000):
    """L-BFGS in log/softmax domain on p-norm objective."""
    n = len(z0)
    z = z0.copy().astype(np.float64)
    t0 = time.monotonic()

    if p_stages is None:
        p_stages = [20, 50, 100, 200, 500, 1000, 2000]

    best_val = float('inf')
    best_z = z.copy()

    for p in p_stages:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        def func_p(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            return 2.0 * n * (np.sum(c_pos ** p)) ** (1.0 / p)

        def jac_p(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_pos = np.maximum(c, 1e-300)
            cp = c_pos ** p
            S = np.sum(cp)
            if S < 1e-300:
                return np.zeros_like(z_val)
            factor = S ** (1.0 / p - 1.0)
            w = c_pos ** (p - 1.0)
            corr = np.correlate(w, a, mode='valid')
            g_a = 4.0 * n * factor * corr
            return a * (g_a - np.dot(g_a, a))

        try:
            res = minimize(
                fun=func_p, jac=jac_p, x0=z,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': max_iter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        cur_val = 2.0 * n * fast_convolve(a_cur).max()
        if cur_val < best_val:
            best_val = cur_val
            best_z = z.copy()
        elapsed = time.monotonic() - t0
        print(f"        L-BFGS-z p={p} ({elapsed:.0f}s): score={cur_val:.6f}")

    return best_z


def z_to_a(z):
    """Convert log-domain to normalized sequence."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}) ***")
        return True
    return False


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def structured_init_asym(n, variant=0):
    """Create asymmetric initial sequences."""
    if variant == 0:
        # Exponential decay (one-sided)
        x = np.arange(n) / n
        a = np.exp(-3.0 * x)
    elif variant == 1:
        # Asymmetric Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2)
    elif variant == 2:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
    elif variant == 3:
        # Bimodal asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.8) ** 2)
    elif variant == 4:
        # Sine (non-symmetric)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.7
    else:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 1.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lse_nesterov(a0, time_limit, beta_stages=None, max_iters_per_stage=30000):
    """Nesterov accelerated projected gradient on LogSumExp objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()

    if beta_stages is None:
        beta_stages = [50, 100, 200, 500, 1000, 2000]

    for beta in beta_stages:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        y = a.copy()
        t_count = 1.0
        lr = 5.0 / (n * np.sqrt(beta))

        for it in range(max_iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            _, grad_y = objective_lse_grad(y, beta)

            a_new = project_simplex(y - lr * grad_y)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count)) / 2.0
            y_new = a_new + ((t_count - 1.0) / t_new) * (a_new - a)
            y_new = np.maximum(y_new, 0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new / sy
            else:
                y_new = a_new.copy()

            a = a_new
            y = y_new
            t_count = t_new
            lr = 2.0 / (it + 3.0) / np.sqrt(beta)

            if (it + 1) % 5000 == 0:
                actual_max = fast_convolve(a).max()
                if actual_max < best_val * (1.0 - 1e-10):
                    best_val = actual_max
                    best_a = a.copy()

            if (it + 1) % 15000 == 0:
                elapsed = time.monotonic() - t0
                print(f"        LSE-Nest beta={beta} iter={it+1} ({elapsed:.0f}s): max_b={best_val:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov p-norm, LogSumExp Nesterov,
    L-BFGS in softmax domain, subgradient max, and active peak shaving.
    Explores multiple lengths around the parent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        parent = structured_init_asym(n_init, variant=0)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])
    a_best = best_state['seq'].copy()

    # === Phase 1: Nesterov p-norm on parent ===
    print(f"\n=== Phase 1: Nesterov p-norm n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()

    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")

    # === Phase 3: LogSumExp Nesterov ===

    rem = deadline - time.monotonic()

        a_lse = lse_nesterov(a_best, t_alloc, beta_stages=[50, 100, 300, 800, 1500])
        try_update(best_state, a_lse, "lse_nesterov_1")
        a_best = best_state['seq']

    # === Phase 4: Subgradient max ===
    print(f"\n=== Phase 4: Subgradient Max n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()

        print(f"  Budget: {t_alloc:.0f}s")
        a_sg = subgradient_max_nesterov(a_best, t_alloc, top_k=5, max_iters=50000)
        try_update(best_state, a_sg, "subgrad_max")
        a_best = best_state['seq']

    # === Phase 5: Active peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()

    print(f"\n=== Phase 6: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = []

        alt_n = base + delta

    # Add some round numbers

        if 384 <= rn <= 2560 and all(abs(rn - c) > 16 for c in candidate_lengths):
            candidate_lengths.append(rn)

        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_to_length(a_best, alt_n)

        # Quick Nesterov p-norm
        t_pn = t_total * 0.5
        a_alt = nesterov_pnorm(init_alt, t_pn, p_stages=[30, 100, 400])
        improved = try_update(best_state, a_alt, f"pn_n{alt_n}")

        if improved:
            a_best = best_state['seq']
            rem = deadline - time.monotonic()
            if rem > 30:
                t_alt2 = min(rem * 0.10, 80.0)
                a_alt2 = lse_nesterov(a_best, t_alt2, beta_stages=[100, 300, 800])
                try_update(best_state, a_alt2, f"lse_n{alt_n}")
                a_best = best_state['seq']

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(a_best)
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = a_best + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 1e-15:
                perturbed = perturbed / ps
            t_sub = min(rem * 0.3, 50.0)
            a_pert = nesterov_pnorm(perturbed, t_sub, p_stages=[50, 200, 800])
            try_update(best_state, a_pert, f"pert_{noise_scale}")
            a_best = best_state['seq']

    # === Phase 8: Final high-p polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 100.0)
        a_fin = nesterov_pnorm(a_best, t_fin, p_stages=[500, 1000, 3000, 5000])
        try_update(best_state, a_fin, "final_pnorm")
        a_best = best_state['seq']

    # === Phase 9: Final peak shaving ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.30, 60.0)
        a_ps2 = active_peak_shaving(a_best, t_ps, n_iters=20000)
        try_update(best_state, a_ps2, "final_peak")
        a_best = best_state['seq']

    # === Phase 10: LSE final ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_lse = min(rem * 0.40, 50.0)
        a_lse2 = lse_nesterov(a_best, t_lse, beta_stages=[500, 1000, 2000])
        try_update(best_state, a_lse2, "final_lse")
        a_best = best_state['seq']

    # Final check
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - (deadline - float(budget_s) + 20.0)
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")

    if best_state['seq'] is None:
        # Fallback
        return [1.0 / 960] * 960
    return [float(x) for x in best_state['seq']]
